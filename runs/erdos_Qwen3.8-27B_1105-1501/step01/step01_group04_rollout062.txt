    # We want to minimize max(corr) * dx
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        k_max = np.argmax(corr)
        
        # Compute gradient of corr[k_max] wrt each h[i]
        # grad[i] = (1-h)[i+k_max-n+1] - h[i-k_max+n-1]
        # (with out-of-bounds = 0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    """Project h onto {x : 0<=x<=1, sum(x)=target}"""
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

def exact_line_search(corr, d, delta_max, iters=50):
    """Find delta in [0, delta_max] minimizing max(corr + d*delta)"""
    if delta_max <= 1e-15:
        return 0.0
    a, b = 0.0, delta_max
    gr = (np.sqrt(5.0)-1.0)/2.0
    c = b - gr*(b-a)
    da_ = a + gr*(b-a)
    fc = float(np.max(corr + d*c))
    fda = float(np.max(corr + d*da_))
    for _ in range(iters):
        if fc < fda:
            b = da_
            da_ = c
            fda = fc
            c = b - gr*(b-a)
            fc = float(np.max(corr + d*c))
        else:
            a = c
            c = da_
            fc = fda
            da_ = a + gr*(b-a)
            fda = float(np.max(corr + d*da_))
    return (a+b)/2.0

        k_max = int(np.argmax(corr))

        grad = np.zeros(n)

        lo2 = max(0, k_max-n+1)
        hi2 = min(n-1, k_max-1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2+1)
            grad[idx] -= h[idx-k_max+n-1]

        pos = np.where((grad > 0) & (h > 1e-10))[0]
        neg = np.where((grad < 0) & (h < 1.0-1e-10))[0]
        
        if len(pos) == 0 or len(neg) == 0:
            no_improve += 1
            if no_improve > max_no_improve:
                break

            h = best_h + 0.01 * rng.randn(n)
            h = project_to_simplex(h, float(n)/2.0)
            corr = np.correlate(h, 1.0-h, mode='full')
            continue
        
        # Try top candidates
        # Score for pair (i,j): grad[j] - grad[i] (we want this to be very negative, 
        # i.e., grad[i] large positive, grad[j] large negative)
        # Actually d_k* = -grad[i] + grad[j], we want d_k* < 0
        # Best reduction: minimize d_k* = grad[j] - grad[i], so maximize grad[i] - grad[j]
        
        # Get top 5 positive and top 5 negative
        pos_sorted = pos[np.argsort(-grad[pos])][:min(10, len(pos))]
        neg_sorted = neg[np.argsort(grad[neg])][:min(10, len(neg))]
        
        best_delta = None
        best_new_c5 = best_c5
        best_i, best_j = -1, -1
        
        for i in pos_sorted:
            for j in neg_sorted:
                if i == j:
                    continue
                delta_max = min(h[i], 1.0-h[j])
                if delta_max < 1e-12:
                    continue
                d = swap_derivative(h, i, j)
                delta = exact_line_search(corr, d, delta_max)
                new_c5 = float(np.max(corr + d*delta)) * dx
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_delta = delta
                    best_i, best_j = i, j
                    best_d = d
        
        if best_i >= 0 and best_new_c5 < best_c5 - 1e-15:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            # Renormalize (shouldn't be needed but safety)
            # Actually the swap preserves the sum, so no renormalization needed
            corr = np.correlate(h, 1.0-h, mode='full')
            best_h = h.copy()
            best_c5 = best_new_c5
            no_improve = 0
            improved_this_pass = True
        else:

                    corr = np.correlate(h, 1.0-h, mode='full')
                    no_improve = 0
                else:
                    break
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_swap_deriv(h, i, j):
    n = len(h)
    d = np.zeros(2*n-1)
    omh = 1.0 - h
    # -omh[i+k-n+1]
    lo1 = max(0, n-1-i); hi1 = min(2*n-2, 2*n-2-i)
    if lo1 <= hi1: d[lo1:hi1+1] -= omh[lo1-i+n-1:hi1-i+n+1]
    # +h[i-k+n-1]
    lo2 = max(0, i-n+1); hi2 = min(2*n-2, i)
    if lo2 <= hi2: d[lo2:hi2+1] += h[lo2-n+1+i:hi2-n+1+i+1]
    # +omh[j+k-n+1]
    lo3 = max(0, n-1-j); hi3 = min(2*n-2, 2*n-2-j)
    if lo3 <= hi3: d[lo3:hi3+1] += omh[lo3-j+n-1:hi3-j+n+1]
    # -h[j-k+n-1]
    lo4 = max(0, j-n+1); hi4 = min(2*n-2, j)
    if lo4 <= hi4: d[lo4:hi4+1] -= h[lo4-n+1+j:hi4-n+1+j+1]
    return d

def line_search_1d(corr, d, delta_max, iters=40):
    if delta_max <= 1e-15: return 0.0
    a, b = 0.0, delta_max
    gr = 0.6180339887
    c = b - gr*(b-a); da = a + gr*(b-a)
    fc = float(np.max(corr + d*c)); fda = float(np.max(corr + d*da))
    for _ in range(iters):
        if fc < fda:
            b = da; da = c; fda = fc
            c = b - gr*(b-a); fc = float(np.max(corr + d*c))
        else:
            a = c; c = da; fc = fda
            da = a + gr*(b-a); fda = float(np.max(corr + d*da))
    return (a+b)*0.5

def greedy_optimize(h0, time_budget, seed=42):
    n = len(h0)
    dx = 2.0 / n
    h = project_to_simplex(h0.astype(float).copy(), float(n)/2.0)
    corr = np.correlate(h, 1.0-h, mode='full')
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    t0 = time.time()
    rng = np.random.RandomState(seed)
    fail_count = 0
    max_fail = 100
    
    while time.time() - t0 < time_budget:
        if time.time() - t0 >= time_budget: break
        kmax = int(np.argmax(corr))
        # gradient of corr[kmax]
        grad = np.zeros(n)
        lo1 = max(0, n-1-kmax); hi1 = min(n-1, 2*n-2-kmax)
        if lo1 <= hi1: grad[lo1:hi1+1] += (1.0-h)[lo1+kmax-n+1:hi1+kmax-n+2]
        lo2 = max(0, kmax-n+1); hi2 = min(n-1, kmax-1)
        if lo2 <= hi2: grad[lo2:hi2+1] -= h[lo2-kmax+n-1:hi2-kmax+n]
        
        pos = np.where((grad > 1e-10) & (h > 1e-10))[0]
        neg = np.where((grad < -1e-10) & (h < 1.0-1e-10))[0]
        
        if len(pos) == 0 or len(neg) == 0:
            fail_count += 1
            if fail_count > max_fail: break
            # Perturb
            h = best_h + 0.01*rng.randn(n)
            h = project_to_simplex(h, float(n)/2.0)
            corr = np.correlate(h, 1.0-h, mode='full')
            continue
        
        # Try candidates
        npos = min(5, len(pos)); nneg = min(5, len(neg))
        pos_top = pos[np.argsort(-grad[pos])][:npos]
        neg_top = neg[np.argsort(grad[neg])][:nneg]
        
        best_new = best_c5; best_i = best_j = -1; best_d_val = 0; best_dd = None
        for i in pos_top:
            for j in neg_top:
                if i == j: continue
                dmax = min(h[i], 1.0-h[j])
                if dmax < 1e-12: continue
                dd = compute_swap_deriv(h, i, j)
                dv = line_search_1d(corr, dd, dmax)
                nc5 = float(np.max(corr + dd*dv)) * dx
                if nc5 < best_new:
                    best_new = nc5; best_i = i; best_j = j; best_d_val = dv; best_dd = dd
        
        if best_i >= 0 and best_new < best_c5 - 1e-14:
            h[best_i] -= best_d_val; h[best_j] += best_d_val
            h = np.clip(h, 0.0, 1.0)
            corr = np.correlate(h, 1.0-h, mode='full')
            best_h = h.copy(); best_c5 = best_new
            fail_count = 0
        else:
            fail_count += 1
            if fail_count > max_fail:
                h = best_h + 0.005*rng.randn(n)
                h = project_to_simplex(h, float(n)/2.0)
                corr = np.correlate(h, 1.0-h, mode='full')
                fail_count = 0
            elif fail_count > max_fail//2:
                break
    
    return best_h, best_c5

                corr = np.correlate(h, 1.0 - h, mode='full')
                fail_count = 0
                pert_idx += 1
            elif fail_count > max_fail // 2:
                # Try random swaps to escape
                for _ in range(50):
                    if time.time() - t0 >= time_budget:
                        break
                    ri = rng.randint(n)
                    rj = rng.randint(n)
                    if ri == rj:
                        continue
                    if h[ri] < 1e-10 or h[rj] > 1.0 - 1e-10:
                        continue
                    dmax = min(h[ri], 1.0 - h[rj])
                    if dmax < 1e-12:
                        continue
                    dd = compute_swap_deriv(h, ri, rj)
                    dv = line_search_1d(corr, dd, dmax)
                    nc5 = float(np.max(corr + dd * dv)) * dx
                    if nc5 < best_c5 - 1e-14:
                        h[ri] -= dv
                        h[rj] += dv
                        h = np.clip(h, 0.0, 1.0)
                        corr = np.correlate(h, 1.0 - h, mode='full')
                        best_h = h.copy()
                        best_c5 = nc5
                        fail_count = 0
                        break

    return best_h, best_c5


def multi_target_swap(h0, time_budget, seed=42, n_targets=3):
    """Optimize using weighted gradient from top correlations simultaneously."""
    n = len(h0)
    dx = 2.0 / n
    h = project_to_simplex(h0.astype(float).copy(), float(n) / 2.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    t0 = time.time()
    rng = np.random.RandomState(seed)
    fail_count = 0
    max_fail = 80

    while time.time() - t0 < time_budget:
        if time.time() - t0 >= time_budget:
            break
        # Get top correlations
        top_idx = np.argsort(corr)[::-1][:n_targets]
        weights = np.exp(5.0 * (corr[top_idx] - corr[top_idx[0]]))
        weights = weights / weights.sum()

        # Combined gradient
        grad = np.zeros(n)
        for w, kmax in zip(weights, top_idx):
            lo1 = max(0, n - 1 - kmax)
            hi1 = min(n - 1, 2 * n - 2 - kmax)
            if lo1 <= hi1:
                grad[lo1:hi1 + 1] += w * (1.0 - h)[lo1 + kmax - n + 1:hi1 + kmax - n + 2]
            lo2 = max(0, kmax - n + 1)
            hi2 = min(n - 1, kmax - 1)
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] -= w * h[lo2 - kmax + n - 1:hi2 - kmax + n]

        pos = np.where((grad > 1e-10) & (h > 1e-10))[0]
        neg = np.where((grad < -1e-10) & (h < 1.0 - 1e-10))[0]

        if len(pos) == 0 or len(neg) == 0:
            fail_count += 1
            if fail_count > max_fail:
                break
            h = best_h + 0.008 * rng.randn(n)
            h = project_to_simplex(h, float(n) / 2.0)
            corr = np.correlate(h, 1.0 - h, mode='full')
            continue

        npos = min(6, len(pos))
        nneg = min(6, len(neg))
        pos_top = pos[np.argsort(-grad[pos])][:npos]
        neg_top = neg[np.argsort(grad[neg])][:nneg]

        best_new = best_c5
        best_i = best_j = -1
        best_dv = 0.0

        for i in pos_top:
            for j in neg_top:
                if i == j:
                    continue
                dmax = min(h[i], 1.0 - h[j])
                if dmax < 1e-12:
                    continue
                dd = compute_swap_deriv(h, i, j)
                dv = line_search_1d(corr, dd, dmax)
                nc5 = float(np.max(corr + dd * dv)) * dx
                if nc5 < best_new:
                    best_new = nc5
                    best_i = i
                    best_j = j
                    best_dv = dv

        if best_i >= 0 and best_new < best_c5 - 1e-14:
            h[best_i] -= best_dv
            h[best_j] += best_dv
            h = np.clip(h, 0.0, 1.0)
            corr = np.correlate(h, 1.0 - h, mode='full')
            best_h = h.copy()
            best_c5 = best_new
            fail_count = 0
        else:
            fail_count += 1
            if fail_count > max_fail:
                h = best_h + 0.005 * rng.randn(n)
                h = project_to_simplex(h, float(n) / 2.0)
                corr = np.correlate(h, 1.0 - h, mode='full')
                fail_count = 0
            elif fail_count > max_fail // 2:
                break

    return best_h, best_c5


def smooth_gradient_optimize(h0, time_budget, seed=42):
    """Smooth max gradient descent with momentum and projection."""
    n = len(h0)
    h = project_to_simplex(h0.astype(float).copy(), float(n) / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    alpha = 8.0
    lr = 0.005
    momentum = 0.95
    v = np.zeros(n)
    stall = 0
    stall_limit = 400

    while time.time() - t0 < time_budget:
        if time.time() - t0 >= time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = float(np.max(corr))
        exp_c = np.exp(alpha * (corr - maxc))
        s_exp = np.sum(exp_c)
        if s_exp < 1e-300:
            break
        p = exp_c / s_exp

        # Compute gradient: grad[i] = sum_k p[k]*((1-h)[i+k-n+1] - h[i-k+n-1])
        # Vectorized using the fact that this is a correlation
        # Part A: sum_k p[k] * (1-h)[i+k-n+1]
        # Part B: sum_k p[k] * h[i-k+n-1]
        # Use np.correlate for efficiency
        omh = 1.0 - h
        # A[i] = sum_k p[k] * omh[i+k-(n-1)]
        # This equals np.correlate(omh, p, 'full')[i + (len(p)-1) - (len(omh)-1)]
        # len(p)=2n-1, len(omh)=n, so index = i + 2n-2 - (n-1) = i + n - 1
        corr_omh_p = np.correlate(omh, p, mode='full')
        # Part B: sum_k p[k] * h[i-k+n-1]
        # B[i] = sum_k p[k] * h[i-k+(n-1)]
        # = np.correlate(p, h, 'full')[i + (len(p)-1) - (len(h)-1)]
        # len(p)=2n-1, len(h)=n, index = i + 2n-2 - (n-1) = i + n - 1
        corr_p_h = np.correlate(p, h, mode='full')

        grad = corr_omh_p[n - 1:2 * n - 1] - corr_p_h[n - 1:2 * n - 1]

        v = momentum * v - lr * grad
        h_new = project_to_simplex(h + v, float(n) / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        alpha = alpha * 1.03 + 0.01
        lr = max(lr * 0.998, 0.0001)

        if stall > stall_limit:
            h = best_h + 0.01 * rng.randn(n)
            h = project_to_simplex(h, float(n) / 2.0)
            v = np.zeros(n)
            stall = 0
            alpha = 8.0
            lr = 0.005

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial values
    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    if h_init is not None:
        n0 = len(h_init)
        # Phase 1: Greedy optimize from initial (main phase)
        alloc1 = min(remaining() * 0.45, 500.0)
        if alloc1 > 30:
            h_best, c5_best = greedy_optimize(h_init, alloc1, seed=seed, top_k=8)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best
                best_n = n0

        # Phase 2: Multi-target swap
        if best_h is not None and remaining() > 50:
            alloc2 = min(remaining() * 0.2, 200.0)
            h_best2, c5_best2 = multi_target_swap(best_h, alloc2, seed=seed + 100, n_targets=3)
            if c5_best2 < best_c5:
                best_c5 = c5_best2
                best_h = h_best2

        # Phase 3: Smooth gradient optimization
        if best_h is not None and remaining() > 50:
            alloc3 = min(remaining() * 0.15, 150.0)
            h_best3, c5_best3 = smooth_gradient_optimize(best_h, alloc3, seed=seed + 200)
            if c5_best3 < best_c5:
                best_c5 = c5_best3
                best_h = h_best3

        # Phase 4: Final greedy polish
        if best_h is not None and remaining() > 30:
            alloc4 = max(remaining() * 0.5, 30.0)
            h_best4, c5_best4 = greedy_optimize(best_h, alloc4, seed=seed + 300, top_k=12)
            if c5_best4 < best_c5:
                best_c5 = c5_best4
                best_h = h_best4
    else:
        # No initial values, create our own
        n0 = 99
        h_init = np.zeros(n0)
        h_init[:n0 // 2] = 1.0
        h_init = project_to_simplex(h_init, float(n0) / 2.0)
        h_best, c5_best = greedy_optimize(h_init, remaining() - 10, seed=seed, top_k=8)
        best_c5 = c5_best
        best_h = h_best
        best_n = n0

    # Final constraint check
    if best_h is not None:
        target = float(best_n) / 2.0
        best_h = project_to_simplex(best_h, target)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        n = 99
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_h = project_to_simplex(best_h, float(n) / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
