```python
def compute_grad_matrix(h):
    n = len(h)
    R = 2*n - 1
    G = np.zeros((R, n))
    one_minus_h = 1.0 - h
    for r in range(R):
        k = r - (n - 1)
        # First term: (1-h[i-k]) * [0 <= i-k < n]
        idx1 = np.arange(n) - k  # i - k for i=0..n-1
        valid1 = (idx1 >= 0) & (idx1 < n)
        G[r, valid1] += one_minus_h[idx1[valid1]]
        # Second term: (-h[i+k]) * [0 <= i+k < n]
        idx2 = np.arange(n) + k  # i + k for i=0..n-1
        valid2 = (idx2 >= 0) & (idx2 < n)
        G[r, valid2] -= h[idx2[valid2]]
    return G
```

```python
def compute_pair_coefficients(h, i, j):
    """Compute c0, c1, c2 for each lag for the pair (i,j)."""
    n = len(h)
    R = 2 * n - 1
    S = h[i] + h[j]
    c0 = np.zeros(R)
    c1 = np.zeros(R)
    c2 = np.zeros(R)
    
    for r in range(R):
        k = r - (n - 1)
        # k is the lag
        # Sum over valid m of h[m+k]*(1-h[m])
        # Valid m: 0<=m<n and 0<=m+k<n
        m_start = max(0, -k)
        m_end = min(n, n - k)
        
        # For each valid m, determine if it involves i, j, or both
        for m in range(m_start, m_end):
            p = m + k  # the other index
            # Term: h[p]*(1-h[m])
            if m == i and p == j:
                # (S-x)*(1-x) = x^2 - (1+S)x + S
                c2[r] += 1.0
                c1[r] += -(1.0 + S)
                c0[r] += S
            elif m == j and p == i:
                # x*(1-(S-x)) = x^2 - (S-1)x
                c2[r] += 1.0
                c1[r] += (1.0 - S)
                c0[r] += 0.0
            elif m == i:
                # h[p]*(1-x), where p≠j
                c1[r] += -h[p]
                c0[r] += h[p] * h[i]
            elif p == i:
                # x*(1-h[m]), where m≠j
                c1[r] += (1.0 - h[m])
                c0[r] += -h[i] * (1.0 - h[m])
            elif m == j:
```

```python
            if U - L < 1e-10:
                continue
            
            c0, c1, c2 = compute_pair_coefficients(h, i, j)
            
            # Ternary search
            lo, hi = L, U
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(c0 + c1*m1 + c2*m1*m1)
                f2 = np.max(c0 + c1*m2 + c2*m2*m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_star = (lo + hi) / 2.0
            
            f_star = np.max(c0 + c1*x_star + c2*x_star*x_star) * dx
            if f_star < score - 1e-10:
                h[i] = x_star
                h[j] = S - x_star
                h[i] = clip(h[i], 0, 1)
                h[j] = clip(h[j], 0, 1)
                score = f_star
                any_improvement = True
                min_improvement = max(min_improvement, score - f_star)
        
        if not any_improvement:
            break
    
    return h, official_score(h)
```

The classification depends on whether m or p equals i or j. For a given k:
- m=i is valid if max(0,-k) <= i <= min(n,n-k)-1
- p=i (i.e., m=i-k) is valid if max(0,-k) <= i-k <= min(n,n-k)-1
- m=j is valid if max(0,-k) <= j <= min(n,n-k)-1
- p=j (i.e., m=j-k) is valid if max(0,-k) <= j-k <= min(n,n-k)-1

For lags where NONE of these conditions hold, the term is just h[p]*(1-h[m]) and contributes to c0 only.
