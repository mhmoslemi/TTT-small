def make_feasible(y, n=None):

def score_exact(h):

def overlap_grad_single(h, k_idx, n):
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:k+m])
            g[k:] -= h[:m]
    else:
        p = -k
        m = n - p
        if m > 0:
            g[p:] += (1.0 - h[:m])
            g[:m] -= h[p:]
    return g

def resample(h_old, n_new):

def _smooth_obj(z, h_ref, lam_q, T, n):
    h = z[:n]
    ov = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    max_ov = float(ov.max())
    ex = np.exp((ov - max_ov) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        return max_ov
    val = max_ov + T * np.log(S)
    if lam_q > 0:
        d = h - h_ref
        val += lam_q * np.dot(d, d)
    return val

def _smooth_grad(z, h_ref, lam_q, T, n):
    h = z[:n]
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    max_ov = float(ov.max())
    ex = np.exp((ov - max_ov) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        w = np.zeros(2 * n - 1)
        mask = ov >= max_ov - 1e-12
        w[mask] = 1.0 / float(np.sum(mask))
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    # Accumulate gradient using overlap_grad_single
    for idx in range(2 * n - 1):
        if w[idx] > 1e-15:
            g += w[idx] * overlap_grad_single(h, idx, n)
    if lam_q > 0:
        g += 2.0 * lam_q * (h - h_ref)
    return g

                                best_h = cur.copy()
                                best_c = cur_c
                        else:
                            cur[i] = orig_i
                            cur[i + 1] = orig_j
            if not improved:
                break

    # Phase 6: Resolution changes (5% time)
    p6_end = t0 + budget_s * 0.93
    if time.time() < p6_end:
        n_cur = len(best_h)
        for dn in (-6, -4, -2, 2, 4, 6):
            if time.time() > p6_end - 5:
                break
            nr = n_cur + dn
            if nr < 20 or nr > 120 or nr % 2 != 0:
                continue
            h_r = resample(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(p6_end, time.time() + 30)
            if time.time() < sub_dl and c_r < best_c + 0.005:
                h_rs, c_rs = active_set_optimize(h_r, sub_dl, n_ks=12, max_cycle=5, lam_q=0.0)
                if c_rs < best_c:
                    best_h, best_c = h_rs, c_rs
                    n = nr

    # Phase 7: Final refinement (7% time)
    p7_end = deadline
    if time.time() < p7_end:
        sub_dl = min(p7_end, time.time() + 60)
        if time.time() < sub_dl:
            h_f, c_f = active_set_optimize(best_h, sub_dl, n_ks=15, max_cycle=10, lam_q=0.0)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
        sub_dl2 = min(p7_end, time.time() + 30)
        if time.time() < sub_dl2:
            h_f2, c_f2 = smooth_optimize(best_h, sub_dl2, T_start=0.01, T_end=0.0003, step0=0.005, T_decay=0.999)
            if c_f2 < best_c:
                best_h, best_c = h_f2, c_f2

    # Ensure feasibility and final evaluation
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
