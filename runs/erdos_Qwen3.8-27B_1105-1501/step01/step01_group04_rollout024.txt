import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_to_simplex(h, target):
    """Project onto {0<=x<=1, sum(x)=target} via binary search on threshold."""
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_fingerprints(h):
    n = len(h)
    m = 2 * n - 1
    f = np.zeros((n, m))
    # For position p, f[p][k] = change in corr[k] when h[p] increases by 1
    # corr[k] = sum_i h[i+k-n+1] * (1-h[i]) (with h=0 outside [0,n-1])
    # h[p] appears in h[i+k-n+1] when i+k-n+1 = p, i.e., i = p-k+n-1
    #   valid when 0 <= p-k+n-1 < n, i.e., k in [p-n+1, p+n-1] intersect [0, 2n-2]
    #   contribution: (1-h[i]) = (1-h[p-k+n-1])
    # h[p] appears in (1-h[i]) when i = p
    #   contribution: -h[p+k-n+1] (valid when 0 <= p+k-n+1 < n)
    
    for p in range(n):
        # First part: h[p] in the h[...] slot
        # i = p - k + n - 1, valid when 0 <= i < n
        # => 0 <= p-k+n-1 < n => p-n+1 <= k <= p (and k in [0, 2n-2])
        k_lo = max(0, p - n + 1)
        k_hi = min(2*n-2, p)
        for k in range(k_lo, k_hi + 1):
            i = p - k + n - 1
            f[p, k] += (1.0 - h[i])
        
        # Second part: h[p] in the (1-h[...]) slot
        # i = p, so h[i+k-n+1] = h[p+k-n+1] must be valid
        # 0 <= p+k-n+1 < n => n-1-p <= k <= 2n-2-p (and k in [0, 2n-2])
        k_lo2 = max(0, n - 1 - p)
        k_hi2 = min(2*n-2, 2*n-2-p)
        for k in range(k_lo2, k_hi2 + 1):
            j = p + k - n + 1
            f[p, k] -= h[j]
    return f
