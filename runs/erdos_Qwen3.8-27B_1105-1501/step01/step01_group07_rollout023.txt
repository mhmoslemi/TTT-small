def make_feasible(y, n=None):
    """Project y to [0,1]^n with sum = n/2."""
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(z)  # most room first
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-14:

        else:
            order = np.argsort(-z)  # most to remove first
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def softmax_objective(h, T):
    """Softmax approximation of max of overlap scores."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = scores - np.max(scores)
    ex = np.exp(shifted / T)
    S = np.sum(ex)
    if S < 1e-300:
        return float(np.max(scores))
    return float(np.max(scores) + T * np.log(S))

def softmax_gradient(h, T):
    """Gradient of softmax objective."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = scores - np.max(scores)
    ex = np.exp(shifted / T)
    S = np.sum(ex)
    if S < 1e-300:
        mask = scores >= np.max(scores) - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):

        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

    orig_j = h[j]
    # h[i] + delta, h[j] - delta

def upsample(h, n_new):
    """Upsample h to n_new points."""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])

```python
def compute_gradient(h, weights):
    """
    h: array of length n
    weights: array of length 2n-1 (weights for each shift in correlation output)
    Returns: gradient array of length n
    """
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    c = n - 1  # center index
    
    for idx in range(2 * n - 1):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        k = idx - c  # shift: k = -(n-1), ..., n-1
        
        if k == 0:
            g += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            # d = k
            d = k
            if d < n:
                g[d:] += w * dx * (1.0 - h[:n-d])
                g[:n-d] -= w * dx * h[d:]
        else:
            # k < 0, d = -k
            d = -k
            if d < n:
                g[:n-d] += w * dx * (1.0 - h[d:])
                g[d:] -= w * dx * h[:n-d]
    
    return g
```

One more thought: I should also try to use the fact that the problem has a natural "balance" property. At the optimum, the gradient of the max (or the subgradient) should be orthogonal to the feasible direction (i.e., the sum constraint is the only thing to worry about). This means the sum of the gradient components should be zero (or close to it) at the optimum.

Actually that's not quite right. The gradient of the objective at the optimum should be orthogonal to the tangent space of the constraint set. The tangent space of {h ∈ [0,1]^n : Σh = n/2} at a point h (where all 0 < h[i] < 1) is {d : Σd = 0}. So at the optimum, the projection of the gradient onto the constraint set should be zero.

This means: the subgradient g should satisfy Σ g[i] = 0 (if all h[i] are strictly between 0 and 1). If some h[i] are at the boundary (0 or 1), the condition is weaker.

This is a useful check: if Σ g[i] ≠ 0, we know the solution isn't optimal (or we're at a boundary point).

Let me incorporate this into my algorithm: after computing the subgradient, I can project it to have zero sum (since we're constrained to the hyperplane Σh = n/2). This gives a better search direction.

Actually, the make_feasible function already handles the projection. But for the gradient direction, it's better to explicitly subtract the mean:
direction = g - mean(g)

This ensures the search direction is in the tangent space of the constraint.

import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(z)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(-z)
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return np.clip(z, 0.0, 1.0)

    n = len(h)

def softmax_gradient(h, T):
    n = len(h)
    scores = overlap_scores(h)
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = compute_gradient(h, w)
    # Project to tangent space (zero sum)
    g = g - np.mean(g)
    return g
