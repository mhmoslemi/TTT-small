import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_rows(h, active_rows):
    n = len(h)
    R = len(active_rows)
    G = np.zeros((R, n))
    for idx, k in enumerate(active_rows):
        shift = k - (n - 1)
        if shift >= 0:
            i_lo, i_hi = 0, n - shift
        else:
            i_lo, i_hi = -shift, n
        for i in range(i_lo, i_hi):
            j = i + shift
            G[idx, i] += (1.0 - h[j])
            G[idx, j] += (-h[i])
    return G

def compute_raw_rows(h, active_rows):
    n = len(h)
    raw = compute_raw_corr(h)
    return raw[active_rows].copy()

def compute_degree(h, active_rows):
    n = len(h)
    R = len(active_rows)
    D = np.ones(R)
    for idx, k in enumerate(active_rows):
        shift = k - (n - 1)
        if shift == 0:
            D[idx] = 1.0
        else:
            D[idx] = 2.0
    return D

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_raw_corr(h) * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_raw_corr(h) * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            i_lo, i_hi = 0, n - shift
        else:
            i_lo, i_hi = -shift, n
        for i in range(i_lo, i_hi):
            j = i + shift
            grad[i] += w * (1.0 - h[j])
            grad[j] += w * (-h[i])
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def smooth_opt(h0, beta, maxiter=200):
    n = len(h0)
    try:
        res = minimize(
            smooth_max_obj, h0, args=(beta,),
            jac=smooth_max_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0.copy()

def cvx_major_step(h0, active_rows, rho, reg_weight, solver=None):
    n = len(h0)
    raw_rows = compute_raw_rows(h0, active_rows)
    G = compute_grad_rows(h0, active_rows)
    D = compute_degree(h0, active_rows)
    R = len(active_rows)

    delta = cp.Variable(n)
    t = cp.Variable()

    sq = cp.sum_squares(delta)
    obj = t + reg_weight * sq
    constraints = [cp.sum(delta) == 0]
    constraints.append(delta >= -h0)
    constraints.append(delta <= 1.0 - h0)
    constraints.append(sq <= rho * rho)

    for idx in range(R):
        constraints.append(raw_rows[idx] + G[idx] @ delta + D[idx] * sq <= t)

    problem = cp.Problem(cp.Minimize(obj), constraints)

    solve_kwargs = {}
    if solver is not None:
        try:
            problem.solve(solver=solver, **solve_kwargs)
        except Exception:
            try:
                problem.solve(solver=cp.ECOS)
            except Exception:
                return None
    else:
        try:
            problem.solve(solver=cp.ECOS)
        except Exception:
            try:
                problem.solve()
            except Exception:
                return None

    if delta.value is None:
        return None
    d = np.asarray(delta.value, dtype=float).ravel()
    if not np.all(np.isfinite(d)):
        return None
    return d

def cvx_smooth_step(h0, beta, maxiter=200):
    n = len(h0)
    try:
        res = minimize(
            smooth_max_obj, h0, args=(beta,),
            jac=smooth_max_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0.copy()

def pick_active_rows(h0, max_active=30):
    n = len(h0)
    raw = compute_raw_corr(h0)
    k_max = len(raw)
    order = np.argsort(raw)[::-1]
    active = list(order[:max_active])
    # Always include the peak
    peak = int(np.argmax(raw))
    if peak not in active:
        active.append(peak)
    return active
