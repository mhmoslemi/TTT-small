from scipy.optimize import minimize
import time


def make_feasible(z):

    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def score_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def interp_binary(parent, parent_n, n, rng):
    """Interpolate parent to length n and threshold to binary."""
    if parent_n == n:
        h = parent.copy()
    else:
        x_src = (np.arange(parent_n) + 0.5) * 2.0 / parent_n
        x_tgt = (np.arange(n) + 0.5) * 2.0 / n
        h = np.interp(x_tgt, x_src, parent)
    # Threshold to binary
    threshold = np.sort(h)[n // 2]
    h_bin = (h >= threshold).astype(float)
    # Fix count
    cnt = int(h_bin.sum())
    if cnt > n // 2:
        # Remove excess ones (from lowest h values)
        one_indices = np.where(h_bin == 1)[0]
        h_vals_at_ones = h[one_indices]
        excess = cnt - n // 2
        to_remove = one_indices[np.argsort(h_vals_at_ones)[:excess]]
        h_bin[to_remove] = 0.0
    elif cnt < n // 2:
        zero_indices = np.where(h_bin == 0)[0]
        h_vals_at_zeros = h[zero_indices]
        deficit = n // 2 - cnt
        to_add = zero_indices[np.argsort(-h_vals_at_zeros)[:deficit]]
        h_bin[to_add] = 1.0
    return h_bin


def binary_anneal(h_bin, n, rng, max_iters, T0, T_end):
    """Simulated annealing on binary sequences."""
    h = h_bin.astype(float).copy()
    sc = score(h)
    best = h.copy()
    best_sc = sc
    T = T0
    cool = (T_end / T0) ** (1.0 / max(max_iters, 1))
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0 or T <= T_end * 1.01:
            T = T_end
            # Continue with greedy
            pass
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_sc = score(h)
        
        if new_sc < sc or rng.random() < np.exp(-(new_sc - sc) / max(T, 1e-12)):
            sc = new_sc
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if new_sc < best_sc:
                best = h.copy()
                best_sc = new_sc
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cool
    
    return best, best_sc


def local_search_binary(h_bin, n, max_rounds=10):
    """Greedy local search: keep swapping if it improves."""
    h = h_bin.astype(float).copy()
    sc = score(h)
    improved = True
    rounds = 0
    
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        # Try all pairs (or a random subset for large n)
        # For n<=80, try all pairs
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_pairs = n_ones * n_zeros
        if n_pairs > 50000:
            # Random subset

            idx = idx[:50000]
        else:
            idx = np.arange(n_pairs)
        
        for pi in idx:
            i = ones[pi // n_zeros]
            j = zeros[pi % n_zeros]
            if h[i] < 0.5 or h[j] > 0.5:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_sc = score(h)
            if new_sc < sc - 1e-15:
                sc = new_sc
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
    
    return h, sc


def smooth_f_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    C = score_arr(x)
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(2 * n - 1) / beta
    return cmax + (mk + np.log(ws)) / beta


def smooth_f_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    C = score_arr(x)
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    # Gradient of C[k] wrt x[i]
    # C[k] = sum_i x[i] * (1-x[i+k]) * dx (for valid i, i+k)
    # dC[k]/dx[i] = (1-x[i+k])*dx - x[i]*dx (if i+k valid) + x[i-k]*dx - (1-x[i-k])*dx (if i-k valid)
    # Actually: C[k] = sum_{valid l} x[l]*(1-x[l+k]) * dx
    # dC[k]/dx[i] = dx * [(1-x[i+k]) if i+k valid] + dx * [-(1-x[i]) if i-k valid... ]
    
    # Let me think more carefully.
    # C[k] = dx * sum_{l: 0<=l<n, 0<=l+k<n} x[l] * (1-x[l+k])
    # dC[k]/dx[i] = dx * [ (1-x[i+k]) * 1_{i+k valid} + (-x[i-k]) * 1_{i-k valid, and the term where l+k=i, i.e., l=i-k} ]
    # = dx * [ (1-x[i+k]) * 1_{0<=i+k<n} + (-x[i-k]) * 1_{0<=i-k<n} ]
    # Wait, the second term: when l+k = i, i.e., l = i-k, the term is x[i-k]*(1-x[i]), and d/dx[i] = -x[i-k]
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if abs(w) < 1e-15:
            continue
        k = k_idx - (n - 1)  # actual lag
        # For this lag k: C[k] = dx * sum_{l=max(0,-k)}^{min(n, n-k)-1} x[l]*(1-x[l+k])
        start = max(0, -k)
        end = min(n, n - k)
        # dC/dx[i] for i in [start, end): (1-x[i+k])*dx
        # dC/dx[i] for i in [start+k, end+k): -x[i-k]*dx
        if start < end:
            grad[start:end] += w * dx * (1.0 - x[start + k:end + k])
        p_start = max(0, start + k)
        p_end = min(n, end + k)
        if p_start < p_end:
            grad[p_start:p_end] += w * dx * (-x[p_start - k:p_end - k])
    return grad


def eq_sum(x):
    return float(np.sum(x) - len(x) / 2.0)


def eq_sum_jac(x):
    return np.ones(len(x))


def smooth_solve(h0, n, beta, maxiter=200):
    h0 = make_feasible(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]
    try:
        res = minimize(smooth_f_obj, h0, args=(beta,), jac=smooth_f_grad,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def score(h):

    dx = 2.0 / n

def sobj(x, beta):
    n = len(x)
    dx = 2.0 / n
    C = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(2 * n - 1) / beta
    return cmax + (mk + np.log(ws)) / beta


def sgrad(x, beta):
    n = len(x)
    dx = 2.0 / n
    C = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if abs(w) < 1e-15:
            continue
        k = k_idx - (n - 1)
        start = max(0, -k)
        end = min(n, n - k)
        if start < end:
            grad[start:end] += w * dx * (1.0 - x[start + k:end + k])
        p_start = max(0, start + k)
        p_end = min(n, end + k)
        if p_start < p_end:
            grad[p_start:p_end] += w * dx * (-x[p_start - k:p_end - k])
    return grad


def eq_sum(x):
    return float(np.sum(x) - len(x) / 2.0)


def eq_sum_jac(x):
    return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]

            return make_feasible(res.x)
    except Exception:
        pass
    return h0
