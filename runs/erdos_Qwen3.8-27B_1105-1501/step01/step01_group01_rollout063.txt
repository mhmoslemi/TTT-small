def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_gradientS(h):
    """Compute gradient of each S_k with respect to h.
    Returns array of shape (2n-1, n) where gradS[k, p] = dS_k/dh_p
    """
    n = len(h)
    gradS = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        shift = k - (n-1)
        # S_k = sum of h[i]*(1-h[j]) where j = i - shift (i.e., i - j = shift)
        # For each pair (i, j) with i - j = shift:
        # d/dh_i: (1-h[j])
        # d/dh_j: -h[i]
        i_lo = max(0, shift)
        i_hi = min(n, n + shift)
        for i in range(i_lo, i_hi):
            j = i - shift
            gradS[k, i] += (1.0 - h[j])
            gradS[k, j] += (-h[i])
    return gradS

    n = len(h)
    h_cur = h.copy()
    best_score = official_score(h_cur)
    t_start = time.time()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        # Compute current S values
        dx = 2.0 / n
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        S = corr  # unscaled
        max_S = np.max(S)
        
        # Select active lags (top active_count)
        n_lags = 2*n - 1
        if active_count >= n_lags:
            active = np.arange(n_lags)
        else:
            # Include top active_count plus some spread
            top_idx = np.argsort(S)[::-1][:active_count]
            active = top_idx
        
        # Compute gradients for active lags
        gradS = compute_gradientS(h_cur)
        # gradS[k, p] = dS_k/dh_p
        
        # LP: minimize t
        # subject to: gradS[active, :] @ h <= t - S[active] + gradS[active,:] @ h_cur
        # i.e., gradS[active, :] @ h - t <= -S[active] + gradS[active,:] @ h_cur
        # sum(h) = n/2
        # 0 <= h <= 1
        
        A_active = gradS[active, :]  # shape (m, n)
        b_lin = -S[active] + A_active @ h_cur  # shape (m,)
        
        # Variables: [h (n), t (1)] total n+1
        # Objective: minimize t → c = [0,...,0, 1]
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Inequality: A_active @ h - t <= b_lin
        # → A_active @ h + (-1)*t <= b_lin
        A_ub = np.hstack([A_active, -np.ones((len(active), 1))])
        b_ub = b_lin
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success and res.x is not None:
                h_lp = res.x[:n]
                t_lp = res.x[n]
                # Direction
                d = h_lp - h_cur

                improved = False
                for alpha in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:

                    h_try = make_feasible(h_try)

                    if s_try < best_score - 1e-14:
                        h_cur = h_try
                        best_score = s_try
                        improved = True
                        break
                if not improved:
                    # Try small random perturbations as fallback

        except Exception:

        lin[ks[valid]] -= h[as_[valid]]
    
    # quadratic: (i,j) pair exists in lag k = i-j+n-1
    kq = i - j + n - 1
    if 0 <= kq < nl:
        quad[kq] = 1.0
    
    return lin, quad

def coord_descent(h, max_passes=5, rng=None, time_limit=None, t_start=None):
    """Coordinate descent with exact 1D line search per pair."""
    n = len(h)
    nl = 2 * n - 1
    h = h.copy()
    S = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(S))
    
    if rng is None:
        rng = np.random.default_rng(42)
    
    for p in range(max_passes):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        
        # Generate random permutation of pairs by shuffling indices
        idx = np.arange(n)
        for _ in range(n):
            j = rng.integers(0, n)
            idx[j], idx[0] = idx[0], idx[j]
        
        improved = False
        for _ in range(n * (n - 1) // 2):
            a = idx[rng.integers(0, n)]
            b = idx[rng.integers(0, n)]
            if a == b:
                continue
            
            d_lo = max(-h[a], h[b] - 1.0)
            d_hi = min(1.0 - h[a], h[b])
            if d_hi - d_lo < 1e-15:
                continue
            
            lin, quad = pair_lin_quad(h, a, b)
            
            # Ternary search
            lo, hi = d_lo, d_hi
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(S + lin * m1 + quad * m1 * m1)
                f2 = np.max(S + lin * m2 + quad * m2 * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-13:
                    break
            
            d_opt = (lo + hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[a] += d_opt
                h[b] -= d_opt
                S = S + lin * d_opt + quad * d_opt * d_opt
                new_max = np.max(S)
                if new_max < best_score:
                    best_score = new_max
                    improved = True
        
        if not improved:
            break
    
    h = make_feasible(h)
    return h

def smooth_lse_objective(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_lse_gradient(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2*n - 1) / (2*n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        a_lo = max(0, shift)
        a_hi = min(n, n + shift)
        for i in range(a_lo, a_hi):
            j = i - shift
            grad[i] += w * (1.0 - x[j])
            grad[j] += w * (-x[i])
    return grad

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def do_smooth_opt(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_lse_objective, h0, args=(beta,),
            jac=smooth_lse_gradient,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def dca_lp_step(h, active_frac=1.0, time_limit=15.0):
    """DCA step using LP linearization."""
    n = len(h)
    nl = 2*n - 1
    h_cur = h.copy()
    best_score = official_score(h_cur)
    t_start = time.time()
    
    for it in range(30):
        if time.time() - t_start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        S = corr
        
        # Select active lags
        n_active = max(10, int(active_frac * nl))
        top_idx = np.argsort(S)[::-1][:n_active]
        
        # Compute gradients for active lags
        # gradS[k, p] = sum over pairs (p, p-shift) of (1-h[p-shift]) - sum over pairs (p+shift, p) of h[p+shift]
        # For lag k, shift = k-(n-1):
        # gradS[k, a] += (1-h[a-shift]) for valid a (a is first index)
        # gradS[k, b] += -h[b+shift] for valid b (b is second index, a=b+shift)
        
        A = np.zeros((n_active, n))
        for row, k in enumerate(top_idx):
            shift = k - (n - 1)
            a_lo = max(0, shift)
            a_hi = min(n, n + shift)
            for a in range(a_lo, a_hi):
                b = a - shift
                A[row, a] += (1.0 - h_cur[b])
                A[row, b] += (-h_cur[a])
        
        b_lin = -S[top_idx] + A @ h_cur
        
        # LP: min t s.t. A@h - t <= b_lin, sum(h) = n/2, 0<=h<=1
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub = np.hstack([A, -np.ones((n_active, 1))])
        b_ub = b_lin
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            from scipy.optimize import linprog as _lp
            res = _lp(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
            if res.success:
                h_lp = res.x[:n]
                d = h_lp - h_cur
                nd = np.linalg.norm(d)
                if nd < 1e-14:
                    break
                improved = False
                for alpha in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1]:
                    h_try = make_feasible(h_cur + alpha * d)
                    s_try = official_score(h_try)
                    if s_try < best_score - 1e-14:
                        h_cur = h_try
                        best_score = s_try
                        improved = True
                        break
                if not improved:
                    break
        except Exception:
            break
    
    return h_cur, best_score

        for i in range(a_lo, a_hi):
            j = i - shift
            grad[i] += w * (1.0 - x[j])
            grad[j] += w * (-x[i])
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def do_smooth_opt(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_obj, h0, args=(beta,),
            jac=smooth_grad, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def binary_push(h, p=1.5):
    return make_feasible(h ** p)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(48) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)
    pn = len(parent)

    n_list = sorted(set([
        max(30, pn - 6), max(30, pn - 4), max(30, pn - 2),
        pn, pn + 1, pn + 2, pn + 3, pn + 4, pn + 5, pn + 6
    ]))
    n_list = [n for n in n_list if 32 <= n <= 66]

    # Phase 1: Coordinate descent from interpolated parent at each n
    for n in n_list:
        if time.time() > deadline - 80:
            break
        h_init = make_feasible(interp_to_n(parent, n))
        remaining = deadline - time.time() - 40
        if remaining < 5:
            break
        t_cd = time.time()
        max_pass = min(8, max(2, int(remaining / (0.3 * n * n))))
        h_cd = coord_descent(h_init, max_passes=max_pass, rng=rng,
                           time_limit=remaining * 0.5, t_start=t_cd)
        s_cd = official_score(h_cd)
        if s_cd < best_score:
            best_h = h_cd.copy()
            best_score = s_cd
            best_n = n
            # Continue improving this n
            h_cd = coord_descent(h_cd, max_passes=3, rng=rng,
                               time_limit=min(20, deadline - time.time() - 20),
                               t_start=time.time())
            s_cd = official_score(h_cd)
            if s_cd < best_score:
                best_h = h_cd.copy()
                best_score = s_cd

    # Phase 2: DCA + smooth on best
    if time.time() < deadline - 30:
        tl = min(20, deadline - time.time() - 15)
        h_dca, s_dca = dca_lp_step(best_h, time_limit=tl)
        if s_dca < best_score:
            best_h = h_dca.copy()
            best_score = s_dca
        h_cd = coord_descent(best_h, max_passes=3, rng=rng,
                           time_limit=min(15, deadline - time.time() - 10),
                           t_start=time.time())
        s_cd = official_score(h_cd)
        if s_cd < best_score:
            best_h = h_cd.copy()
            best_score = s_cd

    # Phase 3: Smooth SLSQP refinement with multiple betas
    if time.time() < deadline - 25:
        for beta in [30, 100, 300, 800]:
            if time.time() > deadline - 15:
                break
            h_sm = do_smooth_opt(best_h, beta, maxiter=150)
            s_sm = official_score(h_sm)
            if s_sm < best_score:
                best_h = h_sm.copy()
                best_score = s_sm
            h_cd = coord_descent(best_h, max_passes=2, rng=rng,
                               time_limit=min(8, deadline - time.time() - 10),
                               t_start=time.time())
            s_cd = official_score(h_cd)
            if s_cd < best_score:
                best_h = h_cd.copy()
                best_score = s_cd

    # Phase 4: Multi-start with perturbation and binary push
    if time.time() < deadline - 20:
        n_trials = min(25, int((deadline - time.time() - 10) / 8))
        for trial in range(n_trials):
            if time.time() > deadline - 10:
                break
            r = trial / max(1, n_trials)
            sigma = 0.005 + 0.06 * (1 - r)
            h_pert = make_feasible(best_h + sigma * rng.standard_normal(best_n))
            if trial % 3 == 0:
                h_pert = binary_push(h_pert, p=1.3 + 0.5 * rng.random())
            h_opt = coord_descent(h_pert, max_passes=2, rng=rng,
                                 time_limit=min(5, deadline - time.time() - 8),
                                 t_start=time.time())
            h_opt = do_smooth_opt(h_opt, 200, maxiter=80)
            h_opt = coord_descent(h_opt, max_passes=2, rng=rng,
                                 time_limit=min(3, deadline - time.time() - 5),
                                 t_start=time.time())
            s = official_score(h_opt)
            if s < best_score:
                best_h = h_opt.copy()
                best_score = s

    # Phase 5: Try different n from best
    if time.time() < deadline - 20:
        for dn in [-2, -1, 1, 2, 3]:
            n_new = best_n + dn
            if n_new < 32 or n_new > 66:
                continue
            if time.time() > deadline - 12:
                break
            h_try = make_feasible(interp_to_n(best_h, n_new))
            h_try = coord_descent(h_try, max_passes=3, rng=rng,
                                 time_limit=min(10, deadline - time.time() - 8),
                                 t_start=time.time())
            h_try = do_smooth_opt(h_try, 300, maxiter=100)
            h_try = coord_descent(h_try, max_passes=2, rng=rng,
                                 time_limit=min(5, deadline - time.time() - 5),
                                 t_start=time.time())
            s = official_score(h_try)
            if s < best_score:
                best_h = h_try.copy()
                best_score = s
                best_n = n_new

    # Phase 6: Final intensive coordinate descent
    if time.time() < deadline - 5:
        tl = max(3, deadline - time.time() - 3)
        h_final = coord_descent(best_h, max_passes=8, rng=rng,
                               time_limit=tl, t_start=time.time())
        s_final = official_score(h_final)
        if s_final < best_score:
            best_h = h_final.copy()
            best_score = s_final
        # One more smooth pass
        if time.time() < deadline - 3:
            h_fin2 = do_smooth_opt(best_h, 500, maxiter=100)
            s_fin2 = official_score(h_fin2)
            if s_fin2 < best_score:
                best_h = h_fin2.copy()
                best_score = s_fin2
            h_fin2 = coord_descent(best_h, max_passes=2, rng=rng,
                                  time_limit=min(3, deadline - time.time() - 1),
                                  t_start=time.time())
            s_fin2 = official_score(h_fin2)
            if s_fin2 < best_score:
                best_h = h_fin2.copy()
                best_score = s_fin2

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
