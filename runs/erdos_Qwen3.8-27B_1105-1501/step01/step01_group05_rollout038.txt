def golden_min(f, a, b, tol=1e-10, max_iter=50):
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2, min(f(c), f(d))

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    # Iteratively fix sum
    for _ in range(1000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            if np.max(room) < 1e-15:
                break
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            if h[idx] < 1e-15:
                break
            delta = min(-residual, h[idx])
            h[idx] -= delta
    # Final fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        # Distribute among all elements
        step = residual / n_target
        h = np.clip(h + step, 0.0, 1.0)

    A[i][r] is the coefficient of delta_i in the change of correlation at output index r.

    C and D are arrays of length 2n-1.

        return (delta_lo + delta_hi) / 2

    deltas = np.linspace(delta_lo, delta_hi, n_grid)
    vals = np.max(C[None, :] + deltas[:, None] * D[None, :], axis=1)
    best_idx = np.argmin(vals)

    elif best_idx == n_grid - 1:

import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(2000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            if np.max(room) < 1e-15:
                break
            idx = int(np.argmax(room))
            h[idx] += min(residual, room[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] < 1e-15:
                break
            h[idx] -= min(-residual, h[idx])
    return h


def compute_A_matrix(h):
    n = len(h)
    A = np.zeros((n, 2 * n - 1))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i1s = max(0, -k)
        i1e = min(n, n - k)
        if i1s < i1e:
            A[i1s:i1e, r] += (1.0 - h[i1s + k:i1e + k])
        i2s = max(0, k)
        i2e = min(n, n + k)
        if i2s < i2e:
            A[i2s:i2e, r] -= h[i2s - k:i2e - k]
    return A


def pair_search(C, D, dlo, dhi):
    """Minimize max(C + d*D) for d in [dlo, dhi]. Returns optimal d."""
    if dhi - dlo < 1e-14:
        return (dlo + dhi) / 2.0
    # Grid
    ng = 25
    deltas = np.linspace(dlo, dhi, ng)
    # Vectorized: C[None,:] + deltas[:,None] * D[None,:]
    vals = np.max(C[None, :] + deltas[:, None] * D[None, :], axis=1)
    bi = int(np.argmin(vals))
    # Golden section in nearest interval
    if bi == 0:
        a, b = dlo, deltas[1]
    elif bi == ng - 1:
        a, b = deltas[-2], dhi
    else:
        a, b = deltas[bi - 1], deltas[bi + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(C + c * D)
    fd = np.max(C + d * D)
    for _ in range(25):
        if a + b < 2 * c or a + b > 2 * d or False:
            break
    for _ in range(25):

            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = np.max(C + c * D)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = np.max(C + d * D)
    return (a + b) / 2.0


def pair_descent(h, time_limit=60.0, seed=42, max_passes=200):

    h = h.copy()
    n = len(h)
    current_score = official_score(h)
    t0 = time.perf_counter()
    
    for path in range(max_passes):
        if time.perf_counter() - t0 > time_limit:
            break

        cmax = float(np.max(C))
        A = compute_A_matrix(h)
        
        best_val = cmax
        best_i = -1
        best_j = -1
        best_d = 0.0
        
        # Try all pairs
        for i in range(n):
            for j in range(i + 1, n):
                S = h[i] + h[j]
                L = max(0.0, S - 1.0)
                U = min(1.0, S)
                if U - L < 1e-12:
                    continue
                dlo = L - h[i]
                dhi = U - h[i]
                D = A[i] - A[j]
                dopt = pair_search(C, D, dlo, dhi)
                val = float(np.max(C + dopt * D))
                if val < best_val:
                    best_val = val
                    best_i = i
                    best_j = j
                    best_d = dopt
        
        if best_i < 0 or best_val >= cmax - 1e-13:
            break
        
        # Apply
        S = h[best_i] + h[best_j]
        h[best_i] = h[best_i] + best_d
        h[best_j] = S - h[best_i]
        h[best_i] = np.clip(h[best_i], 0.0, 1.0)
        h[best_j] = np.clip(h[best_j], 0.0, 1.0)
        
        # Verify sum
        residual = n / 2.0 - np.sum(h)
        if abs(residual) > 1e-10:
            h = repair_vector(h, n)
        
        new_score = official_score(h)
        if new_score < current_score:
            current_score = new_score
        else:
            # Shouldn't happen, but safety
            break
    
    return h, official_score(h)

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(5000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            if np.max(room) < 1e-15:
                break
            idx = int(np.argmax(room))
            h[idx] += min(residual, room[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] < 1e-15:
                break
            h[idx] -= min(-residual, h[idx])
    return h


def compute_A_matrix(h):
    n = len(h)
    A = np.zeros((n, 2 * n - 1))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i1s = max(0, -k)
        i1e = min(n, n - k)
        if i1s < i1e:
            A[i1s:i1e, r] += (1.0 - h[i1s + k:i1e + k])
        i2s = max(0, k)
        i2e = min(n, n + k)
        if i2s < i2e:
            A[i2s:i2e, r] -= h[i2s - k:i2e - k]
    return A


def pair_search(C, D, dlo, dhi):
    if dhi - dlo < 1e-14:
        return (dlo + dhi) / 2.0
    ng = 21
    deltas = np.linspace(dlo, dhi, ng)
    full = C[None, :] + deltas[:, None] * D[None, :]

    m = len(h_src)
    if m == 0 or m == 1:
        return repair_vector(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, h_src, left=h_src[0], right=h_src[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n_target)


def slsqp_polish(h, maxiter=60):
    n = len(h)
    dx = 2.0 / n
    C0 = np.correlate(h, 1.0 - h, mode="full") * dx
    score0 = float(np.max(C0))

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq(x):
        hx = x[:n]
        return x[n] - np.correlate(hx, 1.0 - hx, mode="full") * dx

    def ineq_jac(x):
        hx = x[:n]
        Am = compute_A_matrix(hx)
        Jl = np.zeros((2 * n - 1, n + 1))
        Jl[:, n] = 1.0
        Jl[:, :n] = -dx * Am.T
        return Jl

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    x0 = np.concatenate([h, [score0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
    ]
    try:
        res = optimize.minimize(
            obj, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_out = repair_vector(res.x[:n], n)
            s = official_score(h_out)
            if s < score0 - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, score0
