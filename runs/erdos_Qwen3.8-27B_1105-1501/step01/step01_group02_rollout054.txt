import numpy as np
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_gradient(h, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = corr * dx / temperature
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient of smooth max w.r.t. h[i]
    # d(corr_s)/dh[i] = (1-h[i+s]) - h[i-s] for valid indices
    # s ranges from -(n-1) to n-1
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # Direct term: (1-h[i+s]) for i where 0<=i<n and 0<=i+s<n
        # h[i+s] is valid for i in [max(0,-s), min(n-1, n-1-s)]
        a = max(0, -s)
        b = min(n-1, n-1-s)
        if a <= b:
            grad[a:b+1] += w * (1.0 - h[a+s:b+s+1])
        # Indirect term: -h[i-s] for i where 0<=i<n and 0<=i-s<n
        # h[i-s] is valid for i in [max(0,s), min(n-1, n-1+s)]
        a2 = max(0, s)
        b2 = min(n-1, n-1+s)
        if a2 <= b2:
            grad[a2:b2+1] -= w * h[a2-s:b2-s+1]
    return grad

def thue_morse_h(n, rng):
    """Create a Thue-Morse based initial solution."""
    # Thue-Morse: t(x) = popcount(x) % 2
    t = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # We need sum = n/2
    num_ones = int(np.sum(t))
    target = n // 2
    if num_ones > target:
        # Flip some 1s to 0s
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t

def random_balanced_h(n, rng):
    """Create a random balanced sequence."""
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:n//2]] = 1.0
    return h

def pair_local_search(h, target_sum, max_sweeps=1000, time_limit=None, t0=None):
    """Local search using pairwise moves."""
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        
        improved = False
        # Compute gradient to guide pair selection
        grad = smooth_max_gradient(h, temperature=0.03)
        
        # Sort indices by gradient (we want to move from high grad to low grad)
        order = np.argsort(-grad)  # descending gradient
        
        # Try top pairs
        for i_idx in range(n):
            i = order[i_idx]
            if h[i] < 1e-10:
                continue
            for j_idx in range(i_idx+1, n):
                j = order[j_idx]
                if h[j] > 1.0 - 1e-10:
                    continue
                if grad[i] - grad[j] < 1e-10:
                    continue
                
                # Try moving mass from i to j
                max_delta = min(h[i], 1.0 - h[j])
                # Try a few delta values
                for frac in [1.0, 0.5, 0.25]:
                    delta = max_delta * frac
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-12:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] += delta
                        h[j] -= delta
                        break  # If full move doesn't help, smaller likely won't either
        
        if not improved:
            break
    
    return best_h, best_c5

def simulated_annealing(h, target_sum, t0, time_limit):
    """Simulated annealing with pairwise moves."""
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    temperature = 0.05
    
    while time.time() < time_limit:
        # Pick random pair
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        # Decide direction
        if h[i] > 0.5 or (h[i] > h[j] and np.random.random() > 0.5):
            # Move from i to j
            max_delta = min(h[i], 1.0 - h[j])
        else:
            # Move from j to i
            i, j = j, i
            max_delta = min(h[i], 1.0 - h[j])
        
        if max_delta < 1e-10:
            continue
        
        delta = max_delta * np.random.random()
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        delta_c = new_c5 - current_c5
        
        if delta_c < 0 or np.random.random() < np.exp(-delta_c / temperature):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        
        # Anneal
        temperature = max(0.001, temperature * 0.9995)
        if temperature <= 0.0015:
            # Restart
            temperature = 0.05
            h = best_h + np.random.randn(n) * 0.05
            h = project_onto_constraint(h, target_sum)
            current_c5 = compute_c5(h)
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values

    h0 = project_onto_constraint(h0, target_sum)

    # Phase 2: Simulated annealing (~40% budget)

    if c5 < best_c5:

        best_h = h.copy()
    
    # Phase 3: Final local search (~15% budget)

    h, c5 = pair_local_search(best_h, target_sum, time_limit=ph3_end)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Final: also try from other starting points if time permits
    # (This is a simple check - if we have lots of time left)

**Key insight for efficiency**: Instead of calling np.correlate from scratch each time, I can track the correlation array and update it incrementally.

When we swap h[i]: 1→0 and h[j]: 0→1:

```python
def update_corr_after_swap(corr, h, i, j, n):
    """Update correlation array after swapping h[i]:1->0, h[j]:0->1.
    Returns the new correlation array (modified in place for efficiency)."""
    # For each shift s from -(n-1) to n-1:
    # delta_s = (h[i+s] - h[j+s]) + (h[j-s] - h[i-s])
    # where h[k]=0 if not 0<=k<n
    # The shift s corresponds to index s + (n-1) in the corr array
    for s in range(-(n-1), n):
        # Get h[i+s], h[j+s], h[i-s], h[j-s] with boundary = 0
        h_is = h[i+s] if 0 <= i+s < n else 0.0
        h_js = h[j+s] if 0 <= j+s < n else 0.0
        h_ims = h[i-s] if 0 <= i-s < n else 0.0
        h_jms = h[j-s] if 0 <= j-s < n else 0.0
        delta = (h_is - h_js) + (h_jms - h_ims)
        if delta != 0:
            corr[s + (n-1)] += delta
    return corr
```

This is O(n) per swap, and we can also quickly check if the max changes:
- If the old max was M at position p, and delta_p ≤ 0, then the new max is ≤ M (could be less)
- If delta_p > 0, the new max could be larger

For the acceptance criterion in SA, we need the new max. We can compute it as:
new_max = max(corr[s + (n-1)] for all s) = np.max(corr)

So the update is O(n) + O(n) for the max = O(n) total. Much faster than O(n^2) for np.correlate!
