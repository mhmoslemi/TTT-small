import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(v, target_sum):
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def binary_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr)) * dx

def greedy_binary(n, k, tiebreak_seed=0):

def multi_swap_sa(n, k, t_start, n_iter, rng, t0, time_limit, swap_count=2):
    """SA with multiple swaps per iteration"""
    h = make_binary(n, k, rng)
    best_h = h.copy()
    best_c5 = binary_c5(h)
    cur_c5 = best_c5
    t_end = t_start * 0.0001
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        progress = it / max(n_iter, 1)
        temp = t_start * (t_end / t_start) ** progress
        
        # Perform swap_count random swaps
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Pick swap_count pairs
        swap_i = ones[rng.integers(swap_count, len(ones))]
        swap_j = zeros[rng.integers(swap_count, len(zeros))]
        # Make sure no duplicates
        swap_i = np.unique(swap_i)
        swap_j = np.unique(swap_j)
        m = min(len(swap_i), len(swap_j))
        swap_i = swap_i[:m]
        swap_j = swap_j[:m]
        
        for i, j in zip(swap_i, swap_j):
            h[i], h[j] = h[j], h[i]
        
        new_c5 = binary_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (temp > 1e-12 and rng.random() < np.exp(-delta / temp)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Revert
            for i, j in zip(swap_i, swap_j):
                h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

def improve_continuous(h, n, target_sum, t0, time_limit, beta=1000):
    """Improve a continuous solution using SLSQP"""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_vals = corr * dx
        c_max = float(np.max(c_vals))
        shifted = np.clip(beta * (c_vals - c_max), -500.0, 500.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_vals = corr * dx
        c_max = float(np.max(c_vals))
        shifted = np.clip(beta * (c_vals - c_max), -500.0, 500.0)
        exp_s = np.exp(shifted)
        weights = exp_s / float(np.sum(exp_s))
        
        g = np.zeros(n)
        for d in range(-(n-1), n):
            k_idx = d + (n - 1)
            w = weights[k_idx] * dx
            if abs(w) < 1e-15:
                continue
            if d >= 0:
                i_range = range(0, n - d)
                for i in i_range:
                    g[i] += w * (1.0 - x[i + d])
                for i in range(d, n):
                    g[i] -= w * x[i - d]
            else:
                d2 = -d
                for i in range(d2, n):
                    g[i] += w * (1.0 - x[i - d2])
                for i in range(0, n - d2):
                    g[i] -= w * x[i + d2]
        return g
    
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', jac=grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-10, 'iprint': 0})
    
    h_opt = project_feasible(res.x, target_sum)
    return h_opt

def local_search_continuous(h, n, target_sum, t0, time_limit, n_rounds=50):
    """Local search with random perturbations"""
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    for round_i in range(n_rounds):

        # Random perturbation
        scale = 0.01 / (1.0 + 0.1 * round_i)

        c5 = eval_c5(h_pert)
        
        if c5 < best_c5:

            best_h = h_pert.copy()

def eval_c5(h):
