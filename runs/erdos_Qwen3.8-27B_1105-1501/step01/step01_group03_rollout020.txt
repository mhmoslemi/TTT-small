```python
import numpy as np
import time

def make_feasible(z):
    # Project z onto {0<=h<=1, sum(h)=n/2}
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    # Binary search for offset
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any residual
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        # Distribute error
        if err > 0:
            order = np.argsort(h)[::-1]  # most room first
            for i in order:
                if err <= 1e-14: break
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
        else:
            order = np.argsort(h)  # least first
            for i in order:
                if err >= -1e-14: break
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_C_L_Q(h, v):
    """For direction v (sum=0), compute C_k, L_k, Q_k for all lags k."""
    n = len(h)
    n_lags = 2 * n - 1
    C = np.zeros(n_lags)
    L = np.zeros(n_lags)
    Q = np.zeros(n_lags)
    
    for k in range(-(n-1), n):
        start = max(0, -k)
        end = min(n, n - k)  # exclusive
        # C_k = sum h[i] * (1-h[i+k]) for i in [start, end)
        C[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start+k:end+k]))
        # L_k = sum v[i]*(1-h[i+k]) - h[i]*v[i+k]
        L[k + n - 1] = np.sum(v[start:end] * (1.0 - h[start+k:end+k]) - h[start:end] * v[start+k:end+k])
        # Q_k = -sum v[i]*v[i+k]
        Q[k + n - 1] = -np.sum(v[start:end] * v[start+k:end+k])
    
    return C, L, Q

def feasible_interval(h, v):
    """Find [s_min, s_max] such that h + s*v stays in [0,1]."""
    s_min = 0.0
    s_max = 0.0
    for i in range(len(h)):
        if v[i] > 1e-15:
            s_min = max(s_min, -h[i] / v[i])
            s_max = min(s_max, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-15:
            s_min = max(s_min, (1.0 - h[i]) / v[i])
            s_max = min(s_max, -h[i] / v[i])
    return s_min, s_max

def line_search(h, v, C, L, Q, K=15):
    """Exact 1D line search. Returns best s and best score (raw, before dx)."""
    n = len(h)
    s_min, s_max = feasible_interval(h, v)
    if s_max - s_min < 1e-14:
        return 0.0, np.max(C)
    
    n_lags = 2 * n - 1
    # Find active lags (top K by C value)
    # But we should include lags that might become active
    # For safety, use top 2*K
    top_k = min(2*K, n_lags)
    active_idx = np.argsort(C)[::-1][:top_k]
    
    candidates = set()
    candidates.add(s_min)
    candidates.add(s_max)
    candidates.add(0.0)
    
    # Individual quadratic minima
    for idx in active_idx:
        if Q[idx] > 1e-15:
            s0 = -L[idx] / (2.0 * Q[idx])
            if s_min < s0 < s_max:
                candidates.add(s0)
    
    # Pairwise intersections of linear parts (where Q≈0)
    # and quadratic-linear intersections
    for a in range(len(active_idx)):
        for b in range(a+1, len(active_idx)):
            ia, ib = active_idx[a], active_idx[b]
            # F_ia(s) = F_ib(s)
            # (C_a - C_b) + (L_a - L_b)*s + (Q_a - Q_b)*s^2 = 0
            a2 = Q[ia] - Q[ib]
            b2 = L[ia] - L[ib]
            c2 = C[ia] - C[ib]
            if abs(a2) < 1e-15:
                # Linear: b2*s + c2 = 0
                if abs(b2) > 1e-15:
                    s = -c2 / b2
                    if s_min < s < s_max:
                        candidates.add(s)
            else:
                disc = b2*b2 - 4.0*a2*c2
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    s1 = (-b2 + sqrt_disc) / (2.0 * a2)
                    s2 = (-b2 - sqrt_disc) / (2.0 * a2)
                    for s in [s1, s2]:
                        if s_min < s < s_max:
                            candidates.add(s)
    
    # Add some uniform grid points for robustness
    n_grid = 10
    grid = np.linspace(s_min, s_max, n_grid)
    candidates.update(grid.tolist())
    
    # Evaluate at all candidates
    best_s = 0.0
    best_val = np.max(C)  # at s=0
    for s in candidates:
        s = max(s_min, min(s_max, s))  # clamp
        # F_k(s) = C_k + L_k*s + Q_k*s^2 for all k
        F_s = C + L * s + Q * s * s
        val = np.max(F_s)
        if val < best_val:
            best_val = val
            best_s = s
    
    return best_s, best_val
```

```python
                if s_max - s_min < 1e-13:
                    continue
                
                # Compute L_k for this direction
                L = np.zeros(n_lags)
                for k in range(-(n-1), n):
                    lo_k = max(0, -k)
                    hi_k = min(n, n - k)
                    lv = 0.0
                    if lo_k <= i < hi_k and 0 <= i+k < n:
                        lv += 1.0 - h[i+k]
                    if lo_k <= j < hi_k and 0 <= j+k < n:
                        lv -= 1.0 - h[j+k]
                    ik = i - k
                    if lo_k <= ik < hi_k and 0 <= ik < n:
                        lv -= h[ik]
                    jk = j - k
                    if lo_k <= jk < hi_k and 0 <= jk < n:
                        lv += h[jk]
                    L[offset + k] = lv
                
                # Q_k: only nonzero at k=0, k=j-i, k=i-j
                Q = np.zeros(n_lags)
                Q[offset] = -2.0  # k=0
                d = j - i
                # k=d: Q = I(i in vr_d)
                lo_d = max(0, -d)
                hi_d = min(n, n - d)
                if lo_d <= i < hi_d:
                    L[offset + d]  # just to make sure index is valid
                    Q[offset + d] = 1.0
                    # Wait, I need to recheck. Q_d = -[v_i*v_{i+d}] where i is in valid range
                    # v_i*v_{i+d}: nonzero when i=p or i=q
                    # i=p: v_p*v_{p+d} = 1*v_j = -1 (if p+d=q, i.e., d=q-p)
                    #   But this requires p in valid range.
                    # i=q: v_q*v_{q+d} = -1*v_{q+d}. q+d = q+q-p = 2q-p.
                    #   v_{2q-p} = 0 unless 2q-p=p (q=p) or 2q-p=q (q=0)
                    # So for k=d=q-p: Q_d = -(1*(-1)*I(p in vr) + (-1)*0*I(q in vr)) = I(p in vr)
                    
                # k=-d: Q = -I(j in vr_{-d})
                lo_nd = max(0, d)  # max(0, -(-d)) = max(0, d)
                hi_nd = min(n, n + d)  # min(n, n-(-d)) = min(n, n+d) = n
                if lo_nd <= j < hi_nd:
                    Q[offset - d] = -1.0
                
                # 1D optimization: minimize max_k (C_k + L_k*s + Q_k*s^2) over [s_min, s_max]
                # Candidates: endpoints, breakpoints of top-K lines, vertex of convex quadratic
                candidates = [s_min, s_max, 0.0]
                
                # Vertex of convex quadratic (only Q_d might be positive)
                for idx in range(n_lags):
                    if Q[idx] > 1e-15:
                        s0 = -L[idx] / (2.0 * Q[idx])
                        if s_min < s0 < s_max:
                            candidates.append(s0)
                
                # Breakpoints of top-K linear functions
                K = min(25, n_lags)
                top_idx = np.argsort(C)[::-1][:K]
                for a in range(len(top_idx)):
                    ia = top_idx[a]
                    for b in range(a+1, len(top_idx)):
                        ib = top_idx[b]
                        a2 = Q[ia] - Q[ib]
                        b2 = L[ia] - L[ib]
                        c2 = C[ia] - C[ib]
                        if abs(a2) < 1e-15:
                            if abs(b2) > 1e-15:
                                s = -c2 / b2
                                if s_min < s < s_max:
                                    candidates.append(s)
                        else:
                            disc = b2*b2 - 4.0*a2*c2
                            if disc >= 0:
                                sq = np.sqrt(disc)
                                s1 = (-b2 + sq) / (2.0*a2)
                                s2 = (-b2 - sq) / (2.0*a2)
                                for s in [s1, s2]:
                                    if s_min < s < s_max:
                                        candidates.append(s)
                
                # Grid
                candidates.extend(np.linspace(s_min, s_max, 11).tolist())
                
                # Evaluate
                best_s = 0.0
                best_val = float(np.max(C))
                for s in candidates:
                    s = max(s_min, min(s_max, s))
                    val = float(np.max(C + L*s + Q*s*s))
                    if val < best_val:
                        best_val = val
                        best_s = s
                
                if best_val < float(np.max(C)) - 1e-12:
                    h[i] += best_s
                    h[j] -= best_s
                    improved = True
        
        # After full pass, check score
        new_score = compute_score(h)
        if new_score < score - 1e-12:
            score = new_score
        else:
            # Roll back if score didn't improve (shouldn't happen with exact line search)
            pass
        if not improved:
            break
    
    return h, compute_score(h)
```

```python
        for m in range(n):
            val = 0.0
            if lo_k <= m < hi_k and 0 <= m+k < n:

            mk = m - k
            if lo_k <= mk < hi_k and 0 <= mk < n:
                val += -h[mk]
            G[k + n - 1, m] = val
```
