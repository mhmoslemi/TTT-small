```python
def find_optimal_t(V, c, dx, t_min, t_max, n_grid=50):
    if t_max - t_min < 1e-15:
        return 0.0
    
    # Grid search
    t_grid = np.linspace(t_min, t_max, n_grid)
    f_vals = np.max(V[None, :] + dx * t_grid[:, None] * c[None, :], axis=1)
    best_idx = np.argmin(f_vals)
    best_t = t_grid[best_idx]
    best_f = f_vals[best_idx]
    
    # Refine: check breakpoints near best_t
    # Find top lags at best_t
    vals_at_best = V + dx * best_t * c
    top_lags = np.argsort(vals_at_best)[::-1][:8]
    
    # Check pairwise intersections among top lags
    for i in range(len(top_lags)):
        for j in range(i+1, len(top_lags)):
            pi, pj = top_lags[i], top_lags[j]
            if abs(c[pi] - c[pj]) < 1e-15:
                continue
            t_cross = (V[pj] - V[pi]) / (dx * (c[pi] - c[pj]))
            if t_min - 1e-12 <= t_cross <= t_max + 1e-12:
                t_cross = np.clip(t_cross, t_min, t_max)
                f_cross = np.max(V + dx * t_cross * c)
                if f_cross < best_f - 1e-14:
                    best_f = f_cross
                    best_t = t_cross
    
    # Also check if moving slightly from best_t helps (in case the optimum is between grid points)
    # Subgradient check
    sg = dx * c[np.isclose(V + dx * best_t * c, best_f, atol=1e-10)]
    if len(sg) > 0:
        if sg.min() > 1e-10:  # all positive slope, try smaller t
            t_try = max(t_min, best_t - 0.1 * (t_max - t_min))
        elif sg.max() < -1e-10:  # all negative slope, try larger t
            t_try = min(t_max, best_t + 0.1 * (t_max - t_min))
        else:
            t_try = None  # at optimum
    
    if t_try is not None:
        # Evaluate at t_try and nearby
        for t_cand in [t_try, (t_try + best_t) / 2]:
            f_cand = np.max(V + dx * t_cand * c)
            if f_cand < best_f - 1e-14:
                best_f = f_cand
                best_t = t_cand
    
    # Check boundaries
    for t_cand in [t_min, t_max]:
        f_cand = np.max(V + dx * t_cand * c)
        if f_cand < best_f - 1e-14:
            best_f = f_cand
            best_t = t_cand
    
    return best_t, best_f
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - np.sum(z)
    if residual > 0:
        # Need to add 'residual' total
        order = np.argsort(1.0 - z)
        for idx in order:
            add = min(residual, 1.0 - z[idx])
            if add <= 0:
                break
            z[idx] += add
            residual -= add
            if residual < 1e-15:
                break
    elif residual < 0:

            if sub <= 0:
                break
            z[idx] -= sub
            residual += sub
            if -residual < 1e-15:
                break

def compute_pair_coeff(h, a, b, n):
```
