import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def full_pair_sweep(h, target_sum, n_grid=21):
    """Systematic sweep over all pairs with grid search. Returns (h, c5, num_improvements)."""
    n = len(h)
    c5 = compute_c5(h)
    improvements = 0
    for i in range(n):
        for j in range(i + 1, n):
            # Feasible range for t: h[i]+t in [0,1], h[j]-t in [0,1]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-10:
                continue
            # Grid search
            best_t = 0.0
            best_c5 = c5
            # Include current point
            for val in np.linspace(lo, hi, n_grid):
                h[i] += val
                h[j] -= val
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_t = val
                h[i] -= val
                h[j] += val
            if best_t != 0.0:
                h[i] += best_t
                h[j] -= best_t
                c5 = best_c5
                improvements += 1
    return h, c5, improvements

        dj[idx4] -= h[j + np.arange(s_lo4, s_hi4+1)]

    new_corr = corr_base + t * (di - dj)
