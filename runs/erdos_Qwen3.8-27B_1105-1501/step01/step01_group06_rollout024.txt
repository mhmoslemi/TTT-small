def project_to_target(v, target=None):
    if target is None:
        target = len(v) / 2.0
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(len(v))
    if target >= len(v):
        return np.ones(len(v))
    # Water-filling
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Exact repair
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add mass
            order = np.argsort(-x)  # add to smallest first
            for i in order:
                add = min(1.0 - x[i], residual)
                x[i] += add
                residual -= add
                if residual < 1e-12:
                    break
        else:
            # Need to remove mass
            order = np.argsort(x)  # remove from largest first
            for i in order:
                rem = min(x[i], -residual)
                x[i] -= rem
                residual += rem
                if residual > -1e-12:
                    break
    return x

def lag_grad_vec(h, n, d):

        g[:n-g_] -= h[g_:]
    return g

    offset = n - 1

    n_vars = n + 2  # p (n), r (1), tau (1)
    A_rows = []
    b_rows = []

        g = lag_grad_vec(h, n, d)
        row = np.zeros(n_vars)
        row[:n] = g
        row[n+1] = -1.0  # -tau
        A_rows.append(row)
        b_rows.append(-scores[j])
    
    # |p_i| <= r constraints
    for i in range(n):
        row1 = np.zeros(n_vars)
        row1[i] = 1.0
        row1[n] = -1.0
        A_rows.append(row1)
        b_rows.append(0.0)
        
        row2 = np.zeros(n_vars)
        row2[i] = -1.0
        row2[n] = -1.0
        A_rows.append(row2)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize tau + lam * r
    c = np.zeros(n_vars)
    c[n] = lam
    c[n+1] = 1.0  # tau
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:

            hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, radius))  # r
    bounds.append((-10.0, 10.0))  # tau free-ish

import numpy as np
from scipy.optimize import linprog
import time


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_to_target(v, target):
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(len(v))
    if target >= len(v):
        return np.ones(len(v))
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(x)
            for i in order:
                add = min(1.0 - x[i], residual)
                x[i] += add
                residual -= add
                if residual < 1e-14:
                    break
        else:
            order = np.argsort(-x)
            for i in order:
                rem = min(x[i], -residual)
                x[i] -= rem
                residual += rem
                if residual > -1e-14:
                    break
    return x


def lag_grad_vec(h, n, d):
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[:n - d] += (1.0 - h[d:])
        g[d:] -= h[:n - d]
    else:
        g_ = -d
        g[g_:] += (1.0 - h[:n - g_])
        g[:n - g_] -= h[g_:]
    return g


def compute_scores(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def get_lp_direction(h, n, radius, lam):
    scores = compute_scores(h, n)
    offset = n - 1
    n_vars = n + 2
    A_list = []
    b_list = []

    for j in range(2 * n - 1):
        d = j - offset
        g = lag_grad_vec(h, n, d)
        row = np.zeros(n_vars)
        row[:n] = g
        row[n + 1] = -1.0
        A_list.append(row)
        b_list.append(-scores[j])

    I = np.eye(n)
    A_list.append(np.hstack([I, -np.ones(n), np.zeros(1)]))
    b_list.extend([0.0] * n)
    A_list.append(np.hstack([-I, -np.ones(n), np.zeros(1)]))
    b_list.extend([0.0] * n)

    A_ub = np.array(A_list)
    b_ub = np.array(b_list)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    c = np.zeros(n_vars)
    c[n] = lam
    c[n + 1] = 1.0

    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i] + 1e-10)
        hi_i = min(radius, 1.0 - h[i] - 1e-10)
        if lo_i > hi_i:
            lo_i = 0.0
            hi_i = 0.0
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, radius))
    bounds.append((-100.0, 100.0))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='revised simplex')
        except Exception:
            return None

    if res.status == 0:
        p = res.x[:n].copy()
        p -= np.mean(p)
        return p
    return None


def subgradient_direction(h, n, radius):
    scores = compute_scores(h, n)
    max_score = float(np.max(scores))
    offset = n - 1
    active = scores >= max_score - 0.1 * max_score
    g = np.zeros(n)
    for j in np.where(active)[0]:
        d = j - offset
        g += lag_grad_vec(h, n, d)
    p = -g
    p -= np.mean(p)
    mx = float(np.max(np.abs(p)))
    if mx < 1e-15:
        return None
    p /= mx
    p *= radius
    return p


def do_line_search(h, p, alpha_max, n_grid=21, n_golden=20):
    n = len(h)
    if alpha_max <= 1e-14:
        return 0.0, exact_c5(h)
    alphas = np.linspace(0.0, alpha_max, n_grid)
    best_c5 = float('inf')
    best_alpha = 0.0
    best_idx = 0
    for i, a in enumerate(alphas):
        v = exact_c5(h + a * p)
        if v < best_c5:
            best_c5 = v
            best_alpha = a
            best_idx = i
    if 0 < best_idx < n_grid - 1:
        lo_a = alphas[best_idx - 1]
        hi_a = alphas[best_idx + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(n_golden):
            m1 = hi_a - gr * (hi_a - lo_a)
            m2 = lo_a + gr * (hi_a - lo_a)
            f1 = exact_c5(h + m1 * p)
            f2 = exact_c5(h + m2 * p)
            if f1 < f2:
                hi_a = m2
            else:
                lo_a = m1
            mid = (lo_a + hi_a) / 2.0
            fm = exact_c5(h + mid * p)
            if fm < best_c5:
                best_c5 = fm
                best_alpha = mid
    return best_alpha, best_c5


def local_lp_optimize(h, n, target, deadline, init_radius=0.05):
    best_h = h.copy()
    best_c5 = exact_c5(h)
    radius = init_radius
    lam = 5.0
    stall = 0
    iters = 0
    use_subgrad = False

    while time.time() < deadline:
        if not use_subgrad:
            p = get_lp_direction(h, n, radius, lam)
        else:
            p = subgradient_direction(h, n, radius)

        iters += 1

        if p is None or float(np.max(np.abs(p))) < 1e-14:
            radius *= 0.5
            stall += 1
            if stall > 8:
                rng_l = np.random.default_rng(iters * 7919)
                h = project_to_target(h + rng_l.normal(0, 0.01, n), target)
                c5 = exact_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                radius = init_radius
                stall = 0
                if radius < 0.01:
                    use_subgrad = True
            continue

        pos = p > 1e-14
        neg = p < -1e-14
        alpha_max = 1e10
        if np.any(pos):
            alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
        if np_any_neg := np.any(neg):
            alpha_max = min(alpha_max, float(np.min(h[neg] / (-p[neg]))))
        alpha_max *= 0.999

        if alpha_max < 1e-14:
            radius *= 0.5
            stall += 1
            continue

        best_alpha, c5_new = do_line_search(h, p, alpha_max)

        if c5_new < best_c5 - 1e-15:
            h_new = project_to_target(h + best_alpha * p, target)
            c5_v = exact_c5(h_new)
            if c5_v < best_c5:
                h = h_new
                best_h = h.copy()
                best_c5 = c5_v
                radius = min(radius * 1.25, 0.5)
                stall = 0
                use_subgrad = False
            else:
                radius *= 0.7
                stall += 1
        else:
            radius *= 0.7
            stall += 1
            if stall > 12:
                rng_l = np.random.default_rng(iters * 104729)
                h = project_to_target(h + rng_l.normal(0, 0.015, n), target)
                radius = init_radius
                stall = 0

    return best_h, best_c5


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, left=0.0, right=0.0)
    return project_to_target(h_new, n_new / 2.0)


def binary_push(h, n, target, gamma):
    h_new = h.copy()
    h_new[h_new < 0.5] = np.maximum(h_new[h_new < 0.5] - gamma, 0.0)
    h_new[h_new >= 0.5] = np.minimum(h_new[h_new >= 0.5] + gamma, 1.0)
    return project_to_target(h_new, target)
