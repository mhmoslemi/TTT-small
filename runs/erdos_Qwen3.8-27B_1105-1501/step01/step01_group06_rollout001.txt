import numpy as np
from scipy.optimize import linprog, minimize

def exact_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_target(v, target):
    """Project v onto {0≤x≤1, sum(x)=target} using water-filling."""
    n = len(v)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Exact repair
    residual = target - float(np.sum(x))
    if abs(residual) < 1e-12:
        return x
    if residual > 0:
        # Add to entries not at 1
        idx = np.where(x < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return x
        add = residual / len(idx)
        for i in idx:
            room = 1.0 - x[i]
            x[i] += min(add, room)
            residual -= min(add, room)
            if abs(residual) < 1e-14:
                break
        # Final fix
        if abs(residual) > 1e-14:
            x[idx[0]] += residual
            x[idx[0]] = min(max(x[idx[0]], 0.0), 1.0)
    else:
        # Subtract from entries not at 0
        idx = np.where(x > 1e-12)[0]
        if len(idx) == 0:
            return x
        sub = -residual / len(idx)
        for i in idx:
            x[i] -= min(sub, x[i])
            residual += min(sub, x[i])
            if abs(residual) < 1e-14:
                break
        if abs(residual) > 1e-14:
            x[idx[0]] -= residual
            x[idx[0]] = min(max(x[idx[0]], 0.0), 1.0)

def compute_grads(h, top_k=None):

    Returns: scores (2n-1 array), grads (K x n array), selected_indices"""
    n = len(h)

    scores = corr.copy()
    
    if top_k is not None and top_k < 2*n - 1:

        idx = np.argsort(scores)[::-1][:top_k]
    else:

    grads = np.zeros((len(idx), n))

        g = np.zeros(n)
        if k >= 0:

            # d/dh[j]: as first factor (j=n+k, n=j-k, need 0≤j-k≤n-1-k → k≤j≤n-1): (1-h[j-k])
            #          as second factor (n=j, need 0≤j≤n-1-k): -h[j+k]
            j_range1 = np.arange(k, n)  # j from k to n-1

            j_range2 = np.arange(0, n - k)  # j from 0 to n-1-k

            j_range2 = np.arange(g_lag, n)  # j from g to n-1
            g[j_range2] -= h[j_range2 - g_lag]

def lp_direction(h, radius, top_k=50, mu=5.0, max_iter=200):
    """Find LP direction to reduce max lag score.
    Returns: p (direction), tau_pred (predicted new max)"""
    n = len(h)
    scores, grads, idx = compute_grads(h, top_k=top_k)
    
    # LP: minimize tau + mu*r
    # s.t. score_l + grad_l · p <= tau  →  grad_l · p - tau <= -score_l
    #      sum(p) = 0
    #      -r <= p_i <= r
    #      0 <= h_i + p_i <= 1
    #      0 <= r <= radius
    # Variables: [p_0, ..., p_{n-1}, r, tau]
    nvar = n + 2
    c = np.zeros(nvar)
    c[n] = mu      # coefficient for r
    c[n+1] = 1.0   # coefficient for tau
    
    # Inequality constraints: A_ub @ x <= b_ub
    # grad_l · p - tau <= -score_l  for each selected lag
    A_ub_rows = []
    b_ub_rows = []
    for ii in range(len(idx)):
        row = np.zeros(nvar)
        row[:n] = grads[ii]   # grad_l
        row[n+1] = -1.0       # -tau
        A_ub_rows.append(row)
        b_ub_rows.append(-scores[ii])
    
    # -r <= p_i → p_i + r >= 0 → -p_i - r <= 0
    # r <= p_i → r - p_i <= 0
    for i in range(n):
        row = np.zeros(nvar)
        row[i] = -1.0
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(0.0)
        
        row2 = np.zeros(nvar)
        row2[i] = -1.0
        row2[n] = 1.0
        A_ub_rows.append(row2)
        b_ub_rows.append(0.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, radius))  # r
    bounds.append((None, None))   # tau is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n+1]
    return np.zeros(n), float(np.max(scores))

    # Grid search
    alphas = np.linspace(0.0, a_max, n_evals)
    best_alpha = 0.0
    best_val = exact_c5(h)
    
    for a in alphas[1:]:
        h_try = h + a * p
        h_try = np.clip(h_try, 0.0, 1.0)
        val = exact_c5(h_try)
        if val < best_val:
            best_val = val
            best_alpha = a
    
    # Golden section refinement around best
    if 0 < best_alpha < a_max and n_evals > 1:
        step = alphas[1] - alphas[0]
        lo = best_alpha - step
        hi = best_alpha + step
        lo = max(lo, 0.0)
        hi = min(hi, a_max)
        if hi - lo > 1e-10:
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = exact_c5(np.clip(h + c1 * p, 0, 1))
            f2 = exact_c5(np.clip(h + c2 * p, 0, 1))
            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = exact_c5(np.clip(h + c1 * p, 0, 1))
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = exact_c5(np.clip(h + c2 * p, 0, 1))
            mid = (lo + hi) / 2
            f_mid = exact_c5(np.clip(h + mid * p, 0, 1))
            if f_mid < best_val:
                best_val = f_mid
                best_alpha = mid
    
    return best_alpha, best_val

def subgradient_direction(h, top_k=10, radius=0.1):
    """Simple subgradient direction: negative weighted average of active gradients."""
    n = len(h)
    scores, grads, idx = compute_grads(h, top_k=top_k)
    # Use top k gradients
    d = -np.mean(grads[:top_k], axis=0)
    d -= np.mean(d)  # zero-sum
    norm = np.max(np.abs(d))
    if norm < 1e-15:
        return np.zeros(n)
    d = d / norm * radius
    return d

def interpolate_h(h_old, n_new):
    """Interpolate h from n_old points to n_new points."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    h_new = project_to_target(h_new, target)
    return h_new

def complement_transform(h, alpha):
    """Blend h with its complement: alpha*h + (1-alpha)*(1-h)."""
    h_new = alpha * h + (1.0 - alpha) * (1.0 - h)
    # This preserves sum only if alpha = 0.5
    # For general alpha, sum changes. Re-project.
    target = len(h) / 2.0
    h_new = project_to_target(h_new, target)
    return h_new

`np.correlate(a, v, mode='full')`:
The output is of length N+M-1 where N=len(a), M=len(v).
out[k] = sum_n a[n+k] * conj(v[n]) for all valid n.

For a=h (len n), v=1-h (len n):
out[k] = sum_n h[n+k] * (1-h[n])

Valid n: 0 ≤ n < n and 0 ≤ n+k < n.

If k ≥ 0: 0 ≤ n ≤ n-1-k, so n ranges from 0 to n-1-k.
If k < 0: 0 ≤ n-k < n → k ≤ n < n-k → -k ≤ n ≤ n-1. Let g = -k: g ≤ n ≤ n-1.

So:
k ≥ 0: out[k] = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
k < 0 (g=-k): out[k] = sum_{n=g}^{n-1} h[n-g] * (1-h[n])

The output array has indices 0 to 2n-2. The "middle" (lag 0) is at index n-1.
