def lp_direction(h, dx, active_set):
    """Find a direction d that decreases all active constraints.
    Returns d or None.
    """
    n = len(h)
    L = 2*n - 1
    corr = np.correlate(h, 1.0-h, mode="full") * dx
    G = compute_G(h, dx)
    
    # LP: minimize t
    # s.t. G[r]·d <= t for r in active_set
    #      sum(d) = 0
    #      -1 <= d_i <= 1
    #      (optional: G[r]·d <= (corr[r] - max(corr)) + t for inactive r, to not exceed current max)
    
    # Variables: d[0..n-1], t
    # Total: n+1 variables
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    
    for r in active_set:
        row = np.zeros(n+1)
        row[:n] = G[r]
        row[n] = -1.0  # G[r]·d - t <= 0  i.e., G[r]·d <= t
        A_ub.append(row)
        b_ub.append(0.0)
    
    # Inactive constraints: don't let them exceed current max
    max_corr = np.max(corr)
    for r in range(L):
        if r not in active_set:
            row = np.zeros(n+1)
            row[:n] = G[r]
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(corr[r] - max_corr)  # G[r]·d <= corr[r] - max_corr (negative or zero)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -1 <= d_i <= 1, t unbounded below (or t >= -1)
    bounds = [(-1.0, 1.0)] * n + [(-1.0, 1.0)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success and res.fun < -1e-10:
        return res.x[:n]
    return None

import numpy as np

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def score_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project(v, n=None):
    if n is None:
        n = len(v)
    target = n / 2.0
    # Bisection on uniform shift
    lo, hi = -max(v.max()-0, 1e-9) - 1e-9, -min(v.min()-1, 1e-9) + 1e-9
    # Actually: we want sum(clip(v+lambda, 0, 1)) = target
    # As lambda -> -inf, sum -> 0. As lambda -> +inf, sum -> n.
    lo = -v.max() - 1.0
    hi = -v.min() + 1.0
    w = (v > 0) & (v < 1)  # indices that will be in the "active" range
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_mid = np.clip(v + mid, 0.0, 1.0)
        if np.sum(h_mid) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
            for i in idx:
                d = min(residual, 1.0 - h[i])
                h[i] += d
                residual -= d
                if abs(residual) < 1e-15:
                    break
        else:
            idx = np.argsort(h)[::-1]
            for i in idx:
                d = min(-residual, h[i])
                h[i] -= d
                residual += d
                if abs(residual) < 1e-15:
                    break
    return h

def corr_lags(n):
    return np.arange(-(n-1), n)

def gradient_of_lag(h, s, dx):
    """Gradient of (correlation at lag s) * dx w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s > 0:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
        else:
            g[:] = 1.0 - 2.0 * h
    else:
        a = -s
        g[:n-a] += (1.0 - h[a:])
        g[a:] -= h[:n-a]
    return dx * g

    """Returns G of shape (2n-1, n), G[r] = gradient of corr[r]*dx."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        s = r - (n-1)
        G[r] = gradient_of_lag(h, s, dx)
    return G

        # Weighted gradient: weights = exp(-margins / temperature)
        # Only use the top few
        top_k = min(20, n)
        idx = np.argsort(margins)[:top_k]  # smallest margins = most active
        weights = np.exp(-margins[idx] / temperature)
        weights /= weights.sum()
        
        G = compute_all_gradients(h, dx)
        grad = np.zeros(n)
        for i, r in enumerate(idx):
            grad += weights[i] * G[r]

                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    h_new = project(h_try, n)
                    s = official_score(h_new)
                    if s < score - 1e-12:
                        h = h_new
                        score = s
                        improved = True
                        break
                if not improved:
                    break
    
    return h, official_score(h)

def lp_minimax_step(h, n, dx, temperature=0.02):
    """One LP-based minimax descent step."""
    c = score_all(h)
    max_c = np.max(c)
    margins = max_c - c
    
    # Select active constraints
    tol = max(temperature, 1e-6)
    active = np.where(margins <= tol)[0]
    if len(active) == 0:
        active = np.argsort(margins)[:5]
    if len(active) > 40:
        active = np.argsort(margins)[:40]
    
    G = compute_all_gradients(h, dx)
    
    # LP: minimize t s.t. G[r]·d <= t for r in active, sum(d)=0, -1<=d_i<=1
    nv = n + 1  # n for d, 1 for t
    c_vec = np.zeros(nv)
    c_vec[n] = 1.0
    
    A_ub = np.zeros((len(active), nv))
    for i, r in enumerate(active):
        A_ub[i, :n] = G[r]
        A_ub[i, n] = -1.0
    b_ub = np.zeros(len(active))
    
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(-1.0, 1.0)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
    except:
        return None
    
    if res.success and res.x[n] < -1e-10:
        return res.x[:n]
    return None

def lp_local_search(h, n, max_iters=30, time_limit=15.0):
    """Local search using LP minimax steps."""
    h = project(h, n).copy()
    score = official_score(h)
    dx = 2.0 / n
    start = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        d = lp_minimax_step(h, n, dx)
        if d is None:
            break
        
        # Line search
        best_alpha = 0.0
        best_s = score
        for a in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            for sign in [1, -1]:
                h_new = project(h + sign * a * d, n)
                s = official_score(h_new)
                if s < best_s - 1e-13:
                    best_s = s
                    best_alpha = sign * a
        
        if best_alpha != 0.0:
            h = project(h + best_alpha * d, n)
            score = best_s
        else:
            break
    
    return h, official_score(h)

def binary_swap_search(h, n, max_iters=500, time_limit=30.0, seed=42):
    """Local search for near-binary sequences using swaps."""
    rng = np.random.default_rng(seed)
    h = h.copy()
    score = official_score(h)
    start = time.perf_counter()
    
    # Identify 1-positions and 0-positions
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    no_improve = 0
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        improved = False
        # Try random swaps
        n_tries = min(len(ones) * len(zeros), 300)
        for _ in range(n_tries):
            i = ones[rng.integers(0, len(ones))]
            j = zeros[rng.integers(0, len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            s = official_score(h)
            if s < score - 1e-13:
                score = s
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return h, official_score(h)

def slsqp_polish(h, n, maxiter=60):
    """SLSQP minimax polish."""
    dx = 2.0 / n
    lags = corr_lags(n)
    L = 2 * n - 1
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        j = np.zeros((L, n+1))
        j[:, n] = 1.0
        for r in range(L):
            s = r - (n-1)
            j[r, :n] = -gradient_of_lag(h_x, s, dx)
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - n/2.0
    
    def eq_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    try:
        result = optimize.minimize(
            obj, np.concatenate([h, [official_score(h) + 1e-6]]),
            jac=obj_grad, method='SLSQP',
            bounds=[(0.0, 1.0)]*n + [(0.0, 2.0)],
            constraints=[
                {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
                {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project(result.x[:n], n)
            s = official_score(h_out)
            if s < official_score(h):
                return h_out, s
    except:
        pass
    return h, official_score(h)

def interpolate(h_src, n_target, shift=0):
    """Interpolate h_src (length m) to n_target points."""
    m = len(h_src)
    if m == 0:
        return project(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + shift) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_s = xt[order]
    h_new = np.interp(xt_s, xs, h_src, left=h_src[0], right=h_src[-1])
    h_new = h_new[np.argsort(order)]
    return project(h_new, n_target)

def gen_binary(n, rng):
    """Generate random binary vector with sum = n/2 (n must be even)."""
    if n % 2 != 0:
        return project(rng.uniform(0, 1, n), n)
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h

def gen_continuous(n, rng):
    """Generate random continuous vector."""
    h = rng.uniform(0, 1, n)
    return project(h, n)

def gen_structured(n, rng):
    """Generate structured candidates (sine-based)."""
    t = np.linspace(0, 1, n, endpoint=False)
    freq = rng.uniform(1.5, 6.0)
    phase = rng.uniform(0, 2*np.pi)
    amp = rng.uniform(0.3, 0.5)
    h = 0.5 + amp * np.sin(2*np.pi*freq*t + phase)
    h += rng.normal(0, 0.03, n)
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 5.0
    
    # Load parent
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = project(pv, len(pv))
    except:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        best_h = parent.copy()
    else:
        n_parent = 64
        best_h = project(np.full(64, 0.5), n_parent)
    
    best_score = official_score(best_h)
    best_n = n_parent
    
    rng = np.random.default_rng(seed)
    
    # Phase 1: Generate diverse candidates
    candidates = []
    # Parent variants
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h[::-1]).copy())
    candidates.append((1.0 - best_h).copy())
    
    # Try different resolutions
    n_list = sorted(set([
        n_parent, n_parent - 4, n_parent + 4, n_parent - 8, n_parent + 8,
        n_parent - 1, n_parent + 1
    ]))
    n_list = [n for n in n_list if 32 <= n <= 120]
    
    for n_t in n_list:
        candidates.append(interpolate(best_h, n_t))
        if n_t % 2 == 0:
            # Binary at this resolution
            h_bin = gen_binary(n_t, rng)
            candidates.append(h_bin)
    
    # Additional random candidates
    for n_t in n_list[:6]:
        candidates.append(gen_continuous(n_t, rng))
        candidates.append(gen_structured(n_t, rng))
        if n_t % 2 == 0:
            candidates.append(gen_binary(n_t, rng))
    
    # Phase 2: Score and rank
    scored = []
    for h_c in candidates:
        h_c = project(h_c, len(h_c))
        if not np.all(np.isfinite(h_c)):
            continue
        s = official_score(h_c)
        if np.isfinite(s):
            scored.append((s, h_c))
    
    scored.sort(key=lambda x: (x[0], len(x[1])))
    
    if len(scored) > 0 and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)
    
    # Phase 3: Local optimization on top candidates
    top_k = min(8, len(scored))
    for idx in range(top_k):
        if time.perf_counter() > deadline - 120:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        
        # Choose method based on structure
        h_binary = np.all((h_c < 0.01) | (h_c > 0.99))
        
        if h_binary and n_c % 2 == 0:
            # Binary swap search
            h_loc, s_loc = binary_swap_search(h_c, n_c, max_iters=300,
                                              time_limit=min(25.0, remaining*0.4),
                                              seed=seed+idx)
        else:
            # Weighted gradient descent
            h_loc, s_loc = weighted_gradient_descent(h_c, n_c,
                                                     max_iters=50,
                                                     time_limit=min(25.0, remaining*0.4),
                                                     seed=seed+idx)
        
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c
