def project_feasible(z, target=None):
    # Project z onto {0 <= h <= 1, sum(h) = n/2}
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    target = n / 2.0
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            # c_k = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # d/dh[j]: +（1-h[j+s]) for j in [0,n-1-s], -h[j-s] for j in [s,n-1]
            grad[0:n-s] += wk * (1.0 - h[s:n])
            grad[s:n] -= wk * h[0:n-s]
        else:
            t = -s
            # c_k = sum_{i=0}^{n-1-t} h[i+t]*(1-h[i])
            # d/dh[j]: +(1-h[j-t]) for j in [t,n-1], -h[j+t] for j in [0,n-1-t]
            grad[t:n] += wk * (1.0 - h[0:n-t])
            grad[0:n-t] -= wk * h[t:n]
    return grad
