def correlation_gradient(h, k):
    """Gradient of the k-th correlation value with respect to h.
    
    The k-th correlation (0-indexed in the full vector) corresponds to:
    corr[k] = sum_i h[i] * (1-h[i - (k - (n-1))])
    where the index i - (k-(n-1)) must be in [0, n-1].
    
    Let lag = k - (n-1). Then:
    corr[k] = sum_i h[i] * (1 - h[i - lag]) for valid pairs.
    
    d corr[k] / d h[j] = sum of contributions:
    - If j appears as the first factor: (1 - h[j - lag]) for valid j - lag
    - If j appears as the second factor: -h[j + lag] for valid j + lag
    """
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    
    if lag == 0:
        # corr = sum_i h[i] * (1 - h[i])
        # d/dh[j] = (1 - h[j]) - h[j] = 1 - 2*h[j]
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # corr = sum_{i=lag}^{n-1} h[i] * (1 - h[i-lag])
        # d/dh[j]: 
        #   if j >= lag: contributes (1 - h[j-lag]) as first factor
        #   if j <= n-1-lag: contributes -h[j+lag] as second factor
        for j in range(lag, n):
            grad[j] += (1.0 - h[j - lag])
        for j in range(0, n - lag):
            grad[j] -= h[j + lag]
    else:  # lag < 0
        # corr = sum_{i=0}^{n-1+lag} h[i] * (1 - h[i-lag])
        # where i-lag > i since lag < 0
        # Let lag' = -lag > 0
        # corr = sum_{i=0}^{n-1-lag'} h[i] * (1 - h[i+lag'])
        lag2 = -lag
        for j in range(0, n - lag2):
            grad[j] += (1.0 - h[j + lag2])
        for j in range(lag2, n):
            grad[j] -= h[j - lag2]
    
    return grad

def active_gradient(h, n_active=5):
    """Compute weighted gradient from top active lags."""
    corr = compute_corr(h)
    n = len(h)
    # Find top active lags
    top_idx = np.argsort(corr)[-n_active:][::-1]

    grad = np.zeros(n)
    for i, k in enumerate(top_idx):

def project_gradient(grad, h):

    # Zero-sum projection
    grad = grad - np.mean(grad)
    
    # Zero out components where h is at boundary
    # (if h_i = 0, we can only increase; if h_i = 1, we can only decrease)

    d = d - np.mean(d)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_to_box_eq(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    result = np.clip(h - alpha, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            candidates = np.where(result < 1.0 - 1e-10)[0]
            if len(candidates) > 0:
                idx = candidates[np.argmax(1.0 - result[candidates])]
                add = min(diff, 1.0 - result[idx])
                result[idx] += add
                diff -= add
        else:
            candidates = np.where(result > 1e-10)[0]
            if len(candidates) > 0:
                idx = candidates[np.argmin(result[candidates])]
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
    return result


def corr_gradient_at_lag(h, k):
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        for j in range(lag, n):
            grad[j] += (1.0 - h[j - lag])
        for j in range(0, n - lag):
            grad[j] -= h[j + lag]
    else:
        lag2 = -lag
        for j in range(0, n - lag2):
            grad[j] += (1.0 - h[j + lag2])
        for j in range(lag2, n):
            grad[j] -= h[j - lag2]
    return grad


def feasible_interval(h, d):

    t_min = 0.0
    t_max = 0.0
    pos = d > eps
    neg = d < -eps
    if np.any(pos):

    if np.any(neg):
        lower = np.max(-h[neg] / d[neg])
        t_min = lower
    if t_min > t_max:
        t_min, t_max = t_max, t_min
    return t_min, t_max

def softmax_gradient(h, alpha, n_active):
    corr = compute_corr(h)
    n = len(h)
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)

    top_indices = np.argsort(corr)[-n_active:]
    grad = np.zeros(n)
    for k in top_indices:
        w = p[k]
        if w > 1e-10:
            grad += w * corr_gradient_at_lag(h, k)
    return grad, corr


def project_direction(d, h):
    d = d - np.mean(d)
    d[(h <= 1e-10) & (d < 0)] = 0.0
    d[(h >= 1.0 - 1e-10) & (d > 0)] = 0.0
    d = d - np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return d
    return d / norm
