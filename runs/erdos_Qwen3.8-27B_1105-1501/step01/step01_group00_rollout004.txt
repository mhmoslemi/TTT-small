def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -10.0, 10.0
    while np.sum(np.clip(z + lo, 0, 1)) > target:
        lo -= 5.0
    while np.sum(np.clip(z + hi, 0, 1)) < target:
        hi += 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-13:
        order = np.argsort(1.0 - h)[::-1]
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-15:
                break
    elif residual < -1e-13:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if -residual < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)
