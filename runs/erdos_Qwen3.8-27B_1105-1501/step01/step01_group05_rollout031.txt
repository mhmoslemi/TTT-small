    """Find x in [L,U] minimizing F(h with h[i]=x, h[j]=S-x)"""
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-12:
        return h, official_score(h)
    
    dx = 2.0 / len(h)
    # Golden section search (F is convex)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    invphi2 = invphi * invphi

    c = b - invphi * (b - a)
    d = a + invphi * (b - a)

    n = len(h)
    score = official_score(h)
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]

def basin_hop(h, score, rng, deadline, n_hops=5):

    for _ in range(n_hops):
        if time.perf_counter() > deadline:

        # Zero-sum perturbation
        m = rng.integers(3, min(10, n))
        idx = rng.choice(n, size=m, replace=False)

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(v, n_target):
    h = np.array(v, dtype=float).copy().ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h_out)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = int(np.argmax(1.0 - h_out))
            h_out[idx] = min(1.0, h_out[idx] + residual)
        else:
            idx = int(np.argmin(h_out))
            h_out[idx] = max(0.0, h_out[idx] + residual)
    return h_out

def interp_to_length(h, n_target, offset=0.0):
    n_src = len(h)

    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * (2.0 / n_target)
    xt_sorted = np.sort(xt)
    h_sorted = np.interp(xt_sorted, xs, h, left=h[0], right=h[-1])
    # Map back

    order = np.argsort(xt)
    h_new[order] = h_sorted

def pair_line_min(h, i, j, n_gs=25):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    n = len(h)
    if U - L < 1e-12:
        return h, official_score(h)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = eval_pair(h, i, j, S, c, n)
    fd = eval_pair(h, i, j, S, d, n)
    for _ in range(n_gs):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = eval_pair(h, i, j, S, c, n)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = eval_pair(h, i, j, S, d, n)
    x_best = (a + b) / 2.0
    h_best = h.copy()
    h_best[i] = x_best
    h_best[j] = S - x_best
    s_best = official_score(h_best)
    return h_best, s_best

import numpy as np
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(v, n_target):
    h = np.array(v, dtype=float).copy().ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h_out)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = int(np.argmax(1.0 - h_out))
            h_out[idx] = min(1.0, h_out[idx] + residual)
        else:
            idx = int(np.argmin(h_out))
            h_out[idx] = max(0.0, h_out[idx] + residual)
    return h_out


def interp_to_length(h, n_target, offset=0.0):
    n_src = len(h)
    if n_src == 0:
        return repair_vector(np.full(n_target, 0.5), n_target)
    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * (2.0 / n_target)
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h_sorted = np.interp(xt_sorted, xs, h, left=h[0], right=h[-1])
    h_new = np.empty(n_target)
    h_new[order] = h_sorted
    return repair_vector(h_new, n_target)


def eval_pair(h, i, j, S, x, n):
    tmp = h.copy()
    tmp[i] = x
    tmp[j] = S - x
    dx = 2.0 / n
    corr = np.correlate(tmp, 1.0 - tmp, mode="full")
    return float(np.max(corr)) * dx


def pair_line_min(h, i, j, n_gs=30):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    n = len(h)
    if U - L < 1e-12:
        return h, official_score(h)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = eval_pair(h, i, j, S, c, n)
    fd = eval_pair(h, i, j, S, d, n)
    for _ in range(n_gs):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = eval_pair(h, i, j, S, c, n)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = eval_pair(h, i, j, S, d, n)
    x_best = (a + b) / 2.0
    h_best = h.copy()
    h_best[i] = x_best
    h_best[j] = S - x_best
    s_best = official_score(h_best)
    return h_best, s_best


def full_pair_descent(h, deadline, max_passes=50, gs_iters=30):
    n = len(h)
    score = official_score(h)
    no_improve_count = 0
    for pidx in range(max_passes):
        if time.perf_counter() > deadline:
            break
        improved = False
        for i in range(n):
            if time.perf_counter() > deadline:
                return h, official_score(h)
            for j in range(i + 1, n):
                S = h[i] + h[j]
                L = max(0.0, S - 1.0)
                U = min(1.0, S)
                if U - L < 1e-10:
                    continue
                h_new, s_new = pair_line_min(h, i, j, gs_iters)
                if s_new < score - 1e-13:
                    h = h_new
                    score = s_new
                    improved = True
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 2:
                break
        else:
            no_improve_count = 0
    return h, official_score(h)


def make_perturbation(h, rng, m_min=3, m_max=10, alpha_range=(0.03, 0.2)):
    n = len(h)
    m = int(rng.integers(m_min, min(m_max, n) + 1))
    idx = rng.choice(n, size=m, replace=False)
    d = rng.standard_normal(m)
    d -= d.mean()
    cap = 1.0
    for k in range(m):
        if d[k] > 0:
            cap = min(cap, (1.0 - h[idx[k]]) / d[k] * 0.85)
        elif d[k] < 0:
            cap = min(cap, h[idx[k]] / (-d[k]) * 0.85)
    if cap < 1e-10:
        return None
    a_lo, a_hi = alpha_range
    alpha = float(rng.uniform(min(a_lo, cap * 0.5), min(a_hi, cap)))
    if alpha <= 0:
        return None
    h_pert = h.copy()
    h_pert[idx] += alpha * d
    return repair_vector(h_pert, n)


def basin_hop(h, score, rng, deadline, n_hops=10, expand=False):
    n = len(h)
    best_local_h = h.copy()
    best_local_s = score
    for hop_idx in range(n_hops):
        if time.perf_counter() > deadline:
            break
        m_min = 3
        m_max = min(10, n)
        alpha_range = (0.05, 0.25) if expand else (0.03, 0.15)
        h_pert = make_perturbation(h, rng, m_min, m_max, alpha_range)
        if h_pert is None:
            continue
        t_sub = min(deadline, time.perf_counter() + 70)
        h_new, s_new = full_pair_descent(h_pert, t_sub, max_passes=20, gs_iters=22)
        if s_new < best_local_s:
            best_local_s = s_new
            best_local_h = h_new.copy()
            h = h_new
            score = s_new
    return best_local_h, best_local_s


def gen_binary_pattern(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 1e-10:
        h[idx[m_full]] = frac
    return repair_vector(h, n)


def gen_weights_blend(h1, h2, w, n_target):
    if len(h1) != n_target:
        h1 = interp_to_length(h1, n_target)
    if len(h2) != n_target:
        h2 = interp_to_length(h2, n_target)
    blended = (1.0 - w) * h1 + w * h2
    return repair_vector(blended, n_target)
