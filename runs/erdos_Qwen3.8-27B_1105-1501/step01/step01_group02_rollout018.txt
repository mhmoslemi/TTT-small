def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):

        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

import numpy as np
import time


def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    """Compute C5 = max_k corr_k * dx."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_initial(n, rng, style=0):
    """Create diverse initial 0-1 patterns."""
    h = np.zeros(n)
    if style == 0:
        h[::2] = 1.0
    elif style == 1:
        h[:n // 2] = 1.0
    elif style == 2:
        perm = rng.permutation(n)
        h[perm[:n // 2]] = 1.0
    elif style == 3:
        for i in range(n):
            if (i * 3 + 1) % (n // 3 + 1) < n // 6:
                h[i] = 1.0
    elif style == 4:
        for i in range(n):
            if (i * i + i) % (n // 2 + 1) < n // 4:
                h[i] = 1.0
    return project_onto_constraint(h, n / 2.0)


def random_pair_move(h, cur_c5, dx, T, rng, n_grid=15):
    """One random pair move with SA acceptance."""
    n = len(h)
    i = rng.randint(n)
    j = rng.randint(n)
    if i == j:
        return h, cur_c5, False

    lo = max(h[i] - 1.0, -h[j])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-12:
        return h, cur_c5, False

    best_d = 0.0
    best_local = cur_c5
    grid = np.linspace(lo, hi, n_grid)
    for d in grid:
        if abs(d) < 1e-12:
            continue
        h[i] -= d
        h[j] += d
        c = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        h[i] += d
        h[j] -= d
        if c < best_local:
            best_local = c
            best_d = d

    if abs(best_d) < 1e-12:
        return h, cur_c5, False

    if best_local < cur_c5:
        h[i] -= best_d
        h[j] += best_d
        return h, best_local, True

    # SA acceptance
    delta_c = best_local - cur_c5
    if T > 1e-8 and delta_c < 5.0 * T:
        if rng.random() < np.exp(-delta_c / T):
            h[i] -= best_d
            h[j] += best_d
            return h, best_local, True

    return h, cur_c5, False


def pair_sweep(h, dx, n_grid=10):
    """Systematic sweep over all pairs with greedy updates."""
    n = len(h)
    cur_c5 = compute_c5(h)
    improved = 0
    for i in range(n):
        for j in range(i + 1, n):
            lo = max(h[i] - 1.0, -h[j])
            hi = min(h[i], 1.0 - h[j])
            if hi - lo < 1e-12:
                continue
            best_d = 0.0
            best_local = cur_c5
            grid = np.linspace(lo, hi, n_grid)
            for d in grid:
                if abs(d) < 1e-12:

                h[i] -= d
                h[j] += d
