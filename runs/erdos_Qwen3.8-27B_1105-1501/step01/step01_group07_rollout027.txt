import cvxpy as cp

def lp_step(h, active_lags=None, eps=0.01):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = np.max(c)
    
    if active_lags is None:
        active_lags = np.where(c > c_max - eps)[0]
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for k in active_lags:
        s = k - (n-1)  # shift
        # c[k] = sum_i h[i+s](1-h[i]) * dx (over valid i)
        # Linearize at h: ∇_h c[k]
        # ∂c[k]/∂h[m] = dx * (1{0≤m-s<n}(1-h[m-s]) - 1{0≤m+s<n}h[m+s])
        
        grad = np.zeros(n)
        for m in range(n):
            if 0 <= m-s < n:
                grad[m] += 1.0 - h[m-s]
            if 0 <= m+s < n:
                grad[m] -= h[m+s]
        grad *= dx
        
        # Linear model: c[k] ≈ c[k]_current + grad · (h_var - h)
        constraints.append(c[k] + grad @ (h_var - h) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK)  # or CBC
    
    if h_var.value is None:
        return None, None
    return np.clip(h_var.value, 0, 1), t.value

        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()

            add = min(diff, 1.0 - z[idx])
            z[idx] += add

                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])

def lp_solve(h, all_lags=True):
    """Solve LP to minimize max correlation, linearized at h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    if all_lags:
        lags = range(2 * n - 1)
    else:
        c_max = np.max(c)
        lags = [k for k in range(2 * n - 1) if c[k] > c_max - 0.02]
    
    for k in lags:
        s = k - (n - 1)
        # Compute gradient of c[k] w.r.t. h at current point
        grad = np.zeros(n)
        for m in range(n):
            val = 0.0
            if 0 <= m - s < n:
                val += 1.0 - h[m - s]
            if 0 <= m + s < n:
                val -= h[m + s]
            grad[m] = val * dx
        # Linear constraint: c[k] + grad · (h_var - h) <= t
        cons.append(c[k] + grad @ (h_var - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.GLPK)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None
    
    if h_var.value is None:
        return None
    return h_var.value

    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c

        gj = 0.0

        if 0 <= j + s < n:
            gj -= h[j + s]
        grad_i[k] = gi * dx
        grad_j[k] = gj * dx

    # First order: a * (grad_i[k] - grad_j[k])

    # For small a, first order is good enough for finding the direction

    # Let's use the linear approximation to find promising a values,
    # then verify with exact computation.

    h[j] = orig_j

        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:

            for _ in range(30):

                s2 = score_exact(h)
                if s1 < s2:

                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:

                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)

            h[i] = orig_i + best_a
            h[j] = orig_j - best_a

                h[j] = orig_j
                return None

        return None

    h[j] = orig_j - best_a

            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)


def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def lag_gradient(h, s, dx):
    """Gradient of correlation at shift s w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    for m in range(n):
        val = 0.0
        if 0 <= m - s < n:
            val += 1.0 - h[m - s]
        if 0 <= m + s < n:
            val -= h[m + s]
        g[m] = val * dx
    return g


def compute_all_gradients(h):
    """Compute gradient vectors for all lags. Returns array shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        grads[k] = lag_gradient(h, s, dx)
    return grads


def solve_lp(h, working_set=None, lam=0.0):
    """Solve LP min t + lam*sum(c) s.t. linearized c[k] <= t, box, sum constraint."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx

    if working_set is None:
        working_set = list(range(2 * n - 1))

    h_var = cp.Variable(n)
    t = cp.Variable()

    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]

    for k in working_set:
        s = k - (n - 1)
        g = lag_gradient(h, s, dx)
        cons.append(c[k] + g @ (h_var - h) <= t)

    if lam > 0:
        # Add sum of all lags as secondary objective
        sum_grad = np.zeros(n)
        for k in range(2 * n - 1):
            s = k - (n - 1)
            sum_grad += lag_gradient(h, s, dx)
        obj = cp.Minimize(t + lam * (float(np.sum(c)) + sum_grad @ (h_var - h)))
    else:
        obj = cp.Minimize(t)

    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.GLPK)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return h_var.value


def lp_descent(h, deadline, rng, lam_start=0.0):
    """Iterative LP-based descent with adaptive working set and step size."""
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    n = len(cur)
    step_frac = 1.0
    stall = 0
    lam = lam_start

    while time.time() < deadline:
        # Determine working set: lags near the max
        c_vals = correlations(cur)
        c_max = np.max(c_vals)
        margin = max(0.01, 0.1 * c_max)
        working_set = [k for k in range(2 * n - 1) if c_vals[k] > c_max - margin]
        if len(working_set) < 3:
            working_set = list(range(2 * n - 1))

        h_lp = solve_lp(cur, working_set, lam)
        if h_lp is None:
            break

        h_lp = make_feasible(h_lp, n)
        direction = h_lp - cur

        improved = False
        for frac in [step_frac, step_frac * 0.5, step_frac * 0.25, step_frac * 0.1]:
            if frac < 0.01:
                break
            cand = cur + frac * direction
            cand = make_feasible(cand, n)
            c_cand = score_exact(cand)
            if c_cand < cur_c - 1e-13:
                cur = cand
                cur_c = c_cand
                improved = True
                stall = 0
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break

        if improved:
            step_frac = min(1.0, step_frac * 1.05)
            if lam > 0:
                lam *= 0.9
        else:
            step_frac = max(0.05, step_frac * 0.7)
            lam = max(0.0, lam - 0.01)
            stall += 1
            if stall > 15:
                # Perturb to escape
                pert_size = 0.03 if n > 50 else 0.05
                cur = make_feasible(cur + rng.normal(0, pert_size, n), n)
                cur_c = score_exact(cur)
                stall = 0
                step_frac = 0.5
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c

    return best, best_c


def pair_search_fast(h, i, j, cur_c, n_pts=35):
    """Pair search with golden section refinement."""
    n = len(h)
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None

    best_a = 0.0
    best_s = cur_c

    candidates = np.linspace(lo, hi, n_pts)
    for a in candidates:
        h[i] = orig_i + a
        h[j] = orig_j - a
        s_val = score_exact(h)
        if s_val < best_s:
            best_s = s_val
            best_a = a

    h[i] = orig_i
    h[j] = orig_j

    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / max(1, n_pts - 1)
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(25):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                h[i] = orig_i
                h[j] = orig_j
                return None
    else:
        return None

    h[i] = orig_i + best_a
    h[j] = orig_j - best_a
    return best_s


def deterministic_pair_pass(h, cur_c, active_lags, n):
    """Systematically try all pairs in the active set."""
    best_c = cur_c
    # Get indices appearing in active lag gradients
    active_indices = set()
    for k in active_lags:
        s = k - (n - 1)
        for m in range(n):
            if (0 <= m - s < n) or (0 <= m + s < n):
                active_indices.add(m)
    active_list = list(active_indices)
    if len(active_list) < 2:
        active_list = list(range(n))

    for idx_i in range(len(active_list)):
        for idx_j in range(idx_i + 1, len(active_list)):
            i = active_list[idx_i]
            j = active_list[idx_j]
            res = pair_search_fast(h, i, j, best_c, n_pts=20)
            if res is not None:
                best_c = res
    return best_c


def smooth_grad_descent(h, deadline, T_start, T_end, T_decay, step0, rng):
    """Smooth max gradient descent as backup."""
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    c = n - 1

    while time.time() < deadline:
        scores = np.correlate(cur, 1.0 - cur, mode='full') * (2.0 / n)
        base = float(np.max(scores))
        ex = np.exp((scores - base) / T)
        S = float(np.sum(ex))
        if S < 1e-300:
            mask = scores >= base - 1e-10
            w = np.where(mask, 1.0, 0.0)
            w /= w.sum()
        else:
            w = ex / S

        g = np.zeros(n)
        dx = 2.0 / n
        for d in range(n):
            kd = c + d
            if kd >= 2 * n - 1:
                break
            km = c - d
            if d == 0:
                g += w[kd] * dx * (1.0 - 2.0 * cur)
            else:
                m = n - d
                if m <= 0:
                    continue
                g[:m] += w[km] * dx * (1.0 - cur[d:]) - w[kd] * dx * cur[d:]
                g[d:] += w[kd] * dx * (1.0 - cur[:m]) - w[km] * dx * cur[:m]

        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.3:
                break
            continue

        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 30:
                cur = make_feasible(cur + rng.normal(0, max(0.005, T * 0.3), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.005)
                stall = 0
        else:
            step = min(step * 1.05, T * 2.0)
            stall = 0
        T = max(T_end, T * T_decay)

    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: LP descent (50% of time)
    p1_end = t0 + budget_s * 0.50
    if time.time() < p1_end:
        h1, c1 = lp_descent(best_h, p1_end, rng, lam_start=0.0)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Grad descent refinement (15% of time)
    p2_end = t0 + budget_s * 0.65
    if time.time() < p2_end:
        h2, c2 = smooth_grad_descent(best_h, p2_end, 0.01, 0.0005, 0.999, 0.01, rng)
        if c2 < best_c:
            best_h, best_c = h2, c1 if c1 < best_c else h2, c2

    # Phase 3: Pair search polish (25% of time)
    p3_end = t0 + budget_s * 0.90
    cur = best_h.copy()
    cur_c = best_c
    consecutive_fail = 0
    while time.time() < p3_end:
        c_vals = correlations(cur)
        c_max = np.max(c_vals)
        margin = max(0.02, 0.15 * c_max)
        active = np.where(c_vals > c_max - margin)[0]

        mode = rng.random()
        if mode < 0.5:
            # Random pair from active indices
            active_idx = []
            for k in active:
                s = k - (n - 1)
                for m in range(n):
                    if (0 <= m - s < n) or (0 <= m + s < n):
                        if m not in active_idx:
                            active_idx.append(m)
            if len(active_idx) >= 2:
                i = active_idx[int(rng.integers(0, len(active_idx)))]
                j = active_idx[int(rng.integers(0, len(active_idx)))]
                if i == j:
                    continue
            else:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
        elif mode < 0.7:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
        elif mode < 0.85:
            i = int(rng.integers(0, n - 1))
            j = i + 1
        else:
            # Antipodal pair
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            if i == j:
                continue

        res = pair_search_fast(cur, i, j, cur_c, n_pts=30)
        if res is not None:
            cur_c = res
            consecutive_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            # Chain a few more
            for _ in range(3):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = pair_search_fast(cur, i2, j2, cur_c, n_pts=20)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            consecutive_fail += 1
            if consecutive_fail > 150:
                cur = make_feasible(best_h + rng.normal(0, 0.01, n), n)
                cur_c = score_exact(cur)
                consecutive_fail = 0

    # Phase 4: Second LP pass with different settings (10% of time)
    p4_end = deadline - 2.0
    if time.time() < p4_end:
        h4, c4 = lp_descent(best_h, p4_end, rng, lam_start=0.05)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Final feasibility check
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
