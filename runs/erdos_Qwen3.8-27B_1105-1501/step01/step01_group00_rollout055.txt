            cur[a] += d
            cur[b] -= d
            cur = np.clip(cur, 0.0, 1.0)
            cur[np.argmin(cur)] += (n_sa / 2.0 - np.sum(cur))
            cur = np.clip(cur, 0.0, 1.0)
            sc_new = score_c5(cur)
            delta = sc_new - cur_sc
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_sc = sc_new
                if sc_new < best_score - 1e-14:
                    best_h = cur.copy()
                    best_score = sc_new
            else:
                cur[a] -= d
                cur[b] += d
            T *= cooling

    # Phase 6: Final intensive polish
    if still_going():
        cur_h = best_h.copy()
        n = len(cur_h)
        dx = 2.0 / n
        for pp in range(8):
            if not still_going():
                break
            cur_h, improved = full_pass_pairs(
                cur_h, np.correlate(cur_h, 1.0 - cur_h, mode='full'), dx,
                max_time=min(25.0, deadline - time.time() - 5.0), t_start=time.time())
            if not improved:
                break
            sc_new = score_c5(cur_h)
            if sc_new < best_score - 1e-14:
                best_h = cur_h.copy()
                best_score = sc_new

    # Final validation
    final_h = repair(best_h)
    final_n = len(final_h)
    final_score = score_c5(final_h)
    if not np.isfinite(final_score) or np.any(final_h < -1e-10) or np.any(final_h > 1.0 + 1e-10):
        final_h = np.full(30, 0.5)
        final_h = repair(final_h)
        final_score = score_c5(final_h)
        final_n = len(final_h)
    final_bound = final_score + 2e-13
    return final_h, final_bound, final_n
