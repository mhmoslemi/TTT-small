            h[idx] -= delta
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_tgt, offset=0.0):
    n_src = len(h_src)
    if n_src == 0:
        return repair_vector(np.full(n_tgt, 0.5), n_tgt)
    if n_src == n_tgt and offset == 0:
        return repair_vector(h_src.copy(), n_tgt)
    xs = (np.arange(n_src) + 0.5) / n_src
    xt = ((np.arange(n_tgt) + 0.5) / n_tgt + offset / n_src) % 1.0
    order = np.argsort(xt)
    xt_s = xt[order]
    xs_ext = np.concatenate([xs - 1.0, xs, xs + 1.0])
    h_ext = np.concatenate([h_src, h_src, h_src])
    h_s = np.interp(xt_s, xs_ext, h_ext)
    h_out = np.empty(n_tgt)
    h_out[order] = h_s
    return repair_vector(h_out, n_tgt)

def gen_binary(n, rng):
    target = n / 2.0
    m_int = int(target)
    frac = target - m_int
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_int]] = 1.0
    if frac > 0:
        h[idx[m_int]] = frac
    return repair_vector(h, n)

def gen_step(n, rng, n_steps=None):
    if n_steps is None:
        n_steps = int(rng.integers(2, max(3, n // 4)))
    n_steps = min(n_steps, n)
    if n_steps >= n:
        return repair_vector(rng.uniform(0, 1, n), n)
    boundaries = sorted(rng.choice(n - 1, size=n_steps - 1, replace=False).tolist())
    boundaries = [0] + boundaries + [n]
    vals = rng.uniform(0, 1, n_steps)
    h = np.zeros(n)
    for i in range(n_steps):
        h[boundaries[i]:boundaries[i+1]] = vals[i]
    return repair_vector(h, n)

def gen_sine(n, rng):
    t = np.linspace(0, 1, n, endpoint=False)
    phase = rng.uniform(0, 2 * np.pi)
    freq = float(rng.uniform(1, 6))
    amp = float(rng.uniform(0.3, 0.5))
    h = 0.5 + amp * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.02, n)
    return repair_vector(h, n)

def pair_line_descent(h_in, max_passes=30, n_scan=15, golden_iters=6, deadline=None, seed=42):
    n = len(h_in)
    h = repair_vector(h_in, n).copy()
    score = official_score(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(max_passes):
        if deadline is not None and time.perf_counter() > deadline:
            break
        improved = False
        order = list(range(n))
        rng.shuffle(order)

        for oi in range(n):
            if deadline is not None and time.perf_counter() > deadline:
                break
            i = order[oi]
            for j in range(n):
                if j == i:
                    continue
                if deadline is not None and time.perf_counter() > deadline:
                    break
                S = h[i] + h[j]
                L = max(0.0, S - 1.0)
                U = min(1.0, S)
                if U - L < 1e-10:
                    continue

                h_io = h[i]
                h_jo = h[j]

                x_scan = np.linspace(L, U, n_scan)
                best_x = h_io
                best_val = np.inf

                for x in x_scan:
                    h[i] = x
                    h[j] = S - x
                    val = np.max(np.correlate(h, 1.0 - h, mode="full"))
                    if val < best_val:
                        best_val = val
                        best_x = x

                best_k = int(np.argmin(np.abs(x_scan - best_x)))
                if best_k > 0 and best_k < n_scan - 1:
                    a = x_scan[max(0, best_k - 2)]
                    b = x_scan[min(n_scan - 1, best_k + 2)]
                elif best_k == 0:
                    a, b = x_scan[0], x_scan[min(3, n_scan - 1)]
                else:
                    a, b = x_scan[max(0, n_scan - 4)], x_scan[-1]

                if b - a > 1e-8:
                    c = b - invphi * (b - a)
                    d = a + invphi * (b - a)
                    h[i] = c
                    h[j] = S - c
                    fc = np.max(np.correlate(h, 1.0 - h, mode="full"))
                    h[i] = d
                    h[j] = S - d
                    fd = np.max(np.correlate(h, 1.0 - h, mode="full"))

                    for _ in range(golden_iters):
                        if fc < fd:
                            b = d
                            d = c
                            fd = fc
                            c = b - invphi * (b - a)
                            h[i] = c
                            h[j] = S - c
                            fc = np.max(np.correlate(h, 1.0 - h, mode="full"))
                        else:
                            a = c
                            c = d
                            fc = fd
                            d = a + invphi * (b - a)
                            h[i] = d
                            h[j] = S - d
                            fd = np.max(np.correlate(h, 1.0 - h, mode="full"))

                    x_mid = (a + b) / 2.0
                    h[i] = x_mid
                    h[j] = S - x_mid
                    f_mid = np.max(np.correlate(h, 1.0 - h, mode="full"))
                    if f_mid < best_val:
                        best_val = f_mid
                        best_x = x_mid

                h[i] = best_x
                h[j] = S - best_x
                new_score = best_val * dx

                if new_score < score - 1e-13:
                    score = new_score
                    improved = True
                else:
                    h[i] = h_io
                    h[j] = h_jo

        if not improved:
            break

    h = repair_vector(h, n)
    score = official_score(h)
    return h, score

def block_perturb(h, rng, block_size=None, scale_frac=None):
    n = len(h)
    if block_size is None:
        block_size = int(rng.integers(2, max(3, n // 4)))
    block_size = min(block_size, n)
    if block_size < 2:
        return h.copy()

    block = rng.choice(n, size=block_size, replace=False)
    d = rng.standard_normal(block_size)
    d -= np.mean(d)
    dmax = np.max(np.abs(d))
    if dmax < 1e-10:
        return h.copy()
    d /= dmax

    max_scale = 0.5
    for k in range(block_size):
        if d[k] > 0:
            max_scale = min(max_scale, (1.0 - h[block[k]]) / d[k])
        elif d[k] < 0:
            max_scale = min(max_scale, h[block[k]] / (-d[k]))

    if max_scale < 1e-10:
        return h.copy()

    if scale_frac is None:
        scale_frac = float(rng.uniform(0.1, 0.6))
    scale = scale_frac * max_scale

    h_new = h.copy()
    for k in range(block_size):
        h_new[block[k]] += scale * d[k]
    return repair_vector(h_new, n)

def multi_pair_line_descent(h_in, max_passes=20, n_scan=12, golden_iters=5,
                            deadline=None, seed=42, n_pairs_per_pass=None):
    """Faster variant: only process a subset of pairs per pass for larger n."""
    n = len(h_in)
    h = repair_vector(h_in, n).copy()
    score = official_score(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    for p in range(max_passes):
        if deadline is not None and time.perf_counter() > deadline:
            break
        improved = False
        rng.shuffle(all_pairs)
        n_pp = n_pairs_per_pass if n_pairs_per_pass else len(all_pairs)
        n_pp = min(n_pp, len(all_pairs))

        for (i, j) in all_pairs[:n_pp]:
            if deadline is not None and time.perf_counter() > deadline:
                break
            S = h[i] + h[j]
            L = max(0.0, S - 1.0)
            U = min(1.0, S)
            if U - L < 1e-10:
                continue

            h_io = h[i]
            h_jo = h[j]

            x_scan = np.linspace(L, U, n_scan)
            best_x = h_io
            best_val = np.inf

            for x in x_scan:
                h[i] = x
                h[j] = S - x
                val = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if val < best_val:
                    best_val = val
                    best_x = x

            best_k = int(np.argmin(np.abs(x_scan - best_x)))
            if 0 < best_k < n_scan - 1:
                a = x_scan[max(0, best_k - 2)]
                b = x_scan[min(n_scan - 1, best_k + 2)]
            elif best_k == 0:
                a, b = x_scan[0], x_scan[min(3, n_scan - 1)]
            else:
                a, b = x_scan[max(0, n_scan - 4)], x_scan[-1]

            if b - a > 1e-8:
                c = b - invphi * (b - a)
                d = a + invphi * (b - a)
                h[i] = c
                h[j] = S - c
                fc = np.max(np.correlate(h, 1.0 - h, mode="full"))
                h[i] = d
                h[j] = S - d
                fd = np.max(np.correlate(h, 1.0 - h, mode="full"))
                for _ in range(golden_iters):
                    if fc < fd:
                        b = d
                        d = c
                        fd = fc
                        c = b - invphi * (b - a)
                        h[i] = c
                        h[j] = S - c
                        fc = np.max(np.correlate(h, 1.0 - h, mode="full"))
                    else:
                        a = c
                        c = d
                        fc = fd
                        d = a + invphi * (b - a)
                        h[i] = d
                        h[j] = S - d
                        fd = np.max(np.correlate(h, 1.0 - h, mode="full"))
                x_mid = (a + b) / 2.0
                h[i] = x_mid
                h[j] = S - x_mid
                f_mid = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if f_mid < best_val:
                    best_val = f_mid
                    best_x = x_mid

            h[i] = best_x
            h[j] = S - best_x
            new_score = best_val * dx
            if new_score < score - 1e-13:
                score = new_score
                improved = True
            else:
                h[i] = h_io
                h[j] = h_jo

        if not improved:
            break

    h = repair_vector(h, n)
    score = official_score(h)
    return h, score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 8.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) >= 8 and np.all(np.isfinite(pv)):
            parent = repair_vector(pv, len(pv))
    except Exception:
        parent = None

    if parent is None:
        parent = repair_vector(np.full(64, 0.5), 64)

    n_parent = len(parent)
    best_h = parent.copy()
    best_score = official_score(best_h)
    rng = np.random.default_rng(seed)

    candidates = []
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h).copy())
    candidates.append((1.0 - best_h[::-1]).copy())

    for dn in [-12, -8, -5, -3, -1, 1, 3, 5, 8, 12, 16]:
        n_try = n_parent + dn
        if 20 <= n_try <= 100:
            for off in [0.0, 0.5, 1.0]:
                candidates.append(interp_to_n(parent, n_try, offset=off))

    for n_try in [n_parent, max(24, n_parent - 8), min(90, n_parent + 8), max(20, n_parent - 16)]:
        for _ in range(3):
            candidates.append(gen_binary(n_try, rng))
        for _ in range(3):
            candidates.append(gen_step(n_try, rng))
        for _ in range(2):
            candidates.append(gen_sine(n_try, rng))
        for _ in range(2):
            candidates.append(repair_vector(rng.uniform(0, 1, n_try), n_try))

    scored = []
    for h_c in candidates:
        if not np.all(np.isfinite(h_c)):
            continue
        h_c = repair_vector(h_c, len(h_c))
        s = official_score(h_c)
        if np.isfinite(s) and s >= 0:
            scored.append((s, h_c))

    scored.sort(key=lambda x: (x[0], len(x[1])))

    if len(scored) > 0 and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()

    # Phase 1: Pair-line descent on top candidates
    n_top = min(8, len(scored))
    for idx in range(n_top):
        if time.perf_counter() > deadline - 120:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_lim = min(remaining * 0.35, 25.0)
        dl = time.perf_counter() + t_lim

        h_loc, s_loc = pair_line_descent(h_c, max_passes=20, n_scan=15,
                                         golden_iters=8, deadline=dl, seed=seed + idx * 13)
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()

    # Phase 2: Basin hopping
    stall_count = 0
    for iteration in range(50):
        if time.perf_counter() > deadline - 40:
            break
        remaining = deadline - time.perf_counter()
        if remaining < 8:
            break

        scale_frac = 0.1 + 0.3 * (stall_count / 10.0)
        scale_frac = min(scale_frac, 0.6)
        bs = int(rng.integers(2, max(3, len(best_h) // 3)))
        h_mut = block_perturb(best_h, rng, block_size=bs, scale_frac=scale_frac)

        t_lim = min(remaining * 0.25, 12.0)
        dl = time.perf_counter() + t_lim
        h_loc, s_loc = pair_line_descent(h_mut, max_passes=12, n_scan=12,
                                         golden_iters=6, deadline=dl, seed=seed + 200 + iteration * 7)
        if s_loc < best_score - 1e-12:
            best_score = s_loc
            best_h = h_loc.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 20:
            break

    # Phase 3: Cross-resolution
    n_best = len(best_h)
    for dn in [-4, -2, -1, 1, 2, 4, 6]:
        if time.perf_counter() > deadline - 25:
            break
        n_try = n_best + dn
        if 16 <= n_try <= 100:
            h_interp = interp_to_n(best_h, n_try)
            t_lim = min(12.0, max(4.0, (deadline - time.perf_counter()) * 0.25))
            dl = time.perf_counter() + t_lim
            h_loc, s_loc = pair_line_descent(h_interp, max_passes=15, n_scan=15,
                                             golden_iters=8, deadline=dl, seed=seed + 600 + n_try)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()

    # Phase 4: Final strong polish
    if time.perf_counter() < deadline - 8:
        h_fin, s_fin = pair_line_descent(best_h, max_passes=25, n_scan=21,
                                         golden_iters=12, deadline=deadline - 3,
                                         seed=seed + 999)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = repair_vector(best_h, len(best_h))
    final_score = official_score(best_h)
    if final_score < 0:
        final_score = 0.0

    return (best_h, float(final_score), int(len(best_h)))
