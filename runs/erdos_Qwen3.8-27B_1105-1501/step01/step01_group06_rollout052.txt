def lp_direction(h, radius, lambda_reg=5.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx  # These are the actual C5 values
    R = len(scores)
    
    # Compute gradients
    grad = compute_gradients(h)  # shape (R, n)
    
    # Select active constraints: top K scores
    K = min(R, max(10, R // 2))
    top_idx = np.argsort(scores)[::-1][:K]
    
    # LP: minimize tau + lambda*r
    # Variables: p (n), r (1), tau (1) => n+2 total
    # Constraints:
    # For each active lag l: grad[l]·p - tau <= -scores[l]
    # p_i - r <= 0 for each i (n constraints)
    # -p_i - r <= 0 for each i (n constraints)
    # sum(p) = 0 (equality)
    # Bounds: p_i in [max(-radius, -h_i), min(radius, 1-h_i)]
    #         r in [0, radius]
    #         tau unbounded
    
    n_vars = n + 2  # p, r, tau
    # Objective: minimize tau + lambda*r
    c = np.zeros(n_vars)
    c[n] = lambda_reg  # r
    c[n+1] = 1.0  # tau
    
    A_ub_rows = []
    b_ub_rows = []
    
    # Active constraints: grad[l]·p - tau <= -scores[l]
    for l in top_idx:
        row = np.zeros(n_vars)
        row[:n] = grad[l]
        row[n+1] = -1.0  # -tau
        A_ub_rows.append(row)
        b_ub_rows.append(-scores[l])
    
    # p_i - r <= 0
    for i in range(n):
        row = np.zeros(n_vars)
        row[i] = 1.0
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(0.0)
    
    # -p_i - r <= 0
    for i in range(n):
        row = np.zeros(n_vars)
        row[i] = -1.0
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(0.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, radius))  # r
    bounds.append((None, None))  # tau
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        p = res.x[:n]
        return p
    return None

def line_search(h, p, current_c5):
    """Find best alpha in [0, alpha_max] minimizing C5(h + alpha*p)"""
    n = len(h)
    # Find alpha_max (feasibility bound)
    alpha_max = 1.0
    for i in range(n):
        if p[i] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-12:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    alpha_max = max(alpha_max, 0.0)
    
    if alpha_max < 1e-12:
        return 0.0, current_c5
    
    # Grid search
    alphas = np.geomspace(1e-6, alpha_max, 20)
    alphas = np.append(alphas, alpha_max)
    
    best_alpha = 0.0
    best_c5 = current_c5
    for a in alphas:
        h_trial = h + a * p
        # Project to ensure feasibility (should already be feasible since a <= alpha_max)
        h_trial = np.clip(h_trial, 0.0, 1.0)
        c5_val, _ = exact_c5(h_trial)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_alpha = a
    
    # Golden section refinement around best_alpha
    if best_alpha > 0:
        # Find interval containing best
        idx = np.where(alphas == best_alpha)[0][0]
        if idx > 0 and idx < len(alphas) - 1:
            lo = alphas[idx - 1]
            hi = alphas[idx + 1]
        else:
            lo = max(0, best_alpha * 0.5)
            hi = min(alpha_max, best_alpha * 1.5)
        
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(30):
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            h1 = np.clip(h + c1 * p, 0, 1)
            h2 = np.clip(h + c2 * p, 0, 1)
            c51, _ = exact_c5(h1)
            c52, _ = exact_c5(h2)
            if c51 < c52:
                hi = c2
            else:
                lo = c1
        best_alpha = (lo + hi) / 2
        h_trial = np.clip(h + best_alpha * p, 0, 1)
        best_c5, _ = exact_c5(h_trial)
    
    return best_alpha, best_c5

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx)), corr * dx

def project_to_target(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-10:
        if residual > 0:
            room = 1.0 - x
            room[room < 1e-15] = 0.0
            total_room = room.sum()
            if total_room > 1e-15:
                x = np.clip(x + room * (residual / total_room), 0.0, 1.0)
        else:
            space = x.copy()
            space[space < 1e-15] = 0.0
            total_space = space.sum()
            if total_space > 1e-15:
                x = np.clip(x - space * (-residual / total_space), 0.0, 1.0)
    return x

def compute_gradients(h):
    n = len(h)
    offset = n - 1
    R = 2 * n - 1
    grads = []
    for r in range(R):
        d = r - offset
        L = max(0, -d)
        U = min(n - 1, n - d - 1)
        g = np.zeros(n)
        if L <= U:
            # Term 1: j in [L+d, U+d] ∩ [0, n-1], contribution (1-h[j-d])
            j1_lo = max(0, L + d)
            j1_hi = min(n - 1, U + d)
            if j1_lo <= j1_hi:
                idx1 = np.arange(j1_lo, j1_hi + 1)
                g[idx1] += (1.0 - h[idx1 - d])
            # Term 2: j in [L, U], contribution -h[j+d]
            idx2 = np.arange(L, U + 1)
            g[idx2] += (-h[idx2 + d])
        grads.append(g)
    return np.array(grads)

def lp_local_step(h, radius, top_frac=0.5, lambda_reg=5.0):

    b_ub_list = []

        row[n + 1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-scores[l])

        row[i] = 1.0
        row[n] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(0.0)
        
        row2 = np.zeros(n_vars)
        row2[i] = -1.0
        row2[n] = -1.0
        A_ub_list.append(row2)
        b_ub_list.append(0.0)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

def line_search(h, p, current_c5, max_evals=60):

    for i in range(n):

def subgradient_direction(h, radius, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx

    grads = compute_gradients(h)

    g = np.zeros(n)
    for l in top_idx:
        g += grads[l]
    
    # Descent direction: -g, projected to zero-sum
    d = -g - np.mean(g)
    norm_d = np.max(np.abs(d))
    if norm_d < 1e-12:
        return None
    d = d / norm_d * radius
    # Clip to feasibility
    d = np.clip(d, -h + 1e-12, 1.0 - h - 1e-12)
    d[np.abs(d) < 1e-12] = 0.0
    # Re-enforce sum = 0
    d -= np.mean(d)
    return d
