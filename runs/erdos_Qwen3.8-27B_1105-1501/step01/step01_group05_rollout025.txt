        g[:n] = 1.0
        return g

def gen_block_pattern(n, rng, n_blocks):
    """Generate a pattern with n_blocks distinct values."""
    block_size = n // n_blocks
    remainder = n % n_blocks
    h = np.zeros(n)
    vals = rng.uniform(0, 1, n_blocks)
    pos = 0
    for b in range(n_blocks):
        size = block_size + (1 if b < remainder else 0)
        h[pos:pos+size] = vals[b]
        pos += size
    return repair_vector(h, n)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(300):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            # Spread the addition to elements with most room
            room = 1.0 - h
            for _ in range(10):
                idx = np.argmax(room)
                delta = min(residual, room[idx])
                h[idx] += delta
                residual -= delta
                if abs(residual) < 1e-15:
                    break
        else:
            for _ in range(10):
                idx = np.argmax(h)
                delta = min(-residual, h[idx])
                h[idx] -= delta
                residual += delta
                if abs(residual) < 1e-15:
                    break
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        idx = np.argmax(1.0 - h) if residual > 0 else np.argmax(h)
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient_row(h, k):
    n = len(h)
    p = np.arange(n)
    g = np.zeros(n)
    ok1 = (p - k) >= 0
    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g

def lse_descent(h, n, deadline, seed=42):
    """LSE-annealed gradient descent with momentum."""
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n)
    score = official_score(h)
    momentum = np.zeros(n)
    
    # Anneal beta
    betas = [3, 5, 10, 20, 50, 100, 200, 400, 800]
    
    for beta in betas:
        if time.perf_counter() > deadline - 5:
            break
        lr = 0.05
        mu = 0.5  # momentum
        
        for it in range(300):
            if time.perf_counter() > deadline - 5:
                break
            
            _, grad = lse_smoothed_obj_and_grad(h, beta)
            grad -= np.mean(grad)  # preserve sum constraint
            
            momentum = mu * momentum + (1 - mu) * grad
            direction = -momentum
            norm = np.max(np.abs(direction))
            if norm < 1e-15:
                break
            direction /= norm
            
            # Line search
            best_alpha = 0.0
            best_s = score
            for a in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                h_new = repair_vector(h + a * direction, n)
                s = official_score(h_new)
                if s < best_s - 1e-13:
                    best_s = s
                    best_alpha = a
            
            if best_alpha > 0:
                h = repair_vector(h + best_alpha * direction, n)
                score = best_s
                lr = min(lr * 1.1, 0.3)
            else:
                lr = lr * 0.5
                if lr < 0.001:
                    break
    
    return h, official_score(h)

def local_search(h, n, time_limit=15.0, seed=42):
    """Aggressive local search with multiple move types."""
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n)
    score = official_score(h)
    start = time.perf_counter()
    no_improve_count = 0
    
    while time.perf_counter() - start < time_limit:
        improved = False
        
        # Pair exchanges
        for _ in range(50):
            if time.perf_counter() - start > time_limit:
                return h, official_score(h)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                    continue
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                h_new = repair_vector(h_try, n)
                s = official_score(h_new)
                if s < score - 1e-13:
                    h = h_new
                    score = s
                    improved = True
                    break
        
        # Triple moves
        for _ in range(20):
            if time.perf_counter() - start > time_limit:
                return h, official_score(h)
            i, j, k = rng.integers(0, n, 3)
            if len(set([i, j, k])) < 3:
                continue
            for d1 in [0.05, 0.1, 0.2]:
                for d2 in [0.05, 0.1, 0.2]:
                    if h[i] + d1 > 1 or h[j] - d1 < 0 or h[k] - d2 < 0:
                        continue
                    h_try = h.copy()
                    h_try[i] += d1
                    h_try[j] -= d1
                    h_try[k] -= d2
                    # Need to maintain sum: add d2 somewhere
                    # Actually let's just do: move mass from j to i, and from k to i
                    # Net change: i gets d1+d2, j loses d1, k loses d2
                    # This preserves the sum!
                    h_new = repair_vector(h_try, n)
                    s = official_score(h_new)
                    if s < score - 1e-13:
                        h = h_new
                        score = s
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
    
    return h, official_score(h)

def slsqp_refine(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            j[ki, :n] = -dx * compute_gradient_row(h_x, k)
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 0.05
    
    x0 = np.concatenate([h, [theta0]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 500, 'ftol': 1e-14}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

def gen_perturbed(h, n, rng, scale=0.05):
    noise = rng.normal(0, scale, n)
    noise -= np.mean(noise)

def compute_gradient_row(h, k):
    n = len(h)
    p = np.arange(n)
    g = np.zeros(n)
    ok1 = (p - k) >= 0
    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g


def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    lags = np.arange(-(n - 1), n)
    max_s = np.max(scaled)
    exps = np.exp(beta * (scaled - max_s))
    weights = exps / exps.sum()
    grad = np.zeros(n)
    active = np.where(weights > 1e-9)[0]
    for ki in active:
        grad += weights[ki] * (dx * compute_gradient_row(h, lags[ki]))
    return grad


def lse_descent(h, n, deadline, seed=42):
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n)
    score = official_score(h)
    momentum = np.zeros(n)
    betas = [5, 10, 20, 50, 100, 200, 400, 800]
    for beta in betas:
        if time.perf_counter() > deadline - 5:
            break
        lr = 0.03
        mu = 0.4
        stall = 0
        for it in range(400):
            if time.perf_counter() > deadline - 3:
                break
            grad = lse_gradient(h, beta)
            grad -= np.mean(grad)
            momentum = mu * momentum + (1.0 - mu) * grad
            direction = -momentum
            norm = np.max(np.abs(direction))
            if norm < 1e-15:
                break
            direction /= norm
            best_alpha = 0.0
            best_s = score
            for a in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
                h_new = repair_vector(h + a * direction, n)
                s = official_score(h_new)
                if s < best_s - 1e-14:
                    best_s = s
                    best_alpha = a
            if best_alpha > 0:
                h = repair_vector(h + best_alpha * direction, n)
                score = best_s
                stall = 0
                lr = min(lr * 1.05, 0.2)
            else:
                stall += 1
                lr *= 0.6
                if lr < 0.001:
                    break
                if stall > 30:
                    break
    return h, official_score(h)


def local_search(h, n, time_limit=15.0, seed=42):
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n)
    score = official_score(h)
    start = time.perf_counter()
    no_imp = 0
    deltas = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]
    while time.perf_counter() - start < time_limit:
        improved = False
        for _ in range(80):
            if time.perf_counter() - start > time_limit:
                return h, official_score(h)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for d in deltas:
                if h[i] + d > 1.0 or h[j] - d < 0.0:
                    continue
                h_try = h.copy()
                h_try[i] += d
                h_try[j] -= d
                h_new = repair_vector(h_try, n)
                s = official_score(h_new)
                if s < score - 1e-14:
                    h = h_new
                    score = s
                    improved = True
                    break
        for _ in range(15):
            if time.perf_counter() - start > time_limit:
                return h, official_score(h)
            i, j, k = rng.integers(0, n, 3)
            if len(set([i, j, k])) < 3:
                continue
            d1 = rng.choice([0.03, 0.05, 0.1, 0.15])
            d2 = rng.choice([0.03, 0.05, 0.1, 0.15])
            if h[i] + d1 + d2 > 1.0 or h[j] - d1 < 0.0 or h[k] - d2 < 0.0:
                continue
            h_try = h.copy()
            h_try[i] += d1 + d2
            h_try[j] -= d1
            h_try[k] -= d2
            h_new = repair_vector(h_try, n)
            s = official_score(h_new)
            if s < score - 1e-14:
                h = h_new
                score = s
                improved = True
        if not improved:
            no_imp += 1
            if no_imp > 30:
                break
        else:
            no_imp = 0
    return h, official_score(h)


def slsqp_refine(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            j[ki, :n] = -dx * compute_gradient_row(h_x, k)
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 0.05
    x0 = np.concatenate([h, [theta0]])
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 500, 'ftol': 1e-15}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def gen_perturbed(h, n, rng, scale=0.03):
    noise = rng.normal(0, scale, n)
    noise -= np.mean(noise)
    return repair_vector(h + noise, n)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 5.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = repair_vector(parent, n_parent)
    else:
        n_parent = 65
        best_h = repair_vector(np.full(65, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    candidates = [best_h.copy()]
    n_opts = sorted(set([n_parent - 5, n_parent, n_parent + 5, n_parent + 10,
                         n_parent + 15, n_parent + 20, 55, 70, 80]))
    n_opts = [x for x in n_opts if 16 <= x <= 100]

    for n_try in n_opts:
        candidates.append(gen_interpolated(best_h, n_try, 0))
        candidates.append(gen_interpolated(best_h, n_try, 1))
        candidates.append(gen_perturbed(best_h, min(n_try, n_parent), rng, 0.04))

    scored = []
    seen = set()
    for h_c in candidates:
        key = tuple(np.round(h_c, 4))
        if key in seen:
            continue
        seen.add(key)
        h_c = repair_vector(h_c, len(h_c))
        s = official_score(h_c)
        if np.isfinite(s):
            scored.append((s, h_c))

    scored.sort(key=lambda x: x[0])

    for s_c, h_c in scored[:5]:
        if s_c < best_score:
            best_score = s_c
            best_h = h_c.copy()
            best_n = len(h_c)
            break

    # Phase 1: LSE descent on top candidates
    n_trials = min(6, len(scored))
    for idx in range(n_trials):
        if time.perf_counter() > deadline - 80:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        h_lse, s_lse = lse_descent(h_c, n_c, deadline - 10, seed=seed + idx * 19)
        if s_lse < best_score:
            best_score = s_lse
            best_h = h_lse.copy()
            best_n = n_c

    # Phase 2: Local search on best
    if time.perf_counter() < deadline - 50:
        t_lim = min(20.0, (deadline - time.perf_counter()) * 0.25)
        h_ls, s_ls = local_search(best_h, best_n, time_limit=t_lim, seed=seed + 300)
        if s_ls < best_score:
            best_score = s_ls
            best_h = h_ls.copy()

    # Phase 3: SLSQP
    if time.perf_counter() < deadline - 30:
        h_sl, s_sl = slsqp_refine(best_h, best_n, time_limit=15.0)
        if s_sl < best_score:
            best_score = s_sl
            best_h = h_sl.copy()

    # Phase 4: Local search after SLSQP
    if time.perf_counter() < deadline - 15:
        h_ls2, s_ls2 = local_search(best_h, best_n, time_limit=5.0, seed=seed + 500)
        if s_ls2 < best_score:
            best_score = s_ls2
            best_h = h_ls2.copy()

    # Phase 5: Try higher resolution
    if time.perf_counter() < deadline - 30:
        for n_try in [best_n + 15, best_n + 30]:
            if time.perf_counter() > deadline - 20:
                break
            h_int = gen_interpolated(best_h, n_try)
            h_lse2, s_lse2 = lse_descent(h_int, n_try, deadline - 10, seed=seed + 700 + n_try)
            if s_lse2 < best_score:
                best_score = s_lse2
                best_h = h_lse2.copy()
                best_n = n_try
            h_ls3, s_ls3 = local_search(h_lse2, n_try, time_limit=5.0, seed=seed + 900)
            if s_ls3 < best_score:
                best_score = s_ls3
                best_h = h_ls3.copy()
                best_n = n_try
            if time.perf_counter() < deadline - 10:
                h_sl2, s_sl2 = slsqp_refine(best_h, best_n, time_limit=5.0)
                if s_sl2 < best_score:
                    best_score = s_sl2
                    best_h = h_sl2.copy()

    # Phase 6: Final SLSQP + local polish at current best
    if time.perf_counter() < deadline - 8:
        h_fin, s_fin = slsqp_refine(best_h, best_n, time_limit=5.0)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()
        if time.perf_counter() < deadline - 3:
            h_fin2, s_fin2 = local_search(best_h, best_n, time_limit=3.0, seed=seed + 1100)
            if s_fin2 < best_score:
                best_score = s_fin2
                best_h = h_fin2.copy()

    # Also try a second LSE pass from a slightly perturbed best
    if time.perf_counter() < deadline - 15:
        h_pert = gen_perturbed(best_h, best_n, rng, 0.02)
        h_lse3, s_lse3 = lse_descent(h_pert, best_n, deadline - 5, seed=seed + 1300)
        if s_lse3 < best_score:
            best_score = s_lse3
            best_h = h_lse3.copy()
        h_ls4, s_ls4 = local_search(best_h, best_n, time_limit=3.0, seed=seed + 1400)
        if s_ls4 < best_score:
            best_score = s_ls4
            best_h = h_ls4.copy()

    best_h = repair_vector(best_h, len(best_h))
    final_score = official_score(best_h)
    best_score = min(best_score, final_score)
    if best_score < 0:
        best_score = 0.0

    return (best_h, float(best_score), int(len(best_h)))
