def project_to_target(v, target):

def lp_direction(h, n, radius, mu, max_lags=None):
    """Find LP direction to reduce max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr  # raw scores
    
    # Select lags: top K active + some spread
    if max_lags is None or max_lags >= 2*n - 1:
        selected = list(range(2*n - 1))
    else:

        idx = np.argsort(scores)[::-1]

    for i, m in enumerate(selected):

    # Variables: p (n), r (1), tau (1) -> total n+2
    # Constraints:
    #   grads[i] · p - tau <= -scores[m_i] for each i (K constraints)
    #   p_j - r <= 0 for each j (n constraints)
    #   -p_j - r <= 0 for each j (n constraints)
    # Equality: sum(p) = 0
    # Bounds: p_j in [max(-radius, -h_j), min(radius, 1-h_j)], r in [0, radius], tau unbounded (lower)
    
    c_obj = np.zeros(n + 2)
    c_obj[n] = mu  # coefficient for r
    c_obj[n + 1] = 1.0  # coefficient for tau
    
    A_ub = np.zeros((K + 2*n, n + 2))
    b_ub = np.zeros(K + 2*n)
    
    # Score constraints: grads[i]·p - tau <= -scores[m_i]
    for i in range(K):
        A_ub[i, :n] = grads[i]
        A_ub[i, n+1] = -1.0
        b_ub[i] = -scores[i]
    
    # p_j - r <= 0
    for j in range(n):
        A_ub[K + j, j] = 1.0
        A_ub[K + j, n] = -1.0
        b_ub[K + j] = 0.0
    
    # -p_j - r <= 0
    for j in range(n):
        A_ub[K + n + j, j] = -1.0
        A_ub[K + n + j, n] = -1.0
        b_ub[K + n + j] = 0.0

    A_eq = np.zeros((1, n + 2))

        p -= np.mean(p)
        return p
    return None

            best_c5 = c5_test
            best_alpha = a
    
    # Golden section around best
    if 0 < best_alpha < alpha_max:
        # Refine
        lo = max(0, best_alpha - alpha_max/10)
        hi = min(alpha_max, best_alpha + alpha_max/10)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a = lo
        b = hi
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        fc = exact_c5(h + c * p)
        fd = exact_c5(h + d * p)
        for _ in range(30):
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = b - gr * (b - a)
                fc = exact_c5(h + c * p)
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                fd = exact_c5(h + d * p)
            if abs(b - a) < 1e-10:
                break
        a_mid = (a + b) / 2.0
        h_mid = h + a_mid * p
        c5_mid = exact_c5(h_mid)
        if c5_mid < best_c5:
            best_c5 = c5_mid
            best_h = h_mid
    
    return best_h, best_c5

def lp_direction(h, n, radius, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr
    n_lags = 2 * n - 1
    
    # Use all lags for small n
    selected = list(range(n_lags))
    
    K = len(selected)
    grads = np.zeros((K, n))
    for i, m in enumerate(selected):
        grads[i] = grad_single(h, n, m)
    
    # LP: min tau + mu*r
    # vars: p(n), r(1), tau(1)
    c_obj = np.zeros(n + 2)
    c_obj[n] = mu
    c_obj[n + 1] = 1.0
    
    n_ineq = K + 2 * n
    A_ub = np.zeros((n_ineq, n + 2))
    b_ub = np.zeros(n_ineq)
    
    for i in range(K):
        A_ub[i, :n] = grads[i]
        A_ub[i, n + 1] = -1.0
        b_ub[i] = -scores[i]
    
    for j in range(n):
        A_ub[K + j, j] = 1.0
        A_ub[K + j, n] = -1.0
        b_ub[K + j] = 0.0
        A_ub[K + n + j, j] = -1.0
        A_ub[K + n + j, n] = -1.0
        b_ub[K + n + j] = 0.0
    
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, radius))
    bounds.append((None, None))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        p = res.x[:n].copy()
        p -= np.mean(p)
        if np.max(np.abs(p)) > 1e-12:
            return p
    return None

def subgrad_direction(h, n, radius):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr
    max_s = np.max(scores)
    threshold = max_s - 0.005
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    
    grad_sum = np.zeros(n)
    for m in active:
        grad_sum += grad_single(h, n, m)
    grad_sum /= len(active)
    
    d = -grad_sum
    d -= np.mean(d)
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-15:
        return None
    d = d / max_abs * radius
    # Clip to feasible
    for i in range(n):
        d[i] = min(d[i], 1.0 - h[i])
        d[i] = max(d[i], -h[i])
    return d

def lp_local_opt(h, n, time_limit):

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    old_x = (np.arange(n_old) + 0.5) / n_old
    new_x = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(new_x, old_x, h_old)
    return project_to_target(h_new, n_new / 2.0)
