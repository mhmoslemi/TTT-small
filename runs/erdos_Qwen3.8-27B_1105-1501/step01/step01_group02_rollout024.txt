# Global state for L-BFGS-B callbacks (since we can't use closures)
_g_state = {}

def _smooth_obj(x):

    T = _g_state['T']
    rho = _g_state['rho']
    target_sum = _g_state['target_sum']
    n = len(h)

    # penalty
    pen = rho * (np.sum(h) - target_sum) ** 2
    return log_sum + pen

import numpy as np
import time
from scipy.optimize import minimize

_gs = {}

def _obj(x):
    T = _gs['T']
    rho = _gs['rho']
    target = _gs['target']
    dx = _gs['dx']
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    gm = np.max(g)
    lg = np.logaddexp.reduce((g - gm) / T) - np.log(len(g))
    logsum = gm + T * lg
    pen = rho * (np.sum(x) - target) ** 2
    return logsum + pen

def _grad(x):
    T = _gs['T']
    rho = _gs['rho']
    target = _gs['target']
    dx = _gs['dx']
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    gm = np.max(g)
    ev = np.exp((g - gm) / T)
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        s = ki - (n - 1)
        wi = w[ki] * dx
        if wi < 1e-15:
            continue
        if s >= 0:
            grad[s:] += wi * (1.0 - x[:n - s])
            grad[:n - s] -= wi * x[s:]
        else:
            m = -s
            grad[:n - m] += wi * (1.0 - x[m:])
            grad[m:] -= wi * x[:n - m]
    grad += 2.0 * rho * (np.sum(x) - target)
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(r)
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(r < 1.0 - 1e-9)[0]
        else:
            idx = np.where(r > 1e-9)[0]
        if len(idx) > 0:
            r[idx] += res / len(idx)
            r = np.clip(r, 0.0, 1.0)
    return r

def make_complement(h):
    n = len(h)
    return 1.0 - project(h.copy(), n / 2.0)

def make_shift(h, s):
    return np.roll(h, s)

def smooth_optimize(h, deadline, rng, target, T_list):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best = h.copy()
    for T in T_list:
        if time.time() > deadline:
            break
        _gs['T'] = T
        _gs['rho'] = 50.0
        _gs['target'] = target
        _gs['dx'] = dx
        bounds = [(0.0, 1.0)] * n
        maxiter = min(300, max(30, int(60.0 * n / 70)))
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-10, 'gtol': 1e-7})
            x = np.clip(res.x, 0.0, 1.0)
            x = project(x, target)
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best = x.copy()
            h = x.copy()
        except Exception:
            pass
    return best, best_c5

def compute_b(h, i, j):
    n = len(h)
    b = np.zeros(2 * n - 1)
    for ki in range(2 * n - 1):
        s = ki - (n - 1)
        if s >= 0:
            ci = 0.0
            if s <= i < n:
                ci += (1.0 - h[i - s])
            if i < n - s:
                ci -= h[i + s]
            cj = 0.0
            if s <= j < n:
                cj += (1.0 - h[j - s])
            if j < n - s:
                cj -= h[j + s]
        else:
            m = -s
            ci = 0.0
            if i < n - m:
                ci += (1.0 - h[i + m])
            if m <= i < n:
                ci -= h[i - m]
            cj = 0.0
            if j < n - m:
                cj += (1.0 - h[j + m])
            if m <= j < n:
                cj -= h[j - m]
        b[ki] = ci - cj
    return b

def pair_line_search(h, i, j, corr):
    n = len(h)
    dx = 2.0 / n
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-13:
        return 0.0, np.max(corr) * dx
    b = compute_b(h, i, j)
    a = corr
    for _ in range(35):
        m1 = t_lo + (t_hi - t_lo) / 3.0
        m2 = t_hi - (t_hi - t_lo) / 3.0
        f1 = np.max(a + b * m1)
        f2 = np.max(a + b * m2)
        if f1 < f2:
            t_hi = m2
        else:
            t_lo = m1
    t = (t_lo + t_hi) / 2.0
    c5v = float(np.max(a + b * t) * dx)
    return t, c5v

def phase_pair_search(h, best_c5, deadline, rng, n_iter_per_sweep=500):
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_c5 = best_c5
    improved_count = 0
    total_iters = 0
    while time.time() < deadline:
        batch_end = min(deadline, time.time() + 5.0)
        while time.time() < batch_end:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            corr = np.correlate(h, 1.0 - h, mode='full')
            t, c5_new = pair_line_search(h, i, j, corr)
            if abs(t) > 1e-12 and c5_new < best_c5 - 1e-14:
                h[i] += t
                h[j] -= t
                h = np.clip(h, 0.0, 1.0)
                best_c5 = c5_new
                best = h.copy()
                improved_count += 1
            total_iters += 1
        if improved_count == 0 and total_iters > 2000:
            h = best + rng.randn(n) * 0.04
            h = project(h, n / 2.0)
            best_c5 = compute_c5(best)
            improved_count = 0
            total_iters = 0
    return best, best_c5

def make_fallback(n, rng):
    h = np.zeros(n)
    phi = 0.6180339887498948
    for i in range(n // 2):
        pos = int((i * phi * n) % n)
        h[pos] = 1.0
    return project(h, n / 2.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old)
    x_new = np.arange(n_new) * n_old / n_new
    h_new = np.interp(x_new, x_old, np.concatenate([h, h[:1]]), period=n_old)
    return project(h_new, n_new / 2.0)

def downsample(h, n_new):
    n_old = len(h)
    idx = (np.arange(n_new) * n_old / n_new).astype(int)
    idx = np.minimum(idx, n_old - 1)
    h_new = h[idx]
    return project(h_new, n_new / 2.0)
