- If k ≥ 0: k ≤ j < n+k, i.e., j ∈ [k, n-1+k] ∩ [0, n-1] = [k, n-1] (since n-1+k > n-1 for k>0, so j < n)
  Wait, j < n+k and j < n (since j is an index). So j < min(n, n+k) = n (for k≥0).
  And j ≥ k. So j ∈ [k, n-1].
  
- If k < 0: k ≤ j < n+k. Since k < 0, j ≥ k is always true (j ≥ 0 > k).
  j < n+k. So j ∈ [0, n+k-1].

Combined: the first term is nonzero for j ∈ [max(0,k), min(n-1, n+k-1)].
For k ≥ 0: j ∈ [k, n-1]
For k < 0: j ∈ [0, n+k-1] = [0, n-1+|k|-1]... wait n+k-1 for k<0.
  If k = -1: j ∈ [0, n-2]
  If k = -(n-1): j ∈ [0, 0]

For the second term: -k ≤ j < n-k.
- If k ≥ 0: -k ≤ j is always true (j ≥ 0 ≥ -k). j < n-k. So j ∈ [0, n-k-1].
- If k < 0: -k ≤ j means j ≥ -k = |k|. j < n-k = n+|k|. Since j < n, j ∈ [|k|, n-1].

Combined: the second term is nonzero for j ∈ [max(0,-k), min(n-1, n-k-1)].
For k ≥ 0: j ∈ [0, n-k-1]
For k < 0: j ∈ [-k, n-1]

∂F/∂h[j] = sum_k w_k * [(1-h[j-k]) * 1_{valid1} - h[j+k] * 1_{valid2}]

where w_k = weights[k+(n-1)] is the weight for lag k.

```python
    for k in range(-(n-1), n):
        wk = weights[k + (n-1)]
        if wk < 1e-15:

        # Valid i: max(0, -k) <= i < min(n, n-k)

            # (1-h[i]) contributes -h[i+k] to dc[k]/dh[i]
            j2 = i
            grad[j2] -= wk * h[i+k]
    return grad
```

```python
import numpy as np
import time

def compute_corr(h):
    """Returns correlation values (without dx scaling)"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def compute_c5_and_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    vals = corr * dx
    return vals, float(np.max(vals))

def project_simplex(h, target):
    """Project h to {0<=x<=1, sum(x)=target}"""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_grad(h, weights):
    """Compute gradient of sum_k weights[k] * c_k where c_k = corr values (unscaled)"""
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        wk = weights[k_idx]
        if wk < 1e-15:
            continue
        k = k_idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        for i in range(i_lo, i_hi):
            j1 = i + k
            grad[j1] += wk * (1.0 - h[i])
            grad[i] -= wk * h[i + k]
    return grad

def optimize_target_feasibility(h0, time_budget, seed=42):
    """Optimize using target-feasibility approach"""
    n = len(h0)
    target = float(n) / 2.0
    h = project_simplex(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    # Initial parameters
    alpha = 3.0  # sharpness of soft penalty
    lr = 0.05
    decay = 0.995
    
    # Target starts slightly below current max
    vals, c5_max = compute_c5_and_vals(h)
    tau = c5_max - 0.001  # target slightly below current
    
    v = np.zeros(n)  # momentum
    mom = 0.9
    stall = 0
    best_stall_alpha = alpha
    
    while time.time() - t0 < time_budget:
        # Compute weights: soft penalty for correlations exceeding tau
        vals, c5_max = compute_c5_and_vals(h)
        excess = np.maximum(vals - tau, 0.0)
        raw_w = np.exp(alpha * excess)
        total = raw_w.sum()
        if total < 1e-10:
            weights = np.zeros(len(raw_w))
        else:
            weights = raw_w / total
        
        # Compute gradient
        grad = weighted_grad(h, weights)
        
        # Momentum step
        v = mom * v - lr * grad
        h_new = h + v
        h_new = project_simplex(h_new, target)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            # Update tau to be slightly below new max
            vals_new, _ = compute_c5_and_vals(h_new)
            tau = c5_new - 0.0005
        else:
            stall += 1
            # If stalled, try smaller step
            if stall > 50:
                lr *= 0.8
                v *= 0.5
        
        h = h_new
        
        # Adaptive alpha: increase when making progress, decrease when stalled
        if stall == 0:
            alpha = min(alpha * 1.05, 50.0)
        elif stall > 100:
            alpha = max(alpha * 0.9, 1.0)
            # Try a perturbation
            if stall > 200:
                rng = np.random.RandomState(seed + stall)
                h = best_h + 0.03 * rng.randn(n)
                h = project_simplex(h, target)
                v = np.zeros(n)
                lr = 0.03
                stall = 0
                tau = best_c5 - 0.001
        
        lr *= decay
    
    return best_h, best_c5


def optimize_annealing(h0, time_budget, seed=42):
    """Simulated annealing with informed moves"""
    n = len(h0)
    target = float(n) / 2.0
    h = project_simplex(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    T = 0.01  # initial temperature
    T_min = 0.0001
    cooling = 0.999
    
    while time.time() - t0 < time_budget:
        # Informed move: pick a pair (i, j) based on correlation structure
        # Try to reduce the max correlation by moving mass from positions
        # that contribute to high correlations to positions that don't
        vals, c5_max = compute_c5_and_vals(h)
        max_lag = np.argmax(vals)
        k_max = max_lag - (n - 1)
        
        # Positions that contribute to the max correlation
        i_lo = max(0, -k_max)
        i_hi = min(n, n - k_max)
        # h[i+k_max] * (1-h[i]) for i in [i_lo, i_hi)
        # To reduce this, we want to decrease h[i+k_max] or increase h[i]
        # (subject to sum constraint)
        
        # Pick a random valid i
        i = rng.randint(i_lo, i_hi)
        j = i + k_max
        
        # Move delta from j to i (or vice versa)
        delta = rng.uniform(0.01, 0.1) * (1 if h[j] > h[i] else -1)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            # Try a different move
            delta = -delta * 0.5
            h_new[i] += delta  # now adding 2*delta to i
            h_new[j] -= delta
            if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                h_new = h  # reject
                c5_new = best_c5
            else:
                c5_new = compute_c5(h_new)
        else:
            c5_new = compute_c5(h_new)
        
        # Accept or reject
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            T = max(T * 1.001, 0.005)  # slight reheat on improvement
        else:
            delta_c = c5_new - c5_max
            if delta_c < 0 or rng.expovariate(1.0 / max(T, T_min)) > delta_c:
                h = h_new
        
        T *= cooling
    
    return best_h, best_c5


def multi_start_optimize(n, time_budget, seed=42, n_starts=3):
    """Try multiple random starts and return the best"""
    rng = np.random.RandomState(seed)
    target = float(n) / 2
    best_h = None
    best_c5 = float('inf')
    
    # Split time among starts
    t_per_start = time_budget / n_starts
    t0 = time.time()
    
    for s in range(n_starts):
        if time.time() - t0 >= time_budget - 5:
            break
        
        # Different initialization strategies
        if s == 0:
            # Block initialization: first half 1s, second half 0s, then shuffle
            h = np.zeros(n)
            h[:n//2] = 1.0
            if n % 2 == 1:
                h[n//2] = 0.5
            h = h[rng.permutation(n)]
        elif s == 1:
            # Spread initialization
            h = np.zeros(n)
            positions = np.arange(0, n, max(1, n // (n//2)))
            h[positions[:n//2]] = 1.0
            if n % 2 == 1 and n//2 < len(positions):
                h[positions[n//2]] = 0.5
            # Fill remaining
            need = int(target - h.sum())
            if need > 0:
                zeros = np.where(h == 0)[0]
                h[rng.choice(zeros, size=min(need, len(zeros)), replace=False)] = 1.0
        else:
            # Random
            h = rng.rand(n)
        
        h = project_simplex(h, target)
        
        sub_budget = min(t_per_start, time_budget - (time.time() - t0) - 5)

        h_best, c5_best = optimize_target_feasibility(h, sub_budget, seed + s * 100)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best
    
    return best_h, best_c5
```

```python
def binary_c5(S, n):
    """Compute C5 for a binary vector defined by set S (positions of 1s)"""
    h = np.zeros(n)
    h[S] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def binary_c5_fast(S, n):
    """Faster computation using the fact that h is binary"""
    h = np.zeros(n)
    h[S] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
```

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_simplex(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_binary(h, n):
    """Convert h to a binary (or near-binary) vector with the correct sum"""
    if n % 2 == 0:
        # n/2 ones, n/2 zeros
        idx = np.argsort(h)[::-1]
        out = np.zeros(n)
        out[idx[:n//2]] = 1.0
        return out
    else:
        # (n-1)/2 ones, one 0.5, (n-1)/2 zeros
        idx = np.argsort(h)[::-1]
        out = np.zeros(n)
        out[idx[:n//2]] = 1.0
        out[idx[n//2]] = 0.5
        return out

def sa_binary(h_start, time_budget, seed=42, T_init=0.003, T_min=1e-5, cooling=0.9997):
    """Simulated annealing on binary solutions"""
    n = len(h_start)
    h = make_binary(h_start, n)  # start from binary
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    T = T_init
    ones_idx = np.where(h > 0.9)[0]
    zeros_idx = np.where(h < 0.1)[0]
    
    no_improve = 0
    reheat_threshold = 50000
    
    while time.time() - t0 < time_budget:
        batch = 1000
        for _ in range(batch):
            if time.time() - t0 >= time_budget:
                break
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h)
            
            if c5 <= best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
                # Update indices
                ones_idx[ones_idx == i] = j
                zeros_idx[zeros_idx == j] = i
            else:
                delta = c5 - best_c5
                if rng.expovariate(1.0 / max(T, T_min)) > delta:
                    # Accept
                    ones_idx[ones_idx == i] = j
                    zeros_idx[zeros_idx == j] = i
                else:
                    h[i], h[j] = h[j], h[i]
                no_improve += 1
        
        T = max(T * cooling, T_min)
        
        # Reheat if stuck
        if no_improve > reheat_threshold:
            T = T_init * 0.5
            no_improve = 0
            # Also try a random restart from best
            if rng.random() < 0.3:
                h = best_h + 0.1 * rng.randn(n)
                h = make_binary(h, n)
                ones_idx = np.where(h > 0.9)[0]
                zeros_idx = np.where(h < 0.1)[0]
    
    return best_h, best_c5

def gradient_refine(h0, time_budget, seed=42):
    """Gradient-based refinement allowing non-binary values"""
    n = len(h0)
    target = float(n) / 2
    h = project_simplex(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    alpha = 20.0
    lr = 0.01
    v = np.zeros(n)
    mom = 0.9
    
    while time.time() - t0 < time_budget:
        # Compute weights based on current correlation values
        vals = compute_corr_scaled(h)
        max_val = np.max(vals)
        tau = max_val * 0.999  # target just below max
        excess = np.maximum(vals - tau, 0)
        weights = np.exp(alpha * excess)
        weights = weights / (weights.sum() + 1e-15)
        
        # Compute gradient (O(n²))
        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            wk = weights[k_idx]
            if wk < 1e-15:
                continue
            k = k_idx - (n - 1)
            i_lo = max(0, -k)
            i_hi = min(n, n - k)
            if i_lo >= i_hi:
                continue
            i_arr = np.arange(i_lo, i_hi)
            j_arr = i_arr + k
            grad[j_arr] += wk * (1.0 - h[i_arr])
            grad[i_arr] -= wk * h[j_arr]
        
        # Momentum step
        v = mom * v - lr * grad
        h_new = project_simplex(h + v, target)
        c5 = compute_c5(h_new)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            lr = min(lr * 1.01, 0.05)
        else:
            lr = max(lr * 0.95, 0.001)
        
        h = h_new
        alpha = min(alpha * 1.02, 100.0)
    
    return best_h, best_c5

    """Final polish with random pair moves"""
    n = len(h)
    target = float(n) / 2
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    while time.time() - t0 < time_budget:

            h_test[j] -= delta

                c5 = compute_c5(h_test)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h_test.copy()
    
    return best_h, best_c5
