def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def coord_coeffs(h, i):
    """Compute dO_k/dh[i] for all k, given current h."""
    n = len(h)
    dx = 2.0 / n
    coeffs = np.zeros(2 * n - 1)
    # Forward shifts: k = 0 to n-1
    for k in range(n):
        c = 0.0
        if k <= i:
            c += (1.0 - h[i - k])
        if i + k < n:
            c -= h[i + k]
        coeffs[k] = c * dx
    # Backward shifts: d = 1 to n-1, k = n-1+d
    for d in range(1, n):
        c = 0.0
        if i + d < n:
            c += (1.0 - h[i + d])
        if d <= i:
            c -= h[i - d]
        coeffs[n - 1 + d] = c * dx
    return coeffs

def find_optimal_t(base, coeffs):
    """Find t in [0,1] minimizing max_k (base[k] + coeffs[k]*t)."""
    m = len(base)
    
    def feasible(val):
        lo_t = 0.0
        hi_t = 1.0
        for k in range(m):
            ck = coeffs[k]
            if abs(ck) < 1e-15:
                if base[k] > val:
                    return False
            elif ck > 0:
                bound = (val - base[k]) / ck
                if bound < hi_t:
                    hi_t = bound
                    if hi_t < lo_t - 1e-15:
                        return False
            else:
                bound = (val - base[k]) / ck
                if bound > lo_t:
                    lo_t = bound
                    if lo_t > hi_t + 1e-15:
                        return False
        return lo_t <= hi_t + 1e-12
    
    # Determine search range
    # At t=0: max is max(base)
    # At t=1: max is max(base + coeffs)
    f0 = float(np.max(base))
    f1 = float(np.max(base + coeffs))
    
    # The minimum is at most min(f0, f1)
    # Lower bound: max over k of min(base[k], base[k]+coeffs[k]) (each line's min on [0,1])
    lb = 0.0
    for k in range(m):
        ck = coeffs[k]
        if abs(ck) < 1e-15:
            lb = max(lb, base[k])
        elif ck > 0:
            lb = max(lb, base[k])  # min at t=0
        else:
            lb = max(lb, base[k] + ck)  # min at t=1
    ub = min(f0, f1)
    
    if ub - lb < 1e-15:
        mid_val = ub
    else:
        for _ in range(40):
            mid_val = (lb + ub) * 0.5
            if feasible(mid_val):
                ub = mid_val
            else:
                lb = mid_val
        mid_val = (lb + ub) * 0.5
    
    # Find the optimal t
    lo_t = 0.0
    hi_t = 1.0
    for k in range(m):
        ck = coeffs[k]
        if abs(ck) < 1e-15:
            continue
        elif ck > 0:
            bound = (mid_val - base[k]) / ck
            if bound < hi_t:
                hi_t = bound
        else:
            bound = (mid_val - base[k]) / ck
            if bound > lo_t:
                lo_t = bound
    
    lo_t = max(0.0, lo_t)
    hi_t = min(1.0, hi_t)
    if lo_t > hi_t:
        return 0.5  # fallback
    return (lo_t + hi_t) * 0.5

def coord_descent_sweep(h, order, overlaps):
    """One sweep of coordinate descent. Returns updated h and overlaps."""
    n = len(h)
    for i in order:
        h_i_old = h[i]
        coeffs = coord_coeffs(h, i)
        base = overlaps - coeffs * h_i_old
        t_opt = find_optimal_t(base, coeffs)
        h[i] = t_opt
        delta = t_opt - h_i_old
        if abs(delta) > 1e-15:
            overlaps += coeffs * delta
    return h, overlaps

def coord_descent(h0, max_sweeps, deadline, rng):
    n = len(h0)
    h = h0.copy()
    overlaps = compute_overlaps(h)
    cur_c = float(np.max(overlaps))
    best = h.copy()
    best_c = cur_c
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = list(range(n))
        rng.shuffle(order)
        h, overlaps = coord_descent_sweep(h, order, overlaps)
        cur_c = float(np.max(overlaps))
        if cur_c < best_c - 1e-14:
            best = h.copy()
            best_c = cur_c
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Perturb
                h = make_feasible(h + rng.normal(0, 0.02, n), n)
                overlaps = compute_overlaps(h)
                cur_c = float(np.max(overlaps))
                no_improve = 0
    
    return best, best_c

def dual_coord_descent(h0, deadline, rng, start_T=0.05):
    """Coordinate descent with occasional annealing."""
    n = len(h0)
    h = h0.copy()
    overlaps = compute_overlaps(h)
    cur_c = float(np.max(overlaps))
    best = h.copy()
    best_c = cur_c
    T = start_T
    
    while time.time() < deadline:
        # Do a few sweeps
        for _ in range(3):
            if time.time() > deadline:
                break
            order = list(range(n))
            rng.shuffle(order)
            h, overlaps = coord_descent_sweep(h, order, overlaps)
            cur_c = float(np.max(overlaps))
            if cur_c < best_c - 1e-14:
                best = h.copy()
                best_c = cur_c
        
        # Occasionally perturb
        if rng.random() < 0.05:
            h = make_feasible(h + rng.normal(0, T, n), n)
            overlaps = compute_overlaps(h)
            cur_c = float(np.max(overlaps))
        
        T *= 0.99
        
        if time.time() > deadline:
            break
    
    return best, best_c
