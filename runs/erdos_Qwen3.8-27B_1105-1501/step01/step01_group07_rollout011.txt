    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    alpha = 0.1
    stall = 0

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

                cur_c = score_exact(cur)

                stall = 0

import numpy as np
import time
from scipy.optimize import minimize

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(y, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(z))
            z[idx] = min(1.0, z[idx] + diff)
        else:
            idx = int(np.argmax(z))
            z[idx] = max(0.0, z[idx] + diff)
    return np.clip(z, 0.0, 1.0)

def all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def overlap_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    p = abs(k)
    if p >= n:
        return grad
    m = n - p
    if k >= 0:
        grad[:m] += (1.0 - h[p:])
        grad[p:] -= h[:m]
    else:
        grad[p:] += (1.0 - h[:m])
        grad[:m] -= h[p:]
    return grad

def active_set_grad(h, top_k=5):
    n = len(h)
    cor = all_overlaps(h)
    top_idx = np.argsort(-cor)[:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        grad += overlap_gradient(h, k)
    return cor, grad / top_k

def active_set_descent(h, deadline, rng, top_k=5, alpha_init=0.08):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    alpha = alpha_init

        for s in (alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.04):

            c_new = score_exact(cand)
            if c_new < cur_c - 1e-13:
                cur = cand
                cur_c = c_new

                    best_c = cur_c

        else:

def slsqp_objective(z, h_ref, lam):
    n = len(h_ref)
    h = z[:n]
    t = z[n]
    return t + lam * np.sum((h - h_ref) ** 2)

def slsqp_grad(z, h_ref, lam):
    n = len(h_ref)
    h = z[:n]
    g = np.zeros(n + 1)
    g[n] = 1.0
    if lam > 0:
        g[:n] = 2.0 * lam * (h - h_ref)
    return g

def slsqp_constraints(z):
    n = len(z) - 1
    h = z[:n]
    t = z[n]
    return all_overlaps(h) - t

def slsqp_jacobian(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    jac = np.zeros((2 * n - 1, n + 1))
    center = n - 1
    for idx in range(2 * n - 1):
        k = idx - center
        p = abs(k)
        m = n - p
        if k >= 0:
            jac[idx, :m] += (1.0 - h[p:]) * dx
            jac[idx, p:] -= h[:m] * dx
        else:
            jac[idx, p:] += (1.0 - h[:m]) * dx
            jac[idx, :m] -= h[p:] * dx
        jac[idx, n] = -1.0
    return jac

def run_slsqp(h_start, deadline, rng, lam=0.0, maxiter=150):
    n = len(h_start)
    t0 = score_exact(h_start)
    z0 = np.concatenate([h_start, [t0]])
    h_ref = h_start
    bounds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_constraints, 'jac': slsqp_jacobian},
        {'type': 'eq', 'fun': lambda z, n=n: np.sum(z[:n]) - n / 2.0,
         'jac': lambda z, n=n: np.concatenate([np.ones(n), [0.0]])}
    ]
    try:
        res = minimize(slsqp_objective, z0, args=(h_ref, lam), jac=slsqp_grad,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n], n)
            c_new = score_exact(h_new)
            if c_new < t0 - 1e-12:
                return h_new, c_new
        # Even if not fully converged, check
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n], n)
            c_new = score_exact(h_new)
            if c_new < t0 - 1e-12:
                return h_new, c_new
    except Exception:
        pass
    return h_start, t0

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i, orig_j = h[i], h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a, best_s = 0.0, cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s, best_a = s, a
    if best_s < cur_c - 1e-13 and abs(best_a) > 1e-10:
        span = (hi - lo) / n_pts
        lo2, hi2 = max(lo, best_a - 2 * span), min(hi, best_a + 2 * span)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1, a2 = lo2 + (1 - gr) * (hi2 - lo2), lo2 + gr * (hi2 - lo2)
        for _ in range(20):
            h[i] = orig_i + a1; h[j] = orig_j - a1; s1 = score_exact(h)
            h[i] = orig_i + a2; h[j] = orig_j - a2; s2 = score_exact(h)
            if s1 < s2:
                if s1 < best_s: best_s, best_a = s1, a1
                hi2, a2, a1 = a2, a1, lo2 + (1 - gr) * (hi2 - lo2)
            else:
                if s2 < best_s: best_s, best_a = s2, a2
                lo2, a1, a2 = a1, a2, lo2 + gr * (hi2 - lo2)
        best_a = (lo2 + hi2) * 0.5
        h[i] = orig_i + best_a; h[j] = orig_j - best_a
        best_s = score_exact(h)
        if best_s < cur_c - 1e-13:
            return float(best_s)
    h[i] = orig_i; h[j] = orig_j
    return None

def structured_perturb(h, rng, mode=0):
    n = len(h)
    if mode == 0:
        return make_feasible(h + rng.normal(0, 0.02, n), n)
    elif mode == 1:
        # Sinusoidal perturbation
        f = rng.integers(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        sines = amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
        return make_feasible(h + sines, n)
    elif mode == 2:
        # Block flip
        blen = rng.integers(3, max(4, n // 4))
        start = rng.integers(0, n - blen)
        flip = h.copy()
        flip[start:start + blen] = 1.0 - flip[start:start + blen]
        return make_feasible(flip, n)
    elif mode == 3:
        # Reverse a random block
        blen = rng.integers(4, max(5, n // 3))
        start = rng.integers(0, n - blen)
        rev = h.copy()
        rev[start:start + blen] = h[start + blen - 1:start - 1:-1]
        return make_feasible(rev, n)
    else:
        # Binary quantize then smooth
        binary = (h > 0.5).astype(float)
        smooth = make_feasible(binary + rng.normal(0, 0.15, n), n)
        return smooth

def resample_h(h_old, n_new):

        c_r = score_exact(h_r)
        sub_dl = min(p4_end - 10, time.time() + 30)
        if time.time() < sub_dl:
            h_r, c_r = active_set_descent(h_r, sub_dl, rng, top_k=4, alpha_init=0.03)
        maybe_update(h_r, c_r)

    # Final: one more SLSQP polish on best
    if time.time() < deadline - 10:
        h_f, c_f = run_slsqp(best_h, deadline, rng, lam=0.0, maxiter=60)
        maybe_update(h_f, c_f)

    # Final active-set polish
    if time.time() < deadline - 5:
        h_f2, c_f2 = active_set_descent(best_h, deadline - 2, rng, top_k=3, alpha_init=0.01)
        maybe_update(h_f2, c_f2)

    # Ensure feasibility and final score
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
