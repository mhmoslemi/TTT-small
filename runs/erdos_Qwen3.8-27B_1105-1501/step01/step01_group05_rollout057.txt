def interpolate_to_n(parent, target_n, shift=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(target_n, 0.5), target_n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(target_n) + 0.5 + shift) % target_n) * 2.0 / target_n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, target_n)

def slsqp_polish(h, time_limit=15.0):

        return x[n]

        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        hx = x[:n]
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        return x[n] - dx * corr
    
    def ineq_jac(x):
        hx = x[:n]
        G = compute_grad_matrix(hx)
        J = np.zeros((2*n-1, n+1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    h0 = project_to_target(h, n)
    s0 = official_score(h0)
    theta0 = s0 + 1e-4
    
    x0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,

            s_out = official_score(h_out)
            if s_out < s0 - 1e-10:

    except Exception:
        pass
    return h0, s0

def gen_random(n, rng):
    h = rng.uniform(0.05, 0.95, n)
    return project_to_target(h, n)

def gen_binary(n, rng):
    target = n // 2
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    if target * 2 < n:
        frac = n / 2.0 - target
        if frac > 0:
            h[idx[target]] = frac
    return project_to_target(h, n)

def gen_sine(n, rng):
    t = np.linspace(0, 2, n, endpoint=False)
    freq1 = rng.uniform(1.0, 5.0)
    freq2 = rng.uniform(2.0, 8.0)
    ph1 = rng.uniform(0, 2 * np.pi)
    ph2 = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.3 * np.sin(np.pi * freq1 * t + ph1) + 0.15 * np.sin(np.pi * freq2 * t + ph2)
    h += rng.normal(0, 0.02, n)
    return project_to_target(h, n)

def gen_block(n, rng):
    n_blocks = int(rng.integers(3, 10))
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = (b + 1) * block_size if b < n_blocks - 1 else n
        h[start:end] = rng.uniform(0, 1)
    return project_to_target(h, n)

def lp_local_search(h, max_iters=300, time_limit=35.0, seed=42):
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    stall_count = 0

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        if score < 0.378:
            break
        improved = False

        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        corr_s = corr * dx

        d = lp_minimax_direction(h, corr_s)
        if d is not None:
            h_new, s_new = line_search_lp(h, d, score)
            if s_new < score - 1e-13:
                h = h_new
                score = s_new
                improved = True
                stall_count = 0

        if not improved:
            h_new, s_new = coord_pair_step(h, score, n_pairs=40, rng=rng)
            if s_new < score - 1e-13:
                h = h_new
                score = s_new
                improved = True
                stall_count = 0

        if not improved:
            for _ in range(8):
                scale = 0.03 + 0.05 * min(1.0, stall_count * 0.3)
                pert = rng.normal(0, scale, n)
                pert -= np.mean(pert)
                h_try = project_to_target(h + pert, n)
                s_try = official_score(h_try)
                if s_try < score - 1e-14:
                    h = h_try
                    score = s_try
                    improved = True
                    stall_count = 0
                    break
            if not improved:
                corr_s2 = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
                G2 = compute_grad_matrix(h)
                active = np.argsort(-corr_s2)[:15]
                w = np.ones(len(active)) / len(active)
                g_weights = G2[active, :].T @ w
                g_weights -= np.mean(g_weights)
                for scale in [0.02, 0.05, 0.1, 0.2]:
                    h_try = project_to_target(h - scale * g_weights / (np.max(np.abs(g_weights)) + 1e-15), n)
                    s_try = official_score(h_try)
                    if s_try < score - 1e-14:
                        h = h_try
                        score = s_try
                        improved = True
                        stall_count = 0
                        break

        if not improved:
            stall_count += 1
            if stall_count > 15:
                break

    return h, official_score(h)

def slsqp_polish(h, time_limit=15.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:n]
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        return x[n] - dx * corr

    def ineq_jac(x):
        hx = x[:n]
        G = compute_grad_matrix(hx)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    h0 = project_to_target(h, n)
    s0 = official_score(h0)
    theta0 = s0 + 1e-4
    x0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 120, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_out = project_to_target(res.x[:n], n)
            s_out = official_score(h_out)
            if s_out < s0 - 1e-10:
                return h_out, s_out
    except Exception:
        pass
    return h0, s0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + max(30, budget_s) - 8.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    candidates = []

    if parent is not None:
        n_p = len(parent)
        hp = project_to_target(parent, n_p)
        candidates.append((official_score(hp), hp, n_p))
        for cv in [hp[::-1], 1.0 - hp, 1.0 - hp[::-1]]:
            cvp = project_to_target(cv, n_p)
            candidates.append((official_score(cvp), cvp, n_p))
        h_sym = project_to_target((hp + hp[::-1]) * 0.5, n_p)
        candidates.append((official_score(h_sym), h_sym, n_p))
        for dn in [-5, -3, -1, 0, 1, 3, 5, 8, 10, 15]:
            n_try = n_p + dn
            if 30 <= n_try <= 130:
                for sh in [0, 1, 2, 3]:
                    h_c = interpolate_to_n(parent, n_try, sh)
                    candidates.append((official_score(h_c), h_c, n_try))

    n_list = sorted(set([56, 60, 64, 65, 66, 70, 72, 76, 80, 84, 90, 96, 100]))
    if parent is not None:
        n_p = len(parent)
        n_list = sorted(set(n_list + [max(30, n_p - 4), max(30, n_p - 2), n_p, n_p + 2, n_p + 4, n_p + 6]))

    for n_try in n_list:
        for _ in range(2):
            candidates.append((official_score(gen_random(n_try, rng)), gen_random(n_try, rng), n_try))
        for _ in range(2):
            candidates.append((official_score(gen_binary(n_try, rng)), gen_binary(n_try, rng), n_try))
        for _ in range(2):
            candidates.append((official_score(gen_sine(n_try, rng)), gen_sine(n_try, rng), n_try))
        for _ in range(1):
            candidates.append((official_score(gen_block(n_try, rng)), gen_block(n_try, rng), n_try))

    candidates = [(s, h, n) for s, h, n in candidates if np.isfinite(s) and s > 0]
    candidates.sort(key=lambda x: (x[0], x[2]))

    best_score = candidates[0][0]
    best_h = candidates[0][1].copy()
    best_n = candidates[0][2]

    top_k = min(14, len(candidates))
    results = []
    for idx in range(top_k):
        if time.perf_counter() > deadline - 80:
            break
        s_c, h_c, n_c = candidates[idx]
        remaining = deadline - time.perf_counter()
        t_limit = min(30.0, max(5.0, remaining * 0.45 / max(1, top_k - idx)))
        h_loc, s_loc = lp_local_search(h_c, max_iters=200, time_limit=t_limit, seed=seed + idx * 17)
        results.append((s_loc, h_loc, n_c))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c

    results.sort(key=lambda x: (x[0], x[2]))

    to_polish = results[:min(5, len(results))]
    best_in_list = any(np.allclose(r[1], best_h, atol=1e-10) for r in to_polish)
    if not best_in_list:
        to_polish.append((best_score, best_h.copy(), best_n))

    for idx, (s_c, h_c, n_c) in enumerate(to_polish):
        if time.perf_counter() > deadline - 40:
            break
        h_ref, s_ref = slsqp_polish(h_c, time_limit=12.0)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
            best_n = n_c
        if time.perf_counter() < deadline - 20:
            h_2, s_2 = lp_local_search(h_ref, max_iters=80, time_limit=8.0, seed=seed + 600 + idx)
            if s_2 < best_score:
                best_score = s_2
                best_h = h_2.copy()
                best_n = n_c

    if time.perf_counter() < deadline - 25:
        for n_try in [best_n, best_n + 2, best_n - 2, best_n + 5]:
            if n_try < 30 or n_try > 130:
                continue
            if time.perf_counter() > deadline - 20:
                break
            h_interp = interpolate_to_n(best_h, n_try)
            h_loc, s_loc = lp_local_search(h_interp, max_iters=100, time_limit=10.0, seed=seed + 800 + n_try)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()
                best_n = n_try
            if time.perf_counter() < deadline - 10:
                h_ref, s_ref = slsqp_polish(h_loc, time_limit=8.0)
                if s_ref < best_score:
                    best_score = s_ref
                    best_h = h_ref.copy()
                    best_n = n_try

    if time.perf_counter() < deadline - 15:
        h_fin, s_fin = lp_local_search(best_h, max_iters=300, time_limit=12.0, seed=seed + 999)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()
        if time.perf_counter() < deadline - 8:
            h_fin2, s_fin2 = slsqp_polish(best_h, time_limit=10.0)
            if s_fin2 < best_score:
                best_score = s_fin2
                best_h = h_fin2.copy()

    best_h = project_to_target(best_h, best_n)
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(best_n))
