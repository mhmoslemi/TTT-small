    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = -1.0 - np.max(z) + 1e-12
    hi = 1.0 - np.min(z) - 1e-12
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(z + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo+hi)/2, 0, 1)
    # Fix residual
    res = target - np.sum(h)

        if res > 0:

        else:

Let me think about what's different from the previous program:

1. **Subgradient descent with momentum** (fast, many iterations)
2. **Adaptive beta scheduling** (start low for exploration, increase for exploitation)
3. **Restoration steps** (periodically project back and fix)
4. **Final SLSQP polish** (for the last bits of improvement)

The key advantage of subgradient + momentum over SLSQP:
- Each iteration is O(n²) ≈ 3000 flops for n=50
- SLSQP each iteration involves computing the objective AND gradient AND solving a QP subproblem
- So subgradient can do 10-100x more iterations in the same time
- With momentum, convergence can be much faster

Let me also consider: the problem is non-convex (bilinear), so we might get stuck in local optima. To counter this:
- Multi-start with different initializations
- Simulated annealing-like schedule (high beta early, low beta... wait, low beta = smoother = easier to optimize but less accurate; high beta = sharper = more accurate but harder)
- Actually for our purpose: high beta means the LSE is closer to the max, which is what we want to minimize. But high beta makes the landscape more rugged. So: start with low beta (smooth, easy to optimize), gradually increase to high beta (sharp, accurate).

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def lse_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    return max_c + log_sum_exp / beta

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    # For each output index i (0 to 2n-2), shift s = i - (n-1)
    # dS_i/dh[j] = (1-h[j-s])*I(valid1) + (-h[j+s])*I(valid2)
    for i in range(2 * n - 1):
        wi = w[i] * dx
        if wi < 1e-18:
            continue
        s = i - (n - 1)
        # valid1: max(s,0) <= j <= min(n-1+s, n-1)  [for s>=0: s<=j<=n-1; for s<0: 0<=j<=n-1+s]
        # valid2: max(0,-s) <= j <= min(n-1, n-1-s)  [for s>=0: 0<=j<=n-1-s; for s<0: -s<=j<=n-1]
        if s >= 0:
            j_lo1, j_hi1 = s, n  # [s, n-1]
            j_lo2, j_hi2 = 0, n - s  # [0, n-1-s]
        else:
            j_lo1, j_hi1 = 0, n + s  # [0, n-1+s]
            j_lo2, j_hi2 = -s, n  # [-s, n-1]
        for j in range(j_lo1, j_hi1):
            grad[j] += wi * (1.0 - h[j - s])
        for j in range(j_lo2, j_hi2):
            grad[j] += wi * (-h[j + s])
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)


def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)


def lse_value_and_weights(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = exp_vals / total
    return max_c, w


def lse_gradient(h, w):
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    A = np.zeros(n)
    B = np.zeros(n)
    for m in range(n):
        j_lo = max(0, m - nm1)
        j_hi = min(n - 1, m + nm1)
        idx_arr = np.arange(nm1 + j_lo - m, nm1 + j_hi - m + 1)
        A[j_lo:j_hi + 1] += (1.0 - h[m]) * w[idx_arr]
    for m in range(n):
        j_lo = max(0, nm1 - m)
        j_hi = min(n - 1, 2 * nm1 - m)
        idx_arr = np.arange(nm1 + j_lo - m, nm1 + j_hi - m + 1)
        B[j_lo:j_hi + 1] += h[m] * w[idx_arr]
    return dx * (A - B)


def lse_obj_grad(h, beta):
    n = len(h)
    max_c, w = lse_value_and_weights(h, beta)
    val = max_c + np.log(np.sum(np.exp(np.clip(beta * (np.correlate(h, 1.0 - h, mode='full') - max_c), -500, 0)))) / beta
    grad = lse_gradient(h, w)
    return val, grad


def subgrad_run(h0, n, max_iter, beta_start, beta_end, lr0, momentum_init,
                t_start, deadline):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = official_score(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, max_iter - 1))
        _, grad = lse_obj_grad(h, beta)
        lr = lr0 / (1.0 + 0.01 * it)
        v = momentum_init * v - lr * grad
        h_new = make_feasible(h + v)
        score = official_score(h_new)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)

    def obj_fun(x):
        max_c, w = lse_value_and_weights(x, beta)
        shifted = np.clip(beta * (np.correlate(x, 1.0 - x, mode='full') - max_c), -500, 0)
        return max_c + np.log(np.sum(np.exp(shifted))) / beta

    def obj_jac(x):
        _, w = lse_value_and_weights(x, beta)
        return lse_gradient(x, w)

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_fun, h0, jac=obj_jac, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
        return make_feasible(res['x'])
    return h0

    def _mm_eq(z):
        return np.sum(z[:-1]) - (len(z) - 1) / 2.0

    def _mm_eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    res = minimize(_mm_fun, z0, jac=_mm_fun_jac, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac},
                                {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
        return make_feasible(res['x'][:-1])
    return h0


def chain_improve(h0, deadline, min_remain=15.0):
    n = len(h0)
    cur_h = h0.copy()
    cur_score = official_score(cur_h)

    # Subgradient with momentum (fast, many iterations)
    if time.time() < deadline - min_remain - 80:
        n_iter = 40000
        h_sg, s_sg = subgrad_run(cur_h, n, n_iter, 15.0, 800.0, 0.02, 0.9,
                                 time.time(), deadline - min_remain - 60)
        if s_sg < cur_score - 1e-12:
            cur_h = h_sg
            cur_score = s_sg

    # Further subgradient with smaller step from the new point
    if time.time() < deadline - min_remain - 40:
        h_sg2, s_sg2 = subgrad_run(cur_h, n, 20000, 800.0, 3000.0, 0.005, 0.85,
                                   time.time(), deadline - min_remain - 20)
        if s_sg2 < cur_score - 1e-12:
            cur_h = h_sg2
            cur_score = s_sg2

    # SLSQP smooth polish with increasing beta
    for beta in [200, 600, 1500, 4000]:
        if time.time() > deadline - min_remain - 10:
            break
        h_new = slsqp_smooth(cur_h, beta, maxiter=100)
        s_new = official_score(h_new)
        if s_new < cur_score - 1e-12:
            cur_h = h_new
            cur_score = s_new

    # Direct min-max SLSQP polish
    if time.time() < deadline - min_remain - 5:
        h_new = slsqp_direct(cur_h, maxiter=120)
        s_new = official_score(h_new)
        if s_new < cur_score - 1e-12:
            cur_h = h_new
            cur_score = s_new

    return cur_h, cur_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(40) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(parent)

    rng = np.random.default_rng(seed)

    # Determine sizes to try
    pn = len(parent)
    size_opts = sorted(set([
        max(26, pn - 8), max(26, pn - 4), max(26, pn - 2),
        pn, pn + 2, pn + 4, pn + 6, pn + 8,
    ]))
    size_opts = [n for n in size_opts if 24 <= n <= 76]

    # Phase 1: Optimize parent size with diverse starts
    h_parent = parent.copy()
    for sigma in [0.0, 0.05, 0.1, 0.18]:
        if time.time() > deadline - 60:
            break
        if sigma == 0.0:
            h0 = h_parent.copy()
        else:
            h0 = make_feasible(h_parent + sigma * rng.standard_normal(pn))
        # Binary push variant
        h_bin = make_feasible(h0 ** 1.5)
        for start in [h0, h_bin]:
            if time.time() > deadline - 45:
                break
            h_opt, s_opt = chain_improve(start, deadline)
            if s_opt < best_score - 1e-12:
                best_h = h_opt.copy()
                best_score = s_opt
                best_n = len(h_opt)
                h_parent = best_h.copy()

    # Phase 2: Try other sizes
    for n in size_opts:
        if n == pn:
            continue
        if time.time() > deadline - 80:
            break
        h_interp = make_feasible(interp_to_n(best_h, n))
        h_anti = make_feasible(1.0 - h_interp[::-1].copy())
        for start in [h_interp, h_anti]:
            if time.time() > deadline - 50:
                break
            h_opt, s_opt = chain_improve(start, deadline)
            if s_opt < best_score - 1e-12:
                best_h = h_opt.copy()
                best_score = s_opt
                best_n = n

    # Phase 3: Final refinement around best solution
    remaining = deadline - time.time()
    if remaining > 30:
        n_trials = min(15, int(remaining / 15))
        for tr in range(n_trials):
            if time.time() > deadline - 10:
                break
            h_pert = make_feasible(best_h + 0.015 * rng.standard_normal(best_n))
            h_opt, s_opt = chain_improve(h_pert, deadline, min_remain=8.0)
            if s_opt < best_score - 1e-12:
                best_h = h_opt.copy()
                best_score = s_opt

    # Final polish
    if time.time() < deadline - 3:
        h_final = slsqp_direct(best_h, maxiter=150)
        s_final = official_score(h_final)
        if s_final < best_score:
            best_h = h_final
            best_score = s_final

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    # Ensure we track the true best
    if final_score < best_score:
        best_score = final_score

    return (best_h, float(best_score), int(len(best_h)))
