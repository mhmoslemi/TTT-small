    s = r - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        j = np.arange(s, n)
        g[j] += dx * (1.0 - h[j - s])
        j2 = np.arange(0, n - s)
        g[j2] -= dx * h[j2 + s]
    else:

        j = np.arange(0, n - a)
        g[j] += dx * (1.0 - h[j + a])
        j2 = np.arange(a, n)
        g[j2] -= dx * h[j2 - a]

def all_grads(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))

def lse_weights(c, t):
    cm = np.max(c)
    sh = np.clip(t * (c - cm), -40, 40)
    e = np.exp(sh)

def smooth_gd(h0, n, max_iters=80, t_limit=25.0, seed=42):
    rng = np.random.default_rng(seed)
    h = project_fix(h0, n)
    best_h = h.copy()
    best_s = official_score(h)
    t0 = time.perf_counter()
    L = 2 * n - 1
    no_improve = 0
    t_anneal = [10.0, 30.0, 80.0, 200.0, 500.0]
    seg = max(1, max_iters // len(t_anneal))
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > t_limit:
            break
        ti = t_anneal[min(it // seg, len(t_anneal) - 1)]
        c = corr_scaled(h)
        w = lse_weights(c, ti)
        G = all_grads(h)
        g = G.T @ w
        g -= np.mean(g)
        
        # Add noise early
        if it < max_iters * 0.2:
            g += 0.03 * rng.standard_normal(n)
            g -= np.mean(g)
        
        d = -g
        nm = np.max(np.abs(d))
        if nm < 1e-14:
            break
        d /= nm
        
        improved = False
        for sign in (1, -1):
            for a in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4):
                h_t = project_fix(h + sign * a * d, n)
                s = official_score(h_t)
                if s < best_s - 1e-13:
                    h = h_t
                    best_s = s
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve >= 8:
                break
            # Random kick
            kick = 0.02 * rng.standard_normal(n)
            kick -= np.mean(kick)
            h = project_fix(h + kick, n)
            s = official_score(h)
            if s < best_s:
                best_s = s
                best_h = h.copy()
                no_improve = 0
    
    return best_h, official_score(best_h)

def slsqp_polish(h0, n, t_limit=25.0):
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1
    h = project_fix(h0, n)
    init_score = official_score(h)
    
    def obj(x):
        return x[n]
    
    def obj_g(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_f(x):
        hx = x[:n]
        return x[n] - dx * np.correlate(hx, 1.0 - hx, mode="full")
    
    def ineq_j(x):
        hx = x[:n]
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        G = all_grads(hx)
        J[:, :n] = -G  # G already has dx factor
        return J
    
    def eq_f(x):
        return np.sum(x[:n]) - target
    
    def eq_j(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    theta0 = init_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_f, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_f, 'jac': ineq_j}
    ]
    
    try:
        res = optimize.minimize(
            obj, x0, jac=obj_g, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_out = project_fix(res.x[:n], n)
            s = official_score(h_out)
            if s < init_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, init_score

def interpolate_h(parent, n_target, shift=0):
    m = len(parent)
    if m == 0:
        return np.full(n_target, 0.5)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + shift) * 2.0 / n_target) % 2.0
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_fix(h, n_target)

def gen_random(n, rng):
    h = rng.uniform(0, 1, n)
    return project_fix(h, n)

def gen_binary(n, rng):

    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return project_fix(h, n)

def gen_sine(n, rng):
