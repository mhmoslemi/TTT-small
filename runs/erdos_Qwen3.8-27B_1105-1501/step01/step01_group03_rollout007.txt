    c_vals, G = compute_c_and_grad(h)

    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel().copy()

    dx = 2.0 / n
    nlags = 2 * n - 1
    c_vals = np.zeros(nlags)
    G = np.zeros((nlags, n))

        b = min(n, n - k)
