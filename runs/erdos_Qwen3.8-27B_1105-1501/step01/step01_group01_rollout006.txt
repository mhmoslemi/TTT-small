```python
def cvx_major_step(h, rho, solvers=None, reg_weight=0.01):
    """Take one majorization step."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    grad = compute_grad_matrix(h) * dx  # gradient of scaled objective
    
    # Actually, let me work with unscaled to avoid confusion
    # c_j = raw correlation * dx, so grad is also scaled
    # Let me just work with raw and multiply by dx at the end
    
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    grad_raw = compute_grad_matrix(h)
    
    # CVX problem
    delta = cp.Variable(n)
    t = cp.Variable()
    sq = cp.sum_squares(delta)
    
    constraints = [
        cp.sum(delta) == 0,
        delta >= -h,
        delta <= 1.0 - h,
        sq <= rho * rho,
    ]
    
    for j in range(2 * n - 1):
        constraints.append(raw_corr[j] + grad_raw[j, :] @ delta + sq <= t)
    
    objective = cp.Minimize(t + reg_weight * sq)
    prob = cp.Problem(objective, constraints)
    
    if solvers is None:
        solvers = ['CLARABEL', 'ECOS', 'SCS', 'MOSEK']
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if delta.value is not None and np.all(np.isfinite(delta.value)):
                    return np.asarray(delta.value).ravel()
            break  # only try first solver that gives a result
        except Exception:
            continue
    
    return None
```
