from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    h = np.asarray(v, dtype=float).ravel()
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        h = np.concatenate([h, np.full(n - len(h), 0.5)])
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection shift
    lo = -(np.max(h) + 0.5)
    hi = (1.0 - np.min(h)) + 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        q = np.clip(h + mid, 0.0, 1.0)
        if np.sum(q) < target:
            lo = mid
        else:
            hi = mid
    q = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact mass fix
    for _ in range(500):
        r = target - np.sum(q)
        if abs(r) < 1e-15:
            break
        if r > 0:
            idx = np.argmax(1.0 - q)
            d = min(r, 1.0 - q[idx])
            if d < 1e-16:
                break
            q[idx] += d
        else:
            idx = np.argmax(q)
            d = min(-r, q[idx])
            if d < 1e-16:
                break
            q[idx] -= d
    return q

def grad_matrix(h, dx):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):
        s = r - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            # (1-h[j-s]) for s <= j <= n-1
            for j in range(s, n):
                g[j] += (1.0 - h[j - s])
            # -h[j+s] for 0 <= j <= n-1-s
            for j in range(0, n - s):
                g[j] += (-h[j + s])
        else:
            a = -s
            # (1-h[j+a]) for 0 <= j <= n-1-a
            for j in range(0, n - a):

            # -h[j-a] for a <= j <= n-1
            for j in range(a, n):
                g[j] += (-h[j - a])

        step_caps = [0.5, 0.3, 0.2, 0.1]

    rng = np.random.default_rng(seed)
    stuck = 0
    
    for it in range(max_iter):
        if time.perf_counter() - start > time_limit:
            break
        
        step_cap = step_caps[min(it // 10, len(step_caps) - 1)]
        
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        max_val = np.max(corr)
        
        G = grad_matrix(h, 2.0 / n)
        L = 2 * n - 1
        nvar = 2 * n + 1
        
        # Objective: t + 0.01 * (sum p + sum q)
        c_obj = np.zeros(nvar)
        c_obj[2 * n] = 1.0
        c_obj[:n] = 0.01
        c_obj[n:2 * n] = 0.01

            A_ub_box[i, n + i] = -1.0
        b_ub_box = 1.0 - h
        
        # Lower box: -(p_i - q_i) <= h_i
        A_lb_box = np.zeros((n, nvar))
        for i in range(n):
            A_lb_box[i, i] = -1.0
            A_lb_box[i, n + i] = 1.0
        b_lb_box = h.copy()

        A_step = np.zeros((n, nvar))
        for i in range(n):

            A_step[i, n + i] = 1.0
        b_step = np.full(n, step_cap)
        
        A_ub = np.vstack([A_lag, A_ub_box, A_lb_box, A_step])
        b_ub = np.concatenate([b_lag, b_ub_box, b_lb_box, b_step])
        
        A_eq = np.zeros((1, nvar))
        A_eq[0, :n] = 1.0
        A_eq[0, n:2 * n] = -1.0
        b_eq = np.zeros(1)

                          bounds=bounds, method='highs')
        except Exception:
            break

            stuck += 1
            if stuck > 5:
                break
            continue
        
        delta = res.x[:n] - res.x[n:2 * n]
        
        # Check if step is meaningful
        if np.max(np.abs(delta)) < 1e-10:
            stuck += 1
            if stuck > 5:
                break
            continue
        
        stuck = 0
        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            h_new = project_to_target(h + alpha * delta, n)
            s_new = official_score(h_new)
            if s_new < score - 1e-12:
                h = h_new
                score = s_new
                improved = True
                break

            delta_noisy = delta + rng.normal(0, 0.05, n)
            for alpha in [0.5, 0.3, 0.2, 0.1]:

                s_new = official_score(h_new)
                if s_new < score - 1e-12:
                    h = h_new
                    score = s_new
                    improved = True
                    break

    target = n / 2.0
    lo = -(np.max(h) + 0.5)
    hi = (1.0 - np.min(h)) + 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        q = np.clip(h + mid, 0.0, 1.0)
        if np.sum(q) < target:
            lo = mid
        else:
            hi = mid
    q = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(500):
        r = target - np.sum(q)
        if abs(r) < 1e-15:
            break
        if r > 0:
            idx = np.argmax(1.0 - q)
            d = min(r, 1.0 - q[idx])
            if d < 1e-16:
                break
            q[idx] += d
        else:
            idx = np.argmax(q)
            d = min(-r, q[idx])
            if d < 1e-16:
                break
            q[idx] -= d
    return q

def grad_matrix(h, dx):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n - 1), n):
        r = s + (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] += (-h[s:])
        else:
            a = -s
            g[:n - a] += (1.0 - h[a:])
            g[a:] += (-h[:n - a])
        G[r] = dx * g
    return G

def lp_search(h, max_iter=200, time_limit=200.0, seed=42):
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    start = time.perf_counter()
    rng = np.random.default_rng(seed)
    stuck = 0
    prev_delta = np.zeros(n)
    
    for it in range(max_iter):
        if time.perf_counter() - start > time_limit:
            break
        
        progress = it / max_iter
        step_cap = 0.5 * (1.0 - progress) + 0.05
        reg = 0.005 + 0.02 * progress
        
        G = grad_matrix(h, 2.0 / n)
        L = 2 * n - 1
        nvar = 2 * n + 1
        
        c_obj = np.zeros(nvar)
        c_obj[2 * n] = 1.0
        c_obj[:n] = reg
        c_obj[n:2 * n] = reg
        
        A_lag = np.zeros((L, nvar))
        A_lag[:, :n] = G
        A_lag[:, n:2 * n] = -G
        A_lag[:, 2 * n] = -1.0
        b_lag = np.zeros(L)
        
        A_ub_box = np.zeros((n, nvar))
        for i in range(n):
            A_ub_box[i, i] = 1.0
            A_ub_box[i, n + i] = -1.0
        b_ub_box = 1.0 - h
        
        A_lb_box = np.zeros((n, nvar))
        for i in range(n):
            A_lb_box[i, i] = -1.0
            A_lb_box[i, n + i] = 1.0
        b_lb_box = h.copy()
        
        A_step = np.zeros((n, nvar))
        for i in range(n):
            A_step[i, i] = 1.0
            A_step[i, n + i] = 1.0
        b_step = np.full(n, step_cap)
        
        A_ub = np.vstack([A_lag, A_ub_box, A_lb_box, A_step])
        b_ub = np.concatenate([b_lag, b_ub_box, b_lb_box, b_step])
        
        A_eq = np.zeros((1, nvar))
        A_eq[0, :n] = 1.0
        A_eq[0, n:2 * n] = -1.0
        b_eq = np.zeros(1)
        
        bounds = [(0, None)] * (2 * n) + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0 or res.x is None:
            stuck += 1
            if stuck > 8:
                break
            continue
        
        delta = res.x[:n] - res.x[n:2 * n]
        
        if np.max(np.abs(delta)) < 1e-10:
            stuck += 1
            if stuck > 8:
                break
            continue
        
        stuck = 0
        # Momentum blend
        if it > 0 and np.max(np.abs(prev_delta)) > 1e-10:
            delta = 0.7 * delta + 0.3 * prev_delta
        
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            h_new = project_to_target(h + alpha * delta, n)
            s_new = official_score(h_new)
            if s_new < score - 1e-12:
                h = h_new
                score = s_new
                prev_delta = delta
                improved = True
                break
        
        if not improved:
            # Try noisy direction
            noise = rng.normal(0, 0.08, n)
            noise -= np.mean(noise)
            for alpha in [0.5, 0.3, 0.2, 0.1, 0.05]:
                h_new = project_to_target(h + alpha * (delta + noise), n)
                s_new = official_score(h_new)
                if s_new < score - 1e-12:
                    h = h_new
                    score = s_new
                    prev_delta = delta + noise
                    improved = True
                    break
            if not improved:
                # Try pure random direction
                for _ in range(5):
                    rd = rng.normal(0, 1, n)
                    rd -= np.mean(rd)
                    rd /= np.max(np.abs(rd))
                    for alpha in [0.3, 0.2, 0.1, 0.05]:
                        h_new = project_to_target(h + alpha * rd, n)
                        s_new = official_score(h_new)
                        if s_new < score - 1e-12:
                            h = h_new
                            score = s_new
                            improved = True
                            break
                    if improved:
                        break
                if not improved:
                    stuck += 1
                    if stuck > 10:
                        break
    
    return h, official_score(h)

def slsqp_polish(h, time_limit=60.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_to_target(h, n)
    initial_score = official_score(h)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        G = grad_matrix(h_x, dx)
        L = 2 * n - 1
        j = np.zeros((L, n + 1))
        j[:, n] = 1.0
        j[:, :n] = -G
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.5)]
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    t_start = time.perf_counter()
    max_iter = int(min(80, max(20, time_limit / 2.0)))
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    if time.perf_counter() - t_start < time_limit * 0.5:
        # Try again with different initial theta
        theta0b = initial_score + 1e-6
        x0b = np.concatenate([h, [theta0b]])
        try:
            result = optimize.minimize(
                objective, x0b, jac=objective_grad,
                method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 30, 'ftol': 1e-13}
            )
            if result.x is not None and np.all(np.isfinite(result.x[:n])):
                h_out = project_to_target(result.x[:n], n)
                s = official_score(h_out)
                if s < initial_score - 1e-10:
                    return h_out, s
        except Exception:
            pass
    
    return h, initial_score

def interpolate_to_n(h_parent, n_target, shift=0):
    m = len(h_parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) / m
    xt = ((np.arange(n_target) + 0.5 + shift) / n_target) % 1.0
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, h_parent, left=h_parent[0], right=h_parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)

def gen_binary_from_parent(h_parent, n_target):
    m = len(h_parent)
    if m == 0:
        return project_to_target(np.zeros(n_target), n_target)
    h_interp = interpolate_to_n(h_parent, n_target)
    thresh = np.partition(h_interp, n_target // 2)[n_target // 2]
    h_bin = (h_interp >= thresh).astype(float)
    # Ensure exact count
    n_ones = int(np.sum(h_bin))
    target = n_target // 2
    if n_ones > target:
        vals = np.where(h_bin == 1)[0]
        # Remove the ones with smallest parent values
        order = np.argsort(h_interp[vals])
        for i in range(n_ones - target):
            h_bin[vals[order[i]]] = 0.0
    elif n_ones < target:
        vals = np.where(h_bin == 0)[0]
        order = np.argsort(-h_interp[vals])
        for i in range(target - n_ones):
            h_bin[vals[order[i]]] = 1.0
    return project_to_target(h_bin, n_target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 8.0
    
    # Load parent
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 5 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None
    
    if parent is None:
        n_parent = 65
        parent = np.full(n_parent, 0.5)
    else:
        n_parent = len(parent)
    
    parent = project_to_target(parent, n_parent)
    
    # Generate candidates
    candidates = []
    
    # Parent and symmetries
    candidates.append((official_score(parent), parent.copy(), n_parent))
    candidates.append((official_score(project_to_target(parent[::-1], n_parent)), project_to_target(parent[::-1], n_parent), n_parent))
    h_comp = project_to_target(1.0 - parent, n_parent)
    candidates.append((official_score(h_comp), h_comp, n_parent))
    h_flip_comp = project_to_target(1.0 - parent[::-1], n_parent)
    candidates.append((official_score(h_flip_comp), h_flip_comp, n_parent))
    
    # Binary projection
    n_bin = n_parent if n_parent % 2 == 0 else n_parent + 1
    h_bin = project_to_target(gen_binary_from_parent(parent, n_bin), n_bin)
    candidates.append((official_score(h_bin), h_bin, n_bin))
    
    # Interpolated to nearby resolutions
    n_set = sorted(set([
        max(32, n_parent - 10), max(32, n_parent - 5), n_parent,
        n_parent + 5, n_parent + 10
    ]))
    for n_try in n_set:
        if n_try == n_parent:
            continue
        for sh in [0, 1, 2]:
            h_int = interpolate_to_n(parent, n_try, shift=sh)
            candidates.append((official_score(h_int), h_int, n_try))
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    
    best_score = candidates[0][0]
    best_h = candidates[0][1].copy()
    best_n = len(best_h)
    
    # Run LP search on top candidates
    top_k = min(6, len(candidates))
    for idx in range(top_k):
        s_c, h_c, n_c = candidates[idx]
        remaining = deadline - time.perf_counter()
        if remaining < 30:
            break
        
        # Allocate time: more for top candidates
        if idx == 0:
            t_limit = min(350.0, remaining * 0.35)
        elif idx <= 2:
            t_limit = min(180.0, remaining * 0.2)
        else:
            t_limit = min(100.0, remaining * 0.12)
        
        t_limit = max(t_limit, 30.0)
        h_loc, s_loc = lp_search(h_c, max_iter=300, time_limit=t_limit, seed=seed + idx * 13)
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c
    
    # SLSQP polish on best
    remaining = deadline - time.perf_counter()
    if remaining > 40:
        t_slsqp = min(100.0, remaining * 0.3)
        h_ref, s_ref = slsqp_polish(best_h, time_limit=t_slsqp)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
            best_n = len(best_h)
    
    # Second round: LP from the polished result and nearby interpolations
    remaining = deadline - time.perf_counter()
    if remaining > 60:
        # LP from current best
        t_lp2 = min(150.0, remaining * 0.4)
        h_loc2, s_loc2 = lp_search(best_h, max_iter=200, time_limit=t_lp2, seed=seed + 500)
        if s_loc2 < best_score:
            best_score = s_loc2
            best_h = h_loc2.copy()
        
        # Try interpolating best to other resolutions
        for n_try in [best_n - 5, best_n + 5, best_n + 10]:
            remaining = deadline - time.perf_counter()
            if remaining < 40 or n_try < 32:
                break
            h_int = interpolate_to_n(best_h, n_try)
            t_lp3 = min(100.0, remaining * 0.25)
            h_loc3, s_loc3 = lp_search(h_int, max_iter=150, time_limit=t_lp3, seed=seed + 700 + n_try)
            if s_loc3 < best_score:
                best_score = s_loc3
                best_h = h_loc3.copy()
                best_n = n_try
    
    # Final SLSQP polish
    remaining = deadline - time.perf_counter()
    if remaining > 30:
        t_final = min(60.0, remaining * 0.5)
        h_fin, s_fin = slsqp_polish(best_h, time_limit=t_final)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()
    
    # Final LP polish
    remaining = deadline - time.perf_counter()
    if remaining > 20:
        t_fin2 = min(40.0, remaining * 0.6)
        h_fin2, s_fin2 = lp_search(best_h, max_iter=100, time_limit=t_fin2, seed=seed + 999)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2.copy()
    
    # Ensure feasibility
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)
    best_score = final_score
    
    c5_bound = float(max(best_score, 0.0))
    return (best_h, c5_bound, int(len(best_h)))
