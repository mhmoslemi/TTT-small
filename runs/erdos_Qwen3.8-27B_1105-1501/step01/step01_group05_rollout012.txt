from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(1000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h

def pair_subproblem(h, p, q, n, n_grid=40):
    """Solve the 1D subproblem for pair (p,q) keeping h[p]+h[q] fixed."""
    S = h[p] + h[q]
    L_bound = max(0.0, S - 1.0)
    U_bound = min(1.0, S)
    if U_bound - L_bound < 1e-12:
        return h, official_score(h)
    
    dx = 2.0 / n
    best_x = h[p]
    best_score = official_score(h)
    
    # Grid search
    grid = np.linspace(L_bound, U_bound, n_grid)
    scores = np.zeros(n_grid)
    for idx, x in enumerate(grid):
        h_mod = h.copy()
        h_mod[p] = x
        h_mod[q] = S - x
        scores[idx] = np.max(np.correlate(h_mod, 1.0 - h_mod, mode='full')) * dx
    
    # Find best grid point and refine
    idx_best = np.argmin(scores)
    # Refine with golden section in the best interval
    a = grid[max(0, idx_best - 1)]
    b = grid[min(n_grid - 1, idx_best + 1)]
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    def f_eval(x):
        h_mod = h.copy()
        h_mod[p] = x
        h_mod[q] = S - x
        return np.max(np.correlate(h_mod, 1.0 - h_mod, mode='full')) * dx
    
    fc = f_eval(c)
    fd = f_eval(d)
    for _ in range(20):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_eval(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_eval(d)

    h_opt = h.copy()
    h_opt[p] = x_opt
    h_opt[q] = S - x_opt
    s_opt = f_eval(x_opt)
    
    if s_opt < best_score - 1e-14:
        return h_opt, s_opt
    return h, best_score

import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(5000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            if room[idx] < 1e-15:
                break
            h[idx] += min(residual, room[idx])
        else:
            idx = np.argmax(h)
            if h[idx] < 1e-15:
                break
            h[idx] -= min(-residual, h[idx])
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n):
    """Compute smoothed gradient of F(h) using softmax over active lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    T = max(0.005, 0.03 * scaled.max())
    w = np.exp((scaled - scaled.max()) / T)
    w /= w.sum()
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for idx, L in enumerate(lags):
        wi = w[idx]
        if wi < 1e-8:
            continue
        i_start = max(0, -L)
        i_end = min(n, n - L)
        for i in range(i_start, i_end):
            j = i + L
            grad[j] += wi * (1.0 - h[i])
            grad[i] += wi * (-h[j])
    grad *= dx
    return grad, scaled, w

def pair_subproblem(h, p, q, n, n_golden=25):
    """Solve 1D convex subproblem for pair (p,q). Returns (h_new, score_new, improved)."""
    S = h[p] + h[q]
    Lb = max(0.0, S - 1.0)
    Ub = min(1.0, S)
    span = Ub - Lb
    if span < 1e-10:
        return h.copy(), official_score(h), False
    
    orig = h.copy()
    dx = 2.0 / n
    
    def f_eval(x):
        h[p] = x
        h[q] = S - x
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    
    # Initial: evaluate at current point
    f0 = f_eval(h[p])
    
    # Golden section search on [Lb, Ub]
    gr = (np.sqrt(5) - 1.0) / 2.0
    a, b = Lb, Ub
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_eval(c)
    fd = f_eval(d)
    
    for _ in range(n_golden):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_eval(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_eval(d)
    
    x_opt = (a + b) / 2.0
    f_opt = f_eval(x_opt)
    
    if f_opt < f0 - 1e-13:
        h[p] = x_opt
        h[q] = S - x_opt
        return h.copy(), f_opt, True
    
    # Restore
    h[:] = orig
    return orig, f0, False

def pair_descent_pass(h, n, max_pairs=200, n_golden=25, T=None, time_limit=None):
    """One pass of pair-coordinate descent. Returns (h_new, score_new)."""
    t0 = time.perf_counter()
    score = official_score(h)
    dx = 2.0 / n
    
    # Compute gradient for pair ranking
    grad, scaled, w = compute_grad(h, n)
    
    # Generate candidate pairs ranked by expected improvement
    # Expected improvement for pair (p,q) ≈ |grad[p] - grad[q]| * span(p,q)
    # where span = min(1, h[p]+h[q]) - max(0, h[p]+h[q]-1)
    
    n_pairs_try = min(max_pairs, n * (n-1) // 2)
    
    # Rank pairs
    pair_scores = []
    for p in range(n):
        for q in range(p + 1, n):
            S = h[p] + h[q]
            Lb = max(0.0, S - 1.0)
            Ub = min(1.0, S)
            span = Ub - Lb
            if span < 0.05:
                continue
            gs = abs(grad[p] - grad[q]) * span
            if gs > 1e-8:
                pair_scores.append((gs, p, q))
    
    # Sort by expected improvement
    pair_scores.sort(key=lambda x: -x[0])
    pairs_to_try = [(p, q) for _, p, q in pair_scores[:n_pairs_try]]
    
    # Also add some random pairs for diversity
    rng = np.random.default_rng(42)
    for _ in range(min(50, len(pairs_to_try) // 3)):
        p = rng.integers(0, n)
        q = rng.integers(0, n)
        if p != q:
            S = h[p] + h[q]
            span = min(1.0, S) - max(0.0, S - 1.0)
            if span > 0.05:
                pairs_to_try.append((p, q))
    
    improved = False
    for p, q in pairs_to_try:
        if time_limit is not None and time.perf_counter() - t0 > time_limit:
            break
        h, s, imp = pair_subproblem(h, p, q, n, n_golden=n_golden)
        if imp:
            score = s
            improved = True
    
    return h, score

def smooth_direction_move(h, n, T, rng, time_limit=None):

    t0 = time.perf_counter()

    v += rng.normal(0, noise_scale, n)
    v -= np.mean(v)
    v /= np.max(np.abs(v))
    
    best_h = h.copy()
    best_s = score

def slsqp_polish(h, n, time_limit=30.0, max_active=30):
    """SLSQP polish with active constraint selection."""
    t0 = time.perf_counter()
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    initial_score = official_score(h)
    
    # Select active lags
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    top_idx = np.argsort(scaled)[-max_active:]
    
    def corr_lag(h_x, L):
        i_start = max(0, -L)
        i_end = min(len(h_x), len(h_x) - L)
        i = np.arange(i_start, i_end)
        j = i + L
        return np.sum(h_x[j] * (1.0 - h_x[i]))
    
    def corr_lag_grad(h_x, L):
        i_start = max(0, -L)
        i_end = min(len(h_x), len(h_x) - L)
        i = np.arange(i_start, i_end)
        j = i + L
        g = np.zeros(len(h_x))
        g[j] += (1.0 - h_x[i])
        g[i] += (-h_x[j])
        return g
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    n_lags = 2 * n - 1
    active_lags = lags[top_idx]
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        vals = np.zeros(max_active)
        for i, L in enumerate(active_lags):
            vals[i] = theta - dx * corr_lag(h_x, L)
        return vals
    
    def ineq_jac(x):
        h_x = x[:n]
        j = np.zeros((max_active, n + 1))
        j[:, n] = 1.0
        for i, L in enumerate(active_lags):
            g = corr_lag_grad(h_x, L)
            j[i, :n] = -dx * g
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 0.001
    x0 = np.concatenate([h, [theta0]])
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        if time.perf_counter() - t0 > time_limit * 0.5:
            return h, initial_score
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if time.perf_counter() - t0 > time_limit:
            return h, initial_score
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

def gen_binary(n, rng, frac=None):
    target = n / 2.0
    m_full = int(target)
    frac_val = target - m_full
    if frac is not None:
        m_full = n * frac
        m_full = int(round(m_full))
        frac_val = 0
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac_val > 0:
        h[idx[m_full]] = frac_val
    return repair_vector(h, n)
