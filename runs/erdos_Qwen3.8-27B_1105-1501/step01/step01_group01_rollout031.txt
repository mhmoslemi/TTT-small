                step *= 0.8
    return best_h, best_score

def coordinate_descent(h0, n_sweeps, deadline):
    n = len(h0)
    h = h0.copy()
    best_score = official_score(h0)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try several candidate values for h[i]
            candidates = np.array([max(0, old_val - 0.1), max(0, old_val - 0.03),
                                   old_val, min(1, old_val + 0.03), min(1, old_val + 0.1),
                                   0.0, 1.0, 0.5])
            best_cand = old_val
            best_s = official_score(h)
            for t in candidates:
                if t == old_val or abs(t - old_val) < 1e-10:
                    continue
                h[i] = np.clip(t, 0, 1)
                h = make_feasible(h)
                s = official_score(h)
                if s < best_s - 1e-12:
                    best_s = s
                    best_cand = h[i]
            h[i] = best_cand
            h = make_feasible(h)
            if abs(best_cand - old_val) > 1e-6:
                improved = True
        if not improved:
            break
    s = official_score(h)
    if s < best_score:
        best_score = s
    return h, best_score

def slsqp_smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def slsqp_smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - x[lo1 - shift:hi1 - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-x[lo2 + shift:hi2 + shift])
    return grad

def slsqp_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_eq_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, beta, maxiter=300):
    n = len(h0)
    try:
        res = minimize(
            slsqp_smooth_obj, h0, args=(beta,),
            jac=slsqp_smooth_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def do_slsqp_direct(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    t0 = official_score(h0) + 1e-6
    if t0 > 0.99:
        t0 = 0.99

    def mm_fun(z):
        return z[-1]

    def mm_fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def mm_ineq(z):
        hh = z[:-1]
        t = z[-1]
        c = np.correlate(hh, 1.0 - hh, mode='full') * dx
        return t - c

    def mm_ineq_jac(z):
        hh = z[:-1]
        jj = np.zeros((2 * n - 1, n + 1))
        jj[:, -1] = 1.0
        for k in range(2 * n - 1):
            shift = k - (n - 1)
            lo1 = max(0, shift)
            hi1 = min(n, n + shift)
            if lo1 < hi1:
                jj[k, lo1:hi1] += -dx * (1.0 - hh[lo1 - shift:hi1 - shift])
            lo2 = max(0, -shift)
            hi2 = min(n, n - shift)
            if lo2 < hi2:
                jj[k, lo2:hi2] += -dx * (-hh[lo2 + shift:hi2 + shift])
        return jj

    def mm_eq(z):
        return np.sum(z[:-1]) - len(z[:-1]) / 2.0

    def mm_eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(
            mm_fun, z0, jac=mm_fun_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
                {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0

def binary_pattern(n, k_blocks):
    """Create a binary-like pattern with k_blocks blocks of 1s."""
    h = np.zeros(n)
    block_size = n // (2 * k_blocks)
    for i in range(k_blocks):
        start = i * 2 * block_size
        end = start + block_size
        h[start:end] = 1.0
    return make_feasible(h)

def smooth_pattern(n, k_periods):
    """Create a smooth periodic-like pattern."""
    x = np.linspace(0, 2 * k_periods * np.pi, n)
    h = 0.5 + 0.4 * np.sin(x)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(50) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    parent_n = len(parent)

    # Phase 1: Projected gradient at parent resolution
    beta_schedule = [5, 15, 40, 100, 300, 800, 2000, 5000]
    iters_schedule = [300, 500, 800, 1000, 1200, 1000, 800, 600]

    if time.time() < deadline - 60:
        h1, s1 = projected_gradient(parent, beta_schedule, iters_schedule,
                                     momentum=0.85, deadline=deadline - 30,
                                     score_interval=50)
        if s1 < best_score:
            best_h, best_score = h1, s1

    # SLSQP polish from gradient result
    if time.time() < deadline - 40:
        h2 = do_slsqp(best_h, beta=500, maxiter=400)
        s2 = official_score(h2)
        if s2 < best_score:
            best_h, best_score = h2, s2

    # Direct SLSQP polish
    if time.time() < deadline - 30:
        h3 = do_slsqp_direct(best_h, maxiter=300)
        s3 = official_score(h3)
        if s3 < best_score:
            best_h, best_score = h3, s3

    # Phase 2: Try different n values
    n_candidates = sorted(set([
        max(30, parent_n - 7),
        max(30, parent_n - 5),
        max(30, parent_n - 3),
        parent_n,
        parent_n + 2,
        parent_n + 4,
        parent_n + 6,
        parent_n + 8,
        parent_n + 10,
    ]))
    n_candidates = [n for n in n_candidates if 28 <= n <= 72]

    for n in n_candidates:
        if time.time() > deadline - 30:
            break
        if n == parent_n:
            continue

        h_interp = make_feasible(interp_to_n(best_h, n))

        # Short projected gradient
        bs = [10, 50, 200, 800]
        ins = [200, 400, 600, 500]
        h_opt, s_opt = projected_gradient(h_interp, bs, ins,
                                           momentum=0.85, deadline=deadline - 20,
                                           score_interval=50)
        if s_opt < best_score:
            best_h, best_score = h_opt, s_opt
            best_n = n

        # SLSQP polish
        if time.time() < deadline - 20:
            h_sp = do_slsqp(h_opt, beta=500, maxiter=250)
            s_sp = official_score(h_sp)
            if s_sp < best_score:
                best_h, best_score = h_sp, s_sp
                best_n = n

    # Phase 3: Coordinate descent on best
    if time.time() < deadline - 15:
        h_cd, s_cd = coordinate_descent(best_h, n_sweeps=5, deadline=deadline - 10)
        if s_cd < best_score:
            best_h, best_score = h_cd, s_cd

    # Phase 4: Subgradient method for diversification
    if time.time() < deadline - 30:
        for trial in range(3):
            if time.time() > deadline - 20:
                break
            sigma = 0.08 + 0.04 * trial
            h_pert = make_feasible(best_h + sigma * rng.standard_normal(len(best_h)))
            h_sg, s_sg = subgradient_method(h_pert, n_iter=2000,
                                            deadline=deadline - 15, step_init=0.015)
            h_sg = do_slsqp(h_sg, beta=400, maxiter=200)
            s_final = official_score(h_sg)
            if s_final < best_score:
                best_h, best_score = h_sg, s_final

    # Phase 5: Random restarts with smooth patterns at various n
    if time.time() < deadline - 40:
        for n in [parent_n, parent_n + 4, parent_n - 4]:
            if time.time() > deadline - 30:
                break
            if n < 28 or n > 72:
                continue
            for pat_func, kw in [(binary_pattern, {'k_blocks': 2}),
                                  (binary_pattern, {'k_blocks': 3}),
                                  (smooth_pattern, {'k_periods': 2}),
                                  (smooth_pattern, {'k_periods': 3})]:
                if time.time() > deadline - 25:
                    break
                try:
                    h_pat = pat_func(n, **kw)
                except Exception:
                    continue
                h_opt, s_opt = projected_gradient(h_pat, [20, 100, 500],
                                                   [300, 500, 400],
                                                   momentum=0.85,
                                                   deadline=deadline - 20,
                                                   score_interval=80)
                h_opt = do_slsqp(h_opt, beta=400, maxiter=200)
                s_f = official_score(h_opt)
                if s_f < best_score:
                    best_h, best_score = h_opt, s_f

    # Phase 6: Final intensive polish
    if time.time() < deadline - 10:
        # Multiple SLSQP passes with different betas
        for beta in [200, 1000, 5000]:
            if time.time() > deadline - 8:
                break
            h_p = do_slsqp(best_h, beta=beta, maxiter=300)
            s_p = official_score(h_p)
            if s_p < best_score:
                best_h, best_score = h_p, s_p

        # Direct method
        if time.time() < deadline - 8:
            h_d = do_slsqp_direct(best_h, maxiter=300)
            s_d = official_score(h_d)
            if s_d < best_score:
                best_h, best_score = h_d, s_cd if False else (h_d, s_d)

    # Final verification
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)

    return (best_h, final_score, len(best_h))
