def corr_value_and_grad(h, r, n):
    """Compute correlation at index r and its gradient.
    
    np.correlate(h, 1-h, mode='full')[r]
    = sum over valid i of h[i] * (1-h[i + r - (n-1)])
    """
    offset = r - (n - 1)

    # valid i: 0 <= i < n and 0 <= i+offset < n
    # i ranges from max(0, -offset) to min(n-1, n-1-offset)
    
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    
    if i_lo > i_hi:

    indices = np.arange(i_lo, i_hi + 1)

import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_target(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(160):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(x < 1.0)[0]
            for i in idx:
                add = min(1.0 - x[i], residual)
                x[i] += add
                residual -= add
                if residual < 1e-14:
                    break
        else:
            idx = np.where(x > 0.0)[0]
            for i in idx:
                rem = min(x[i], -residual)
                x[i] -= rem
                residual += rem
                if residual > -1e-14:
                    break
    return x

def corr_val_grad(h, r, n):
    offset = r - (n - 1)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return 0.0, np.zeros(n)
    idx = np.arange(i_lo, i_hi + 1)
    val = float(np.sum(h[idx] * (1.0 - h[idx + offset])))
    grad = np.zeros(n)
    if offset == 0:
        grad[idx] = 1.0 - 2.0 * h[idx]
    else:
        grad[idx] = (1.0 - h[idx + offset])
        grad[idx + offset] -= h[idx]
    return val, grad

def verify_gradient_formula(h, n):
    eps = 1e-6
    max_err = 0.0
    for r in range(2 * n - 1):
        val, grad = corr_val_grad(h, r, n)
        for i in range(0, n, 7):
            h_p = h.copy()
            h_p[i] += eps
            val_p, _ = corr_val_grad(h_p, r, n)
            num_g = (val_p - val) / eps
            max_err = max(max_err, abs(num_g - grad[i]))
    return max_err

def compute_grads_for_lags(h, n, lags):
    grads = np.zeros((len(lags), n))
    vals = np.zeros(len(lags))
    for i, r in enumerate(lags):
        vals[i], grads[i] = corr_val_grad(h, r, n)
    return vals, grads

def lp_direction(h, n, radius, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    idx = list(idx)
    n_lags = len(idx)
    _, grads_sel = compute_grads_for_lags(h, n, idx)
    vals_sel = np.array([corr[r] for r in idx])

    nv = n + 2
    c = np.zeros(nv)
    c[n] = 5.0 / n
    c[n + 1] = 1.0

    A_rows = []
    b_rows = []
    for i in range(n_lags):
        row = np.zeros(nv)
        row[:n] = grads_sel[i]
        row[n + 1] = -1.0
        A_rows.append(row)
        b_rows.append(-vals_sel[i])
    for j in range(n):
        row = np.zeros(nv)
        row[j] = 1.0
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
        row2 = np.zeros(nv)
        row2[j] = -1.0
        row2[n] = -1.0
        A_rows.append(row2)
        b_rows.append(0.0)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for j in range(n):
        lo_p = max(-radius, -h[j])
        hi_p = min(radius, 1.0 - h[j])
        bounds.append((lo_p, hi_p))
    bounds.append((0.0, radius))
    bounds.append((None, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            p = res.x[:n].copy()
            p -= np.mean(p)
            pn = np.max(np.abs(p))
            if pn > 1e-12:
                return p / pn
            return None
    except Exception:
        pass
    return None

def subgradient_direction(h, n, top_k, radius):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    active = np.where(corr >= cmax - 0.05 * cmax)[0]
    if len(active) == 0:
        active = np.argsort(corr)[-top_k:]
    active = active[:top_k]
    _, grads = compute_grads_for_lags(h, n, active)
    w = np.ones(len(active)) / len(active)
    total_grad = np.sum(grads * w[:, None], axis=0)
    p = -total_grad
    p -= np.mean(p)
    pn = np.max(np.abs(p))
    if pn < 1e-14:
        return None
    p = p / pn * radius
    return p

def line_search(h, p, n, a_max):

    a_upper = a_max
    pos = p > 1e-14

    if np.any(pos):
        a_upper = min(a_upper, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        a_upper = min(a_upper, float(np.min(h[neg] / (-p[neg]))))
    if a_upper <= 1e-12:
        return 0.0, exact_c5(h)

    current_c5 = exact_c5(h)
    best_alpha = 0.0
    best_c5 = current_c5
    alphas = np.linspace(0, a_upper, 30)

        if c5_try < best_c5:

            best_alpha = a
    if best_alpha > 1e-10 and best_alpha < a_upper * 0.95:
        a_lo = max(0.0, best_alpha * 0.5)
        a_hi = min(a_upper, best_alpha * 1.5)
        alphas2 = np.linspace(a_lo, a_hi, 20)

            h_try = project_to_target(h + a * p, n / 2.0)
            c5_try = exact_c5(h_try)
            if c5_try < best_c5:

                best_alpha = a
    return best_alpha, best_c5

def local_optimize(h, n, target_sum, max_iters, deadline, use_lp=True):
    best_h = h.copy()
    best_c5 = exact_c5(h)
    radius = 0.3
    top_k = 8
    stalls = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        p = None
        if use_lp:
            p = lp_direction(best_h, n, radius, top_k)
        if p is None:
            p = subgradient_direction(best_h, n, top_k, radius)
        if p is None:
            radius *= 0.5
            stalls += 1
            if stalls > 20:
                break
            continue
        alpha, c5_new = line_search(best_h, p, n, radius)
        if c5_new < best_c5 - 1e-11:
            best_h = project_to_target(best_h + alpha * p, target_sum)
            best_c5 = exact_c5(best_h)
            radius = min(radius * 1.1, 0.8)
            stalls = 0
            top_k = min(top_k + 1, 25)
        else:
            radius *= 0.6
            stalls += 1
            if stalls > 30:
                break
    return best_h, best_c5

def random_binary_like(n, rng):
    target = n / 2.0
    h = np.zeros(n)
    n_ones = int(target)
    h[:n_ones] = 1.0
    rem = target - n_ones
    if rem > 1e-10:
        h[n_ones] = rem
    h = h[rng.permutation(n)]
    return project_to_target(h, target)

def interpolate_to_size(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_target(h_new, n_new / 2.0)

    h2[h2 < 0.5] = 0.0
