import numpy as np
import time


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out


def compute_F(h):
    n = len(h)
    s_vals = np.arange(-(n - 1), n)
    idx_plus = np.arange(n)[:, None] + s_vals[None, :]
    mask_p = (idx_plus >= 0) & (idx_plus < n)
    Fp = np.where(mask_p, h[np.clip(idx_plus, 0, n - 1)], 0.0)
    idx_minus = np.arange(n)[:, None] - s_vals[None, :]
    mask_m = (idx_minus >= 0) & (idx_minus < n)
    Fm = np.where(mask_m, h[np.clip(idx_minus, 0, n - 1)], 0.0)
    return Fp + Fm


def binary_local_search(h, rng, t_end):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = compute_F(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    best_h = h.copy()
    best_c5 = np.max(corr) * dx
    stall = 0
    temp = 0.5
    cool = 0.9997

    while time.time() < t_end:
        if len(ones) == 0 or len(zeros) == 0:
            break

        F_ones = F[ones]
        F_zeros = F[zeros]
        diff = F_ones[:, None, :] - F_zeros[None, :, :]
        new_corr_mat = diff + corr[None, None, :]
        new_max = np.max(new_corr_mat, axis=2)

        old_max = np.max(corr)
        min_new = np.min(new_max)
        best_idx = np.argmin(new_max)
        bi = ones[best_idx // len(zeros)]
        bj = zeros[best_idx % len(zeros)]

        if old_max - min_new > 1e-10:
            corr = corr + F[bi] - F[bj]
            h[bi] = 0.0
            h[bj] = 1.0
            F = compute_F(h)
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            stall = 0
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stall += 1
            accept = False
            if stall > 3:
                r = rng.rand()
                idx2 = rng.randint(new_max.size)
                di = ones[idx2 // len(zeros)]
                dj = zeros[idx2 % len(zeros)]
                delta_c5 = (np.max(corr + F[di] - F[dj]) - old_max) * dx
                if delta_c5 < 0 or r < np.exp(-delta_c5 / max(temp, 1e-6)):
                    corr = corr + F[di] - F[dj]
                    h[di] = 0.0
                    h[dj] = 1.0
                    F = compute_F(h)
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    accept = True
            if accept:
                stall = 0
            else:
                temp *= cool
                if stall > 100:

                        i2 = ones[rng.randint(len(ones))]
                        j2 = zeros[rng.randint(len(zeros))]
                        corr = corr + F[i2] - F[j2]
                        h[i2] = 0.0
                        h[j2] = 1.0
                    F = compute_F(h)
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    stall = 0
                    temp = 0.3

    return best_h, best_c5


def make_initial_patterns(n, rng):
    patterns = []
    h1 = np.zeros(n)
    h1[::2] = 1.0
    patterns.append(h1)

    h2 = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        pos = int((i * (phi - 1.0) * n) % n)
        h2[pos] = 1.0
    if np.sum(h2) != n // 2:
        h2 = make_binary(h2 + rng.randn(n) * 0.2)
    patterns.append(h2)

    h3 = make_binary(rng.rand(n) + 0.3)
    patterns.append(h3)

    h4 = np.zeros(n)
    block = n // 7
    for i in range(n):
        if (i // block) % 2 == 0:
            h4[i] = 1.0
    if np.sum(h4) != n // 2:
        h4 = make_binary(h4 + rng.randn(n) * 0.3)
    patterns.append(h4)

    return patterns


def continuous_refine(h_best, n, target_sum, t_end, rng):
    hc = h_best.copy()
    best_c5 = compute_c5(hc)
    best_h = hc.copy()

    for iteration in range(300):
        if time.time() > t_end:
            break
        corr_c = np.correlate(hc, 1.0 - hc, mode="full")
        m = np.max(corr_c)
        grad = np.zeros(n)
        active = np.where(corr_c >= m - 1e-6)[0]
        for k in active:
            s = k - (n - 1)
            a = max(0, s)
            b = min(n - 1, n - 1 + s)
            if a <= b:
                grad[a:b + 1] += (1.0 - hc[a - s:b - s + 1])
            a2 = max(0, -s)
            b2 = min(n - 1, n - 1 - s)
            if a2 <= b2:
                grad[a2:b2 + 1] -= hc[a2 + s:b2 + s + 1]

        lr = 0.5 / (1.0 + iteration * 0.05)
        h_new = hc - lr * grad * (2.0 / n) * 0.05
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        hc = h_new

    return best_h, best_c5
