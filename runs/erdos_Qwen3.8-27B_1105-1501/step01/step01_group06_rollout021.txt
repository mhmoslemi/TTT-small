def c5_official(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_target(v, target):
    v = np.asarray(v, dtype=float).copy()
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-13:
        if residual > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        if np.any(mask):
            idxs = np.where(mask)[0]
            if residual > 0:
                room = 1.0 - x[idxs]
                total_room = float(np.sum(room))
                if total_room > 1e-15:
                    x[idxs] += residual * room / total_room
                    x[idxs] = np.minimum(x[idxs], 1.0)
            else:
                total = float(np.sum(x[idxs]))
                if total > 1e-15:
                    x[idxs] *= (total + residual) / total
                    x[idxs] = np.maximum(x[idxs], 0.0)
    return x

def line_search_fast(C, A, R, T):
    if T <= 1e-14:
        return 0.0
    ng = 21
    sv = np.linspace(0.0, T, ng + 1)
    best_s = 0.0
    best_val = float(np.max(C))
    for k in range(1, ng + 1):
        s = sv[k]
        val = float(np.max(C + s * A - s * s * R))
        if val < best_val:
            best_val = val
            best_s = s
    if best_s <= 1e-14 or best_s >= T - 1e-14:
        return best_s
    idx = int(round(best_s / T * ng))
    lo_s = max(0.0, sv[max(0, idx - 1)])
    hi_s = min(T, sv[min(ng, idx + 1)])
    gr = 0.6180339887

        s1 = lo_s + gr * (hi_s - lo_s)
