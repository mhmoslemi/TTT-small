Let me design a cleaner, more focused implementation:

1. **Core LP-based local solver**: At point h, compute all C_k values and their gradients. Solve an LP to minimize t + λ*||d||_1 subject to C_k(h) + ∇C_k(h)·d ≤ t, sum(d) = 0, |d_i| ≤ rad. Then do exact line search.

2. **Diverse initialization**: 
   - Parent (if available)
   - Area transfer to different lengths
   - Binarization (top-n/2 entries to 1, rest 0)
   - Random perturbations
   - Cyclic shifts of parent

3. **Iterative refinement**: Run LP solver from multiple starting points, keep best.

4. **Pool recombination**: Mix top candidates.

Let me think about what could improve over the previous approach:

The previous approach used soft-max (log-sum-exp) smoothing with SLSQP. The key weakness is that SLSQP with many constraints (2n-1 inequality + 1 equality + bounds) can get stuck. The LP approach is cleaner - it directly handles the minimax structure via the epigraph variable t.

Key improvements I'll make:
1. More aggressive LP radii and multiple LP passes per candidate
2. Better candidate diversity (cyclic shifts are free and preserve constraints)
3. Shorter vectors explored more aggressively
4. Binary seeds that break local optima

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(z):
    """Project z onto {0<=h<=1, sum(h)=n/2}"""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-9
    hi = 1.0 - np.min(z) + 1e-9
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix remaining sum error
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        # Distribute error
        if err > 0:
            order = np.argsort(h)  # from smallest to largest
            for i in order:
                if err < 1e-15:
                    break
                if h[i] < 1.0:
                    d = min(err, 1.0 - h[i])
                    h[i] += d
                    err -= d
        else:
            order = np.argsort(h)[::-1]  # from largest to smallest
            for i in order:
                if err > -1e-15:
                    break
                if h[i] > 0.0:
                    d = min(-err, h[i])
                    h[i] -= d
                    err += d
    return np.clip(h, 0.0, 1.0)

def compute_overlap_values(h):
    """Compute all C_k values for k = -(n-1), ..., n-1"""
    n = len(h)
    dx = 2.0 / n
    F = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start+k:end+k])) * dx
    return F

def compute_gradient_row(h, k, n):
    """Compute gradient of C_k with respect to h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start < end:
        grad[start:end] += (1.0 - h[start+k:end+k])
        p_start = start + k
        p_end = end + k
        grad[p_start:p_end] += (-h[start:end])
    return grad * dx

def compute_all_gradients(h):
    """Compute gradient matrix (2n-1) x n"""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        G[k + n - 1] = compute_gradient_row(h, k, n)
    return G

def lp_refine(h, rad=0.3, lam=0.1, max_iters=50, time_limit=20.0):
    """LP-based local refinement"""
    n = len(h)
    h = project_feasible(h.copy())
    current_score = official_score(h)
    t_start = time.time()
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if rad < 1e-6:
            break
        
        # Compute current values and gradients
        c_vals = compute_overlap_values(h)
        G = compute_all_gradients(h)
        
        # Build LP: minimize t + lam/n * sum(p_i)
        # Variables: d_0,...,d_{n-1}, p_0,...,p_{n-1}, t
        # Constraints:
        #   G @ d - t <= -c_vals  (for each lag)
        #   d_i - p_i <= 0
        #   -d_i - p_i <= 0
        #   sum(d_i) = 0
        # Bounds:
        #   delta_lo_i <= d_i <= delta_hi_i
        #   0 <= p_i <= 2*rad
        #   0 <= t <= 2.0
        
        m = 2 * n + 1  # total variables
        c_obj = np.zeros(m)
        c_obj[n:2*n] = lam / n  # penalty on p
        c_obj[-1] = 1.0  # minimize t
        
        # Inequality constraints: A_ub @ x <= b_ub
        # Row block 1: G @ d - t <= -c_vals  =>  [G, 0, -1] @ x <= -c_vals
        A1 = np.zeros((2*n-1, m))
        A1[:, :n] = G
        A1[:, -1] = -1.0
        b1 = -c_vals
        
        # Row block 2: d_i - p_i <= 0 for each i
        A2 = np.zeros((2*n, m))
        for i in range(n):
            A2[i, i] = 1.0
            A2[i, n+i] = -1.0
        b2 = np.zeros(2*n)
        
        # Row block 3: -d_i - p_i <= 0 for each i
        A3 = np.zeros((2*n, m))
        for i in range(n):
            A3[i, i] = -1.0
            A3[i, n+i] = -1.0
        b3 = np.zeros(2*n)
        
        A_ub = np.vstack([A1, A2, A3])
        b_ub = np.concatenate([b1, b2, b3])
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, m))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds
        bounds = []
        for i in range(n):
            lo_i = max(-rad, -h[i] - 1e-9)
            hi_i = min(rad, 1.0 - h[i] + 1e-9)
            bounds.append((lo_i, hi_i))
        for i in range(n):
            bounds.append((0.0, 2.0 * rad + 1e-9))
        bounds.append((0.0, 2.0))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success and res.x is not None:
                d = res.x[:n]
                # Remove mean to ensure sum ≈ 0 (safety)
                d = d - np.mean(d)
                
                # Line search
                for alpha in [1.0, 0.6, 0.35, 0.2]:
                    h_try = project_feasible(h + alpha * d)
                    sc_try = official_score(h_try)
                    if sc_try < current_score - 1e-14:
                        h = h_try
                        current_score = sc_try
                        rad = min(rad * 1.15, 0.5)
                        break
                else:
                    rad *= 0.5
            else:
                rad *= 0.5
        except Exception:
            rad *= 0.5
    
    return h, current_score

def area_transfer(h_src, n_tgt):
    """Transfer h from its length to n_tgt using area integration"""
    n_src = len(h_src)
    dxs = 2.0 / n_src
    dxt = 2.0 / n_tgt
    x_src = (np.arange(n_src) + 0.5) * dxs
    x_tgt = (np.arange(n_tgt) + 0.5) * dxt
    # Simple: use interpolation then project
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_tgt)

def binary_round(h, n):
    """Create a binary candidate from h"""
    h = project_feasible(h)
    n = len(h)
    target = n // 2
    # For even n: exactly n/2 ones
    # For odd n: (n-1)/2 ones and one 0.5
    if n % 2 == 0:

        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0

        h_bin[idx[:target_floor]] = 1.0
        h_bin[idx[target_floor]] = 0.5
        # Need sum = n/2, so need target_floor ones + one 0.5 = target_floor + 0.5
        # But n/2 for odd n is not integer... wait
        # sum = n/2 means for n=61, sum = 30.5

    return project_feasible(h_bin)

def cyclic_shift(h, shift):
    return project_feasible(np.roll(h, shift))

import numpy as np
from scipy.optimize import linprog
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        if err > 0:
            order = np.argsort(h)
            for i in order:
                if err < 1e-15:
                    break
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                if err > -1e-15:
                    break
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_all_C(h):
    n = len(h)
    dx = 2.0 / n
    C = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        s = max(0, -k)
        e = min(n, n - k)
        if s < e:
            C[k + n - 1] = np.sum(h[s:e] * (1.0 - h[s + k:e + k])) * dx
    return C


def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        s = max(0, -k)
        e = min(n, n - k)
        if s >= e:
            continue
        G[k + n - 1, s:e] += (1.0 - h[s + k:e + k])
        G[k + n - 1, s + k:e + k] += (-h[s:e])
    G *= dx
    return G


def lp_step(h, rad, lam, time_limit=10.0):
    n = len(h)
    C = compute_all_C(h)
    G = build_gradient_matrix(h)
    m = 2 * n + 1
    c_obj = np.zeros(m)
    c_obj[n:2 * n] = lam / n
    c_obj[-1] = 1.0

    n_rows = (2 * n - 1) + 2 * n + 2 * n
    A_ub = np.zeros((n_rows, m))
    b_ub = np.zeros(n_rows)

    A_ub[:2 * n - 1, :n] = G
    A_ub[:2 * n - 1, -1] = -1.0
    b_ub[:2 * n - 1] = -C

    r = 2 * n - 1
    for i in range(n):
        A_ub[r, i] = 1.0
        A_ub[r, n + i] = -1.0
        r += 1
    for i in range(n):
        A_ub[r, i] = -1.0
        A_ub[r, n + i] = -1.0
        r += 1

    A_eq = np.zeros((1, m))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = []
    for i in range(n):
        lo_i = max(-rad, -h[i] - 1e-9)
        hi_i = min(rad, 1.0 - h[i] + 1e-9)
        bounds.append((lo_i, hi_i))
    for _ in range(n):
        bounds.append((0.0, 2.0 * rad + 1e-9))
    bounds.append((0.0, 2.0))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n] - np.mean(res.x[:n])
            return d
    except Exception:
        pass
    return None


def lp_refine(h, rad_init=0.35, max_iters=60, time_limit=30.0):
    n = len(h)
    h = project_feasible(h.copy())
    cur_score = official_score(h)
    rad = rad_init
    t0 = time.time()

    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if rad < 1e-7:
            break

        d = lp_step(h, rad, 0.1)
        if d is None or np.max(np.abs(d)) < 1e-10:
            rad *= 0.4
            continue

        improved = False
        for alpha in [1.0, 0.6, 0.35, 0.18]:
            h_try = project_feasible(h + alpha * d)
            sc = official_score(h_try)
            if sc < cur_score - 1e-14:
                h = h_try
                cur_score = sc
                improved = True
                break
        if improved:
            rad = min(rad * 1.1, 0.5)
        else:
            rad *= 0.45

    return h, cur_score


def coord_descent(h, n_iter=200, time_limit=10.0):
    n = len(h)
    h = project_feasible(h.copy())
    cur_score = official_score(h)
    t0 = time.time()
    dx = 2.0 / n

    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        i = it % n
        best_delta = 0.0
        best_sc = cur_score
        for sign in [1.0, -1.0]:
            for step in [0.01, 0.03, 0.08, 0.15]:
                d_val = sign * step
                if h[i] + d_val < -1e-9 or h[i] + d_val > 1.0 + 1e-9:
                    continue
                # Find j to compensate: pick the coordinate with most room
                if d_val > 0:
                    j_candidates = np.where(h > step + 1e-9)[0]
                    j_candidates = j_candidates[j_candidates != i]
                else:
                    j_candidates = np.where(h < 1.0 - step - 1e-9)[0]
                    j_candidates = j_candidates[j_candidates != i]
                if len(j_candidates) == 0:
                    continue
                # Pick j with most "room"
                if d_val > 0:
                    j = j_candidates[np.argmax(h[j_candidates])]
                else:
                    j = j_candidates[np.argmin(h[j_candidates])]
                h_try = h.copy()
                h_try[i] += d_val
                h_try[j] -= d_val
                h_try = project_feasible(h_try)
                sc = official_score(h_try)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = d_val
                    best_j = j
        if best_delta != 0.0:
            h[i] += best_delta
            h[best_j] -= best_delta
            h = project_feasible(h)
            cur_score = best_score if 'best_score' in dir() else best_sc
            cur_score = best_sc

    return h, cur_score


def area_transfer(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_feasible(h_src.copy())
    dxs = 2.0 / n_src
    x_src = (np.arange(n_src) + 0.5) * dxs
    x_tgt = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_tgt)


def try_update(h_cand, best_state):
    h_cand = project_feasible(h_cand)
    if not np.all(np.isfinite(h_cand)):
        return
    sc = official_score(h_cand)
    if not np.isfinite(sc):
        return
    if sc < best_state['score'] - 1e-15:
        best_state['h'] = h_cand.copy()
        best_state['score'] = sc
    elif abs(sc - best_state['score']) < 5e-14 and len(h_cand) < len(best_state['h']):
        best_state['h'] = h_cand.copy()
        best_state['score'] = sc
