    """Compute the max overlap for a binary (or 0-1 fraction) array."""
    dx = 2.0 / n

def binary_local_search(h_bin, n, max_iters, rng, temperature=0.0):
    """Local search on binary patterns with exactly n/2 ones."""
    h = h_bin.copy().astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_score = binary_score(h, n)
    
    for it in range(max_iters):
        # Pick a random 1 and a random 0 to swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        
        # Try the swap
        h[i] = 0
        h[j] = 1
        new_score = binary_score(h, n)
        
        if new_score < best_score or (temperature > 0 and rng.random() < np.exp((best_score - new_score) / temperature)):
            best_score = new_score
            # Update ones and zeros
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
        else:
            # Revert
            h[i] = 1
            h[j] = 0
    
    return h, best_score

    n = len(h)
    dx = 2.0 / n

    grad = np.zeros(n)

        w = weights[idx]
        if w < 1e-15:
            continue
        k = idx - (n-1)  # shift
        # dc[k]/dh[m] = (1-h[m-k]) if valid + (-h[m+k]) if valid
        # For shift k: c[k] = sum_{n'} h[n'+k]*(1-h[n'])
        # dc/dh[m]: m appears as n'+k (first factor) or n' (second factor)

        #   contribution: (1-h[m-k])
        # As second factor: n' = m, valid if 0 <= m+k < n
        #   contribution: -h[m+k]

        # First term: (1-h[m-k]) for m in range where 0 <= m-k < n, i.e., k <= m < k+n
        #   But m is always in [0, n), so: max(0,k) <= m < min(n, k+n)
        #   Wait, m-k >= 0 means m >= k, and m-k < n means m < k+n
        #   So m ranges from max(0,k) to min(n-1, k+n-1)
        
        # Second term: -h[m+k] for m in range where 0 <= m+k < n, i.e., -k <= m < n-k
        #   m ranges from max(0,-k) to min(n-1, n-k-1)
        
        lo1 = max(0, k)
        hi1 = min(n, k + n)  # exclusive
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1-k:hi1-k])
        
        lo2 = max(0, -k)
        hi2 = min(n, n - k)  # exclusive
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2+k:hi2+k])
    
    grad *= dx
    return obj, grad

    """Hill climbing on binary patterns. Returns improved binary pattern."""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    score = official_score(h)

        best_improve = 0
        best_swap = None
        for i in ones:

                h[j] = 1

        h[j] = 1

        zeros.append(i)

```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    C = c * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    obj = cmax + (mk + np.log(ws)) / beta
    weights = es / ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2 + k:hi2 + k])
    grad *= dx
    return obj, grad

def momentum_proj_grad(h0, beta, max_iter=500, step_init=0.1, momentum=0.9):
    n = len(h0)
    h = h0.copy().astype(float)
    z = h.copy()  # momentum accumulator
    step = step_init
    best_h = h.copy()
    best_score = official_score(h0)
    prev_obj, _ = smooth_obj_grad(h, beta)
    
    for it in range(max_iter):
        obj, grad = smooth_obj_grad(h, beta)
        if not np.isfinite(obj):
            break
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Momentum update
        z = momentum * z - step * grad
        h_new = make_feasible(z)
        new_obj, _ = smooth_obj_grad(h_new, beta)
        
        if new_obj < prev_obj - 1e-14:
            h = h_new
            prev_obj = new_obj
            step = min(step * 1.01, step_init * 3)
            sc = official_score(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
        else:
            # Revert momentum
            z = h.copy()
            step *= 0.5
            if step < 1e-8:
                break
    
    sc = official_score(h)
    if sc < best_score:
        best_score = sc
        best_h = h.copy()
    
    return best_h, best_score

def slsqp_opt(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj_grad(x, beta)[0]
    
    def grad(x):
        return smooth_obj_grad(x, beta)[1]
    
    def eq_fun(x):
        return float(np.sum(x) - n / 2.0)
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0

def binary_hill_climb(h_bin, max_rounds=100):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    score = official_score(h)
    
    for r in range(max_rounds):
        best_improve = 1e-10
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_score = official_score(h)
                improve = score - new_score
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
        score -= best_improve
    
    return h, score

def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def thue_morse_binary(n, rng):
    """Generate a Thue-Morse based binary pattern with exactly n/2 ones."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    # Balance if needed
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h == 1)[0]
        to_flip = one_idx[:n_ones - target]
        h[to_flip] = 0
    elif n_ones < target:
        zero_idx = np.where(h == 0)[0]
        to_flip = zero_idx[:target - n_ones]
        h[to_flip] = 1
    return h
