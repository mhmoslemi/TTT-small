    corr = np.correlate(h, 1-h, mode='full') * dx
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    
    for k in range(2*n-1):

        j1 = p + shift
        j2 = p - shift
        term1 = h[p] * (1-h[j1]) if 0 <= j1 < n else 0.0
        term2 = h[j2] * (1-h[p]) if 0 <= j2 < n else 0.0

            B[k] = C_k + dx * h[j2]

        else:

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -z.max() - 2.0
    hi = 1.0 - z.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def find_optimal_hp(h, p, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    j1 = p + shifts
    j2 = p - shifts
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    
    A = np.zeros(2 * n - 1)
    B = corr.copy()
    
    idx1 = np.where(valid1)[0]
    if len(idx1) > 0:
        A[idx1] = dx * (1.0 - h[j1[idx1]])
        B[idx1] -= h[p] * (1.0 - h[j1[idx1]]) * dx
    
    idx2 = np.where(valid2)[0]
    if len(idx2) > 0:
        A[idx2] -= dx * h[j2[idx2]]
        B[idx2] -= h[j2[idx2]] * (1.0 - h[p]) * dx
        B[idx2] += h[j2[idx2]] * dx
    
    # Minimize max_k (A[k]*t + B[k]) over t in [0,1]
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = A * mid + B
        M = vals.max()
        active = vals >= M - 1e-14
        gmin = A[active].min()
        gmax = A[active].max()
        if gmin >= -1e-14:
            hi = mid
        elif gmax <= 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, max_sweeps=200, rng=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_score = official_score(h)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        improved = False
        for p in order:
            opt = find_optimal_hp(h, p, dx)
            if abs(opt - h[p]) > 1e-14:

                improved = True
        
        h = make_feasible(h)
        score = official_score(h)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
        
        if not improved and sweep > 3:
            break
    
    return best_h, best_score
