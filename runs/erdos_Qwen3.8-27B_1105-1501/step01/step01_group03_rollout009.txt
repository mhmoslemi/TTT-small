import numpy as np
from scipy.optimize import linprog, minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(z):
    """Project z to the feasible set: 0<=h<=1, sum(h)=n/2"""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    # Bisection for the shift
    lo = -1.5
    hi = 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        # Distribute error to indices where we have room
        if err > 0:
            idx = np.where(h < 1.0)[0]
            per = err / len(idx)
            for i in idx:
                d = min(per, 1.0 - h[i])
                h[i] += d
                err -= d
                if abs(err) < 1e-14:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            per = -err / len(idx)
            for i in idx:
                d = min(per, h[i])
                h[i] -= d
                err += d
                if abs(err) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def compute_F(h):
    """Compute all overlap values C_k * (1/dx) = F_k for all k"""
    n = len(h)
    # Use np.correlate to get all lags at once
    # np.correlate(h, 1-h, 'full') gives lags from -(n-1) to n-1
    return np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)

def compute_F_dx(h):
    """Compute all C_k = dx * F_k"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient_k(h, k, dx):
    """Compute gradient of C_k with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start >= end:
        return grad
    # Contribution as "first" index: h_j * (1 - h_{j+k}), j in [start, end)
    # d/dh_j = (1 - h_{j+k})
    grad[start:end] += (1.0 - h[start + k:end + k])
    # Contribution as "second" index: h_{j} * (1 - h_{j+k})
    # For index i = j+k, j = i-k, so i in [start+k, end+k)
    # d/dh_i = -h_{i-k}
    p_start = max(0, start + k)
    p_end = min(n, end + k)
    if p_start < p_end:
        grad[p_start:p_end] += (-h[p_start - k:p_end - k])
    return grad * dx

def compute_all_gradients(h, dx):
    """Compute gradient matrix: (2n-1) x n, where row k+n-1 is ∇C_{k}"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        G[k + n - 1] = compute_gradient_k(h, k, dx)
    return G

def lp_direction(h, active_lags, R):
    """
    Solve LP to find direction d that reduces the max overlap.
    Variables: d[0..n-1], t
    Minimize: t + mu * sum(|d_i|)  [use |d_i| via p_i variables]
    Constraints:
      C_k + G_k·d <= t  for k in active_lags
      sum(d) = 0
      -R <= d_i <= R
    """
    n = len(h)
    dx = 2.0 / n
    F = compute_F(h)  # raw (unscaled)
    C = F * dx  # scaled
    
    G = np.zeros((len(active_lags), n))
    for i, k in enumerate(active_lags):
        G[i] = compute_gradient_k(h, k, dx)
    
    # LP variables: d[0..n-1], p[0..n-1] (where p_i >= |d_i|), t
    # Total: 2n + 1
    # Minimize: t + (mu/n) * sum(p_i)
    mu = 0.1
    c_obj = np.zeros(2 * n + 1)
    c_obj[n:2*n] = mu / n  # penalty on p
    c_obj[2*n] = 1.0  # minimize t
    
    # Constraints (inequality form: A_ub @ x <= b_ub)
    # For each active lag j: G_j·d - t <= -C_j
    # => G_j·d + 0·p - 1·t <= -C_j
    A_ub_rows = []
    b_ub = []
    for j in range(len(active_lags)):
        row = np.zeros(2 * n + 1)
        row[0:n] = G[j]
        row[2*n] = -1.0
        A_ub_rows.append(row)
        b_ub.append(-C[j])
    
    # d_i - p_i <= 0 (i.e., d_i <= p_i)
    for i in range(n):
        row = np.zeros(2 * n + 1)
        row[i] = 1.0
        row[n + i] = -1.0
        A_ub_rows.append(row)
        b_ub.append(0.0)
    
    # -d_i - p_i <= 0 (i.e., -d_i <= p_i)
    for i in range(n):
        row = np.zeros(2 * n + 1)
        row[i] = -1.0
        row[n + i] = -1.0
        A_ub_rows.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -R <= d_i <= R, 0 <= p_i <= 2R, -inf <= t <= inf
    bounds = [(-R, R)] * n + [(0, 2 * R)] * n + [(-0.5, 2.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        d = res.x[0:n]
        # Center d to ensure sum = 0 (fix numerical issues)
        d = d - np.mean(d)
        return d, res.x[2*n]
    
    return None, None

def subgradient_step(h, R, alpha_seq):
    """
    One subgradient/LP step: find direction, line search.
    Returns improved h or None.
    """
    n = len(h)
    dx = 2.0 / n
    C = compute_F_dx(h)
    max_val = np.max(C)
    
    # Active set: lags within tolerance of max
    tol = max(1e-4, 0.02 * max_val)
    active_mask = C >= max_val - tol
    active_lags = np.where(active_mask)[0].tolist()
    
    if len(active_lags) == 0:
        active_lags = [int(np.argmax(C))]
    
    d, t_pred = lp_direction(h, active_lags, R)
    if d is None:
        return None
    
    # Line search
    best_h = None
    best_score = official_score(h)
    for alpha in alpha_seq:
        h_try = project_feasible(h + alpha * d)
        sc = official_score(h_try)
        if sc < best_score - 1e-12:
            best_h = h_try
            best_score = sc
    
    return best_h

def binary_round(h, rng):
    """Round h to a near-binary feasible vector"""
    n = len(h)
    target = n / 2.0
    # Sort by value descending
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    if target == int(target):
        # Even n: set top n/2 to 1
        h_bin[order[:int(target)]] = 1.0
    else:
        # Odd n: set top floor(n/2) to 1, one to 0.5
        k = int(target)
        h_bin[order[:k]] = 1.0
        h_bin[order[k]] = 0.5
    return h_bin

def coordinate_descent(h, n_iter=100, tol=1e-10):
    """
    Coordinate descent: for each coordinate i, find the best h_i that
    minimizes the max overlap, given all other coordinates fixed.
    """
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    score_cur = official_score(h_cur)
    
    for it in range(n_iter):
        improved = False
        for i in range(n):
            # Try to find best h_i
            # The overlap values as a function of h_i are piecewise linear
            # (since each C_k is linear in h_i when others are fixed)
            # So the max is a piecewise linear function of h_i, and
            # we want to minimize it over [0,1] while maintaining sum constraint
            # (but we handle sum constraint separately via projection)
            # Simple approach: try a few candidate values
            best_val = h_cur[i]
            best_sc = score_cur
            candidates = [0.0, 1.0, h_cur[i], 
                         h_cur[i] + 0.1, h_cur[i] - 0.1,
                         h_cur[i] + 0.05, h_cur[i] - 0.05]
            # Also try the "balanced" value
            for v in candidates:
                v = np.clip(v, 0.0, 1.0)
                if abs(v - h_cur[i]) < 1e-10:
                    continue
                trial = h_cur.copy()
                trial[i] = v
                # Need to maintain sum: adjust another coordinate
                # Actually, let's just evaluate and use projection later
                # For now, just evaluate the raw change
                F = np.correlate(trial, 1.0 - trial, mode='full')
                sc_trial = float(np.max(F) * dx)
                if sc_trial < best_sc - 1e-12:
                    best_sc = sc_trial
                    best_val = v
            if abs(best_val - h_cur[i]) > 1e-10:
                h_cur[i] = best_val
                improved = True
        
        if not improved:
            break
    
    # Project to feasible set
    h_final = project_feasible(h_cur)
    return h_final
