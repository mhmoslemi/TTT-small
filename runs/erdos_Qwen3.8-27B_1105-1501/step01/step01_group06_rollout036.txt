import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_c5(h):
    """Compute the exact C5 bound."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_target(v, target):
    """Project v to [0,1]^n with sum = target using bisection."""
    n = len(v)
    v = np.clip(v, -100.0, 100.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-10:
        # Simple fix: adjust largest elements
        if residual > 0:
            idx = np.argsort(-result)
            for i in idx:
                add = min(1.0 - result[i], residual)
                result[i] += add
                residual -= add
                if residual < 1e-12:
                    break
        else:
            idx = np.argsort(result)
            for i in idx:
                sub = min(result[i], -residual)
                result[i] -= sub
                residual += sub
                if residual > -1e-12:
                    break
    return result

def compute_scores_and_grads(h):
    """Compute all correlation scores and their gradients."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx  # length 2n-1
    
    # Compute gradients for each lag
    # grad[r, j] = d(scores[r])/d(h[j])
    # lag = r - (n-1)
    # d/dh[j] = (1 - h[j-lag]) * [0 <= j-lag <= n-1] - h[j+lag] * [0 <= j+lag <= n-1]
    # times dx
    
    grads = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        lag = r - (n - 1)
        # First term: (1 - h[j-lag]) where 0 <= j-lag <= n-1
        # j ranges from lag to n-1+lag, intersect [0, n-1]
        if lag >= 0:
            j_start1 = lag
            j_end1 = n  # exclusive
        else:
            j_start1 = 0
            j_end1 = n + lag  # = n - |lag|
        if j_end1 > j_start1:
            grads[r, j_start1:j_end1] += (1.0 - h[j_start1 - lag : j_end1 - lag]) * dx
        
        # Second term: -h[j+lag] where 0 <= j+lag <= n-1
        # j ranges from -lag to n-1-lag, intersect [0, n-1]
        if lag >= 0:
            j_start2 = 0
            j_end2 = n - lag
        else:
            j_start2 = -lag
            j_end2 = n
        if j_end2 > j_start2:
            grads[r, j_start2:j_end2] -= h[j_start2 + lag : j_end2 + lag] * dx
    
    return scores, grads

def lp_direction(h, scores, grads, radius, top_k):
    """Find LP descent direction."""
    n = len(h)
    M = float(np.max(scores))
    
    # Select active lags (top_k closest to max)
    sorted_idx = np.argsort(-scores)
    active = sorted_idx[:top_k]
    
    # LP: minimize tau + lambda * r
    # subject to: grads[r] @ p - tau <= -scores[r] for r in active
    #             sum(p) = 0
    #             -radius <= p_i <= radius
    #             -h_i <= p_i <= 1-h_i
    #             -r <= p_i <= r  (L-inf bound via r variable)
    
    n_active = len(active)
    # Variables: p (n), tau (1), r (1) => total n+2
    c = np.zeros(n + 2)
    c[n] = 1.0      # minimize tau
    c[n+1] = 5.0    # + lambda * r (regularization)
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active r: grads[r] @ p - tau <= -scores[r]
    # => [grads[r], -1, 0] @ [p, tau, r] <= -scores[r]
    A_ub = np.zeros((n_active + 2*n, n + 2))
    b_ub = np.zeros(n_active + 2*n)
    
    for i, ar in enumerate(active):
        A_ub[i, :n] = grads[ar]
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[ar]
    
    # L-inf constraints: p_i <= r and -p_i <= r
    base = n_active
    for i in range(n):
        A_ub[base + i, i] = 1.0
        A_ub[base + i, n+1] = -1.0
        b_ub[base + i] = 0.0
        
        A_ub[base + n + i, i] = -1.0
        A_ub[base + n + i, n+1] = -1.0
        b_ub[base + n + i] = 0.0
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i])
        hi_i = min(radius, 1.0 - h[i])
        if lo_i > hi_i:
            lo_i = hi_i = 0.0
        bounds.append((lo_i, hi_i))
    bounds.append((None, None))  # tau free
    bounds.append((0.0, radius))  # r >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        p = res.x[:n]
        tau = res.x[n]
        if tau < M - 1e-10 and np.max(np.abs(p)) > 1e-10:
            return p, tau
    return None, M

def interpolate_h(h_old, n_new):
    """Interpolate h from n_old points to n_new points."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_to_target(h_new, n_new / 2.0)
    return h_new

def subgradient_direction(h, scores, grads, top_k):
    """Compute subgradient direction (average of active gradients)."""
    n = len(h)
    sorted_idx = np.argsort(-scores)
    active = sorted_idx[:top_k]
    g = np.mean(grads[active], axis=0)
    p = -g
    p -= np.mean(p)  # Make sum zero
    p_norm = np.max(np.abs(p))
    if p_norm > 1e-15:
        p = p / p_norm * 0.1  # Scale to reasonable size
    else:
        p = np.zeros(n)
    return p

    radius = radius_init
    stagnation = 0

        M = float(np.max(scores))

            p = subgradient_direction(h, scores, grads, top_k)
        
        if np.max(np.abs(p)) < 1e-12:
            # No useful direction
            radius *= 0.5
            if radius < 1e-6:
                break
            continue
        
        # Line search
        alpha, c5_new = line_search(h, p, c5, max_alpha=radius / (np.max(np.abs(p)) + 1e-15))

            h = np.clip(h, 0.0, 1.0)
            h = project_to_target(h, n / 2.0)
            c5 = exact_c5(h)
            best_h = h.copy()
            best_c5 = c5
            stagnation = 0
            # Expand radius slightly

        else:
            stagnation += 1
            radius *= 0.7

import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_target(v, target):
    n = len(v)
    v = np.clip(np.nan_to_num(v), -50.0, 50.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(-result)
            for i in idx:
                add = min(1.0 - result[i], residual)
                result[i] += add
                residual -= add
                if residual < 1e-12:
                    break
        else:
            idx = np.argsort(result)
            for i in idx:
                sub = min(result[i], -residual)
                result[i] -= sub
                residual += sub
                if residual > -1e-12:
                    break
    return result

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def lag_gradient(h, lag):
    """Analytical gradient of a single lag score w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # d/dh[j] = (1-h[j-lag])*[0<=j-lag<n] - h[j+lag]*[0<=j+lag<n]
    if lag >= 0:
        j1_start, j1_end = lag, n
        if j1_end > j1_start:
            grad[j1_start:j1_end] += (1.0 - h[j1_start - lag:j1_end - lag]) * dx
        j2_start, j2_end = 0, n - lag
        if j2_end > j2_start:
            grad[j2_start:j2_end] -= h[j2_start + lag:j2_end + lag] * dx
    else:
        L = -lag
        j1_start, j1_end = 0, n - L
        if j1_end > j1_start:
            grad[j1_start:j1_end] += (1.0 - h[j1_start + L:j1_end + L]) * dx
        j2_start, j2_end = L, n
        if j2_end > j2_start:
            grad[j2_start:j2_end] -= h[j2_start - L:j2_end - L] * dx
    return grad

def verify_gradients(h):
    n = len(h)
    dx = 2.0 / n
    eps = 1e-6
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    rng = np.random.default_rng(123)
    for _ in range(8):
        r = int(rng.integers(0, 2 * n - 1))
        j = int(rng.integers(0, n))
        h_plus = h.copy()
        h_plus[j] += eps
        corr_plus = np.correlate(h_plus, 1.0 - h_plus, mode='full') * dx
        fd = (corr_plus[r] - corr[r]) / eps
        lag = r - (n - 1)
        g = lag_gradient(h, lag)
        if abs(g[j] - fd) > 1e-3:
            return False
    return True

def lp_find_direction(h, scores, radius, top_k):
    """Find LP-based minimax descent direction."""
    n = len(h)
    M = float(np.max(scores))
    sorted_idx = np.argsort(-scores)
    active = sorted_idx[:top_k]
    
    n_active = len(active)
    # Variables: p(n), tau(1), r(1)
    c = np.zeros(n + 2)
    c[n] = 1.0
    c[n + 1] = 3.0  # regularization on L-inf norm
    
    n_ineq = n_active + 2 * n
    A_ub = np.zeros((n_ineq, n + 2))
    b_ub = np.zeros(n_ineq)
    
    for i, ar in enumerate(active):
        lag = ar - (n - 1)
        g = lag_gradient(h, lag)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[ar]
    
    base = n_active
    for i in range(n):
        A_ub[base + i, i] = 1.0
        A_ub[base + i, n + 1] = -1.0
        b_ub[base + i] = 0.0
        A_ub[base + n + i, i] = -1.0
        A_ub[base + n + i, n + 1] = -1.0
        b_ub[base + n + i] = 0.0
    
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i])
        hi_i = min(radius, 1.0 - h[i])
        if lo_i > hi_i:
            lo_i = hi_i = 0.0
        bounds.append((lo_i, hi_i))
    bounds.append((None, None))
    bounds.append((0.0, radius))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            p = res.x[:n]
            tau = res.x[n]
            if tau < M - 1e-8 and np.max(np.abs(p)) > 1e-8:
                return p
    except Exception:
        pass
    return None

def subgradient_direction(h, scores, top_k):
    n = len(h)

    g_sum = np.zeros(n)
