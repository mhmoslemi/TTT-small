def project_feasible(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    z = np.clip(z, -10, 10)  # prevent overflow
    
    # Bisection on lambda
    lo = -1.0 - z.max()  # ensures sum(clip) = 0
    hi = 1.0 - z.min()   # ensures sum(clip) = n
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            # Need to add to some entries
            order = np.argsort(1.0 - h)  # most room first (actually least room first for stability)
            # Actually, we want to add to entries with most room
            order = np.argsort(-(1.0 - h))  # descending room
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)  # ascending h, subtract from smallest first
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, linprog
import time

# Constants
ACC = 1e-12

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad(h, p):
    """Gradient of (V_p * dx) w.r.t. h. p is index in full corr (0..2n-2)."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        g[:L] += (1.0 - h[s:])
        g[s:] -= h[:L]
    else:
        t = -s
        L = n - t
        g[t:] += (1.0 - h[:L])
        g[:L] -= h[t:]
    return dx * g

def project_feasible(z):
    """Project z to [0,1]^n with sum = n/2."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float)
    
    lo = -z.max() - 1.0
    hi = 1.0 - z.min()
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(-(1.0 - h))
            for idx in order:
                add = min(residual, 1.0 - h[idx] + 1e-15)
                if add <= 0: break
                h[idx] = min(1.0, h[idx] + add)
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx] - 1e-15)
                if sub <= 0: break
                h[idx] = max(0.0, h[idx] - sub)
                residual += sub
                if abs(residual) < 1e-15: break
    
    return np.clip(h, 0.0, 1.0)

def active_set_lp(h, k_active=30):
    """Solve active set LP to get a direction."""
    n = len(h)
    L = 2 * n - 1
    vals = compute_values(h)
    
    # Select active set: top k by value + all within epsilon of max
    v_max = vals.max()
    eps = 0.01
    active_mask = (vals >= v_max - eps)
    # Also include top k
    top_k = np.argsort(vals)[-k_active:]
    active = np.union1d(np.where(active_mask)[0], top_k)
    active = np.sort(active)
    
    # Build LP: min tau s.t. grad_p . h - tau <= rhs_p, sum(h)=n/2, 0<=h<=1
    # x = [h_0, ..., h_{n-1}, tau]
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    
    for i, p in enumerate(active):
        g = compute_grad(h, p)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = np.dot(g, h) - vals[p]  # g.(h-h*) <= tau - V_p => g.h - tau <= g.h* - V_p
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]  # tau in [0,1] should be fine
    
    try:
        res = linprog(c=np.zeros(n+1), A_ub=A_ub, b_ub=b_ub, 
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_lp = res.x[:n]
            direction = h_lp - h
            return direction, active, vals
    except Exception:
        pass
    return None, active, vals

def line_search(h, direction, alphas=None):
    """Try different step sizes and return the best h."""
    if alphas is None:
        alphas = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]

def smooth_obj_and_grad(h, beta):
    """Compute log-sum-exp smooth objective and gradient."""
    n = len(h)
    vals = compute_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_shift = shifted.max()
    exp_shifted = np.exp(shifted - max_shift)
    logsum = max_shift + np.log(exp_shifted.sum())
    obj = cmax + logsum / beta
    
    w = exp_shifted / exp_shifted.sum()
    # Gradient: sum_p w_p * grad_p
    grad = np.zeros(n)
    for p in range(len(vals)):
        if w[p] > 1e-15:
            grad += w[p] * compute_grad(h, p)
    
    return obj, grad

def try_smooth(h0, beta, maxiter=200, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        return smooth_obj_and_grad(h, beta)[0]
    
    def grad(h):
        return smooth_obj_and_grad(h, beta)[1]
    
    def eq_fun(h):
        return np.sum(h) - n / 2.0
    
    def eq_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or (res.x is not None and np.all(np.isfinite(res.x))):
            h_new = project_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

But actually, I can vectorize the gradient computation. Let me think...

For the smooth gradient: grad = sum_p w_p * grad_p

For s >= 0 shifts: grad_p has the form:
g[:L] += (1-h[s:])  => contributes to positions 0..L-1
g[s:] -= h[:L]     => contributes to positions s..n-1

For s < 0 (t=-s) shifts: grad_p has the form:
g[t:] += (1-h[:L])  => contributes to positions t..n-1
g[:L] -= h[t:]      => contributes to positions 0..L-1

I can vectorize this using cumulative sums or direct indexing. But for n=60, the loop is fast enough. Let me not over-optimize.
