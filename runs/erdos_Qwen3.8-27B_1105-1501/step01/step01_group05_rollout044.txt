def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n_target):
    h = np.array(v, dtype=float).ravel().copy()
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    else:
        h = h[:n_target]
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for m in range(L):
        s = m - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i - s < n:
                g += (1.0 - h[i - s])
            if 0 <= i + s < n:
                g += (-h[i + s])
            G[m, i] = g
    return G


def gen_interpolated(parent, n_target, shift=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + shift) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)

def discrete_sa(h_init, n, time_limit=60.0, seed=42, T_init=None, T_min=None):
    rng = np.random.default_rng(seed)
    if n % 2 != 0:
        n = n + 1
    target = n // 2
    h = project_to_target(h_init, n)
    # Convert to binary
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    score = official_score(b)
    best_b = b.copy()
    best_score = score

    if T_init is None:
        T_init = 0.1
    if T_min is None:
        T_min = 0.001

    start_time = time.perf_counter()
    total_iters = 0
    no_improve = 0

    while time.perf_counter() - start_time < time_limit:
        total_iters += 1
        T = T_init * (T_min / T_init) ** (total_iters / 5000.0)
        T = max(T, T_min)

        # Move: swap a 1 with a 0
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]

        b_new = b.copy()
        b_new[i] = 0.0
        b_new[j] = 1.0
        s_new = official_score(b_new)

        delta_s = s_new - score
        if delta_s < 0:
            accept = True
        else:
            accept = rng.random() < math.exp(-delta_s / max(T, 1e-10))

        if accept:
            b = b_new
            score = s_new
            if score < best_score:
                best_b = b.copy()
                best_score = score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3000:
            # Restart with perturbation
            for _ in range(10):
                ones = np.where(b == 1.0)[0]
                zeros = np.where(b == 0.0)[0]
                i = ones[rng.integers(0, len(ones))]
                j = zeros[rng.integers(0, len(zeros))]
                b[i] = 0.0
                b[j] = 1.0
            score = official_score(b)
            no_improve = 0

    # Convert back to continuous
    h_out = project_to_target(best_b, n)
    return h_out, official_score(h_out)


def slsqp_polish(h, time_limit=20.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        G = compute_grad_matrix(h_x) * dx
        G = G * (-1.0)
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, :n] = G
        j[:, n] = 1.0
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass

    return h, initial_score


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 6.0

    # Get parent
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    rng = np.random.default_rng(seed)

    if parent is not None:
        n_parent = len(parent)
        best_h = project_to_target(parent, n_parent)
    else:
        n_parent = 64
        best_h = project_to_target(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)

    # Build candidate pool
    candidates = []
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h).copy())
    candidates.append((1.0 - best_h[::-1]).copy())

    # Interpolations to nearby resolutions
    n_list = sorted(set([
        best_n, max(32, best_n - 5), best_n + 5, best_n + 10,
        max(32, best_n - 10), best_n + 15
    ]))
    n_list = [n for n in n_list if 32 <= n <= 140]

    for n_try in n_list:
        if time.perf_counter() > deadline - 60:
            break
        candidates.append(gen_interpolated(best_h, n_try, shift=0))
        candidates.append(gen_interpolated(best_h, n_try, shift=1))
        candidates.append(gen_interpolated(best_h, n_try, shift=2))

    # Random candidates
    for n_try in n_list[:5]:
        if time.perf_counter() > deadline - 60:
            break
        for _ in range(3):
            candidates.append(gen_random_continuous(n_try, rng))
        for _ in range(3):
            candidates.append(gen_random_binary(n_try, rng))
        for _ in range(2):
            candidates.append(gen_sine(n_try, rng))
        for _ in range(2):
            candidates.append(gen_step_family(n_try, rng, num_steps=5))

    # Score all candidates
    scored = []
    for h_c in candidates:
        if not np.all(np.isfinite(h_c)):
            continue
        h_c = project_to_target(h_c, len(h_c))
        s = official_score(h_c)
        if np.isfinite(s):
            scored.append((s, h_c))

    scored.sort(key=lambda x: (x[0], len(x[1])))

    if len(scored) > 0 and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    # Phase 1: LP local descent on top candidates
    n_top = min(8, len(scored))
    results = []
    for idx in range(n_top):
        if time.perf_counter() > deadline - 200:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_limit = min(80.0, max(10.0, remaining / max(1, n_top - idx) * 0.5))
        if t_limit < 8.0:
            t_limit = 8.0
        h_loc, s_loc = lp_local_descent(h_c, max_iters=60, time_limit=t_limit, seed=seed + idx * 13)
        results.append((s_loc, h_loc))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = len(h_loc)

    results.sort(key=lambda x: x[0])

    # Phase 2: Discrete binary search
    if time.perf_counter() < deadline - 150:
        # Use best continuous to seed binary
        n_disc = best_n if best_n % 2 == 0 else best_n + 1
        if n_disc > 100:
            n_disc = 96 if n_disc > 100 else n_disc
        h_seed_bin = gen_interpolated(best_h, n_disc)
        h_disc, s_disc = discrete_sa(h_seed_bin, n_disc,
                                     time_limit=min(40.0, max(10.0, deadline - time.perf_counter() - 80)),
                                     seed=seed + 100)
        if s_disc < best_score:
            best_score = s_disc
            best_h = h_disc.copy()
            best_n = len(h_disc)

        # Second discrete run with different init
        if time.perf_counter() < deadline - 100:
            h_seed2 = gen_random_binary(n_disc, rng)
            h_disc2, s_disc2 = discrete_sa(h_seed2, n_disc,
                                           time_limit=min(30.0, max(5.0, deadline - time.perf_counter() - 60)),
                                           seed=seed + 200)
            if s_disc2 < best_score:
                best_score = s_disc2
                best_h = h_disc2.copy()
                best_n = len(h_disc2)

    # Phase 3: SLSQP polish on best results
    to_refine = results[:min(5, len(results))]
    if best_h is not None:
        to_refine.append((best_score, best_h.copy()))

    for idx in range(len(to_refine)):
        if time.perf_counter() > deadline - 40:
            break
        s_c, h_c = to_refine[idx]
        h_ref, s_ref = slsqp_polish(h_c, time_limit=15.0)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
            best_n = len(h_ref)
        # Follow up with local descent
        if time.perf_counter() < deadline - 25:
            h_2, s_2 = lp_local_descent(h_ref, max_iters=20, time_limit=8.0, seed=seed + 300 + idx)
            if s_2 < best_score:
                best_score = s_2
                best_h = h_2.copy()
                best_n = len(h_2)

    # Phase 4: Additional interpolated runs from best
    if time.perf_counter() < deadline - 50:
        for n_try in [best_n, max(32, best_n - 5), best_n + 5, best_n + 10]:
            if time.perf_counter() > deadline - 20:
                break
            if n_try < 32 or n_try > 140:
                continue
            h_interp = gen_interpolated(best_h, n_try)
            h_loc, s_loc = lp_local_descent(h_interp, max_iters=30, time_limit=12.0, seed=seed + 400 + n_try)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()
                best_n = len(h_loc)
            if time.perf_counter() < deadline - 15:
                h_ref, s_ref = slsqp_polish(h_loc, time_limit=10.0)
                if s_ref < best_score:
                    best_score = s_ref
                    best_h = h_ref.copy()
                    best_n = len(h_ref)

    # Phase 5: Final polish
    if time.perf_counter() < deadline - 15:
        h_fin, s_fin = slsqp_polish(best_h, time_limit=10.0)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()
        if time.perf_counter() < deadline - 8:
            h_fin2, s_fin2 = lp_local_descent(best_h, max_iters=15, time_limit=5.0, seed=seed + 999)
            if s_fin2 < best_score:
                best_score = s_fin2
                best_h = h_fin2.copy()

    # Final validation
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)

    # Ensure validity
    if not np.all(np.isfinite(best_h)) or np.any(best_h < -1e-10) or np.any(best_h > 1 + 1e-10):
        if parent is not None:
            best_h = project_to_target(parent, len(parent))
            final_score = official_score(best_h)
        else:
            best_h = project_to_target(np.full(64, 0.5), 64)
            final_score = official_score(best_h)

    c5_bound = float(final_score)
    if c5_bound < 0:
        c5_bound = 0.0

    return (best_h, c5_bound, int(len(best_h)))
