import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n):
    """Project v to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(np.asarray(v, dtype=float).ravel()[:n], 0.0, 1.0)
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    
    # Bisection for uniform shift
    lo = -(max(h) + 0.1)
    hi = (1.0 - min(h) + 0.1)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        shifted = np.clip(h + mid, 0.0, 1.0)
        if np.sum(shifted) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Exact mass correction
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_matrix(h):
    """Compute the (2n-1) x n gradient matrix for scaled correlations.
    Row m corresponds to lag s = m - (n-1)."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n-1), n):
        m = s + n - 1
        # c_s = sum_{i: 0<=i<n, 0<=i+s<n} h[i] * (1-h[i+s])
        # dc_s/dh[j] = (1-h[j+s]) if 0<=j+s<n, plus (-h[j-s]) if 0<=j-s<n
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g += (-h[j - s])
            G[m, j] = dx * g
    return G


def lp_minimax_direction(h, c_scaled, G, max_active=50):
    """Find a descent direction using LP minimax formulation.
    Returns direction d (length n) or None."""
    n = len(h)
    L = len(c_scaled)
    score = np.max(c_scaled)
    
    # Active constraints: those near the maximum
    margins = score - c_scaled
    active_tol = max(0.01, 0.05 * score)
    active_idx = np.where(margins <= active_tol)[0]
    
    if len(active_idx) == 0:
        active_idx = np.argsort(margins)[:max_active]
    elif len(active_idx) > max_active:
        # Keep the most active ones
        order = np.argsort(margins[active_idx])
        active_idx = active_idx[order[:max_active]]
    
    # LP: minimize t
    # s.t. G[active] @ d + c_scaled[active] <= t
    #      sum(d) = 0
    #      -h[j] <= d[j] <= 1-h[j]
    #      t >= 0 (optional, to ensure we're looking for improvement)
    
    # Variables: [d_0, ..., d_{n-1}, t]
    n_vars = n + 1
    t_idx = n
    
    # Objective: minimize t
    c_obj = np.zeros(n_vars)
    c_obj[t_idx] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag m: G[m,:] @ d - t <= -c_scaled[m]
    n_active = len(active_idx)
    A_ub = np.zeros((n_active + 2*n, n_vars))
    b_ub = np.zeros(n_active + 2*n)
    
    for i, m in enumerate(active_idx):
        A_ub[i, :n] = G[m, :]
        A_ub[i, t_idx] = -1.0
        b_ub[i] = -c_scaled[m]
    
    # Box constraints on d: d[j] <= 1-h[j] and -d[j] <= h[j]
    for j in range(n):
        A_ub[n_active + j, j] = 1.0
        b_ub[n_active + j] = 1.0 - h[j]
        A_ub[n_active + n + j, j] = -1.0
        b_ub[n_active + n + j] = h[j]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds_lb = np.zeros(n_vars)
    bounds_ub = np.ones(n_vars) * np.inf
    
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                               bounds=(bounds_lb, bounds_ub), method='highs')
        if res.status == 0 and res.x is not None:
            d = res.x[:n]
            # Check if this is actually a descent direction
            # The predicted max change should be negative
            pred_change = np.max(c_scaled[active_idx] + G[active_idx] @ d) - score
            if pred_change < -1e-8:
                return d
    except Exception:
        pass
    return None


def fallback_direction(h, c_scaled, G):
    """Fallback: negative sum of gradients of active constraints."""
    n = len(h)
    score = np.max(c_scaled)
    margins = score - c_scaled
    active_tol = max(0.02, 0.1 * score)
    active_idx = np.where(margins <= active_tol)[0]
    
    if len(active_idx) == 0:
        active_idx = np.argsort(margins)[:5]
    
    d = -np.sum(G[active_idx], axis=0)
    d -= np.mean(d)  # Ensure sum ≈ 0
    
    norm = np.max(np.abs(d))
    if norm < 1e-12:
        return None
    d /= norm
    
    # Check descent
    pred_max = np.max(c_scaled + G @ d)
    if pred_max > score + 1e-8:
        d = -d
        pred_max = np.max(c_scaled + G @ d)
        if pred_max > score + 1e-8:
            return None
    return d


def line_search(h, d, score, c_scaled, G, n):
    """Line search: find best alpha to move along direction d."""
    alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
    best_alpha = 0.0
    best_score = score
    best_h = h
    
    for a in alphas:
        for sign in [1, -1]:
            h_try = project_to_target(h + sign * a * d, n)
            s = official_score(h_try)
            if s < best_score - 1e-10:
                best_score = s
                best_alpha = sign * a
                best_h = h_try
    return best_h, best_score


def lp_local_search(h, n, max_iters=80, time_limit=30.0):
    """LP-based minimax local search."""
    h = project_to_target(h, n)
    score = official_score(h)
    dx = 2.0 / n
    start_time = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start_time > time_limit:
            break
        
        c_scaled = np.correlate(h, 1.0 - h, mode="full") * dx
        G = compute_grad_matrix(h)
        
        # Get LP direction
        d = lp_minimax_direction(h, c_scaled, G, max_active=40)
        if d is None:
            d = fallback_direction(h, c_scaled, G)
        if d is None:
            break
        
        # Line search
        h_new, s_new = line_search(h, d, score, c_scaled, G, n)
        if s_new < score - 1e-10:
            h = h_new
            score = s_new
        else:
            # Try smaller steps or different direction
            break
    
    return h, official_score(h)


def discrete_sa(n, n_iters=50000, time_limit=30.0, seed=42):
    """Simulated annealing over binary patterns with exactly n/2 ones (n even)."""
    if n % 2 != 0:
        return None, None
    
    rng = np.random.default_rng(seed)
    ones_count = n // 2
    
    # Initial: random balanced binary
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:ones_count]] = 1.0
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    
    T0 = 0.3
    T1 = 0.01
    start_time = time.perf_counter()
    
    for it in range(n_iters):
        if time.perf_counter() - start_time > time_limit:
            break
        
        # Temperature schedule
        frac = it / max(1, n_iters)
        T = T0 * (T1 / T0) ** frac
        
        # Move: swap a 1 with a 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_score = official_score(h)
        
        delta = new_score - score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_score


def gen_interpolated(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)


def gen_structured(n, rng):
    """Generate various structured candidates."""
    t = np.linspace(0, 1, n)
    
    # Try different structures
    candidates = []
    
    # Sine wave
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(1.5, 6.0)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t + phase)
    candidates.append(h)
    
    # Triangle wave
    h2 = 0.5 + 0.4 * np.abs(2 * ((t * freq + phase / (2*np.pi)) % 1) - 1) - 0.4
    candidates.append(h2)
    
    # Block structure
    n_blocks = rng.integers(2, 6)
    block_size = n // n_blocks
    vals = rng.uniform(0.1, 0.9, n_blocks)
    h3 = np.repeat(vals, block_size)[:n]
    if len(h3) < n:
        h3 = np.pad(h3, (0, n - len(h3)))
    candidates.append(h3)
    
    # Alternating pattern with noise
    h4 = np.where(np.arange(n) % 2 == 0, 0.8, 0.2) + rng.normal(0, 0.1, n)
    candidates.append(h4)
    
    # Ramp with wrap
    h5 = np.linspace(0.0, 1.0, n) * rng.uniform(0.3, 0.7) + 0.5
    candidates.append(h5)
    
    best = None
    best_s = np.inf
    for h_c in candidates:
        h_c = project_to_target(h_c, n)
        s = official_score(h_c)
        if s < best_s:
            best_s = s
            best = h_c
    return best


def slsqp_polish(h, n, maxiter=50):
    """Apply SLSQP to minimize the maximum correlation."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    
    x0 = np.concatenate([h, [theta0]])
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        hx = x[:n]
        corr = np.correlate(hx, 1.0 - hx, mode="full") * dx
        return x[n] - corr
    
    def ineq_jac(x):
        hx = x[:n]
        G = compute_grad_matrix(hx)
        j = np.zeros((2*n-1, n+1))
        j[:, :n] = -G
        j[:, n] = 1.0
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': maxiter, 'ftol': 1e-10})
        if result.x is not None:
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score:
                return h_out, s
    except Exception:
        pass
    return h, initial_score

def project_to_target(v, n):

        else:
            hi = mid

def gen_interpolated(parent, n_target, offset=0):

        return project_to_target(np.full(n_target, 0.5), n_target)

        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': maxiter, 'ftol': 1e-10})
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + max(60, budget_s) - 12.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 10 and np.all(np.isfinite(pv)) and np.all(pv >= -0.01) and np.all(pv <= 1.01):
            parent = project_to_target(pv, len(pv))
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = parent.copy()
    else:
        n_parent = 64
        best_h = project_to_target(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    # Phase 1: Build diverse candidate pool
    candidates = []
    candidates.append((best_score, best_h.copy()))

    transforms = [best_h[::-1].copy(), (1.0 - best_h).copy(), (1.0 - best_h[::-1]).copy()]
    for tr in transforms:
        tr = project_to_target(tr, best_n)
        s = official_score(tr)
        candidates.append((s, tr))
        if s < best_score:
            best_score = s
            best_h = tr.copy()

    n_list = sorted(set([best_n, best_n - 4, best_n + 4, best_n + 8,
                         best_n + 12, max(32, best_n - 8), best_n - 12 if best_n > 40 else 32,
                         best_n + 16]))
    n_list = [n for n in n_list if 28 <= n <= 150]

    for n_try in n_list:
        if time.perf_counter() > deadline - 200:
            break
        for offset in [0, 1, 2, 3]:
            h_interp = gen_interpolated(best_h, n_try, offset)
            s = official_score(h_interp)
            candidates.append((s, h_interp))

    for n_try in n_list[:6]:
        if time.perf_counter() > deadline - 180:
            break
        for _ in range(4):
            for h_c in gen_diverse(n_try, rng):
                candidates.append((official_score(h_c), h_c))

    candidates.sort(key=lambda x: x[0])
    if candidates[0][0] < best_score:
        best_score = candidates[0][0]
        best_h = candidates[0][1].copy()
        best_n = len(best_h)

    # Phase 2: LP local search on top candidates
    top_k = min(10, len(candidates))
    local_results = []

    for idx in range(top_k):
        if time.perf_counter() > deadline - 150:
            break
        h_c = candidates[idx][1].copy()
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_limit = min(30.0, max(5.0, remaining * 0.5 / max(1, top_k - idx)))

        h_loc, s_loc = lp_local_search(h_c, n_c, max_iters=60, time_limit=t_limit, seed=seed + idx * 13)
        local_results.append((s_loc, h_loc))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c

        # Coordinate descent follow-up
        if time.perf_counter() < deadline - 130:
            h_cd, s_cd = coordinate_descent(h_loc, n_c, max_iters=100, time_limit=min(8.0, t_limit * 0.3), seed=seed + 100 + idx)
            if s_cd < best_score:
                best_score = s_cd
                best_h = h_cd.copy()
                best_n = n_c
                local_results.append((s_cd, h_cd))

    local_results.sort(key=lambda x: x[0])
    if local_results[0][0] < best_score:
        best_score = local_results[0][0]
        best_h = local_results[0][1].copy()
        best_n = len(best_h)

    # Phase 3: Discrete SA
    if time.perf_counter() < deadline - 80:
        even_ns = sorted(set([n for n in n_list if n % 2 == 0] +
                             [best_n if best_n % 2 == 0 else best_n + 1,
                              best_n if best_n % 2 == 0 else best_n - 1]))
        even_ns = [n for n in even_ns if 32 <= n <= 130][:3]

        for n_disc in even_ns:
            if time.perf_counter() > deadline - 50:
                break
            h_cont_n = gen_interpolated(best_h, n_disc) if len(best_h) != n_disc else best_h.copy()
            h_bin_init = (h_cont_n >= 0.5).astype(float)
            # Fix to exactly n/2 ones
            ones = int(np.sum(h_bin_init == 1))
            if ones > n_disc // 2:
                one_idx = np.where(h_bin_init == 1)[0]
                keep = h_bin_init[one_idx].argsort()[-(n_disc // 2):]
                h_bin_init[one_idx] = 0
                h_bin_init[one_idx[keep]] = 1
            elif ones < n_disc // 2:
                zero_idx = np.where(h_bin_init == 0)[0]
                need = n_disc // 2 - ones
                h_bin_init[zero_idx[:need]] = 1

            h_disc, s_disc = discrete_sa(n_disc, n_iters=50000, time_limit=min(25.0, (deadline - time.perf_counter()) * 0.25), seed=seed + 200, h_init=h_bin_init)

            if s_disc is not None and s_disc < best_score:
                best_score = s_disc
                best_h = h_disc.copy()
                best_n = n_disc

            if h_disc is not None:
                h_conv = project_to_target(h_disc.astype(float), n_disc)
                if time.perf_counter() < deadline - 40:
                    h_loc2, s_loc2 = lp_local_search(h_conv, n_disc, max_iters=40, time_limit=min(15.0, (deadline - time.perf_counter()) * 0.3), seed=seed + 300)
                    if s_loc2 < best_score:
                        best_score = s_loc2
                        best_h = h_loc2.copy()
                        best_n = n_disc

    # Phase 4: SLSQP polish
    if time.perf_counter() < deadline - 30:
        h_ref, s_ref = slsqp_polish(best_h, best_n, maxiter=70)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()

        # LP after SLSQP
        if time.perf_counter() < deadline - 15:
            h_post, s_post = lp_local_search(best_h, best_n, max_iters=30, time_limit=min(12.0, (deadline - time.perf_counter()) * 0.5), seed=seed + 400)
            if s_post < best_score:
                best_score = s_post
                best_h = h_post.copy()

    # Phase 5: Try more resolutions from best
    if time.perf_counter() < deadline - 20:
        for n_try in [best_n + 5, best_n - 5, best_n + 10]:
            if time.perf_counter() > deadline - 10 or n_try < 28 or n_try > 150:
                continue
            h_interp = gen_interpolated(best_h, n_try)
            h_loc3, s_loc3 = lp_local_search(h_interp, n_try, max_iters=30, time_limit=min(10.0, (deadline - time.perf_counter()) * 0.4), seed=seed + 500)
            if s_loc3 < best_score:
                best_score = s_loc3
                best_h = h_loc3.copy()
                best_n = n_try

    # Phase 6: Final SLSQP
    if time.perf_counter() < deadline - 5:
        h_fin, s_fin = slsqp_polish(best_h, best_n, maxiter=40)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)
    assert abs(np.sum(best_h) * (2.0 / len(best_h)) - 1.0) < 1e-9
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)

    return (best_h, float(final_score), int(len(best_h)))
