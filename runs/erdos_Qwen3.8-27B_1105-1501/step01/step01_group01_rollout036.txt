def find_1d_minimax(A, B):
    """Find t in [0,1] minimizing max_k (A[k] + B[k]*t)."""
    t_lo, t_hi = 0.0, 1.0
    for _ in range(50):
        t_mid = (t_lo + t_hi) / 2.0
        vals = A + B * t_mid
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        min_b = float(np.min(B[active]))
        max_b = float(np.max(B[active]))
        if min_b > 1e-12:
            t_hi = t_mid
        elif max_b < -1e-12:
            t_lo = t_mid
        else:
            return t_mid
    return (t_lo + t_hi) / 2.0

def coord_descent(h0, max_sweeps=300, seed=42, tol=1e-14):
    """Coordinate descent for the Erdős overlap problem."""
    rng = np.random.default_rng(seed)
    h = make_feasible(h0)
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    stall = 0
    
    for sweep in range(max_sweeps):
        order = rng.permutation(n)
        h_new = h.copy()
        improved = False
        for p in order:
            corr = np.correlate(h_new, 1.0 - h_new, mode='full')
            A = np.zeros(2*n-1)
            B = np.zeros(2*n-1)
            for k in range(2*n-1):
                s = k - (n-1)
                c_val = corr[k]
                # Subtract terms involving h_new[p]
                if 0 <= p+s < n:
                    c_val -= h_new[p] * (1.0 - h_new[p+s])
                if 0 <= p-s < n:
                    c_val -= h_new[p-s] * (1.0 - h_new[p])
                A[k] = c_val
                if 0 <= p-s < n:
                    A[k] += h_new[p-s]
                if 0 <= p+s < n:
                    B[k] += (1.0 - h_new[p+s])
                if 0 <= p-s < n:
                    B[k] -= h_new[p-s]
            t_opt = find_1d_minimax(A, B)
            if abs(t_opt - h_new[p]) > 1e-10:
                improved = True
            h_new[p] = t_opt
        
        h_new = make_feasible(h_new)
        score = official_score(h_new)
        if score < best_score - tol:
            best_score = score
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 50:
            break
    
    return best_h, best_score
