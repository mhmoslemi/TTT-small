```python
def lag_gradient(h, k, n):
    """Gradient of F_k with respect to h, where k is the lag index (-n+1 to n-1)."""
    grad = np.zeros(n)
    if k >= 0:
        # F_k = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])

        for i in range(j, n):
```

```python
def make_feasible(z):
    """Project z onto {0<=h<=1, sum(h)=n/2}."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    # Binary search on shift
    lo = np.min(z) - 10.0
    hi = np.max(z) + 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        if err > 0:
            order = np.argsort(-h)  # prefer to add to smaller values
            # Actually, to minimize perturbation, add to values closest to 0.5
            order = np.argsort(np.abs(h - 0.5))

                d = min(err, 1.0 - h[i])

                err -= d

def lag_gradient_vec(h, k, n):
    """Vectorized gradient of F_k w.r.t. h."""
    grad = np.zeros(n)
    if k >= 0:
        # F_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # dF_k/dh[i] = (1-h[i-k]) for i in [k, n-1]
        #             - h[i+k] for i in [0, n-1-k]
        if n - k > 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
    else:
        j = -k
        # F_{-j} = sum_{i=0}^{n-1-j} h[i]*(1-h[i+j])
        # dF_{-j}/dh[i] = (1-h[i+j]) for i in [0, n-1-j]
        #                - h[i-j] for i in [j, n-1]
        if n - j > 0:
            grad[:n-j] += (1.0 - h[j:])
            grad[j:] -= h[:n-j]
    return grad

def smooth_obj_and_grad(h, beta):
    """Compute smooth log-sum-exp objective and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    C = compute_all_F(h) * dx  # shape (2n-1,)
    M = np.max(C)
    shifted = beta * (C - M)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    es = np.exp(shifted)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = es / ws
    obj = M + np.log(ws) / beta
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2*n-1):
        w = weights[idx]
        if abs(w) < 1e-12:
            continue
        k = idx - (n - 1)
        grad += w * lag_gradient_vec(h, k, n)
    grad *= dx
    return obj, grad

def safe_sigmoid(x):
    """Numerically stable sigmoid."""
    out = np.zeros_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def safe_logit(h):
    """Numerically stable logit (inverse sigmoid)."""
    eps = 1e-15
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def solve_logit_bias(y):
    """Find c such that sum(sigmoid(y_i + c)) = len(y)/2."""
    n = len(y)
    target = n / 2.0
    lo = np.min(y) - 50.0
    hi = np.max(y) + 50.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(safe_sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) * 0.5

    """Convert logit coordinates to h, enforcing sum constraint."""
    c = solve_logit_bias(y)
    h = safe_sigmoid(y + c)
    # Small correction for exact sum
    err = len(h) / 2.0 - np.sum(h)
    if abs(err) > 1e-14:
        # Distribute correction proportional to h(1-h)
        w = h * (1.0 - h)
        w_sum = np.sum(w)
        if w_sum > 1e-15:
            h += err * w / w_sum
    return np.clip(h, 0.0, 1.0)

def logit_obj(y, beta):
    """Objective in logit space."""
    h = logit_h(y)
    obj, _ = smooth_obj_and_grad(h, beta)
    return obj

    h = logit_h(y)
    c = solve_logit_bias(y)

    w = h * (1.0 - h)
    W = np.sum(w)
    if W < 1e-15:
        return np.zeros_like(y)
    grad_y = w * (grad_h - np.sum(grad_h * w) / W)
    return grad_y
```

Now let me think about the active-set refinement more carefully.

The idea is:
1. At the current h, identify the Q lags with the highest overlap
2. Compute the average (or weighted) gradient of those Q lags
3. Take a step in the negative gradient direction
4. Project back to the feasible set
5. Use backtracking line search to ensure improvement

The key is that by focusing on the active lags, we get a smoother direction that's more likely to decrease the max without increasing other lags too much.

For the step size, I'll start with a reasonable value (based on the gradient magnitude) and backtrack if the score doesn't decrease.

```python
def active_set_step(h, Q, alpha0=None):
    """One active-set gradient step."""
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(C)[-Q:]
    g = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        g += lag_grad(h, k, n)
    g *= dx / Q
    
    # Project gradient onto tangent space (zero sum)
    g = g - g.mean()
    
    if alpha0 is None:
        alpha0 = 0.1 / (np.linalg.norm(g) + 1e-10)
    
    # Backtracking line search
    current = official_score(h)
    alpha = alpha0
    for _ in range(20):
        h_new = make_feasible(h - alpha * g)
        sc = official_score(h_new)
        if sc < current - 1e-15:
            return h_new, sc
        alpha *= 0.5
    return h, current
```

```python
    for dn in [-15, -10, -6, -4, -2, 2, 4, 6, 10, 15]:
        nn = len(best_h) + dn
        if 18 <= nn <= 80 and nn not in try_ns:
            try_ns.append(nn)
    try_ns.sort()

    for nn in try_ns:
        if time.time() > deadline - 120:
            break
        h_n = interp_to_length(best_h, nn)
        track(h_n)
        for beta in [100, 500]:
            if time.time() > deadline - 30:
                break
            h_opt = logit_optimize(h_n, beta, maxiter=150, gtol=1e-8)
            track(h_opt)
        # Active set on this resolution
        if time.time() > deadline - 20:
            break
        h_as, _ = active_set_refine(h_opt, 60, Q=3, max_time_s=8.0)
        track(h_as)

    # Phase 3: Coarse-to-fine
    if time.time() < deadline - 80:
        h_coarse = interp_to_length(best_h, max(20, len(best_h) // 3))
        for beta in [100, 500]:
            if time.time() > deadline - 30:
                break
            h_coarse = logit_optimize(h_coarse, beta, maxiter=150, gtol=1e-8)
        h_fine = interp_to_length(h_coarse, len(best_h))
        for beta in [500, 2000]:
            if time.time() > deadline - 20:
                break
            h_fine = logit_optimize(h_fine, beta, maxiter=200, gtol=1e-9)
        track(h_fine)

    # Phase 4: Fine-to-coarse-to-fine (escape local min)
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        h_down = interp_to_length(best_h, max(25, n_cur // 2))
        h_down = logit_optimize(h_down, 200, maxiter=100, gtol=1e-7)
        h_back = interp_to_length(h_down, n_cur)
        h_back = logit_optimize(h_back, 500, maxiter=200, gtol=1e-9)
        track(h_back)

    # Phase 5: Active-set refinement on best
    if time.time() < deadline - 40:
        for Q in [1, 2, 3, 5]:
            if time.time() > deadline - 15:
                break
            h_as, _ = active_set_refine(best_h, 80, Q=Q, max_time_s=10.0)
            track(h_as)

    # Phase 6: Final high-beta polish
    if time.time() < deadline - 20:
        for beta in [3000, 8000]:
            if time.time() > deadline - 10:
                break
            h_final = logit_optimize(best_h, beta, maxiter=200, gtol=1e-10)
            track(h_final)

    # Phase 7: One more active set round
    if time.time() < deadline - 10:
        h_as, _ = active_set_refine(best_h, 100, Q=3, max_time_s=10.0)
        track(h_as)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score >= 0.5:
        if parent is not None:
            best_h = make_feasible(parent.copy())
        else:
            best_h = make_feasible(np.full(pn, 0.5))
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
```
