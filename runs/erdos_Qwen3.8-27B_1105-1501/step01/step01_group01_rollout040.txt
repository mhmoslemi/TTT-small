def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_gradients(h):
    n = len(h)
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        # p as first index (i+s=p, i=p-s): max(0,-s)<=i<n and 0<=i+s<n
        # i = p-s, so we need max(0,-s) <= p-s < n and 0 <= p < n
        # => max(s,0) <= p < n + s and 0 <= p < n
        # => max(s,0) <= p <= min(n-1, n-1+s)
        p_lo = max(s, 0)
        p_hi = min(n-1, n-1+s)
        for p in range(p_lo, p_hi+1):
            i = p - s
            grad[k, p] += (1.0 - h[i])
        # p as second index (i=p): max(0,-s) <= p < n and 0 <= p+s < n
        # => max(0,-s) <= p <= min(n-1, n-1-s)
        p_lo2 = max(0, -s)
        p_hi2 = min(n-1, n-1-s)
        for p in range(p_lo2, p_hi2+1):
            j = p + s
            grad[k, p] += (-h[j])
    return grad

def eta_values(n):
    k = np.arange(2*n-1)
    s = k - (n-1)
    return (n - np.abs(s)).astype(float)

def verify_gradients(h, grad, tol=1e-6):
    n = len(h)
    # Check a few random entries
    rng = np.random.default_rng(0)
    for _ in range(5):
        k = rng.integers(0, 2*n-1)
        p = rng.integers(0, n)
        eps = 1e-8
        h_plus = h.copy()
        h_plus[p] += eps
        h_minus = h.copy()
        h_minus[p] -= eps
        corr_plus = np.correlate(h_plus, 1.0-h_plus, mode='full')
        corr_minus = np.correlate(h_minus, 1.0-h_minus, mode='full')
        numeric_grad = (corr_plus[k] - corr_minus[k]) / (2*eps)
        if abs(numeric_grad - grad[k,p]) > tol:
            return False
    return True

def majorization_solve(h, R, n_active=None, solvers=None):
    n = len(h)
    if n_active is None:
        n_active = min(2*n-1, 40)
    if solvers is None:
        solvers = ['ECOS', 'SCS', 'CLARABEL']
    
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    r = corr
    grad = compute_gradients(h)
    eta = eta_values(n)
    
    # Select active rows: top n_active by correlation value
    if n_active < 2*n-1:
        active_idx = np.argsort(r)[::-1][:n_active]
    else:
        active_idx = np.arange(2*n-1)
    
    # Build CVX problem
    delta = cvx.Variable(n)
    t = cvx.Variable()
    
    constraints = []
    constraints.append(cvx.sum(delta) == 0)
    constraints.append(delta >= -h + 1e-12)
    constraints.append(delta <= 1.0 - h - 1e-12)
    constraints.append(cvx.sum_squares(delta) <= R**2)
    
    for k in active_idx:
        # r[k] + grad[k] @ delta + eta[k] * sum_squares(delta) <= t
        constraints.append(
            r[k] + grad[k] @ delta + eta[k] * cvx.sum_squares(delta) <= t
        )
    
    objective = cvx.Minimize(t + 0.005 * cvx.sum_squares(delta))
    prob = cvx.Problem(objective, constraints)
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    d_val = delta.value
    t_val = t.value
    
    if d_val is None or not np.all(np.isfinite(d_val)):
        return t_val, None
    
    h_new = make_feasible(h + d_val)
    return t_val, h_new

def smooth_objective(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_gradient(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = exp_vals / exp_sum
    
    # Use the gradient matrix
    grad_mat = compute_gradients(x)  # shape (2n-1, n), gradient of raw corr
    # Scaled gradient: d/dx of (corr * dx) = dx * grad_mat (but we need to be careful)
    # Actually: corr_s[k] = dx * corr[k], so d(corr_s[k])/dx[p] = dx * grad_mat[k,p]
    # The smooth gradient = sum_k weights[k] * dx * grad_mat[k,p]
    # But we also need the derivative of max_c term... 
    # Actually, the smooth objective is max_c + log(sum exp(beta*(corr-max_c)))/beta
    # The gradient is: sum_k p_k * corr_s_grad_k where p_k = exp(beta*(corr_k - max_c)) / sum
    # = sum_k p_k * dx * grad_mat[k, :]
    
    total_grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        total_grad += w * grad_mat[k]
    return total_grad

def do_smooth_opt(h0, beta, maxiter=200):
    n = len(h0)
    try:
        res = minimize(
            smooth_objective, h0, args=(beta,),
            jac=smooth_gradient,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:

            else:

            if abs(add) < 1e-14:

def compute_gradients(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad_mat = compute_gradients(x)
    return weights @ (dx * grad_mat)

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def do_smooth_opt(h0, beta, maxiter=150):
    n = len(h0)
    try:
        res = minimize(
            smooth_obj, h0, args=(beta,),
            jac=smooth_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0
