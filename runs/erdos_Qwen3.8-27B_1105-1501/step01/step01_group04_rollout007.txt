def compute_c5(h):

def exact_waterfill(z, target):
    """Project z onto {0 <= x <= 1, sum(x) = target}"""
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    result = np.clip(z - alpha, 0.0, 1.0)
    # Fix residual
    residual = target - result.sum()
    if abs(residual) > 1e-15:
        if residual > 0:
            # Need to add mass
            slack = 1.0 - result
            idx = np.argsort(-slack)
            for i in idx:
                add = min(slack[i], residual)
                if add > 0:
                    result[i] += add
                    residual -= add
                    if abs(residual) < 1e-15:
                        break
        else:
            # Need to remove mass
            idx = np.argsort(result)

                rem = min(result[i], -residual)
                if rem > 0:
                    result[i] -= rem
                    residual += rem
                    if abs(residual) < 1e-15:
                        break
    return result

def active_lag_weights(h, alpha=20.0):
    """Get soft-max weights on lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    # Avoid overflow
    shifted = alpha * (corr - max_c)
    np.clip(shifted, -100, 0, out=shifted)
    exp_c = np.exp(shifted)
    total = np.sum(exp_c)
    if total < 1e-30:
        return np.zeros_like(corr)
    return exp_c / total

def gradient_from_weights(h, weights):
    """Compute gradient of sum_k weights[k] * corr[k] * dx"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    b = 1.0 - h
    # For lag L >= 0 (offset n-1+L in full correlate):
    # corr[n-1+L] = sum_{m=L}^{n-1} h[m] * (1-h[m-L])  for L>0
    # corr[n-1] = sum h[m]*(1-h[m])
    # Derivative wrt h[i]:
    #   from h[m] term: +weights*dx*b[m-L] for m=i, so L <= i, contributes b[i-L]
    #   from (1-h[m-L]) term: -weights*dx*h[m] for m-L=i, so m=i+L, contributes -h[i+L]
    
    # Actually let me re-derive. For np.correlate(h, 1-h, 'full'):
    # result[k] = sum_m h[m] * (1-h[m-(k-(n-1))])
    # where the sum is over valid m.
    # Let L = k - (n-1). Then result[k] = sum_m h[m] * (1-h[m-L])
    # valid range: 0 <= m < n and 0 <= m-L < n
    
    # d/dh[i] of result[k]:
    #   Term 1: h[i] appears as h[m] when m=i, contribution: (1-h[i-L]) if 0<=i-L<n
    #   Term 2: h[i] appears in (1-h[m-L]) when m-L=i, i.e., m=i+L, contribution: -h[i+L] if 0<=i+L<n
    
    # So grad[i] = dx * sum_L weights[n-1+L] * [(1-h[i-L])*I(0<=i-L<n) - h[i+L]*I(0<=i+L<n)]
    
    # This is a convolution-like operation. Let me implement it directly.
    for L in range(0, n):
        w = weights[n-1+L] * dx
        if w < 1e-15:
            continue
        # For m from L to n-1: result contributes h[m]*(1-h[m-L])
        # d/dh[i]: if i >= L: term (1-h[i-L]); if i+L < n: term -h[i+L]
        if L == 0:
            grad += w * (b - h)  # (1-h[i]) - h[i] = 1-2h[i]
        else:
            # i >= L: grad[i] += w * (1-h[i-L])
            grad[L:] += w * b[:n-L]
            # i+L < n, i < n-L: grad[i] += -w * h[i+L]
            grad[:n-L] -= w * h[L:]
    
    for L in range(1, n):
        w = weights[n-1-L] * dx
        if w < 1e-15:
            continue
        # For m from 0 to n-1-L: result contributes h[m]*(1-h[m+L])
        # d/dh[i]: if i < n-L: term (1-h[i+L]); if i-L >= 0: term -h[i-L]
        # i < n-L: grad[i] += w * (1-h[i+L])
        grad[:n-L] += w * b[L:]
        # i >= L: grad[i] += -w * h[i-L]
        grad[L:] -= w * h[:n-L]
    
    return grad

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_simplex(z, target):
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    result = np.clip(z - alpha, 0.0, 1.0)
    residual = target - result.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            slack = 1.0 - result
            order = np.argsort(-slack)
            for idx in order:
                add = min(slack[idx], residual)
                if add > 0:
                    result[idx] += add
                    residual -= add
                    if abs(residual) < 1e-14:
                        break
        else:
            order = np.argsort(result)
            for idx in order:
                rem = min(result[idx], -residual)
                if rem > 0:
                    result[idx] -= rem
                    residual += rem
                    if abs(residual) < 1e-14:
                        break
    return result


def pair_line_search(h, i, j):
    """Exact 1D search for moving mass between coordinates i and j."""
    hi_val = h[i]
    hj_val = h[j]
    t_min = -hi_val
    t_max = 1.0 - hi_val
    t_min = max(t_min, hj_val - 1.0)
    t_max = min(t_max, hj_val)
    if t_min >= t_max - 1e-12:
        return h, compute_c5(h)
    pad = (t_max - t_min) * 1e-10
    t_min += pad
    t_max -= pad
    if t_min >= t_max:
        return h, compute_c5(h)
    
    n = len(h)
    target = float(n) / 2.0
    
    # Grid evaluation
    n_grid = 24
    ts = np.linspace(t_min, t_max, n_grid)
    best_c5 = float('inf')
    best_t = 0.0
    for t in ts:
        h_candidate = h.copy()
        h_candidate[i] = hi_val + t
        h_candidate[j] = hj_val - t
        c5 = compute_c5(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    # Refine if interior
    if t_min + 1e-10 < best_t < t_max - 1e-10:
        lo = best_t - (t_max - t_min) / n_grid * 1.5
        hi = best_t + (t_max - t_min) / n_grid * 1.5
        lo = max(lo, t_min)
        hi = min(hi, t_max)
        fine_ts = np.linspace(lo, hi, 12)

            h_candidate[i] = hi_val + t

def gradient_guided_search(h0, time_budget, seed=42):
    """Use smoothed-max gradient to select promising pairs, then exact line search."""
    n = len(h0)
    target = float(n) / 2.0
    h = h0.astype(float).copy()
    h = project_simplex(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    alpha = 15.0
    
    while time.time() - t0 < time_budget:
        # Compute smoothed weights
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        shifted = alpha * (corr - max_c)
        np.clip(shifted, -80, 0, out=shifted)
        exp_c = np.exp(shifted)
        w = exp_c / np.sum(exp_c)
        
        # Compute gradient
        grad = np.zeros(n)
        b = 1.0 - h
        for L in range(0, n):
            wi = w[n - 1 + L] * dx
            if wi < 1e-14:
                continue
            if L == 0:
                grad += wi * (b - h)
            else:
                grad[L:] += wi * b[:n - L]
                grad[:n - L] -= wi * h[L:]
        for L in range(1, n):
            wi = w[n - 1 - L] * dx
            if wi < 1e-14:
                continue
            grad[:n - L] += wi * b[L:]
            grad[L:] -= wi * h[:n - L]
        
        # Select top pairs by gradient difference
        n_pairs = min(60, n * (n - 1) // 3)
        # Score each pair by |grad[i] - grad[j]| (potential for improvement)
        # Only consider pairs where both can move
        candidates = []
        top_k = min(40, n)
        for i in range(n):
            if h[i] < 1e-8 or h[i] > 1.0 - 1e-8:
                continue
            for j in np.argsort(-np.abs(grad))[:top_k]:
                if j == i:
                    continue
                if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                    continue
                score = abs(grad[i] - grad[j])
                candidates.append((score, i, j))
        
        candidates.sort(key=lambda x: -x[0])
        
        # Take top N unique pairs
        seen = set()
        selected = []
        for score, i, j in candidates:
            key = (min(i, j), max(i, j))
            if key not in seen:
                seen.add(key)
                selected.append((i, j))
            if len(selected) >= n_pairs:
                break
        
        # Add some random pairs

            i = rng.randint(n)

                best_h = new_h.copy()
            if new_c5 < compute_c5(h) - 1e-14:

def resample_to_n(h_old, n_new):
    """Resample piecewise-constant function to new grid preserving integral."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    new_h = np.zeros(n_new)
    for i in range(n_new):
        left = i * dx_new
        right = (i + 1) * dx_new
        mass = 0.0
        j_start = max(0, int(np.floor(left / dx_old)))
        j_end = min(n_old, int(np.ceil(right / dx_old)))
        for j in range(j_start, j_end):
            cl = j * dx_old
            cr = (j + 1) * dx_old
            ol = max(left, cl)
            orr = min(right, cr)
            if orr > ol:
                mass += h_old[j] * (orr - ol)
        new_h[i] = mass / dx_new
    
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = project_simplex(new_h, float(n_new) / 2.0)
    return new_h


def structured_init(n, seed=0):
    """Create a structured initial guess using golden-ratio spacing."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = n // 2
    if n % 2 == 1:
        h[n // 2] = 0.5
        remaining_ones = n // 2
    else:
        remaining_ones = ones
    
    # Golden ratio spacing
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    for k in range(remaining_ones * 2):
        pos = int((k * (1.0 / phi)) % n)
        pos = pos % n
        if pos not in positions and h[pos] == 0:
            positions.add(pos)
            h[pos] = 1.0
        if len(positions) >= remaining_ones:
            break
    # Fill remaining
    for pos in range(n):
        if len(positions) >= remaining_ones:
            break
        if h[pos] == 0:
            h[pos] = 1.0
            positions.add(pos)
    
    h = h[rng.permutation(n)]
    return project_simplex(h, float(n) / 2.0)

            if n_init - delta >= 20:
                candidate_ns.add(n_init - delta)
            if n_init + delta <= 250:
                candidate_ns.add(n_init + delta)
        candidate_ns.update([31, 47, 63, 81, 99, 127, 163, 199])
        candidate_ns = sorted(candidate_ns)
        
        for n_new in candidate_ns:
            if remaining() < 25:
                break
            alloc = min(remaining() * 0.06, 45.0)
            if alloc < 8:
                break
            
            h_new = resample_to_n(h_init, n_new)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                try_update(h_new, c5_new, n_new)
            
            # Brief optimization
            if remaining() > 15:
                sub_alloc = min(alloc * 0.7, remaining() - 10)
                if sub_alloc > 5:
                    h_opt, c5_opt = gradient_guided_search(h_new, sub_alloc, seed + n_new * 13)
                    try_update(h_opt, c5_opt, n_new)
    
    # Phase 4: More sweeps on best
    if best_h is not None and remaining() > 40:
        alloc = min(remaining() * 0.35, 200.0)
        h_best, c5_best = exhaustive_pair_search(best_h, alloc, seed + 200)
        try_update(h_best, c5_best, best_n)
    
    # Phase 5: Random restarts
    if remaining() > 30:
        rng = np.random.RandomState(seed + 777)
        for n_try in [n_init, n_init - 2, n_init + 2, 63, 81]:
            if remaining() < 15:
                break
            if n_try < 20 or n_try > 250:
                continue
            alloc = min(remaining() * 0.08, 30.0)
            if alloc < 5:
                break
            
            # Start from perturbed best
            h_pert = best_h.copy() if best_h is not None and len(best_h) == n_try else structured_init(n_try, seed + n_try)
            if len(h_pert) == n_try:
                pert = rng.randn(n_try) * 0.03
                pert -= pert.mean()
                h_pert = project_simplex(h_pert + pert, float(n_try) / 2.0)
            
            h_opt, c5_opt = gradient_guided_search(h_pert, alloc, seed + n_try * 31 + 500)
            try_update(h_opt, c5_opt, n_try)
    
    # Phase 6: Final polish
    if best_h is not None and remaining() > 20:
        alloc = min(remaining() * 0.6, 150.0)
        h_best, c5_best = exhaustive_pair_search(best_h, alloc, seed + 999)
        try_update(h_best, c5_best, best_n)
    
    # Fallback
    if best_h is None:
        best_h = h_init.copy()
        best_c5 = compute_c5(best_h)
        best_n = len(best_h)
    
    # Final feasibility check
    best_h = project_simplex(best_h, float(best_n) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, best_n
