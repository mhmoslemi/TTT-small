def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(v, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, beta):
    """Vectorized smooth gradient computation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(beta * (corr - c_max), -500.0, 500.0)
    ev = np.exp(shifted)
    w = ev / np.sum(ev)
    
    grad = w[n-1] * (1.0 - 2.0 * h)
    for L in range(1, n):
        wp = w[n-1+L]
        if wp > 1e-14:
            grad[:n-L] += wp * (1.0 - h[L:])
            grad[L:] -= wp * h[:n-L]
        wn = w[n-1-L]
        if wn > 1e-14:
            grad[L:] += wn * (1.0 - h[:n-L])
            grad[:n-L] -= wn * h[L:]
    return grad * dx

def quick_pair_eval(h, i, j, n_fractions=5):
    """Quick evaluation of a pair transfer at several fractions of max transfer."""
    T = min(1.0 - h[i], h[j])
    if T < 1e-12:
        return 0.0, eval_c5(h)
    base_c5 = eval_c5(h)
    best_delta = 0.0
    best_val = base_c5
    for frac in np.linspace(0.1, 1.0, n_fractions):
        d = T * frac
        h[i] += d
        h[j] -= d
        v = eval_c5(h)
        h[i] -= d
        h[j] += d
        if v < best_val:
            best_val = v
            best_delta = d
    return best_delta, best_val

def refine_pair(h, i, j, d_start, d_end, n_iters=25):
    """Golden section search for pair (i,j) in [d_start, d_end]."""
    if d_end - d_start < 1e-12:
        return (d_start + d_end) / 2
    gr = (np.sqrt(5) - 1) / 2
    a, b = d_start, d_end
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] += c; h[j] -= c; fc = eval_c5(h); h[i] -= c; h[j] += c
    h[i] += d; h[j] -= d; fd = eval_c5(h); h[i] -= d; h[j] += d
    best_d = c if fc < fd else d
    best_v = min(fc, fd)
    for _ in range(n_iters):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            h[i] += c; h[j] -= c; fc = eval_c5(h); h[i] -= c; h[j] += c
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            h[i] += d; h[j] -= d; fd = eval_c5(h); h[i] -= d; h[j] += d
        if fc < best_v:
            best_v = fc; best_d = c
        if fd < best_v:
            best_v = fd; best_d = d
    return best_d, best_v
