                break

                break
            continue

        # Remove mean to ensure sum = 0 (numerical safety)

                break

        else:
            rad *= 0.5
            if rad < 1e-5:

def compute_grad_matrix(h, n):
    """Returns (2n-1) x n gradient matrix for all C_k"""
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):

        s = max(0, -k)

            jj = ii + k
            G[idx, ii] += (1.0 - h[jj]) * dx
            G[idx, jj] += (-h[ii]) * dx
