import numpy as np
from scipy.optimize import minimize, linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad_rows(h):
    """Compute gradients of all overlap functionals w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    num_lags = 2 * n - 1
    grads = np.zeros((num_lags, n))
    for p in range(num_lags):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                grads[p, :L] -= dx * h[s:n]
                grads[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            L = n - t
            if L > 0:
                grads[p, :L] += dx * (1.0 - h[t:n])
                grads[p, t:] -= dx * h[:L]
    return grads

def repair(z):
    """Project z onto {h : 0<=h<=1, sum(h)=n/2}."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    # Binary search for shift
    lo = -2.0
    hi = 2.0
    # Expand bounds if needed
    while np.sum(np.clip(z + lo, 0, 1)) > target:
        lo *= 2
    while np.sum(np.clip(z + hi, 0, 1)) < target:
        hi *= 2
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, values, grads, top_k, step_size):
    """One subgradient step."""
    n = len(h)
    # Select active constraints
    m = min(top_k, len(values))
    idx = np.argsort(values)[-m:]
    w = np.zeros(len(values))
    w[idx] = 1.0 / m
    grad = w @ grads  # weighted gradient
    # Step
    h_new = h - step_size * grad
    return repair(h_new)

def active_set_lp(h, values, grads, top_k, radius):
    """Solve linearized active-set LP. Returns direction or None."""
    n = len(h)
    m = min(top_k, len(values))
    idx = np.argsort(values)[-m:]
    v_active = values[idx]
    J = grads[idx]  # shape (m, n)
    
    # LP: minimize t
    # subject to: J @ d + v_active <= t  =>  J @ d - t <= -v_active  =>  -J @ d + t >= v_active
    #          sum(d) = 0
    #          -radius <= d <= radius
    #          -radius <= t <= radius (bound t)
    #
    # Standard form: minimize c @ x
    # subject to: A_ub @ x <= b_ub
    #            A_eq @ x = b_eq
    #            x_lo <= x <= x_hi
    #
    # x = [d, t] (n+1 variables)
    #
    # Constraints:
    # J @ d - t <= -v_active  (m constraints)
    # sum(d) = 0  (1 equality)
    # -radius <= d_i <= radius
    # -radius <= t <= radius (or 0 <= t, but let's use symmetric)
    
    # Actually, to find a descent direction, I want to minimize t where t >= max of linearized values.
    # Linearized value at lag k: v_k + J_k @ d
    # Want: v_k + J_k @ d <= t for all active k
    # => J_k @ d - t <= -v_k
    
    # Objective: minimize t (i.e., minimize max of linearized values)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = J  # J @ d
    A_ub[:, n] = -1.0  # -t
    b_ub = -v_active  # <= -v_k
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(d) = 0
    b_eq = np.zeros(1)
    
    x_lo = np.concatenate([-radius * np.ones(n), [-radius]])
    x_hi = np.concatenate([radius * np.ones(n), [radius]])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(x_lo, x_hi)), method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            t_val = res.x[n]
            # Check if this is an improvement
            current_max = max(v_active)
            if t_val < current_max - 1e-12 and np.abs(d).max() > 1e-10:
                return d
    except Exception:
        pass
    return None

def line_search(h, direction, max_alpha=2.0):
    """Find best alpha along direction. Returns (best_h, best_score)."""
    current_score = official_score(h)
    best_h = h
    best_s = current_score
    alphas = np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0])
    for a in alphas:
        if a * np.abs(direction).max() > max_alpha:
            break
        h_try = repair(h + a * direction)
        s_try = official_score(h_try)
        if s_try < best_s:
            best_s = s_try
            best_h = h_try
    return best_h, best_s

def smooth_obj(x, beta):
    n = len(x)
    vals = compute_all_overlaps(x)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)  # numerical stability
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_grad(x, beta):
    n = len(x)
    vals = compute_all_overlaps(x)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    grads = compute_grad_rows(x)
    return grads.T @ w  # shape (n,)

def try_slsqp_smooth(h0, beta, maxiter=200):

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad_rows(h):
    n = len(h)
    dx = 2.0 / n
    num_lags = 2 * n - 1
    grads = np.zeros((num_lags, n))
    for p in range(num_lags):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                grads[p, :L] -= dx * h[s:n]
                grads[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            L = n - t
            if L > 0:
                grads[p, :L] += dx * (1.0 - h[t:n])
                grads[p, t:] -= dx * h[:L]
    return grads

    except Exception:
        pass
    return None, None

def active_set_lp(h, values, grads, top_k, radius):
    n = len(h)
    m = min(top_k, len(values))
    idx = np.argsort(values)[-m:]
    v_act = values[idx]
    J = grads[idx]
    c = np.zeros(n + 1); c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = J
    A_ub[:, n] = -1.0
    b_ub = -v_act
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    x_lo = np.concatenate([-radius * np.ones(n), [-radius]])
    x_hi = np.concatenate([radius * np.ones(n), [radius]])
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(x_lo, x_hi)), method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.abs(d).max() > 1e-10:
                return d
    except Exception:
        pass
    return None

def line_search(h, direction, max_alpha=1.5):
    cur_s = official_score(h)
    best_h, best_s = h, cur_s
    alphas = np.array([0.01, 0.02, 0.04, 0.08, 0.15, 0.25, 0.4, 0.6, 0.9, 1.2])
    max_d = np.abs(direction).max()
    for a in alphas:
        if a * max_d > max_alpha: break
        h_try = repair(h + a * direction)
        s_try = official_score(h_try)
        if s_try < best_s - 1e-14:
            best_s = s_try
            best_h = h_try
    return best_h, best_s

def proj_grad_step(h, vals, grads, top_k, step, momentum_vec, mu):
    n = len(h)
    m = min(top_k, len(vals))
    idx = np.argsort(vals)[-m:]
    w = np.zeros(len(vals)); w[idx] = 1.0 / m
    g = grads.T @ w
    momentum_vec[:] = mu * momentum_vec + g
    h_new = h - step * momentum_vec
    return repair(h_new), momentum_vec

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return repair(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = repair(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
        h_init = 0.5 + 0.28 * np.sin(np.pi * x) + 0.08 * np.sin(2 * np.pi * x)
        best_h = repair(h_init)
        best_score = official_score(best_h)

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-13:
            best_h = h.copy()
            best_score = sc

    update(best_h, best_score)

    def time_left():
        return deadline - time.time()

    # Phase A: Projected gradient with momentum (fast, many iterations)
    if time_left() > 50:
        n = len(best_h)
        h_curr = best_h.copy()
        momentum = np.zeros(n)
        mu = 0.7
        phase_configs = [
            (30, 0.003, 500),
            (20, 0.005, 800),
            (15, 0.003, 1000),
            (10, 0.002, 1200),
            (7, 0.001, 1500),
            (5, 0.0008, 2000),
            (3, 0.0004, 2500),
        ]
        for top_k, step0, iters in phase_configs:
            if time_left() < 10: break
            for it in range(iters):
                if time_left() < 5: break
                vals = compute_all_overlaps(h_curr)
                grads = compute_grad_rows(h_curr)
                step = step0 * max(0.2, 1.0 - it / iters)
                h_new, momentum = proj_grad_step(h_curr, vals, grads, top_k, step, momentum, mu)
                s_new = official_score(h_new)
                s_old = official_score(h_curr)
                if s_new < s_old - 1e-13:
                    h_curr = h_new
                    if s_new < best_score - 1e-13:
                        update(h_curr, s_new)
                elif s_new < s_old + 1e-12:
                    # Accept small worsening for exploration
                    if rng.random() < 0.3:
                        h_curr = h_new
                else:
                    step *= 0.3
                    h_new, momentum = proj_grad_step(h_curr, vals, grads, top_k, step, momentum, mu)
                    s_new2 = official_score(h_new)
                    if s_new2 < s_old - 1e-13:
                        h_curr = h_new
                        if s_new2 < best_score - 1e-13:
                            update(h_curr, s_new2)
                    momentum *= 0.5

    # Phase B: SLSQP smooth optimization
    if time_left() > 60:
        for beta in [100, 500, 2000, 8000, 30000]:
            if time_left() < 30: break
            h_new, sc = try_slsqp_smooth(best_h, beta, maxiter=250)
            if h_new is not None and sc < best_score - 1e-13:
                update(h_new, sc)

    # Phase C: Active-set LP with line search
    if time_left() > 40:
        h_curr = best_h.copy()
        max_iter = 300
        for iteration in range(max_iter):
            if time_left() < 10: break
            vals = compute_all_overlaps(h_curr)
            grads = compute_grad_rows(h_curr)
            cur_s = official_score(h_curr)
            found = False
            for tk in [60, 40, 30, 20]:
                if time_left() < 5: break
                for rad in [0.08, 0.04, 0.02, 0.01, 0.005]:
                    if time_left() < 3: break
                    d = active_set_lp(h_curr, vals, grads, tk, rad)
                    if d is not None:
                        h_new, s_new = line_search(h_curr, d)
                        if s_new < cur_s - 1e-13:
                            h_curr = h_new
                            if s_new < best_score - 1e-13:
                                update(h_curr, s_new)
                            found = True
                            break
                if found: break
            if not found:
                # Escape: random perturbation
                sigma = 0.015 + 0.03 * rng.random()
                h_try = repair(h_curr + rng.normal(0, sigma, len(h_curr)))
                s_try = official_score(h_try)
                if s_try < cur_s:
                    h_curr = h_try
                    if s_try < best_score - 1e-13:
                        update(h_curr, s_try)
                else:
                    # Try smaller perturbation
                    h_try = repair(h_curr + rng.normal(0, 0.005, len(h_curr)))
                    s_try = official_score(h_try)
                    if s_try < cur_s:
                        h_curr = h_try
                        if s_try < best_score - 1e-13:
                            update(h_curr, s_try)
                    else:
                        break

    # Phase D: SLSQP direct minimax
    if time_left() > 60:
        h_new, sc = try_slsqp_direct(best_h, maxiter=350)
        if h_new is not None and sc < best_score - 1e-13:
            update(h_new, sc)

    # Phase E: Fine project
