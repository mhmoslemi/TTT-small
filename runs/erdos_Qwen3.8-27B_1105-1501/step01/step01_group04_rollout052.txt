            if 0 <= i - k < n:

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target):
    lo_t = float(h.min()) - 1.0
    hi_t = float(h.max()) + 1.0
    for _ in range(50):

        if s > target:
            lo_t = mid
        else:
            hi_t = mid
    return np.clip(h - (lo_t + hi_t) / 2.0, 0.0, 1.0)

def smooth_max_grad_vec(h, alpha):
    """Vectorized smoothed max and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    max_val = float(np.max(vals))
    
    shifted = vals - max_val
    # For numerical stability
    if alpha * (-shifted).max() < 700:
        exp_vals = np.exp(alpha * shifted)
        p = exp_vals / np.sum(exp_vals)
    else:
        threshold = max_val - 3.0 / alpha
        mask = vals >= threshold
        count = np.sum(mask)
        p = np.where(mask, 1.0 / count, 0.0)
    
    A = np.correlate(1.0 - h, p, mode='full')[n-1 : 2*n-1]
    B = np.convolve(h, p, mode='full')[n-1 : 2*n-1]
    grad = dx * (A - B)
    return max_val, grad

def local_search(h0, time_budget, seed=42):
    """Efficient local search using pair moves and correlation updates."""
    n = len(h0)
    dx = 2.0 / n
    target = float(n) / 2.0
    h = h0.copy()
    corr = compute_corr(h)
    vals = corr * dx
    best_c5 = float(np.max(vals))
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Precompute which lags are "active" (near max)
    for iteration in range(100000):
        if time.time() - t0 >= time_budget:
            break
        
        # Find current max
        max_val = float(np.max(vals))
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        # Choose move strategy
        if rng.random() < 0.7:
            # Pair move: change h[i] by +delta, h[j] by -delta
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.choice([0.01, 0.02, 0.05, 0.1]) * rng.choice([-1, 1])
        else:
            # Single coordinate with adjustment elsewhere
            i = rng.randint(n)
            delta = rng.choice([0.01, 0.02, 0.05]) * rng.choice([-1, 1])
            j = i  # will handle as pair with self-cancel
        
        if i == j:
            continue
        
        if h[i] + delta < -1e-10 or h[i] + delta > 1.0 + 1e-10:
            continue
        if h[j] - delta < -1e-10 or h[j] - delta > 1.0 + 1e-10:
            continue
        
        # Compute change in correlation
        # Δcorr[m] = delta*(1-h[i+k]) - h[i-k]*delta - delta*(1-h[j+k]) + h[j-k]*delta
        # where k = m - n + 1
        # = delta * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
        
        # For each m, k = m-n+1
        # Valid ranges: 0≤i+k<n for first term, 0≤i-k<n for second, etc.
        
        # This is O(n) per move
        k_range = np.arange(-(n-1), n)
        m_indices = n - 1 + k_range
        
        delta_corr = np.zeros_like(corr)
        
        # Contribution from i (positive delta)
        valid_ik = (i + k_range >= 0) & (i + k_range < n)
        valid_ik_m = (i - k_range >= 0) & (i - k_range < n)
        delta_corr[m_indices[valid_ik]] += delta * (1.0 - h[i + k_range[valid_ik]])
        delta_corr[m_indices[valid_ik_m]] -= delta * h[i - k_range[valid_ik_m]]
        
        # Contribution from j (negative delta)
        valid_jk = (j + k_range >= 0) & (j + k_range < n)
        valid_jk_m = (j - k_range >= 0) & (j - k_range < n)
        delta_corr[m_indices[valid_jk]] -= delta * (1.0 - h[j + k_range[valid_jk]])
        delta_corr[m_indices[valid_jk_m]] += delta * h[j - k_range[valid_jk_m]]
        
        new_corr = corr + delta_corr
        new_max = float(np.max(new_corr * dx))
        
        if new_max < max_val - 1e-12:
            h[i] += delta
            h[j] -= delta
            corr = new_corr
            vals = corr * dx
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    lo_t = float(h.min()) - 1.0
    hi_t = float(h.max()) + 1.0
    for _ in range(50):
        mid = (lo_t + hi_t) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo_t = mid
        else:
            hi_t = mid
    return np.clip(h - (lo_t + hi_t) / 2.0, 0.0, 1.0)

def smooth_max_grad_vec(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    max_val = float(np.max(vals))
    shifted = vals - max_val
    max_neg = float(-shifted.min())
    if alpha * max_neg < 700:
        exp_vals = np.exp(alpha * shifted)
        p = exp_vals / np.sum(exp_vals)
    else:
        threshold = max_val - 3.0 / alpha
        mask = vals >= threshold
        count = max(float(np.sum(mask)), 1.0)
        p = np.where(mask, 1.0 / count, 0.0)
    A = np.correlate(1.0 - h, p, mode='full')[n-1:2*n-1]
    B = np.convolve(h, p, mode='full')[n-1:2*n-1]
    grad = dx * (A - B)
    return max_val, grad

def optimize_smoothed(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = np.clip(h0.astype(float).copy(), 0.0, 1.0)
    h = project_to_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    alpha = 20.0
    lr = 0.005
    momentum = 0.95
    v = np.zeros(n)
    stall_count = 0
    while time.time() - t0 < time_budget:
        batch = 50
        for _ in range(batch):
            if time.time() - t0 >= time_budget:
                break
            max_val, grad = smooth_max_grad_vec(h, alpha)
            v = momentum * v - lr * grad
            h_new = project_to_sum(h + v, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                stall_count = 0
            else:
                stall_count += 1
            h = h_new
        if stall_count > 100:
            rng = np.random.RandomState(seed + stall_count)
            h = best_h + 0.02 * rng.randn(n)
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target)
            v = np.zeros(n)
            stall_count = 0
            lr = max(lr * 0.6, 0.0003)
            alpha = min(alpha * 2.0, 1000.0)
        alpha = min(alpha * 1.02 + 0.1, 500.0)
        lr = max(lr * 0.995, 0.0002)
    return best_h, best_c5

def local_search(h0, time_budget, seed=42):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    best_c5 = float(np.max(vals))
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed)
    k_range = np.arange(-(n-1), n)
    m_indices = n - 1 + k_range
    i_vals = np.arange(n)
    
    while time.time() - t0 < time_budget:
        max_val = float(np.max(vals))
        if max_val < best_c5 - 1e-15:
            best_c5 = max_val
            best_h = h.copy()
        
        # Pick active lag
        active = vals >= max_val - 0.002

            active = vals == max_val
        active_idx = np.where(active)[0]
        m_star = active_idx[rng.randint(len(active_idx))]
        k_star = int(m_star - n + 1)
        
        # Compute gradient for this lag
        ip = i_vals + k_star
        im = i_vals - k_star
        grad_k = np.zeros(n)

        mm = (im >= 0) & (im < n)
        grad_k[mp] += (1.0 - h[ip[mp]])
        grad_k[mm] -= h[im[mm]]
        
        # Find best pair move

        dec_candidates = np.where((grad_k > 0.01) & (h > 0.01))[0]
        
        if len(inc_candidates) == 0 or len(dec_candidates) == 0:

        else:
            i = inc_candidates[rng.randint(len(inc_candidates))]
            j = dec_candidates[rng.randint(len(dec_candidates))]

            vm = (i - k_range >= 0) & (i - k_range < n)
            delta_corr[m_indices[vp]] += delta * (1.0 - h[i + k_range[vp]])
            delta_corr[m_indices[vm]] -= delta * h[i - k_range[vm]]
            vp = (j + k_range >= 0) & (j + k_range < n)

            delta_corr[m_indices[vp]] -= delta * (1.0 - h[j + k_range[vp]])
            delta_corr[m_indices[vm]] += delta * h[j - k_range[vm]]
            
            new_vals = (corr + delta_corr) * dx
            if float(np.max(new_vals)) < max_val:
                h[i] += delta
                h[j] -= delta
                corr += delta_corr
                vals = new_vals
                break
    
    return best_h, best_c5
