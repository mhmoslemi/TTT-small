        valid2 = (m + k >= 0) & (m + k < n)
        contrib1 = np.where(valid1, 1 - h[m - k], 0.0)

        effect[m] = (contrib1 + contrib2) * dx

    # new_lags[k] = lags[k] + delta[j, k]
    
    for idx_i in range(len(ones)):
        i = ones[idx_i]
        # delta shape: (n_zeros, 2n-1)
        delta = -effect[i][None, :] + effect[zeros][:, :]
        new_lags = lags[None, :] + delta  # (n_zeros, 2n-1)

        if maxes[min_idx] < best_c - 1e-15:
            best_c = maxes[min_idx]
            best_i = i
            best_j = zeros[min_idx]
    
    if best_i is not None:
        h[best_i] = 0.0
        h[best_j] = 1.0
        return h, best_c, True

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid

    diff = target - np.sum(z)
    if abs(diff) > 1e-13:

            add = min(diff, 1.0 - z[idx])
            z[idx] += add

                z[idx2] += min(rem, 1.0 - z[idx2])

            sub = min(-diff, z[idx])

def to_binary(h):

    target = int(n / 2)

    # Then adjust to get exactly target ones
    b = (h > 0.5).astype(float)
    n_ones = int(b.sum())
    if n_ones == target:
        return b
    elif n_ones > target:
        # Need to flip some 1s to 0s: pick the smallest h values among 1s
        one_positions = np.where(b == 1)[0]
        # Sort by h value ascending (least confident 1s first)
        order = np.argsort(h[one_positions])
        flip_count = n_ones - target
        for i in range(flip_count):
            b[one_positions[order[i]]] = 0
    else:
        # Need to flip some 0s to 1s: pick the largest h values among 0s
        zero_positions = np.where(b == 0)[0]
        order = np.argsort(-h[zero_positions])
        flip_count = target - n_ones
        for i in range(flip_count):
            b[zero_positions[order[i]]] = 1
    return b

def compute_effect(h):
    """Compute effect matrix: effect[m, k_idx] = change in lag k_idx when h[m] goes 0->1."""
    n = len(h)
    dx = 2.0 / n
    effect = np.zeros((n, 2*n-1))
    k = np.arange(-(n-1), n)  # actual lags
    for m in range(n):

        valid2 = (m + k >= 0) & (m + k < n)
        contrib1 = np.where(valid1, 1.0 - h[m - k], 0.0)
        contrib2 = np.where(valid2, -h[m + k], 0.0)
        effect[m] = (contrib1 + contrib2) * dx
    return effect

            g[m] = (grad1 + grad2).sum()

                    best_c = cur_c

        else:

        T = max(0.001, T * 0.999)

After flipping position i from 1 to 0 and position j from 0 to 1:
new_lags = old_lags - effect[i] + effect[j]

Where effect[m] is the change when position m goes from 0 to 1. Since we're going 1→0 at i, the change is -effect[i]. And 0→1 at j gives +effect[j].

But wait, the effect matrix was computed for the CURRENT state of h. After the flip, h changes, so the effect matrix for the new state is different. However, for computing the new lags from the old lags, we only need the OLD effect matrix (since the change from old to new state is exactly -effect_old[i] + effect_old[j]).

Then for the next iteration, we need to recompute the effect matrix for the new state. Alternatively, we can update it incrementally.

The effect of flipping h[i] and h[j] on the effect matrix:
- effect[m, k] depends on h[m-k] and h[m+k]
- If i or j changes, it affects effect[m, k] for m where m-k=i or m+k=i (or j)

This is getting complex. Let me just recompute the effect matrix each time. For n=70, it's O(n * 2n) = O(n^2) ≈ 10K operations, very fast.

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    target = int(n / 2)
    order = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def compute_effect(h):
    n = len(h)
    dx = 2.0 / n
    effect = np.zeros((n, 2*n-1))
    k = np.arange(-(n-1), n)
    for m in range(n):
        valid1 = (m - k >= 0) & (m - k < n)
        valid2 = (m + k >= 0) & (m + k < n)
        contrib1 = np.where(valid1, 1.0 - h[m - k], 0.0)
        contrib2 = np.where(valid2, -h[m + k], 0.0)
        effect[m] = (contrib1 + contrib2) * dx
    return effect

def pair_flip_best(h, lags, effect):
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return (-1, -1), float(lags.max())
    best_c = float(lags.max())
    best_i, best_j = -1, -1
    n = len(h)
    for ii in range(len(ones)):
        i = int(ones[ii])
        delta = -effect[i][None, :] + effect[zeros][:, :]
        new_lags = lags[None, :] + delta
        maxes = new_lags.max(axis=1)
        min_idx = int(np.argmin(maxes))
        if maxes[min_idx] < best_c - 1e-15:
            best_c = float(maxes[min_idx])
            best_i = i
            best_j = int(zeros[min_idx])
    return (best_i, best_j), best_c

def continuous_refine(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    cur = make_feasible(h.copy())
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = 0.05
    lr = 0.03
    k_arr = np.arange(-(n-1), n)
    
    for step in range(5000):
        if time.time() > deadline:
            break
        lags = all_lags(cur)
        max_val = float(lags.max())
        ex = np.exp((lags - max_val) / T)
        S = float(ex.sum())
        w = ex / S
        
        g = np.zeros(n)
        for m in range(n):
            v1 = (m - k_arr >= 0) & (m - k_arr < n)
            v2 = (m + k_arr >= 0) & (m + k_arr < n)
            t1 = np.where(v1, w * (1.0 - cur[m - k_arr]), 0.0)
            t2 = np.where(v2, -w * cur[m + k_arr], 0.0)
            g[m] = float((t1 + t2).sum())
        
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(0.001, T * 0.7)
            continue
        
        direction = -g / gn
        improved = False
        for s in (lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05):

            c_new = score_exact(cand)

                cur = cand
                cur_c = c_new
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                improved = True
                break

                # Random restart
                cur = make_feasible(best + rng.normal(0, 0.02, n))

                lr = 0.03
            T = max(0.001, T * 0.9)

    return best, best_c

def resample_func(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    if n_new < n_old:
        ratio = n_old / n_new
        h_new = np.zeros(n_new)
        for i in range(n_new):
            start = int(i * ratio)
            end = min(int((i+1) * ratio), n_old)
            if end > start:
                h_new[i] = h_old[start:end].mean()
        return make_feasible(h_new, n_new)
    else:
        x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
        x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
        h_new = np.interp(x_new, x_old, h_old)
        return make_feasible(h_new, n_new)
