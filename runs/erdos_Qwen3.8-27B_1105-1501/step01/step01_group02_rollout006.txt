def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute residual
        n = len(result)
        per = residual / n
        result = np.clip(result + per, 0.0, 1.0)
    return result
