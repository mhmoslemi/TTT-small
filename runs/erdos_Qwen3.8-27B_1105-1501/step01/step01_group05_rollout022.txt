import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(v, n):
    """Project v to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.array(v, dtype=float).ravel()[:n].copy()
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        res = target - s
        if abs(res) < 1e-14:
            break
        if res > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            if room[idx] < 1e-15:
                break
            h[idx] += min(res, room[idx])
        else:
            idx = np.argmax(h)
            if h[idx] < 1e-15:
                break
            h[idx] -= min(-res, h[idx])
    # Final fix
    res = target - np.sum(h)
    if abs(res) > 1e-15:
        if res > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += res
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(h, k):
    """Gradient of C_k with respect to h, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    p = np.arange(n)
    # For C_k = dx * sum_i h[i+k]*(1-h[i]) over valid i
    # d/dh[m] C_k:
    # m appears as i+k when i = m-k: contributes dx*(1-h[m-k]) if valid
    # m appears as i when i = m: contributes -dx*h[m+k] if valid
    ok1 = (p - k) >= 0
    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g * dx

def compute_all_grads(h):
    """Compute gradient matrix: grad[q_idx, i] = d(C_q)/d(h[i]) * dx"""
    n = len(h)
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    grad = np.zeros((2*n-1, n))
    for qi, k in enumerate(lags):
        grad[qi] = corr_grad(h, k)
    return grad, lags

def pair_search(h, i, j, grid_n=25, golden_iters=15):
    """Find optimal x for pair (i,j) where h[i]=x, h[j]=S-x."""
    n = len(h)
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-10:
        return h, official_score(h)
    
    def eval_at(x):
        h2 = h.copy()
        h2[i] = x
        h2[j] = S - x
        return official_score(h2)
    
    # Grid search
    grid = np.linspace(L, U, grid_n)
    vals = np.array([eval_at(x) for x in grid])
    best_idx = np.argmin(vals)
    best_val = vals[best_idx]
    best_x = grid[best_idx]
    
    # Golden section refinement in neighborhood
    # Find the interval containing the minimum
    if best_idx == 0:
        lo, hi = grid[0], grid[1]
    elif best_idx == grid_n - 1:
        lo, hi = grid[-2], grid[-1]
    else:
        lo, hi = grid[best_idx - 1], grid[best_idx + 1]
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_at(c)
    fd = eval_at(d)
    
    for _ in range(golden_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_at(d)
    
    x_mid = (lo + hi) / 2
    v_mid = eval_at(x_mid)
    
    if v_mid < best_val:
        best_val = v_mid
        best_x = x_mid
    
    h_new = h.copy()
    h_new[i] = best_x
    h_new[j] = S - best_x
    # Project to ensure feasibility (should already be feasible)
    h_new = project(h_new, n)
    final_score = official_score(h_new)
    return h_new, final_score

def select_pairs(h, top_count=100):

    n = len(h)
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    
    corr = np.correlate(h, 1.0-h, mode="full") * dx
    theta = np.max(corr)
    
    # Compute gradient of the max (using active constraints)
    # Use softmax weighting over active constraints
    T = max(0.003, 0.01 * theta)
    w = np.exp((corr - theta) / T)
    w /= np.sum(w)
    
    # Composite gradient
    g = np.zeros(n)
    for qi, k in enumerate(lags):
        if w[qi] > 1e-6:
            g += w[qi] * corr_grad(h, k)
    g -= np.mean(g)
    
    # Score pairs by |g[i] - g[j]|
    # For efficiency, use the top coordinates
    g_abs = np.abs(g)
    top_idx = np.argsort(g_abs)[-min(40, n):]
    
    pairs = []
    seen = set()
    for a in range(len(top_idx)):
        for b in range(a+1, len(top_idx)):
            i, j = top_idx[a], top_idx[b]
            key = (int(i), int(j)) if i < j else (int(j), int(i))
            if key not in seen:
                seen.add(key)
                pairs.append((abs(g[i] - g[j]), i, j))
    
    # Add some random pairs

        j = rng.integers(0, n)

            key = (int(i), int(j)) if i < j else (int(j), int(i))
            if key not in seen:
                seen.add(key)
                pairs.append((abs(g[i] - g[j]), i, j))
    
    pairs.sort(key=lambda x: -x[0])
    return pairs[:top_count]

def slsqp_refine(h, n, time_limit=20.0):
    """SLSQP refinement with active constraints."""
    start_time = time.perf_counter()
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    
    corr = np.correlate(h, 1.0-h, mode="full") * dx
    theta = np.max(corr)
    
    # Select top M active constraints
    M = min(12, 2*n-1)
    active_idx = np.argsort(corr)[-M:]
    active_lags = lags[active_idx]
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        th = x[n]
        corr_x = np.correlate(h_x, 1.0-h_x, mode="full") * dx
        return th - corr_x[active_idx]
    
    def ineq_jac(x):
        h_x = x[:n]
        j = np.zeros((M, n + 1))
        j[:, n] = 1.0
        for mi, qi in enumerate(active_idx):
            j[mi, :n] = -corr_grad(h_x, lags[qi])
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    theta0 = theta + 1e-5
    x0 = np.concatenate([h, [theta0]])
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project(result.x[:n], n)
            s = official_score(h_out)
            s0 = official_score(h)
            if s < s0 - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, official_score(h)

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project(h, n)

def corr_grad(h, k):

    p = np.arange(n)

    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g * dx

def pair_descent(h, n, max_passes=20, time_limit=60.0, seed=42):
    start = time.perf_counter()
    h = project(h, n)
    score = official_score(h)
    rng = np.random.default_rng(seed)
    stagnant = 0
    for p in range(max_passes):
        if time.perf_counter() - start > time_limit:
            break
        pairs = select_pairs(h, n, top_count=130, seed=seed + p)
        pass_improved = False
        for idx, (_, i, j) in enumerate(pairs):
            if time.perf_counter() - start > time_limit:
                break
            if idx % 20 == 0:
                score = official_score(h)
            h_new, s_new = pair_search(h, i, j, grid_n=19, golden_iters=11)
            if s_new < score - 1e-12:
                h = h_new
                score = s_new
                pass_improved = True
        for _ in range(25):
            if time.perf_counter() - start > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            h_new, s_new = pair_search(h, i, j, grid_n=15, golden_iters=9)
            if s_new < score - 1e-12:
                h = h_new
                score = s_new
                pass_improved = True
        if pass_improved:
            stagnant = 0
        else:
            stagnant += 1
            pert = h + rng.normal(0, 0.03, n)
            h_p = project(pert, n)
            s_p = official_score(h_p)
            if s_p < score:
                h = h_p
                score = s_p
                stagnant = 0
            elif stagnant >= 3:
                break
    h = project(h, n)
    return h, official_score(h)


def smooth_descent(h, n, time_limit=15.0, seed=0):
    start = time.perf_counter()
    rng = np.random.default_rng(seed)
    h = project(h, n)
    score = official_score(h)
    for _ in range(8):
        if time.perf_counter() - start > time_limit:
            break
        g, theta, corr, lags = compute_composite_grad(h)
        norm = np.max(np.abs(g))
        if norm < 1e-12:
            break
        direction = -g / norm
        direction += rng.normal(0, 0.08, n)
        direction -= np.mean(direction)
        best_alpha = 0.0
        best_s = score
        for a in [0.01, 0.02, 0.04, 0.08, 0.15, 0.25, 0.4]:
            for sgn in [1, -1]:
                h_new = project(h + sgn * a * direction, n)
                sv = official_score(h_new)
                if sv < best_s - 1e-12:
                    best_s = sv
                    best_alpha = sgn * a
        if best_alpha != 0.0:
            h = project(h + best_alpha * direction, n)
            score = official_score(h)
        else:
            break
    return h, official_score(h)


def slsqp_refine(h, n, time_limit=25.0):
    start = time.perf_counter()
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    theta = float(np.max(corr))
    M = min(14, 2 * n - 1)
    active_idx = np.argsort(corr)[-M:]

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:n]
        th = x[n]
        cx = np.correlate(hx, 1.0 - hx, mode="full") * dx
        return th - cx[active_idx]

    def ineq_jac(x):
        hx = x[:n]
        j = np.zeros((M, n + 1))
        j[:, n] = 1.0
        for mi, qi in enumerate(active_idx):
            j[mi, :n] = -corr_grad(hx, lags[qi])
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    x0 = np.concatenate([h, [theta + 1e-5]])
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project(result.x[:n], n)
            s = official_score(h_out)
            s0 = official_score(h)
            if s < s0 - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, official_score(h)


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return project(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project(h, n)


def gen_quantized(h, n):
    m = len(h)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = (np.arange(n) + 0.5) * 2.0 / n
    h_interp = np.interp(xt, xs, h)
    h_q = np.where(h_interp > 0.6, 1.0, np.where(h_interp < 0.4, 0.0, 0.5))
    return project(h_q, n)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 12.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n0 = len(parent)
        best_h = project(parent, n0)
    else:
        n0 = 65
        best_h = project(np.full(n0, 0.5), n0)

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    candidates = [(best_score, best_h.copy())]
    for sym in [best_h[::-1], 1.0 - best_h, 1.0 - best_h[::-1]]:
        hs = project(sym, n0)
        candidates.append((official_score(hs), hs))

    n_list = sorted(set([max(32, n0 - 10), max(32, n0 - 5), n0, n0 + 5, n0 + 10, n0 + 15]))
    for nt in n_list:
        if time.perf_counter() > deadline - 180:
            break
        for off in [0, 1, 2]:
            hi = gen_interpolated(best_h, nt, off)
            candidates.append((official_score(hi), hi))
        if nt != n0:
            hq = gen_quantized(best_h, nt)
            candidates.append((official_score(hq), hq))

    candidates.sort(key=lambda x: x[0])
    num_top = min(10, len(candidates))

    for idx in range(num_top):
        if time.perf_counter() > deadline - 60:
            break
        sc, hc = candidates[idx]
        nc = len(hc)
        remaining = deadline - time.perf_counter()
        if remaining < 20:
            break
        t_alloc = min(150.0, max(20.0, remaining / max(1, num_top - idx) * 0.55))

        hp, sp = pair_descent(hc, nc, max_passes=18, time_limit=t_alloc, seed=seed + idx * 17)
        if sp < best_score:
            best_score = sp
            best_h = hp.copy()
            best_n = nc

        if time.perf_counter() > deadline - 40:
            break
        hs2, ss2 = smooth_descent(hp, nc, time_limit=min(15, t_alloc * 0.25), seed=seed + 200 + idx)
        if ss2 < best_score:
            best_score = ss2
            best_h = hs2.copy()
            best_n = nc

        if time.perf_counter() > deadline - 30:
            break
        hp2, sp2 = pair_descent(hs2, nc, max_passes=10, time_limit=min(25, t_alloc * 0.35), seed=seed + 400 + idx)
        if sp2 < best_score:
            best_score = sp2
            best_h = hp2.copy()
            best_n = nc

    # Second round: interpolate current best and optimize
    if time.perf_counter() < deadline - 50:
        for nt in [best_n, best_n + 5, best_n - 5]:
            if time.perf_counter() > deadline - 30:
                break
            if nt < 32 or nt > 120:
                continue
            hi = gen_interpolated(best_h, nt)
            hp, sp = pair_descent(hi, nt, max_passes=12, time_limit=min(40, (deadline - time.perf_counter()) * 0.3), seed=seed + 600)
            if sp < best_score:
                best_score = sp
                best_h = hp.copy()
                best_n = nt
            hs, ss = smooth_descent(hp, nt, time_limit=10, seed=seed + 700)
            if ss < best_score:
                best_score = ss
                best_h = hs.copy()
                best_n = nt
            hp2, sp2 = pair_descent(hs, nt, max_passes=8, time_limit=min(20, (deadline - time.perf_counter()) * 0.2), seed=seed + 800)
            if sp2 < best_score:
                best_score = sp2
                best_h = hp2.copy()
                best_n = nt

    # SLSQP polish
    if time.perf_counter() < deadline - 25:
        href, sref = slsqp_refine(best_h, best_n, time_limit=20.0)
        if sref < best_score:
            best_score = sref
            best_h = href.copy()

    # Final pair pass
    if time.perf_counter() < deadline - 10:
        hfin, sfin = pair_descent(best_h, best_n, max_passes=6, time_limit=min(15, max(5, deadline - time.perf_counter() - 8)), seed=seed + 999)
        if sfin < best_score:
            best_score = sfin
            best_h = hfin.copy()

    # Final validation
    best_h = project(best_h, len(best_h))
    final_score = official_score(best_h)
    return (best_h, float(final_score), int(len(best_h)))
