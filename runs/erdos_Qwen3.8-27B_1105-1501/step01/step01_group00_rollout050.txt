    """Try to improve h by exchanging mass between positions i and j."""
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return None

    # Include endpoints and some interior points
    points = set()
    points.add(0.0)

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

    return np.sum(x) - len(x) / 2.0

    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    npts = 2 * n - 1
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(npts) / npts
    else:
        w /= ws
    grad = np.zeros(n)
    for p in range(npts):
        s = p - (n - 1)
        if s >= 0:
            for j in range(n - s):
                grad[j] += w[p] * (1.0 - x[j + s])
                grad[j + s] -= w[p] * x[j]
        else:
            t = -s
            for j in range(n - t):
                grad[j + t] += w[p] * (1.0 - x[j])
                grad[j] -= w[p] * x[j + t]
    return dx * grad

def try_slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None
