import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(500):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient_for_lag(h, k, n):
    """Compute ∂c_k/∂h_i for the k-th lag correlation."""
    g = np.zeros(n)
    p = np.arange(n)
    ok1 = (p + k) < n
    g[ok1] += (1.0 - h[p[ok1] + k])
    ok2 = (p - k) >= 0
    g[ok2] += (-h[p[ok2] - k])
    return g

def logsumexp_gradient(h, theta):
    """Compute gradient of logsumexp(theta * corr * dx) / theta with respect to h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    lags = np.arange(-(n-1), n)
    scaled = theta * corr * dx
    
    # Numerical stability
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp  # softmax weights
    
    grad = np.zeros(n)
    for ki in range(len(lags)):
        if weights[ki] < 1e-10:
            continue
        k = int(lags[ki])
        grad += weights[ki] * dx * compute_gradient_for_lag(h, k, n)
    
    return grad / theta * theta  # The gradient of (1/theta)*log(sum exp(theta*x)) is sum softmax * grad x
    # Actually: d/dh [(1/theta) * log(sum_k exp(theta * c_k * dx))]
    # = (1/theta) * sum_k (theta * dx * ∂c_k/∂h * exp(theta*c_k*dx)) / sum_j exp(theta*c_j*dx)
    # = sum_k (dx * ∂c_k/∂h * softmax_k)
    # So the /theta and *theta cancel. Good.

def logsumexp_value(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = theta * corr * dx
    max_scaled = np.max(scaled)
    return (max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))) / theta

def project_and_repair(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    residual = target - np.sum(h)
    if abs(residual) < 1e-14:
        return h
    # Distribute residual proportionally to available room
    if residual > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-14:
            h += residual * room / total_room
        else:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
    else:
        total_h = np.sum(h)
        if total_h > 1e-14:
            h += residual * h / total_h
        else:
            idx = np.argmax(h)
            h[idx] += residual
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            room = 1.0 - h
            idx = np.argsort(room)[-int(np.ceil(abs(residual)))]
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argsort(h)[int(np.ceil(abs(residual)))]
            h[idx] = max(0.0, h[idx] + residual)
    return h

def lse_optimize(h, n, theta=50.0, max_iters=200, lr=0.05, momentum=0.9, time_limit=30.0):
    """Optimize using logsumexp smoothed objective with projected gradient + momentum."""
    start_time = time.perf_counter()
    h = h.copy()
    score = official_score(h)
    v = np.zeros(n)  # momentum
    
    for it in range(max_iters):
        if time.perf_counter() - start_time > time_limit:
            break
        
        grad = logsumexp_gradient(h, theta)
        # Remove mean component (constraint: sum = n/2)
        grad = grad - np.mean(grad)
        
        v = momentum * v - lr * grad
        h_new = project_and_repair(h + v, n)
        s_new = official_score(h_new)
        
        if s_new < score:
            h = h_new
            score = s_new
            # Decay momentum on success
            v *= 0.5
        else:
            # Line search
            v = 0.5 * v
            found = False
            for attempt in range(5):
                h_try = project_and_repair(h + v, n)
                s_try = official_score(h_try)
                if s_try < score:
                    h = h_try
                    score = s_try
                    found = True
                    break
                v *= 0.5
            if not found:
                # Increase theta for sharper approximation
                theta *= 1.5
                if theta > 500:
                    break
    
    return h, official_score(h)

def slsqp_refine(h, n, time_limit=20.0):
    """SLSQP refinement for small n."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            k_int = int(k)
            j[ki, :n] = -dx * compute_gradient_for_lag(h_x, k_int, n)
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(500):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, k, n):
    g = np.zeros(n)
    p = np.arange(n)
    ok1 = (p + k) < n
    g[ok1] += (1.0 - h[p[ok1] + k])
    ok2 = (p - k) >= 0
    g[ok2] += (-h[p[ok2] - k])
    return g


def lse_gradient(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    lags = np.arange(-(n - 1), n)
    scaled = theta * corr * dx
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for ki in range(len(lags)):
        if weights[ki] < 1e-8:
            continue
        grad += weights[ki] * dx * grad_for_lag(h, int(lags[ki]), n)
    return grad


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    residual = target - np.sum(h)
    if abs(residual) < 1e-13:
        return h
    if residual > 0:
        room = 1.0 - h
        tr = np.sum(room)
        if tr > 1e-14:
            h += residual * room / tr
    else:
        th = np.sum(h)
        if th > 1e-14:
            h += residual * h / th
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + residual)
    return h


def lse_optimize(h, n, theta0=50.0, max_iters=200, lr0=0.04, time_limit=25.0, seed=0):
    rng = np.random.default_rng(seed)
    start_time = time.perf_counter()
    h = h.copy()
    score = official_score(h)
    theta = theta0
    v = np.zeros(n)
    lr = lr0
    stall = 0

    for it in range(max_iters):
        if time.perf_counter() - start_time > time_limit:
            break
        grad = lse_gradient(h, theta)
        grad -= np.mean(grad)
        v = 0.9 * v - lr * grad
        h_new = project_feasible(h + v, n)
        s_new = official_score(h_new)
        if s_new < score - 1e-12:
            h = h_new
            score = s_new
            v *= 0.5
            stall = 0
            lr = min(lr * 1.1, 0.15)
        else:
            stall += 1
            v *= 0.5
            found = False
            for _ in range(6):
                h_try = project_feasible(h + v, n)
                s_try = official_score(h_try)
                if s_try < score - 1e-12:
                    h = h_try
                    score = s_try
                    found = True
                    break
                v *= 0.5
            if not found:
                theta *= 1.4
                if theta > 800:
                    break
                lr *= 0.7
                v = np.zeros(n)

    return h, official_score(h)


def slsqp_refine(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    initial_score = official_score(h)

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq(x):
        hx = x[:n]
        c = np.correlate(hx, 1.0 - hx, mode="full")
        return x[n] - dx * c

    def ineq_jac(x):
        hx = x[:n]
        nl = 2 * n - 1
        j = np.zeros((nl, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            j[ki, :n] = -dx * grad_for_lag(hx, int(k), n)
        return j

    def eq(x):
        return np.sum(x[:n]) - target_sum

    def eq_j(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h, [initial_score + 1e-4]])
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_j},
            {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac}]
    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 120, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_out = repair_vector(res.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-11:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def pair_descent(h, n, max_rounds=200, time_limit=15.0, seed=0):
    rng = np.random.default_rng(seed)
    start_time = time.perf_counter()
    h = h.copy()
    score = official_score(h)
    improved = True
    round_num = 0

    while improved and round_num < max_rounds:
        if time.perf_counter() - start_time > time_limit:
            break
        improved = False
        round_num += 1
        for _ in range(n * 2):
            if time.perf_counter() - start_time > time_limit:
                break
            i, j = rng.choice(n, 2, replace=False)
            for delta in [0.02, 0.05, 0.1, 0.2, 0.3]:
                for sign in [1, -1]:
                    h_try = h.copy()
                    h_try[i] += sign * delta
                    h_try[j] -= sign * delta
                    h_try = repair_vector(h_try, n)
                    s_try = official_score(h_try)
                    if s_try < score - 1e-12:
                        h = h_try
                        score = s_try
                        improved = True
                        break
                if improved:
                    break

    return h, official_score(h)


def simulated_annealing(h, n, time_limit=20.0, seed=0):
    rng = np.random.default_rng(seed)
    start_time = time.perf_counter()
    h = h.copy()
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    T = 0.08
    nparts = max(150, n * 4)
    cooling = (0.002 / T) ** (1.0 / nparts)

    for _ in range(nparts):
        if time.perf_counter() - start_time > time_limit:
            break
        T *= cooling
        h_try = h.copy()
        mt = rng.integers(0, 4)
        if mt == 0:
            i = rng.integers(0, n)
            h_try[i] += rng.uniform(-0.4, 0.4)
        elif mt == 1:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.3, 0.3)
            h_try[i] += d
            h_try[j] -= d
        elif mt == 2:
            w = rng.integers(2, max(3, n // 5))
            s = rng.integers(0, max(1, n - w))
            h_try[s:s + w] += rng.uniform(-0.15, 0.15)
        else:
            i = rng.integers(0, n)
            h_try[i] = rng.uniform(0, 1)

        h_try = repair_vector(h_try, n)
        s_try = official_score(h_try)
        if s_try < score or rng.uniform() < np.exp(-(s_try - score) / max(T, 1e-12)):
            h = h_try
            score = s_try
            if score < best_score:
                best_score = score
                best_h = h.copy()

    return best_h, best_score


def gen_interp(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def gen_binary(n, rng):
    h = np.zeros(n)
    k = n // 2
    idx = rng.permutation(n)[:k]
    h[idx] = 1.0
    frac = n / 2.0 - k
    if frac > 0:
        h[rng.integers(0, n)] = frac * (1 - h[rng.integers(0, n)])
    return repair_vector(h, n)


def gen_wave(n, rng):
    t = np.linspace(0, 2, n)
    f = rng.uniform(1.5, 5)
    p = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * f * t + p) + 0.1 * np.sin(4 * np.pi * f * t + p * 0.7)
    return repair_vector(h, n)
