import numpy as np
import time
from scipy.optimize import linprog, minimize

def make_feasible(y, n=None):
    # Same as before - bisect to find the right threshold
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        # Simple fix
        idx_order = np.argsort(z) if diff > 0 else np.argsort(-z)
        remaining = abs(diff)
        for idx in idx_order:
            if diff > 0:
                add = min(remaining, 1.0 - z[idx])
                z[idx] += add
                remaining -= add
            else:
                sub = min(remaining, z[idx])
                z[idx] -= sub
                remaining -= sub
            if remaining < 1e-13:
                break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lp_step(h, scores=None, n_active=10):
    """LP-based step: find direction that reduces active lags simultaneously"""
    n = len(h)
    if scores is None:
        scores = all_scores(h)
    
    base = float(np.max(scores))
    # Find active lags (those close to max)
    active = np.where(scores >= base - 0.01)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    
    # Compute gradients for active lags
    # scores[k] where k is index in full correlation, actual lag = k - (n-1)
    grads = []
    for k in active:
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            # c[lag] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # d/dh[j] = (1-h[j-lag])*1_{j>=lag} - h[j]*1_{j<=n-1-lag}
            for j in range(lag, n):
                g[j] += (1.0 - h[j - lag])
            for j in range(0, n - lag):
                g[j] -= h[j]
        else:
            d = -lag
            # c[-d] = sum_{i=d}^{n-1} h[i-d]*(1-h[i])
            # d/dh[j] = (1-h[j+d])*1_{j<=n-1-d} - h[j-d]*1_{j>=d}
            for j in range(0, n - d):
                g[j] += (1.0 - h[j + d])
            for j in range(d, n):
                g[j] -= h[j - d]
        grads.append(g)
    
    grads = np.array(grads)  # shape (m, n)
    m = len(grads)
    
    # LP: minimize t
    # subject to: scores[k] + grads[k] @ d <= t  for all k in active
    #             sum(d) = 0
    #             h[j] + d[j] >= 0  =>  -d[j] <= h[j]
    #             h[j] + d[j] <= 1  =>  d[j] <= 1 - h[j]
    #
    # Variables: [d_0, ..., d_{n-1}, t]  (n+1 variables)
    #
    # Minimize: [0, ..., 0, 1] @ [d, t]
    #
    # Inequality constraints (A_ub @ vars <= b_ub):
    #   For each active lag k: grads[k] @ d - t <= -scores[k]
    #     => [grads[k], -1] @ [d, t] <= -scores[k]
    #   For each j: -d[j] <= h[j] => [-e_j, 0] @ [d,t] <= h[j]
    #   For each j: d[j] <= 1-h[j] => [e_j, 0] @ [d,t] <= 1-h[j]
    #
    # Equality constraints (A_eq @ vars = b_eq):
    #   sum(d) = 0 => [1, 1, ..., 1, 0] @ [d, t] = 0
    
    # c vector (objective)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints
    A_ub = np.zeros((m + 2*n, n + 1))
    b_ub = np.zeros(m + 2*n)
    
    for i, k in enumerate(active):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[k]
    
    for j in range(n):
        A_ub[m + j, j] = -1.0  # -d[j] <= h[j]
        b_ub[m + j] = h[j]
        A_ub[m + n + j, j] = 1.0  # d[j] <= 1-h[j]
        b_ub[m + n + j] = 1.0 - h[j]
    
    # Equality constraint
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: already handled by inequalities, but add explicit bounds for solver
    bounds = [(0, None)] * (n + 1)  # t >= 0 (actually t can be negative, but that's fine)
    # Actually t should be free or >= -inf. Let's not bound it.
    bounds = [(None, None)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    d = result.x[:n]
    t_new = result.x[n]
    
    # Check if this is a descent direction
    # The new scores would be approximately scores + grads @ d
    # We want max(scores + grads @ d) < max(scores) = base
    # The LP ensures scores[k] + grads[k]@d <= t for active k
    # But we need to check all k
    
    return d

def block_optimize(h, blocks, deadline):
    """Optimize blocks of variables jointly"""
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    
    while time.time() < deadline:
        improved = False
        for block in blocks:
            if time.time() > deadline:
                break
            if len(block) < 2:
                continue
            # Optimize h[block] jointly
            def obj(x):
                tmp = cur.copy()
                tmp[block] = x
                # Don't enforce feasibility here, just evaluate
                # But we need to handle the constraint...
                # Actually for block optimization, we can't easily maintain the sum constraint
                # Let's instead optimize with a penalty
                return score_exact(tmp)
            
            x0 = cur[block]
            bounds_b = [(0, 1)] * len(block)
            # Use L-BFGS-B
            res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds_b,
                          options={'maxiter': 50, 'ftol': 1e-12})
            if res.success and res.fun < cur_c - 1e-12:
                cur[block] = res.x
                # Re-project to maintain constraint
                cur = make_feasible(cur, n)
                cur_c = score_exact(cur)
                if cur_c < cur_c + 1e-12:  # sanity check
                    improved = True
                # Update cur_c after projection
                cur_c = score_exact(cur)
    
    return cur, cur_c

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

def grad_descent(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        w, g = smooth_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 30:
                cur = make_feasible(cur + rng.normal(0, max(0.003, T * 0.2), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.003)
                stall = 0
        else:
            step = min(step * 1.1, T * 2.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                best_a = 0.0
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        idx = np.argsort(z) if diff > 0 else np.argsort(-z)
        rem = abs(diff)
        for i in idx:
            if diff > 0:
                a = min(rem, 1.0 - z[i])
                z[i] += a
                rem -= a
            else:
                a = min(rem, z[i])
                z[i] -= a
                rem -= a
            if rem < 1e-13:
                break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        ws = w.sum()
        if ws > 0:
            w = w / ws
        else:
            w = np.ones(len(scores)) / len(scores)
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

def fw_method(h, deadline, T_start=0.15, T_end=0.002, T_decay=0.9998, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    k = 0
    stall = 0
    k_half = n // 2
    while time.time() < deadline:
        w, g = smooth_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        idx = np.argsort(g)
        s = np.zeros(n)
        s[idx[:k_half]] = 1.0
        base_gamma = 2.0 / (k + 2)
        improved = False
        for gamma in (base_gamma, 1.5 * base_gamma, 2.0 * base_gamma, 0.5 * base_gamma, min(1.0, 4.0 * base_gamma)):
            if gamma <= 0 or gamma > 1.0:
                continue
            cand = (1.0 - gamma) * cur + gamma * s
            c_try = score_exact(cand)
            if c_try < cur_c - 1e-13:
                cur = cand
                cur_c = c_try
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            stall += 1
            if stall > 40:
                noise = rng.normal(0, max(0.005, T * 0.3), n)
                cur = make_feasible(cur + noise, n)
                cur_c = score_exact(cur)
                T = max(T_end, T * 0.5)
                stall = 0
        else:
            stall = max(0, stall - 1)
        k += 1
        T = max(T_end, T * T_decay)
    return best, best_c

def lp_step(h, scores, n_active=6):
    n = len(h)
    base = float(np.max(scores))
    active = np.where(scores >= base - 0.008)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    active = active[:n_active]
    grads = []
    for k in active:
        lag = int(k) - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            for j in range(lag, n):
                g[j] += (1.0 - h[j - lag])
            for j in range(0, n - lag):
                g[j] -= h[j]
        else:
            d = -lag
            for j in range(0, n - d):
                g[j] += (1.0 - h[j + d])
            for j in range(d, n):
                g[j] -= h[j - d]
        grads.append(g)
    grads = np.array(grads)
    m = len(grads)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    for i in range(m):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[active[i]]
    for j in range(n):
        A_ub[m + j, j] = -1.0
        b_ub[m + j] = h[j]
        A_ub[m + n + j, j] = 1.0
        b_ub[m + n + j] = 1.0 - h[j]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(None, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def grad_descent(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        w, g = smooth_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.82
            stall += 1
            if stall > 35:
                cur = make_feasible(cur + rng.normal(0, max(0.004, T * 0.25), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.004)
                stall = 0
        else:
            step = min(step * 1.08, T * 3.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                best_a = 0.0
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def triplet_search(h, i, j, k, cur_c, n_pts=12):
    oi, oj, ok = h[i], h[j], h[k]
    lo_i = max(-oi, oj + ok - 1.0)
    hi_i = min(1.0 - oi, oj + ok)
    if hi_i - lo_i < 1e-12:
        return None
    lo_j = max(-oj, oi + ok - 1.0)
    hi_j = min(1.0 - oj, oi + ok)
    if hi_j - lo_j < 1e-12:
        return None
    best_s = cur_c
    best_ai, best_aj = 0.0, 0.0
    ts = np.linspace(max(lo_i, -oj - ok + 1.0), min(hi_i, 1.0 - oj - ok + 1.0), n_pts) if n_pts > 1 else np.array([0.0])
    for ai in ts:
        h[i] = oi + ai
        rem_lo = max(-oj, (ok - 1.0) - ai, -oi - ai)
        rem_hi = min(1.0 - oj, ok - ai, 1.0 - oi - ai)
        if rem_hi - rem_lo < 1e-12:
            continue
        for aj in np.linspace(rem_lo, rem_hi, n_pts):
            h[j] = oj + aj
            h[k] = ok - ai - aj
            if h[k] < -1e-12 or h[k] > 1.0 + 1e-12:
                continue
            s = score_exact(h)
            if s < best_s:
                best_s = s
                best_ai, best_aj = ai, aj
    h[i], h[j], h[k] = oi, oj, ok
    if best_s < cur_c - 1e-14 and (abs(best_ai) > 1e-10 or abs(best_aj) > 1e-10):
        h[i] = oi + best_ai
        h[j] = oj + best_aj
        h[k] = ok - best_ai - best_aj
        h[k] = np.clip(h[k], 0.0, 1.0)
        h = make_feasible(h)
        s_final = score_exact(h)
        if s_final < cur_c - 1e-14:
            return float(s_final)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 16:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        n_init = 70
        x = (np.arange(n_init) + 0.5) * (2.0 / n_init)
        best_h = make_feasible(np.full(n_init, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: FW method for exploration (12%)
    p1 = t0 + budget_s * 0.12
    if time.time() < p1:
        h1, c1 = fw_method(best_h, p1, T_start=0.2, T_end=0.005, rng=rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
            n = len(best_h)

    # Phase 2: Gradient descent (18%)
    p2 = t0 + budget_s * 0.30
    if time.time() < p2:
        h2, c2 = grad_descent(best_h, p2, 0.012, 0.0005, 0.9995, 0.006, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2
        n = len(best_h)

    # Phase 3: LP-based steps (10%)
    p3 = t0 + budget_s * 0.40
    cur = best_h.copy()
    cur_c = best_c
    while time.time() < p3:
        scores = all_scores(cur)
        d = lp_step(cur, scores, n_active=6)
        if d is not None:
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                d = d / dn
                improved = False
                for step in (0.4, 0.2, 0.1, 0.05, 0.02):
                    cand = make_feasible(cur + step * d, n)
                    c_new = score_exact(cand)
                    if c_new < cur_c - 1e-14:
                        cur = cand
                        cur_c = c_new
                        improved = True
                        if cur_c < best_c:
                            best_h, best_c = cur.copy(), cur_c
                        break
                if not improved:
                    w, g = smooth_grad(cur, 0.005)
                    gn = np.linalg.norm(g)
                    if gn > 1e-12:
                        for step in (0.008, 0.004, 0.002):
                            cand = make_feasible(cur - step * g / gn, n)
                            c_new = score_exact(cand)
                            if c_new < cur_c - 1e-14:
                                cur = cand
                                cur_c = c_new
                                if cur_c < best_c:
                                    best_h, best_c = cur.copy(), cur_c
                                break

    # Phase 4: Aggressive pair + triplet search (35%)
    p4 = t0 + budget_s * 0.75
    cur = best_h.copy()
    cur_c = best_c
    stall_count = 0
    iter_count = 0
    while time.time() < p4:
        iter_count += 1
        mode = rng.random()
        if mode < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=30)
        elif mode < 0.75:
            i = int(rng.integers(0, n - 1))
            j = i + 1
            res = pair_search(cur, i, j, cur_c, n_pts=35)
        elif mode < 0.85:
            i = int(rng.integers(0, n))
            j = (i + 1) % n
            k = (i + 2) % n
            if len(set([i, j, k])) < 3:
                continue
            res = triplet_search(cur, i, j, k, cur_c, n_pts=10)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=40)
        if res is not None:
            cur_c = res
            stall_count = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
                follow = min(5, int((p4 - time.time()) / 5.0))
                for _ in range(max(1, follow)):
                    if time.time() > p4:
                        break
                    i2 = int(rng.integers(0, n))
                    j2 = int(rng.integers(0, n))
                    if i2 == j2:
                        continue
                    res2 = pair_search(cur, i2, j2, cur_c, n_pts=20)
                    if res2 is not None:
                        cur_c = res2
                        if cur_c < best_c:
                            best_h, best_c = cur.copy(), cur_c
                    else:
                        break
        else:
            stall_count += 1
            if stall_count > 120:
                cur = make_feasible(best_h + rng.normal(0, 0.006, n), n)
                cur_c = score_exact(cur)
                stall_count = 0

    # Phase 5: Fine gradient descent (8%)
    p5 = t0 + budget_s * 0.83
    if time.time() < p5:
        h5, c5 = grad_descent(best_h, p5, 0.004, 0.00005, 0.999, 0.002, rng)
        if c5 < best_c:
            best_h, best_c = h5, c5
        n = len(best_h)

    # Phase 6: Resolution sweeping (7%)
    p6 = t0 + budget_s * 0.90
    n_cur = len(best_h)
    tried = set()
    for dn in (-6, -8, -12, 6, 8, 12, 16, -16):
        if time.time() > p6 - 3:
            break
        nr = n_cur + dn
        if nr in tried or nr < 20 or nr > 160:
            continue
        tried.add(nr)
        h_r = resample(best_h, nr)
        if h_r is None:
            continue
        c_r = score_exact(h_r)
        sub_dl = min(p6 - 1, time.time() + 35)
        if time.time() < sub_dl:
            h_r, c_r = grad_descent(h_r, sub_dl, 0.008, 0.0002, 0.999, 0.004, rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
            n = len(best_h)

    # Phase 7: Final polish - pair search + gradient (10%)
    p7 = deadline
    cur = best_h.copy()
    cur_c = best_c
    n = len(cur)
    while time.time() < p7:
        # Quick gradient step
        w, g = smooth_grad(cur, 0.002)
        gn = np.linalg.norm(g)
        if gn > 1e-12:
            for step in (0.003, 0.001, 0.0005):
                cand = make_feasible(cur - step * g / gn, n)
                c_new = score_exact(cand)
                if c_new < cur_c - 1e-14:
                    cur = cand
                    cur_c = c_new
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                    break
        # Pair search
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        res = pair_search(cur, i, j, cur_c, n_pts=25)
        if res is not None:
            cur_c = res
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
