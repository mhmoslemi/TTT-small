def _smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        j_lo = max(0, shift)
        j_hi = min(n, n + shift)
        for p in range(j_lo, j_hi):
            grad[p] += w * (1.0 - x[p - shift])
        p_lo2 = max(0, j_lo - shift)
        p_hi2 = min(n, j_hi - shift)
        for p in range(p_lo2, p_hi2):
            grad[p] += w * (-x[p + shift])
    return grad

        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    # Stage 3: Fista subgradient averaging
    if time.time() < deadline - 10:
        cur, cur_s = _subgrad_fista(cur, beta=200, n_iter=8000, alpha_init=0.006, mu=0.85)

    # Stage 4: Another SLSQP polish
    if time.time() < deadline - 10:
        new = _slsqp_smooth(cur, beta=800, maxiter=150)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    # Stage 5: Direct min-max
    if time.time() < deadline - 8:
        new = _slsqp_direct(cur, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    # Stage 6: Final Fista
    if time.time() < deadline - 5:
        cur, cur_s = _subgrad_fista(cur, beta=500, n_iter=5000, alpha_init=0.003, mu=0.9)

    return cur, cur_s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(40) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)

    rng = np.random.default_rng(seed)

    # Determine n values to try
    n_candidates = set()
    for n in range(max(24, parent_n - 14), min(68, parent_n + 16) + 1, 2):
        n_candidates.add(n)
    # Ensure parent_n is included
    n_candidates.add(parent_n)
    n_sorted = sorted(n_candidates)

    # Phase 1: Quick screening of all n values
    phase1_results = []
    for n in n_sorted:
        if time.time() > deadline - 60:
            break
        h_interp = _interp(parent, n)
        s0 = _score(h_interp)
        # Quick SLSQP
        if time.time() < deadline - 40:
            h_opt = _slsqp_smooth(h_interp, beta=80, maxiter=80)
            s1 = _score(h_opt)
            if s1 < s0:
                phase1_results.append((s1, n, h_opt))
            else:
                phase1_results.append((s0, n, h_interp))
        else:
            phase1_results.append((s0, n, h_interp))
        # Update best
        sc = _score(phase1_results[-1][2])
        if sc < best_score - 1e-12:
            best_h = phase1_results[-1][2].copy()
            best_score = sc
            best_n = n

    # Phase 2: Focused optimization on top n values
    phase1_results.sort(key=lambda x: x[0])
    top_ns = []
    for s, n, h in phase1_results[:5]:
        if n not in top_ns:
            top_ns.append(n)

    for n in top_ns:
        if time.time() > deadline - 30:
            break

        # Get the best h for this n from phase 1
        h_base = None
        for s, nn, hh in phase1_results:
            if nn == n:
                h_base = hh.copy()
                break
        if h_base is None:
            h_base = _interp(parent, n)

        # Generate diverse starting points
        starts = [h_base.copy()]
        starts.append(_mirror(h_base))
        for sigma in [0.08, 0.15, 0.25]:
            starts.append(_project(h_base + sigma * rng.standard_normal(n)))
        for p in [1.3, 2.0, 3.0]:
            starts.append(_binary_push(h_base, p))
        # Structured: sinusoidal perturbation
        for freq in [1, 2, 3]:
            phase = rng.uniform(0, 6.28)
            pert = 0.1 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            starts.append(_project(h_base + pert))

        for h0 in starts:
            if time.time() > deadline - 20:
                break
            cur, cur_s = _chain_optimize(h0, deadline, rng)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)

    # Phase 3: Intensive refinement on best solution
    remaining = deadline - time.time()
    if remaining > 40:
        # Multiple Fista passes with different betas
        for beta in [100, 400, 1000, 2000]:
            if time.time() > deadline - 10:
                break
            best_h, best_score = _subgrad_fista(
                best_h, beta=beta, n_iter=min(15000, int(remaining / 6)),
                alpha_init=0.004, mu=0.88)
            remaining = deadline - time.time()

    if time.time() < deadline - 30:
        # SLSQP polish
        for beta in [500, 1500]:
            if time.time() > deadline - 15:
                break
            new = _slsqp_smooth(best_h, beta=beta, maxiter=150)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h = new
                best_score = ns

    if time.time() < deadline - 20:
        # Direct min-max final
        new = _slsqp_direct(best_h, maxiter=150)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h = new
            best_score = ns

    # Phase 4: Random restarts around best
    remaining = deadline - time.time()
    n_restarts = min(15, int(remaining / 25))
    for _ in range(n_restarts):
        if time.time() > deadline - 10:
            break
        h_pert = _project(best_h + 0.03 * rng.standard_normal(best_n))
        cur, cur_s = _chain_optimize(h_pert, deadline, rng)
        if cur_s < best_score - 1e-12:
            best_h = cur.copy()
            best_score = cur_s
            best_n = len(cur)

    # Final single Fista pass
    if time.time() < deadline - 5:
        best_h, best_score = _subgrad_fista(best_h, beta=1000, n_iter=5000,
                                            alpha_init=0.002, mu=0.92)

    # Final feasibility check
    best_h = _project(best_h)
    final_score = _score(best_h)

    return (best_h, final_score, len(best_h))
