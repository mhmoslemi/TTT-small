import numpy as np
import time
from scipy.optimize import linprog

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def scores_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lp_step(h, trust_radius=0.15, top_k=None):
    n = len(h)
    dx = 2.0 / n
    scores = scores_all(h)
    cur_max = float(np.max(scores))
    
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(scores)[::-1][:top_k]
        lag_indices = idx
    else:
        lag_indices = np.arange(2 * n - 1)
    
    n_vars = n + 1
    A_rows = np.zeros((len(lag_indices), n_vars))
    b_rows = np.zeros(len(lag_indices))
    
    for ii, m in enumerate(lag_indices):
        d = int(m) - (n - 1)
        k = np.arange(n)
        kd = k + d
        valid_pos = (kd >= 0) & (kd < n)
        grad = np.zeros(n)
        grad[valid_pos] += dx * (1.0 - h[kd[valid_pos]])
        kd2 = k - d
        valid_neg = (kd2 >= 0) & (kd2 < n)
        grad[valid_neg] -= dx * h[kd2[valid_neg]]
        A_rows[ii, :n] = grad
        A_rows[ii, n] = -1.0
        b_rows[ii] = -scores[int(m)]
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo_b = -min(h[i], trust_radius)
        hi_b = min(1.0 - h[i], trust_radius)
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_rows, b_ub=b_rows, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status != 0 or not hasattr(res, 'x'):
        return None
    
    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-12:
        return None
    
    best_h = None
    best_c = cur_max
    for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        candidate = make_feasible(h + frac * delta, n)
        c_new = score_exact(candidate)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = candidate
    
    return best_h

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

def grad_descent(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        w, g = smooth_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.85
            stall += 1
            if stall > 40:
                cur = make_feasible(cur + rng.normal(0, max(0.005, T * 0.3), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.005)
                stall = 0
        else:
            step = min(step * 1.05, T * 2.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=35):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(25):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                best_a = 0.0
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=30):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def binary_push(h, n):
    """Try pushing values toward 0 or 1 to see if it helps."""
    h2 = h.copy()
    for i in range(n):
        if 0.15 < h2[i] < 0.85:
            h2[i] = 0.0 if h2[i] < 0.5 else 1.0
    h2 = make_feasible(h2, n)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: LP-based descent (55% of time)
    p1_end = t0 + budget_s * 0.55
    cur_h = best_h.copy()
    cur_c = best_c
    rho = 0.2
    stall = 0
    lp_count = 0
    
    while time.time() < p1_end:
        new_h = lp_step(cur_h, trust_radius=rho, top_k=None)
        lp_count += 1
        if new_h is not None:
            cur_h = new_h
            cur_c = score_exact(cur_h)
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
            rho = min(rho * 1.03, 0.35)
            stall = 0
        else:
            rho *= 0.75
            stall += 1
            if stall > 15:
                # Try pair search to escape
                esc = False
                for _ in range(60):
                    if time.time() > p1_end:
                        break
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
                    res = pair_search(cur_h, i, j, cur_c, n_pts=30)
                    if res is not None:
                        cur_c = res
                        esc = True
                        if cur_c < best_c:
                            best_h, best_c = cur_h.copy(), cur_c
                if esc:
                    stall = 0
                    rho = 0.15
                else:
                    # Try binary push
                    bp = binary_push(cur_h, n)
                    bp_c = score_exact(bp)
                    if bp_c < cur_c:
                        cur_h = bp
                        cur_c = bp_c
                        if cur_c < best_c:
                            best_h, best_c = cur_h.copy(), cur_c
                        stall = 0
                        rho = 0.15
                    else:
                        cur_h = make_feasible(best_h + rng.normal(0, 0.015, n), n)
                        cur_c = score_exact(cur_h)
                        stall = 0
                        rho = 0.2
                if rho < 0.003:
                    # LP is stuck, switch to pair search mode temporarily
                    for _ in range(40):
                        if time.time() > p1_end:
                            break
                        i = int(rng.integers(0, n))
                        j = int(rng.integers(0, n))
                        if i == j:
                            continue
                        res = pair_search(cur_h, i, j, cur_c, n_pts=25)
                        if res is not None:
                            cur_c = res
                            if cur_c < best_c:
                                best_h, best_c = cur_h.copy(), cur_c
                    rho = 0.1

    # Phase 2: Gradient descent polish (15% of time)
    p2_end = t0 + budget_s * 0.70
    if time.time() < p2_end:
        h2, c2 = grad_descent(best_h, p2_end, 0.01, 0.0002, 0.999, 0.003, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2
            n = len(best_h)

    # Phase 3: Systematic pair + LP sweep (20% of time)
    p3_end = t0 + budget_s * 0.90
    cur_h = best_h.copy()
    cur_c = best_c
    sweep = 0
    while time.time() < p3_end:
        # Alternating: one LP step, then a few pair searches
        new_h = lp_step(cur_h, trust_radius=0.08, top_k=50)
        if new_h is not None:
            cur_h = new_h
            cur_c = score_exact(cur_h)
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # Adjacent pair sweep
        for i in range(0, n - 1, max(1, n // 40)):
            if time.time() > p3_end:
                break
            res = pair_search(cur_h, i, min(i + 1, n - 1), cur_c, n_pts=25)
            if res is not None:
                cur_c = res
                if cur_c < best_c:
                    best_h, best_c = cur_h.copy(), cur_c
        # Random pairs
        for _ in range(5):
            if time.time() > p3_end:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur_h, i, j, cur_c, n_pts=20)
            if res is not None:
                cur_c = res
                if cur_c < best_c:
                    best_h, best_c = cur_h.copy(), cur_c
        sweep += 1

    # Phase 4: Try different resolutions (8% of time)
    p4_end = t0 + budget_s * 0.98
    n_cur = len(best_h)
    for dn in [-4, 4, -6, 6, -8, 3, 5]:
        if time.time() > p4_end - 5:
            break
        nr = n_cur + dn
        if nr < 30 or nr > 150:
            continue
        h_r = resample(best_h, nr)
        if h_r is None:
            continue
        c_r = score_exact(h_r)
        sub_dl = min(p4_end - 3, time.time() + 50)
        rr = 0.1
        for _ in range(50):
            if time.time() > sub_dl:
                break
            new_r = lp_step(h_r, trust_radius=rr, top_k=40)
            if new_r is not None:
                h_r = new_r
                c_r = score_exact(h_r)
                rr = min(rr * 1.02, 0.2)
            else:
                rr *= 0.7
                if rr < 0.005:
                    break
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Final polish
    if time.time() < deadline - 1:
        h_f, c_f = grad_descent(best_h, deadline - 1, 0.003, 0.00003, 0.998, 0.002, rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
