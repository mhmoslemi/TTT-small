```python
    n = len(h)
    dx = 2.0 / n

    one_minus_h = 1.0 - h

    # overlap_s = a_s * t + b_s where t = h[j]
    a = np.zeros(2*n - 1)
    b = np.zeros(2*n - 1)
    for s in range(2*n - 1):
        shift = s - (n - 1)
        if shift >= 0:
            # overlap = sum_{i=0}^{n-shift-1} h[i+shift] * (1-h[i])
            # h[j] appears as h[i+shift] when i = j-shift (need j-shift >= 0 and j-shift <= n-shift-1)
            # h[j] appears in (1-h[i]) when i = j (need j <= n-shift-1)
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - np.sum(z)
    if residual > 0:
        order = np.argsort(1.0 - z)
        for idx in order:
            add = min(residual, 1.0 - z[idx])
            if add <= 0:
                break
            z[idx] += add
            residual -= add
            if residual < 1e-15:
                break

            if sub <= 0:
```

```python
            # Find d minimizing max_s (overlap_s + (a_s^j - a_s^k) * d)
            # subject to: 0 ≤ h[j] + d ≤ 1 and 0 ≤ h[k] - d ≤ 1
            # i.e., -h[j] ≤ d ≤ 1 - h[j] and h[k] - 1 ≤ d ≤ h[k]
            # Combined: max(-h[j], h[k]-1) ≤ d ≤ min(1-h[j], h[k])
            
            a_j = np.zeros(2*n - 1)
            a_k = np.zeros(2*n - 1)
            # Compute a_j (derivative of overlap w.r.t. h[j])
            for s in range(2*n - 1):
                shift = s - (n - 1)
                if shift >= 0:
                    if j >= shift:
                        a_j[s] += (1.0 - h[j-shift])
                    if j + shift < n:
                        a_j[s] += (-h[j+shift])
                else:
                    t_shift = -shift
                    if j + t_shift < n:
                        a_j[s] += (1.0 - h[j+t_shift])
                    if j >= t_shift:
                        a_j[s] += (-h[j-t_shift])
            
            for s in range(2*n - 1):
                shift = s - (n - 1)
                if shift >= 0:
                    if k >= shift:
                        a_k[s] += (1.0 - h[k-shift])
                    if k + shift < n:
                        a_k[s] += (-h[k+shift])
                else:
                    t_shift = -shift
                    if k + t_shift < n:
                        a_k[s] += (1.0 - h[k+t_shift])
                    if k >= t_shift:
                        a_k[s] += (-h[k-t_shift])
            
            da = a_j - a_k
            overlaps = np.correlate(h, 1.0 - h, mode='full')
            
            d_lb = max(-h[j], h[k] - 1.0)
            d_ub = min(1.0 - h[j], h[k])
            
            if d_lb >= d_ub:
                continue
            
            def f_d(d):
                return (overlaps + da * d).max()
            
            res = minimize_scalar(f_d, bounds=(d_lb, d_ub), method='brent',
                                  options={'xatol': 1e-13})
            best_d = res.x
            max_change = max(max_change, abs(best_d))
            h[j] += best_d
            h[k] -= best_d
        
        if max_change < tol:
            break
    
    return h
```

```python
    return a

def pairwise_update(h, j, k):
    """Find best d to update h[j]+=d, h[k]-=d. Returns (h_new, improvement)."""
    n = len(h)
    a_j = get_derivatives(h, j)
    a_k = get_derivatives(h, k)
    da = a_j - a_k
    overlaps = np.correlate(h, 1.0 - h, mode='full')
    
    d_lb = max(-h[j], h[k] - 1.0)
    d_ub = min(1.0 - h[j], h[k])
    if d_lb >= d_ub - 1e-15:
        return h, 0.0
    
    if abs(da).max() < 1e-15:
        return h, 0.0
    
    def f_d(d):
        return (overlaps + da * d).max()
    
    # Current value
    f_cur = overlaps.max()
    
    res = minimize_scalar(f_d, bounds=(d_lb, d_ub), method='brent',
                          options={'xatol': 1e-13})
    best_d = res.x
    improvement = f_cur - res.fun
    h_new = h.copy()
    h_new[j] += best_d
    h_new[k] -= best_d
    return h_new, improvement

        if strategy == 'cyclic':

                h_new, imp = pairwise_update(h, j, k)
                if imp > 0:
                    h = h_new
                    max_imp = max(max_imp, imp)
        elif strategy == 'best':
            # Best improvement: try all pairs, pick the best
            best_imp = 0.0
            best_jk = None
            for j in range(n):
                for k in range(j+1, n):
                    h_new, imp = pairwise_update(h, j, k)
                    if imp > best_imp:
                        best_imp = imp
                        best_jk = (j, k)
            if best_jk is not None:
                h, _ = pairwise_update(h, best_jk[0], best_jk[1])
                max_imp = best_imp
        elif strategy == 'random':
            # Random pairs
            for _ in range(2 * n):
                j = np.random.randint(n)
                k = np.random.randint(n)
                if j == k:
                    continue
                h_new, imp = pairwise_update(h, j, k)
                if imp > 0:
                    h = h_new
                    max_imp = max(max_imp, imp)
        
        if max_imp < tol:
            break
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):

    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()

    residual = target - np.sum(z)
    if residual > 1e-14:
        order = np.argsort(1.0 - z)
        for idx in order:
            add = min(residual, 1.0 - z[idx])
            if add <= 1e-15:
                break
            z[idx] += add
            residual -= add
    elif residual < -1e-14:
        order = np.argsort(z)
        for idx in order:
            sub = min(-residual, z[idx])
            if sub <= 1e-15:
                break
            z[idx] -= sub
            residual += sub
    return z

def get_derivatives(h, j):
    n = len(h)
    a = np.zeros(2 * n - 1)
    for s in range(2 * n - 1):
        shift = s - (n - 1)
        if shift >= 0:
            if j >= shift:
                a[s] += (1.0 - h[j - shift])
            if j + shift < n:
                a[s] += (-h[j + shift])
        else:
            t = -shift
            if j + t < n:
                a[s] += (1.0 - h[j + t])
            if j >= t:
                a[s] += (-h[j - t])
    return a

def pairwise_update(h, j, k):
    n = len(h)
    a_j = get_derivatives(h, j)
    a_k = get_derivatives(h, k)
    da = a_j - a_k
    if abs(da).max() < 1e-15:
        return h, 0.0
    overlaps = np.correlate(h, 1.0 - h, mode='full')
    d_lb = max(-h[j], h[k] - 1.0) + 1e-12
    d_ub = min(1.0 - h[j], h[k]) - 1e-12
    if d_lb >= d_ub:
        return h, 0.0
    
    def f_d(d):
        return (overlaps + da * d).max()
    
    f_cur = overlaps.max()
    res = minimize_scalar(f_d, bounds=(d_lb, d_ub), method='brent',
                          options={'xatol': 1e-13})
    imp = f_cur - res.fun
    if imp < 1e-15:
        return h, 0.0
    h_new = h.copy()
    h_new[j] += res.x
    h_new[k] -= res.x
    h_new[j] = np.clip(h_new[j], 0.0, 1.0)
    h_new[k] = np.clip(h_new[k], 0.0, 1.0)
    return h_new, imp

def coord_descent(h, max_sweeps=1000, tol=1e-13, strategy='cyclic', rng=None):
    n = len(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        max_imp = 0.0
        if strategy == 'cyclic':
            for j in range(n):
                k = (j + 1) % n
                h_new, imp = pairwise_update(h, j, k)
                if imp > 0:
                    h = h_new
                    max_imp = max(max_imp, imp)
        elif strategy == 'random':
            for _ in range(2 * n):

                k = rng.integers(n)

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    # Gradient of sum of overlaps
    # d/dx[j] of overlap_s at shift s
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        ws_val = w[s]
        if ws_val < 1e-15:
            continue
        shift = s - (n - 1)
        if shift >= 0:
            for j in range(n):
                if j >= shift:
                    grad[j] += ws_val * (1.0 - x[j - shift])
                if j + shift < n:
                    grad[j] += ws_val * (-x[j + shift])
        else:
            t = -shift
            for j in range(n):
                if j + t < n:
                    grad[j] += ws_val * (1.0 - x[j + t])
                if j >= t:
                    grad[j] += ws_val * (-x[j - t])
    return dx * grad

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

    return None, None
```

```python
        if sc < best_score - 1e-14:
            update(cur, sc)

    # Phase 5: Random restarts with CD
    if not is_deadline():
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(4, min(25, int(remaining / 12)))
        for i in range(n_starts):
            if is_deadline():
                break
            sigma = [0.02, 0.04, 0.08, 0.12, 0.03, 0.06, 0.15][i % 7]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_cand = coord_descent(h_cand, max_sweeps=400, tol=1e-14, rng=rng)
            sc = official_score(h_cand)
            if sc < best_score - 1e-14:
                update(h_cand, sc)
            if not is_deadline():
                h_new, sc2 = try_smooth(h_cand, 500, maxiter=100)
                if h_new is not None:
                    cur2 = coord_descent(h_new, max_sweeps=200, tol=1e-14, rng=rng)
                    sc2 = official_score(cur2)
                    if sc2 < best_score - 1e-14:
                        update(cur2, sc2)

    # Phase 6: Resolution changes
    if not is_deadline():
        n_base = len(best_h)
        opts = sorted(set([max(30, n_base - 10), max(30, n_base - 5),
                           n_base + 5, n_base + 10, n_base + 15]))
        for n_opt in opts:
            if is_deadline():
                break
            if n_opt < 20 or n_opt > 100:
                continue
            h_int = interpolate_to(best_h, n_opt)
            h_int = coord_descent(h_int, max_sweeps=500, tol=1e-14, rng=rng)
            sc = official_score(h_int)
            if sc < best_score - 1e-14:
                update(h_int, sc)
            if not is_deadline():
                h_new, sc2 = try_smooth(h_int, 300, maxiter=150)
                if h_new is not None:
                    cur2 = coord_descent(h_new, max_sweeps=200, tol=1e-14, rng=rng)
                    sc2 = official_score(cur2)
                    if sc2 < best_score - 1e-14:
                        update(cur2, sc2)

    # Phase 7: Direct minimax SLSQP
    if not is_deadline():
        h_new, sc = try_direct_mm(best_h, maxiter=300)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 8: Final intensive CD + polish
    if not is_deadline():
        cur = coord_descent(best_h, max_sweeps=2000, tol=1e-15, rng=rng)
        sc = official_score(cur)
        if sc < best_score - 1e-14:
            best_h = cur.copy()
            best_score = sc
        for beta in [2000, 10000]:
            if is_deadline():
                break
            h_new, sc = try_smooth(best_h, beta, maxiter=200)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
        if not is_deadline():
            cur = coord_descent(best_h, max_sweeps=500, tol=1e-15, rng=rng)
            sc = official_score(cur)
            if sc < best_score - 1e-14:
                best_h = cur.copy()
                best_score = sc

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
```
