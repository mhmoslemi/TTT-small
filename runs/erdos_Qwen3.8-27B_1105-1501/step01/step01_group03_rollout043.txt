from scipy.fft import fft, ifft
import time
import cvxpy as cp

def fast_correlate(h):
    """Compute full cross-correlation h * (1-h) via FFT."""
    n = len(h)
    N = 2 * n - 1
    H = fft(h, N)
    G = fft(1.0 - h, N)
    return ifft(np.conj(H) * G).real

def fast_score(h):
    """Fast C5 evaluation using FFT."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(fast_correlate(h)) * dx)

def official_score(h):
    """Exact C5 evaluation (for final verification)."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    """Project z to feasible set: 0<=h<=1, sum(h)=n/2."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        idx = np.argsort(h) if err > 0 else np.argsort(-h)
        for i in idx:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1:
                d = min(err, 1 - h[i]); h[i] += d; err -= d
            elif err < 0 and h[i] > 0:
                d = min(-err, h[i]); h[i] -= d; err += d
    return h

def lse_obj(h, beta):
    """Log-sum-exp smoothed max correlation."""
    dx = 2.0 / len(h)
    C = fast_correlate(h) * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    ws = np.sum(np.exp(shifted - mk))
    if ws < 1e-300:
        return cmax + np.log(2*len(h)-1) / beta
    return cmax + (mk + np.log(ws)) / beta

    """Gradient of LSE objective."""
    n = len(h)
    dx = 2.0 / n
    C = fast_correlate(h) * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = es / ws

    for k in range(2*n-1):

        if wk < 1e-12:

        s = k - (n-1)  # shift
        # (1-h[j-s]) part: j in [max(0,s), min(n-1, n-1+s)]
        lo1, hi1 = max(0, s), min(n, n+s)
        if lo1 < hi1:
            if s >= 0:
                grad[lo1:hi1] += wk * (1.0 - h[:n-s])
            else:
                grad[lo1:hi1] += wk * (1.0 - h[-s:])
        # -h[j+s] part: j in [max(0,-s), min(n-1, n-1-s)]
        lo2, hi2 = max(0, -s), min(n, n-s)
        if lo2 < hi2:
            if s >= 0:
                grad[lo2:hi2] += wk * (-h[s:])
            else:
                grad[lo2:hi2] += wk * (-h[:n+s])
    return grad * dx

    return float(np.sum(h) - len(h)/2)

def sum_eq_jac(h):
    return np.ones(len(h))

    h = make_feasible(h0.copy())
    n = len(h)

                          options={'maxiter': maxiter_each, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h = make_feasible(res.x)
        except:
            pass
    return h

    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return make_feasible(np.interp(x_tgt, x_src, h_src))

def solve_milp_binary(n, time_limit=30):

**Frank-Wolfe (conditional gradient) method**:
- The feasible set is a simplex-like set (intersection of box with hyperplane)
- The objective is convex? No, it's NOT convex (max of non-convex functions).

So Frank-Wolfe doesn't directly apply.

**Alternating minimization**: 
- Fix the "active set" of shifts (those where F(k) = max)
- Minimize the sum of active F(k) values
- Update the active set
- Repeat

This is like a block coordinate method on the "shift dimension" and the "function dimension" simultaneously. Let me try this:

```
Active = {k: F(k) = max}
Repeat:
  Minimize sum_{k in Active} F(k) subject to constraints
  (This is a quadratic program in h! F(k) = sum h_i(1-h_{i-k}) is quadratic)
  Update Active = {k: F(k) >= max - epsilon}
```

The inner QP: minimize ∑_{k∈Active} ∑_i h_i(1-h_{i-k}) = ∑_i h_i - ∑_i ∑_{k∈Active} h_i h_{i-k}

This is a quadratic function of h. The Hessian would be:
H[i,j] = -∑_{k∈Active} [1 if j = i-k or j = i+k] (approximately)

This is a dense matrix, so solving the QP is O(n³) per iteration. For n=60, that's ~216K operations per iteration, which is fast.

But the QP might not be convex (the Hessian is not necessarily positive semidefinite). In that case, I can use a trust-region method or just take a line search step.

Actually, ∑ h_i(1-h_{i-k}) = ∑ h_i - ∑ h_i h_{i-k}
The Hessian of -∑ h_i h_{i-k} w.r.t. h is: -2*(off-diagonal terms for pairs (i, i-k))

This is generally indefinite, so the QP is non-convex. I'll use a trust-region approach:
1. Compute gradient g = ∇(sum_{active} F(k))
2. Solve the trust-region subproblem: min g^T d + (1/2) d^T H d, s.t. ||d|| ≤ Δ
3. Take the step
4. Update

This is essentially what L-BFGS-B does (approximately), but with the explicit Hessian structure.

I think the cleanest implementation is to just use L-BFGS-B with the "active set" objective:
minimize ∑_{k∈Active} F(k) (or the log-sum-exp over the active set)

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
import cvxpy as cp

def fast_correlate(h):
    n = len(h)
    N = 2 * n - 1
    H = fft(h, N)
    G = fft(1.0 - h, N)
    return ifft(np.conj(H) * G).real

def fast_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(fast_correlate(h)) * dx)

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        idx_order = np.argsort(h) if err > 0 else np.argsort(-h)
        for i in idx_order:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1:
                d = min(err, 1 - h[i]); h[i] += d; err -= d
            elif err < 0 and h[i] > 0:
                d = min(-err, h[i]); h[i] -= d; err += d
    return np.clip(h, 0, 1)

def lse_obj(h, beta):
    dx = 2.0 / len(h)
    C = fast_correlate(h) * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    ws = np.sum(np.exp(shifted - mk))
    if ws < 1e-300:
        return cmax + np.log(2 * len(h) - 1) / beta
    return cmax + (mk + np.log(ws)) / beta

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    C = fast_correlate(h) * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = es / ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        lo1, hi1 = max(0, s), min(n, n + s)
        if lo1 < hi1:
            if s >= 0:
                grad[lo1:hi1] += wk * (1.0 - h[:n - s])
            else:
                grad[lo1:hi1] += wk * (1.0 - h[-s:])
        lo2, hi2 = max(0, -s), min(n, n - s)
        if lo2 < hi2:
            if s >= 0:
                grad[lo2:hi2] += wk * (-h[s:])
            else:
                grad[lo2:hi2] += wk * (-h[:n + s])
    return grad * dx

def sum_eq(h):
    return float(np.sum(h) - len(h) / 2)

def sum_eq_jac(h):
    return np.ones(len(h))

def lse_optimize(h0, betas, maxiter_each=150, tol=1e-12):
    h = make_feasible(h0.copy())
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    for beta in betas:
        try:
            res = minimize(lse_obj, h, args=(beta,), jac=lse_grad,
                          bounds=bounds, constraints=cons, method='SLSQP',
                          options={'maxiter': maxiter_each, 'ftol': tol})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = make_feasible(res.x)
                if fast_score(h_new) <= fast_score(h) + 1e-10:
                    h = h_new
        except Exception:
            pass
    return h

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_s = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_t = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return make_feasible(np.interp(x_t, x_s, h_src))

def random_binary(n, rng):
    if n % 2 != 0:
        n += 1
    m = n // 2
    h = np.zeros(n)
    h[rng.choice(n, m, replace=False)] = 1.0
    return h

def structured_binary(n, typ, rng=None):
    if n % 2 != 0:
        n += 1
    m = n // 2
    h = np.zeros(n)
    if typ == 'alt':
        h[::2] = 1.0
    elif typ == 'block':
        h[:m] = 1.0
    elif typ == 'arithmetic':
        for step in [3, 5, 7, 11, 13]:
            if np.gcd(step, n) == 1:
                break
        h = ((np.arange(n) * step) % n) < m
    elif typ == 'golden':
        phi = (1 + np.sqrt(5)) / 2
        h = (np.arange(n) * phi % n) < m
    elif typ == 'two_block':
        h[:n // 4] = 1.0
        h[n // 2:3 * n // 4] = 1.0
    return h

def solve_milp(n, time_limit=20):
    if n % 2 != 0:
        n += 1
    m = n // 2
    h_var = cp.Variable(n, boolean=True)
    t_var = cp.Variable(nonneg=True)
    cons = [cp.sum(h_var) == m]
    for s in range(-(n - 1), n):
        vi = [i for i in range(n) if 0 <= i - s < n]
        if not vi:
            continue
        vj = [i - s for i in vi]
        yv = [cp.Variable(nonneg=True) for _ in vi]
        for i, j, y in zip(vi, vj, yv):
            cons += [y <= h_var[i], y <= h_var[j], y >= h_var[i] + h_var[j] - 1]
        cons.append(cp.sum(h_var[vi]) - cp.sum(yv) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.CBC, cp.SCIP, cp.GLPK_MI]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_val = np.round(h_var.value).astype(float)
    s = int(np.sum(h_val))
    if s != m:
        if s > m:
            idx = np.where(h_val == 1)[0][:s - m]
            h_val[idx] = 0
        else:
            idx = np.where(h_val == 0)[0][:m - s]
            h_val[idx] = 1
    return h_val

def nelder_mead_polish(h0, maxiter=200, step=0.05):
    from scipy.optimize import minimize as sp_min
    h0 = make_feasible(h0.copy())
    n = len(h0)
    x0 = h0.copy()
    x0[n - 1] = 0.0  # make it a free variable to use in optimization

    def obj(x):
        h = make_feasible(x)
        return fast_score(h)

    try:
        res = sp_min(obj, x0, method='Nelder-Mead',
                     options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10,
                              'initial_simplex': np.column_stack([x0, step * np.eye(n) * 0.1])})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0

    t_start = time.time()

    rng = np.random.default_rng(seed)
