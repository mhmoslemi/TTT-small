        n_lo = max(0, -s)
        n_hi = min(n - 1, n - 1 - s)
        if n_lo > n_hi:
            continue
        ix1 = np.arange(n_lo, n_hi + 1)
        ix2 = ix1 + s
        g_full[ix1] += wi * (1.0 - h[ix2])
        g_full[ix2] += wi * (-h[ix1])
    g_full *= dx
    m = (n - 1) // 2
    return g_full[:m] - g_full[n - m:n][::-1]


def subgradient_optimize(h0, n, max_iter=400, alpha0=0.08, decay_tau=80.0):
    m = (n - 1) // 2
    h_cur = h0.copy()
    best_local = h_cur.copy()
    best_score = official_score(build_symmetric(h_cur, n))
    for it in range(max_iter):
        h_full = build_symmetric(h_cur, n)
        F = np.correlate(h_full, 1.0 - h_full, mode='full')
        cmax = np.max(F)
        tol = 1e-4 * cmax + 1e-9
        active = np.where(F >= cmax - tol)[0]
        shifts = active - (n - 1)
        g_half = compute_sym_gradient(h_cur, n, shifts)
        alpha = alpha0 / np.sqrt(1.0 + it / decay_tau)
        h_new = np.clip(h_cur - alpha * g_half, 0.0, 1.0)
        h_cur = h_new
        sc = official_score(build_symmetric(h_cur, n))
        if sc < best_score - 1e-15:
            best_score = sc
            best_local = h_cur.copy()
    return best_local


def nelder_mead_optimize(h0, n, max_iter=800):
    m = (n - 1) // 2

    def obj(x):
        return official_score(build_symmetric(x, n))

    try:
        res = minimize(obj, h0, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10,
                                'initial_simplex': None})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h0


def slsqp_optimize(h0, n, beta=500, maxiter=200):
    m = (n - 1) // 2
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(sym_lse_obj, h0, args=(n, beta), jac=sym_lse_grad,
                       bounds=bounds, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h0


def thue_morse(m, rng):
    tm = np.array([bin(i).count('1') % 2 for i in range(m)], dtype=float)
    shift = rng.integers(0, m)
    tm = np.roll(tm, shift)
    return tm


def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return np.interp(x_tgt, x_src, h_src)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    best_h = None
    best_score = float('inf')

    def try_update(h):
        nonlocal best_h, best_score
        h = make_feasible(h)
        if not np.all(np.isfinite(h)):
            return
        sc = official_score(h)
        if np.isfinite(sc) and sc < best_score:
            best_h = h.copy()
            best_score = sc

    if parent is not None:
        try_update(parent)
        parent_n = len(parent)
    else:
        parent_n = 61
        best_h = make_feasible(np.full(parent_n, 0.5))
        best_score = official_score(best_h)

    odd_lengths = [41, 45, 49, 51, 53, 55, 57, 59, 61, 63, 65, 67, 69, 71, 73, 75, 77]
    for dl in [-6, -4, -2, 2, 4, 6]:
        nl = parent_n + dl
        if nl % 2 == 1 and 21 <= nl <= 80:
            if nl not in odd_lengths:
                odd_lengths.append(nl)
    odd_lengths = sorted(set(odd_lengths))

    # Prioritize lengths near parent
    odd_lengths.sort(key=lambda x: abs(x - parent_n))

    for n in odd_lengths:
        if time.time() > deadline - 40:
            break
        m = (n - 1) // 2

        inits = []

        if parent is not None:
            h_interp = interp_to_length(parent, parent_n, n)
            h_half_parent = project_to_symmetric(h_interp, n)
            inits.append(h_half_parent.copy())
            # Also try the "anti-symmetric" projection
            h_half_anti = 1.0 - (h_interp[:m] + 1.0 - h_interp[n - 1 - np.arange(m)]) / 2.0
            inits.append(np.clip(h_half_anti, 0, 1))

        # Thue-Morse
        inits.append(thue_morse(m, rng))
        inits.append(1.0 - thue_morse(m, rng))

        # Random
        for _ in range(2):
            inits.append(rng.uniform(0, 1, size=m))

        # Binary
        for _ in range(2):
            inits.append(rng.choice([0, 1], size=m).astype(float))

        # Smooth sine patterns
        for freq in [1, 2, 3]:
            x = np.linspace(0, 1, m)
            phase = rng.uniform(0, 2 * np.pi)
            inits.append(np.clip(0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase), 0, 1))

        # Cap initializations to save time
        max_inits = 8 if n > 60 else 10
        inits = inits[:max_inits]

        for h0 in inits:
            if time.time() > deadline - 30:
                break
            h0 = np.clip(h0, 0.0, 1.0)
            h_cur = h0.copy()

            # Phase 1: Subgradient
            h_cur = subgradient_optimize(h_cur, n, max_iter=300, alpha0=0.06, decay_tau=60.0)

            # Phase 2: SLSQP polish
            if time.time() < deadline - 10:
                h_cur = slsqp_optimize(h_cur, n, beta=400, maxiter=150)

            # Phase 3: Nelder-Mead for non-smooth max
            if time.time() < deadline - 10:
                h_cur2 = nelder_mead_optimize(h_cur, n, max_iter=500)
                sc2 = official_score(build_symmetric(h_cur2, n))
                sc1 = official_score(build_symmetric(h_cur, n))
                if sc2 < sc1:
                    h_cur = h_cur2

            # Re-polish with SLSQP after Nelder-Mead
            if time.time() < deadline - 8:
                h_cur = slsqp_optimize(h_cur, n, beta=800, maxiter=100)

            h_full = build_symmetric(h_cur, n)
            try_update(h_full)

    # Final refinement on best solution found
    if best_h is not None and time.time() < deadline - 10:
        bn = len(best_h)
        if bn % 2 == 1:
            m = (bn - 1) // 2
            h_half_best = project_to_symmetric(best_h, bn)
            # Fine SLSQP
            h_half_refined = slsqp_optimize(h_half_best, bn, beta=1500, maxiter=150)
            h_refined = build_symmetric(h_half_refined, bn)
            try_update(h_refined)
            # Tiny Nelder-Mead
            if time.time() < deadline - 5:
                h_half_nm = nelder_mead_optimize(h_half_refined, bn, max_iter=300)
                h_nm = build_symmetric(h_half_nm, bn)
                try_update(h_nm)

    # Try one more thing: slightly different length
    if best_h is not None and time.time() < deadline - 10:
        bn = len(best_h)
        for alt_n in [bn - 2, bn + 2]:
            if alt_n < 21 or alt_n > 80 or alt_n % 2 != 1:
                continue
            if time.time() > deadline - 5:
                break
            m = (alt_n - 1) // 2
            h_alt = interp_to_length(best_h, bn, alt_n)
            h_half_alt = project_to_symmetric(h_alt, alt_n)
            h_half_alt = subgradient_optimize(h_half_alt, alt_n, max_iter=200, alpha0=0.05, decay_tau=50.0)
            h_half_alt = slsqp_optimize(h_half_alt, alt_n, beta=600, maxiter=100)
            h_alt_full = build_symmetric(h_half_alt, alt_n)
            try_update(h_alt_full)

    if best_h is None:
        best_h = make_feasible(np.full(parent_n, 0.5))
        best_score = official_score(best_h)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score < 0:
        best_h = make_feasible(np.full(parent_n, 0.5))
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
