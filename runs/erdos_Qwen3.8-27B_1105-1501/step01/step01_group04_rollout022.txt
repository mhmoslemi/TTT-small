def project_exact(z, target):
    """Project z onto {x: 0<=x_i<=1, sum(x)=target} using water-filling."""
    n = len(z)
    z = np.array(z, dtype=float)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(100):

        if s > target:

        else:

            idx = np.argmax(1.0 - y)
        else:

        y = np.clip(y, 0.0, 1.0)

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_exact(z, target):
    z = np.asarray(z, dtype=np.float64)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) * 0.5
    y = np.clip(z - alpha, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-12:
        for _ in range(10):
            if r > 0:
                idx = np.argmin(y)
            else:
                idx = np.argmax(y)
            y[idx] += r
            y = np.clip(y, 0.0, 1.0)
            r = target - y.sum()
            if abs(r) < 1e-12:
                break
    return y


def line_search_pair(h, i, j):
    """Exact 1D search along h + t*(e_i - e_j). Returns (t_opt, h_opt)."""
    t_lo = max(h[j] - 1.0, -h[i])
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return 0.0, h.copy()

    best_t = 0.0
    best_h = h.copy()
    best_val = compute_c5(h)

    # Coarse grid
    ts = np.linspace(t_lo, t_hi, 12)
    for t in ts:
        h_t = h.copy()
        h_t[i] += t
        h_t[j] -= t
        v = compute_c5(h_t)
        if v < best_val:
            best_val = v
            best_t = t
            best_h = h_t.copy()

    # Refine around best
    spread = (t_hi - t_lo) / 12.0
    if best_t > t_lo and best_t < t_hi:
        ts2 = np.linspace(max(t_lo, best_t - spread), min(t_hi, best_t + spread), 10)
    else:
        ts2 = np.linspace(t_lo, t_hi, 6)
    for t in ts2:
        h_t = h.copy()
        h_t[i] += t
        h_t[j] -= t
        v = compute_c5(h_t)
        if v < best_val:
            best_val = v
            best_t = t
            best_h = h_t.copy()

    return best_t, best_h


def generate_directions(h, corr, rng, n_max=40):
    """Generate candidate (i,j) swap directions focused on active lags."""
    n = len(h)
    max_c = np.max(corr)
    directions = []
    seen = set()

    # Focus on lags with high correlation
    for lag in range(1, n):
        w_pos = corr[n - 1 + lag]
        w_neg = corr[n - 1 - lag]
        w = max(w_pos, w_neg)
        if w < max_c * 0.7:
            continue

        # Top contributing terms for positive lag
        m = np.arange(lag, n)
        terms_p = h[m] * (1.0 - h[m - lag])
        top_k = min(4, len(terms_p))
        for idx in np.argsort(terms_p)[-top_k:]:
            i = lag + idx
            j = i - lag
            if 0 <= i < n and 0 <= j < n and i != j and h[i] > 0.01 and h[j] < 0.99:
                key = (i, j)
                if key not in seen:
                    seen.add(key)
                    directions.append(key)

        # Top contributing terms for negative lag
        m2 = np.arange(0, n - lag)
        terms_n = h[m2] * (1.0 - h[m2 + lag])
        top_k2 = min(4, len(terms_n))
        for idx in np.argsort(terms_n)[-top_k2:]:
            i = idx
            j = idx + lag
            if 0 <= i < n and 0 <= j < n and i != j and h[i] > 0.01 and h[j] < 0.99:
                key = (i, j)
                if key not in seen:
                    seen.add(key)
                    directions.append(key)

        if len(directions) >= n_max * 2:
            break

    # Add random exploration directions
    for _ in range(8):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j and h[i] > 0.02 and h[j] < 0.98:
            key = (i, j)
            if key not in seen:
                seen.add(key)
                directions.append(key)

    # Add directions involving fractional values (push toward 0 or 1)
    frac = np.where((h > 0.05) & (h < 0.95))[0]
    for fi in frac[:6]:
        for _ in range(3):
            fj = rng.randint(n)
            if fj != fi and h[fj] > 0.05 and h[fi] < 0.95:
                key = (fj, fi)
                if key not in seen:
                    seen.add(key)
                    directions.append(key)
            if fj != fi and h[fi] > 0.05 and h[fj] < 0.95:
                key = (fi, fj)
                if key not in seen:
                    seen.add(key)
                    directions.append(key)

    return directions[:n_max]


def optimize_exact(h0, time_limit, seed=42):
    """Main optimization using exact line search on pair swaps."""
    n = len(h0)
    target = n / 2.0
    h = project_exact(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr_scaled(h)
    current_c5 = best_c5

    t0 = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    stall_limit = 400
    n_dirs = 35

    while time.time() - t0 < time_limit:
        improved = False
        directions = generate_directions(h, corr, rng, n_max=n_dirs)

        for (i, j) in directions:
            if time.time() - t0 >= time_limit:
                break
            t_opt, h_opt = line_search_pair(h, i, j)
            if abs(t_opt) < 1e-10:
                continue
            c5_new = compute_c5(h_opt)
            if c5_new < current_c5 - 1e-12:
                h = h_opt.copy()
                current_c5 = c5_new
                corr = compute_corr_scaled(h)
                improved = True
            if c5_new < best_c5 - 1e-12:
                best_h = h_opt.copy()
                best_c5 = c5_new

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                # Escape local optimum
                amplitude = 0.015
                h = best_h + amplitude * rng.randn(n)
                h = project_exact(h, target)
                current_c5 = compute_c5(h)
                corr = compute_corr_scaled(h)
                stall = 0
                n_dirs = max(20, n_dirs - 5)

    return best_h, best_c5


def random_init(n, rng):
    h = np.zeros(n)
    ones_count = n // 2
    h[:ones_count] = 1.0
    if n % 2 == 1:
        h[ones_count] = 0.5
    idx = rng.permutation(n)
    return h[idx]


def structured_init(n, rng):
    """Create a balanced structured initialization."""
    h = np.zeros(n)
    ones_count = n // 2
    # Spread ones roughly evenly
    positions = np.linspace(0, n - 1, ones_count, dtype=int)
    for p in positions:
        h[p % n] = 1.0
    if n % 2 == 1:
        remaining = 49.5 - h.sum() if n == 99 else n / 2.0 - h.sum()
        zero_pos = np.where(h == 0)[0]
        if len(zero_pos) > 0:
            h[zero_pos[0]] = min(1.0, remaining)
    idx = rng.permutation(n)
    return h[idx]


def resample_h(h_old, n_new):
    """Resample a step function to a new grid, preserving integral."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new

    # Create fine grid interpolation
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=0.0, right=0.0)

    # Adjust to meet integral constraint
    target = n_new / 2.0
    scale = target / (h_new * dx_new).sum() if h_new.sum() > 0 else 1.0
    h_new = h_new * scale
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_exact(h_new, target)
    return h_new
