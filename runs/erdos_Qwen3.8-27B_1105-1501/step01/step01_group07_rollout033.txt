def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-10:

            order = np.argsort(z)
            idx = order[0]
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            diff -= add
            if diff > 1e-10:

                z[idx] += min(diff, 1.0 - z[idx])

    m = float(np.max(scores_top))

**Key insight about the problem structure:**

The problem is: minimize max_k ∑ᵢ h[i]*(1-h[i+k])·dx subject to ∑h = n/2, 0≤h≤1.

For a given h, the overlap at lag k is essentially measuring how "misaligned" h is with its k-shift. If h is "periodic" with period p, then at lags that are multiples of p, the overlap is 0 (or very small). At other lags, it's larger.

import numpy as np
import time


def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(z)
            idx = int(order[0])
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            diff -= add
            if diff > 1e-10:
                idx = int(order[1])
                z[idx] += min(diff, 1.0 - z[idx])
        else:
            order = np.argsort(-z)
            idx = int(order[0])
            sub = min(-diff, z[idx])
            z[idx] -= sub
            diff += sub
            if diff < -1e-10:
                idx = int(order[1])
                z[idx] -= min(-diff, z[idx])
    return np.clip(z, 0.0, 1.0)


def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def compute_grad(h, w, n):
    g = np.zeros(n)
    c = n - 1
    wc = w[c]
    if wc > 1e-15:
        g += wc * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[c + d]
        wn = w[c - d]
        if wp < 1e-15 and wn < 1e-15:
            continue
        m = n - d
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g


def active_set_weights(scores, top_m, T):
    n = len(scores)
    if top_m >= n:
        m = float(np.max(scores))
        ex = np.exp((scores - m) / T)
        return ex / ex.sum()
    idx = np.argsort(scores)[-top_m:]
    w = np.zeros(n)
    scores_top = scores[idx]
    m = float(np.max(scores_top))
    ex = np.exp((scores_top - m) / T)
    w[idx] = ex / ex.sum()
    return w


def pair_refine(h, i, j, cur_c, iters=25):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = lo + (1.0 - gr) * (hi - lo)
    a2 = lo + (hi - lo) * gr
    h[i] = orig_i + a1
    h[j] = orig_j - a1
    s1 = score_exact(h)
    h[i] = orig_i + a2
    h[j] = orig_j - a2
    s2 = score_exact(h)
    best_s = cur_c
    best_a = 0.0
    for _ in range(iters):
        if s1 < s2:
            if s1 < best_s:
                best_s = s1
                best_a = a1
            hi = a2
            a2 = a1
            a1 = lo + (1.0 - gr) * (hi - lo)
            h[i] = orig_i + a1
            h[j] = orig_j - a1
            s1 = score_exact(h)
        else:
            if s2 < best_s:
                best_s = s2
                best_a = a2
            lo = a1
            a1 = a2
            a2 = lo + (hi - lo) * gr
            h[i] = orig_i + a2
            h[j] = orig_j - a2
            s2 = score_exact(h)
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None


def projected_gradient(h, deadline, T_start, T_end, T_decay, step0, rng, top_m=8):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    iters = 0
    while time.time() < deadline:
        scores = np.correlate(cur, 1.0 - cur, mode='full') * dx
        w = active_set_weights(scores, top_m, T)
        g = compute_grad(cur, w, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 2.0:
                break
            continue
        direction = -g / g_norm
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 2.0, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.7
            stall += 1
            if stall > 40:
                cur = make_feasible(best + rng.normal(0, max(0.008, T * 0.5), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.008)
                stall = 0
        else:
            step = min(step * 1.15, 0.3)
            stall = 0
        T = max(T_end, T * T_decay)
        iters += 1
    return best, best_c


def gradient_guided_pairs(h, cur_c, n, top_m, n_trials, deadline):
    dx = 2.0 / n
    c = n - 1
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        w = active_set_weights(scores, top_m, 0.005)
        g = compute_grad(h, w, n)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if i == j or g[i] < 1e-10 or -g[j] < 1e-10:
            break
        res = pair_refine(h, i, j, cur_c, iters=30)
        if res is None:
            # Try second-best pair
            g2 = g.copy()
            g2[i] = -np.inf
            i2 = int(np.argmax(g2))
            g3 = g.copy()
            g3[j] = np.inf
            j2 = int(np.argmin(g3))
            if i2 != j2 and g[i2] > 1e-10 and -g[j2] > 1e-10:
                res = pair_refine(h, i2, j2, cur_c, iters=25)
        if res is not None:
            cur_c = res
    return cur_c

def smooth_h(h, n, sigma_idx):
    kernel_size = int(2 * sigma_idx + 1)
    if kernel_size < 3:
        return h.copy()
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / max(sigma_idx, 0.5)) ** 2)
    kernel /= kernel.sum()
    h_padded = np.pad(h, (kernel_size // 2, kernel_size // 2), mode='edge')
    h_smooth = np.convolve(h_padded, kernel, mode='valid')[:n]
    return make_feasible(h_smooth, n)


def fourier_sample(params, n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    K = len(params) // 2
    h = np.full(n, 0.5)
    for k in range(K):
        h += params[2 * k] * np.cos(np.pi * (k + 1) * x)
        h += params[2 * k + 1] * np.sin(np.pi * (k + 1) * x)
    return make_feasible(h, n)
