```python
def compute_grad_matrix(h):
    n = len(h)
    R = 2 * n - 1
    G = np.zeros((R, n))
    for k in range(R):
        s = k - (n - 1)
        if s >= 0:
            # c[k] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # ∂c[k]/∂h[p] = (1-h[p-s]) if s <= p <= n-1
            #                - h[p+s] if 0 <= p <= n-1-s
            for p in range(s, n):
                G[k, p] += (1.0 - h[p - s])
            for p in range(0, n - s):
                G[k, p] -= h[p + s]
        else:
            # s < 0, let ss = -s > 0
            # c[k] = sum_{i=ss}^{n-1} h[i-ss]*(1-h[i])
            # ∂c[k]/∂h[p] = (1-h[p+ss]) if 0 <= p <= n-1-ss
            #                - h[p-ss] if ss <= p <= n-1
            ss = -s
            for p in range(0, n - ss):
                G[k, p] += (1.0 - h[p + ss])
            for p in range(ss, n):
                G[k, p] -= h[p - ss]
    return G
```

```python
def majorize_step(h0, rho, active_rows=None, solver=None):
    n = len(h0)
    dx = 2.0 / n
    R = 2 * n - 1
    
    # Compute current correlations and gradients
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    G = compute_grad_matrix(h0)
    
    # Select active rows
    if active_rows is None:
        active_rows = range(R)
    
    # Build CVXPY problem
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(delta) == 0,
        delta >= -h0,
        delta <= 1.0 - h0,
        cp.sum_squares(delta) <= rho**2
    ]
    
    # Add row constraints
    # r_k(h0) + G[k]·delta + eta * ||delta||² <= t
    # where eta = 1 (max degree / 2, and max degree is 2)
    eta = 1.0  # safe upper bound
    sq = cp.sum_squares(delta)
    
    for k in active_rows:
        constraints.append(corr[k] + G[k, :] @ delta + eta * sq <= t)
    
    # Objective: minimize t (scaled) + small regularization
    objective = cp.Minimize(t / dx + 0.01 * sq)
    
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        if solver:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception:
        return None
```

```python
def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:

            else:

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_grad_matrix(h):
    n = len(h)
    R = 2 * n - 1
    G = np.zeros((R, n))
    for k in range(R):
        s = k - (n - 1)
        if s >= 0:
            for p in range(s, n):
                G[k, p] += (1.0 - h[p - s])
            for p in range(0, n - s):
                G[k, p] -= h[p + s]
        else:
            ss = -s
            for p in range(0, n - ss):
                G[k, p] += (1.0 - h[p + ss])
            for p in range(ss, n):
                G[k, p] -= h[p - ss]
    return G

def compute_degrees(n):
    R = 2 * n - 1
    degs = np.zeros((R, n))

        s = k - (n - 1)
        if s >= 0:

                degs[k, p] += 1.0
            for p in range(0, n - s):
                degs[k, p] += 1.0
        else:
            ss = -s
            for p in range(0, n - ss):
                degs[k, p] += 1.0
            for p in range(ss, n):
                degs[k, p] += 1.0
    return degs

def majorize_step(h0, rho, max_rows=None, solver_list=None):
    n = len(h0)
    R = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    G = compute_grad_matrix(h0)
    degs = compute_degrees(n)
    eta = np.max(degs, axis=1) / 2.0
    eta = np.maximum(eta, 0.5)

    if max_rows is not None and max_rows < R:
        order = np.argsort(-corr)
        row_idx = order[:max_rows]
        row_idx = np.unique(row_idx)
    else:
        row_idx = np.arange(R)

    delta = cp.Variable(n)
    t = cp.Variable()
    sq = cp.sum_squares(delta)
    constraints = [
        cp.sum(delta) == 0,
        delta >= -h0 - 1e-10,
        delta <= 1.0 - h0 + 1e-10,
        sq <= rho * rho
    ]
    for k in row_idx:
        constraints.append(corr[k] + G[k, :] @ delta + eta[k] * sq <= t)

    objective = cp.Minimize(t + 0.05 * sq)
    prob = cp.Problem(objective, constraints)

    solvers = solver_list if solver_list else ['ECOS', 'CLARABEL', 'MOSEK', 'SCS', 'PDLP']
    last_err = None

    max_c = np.max(corr)

    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta
```
