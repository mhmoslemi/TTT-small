import numpy as np
from scipy.optimize import minimize
import time

def project_to_target(v, target):
    """Water-filling projection to [0,1]^n with exact sum = target."""
    v = np.nan_to_num(v.astype(float), nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    result = np.clip(v - theta, 0.0, 1.0)
    # Exact sum correction
    residual = float(np.sum(result)) - target
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(result)[::-1]
        else:
            order = np.argsort(result)
        for i in order:
            if residual > 0:
                if result[i] > 0:
                    dec = min(residual, result[i])
                    result[i] -= dec
                    residual -= dec
            else:
                if result[i] < 1.0:
                    inc = min(-residual, 1.0 - result[i])
                    result[i] += inc
                    residual += inc
            if abs(residual) < 1e-14:
                break
    return result

def c5_official(h):
    """Compute the exact C5 value."""
    n = len(h)
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_weights(F, n_active_frac=0.15):
    """Compute soft active-set weights for the correlation vector."""
    L = len(F)
    raw_max = float(np.max(F))
    # Use top fraction as "active"
    n_active = max(4, int(L * n_active_frac))
    threshold = np.sort(F)[-n_active]
    active_mask = F >= threshold
    if not np.any(active_mask):
        active_mask = np.zeros(L, dtype=bool)
        active_mask[np.argmax(F)] = True
    weights = np.zeros(L)
    # Soft weight: exponential decay from the max
    spread = max(raw_max - threshold, 1e-12)
    beta = 8.0 / spread
    weights[active_mask] = np.exp(-beta * (raw_max - F[active_mask]))
    total = float(np.sum(weights))
    if total < 1e-15:
        weights[active_mask] = 1.0
        total = float(np.sum(weights))
    weights /= total
    return weights

def compute_Q(h, g, weights, n):
    """Compute Q vector for direction scoring.
    Q[t] = sum_K weights[K] * (g[t+n-1-K] - h[t+K-n+1]) [with bounds -> 0]
    """
    L = 2 * n - 1
    Q = np.zeros(n)
    for t in range(n):
        s = 0.0
        for K in range(L):
            # g index: t + n - 1 - K
            gi = t + n - 1 - K
            if 0 <= gi < n:
                s += weights[K] * g[gi]
            # h index: t + K - n + 1
            hi2 = t + K - n + 1
            if 0 <= hi2 < n:
                s -= weights[K] * h[hi2]
        Q[t] = s
    return Q

def line_search(F, A, R, T, n_coarse=30, n_fine=12):
    """Find delta in [0, T] minimizing max(F + delta*A - delta^2*R).
    Returns (best_delta, best_value)."""
    if T < 1e-14:
        return 0.0, float(np.max(F))
    
    # Coarse grid
    deltas = np.linspace(0, T, n_coarse)
    best_val = float('inf')
    best_d = 0.0
    vals_arr = np.empty(n_coarse)
    for idx, d in enumerate(deltas):
        v = float(np.max(F + d * A - d * d * R))
        vals_arr[idx] = v
        if v < best_val:
            best_val = v
            best_d = d
    
    # Fine grid around best
    step = T / (n_coarse - 1)
    lo = max(0.0, best_d - 2 * step)
    hi = min(T, best_d + 2 * step)
    if hi > lo:
        fine_deltas = np.linspace(lo, hi, n_fine)
        for d in fine_deltas:
            v = float(np.max(F + d * A - d * d * R))
            if v < best_val:
                best_val = v
                best_d = d
    
    return best_d, best_val

def local_exchange_pass(h, F, weights, n, max_cand=50, grid_coarse=30, grid_fine=12, rng=None):
    """One pass of local exchange. Returns (h_new, improved, c5_raw_old, c5_raw_new)."""
    g = 1.0 - h
    raw_max = float(np.max(F))
    
    # Compute Q for direction scoring
    Q = compute_Q(h, g, weights, n)
    
    # Sort by Q: low Q = good target (receive mass), high Q = good donor (give mass)
    order = np.argsort(Q)
    n_sel = min(15, n // 3)
    targets = order[:n_sel]
    donors = order[::-1][:n_sel]
    
    # Build candidate list
    candidates = []
    for i in targets:
        for j in donors:
            if i != j:
                candidates.append((int(i), int(j)))
    
    # Add some random candidates for diversity
    if rng is not None:
        n_rand = max(5, n_sel // 2)
        for _ in range(n_rand):
            ri = int(rng.integers(0, n))
            rj = int(rng.integers(0, n))
            if ri != rj:
                candidates.append((ri, rj))
    
    # Cap candidates
    candidates = candidates[:max_cand]
    
    # Evaluate each candidate
    best_val = raw_max
    best_d = 0.0
    best_i = -1
    best_j = -1
    
    for (i, j) in candidates:
        T = min(1.0 - h[i], h[j])
        if T < 1e-12:
            continue
        
        # Compute A and R
        p = np.zeros(n)
        p[i] = 1.0
        p[j] = -1.0
        A = np.correlate(p, g, mode='full') - np.correlate(h, p, mode='full')
        R = np.correlate(p, p, mode='full')
        
        # Line search
        d, val = line_search(F, A, R, T, grid_coarse, grid_fine)
        if val < best_val and d > 1e-12:
            best_val = val
            best_d = d
            best_i = i
            best_j = j
    
    if best_i < 0:
        return h, False, raw_max, raw_max
    
    # Apply move
    h_new = h.copy()
    h_new[best_i] += best_d
    h_new[best_j] -= best_d
    h_new = np.clip(h_new, 0.0, 1.0)
    
    return h_new, True, raw_max, best_val

def symmetrize(h):
    """Try symmetrization and return best variant."""
    h_rev = h[::-1].copy()
    h_sym = 0.5 * (h + h_rev)
    h_sym = project_to_target(h_sym, len(h) / 2.0)
    
    best_c5 = c5_official(h)
    best_h = h
    for candidate in [h_rev, h_sym]:
        c5 = c5_official(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate
    return best_h, best_c5

def sharpen(h, alpha):
    """Push values toward 0 or 1."""
    n = len(h)
    h_sharp = h.copy()
    below = h < 0.5
    above = ~below
    h_sharp[below] = h[below] ** alpha
    h_sharp[above] = 1.0 - (1.0 - h[above]) ** alpha
    h_sharp = project_to_target(h_sharp, n / 2.0)
    return h_sharp

def resample(h, new_n):
    """Resample h to new_n points via linear interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_to_target(h_new, new_n / 2.0)
    return h_new

1. The linear max (which increases with δ if any A[k] > 0 for active k)
2. The concave quadratic at the center (which increases then decreases)
3. The two convex quadratics (which generally increase)

import numpy as np
from scipy.optimize import minimize

def project_to_target(v, target):
    v = np.nan_to_num(np.asarray(v, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    result = np.clip(v - theta, 0.0, 1.0)
    residual = float(np.sum(result)) - target
    if abs(residual) > 1e-12:
        order = np.argsort(result)[::-1] if residual > 0 else np.argsort(result)
        for idx in order:
            if residual > 0 and result[idx] > 0:
                d = min(residual, result[idx])
                result[idx] -= d
                residual -= d
            elif residual < 0 and result[idx] < 1.0:
                d = min(-residual, 1.0 - result[idx])
                result[idx] += d
                residual += d
            if abs(residual) < 1e-14:
                break
    return result


def c5_official(h):
    n = len(h)
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def get_weights(F):
    L = len(F)
    raw_max = float(np.max(F))
    n_active = max(4, int(L * 0.15))
    sorted_F = np.sort(F)
    threshold = sorted_F[-n_active]
    spread = max(raw_max - threshold, 1e-12)
    beta = 8.0 / spread
    shifted = np.clip(beta * (F - raw_max), -300.0, 0.0)
    weights = np.exp(shifted)
    total = float(np.sum(weights))
    if total < 1e-15:
        weights = np.zeros(L)
        top_idx = np.argpartition(F, -4)[len(F) - 4:]
        weights[top_idx] = 1.0
        weights /= 4.0
    else:
        weights /= total
    return weights


def compute_Q_fast(h, g, weights, n):
    L = 2 * n - 1
    K = np.arange(L)
    Q = np.zeros(n)
    for t in range(n):
        gi = t + n - 1 - K
        hi = t + K - n + 1
        val = np.zeros(L)
        m1 = (gi >= 0) & (gi < n)
        m2 = (hi >= 0) & (hi < n)
        if np.any(m1):
            val[m1] += g[gi[m1]]
        if np.any(m2):
            val[m2] -= h[hi[m2]]
        Q[t] = float(np.dot(weights, val))
    return Q

    R = np.zeros(L)
    R[n - 1] = 2.0

def line_search(F, A, R, T):
    if T < 1e-14:
        return 0.0, float(np.max(F))
    n_coarse = 30
    n_fine = 15
    deltas = np.linspace(0.0, T, n_coarse)
    best_val = float('inf')
    best_d = 0.0
    for d in deltas:
        v = float(np.max(F + d * A - d * d * R))
        if v < best_val:
            best_val = v
            best_d = d
    step = T / (n_coarse - 1)
    lo = max(0.0, best_d - 2.0 * step)
    hi = min(T, best_d + 2.0 * step)
    if hi - lo > 1e-14:
        fine_deltas = np.linspace(lo, hi, n_fine)
        for d in fine_deltas:
            v = float(np.max(F + d * A - d * d * R))
            if v < best_val:
                best_val = v
                best_d = d
    return best_d, best_val


def local_exchange_pass(h, n, max_cand, rng):
    g = 1.0 - h
    F = np.correlate(h, g, mode='full')
    raw_max = float(np.max(F))
    weights = get_weights(F)
    Q = compute_Q_fast(h, g, weights, n)
    order = np.argsort(Q)
    n_sel = min(max(8, n // 4), 20)
    targets = order[:n_sel]
    donors = order[::-1][:n_sel]
    candidates = []
    for i in targets:
        for j in donors:
            if i != j:
                candidates.append((int(i), int(j)))
    n_rand = max(5, n_sel // 3)
    for _ in range(n_rand):
        ri = int(rng.integers(0, n))
        rj = int(rng.integers(0, n))
        if ri != rj:
            candidates.append((ri, rj))
    candidates = candidates[:max_cand]
    best_val = raw_max
    best_d = 0.0
    best_i = -1
    best_j = -1
    for (i, j) in candidates:
        T = min(1.0 - h[i], h[j])
        if T < 1e-12:
            continue
        A, R = compute_A_R_fast(i, j, h, g, n)
        d, val = line_search(F, A, R, T)
        if d > 1e-12 and val < best_val - 1e-12:
            best_val = val
            best_d = d
            best_i = i
            best_j = j
    if best_i < 0:
        return h, False, raw_max, raw_max
    h_new = h.copy()
    h_new[best_i] += best_d
    h_new[best_j] -= best_d
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, True, raw_max, best_val


def run_exchange_opt(h_init, max_time, seed):
    t_start = time.time()
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_raw = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 40
    while no_improve < max_no_improve:
        if time.time() - t_start > max_time:
            break
        h_new, improved, old_raw, new_raw = local_exchange_pass(h, n, 60, rng)
        if improved and new_raw < best_raw - 1e-10:
            h = h_new
            best_raw = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                sigma = 0.02 * (1.0 + no_improve * 0.1)
                h_pert = h + rng.normal(0, min(sigma, 0.15), n)
                h_pert = project_to_target(h_pert, n * 0.5)
                raw_pert = float(np.max(np.correlate(h_pert, 1.0 - h_pert, mode='full')))
                if raw_pert < best_raw:
                    h = h_pert
                    best_raw = raw_pert
                    best_h = h.copy()
                    no_improve = 0
    return best_h, best_raw * dx


def symmetrize_h(h):
    n = len(h)
    h_rev = h[::-1].copy()
    h_sym = 0.5 * (h + h_rev)
    h_sym = project_to_target(h_sym, n * 0.5)
    best_h = h.copy()
    best_c5 = c5_official(h)
    for cand in [h_rev, h_sym]:
        c5c = c5_official(cand)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = cand
    return best_h, best_c5


def sharpen_h(h, alpha):
    n = len(h)
    h_s = h.copy()
    below = h < 0.5
    h_s[below] = h[below] ** alpha
    h_s[~below] = 1.0 - (1.0 - h[~below]) ** alpha
    return project_to_target(h_s, n * 0.5)


def resample_to(h, new_n):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, new_n * 0.5)

        try:
            res = minimize(obj, best_x, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.success or float(res.fun) < best_val:
                rx = project_to_target(res.x, target)
                cv = obj(rx)
                if cv < best_val:
                    best_x = rx
                    best_val = cv
        except Exception:
            pass
    return best_x, best_val


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s * 0.98

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project_to_target(h0, n0 * 0.5)
    best_h = h.copy()
    best_c5 = c5_official(h)

    h_sym, c5_sym = symmetrize_h(best_h)
    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_sym

    # Phase 1: Main exchange optimization
    time_allocate = budget_s * 0.50
    h_opt, c5_opt = run_exchange_opt(best_h, time_allocate, seed=seed)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Symmetrize after main optimization
    h_sym, c5_sym = symmetrize_h(best_h)
    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_sym

    # Phase 2: Sharpening + re-optimize
    remaining = t_end - time.time()
    if remaining > 20:
        for alpha in [2.0, 1.5, 3.0, 4.0]:
            if time.time() > t_end:
                break
            h_sh = sharpen_h(best_h, alpha)
            c5_sh = c5_official(h_sh)
            if c5_sh < best_c5:
                best_c5 = c5_sh
                best_h = h_sh
                rem = t_end - time.time()
                if rem > 10:
                    h_o2, c5_o2 = run_exchange_opt(best_h, min(rem * 0.5, 60.0), seed=seed + 100)
                    if c5_o2 < best_c5:
                        best_c5 = c5_o2
                        best_h = h_o

    # Phase 3: SLSQP polish
    remaining = t_end - time.time()
    if remaining > 15:
        h_pol, raw_pol = slsqq_polish(best_h, len(best_h), len(best_h) * 0.5, maxiter=300)
        c5_pol = c5_official(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 4: Multi-resolution search
    remaining = t_end - time.time()
    if remaining > 30:
        n_candidates = [n0 + 1, n0 - 1, n0 + 2, n0 + 5, 64, 72, 80, 96, 48, 56]
        n_candidates = [x for x in n_candidates if
                        16 <= x <= 200 and x != n0]
        n_seen = set()
        n_unique = []
        for x in n_candidates:
            if x not in n_seen:
                n_seen.add(x)
                n_unique.append(x)
        time_per = min(remaining * 0.7 / max(len(n_unique), 1), 30.0)
        for nn in n_unique:
            if time.time() > t_end:
                break
            h_res = resample_to(best_h, nn)
            c5_res = c5_official(h_res)
            rem = t_end - time.time()
            if rem < 3:
                break
            t_alloc = min(time_per, rem * 0.4)
            h_o3, c5_o3 = run_exchange_opt(h_res, t_alloc, seed=seed + 300 + nn)
            if c5_o3 < best_c5:
                best_c5 = c5_o3
                best_h = h_o3

    # Phase 5: Final exchange optimization
    remaining = t_end - time.time()
    if remaining > 5:
        h_fin, c5_fin = run_exchange_opt(best_h, remaining * 0.9, seed=seed + 700)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final SLSQP polish
    remaining = t_end - time.time()
    if remaining > 5:
        h_pol2, raw_pol2 = slsqq_polish(best_h, len(best_h), len(best_h) * 0.5, maxiter=100)
        c5_pol2 = c5_official(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2

    # Final cleanup
    best_h = project_to_target(best_h, len(best_h) * 0.5)
    best_c5 = c5_official(best_h)
    return best_h, best_c5, int(len(best_h))
