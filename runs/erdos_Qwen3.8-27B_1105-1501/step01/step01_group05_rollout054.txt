from scipy import optimize
import time

def official_score(h):

    if n == 0:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_to_target(v, n_target):
    h = np.array(v, dtype=float).ravel()[:n_target].copy()
    if len(h) < n_target:
        h = np.concatenate([h, np.full(n_target - len(h), 0.5)])
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    lo = -max(1.0, float(np.max(h))) - 0.5
    hi = 1.5 - float(np.min(h))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(5000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
            d = min(residual, 1.0 - h[idx])
            if d <= 0:
                break
            h[idx] += d
        else:
            idx = int(np.argmax(h))
            d = min(-residual, h[idx])
            if d <= 0:
                break
            h[idx] -= d
    return np.clip(h, 0.0, 1.0)

def grad_row(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += dx * (1.0 - h[s:])
        g[s:] -= dx * h[:n-s]
    else:
        a = -s
        g[a:] += dx * (1.0 - h[:n-a])
        g[:n-a] -= dx * h[a:]
    return g

def build_G(h):
    """Build full gradient matrix (2n-1) x n"""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        s = r - (n-1)
        G[r] = grad_row(h, s)
    return G

def lp_direction(h, G, corr_scaled):
    """Find search direction via LP"""
    n = len(h)
    L = 2*n - 1
    max_val = np.max(corr_scaled)
    margins = max_val - corr_scaled
    
    # Active set
    tol = max(1e-6, 0.01 * max_val)
    active = np.where(margins <= tol)[0]
    if len(active) == 0:
        return None
    
    # Take top-a few active for hard constraints
    max_hard = min(len(active), 15)
    top_active = active[np.argsort(margins[active])[:max_hard]]
    
    # LP variables: d+ (n), d- (n), rho (1)
    nvar = 2*n + 1
    c_obj = np.zeros(nvar)
    c_obj[2*n] = -1.0
    
    A_rows = []
    b_vals = []
    
    # Hard: G[r] d <= -rho for top active
    for r in top_active:
        row = np.zeros(nvar)
        row[:n] = G[r]
        row[n:2*n] = -G[r]
        row[2*n] = 1.0
        A_rows.append(row)
        b_vals.append(0.0)
    
    # Soft: G[r] d <= margins[r] + tol for all others
    for r in range(L):
        if r not in top_active:
            row = np.zeros(nvar)
            row[:n] = G[r]
            row[n:2*n] = -G[r]
            A_rows.append(row)
            b_vals.append(margins[r] + tol)
    
    # L1 norm: sum(d+ + d-) <= 1
    row = np.zeros(nvar)
    row[:2*n] = 1.0
    A_rows.append(row)
    b_vals.append(1.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.array([0.0])
    
    bounds = [(0.0, None)] * nvar
    
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                               A_eq=A_eq, b_eq=b_eq,
                               bounds=bounds, method='highs')
        if res.status == 0 and res.fun < -1e-10:
            d = res.x[:n] - res.x[n:2*n]
            if np.max(np.abs(d)) > 1e-10:
                return d
    except Exception:
        pass
    return None

def gradient_direction(h, G, corr_scaled):
    """Weighted gradient direction (fallback when LP fails)"""
    n = len(h)
    L = 2*n - 1
    max_val = np.max(corr_scaled)
    margins = max_val - corr_scaled
    tol = max(1e-6, 0.02 * max_val)
    weights = np.zeros(L)
    active = margins <= tol
    weights[active] = (tol - margins[active] + 0.01) ** 2
    d = -G.T @ weights
    d -= np.mean(d)
    norm = np.max(np.abs(d))
    if norm > 1e-12:
        d /= norm
    return d

def local_search_lp(h, n, max_iters=200, time_limit=120.0, seed=42):
    """Main local search using LP directions"""
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    dx = 2.0 / n
    L = 2*n - 1
    start = time.perf_counter()
    n_stuck = 0
    
    alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.05, 0.15]
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        score = float(np.max(corr))
        
        # Get direction
        G = build_G(h)
        d = lp_direction(h, G, corr)
        if d is None:
            d = gradient_direction(h, G, corr)
            if d is None or np.max(np.abs(d)) < 1e-12:
                # Add random perturbation
                d = rng.normal(0, 1, n)
                d -= np.mean(d)
                d /= np.max(np.abs(d))
        
        # Line search
        best_alpha = 0.0
        best_score = score
        for sign in [1, -1]:
            for a in alphas:

                h_new = project_to_target(h + alpha * d, n)
                s_new = official_score(h_new)

                    best_alpha = alpha

                n_stuck = 0

def discrete_sa(n, max_iters=50000, time_limit=120.0, seed=42):
    """Simulated annealing for binary sequences of even length n"""
    if n % 2 != 0:
        return None, None
    m = n // 2
    rng = np.random.default_rng(seed)
    
    # Initial state: random balanced
    h = np.zeros(n)
    ones = rng.choice(n, size=m, replace=False)
    h[ones] = 1.0
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    dx = 2.0 / n
    
    T = 0.05
    T_min = 0.001
    cooling = (T_min / T) ** (1.0 / max(1, max_iters))
    
    start = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        # Random swap: swap a 1 and a 0
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_score = official_score(h)
        
        delta = new_score - score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cooling
    
    return best_h, best_score

def slsqp_polish(h, n, time_limit=30.0):
    """NLP polish using SLSQP"""
    h = project_to_target(h, n)
    dx = 2.0 / n
    target_sum = n / 2.0
    L = 2 * n - 1
    lags = np.arange(-(n-1), n)
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def c_ineq(x):
        hx = x[:n]
        theta = x[n]
        corr = np.correlate(hx, 1.0 - hx, mode="full") * dx
        return theta - corr
    
    def c_ineq_jac(x):
        hx = x[:n]
        J = np.zeros((L, n+1))
        J[:, n] = 1.0
        for r in range(L):
            s = r - (n-1)
            J[r, :n] = -grad_row(hx, s)
        return J
    
    def c_eq(x):
        return np.sum(x[:n]) - target_sum
    
    def c_eq_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac},
        {'type': 'ineq', 'fun': c_ineq, 'jac': c_ineq_jac}
    ]
    
    start = time.perf_counter()
    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 100, 'ftol': 1e-10})
        if time.perf_counter() - start > time_limit:
            raise TimeoutError
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_out = project_to_target(res.x[:n], n)
            s_out = official_score(h_out)
            if s_out < initial_score - 1e-10:
                return h_out, s_out
    except Exception:
        pass
    return h, initial_score

def interpolate_to_n(parent, n_target, shift=0):
    """Interpolate parent to target length"""
    n_p = len(parent)
    if n_p == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = ((np.arange(n_p) + 0.5) / n_p) * 2.0
    xt = (((np.arange(n_target) + 0.5 + shift) % n_target) / n_target) * 2.0
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)

        n_parent = len(parent)

    candidates.append((1.0 - p[::-1]).copy())
    candidates.append((p + p[::-1]) / 2.0)
    
    # Interpolate to nearby resolutions
    n_vals = sorted(set([n_parent, n_parent-1, n_parent+1,
                         n_parent-4, n_parent+4,
                         n_parent+8, max(48, n_parent-8)]))
    for n_try in n_vals:
        if 32 <= n_try <= 130:
            for shift in [0, 1, 2]:
                candidates.append(interpolate_to_n(parent, n_try, shift))

            candidates.append(project_to_target(v, n_try))

    scored = []

        if np.isfinite(s):
