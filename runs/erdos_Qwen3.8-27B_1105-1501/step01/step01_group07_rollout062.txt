import numpy as np
import time
from scipy.optimize import linprog

def project_feasible(y, target):
    """Project y onto {0<=z<=1, sum(z)=target} using bisection."""
    n = len(y)
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(z)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(-z)
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return z

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def score_components(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad_matrix(h):
    """Compute G where G[m,j] = d(c[m])/d(h[j]), c = correlate(h,1-h,'full')*dx."""
    n = len(h)
    dx = 2.0 / n
    nm1 = 2 * n - 1
    G = np.zeros((nm1, n))
    for d in range(-(n - 1), n):
        m = d + (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[m, idx] += dx * (1.0 - h[idx + d])
        G[m, idx + d] += dx * (-h[idx])
    return G

def one_lp_step(h0):
    """Solve one LP cutting plane step. Returns h_new or None."""
    n = len(h0)
    target = n / 2.0
    nm1 = 2 * n - 1
    
    c_vals = score_components(h0)
    G = compute_grad_matrix(h0)
    
    # LP: min t s.t. G[m,:]@h - t <= b_m, 1@h = target, 0<=h<=1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((nm1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = c_vals - G @ h0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def lp_phase(h0, deadline, max_no_improve=50):
    """LP cutting plane phase."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    c = score_exact(h)
    stall = 0
    
    while time.time() < deadline:
        h_lp = one_lp_step(h)
        if h_lp is None:
            stall += 1
            if stall > max_no_improve:
                break
            continue
        
        # Line search: try t in [0,1]
        best_t = 0.0
        best_c = c
        for t in np.linspace(0.0, 1.0, 11):
            cand = project_feasible(h + t * (h_lp - h), target)
            c_new = score_exact(cand)
            if c_new < best_c:
                best_c = c_new
                best_t = t
                h = cand
        
        if best_t == 0.0:
            stall += 1
            if stall > max_no_improve:
                # Perturb and restart
                h = project_feasible(h + np.random.randn(n) * 0.01, target)
                c = score_exact(h)
                stall = 0
        else:
            c = best_c
            stall = 0
    
    return h, c

def subgrad_phase(h0, deadline, rng):
    """Subgradient method with active set."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    c = score_exact(h)
    alpha = 0.02
    tol = 0.001
    
    while time.time() < deadline:
        c_vals = score_components(h)
        max_c = float(np.max(c_vals))
        # Active set: within tol of max
        active = np.where(c_vals > max_c - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_vals))])
        
        # Average subgradient from active set
        G = compute_grad_matrix(h)
        g = G[active].mean(axis=0)
        
        # Step
        step_size = alpha / (1.0 + 0.01 * (deadline - time.time()))
        # Actually use decreasing step
        # h_new = h - step_size * g
        # Project
        h_new = project_feasible(h - step_size * g, target)
        c_new = score_exact(h_new)
        
        if c_new < c:
            h = h_new
            c = c_new
            alpha = min(alpha * 1.01, 0.1)
        else:
            alpha *= 0.9
            if alpha < 1e-6:
                break
    
    return h, c

def pair_search(h, i, j, cur_c, n_pts=30):
    """Try to improve by adjusting h[i] and h[j] while preserving sum."""
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        # Golden section refinement
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(20):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:

    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[m, idx] += dx * (1.0 - h[idx + s])
        G[m, idx + s] += dx * (-h[idx])
    return G

def one_lp_step(h0):
    n = len(h0)
    target = n / 2.0
    nm1 = 2 * n - 1
    c_vals = score_components(h0)
    G = compute_grad_matrix(h0)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((nm1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = c_vals - G @ h0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_phase(h0, deadline, rng, max_stall=80):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    c = score_exact(h)
    best_h = h.copy()
    best_c = c
    stall = 0
    it = 0
    while time.time() < deadline:
        h_lp = one_lp_step(h)
        if h_lp is None:
            stall += 1
            if stall > max_stall:
                break
            continue
        best_t = 0.0
        best_c_trial = c
        n_search = 15
        for t in np.linspace(0.0, 1.0, n_search):
            if t == 0.0:
                continue
            cand = project_feasible(h + t * (h_lp - h), target)
            c_new = score_exact(cand)
            if c_new < best_c_trial:
                best_c_trial = c_new
                best_t = t
                h = cand
        if best_t > 0.0:
            c = best_c_trial
            stall = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                h = project_feasible(h + rng.normal(0, 0.02, n), target)
                c = score_exact(h)
                stall = 0
        it += 1
    return best_h, best_c

def subgrad_phase(h0, deadline, rng):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    c = score_exact(h)
    best_h = h.copy()
    best_c = c
    alpha = 0.015
    tol_factor = 0.002
    it = 0
    while time.time() < deadline and alpha > 1e-7:
        c_vals = score_components(h)
        max_c = float(np.max(c_vals))
        threshold = max_c - tol_factor
        active = np.where(c_vals >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_vals))])
        G = compute_grad_matrix(h)
        g = G[active].mean(axis=0)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        h_new = project_feasible(h - alpha * g, target)
        c_new = score_exact(h_new)
        if c_new < c - 1e-14:
            h = h_new
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
            alpha = min(alpha * 1.005, 0.08)
        else:
            alpha *= 0.92
        it += 1
    return best_h, best_c

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(22):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                best_a = 0.0
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new / 2.0)

def target_activity_search(h, deadline, rng, max_no_improve=150):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    c_cur = score_exact(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    stall = 0
    while time.time() < deadline:
        c_vals = score_components(h_cur)
        max_c = float(np.max(c_vals))
        active = np.where(c_vals >= max_c - 0.001)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_vals))])
        G = compute_grad_matrix(h_cur)
        g_sub = G[active].mean(axis=0)
        gn = float(np.linalg.norm(g_sub))
        if gn < 1e-14:
            stall += 1
            if stall > 20:
                h_cur = project_feasible(h_cur + rng.normal(0, 0.01, n), target)
                c_cur = score_exact(h_cur)
                stall = 0
            continue
        direction = -g_sub / gn
        if direction.sum() != 0:
            adj = direction - direction.mean()
            if np.linalg.norm(adj) > 1e-14:
                direction = adj / np.linalg.norm(adj)
        improved = False
        for s in (0.03, 0.02, 0.01, 0.005, 0.002):
            cand = project_feasible(h_cur + s * direction, target)
            c_new = score_exact(cand)
            if c_new < c_cur - 1e-14:
                h_cur = cand
                c_cur = c_new
                improved = True
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h_cur.copy()
                break
        if not improved:
            stall += 1
            if stall > max_no_improve:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        n_init = len(initial_h_values)
        best_h = project_feasible(np.array(initial_h_values, dtype=np.float64).copy(), n_init / 2.0)
    except (NameError, Exception):
        n_init = 70
        best_h = project_feasible(np.full(n_init, 0.5), n_init / 2.0)
    best_c = score_exact(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: LP cutting plane (35% of time)
    p1_end = t0 + budget_s * 0.35
    if time.time() < p1_end:
        h1, c1 = lp_phase(best_h, p1_end, rng, max_stall=100)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Targeted subgradient on active lags (15% of time)
    p2_end = t0 + budget_s * 0.50
    if time.time() < p2_end:
        h2, c2 = target_activity_search(best_h, p2_end, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: LP restarts with perturbations (15% of time)
    p3_end = t0 + budget_s * 0.65
    attempts = 0
    while time.time() < p3_end and attempts < 40:
        scale = 0.04 * (1.0 - attempts / 40.0) + 0.005
        h_try = project_feasible(best_h + rng.normal(0, scale, n), target)
        c_try = score_exact(h_try)
        sub_dl = min(p3_end - 1, time.time() + 12)
        if time.time() < sub_dl:
            h_try, c_try = lp_phase(h_try, sub_dl, rng, max_stall=40)
        if c_try < best_c:
            best_h, best_c = h_try, c_try
        attempts += 1

    # Phase 4: Pair and coordinate search (25% of time)
    p4_end = t0 + budget_s * 0.90
    cur = best_h.copy()
    cur_c = best_c
    consecutive_fail = 0
    while time.time() < p4_end:
        mode = rng.random()
        if mode < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=35)
        elif mode < 0.75:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=25)
        elif mode < 0.85:
            i = int(rng.integers(0, n - 1))
            res = pair_search(cur, i, i + 1, cur_c, n_pts=45)
        else:
            i = int(rng.integers(0, n))
            res = coord_search(cur, i, cur_c, n_pts=30)
        if res is not None:
            cur_c = res
            consecutive_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
                for _ in range(6):
                    i2 = int(rng.integers(0, n))
                    j2 = int(rng.integers(0, n))
                    if i2 == j2:
                        continue
                    res2 = pair_search(cur, i2, j2, cur_c, n_pts=25)
                    if res2 is not None:
                        cur_c = res2
                        if cur_c < best_c:
                            best_h, best_c = cur.copy(), cur_c
                    else:
                        break
        else:
            consecutive_fail += 1
            if consecutive_fail > 250:
                cur = project_feasible(best_h + rng.normal(0, 0.008, n), target)
                cur_c = score_exact(cur)
                consecutive_fail = 0

    # Phase 5: Resolution changes (5% of time)
    p5_end = deadline - 5
    n_cur = len(best_h)
    for dn in [6, -6, 12, -12, 4, -4, 18, -18]:
        if time.time() > p5_end:
            break
        nr = n_cur + dn
        if nr < 20 or nr > 250:
            continue
        h_r = resample(best_h, nr)
        if h_r is None:
            continue
        c_r = score_exact(h_r)
        sub_dl = min(p5_end - 2, time.time() + 25)
        if time.time() < sub_dl:
            h_r, c_r = lp_phase(h_r, sub_dl, rng, max_stall=30)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Final refinement
    if time.time() < deadline - 2:
        h_f, c_f = lp_phase(best_h, deadline - 2, rng, max_stall=20)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
