For h of length n, this gives output of length 2n-1. The k-th element (0-indexed) is:
sum_i h[i] * (1-h[i + k - (n-1)])

where the valid range is where both indices are in [0, n-1].

So for lag m = k - (n-1), the sum is over i where 0 ≤ i < n and 0 ≤ i+m < n.

For m ≥ 0: i ranges from 0 to n-1-m, sum = Σ_{i=0}^{n-1-m} h[i](1-h[i+m])
For m < 0: let m' = -m, i ranges from m' to n-1, sum = Σ_{i=m'}^{n-1} h[i](1-h[i-m'])

This is symmetric: S(m) = S(-m) in expectation for random h, but not exactly.

Wait, actually np.correlate(h, 1-h) is not symmetric in general. Let me check:
np.correlate(a, b, mode='full')[k] = Σ_i a[i+k] * b[i] for valid i.

So np.correlate(h, 1-h)[k] = Σ_i h[i+k] * (1-h[i]) for valid i.

This means for lag m = k - (n-1):
- If m ≥ 0: Σ_{i=0}^{n-1-m} h[i+m](1-h[i])
- If m < 0: let m' = -m: Σ_{i=m'}^{n-1} h[i](1-h[i+m'])

So S(m) for m≥0 = Σ h[i+m](1-h[i]) and S(-m) for m≥0 = Σ h[i](1-h[i+m])

These are NOT the same in general! S(m) + S(-m) for a given m would be:
Σ h[i+m](1-h[i]) + Σ h[i](1-h[i+m]) = Σ h[i+m] - h[i+m]h[i] + h[i] - h[i]h[i+m]

```python
    n = len(x0)
    target = n / 2.0
    x = x0.copy()
    y = x0.copy()
    t_k = 1.0
    
    if lr is None:
        lr = 0.1 / beta  # rough Lipschitz estimate
    
    for step in range(steps):

            break
        
        gy = smooth_grad(y, beta)
        y_new = project_feasible(y - lr * gy, target)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - x)
        x = y_new

    return project_feasible(x, target)
```

```python
def compute_grad(x, weights, dx):
    n = len(x)
    grad = np.zeros(n)
    # For each k in 0..2n-2:
    #   term A: w[k] * (1-x[p-k]) for p in [max(0,k), min(n-1, n+k-1)]
    #   term B: -w[k] * x[p] for p in [max(0,-k), min(n-1, n-k-1)]
    
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # Term A
        p_lo = max(0, k)
        p_hi = min(n, n + k)
        if p_lo < p_hi:
            grad[p_lo:p_hi] += w * (1.0 - x[p_lo-k:p_hi-k])
        # Term B
        p_lo2 = max(0, -k)
        p_hi2 = min(n, n - k)
        if p_lo2 < p_hi2:
            grad[p_lo2:p_hi2] -= w * x[p_lo2:p_hi2]
    return grad
```

This is vectorized in the inner loop (slicing), so for each k it's O(n) but done in C. Total is O(n * 2n) = O(n^2) C-level operations, which is very fast.

```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-10:
        idx = np.argsort(1.0 - h)[::-1]
        for i in idx:
            add = min(residual, 1.0 - h[i])
            if add <= 0: break
            h[i] += add
            residual -= add
            if residual < 1e-13: break

        idx = np.argsort(h)[::-1]
        for i in idx:
            sub = min(-residual, h[i])
            if sub <= 0: break
            h[i] -= sub
            residual += sub
            if -residual < 1e-13: break
    return np.clip(h, 0.0, 1.0)

    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)

        val = max_c * dx
        w = np.ones(2*n-1) / (2*n-1)
    else:
        val = (max_c + np.log(s) / beta) * dx
        w = e / s
    g = compute_grad(x, w, dx)
    return val, g

def fista(x0, beta, steps=2000, lr=None, time_limit=10.0):
    t0 = time.time()

    target = n / 2.0
    x = x0.copy()
    y = x0.copy()
    t_k = 1.0
    if lr is None:

        _, g0 = lse_fun_and_grad(x0, beta)
        gn = np.sum(g0**2) + 1e-12
        lr = 1.0 / (beta * 2 * n)  # rough
    for step in range(steps):
        if time.time() - t0 > time_limit:

        _, gy = lse_fun_and_grad(y, beta)
        x_new = project_feasible(y - lr * gy, target)

        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        x = x_new
        t_k = t_new

def lbfgs_penalty(h0, mu=50.0, beta=100, maxiter=200, time_limit=20.0):

    n = len(h0)
    def obj(z):
        if time.time() - t0 > time_limit:
            raise TimeoutError()
        v, _ = lse_fun_and_grad(z, beta)

                       bounds=[(0.0, 1.0)]*n,

        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):

    return h0
```
