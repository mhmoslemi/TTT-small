def project_onto_constraint(h, target_sum):

    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_initial(n, rng):
    """Create a structured initial h vector."""
    h = np.zeros(n)
    # Use Thue-Morse-like pattern mixed with randomness
    def thue_morse(k):
        return bin(k).count('1') % 2
    
    for i in range(n):
        h[i] = 1.0 - thue_morse(i)
    
    # Add some randomness
    h = h + (rng.rand(n) - 0.5) * 0.3
    h = project_onto_constraint(h, n / 2.0)

def pair_sweep(h, n_grid=11, time_budget=None, t0=None):
    """One sweep of pair-exchange line search."""
    n = len(h)
    improved = 0
    c5 = compute_c5(h)
    
    for i in range(n):
        if time_budget is not None and time.time() - t0 > time_budget:
            return h, improved, c5
        for j in range(i + 1, n):
            hi_val = h[i]
            hj_val = h[j]
            L = max(-hi_val, hj_val - 1.0)
            U = min(1.0 - hi_val, hj_val)
            if U - L < 1e-10:
                continue
            
            old_i, old_j = h[i], h[j]
            best_t = 0.0
            best_c5_local = c5
            
            # Coarse grid
            t_vals = np.linspace(L, U, n_grid)
            for t in t_vals:
                if abs(t) < 1e-12:
                    continue
                h[i] = old_i + t
                h[j] = old_j - t
                c = compute_c5(h)
                if c < best_c5_local:
                    best_c5_local = c
                    best_t = t

                lo_r = max(L, best_t - margin)

                t_fine = np.linspace(lo_r, hi_r, 9)

def optimize_manifold(h_init, n, rng, deadline, temperature_seq=None):
    """Optimize on the shift-complement manifold using smooth L-BFGS-B."""
    from scipy.optimize import minimize
    
    if n % 2 != 0:
        return h_init
    
    m = n // 2
    
    # Initialize on manifold
    a = np.zeros(m)
    for p in range(m):
        a[p] = 0.5 * (h_init[p] + 1.0 - h_init[p + m])
    a = np.clip(a, 0.0, 1.0)
    
    # Manifold: h[:m] = a, h[m:] = 1-a
    def obj_and_grad(a_vec, temp):
        h = np.empty(n)
        h[:m] = a_vec
        h[m:] = 1.0 - a_vec
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        maxc = np.max(corr)
        
        # Smooth max
        exp_vals = np.exp((corr - maxc) / temp)
        Z = np.sum(exp_vals)
        weights = exp_vals / Z
        
        # Gradient of smooth max w.r.t. h
        grad_h = np.zeros(n)
        for k in range(2 * n - 1):
            s = k - (n - 1)
            w = weights[k] * dx
            if w < 1e-14:
                continue
            a_s = max(0, s)
            b_s = min(n - 1, n - 1 + s)
            if a_s <= b_s:
                idx = np.arange(a_s, b_s + 1)
                grad_h[idx] += w * (1.0 - h[idx - s])
            a2 = max(0, -s)
            b2 = min(n - 1, n - 1 - s)
            if a2 <= b2:
                idx = np.arange(a2, b2 + 1)
                grad_h[idx] -= w * h[idx + s]
        
        # Chain rule: a[p] appears in h[p] (coeff +1) and h[p+m] (coeff -1)
        grad_a = grad_h[:m] - grad_h[m:]
        
        val = (maxc + temp * np.log(Z)) * dx
        return val, grad_a
    
    # Anneal temperature

            v, g = obj_and_grad(aval, temp)

    h_out = np.concatenate([best_a, 1.0 - best_a])
    h_out = project_onto_constraint(h_out, m)
    return h_out

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def make_initial(n, rng):
    h = np.zeros(n)
    phi = 0.6180339887498949

def pair_sweep(h, n_grid=11, refine_grid=9, deadline=None):
    n = len(h)
    improved = 0
    c5 = compute_c5(h)
    for i in range(n):
        if deadline is not None and time.time() > deadline:
            return h, improved, c5
        for j in range(i + 1, n):
            hi_val = h[i]
            hj_val = h[j]
            L = max(-hi_val, hj_val - 1.0)
            U = min(1.0 - hi_val, hj_val)
            if U - L < 1e-10:
                continue
            old_i, old_j = h[i], h[j]
            best_t = 0.0
            best_local = c5
            t_vals = np.linspace(L, U, n_grid)
            for t in t_vals:
                if abs(t) < 1e-12:
                    continue
                h[i] = old_i + t
                h[j] = old_j - t

            if best_t != 0.0:

                lo_r = max(L, best_t - margin)
                hi_r = min(U, best_t + margin)

                    h[i] = old_i + t
                    h[j] = old_j - t

                    if c < best_local:

                        best_t = t
