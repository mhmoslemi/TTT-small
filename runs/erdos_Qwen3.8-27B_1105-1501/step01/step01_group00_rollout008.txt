import numpy as np
from scipy.optimize import minimize
import time

def score_eval(h):
    """Compute C5 = max_k sum(h[i]*(1-h[i+k])) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def make_feasible(z):
    """Project z to [0,1]^n with sum = n/2"""
    n = len(z)
    target = n / 2.0
    h = np.clip(z, 0.0, 1.0)
    # Binary search for shift to make sum = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)[::-1]  # most room first
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-14:
                break
    elif residual < -1e-12:
        order = np.argsort(h)  # least value first
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if residual > -1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def pair_sweep(h, n_pairs_per_iter, n_samples, rng, max_time):
    """Iterate pair moves to improve h"""
    n = len(h)
    best_score = score_eval(h)
    best_h = h.copy()
    t0 = time.time()
    
    all_pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    
    while time.time() - t0 < max_time:
        # Shuffle pairs
        rng.shuffle(all_pairs)
        pairs_to_try = all_pairs[:n_pairs_per_iter]
        improved = False
        
        for i, j in pairs_to_try:
            if time.time() >= t0 + max_time:
                break
            # Feasible range for t: h[i]+t in [0,1], h[j]-t in [0,1]
            t_min = max(-h[i], h[j] - 1.0)
            t_max = min(1.0 - h[i], h[j])
            if t_max <= t_min:
                continue
            
            # Sample t values
            ts = np.linspace(t_min, t_max, n_samples)
            # Also add midpoint and some random points
            ts = np.concatenate([ts, [t_min + 0.5*(t_max-t_min)]])
            
            best_t = 0.0
            best_sc = best_score
            for t in ts:
                if t < t_min - 1e-12 or t > t_max + 1e-12:
                    continue
                h_new = best_h.copy()
                h_new[i] += t
                h_new[j] -= t
                if h_new[i] < -1e-12 or h_new[i] > 1 + 1e-12:
                    continue
                if h_new[j] < -1e-12 or h_new[j] > 1 + 1e-12:
                    continue
                h_new = np.clip(h_new, 0.0, 1.0)
                sc = score_eval(h_new)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_t = t
            
            if best_t != 0.0:
                best_h[i] += best_t
                best_h[j] -= best_t
                best_h = np.clip(best_h, 0.0, 1.0)
                best_score = best_sc
                improved = True
        
        if not improved:
            break
    
    return best_h, best_score

def score_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-14:
                break
    elif residual < -1e-12:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if residual > -1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def pair_sweep(h, time_limit, n_samples=30, rng=None):
    """Iterative pair moves to minimize C5"""
    n = len(h)

        rng = np.random.default_rng(42)
    
    best_h = h.copy()
    best_score = score_eval(h)
    t0 = time.time()
    n_stalled = 0
    
    while time.time() - t0 < time_limit and n_stalled < 10:
        n_pairs = n * (n - 1)  # allow duplicates, ~2x coverage
        any_improve = False
        
        for _ in range(n_pairs):
            if time.time() - t0 > time_limit:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Feasible range for t: h[i]+t in [0,1], h[j]-t in [0,1]
            t_min = max(-best_h[i], best_h[j] - 1.0)

            if t_max <= t_min + 1e-14:
                continue
            
            # Sample t values
            ts = np.linspace(t_min, t_max, n_samples)
            # Add some targeted samples
            mid = (t_min + t_max) / 2.0
            ts = np.concatenate([ts, [mid, mid*0.5, mid*1.5 + (t_max-t_min)*0.1]])
            ts = ts[(ts >= t_min - 1e-12) & (ts <= t_max + 1e-12)]
            
            best_t = 0.0
            local_best = best_score
            
            for t in ts:
                t = float(np.clip(t, t_min, t_max))
                h_new = best_h.copy()
                h_new[i] += t
                h_new[j] -= t
                # Quick check
                if h_new[i] < -1e-10 or h_new[i] > 1 + 1e-10:
                    continue
                if h_new[j] < -1e-10 or h_new[j] > 1 + 1e-10:
                    continue

                    local_best = sc

                best_h[j] -= best_t
                best_score = local_best
                any_improve = True
        
        if not any_improve:

        else:
            n_stalled = 0
