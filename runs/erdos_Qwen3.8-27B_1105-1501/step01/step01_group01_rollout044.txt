def official_score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_grad_matrix(h):
    n = len(h)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        shift = m - (n - 1)
        for p in range(n):
            if 0 <= p + shift < n:
                A[m, p] += (1.0 - h[p + shift])
            if 0 <= p - shift < n:
                A[m, p] += (-h[p - shift])
    return A

def dca_optimize(h0, max_iter=80, deadline=0.0):
    h = h0.copy()
    best_h = h.copy()
    best_score = official_score(h)
    n = len(h)
    stalled = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        A = compute_grad_matrix(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        b = A @ h - corr
        c_lp = np.zeros(n + 1)
        c_lp[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, -1] = -1.0
        b_ub = b.copy()
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(-1.0, None)]
        try:
            res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                break
            h_lp = res.x[:n]
            d = h_lp - h
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-10:
                break
            improved = False
            for alpha in [1.0, 0.85, 0.7, 0.55, 0.4, 0.3, 0.2, 0.12, 0.06, 0.03]:
                h_try = np.clip(h + alpha * d, 0.0, 1.0)
                h_try = make_feasible(h_try)
                s = official_score(h_try)
                if s < best_score - 1e-13:
                    h = h_try
                    best_h = h_try.copy()
                    best_score = s
                    improved = True
                    stalled = 0
                    break
            if not improved:
                stalled += 1
                if stalled >= 3:
                    break
        except Exception:
            break
    return best_h, best_score

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    A = compute_grad_matrix(x)
    grad = A.T @ (weights * dx)
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def smooth_optimize(h0, beta, maxiter=200, deadline=0.0):
    n = len(h0)
    try:
        res = minimize(
            smooth_obj, h0.copy(), args=(beta,),
            jac=smooth_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def binary_push(h, p=2.0):
    centered = 2.0 * h - 1.0
    pushed = np.sign(centered) * np.abs(centered) ** (1.0 / p)
    h_new = 0.5 * (pushed + 1.0)
    return make_feasible(h_new)

def hard_threshold(h):
    h_bin = (h > 0.5).astype(float)
    return make_feasible(h_bin)

def perturbed_start(h, sigma, rng):
    return make_feasible(h + sigma * rng.standard_normal(len(h)))

def coordinate_refine(h, n_sweeps=5, deadline=0.0):
    h = h.copy()
    best_score = official_score(h)
    n = len(h)
    rng = np.random.default_rng(12345)
    deltas = [0.05, 0.03, 0.015, 0.008, 0.004]
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            improved = False
            for d in deltas:
                for sign in [1, -1]:
                    h_try = h.copy()
                    old_val = h_try[i]
                    new_val = np.clip(old_val + sign * d, 0.0, 1.0)
                    if new_val == old_val:
                        continue
                    delta = new_val - old_val
                    h_try[i] = new_val
                    # Compensate sum: find j != i to adjust
                    if abs(delta) > 1e-15:
                        # Adjust the coordinate with most room in opposite direction
                        if delta > 0:
                            other = np.argsort(h_try)
                            for j in other:
                                if j == i:
                                    continue
                                room = h_try[j]
                                adj = min(delta, room)
                                h_try[j] -= adj
                                delta -= adj
                                if delta < 1e-15:
                                    break
                        else:
                            other = np.argsort(1.0 - h_try)
                            for j in other:
                                if j == i:
                                    continue
                                room = 1.0 - h_try[j]
                                adj = min(-delta, room)
                                h_try[j] += adj
                                delta += adj
                                if delta > -1e-15:
                                    break
                    h_try = make_feasible(h_try)
                    s = official_score(h_try)
                    if s < best_score - 1e-13:
                        h = h_try
                        best_score = s
                        improved = True
                        break
                if improved:
                    break
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(48) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(parent)
    rng = np.random.default_rng(seed)
    parent_n = len(parent)

    # Candidate sizes: focus on even sizes for binary feasibility
    n_options = []
    for n in range(max(24, parent_n - 14), min(80, parent_n + 14) + 1):
        n_options.append(n)
    # Ensure we have some key sizes
    for n in [48, 50, 52, 54, 56, 46, 44]:
        if n not in n_options and 24 <= n <= 76:
            n_options.append(n)
    n_options = sorted(set(n for n in n_options if 24 <= n <= 76))

    # Phase 1: DCA + binary push on parent-sized and nearby
    for n in n_options:
        if time.time() > deadline - 15:
            break
        h_base = make_feasible(interp_to_n(parent, n))
        base_score = official_score(h_base)

        # Generate diverse starting points
        starts = []
        starts.append(('interp', h_base))
        starts.append(('binary2', binary_push(h_base, 2.0)))
        starts.append(('binary3', binary_push(h_base, 3.0)))
        starts.append(('binary15', binary_push(h_base, 1.5)))
        starts.append(('thresh', hard_threshold(h_base)))
        starts.append(('thresh_bin2', binary_push(hard_threshold(h_base), 2.0)))
        for sigma in [0.08, 0.15, 0.25]:
            starts.append(('pert', perturbed_start(h_base, sigma, rng)))
        # Anti-symmetric
        h_anti = make_feasible(1.0 - h_base[::-1].copy())
        starts.append(('anti', h_anti))
        starts.append(('anti_bin', binary_push(h_anti, 2.0)))

        for name, h0 in starts:
            if time.time() > deadline - 10:
                break
            h0 = make_feasible(h0)
            # DCA optimization
            h_dca, s_dca = dca_optimize(h0, max_iter=60, deadline=deadline - 5)
            if s_dca < best_score - 1e-13:
                best_h = h_dca
                best_score = s_dca
                best_n = len(h_dca)
            # Smooth opt on the DCA result if it improved
            if time.time() < deadline - 8:
                h_sm = smooth_optimize(h_dca, 200, maxiter=100, deadline=deadline - 3)
                s_sm = official_score(h_sm)
                if s_sm < best_score - 1e-13:
                    best_h = h_sm
                    best_score = s_sm
                    best_n = len(h_sm)

    # Phase 2: Coordinate descent refinement on best
    if time.time() < deadline - 10:
        h_cd, s_cd = coordinate_refine(best_h, n_sweeps=8, deadline=deadline - 5)
        if s_cd < best_score - 1e-13:
            best_h = h_cd
            best_score = s_cd

    # Phase 3: DCA from best with more iterations
    if time.time() < deadline - 10:
        h_dca2, s_dca2 = dca_optimize(best_h, max_iter=100, deadline=deadline - 5)
        if s_dca2 < best_score - 1e-13:
            best_h = h_dca2
            best_score = s_dca2

    # Phase 4: Try perturbing the best and re-optimizing
    if time.time() < deadline - 15:
        for trial in range(5):
            if time.time() > deadline - 8:
                break
            h_pert = perturbed_start(best_h, 0.04, rng)
            h_r, s_r = dca_optimize(h_pert, max_iter=50, deadline=deadline - 5)
            if s_r < best_score - 1e-13:
                best_h = h_r
                best_score = s_r
            h_sm2 = smooth_optimize(best_h, 500, maxiter=80, deadline=deadline - 3)
            s_sm2 = official_score(h_sm2)
            if s_sm2 < best_score - 1e-13:
                best_h = h_sm2
                best_score = s_sm2

    # Phase 5: Final DCA polish
    if time.time() < deadline - 5:
        h_final, s_final = dca_optimize(best_h, max_iter=80, deadline=deadline - 2)
        if s_final < best_score - 1e-13:
            best_h = h_final
            best_score = s_final

    # Final refinement: coordinate descent + DCA
    if time.time() < deadline - 3:
        h_cd2, s_cd2 = coordinate_refine(best_h, n_sweeps=3, deadline=deadline - 2)
        if s_cd2 < best_score - 1e-13:
            best_h = h_cd2
            best_score = s_cd2
            h_f2, s_f2 = dca_optimize(best_h, max_iter=40, deadline=deadline - 1)
            if s_f2 < best_score - 1e-13:
                best_h = h_f2
                best_score = s_f2

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    # Verify: if somehow worse than parent, return parent
    parent_score = official_score(parent)
    if final_score > parent_score:
        best_h = parent
        final_score = parent_score

    return (best_h, final_score, len(best_h))
