def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    p = np.exp(alpha * (corr - max_c))
    p /= p.sum()

    for idx in range(2*n - 1):
        k = idx - (n-1)
        if k >= 0:
            grad[k:] += p[idx] * (1.0 - h[:n-k])
            grad[:n-k] -= p[idx] * h[k:]
        else:
            r = -k
            grad[:n-r] += p[idx] * (1.0 - h[r:])
            grad[r:] -= p[idx] * h[:n-r]
    return grad

def project(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def pair_move_improve(h, n_trials=200, deltas=None, rng=None):
    """Try random pair moves to improve C5."""
    n = len(h)
    target = float(n) / 2.0
    best_c5 = compute_c5(h)
    if deltas is None:
        deltas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            for sign in [1, -1]:
                delta = sign * d
                hi_test = h[i] + delta
                hj_test = h[j] - delta
                if hi_test < -1e-10 or hi_test > 1.0 + 1e-10:
                    continue
                if hj_test < -1e-10 or hj_test > 1.0 + 1e-10:
                    continue
                h_test = h.copy()
                h_test[i] = np.clip(hi_test, 0.0, 1.0)
                h_test[j] = np.clip(hj_test, 0.0, 1.0)
                c5 = compute_c5(h_test)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    h = h_test.copy()
    return h, best_c5

    n = len(h0)

    h = project(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    alpha = 10.0
    max_iter = 10000
    stalled = 0
    stall_limit = 200
    
    while time.time() - t0 < time_budget * 0.6:
        grad = soft_grad(h, alpha)
        # Backtracking line search
        step = 0.5
        improved = False
        for _ in range(20):
            h_trial = project(h - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c5_trial
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            stalled += 1
            # Try a smaller step in a random direction too
            h_trial = project(h - 0.01 * grad + 0.01 * rng.randn(n), target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                h = h_trial
                best_c5 = c5_trial
                best_h = h.copy()
                improved = True
            stalled += 1 if not improved else 0
        
        if stalled > stall_limit:
            # Restart
            h = best_h + 0.03 * rng.randn(n)
            h = project(h, target)
            alpha = 5.0
            stalled = 0
        
        # Adapt alpha
        if stalled % 50 == 0 and stalled > 0:
            alpha = min(alpha * 1.3, 100.0)
        elif improved:
            alpha = min(alpha * 1.02, 50.0)
    
    # Phase B: Pair moves
    if time.time() - t0 < time_budget:
        remaining = time_budget - (time.time() - t0)
        h, best_c5 = pair_move_improve(best_h, n_trials=5000, rng=rng)
        best_h = h
    
    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / exp_c.sum()

    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_optimize(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = project(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    alpha = 12.0
    lr = 0.04
    momentum = 0.88
    v = np.zeros(n)
    no_improve = 0
    mode = 0
    while time.time() - t0 < time_budget:
        if mode == 0:
            grad = soft_grad(h, alpha)
        else:
            tau = best_c5 * n / 2.0 - 0.05
            grad = threshold_grad(h, tau)
            if grad.max() < 1e-15:
                tau = best_c5 * n / 2.0 + 0.05
                grad = threshold_grad(h, tau)
                if grad.max() < 1e-15:
                    mode = 0
                    grad = soft_grad(h, alpha)
        v = momentum * v - lr * grad
        h_trial = project(h + v, target)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-14:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
            no_improve = 0
        else:
            h = h_trial
            no_improve += 1
        if no_improve > 200:
            h = best_h + 0.02 * rng.randn(n)
            h = project(h, target)
            v = np.zeros(n)
            no_improve = 0
            lr = max(lr * 0.8, 0.003)
            alpha = max(alpha * 0.7, 5.0)
            mode = 1 - mode
        elif no_improve > 80:
            alpha = min(alpha * 1.15, 120.0)
            mode = 1 - mode
        lr = max(lr * 0.9998, 0.002)
    return best_h, best_c5

def pair_optimize(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    d_idx = 0
    stuck = 0
    while time.time() - t0 < time_budget:
        improved = False
        for _ in range(50):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas[max(d_idx, len(deltas)-4):]:
                for sign in [1, -1]:
                    delta = sign * d
                    vi = h[i] + delta
                    vj = h[j] - delta
                    if vi < -1e-12 or vi > 1.0+1e-12 or vj < -1e-12 or vj > 1.0+1e-12:
                        continue
                    h_test = h.copy()
                    h_test[i] = vi
                    h_test[j] = vj
                    c5 = compute_c5(h_test)
                    if c5 < best_c5 - 1e-14:
                        h = h_test
                        best_c5 = c5
                        improved = True
                        d_idx = min(d_idx + 1, len(deltas) - 4)
                        break
                if improved:
                    break
        if not improved:
            stuck += 1
            if stuck > 300:
                stuck = 0
                d_idx = max(d_idx - 1, 0)
                h_pert = h + 0.008 * rng.randn(n)
                h_pert = project(h_pert, target)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    h = h_pert
                    best_c5 = c5_pert
    return h, best_c5

def targeted_pair(h, rng, n):
    """Try pairs near the active set of correlations."""
    target = float(n) / 2.0
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    active = np.where(corr > max_c - 0.3)[0]
    for _ in range(200):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in [0.01, 0.02, 0.05]:
            for sign in [1, -1]:
                delta = sign * d
                vi = h[i] + delta
                vj = h[j] - delta
                if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                    continue
                h[i] = vi
                h[j] = vj
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                else:
                    h[i] -= delta
                    h[j] += delta
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass
    if h_init is None:
        n0 = 99
        h_init = 0.5 * np.ones(n0)

    n0 = len(h_init)

    # Phase 1: Main gradient optimization
    alloc = min(remaining() * 0.40, 400.0)
    if alloc > 20:
        h1, c5_1 = gradient_optimize(h_init, alloc, seed)
        best_h = h1
        best_c5 = c5_1
        best_n = n0

    # Phase 2: Pair optimization
    if best_h is not None and remaining() > 20:
        alloc = min(remaining() * 0.12, 100.0)
        h2, c5_2 = pair_optimize(best_h, alloc, seed + 100)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 3: Second gradient pass from perturbed
    if best_h is not None and remaining() > 30:
        alloc = min(remaining() * 0.20, 180.0)
        perturbed = best_h + 0.012 * rng.randn(len(best_h))
        h3, c5_3 = gradient_optimize(perturbed, alloc, seed + 200)
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3

    # Phase 4: More pair optimization
    if best_h is not None and remaining() > 20:
        alloc = min(remaining() * 0.15, 120.0)
        h4, c5_4 = pair_optimize(best_h, alloc, seed + 300)
        if c5_4 < best_c5:
            best_h = h4
            best_c5 = c5_4

    # Phase 5: Third gradient pass
    if best_h is not None and remaining() > 40:
        alloc = min(remaining() * 0.15, 130.0)
        perturbed2 = best_h + 0.008 * rng.randn(len(best_h))
        h5, c5_5 = gradient_optimize(perturbed2, alloc, seed + 400)
        if c5_5 < best_c5:
            best_h = h5
            best_c5 = c5_5

    # Phase 6: Final pair polish
    if best_h is not None and remaining() > 15:
        alloc = min(remaining() * 0.18, 150.0)
        h6, c5_6 = pair_optimize(best_h, alloc, seed + 500)
        if c5_6 < best_c5:
            best_h = h6
            best_c5 = c5_6

    # Phase 7: Try n=63 if time allows
    if remaining() > 100:
        n63 = 63
        h63 = 0.5 + 0.3 * np.sin(2 * np.pi * 1.7 * np.arange(n63) / n63)
        h63 = project(h63, float(n63) / 2.0)
        alloc = min(remaining() * 0.25, 120.0)
        h7, c5_7 = gradient_optimize(h63, alloc, seed + 63)
        if remaining() > 20:
            alloc2 = min(remaining() * 0.1, 40.0)
            h7b, c5_7b = pair_optimize(h7, alloc2, seed + 64)
            if c5_7b < c5_7:
                h7, c5_7 = h7b, c5_7b
        if c5_7 < best_c5:
            best_h = h7
            best_c5 = c5_7
            best_n = n63

    # Final: ensure constraint
    if best_h is None:
        best_h = h_init
        best_n = n0
    target = float(best_n) / 2.0
    best_h = project(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
