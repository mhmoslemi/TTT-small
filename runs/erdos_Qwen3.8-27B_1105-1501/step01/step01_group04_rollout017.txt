                    if direction:

                        h[j2] += amt2
                    else:
                        h[i2] += amt2
                        h[j2] -= amt2
                    h = np.clip(h, 0.0, 1.0)
                    h = project_simplex(h, target)
                    best_h = h.copy()
                    best_c5 = best_local_c5
                    improved = True

        if not improved and time.time() - sweep_start > 2.0:
            # Perturb and restart
            h = best_h + 0.003 * rng.randn(n)
            h = project_simplex(h, target)

    return best_h, best_c5


def slsqp_attempt(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    dx = 2.0 / n
    c0 = compute_corr(h0) * dx
    t0_init = float(np.max(c0))

    def obj(y):
        t = y[0]
        h = y[1:]
        return t

    def jac_obj(y):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def constr_func(y):
        t = y[0]
        h = y[1:]
        c = compute_corr(h) * dx
        # Only include the top constraints to reduce dim
        top_k = min(30, len(c))
        idx = np.argsort(c)[::-1][:top_k]
        return np.concatenate([t - c[idx], [np.sum(h) - target]])

    # SLSQP needs scipy constraints format
    from scipy.optimize import NonlinearConstraint

    c_full = compute_corr(h0) * dx
    top_k = min(25, len(c_full))
    idx = np.argsort(c_full)[::-1][:top_k]

    def con_func(y):
        t = y[0]
        h = y[1:]
        c = compute_corr(h) * dx
        return t - c[idx]

    def con_jac(y):
        h = y[1:]
        dx_local = 2.0 / n
        a = 1.0 - 2.0 * h
        J = np.zeros((top_k, n + 1))
        for ii, k in enumerate(idx):
            row = np.correlate(np.ones(1) if False else np.zeros(2*n-1), a, mode='valid')
            # Actually we need d(t - c[k])/dh[p] = -dc[k]/dh[p] = -(1-2h[k-p])*I(...)
            # which is -(grad of c[k])
            # grad of c[k] is (1-2h) shifted
            q = k
            # dc[k]/dh[p] = (1-2*h[q-p])*I(0<=q-p<=n-1)
            lo_p = max(0, q - (n - 1))
            hi_p = min(n - 1, q)
            for p in range(lo_p, hi_p + 1):
                J[ii, p + 1] = -(1.0 - 2.0 * h[q - p])
        # Constraint: sum(h) - target = 0 handled separately
        return J

    # This is getting complex with mixed constraints. Let me use a simpler approach.
    # Minimize t + penalty for violations
    def penalized_obj(y):
        t = y[0]
        h = y[1:]
        c = compute_corr(h) * dx
        viol = np.maximum(0, c - t)
        return t + 100.0 * np.sum(viol ** 2)

    def penalized_jac(y):
        h = y[1:]
        grads = compute_corr(h) * dx
        c = compute_corr(h) * dx
        t = y[0]
        J = np.zeros(n + 1)
        J[0] = 1.0 + 200.0 * np.sum(np.maximum(0, c - t))
        # This is an approximation - SLSQP can handle inexact derivatives
        return J

    y0 = np.zeros(n + 1)
    y0[0] = t0_init + 0.005
    y0[1:] = h0

    bounds = [(y0[0] - 0.1, y0[0] + 0.01)] + [(0.0, 1.0)] * n

    constraints = [{'type': 'eq', 'fun': lambda y: np.sum(y[1:]) - target}]

    t_start = time.time()
    try:
        res = minimize(penalized_obj, y0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 50, 'ftol': 1e-10})
        if res.success or res.fun < t0_init:
            h_opt = np.clip(res.x[1:], 0.0, 1.0)
            h_opt = project_simplex(h_opt, target)
            c5_opt = compute_c5(h_opt)
            if c5_opt < t0_init:
                return h_opt, c5_opt
    except Exception:
        pass

    return h0, compute_c5(h0)


def top_k_spread_objective_grad(h, top_k=10, alpha=50.0):
    """Gradient of: max_c + penalty * variance of top-k correlations."""
    n = len(h)
    dx = 2.0 / n
    c = compute_corr(h)
    c_max = np.max(c)
    w = np.exp(alpha * (c - c_max))
    w = w / np.sum(w)
    g = dx * np.correlate(w, 1.0 - 2.0 * h, mode='valid')

    # Add variance-spreading term for top correlations
    idx = np.argsort(c)[::-1][:top_k]
    c_top = c[idx]
    c_mean_top = np.mean(c_top)
    # Gradient of sum of (c[k] - mean_top)^2 for k in top_k
    # This pushes top correlations toward their mean
    delta = c_top - c_mean_top
    spread_w = np.zeros(len(c))
    for ii, k in enumerate(idx):
        spread_w[k] = 2.0 * delta[ii]
    spread_grad = dx * np.correlate(spread_w, 1.0 - 2.0 * h, mode='valid')

    return g + 0.1 * spread_grad


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial values
    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_simplex(h_init, float(n0) / 2.0)
        c5_init = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
        best_n = n0
    else:
        # Fallback: create a random initial
        n0 = 99
        h_init = np.zeros(n0)
        h_init[:n0 // 2] = 1.0
        if n0 % 2 == 1:
            h_init[n0 // 2] = 0.5
        h_init = h_init[rng.permutation(n0)]
        h_init = project_simplex(h_init, float(n0) / 2.0)
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)
        best_n = n0

    # Phase 1: Adam with smooth max - moderate alpha
    if remaining() > 80:
        alloc = min(remaining() * 0.22, 220.0)
        h_opt, c5_opt = adam_optimize(best_h, alloc, seed + 1,
                                      alpha_start=3.0, alpha_end=40.0,
                                      lr_start=0.04, lr_end=0.003, mode='smooth')
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(h_opt)

    # Phase 2: P-norm optimization with different p values
    for p in [16, 32, 64]:
        if remaining() < 30:
            break
        alloc = min(remaining() * 0.08, 60.0)
        if alloc < 15:
            break
        h_pn, c5_pn = adam_optimize(best_h, alloc, seed + 100 + p,
                                    mode='pnorm', p=p,
                                    lr_start=0.02, lr_end=0.002)
        if c5_pn < best_c5:
            best_c5 = c5_pn
            best_h = h_pn
            best_n = len(h_pn)

    # Phase 3: High-alpha smooth max for fine-tuning
    if remaining() > 40:
        alloc = min(remaining() * 0.18, 180.0)
        h_opt, c5_opt = adam_optimize(best_h, alloc, seed + 200,
                                      alpha_start=30.0, alpha_end=150.0,
                                      lr_start=0.01, lr_end=0.001, mode='smooth')
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(h_opt)

    # Phase 4: SLSQP attempt
    if remaining() > 30:
        alloc = min(remaining() * 0.1, 60.0)
        h_slq, c5_slq = slsqp_attempt(best_h, alloc, seed + 300)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq
            best_n = len(h_slq)

    # Phase 5: Coordinate descent / local search
    if remaining() > 25:
        alloc = min(remaining() * 0.15, 120.0)
        h_cd, c5_cd = coordinate_descent(best_h, alloc, seed + 400)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            best_n = len(h_cd)

    # Phase 6: Try nearby dimensions
    if remaining() > 25:
        for n_try in [97, 101, 95, 103]:
            if remaining() < 20:
                break
            n_best = len(best_h)
            if n_try == n_best:
                continue
            h_interp = resample(best_h, n_try)
            alloc = min(remaining() * 0.05, 40.0)
            if alloc < 10:
                break
            h_opt, c5_opt = adam_optimize(h_interp, alloc, seed + 500 + n_try,
                                          alpha_start=20.0, alpha_end=100.0,
                                          lr_start=0.015, lr_end=0.001, mode='smooth')
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try
            # Quick coordinate descent on the new dimension
            if remaining() > 15 and c5_opt < best_c5 + 0.001:
                alloc2 = min(remaining() * 0.03, 20.0)
                h_cd2, c5_cd2 = coordinate_descent(h_opt, alloc2, seed + 600 + n_try)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2
                    best_n = n_try

    # Phase 7: Final high-alpha polish
    if remaining() > 15:
        alloc = min(remaining() * 0.2, 100.0)
        h_opt, c5_opt = adam_optimize(best_h, alloc, seed + 700,
                                      alpha_start=80.0, alpha_end=300.0,
                                      lr_start=0.005, lr_end=0.0005, mode='smooth')
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(h_opt)

    # Final: ensure feasibility
    target = float(best_n) / 2.0
    best_h = project_simplex(best_h, target)
    best_c5 = compute_c5(best_h)

    # Final coordinate descent if time remains
    if remaining() > 10:
        h_fd, c5_fd = coordinate_descent(best_h, remaining() * 0.4, seed + 800)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = project_simplex(h_fd, target)
            best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
