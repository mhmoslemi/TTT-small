```python
def trust_region_step(h, T, delta, solver='CLARABEL'):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    t_current = float(np.max(overlaps))
    
    # Active set: lags with overlap > t_current - T
    active_mask = overlaps > t_current - T
    active_lags = np.where(active_mask)[0]
    
    if len(active_lags) == 0:
        active_lags = np.array([np.argmax(overlaps)])
    
    # Compute gradients for active lags
    grads = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        m = k - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + m < n:
                g += (1.0 - h[i + m])
            if 0 <= i - m < n:
                g -= h[i - m]
            grads[idx, i] = dx * g
    
    # QP: minimize t s.t. overlaps[k] + grads[k] · d <= t, sum(d)=0, 0<=h+d<=1
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        h + d >= 0,
        h + d <= 1
    ]
    for idx, k in enumerate(active_lags):
        constraints.append(overlaps[k] + grads[idx] @ d <= t)
    
    # Add trust region (L_infinity norm)
    constraints.append(d <= delta)
    constraints.append(-d <= delta)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver='ECOS')
        except:
            return None, t_current
    
    if d.value is None:
        return None, t_current
    
    d_val = d.value
    if not np.all(np.isfinite(d_val)):
        return None, t_current
    
    h_new = make_feasible(h + d_val)
    c_new = score_exact(h_new)
    
    if c_new < t_current - 1e-12:
        return h_new, c_new
    return None, t_current

def compute_one_gradient(h, k, n, dx):
    """Gradient of output[k] w.r.t. h, times dx."""
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # Term 1: (1-h[i-lag]) for i in [lag, n-1]
        grad[lag:] += 1.0 - h[:n-lag][::-1]  # h[i-lag] for i in [lag, n-1] is h[0], h[1], ..., h[n-1-lag]
        # Term 2: -h[i+lag] for i in [0, n-1-lag]
        grad[:n-lag] -= h[lag:]
    else:
        p = -lag
        # Term 1: (1-h[i+p]) for i in [0, n-1-p]
        grad[:n-p] += 1.0 - h[p:]
        # Term 2: -h[i-p] for i in [p, n-1]
        grad[p:] -= h[:n-p]
    return grad * dx
```

import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        p = -lag
        if p < n:
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
    return grad * dx

def compute_active_gradients(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        lag = k - (n - 1)
        grads[idx] = compute_gradient_for_lag(h, lag, n, dx)
    return grads

def trust_region_step(h, T_active, delta, solver_name):
    n = len(h)
    overlaps = get_all_overlaps(h)
    t_cur = float(np.max(overlaps))
    active_mask = overlaps > t_cur - T_active
    active_idx = np.where(active_mask)[0]
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(overlaps))])
    if len(active_idx) > 60:
        order = np.argsort(overlaps[active_idx])[::-1]
        active_idx = active_idx[order[:60]]
    grads = compute_active_gradients(h, active_idx)
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(d) == 0,
        d <= delta,
        -d <= delta
    ]
    for idx in range(len(active_idx)):
        constraints.append(overlaps[active_idx[idx]] + grads[idx] @ d <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver_name)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return None, t_cur, active_idx
    if d.value is None:
        return None, t_cur, active_idx
    d_val = np.asarray(d.value, dtype=np.float64).flatten()
    if not np.all(np.isfinite(d_val)):
        return None, t_cur, active_idx
    h_new = h + d_val
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    c_new = score_exact(h_new)
    if c_new < t_cur - 1e-14:
        return h_new, c_new, active_idx
    return None, t_cur, active_idx

def smooth_grad_direction(h, T):
    n = len(h)
    overlaps = get_all_overlaps(h)
    base = float(np.max(overlaps))
    ex = np.exp((overlaps - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        w = np.zeros(len(overlaps))
        w[np.argmax(overlaps)] = 1.0
    else:
        w = ex / S
    g = np.zeros(n)
    for idx in range(len(overlaps)):
        if w[idx] < 1e-15:
            continue
        lag = idx - (n - 1)
        g += w[idx] * compute_gradient_for_lag(h, lag, n, 2.0 / n)
    return g

def project_subgradient(h, deadline, T, step_init, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    step = step_init
    stall = 0
    while time.time() < deadline:
        g = smooth_grad_direction(cur, T)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1, step * 0.05):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.7
            stall += 1
            if stall > 30:
                cur = make_feasible(cur + rng.normal(0, 0.01, n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.005)
                stall = 0
        else:
            step = min(step * 1.1, 0.1)
            stall = 0
        T = max(0.0005, T * 0.999)
    return best, best_c

def pair_move(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / max(n_pts - 1, 1)
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(20):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                h[i] = orig_i
                h[j] = orig_j
                return None
            return float(best_s)
    return None

def coord_move(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 20:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    solver_name = 'CLARABEL'
    try:
        d_test = cp.Variable(n)
        cp.Problem(cp.Minimize(cp.sum(d_test ** 2)), [cp.sum(d_test) == 0]).solve(solver=solver_name)
    except Exception:
        solver_name = 'ECOS'

    # Phase 1: Trust-region QP (50% of budget)
    p1_end = t0 + budget_s * 0.50
    delta = 0.05
    T_active = 0.02
    cur = best_h.copy()
    cur_c = best_c
    stall_count = 0
    iter_count = 0
    while time.time() < p1_end:
        h_new, c_new, active_idx = trust_region_step(cur, T_active, delta, solver_name)
        if h_new is not None:
            cur = h_new
            cur_c = c_new
            stall_count = 0
            delta = min(delta * 1.3, 0.2)
            T_active = max(0.001, T_active * 0.95)
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
        else:
            stall_count += 1
            if stall_count > 15:
                delta = max(delta * 0.5, 0.005)
                T_active = min(T_active * 1.5, 0.5)
                if delta < 0.006:
                    # Fallback to subgradient for a few steps
                    sub_dl = min(time.time() + 20, p1_end)
                    cur, cur_c = project_subgradient(cur, sub_dl, max(T_active, 0.01), 0.02, rng)
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                    delta = 0.03
                    T_active = 0.02
                    stall_count = 0
        iter_count += 1

    # Phase 2: Local polish with pair/coordinate moves (25% of budget)
    p2_end = t0 + budget_s * 0.75
    cur = best_h.copy()
    cur_c = best_c
    consec_fail = 0
    while time.time() < p2_end:
        mode = rng.random()
        if mode < 0.6:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_move(cur, i, j, cur_c, n_pts=30)
        elif mode < 0.8:
            i = int(rng.integers(0, n - 1))
            res = pair_move(cur, i, i + 1, cur_c, n_pts=35)
        elif mode < 0.9:
            i = int(rng.integers(0, n))
            res = coord_move(cur, i, cur_c, n_pts=25)
        else:
            # Block move: shift mass from a pair of points
            i = int(rng.integers(0, n // 2))
            j = int(rng.integers(n // 2, n))
            res = pair_move(cur, i, j, cur_c, n_pts=30)
        if res is not None:
            cur_c = res
            consec_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            # Follow-up moves
            for _ in range(3):
                if time.time() > p2_end:
                    break
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = pair_move(cur, i2, j2, cur_c, n_pts=20)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            consec_fail += 1
            if consec_fail > 300:
                cur = make_feasible(best_h + rng.normal(0, 0.005, n), n)
                cur_c = score_exact(cur)
                consec_fail = 0

    # Phase 3: Second round of QP with refined parameters (12% of budget)
    p3_end = t0 + budget_s * 0.87
    if time.time() < p3_end:
        cur = best_h.copy()
        cur_c = best_c
        delta = 0.02
        T_active = 0.005
        stall_count = 0
        while time.time() < p3_end:
            h_new, c_new, _ = trust_region_step(cur, T_active, delta, solver_name)
            if h_new is not None:
                cur = h_new
                cur_c = c_new
                stall_count = 0
                delta = min(delta * 1.2, 0.05)
                T_active = max(0.0005, T_active * 0.9)
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c
            else:
                stall_count += 1
                if stall_count > 20:
                    delta = max(delta * 0.5, 0.002)
                    if delta < 0.003:
                        break

    # Phase 4: Try different resolutions (7% of budget)
    p4_end = deadline
    if time.time() < p4_end:
        n_cur = len(best_h)
        for dn in (-4, -2, 2, 4, 6):
            if time.time() > p4_end - 8:
                break
            nr = n_cur + dn
            if nr < 20 or nr > 150:
                continue
            h_r = resample_h(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(p4_end - 3, time.time() + 50)
            if time.time() < sub_dl:
                delta_r = 0.02
                T_r = 0.01
                cur_r = h_r.copy()
                cur_cr = c_r
                stall_r = 0
                while time.time() < sub_dl:
                    h_new_r, c_new_r, _ = trust_region_step(cur_r, T_r, delta_r, solver_name)
                    if h_new_r is not None:
                        cur_r = h_new_r
                        cur_cr = c_new_r
                        stall_r = 0
                        delta_r = min(delta_r * 1.2, 0.06)
                        if cur_cr < c_r:
                            c_r = cur_cr
                    else:
                        stall_r += 1
                        if stall_r > 15:
                            break
                if c_r < best_c:
                    best_h, best_c = cur_r, c_r

    # Final polish
    if time.time() < deadline - 1:
        sub_dl = deadline - 1
        cur_f = best_h.copy()
        cur_fc = best_c
        for _ in range(min(200, int((sub_dl - time.time()) / 0.1))):
            if time.time() > sub_dl:
                break
            i = int(rng.integers(0, len(cur_f)))
            j = int(rng.integers(0, len(cur_f)))
            if i == j:
                continue
            res = pair_move(cur_f, i, j, cur_fc, n_pts=20)
            if res is not None:
                cur_fc = res
                if cur_fc < best_c:
                    best_h, best_c = cur_f.copy(), cur_fc

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
