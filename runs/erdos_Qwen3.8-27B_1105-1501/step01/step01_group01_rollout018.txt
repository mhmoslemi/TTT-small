        for ls in range(25):
            alpha = step * (0.5 ** ls)
            h_new = make_feasible(h - alpha * g)
            s = official_score(h_new)
            if s < best_score - 1e-14:
                h = h_new
                best_score = s
                refreshed = True
                break
        if not refreshed:
            step *= 0.5
            if step < 1e-8:
                break
    return h, best_score

def coordinate_descent(h0, max_sweeps=20):
    n = len(h0)
    h = h0.copy()
    best_score = official_score(h)
    for sweep in range(max_sweeps):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            for d in [0.02, 0.01, 0.04, 0.005]:
                for sign in [1, -1]:
                    val = h[i] + sign * d
                    if val < 0 or val > 1:
                        continue
                    h[i] = val
                    h_fixed = make_feasible(h)
                    s = official_score(h_fixed)
                    if s < best_score - 1e-12:
                        h = h_fixed
                        best_score = s
                        improved = True
                    else:
                        h[i] = val - sign * d
        if not improved:
            break
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(50) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)

    parent_n = len(parent)

    n_values = sorted(set([
        max(10, parent_n - 7), max(10, parent_n - 5), max(10, parent_n - 3),
        max(10, parent_n - 1), parent_n, parent_n + 1, parent_n + 3,
        parent_n + 5, parent_n + 7
    ]))
    n_values = [n for n in n_values if 20 <= n <= 70]

    candidates = []

    for n in n_values:
        if time.time() > deadline - 40:
            break
        h_interp = make_feasible(interp_to_n(parent, n))
        s0 = official_score(h_interp)
        candidates.append((s0, h_interp.copy(), n))

        h_anti = make_feasible(1.0 - h_interp[::-1].copy())
        s1 = official_score(h_anti)
        candidates.append((s1, h_anti.copy(), n))

        for sigma in [0.08, 0.15]:
            hp = make_feasible(h_interp + sigma * rng.standard_normal(n))
            sp = official_score(hp)
            candidates.append((sp, hp.copy(), n))

        hb = make_feasible(h_interp ** 1.4)
        sb = official_score(hb)
        candidates.append((sb, hb.copy(), n))

    candidates.sort(key=lambda x: x[0])

    top_k = min(12, len(candidates))
    for i in range(top_k):
        if time.time() > deadline - 30:
            break
        sc, h0, n = candidates[i]
        cur_h = h0.copy()
        cur_score = sc

        for beta in [50, 200, 800]:
            if time.time() > deadline - 20:
                break
            mi = 250 if n <= 54 else 150
            new_h = do_smooth_opt(cur_h, beta, maxiter=mi)
            new_score = official_score(new_h)
            if new_score < cur_score - 1e-12:
                cur_h = new_h
                cur_score = new_score

        if time.time() < deadline - 15:
            ah = do_active_opt(cur_h, n_active=6, maxiter=100)
            as = official_score(ah)
            if as < cur_score - 1e-12:
                cur_h = ah
                cur_score = as

        if time.time() < deadline - 10:
            gh, gs = gradient_polish(cur_h, beta=3000, max_iter=150, initial_step=0.03)
            if gs < cur_score - 1e-12:
                cur_h = gh
                cur_score = gs

        if cur_score < best_score - 1e-12:
            best_h = cur_h.copy()
            best_score = cur_score
            best_n = len(cur_h)

    remaining = deadline - time.time()
    if remaining > 15:
        n_trials = min(20, int(remaining / 15))
        for t in range(n_trials):
            if time.time() > deadline - 8:
                break
            pert = make_feasible(best_h + 0.015 * rng.standard_normal(best_n))
            ph = do_smooth_opt(pert, 400, maxiter=120)
            ps = official_score(ph)
            if ps < best_score - 1e-12:
                ph2, gs2 = gradient_polish(ph, beta=5000, max_iter=100, initial_step=0.02)
                if gs2 < ps:
                    ph = ph2
                    ps = gs2
                best_h = ph.copy()
                best_score = ps
                best_n = len(ph)

    if time.time() < deadline - 5:
        fh, fs = gradient_polish(best_h, beta=8000, max_iter=200, initial_step=0.02)
        if fs < best_score:
            best_h = fh
            best_score = fs

    if time.time() < deadline - 3:
        ah = do_active_opt(best_h, n_active=5, maxiter=80)
        as = official_score(ah)
        if as < best_score:
            best_h = ah
            best_score = as

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
