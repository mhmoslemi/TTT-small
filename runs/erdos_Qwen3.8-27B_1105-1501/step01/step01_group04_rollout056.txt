    n = len(h)
    m = n - 1
    a = np.zeros((n, 2*n-1))
    i_idx = np.arange(n)[:, None]  # (n, 1)
    k_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    
    t1_idx = i_idx - k_idx + m  # (n, 2n-1)
    t2_idx = i_idx + k_idx - m  # (n, 2n-1)
    
    t1_valid = (t1_idx >= 0) & (t1_idx < n)
    t2_valid = (t2_idx >= 0) & (t2_idx < n)
    
    # Clip for safe indexing
    t1_safe = np.clip(t1_idx, 0, n-1)
    t2_safe = np.clip(t2_idx, 0, n-1)
    
    term1 = np.where(t1_valid, 1.0 - h[t1_safe], 0.0)
    term2 = np.where(t2_valid, h[t2_safe], 0.0)
    
    a = term1 - term2
    return a
