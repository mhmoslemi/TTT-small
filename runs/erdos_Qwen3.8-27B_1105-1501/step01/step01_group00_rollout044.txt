            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def greedy_pairs(h, n_sweeps, n_samples, deadline):
    n = len(h)
    h = h.copy()
    current_score = score_exact(h)
    best_score = current_score
    pairs = [(a, b) for a in range(n) for b in range(a + 1, n)]
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for a, b in pairs:
            lo = max(-h[a], h[b] - 1.0)
            hi = min(1.0 - h[a], h[b])
            if hi - lo < 1e-10:
                continue
            if n_samples <= 1:
                deltas = np.array([(lo + hi) / 2.0])
            else:
                margins = (hi - lo) * 0.005
                deltas = np.linspace(lo + margins, hi - margins, n_samples)
            best_d = None
            best_s = current_score
            for d in deltas:
                h[a] += d
                h[b] -= d
                s = score_exact(h)
                h[a] -= d
                h[b] += d
                if s < best_s - 1e-14:
                    best_s = s
                    best_d = d
            if best_d is not None:
                h[a] += best_d
                h[b] -= best_d
                current_score = best_s
                if current_score < best_score:
                    best_score = current_score
                improved = True
        if not improved:
            break
    return h, best_score

def active_lag_indices(h, k_top):
    n = len(h)
    vals = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argpartition(vals, -k_top)[-k_top:]
    indices = set()
    for p in top_idx:
        s = p - (n - 1)
        if s >= 0:
            for i in range(n - s):
                indices.add(i + s)
                indices.add(i)
        else:
            t = -s
            for i in range(n - t):
                indices.add(i)
                indices.add(i + t)
    return sorted(indices)

def sa_focused(h, n_iter, T0, Tend, deadline, seed, k_top=15):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    current_score = score_exact(h)
    best_h = h.copy()
    best_score = current_score
    active = active_lag_indices(h, k_top)
    n_active = max(3, min(n, len(active)))
    for i in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (Tend / T0) ** (i / max(n_iter - 1, 1))
        if i % 200 == 0 and len(active) > n_active:
            active = active_lag_indices(h, k_top)
            n_active = max(3, min(n, len(active)))
        if rng.random() < 0.7:
            a = active[rng.integers(n_active)]
            b = active[rng.integers(n_active)]
        else:
            a = rng.integers(n)
            b = rng.integers(n)
        if a == b:
            continue
        lo = max(-h[a], h[b] - 1.0)
        hi = min(1.0 - h[a], h[b])
        if hi - lo < 1e-10:
            continue
        d = rng.uniform(lo, hi)
        h[a] += d
        h[b] -= d
        s_new = score_exact(h)
        delta = s_new - current_score
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            current_score = s_new
            if s_new < best_score - 1e-14:
                best_score = s_new
                best_h = h.copy()
        else:
            h[a] -= d
            h[b] += d
    return best_h, best_score

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def antisym_transform(h):
    h_new = 1.0 - h[::-1].copy()
    return make_feasible(h_new)

def mirror_avg(h):
    h_new = (h + 1.0 - h[::-1]) / 2.0
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve
    rng_main = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            best_h = make_feasible(parent)
            best_score = score_exact(best_h)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n = 60
        x = np.arange(n)
        best_h = make_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x / n) + 0.1 * np.cos(4 * np.pi * x / n))
        best_score = score_exact(best_h)

    def try_update(h_cand):
        nonlocal best_h, best_score
        if h_cand is None:
            return
        sc = score_exact(h_cand)
        if sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc

    # Phase 1: Initial greedy from parent
    h_new, sc = greedy_pairs(best_h, 80, 9, deadline)
    if sc < best_score - 1e-14:
        best_h = h_new
        best_score = sc

    # Phase 2: Finer greedy
    h_new, sc = greedy_pairs(best_h, 80, 15, deadline)
    if sc < best_score - 1e-14:
        best_h = h_new
        best_score = sc

    # Phase 3: Resolution sweep
    n_base = len(best_h)
    res_list = sorted(set([
        max(20, n_base - 10), max(20, n_base - 5),
        n_base,
        min(80, n_base + 5), min(80, n_base + 10), min(80, n_base + 15)
    ]))
    for n_opt in res_list:
        if time.time() > deadline:
            break
        h_int = interpolate_to(best_h, n_opt)
        h_new2, sc2 = greedy_pairs(h_int, 60, 9, deadline)
        if sc2 < best_score - 1e-14:
            best_h = h_new2
            best_score = sc2
            h_new, sc = greedy_pairs(best_h, 40, 13, deadline)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 4: Focus greedy with more samples on best
    h_new, sc = greedy_pairs(best_h, 60, 21, deadline)
    if sc < best_score - 1e-14:
        best_h = h_new
        best_score = sc

    # Phase 5: SA exploration
    h_new, sc = sa_focused(best_h, 10000, 0.008, 0.0001, deadline, seed)
    if sc < best_score - 1e-14:
        best_h = h_new
        best_score = sc
    h_new, sc = greedy_pairs(best_h, 50, 11, deadline)
    if sc < best_score - 1e-14:
        best_h = h_new
        best_score = sc

    # Phase 6: Anti-symmetric and mirror transforms
    if time.time() < deadline:
        h_anti = antisym_transform(best_h)
        try_update(h_anti)
        h_new, sc = greedy_pairs(h_anti, 50, 9, deadline)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
    if time.time() < deadline:
        h_mir = mirror_avg(best_h)
        try_update(h_mir)
        h_new, sc = greedy_pairs(h_mir, 50, 9, deadline)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 7: Random perturbations + re-optimize
    n = len(best_h)
    remaining = deadline - time.time()
    n_pert = max(2, min(15, int(remaining / 15)))
    for i in range(n_pert):
        if time.time() > deadline:
            break
        sigma = 0.03 + 0.07 * rng_main.random()
        h_pert = make_feasible(best_h + rng_main.normal(0, sigma, size=n))
        h_new, sc = greedy_pairs(h_pert, 40, 9, deadline)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 8: More resolution sweeps from updated best
    if time.time() < deadline:
        n_base2 = len(best_h)
        res_list2 = sorted(set([
            max(20, n_base2 - 8), max(20, n_base2 - 3),
            min(80, n_base2 + 3), min(80, n_base2 + 8)
        ]))
        for n_opt in res_list2:
            if time.time() > deadline:
                break
            h_int = interpolate_to(best_h, n_opt)
            h_new2, sc2 = greedy_pairs(h_int, 50, 11, deadline)
            if sc2 < best_score - 1e-14:
                best_h = h_new2
                best_score = sc2
                h_new, sc = greedy_pairs(best_h, 30, 15, deadline)
                if sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc

    # Phase 9: Final intensive polish
    if time.time() < deadline:
        h_new, sc = greedy_pairs(best_h, 40, 25, deadline)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 10: Last SA + greedy
    if time.time() < deadline:
        h_new, sc = sa_focused(best_h, 5000, 0.004, 0.00005, deadline, seed + 1)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
        h_new, sc = greedy_pairs(best_h, 30, 15, deadline)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-13

    return final_h, final_bound, final_n
