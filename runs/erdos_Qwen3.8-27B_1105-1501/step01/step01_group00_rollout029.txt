```python
def smooth_grad_vec2(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    g = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-8:
            continue
        s = idx - (n - 1)
        g += w[idx] * grad_lag(x, s)
    return dx * g

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def try_slsqp(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_obj_vec, h0, args=(beta, n), jac=smooth_grad_vec2,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None
```

```python
def grad_lag(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        # j < m: g[j] += (1 - h[j+s])
        g[:m] += (1.0 - h[s:s+m])  # h[s:s+m] = h[s], h[s+1], ..., h[s+m-1] = h[s], ..., h[n-1]
        # s <= j < n: g[j] -= h[j-s]
        g[s:] -= h[0:n-s]  # h[0], h[1], ..., h[n-s-1]
    else:
        t = -s
        m = n - t
        # j >= t, j < n: g[j] += (1 - h[j-t])
        g[t:] += (1.0 - h[0:m])
        # j < n-t: g[j] -= h[j+t]
        g[:m] -= h[t:n]
    return g
```
