            # j+s < n means j < n-s
            # Contribution to h[j] from first term: (1-h[j+s]) for j in [0, n-s-1]
            grad[:n-s] += w * dx * (1.0 - h[s:])
            # Contribution to h[j] from second term: -h[j-s] for j in [s, n-1]
            grad[s:] += w * dx * (-h[:n-s])
        else:
            s_pos = -s
            # j+s < n is always true (since j < n and s < 0, j+s < n-s_pos < n)
            # Actually j+s >= 0 means j >= -s = s_pos
            # First term: (1-h[j+s]) for j in [s_pos, n-1]
            grad[s_pos:] += w * dx * (1.0 - h[:n-s_pos])
            # Second term: -h[j-s] = -h[j+s_pos] for j in [0, s_pos-1]
            grad[:s_pos] += w * dx * (-h[s_pos:])
    return grad

def score_exact(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(x, target=None):
    n = len(x)
    if target is None:
        target = n / 2.0
    x = np.nan_to_num(np.asarray(x, dtype=float).ravel(), nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

            sp = -s

            grad[:sp] += wi * dx * (-h[sp:])

    e = np.sum(np.exp(shifted))

        return max_c
    return max_c + np.log(e) / beta

def smooth_lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx

    if total < 1e-300:

        w = e / total

            hs = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x / 2.0)
            hs = project_feasible(hs)
            all_starts.append((n, hs, score_exact(hs)))
    # Random starts
    for n in [28, 32, 36, 40]:
        for _ in range(3):
            hr = project_feasible(rng.random(n))
            all_starts.append((n, hr, score_exact(hr)))

    # Sort by score, take top candidates
    all_starts.sort(key=lambda x: x[2])
    top_starts = all_starts[:20]

    # Run improvement chains
    for n, h0, s0 in top_starts:
        if time.time() > deadline - 30:
            break
        h_opt, s_opt = improve_chain(h0, deadline)
        if s_opt < best_score - 1e-13:
            best_h = h_opt.copy()
            best_score = s_opt
            best_n = n

    # Phase C: Iterated local search around best
    remaining = deadline - time.time()
    n_trials = min(40, int(remaining / 15))
    for trial in range(n_trials):
        if time.time() > deadline - 10:
            break
        sigma = 0.02 + 0.03 * rng.random()
        h_pert = project_feasible(best_h + sigma * rng.standard_normal(best_n))
        h_opt, s_opt = improve_chain(h_pert, deadline)
        if s_opt < best_score - 1e-13:
            best_h = h_opt.copy()
            best_score = s_opt
            best_n = best_n

    # Phase D: Try different n around best
    for dn in [-2, -1, 1, 2, 3]:
        if time.time() > deadline - 15:
            break
        n_new = best_n + dn
        if n_new < 20 or n_new > 70:
            continue
        h_interp = project_feasible(interp_resize(best_h, n_new))
        h_opt, s_opt = improve_chain(h_interp, deadline)
        if s_opt < best_score - 1e-13:
            best_h = h_opt.copy()
            best_score = s_opt
            best_n = n_new

    # Final polish
    if time.time() < deadline - 5:
        final_h = direct_minimax(best_h, maxiter=200)
        fs = score_exact(final_h)
        if fs < best_score:
            best_h = final_h
            best_score = fs
        final_h2 = slsqp_polish(best_h, maxiter=200, beta=800)
        fs2 = score_exact(final_h2)
        if fs2 < best_score:
            best_h = final_h2
            best_score = fs2

    best_h = project_feasible(best_h)
    final_score = score_exact(best_h)

    return (best_h, final_score, len(best_h))
