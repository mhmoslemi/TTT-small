def exact_pair_delta(c, d, lo, hi):
    """Find delta in [lo, hi] minimizing max_k (c[k] + d[k]*delta).
    Returns (delta, opt_value)."""
    # Feasibility check
    if hi - lo < 1e-14:
        return lo, float(np.max(c + d * lo))
    
    # Binary search on V
    # V_hi = min(f(lo), f(hi))
    f_lo = float(np.max(c + d * lo))
    f_hi = float(np.max(c + d * hi))
    V_hi = min(f_lo, f_hi)
    V_lo = 0.0
    
    for _ in range(40):
        V_mid = (V_lo + V_hi) / 2.0
        # Check feasibility: exists delta in [lo,hi] with c[k]+d[k]*delta <= V_mid for all k
        lower = lo
        upper = hi
        feasible = True
        for k in range(len(d)):
            dk = d[k]
            ck = c[k]
            if dk > 1e-15:
                u = (V_mid - ck) / dk
                if u < upper:
                    upper = u
            elif dk < -1e-15:
                l = (V_mid - ck) / dk
                if l > lower:
                    lower = l
            else:
                if ck > V_mid:
                    feasible = False
                    break
        if lower > upper:
            feasible = False
        if feasible:
            V_hi = V_mid
        else:
            V_lo = V_mid
        if V_hi - V_lo < 1e-10:
            break
    
    # Find delta at V_hi
    lower = lo
    upper = hi
    for k in range(len(d)):
        dk = d[k]
        ck = c[k]
        if dk > 1e-15:
            u = (V_hi - ck) / dk
            if u < upper:
                upper = u
        elif dk < -1e-15:
            l = (V_hi - ck) / dk
            if l > lower:
                lower = l
    
    if lower > upper:
        delta = (lower + upper) / 2.0
    else:
        delta = (lower + upper) / 2.0
    delta = max(lo, min(hi, delta))
    
    return delta, float(np.max(c + d * delta))
