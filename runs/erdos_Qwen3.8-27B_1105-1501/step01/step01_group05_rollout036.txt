    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for ki in range(2*n - 1):
        wi = w[ki]
        if wi < 1e-10:
            continue
        k = lags[ki]
        scale = wi * dx
        if k >= 0:
            if k == 0:
                # c_0 = dx * sum h[i]*(1-h[i])
                # d/dh[m] = dx * (1 - 2*h[m])
                grad += scale * (1.0 - 2.0 * h)
            else:
                # m from 0 to n-1-k: contributes (1-h[m+k])
                m1 = np.arange(0, n - k)
                grad[m1] += scale * (1.0 - h[m1 + k])
                # m from k to n-1: contributes -h[m-k]
                m2 = np.arange(k, n)
                grad[m2] -= scale * h[m2 - k]
        else:
            kp = -k
            # m from kp to n-1: contributes (1-h[m-kp])
            m1 = np.arange(kp, n)
            grad[m1] += scale * (1.0 - h[m1 - kp])
            # m from 0 to n-1-kp: contributes -h[m+kp]
            m2 = np.arange(0, n - kp)
            grad[m2] -= scale * h[m2 + kp]
    return grad

def lse_optimize(h, tau_start, tau_end, n_max_iter, time_limit, seed):
    """Optimize LSE-smoothed objective with projected gradient + momentum."""
    n = len(h)
    sum_target = n / 2.0
    proj = project_h(sum_target)
    h = proj(h)
    rng = np.random.default_rng(seed)
    
    tau = tau_start
    best_h = h.copy()
    best_score = official_score(h)
    start_time = time.perf_counter()
    
    moment = np.zeros(n)
    beta1, beta2 = 0.9, 0.99
    
    while tau <= tau_end:
        # Run gradient descent at this tau
        step_size = 0.05
        for it in range(n_max_iter):
            if time.perf_counter() - start_time > time_limit:
                break
            
            grad = lse_gradient(h, tau)
            # Remove mean component (to help with sum constraint)
            grad -= np.mean(grad)
            
            # Momentum
            moment = beta1 * moment + (1 - beta1) * grad
            direction = -moment
            
            # Line search
            best_alpha = step_size
            best_f = lse_value(compute_scaled_corr(h), tau)
            for alpha in [step_size * 0.1, step_size * 0.3, step_size, step_size * 3, step_size * 10]:
                h_try = proj(h + alpha * direction)
                f_try = lse_value(compute_scaled_corr(h_try), tau)
                if f_try < best_f:
                    best_f = f_try
                    best_alpha = alpha
            
            h = proj(h + best_alpha * direction)
            
            # Check convergence
            if it > 0:
                pass  # Could check gradient norm
        
        # Update best
        score = official_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        # Decrease tau
        tau = min(tau * 2.0, tau_end)
    
    return best_h, best_score

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")  # raw, not scaled
    corr = corr_raw * dx  # scaled
    max_c = np.max(corr)
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # j indexes 0 to 2n-2, corresponding to lag k = j - (n-1)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        k = j - (n - 1)  # the actual lag
        scale = wj * dx  # since c_k = dx * (raw correlation at this position)
        
        if k == 0:
            grad += scale * (1.0 - 2.0 * h)
        elif k > 0:
            # c_k = dx * sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
            # dc_k/dh[m] = (1-h[m-k]) * 1_{k<=m<=n-1} - h[m+k] * 1_{0<=m<=n-1-k}
            # Term 1: (1-h[m-k]) for m=k,...,n-1
            if k < n:
                m_range = np.arange(k, n)
                grad[m_range] += scale * (1.0 - h[m_range - k])
            # Term 2: -h[m+k] for m=0,...,n-1-k
            if k < n:
                m_range = np.arange(0, n - k)
                grad[m_range] -= scale * h[m_range + k]
        else:
            # k < 0, let m' = -k
            mp = -k
            # c_{-mp} = dx * sum_{i=mp}^{n-1} h[i-mp] * (1-h[i])
            # dc/dh[p] = (1-h[p+mp]) * 1_{0<=p<=n-1-mp} - h[p-mp] * 1_{mp<=p<=n-1}
            # Term 1: (1-h[p+mp]) for p=0,...,n-1-mp
            m_range = np.arange(0, n - mp)
            grad[m_range] += scale * (1.0 - h[m_range + mp])
            # Term 2: -h[p-mp] for p=mp,...,n-1
            m_range = np.arange(mp, n)
            grad[m_range] -= scale * h[m_range - mp]
    
    return grad

Now let me think about the optimization strategy more carefully.

The key challenge is that as tau increases, the LSE objective becomes closer to the true max, but the gradient becomes more concentrated on the active constraints. This can lead to:
1. Large steps that overshoot
2. Oscillation between active constraints

To handle this, I'll use:
1. Adaptive step size with backtracking
2. Momentum (Adam-like) to smooth the direction

import numpy as np

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(1000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    return h


def project_affine_box(v, sum_target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > sum_target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def compute_scaled_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def lse_grad(h, tau):
    n = len(h)

    corr = compute_scaled_corr(h)
    max_c = np.max(corr)
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -80, 80)
    w = np.exp(shifted)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-10:
            continue
        k = j - nm1
        scale = wj * dx
        if k == 0:
            grad += scale * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[k:] += scale * (1.0 - h[:n - k])
                grad[:n - k] -= scale * h[k:]
        else:
            mp = -k
            if mp < n:
                grad[:n - mp] += scale * (1.0 - h[mp:])
                grad[mp:] -= scale * h[:n - mp]
    return grad

def slsqp_refine(h, n_target, time_limit=30.0):
    n = len(h)
    dx = 2.0 / n
    sum_target = n / 2.0
    init_score = official_score(h)

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq(x):
        hx = x[:n]
        return x[n] - dx * np.correlate(hx, 1.0 - hx, mode="full")

    lags = np.arange(-(n - 1), n)

    def ineq_jac(x):
        hx = x[:n]
        n_lags = 2 * n - 1
        J = np.zeros((n_lags, n + 1))
        J[:, n] = 1.0
        for j in range(n_lags):
            k = j - (n - 1)
            scale = dx
            if k == 0:
                J[j, :n] = -scale * (1.0 - 2.0 * hx)
            elif k > 0:
                J[j, k:n] += -scale * (1.0 - hx[:n - k])
                J[j, :n - k] += -scale * (-hx[k:])
            else:
                mp = -k
                J[j, :n - mp] += -scale * (1.0 - hx[mp:])
                J[j, mp:n] += -scale * (-hx[:n - mp])
        return J

    def eq(x):
        return np.sum(x[:n]) - sum_target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(init_score - 0.01, init_score + 0.01)]
    x0 = np.concatenate([h, [init_score + 0.001]])
    cons = [
        {'type': 'eq', 'fun': eq, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac}
    ]
    try:
        res = optimize.minimize(
            obj, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_out = repair_vector(res.x[:n], n)
            s = official_score(h_out)
            if s < init_score - 1e-12:
                return h_out, s
    except Exception:
        pass
    return h, init_score


def interpolate_h(parent, n_target):
    m = len(parent)
    if m == n_target:
        return parent.copy()
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = (np.arange(n_target) + 0.5) * 2.0 / n_target
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return h
