            if time.time() >= dl:
                break
            hi_ = interp_res(best_h, nt)
            h, sc = pair_greedy(hi_, dl, max_passes=80, n_samp=10, rng=rng)
            if sc < best_sc:
                best_h, best_sc = h, sc

    # Phase 5: Random restarts with greedy
    n = len(best_h)
    while time.time() < dl:
        sig = 0.02 + 0.15 * rng.random()
        hc = repair_exact(best_h + rng.normal(0, sig, n))
        h, sc = pair_greedy(hc, dl, max_passes=60, n_samp=10, rng=rng)
        if sc < best_sc:
            best_h, best_sc = h, sc
        # Also try smaller perturbations occasionally
        if time.time() < dl:
            hc2 = repair_exact(best_h + rng.normal(0, 0.02, n))
            h, sc = pair_greedy_fine(hc2, dl, n_samp=20, rng=rng)
            if sc < best_sc:
                best_h, best_sc = h, sc

    # Phase 6: Final intensive polishing
    if time.time() < dl:
        h, sc = pair_greedy_fine(best_h, dl, n_samp=25, rng=rng)
        if sc < best_sc:
            best_h, best_sc = h, sc
        h, sc = pair_greedy(best_h, dl, max_passes=200, n_samp=18, rng=rng)
        if sc < best_sc:
            best_h, best_sc = h, sc

    fh = repair_exact(best_h)
    fs = score(fh)
    return fh, fs + 1e-13, len(fh)
