def lse_smooth_and_grad(h, beta):
    """Compute log-sum-exp of lag values and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

        if w < 1e-15:
            continue

            # pairs: (i, i+shift) for i = 0..n-1-shift
            ell = n - shift
            # d/dh[i] = (1 - h[i+shift]) for i in 0..ell-1
            grad[:ell] += w * (1.0 - h[shift:])
            # d/dh[i+shift] = -h[i] for i in 0..ell-1, i.e., h[shift:]
            grad[shift:] -= w * h[:ell]

            # pairs: (i, i-t) for i = t..n-1, equivalently (j+t, j) for j=0..n-1-t

    return phi, grad

        if s_cur < best_score:
            best_h = h_cur.copy()
            best_score = s_cur
    
    # Phase 2: Optimize parent directly at its resolution and nearby
    for n in [max(24, parent_n - 4), parent_n, min(68, parent_n + 4)]:
        if time.time() > deadline - 50:
            break
        if n % 2 != 0:
            n = max(24, n - 1)
        h_i = project_feasible(interp_to_n(parent, n))
        s_i = official_score(h_i)
        
        h_cur = h_i.copy()
        s_cur = s_i
        for beta in [20, 60, 200, 600, 1500]:
            if time.time() > deadline - 40:
                break
            iters = min(300, max(60, int(5000 / max(n, 1))))
            h_new, s_new = plsed(h_cur, beta, max_iters=iters)
            if s_new < s_cur - 1e-14:
                h_cur = h_new
                s_cur = s_new
        
        if n <= 54 and time.time() < deadline - 30:
            h_sq = slsqp_direct(h_cur, maxiter=120)
            s_sq = official_score(h_sq)
            if s_sq < s_cur:
                h_cur = h_sq
                s_cur = s_sq
        
        if s_cur < best_score:
            best_h = h_cur.copy()
            best_score = s_cur
    
    # Phase 3: SLSQP smooth continuation on best
    if time.time() < deadline - 30:
        for beta in [50, 200, 600]:
            if time.time() > deadline - 20:
                break
            h_sq = slsqp_smooth(best_h, beta, maxiter=150)
            s_sq = official_score(h_sq)
            if s_sq < best_score:
                best_h = h_sq
                best_score = s_sq
    
    # Phase 4: Local refinement with small perturbations
    remaining = deadline - time.time()
    if remaining > 25:
        best_n = len(best_h)
        n_trials = min(40, int(remaining / 6))
        for trial in range(n_trials):
            if time.time() > deadline - 15:
                break
            sigma = 0.01 + 0.015 * (trial % 8)
            h_pert = project_feasible(best_h + sigma * rng.standard_normal(best_n))
            for beta in [150, 500]:
                if time.time() > deadline - 15:
                    break
                h_new, s_new = plsed(h_pert, beta, max_iters=120, step0=0.03)
                if s_new < best_score:
                    best_h = h_new.copy()
                    best_score = s_new
    
    # Phase 5: Final high-beta polish
    if time.time() < deadline - 12:
        for beta in [2000, 5000]:
            if time.time() > deadline - 12:
                break
            h_new, s_new = plsed(best_h, beta, max_iters=200, step0=0.01)
            if s_new < best_score:
                best_h = h_new.copy()
                best_score = s_new
    
    # Phase 6: One final SLSQP direct if time allows
    if time.time() < deadline - 10 and len(best_h) <= 54:
        h_sq = slsqp_direct(best_h, maxiter=150)
        s_sq = official_score(h_sq)
        if s_sq < best_score:
            best_h = h_sq
            best_score = s_sq
    
    # Final sanitization
    best_h = project_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
