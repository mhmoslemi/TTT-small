**Local search strategy**:

2. For each active shift k, identify "bad" pairs: positions i where h[i] is high and h[i+k] is low (or vice versa)

Actually, for continuous h, a "swap" isn't quite right. Let me think of better moves:

**Move types**:
1. **Push-pull**: Pick i and j, increase h[i] by δ and decrease h[j] by δ (preserves sum). This is a "flow" move.
2. **Squeeze**: Pick i, increase h[i] by δ and decrease all other h by δ*(1/(n-1)) (preserves sum). Less useful.
3. **Swap**: If h is near-binary, swap a 1 and a 0 (i.e., set h[i]=1, h[j]=0 if they were 0 and 1).

For the continuous case, move type 1 is most natural. The question is: which (i, j, δ) to try?

**Smart move selection**:
- For each active shift k, the "cost" of having h[i] high and h[i+k] low is h[i]*(1-h[i+k]).
- To reduce this, we want to either decrease h[i] or increase h[i+k].
- A good move: pick (i, i+k) for an active k, and move mass from h[i] to h[i+k] (or vice versa).

def local_search(h0, time_budget, seed=42):

    target = n / 2.0
    h = h0.astype(float).copy()

    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    while time.time() - t0 < time_budget:
        # Get active shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        tol = 0.02
        active_shifts = [int(m) - (n-1) for m in np.where(corr >= mx - tol)[0]]
        
        improved = False
        for _ in range(10000):
            if time.time() - t0 >= time_budget:
                break
            # Pick a random active shift
            k = rng.choice(active_shifts)
            # Find a "bad" pair at this shift
            if k >= 0:
                valid_i = range(n - k)
            else:
                valid_i = range(n + k)
            
            i = rng.choice(list(valid_i))
            j = i + k

            if h[i] > h[j]:

            h_test[i] += delta
            h_test[j] -= delta
            if h_test[i] < 0 or h_test[i] > 1 or h_test[j] < 0 or h_test[j] > 1:
                continue
            # Project to maintain sum constraint (the move preserves sum, so just clip)
            h_test = np.clip(h_test, 0, 1)

            h_test = project_simplex(h_test, target)

            if c5_test < best_c5 - 1e-10:

                h = h_test.copy()

                    corr_full[m] += h1 * (1.0 - h[j])
                j2 = i - k
                if 0 <= j2 < n:
                    corr_full[m] += h[i] * (1.0 - h[j2]) * 0  # already counted

    # Simpler: just compute corr from scratch each time (n=99 is fast)
    for step in range(n_ones):
        best_pos = -1
        best_val = float('inf')
        for pos in range(n):
            h[pos] = 1.0
            c = compute_c5(h)
            if c < best_val:
                best_val = c
                best_pos = pos
            h[pos] = 0.0
        h[best_pos] = 1.0

    if frac > 0.01:
        best_pos = -1
        best_val = float('inf')
        for pos in range(n):
            if h[pos] == 0:
                h[pos] = frac
                c = compute_c5(h)
                if c < best_val:
                    best_val = c
                    best_pos = pos
                h[pos] = 0.0
        if best_pos >= 0:
            h[best_pos] = frac

    h = project_simplex(h, target)
    return h


def try_rounding(h, target):
    n = len(h)
    c5_orig = compute_c5(h)

    # Round to 0/1 with one fractional
    sorted_idx = np.argsort(h)
    n_ones = int(target)
    frac = target - n_ones

    candidates = []
    # Standard rounding
    h_r = np.zeros(n)
    h_r[sorted_idx[n_ones:]] = 1.0
    if frac > 0.01:
        h_r[sorted_idx[n_ones - 1]] = frac
    h_r = project_simplex(h_r, target)
    candidates.append(h_r)

    # Alternative: threshold at 0.5
    h_t = (h >= 0.5).astype(float)
    h_t = project_simplex(h_t, target)
    candidates.append(h_t)

    best_h = h.copy()
    best_c5 = c5_orig
    for hc in candidates:
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc.copy()

    return best_h, best_c5


def create_init(n, rng):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * 5 // n) % 2 == 0 else 0.0
    h = project_simplex(h, n / 2.0)
    h += 0.15 * rng.randn(n)
    h = project_simplex(h, n / 2.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    best_h = None
    best_c5 = float('inf')

    if h_init is not None:
        best_h = h_init.astype(float).copy()
        best_c5 = compute_c5(best_h)
    else:
        n = 99
        best_h = create_init(n, rng)
        best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Main subgradient optimization (35% of time)
    if remaining() > 60:
        alloc = min(remaining() * 0.35, 350.0)
        h, c5 = opt_subgrad(best_h, alloc, seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 2: Targeted local search (25% of time)
    if best_h is not None and remaining() > 30:
        alloc = min(remaining() * 0.25, 220.0)
        h, c5 = local_search(best_h, alloc, seed + 100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 3: Try rounding
    if best_h is not None:
        h_r, c5_r = try_rounding(best_h, len(best_h) / 2.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 4: Another round of subgradient with different params (15%)
    if best_h is not None and remaining() > 40:
        alloc = min(remaining() * 0.15, 130.0)
        h, c5 = opt_subgrad(best_h, alloc, seed + 500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 5: Final local search (remaining time)
    if best_h is not None and remaining() > 15:
        alloc = remaining() - 10.0
        h, c5 = local_search(best_h, alloc, seed + 999)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Try greedy binary as alternative
    if remaining() > 30:
        n = len(best_h)
        h_g = greedy_binary(n, seed + 777)
        c5_g = compute_c5(h_g)
        if c5_g < best_c5:
            # Refine the greedy solution
            alloc = min(remaining() - 15.0, 100.0)
            h_r, c5_r = local_search(h_g, alloc, seed + 888)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
        elif c5_g < best_c5 + 0.005:
            # Close enough, refine it
            alloc = min(remaining() - 15.0, 80.0)
            if alloc > 10:
                h_r, c5_r = local_search(h_g, alloc, seed + 666)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r

    # Final: ensure constraint
    target = len(best_h) / 2.0
    best_h = project_simplex(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
