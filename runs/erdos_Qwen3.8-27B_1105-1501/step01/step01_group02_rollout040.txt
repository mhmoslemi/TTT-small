import numpy as np
import time


def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_effect_matrix(h):
    n = len(h)
    n_lags = 2 * n - 1
    eff = np.zeros((n, n_lags))
    idx = np.arange(n)
    for k in range(n_lags):
        s = k - (n - 1)
        ip = idx + s
        im = idx - s
        mp = (ip >= 0) & (ip < n)
        mm = (im >= 0) & (im < n)
        t1 = np.zeros(n)
        t1[mp] = 1.0 - h[ip[mp]]
        t2 = np.zeros(n)
        t2[mm] = -h[im[mm]]
        eff[:, k] = t1 + t2
    return eff


def find_best_delta(corr_vals, c, dfwd, drev, n_eval=25):
    best_c5 = float(np.max(corr_vals))
    best_delta = 0.0
    if dfwd > 1e-10:
        deltas = np.linspace(0.0, dfwd, n_eval)
        c5s = np.max(corr_vals[None, :] + deltas[:, None] * c[None, :], axis=1)
        idx = np.argmin(c5s)
        if c5s[idx] < best_c5 - 1e-12:
            best_c5 = float(c5s[idx])
            best_delta = float(deltas[idx])
    if drev > 1e-10:
        deltas = np.linspace(0.0, drev, n_eval)
        c5s = np.max(corr_vals[None, :] - deltas[:, None] * c[None, :], axis=1)
        idx = np.argmin(c5s)
        if c5s[idx] < best_c5 - 1e-12:
            best_c5 = float(c5s[idx])
            best_delta = -float(deltas[idx])
    return best_delta, best_c5


def optimize_n(h, target_sum, time_limit, rng, n_eval=25):
    n = len(h)
    best_h = h.copy()
    best_c5 = float(np.max(compute_corr_vals(h)))
    t0 = time.time()
    no_improve_rounds = 0

    while time.time() - t0 < time_limit - 1:
        corr_vals = compute_corr_vals(h)
        curr_c5 = float(np.max(corr_vals))
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            no_improve_rounds = 0

        eff = compute_effect_matrix(h)

        pairs = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
        rng.shuffle(pairs)

        round_improved = False
        updates_since_recompute = 0

        for idx in range(len(pairs)):
            if time.time() - t0 > time_limit - 1:
                break
            i, j = int(pairs[idx, 0]), int(pairs[idx, 1])

            c = eff[i] - eff[j]
            dfwd = min(h[j], 1.0 - h[i])
            drev = min(h[i], 1.0 - h[j])

            if dfwd < 1e-8 and drev < 1e-8:
                continue

            delta, new_c5 = find_best_delta(corr_vals, c, dfwd, drev, n_eval)

            if abs(delta) > 1e-7 and new_c5 < curr_c5 - 5e-10:
                h[i] += delta
                h[j] -= delta
                curr_c5 = new_c5
                round_improved = True
                updates_since_recompute += 1

                if updates_since_recompute >= 15:
                    corr_vals = compute_corr_vals(h)
                    eff = compute_effect_matrix(h)
                    curr_c5 = float(np.max(corr_vals))
                    updates_since_recompute = 0
                    if curr_c5 < best_c5 - 1e-12:
                        best_c5 = curr_c5
                        best_h = h.copy()

        if not round_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 2:
                scale = 0.02 + 0.02 * min(no_improve_rounds, 5)
                h = best_h.copy()
                h += rng.randn(n) * scale
                h = project_feasible(h, target_sum)
                no_improve_rounds = 0

    final_c5 = float(np.max(compute_corr_vals(best_h)))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5


def make_structured_h(n, rng):
    h = np.zeros(n)
    phi = 0.6180339887498949
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    h = project_feasible(h, n / 2.0)
    return h


def make_balanced_h(n, rng):
    h = np.zeros(n)
    for i in range(n):
        val = (i * 0.6180339887498949 + rng.uniform(0, 0.1)) % 1.0
        h[i] = 1.0 if val < 0.5 else 0.0
    h = project_feasible(h, n / 2.0)
    return h
