The objective is: min_h max_k c_k(h) subject to sum(h) = n/2, 0 ≤ h ≤ 1.

This is a min-max problem. I can use the von Neumann minimax theorem / game theory approach:
- Introduce a probability distribution p over the shifts k
- The Lagrangian dual is: min_h max_p sum_k p_k c_k(h) = min_h E_p[c_k(h)]
- The inner max over p is achieved by putting all mass on the max, which gives back the original.
- But we can use the LORENTZ/ROBINSON approach: alternate between:
  - Given p, minimize E_p[c_k(h)] over h (this is a QP since c_k is bilinear)
  - Given h, update p to put more weight on large c_k

This is the classical approach for min-max problems. Let me implement this.

Actually, c_k(h) = dx * sum_i h[i+offset_k] * (1-h[i]) is bilinear (not linear) in h. So minimizing E_p[c_k(h)] is a bilinear objective, which is nonconvex.

This gives us a cutting-plane / linearization approach:

4. Solve the resulting (nonlinear) feasibility problem.
5. Iterate.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr) * dx)
    return max_val, corr * dx

def compute_subgrad_top(h, m=5):
    """Compute average subgradient over top m active shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    max_c = np.max(scaled)
    
    # Get top m shifts
    top_vals, top_idx = np.sort(scaled), np.argsort(scaled)
    top_idx = top_idx[-m:]
    
    # Weight by how close to max (exponential weighting)
    weights = np.exp(50.0 * (scaled[top_idx] - max_c))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        offset = int(k) - (n - 1)
        # c_k = sum_i h[i+offset] * (1-h[i]) for valid i
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        
        # grad_k[j] = [1 if j in [i_lo+offset, i_hi+offset]] 
        #            - h[j+offset] * [j in [i_lo, i_hi]]
        #            - h[j-offset] * [j-offset in [i_lo, i_hi]]
        
        # Term 1
        t1_lo = i_lo + offset
        t1_hi = i_hi + offset
        t1_lo = max(0, t1_lo)
        t1_hi = min(n-1, t1_hi)
        if t1_lo <= t1_hi:
            grad[t1_lo:t1_hi+1] += w
        
        # Term 2: -h[j+offset] for j in [i_lo, i_hi]
        if i_lo <= i_hi:
            js = np.arange(i_lo, i_hi + 1)
            js_off = js + offset
            valid = (js_off >= 0) & (js_off < n)
            grad[js[valid]] -= w * h[js_off[valid]]

        if s > target:
            lo = mid
        else:
            hi = mid

        else:
            stall += 1
        h = h_new
        iterations += 1

        if stall > 30:
            t_left = time_budget - (time.time() - t0)
            if t_left > 3:
                h, c5_pd = pair_refine(h, 2, seed + iterations)
                if c5_pd < best_c5:
                    best_c5 = c5_pd
                    best_h = h.copy()
            stall = 0
            v = np.zeros(n)
            momentum = min(momentum * 1.003, 0.97)
            m_active = min(m_active + 1, 12)
            temp = max(temp * 0.92, 2.0)

        if stall > 250:
            rng_r = np.random.RandomState(seed + iterations * 13)
            h = best_h + 0.02 * rng_r.randn(n)
            h = project_simplex(h, target)
            v = np.zeros(n)
            stall = 0

    t_left = time_budget - (time.time() - t0)
    if t_left > 3:
        n_rounds = max(1, int(t_left * 0.8))
        h, c5_fin = pair_refine(h, n_rounds, seed + 99991)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h

    return best_h, best_c5


def make_golden_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    ones = n // 2
    used = set()
    for i in range(ones * 2):
        pos = int((i * phi) % 1.0 * n) % n
        if pos not in used:
            h[pos] = 1.0
            used.add(pos)
            if len(used) >= ones:
                break
    if len(used) < ones:
        for pos in range(n):
            if h[pos] == 0:
                h[pos] = 1.0
                used.add(pos)
                if len(used) >= ones:
                    break
    if n % 2 == 1 and np.sum(h) == ones:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = 0.5
    return h


def make_random_01(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = 0.5
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    # Phase 1: Heavy optimization of initial_h_values
    if h_init is not None:
        n0 = len(h_init)
        alloc = min(remaining() * 0.5, 500.0)
        if alloc > 20:
            h_best, c5_best = optimize_from(h_init, alloc, seed)
            best_c5 = c5_best
            best_h = h_best
            best_n = n0

    # Phase 2: Try nearby n with golden init
    if h_init is not None:
        n_cands = [len(h_init) - 2, len(h_init) - 1, len(h_init) + 1, len(h_init) + 2]
    else:
        n_cands = [98, 100, 96, 102]
    seen = set()
    n_cands = [n for n in n_cands if n > 40 and n not in seen and not seen.add(n)]

    for n in n_cands:
        if remaining() < 25:
            break
        alloc = min(remaining() * 0.13, 120.0)
        if alloc < 12:
            break
        h_g = make_golden_init(n)
        h_g = project_simplex(h_g, float(n) / 2.0)
        c5_g = compute_c5(h_g)
        if c5_g < best_c5 + 0.03:
            h_best, c5_best = optimize_from(h_g, alloc, seed + n * 37)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best
                best_n = n

    # Phase 3: Random 0/1 attempts
    for trial in range(2):
        if remaining() < 20:
            break
        n = best_n if best_h is not None else 100
        if n % 2 != 0:
            n += 1
        h_r = make_random_01(n, rng)
        alloc = min(remaining() * 0.08, 60.0)
        if alloc < 8:
            break
        h_best, c5_best = optimize_from(h_r, alloc, seed + 400 + trial * 77)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best
            best_n = n

    # Phase 4: Final intensive pairing on best
    if best_h is not None and remaining() > 8:
        alloc = min(remaining() * 0.35, 150.0)
        n_rounds = max(1, int(alloc * 1.2))
        h_best, c5_best = pair_refine(best_h, n_rounds, seed + 31337)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best

    # Final safety
    if best_h is None:
        n = 99
        best_h = make_golden_init(n)
        best_n = n

    target = float(best_n) / 2.0
    best_h = project_simplex(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
