```python
            g = -G[max_idx, :]
            g -= np.mean(g)
            norm = np.max(np.abs(g))
            if norm < 1e-12:
                break
            d = g / norm * step_size * 0.1
        else:
            # Scale d to use the full step budget
            d_norm = np.sum(np.abs(d))
            if d_norm > step_size:
                d *= step_size / d_norm
            elif d_norm > 1e-10:
                # Try to scale up a bit
                scale = step_size / d_norm
                if scale < 10:  # don't scale up too much
                    d *= scale
        
        # Line search
        best_alpha = 0.0
        best_local = score
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, -0.1, -0.3, -0.5]:
            h_new = project_to_target(h + alpha * d, n)
            s = official_score(h_new)
            if s < best_local - 1e-12:
                best_local = s
                best_alpha = alpha
        
        if best_alpha != 0.0:
            h = project_to_target(h + best_alpha * d, n)
            score = best_local
        
        # Random perturbation to escape local minima
        if it % 10 == 9:
            noise = rng.normal(0, 0.02, n)
            noise -= np.mean(noise)
            h = project_to_target(h + noise, n)
            new_score = official_score(h)
            if new_score < score + 0.005:  # allow slight worsening
                score = new_score
    
    return h, official_score(h)
```

```python
import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    h = np.clip(np.asarray(v, dtype=float).ravel(), 0.0, 1.0)
    target = n / 2.0
    lo = 1.0 - h.max() - 1.0
    hi = 2.0 - h.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        residual = target - h.sum()
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + residual, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + residual, 0.0)
    return h

def score_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    p = np.arange(n)
    for r in range(L):
        k = r - (n - 1)
        ok1 = (p - k >= 0) & (p - k < n)
        G[r, ok1] += (1.0 - h[p[ok1] - k])
        ok2 = (p + k >= 0) & (p + k < n)
        G[r, ok2] += (-h[p[ok2] + k])
    G *= dx
    return G

def lp_find_direction(h, step_size=0.5):
    n = len(h)
    c = score_all(h)
    G = compute_gradient_matrix(h)
    L = len(c)
    num_vars = 2 * n + 1
    idx_t = 2 * n
    
    c_obj = np.zeros(num_vars)
    c_obj[idx_t] = 1.0
    
    # Build A_ub: for each r, G[r,:] @ d - t <= -c[r]
    A_ub = np.zeros((L + 1, num_vars))
    b_ub = np.zeros(L + 1)
    for r in range(L):
        A_ub[r, :n] = G[r, :]
        A_ub[r, n:2*n] = -G[r, :]
        A_ub[r, idx_t] = -1.0
        b_ub[r] = -c[r]
    # L1 norm constraint
    A_ub[L, :n] = 1.0
    A_ub[L, n:2*n] = 1.0
    b_ub[L] = step_size
    
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)
    
    bounds = [(0, None)] * (2*n) + [(0.0, None)]
    # t should be at least min(c) to be meaningful, but let's leave it unbounded below
    # Actually t >= 0 is a safe lower bound since correlations are non-negative
    bounds[idx_t] = (0.0, None)
    
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                               bounds=bounds, method='highs',
                               options={'presolve': True})
        if res.status == 0:
            d = res.x[:n] - res.x[n:2*n]
            return d
    except Exception:
        pass
    return None

def lp_local_search(h, max_iters=40, time_limit=30.0, seed=42):
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    start = time.perf_counter()
    rng = np.random.default_rng(seed)
    step_size = 0.5
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        d = lp_find_direction(h, step_size)
        
        if d is None:
            # Fallback: negative gradient of max correlation
            c = score_all(h)
            max_idx = np.argmax(c)
            G = compute_gradient_matrix(h)
            d = -G[max_idx, :]
            d -= np.mean(d)
            norm = np.max(np.abs(d))
            if norm < 1e-12:
                break
            d = d / norm * 0.1
        
        # Line search with multiple step sizes
        best_alpha = 0.0
        best_local = score
        alphas = [1.0, 0.5, 0.3, 0.1, 0.7, 0.2, -0.1, -0.3]
        for a in alphas:
            h_new = project_to_target(h + a * d, n)
            s = official_score(h_new)
            if s < best_local - 1e-12:
                best_local = s
                best_alpha = a
        
        if best_alpha != 0.0:
            h = project_to_target(h + best_alpha * d, n)
            score = best_local
            no_improve_count = 0
            # Reduce step size slightly as we converge
            step_size = max(0.1, step_size * 0.95)
        else:
            no_improve_count += 1
            if no_improve_count >= 5:
                # Try random perturbation
                noise = rng.normal(0, 0.05, n)
                noise -= np.mean(noise)
                h = project_to_target(h + noise, n)
                s = official_score(h)
                if s < score + 0.01:
                    score = s
                no_improve_count = 0
                step_size = 0.5
            if no_improve_count >= 15:
                break
    
    return h, official_score(h)

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) * 1.0) * 2.0 / n
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return project_to_target(h, n)

def gen_random_binary(n, rng):
    if n % 2 == 0:
        target_ones = n // 2
        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:target_ones]] = 1.0

        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:target_ones]] = 1.0
        h[idx[target_ones]] = 0.5
    return project_to_target(h, n)

def gen_random_continuous(n, rng):
    h = rng.uniform(0.1, 0.9, n)

def gen_sine(n, rng):
    t = np.linspace(0, 1, n)

    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * t + phase)

def slsqp_polish(h, time_limit=15.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    L = 2 * n - 1
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        G = compute_gradient_matrix(h_x)  # already includes dx
        j = np.zeros((L, n + 1))
        j[:, n] = 1.0
        j[:, :n] = -G
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    initial_score = official_score(h)
    theta0 = initial_score + 1e-5
    x0 = np.concatenate([h, [theta0]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if result.success or result.status == 0:
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score
```

```python
def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n):
    h = np.clip(np.asarray(v, dtype=float).ravel(), 0.0, 1.0)
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    target = n / 2.0
    lo = 1.0 - h.max() - 1.0
    hi = 2.0 - h.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(300):
        residual = target - h.sum()
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + residual, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + residual, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def score_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    p = np.arange(n)
    for r in range(L):
        k = r - (n - 1)
        ok1 = (p - k >= 0) & (p - k < n)
        G[r, ok1] += (1.0 - h[p[ok1] - k])
        ok2 = (p + k >= 0) & (p + k < n)
        G[r, ok2] += (-h[p[ok2] + k])
    G *= dx
    return G

            break

        d = lp_find_direction(h, step_size)

        if d is None:
            c = score_all(h)
            max_idx = int(np.argmax(c))
            G = compute_gradient_matrix(h)
            active_mask = c >= c[max_idx] - 0.01
            d = -np.sum(G[active_mask, :], axis=0)
            d -= np.mean(d)
            norm = np.max(np.abs(d))

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = (np.arange(n) + 0.5 + offset) * 2.0 / n
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return project_to_target(h, n)

def gen_sine(n, rng):
    t = np.linspace(0, 1, n)
    freq = int(rng.integers(2, 9))
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.03, n)
    return project_to_target(h, n)


def gen_step_pattern(n, rng):
    n_steps = int(rng.integers(3, 8))
    if n_steps >= n:
        return gen_random_continuous(n, rng)
    boundaries = sorted(rng.choice(n - 1, size=n_steps - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    vals = rng.uniform(0.0, 1.0, n_steps)
    h = np.zeros(n)
    for i in range(n_steps):
        h[boundaries[i]:boundaries[i + 1]] = vals[i]
    return project_to_target(h, n)

def slsqp_polish(h, time_limit=15.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    L = 2 * n - 1

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        G = compute_gradient_matrix(h_x)
        j = np.zeros((L, n + 1))
        j[:, n] = 1.0
        j[:, :n] = -G
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass

    return h, initial_score

    for n_try in n_list:
        candidates.append(gen_interpolated(best_h, n_try))
        candidates.append(gen_interpolated(best_h, n_try, offset=1))
        candidates.append(gen_interpolated(best_h, n_try, offset=2))
        candidates.append(gen_interpolated(best_h[::-1], n_try))

    for n_try in n_list[:5]:
        for _ in range(3):
            candidates.append(gen_random_continuous(n_try, rng))
        for _ in range(4):
            candidates.append(gen_random_binary(n_try, rng))
        for _ in range(2):
            candidates.append(gen_sine(n_try, rng))
        for _ in range(2):
            candidates.append(gen_step_pattern(n_try, rng))

    scored = []
    seen_scores = set()
    for h_c in candidates:
        n_c = len(h_c)
        if n_c < 20:
            continue
        h_c = project_to_target(h_c, n_c)
        if not np.all(np.isfinite(h_c)):
            continue
        s = official_score(h_c)
        if np.isfinite(s) and s > 0:
            scored.append((s, h_c))

    scored.sort(key=lambda x: (x[0], len(x[1])))

    if scored and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    n_top = min(10, len(scored))
    for idx in range(n_top):
        if time.perf_counter() > deadline - 150:
            break
        s_c, h_c = scored[idx]
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_limit = min(35.0, max(8.0, remaining * 0.55 / max(1, n_top - idx)))
        h_loc, s_loc = lp_local_search(h_c, max_iters=50, time_limit=t_limit, seed=seed + idx * 17)
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c
            # Try SLSQP immediately after good LP result
            if time.perf_counter() < deadline - 30:
                h_sp, s_sp = slsqp_polish(h_loc, time_limit=10.0)
                if s_sp < best_score:
                    best_score = s_sp
                    best_h = h_sp.copy()

    # Second round: interpolate best to new resolutions and refine
    if time.perf_counter() < deadline - 100:
        second_candidates = []
        for n_try in [best_n, best_n + 4, best_n + 8, best_n - 4, best_n + 12]:
            if n_try < 24 or n_try > 140:
                continue
            second_candidates.append(gen_interpolated(best_h, n_try))
            second_candidates.append(gen_interpolated(best_h, n_try, offset=1))
            second_candidates.append(gen_interpolated(best_h[::-1], n_try))

        scored2 = []
        for h_c in second_candidates:
            n_c = len(h_c)
            h_c = project_to_target(h_c, n_c)
            s = official_score(h_c)
            if np.isfinite(s) and s > 0:
                scored2.append((s, h_c))
        scored2.sort(key=lambda x: (x[0], len(x[1])))

        n_top2 = min(5, len(scored2))
        for idx in range(n_top2):
            if time.perf_counter() > deadline - 80:
                break
            s_c, h_c = scored2[idx]
            n_c = len(h_c)
            remaining = deadline - time.perf_counter()
            t_limit = min(20.0, max(5.0, remaining * 0.5 / max(1, n_top2 - idx)))
            h_loc, s_loc = lp_local_search(h_c, max_iters=40, time_limit=t_limit, seed=seed + 500 + idx * 23)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()
                best_n = n_c

    # Discrete search
    if time.perf_counter() < deadline - 70:
        n_disc = best_n if best_n % 2 == 0 else best_n + 1
        if 30 <= n_disc <= 110:
            h_disc_init = gen_interpolated(best_h, n_disc)
            h_d, s_d = discrete_search(h_disc_init, time_limit=25.0, seed=seed + 200)
            if s_d < best_score:
                best_score = s_d
                best_h = h_d.copy()
                best_n = n_disc
            # LP polish the discrete result
            if time.perf_counter() < deadline - 40:
                h_lp, s_lp = lp_local_search(h_d, max_iters=25, time_limit=15.0, seed=seed + 400)
                if s_lp < best_score:
                    best_score = s_lp
                    best_h = h_lp.copy()
                    best_n = n_disc

        # Also try a different n for discrete
        n_disc2 = 64 if best_n != 64 else 66
        if time.perf_counter() < deadline - 40 and 30 <= n_disc2 <= 110:
            h_d2_init = gen_interpolated(best_h, n_disc2)
            h_d2, s_d2 = discrete_search(h_d2_init, time_limit=15.0, seed=seed + 250)
            if s_d2 < best_score:
                best_score = s_d2
                best_h = h_d2.copy()
                best_n = n_disc2
            if time.perf_counter() < deadline - 25:
                h_lp2, s_lp2 = lp_local_search(h_d2, max_iters=20, time_limit=10.0, seed=seed + 450)
                if s_lp2 < best_score:
                    best_score = s_lp2
                    best_h = h_lp2.copy()
                    best_n = n_disc2

    # Final polish
    if time.perf_counter() < deadline - 20:
        h_pol, s_pol = slsqp_polish(best_h, time_limit=12.0)
        if s_pol < best_score:
            best_score = s_pol
            best_h = h_pol.copy()
        if time.perf_counter() < deadline - 10:
            h_lp3, s_lp3 = lp_local_search(best_h, max_iters=15, time_limit=7.0, seed=seed + 999)
            if s_lp3 < best_score:
                best_score = s_lp3
                best_h = h_lp3.copy()

    # Additional: try one more SLSQP + LP cycle if time allows
    if time.perf_counter() < deadline - 15:
        h_pol2, s_pol2 = slsqp_polish(best_h, time_limit=8.0)
        if s_pol2 < best_score:
            best_score = s_pol2
            best_h = h_pol2.copy()

    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)
    best_h = project_to_target(best_h, len(best_h))

    return (best_h, float(final_score), int(len(best_h)))
```
