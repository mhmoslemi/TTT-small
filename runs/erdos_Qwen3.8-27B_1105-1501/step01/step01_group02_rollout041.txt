def compute_c5(h):

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual
        n = len(result)
        # Find indices where we can add/subtract
        if residual > 0:
            # Need to add: find indices with h < 1
            can_add = result < 1.0 - 1e-10
        else:
            can_add = result > 1e-10
        if np.any(can_add):
            indices = np.where(can_add)[0]
            amount = residual / len(indices)
            result[indices] += amount
            result = np.clip(result, 0.0, 1.0)

            if abs(residual2) > 1e-12 and len(indices) > 0:

    return result

def make_binary_pattern(n, rng):
    """Create a random binary pattern with exactly n//2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def make_structured_pattern(n, rng, style=0):
    """Create structured patterns."""
    h = np.zeros(n)
    half = n // 2
    if style == 0:
        # Alternating blocks
        block_size = max(1, n // 20)
        positions = []
        i = 0
        while i < n:
            positions.append(i)
            i += 2 * block_size
        h[positions[:half]] = 1.0
    elif style == 1:
        # Golden ratio
        phi = 0.6180339887
        positions = np.array([int((i * phi * n) % n) for i in range(n)])
        positions = np.unique(positions)
        h[positions[:half]] = 1.0
    elif style == 2:
        # Striped
        h[::2] = 1.0
        if np.sum(h) != half:
            # Adjust
            diff = half - np.sum(h)
            idx = np.where(h == 0)[0]
            for k in range(abs(int(diff))):
                if diff > 0:
                    h[idx[k]] = 1
                else:
                    h[n//2 + k] = 0
    else:
        # Random binary
        h = make_binary_pattern(n, rng)
    return h

def pair_line_search(h, i, j, n_samples=51):
    """Do a line search along the pair direction (h[i]+t, h[j]-t)."""
    n = len(h)
    target_sum = n / 2.0
    hi_val = h[i]
    hj_val = h[j]
    
    lo = max(-hi_val, hj_val - 1.0) + 1e-10
    hi = min(1.0 - hi_val, hj_val) - 1e-10
    if hi <= lo:
        return h, False
    
    best_t = 0.0
    best_c5 = compute_c5(h)
    
    # Coarse search
    ts = np.linspace(lo, hi, n_samples)
    for t in ts:
        h[i] = hi_val + t
        h[j] = hj_val - t
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    if best_t == 0.0:
        h[i] = hi_val
        h[j] = hj_val
        return h, False
    
    # Fine search around best_t
    t_range = (hi - lo) / n_samples
    fine_lo = max(lo, best_t - t_range)
    fine_hi = min(hi, best_t + t_range)
    ts_fine = np.linspace(fine_lo, fine_hi, 21)
    for t in ts_fine:
        h[i] = hi_val + t
        h[j] = hj_val - t
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    h[i] = hi_val + best_t
    h[j] = hj_val - best_t
    return h, True

                c5 = compute_c5(x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x
        except Exception:
            pass

def coord_descent_sweep(h, target_sum, n_test=32):
    """One sweep of coordinate descent with line search."""
    n = len(h)
    improved = 0
    for j in range(n):
        sum_others = np.sum(h) - h[j]
        lo = max(0.0, target_sum - sum_others)
        hi = min(1.0, target_sum - sum_others)
        if hi - lo < 1e-10:
            continue
        cur_c5 = compute_c5(h)
        best_val = h[j]
        best_c5_j = cur_c5
        test_vals = np.linspace(lo, hi, n_test)
        for val in test_vals:
            h[j] = val
            c5_j = compute_c5(h)
            if c5_j < best_c5_j:
                best_c5_j = c5_j
                best_val = val
        h[j] = best_val
        if best_c5_j < cur_c5 - 1e-12:
            improved += 1
    return h, improved

import numpy as np
import time
from scipy.optimize import minimize

# Global state for scipy optimizer
_g_temp = 0.1
_g_dx = 0.0
_g_target = 0.0
_g_rho = 50.0

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(x):
    dx = _g_dx
    T = max(_g_temp, 1e-6)
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    v = vals / T
    v -= np.max(v)
    lse = T * (np.max(v) + np.log(np.sum(np.exp(v))))
    resid = np.sum(x) - _g_target
    return lse + _g_rho * resid * resid

def smooth_grad(x):
    dx = _g_dx
    T = max(_g_temp, 1e-6)
    rho = _g_rho
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    v = vals / T
    v -= np.max(v)
    expv = np.exp(v)
    w = expv / np.sum(expv)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if s >= 0:
            # First term: i in [s, n-1], contrib (1-x[i-s])
            if s < n:
                grad[s:n] += wk * (1.0 - x[0:n-s])
            # Second term: i in [0, n-s-1], contrib -x[i+s]
            if s < n:
                grad[0:n-s] -= wk * x[s:n]
        else:
            # s < 0, let r = -s > 0
            r = -s
            # First term: i in [0, n-r-1], contrib (1-x[i+r])
            grad[0:n-r] += wk * (1.0 - x[r:n])
            # Second term: i in [r, n-1], contrib -x[i-r]
            grad[r:n] -= wk * x[0:n-r]
    
    resid = np.sum(x) - _g_target
    grad += 2.0 * rho * resid
    
    return grad

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        prog = np.sum(np.clip(h - mid, 0.0, 1.0))
        if prog > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix exact sum
    resid = target_sum - np.sum(result)
    if abs(resid) > 1e-13:
        if resid > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            result[idx] += resid / len(idx)
            result = np.clip(result, 0.0, 1.0)
            resid2 = target_sum - np.sum(result)
            if abs(resid2) > 1e-13:
                result[idx[0]] = np.clip(result[idx[0]] + resid2, 0.0, 1.0)
    return result

def lbfgs_opt(h, n, target_sum, t_limit, temps):
    global _g_temp, _g_dx, _g_target, _g_rho
    dx = 2.0 / n
    _g_dx = dx
    _g_target = target_sum
    _g_rho = 80.0
    bounds = [(0.0, 1.0)] * n
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for T in temps:
        if time.time() >= t_limit:
            break
        _g_temp = T
        try:
            res = minimize(smooth_obj, best_h.copy(), jac=smooth_grad,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 150, 'ftol': 1e-9, 'gtol': 1e-7})
            x = project(res.x, target_sum)
            c5 = compute_c5(x)
            if c5 < best_c5:

                best_h = x
        except Exception:

    return best_h, best_c5

def pair_swap_sa(h, n, target_sum, t_limit, rng, init_temp=1.0, cool_rate=0.9995):
    """Simulated annealing with pair swaps on binary patterns."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    temp = init_temp
    
    # For binary, track ones and zeros
    ones = set(np.where(h >= 0.999)[0].tolist())
    zeros = set(np.where(h <= 0.001)[0].tolist())
    
    while time.time() < t_limit:
        # Choose a random pair (one, zero)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]
        
        # Swap: h[i] 1->0, h[j] 0->1
        old_i = h[i]
        old_j = h[j]
        h[i] = old_j  # becomes 0
        h[j] = old_i  # becomes 1
        new_c5 = compute_c5(h)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            delta = new_c5 - cur_c5
            if temp > 0.001 and rng.random() < np.exp(-delta / (temp * 0.01)):
                # Accept worse move
                cur_c5 = new_c5
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
                zeros.add(i)
            else:
                # Revert
                h[i] = old_i
                h[j] = old_j
        
        temp *= cool_rate
    
    return best_h, best_c5

    if abs(resid) > 1e-13:
        if resid > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            result[idx] += resid / len(idx)
            result = np.clip(result, 0.0, 1.0)
            resid2 = target_sum - np.sum(result)
            if abs(resid2) > 1e-13:
                result[idx[0]] = np.clip(result[idx[0]] + resid2, 0.0, 1.0)
    return result

def lbfgs_opt(h, n, target_sum, t_limit, temps):
    global _g_temp, _g_dx, _g_target, _g_rho
    _g_dx = 2.0 / n
    _g_target = target_sum
    _g_rho = 80.0
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for T in temps:
        if time.time() >= t_limit:
            break
        _g_temp = T
        try:
            res = minimize(smooth_obj, best_h.copy(), jac=smooth_grad,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 120, 'ftol': 1e-9, 'gtol': 1e-7})
            x = project(res.x, target_sum)
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x
        except Exception:
            pass
    return best_h, best_c5

def swap_delta_binary(h, corr, i, j):
    """Compute delta of correlation vector when swapping h[i]=1 to 0 and h[j]=0 to 1.
    For binary h, s=0 no change. For s!=0: delta[s] = h[i+s]+h[i-s]-h[j+s]-h[j-s] (0 if out of range).
    Returns (new_max_dx, changed) where new_max_dx is dx * new_max_corr."""
    n = len(h)
    dx = 2.0 / n
    # Build delta vector for all shifts s in [-(n-1), n-1]
    # corr[k] corresponds to s = k - (n-1)
    # delta for shift s: h[i+s] + h[i-s] - h[j+s] - h[j-s] (each term 0 if index out of [0,n-1])
    # We only need to compute the new max, so:
    new_corr = corr.copy()
    # Shifts affected: s where any of i+s, i-s, j+s, j-s is in [0,n-1]
    # For binary h, s=0 gives 0 change (verified).
    # s ranges from -(n-1) to n-1, index k = s + (n-1)
    for s in range(-(n-1), n):
        if s == 0:
            continue
        di = 0.0
        if 0 <= i + s < n:
            di += h[i + s]
        if 0 <= i - s < n:
            di += h[i - s]
        if 0 <= j + s < n:
            di -= h[j + s]
        if 0 <= j - s < n:
            di -= h[j - s]
        if di != 0.0:
            new_corr[s + (n-1)] += di
    new_max = np.max(new_corr)
    return float(new_max * dx), new_corr

def binary_sa(h, n, target, t_limit, rng, init_temp=0.8, cool=0.9997):
    """Simulated annealing on binary patterns with O(n) swap evaluation."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    ones = np.where(h >= 0.999)[0]
    zeros = np.where(h <= 0.001)[0]
    temp = init_temp
    best_no_change = 0
    
    while time.time() < t_limit and temp > 0.005:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        new_c5, _ = swap_delta_binary(h, corr, i, j)
        delta_c5 = new_c5 - cur_c5
        
        if delta_c5 < -1e-12:
            # Accept improvement
            h[i], h[j] = 0.0, 1.0
            cur_c5 = new_c5
            # Update corr
            for s in range(-(n-1), n):
                if s == 0:
                    continue
                di = 0.0
                if 0 <= i + s < n:
                    di += h[i + s]
                if 0 <= i - s < n:
                    di += h[i - s]
                if 0 <= j + s < n:
                    di -= h[j + s]
                if 0 <= j - s < n:
                    di -= h[j - s]
                if di != 0.0:
                    corr[s + (n-1)] += di
            # Update sets
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                best_no_change = 0
            else:
                best_no_change += 1
        else:
            # Maybe accept worse
            if temp > 0.01 and rng.random() < np.exp(-delta_c5 / (temp * dx * 0.5)):
                h[i], h[j] = 0.0, 1.0
                cur_c5 = new_c5
                for s in range(-(n-1), n):
                    if s == 0:
                        continue
                    di = 0.0
                    if 0 <= i + s < n:
                        di += h[i + s]
                    if 0 <= i - s < n:
                        di += h[i - s]
                    if 0 <= j + s < n:
                        di -= h[j + s]
                    if 0 <= j - s < n:
                        di -= h[j - s]
                    if di != 0.0:
                        corr[s + (n-1)] += di
                ones = ones[ones != i]
                ones = np.append(ones, j)
                zeros = zeros[zeros != j]
                zeros = np.append(zeros, i)
            # else: reject, do nothing
        
        temp *= cool
    
    return best_h, best_c5

def pair_move_refine(h, n, target, t_limit, rng, n_coarse=12, n_fine=25):
    """Refine with pair moves and adaptive line search."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() < t_limit:
        any_improve = False
        # Try a batch of random pairs
        n_tries = 80
        for _ in range(n_tries):
            if time.time() >= t_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi = h[i]
            hj = h[j]
            lo_t = max(-hi, hj - 1.0) + 1e-10
            hi_t = min(1.0 - hi, hj) - 1e-10
            if hi_t <= lo_t:
                continue
            # Coarse line search
            cur_c5 = compute_c5(h)
            best_t = 0.0
            best_c = cur_c5
            ts = np.linspace(lo_t, hi_t, n_coarse)
            for t in ts:
                h[i] = hi + t
                h[j] = hj - t
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_t = t
            if best_t != 0.0:
                # Fine search
                r = (hi_t - lo_t) / n_coarse
                fl = max(lo_t, best_t - r)
                fh = min(hi_t, best_t + r)
                for t in np.linspace(fl, fh, n_fine):
                    h[i] = hi + t
                    h[j] = hj - t
                    c = compute_c5(h)
                    if c < best_c:
                        best_c = c
                        best_t = t
                h[i] = hi + best_t
                h[j] = hj - best_t
                if best_c < best_c5 - 1e-11:
                    best_c5 = best_c
                    best_h = h.copy()
                    any_improve = True
            else:
                h[i] = hi
                h[j] = hj
        
        if not any_improve:
            # Perturb to escape
            h = best_h + rng.randn(n) * 0.025
            h = np.clip(h, 0, 1)
            h = project(h, target)
    
    return best_h, best_c5

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def binarize(h, target):
    """Convert near-binary h to exact binary."""
    h_b = (h >= 0.5).astype(float)
    cur_sum = int(np.sum(h_b))
    need = int(target)
    if cur_sum > need:
        idx_ones = np.where(h_b == 1)[0]
        # Remove ones with smallest h value (least committed)
        order = idx_ones[np.argsort(h[idx_ones])]
        for idx in order[:cur_sum - need]:
            h_b[idx] = 0
    elif cur_sum < need:
        idx_zeros = np.where(h_b == 0)[0]
        order = idx_zeros[np.argsort(-h[idx_zeros])]
        for idx in order[:need - cur_sum]:
            h_b[idx] = 1
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    global _g_temp, _g_dx, _g_target, _g_rho
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get parent
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass
    
    best_h = None
    best_c5 = float('inf')
    
    if h0 is not None and len(h0) > 0:
        n = len(h0)
        target = n / 2.0
        h0 = project(h0, target)
        c5_0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_0
    else:
        n = 70
        target = 2.0 * 35.0
        best_h = make_binary(n, rng)
        best_c5 = compute_c5(best_h)
    
    n_init = len(best_h)
    target_init = n_init / 2.0
    
    # Time allocation
    t1 = t0 + budget_s * 0.30   # L-BFGS phase
    t2 = t0 + budget_s * 0.55   # Pair move refinement
    t3 = t0 + budget_s * 0.80   # Binary SA
    t4 = t0 + budget_s - 8      # Other n / final polish
    
    # Phase 1: L-BFGS on parent
    temps = [0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002]
    h1, c5_1 = lbfgs_opt(best_h, n_init, target_init, t1, temps)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 1b: L-BFGS on binarized version
    h_bin = binarize(best_h, target_init)
    h1b, c5_1b = lbfgs_opt(h_bin, n_init, target_init, min(t1, t0 + budget_s * 0.18), temps[:6])
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b
    
    # Phase 2: Pair move refinement on best
    h2, c5_2 = pair_move_refine(best_h, n_init, target_init, t2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2
    
    # Phase 2b: Also refine binary version
    h_bin2 = binarize(best_h, target_init)
    h2b, c5_2b = pair_move_refine(h_bin2, n_init, target_init, min(t2, t0 + budget_s * 0.70), rng)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b
    
    # Phase 3: Binary SA on current best (binarized)
    h_bin3 = binarize(best_h, target_init)
    h3, c5_3 = binary_sa(h_bin3, n_init, target_init, t3, rng,
                         init_temp=0.5, cool=0.9997)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3
    
    # Phase 3b: SA on fresh random binary
    h_rand = make_binary(n_init, rng)
    h3b, c5_3b = binary_sa(h_rand, n_init, target_init, min(t3, time.time() + 30), rng,
                           init_temp=0.4, cool=0.9995)
    if c5_3b < best_c5:
        best_c5 = c5_3b
        best_h = h3b
    
    # Phase 3c: SA from parent binarized
    if h0 is not None and len(h0) > 0:
        h_pbin = binarize(h0, target_init)
        h3c, c5_3c = binary_sa(h_pbin, n_init, target_init, min(t3, time.time() + 25), rng,
                               init_temp=0.6, cool=0.9996)
        if c5_3c < best_c5:
            best_c5 = c5_3c
            best_h = h3c
    
    # Phase 4: Try different n values
    for n_try in [80, 60, 56, 72, 90]:
        if time.time() > t4:
            break
        target_try = n_try / 2.0
        # Binary start
        h_try = make_binary(n_try, rng)
        t_sub = min(t4, time.time() + 20)
        h_try, c5_try = lbfgs_opt(h_try, n_try, target_try, t_sub, [0.15, 0.08, 0.04, 0.02])
        if time.time() > t4:
            break
        t_sub2 = min(t4, time.time() + 12)
        h_try, c5_try = pair_move_refine(h_try, n_try, target_try, t_sub2, rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    # Final polish: pair moves on best
    if time.time() < t4 - 2:
        n_f = len(best_h)
        t_f = n_f / 2.0
        h_f, c5_f = pair_move_refine(best_h, n_f, t_f, t4, rng, n_coarse=15, n_fine=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    
    # Ensure feasibility
    final_n = len(best_h)
    final_target = final_n / 2.0
    best_h = project(best_h, final_target)
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, final_n
