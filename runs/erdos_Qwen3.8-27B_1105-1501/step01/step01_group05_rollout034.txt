    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)

        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)

    for _ in range(100):

        if abs(resid) < 1e-15:

        if resid > 0:

        else:
            idx = np.argmax(h)
            d = min(-resid, h[idx])
            h[idx] -= d

The key insight for discrete: for a binary vector, the score at lag s is the number of transitions from 1 to 0 at that offset. Minimizing the maximum over all lags is equivalent to finding a binary sequence that spreads out its 1s as evenly as possible relative to all shifts.

import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    h = np.asarray(v, dtype=float).ravel()
    if len(h) < n:
        h = np.concatenate([h, np.full(n - len(h), 0.5)])
    else:
        h = h[:n].copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection for shift
    lo, hi = -max(h) - 0.1, 1.0 - min(h) + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h + lam, 0.0, 1.0)
    # Exact correction
    for _ in range(200):
        resid = target - np.sum(h)
        if abs(resid) < 1e-15:
            break
        if resid > 0:
            idx = int(np.argmax(1.0 - h))
            d = min(resid, 1.0 - h[idx])
            h[idx] += d
        else:
            idx = int(np.argmax(h))
            d = min(-resid, h[idx])
            h[idx] -= d
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):
        s = r - (n - 1)
        for i in range(n):
            g = 0.0
            js = i - s
            if 0 <= js < n:
                g += 1.0 - h[js]
            ip = i + s
            if 0 <= ip < n:
                g -= h[ip]
            G[r, i] = dx * g
    return G

def lp_descent(h, max_iters=300, time_limit=30.0, seed=42):
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    stalled = 0
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        s_vals = corr * (2.0 / n)
        theta = float(np.max(s_vals))
        L = len(s_vals)
        
        if abs(theta - score) > 1e-10:
            score = theta
        
        G = compute_grad_matrix(h)
        
        # LP: min t s.t. G@t - t_var <= -s, sum(d)=0, |d|<=1
        # Actually: min t s.t. s_r + G[r,:]@d <= t
        # In standard form: G[r,:]@d - t <= -s_r
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_ub = np.zeros((L + 2 * n, n + 1))
        b_ub = np.zeros(L + 2 * n)
        for r in range(L):
            A_ub[r, :n] = G[r, :]
            A_ub[r, n] = -1.0
            b_ub[r] = -s_vals[r]
        for i in range(n):
            A_ub[L + i, i] = 1.0
            b_ub[L + i] = 1.0
            A_ub[L + n + i, i] = -1.0
            b_ub[L + n + i] = 1.0
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        try:
            res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                                   A_eq=A_eq, b_eq=b_eq, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            # Fallback: use negative gradient of max
            active = np.where(s_vals >= theta - 1e-6)[0]
            grad = np.zeros(n)
            for r in active:
                grad += G[r, :]
            grad -= np.mean(grad)
            norm = np.max(np.abs(grad))
            if norm < 1e-15:
                break
            d = -grad / norm
            best_alpha = 0.0
            for a in [0.1, 0.05, 0.02, 0.01]:
                h_new = project_to_target(h + a * d, n)
                s_new = official_score(h_new)
                if s_new < score - 1e-13:
                    h = h_new
                    score = s_new
                    best_alpha = a
                    break
            if best_alpha == 0:
                stalled += 1
                if stalled > 10:
                    # Perturb
                    h = project_to_target(h + rng.normal(0, 0.01, n), n)
                    score = official_score(h)
                    stalled = 0
            continue
        
        t_opt = res.x[n]
        if t_opt >= theta - 1e-10:
            # No improvement from LP
            stalled += 1
            if stalled > 20:
                # Perturb and restart
                h = project_to_target(h + rng.normal(0, 0.02, n), n)
                score = official_score(h)
                stalled = 0
                continue
            break
        
        d = res.x[:n]
        stalled = 0
        
        # Line search
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if np.any(pos_mask):
            alpha_max = min(alpha_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
        if np.any(neg_mask):
            alpha_max = min(alpha_max, float(np.min(-h[neg_mask] / d[neg_mask])))
        alpha_max = max(alpha_max, 0.001)
        
        best_alpha = 0.0
        best_score_local = score
        n_trials = 8
        alphas = np.linspace(0.05 * alpha_max, alpha_max, n_trials)
        for a in alphas:
            h_new = project_to_target(h + a * d, n)
            s_new = official_score(h_new)
            if s_new < best_score_local - 1e-13:
                best_score_local = s_new
                best_alpha = a
        
        if best_alpha > 0:
            h = project_to_target(h + best_alpha * d, n)
            score = best_score_local
        else:
            # Even the LP direction doesn't help with actual bilinear eval
            # Try very small step
            for a in [0.01, 0.005, 0.001]:
                h_new = project_to_target(h + a * d, n)
                s_new = official_score(h_new)
                if s_new < score - 1e-14:
                    h = h_new
                    score = s_new
                    break
            else:
                stalled += 1
                if stalled > 15:
                    h = project_to_target(h + rng.normal(0, 0.015, n), n)
                    score = official_score(h)
                    stalled = 0
    
    h = project_to_target(h, n)
    score = official_score(h)
    return h, score

def discrete_local_search(h, n, time_limit=60.0, seed=42):
    """Binary local search for even n."""
    if n % 2 != 0:
        n = n - 1  # Make even
    rng = np.random.default_rng(seed)
    target = n // 2
    
    # Initialize from h (threshold)
    idx = np.argsort(-h[:n])
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    
    score = official_score(h_bin)
    best_h = h_bin.copy()
    best_score = score
    
    start = time.perf_counter()
    T = 0.05
    T_min = 0.001
    no_improve = 0
    
    while time.perf_counter() - start < time_limit:
        # Pick random 1 and 0
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_score = official_score(h_bin)
        
        delta = new_score - score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            score = new_score
            no_improve = 0
            if score < best_score:
                best_score = score
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1
        
        # Cool down
        T *= 0.9995
        if no_improve > 5000:
            T = max(T, 0.02)
            no_improve = 0
    
    return best_h, best_score

def interpolate_to(h_src, n_target, shift=0):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    xs = (np.arange(n_src) + 0.5) / n_src
    xt = ((np.arange(n_target) + 0.5 + shift) % n_target) / n_target
    xt_sorted = np.sort(xt)
    order = np.argsort(xt)
    h_interp = np.interp(xt_sorted, xs, h_src, left=h_src[0], right=h_src[-1])
    h_out = np.empty(n_target)
    h_out[order] = h_interp
    return h_out

def slsqp_polish(h, time_limit=20.0):
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1
    lags = np.arange(-(n-1), n)
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        hx = x[:n]
        theta = x[n]
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        hx = x[:n]
        G = compute_grad_matrix(hx)
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        J[:, :n] = -G
        return J
    
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    theta0 = score + 1e-6
    x0 = np.concatenate([h, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 100, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_new = project_to_target(res.x[:n], n)
            s_new = official_score(h_new)
            if s_new < score - 1e-10:
                return h_new, s_new
    except Exception:
        pass
    return h, score

import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n):
    h = np.asarray(v, dtype=float).ravel()
    if len(h) < n:
        h = np.concatenate([h, np.full(n - len(h), 0.5)])
    else:
        h = h[:n].copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -max(h) - 0.1, 1.0 - min(h) + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h + lam, 0.0, 1.0)
    for _ in range(200):
        resid = target - np.sum(h)
        if abs(resid) < 1e-15:
            break
        if resid > 0:
            idx = int(np.argmax(1.0 - h))
            d = min(resid, 1.0 - h[idx])
            h[idx] += d
        else:
            idx = int(np.argmax(h))
            d = min(-resid, h[idx])
            h[idx] -= d
    return np.clip(h, 0.0, 1.0)


def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):
        s = r - (n - 1)
        for i in range(n):
            g = 0.0
            js = i - s
            if 0 <= js < n:
                g += 1.0 - h[js]
            ip = i + s
            if 0 <= ip < n:
                g -= h[ip]
            G[r, i] = dx * g
    return G


def lp_descent(h, max_iters=300, time_limit=30.0, seed=42):
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    stalled = 0

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        s_vals = corr * (2.0 / n)
        theta = float(np.max(s_vals))
        L = len(s_vals)

        if abs(theta - score) > 1e-12:
            score = theta

        G = compute_grad_matrix(h)

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((L + 2 * n, n + 1))
        b_ub = np.zeros(L + 2 * n)
        for r in range(L):
            A_ub[r, :n] = G[r, :]
            A_ub[r, n] = -1.0
            b_ub[r] = -s_vals[r]
        for i in range(n):
            A_ub[L + i, i] = 1.0
            b_ub[L + i] = 1.0
            A_ub[L + n + i, i] = -1.0
            b_ub[L + n + i] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        d = None
        try:
            res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                                   A_eq=A_eq, b_eq=b_eq, method='highs')
            if res.status == 0:
                t_opt = res.x[n]
                if t_opt < theta - 1e-10:
                    d = res.x[:n]
                else:
                    stalled += 1
            else:
                stalled += 1
        except Exception:
            stalled += 1

        if d is None:
            if stalled > 15:
                h = project_to_target(h + rng.normal(0, 0.02, n), n)
                score = official_score(h)
                stalled = 0
                continue
            if stalled > 40:
                break
            # Fallback: negative gradient of active set
            active = np.where(s_vals >= theta - 0.02)[0]
            if len(active) == 0:
                active = np.array([np.argmax(s_vals)])
            grad = np.zeros(n)
            for r in active[:40]:
                grad += G[r, :]
            grad -= np.mean(grad)
            norm = np.max(np.abs(grad))
            if norm < 1e-15:
                break
            d = -grad / norm
            stalled += 1
            if stalled > 50:
                break

        # Line search
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if np.any(pos_mask):
            alpha_max = min(alpha_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
        if np.any(neg_mask):
            alpha_max = min(alpha_max, float(np.min(-h[neg_mask] / d[neg_mask])))
        alpha_max = max(alpha_max, 0.001)

        best_alpha = 0.0
        best_score_local = score
        n_trials = 10
        alphas = np.linspace(0.1 * alpha_max, alpha_max, n_trials)
        for a in alphas:
            h_new = project_to_target(h + a * d, n)
            s_new = official_score(h_new)
            if s_new < best_score_local - 1e-13:
                best_score_local = s_new
                best_alpha = a

        if best_alpha > 0:
            h = project_to_target(h + best_alpha * d, n)
            score = best_score_local
            stalled = 0
        else:
            for a in [0.01, 0.005, 0.002]:
                h_new = project_to_target(h + a * d, n)
                s_new = official_score(h_new)
                if s_new < score - 1e-14:
                    h = h_new
                    score = s_new
                    stalled = 0
                    break
            else:
                stalled += 1

    h = project_to_target(h, n)
    score = official_score(h)
    return h, score


def discrete_search(h_init, n_target, time_limit=60.0, seed=42):
    if n_target % 2 != 0:
        n_target -= 1
    n = n_target
    rng = np.random.default_rng(seed)
    target = n // 2

    idx = np.argsort(-h_init[:n])
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    score = official_score(h_bin)
    best_h = h_bin.copy()
    best_score = score

    start = time.perf_counter()
    T = 0.04
    T_floor = 0.0005

    while time.perf_counter() - start < time_limit:
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_score = official_score(h_bin)
        delta = new_score - score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_floor)):
            score = new_score
            if score < best_score:
                best_score = score
                best_h = h_bin.copy()
                T = max(T * 0.98, T_floor)
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            T *= 0.999

    return h_bin if best_score > score else best_h, best_score


def interpolate_to(h_src, n_target, shift=0):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    xs = (np.arange(n_src) + 0.5) / n_src
    xt = ((np.arange(n_target) + 0.5 + shift) % n_target) / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h_interp = np.interp(xt_sorted, xs, h_src, left=h_src[0], right=h_src[-1])
    h_out = np.empty(n_target)
    h_out[order] = h_interp
    return h_out


def slsqp_polish(h, time_limit=25.0):
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:n]
        theta = x[n]
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        hx = x[:n]
        G = compute_grad_matrix(hx)
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        J[:, :n] = -G
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    theta0 = score + 1e-6
    x0 = np.concatenate([h, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 120, 'ftol': 1e-11})
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_new = project_to_target(res.x[:n], n)
            s_new = official_score(h_new)
            if s_new < score - 1e-10:
                return h_new, s_new
    except Exception:
        pass
    return h, score
