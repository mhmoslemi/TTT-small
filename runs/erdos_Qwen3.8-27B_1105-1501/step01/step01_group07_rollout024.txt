from scipy.optimize import minimize

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    # Bisection on shift
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            # Add to smallest elements
            idx = np.argsort(z)[:max(1, int(abs(diff))+2)]
            for i in idx:
                add = min(diff, 1.0 - z[i])
                z[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            idx = np.argsort(-z)[:max(1, int(abs(diff))+2)]
            for i in idx:
                sub = min(-diff, z[i])
                z[i] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def epigraph_objective(z, n, lam_q, h_ref):
    h = z[:n]
    t = z[n]
    if np.any(h < -1e-6) or np.any(h > 1 + 1e-6):
        return 1e10
    obj = t
    if lam_q > 0:
        obj += lam_q * np.sum((h - h_ref)**2)
    return obj

def epigraph_constraints(z, n):
    h = z[:n]
    t = z[n]
    npts = len(h)
    dx = 2.0 / npts
    # Compute all correlation values
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Constraints: corr[k] * dx - t <= 0 for all k
    return corr * dx - t

def epigraph_jacobian(z, n):
    h = z[:n]
    t = z[n]
    npts = len(h)
    dx = 2.0 / npts
    m = 2 * npts - 1
    jac = np.zeros((m, npts + 1))
    # For each correlation index k (0 to 2n-2), shift s = k - (n-1)
    c = npts - 1
    for s in range(-(npts-1), npts):
        k = c + s
        m_s = npts - abs(s)
        for j in range(m_s):
            if s >= 0:
                # F_s = sum h[j]*(1-h[j+s])

                # dF/dh[j] += -h[j+p]
                jac[k, j] += dx * (-h[j+p])
        # dt component
        jac[k, npts] = -1.0
    return jac

def pair_line_search(h, i, j, n_pts=50):
    """Find best a such that h[i]+=a, h[j]-=a minimizes score."""
    n = len(h)
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return 0.0, score_exact(h)
    
    best_a = 0.0
    best_s = score_exact(h)
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    
    # Golden section refinement
    if abs(best_a) > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0

    diff = target - np.sum(z)

                z[i] += add
                diff -= add

        else:

                z[i] -= sub
                diff += sub

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _eg_obj(z, n, h_ref, lam_q):
    h = z[:n]
    t = z[n]
    obj = t
    if lam_q > 0:
        obj += lam_q * np.sum((h - h_ref) ** 2)
    return obj

def _eg_jac(z, n, h_ref, lam_q):
    h = z[:n]
    grad = np.zeros(n + 1)
    grad[n] = 1.0
    if lam_q > 0:
        grad[:n] = 2.0 * lam_q * (h - h_ref)
    return grad

    h = z[:n]
    t = z[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k_list is not None:
        return -(corr[k_list] * dx - t)
    return -(corr * dx - t)

def _ineg_jac(z, n, k_list):
    h = z[:n]
    npts = n
    dx = 2.0 / npts
    if k_list is not None:
        ks = k_list
        m = len(ks)
    else:
        ks = np.arange(2 * npts - 1)
        m = len(ks)
    c = npts - 1
    jac = np.zeros((m, npts + 1))
    for idx, k in enumerate(ks):
        s = k - c
        m_s = npts - abs(s)
        j = np.arange(m_s)
        if s >= 0:
            jac[idx, j] += dx * (1.0 - h[j + s])
            jac[idx, j + s] += dx * (-h[j])
        else:
            p = -s
            jac[idx, j + p] += dx * (1.0 - h[j])
            jac[idx, j] += dx * (-h[j + p])
        jac[idx, npts] = 1.0  # negative of -1 (since we negate)
    return jac
