    grad *= dx
    return loss, grad


def proj_grad_step(h, grad, lr):
    return make_feasible(h - lr * grad)


def backtracking_line(h, grad, lr, beta, score_fn):
    f0 = score_fn(h)
    c = 0.99
    for _ in range(30):
        h_new = proj_grad_step(h, grad, lr * c)
        f_new = score_fn(h_new)
        if f_new < f0 - 1e-14:
            return h_new, f_new
        c *= 0.5
    return h, f0


def momentum_pg(h, score_fn, n_iter=200, beta=200, lr0=0.01):
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    v = np.zeros(n)
    momentum = 0.8
    lr = lr0
    best_local = h_cur.copy()
    best_score = score_fn(h_cur)
    stagnation = 0
    for it in range(n_iter):
        loss, grad = smooth_loss_grad(h_cur, beta)
        v = momentum * v + (1 - momentum) * grad
        h_new = make_feasible(h_cur - lr * v)
        s_new = score_fn(h_new)
        if s_new < best_score - 1e-12:
            best_score = s_new
            best_local = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1
            lr *= 0.7
        h_prev = h_cur.copy()
        h_cur = h_new
        if stagnation > 40:
            break
        if lr < 1e-8:
            break
    return best_local, best_score


def epigraph_solve(h0, n, maxiter=200, ftol=1e-12):
    h0 = make_feasible(h0)
    dx = 2.0 / n
    score0 = official_score(h0)
    theta0 = score0 + 1e-4
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.75)]

    def obj(z):
        return z[-1]

    def obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def ineq_fun(z):
        hh = z[:n]
        F = compute_F(hh, n)
        return z[-1] - dx * F

    def ineq_jac(z):
        hh = z[:n]
        J = np.zeros((2*n-1, n+1))
        J[:, -1] = 1.0
        for k in range(-(n-1), n):
            J[k + n - 1, :n] = -dx * grad_single_k(hh, k, n)
        return J

    def eq_fun(z):
        return float(np.sum(z[:n]) - n / 2.0)

    def eq_jac(z):
        g = np.ones(n + 1)
        g[-1] = 0.0
        return g

    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, z0, jac=obj_jac, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x[:n])
    except Exception:
        pass
    return h0


def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def symmetry_transform(h):
    n = len(h)
    return make_feasible(1.0 - h[::-1])


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    if parent is not None:
        parent = make_feasible(parent)
        parent_n = len(parent)
        best_h = parent.copy()
        best_score = official_score(best_h)
    else:
        parent_n = 50
        parent = make_feasible(np.full(parent_n, 0.5))
        best_h = parent.copy()
        best_score = official_score(best_h)

    def check_time():
        return time.time() < deadline - 5

    def try_update(cand):
        nonlocal best_h, best_score
        cand = make_feasible(cand)
        if not np.all(np.isfinite(cand)):
            return
        sc = official_score(cand)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = cand.copy()
            best_score = sc

    # Phase 1: Momentum projected gradient at parent resolution
    if check_time():
        for beta in [50, 100, 200, 500]:
            if not check_time():
                break
            lr = 0.02 / np.sqrt(max(beta, 1))
            h_pg, _ = momentum_pg(best_h, official_score, n_iter=300, beta=beta, lr0=lr)
            try_update(h_pg)

    # Phase 2: Epigraph SLSQP at parent resolution
    if check_time():
        h_ep = epigraph_solve(best_h, len(best_h), maxiter=300, ftol=1e-12)
        try_update(h_ep)

    # Phase 3: Try nearby resolutions
    n_best = len(best_h)
    resolutions = []
    for dn in [0, -4, 4, -8, 8, -12, 12, -16, 16, 20, -20]:
        nl = n_best + dn
        if 24 <= nl <= 100:
            resolutions.append(nl)
    resolutions = sorted(set(resolutions))

    for nl in resolutions:
        if not check_time():
            break
        if nl == n_best:
            continue
        h_interp = interp_to_length(best_h, n_best, nl)
        # PG at new resolution
        h_pg, _ = momentum_pg(h_interp, official_score, n_iter=200, beta=200, lr0=0.01)
        try_update(h_pg)
        if check_time():
            h_ep = epigraph_solve(h_pg, nl, maxiter=150, ftol=1e-12)
            try_update(h_ep)

    # Phase 4: Active-set targeted perturbation
    if check_time():
        for attempt in range(8):
            if not check_time():
                break
            h_cur = best_h.copy()
            n = len(h_cur)
            F = compute_F(h_cur, n)
            C = F * (2.0 / n)
            cmax = C.max()
            # Find active set
            active_thresh = cmax - 0.002
            active_mask = C >= active_thresh
            n_active = active_mask.sum()
            # Try to find a direction that reduces active constraints
            # Random direction projected to reduce active constraints
            for trial in range(20):
                d = rng.standard_normal(n)
                d -= d.mean()  # zero mean to preserve sum
                # Check if this direction reduces active constraints
                decrease = 0
                for k in range(-(n-1), n):
                    if active_mask[k + n - 1]:
                        gk = grad_single_k(h_cur, k, n)
                        dk = np.dot(d, gk) * (2.0 / n)
                        decrease += -dk
                if decrease > 0:
                    # Take a small step in this direction
                    for alpha in [0.01, 0.02, 0.05, 0.1]:
                        h_try = make_feasible(h_cur + alpha * d)
                        sc = official_score(h_try)
                        if sc < best_score - 1e-13:
                            try_update(h_try)
                            h_cur = best_h.copy()
                            break

    # Phase 5: Symmetry exploitation + local search
    if check_time():
        h_sym = symmetry_transform(best_h)
        sc_sym = official_score(h_sym)
        # Average with symmetric version (maintain feasibility)
        h_avg = make_feasible(0.5 * (best_h + h_sym))
        try_update(h_avg)
        if check_time():
            h_pg, _ = momentum_pg(h_avg, official_score, n_iter=200, beta=300, lr0=0.008)
            try_update(h_pg)

    # Phase 6: Multi-start from perturbed versions
    if check_time():
        for sigma in [0.01, 0.02, 0.04, 0.08]:
            if not check_time():
                break
            for _ in range(3):
                if not check_time():
                    break
                h_pert = make_feasible(best_h + sigma * rng.standard_normal(len(best_h)))
                h_pg, _ = momentum_pg(h_pert, official_score, n_iter=150, beta=300, lr0=0.008)
                try_update(h_pg)

    # Phase 7: Final polish with epigraph
    if check_time():
        h_final = epigraph_solve(best_h, len(best_h), maxiter=400, ftol=1e-13)
        try_update(h_final)

    # Phase 8: One more resolution try at the best
    if check_time():
        n_best = len(best_h)
        for nl in [n_best + 2, n_best - 2, n_best + 4]:
            if not check_time():
                break
            if nl < 20 or nl > 100:
                continue
            h_interp = interp_to_length(best_h, n_best, nl)
            h_pg, _ = momentum_pg(h_interp, official_score, n_iter=150, beta=400, lr0=0.005)
            try_update(h_pg)
            if check_time():
                h_ep = epigraph_solve(h_pg, nl, maxiter=200, ftol=1e-13)
                try_update(h_ep)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score < 0:
        best_h = parent.copy()
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
