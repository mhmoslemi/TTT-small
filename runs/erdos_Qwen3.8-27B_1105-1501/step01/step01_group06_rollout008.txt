def compute_pair_coefficients(h, i, j):
    """Compute A and R for transfer from j to i (h[i] += d, h[j] -= d)"""
    n = len(h)
    g = 1.0 - h
    p = np.zeros(n)
    p[i] = 1.0
    p[j] = -1.0
    A = np.correlate(p, g, mode='full') - np.correlate(h, p, mode='full')
    R = np.correlate(p, p, mode='full')
    return A, R
