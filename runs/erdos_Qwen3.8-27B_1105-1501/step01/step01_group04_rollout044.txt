import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    T1 = np.convolve(1.0 - h, p, mode='full')[n - 1:2 * n - 1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2


def target_penalty_grad(h, tau, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    z = alpha * (scaled - tau)
    z_clipped = np.clip(z, -500, 500)
    softplus = np.logaddexp(0, z_clipped)
    sig = 1.0 / (1.0 + np.exp(-z_clipped))
    weight = 2.0 * softplus * sig * alpha
    total = np.sum(weight)
    if total < 1e-10:
        weight = np.ones_like(weight) / len(weight)
    else:
        weight = weight / total
    T1 = np.convolve(1.0 - h, weight, mode='full')[n - 1:2 * n - 1]
    T2 = np.correlate(weight, h, mode='valid')[::-1]
    return dx * (T1 - T2)


def targeted_swap_step(h, rng):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    k_star = max_idx - (n - 1)
    
    if k_star >= 0:
        # Need positions i where h[i+k_star]>0 and h[i] is small
        candidates = []
        for i in range(n - k_star):
            if h[i + k_star] > 0.3 and h[i] < 0.7:
                candidates.append(i)
        if not candidates:
            return None
        i = candidates[rng.randint(len(candidates))]
        j = i + k_star
        amt = min(h[j] - 0.0, 1.0 - h[i])
        if amt < 0.01:
            return None
    else:
        ks = -k_star
        candidates = []
        for i in range(ks, n):
            if h[i] > 0.3 and h[i - ks] < 0.7:
                candidates.append(i)
        if not candidates:
            return None
        i = candidates[rng.randint(len(candidates))]
        j = i
        i_pos = i - ks
        amt = min(h[j] - 0.0, 1.0 - h[i_pos])
        if amt < 0.01:
            return None
        j = i
        i = i_pos
    
    h_new = h.copy()
    scale = rng.choice([0.5, 0.8, 1.0])
    h_new[i] += amt * scale
    h_new[j] -= amt * scale
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

    cool = 0.999
    
    while time.time() - t0 < time_budget:

            # Random swap

            j = rng.randint(n)

                continue
            delta = rng.uniform(0.02, 0.15) * rng.choice([-1, 1])
            h_try = current_h.copy()
            h_try[i] = np.clip(current_h[i] + delta, 0.0, 1.0)
            h_try[j] = np.clip(current_h[j] - delta, 0.0, 1.0)
            if abs(np.sum(h_try) - target) > 0.01:

            diff = c5_try - current_c5

                current_c5 = c5_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        temp *= cool
    
    return best_h, best_c5
