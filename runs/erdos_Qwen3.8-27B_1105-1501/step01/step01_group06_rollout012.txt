import numpy as np
import time

def c5_official(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_target(v, target):
    v = np.asarray(v, dtype=float).copy()
    v = np.nan_to_num(v, nan=0.5, posinf=2.0, neginf=-2.0)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual exactly
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            room = 1.0 - x
            order = np.argsort(-room, kind='stable')
            for idx in order:
                add = min(room[idx], residual)
                if add > 0:
                    x[idx] += add
                    residual -= add
                if residual <= 1e-12:
                    break
        else:
            order = np.argsort(x, kind='stable')
            for idx in order:
                rem = min(x[idx], -residual)
                if rem > 0:
                    x[idx] -= rem
                    residual += rem
                if residual >= -1e-12:
                    break
    return x

            break
        
        g = 1.0 - h
        C0 = np.correlate(h, g, mode='full')
        max_C0_raw = float(np.max(C0))
        
        best_improvement = 0.0
        best_i, best_j, best_s = -1, -1, 0.0
        
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                T = min(1.0 - h[i], h[j])
                if T < 1e-10:
                    continue
                
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                A = np.correlate(d, g, mode='full') - np.correlate(h, d, mode='full')
                R = np.correlate(d, d, mode='full')
                
                # Vectorized line search
                ss = np.linspace(0.0, T, n_samples)
                C_s = C0[None, :] + ss[:, None] * A[None, :] - (ss[:, None] ** 2) * R[None, :]
                vals = np.max(C_s, axis=1)
                best_idx = int(np.argmin(vals))
                best_s_local = ss[best_idx]
                best_val_local = float(vals[best_idx])
                
                # Refinement
                if 0 < best_idx < n_samples - 1:
                    lo_s = ss[max(0, best_idx - 1)]
                    hi_s = ss[min(n_samples - 1, best_idx + 1)]
                    for _ in range(4):
                        ss_f = np.linspace(lo_s, hi_s, 16)
                        C_f = C0[None, :] + ss_f[:, None] * A[None, :] - (ss_f[:, None] ** 2) * R[None, :]
                        vals_f = np.max(C_f, axis=1)
                        best_f = int(np.argmin(vals_f))
                        if best_f > 0:
                            lo_s = ss_f[best_f - 1]
                        if best_f < len(ss_f) - 1:
                            hi_s = ss_f[best_f + 1]
                        best_s_local = ss_f[best_f]
                        best_val_local = float(vals_f[best_f])
                
                improvement = max_C0_raw - best_val_local
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i, best_j, best_s = i, j, best_s_local
        
        if best_i >= 0 and best_improvement > 1e-13:
            h[best_i] = min(1.0, h[best_i] + best_s)
            h[best_j] = max(0.0, h[best_j] - best_s)
            h = project_to_target(h, target)
            new_c5 = c5_official(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            break
    
    return best_h, best_c5
