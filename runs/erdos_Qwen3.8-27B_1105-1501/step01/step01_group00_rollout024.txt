import time

def repair_exact(z):
    """Clip to [0,1] and adjust sum to n/2."""
    n = len(z)
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    # Binary search for shift
    lo, hi = -1.0, 1.0
    # Ensure sum is achievable
    while np.sum(np.clip(z + hi, 0, 1)) < target:
        hi *= 2
    while np.sum(np.clip(z + lo, 0, 1)) > target:
        lo *= 2
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1 - h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            else:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-15:
                break
    return h

def score_h(h):
    """Compute exact C5 objective."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def score_full(h):
    """Compute full correlation vector * dx."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

```python
def sa_phase(h, score, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_score = score
    T = 0.001  # Initial temperature
    cooling = 0.9999
    
    step = 0
    while time.time() < deadline:
        step += 1
        # Pick random pair
        a = rng.integers(n)
        b = rng.integers(n)
        if a == b:
            continue
        lo = max(-h[a], h[b] - 1.0)
        hi = min(1.0 - h[a], h[b])
        if hi - lo < 1e-6:
            continue
        # Random delta
        d = rng.uniform(lo, hi)
        
        h[a] += d
        h[b] -= d
        new_score = score_h(h)
        
        if new_score < score - 1e-13:
            # Always accept improvement
            score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_h = h.copy()
        else:
            # SA acceptance
            delta = new_score - score
            if delta < 0.01 and rng.random() < np.exp(-delta / max(T, 1e-8)):
                score = new_score
            else:
                # Reject
                h[a] -= d
                h[b] += d
        
        T *= cooling
    
    return best_h, best_score
```

- Initial temperature T: should be large enough to explore but small enough to converge
- Cooling rate: slow cooling allows more exploration

For this problem, the "energy scale" is O(0.01) (improvements of at most ~0.01 per move), so T should start around 0.001-0.01 and cool to ~1e-6.

The perturbation for a pair move (a, b, d) on correlation position p is:
ΔC[p] = d * (dC[p]/dh[a] - dC[p]/dh[b]) + d² * (second order terms)

If I ignore second-order terms, I can compute all candidate scores without calling np.correlate:
C_new[p] ≈ C[p] + d * (S[a,p] - S[b,p])

Then the new max is max(C_new[p]) * dx.
