    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)

            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_val(h, beta):
    n = len(h)
    C = compute_F(h, n) * (2.0 / n)
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(len(C)) / beta
    return cmax + (mk + np.log(ws)) / beta

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    C = compute_F(h, n) * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(len(C)) / len(C)
    else:
        weights = es / ws
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[k + n - 1]
        if w < 1e-14:
            continue
        if k >= 0:
            # First factor: j in [k, n-1]
            grad[k:n] += w * (1.0 - h[0:n-k])
            # Second factor: j in [0, n-1-k]
            grad[0:n-k] += w * (-h[k:n])
        else:
            k2 = -k
            # First factor: j in [0, n-1-k2]
            grad[0:n-k2] += w * (1.0 - h[k2:n])
            # Second factor: j in [k2, n-1]
            grad[k2:n] += w * (-h[0:n-k2])
    return grad * dx

def projected_gradient_descent(h0, n, beta, n_iters, alpha0):
    h = h0.copy()
    for t in range(n_iters):
        grad = smooth_grad(h, beta)
        alpha = alpha0 / (1.0 + t * 0.01)
        h_new = make_feasible(h - alpha * grad)
        if not np.all(np.isfinite(h_new)):
            break
        f_new = smooth_obj_val(h_new, beta)
        f_old = smooth_obj_val(h, beta)
        if f_new < f_old or t < 10:
            h = h_new
        else:
            # Halve step size
            alpha = alpha * 0.5
            h_new = make_feasible(h - alpha * grad)
            h = h_new
    return h

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        for i in idx:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_F(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')


def smooth_obj(h, beta):
    n = len(h)
    C = compute_F(h) * (2.0 / n)
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(len(C)) / beta
    return cmax + (mk + np.log(ws)) / beta


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    C = compute_F(h) * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(len(C)) / len(C)
    else:
        weights = es / ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[k + n - 1]
        if w < 1e-14:
            continue
        if k >= 0:
            grad[k:n] += w * (1.0 - h[0:n - k])
            grad[0:n - k] += -w * h[k:n]
        else:
            k2 = -k
            grad[0:n - k2] += w * (1.0 - h[k2:n])
            grad[k2:n] += -w * h[0:n - k2]
    return grad * dx


def eq_sum(h):
    return float(np.sum(h) - len(h) / 2.0)


def eq_sum_jac(h):
    return np.ones(len(h))


def slsqp_smooth(h0, beta, maxiter=300, ftol=1e-13):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0


def projected_gradient(h0, beta, n_iters, alpha0, time_limit=None, t_start=0.0):
    h = h0.copy()
    alpha = alpha0
    f_prev = smooth_obj(h, beta)
    for t in range(n_iters):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        grad = smooth_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = alpha / (1.0 + t * 1e-4)
        h_new = make_feasible(h - step * grad)
        f_new = smooth_obj(h_new, beta)
        if f_new < f_prev:
            h = h_new
            f_prev = f_new
        else:
            alpha = alpha * 0.7
    return h


def direct_ineq_val(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    F = compute_F(h)
    return z[-1] - dx * F


def direct_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    nvar = n + 1
    nF = 2 * n - 1
    J = np.zeros((nF, nvar))
    J[:, -1] = 1.0
    for k in range(-(n - 1), n):
        row = k + n - 1
        g = np.zeros(n)
        if k >= 0:
            g[k:n] += (1.0 - h[0:n - k])
            g[0:n - k] += (-h[k:n])
        else:
            k2 = -k
            g[0:n - k2] += (1.0 - h[k2:n])
            g[k2:n] += (-h[0:n - k2])
        J[row, :n] = -dx * g
    return J


def direct_eq_val(z):
    n = len(z) - 1
    return float(np.sum(z[:n]) - n / 2.0)


def direct_eq_jac(z):
    n = len(z) - 1
    g = np.ones(n + 1)
    g[-1] = 0.0
    return g


def direct_obj_val(z):
    return float(z[-1])


def direct_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def slsqp_direct(h0, maxiter=300, ftol=1e-13):
    n = len(h0)
    dx = 2.0 / n
    score0 = official_score(h0)
    z0 = np.concatenate([h0, [score0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': direct_ineq_val, 'jac': direct_ineq_jac},
        {'type': 'eq', 'fun': direct_eq_val, 'jac': direct_eq_jac}
    ]
    try:
        res = minimize(direct_obj_val, z0, jac=direct_obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x[:n])
    except Exception:
        pass
    return h0


def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def symmetry_transform(h):
    return make_feasible(1.0 - h[::-1])


def random_perturb(h, sigma, rng):
    noise = rng.normal(0.0, sigma, size=len(h))
    return make_feasible(h + noise)


def coordinate_descent(h, n_iter, rng):
    n = len(h)
    best = h.copy()
    best_score = official_score(best)
    for _ in range(n_iter):
        i = rng.integers(0, n)
        for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.05, -0.05]:
            trial = best.copy()
            trial[i] += delta
            trial = make_feasible(trial)
            sc = official_score(trial)
            if sc < best_score - 1e-15:
                best = trial
                best_score = sc
                break
    return best, best_score
