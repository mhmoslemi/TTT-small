import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_only(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def score_with_corr(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return float(np.log(S) + base), g

def grad_descent(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_only(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        _, g = smooth_obj_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_only(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 50:
                cur = make_feasible(cur + rng.normal(0, max(0.008, T * 0.5), n), n)
                cur_c = score_only(cur)
                step = max(T * 0.5, 0.003)
                stall = 0
        else:
            step = min(step * 1.08, T * 3.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=35):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    grid = np.linspace(lo, hi, n_pts)
    for a in grid:
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_only(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / max(n_pts - 1, 1)
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-14:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_only(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_only(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1

                else:

def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_only(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:

def directed_bottleneck_move(h, cur_c, rng):
    """Target the current bottleneck lag with a directed pair move."""
    n = len(h)
    _, corr = score_with_corr(h)
    k_star = int(np.argmax(corr))
    d = k_star - (n - 1)
    if d == 0:
        return None
    if d > 0:
        # F(d) = sum_{j=d}^{n-1} h[j]*(1-h[j-d])
        # Want to reduce: decrease h[j], increase h[j-d] for problematic j
        js = np.arange(d, n)
        contribs = h[js] * (1.0 - h[js - d])
        order = np.argsort(-contribs)
        candidates = [(js[order[t]], js[order[t]] - d) for t in range(min(8, len(js)))]
    else:
        ad = -d
        js = np.arange(0, n - ad)
        contribs = h[js] * (1.0 - h[js + ad])
        order = np.argsort(-contribs)
        candidates = [(js[order[t]], js[order[t]] + ad) for t in range(min(8, len(js)))]
    
    best_s = cur_c
    best_move = None
    rng.shuffle(candidates)
    for i, j in candidates:
        orig_i = h[i]
        orig_j = h[j]
        # Try moving from i to j (decrease h[i], increase h[j])
        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-12:
            continue
        for frac in (0.4, 0.7, 0.5, 0.3, 0.9, 0.2):
            delta = max_d * frac
            h[i] = orig_i - delta
            h[j] = orig_j + delta
            s = score_only(h)
            if s < best_s:
                best_s = s
                best_move = (i, j, -delta)
        # Try moving from j to i
        max_d2 = min(h[j], 1.0 - h[i])
        if max_d2 > 1e-12:
            for frac in (0.4, 0.7, 0.5, 0.3, 0.9, 0.2):
                delta = max_d2 * frac
                h[i] = orig_i + delta
                h[j] = orig_j - delta
                s = score_only(h)
                if s < best_s:
                    best_s = s
                    best_move = (i, j, delta)
        h[i] = orig_i
        h[j] = orig_j
    if best_move is not None and best_s < cur_c - 1e-14:
        i, j, delta = best_move
        h[i] += delta
        h[j] -= delta
        return float(best_s)
    return None

def block_move(h, cur_c, rng):
    """Try shifting a contiguous block."""
    n = len(h)
    blen = int(rng.integers(3, max(4, n // 4)))
    start = int(rng.integers(0, n - blen))
    end = start + blen
    block = h[start:end].copy()
    best_s = cur_c
    best_delta = 0.0
    max_up = min(1.0 - block).item()
    for delta in (0.1, 0.2, 0.4, -0.1, -0.2, -0.4, 0.3, -0.3):
        if delta > 0 and delta > max_up:
            continue
        if delta < 0 and -delta > block.min():
            continue
        h[start:end] = block + delta
        # Compensate: spread -delta*blen over the rest
        rest_mask = np.ones(n, dtype=bool)
        rest_mask[start:end] = False
        n_rest = n - blen
        comp = -delta * blen / n_rest
        # Apply compensation only if it keeps values in [0,1]
        rest_vals = h[rest_mask]
        if np.min(rest_vals + comp) < -1e-12 or np.max(rest_vals + comp) > 1.0 + 1e-12:
            continue
        h[rest_mask] += comp
        s = score_only(h)
        if s < best_s:
            best_s = s
            best_delta = delta
    h[start:end] = block
    if best_delta != 0.0 and best_s < cur_c - 1e-14:
        h[start:end] = block + best_delta
        rest_mask = np.ones(n, dtype=bool)
        rest_mask[start:end] = False
        n_rest = n - blen
        comp = -best_delta * blen / n_rest
        h[rest_mask] += comp
        h = make_feasible(h)
        best_s = score_only(h)
        return float(best_s)
    return None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 20:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

    n_main = 140
    inits = []
    
    # Init 0: Parent upscaled
    h0 = resample_h(parent, n_main)
    if h0 is not None:
        inits.append(h0)
    
    # Init 1: Smooth sinusoidal
    inits.append(make_smooth_init(n_main, 1, 0.45, rng))
    inits.append(make_smooth_init(n_main, 2, 0.35, rng))
    
    # Init 2: Binary random
    inits.append(make_binary_init(n_main, rng))
    
    # Init 3: Block + noise
    hb = make_block_init(n_main)
    hn = make_feasible(hb + rng.normal(0, 0.05, n_main), n_main)
    inits.append(hn)
    
    # Quick evaluation of initializations
    best_h = inits[0]
    best_c = score_only(best_h)
    for h_init in inits[1:]:
        c_init = score_only(h_init)
        if c_init < best_c:
            best_h = h_init
            best_c = c_init
    
    # Phase 1: Gradient descent from best init (20% time)
    p1_end = t0 + budget_s * 0.20
    if time.time() < p1_end:
        h1, c1 = grad_descent(best_h, p1_end, 0.12, 0.001, 0.9995, 0.02, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
    
    # Also try gradient descent from second-best init
    if time.time() < t0 + budget_s * 0.28:
        other_inits = sorted(inits, key=lambda x: score_only(x))
        if len(other_inits) > 1:
            h_alt = other_inits[1].copy()
            c_alt = score_only(h_alt)
            if c_alt < best_c + 0.01:
                sub_dl = min(t0 + budget_s * 0.28, time.time() + 30)
                h2, c2 = grad_descent(h_alt, sub_dl, 0.1, 0.002, 0.9995, 0.015, rng)
                if c2 < best_c:
                    best_h, best_c = h2, c2
    
    # Phase 2: Main local search (50% time)
    p2_end = t0 + budget_s * 0.72
    cur = best_h.copy()
    cur_c = best_c
    n = len(cur)
    stall_count = 0
    
    while time.time() < p2_end:
        mode = rng.random()
        res = None
        if mode < 0.35:
            # Directed bottleneck move
            res = directed_bottleneck_move(cur, cur_c, rng)
        elif mode < 0.60:
            # Random pair search
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                res = pair_search(cur, i, j, cur_c, n_pts=30)
        elif mode < 0.75:
            # Adjacent pair
            i = int(rng.integers(0, n - 1))
            res = pair_search(cur, i, i + 1, cur_c, n_pts=35)
        elif mode < 0.85:
            # Coordinate
            i = int(rng.integers(0, n))
            res = coord_search(cur, i, cur_c, n_pts=25)
        else:
            # Block move
            res = block_move(cur, cur_c, rng)
        
        if res is not None:
            cur_c = res
            stall_count = 0
            if cur_c < best_c:
                best_h = cur.copy()
                best_c = cur_c
            # Do a few extra quick moves after improvement
            for _ in range(3):
                if time.time() > p2_end:
                    break
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = pair_search(cur, i2, j2, cur_c, n_pts=20)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h = cur.copy()
                        best_c = cur_c
                else:
                    break
        else:
            stall_count += 1
            if stall_count > 300:
                # Perturb and restart
                cur = make_feasible(best_h + rng.normal(0, 0.015, n), n)
                cur_c = score_only(cur)
                stall_count = 0
            elif stall_count > 100:
                # Moderate perturbation
                cur = make_feasible(best_h + rng.normal(0, 0.008, n), n)
                cur_c = score_only(cur)
                if cur_c > best_c + 0.005:
                    cur = best_h.copy()
                    cur_c = best_c
                stall_count = 50
    
    # Phase 3: Fine gradient descent (12% time)
    p3_end = t0 + budget_s * 0.84
    if time.time() < p3_end:
        h3, c3 = grad_descent(best_h, p3_end, 0.005, 0.0001, 0.999, 0.002, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3
    
    # Phase 4: Systematic adjacent pair sweep (8% time)
    p4_end = t0 + budget_s * 0.92
    cur = best_h.copy()
    cur_c = best_c
    n = len(cur)
    improved = True
    while improved and time.time() < p4_end:
        improved = False
        for i in range(n - 1):
            if time.time() > p4_end:
                break
            res = pair_search(cur, i, i + 1, cur_c, n_pts=25)
            if res is not None:
                cur_c = res
                improved = True
                if cur_c < best_c:
                    best_h = cur.copy()
                    best_c = cur_c
        # Also sweep random pairs
        for _ in range(30):
            if time.time() > p4_end:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=25)
            if res is not None:
                cur_c = res
                if cur_c < best_c:
                    best_h = cur.copy()
                    best_c = cur_c
                improved = True
    
    # Phase 5: Try higher resolution (8% time)
    p5_end = deadline - 5.0
    n_cur = len(best_h)
    for dn in (30, 60, -30):
        if time.time() > p5_end:
            break
        nr = n_cur + dn
        if nr < 30 or nr > 250:
            continue
        h_r = resample_h(best_h, nr)
        if h_r is None:
            continue
        c_r = score_only(h_r)
        sub_dl = min(p5_end - 2, time.time() + 30)
        if time.time() < sub_dl:
            h_r, c_r = grad_descent(h_r, sub_dl, 0.008, 0.0002, 0.999, 0.004, rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
    
    # Final refinement
    if time.time() < deadline - 3:
        h_f, c_f = grad_descent(best_h, deadline - 3, 0.003, 0.00005, 0.9985, 0.001, rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    
    # Final local polish
    if time.time() < deadline - 1:
        cur = best_h.copy()
        cur_c = best_c
        n = len(cur)
        for _ in range(50):
            if time.time() > deadline - 1:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=20)
            if res is not None:
                cur_c = res
                if cur_c < best_c:
                    best_h = cur.copy()
                    best_c = cur_c
    
    # Ensure feasibility
    best_h = make_feasible(best_h)
    best_c = score_only(best_h)
    return best_h, best_c, len(best_h)
