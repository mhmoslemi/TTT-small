def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def repair_exact(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def pair_line_best(h, a, b, n_samples):
    lo = max(-h[a], h[b] - 1.0) + 1e-12
    hi = min(1.0 - h[a], h[b]) - 1e-12
    if hi - lo < 1e-12:
        return 0.0, None

    best_d = 0.0

    deltas = np.linspace(lo, hi, n_samples)

        h[b] -= d

        h[a] -= d
        h[b] += d
        if sc < best_sc:

def greedy_pair(h0, max_passes, n_samples, t_start, deadline):

    n = len(h)

        if time.time() >= deadline:
            break
        np.random.shuffle(idx)
        improved = False

            for j in range(i + 1, n):
                a, b = idx[i], idx[j]
                d, new_sc = pair_line_best(h, a, b, n_samples)
                if d != 0.0 and new_sc < sc - 1e-13:
                    h[a] += d
                    h[b] -= d
                    h = np.clip(h, 0.0, 1.0)
                    sc = new_sc
                    improved = True
            if time.time() >= deadline:
                break
        if not improved:
            break
    return h, sc

def interp_h(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    return repair_exact(np.interp(tgt_c, src_c, h_src))

def sa_pair(h0, t_start, deadline, t0, cooling, max_iters):
    h = h0.copy()
    sc = official_score(h)
    best_h = h.copy()
    best_sc = sc
    n = len(h)
    T = t0
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        a, b = np.random.choice(n, 2, replace=False)
        lo = max(-h[a], h[b] - 1.0) + 1e-12
        hi = min(1.0 - h[a], h[b]) - 1e-12
        if hi - lo < 1e-12:
            T *= cooling
            continue
        d = np.random.uniform(lo, hi)
        h[a] += d
        h[b] -= d
        h = np.clip(h, 0.0, 1.0)
        new_sc = official_score(h)
        if new_sc < sc or np.random.random() < np.exp(-(new_sc - sc) / max(T, 1e-12)):
            sc = new_sc
            if sc < best_sc:
                best_sc = sc
                best_h = h.copy()
        else:
            h[a] -= d
            h[b] += d
            h = np.clip(h, 0.0, 1.0)
        T *= cooling
    return best_h, best_sc

def antisym_transform(h):
    h_new = 1.0 - h[::-1].copy()
    return repair_exact(h_new)
