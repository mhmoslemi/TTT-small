                true_c = float(np.max(true_scores))
                if true_c > check_c + 1e-10:
                    # Revert to last good state
                    h[:] = check_h
                    scores = check_scores.copy()
                    check_c = float(np.max(scores))
                    G = compute_influence(h)
                else:
                    check_h = h.copy()
                    check_scores = true_scores.copy()
                    check_c = true_c
                    scores = true_scores
                    G = compute_influence(h)
                moves_since_check = 0
    
    return h, scores, float(np.max(overlap_scores(h)))

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=0
