def project_to_target(v, target):
    """Project v to [0,1]^n with sum = target using water-filling bisection."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)

    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid
    theta = (lo + hi) / 2.0

def project_to_target(v, target):

    n = len(h)

def compute_all_gradients(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            if n - d > 0:
                G[k, :n - d] += (1.0 - h[d:])
            if d < n:
                G[k, d:] -= h[:n - d]
        else:
            g = -d
            G[k, g:] += (1.0 - h[:n - g])
            G[k, :n - g] -= h[g:]
    return G

def project_to_target(v, target):
