def solve_qp(h0, active_set, lam=1.0):
    n = len(h0)
    dx = 2.0/n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Get gradients for active set
    grads = manual_grad(h0)  # shape (2n-1, n)
    vals = manual_correlate(h0) * dx

import time

    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

    """Gradient of f_s (raw, unscaled) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for k in range(n):
            if k >= s and k < n:
                g[k] += 1.0 - h[k - s]
            if k < n - s:
                g[k] -= h[k + s]
    else:
        t = -s
        for k in range(n):
            if k < n - t:
                g[k] += 1.0 - h[k + t]
            if k >= t:
                g[k] -= h[k - t]
    return g

    n = len(h)
    dx = 2.0 / n

    for p in range(2 * n - 1):
        s = p - (n - 1)
        result[p] = dx * gradient_p(h, s)
    return result

def qp_step(h0, k_active=5, reg=0.5, solver=None):
    """One active-set QP step. Returns new h or None."""
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    dx = 2.0 / n
    for p in top_idx:
        s = p - (n - 1)
        g = dx * gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)
    
    obj = cp.Minimize(t_var + reg / 2.0 * cp.quad_form(h_var - h0, cp.eye(n)))
    prob = cp.Problem(obj, constraints)
    
    if solver is None:

    try:

        try:
            prob.solve(solver=cp.ECOS)
        except Exception:

        return None
    
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    
    return make_feasible(h_new)

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_p(h, s):
    """Gradient of raw f_s with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for k in range(n):
            if k >= s:
                g[k] += 1.0 - h[k - s]
            if k < n - s:
                g[k] -= h[k + s]
    else:
        t = -s
        for k in range(n):
            if k < n - t:
                g[k] += 1.0 - h[k + t]
            if k >= t:
                g[k] -= h[k - t]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:

                h[idx] -= sub
                residual += sub

def qp_step(h0, k_active, reg):
    """One active-set QP step using cvxpy."""
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = dx * gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)

    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue

        return None

    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)
