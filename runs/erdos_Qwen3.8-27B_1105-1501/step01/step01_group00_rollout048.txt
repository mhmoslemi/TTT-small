        vals = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = vals.max()

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # First try to fix by uniform shift
    s = z.sum()
    if s < target:
        room = 1.0 - z
        total_room = room.sum()
        if total_room > 0:
            deficit = target - s
            if deficit <= total_room:
                z = z + deficit / total_room * room
                z = np.clip(z, 0.0, 1.0)
    else:
        excess = s - target
        if excess > 0:
            z = z - excess / s * z
            z = np.clip(z, 0.0, 1.0)
    # Fix residual by greedy adjustment

    if abs(residual) > 1e-12:

            for idx in order:

                if add <= 0:

                z[idx] += add
                residual -= add

                residual += sub

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    s = z.sum()
    if abs(s - target) > 1e-10:
        if s < target:
            room = 1.0 - z
            tr = room.sum()
            if tr > 1e-15:
                z = np.clip(z + (target - s) / tr * room, 0.0, 1.0)
        else:
            z = np.clip(z * target / s, 0.0, 1.0)
    residual = target - z.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - z)
            for idx in order:
                add = min(residual, 1.0 - z[idx])
                if add <= 0:
                    break
                z[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(z)
            for idx in order:
                sub = min(-residual, z[idx])
                if sub <= 0:
                    break
                z[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(z, 0.0, 1.0)

def lag_gradient(h, s):
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    if s >= 0:
        m1 = j >= s
        m2 = j < n - s
        g[m1] += (1.0 - h[j[m1] - s])
        g[m2] -= h[j[m2] + s]
    else:
        t = -s
        m1 = j < n - t
        m2 = j >= t
        g[m1] += (1.0 - h[j[m1] + t])
        g[m2] -= h[j[m2] - t]
    return g

def smooth_obj_fn(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = vals.max()
    shifted = beta * (vals - cm)
    shifted -= shifted.max()
    return cm + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fn(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def smooth_grad_fn(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = vals.max()
    shifted = beta * (vals - cm)
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    grads = np.zeros((len(vals), n))
    lags = np.arange(-(n - 1), n)
    for idx in range(len(vals)):
        grads[idx] = lag_gradient(x, lags[idx])
    return dx * (w @ grads)

def try_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fn, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj_fn, h0, args=(beta,), jac=smooth_grad_fn,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def active_set_optimize(h0, n_iters=3000, K=3, step0=0.003, momentum=0.7):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_sc = official_score(h)
    step = step0
    
    for it in range(n_iters):
        vals = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = vals.max()
        top_k = np.argsort(vals)[-K:]
        lags = top_k - (n - 1)
        
        grads = np.zeros((K, n))
        for i in range(K):
            grads[i] = lag_gradient(h, lags[i])
        avg_grad = (dx * grads.mean(axis=0))
        
        # Project onto sum=0
        avg_grad -= avg_grad.mean()
        
        # Momentum
        v = momentum * v - step * avg_grad
        v -= v.mean()
        
        h_new = h + v
        h_new = make_feasible(h_new)
        sc_new = official_score(h_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new.copy()
            step = min(step * 1.05, 0.02)
        else:
            step *= 0.9
            v *= momentum * 0.5
        
        h = h_new
    
    return best_h, best_sc

def smooth_optimize_ladder(h0, betas, maxiter_per=150):
    cur = h0.copy()
    best = cur.copy()
    best_sc = official_score(cur)
    for beta in betas:
        h_new, sc = try_smooth(cur, beta, maxiter_per)
        if h_new is not None:
            if sc < best_sc:
                best_sc = sc
                best = h_new.copy()
                cur = h_new
    return best, best_sc

def sinusoidal_perturb(h, freqs, amplitudes, rng):
    n = len(h)
    x = np.arange(n)
    pert = np.zeros(n)
    for f, a in zip(freqs, amplitudes):
        phase = rng.uniform(0, 2 * np.pi)
        pert += a * np.sin(2 * np.pi * f * x / n + phase)
    return make_feasible(h + pert)

    n_src = len(h_src)

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def two_level_equality(h0, s1, s2, maxiter=200):
    """Minimize t s.t. f_{s1} <= t, f_{s2} <= t, 0<=h<=1, sum(h)=n/2"""
    n = len(h0)
    dx = 2.0 / n
    theta0 = max(official_score(h0), 
                 float(np.dot(h0[max(0,0):n-abs(s1)], 1.0-h0[max(abs(s1),0):n])) * dx) + 0.01
    
    def obj(z):
        return z[-1]
    
    def obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def ineq1(z):
        h = z[:-1]
        return z[-1] - float(np.correlate(h, 1.0-h, mode='full')[s1+(n-1)] * dx)
    
    def ineq1_jac(z):
        h = z[:-1]
        g = -dx * lag_gradient(h, s1)
        return np.append(g, 1.0)
    
    def ineq2(z):
        h = z[:-1]
        return z[-1] - float(np.correlate(h, 1.0-h, mode='full')[s2+(n-1)] * dx)
    
    def ineq2_jac(z):
        h = z[:-1]
        g = -dx * lag_gradient(h, s2)
        return np.append(g, 1.0)
    
    def eq(z):
        return np.sum(z[:-1]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    z0 = np.append(h0, theta0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq1, 'jac': ineq1_jac},
        {'type': 'ineq', 'fun': ineq2, 'jac': ineq2_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, z0, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:-1])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None
