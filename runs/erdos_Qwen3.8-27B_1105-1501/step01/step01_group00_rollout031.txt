    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    grads = compute_grads(x)
    return dx * (w @ grads)

def pnorm_obj(x, p):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    return np.sum(vals ** p) ** (1.0 / p)

def pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    sp = np.sum(vals ** p)
    base = (1.0 / p) * sp ** (1.0 / p - 1.0)
    w = p * vals ** (p - 1) * dx
    w = np.maximum(w, 0.0)
    w /= w.sum() if w.sum() > 0 else 1.0
    grads = compute_grads(x)
    return base * (w @ grads)

def try_smooth(h0, beta, maxiter=250):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def try_pnorm(h0, p, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(pnorm_obj, h0, args=(p,), jac=pnorm_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def las_step(h, tol_frac=0.98, max_lags=12):
    n = len(h)
    c_vals = np.correlate(h, 1.0 - h, mode='full')
    C = c_vals.max()
    threshold = C * tol_frac
    active = np.where(c_vals >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vals)])
    if len(active) > max_lags:
        order = np.argsort(c_vals[active])[::-1]
        active = active[order[:max_lags]]
    grads = compute_grads(h)
    G = grads[active]
    c_active = c_vals[active]
    m = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i in range(m):
        A_ub[i, :n] = G[i]
        A_ub[i, n] = -1.0
        b_ub[i] = G[i] @ h - c_active[i]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = make_feasible(res.x[:n])
            return h_new
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='interior-point')
        if res.success and res.x is not None:
            h_new = make_feasible(res.x[:n])
            return h_new
    except Exception:
        pass
    return None

def las_with_backtrack(h, tol_frac=0.98, max_lags=12, max_backtrack=4):
    n = len(h)
    current_score = official_score(h)
    for bt in range(max_backtrack):
        tf = tol_frac - 0.02 * bt
        ml = max_lags + 2 * bt
        h_new = las_step(h, tol_frac=tf, max_lags=ml)
        if h_new is None:
            continue
        sc = official_score(h_new)
        if sc < current_score - 1e-15:
            return h_new, sc
    return None, current_score

def projected_grad_descent(h, lr=0.01, n_steps=50):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_sc = official_score(h)
    for step in range(n_steps):
        c_vals = np.correlate(h, 1.0 - h, mode='full')
        C = c_vals.max()
        thresh = C * 0.99
        active = np.where(c_vals >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vals)])
        grads = compute_grads(h)
        g = np.mean(grads[active], axis=0)
        h_new = make_feasible(h - lr * g)
        sc = official_score(h_new)
        if sc < best_sc - 1e-15:
            best = h_new.copy()
            best_sc = sc
        h = h_new
        lr *= 0.98
    return best, best_sc

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')
    best_n = 0

    def is_deadline():
        return time.time() >= deadline

    def update(h, sc):
        nonlocal best_h, best_score, best_n
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)
        elif abs(sc - best_score) <= 1e-14 and len(h) < best_n:
            best_h = h.copy()
            best_n = len(h)

    # Phase 0: Initialize
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                best_n = len(parent)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = np.arange(n0)
        h_init = 0.5 + 0.25 * np.sin(2 * np.pi * 7 * x / n0) + 0.15 * np.cos(2 * np.pi * 3 * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)
        best_n = len(best_h)
    update(best_h, best_score)

    # Phase 1: LAS iterations (primary new method)
    if not is_deadline():
        cur = best_h.copy()
        cur_sc = best_score
        for it in range(200):
            if is_deadline():
                break
            h_new, sc = las_with_backtrack(cur, tol_frac=0.98, max_lags=12)
            if h_new is not None and sc < cur_sc - 1e-13:
                cur = h_new
                cur_sc = sc
                if sc < best_score - 1e-14:
                    update(h_new, sc)
            else:
                # Try with tighter active set
                h_direct = las_step(cur, tol_frac=0.995, max_lags=6)
                if h_direct is not None:
                    sc_d = official_score(h_direct)
                    if sc_d < cur_sc - 1e-13:
                        cur = h_direct
                        cur_sc = sc_d
                        if sc_d < best_score - 1e-14:
                            update(h_direct, sc_d)
                    else:
                        break

    # Phase 2: Smooth max polish
    if not is_deadline():
        cur = best_h.copy()
        for beta in [50, 200, 1000, 5000, 20000, 80000]:
            if is_deadline():
                break
            h_new, sc = try_smooth(cur, beta, maxiter=200)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
                cur = h_new
            elif h_new is not None and sc < official_score(cur):
                cur = h_new

    # Phase 3: p-norm approach
    if not is_deadline():
        cur = best_h.copy()
        for p in [8, 16, 32, 64]:
            if is_deadline():
                break
            h_new, sc = try_pnorm(cur, p, maxiter=150)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
                cur = h_new
            elif h_new is not None and sc < official_score(cur):
                cur = h_new

    # Phase 4: LAS again from improved point
    if not is_deadline():
        cur = best_h.copy()
        for it in range(100):
            if is_deadline():
                break
            h_new, sc = las_with_backtrack(cur, tol_frac=0.97, max_lags=15)
            if h_new is not None and sc < official_score(cur) - 1e-13:
                cur = h_new
                if sc < best_score - 1e-14:
                    update(h_new, sc)
            else:
                break

    # Phase 5: Projected gradient
    if not is_deadline():
        h_pgd, sc_pgd = projected_grad_descent(best_h, lr=0.02, n_steps=80)
        if sc_pgd < best_score - 1e-14:
            update(h_pgd, sc_pgd)
            # Polish
            h_new, sc = try_smooth(h_pgd, 500, maxiter=150)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)

    # Phase 6: Resolution sweep
    if not is_deadline():
        n_base = len(best_h)
        n_opts = sorted(set([max(20, n_base - 15), n_base - 5, n_base,
                             n_base + 5, min(80, n_base + 15), min(70, n_base + 20)]))
        for n_opt in n_opts:
            if is_deadline():
                break
            if n_opt == n_base:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-14:
                update(h_int, sc_int)
            h_new, sc = try_smooth(h_int, 300, maxiter=120)
            if h_new is not None:
                if sc < best_score - 1e-14:
                    update(h_new, sc)
                    h_int = h_new
            if not is_deadline():
                for it in range(30):
                    if is_deadline():
                        break
                    h_l, sc_l = las_with_backtrack(h_int, tol_frac=0.97, max_lags=10)
                    if h_l is not None and sc_l < official_score(h_int) - 1e-13:
                        h_int = h_l
                        if sc_l < best_score - 1e-14:
                            update(h_l, sc_l)
                    else:
                        break

    # Phase 7: Random restarts + LAS
    if not is_deadline():
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(3, min(20, int(remaining / 15)))
        for i in range(n_starts):
            if is_deadline():
                break
            sigma = [0.01, 0.03, 0.05, 0.08, 0.12][i % 5]
            noise = rng.normal(0, sigma, size=n)
            h_cand = make_feasible(best_h + noise)
            sc_cand = official_score(h_cand)
            # LAS from perturbed point
            h_new, sc = las_with_backtrack(h_cand, tol_frac=0.96, max_lags=15)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
            elif sc_cand < best_score - 1e-14:
                update(h_cand, sc_cand)
            # Smooth polish from perturbed
            if not is_deadline():
                h_s, sc_s = try_smooth(h_cand, 500, maxiter=100)
                if h_s is not None and sc_s < best_score - 1e-14:
                    update(h_s, sc_s)

    # Phase 8: Intensive final polish
    if not is_deadline():
        cur = best_h.copy()
        for beta in [1000, 5000, 20000, 100000]:
            if is_deadline():
                break
            h_new, sc = try_smooth(cur, beta, maxiter=300)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
                cur = h_new
        if not is_deadline():
            for it in range(50):
                if is_deadline():
                    break
                h_new, sc = las_with_backtrack(cur, tol_frac=0.99, max_lags=8)
                if h_new is not None and sc < official_score(cur) - 1e-13:
                    cur = h_new
                    if sc < best_score - 1e-14:
                        update(h_new, sc)
                else:
                    break

    # Phase 9: Very fine LES with alternating tight/loose active sets
    if not is_deadline():
        cur = best_h.copy()
        for it in range(80):
            if is_deadline():
                break
            if it % 2 == 0:
                h_new, sc = las_with_backtrack(cur, tol_frac=0.99, max_lags=6)
            else:
                h_new, sc = las_with_backtrack(cur, tol_frac=0.94, max_lags=20)
            if h_new is not None and sc < official_score(cur) - 1e-14:
                cur = h_new
                if sc < best_score - 1e-14:
                    update(h_new, sc)
            else:
                h_new, sc = try_smooth(cur, 10000, maxiter=100)
                if h_new is not None and sc < official_score(cur) - 1e-14:
                    cur = h_new
                    if sc < best_score - 1e-14:
                        update(h_new, sc)
                else:
                    break

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
