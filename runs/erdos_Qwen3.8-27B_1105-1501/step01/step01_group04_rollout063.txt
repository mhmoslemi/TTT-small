    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_simplex(h, target):
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        w = p[j] * dx
        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + m:hi2 + m]
    return grad, max_c * dx

def feasibility_grad(h, alpha, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    s = alpha * (dx * corr - tau)
    # sigmoid(s) = 1/(1+exp(-s))
    w = alpha * dx / (1.0 + np.exp(-s))
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        wi = w[j]
        if abs(wi) < 1e-15:
            continue
        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + m:hi2 + m]
    # Also compute the objective value
    obj = np.sum(np.log1p(np.exp(s))) * dx  # softplus(s) = log(1+exp(s))
    return grad, obj

def optimize_ensemble(h0, time_budget, seed=42):
    """Main optimizer combining smoothed max and feasibility approaches."""
    n = len(h0)
    target_sum = float(n) / 2.0
    h = project_to_simplex(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    # Phase 1: Smoothed max with increasing alpha
    alpha = 10.0
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    phase1_time = time_budget * 0.4
    phase2_time = time_budget * 0.35
    phase3_time = time_budget * 0.25
    
    phase1_end = t0 + phase1_time
    while time.time() < phase1_end:
        batch = 100
        for _ in range(batch):
            if time.time() >= phase1_end:
                break
            grad, _ = smooth_max_grad(h, alpha)
            v = momentum * v - lr * grad
            h_new = project_to_simplex(h + v, target_sum)
            c5 = compute_c5(h_new)
            h = h_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        alpha *= 1.05
        lr = max(lr * 0.99, 0.001)
    
    # Phase 2: Feasibility-based optimization
    phase2_end = t0 + phase1_time + phase2_time
    tau = best_c5 * 0.999  # slightly below current best
    alpha_f = 50.0
    lr_f = 0.005
    momentum_f = 0.85
    v_f = np.zeros(n)
    
    while time.time() < phase2_end:
        batch = 50
        for _ in range(batch):
            if time.time() >= phase2_end:
                break
            grad_f, _ = feasibility_grad(h, alpha_f, tau)
            v_f = momentum_f * v_f - lr_f * grad_f
            h_new = project_to_simplex(h + v_f, target_sum)
            c5 = compute_c5(h_new)
            h = h_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                # Tighten tau
                tau = max(best_c5 * 0.9995, best_c5 - 0.001)
        
        alpha_f = min(alpha_f * 1.02, 500.0)
        lr_f = max(lr_f * 0.995, 0.0005)
    
    # Phase 3: Local refinement
    phase3_end = t0 + time_budget
    h = best_h.copy()
    rng = np.random.RandomState(seed + 777)
    
    while time.time() < phase3_end:
        # Try random 2-point moves
        for _ in range(10000):
            if time.time() >= phase3_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            delta = rng.choice([0.01, 0.02, 0.05, 0.1]) * rng.choice([-1, 1])
            if delta > 0:
                # Move from j to i: h[i] += delta, h[j] -= delta
                if h[i] + delta > 1.0:
                    continue
                if h[j] - delta < 0.0:
                    continue
            else:
                # Move from i to j: h[i] += delta (negative), h[j] -= delta (positive change)
                if h[i] + delta < 0.0:
                    continue
                if h[j] - delta > 1.0:
                    continue
            
            h_test = h.copy()
            h_test[i] += delta
            h_test[j] -= delta
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-12:
                c5 = c5_test
                h = h_test
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):

    return float(np.max(corr) * dx)

    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr_weighted(h, w):
    """Gradient of sum_j w[j] * corr[j] with respect to h.
    w has length 2n-1."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        wi = w[j]
        if abs(wi) < 1e-15:

        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + m:hi2 + m]
    return grad

    corr = get_corr(h)
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))

    return grad_corr_weighted(h, p * dx), max_c * dx

    s = alpha * (dx * corr - tau)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_simplex(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradweighted(h, w):
    n = len(h)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        wi = w[j]
        if abs(wi) < 1e-15:
            continue
        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + m:hi2 + m]
    return grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_simplex(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradweighted(h, w):
    n = len(h)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        wi = w[j]
        if abs(wi) < 1e-15:
            continue
        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + m:hi2 + m]
    return grad

def opt_gradient(h0, target, t_end, seed, mode='smooth', alpha_init=15.0, lr_init=0.008):
    n = len(h0)
    dx = 2.0 / n
    h = project_simplex(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = best_c5 * 0.999
    alpha = alpha_init
    lr = lr_init
    mu = 0.90
    v = np.zeros(n)
    stall = 0
    rng = np.random.RandomState(seed)

    while time.time() < t_end:
        for _ in range(200):
            if time.time() >= t_end:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            if mode == 'smooth':
                max_c = np.max(corr)
                exp_c = np.exp(alpha * (corr - max_c))
                w = (exp_c / np.sum(exp_c)) * dx
            else:
                s = alpha * (dx * corr - tau)
                w = alpha * dx / (1.0 + np.exp(-np.clip(s, -300, 300)))
            grad = gradweighted(h, w)
            v = mu * v - lr * grad
            h_new = project_simplex(h + v, target)
            c5 = compute_c5(h_new)
            h = h_new
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
                if mode == 'feas':
                    tau = best_c5 * 0.9995

                stall += 1

            v = np.zeros(n)
            lr = max(lr * 0.5, 0.0003)
            stall = 0
        alpha = min(alpha * 1.015, 400.0)
        lr = max(lr * 0.996, 0.0002)
    return best_h, best_c5

def local_search(h0, target, t_end, seed):

    else:
