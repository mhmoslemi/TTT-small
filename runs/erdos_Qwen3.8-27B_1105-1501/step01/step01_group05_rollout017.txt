import numpy as np
import time

def objective(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n_target):
    """Clip to [0,1] and adjust sum to n_target/2."""
    h = np.array(h, dtype=float).ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        trial = np.clip(h + mid, 0.0, 1.0)
        if trial.sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact correction
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            delta = min(residual, 1.0 - h[idx])
            h[idx] += delta
            residual -= delta
            if abs(residual) > 1e-14:
                idx2 = np.argmax(1.0 - h)
                h[idx2] += residual
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
            residual += delta
            if abs(residual) > 1e-14:
                idx2 = np.argmax(h)
                h[idx2] -= residual
    h = np.clip(h, 0.0, 1.0)
    return h

def pair_line_search(h, i, j, n_evals=25):
    """Minimize along pair line (i,j) using golden section."""
    n = len(h)
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-12:
        return h, objective(h)
    
    # Check if currently at a boundary that's optimal
    dx = 2.0 / n
    h[i] = L
    h[j] = S - L
    fL = objective(h)
    h[i] = U
    h[j] = S - U
    fU = objective(h)
    
    # Restore
    hi_save, hj_save = h[i], h[j]
    
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    
    h[i] = c
    h[j] = S - c
    fc = objective(h)
    h[i] = d
    h[j] = S - d
    fd = objective(h)
    
    for _ in range(n_evals):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            h[i] = c
            h[j] = S - c
            fc = objective(h)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            h[i] = d
            h[j] = S - d
            fd = objective(h)
    
    # Pick best
    best_x, best_f = None, float('inf')
    for x in [L, U, c, d, (L+U)/2, hi_save]:
        h[i] = x
        h[j] = S - x
        if 0 <= x <= 1 and 0 <= S-x <= 1:

            if f < best_f:

                best_x = x
    
    h[i] = best_x
    h[j] = S - best_x
    return h, objective(h)

def full_pair_pass(h, rng, n_evals=20, time_limit=None, start_time=None):
    """Do one pass over all pairs, greedily improving."""
    n = len(h)
    pairs = list(range(n * (n-1) // 2))
    # Generate all pairs
    all_pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    rng.shuffle(all_pairs)
    
    score = objective(h)
    improved = 0
    
    for (i, j) in all_pairs:
        if time_limit is not None:
            if time.perf_counter() - start_time > time_limit:
                break
        h_new, s_new = pair_line_search(h, i, j, n_evals=n_evals)
        if s_new < score - 1e-13:
            h = h_new
            score = s_new
            improved += 1
    
    return h, score, improved

def basin_hop(h, rng, n_kicks=5, kick_size=None, n_evals=15):
    """Try basin hopping: kick with zero-sum block, then local optimize."""
    n = len(h)
    if kick_size is None:
        kick_size = max(4, n // 8)
    
    best_h = h.copy()
    best_score = objective(h)
    
    for _ in range(n_kicks):
        # Select random block
        block_size = rng.integers(kick_size, min(2*kick_size, n) + 1)
        block = rng.choice(n, size=block_size, replace=False)
        
        # Generate zero-sum direction
        d = rng.normal(0, 1, block_size)
        d -= d.mean()
        norm = np.max(np.abs(d))
        if norm < 1e-12:
            continue
        d /= norm
        
        # Compute max step
        h_block = h[block]
        positive = d > 0
        negative = d < 0
        max_step = 0.5
        if np.any(positive):
            max_step = min(max_step, (1.0 - h_block[positive]).min() / d[positive].max())
        if np.any(negative):
            max_step = min(max_step, h_block[negative].min() / (-d[negative]).max())
        
        if max_step < 1e-6:
            continue
        
        # Try several step sizes
        for step_frac in [0.3, 0.5, 0.7, 1.0]:
            step = step_frac * max_step
            h_try = h.copy()
            h_try[block] += step * d
            
            # Repair (should be close to feasible already)
            h_try = repair(h_try, n)
            
            # Local optimize
            h_opt, s_opt, _ = full_pair_pass(h_try, rng, n_evals=n_evals)
            
            if s_opt < best_score - 1e-12:
                best_h = h_opt
                best_score = s_opt
                
    return best_h, best_score

    if np.any(neg): ms = min(ms, hb[neg].min() / (-d[neg]).max())
    step = min(step_fraction, ms * 0.9)
    if step < 1e-6:
        return h
    h_new = h.copy()
    h_new[block] += step * d
    return repair(h_new, n)

def basin_hop(h, rng, time_limit=15.0):
    t_start = time.perf_counter()
    best_h = h.copy()
    best_score = objective(h)
    n = len(h)
    while time.perf_counter() - t_start < time_limit:
        bs = int(rng.integers(max(3, n // 15), min(n // 2, 25)))
        sf = rng.uniform(0.15, 0.9)
        h_k = zero_sum_kick(h, rng, bs, sf)
        h_o, s_o = local_optimize(h_k, rng, max_passes=6, n_evals=15, time_limit=8.0)
        if s_o < best_score - 1e-12:
            best_score = s_o; best_h = h_o
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 10.0
    rng = np.random.default_rng(seed)
    
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) >= 8 and np.all(np.isfinite(pv)):
            parent = pv
    except Exception:
        parent = None
    
    if parent is not None:
        pn = len(parent)
    else:
        pn = 64
    best_h = repair(parent if parent is not None else np.full(pn, 0.5), pn)
    best_score = objective(best_h)
    
    candidates = [(best_score, best_h.copy(), pn)]
    
    for dn in (-8, -5, -3, -1, 1, 3, 5, 8, 12):
        nc = pn + dn
        if nc < 24 or nc > 120:
            continue
        if time.perf_counter() > deadline - 60:
            break
        for off in (0, 1):
            h_i = interpolate_to(best_h, nc, off)
            candidates.append((objective(h_i), h_i, nc))
    
    # Random seeds
    for nc in (pn, pn + 5, pn - 5):
        if nc < 24 or nc > 120 or time.perf_counter() > deadline - 40:
            break
        h_r = repair(rng.uniform(0, 1, nc), nc)
        candidates.append((objective(h_r), h_r, nc))
        # Binary
        h_b = np.zeros(nc)
        idx = rng.permutation(nc)
        h_b[idx[:nc//2]] = 1.0
        if nc % 2 == 1:
            h_b[idx[nc//2]] = 1.0
        h_b = repair(h_b, nc)
        candidates.append((objective(h_b), h_b, nc))
    
    candidates.sort(key=lambda x: (x[0], x[2]))
    
    # Local optimize top candidates
    for idx in range(min(8, len(candidates))):
        if time.perf_counter() > deadline - 120:
            break
        s_c, h_c, n_c = candidates[idx]
        rem = deadline - time.perf_counter()
        tl = min(40.0, max(5.0, rem * 0.25))
        h_o, s_o = local_optimize(h_c, rng, max_passes=30, n_evals=22, time_limit=tl)
        if s_o < best_score:
            best_score = s_o; best_h = h_o
    
    # Basin hopping
    stagnation = 0
    while time.perf_counter() < deadline - 50:
        rem = deadline - time.perf_counter()
        tl = min(18.0, max(3.0, rem * 0.12))
        h_n, s_n = basin_hop(best_h, rng, time_limit=tl)
        if s_n < best_score - 1e-12:
            best_score = s_n; best_h = h_n; stagnation = 0
        else:
            stagnation += 1
        if stagnation > 8 and time.perf_counter() < deadline - 30:
            nc = int(rng.integers(max(25, len(best_h) - 10), min(110, len(best_h) + 10)))
            h_i = interpolate_to(best_h, nc)
            h_o, s_o = local_optimize(h_i, rng, max_passes=10, n_evals=20, time_limit=12.0)
            if s_o < best_score:
                best_score = s_o; best_h = h_o; stagnation = 0
            else:
