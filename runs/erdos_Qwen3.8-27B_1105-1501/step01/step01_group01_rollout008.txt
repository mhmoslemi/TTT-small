    """Compute gradient of LSE-smoothed max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = exp_vals / exp_sum
    
    grad = np.zeros(n)

        if w < 1e-15:
            continue
        shift = k - (n-1)
        if shift >= 0:
            # d(c_k)/d(h[i]) = dx * [(1-h[i-shift])*1_{i>=shift} - h[i+shift]*1_{i<=n-1-shift}]

            # For i in [0, n-1-shift]: contribution -h[i+shift]*dx*w
            # Combined for i in [shift, n-1-shift]: dx*w*(1-h[i-shift] - h[i+shift])
            # For i in [n-shift, n-1]: dx*w*(1-h[i-shift])

            for i in range(n - shift, n):
                grad[i] += w * (1.0 - h[i - shift])

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection for water-fill
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def lse_grad(h, beta):
    """Compute gradient of LSE-smoothed max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        if shift > 0:
            s = shift
            grad[s:] += w * (1.0 - h[s:])
            grad[:n-s] += w * (-h[:n-s])
        elif shift < 0:
            t = -shift
            grad[:n-t] += w * (1.0 - h[t:])
            grad[t:] += w * (-h[:-t])
        else:
            grad += w * (1.0 - 2.0 * h)
    
    return grad

def lse_descent(h0, beta_list, max_iter, step_init, rng, deadline, noise_schedule=None):
    """LSE gradient descent with Armijo line search and annealing."""
    n = len(h0)
    h = h0.copy()
    h = make_feasible(h)
    step = step_init
    best_h = h.copy()
    best_score = official_score(h)
    
    for beta in beta_list:
        iters = max_iter
        for it in range(iters):
            if time.time() > deadline:
                return best_h, best_score
            
            # Perturbation (annealing)
            if noise_schedule is not None:
                noise_level = noise_schedule(it, len(beta_list), beta, iters)
                if noise_level > 0:
                    h = make_feasible(h + noise_level * rng.standard_normal(n))

            scaled_grad = lse_grad(h, beta)  # gradient of LSE value
            
            # Project gradient onto feasible direction
            # Remove the component along the constraint (sum=0 direction)
            grad_proj = scaled_grad - np.mean(scaled_grad)
            
            # Check if gradient is essentially zero
            if np.max(np.abs(grad_proj)) < 1e-12:
                break
            
            # Armijo line search
            g_norm_sq = np.sum(grad_proj ** 2)
            if g_norm_sq < 1e-30:
                break
            
            alpha = step
            cur_lse, _, _ = lse_values(h, beta)
            
            improved = False
            for _ in range(30):
                h_trial = make_feasible(h - alpha * grad_proj)
                trial_lse, _, _ = lse_values(h_trial, beta)
                if trial_lse <= cur_lse - 1e-4 * alpha * g_norm_sq:
                    h = h_trial
                    improved = True
                    break
                alpha *= 0.5
            
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break
            
            # Check exact score improvement
            cur_score = official_score(h)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_h = h.copy()
            
            # Reset step occasionally
            if it % 100 == 0:
                step = step_init
        
        # Update step for next beta
        step = step_init * 0.8
    
    # Final exact score check
    final_score = official_score(best_h)
    if final_score < best_score:
        best_score = final_score
    
    return best_h, best_score
