import numpy as np
from scipy import optimize
from scipy.optimize import linprog
import time
import math

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    """Project v to exact target sum = n/2, with 0 <= h <= 1."""
    h = np.array(v, dtype=float).copy().ravel()[:n]
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection on shift
    lo = 1.0 - h.max() - 1.0
    hi = 2.0 - h.min() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    # Exact mass correction
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h

def grad_row(h, k):
    """Gradient of the correlation at lag k (where k = r - (n-1))."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    p = np.arange(n)
    # h_i appears as h_{n+k} when n = i-k: contributes (1-h_{i-k})
    ok1 = (p - k >= 0) & (p - k < n)
    g[ok1] += (1.0 - h[p[ok1] - k])
    # h_i appears as (1-h_n) when n = i: contributes -h_{i+k}
    ok2 = (p + k >= 0) & (p + k < n)
    g[ok2] += (-h[p[ok2] + k])
    return g * dx

def compute_all_grads(h):
    """Compute gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):
        k = r - (n - 1)
        G[r] = grad_row(h, k)
    return G

def lp_direction(h, G, active_set, inactive_margin, l1_bound=1.0):
    """Find LP tangent direction. Returns d or None."""
    n = len(h)
    # Variables: p (n), q (n), rho (1) -> total 2n+1
    # d = p - q
    # Minimize: -rho
    # Subject to:
    #   For active r: G[r]·(p-q) + rho <= 0
    #   For inactive r: G[r]·(p-q) <= inactive_margin
    #   sum(p) = sum(q)
    #   sum(p) + sum(q) <= l1_bound
    #   p, q, rho >= 0
    
    c_obj = np.zeros(2 * n + 1)
    c_obj[-1] = -1.0  # minimize -rho
    
    # Inequality constraints: A_ub @ x <= b_ub
    A_rows = []
    b_rows = []
    
    for r in active_set:
        row = np.zeros(2 * n + 1)
        row[:n] = G[r]       # G[r]·p
        row[n:2*n] = -G[r]   # -G[r]·q
        row[-1] = 1.0        # + rho
        A_rows.append(row)
        b_rows.append(0.0)
    
    # Inactive constraints (only a subset if too many)
    inactive = [r for r in range(len(active_set) + (len(G) - len(active_set))) if r not in set(active_set)]
    # Actually, let me just include all inactive
    inactive_set = set(range(len(G))) - set(active_set)
    # For speed, only include the top few inactive (closest to active)
    if len(inactive_set) > 50:
        inactive_list = sorted(inactive_set, key=lambda r: -G[r].sum()**2)[:50]
    else:
        inactive_list = list(inactive_set)
    
    for r in inactive_list:
        row = np.zeros(2 * n + 1)
        row[:n] = G[r]
        row[n:2*n] = -G[r]
        A_rows.append(row)
        b_rows.append(inactive_margin)
    
    A_ub = np.array(A_rows) if A_rows else None
    b_ub = np.array(b_rows) if b_rows else None
    
    # Equality: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.array([0.0])
    
    # L1 constraint: sum(p) + sum(q) <= l1_bound
    # Add as inequality
    l1_row = np.zeros(2 * n + 1)
    l1_row[:n] = 1.0
    l1_row[n:2*n] = 1.0
    if A_ub is not None:
        A_ub = np.vstack([A_ub, l1_row.reshape(1, -1)])
        b_ub = np.append(b_ub, l1_bound)
    else:
        A_ub = l1_row.reshape(1, -1)
        b_ub = np.array([l1_bound])
    
    bounds = [(0, None)] * (2 * n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            p = res.x[:n]
            q = res.x[n:2*n]
            d = p - q
            if np.max(np.abs(d)) > 1e-10:
                return d
    except Exception:
        pass
    return None

def local_lp_search(h, max_iters=50, time_limit=30.0, seed=42):
    """LP-based local minimax search."""
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_val = np.max(corr)
        
        # Active set
        tol = max(0.005, 0.01 * max_val)
        active = np.where(corr >= max_val - tol)[0]
        
        # Compute gradients
        G = compute_all_grads(h)
        
        # LP direction
        d = lp_direction(h, G, active.tolist(), tol)
        
        if d is None:
            # Fallback: negative sum of active gradients
            d = -np.sum(G[active], axis=0)
            d -= np.mean(d)
        
        # Normalize
        d_norm = np.max(np.abs(d))
        if d_norm < 1e-12:
            break
        d /= d_norm
        
        # Line search
        best_alpha = 0.0
        best_s = score
        alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for sign in [1, -1]:
            for a in alphas:
                alpha = sign * a
                h_new = project_to_target(h + alpha * d, n)
                s = official_score(h_new)
                if s < best_s - 1e-14:
                    best_s = s
                    best_alpha = alpha
        
        if best_alpha == 0.0:
            # Try random perturbation
            for _ in range(10):
                rnd = rng.normal(0, 0.1, n)
                rnd -= np.mean(rnd)
                rnd /= max(np.max(np.abs(rnd)), 1e-10)
                for a in [0.01, 0.05, 0.1]:
                    h_new = project_to_target(h + a * rnd, n)
                    s = official_score(h_new)
                    if s < best_s - 1e-14:
                        best_s = s
                        best_alpha = a
                        d = rnd
                        break
                if best_alpha != 0.0:
                    break
        
        if best_alpha != 0.0:
            h = project_to_target(h + best_alpha * d, n)
            score = best_s
        else:
            break
    
    return h, official_score(h)

def discrete_sa(h_init, n, time_limit=30.0, seed=42):
    """Simulated annealing on discrete (binary) patterns."""
    if n % 2 != 0:
        return None, None
    
    rng = np.random.default_rng(seed)
    # Initialize: threshold h_init
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(h_init)[::-1]
    h_bin[sorted_idx[:n//2]] = 1.0
    
    cur = h_bin.copy()
    cur_score = official_score(cur)
    best = cur.copy()
    best_score = cur_score
    
    T_start = 0.15
    T_end = 0.001
    start = time.perf_counter()
    max_iters = int(time_limit / 0.005)
    n_iters = 0
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        n_iters += 1
        
        # Temperature
        frac = n_iters / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac
        
        # Move: swap a random 1 with a random 0
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        cur[i] = 0.0
        cur[j] = 1.0
        new_score = official_score(cur)
        
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # Reject
            cur[i] = 1.0
            cur[j] = 0.0
    
    return best, best_score

def gen_mechanical(n, alpha, seed=42):
    """Generate a mechanical word (Sturmian sequence) of length n with slope alpha."""
    # Mechanical word: a_i = floor((i+1)*alpha) - floor(i*alpha)
    a = alpha
    w = np.zeros(n)
    s = 0.0
    for i in range(n):
        s += a
        w[i] = 1.0 if s >= 1.0 else 0.0
        s -= np.floor(s)
    # Adjust to exact target
    return project_to_target(w, n)

def gen_interpolated(parent, n, offset=0):
    """Interpolate parent to new resolution n."""
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n)

def slsqp_refine(h, n, time_limit=20.0):
    """SLSQP minimax refinement."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        L = 2 * n - 1
        j = np.zeros((L, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            j[ki, :n] = -grad_row(h_x, k)
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-5
    
    x0 = np.concatenate([h, [theta0]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 100, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def grad_row(h, k):
    n = len(h)

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n)


def slsqp_refine(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        L = 2 * n - 1
        j = np.zeros((L, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            j[ki, :n] = -grad_row(h_x, k)
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-5
    x0 = np.concatenate([h, [theta0]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 120, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score

        key = (n_c, tuple(np.round(h_c, 3)))
        if key in seen:
            continue
        seen.add(key)
        s = official_score(h_c)
        if np.isfinite(s):
            scored.append((s, h_c))
    
    scored.sort(key=lambda x: x[0])
    
    if scored and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)
    
    # Phase 1: LP local search on top candidates
    top_k = min(10, len(scored))
    local_results = []
    for idx in range(top_k):
        remaining = deadline - time.perf_counter()
        if remaining < 30:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        t_limit = min(45.0, max(8.0, remaining / max(1, top_k - idx) * 0.55))
        h_loc, s_loc = local_lp_search(h_c, max_iters=60, time_limit=t_limit, seed=seed + idx * 17)
        local_results.append((s_loc, h_loc))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c
    
    local_results.sort(key=lambda x: x[0])
    
    # Phase 2: Discrete search
    if time.perf_counter() < deadline - 180:
        disc_ns = [n for n in [best_n if best_n % 2 == 0 else best_n + 1,
                                best_n + 2, best_n - 2, best_n + 4, best_n + 6]
                   if n % 2 == 0 and 40 <= n <= 130]
        for n_disc in disc_ns[:3]:
            if time.perf_counter() > deadline - 140:
                break
            # From best continuous
            h_seed = best_h[:n_disc] if len(best_h) >= n_disc else np.full(n_disc, 0.5)
            h_disc, s_disc = discrete_sa(h_seed, n_disc, time_limit=20.0, seed=seed + 100)
            if h_disc is not None and s_disc < best_score:
                best_score = s_disc
                best_h = h_disc.copy()
                best_n = n_disc
            # From random binary
            for s_init in [seed + 200, seed + 300]:
                if time.perf_counter() > deadline - 120:
                    break
                h_init = np.zeros(n_disc)
                idx_sel = rng.choice(n_disc, size=n_disc//2, replace=False)
                h_init[idx_sel] = 1.0
                h_d2, s_d2 = discrete_sa(h_init, n_disc, time_limit=12.0, seed=s_init)
                if h_d2 is not None and s_d2 < best_score:
                    best_score = s_d2
                    best_h = h_d2.copy()
                    best_n = n_disc
    
    # Phase 3: Polish discrete winners with continuous LP
    if time.perf_counter() < deadline - 100:
        for n_try in [best_n, best_n + 5, best_n - 5]:
            if time.perf_counter() > deadline - 50:
                break
            if 30 <= n_try <= 140:
                h_interp = gen_interpolated(best_h, n_try)
                h_loc, s_loc = local_lp_search(h_interp, max_iters=50, time_limit=30.0, seed=seed + 500 + n_try)
                if s_loc < best_score:
                    best_score = s_loc
                    best_h = h_loc.copy()
                    best_n = n_try
    
    # Phase 4: SLSQP refinement
    if time.perf_counter() < deadline - 40:
        h_ref, s_ref = slsqp_refine(best_h, best_n, time_limit=25.0)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
    
    # Phase 5: Another round of LP after SLSQP
    if time.perf_counter() < deadline - 20:
        h_fin, s_fin = local_lp_search(best_h, max_iters=40, time_limit=12.0, seed=seed + 900)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()
    
    # Phase 6: Final SLSQP
    if time.perf_counter() < deadline - 10:
        h_ref2, s_ref2 = slsqp_refine(best_h, len(best_h), time_limit=6.0)
        if s_ref2 < best_score:
            best_score = s_ref2
            best_h = h_ref2.copy()
    
    # Final validation
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)
    
    return (best_h, float(final_score), int(len(best_h)))
