import numpy as np

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def pair_move_gradient(h, i, j):
    n = len(h)
    s = np.arange(-(n - 1), n)
    k1 = i + s
    k2 = j + s
    k3 = i - s
    k4 = j - s
    g = np.zeros(2 * n - 1)
    m1 = (k1 >= 0) & (k1 < n)
    m2 = (k2 >= 0) & (k2 < n)
    m3 = (k3 >= 0) & (k3 < n)
    m4 = (k4 >= 0) & (k4 < n)
    g[m1] += h[k1[m1]]
    g[m2] -= h[k2[m2]]
    g[m3] += h[k3[m3]]
    g[m4] -= h[k4[m4]]
    return g


def find_optimal_delta(corr, g, delta_lo, delta_hi):
    if delta_hi - delta_lo < 1e-14:
        return (delta_lo + delta_hi) / 2.0
    a, b = delta_lo, delta_hi
    fa = np.max(corr + a * g)
    fb = np.max(corr + b * g)
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + m1 * g)
        f2 = np.max(corr + m2 * g)
        if f1 < f2:
            b = m2
            fb = f2
        else:
            a = m1
            fa = f1
    return (a + b) / 2.0


def try_pair(h, corr, i, j):
    n = len(h)

    new_c5 = np.max(corr + delta_opt * g)
    current_c5 = np.max(corr)
    if new_c5 < current_c5 - 1e-12:
        h[i] -= delta_opt
        h[j] += delta_opt
        corr = corr + delta_opt * g
