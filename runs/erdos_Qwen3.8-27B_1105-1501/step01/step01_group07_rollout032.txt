def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def pair_scan_fast(h, i, j, n_pts=35):
    n = len(h)
    dx = 2.0 / n
    c = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr)) * dx
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None, cur_max
    delta_pair = np.zeros(2 * n - 1)
    lo_i = max(0, c - i)
    hi_i = min(2 * n - 2, c - i + n - 1)
    if lo_i <= hi_i:
        idxs = np.arange(lo_i, hi_i + 1)
        delta_pair[lo_i:hi_i + 1] += 1.0 - 2.0 * h[i + idxs - c]
    lo_j = max(0, c - j)
    hi_j = min(2 * n - 2, c - j + n - 1)
    if lo_j <= hi_j:
        idxs = np.arange(lo_j, hi_j + 1)
        delta_pair[lo_j:hi_j + 1] -= 1.0 - 2.0 * h[j + idxs - c]
    best_a = 0.0
    best_s = cur_max
    for a in np.linspace(lo, hi, n_pts):
        new_max = float(np.max(corr + a * delta_pair)) * dx
        if new_max < best_s:
            best_s = new_max
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_max - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return best_a, best_s
    return None, cur_max

    n = len(h)

        if d > 0 and d < n:
            lags.append(d)
    return lags

def gradient_step(h, T, step):
    n = len(h)
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode='full') * dx

    ex = np.exp((lags - base) / T)

        w = ex / S

    return g
