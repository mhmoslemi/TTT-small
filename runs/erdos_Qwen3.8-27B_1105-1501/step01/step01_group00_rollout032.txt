from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.clip(z, 0.0, 1.0)
    # Binary search for lambda such that sum(clip(z + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if residual < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if -residual < 1e-14:
                    break
    h = np.clip(h, 0.0, 1.0)
    return h

def pair_move_search(h, max_iter=100, n_samples=16, time_limit=None, t_start=None):
    """Perform pair-move optimization."""
    n = len(h)
    h = h.copy()
    current_score = official_score(h)
    
    for it in range(max_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        
        improved = False
        # Try all pairs
        for a in range(n):
            for b in range(a+1, n):
                if h[a] <= 1e-10 and h[b] <= 1e-10:
                    continue
                if 1.0 - h[a] <= 1e-10 and 1.0 - h[b] <= 1e-10:
                    continue
                
                # Feasible interval for delta
                # h[a] + delta in [0,1] => delta in [-h[a], 1-h[a]]
                # h[b] - delta in [0,1] => delta in [h[b]-1, h[b]]
                lo = max(-h[a], h[b] - 1.0)
                hi = min(1.0 - h[a], h[b])
                
                if hi <= lo + 1e-10:
                    continue
                
                # Sample the interval
                deltas = np.linspace(lo + (hi-lo)*0.01, hi - (hi-lo)*0.01, n_samples)
                # Add 0 as a reference point
                best_delta = 0.0
                best_score = current_score
                
                d0 = h[a]
                d1 = h[b]
                
                for d in deltas:
                    h[a] = d0 + d
                    h[b] = d1 - d
                    s = official_score(h)
                    if s < best_score:
                        best_score = s
                        best_delta = d
                
                # Restore
                h[a] = d0
                h[b] = d1
                
                if best_delta != 0.0:
                    h[a] = d0 + best_delta
                    h[b] = d1 - best_delta
                    current_score = best_score
                    improved = True
        
        if not improved:
            break
    
    return h, current_score

import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if residual < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if -residual < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def compute_lq(h, a, b):
    """Compute L_p and Q_p for all lags p=0..2n-2 for pair (a,b)."""
    n = len(h)
    m = 2 * n - 1
    L = np.zeros(m)
    Q = np.zeros(m)
    
    for p in range(m):
        s = p - (n - 1)
        if s >= 0:
            # Positive lag s
            # L = I(a>=s)*(1-h[a-s]) - I(a<=n-1-s)*h[a+s] - I(b>=s)*(1-h[b-s]) + I(b<=n-1-s)*h[b+s]
            if a >= s:
                L[p] += (1.0 - h[a - s])
            if a <= n - 1 - s:
                L[p] -= h[a + s]
            if b >= s:
                L[p] -= (1.0 - h[b - s])
            if b <= n - 1 - s:

            if abs(a - b) == s:

            if a <= n - 1 - t:

            if b >= t:
                L[p] += h[b - t]

                Q[p] = 1.0

def pair_move_pass(h, n_samples=32, time_limit=None, t_start=None):
    """One pass of pair-move optimization."""
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    current_score = official_score(h)
    improved = True
    passes = 0
    
    while improved and passes < 20:
        passes += 1
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')  # unscaled
        
        for a in range(n):
            for b in range(a + 1, n):
                if h[a] < 1e-10 and h[b] < 1e-10:
                    continue
                if h[a] > 1 - 1e-10 and h[b] > 1 - 1e-10:
                    continue
                
                # Feasible interval for delta
                lo_d = max(-h[a], h[b] - 1.0)
                hi_d = min(1.0 - h[a], h[b])
                if hi_d <= lo_d + 1e-12:
                    continue
                
                L, Q = compute_lq(h, a, b)
                
                # Sample delta values
                interior_lo = lo_d + 0.01 * (hi_d - lo_d)
                interior_hi = hi_d - 0.01 * (hi_d - lo_d)
                d_samples = np.linspace(interior_lo, interior_hi, n_samples)
                
                # Evaluate: max_p (corr[p] + L[p]*d + Q[p]*d^2) * dx
                # Vectorized: (n_samples, m) matrix
                vals = corr[None, :] + L[None, :] * d_samples[:, None] + Q[None, :] * d_samples[:, None] ** 2
                scores = vals.max(axis=1) * dx
                
                best_idx = np.argmin(scores)
                if scores[best_idx] < current_score - 1e-12:
                    h[a] += d_samples[best_idx]
                    h[b] -= d_samples[best_idx]
                    current_score = scores[best_idx]
                    improved = True
    
    return h, current_score

import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if residual < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if -residual < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def precompute_f(h):
    n = len(h)
    m = 2 * n - 1
    f = np.zeros((m, n))
    for p in range(m):
        s = p - (n - 1)
        if s >= 0:
            for j in range(n):
                val = 0.0
                if j >= s:
                    val += (1.0 - h[j - s])
                if j <= n - 1 - s:
                    val -= h[j + s]
                f[p, j] = val
        else:
            t = -s
            for j in range(n):
                val = 0.0
                if j <= n - 1 - t:
                    val += (1.0 - h[j + t])
                if j >= t:
                    val -= h[j - t]
                f[p, j] = val
    return f

def precompute_q(n):
    m = 2 * n - 1
    q = np.zeros((n, m))
    for d in range(1, n):
        p1 = (n - 1) + d
        p2 = (n - 1) - d
        if 0 <= p1 < m:
            q[d, p1] = 1.0
        if 0 <= p2 < m:
            q[d, p2] = 1.0
    return q

def pair_move_optimize(h, n_samples=32, max_passes=30, time_budget=300.0, t_start=None):
    if t_start is None:
        t_start = time.time()
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    h = h.copy()
    current_score = official_score(h)
    q_matrix = precompute_q(n)

    for pass_num in range(max_passes):
        if time.time() - t_start > time_budget:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        f_mat = precompute_f(h)

        for a in range(n):
            for b in range(a + 1, n):
                if h[a] < 1e-10 and h[b] < 1e-10:
                    continue
                if h[a] > 1.0 - 1e-10 and h[b] > 1.0 - 1e-10:
                    continue

                lo_d = max(-h[a], h[b] - 1.0)
                hi_d = min(1.0 - h[a], h[b])
                if hi_d <= lo_d + 1e-12:
                    continue

                L = f_mat[:, a] - f_mat[:, b]
                d = abs(a - b)
                Q = q_matrix[d]

                span = hi_d - lo_d
                d_lo = lo_d + 0.02 * span
                d_hi = hi_d - 0.02 * span
                d_samples = np.linspace(d_lo, d_hi, n_samples)

                delta_sq = d_samples ** 2
                vals = corr[None, :] + L[None, :] * d_samples[:, None] + Q[None, :] * delta_sq[:, None]
                scores = vals.max(axis=1) * dx

                best_idx = np.argmin(scores)
                if scores[best_idx] < current_score - 1e-13:
                    best_d = d_samples[best_idx]
                    h[a] += best_d
                    h[b] -= best_d
                    current_score = scores[best_idx]
                    improved = True

        if not improved:
            break

    return h, current_score
