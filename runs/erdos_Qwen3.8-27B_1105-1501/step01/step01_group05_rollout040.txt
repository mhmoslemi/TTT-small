import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(z, n):
    """Project z onto {h: 0≤h[i]≤1, sum(h)=n/2}"""
    z = np.asarray(z, dtype=float).ravel()[:n].copy()
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(z - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(z - mid, 0.0, 1.0)
    # Handle tiny residual
    residual = target - np.sum(proj)
    if abs(residual) > 1e-14:
        if residual > 0:
            # Find element with most room
            room = 1.0 - proj
            # Add to elements with most room
            idx = np.argsort(-room)[:max(1, int(abs(residual)))]
            # Distribute
            per = residual / len(idx)
            for i in idx:
                add = min(per, room[i])
                proj[i] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            # Remove from elements with most value
            idx = np.argsort(-proj)[:max(1, int(abs(residual)))]
            per = -residual / len(idx)
            for i in idx:
                rem = min(per, proj[i])
                proj[i] -= rem
                residual += rem
                if abs(residual) < 1e-15:
                    break
    return np.clip(proj, 0.0, 1.0)

def lse_score(h, tau):
    """Log-sum-exp smoothed score"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    return m + tau * np.log(np.sum(np.exp((scaled - m) / tau)))

def lse_weights(h, tau):
    """Get softmax weights for LSE"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    w = np.exp((scaled - m) / tau)
    w /= np.sum(w)
    return w, scaled

def lse_gradient(h, tau):
    """Gradient of LSE smoothed score"""
    n = len(h)
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    w, _ = lse_weights(h, tau)
    grad = np.zeros(n)
    for ki in range(len(lags)):
        if w[ki] > 1e-12:
            grad += w[ki] * dx * compute_gradient_row(h, lags[ki])
    return grad

def compute_gradient_row(h, k):
    n = len(h)
    p = np.arange(n)
    g = np.zeros(n)
    ok1 = (p - k) >= 0
    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g

def lse_descent(h, n, tau_start=0.1, tau_end=0.01, n_iter=200, time_limit=30.0):
    """LSE-smoothed projected gradient descent"""
    start = time.perf_counter()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_score = official_score(h)
    
    for it in range(n_iter):
        if time.perf_counter() - start > time_limit:
            break
        # Linearly anneal temperature
        frac = it / max(1, n_iter - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute gradient
        grad = lse_gradient(h, tau)
        
        # Remove mean component (to preserve sum constraint approximately)
        grad -= np.mean(grad)
        
        # Line search
        best_alpha = 0.0
        best_local = lse_score(h, tau)
        for lr in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            h_new = project_feasible(h - lr * grad, n)
            s = lse_score(h_new, tau)
            if s < best_local - 1e-10:
                best_local = s
                best_alpha = lr
        
        if best_alpha > 0:
            h = project_feasible(h - best_alpha * grad, n)
            exact_s = official_score(h)
            if exact_s < best_score:
                best_score = exact_s
                best_h = h.copy()
    
    return best_h, best_score

        else:

            order = np.argsort(-room)
            for i in order:

                proj[i] += add
                residual -= add

            for i in order:
                rem = min(-residual, proj[i])
                proj[i] -= rem
                residual += rem
                if abs(residual) < 1e-15:

def corr_scaled(h):
    """Return dx * correlate(h, 1-h, mode='full')"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lse_value(scaled, tau):
    m = np.max(scaled)
    return m + tau * np.log(np.sum(np.exp((scaled - m) / tau)))

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    scaled = corr_scaled(h)
    m = np.max(scaled)
    w = np.exp((scaled - m) / tau)
    w /= np.sum(w)
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for ki in range(len(lags)):
        if w[ki] > 1e-14:
            k = lags[ki]
            p = np.arange(n)
            g = np.zeros(n)
            ok1 = (p - k) >= 0
            ok1 &= (p - k) < n
            g[ok1] += (1.0 - h[p[ok1] - k])
            ok2 = (p + k) >= 0
            ok2 &= (p + k) < n
            g[ok2] += (-h[p[ok2] + k])
            grad += w[ki] * dx * g
    return grad

def lse_descent(h0, n, tau_start, tau_end, n_iters, step_scale, time_limit):
    start = time.perf_counter()
    h = project_feasible(h0, n).copy()
    best_h = h.copy()
    best_exact = official_score(h)
    
    lr = step_scale
    for it in range(n_iters):
        if time.perf_counter() - start > time_limit:
            break
        frac = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad = lse_gradient(h, tau)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        best_alpha = 0.0
        best_lse = lse_value(corr_scaled(h), tau)
        for trial in range(12):
            alpha = lr / (2 ** trial)
            h_new = project_feasible(h - alpha * grad, n)
            lse_new = lse_value(corr_scaled(h_new), tau)
            if lse_new < best_lse - 1e-12:
                best_alpha = alpha
                best_lse = lse_new
                break
        
        if best_alpha > 0:
            h = project_feasible(h - best_alpha * grad, n)
            lr = min(1.0, lr * 1.5)
            exact = official_score(h)
            if exact < best_exact:
                best_exact = exact
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_exact

def balanced_descent(h0, n, top_k, n_iters, time_limit):
    start = time.perf_counter()
    h = project_feasible(h0, n).copy()
    best_h = h.copy()
    best_exact = official_score(h)
    lags = np.arange(-(n-1), n)
    
    for it in range(n_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        scaled = corr_scaled(h)
        # Find top-k active
        top_idx = np.argsort(-scaled)[:top_k]
        
        # Compute gradient rows for active constraints
        grads = np.zeros((top_k, n))
        for i, ki in enumerate(top_idx):
            k = lags[ki]
            p = np.arange(n)
            g = np.zeros(n)
            ok1 = (p - k) >= 0
            ok1 &= (p - k) < n
            g[ok1] += (1.0 - h[p[ok1] - k])
            ok2 = (p + k) >= 0
            ok2 &= (p + k) < n
            g[ok2] += (-h[p[ok2] + k])
            grads[i] = g
        
        # Min-norm direction: d = G^T (GG^T + eps*I)^{-1} (-1)
        G = grads  # shape (top_k, n)
        # We want G @ d <= -eps for all rows (decrease all active)
        # Min-norm solution: d = -G^T (G G^T + reg*I)^{-1} ones
        GTG = G @ G.T + 1e-8 * np.eye(top_k)
        try:
            rhs = np.ones(top_k)
            coeffs = np.linalg.solve(GTG, rhs)
            d = -G.T @ coeffs
        except np.linalg.LinAlgError:
            d = -np.mean(grads, axis=0)
        
        d -= np.mean(d)
        
        # Line search
        best_alpha = 0.0
        best_val = np.max(scaled)
        for a in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_new = project_feasible(h - a * d, n)
            s = official_score(h_new)
            if s < best_val - 1e-12:
                best_val = s
                best_alpha = a
        
        if best_alpha > 0:
            h = project_feasible(h - best_alpha * d, n)
            if best_val < best_exact:
                best_exact = best_val
                best_h = h.copy()
        else:
            # Try smaller
            for a in [0.0001, 0.0005]:
                h_new = project_feasible(h - a * d, n)
                s = official_score(h_new)
                if s < best_exact - 1e-12:
                    h = h_new
                    best_exact = s
                    best_h = h_new.copy()
                    break
            else:
                break
    
    return best_h, best_exact

import numpy as np

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(z, n):
    z = np.asarray(z, dtype=float).ravel()[:n].copy()
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(proj)
    if abs(residual) > 1e-15:
        if residual > 0:
            order = np.argsort(-(1.0 - proj))
            for i in order:
                add = min(residual, 1.0 - proj[i])
                proj[i] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(-proj)
            for i in order:
                rem = min(-residual, proj[i])
                proj[i] -= rem
                residual += rem
                if abs(residual) < 1e-15:
                    break
    return np.clip(proj, 0.0, 1.0)

def lse_value(scaled, tau):
    m = np.max(scaled)
    return m + tau * np.log(np.sum(np.exp((scaled - m) / tau)))

def grad_single_lag(h, k):
    n = len(h)
    p = np.arange(n)
    g = np.zeros(n)
    ok1 = (p - k) >= 0
    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g

def lse_weights(scaled, tau):
    m = np.max(scaled)
    w = np.exp((scaled - m) / tau)
    w /= np.sum(w)
    return w

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    scaled = corr_scaled(h)
    w = lse_weights(scaled, tau)
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    # Only compute for significant weights
    significant = w > 1e-10
    for ki in np.where(significant)[0]:
        grad += w[ki] * dx * grad_single_lag(h, lags[ki])
    return grad

def interpolate_to(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_feasible(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_feasible(h, n_target)

def gen_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 0.5:
        m_full += 1
        frac -= 1.0
    if frac > 1e-10:
        h[idx[m_full]] = frac
    return project_feasible(h, n)

def gen_smooth(n, rng):
    # Generate a smooth periodic-like function
    n_modes = rng.integers(2, 5)
    h = np.full(n, 0.5)
    for m in range(n_modes):
        freq = rng.uniform(1, 4) * (m + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3) / (m + 1)
        t = np.arange(n)
        h += amp * np.sin(2 * np.pi * freq * t / n + phase)
    return project_feasible(h, n)

= h[i] * (1 - h[i+k] - h[i-k]) + h[i-k]

This is LINEAR in h[i] for k≠0!

But for k=0:
c_0 = (terms not involving h[i]) + h[i] - h[i]^2
= (terms not involving h[i]) - h[i]^2 + h[i]

This is QUADRATIC in h[i] (specifically, a concave quadratic).

So the function f_i(t) = max_k (contribution of k at value t) is:
- For k=0: B_0 - t^2 + t (quadratic, concave)
- For k≠0: A_k * t + B_k (linear)

The max of a concave quadratic and several linear functions is... not necessarily convex! The max of concave and linear is not convex in general.
