import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        # Distribute the correction to the elements furthest from the boundary
        if diff > 0:
            order = np.argsort(z)  # ascending, smallest first (most room to increase)
            for idx in order:
                room = 1.0 - z[idx]
                if room > 1e-12:
                    add = min(diff, room)
                    z[idx] += add
                    diff -= add
                    if abs(diff) < 1e-12:
                        break
        else:
            order = np.argsort(-z)  # descending, largest first (most room to decrease)
            for idx in order:
                if z[idx] > 1e-12:
                    sub = min(-diff, z[idx])
                    z[idx] -= sub
                    diff += sub
                    if abs(diff) < 1e-12:
                        break
    return np.clip(z, 0.0, 1.0)

    n = len(h)

    """
    Build and solve the trust-region SOCP.
    active_idx: indices into the full correlation array
    delta: trust region radius
    Returns: (new_h, new_score) or (None, inf)
    """
    n = len(h)
    m = n - 1  # index offset for the "zero lag"
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    cons = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n / 2.0,
        cp.norm(x - h) <= delta,
    ]
    
    lv = lag_values(h)
    
    for idx in active_idx:
        d = idx - m  # actual lag (can be negative)
        # Compute gradient of G[d] at h
        ad = abs(d)
        nn = n - ad
        g = np.zeros(n)
        if d >= 0:
            g[:nn] += (1.0 - h[ad:ad+nn])
            g[ad:ad+nn] -= h[:nn]
        else:
            g[ad:ad+nn] += (1.0 - h[:nn])
            g[:nn] -= h[ad:ad+nn]
        cons.append(lv[idx] + g @ (x - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), cons)
    
    solvers = [cp.SCS, cp.ECOS, cp.CLARABEL]

        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except:
            continue

        return None, float('inf')
    
    x_val = np.clip(x.value, 0.0, 1.0)
    x_val = make_feasible(x_val)
    c = score_exact(x_val)
    return x_val, c

def softmax_gradient(h, T):
    """Compute soft-max gradient (same as previous method)"""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        w = np.zeros_like(scores)
        w[scores >= base - 1e-10] = 1.0
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g

import numpy as np
import time
import cvxpy as cp

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(z)
            for idx in order:
                room = 1.0 - z[idx]
                if room > 1e-12:
                    add = min(diff, room)
                    z[idx] += add
                    diff -= add
                    if abs(diff) < 1e-12:
                        break
        else:
            order = np.argsort(-z)
            for idx in order:
                if z[idx] > 1e-12:
                    sub = min(-diff, z[idx])
                    z[idx] -= sub
                    diff += sub
                    if abs(diff) < 1e-12:
                        break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h, idx):
    N = len(h)
    d = idx - (N - 1)
    g = np.zeros(N)
    if d >= 0:
        m = N - d
        g[d:d+m] += (1.0 - h[:m])
        g[:m] -= h[d:d+m]
    else:
        e = -d
        m = N - e
        g[:m] += (1.0 - h[e:e+m])
        g[e:e+m] -= h[:m]
    return g

def solve_sosoc(h, active_idx, delta):
    N = len(h)
    x = cp.Variable(N)
    t = cp.Variable()
    cons = [
        x >= 0,
        x <= 1,
        cp.sum(x) == N / 2.0,
        cp.norm(x - h, 2) <= delta,
    ]
    lv = lag_values(h)
    for idx in active_idx:
        g = corr_gradient(h, idx)
        cons.append(lv[idx] + g @ (x - h) <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    x_val = None
    for solver in (cp.SCS, cp.CLARABEL, cp.ECOS):
        try:
            prob.solve(solver=solver, **{'max_iters': 10000, 'eps': 1e-8} if solver == cp.SCS else {})
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = x.value
                break
        except Exception:
            continue
    if x_val is None:
        return None, float('inf')
    x_val = np.clip(x_val, 0.0, 1.0)
    x_val = make_feasible(x_val)
    return x_val, score_exact(x_val)

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        w = np.zeros_like(scores)
        idxs = np.where(scores >= base - 1e-10)[0]
        w[idxs] = 1.0
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g

def grad_step(h, deadline, T0, T_min, decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T0
    step = step0
    stall = 0
    while time.time() < deadline:
        g = softmax_gradient(cur, T)
        gn = float(np.linalg.norm(g))
        if gn < 1e-18:
            break
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 30:
                cur = make_feasible(cur + rng.normal(0, 0.005, n))
                cur_c = score_exact(cur)
                step = max(T, 0.003)
                stall = 0
        else:
            step = min(step * 1.03, 0.1)
            stall = 0
        T = max(T_min, T * decay)
    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)

                h[j] = orig_j - a1

                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:

def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def select_active(h, lags, frac=0.92, max_k=25):
    n = len(h)
    cur_max = float(np.max(lags))
    threshold = cur_max * frac
    active = np.where(lags >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(lags))])
    if len(active) > max_k:
        order = active[np.argsort(lags[active])[::-1]]
        active = order[:max_k]
    return active

def gradient_informed_pairs(h, lags, n_pairs=10):
    """Select pairs (i,j) based on gradient information for the max lag."""
    n = len(h)
    max_idx = int(np.argmax(lags))
    g = corr_gradient(h, max_idx)
    # We want to increase h where g < 0 and decrease h where g > 0
    # (since decreasing the max lag means moving in direction -g)
    # So good source: high h and positive g (decreasing it helps)
    # Good target: low h and negative g (increasing it helps)
    sources = np.argsort(-h * g)[::-1]  # high h*g means reducing h helps most
    targets = np.argsort(h * g)         # low h*g (negative) means increasing h helps
    pairs = []
    for s in sources[:n_pairs * 2]:
        for t in targets[:n_pairs * 2]:
            if s != t and h[s] > 0.1 and h[t] < 0.9:
                pairs.append((int(s), int(t)))
                if len(pairs) >= n_pairs:
                    return pairs
    return pairs

                cur = x_new
                cur_c = c_new
                stall = 0
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c
            else:
                delta *= 0.75
                stall += 1
                if stall > 8:
                    cur = make_feasible(cur + rng.normal(0, 0.003, n))
                    cur_c = score_exact(cur)
                    delta = 0.12
                    stall = 0

    # Phase 5: Multi-resolution exploration (3% of time)
    p5_end = deadline
    if time.time() < p5_end:
        n_cur = len(best_h)
        for dn in (-5, 5, -10, 10):
            if time.time() > p5_end - 10:
                break
            nr = n_cur + dn
            if nr < 20 or nr > 150:
                continue
            h_r = resample_h(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(p5_end - 3, time.time() + 40)
            if time.time() < sub_dl:
                h_r, c_r = grad_step(h_r, sub_dl, 0.005, 0.0001, 0.999, 0.003, rng)
            sub_dl2 = min(p5_end - 3, time.time() + 30)
            if time.time() < sub_dl2:
                for _ in range(40):
                    if time.time() > sub_dl2:
                        break
                    i = int(rng.integers(0, nr))
                    j = int(rng.integers(0, nr))
                    if i != j:
                        res = pair_search(h_r, i, j, c_r, n_pts=20)
                        if res is not None:
                            c_r = res
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    # Final: ensure feasibility and compute exact score
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
