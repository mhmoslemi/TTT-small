import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # Fine correction
    err = target - np.sum(h)
    if abs(err) > 1e-14:
        # Distribute the error among non-saturated entries
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            adjust = err / np.sum(free)
            h[free] = np.clip(h[free] + adjust, 0.0, 1.0)
            # Second pass
            err2 = target - np.sum(h)
            if abs(err2) > 1e-14:
                h = np.clip(h + err2 / n, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_overlaps(h):
    """Compute all overlap values (before multiplying by dx)."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_at_k(h, k, n):
    """Gradient of overlap_k with respect to h.
    k is the shift index in the full correlation array (0 to 2n-2).
    The actual shift is k - (n-1)."""
    shift = k - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        # overlap = sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i])
        # d/dh[i+shift] = (1-h[i]) for i=0..n-1-shift
        # d/dh[i] = -h[i+shift] for i=0..n-1-shift
        end = n - shift
        grad[shift:] += (1.0 - h[:-shift]) if shift > 0 else (1.0 - h)
        if shift > 0:
            grad[:-shift] += (-h[shift:])
    else:
        # shift < 0, let s = -shift
        s = -shift
        # overlap = sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
        end = n - s
        grad[:end] += (1.0 - h[s:])
        grad[s:] += (-h[:end])
    return grad

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = compute_overlaps(h)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax
    return cmax + (mk + np.log(ws)) / beta

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = compute_overlaps(h)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        grad += weights[k] * compute_grad_at_k(h, k, n)
    return grad * dx

def project_feasible(y):
    """Project y onto {0<=h<=1, sum(h)=n/2}"""
    return make_feasible(y)

def projected_gradient_nesterov(h0, beta, step0, n_iter, min_step=1e-6):
    """Nesterov accelerated projected gradient for the smoothed objective."""
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    step = step0
    best_h = h.copy()
    best_val = smooth_objective(h, beta)
    
    for it in range(n_iter):
        # Nesterov extrapolation
        beta_m = (t - 1.0) / (t + 1.0 + np.sqrt(1.0 + 4.0 * t * t))
        # Actually: t_new = (1 + sqrt(1+4t^2))/2, beta = (t-1)/t_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_m = (t - 1.0) / t_new
        y = h + beta_m * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)  # Keep in box for evaluation
        
        # Compute gradient at y (projected)
        y_proj = project_feasible(y)
        g = smooth_gradient(y_proj, beta)
        
        # Step
        h_new = y_proj - step * g
        h_new = project_feasible(h_new)
        
        val = smooth_objective(h_new, beta)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        t = t_new
        step = max(min_step, step0 / (1.0 + 0.01 * it))
    
    return best_h

        return J
    
    def eq(z):
        return np.sum(z[:n]) - n/2.0
    
    def eq_jac(z):
        g = np.ones(n+1)
        g[-1] = 0.0
        return g
    
    theta0 = score0 + 1e-4

    bounds = [(0.0, 1.0)]*n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    try:

    except Exception:
        pass
    return h0

def active_eq_fun(z):
    n = len(z) - 1
    return float(np.sum(z[:n]) - n / 2.0)

def active_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g

def _direct_obj(z):
    return z[-1]

def _direct_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def direct_solve(h0, maxiter=200, use_active=False, max_active=8, tol=1e-10):
    global _active_idx
    n = len(h0)
    h = h0.copy()
    for _ in range(50):
        if use_active:
            F = get_overlaps(h)
            dx = 2.0 / n
            C = F * dx
            cmax = C.max()
            active = np.where(C >= cmax - tol)[0]
            if len(active) > max_active:
                active = np.argsort(C)[::-1][:max_active]
            _active_idx = np.sort(active)
            cons = [
                {'type': 'ineq', 'fun': active_ineq_fun, 'jac': active_ineq_jac},
                {'type': 'eq', 'fun': active_eq_fun, 'jac': active_eq_jac}
            ]
        else:
            cons = [
                {'type': 'ineq', 'fun': full_ineq_fun, 'jac': full_ineq_jac},
                {'type': 'eq', 'fun': full_eq_fun, 'jac': full_eq_jac}
            ]
        theta0 = official_score(h) + 1e-4
        z0 = np.concatenate([h, [theta0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        try:
            res = minimize(_direct_obj, z0, jac=_direct_obj_jac, bounds=bounds,
                           constraints=cons, method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = make_feasible(res.x[:n])
                sc_new = official_score(h_new)
                sc_old = official_score(h)
                if sc_new < sc_old - 1e-14:
                    h = h_new
                else:
                    break
            else:
                break
        except Exception:
            break
    return h

def smooth_solve(h0, beta, maxiter=200):
    n = len(h0)
    h = h0.copy()
    if n > 80:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(_smooth_obj, h, args=(beta,), jac=_smooth_obj_jac,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h

def proj_grad_nesterov(h0, beta, step0, n_iter):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    step = step0
    for it in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_m = (t - 1.0) / t_new
        y = h + beta_m * (h - h_prev)
        y_proj = make_feasible(y)
        g = smooth_obj_jac(y_proj, beta)
        h_new = make_feasible(y_proj - step * g)
        h_prev = h
        h = h_new
        t = t_new
        step = max(1e-7, step0 / (1.0 + 0.02 * it))
    return h

def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def symmetry_transform(h):
    return make_feasible(1.0 - h[::-1])
