```python
def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

            if 0 <= j2 < n:
```

```python
def coord_descent_pass_fast(h, lam, n_iter_inner=40):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for i in range(n):
        # Compute how corr changes when h[i] changes
        # For each k, delta_corr[k] = delta * coeff[k] where:
        # coeff[k] = (1-h[i+lag_k]) if 0<=i+lag_k<n else 0 
        #           - h[i-lag_k] if 0<=i-lag_k<n else 0
        coeff = np.zeros(2*n - 1)
        for idx in range(2*n - 1):
            lag = idx - (n - 1)
            c = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                c += (1.0 - h[j1])
            j2 = i - lag
            if 0 <= j2 < n:
                c -= h[j2]
            coeff[idx] = c
        
        # corr_new = corr + (t - h[i]) * coeff
        # Want to minimize max(corr + (t-h[i])*coeff) + lam*t
        # Let u = t - h[i], then t = h[i] + u, and u ∈ [0-h[i], 1-h[i]]
        # Objective: max(corr + u*coeff) + lam*(h[i] + u)
        # = max(corr + u*coeff) + lam*u + const
        # = max_k(corr[k] + u*coeff[k]) + lam*u
        
        # 1D convex minimization over u ∈ [-h[i], 1-h[i]]
        # g(u) = max_k(corr[k] + u*coeff[k]) + lam*u
        
        lo_u, hi_u = -h[i], 1.0 - h[i]
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_u = lo_u + (1.0 - gr) * (hi_u - lo_u)
        d_u = lo_u + gr * (hi_u - lo_u)
        
        def g(u):
            return np.max(corr + u * coeff) + lam * u
        
        fc, fd = g(c_u), g(d_u)
        for _ in range(n_iter_inner):
            if fc <= fd:
                hi_u, d_u, fd = d_u, c_u, fc
                c_u = lo_u + (1.0 - gr) * (hi_u - lo_u)
                fc = g(c_u)
            else:
                lo_u, c_u, fc = c_u, d_u, fc
                d_u = lo_u + gr * (hi_u - lo_u)
                fd = g(d_u)
        
        u_star = (lo_u + hi_u) / 2.0
        delta = u_star
        h[i] += delta
        corr += delta * coeff
    
    return h
```

```python
def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-14:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def coord_pass(h, lam, n_inner=50, order=None):
    """One full pass of coordinate descent."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    if order is None:
        order = range(n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for i in order:
        coeff = np.zeros(2 * n - 1)
        for idx in range(2 * n - 1):
            lag = idx - (n - 1)
            c = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                c += 1.0 - h[j1]
            j2 = i - lag
            if 0 <= j2 < n:
                c -= h[j2]
            coeff[idx] = c
        
        lo_u = -h[i]
        hi_u = 1.0 - h[i]
        if lo_u == hi_u:
            continue
        
        c_u = lo_u + (1.0 - gr) * (hi_u - lo_u)
        d_u = lo_u + gr * (hi_u - lo_u)
        fc = np.max(corr + c_u * coeff) + lam * (h[i] + c_u)
        fd = np.max(corr + d_u * coeff) + lam * (h[i] + d_u)
        
        for _ in range(n_inner):
            if fc <= fd:
                hi_u = d_u
                d_u = c_u
                fd = fc
                c_u = lo_u + (1.0 - gr) * (hi_u - lo_u)
                fc = np.max(corr + c_u * coeff) + lam * (h[i] + c_u)
            else:
                lo_u = c_u
                c_u = d_u
                fc = fd
                d_u = lo_u + gr * (hi_u - lo_u)
                fd = np.max(corr + d_u * coeff) + lam * (h[i] + d_u)
        
        u_star = (lo_u + hi_u) / 2.0
        h[i] += u_star
        corr += u_star * coeff
    
    return h

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        j_lo = max(0, shift)
        j_hi = min(n, n + shift)
        for p in range(j_lo, j_hi):
            grad[p] += w * (1.0 - x[p - shift])
        p_lo2 = max(0, j_lo - shift)
        p_hi2 = min(n, j_hi - shift)
        for p in range(p_lo2, p_hi2):
            grad[p] += w * (-x[p + shift])
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

    try:

                       method='SLSQP', bounds=bounds,

                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):

    return h0
```

However, there's a subtle issue: the golden section search finds the optimum of the 1D function, but this assumes the other coordinates stay fixed. In Gauss-Seidel, after updating h[i], the next coordinate h[i+1] is optimized with the new h[i]. This is standard and should work.

But wait - I'm using a random order each pass. This is more like a "randomized Gauss-Seidel" which can help escape local structure.

Let me also think about whether the golden section search is the best 1D solver here. The function g(u) = max_k(corr[k] + u*coeff[k]) + lam*(h[i]+u) is convex and piecewise linear. Its minimum is at a vertex of the upper envelope or at a boundary.

```python
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-14:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def _coeff_vec(i, h, n):
    m = 2 * n - 1
    coeff = np.zeros(m)
    base = n - 1
    for idx in range(m):
        lag = idx - base
        c = 0.0
        j1 = i + lag
        if 0 <= j1 < n:
            c += 1.0 - h[j1]
        j2 = i - lag
        if 0 <= j2 < n:
            c -= h[j2]
        coeff[idx] = c
    return coeff

def coord_pass(h, lam, n_inner=45, order=None):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    if order is None:
        order = np.arange(n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    m = 2 * n - 1
    for i in order:
        coeff = _coeff_vec(i, h, n)
        lo_u = -h[i]
        hi_u = 1.0 - h[i]
        if hi_u - lo_u < 1e-15:
            continue
        c_u = lo_u + (1.0 - gr) * (hi_u - lo_u)
        d_u = lo_u + gr * (hi_u - lo_u)
        fc = np.max(corr + c_u * coeff) + lam * (h[i] + c_u)
        fd = np.max(corr + d_u * coeff) + lam * (h[i] + d_u)
        for _ in range(n_inner):
            if fc <= fd:
                hi_u = d_u
                d_u = c_u
                fd = fc
                c_u = lo_u + (1.0 - gr) * (hi_u - lo_u)
                fc = np.max(corr + c_u * coeff) + lam * (h[i] + c_u)
            else:
                lo_u = c_u
                c_u = d_u
                fc = fd
                d_u = lo_u + gr * (hi_u - lo_u)
                fd = np.max(corr + d_u * coeff) + lam * (h[i] + d_u)
        u_star = (lo_u + hi_u) / 2.0
        h[i] += u_star
        corr += u_star * coeff
    return h

    target = n // 2

    else:
        target = (n - 1) // 2
    idx = np.argsort(h)[::-1]

    return h_bin

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        j_lo = max(0, shift)
        j_hi = min(n, n + shift)
        for p in range(j_lo, j_hi):
            grad[p] += w * (1.0 - x[p - shift])
        p_lo2 = max(0, j_lo - shift)
        p_hi2 = min(n, j_hi - shift)
        for p in range(p_lo2, p_hi2):
            grad[p] += w * (-x[p + shift])
    return grad

    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def do_smooth_opt(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))

        if 0 <= j2 < n:
            g -= h[j2]
        grad[i] = g

            if time.time() > deadline - 8:

            h_pert = make_feasible(best_h + 0.03 * rng.standard_normal(best_n))
            h_p, s_p, _ = run_cd(h_pert, max_passes=300, rng=rng,
                                  lam0=0.0, n_inner=45,
                                  time_limit=5.0, t_start=time.time())
            s_p = official_score(h_p)
            if s_p < best_score - 1e-13:
                best_h = h_p.copy()
                best_score = s_p
                best_n = len(h_p)
            # SLSQP on perturbed
            if time.time() < deadline - 8:
                h_sp = do_smooth_opt(best_h, 500, maxiter=100)
                s_sp = official_score(h_sp)
                if s_sp < best_score - 1e-13:
                    best_h = h_sp.copy()
                    best_score = s_sp

    # Very final SLSQP
    if time.time() < deadline - 5:
        h_final = do_smooth_opt(best_h, 2000, maxiter=100)
        s_final = official_score(h_final)
        if s_final < best_score:
            best_h = h_final
            best_score = s_final

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)

    return (best_h, final_score, len(best_h))
```
