import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    # Bisection on shift to satisfy sum constraint
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y + mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(z)  # smallest first (most room)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(-z)  # largest first (most to remove)
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:

    grad = np.zeros(n)
    if k >= 0:

        grad[p:p+m] += dx * (1.0 - h[:m])
        grad[:m] -= dx * h[p:p+m]

    t = z[n]

    n = len(h)
    gx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += gx * (1.0 - h[k:])
            g[k:k+m] -= gx * h[:m]
    else:

        m = n - p
        if m > 0:
            g[p:p+m] += gx * (1.0 - h[:m])
            g[:m] -= gx * h[p:p+m]
    return g

    n = len(h)
    dx = 2.0 / n
    if k >= 0:
        m = n - k
        if m <= 0:
            return -t
        return float(np.sum(h[:m] * (1.0 - h[k:]))) * dx - t

        p = -k
        m = n - p
        if m <= 0:
            return -t
        return float(np.sum(h[p:p+m] * (1.0 - h[:m]))) * dx - t

    """Solve the epigraph minimax problem with active constraints."""
    # Map np.correlate index to lag
    # np.correlate(h, 1-h, 'full') has length 2n-1
    # Index i corresponds to lag k = i - (n-1)
    
    num_con = len(active_k)
    
    def objective(z):
        return z[n]  # minimize t
    
    def objective_grad(z):
        gz = np.zeros(n + 1)
        gz[n] = 1.0
        return gz
    
    constraints = []
    for k in active_k:
        # Create constraint dict
        def cons(z, kk=k):
            h = z[:n]
            t = z[n]
            return con_val_fn(h, t, kk)
        def cons_grad(z, kk=k):
            h = z[:n]
            t = z[n]
            gz = np.zeros(n + 1)
            gz[:n] = con_grad_fn(h, t, kk)
            gz[n] = -1.0
            return gz
        constraints.append({'type': 'ineq', 'fun': cons, 'jac': cons_grad})
    
    # Equality constraint: sum(h) - n/2 = 0
    def eq_con(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_con_grad(z):
        gz = np.zeros(n + 1)
        gz[:n] = 1.0
        return gz
    
    constraints.append({'type': 'eq', 'fun': eq_con, 'jac': eq_con_grad})
    
    bounds = [(0, 1)] * n + [(0, 1.0)]
    x0 = np.concatenate([h_init, [score_exact(h_init)]])
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': tol, 'iprint': 0})
        if res.success or res.x is not None:
            h_sol = res.x[:n]
            h_sol = make_feasible(h_sol, n)
            c_sol = score_exact(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return None, None

def grad_overlap(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return grad
        # First factor: h[j+k] for j=0..m-1, indices k..n-1
        # d/dh[i] for i in [k, n-1]: (1 - h[i-k])
        grad[k:] += (1.0 - h[:m])
        # Second factor: h[j] for j=0..m-1, indices 0..m-1
        # d/dh[i] for i in [0, m-1]: -h[i+k]
        grad[:m] -= h[k:]

        m = n - p
        if m <= 0:
            return grad
