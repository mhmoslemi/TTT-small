```python
from scipy.optimize import linprog
import time

def exact_score(h):
    """Compute C5 = max_k ∫h(x)(1-h(x+k))dx using np.correlate."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(z, n):
    """Project z onto {h: 0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    # Binary search for shift
    lo = -1.0 - np.max(z) - 0.1
    hi = 1.0 - np.min(z) + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        # Distribute error
        for _ in range(200):
            if abs(err) < 1e-14:
                break
            if err > 0:
                mask = h < 1.0
                cnt = np.sum(mask)
                if cnt == 0:
                    break
                add = min(err, cnt * 1e-12)
                h[mask] += add / cnt
            else:
                mask = h > 0.0
                cnt = np.sum(mask)
                if cnt == 0:
                    break
                sub = min(-err, cnt * 1e-12)
                h[mask] -= sub / cnt
    return np.clip(h, 0.0, 1.0)

def compute_all_C(h, n):
    """Compute all C_k values (scaled by dx)."""
    dx = 2.0 / n
    C = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):

        e = min(n, n - k)
        C[k + n - 1] = dx * np.sum(h[s:e] * (1.0 - h[s+k:e+k]))
    return C

def compute_gradient_matrix(h, n):
    """Compute full gradient matrix G of shape (2n-1, n)."""
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n-1), n):
        s = max(0, -k)
        e = min(n, n - k)
        # First term: for j in [s, e), G[k,j] += dx * (1 - h[j+k])
        G[k + n - 1, s:e] += dx * (1.0 - h[s+k:e+k])
        # Second term: for j in [s, e), h[j-k] contributes -dx*h[j-k] to G[k,j]
        # j-k ranges from s-k to e-1-k
        js = s  # j range start
        je = e   # j range end
        # The index j-k must be in [s, e)
        # j ranges from max(s, s+k) to min(e, e+k) - 1... 
        # Actually, the second term: for each i in [s,e), the index i+k contributes -dx*h[i] to position j=i+k
        # So G[k, i+k] += -dx * h[i] for i in [s, e)
        # which means for j = i+k, G[k, j] += -dx * h[j-k] where j-k in [s, e)
        # So j ranges from s+k to e-1+k, and we need j in [0, n)
        j_start = max(0, s + k)
        j_end = min(n, e + k)
        if j_start < j_end:
            G[k + n - 1, j_start:j_end] += -dx * h[j_start - k:j_end - k]
    return G

def lp_step(h, n, rad, mu, deadline):
    """Perform one LP-based step from h."""
    C_vals = compute_all_C(h, n)
    G = compute_gradient_matrix(h, n)
    cmax = np.max(C_vals)
    
    m = 2 * n - 1
    # Variables: d[0..n-1], p[0..n-1], t
    # Total: 2n + 1 variables
    
    # Objective: minimize t + (mu/n) * sum(p)
    c_obj = np.zeros(2 * n + 1)
    c_obj[n:2*n] = mu / n
    c_obj[2*n] = 1.0  # coefficient of t
    
    # Inequality constraints: C_k + G[k,:]·d - t ≤ 0 for all k
    # i.e., G[k,:]·d - t ≤ -C[k]
    A_ub = np.zeros((m, 2 * n + 1))
    b_ub = np.zeros(m)
    for k in range(m):
        A_ub[k, :n] = G[k, :]
        A_ub[k, 2*n] = -1.0
        b_ub[k] = -C_vals[k]
    
    # Absolute value constraints: d_i - p_i ≤ 0 and -d_i - p_i ≤ 0
    abs_rows = 2 * n
    A_abs = np.zeros((abs_rows, 2 * n + 1))
    b_abs = np.zeros(abs_rows)
    for i in range(n):
        A_abs[2*i, i] = 1.0
        A_abs[2*i, n+i] = -1.0
        A_abs[2*i+1, i] = -1.0
        A_abs[2*i+1, n+i] = -1.0
    
    A_full = np.vstack([A_ub, A_abs])
    b_full = np.concatenate([b_ub, b_abs])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        d_lo = max(-rad, -h[i] - 1e-10)
        d_hi = min(rad, 1.0 - h[i] + 1e-10)
        bounds.append((d_lo, d_hi))
    for i in range(n):
        bounds.append((0.0, 2 * rad + 1e-10))
    bounds.append((0.0, 2.0))  # t bound
    
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0 or not np.all(np.isfinite(res.x)):
            return None, cmax
        d = res.x[:n]
        # Correct sum
        d = d - np.mean(d)
        return d, cmax
    except Exception:
        return None, cmax
```

```python
def compute_C_and_G(h, n):
    dx = 2.0 / n
    C = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n-1), n):
        idx = k + n - 1
        s = max(0, -k)
        e = min(n, n - k)
        # C_k = dx * sum_{i=s}^{e-1} h[i] * (1-h[i+k])
        C[idx] = dx * np.dot(h[s:e], 1.0 - h[s+k:e+k])
        # Gradient
        # dC/dh[j] for j in [s,e): dx * (1 - h[j+k])
        G[idx, s:e] += dx * (1.0 - h[s+k:e+k])
        # dC/dh[j] for j where j-k in [s,e): dx * (-h[j-k])
        # j = i+k for i in [s,e), so j in [s+k, e+k)
        j_lo = max(0, s + k)
        j_hi = min(n, e + k)
        if j_lo < j_hi:
            G[idx, j_lo:j_hi] += -dx * h[j_lo - k:j_hi - k]
    return C, G
```

- j_lo = max(0, 0) = 0, j_hi = min(n, n) = n

```python
def lp_step(h, n, rad, mu, deadline):
    C, G = compute_C_and_G(h, n)
    cmax = np.max(C)
    m = 2 * n - 1
    # Variables: [d_0,...,d_{n-1}, p_0,...,p_{n-1}, t]
    nv = 2 * n + 1
    
    # Objective: min t + (mu/n) * sum(p)
    c_obj = np.zeros(nv)
    c_obj[n:2*n] = mu / n
    c_obj[2*n] = 1.0
    
    # Inequality constraints: G[k,:]·d - t <= -C[k]
    # Plus: d_i - p_i <= 0, -d_i - p_i <= 0
    n_ineq = m + 2 * n
    A_ub = np.zeros((n_ineq, nv))
    b_ub = np.zeros(n_ineq)
    
    # Linearization constraints
    A_ub[:m, :n] = G
    A_ub[:m, 2*n] = -1.0
    b_ub[:m] = -C
    
    # Absolute value constraints
    for i in range(n):
        A_ub[m + 2*i, i] = 1.0       # d_i - p_i <= 0
        A_ub[m + 2*i, n + i] = -1.0
        A_ub[m + 2*i + 1, i] = -1.0  # -d_i - p_i <= 0
        A_ub[m + 2*i + 1, n + i] = -1.0
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        d_lo = max(-rad, -h[i] - 1e-10)
        d_hi = min(rad, 1.0 - h[i] + 1e-10)
        bounds.append((d_lo, d_hi))
    for i in range(n):
        bounds.append((0.0, 2.0 * rad + 1e-10))
    bounds.append((0.0, 2.0))  # t
    
    if time.time() > deadline:
        return None, cmax
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True})
        if res.status != 0:
            return None, cmax
        d = res.x[:n].copy()
        d -= np.mean(d)  # fix any tiny equality violation
        return d, cmax
    except Exception:
        return None, cmax
```
