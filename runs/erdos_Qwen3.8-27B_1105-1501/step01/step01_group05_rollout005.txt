    xs = np.linspace(L, U, n_samples)
    for x in xs:
        h[i] = x
        h[j] = S - x
        s = objective(h)
        if s < best_score:
            best_score = s
            best_x = x
            best_j = S - x
    # Fine refinement
    if U - L > 1e-8:
        span = (U - L) / max(n_samples - 1, 1)
        lo_f = max(L, best_x - 2 * span)
        hi_f = min(U, best_x + 2 * span)
        if hi_f - lo_f > 1e-10:
            fine_xs = np.linspace(lo_f, hi_f, 9)
            for x in fine_xs:
                h[i] = x
                h[j] = S - x
                s = objective(h)
                if s < best_score:
                    best_score = s
                    best_x = x
                    best_j = S - x
    h[i] = best_x
    h[j] = best_j
    return best_score

import numpy as np
import time


def objective(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair(v, n):
    h = np.asarray(v, dtype=float).ravel()
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - np.sum(h)
    for _ in range(n * 3):
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta < 1e-15:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta < 1e-15:
                break
            h[idx] -= delta
    return h


def interpolate_to(h_src, n_target, offset=0):
    m = len(h_src)
    if m < 2:
        return repair(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) / n_target
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, h_src, left=h_src[0], right=h_src[-1])
    h = h[np.argsort(order)]
    return repair(h, n_target)


def pair_line_min_inplace(h, i, j, n_samples=15):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-10:
        return objective(h)
    cur_score = objective(h)
    best_score = cur_score
    best_x = h[i]
    xs = np.linspace(L, U, n_samples)
    for x in xs:
        h[i] = x
        h[j] = S - x
        s = objective(h)
        if s < best_score:
            best_score = s
            best_x = x
    if U - L > 1e-8:
        span = (U - L) / max(n_samples - 1, 1)
        lo_f = max(L, best_x - 2.5 * span)
        hi_f = min(U, best_x + 2.5 * span)
        if hi_f - lo_f > 1e-10:
            fine_xs = np.linspace(lo_f, hi_f, 11)
            for x in fine_xs:
                h[i] = x
                h[j] = S - x
                s = objective(h)
                if s < best_score:
                    best_score = s
                    best_x = x
    h[i] = best_x
    h[j] = S - best_x
    return best_score


def local_pair_descent(h0, deadline, n_passes=8, n_samples=15, seed=0):
    n = len(h0)
    h = h0.copy()
    score = objective(h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    npairs = len(pairs)
    for p in range(n_passes):
        if time.perf_counter() > deadline:
            break
        offset = (p * 7919) % npairs
        improved = False
        for pidx in range(npairs):
            if time.perf_counter() > deadline:
                break
            k = (pidx + offset) % npairs
            i, j = pairs[k]
            si, sj = h[i], h[j]
            s_new = pair_line_min_inplace(h, i, j, n_samples)
            if s_new < score - 1e-13:
                score = s_new
                improved = True
            else:
                h[i] = si
                h[j] = sj
        if not improved:
            break
    return h, score


def block_mutation(h, rng, min_bs=3, max_bs=8):
    n = len(h)
    bs = int(rng.integers(min_bs, min(max_bs + 1, n)))
    if bs < 2:
        return None
    indices = rng.choice(n, size=bs, replace=False)
    d = rng.normal(0, 1, bs)
    d -= d.mean()
    dn = np.max(np.abs(d))
    if dn < 1e-12:
        return None
    d /= dn
    lam_max = np.inf
    for k in range(bs):
        if d[k] > 0:
            room = (1.0 - h[indices[k]]) / d[k]
            lam_max = min(lam_max, room)
        elif d[k] < 0:
            room = h[indices[k]] / (-d[k])
            lam_max = min(lam_max, room)
    if lam_max < 1e-8:
        return None
    h_new = h.copy()
    for k in range(bs):
        h_new[indices[k]] += lam_max * d[k]
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair(h_new, n)


def gen_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 1e-10:
        h[idx[m_full]] = frac
    return repair(h, n)


def gen_smooth(n, rng):
    t = np.arange(n) / n
    freq = rng.uniform(1.0, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.3, 0.48)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.02, n)
    return repair(h, n)


def gen_blocks(n, rng):
    n_b = int(rng.integers(3, min(9, n)))
    if n_b >= n:
        n_b = max(2, n - 1)
    bnd = sorted(rng.choice(range(1, n), size=n_b - 1, replace=False).tolist())
    bnd = [0] + bnd + [n]
    h = np.zeros(n)
    for b in range(n_b):
        h[bnd[b]:bnd[b + 1]] = rng.uniform(0.05, 0.95)
    return repair(h, n)
