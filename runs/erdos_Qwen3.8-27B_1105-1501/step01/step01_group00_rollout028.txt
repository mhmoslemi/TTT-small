    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)

    if abs(residual) > 1e-13:
        if residual > 0:

        else:

            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:

                h[idx] -= sub
                residual += sub

    target = n / 2.0

    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def pg_step(h, beta, alpha_max=0.8, max_ls=18):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = fast_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.04 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        h = np.asarray(initial_h_values, dtype=float).ravel()
        h = make_feasible(h)
    except (NameError, AttributeError):
        n = 60
        x = np.arange(n) / n
        h = 0.5 + 0.2 * np.sin(np.pi * x) + 0.08 * np.sin(2 * np.pi * x)
        h = make_feasible(h)

    best_h = h.copy()
    best_score = official_score(h)

    def time_left():
        return deadline - time.time()

    # Phase 1: Projected gradient with adaptive beta
    h_cur = h.copy()
    beta = 20.0
    stall = 0
    while time_left() > 250:
        h_new, sc = pg_step(h_cur, beta, alpha_max=0.8, max_ls=16)
        if sc < best_score - 1e-12:
            best_h = h_new.copy()
            best_score = sc
            stall = 0
            h_cur = h_new
        elif sc < official_score(h_cur) - 1e-12:
            stall = 0
            h_cur = h_new
        else:
            stall += 1
            h_cur = h_new
        beta = min(beta * 1.04, 400.0)
        if stall > 100:
            h_cur = make_feasible(best_h + rng.normal(0, 0.03, len(best_h)))
            beta = 15.0
            stall = 0

    # Phase 2: Active set LP polishing
    h_cur = best_h.copy()
    trust = 0.25
    for eps in [0.01, 0.005, 0.002, 0.001]:
        if time_left() < 80:
            break
        for _ in range(40):
            if time_left() < 80:
                break
            h_new, sc = active_set_lp(h_cur, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                    h_cur = h_new
                    trust = min(trust * 1.1, 0.4)
                else:
                    h_cur = h_new
                    trust = max(trust * 0.8, 0.03)

    # Phase 3: Fine projected gradient with high beta
    h_cur = best_h.copy()
    beta = 100.0
    while time_left() > 50:
        h_new, sc = pg_step(h_cur, beta, alpha_max=0.4, max_ls=14)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
        h_cur = h_new
        beta = min(beta * 1.015, 500.0)

    # Phase 4: Interpolation to different n and re-optimize
    if time_left() > 60:
        n_base = len(best_h)
        for n_try in [max(30, n_base - 10), max(30, n_base - 5), min(80, n_base + 5)]:
            if time_left() < 40:
                break
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int.copy()
                best_score = sc_int
            h_cur = h_int.copy()
            beta = 50.0
            for _ in range(200):
                if time_left() < 40:
                    break
                h_new, sc = pg_step(h_cur, beta, alpha_max=0.5, max_ls=12)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
                beta = min(beta * 1.02, 200.0)

    # Phase 5: Final LP polish
    if time_left() > 15:
        h_cur = best_h.copy()
        for eps in [0.002, 0.001, 0.0005]:
            if time_left() < 15:
                break
            for _ in range(20):
                if time_left() < 15:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.08)
                if h_new is not None:
                    if sc < best_score - 1e-13:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    # Phase 6: Final fine PG
    if time_left() > 10:
        h_cur = best_h.copy()
        beta = 200.0
        for _ in range(500):
            if time_left() < 10:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.2, max_ls=10)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
