    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def compute_pair_slopes(h, i, j):
    n = len(h)
    n_m = 2 * n - 1
    nm1 = n - 1
    
    def pos_slope(p):
        sl = np.zeros(n_m)
        # Term 1: (1-h[p+s]) valid for m in [nm1-p, nm1+p]
        m1_lo = max(0, nm1 - p)
        m1_hi = min(2*n-2, nm1 + p)
        if m1_lo <= m1_hi:
            s_vals = np.arange(m1_lo, m1_hi + 1) - nm1
            sl[m1_lo:m1_hi+1] = 1.0 - h[p + s_vals]
        # Term 2: -h[p-s] valid for m in [p, p+nm1]
        m2_lo = p
        m2_hi = min(2*n-2, p + nm1)
        if m2_lo <= m2_hi:
            s_vals = np.arange(m2_lo, m2_hi + 1) - nm1
            sl[m2_lo:m2_hi+1] -= h[p - s_vals]
        return sl
    
    return pos_slope(i) - pos_slope(j)

def find_optimal_delta(corr, slopes, lo, hi, K=30):
    """Find delta in [lo,hi] minimizing max(corr + delta*slopes)."""
    if hi - lo < 1e-12:
        return 0.0, np.max(corr)
    
    n_m = len(corr)
    K = min(K, n_m)
    top = np.argsort(corr)[-K:]
    
    candidates = [lo, hi]
    for a in range(K):
        for b in range(a+1, K):
            ia, ib = top[a], top[b]
            ds = slopes[ia] - slopes[ib]
            if abs(ds) < 1e-12:
                continue
            d = (corr[ib] - corr[ia]) / ds
            if lo <= d <= hi:
                candidates.append(d)
    
    best_d = 0.0
    best_val = np.max(corr)
    for d in candidates:
        val = np.max(corr + d * slopes)
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass
    
    if h0 is None or len(h0) == 0:
        n_init = 70
        h0 = np.zeros(n_init)
        # Create a structured initial pattern
        h0[::2] = 1.0
        h0[-1] = 0.0  # ensure sum = n/2
        # Actually for n=70, h0[::2] gives 35 ones, which is n/2. Good.
    
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_corr = compute_corr(best_h)
    
    # Phase 1: Pair line search with focus on hard pairs (~35% budget)
    ph1_end = t0 + budget_s * 0.35
    h = best_h.copy()
    corr = compute_corr(h)
    
    # Pre-compute which pairs are "interesting"
    # A pair (i,j) is interesting if moving mass from i to j (or vice versa) 
    # could reduce the max correlation.
    
    num_sweeps = 0
    stagnation_count = 0
    
    while time.time() < ph1_end:
        # Pick a random pair
        if rng.rand() < 0.7:
            # Focus on "extreme" pairs: high h at i, low h at j
            i = rng.choice(np.argsort(h)[-n//4:])  # top 25% highest
            j = rng.choice(np.argsort(h)[:-n//4-1])  # bottom 25% lowest
            if h[i] <= h[j]:
                i, j = j, i
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        
        slopes = compute_pair_slopes(h, i, j)
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        
        best_d, best_val = find_optimal_delta(corr, slopes, lo, hi)
        
        if best_val < np.max(corr) - 1e-10:
            h[i] += best_d
            h[j] -= best_d
            corr = corr + best_d * slopes
            if np.max(corr) < best_c5:
                best_c5 = np.max(corr) * (2.0/n)
                best_h = h.copy()
                best_corr = corr.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        num_sweeps += 1
        if num_sweeps % 500 == 0:

            corr = compute_corr(h)
    
    # Phase 2: Simulated annealing with pair moves (~40% budget)
    ph2_end = t0 + budget_s * 0.75

    min_temp = 0.0001

        j = rng.randint(n)
        if i == j:

        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-6:

            if max_d < 1e-6:
