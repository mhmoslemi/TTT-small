    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection to find shift that makes sum(clip(z+shift, 0, 1)) = target
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_grad_all(h):
    """Compute gradient of each full[k] w.r.t. h. Returns array of shape (2n-1, n)."""
    n = len(h)
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            # First factor: h[p-s] for p = s, ..., n-1
            p_range = range(s, n)
            for p in p_range:
                grad[k, p] += (1.0 - h[p-s])
            # Second factor: -h[p+s] for p = 0, ..., n-1-s
            p_range = range(0, n-s)
            for p in p_range:
                grad[k, p] -= h[p+s]
        else:
            # s < 0, r = -s
            # First factor: h[p-s] = h[p+r] for p = 0, ..., n-1-r = n-1+s
            p_range = range(0, n+s)  # n+s = n-|s|
            for p in p_range:
                grad[k, p] += (1.0 - h[p-s])
            # Second factor: -h[p+s] for p = -s, ..., n-1
            p_range = range(-s, n)
            for p in p_range:
                grad[k, p] -= h[p+s]
    return grad

def solve_lp_direction(h0, h0_full, active_idx):
    """Solve LP: min t s.t. t >= L_k(h) for k in active, sum(h)=n/2, 0<=h<=1
    Returns proposed h (or None if LP fails)."""
    n = len(h0)
    grad = compute_grad_all(h0)
    
    # Linearize: L_k(h) = h0_full[k] + grad[k] @ (h - h0)
    # = h0_full[k] - grad[k] @ h0 + grad[k] @ h
    # So constraint: t - grad[k] @ h >= h0_full[k] - grad[k] @ h0
    # i.e., -t + grad[k] @ h <= h0_full[k] - grad[k] @ h0... 
    # Wait: t >= L_k(h) means t - L_k(h) >= 0
    # t - h0_full[k] + grad[k]@h0 - grad[k]@h >= 0
    # -t + grad[k]@h <= grad[k]@h0 - h0_full[k]
    
    # LP: min t
    # s.t. for each active k: -t + grad[k] @ h <= grad[k] @ h0 - h0_full[k]
    #      sum(h) = n/2
    #      0 <= h <= 1
    
    m = len(active_idx)
    # c vector: [0, ..., 0, 1] (t is last variable? or first?)
    # Let's use variable order: [h_0, ..., h_{n-1}, t]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: -t + grad[k] @ h <= grad[k]@h0 - h0_full[k]
    A_ub = np.zeros((m, n+1))
    b_ub = np.zeros(m)
    for i, k in enumerate(active_idx):
        A_ub[i, :n] = grad[k]
        A_ub[i, n] = -1.0
        b_ub[i] = grad[k] @ h0 - h0_full[k]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds: 0 <= h_i <= 1, t unbounded below (or t >= 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def do_lp_iteration(h, max_iter=50, active_frac=0.3):
    """DCA-like LP iteration."""
    n = len(h)
    h = h.copy()
    best_score = score_exact(h)
    
    for it in range(max_iter):
        full = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(full)
        
        # Select active lags
        threshold = max_val * (1 - active_frac)  # or max_val - some_eps
        # Actually, let's use a fraction of the range
        active_idx = np.where(full >= threshold)[0]
        
        # Solve LP
        h_new = solve_lp_direction(h, full, active_idx)
        if h_new is None:
            break
        
        # Line search
        improved = False
        for alpha in [1.0, 0.9, 0.75, 0.5, 0.35, 0.2, 0.1]:
            h_try = make_feasible(h + alpha * (h_new - h))
            s_try = score_exact(h_try)
            if s_try < best_score - 1e-14:
                h = h_try
                best_score = s_try
                improved = True
                break
        
        if not improved:
            # Try with more active constraints
            active_idx = np.where(full >= max_val * 0.5)[0]
            h_new = solve_lp_direction(h, full, active_idx)
            if h_new is not None:
                for alpha in [0.5, 0.3, 0.1]:
                    h_try = make_feasible(h + alpha * (h_new - h))
                    s_try = score_exact(h_try)
                    if s_try < best_score - 1e-14:
                        h = h_try
                        best_score = s_try
                        improved = True
                        break
            if not improved:
                break
    
    return h, best_score

    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)

def compute_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag_k(h, k):
    """Compute gradient of full[k] w.r.t. h."""
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        for p in range(s, n):
            g[p] += (1.0 - h[p - s])
        for p in range(0, n - s):
            g[p] -= h[p + s]
    else:
        for p in range(0, n + s):
            g[p] += (1.0 - h[p - s])
        for p in range(-s, n):
            g[p] -= h[p + s]
    return g

def grad_all(h):
    """Compute gradient of all lags. Returns (2n-1, n) array."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        G[k] = grad_lag_k(h, k)
    return G

def solve_lp(h0, full0, active_idx):
    """Solve LP for direction. Returns proposed h or None."""
    n = len(h0)
    m = len(active_idx)
    
    # Variable: [h_0, ..., h_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    G = grad_all(h0)
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, k in enumerate(active_idx):
        A_ub[i, :n] = G[k]
        A_ub[i, n] = -1.0
        b_ub[i] = G[k] @ h0 - full0[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

        if res.success:

    except Exception:
        pass
    return None

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    G = grad_all(x) * dx
    grad = G.T @ weights
    return grad

def smooth_opt(h0, beta, maxiter=100):
    n = len(h0)
    try:
        res = minimize(
            smooth_obj, h0, args=(beta,),
            jac=smooth_grad,
            method='SLSQP',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

            # a_k = grad of full[k] w.r.t. h[m]
            a = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                s_val = k - (n - 1)
                if s_val >= 0:
                    if s_val <= m <= n - 1:
                        a[k] = (1.0 - h[m - s_val])
                    if 0 <= m <= n - 1 - s_val:
                        a[k] -= h[m + s_val]
                else:
                    if 0 <= m <= n - 1 + s_val:
                        a[k] = (1.0 - h[m - s_val])
                    if -s_val <= m <= n - 1:
                        a[k] -= h[m + s_val]
            
            b = full - a * h[m]
            
            # Minimize max_k (a[k]*x + b[k]) for x in [0,1]
            # Use bisection on subgradient
            def obj_1d(x):
                return np.max(a * x + b)
            
            # Find minimum of convex function (max of lines)
            # Subgradient at x is in [min_{k in active} a[k], max_{k in active} a[k]]
            # where active = argmax_k (a[k]*x + b[k])
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = a * mid + b
                active = np.where(vals == np.max(vals))[0]
                subgrad_min = np.min(a[active])
                subgrad_max = np.max(a[active])
                if subgrad_min > 0:
                    hi = mid  # function increasing, go left
                elif subgrad_max < 0:
                    lo = mid  # function decreasing, go right
                else:
                    break  # 0 is in subgradient, found min
            x_opt = (lo + hi) / 2.0
            
            # Apply and check
            h[m] = x_opt
            h_try = make_feasible(h)
            s_try = score_exact(h_try)
            if s_try < best_score - 1e-14:
                h = h_try
                best_score = s_try
                improved = True
            else:
                h[m] = x_opt
                # Revert - we'll reproject anyway, so just leave it
                h = make_feasible(h.copy())
                # Actually this is wasteful. Let me just track the original h[m]
        
        if not improved:
            break
    
    return h, best_score
