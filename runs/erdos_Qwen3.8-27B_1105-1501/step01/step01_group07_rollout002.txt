def scores_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient(h, T, k_top=0):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    if k_top > 0 and k_top < len(scores):
        idx = np.argpartition(scores, -k_top)[-k_top:]
        w = np.zeros(len(scores))
        active_scores = scores[idx]
        ex = np.exp((active_scores - base) / T)
        S = float(np.sum(ex))
        if S < 1e-300:
            w[idx] = 1.0 / k_top
        else:
            w[idx] = ex / S
    else:
        ex = np.exp((scores - base) / T)
        S = float(np.sum(ex))
        if S < 1e-300:
            mask = scores >= base - 1e-10
            w = np.where(mask, 1.0, 0.0)
            w /= w.sum()
        else:
            w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        if wp > 0 or wn > 0:
            g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
            g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

def momentum_descent(h, deadline, T_start, T_end, beta, lr_init, rng, k_top_init=7):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    lr = lr_init
    vel = np.zeros(n)
    k_top = k_top_init
    stall = 0
    last_improve = time.time()
    while time.time() < deadline:
        w, g = compute_gradient(cur, T, k_top)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        direction = -g / gn
        vel = beta * vel + (1.0 - beta) * direction
        improved = False
        for s in (lr, lr * 0.5, lr * 0.25, lr * 0.1):
            cand = make_feasible(cur + s * vel, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                    last_improve = time.time()
                break
        if improved:
            lr = min(lr * 1.03, T * 1.5)
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 30:
                cur = make_feasible(best + rng.normal(0, max(0.003, T * 0.2), n), n)
                cur_c = score_exact(cur)
                vel = np.zeros(n)
                lr = max(T, 0.003)
                stall = 0
                if cur_c > best_c + 0.01:
                    cur = best.copy()
                    cur_c = best_c
        T = max(T_end, T * 0.999)
        k_top = max(3, min(n, k_top + (1 if T < 0.01 else -1)))
    return best, best_c

def pair_search_golden(h, i, j, cur_c, n_pts=20):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(20):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                h[i] = orig_i
                h[j] = orig_j
                return None
            return float(best_s)
    return None

def systematic_pair_refine(h, deadline, max_passes=10):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    for p in range(max_passes):
        if time.time() > deadline:
            break
        any_improved = False
        step = max(1, n // 15)
        for i in range(0, n, step):
            if time.time() > deadline:
                break
            for j in range(i + 1, n, step):
                if time.time() > deadline:
                    break
                res = pair_search_golden(cur, i, j, cur_c, n_pts=15)
                if res is not None:
                    cur_c = res
                    any_improved = True
                    if cur_c < best_c:
                        best = cur.copy()
                        best_c = cur_c
        if not any_improved:
            break
    return best, best_c

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new <= n_old or n_new < 14:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def coord_search_golden(h, i, cur_c, n_pts=20):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        lo2 = max(-orig, best_a - (1.0 - 2.0 * orig) / n_pts * 1.5)
        hi2 = min(1.0 - orig, best_a + (1.0 - 2.0 * orig) / n_pts * 1.5)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
        a2 = lo2 + gr * (hi2 - lo2)
        for _ in range(18):
            h[i] = orig + a1
            s1 = score_exact(h)
            h[i] = orig + a2
            s2 = score_exact(h)
            if s1 < s2:
                if s1 < best_s:
                    best_s = s1
                    best_a = a1
                hi2 = a2
                a2 = a1
                a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            else:
                if s2 < best_s:
                    best_s = s2
                    best_a = a2
                lo2 = a1
                a1 = a2
                a2 = lo2 + gr * (hi2 - lo2)
        best_a = (lo2 + hi2) * 0.5
        h[i] = orig + best_a
        best_s = score_exact(h)
        if best_s < cur_c:
            return float(best_s)
        h[i] = orig
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: Momentum gradient descent (30% of time)
    p1_end = t0 + budget_s * 0.30
    if time.time() < p1_end:
        h1, c1 = momentum_descent(best_h, p1_end, 0.12, 0.001, 0.85, 0.025, rng, k_top_init=7)
        if c1 < best_c:
            best_h, best_c = h1, c1
        n = len(best_h)

    # Phase 2: Coarse systematic pair refinement (20% of time)
    p2_end = t0 + budget_s * 0.50
    if time.time() < p2_end:
        h2, c2 = systematic_pair_refine(best_h, p2_end, max_passes=8)
        if c2 < best_c:
            best_h, best_c = h2, c2
        n = len(best_h)

    # Phase 3: Fine momentum gradient (15% of time)
    p3_end = t0 + budget_s * 0.65
    if time.time() < p3_end:
        h3, c3 = momentum_descent(best_h, p3_end, 0.006, 0.0001, 0.9, 0.004, rng, k_top_init=5)
        if c3 < best_c:
            best_h, best_c = h3, c3
        n = len(best_h)

    # Phase 4: Targeted pair + coordinate search (20% of time)
    p4_end = t0 + budget_s * 0.85
    cur = best_h.copy()
    cur_c = best_c
    consecutive_fail = 0
    while time.time() < p4_end:
        r = rng.random()
        if r < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search_golden(cur, i, j, cur_c, n_pts=25)
        elif r < 0.75:
            i = int(rng.integers(0, n))
            res = coord_search_golden(cur, i, cur_c, n_pts=25)
        else:
            i = int(rng.integers(0, n - 1))
            res = pair_search_golden(cur, i, i + 1, cur_c, n_pts=30)
        if res is not None:
            cur_c = res
            consecutive_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            for _ in range(3):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    break
                res2 = pair_search_golden(cur, i2, j2, cur_c, n_pts=18)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            consecutive_fail += 1
            if consecutive_fail > 150:
                cur = make_feasible(best_h + rng.normal(0, 0.008, n), n)
                cur_c = score_exact(cur)
                consecutive_fail = 0

    # Phase 5: Multi-resolution refinement (10% of time)
    p5_end = deadline
    n_cur = len(best_h)
    resolutions = []
    for dn in (-10, -5, 5, 10, 15):
        nr = n_cur + dn
        if 20 <= nr <= 150 and nr != n_cur:
            resolutions.append(nr)
    resolutions.sort()
    for nr in resolutions:
        if time.time() > p5_end - 10:
            break
        h_r = resample(best_h, nr)
        if h_r is None:
            continue
        c_r = score_exact(h_r)
        sub_dl = min(p5_end - 3, time.time() + min(60, (p5_end - time.time()) * 0.4))
        if time.time() < sub_dl:
            h_r, c_r = momentum_descent(h_r, sub_dl, 0.01, 0.0003, 0.85, 0.008, rng, k_top_init=5)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
            n = c_r  # dummy, will update n below
            n = len(best_h)

    # Final polish
    if time.time() < deadline - 3:
        h_f, c_f = momentum_descent(best_h, deadline - 3, 0.002, 0.00005, 0.9, 0.001, rng, k_top_init=4)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final feasibility and evaluation
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
