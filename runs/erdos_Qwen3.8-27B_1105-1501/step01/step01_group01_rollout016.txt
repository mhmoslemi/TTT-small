            grads[j, p2] -= h[p1]

    # Phase 1: Generate binary candidates and optimize
    for n in n_options:
        if time.time() > deadline - 60:
            break
        t_phase = time.time()
        
        h_interp = make_feasible(interp_to_n(parent, n))
        
        # Generate binary candidates
        bin_candidates = generate_binary_patterns(n, rng)
        
        # Add thresholded versions of interpolated parent
        for thresh in [0.4, 0.5, 0.6]:
            hb = push_towards_binary(h_interp, thresh)
            bin_candidates.append(hb)
        
        # Evaluate and keep best few
        scored = []
        for bc in bin_candidates:
            bc = make_feasible(bc)
            sc = official_score(bc)
            scored.append((sc, bc))
        scored.sort(key=lambda x: x[0])
        
        # Local search on top candidates
        for sc, bc in scored[:5]:
            if time.time() > deadline - 30:
                break
            hl, sl = binary_local_search(bc, n, max_iters=3000, time_limit=20.0)
            if sl < best_score - 1e-12:
                best_h = hl.copy()
                best_score = sl
        
        # Smooth optimization from best binary
        if time.time() < deadline - 40:
            best_bin = scored[0][1] if scored else h_interp
            for beta in [50, 200, 500]:
                if time.time() > deadline - 15:
                    break
                h_opt = do_smooth_opt(best_bin, beta, maxiter=150)
                s_opt = official_score(h_opt)
                if s_opt < best_score - 1e-12:
                    best_h = h_opt
                    best_score = s_opt
                    best_bin = h_opt.copy()
    
    # Phase 2: Smooth optimization from parent interpolation
    for n in n_options[:4]:
        if time.time() > deadline - 60:
            break
        h_interp = make_feasible(interp_to_n(best_h, n)) if len(best_h) != n else best_h.copy()
        
        starts = [h_interp]
        for sigma in [0.03, 0.08, 0.15]:
            starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))
        
        for h0 in starts:
            if time.time() > deadline - 20:
                break
            for beta in [30, 100, 400, 1200]:
                if time.time() > deadline - 10:
                    break
                h_opt = do_smooth_opt(h0, beta, maxiter=200)
                s_opt = official_score(h_opt)
                if s_opt < best_score - 1e-12:
                    best_h = h_opt
                    best_score = s_opt
    
    # Phase 3: SOCP-based refinement
    for attempt in range(8):
        if time.time() > deadline - 30:
            break
        n_b = len(best_h)
        r = 0.08
        for ri in range(5):
            if time.time() > deadline - 10:
                break
            r = r * (0.7 if ri > 0 else 1.0)
            if r < 0.002:
                break
            h_new = socp_step(best_h, r, max_socs=50)
            s_new = official_score(h_new)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new
                break
    
    # Phase 4: Fine-tuning with very small perturbations
    if time.time() < deadline - 20:
        n_b = len(best_h)
        for _ in range(15):
            if time.time() > deadline - 10:
                break
            h_pert = make_feasible(best_h + 0.01 * rng.standard_normal(n_b))
            h_opt = do_smooth_opt(h_pert, 800, maxiter=100)
            s_opt = official_score(h_opt)
            if s_opt < best_score - 1e-12:
                best_h = h_opt
                best_score = s_opt
    
    # Phase 5: Try different n for final best
    for n_try in n_options:
        if time.time() > deadline - 15:
            break
        if n_try == len(best_h):
            continue
        h_try = make_feasible(interp_to_n(best_h, n_try))
        for beta in [200, 800]:
            if time.time() > deadline - 8:
                break
            h_opt = do_smooth_opt(h_try, beta, maxiter=120)
            s_opt = official_score(h_opt)
            if s_opt < best_score - 1e-12:
                best_h = h_opt
                best_score = s_opt
    
    # Final polish
    if time.time() < deadline - 5:
        h_final = do_smooth_opt(best_h, 2000, maxiter=300)
        s_final = official_score(h_final)
        if s_final < best_score:
            best_h = h_final
            best_score = s_final
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
