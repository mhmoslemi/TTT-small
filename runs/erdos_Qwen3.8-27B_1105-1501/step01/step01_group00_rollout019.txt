from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    # Bisection to find shift that makes sum = target after clipping
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def vec_grads(h):
    """Vectorized computation of all 2n-1 gradients.
    Returns shape (2n-1, n).
    """
    n = len(h)
    grads = np.zeros((2*n-1, n))
    # Lag 0
    grads[n-1] = 1.0 - 2.0*h
    # Lags 1 to n-1
    for t in range(1, n):
        # +t: ∂/∂h[i] = [i>=t](1-h[i-t]) - [i<=n-1-t]h[i+t]
        gp = np.zeros(n)
        gp[t:] += (1.0 - h[:n-t])
        gp[:n-t] -= h[t:]
        grads[n-1+t] = gp

        gm = np.zeros(n)
        gm[:n-t] += (1.0 - h[t:])
        gm[t:] -= h[:n-t]
        grads[n-1-t] = gm

def smooth_max_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # For numerical stability, subtract max (which is 0)
    logsum = np.log(np.sum(np.exp(shifted)))

            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                best_n = len(h_new)
                cur = h_new
            elif h_new is not None:
                sc_cur = official_score(cur)
                if sc < sc_cur:
                    cur = h_new

    # Phase 2: L-BFGS-B with penalty (faster for large n)
    if time.time() < deadline:
        cur = best_h.copy()
        mu_schedule = [1.0, 10.0, 100.0, 1000.0]
        for mu in mu_schedule:
            if time.time() >= deadline:
                break
            for beta in [500, 2000]:
                if time.time() >= deadline:
                    break
                h_new, sc = run_lbfgsb(cur, beta, mu, 500, deadline)
                if h_new is not None and sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc
                    best_n = len(h_new)
                    cur = h_new
                elif h_new is not None:
                    sc_cur = official_score(cur)
                    if sc < sc_cur:
                        cur = h_new

    # Phase 3: Subgradient polishing
    if time.time() < deadline:
        h_sg, sc_sg = subgradient_polish(best_h, 800, 0.005, 1e-7, deadline, reserve)
        if sc_sg < best_score - 1e-14:
            best_h = h_sg.copy()
            best_score = sc_sg
            best_n = len(h_sg)

    # Phase 4: More SLSQP from improved point
    if time.time() < deadline:
        cur = best_h.copy()
        for beta in [1000, 5000, 20000, 100000, 300000]:
            if time.time() >= deadline:
                break
            h_new, sc = run_slsqp(cur, beta, 300, 1e-15, deadline)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                best_n = len(h_new)
                cur = h_new

    # Phase 5: Trust-constr attempt
    if time.time() < deadline:
        for beta in [500, 2000, 10000]:
            if time.time() >= deadline:
                break
            h_new, sc = run_trust(best_h, beta, 200, deadline)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                best_n = len(h_new)

    # Phase 6: Resolution sweep
    if time.time() < deadline:
        n_base = best_n
        n_opts = []
        for dn in range(-15, 21, 5):
            n_try = n_base + dn
            if 35 <= n_try <= 80:
                n_opts.append(n_try)
        n_opts = sorted(set(n_opts))
        for n_opt in n_opts:
            if time.time() >= deadline:
                break
            if n_opt == n_base:
                continue
            remaining = deadline - time.time()
            if remaining < 15:
                break
            h_int = interpolate_to(best_h, n_opt)
            h_new, sc = run_slsqp(h_int, 500, 150, 1e-13, deadline)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                best_n = len(h_new)
            if time.time() < deadline:
                h_new2, sc2 = run_slsqp(
                    h_new if h_new is not None else h_int, 5000, 200, 1e-14, deadline)
                if h_new2 is not None and sc2 < best_score - 1e-14:
                    best_h = h_new2.copy()
                    best_score = sc2
                    best_n = len(h_new2)

    # Phase 7: Random restarts with quick polish
    if time.time() < deadline:
        n = best_n
        remaining = deadline - time.time()
        n_starts = max(2, min(15, int(remaining / 20)))
        for i in range(n_starts):
            if time.time() >= deadline:
                break
            sigma = 0.02 + 0.15 * rng.random()
            noise = rng.normal(0, sigma, size=n)
            h_cand = make_feasible(best_h + noise)
            sc_cand = official_score(h_cand)
            h_new, sc = run_slsqp(h_cand, 2000, 120, 1e-13, deadline)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                best_n = len(h_new)
            elif sc_cand < best_score - 1e-14:
                best_h = h_cand.copy()
                best_score = sc_cand
                best_n = n

    # Phase 8: Final intensive polishing
    if time.time() < deadline:
        for beta in [5000, 20000, 100000, 500000]:
            if time.time() >= deadline:
                break
            h_new, sc = run_slsqp(best_h, beta, 400, 1e-15, deadline)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                best_n = len(h_new)

        # Subgradient polish again with tighter tolerance
        if time.time() < deadline:
            h_sg, sc_sg = subgradient_polish(best_h, 1000, 0.002, 1e-8, deadline, reserve)
            if sc_sg < best_score - 1e-14:
                best_h = h_sg.copy()
                best_score = sc_sg
                best_n = len(h_sg)

        # L-BFGS-B final
        if time.time() < deadline:
            h_new, sc = run_lbfgsb(best_h, 5000, 100.0, 1000, deadline)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                best_n = len(h_new)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-13

    return final_h, final_bound, final_n
