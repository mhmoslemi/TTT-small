        h[j_idx] = old_i
        
        c5 = eval_c5(h)
        
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
            # Update high/low
            if h[i_idx] > 0.5:
                high = np.append(high, i_idx)
                low = np.delete(low, np.where(low == j_idx))
            else:
                low = np.append(low, i_idx)
                high = np.delete(high, np.where(high == j_idx))
            if h[j_idx] > 0.5:
                high = np.append(high, j_idx)
                low = np.delete(low, np.where(low == i_idx))
            else:
                low = np.append(low, j_idx)
                high = np.delete(high, np.where(high == j_idx))
        else:
            h[i_idx] = old_i
            h[j_idx] = old_j
            no_improve += 1
        
        total_swaps += 1
        if no_improve > max_no_improve:
            break
    
    return best_h, best_c5

def simulated_annealing(h, rng, time_limit, t0, T0=0.5, cooling=0.9999):
    """Simulated annealing with swap moves."""
    n = len(h)
    h = h.copy()
    best_c5 = eval_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    T = T0
    
    high = np.where(h > 0.5)[0].tolist()
    low = np.where(h < 0.5)[0].tolist()
    
    if not high or not low:
        return best_h, best_c5
    
    iterations = 0
    while time.time() - t0 < time_limit:
        iterations += 1
        if iterations % 5000 == 0:
            T *= cooling
            high = np.where(h > 0.5)[0].tolist()
            low = np.where(h < 0.5)[0].tolist()
            if not high or not low:
                break
        
        i = high[rng.integers(len(high))]
        j = low[rng.integers(len(low))]
        if i == j:
            continue
        
        old_i, old_j = h[i], h[j]
        h[i] = old_j
        h[j] = old_i
        
        new_c5 = eval_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = old_i
            h[j] = old_j
    
    return best_h, best_c5

def continuous_polish(h, n, budget_s, t0):
    """Polish a near-binary solution using projected gradient on non-binary entries."""
    target = n / 2.0
    h = project_to_sum(h, target)
    
    best_c5 = eval_c5(h)
    best_h = h.copy()
    
    # Use SLSQP with smooth max approximation
    def smooth_obj(x, beta):
        corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        cmax = float(np.max(corr))
        shifted = beta * (corr - cmax)
        shifted = np.clip(shifted, -500, 500)
        return cmax + float(np.log(np.sum(np.exp(shifted)))) / beta
    
    def smooth_grad(x, beta):
        corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        cmax = float(np.max(corr))
        shifted = np.clip(beta * (corr - cmax), -500, 500)
        exps = np.exp(shifted)
        w = exps / float(np.sum(exps))
        grad = np.zeros(n)
        for d in range(-(n - 1), n):
            wi = w[d + n - 1] * (2.0 / n)
            if abs(wi) < 1e-15:
                continue
            if d >= 0:
                for i in range(n - d):
                    grad[i] += wi * (1.0 - x[i + d])
                    grad[i + d] -= wi * x[i + d]
            else:
                g = -d
                for i in range(g, n):
                    grad[i] += wi * (1.0 - x[i - g])
                    grad[i - g] -= wi * x[i - g]
        return grad
    
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    
    for beta in [10, 50, 200, 1000, 5000]:
        if time.time() - t0 > t0 + budget_s:
            break
        try:
            res = minimize(smooth_obj, h, jac=lambda x: smooth_grad(x, beta),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-14})
            h_new = project_to_sum(res.x, target)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            break
    
    return best_h, best_c5

def multi_start_optimize(n, rng, time_limit, t0, h_init=None):
    """Run multiple starts of local search/SA for given n."""
    target = n / 2.0
    best_h = None
    best_c5 = float('inf')
    
    # Generate initial solutions
    inits = []
    
    # 1. Random balanced binary
    for _ in range(3):
        inits.append(make_binary_balanced(n, rng))
    
    # 2. Thue-Morse (if n is power of 2)
    if n in [64, 128, 256] or (n & (n - 1) == 0):
        inits.append(make_thue_morse(n))
    
    # 3. Block patterns
    for k in [1, 2, 3, 4]:
        inits.append(make_block_pattern(n, k, rng))
    
    # 4. From initial_h_values if available
    if h_init is not None and len(h_init) > 0:
        h_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_init)), h_init)
        h_interp = project_to_sum(h_interp, target)
        inits.append(h_interp)
    
    # 5. Spread (evenly spaced ones)
    h_spread = np.zeros(n)
    ones_pos = np.arange(0, n, n // max(1, n // 2))[:n // 2]
    if n % 2 != 0:
        ones_pos = np.arange(0, n, 2)[:((n - 1) // 2)]
        h_spread[ones_pos] = 1.0
        h_spread[-1] = 0.5
    else:
        h_spread[ones_pos] = 1.0
    inits.append(h_spread)
    
    # Distribute time among inits
    time_per_init = time_limit / len(inits)
    
    for idx, h0 in enumerate(inits):
        remaining = time_limit - (time.time() - t0)
        if remaining < 1.0:
            break
        
        h0 = project_to_sum(h0, target)
        
        # Quick local search
        t_sub = min(time_per_init * 0.6, remaining * 0.7)
        h_ls, c5_ls = local_search_binary(h0, rng, t_sub, time.time())
        
        # SA on the result
        remaining2 = time_limit - (time.time() - t0)
        if remaining2 > 2.0:
            t_sa = min(time_per_init * 0.4, remaining2 * 0.5)
            h_sa, c5_sa = simulated_annealing(h_ls, rng, t_sa, time.time())
            if c5_sa < c5_ls:
                h_ls, c5_ls = h_sa, c5_sa
        
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Continuous polish on the best
    remaining3 = time_limit - (time.time() - t0)
    if remaining3 > 5.0 and best_h is not None:
        h_pol, c5_pol = continuous_polish(best_h, n, remaining3 - 2.0, time.time())
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    return best_h, best_c5

def nested_perturb_search(h_best, n, rng, time_limit, t0):
    """Perturb the best solution and re-optimize."""
    target = n / 2.0
    best_h = h_best.copy()
    best_c5 = eval_c5(h_best)
    
    high = np.where(h_best > 0.5)[0]
    low = np.where(h_best < 0.5)[0]
    
    if len(high) == 0 or len(low) == 0:
        return best_h, best_c5
    
    n_pert = len(high) * len(low)
    attempts = 0
    
    while time.time() - t0 < time_limit:
        attempts += 1
        # Random perturbation: swap k random pairs
        k = rng.integers(1, 4)
        h_pert = h_best.copy()
        for _ in range(k):
            i = high[rng.integers(len(high))]
            j = low[rng.integers(len(low))]
            h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
        
        h_pert = project_to_sum(h_pert, target)
        
        # Quick local search from perturbed point
        t_sub = min(2.0, (time_limit - (time.time() - t0)) * 0.3)
        h_ls, c5_ls = local_search_binary(h_pert, rng, t_sub, time.time())
        
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            high = np.where(best_h > 0.5)[0]
            low = np.where(best_h < 0.5)[0]
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get initial values
    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    
    # Candidate n values (prefer sizes where sum is integer for binary solutions)
    n_candidates = [64, 63, 66, 65, 68, 70, 72, 80, 96, 128]
    
    # Allocate time: more time to promising sizes
    # Each n gets a share, with the first few getting more
    total_time = budget_s * 0.85  # Keep 15% for final polish
    n_alloc = {}
    weights = [3, 2, 2, 2, 2, 1.5, 1.5, 1.5, 1, 1]
    wsum = sum(weights)
    for i, n in enumerate(n_candidates):
        n_alloc[n] = total_time * weights[i] / wsum
    
    for n in n_candidates:
        remaining = total_time - (time.time() - t0)
        if remaining < 5.0:
            break
        
        alloc = min(n_alloc.get(n, 30), remaining)
        
        h_opt, c5_opt = multi_start_optimize(n, rng, alloc, t0, h_init)
        
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
    
    # Final: nested perturbation search on the best
    remaining = total_time - (time.time() - t0)
    if remaining > 10.0 and best_h is not None:
        h_np, c5_np = nested_perturb_search(best_h, best_n, rng, remaining - 2.0, time.time())
        if c5_np < best_c5:
            best_c5 = c5_np
            best_h = h_np.copy()
    
    # Final continuous polish
    remaining = total_time - (time.time() - t0)
    if remaining > 5.0 and best_h is not None:
        h_pol, c5_pol = continuous_polish(best_h, best_n, remaining - 1.0, time.time())
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Ensure feasibility
    if best_h is not None:
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = eval_c5(best_h)
    else:
        # Fallback
        best_n = 64
        best_h = make_binary_balanced(best_n, rng)
        best_c5 = eval_c5(best_h)
    
    return best_h, best_c5, best_n
