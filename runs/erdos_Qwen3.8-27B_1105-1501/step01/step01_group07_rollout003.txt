def make_feasible(y, n=None):
    """Optimal projection onto {0<=h<=1, sum(h)=n/2}."""
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(z)  # fill smallest first (most room)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:
            order = np.argsort(-z)  # reduce largest first
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-15:
                    break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_grad(h, k, weight=1.0):
    """Return weight * dF_k/dh as a new array."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += weight * (1.0 - h[k:])
            g[k:] -= weight * h[:m]
    else:
        p = -k
        m = n - p
        if m > 0:
            g[p:] += weight * (1.0 - h[:m])
            g[:m] -= weight * h[p:]
    return g

def smooth_max_grad(h, T):
    """Return (value, gradient) of smooth max of F_k * dx."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    base = np.max(vals)
    sh = (vals - base) / T
    ex = np.exp(sh)
    S = np.sum(ex)
    if S < 1e-300:
        val = base * dx
        w = np.zeros(len(vals))
        w[np.argmax(vals)] = 1.0
    else:
        val = (base + T * np.log(S)) * dx
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    for d in range(-(n-1), n):
        wd = w[c + d]
        if wd > 1e-14:
            g += lag_grad(h, d, dx * wd)
    return val, g

def projected_descent(h, deadline, T_start, T_end, T_decay, step0, rng):
    """Projected gradient descent with smooth max."""
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    
    while time.time() < deadline:
        val, g = smooth_max_grad(cur, T)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        direction = -g / gn
        improved = False
        # Backtracking line search
        s = step
        for _ in range(20):

            c_new = score_exact(cand)
            if c_new < cur_c - 1e-15:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c

def slsqp_solve(h0, deadline, max_iter=200):
    """Solve the epigraph NLP using SLSQP."""
    n = len(h0)
    dx = 2.0 / n
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def eq_con(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    def ineq_con(x):
        h = x[:n]
        t = x[n]
        vals = np.correlate(h, 1.0 - h, mode='full')
        return t - vals
    
    def ineq_grad(x):
        h = x[:n]
        J = np.zeros((2*n - 1, n + 1))
        c = n - 1
        for d in range(-(n-1), n):
            J[c + d, :n] = -lag_grad(h, d, 1.0)
            J[c + d, n] = 1.0
        return J
    
    x0 = np.concatenate([h0, [score_exact(h0) / dx + 1e-6]])
    bounds = [(0, 1)] * n + [(0, 100)]
    
    cons = [
        {'type': 'eq', 'fun': eq_con, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_grad},
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.success or res.fun < x0[n]:
            h_sol = make_feasible(res.x[:n], n)
            c_sol = score_exact(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return None, None
