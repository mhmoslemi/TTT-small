            else:
                step *= 0.7
    return h, official_score(h)

def _slsqp_obj(x, active_lags, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    return float(np.max(corr))

def _slsqp_jac(x, active_lags, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(500.0 * (corr - m), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-18:
            continue
        shift = k - (n - 1)
        if shift == 0:
            grad += w * (1.0 - 2.0 * x)
        else:
            for p in range(n):
                g = 0.0
                a = p - shift
                b = p + shift
                if 0 <= a < n:
                    g += 1.0 - x[a]
                if 0 <= b < n:
                    g -= x[b]
                grad[p] += w * g
    return grad

def slsqp_polish(h0, maxiter=200):
    n = len(h0)
    active_lags = None
    bounds = [(0.0, 1.0)] * n

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(
            _slsqp_obj, h0, args=(active_lags, n),
            jac=_slsqp_jac, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0.copy()

def coord_descent(h0, deadline, rng):
    n = len(h0)
    h = h0.copy()
    score = official_score(h)
    improved_any = True
    while improved_any and time.time() < deadline - 5:
        improved_any = False
        for p in range(n):
            if time.time() > deadline - 5:
                break
            for q in rng.choice(n, size=min(5, n), replace=False):
                if q == p:
                    continue
                s = h[p] + h[q]
                best_local = h.copy()
                best_local_s = score
                for frac in np.linspace(0.0, 1.0, 21):
                    val_p = frac * s
                    val_q = s - val_p
                    if val_p < -1e-12 or val_p > 1 + 1e-12 or val_q < -1e-12 or val_q > 1 + 1e-12:
                        continue
                    trial = h.copy()
                    trial[p] = val_p
                    trial[q] = val_q
                    trial = make_feasible(trial)
                    s_trial = official_score(trial)
                    if s_trial < best_local_s - 1e-14:
                        best_local = trial
                        best_local_s = s_trial
                if best_local_s < score - 1e-14:
                    h = best_local
                    score = best_local_s
                    improved_any = True
    return h, official_score(h)

def binary_push(h, power=1.5):
    h_new = np.power(h, power)
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(50) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    n_options = [46, 48, 50, 51, 52, 54, 56, 58]
    parent_n = len(parent)
    if parent_n in n_options:
        n_options.remove(parent_n)
    n_options.insert(0, parent_n)
    n_options = [n for n in n_options if 40 <= n <= 64]

    for n in n_options:
        if time.time() > deadline - 80:
            break

        h_interp = make_feasible(interp_to_n(parent, n))
        cur_score = official_score(h_interp)
        if cur_score < best_score - 1e-12:
            best_h = h_interp.copy()
            best_score = cur_score
            best_n = n

        starts = [h_interp.copy()]
        h_anti = make_feasible(1.0 - h_interp[::-1].copy())
        starts.append(h_anti)
        for sigma in [0.03, 0.08, 0.15]:
            starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))
        starts.append(binary_push(h_interp, 1.3))
        starts.append(binary_push(h_interp, 2.0))

        for h0 in starts:
            if time.time() > deadline - 40:
                break

            h_curr = h0.copy()
            s_curr = official_score(h_curr)

            h_curr, s_curr = pg_optimize(h_curr, deadline)

            if time.time() < deadline - 15:
                h_pol = slsqp_polish(h_curr, maxiter=150)
                s_pol = official_score(h_pol)
                if s_pol < s_curr - 1e-12:
                    h_curr = h_pol
                    s_curr = s_pol

            if s_curr < best_score - 1e-12:
                best_h = h_curr.copy()
                best_score = s_curr
                best_n = n
                # Further refinement on new best
                if time.time() < deadline - 20:
                    h_cd, s_cd = coord_descent(best_h, deadline, rng)
                    if s_cd < best_score - 1e-12:
                        best_h = h_cd
                        best_score = s_cd

    # Final refinement phase
    remaining = deadline - time.time()
    if remaining > 25:
        num_trials = min(20, int(remaining / 15))
        for trial in range(num_trials):
            if time.time() > deadline - 10:
                break
            sigma = 0.015 + 0.02 * trial / max(1, num_trials)
            h_pert = make_feasible(best_h + sigma * rng.standard_normal(best_n))
            h_opt, s_opt = pg_optimize(h_pert, deadline)
            if time.time() < deadline - 8:
                h_opt = slsqp_polish(h_opt, maxiter=100)
                s_opt = official_score(h_opt)
            if s_opt < best_score - 1e-12:
                best_h = h_opt
                best_score = s_opt

    # Coordinate descent on best
    if time.time() < deadline - 8:
        h_cd, s_cd = coord_descent(best_h, deadline - 3, rng)
        if s_cd < best_score - 1e-12:
            best_h = h_cd
            best_score = s_cd

    # Final SLSQP polish
    if time.time() < deadline - 5:
        h_final = slsqp_polish(best_h, maxiter=200)
        s_final = official_score(h_final)
        if s_final < best_score:
            best_h = h_final
            best_score = s_final

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
