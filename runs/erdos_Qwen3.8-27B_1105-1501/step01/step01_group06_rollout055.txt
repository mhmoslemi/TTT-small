def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_scores(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # raw scores, length 2n-1

def project_to_target(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    diff = target - float(np.sum(result))
    if abs(diff) > 1e-10:
        step = 1.0 / n
        if diff > 0:
            for i in range(n):
                if result[i] < 1.0:
                    amt = min(step, 1.0 - result[i], diff)
                    result[i] += amt
                    diff -= amt
                    if abs(diff) < 1e-12:
                        break
        else:
            for i in range(n):
                if result[i] > 0.0:
                    amt = min(step, result[i], -diff)
                    result[i] -= amt
                    diff += amt
                    if abs(diff) < 1e-12:
                        break
    return result

def compute_lag_gradients(h):
    """Compute gradient of each lag score w.r.t. h.
    Returns array of shape (2n-1, n) where row k is gradient of score[k]."""
    n = len(h)
    num_scores = 2 * n - 1
    offset = n - 1
    grads = np.zeros((num_scores, n))
    
    for k in range(num_scores):
        d = k - offset  # lag
        if d >= 0:
            # c_k = sum_{j=0}^{n-1-d} h[j+d] * (1-h[j])
            # grad wrt h[i]:
            #   (1-h[i-d]) if d <= i <= n-1
            #   -h[i+d] if 0 <= i <= n-1-d
            if d == 0:
                grads[k] = 1.0 - 2.0 * h
            else:
                idx1 = np.arange(d, n)  # where (1-h[i-d]) contributes
                grads[k, idx1] += (1.0 - h[idx1 - d])
                idx2 = np.arange(0, n - d)  # where -h[i+d] contributes
                grads[k, idx2] -= h[idx2 + d]
        else:
            g = -d
            # c_k = sum_{j=g}^{n-1} h[j-g] * (1-h[j])
            # grad wrt h[i]:
            #   (1-h[i+g]) if 0 <= i <= n-1-g
            #   -h[i-g] if g <= i <= n-1
            idx1 = np.arange(0, n - g)  # where (1-h[i+g]) contributes
            grads[k, idx1] += (1.0 - h[idx1 + g])
            idx2 = np.arange(g, n)  # where -h[i-g] contributes
            grads[k, idx2] -= h[idx2 - g]
    
    return grads

def lp_trust_region_step(h, radius, mu, k_active=None):
    """Solve LP to find trust region step direction.
    Returns (p, predicted_tau) or None if infeasible."""
    n = len(h)
    scores = get_scores(h)
    grads = compute_lag_gradients(h)
    
    if k_active is not None and k_active < len(scores):
        # Select top k_active lags
        top_idx = np.argsort(scores)[::-1][:k_active]
    else:
        top_idx = np.arange(len(scores))
    
    # LP variables: [p (n), r (1), tau (1)]
    n_vars = n + 2
    # Objective: minimize tau + mu * r
    c_obj = np.zeros(n_vars)
    c_obj[n] = mu       # r coefficient
    c_obj[n + 1] = 1.0  # tau coefficient
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag: grad_l . p - tau <= -score_l
    # p_i - r <= 0
    # -p_i - r <= 0
    num_constraints = len(top_idx) + 2 * n
    
    A_ub = np.zeros((num_constraints, n_vars))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    for idx in top_idx:
        A_ub[row, :n] = grads[idx]
        A_ub[row, n + 1] = -1.0  # -tau
        b_ub[row] = -scores[idx]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n] = -1.0
        b_ub[row] = 0.0
        row += 1
        A_ub[row, i] = -1.0
        A_ub[row, n] = -1.0
        b_ub[row] = 0.0
        row += 1
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i])
        hi_i = min(radius, 1.0 - h[i])
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, radius))  # r
    bounds.append((-1e6, 1e6))    # tau
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='interior-point')
        except Exception:
            return None, None
    
    if res.status != 0:
        return None, None
    
    p = res.x[:n]
    tau = res.x[n + 1]
    return p, tau

def subgradient_direction(h):
    """Fallback: negative weighted gradient of top active scores."""
    n = len(h)
    scores = get_scores(h)
    max_score = np.max(scores)
    # Weights: exponential decay from max
    diffs = max_score - scores
    weights = np.exp(-diffs / (0.1 * max_score + 1e-10))
    weights /= np.sum(weights)
    
    grads = compute_lag_gradients(h)
    # Weighted gradient
    g = weights @ grads  # shape (n,)
    # Descent direction: -g, projected to sum zero
    p = -g
    p -= np.mean(p)
    # Normalize to unit L-infinity
    max_abs = np.max(np.abs(p))
    if max_abs < 1e-12:
        return None
    p /= max_abs
    return p

    h_new = np.interp(x_new, x_old, h_old)

    best_h = h.copy()
    best_c5 = current_c5
    for f in fractions:
        alpha = f * max_alpha
        if alpha < 1e-10:
            continue
        h_new = h + alpha * p
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_target(h_new, n / 2.0)
        c5_new = exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def lp_local_opt(h, deadline, radius0, mu, max_iters=500, shrink=0.6, grow=1.2):
    n = len(h)
    current_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    radius = radius0
    stalled = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        p = lp_step(h, radius, mu)
        if p is None:
            p = subgradient_dir(h)
            if p is None:
                break
            radius_curr = 0.03
            p = p * radius_curr
        else:
            radius_curr = radius
        h_new, c5_new = line_search_exact(h, p, current_c5)
        if c5_new < current_c5 - 1e-11:
            h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            radius = min(radius * grow, 0.4)
            stalled = 0
        else:
            radius *= shrink
            stalled += 1
            if radius < 1e-6 or stalled > 20:
                break
    return best_h, best_c5

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_target(h_new, n_new / 2.0)

def reversed_complement(h):
    return project_to_target(1.0 - h[::-1].copy(), len(h) / 2.0)

def binary_push(h, threshold):
    n = len(h)
    h_b = h.copy()
    h_b[h_b < threshold] = 0.0
    h_b[h_b > 1.0 - threshold] = 1.0
    return project_to_target(h_b, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline_total = t0 + budget_s

    h0 = project_to_target(np.array(initial_h_values, dtype=float), len(initial_h_values) / 2.0)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    best_n = n0

    # Phase 1: Core local optimization
    dl = t0 + budget_s * 0.25
    h_opt, c5_opt = lp_local_opt(h0, dl, radius0=0.12, mu=2.0, max_iters=400)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Reversed complement
    dl = t0 + budget_s * 0.35
    h_rc = reversed_complement(best_h)
    h_opt, c5_opt = lp_local_opt(h_rc, dl, radius0=0.1, mu=3.0, max_iters=200)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        best_n = len(h_opt)

    # Phase 3: Resolution sweep
    dl = t0 + budget_s * 0.65
    n_try_list = []
    for dn in [-10, -8, -6, -4, -2, -1, 1, 2, 4, 6, 8, 10, 12, 16, 20]:
        nc = best_n + dn
        if 18 <= nc <= 160:
            n_try_list.append(nc)
    n_try_list = sorted(set(n_try_list))
    per_try = (budget_s * 0.30) / max(1, len(n_try_list))
    for nc in n_try_list:
        if time.time() > dl:
            break
        h_interp = interpolate_to_n(best_h, nc)
        sub_dl = min(dl, time.time() + per_try)
        h_opt, c5_opt = lp_local_opt(h_interp, sub_dl, radius0=0.08, mu=3.0, max_iters=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = nc

    # Phase 4: Binary push attempts
    dl = t0 + budget_s * 0.80
    for thr in [0.1, 0.15, 0.2, 0.25, 0.3]:
        if time.time() > dl:
            break
        h_bp = binary_push(best_h, thr)
        sub_dl = min(dl, time.time() + budget_s * 0.02)
        h_opt, c5_opt = lp_local_opt(h_bp, sub_dl, radius0=0.06, mu=5.0, max_iters=60)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Random perturbations with LP polish
    dl = t0 + budget_s * 0.93
    base_ps = [0.02, 0.04, 0.06, 0.01, 0.03, 0.05, 0.08, 0.12]
    for i in range(len(base_ps)):
        if time.time() > dl:
            break
        ps = base_ps[i]
        h_pert = best_h + rng.normal(0, ps, best_n)
        h_pert = project_to_target(h_pert, best_n / 2.0)
        sub_dl = min(dl, time.time() + budget_s * 0.015)
        h_opt, c5_opt = lp_local_opt(h_pert, sub_dl, radius0=0.04, mu=5.0, max_iters=50)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Fine resolution around best_n
    if time.time() < dl:
        for dn in [-2, -1, 1, 2]:
            nc = best_n + dn
            if nc < 16:
                continue
            if time.time() > dl:
                break
            h_interp = interpolate_to_n(best_h, nc)
            sub_dl = min(dl, time.time() + 3.0)
            h_opt, c5_opt = lp_local_opt(h_interp, sub_dl, radius0=0.05, mu=4.0, max_iters=40)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = nc

    # Final polish on best
    remaining = deadline_total - time.time()
    if remaining > 5.0:
        h_opt, c5_opt = lp_local_opt(best_h, deadline_total - 3.0, radius0=0.03, mu=5.0, max_iters=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = project_to_target(best_h, best_n / 2.0)
    best_c5 = exact_c5(best_h)
    return best_h, best_c5, best_n
