def make_feasible(z, target=None):

    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)

    hi = 2.0 - np.min(z)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]

            else:

            best_x = old_val
            best_c = np.max(corr0 + old_val * slopes)  # approximation for initial

def coord_opt_1d(corr0, slopes, n):
    """Find x in [0,1] minimizing max_p c_p(x) where c_p is linear or quadratic."""
    # Candidates: 0, 1, and intersections of quadratic with each linear
    candidates = [0.0, 1.0]
    
    # Quadratic: c[n-1] = corr0[n-1] + x - x^2
    # Linear p: c[p] = corr0[p] + x * slopes[p]
    # Intersection: corr0[n-1] + x - x^2 = corr0[p] + x * slopes[p]
    # => x^2 + (slopes[p] - 1)*x + (corr0[p] - corr0[n-1]) = 0
    
    quad0 = corr0[n-1]
    for p in range(2*n - 1):
        if p == n-1:
            continue
        # x^2 + (slopes[p] - 1)*x + (corr0[p] - quad0) = 0
        a = 1.0
        b = slopes[p] - 1.0
        c = corr0[p] - quad0
        disc = b*b - 4*a*c
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            x1 = (-b + sqrt_disc) / (2*a)
            x2 = (-b - sqrt_disc) / (2*a)
            for x in [x1, x2]:
                if 0.0 <= x <= 1.0:
                    candidates.append(x)
    
    # Also add vertices of the upper envelope of linear functions
    # (This is more complex; skip for now, the grid will catch nearby points)

        c[n-1] = quad0 + x * (1.0 - x)
        val = np.max(c)
        if val < best_val:

import numpy as np
from scipy.optimize import minimize
import time

def score_fn(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def cardes_coord_opt(h, i, n):
    """Optimize coordinate i exactly using the 1D structure."""
    old = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    slopes = corr1 - corr0
    quad0 = corr0[n - 1]
    candidates = [0.0, 1.0, old]
    nfull = 2 * n - 1
    for p in range(nfull):
        if p == n - 1:
            continue
        a = 1.0
        b = slopes[p] - 1.0
        c = corr0[p] - quad0
        disc = b * b - 4.0 * a * c
        if disc >= 0:
            sd = np.sqrt(disc)
            x1 = (-b + sd) / (2.0 * a)
            x2 = (-b - sd) / (2.0 * a)
            if 0.0 <= x1 <= 1.0:
                candidates.append(x1)
            if 0.0 <= x2 <= 1.0:
                candidates.append(x2)
    best_x = old
    best_val = np.inf
    for x in candidates:
        c = corr0 + x * slopes
        c[n - 1] = quad0 + x * (1.0 - x)
        val = float(np.max(c))
        if val < best_val:
            best_val = val
            best_x = x
    return best_x

def coord_descent(h, sweeps=30):
    """Coordinate descent with exact 1D subproblem solution."""
    h = h.copy()
    n = len(h)
    cur_score = score_fn(h)
    for sw in range(sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            new_val = cardes_coord_opt(h, i, n)
            if abs(new_val - old_val) > 1e-10:
                h[i] = new_val
                improved = True
        h = make_feasible(h)
        new_score = score_fn(h)
        if new_score < cur_score - 1e-10:
            cur_score = new_score
        elif not improved:
            break
    return h, cur_score

def binary_local_search(h, max_iters=200):
    """Discrete local search on binary patterns (for even n)."""
    n = len(h)
    if n % 2 != 0:
        return h
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    if h_bin.sum() > target:
        order = np.argsort(h_bin - h)
        for idx in order:
            if h_bin.sum() <= target:
                break
            h_bin[idx] = 0.0
    elif h_bin.sum() < target:
        order = np.argsort(h - h_bin)
        for idx in order:
            if h_bin.sum() >= target:
                break
            h_bin[idx] = 1.0
    cur_score = score_fn(h_bin)
    for _ in range(max_iters):
        improved = False
        zeros = np.where(h_bin < 0.5)[0]
        ones = np.where(h_bin > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_gain = 1e-10
        best_pair = None
        n_z = len(zeros)
        n_o = len(ones)
        step = max(1, n_z // 8)
        step_o = max(1, n_o // 8)
        for zi in range(0, n_z, step):
            for oi in range(0, n_o, step_o):
                h_try = h_bin.copy()
                h_try[zeros[zi]] = 1.0
                h_try[ones[oi]] = 0.0
                s = score_fn(h_try)
                if cur_score - s > best_gain:
                    best_gain = cur_score - s
                    best_pair = (zeros[zi], ones[oi])
        if best_pair is not None:
            h_bin[best_pair[0]] = 1.0
            h_bin[best_pair[1]] = 0.0
            cur_score -= best_gain
            improved = True
        if not improved:
            break
    return h_bin

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

def smooth_obj(x, beta, n, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500.0, 0.0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_grad(x, beta, n, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            j_lo = shift
            j_hi = n
        else:
            j_lo = 0
            j_hi = n + shift
        for j in range(j_lo, j_hi):
            grad[j] += w * (1.0 - x[j - shift] if shift >= 0 else x[j - shift])
    return grad

def slsqp_smooth(h0, beta, maxiter=200, n=None, dx=None):
    if n is None:
        n = len(h0)
    if dx is None:
        dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj(x, beta, n, dx)

    def grad(x):
        return smooth_grad(x, beta, n, dx)

    def eqfun(x):
        return np.sum(x) - n / 2.0

    def eqjac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eqfun, 'jac': eqjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def min_max_obj(x, n, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    return np.max(corr)

def do_direct_minimax(h0, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    t0 = score_fn(h0) + 1e-6
    if t0 > 0.99:
        t0 = 0.99
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def obj(z):
        return z[-1]

    def obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def ineq_fun(z):
        h = z[:-1]
        t = z[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr

    def ineq_jac(z):
        h = z[:-1]
        nfull = 2 * n - 1
        jac = np.zeros((nfull, n + 1))
        jac[:, -1] = 1.0
        for k in range(nfull):
            shift = k - (n - 1)
            if shift >= 0:
                j_lo = shift
                j_hi = n
            else:
                j_lo = 0
                j_hi = n + shift
            for j in range(j_lo, j_hi):
                if shift >= 0:
                    p = j
                    q = j - shift
                else:
                    p = j - shift
                    q = j
                jac[k, p] += -dx * (1.0 - h[q])
                jac[k, q] += -dx * (-h[p])
        return jac

    def eqfun(z):
        return np.sum(z[:-1]) - n / 2.0

    def eqjac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(obj, z0, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
                                    {'type': 'eq', 'fun': eqfun, 'jac': eqjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0
