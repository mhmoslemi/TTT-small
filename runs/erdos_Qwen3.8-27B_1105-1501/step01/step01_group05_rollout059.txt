def official_score(h):
    n = len(h)

def project_to_target(v, n):

        else:

    else:
        a = -k
        for j in range(a, n):
            g[j] += (1.0 - h[j - a])
        for j in range(0, n - a):
            g[j] += (-h[j + a])
    return dx * g


def compute_grad_batch(h, dx, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, k in enumerate(lags):
        G[idx] = compute_grad(h, k, dx)
    return G


def subgradient_momentum(h, n, max_iters=120, time_limit=30.0, seed=42, momentum=0.7):
    rng = np.random.default_rng(seed)
    dx = 2.0 / n
    h = project_to_target(h, n).copy()
    score = official_score(h)
    start = time.perf_counter()
    alpha = 0.15
    v = np.zeros(n)
    no_improve = 0
    best_local = score
    best_local_h = h.copy()

    all_lags = np.arange(-(n - 1), n)

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        if no_improve > 30:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = corr * dx
        max_val = scaled.max()

        tol = max(1e-10, 0.015 * max_val)
        active = np.where(scaled >= max_val - tol)[0]
        if len(active) > 40:
            nearest = np.argsort(scaled[active])[::-1][:40]
            active = active[nearest]

        grad = np.zeros(n)
        for r in active:
            k = int(all_lags[r])
            grad += compute_grad(h, k, dx)
        grad /= len(active)
        grad -= np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad
        direction = -v
        norm = np.max(np.abs(direction))
        if norm < 1e-14:
            break
        direction = direction / norm

        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05]:
            h_new = project_to_target(h + a * direction, n)
            s_new = official_score(h_new)
            if s_new < score - 1e-13:
                h = h_new
                score = s_new
                no_improve = 0
                improved = True
                alpha = min(0.4, alpha * 1.15)
                if score < best_local:
                    best_local = score
                    best_local_h = h.copy()
                break

        if not improved:
            no_improve += 1
            alpha = max(0.01, alpha * 0.6)
            if no_improve > 10:
                noise = rng.normal(0, 0.02, n)
                noise -= np.mean(noise)
                h = project_to_target(h + noise, n)
                score = official_score(h)
                no_improve = 0
                v = np.zeros(n)
                if score > best_local + 0.005:
                    h = best_local_h.copy()
                    score = best_local

    return h, official_score(h)


def coordinate_swap_descent(h, n, max_sweeps=8, time_limit=25.0, seed=42):
    rng = np.random.default_rng(seed)
    dx = 2.0 / n
    h = project_to_target(h, n).copy()
    score = official_score(h)
    start = time.perf_counter()
    all_lags = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.perf_counter() - start > time_limit:
            break
        improved_any = False
        order = rng.permutation(n)
        for i in order:
            if time.perf_counter() - start > time_limit:
                break
            for j in range(n):
                if i == j:
                    continue
                if h[i] <= 0.01 or h[i] >= 0.99:
                    continue
                for delta in [0.05, 0.1, 0.2, -0.05, -0.1, -0.2]:
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    h_new = project_to_target(h_try, n)
                    s_new = official_score(h_new)
                    if s_new < score - 1e-12:
                        h = h_new
                        score = s_new
                        improved_any = True
                        break
        if not improved_any:
            break

    return h, official_score(h)


def slsqp_polish(h, n, maxiter=100):
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1
    initial_score = official_score(h)
    theta0 = initial_score + 1e-5

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:n]
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        return x[n] - dx * corr

    def ineq_jac(x):
        hx = x[:n]
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        for r in range(L):
            k = r - (n - 1)
            J[r, :n] = -compute_grad(hx, k, dx)
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h, [theta0]])
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]

    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_new = project_to_target(res.x[:n], n)
            s_new = official_score(h_new)
            if s_new < initial_score - 1e-10:
                return h_new, s_new
    except Exception:
        pass
    return h, initial_score


def interpolate(parent, n_target, shift=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + shift) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)


def discrete_sa(h_init, n, time_limit=30.0, seed=42):
    if n % 2 != 0:
        return project_to_target(h_init, n), official_score(h_init)
    rng = np.random.default_rng(seed)
    h_bin = (h_init >= 0.5).astype(float)
    if int(h_bin.sum()) != n // 2:
        h_sorted = np.argsort(h_init)[::-1]
        h_bin = np.zeros(n)
        h_bin[h_sorted[:n // 2]] = 1.0
    score = official_score(h_bin)
    best_h = h_bin.copy()
    best_score = score
    cur_temp = 0.15
    start = time.perf_counter()

    while time.perf_counter() - start < time_limit and cur_temp > 0.002:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        s_new = official_score(h_bin)
        delta = s_new - score
        if delta < 0 or rng.random() < np.exp(-delta / max(cur_temp, 1e-6)):
            score = s_new
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_bin.copy()
        cur_temp *= 0.9985

    return project_to_target(best_h, n), best_score


def continuous_sa(h_init, n, time_limit=20.0, seed=42):
    rng = np.random.default_rng(seed)
    h = project_to_target(h_init, n).copy()
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    cur_temp = 0.05
    start = time.perf_counter()

    while time.perf_counter() - start < time_limit and cur_temp > 0.003:
        move_type = rng.integers(0, 3)
        h_try = h.copy()
        if move_type == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = rng.uniform(-0.3, 0.3)
                h_try[i] += delta
                h_try[j] -= delta
        elif move_type == 1:
            idx = rng.integers(0, n)
            h_try[idx] = rng.uniform(0, 1)
        else:
            num = rng.integers(2, 5)
            idxs = rng.choice(n, num, replace=False)
            vals = h[idxs] + rng.normal(0, 0.1, num)
            h_try[idxs] = vals
        h_try = project_to_target(h_try, n)
        s_new = official_score(h_try)
        delta = s_new - score
        if delta < 0 or rng.random() < np.exp(-delta / max(cur_temp, 1e-6)):
            h = h_try
            score = s_new
            if score < best_score - 1e-12:
                best_score = score
                best_h = h.copy()
        cur_temp *= 0.997

    return best_h, best_score


def generate_step(n, rng, n_steps=5):
    if n_steps >= n:
        h = rng.uniform(0, 1, n)
    else:
        boundaries = sorted(rng.choice(n - 1, size=n_steps - 1, replace=False))
        boundaries = [0] + list(boundaries) + [n]
        vals = rng.uniform(0.2, 0.8, n_steps)
        h = np.zeros(n)
        for i in range(n_steps):
            h[boundaries[i]:boundaries[i + 1]] = vals[i]
    return project_to_target(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 6.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 10 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = project_to_target(parent, n_parent)
    else:
        n_parent = 64
        best_h = project_to_target(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    rng = np.random.default_rng(seed)

    # Build candidate pool
    candidates = []
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h).copy())
    candidates.append((1.0 - best_h)[::-1].copy())
    h_sym = (best_h + best_h[::-1]) * 0.5
    candidates.append(project_to_target(h_sym, n_parent))

    n_list = sorted(set([n_parent, n_parent + 2, n_parent - 2, n_parent + 4, n_parent - 4]))
    n_list = [x for x in n_list if 32 <= x <= 128]
    for n_try in n_list:
        for sh in [0, 1, 2]:
            candidates.append(interpolate(best_h, n_try, sh))

    for n_try in n_list[:4]:
        for _ in range(4):
            candidates.append(generate_step(n_try, rng, n_steps=rng.integers(3, 8)))
        for _ in range(3):
            h_rand = rng.uniform(0.1, 0.9, n_try)
            candidates.append(project_to_target(h_rand, n_try))
        if n_try % 2 == 0:
            for _ in range(3):
                h_bin = np.zeros(n_try)
                idx = rng.choice(n_try, n_try // 2, replace=False)
                h_bin[idx] = 1.0
                candidates.append(project_to_target(h_bin, n_try))

    # Score and sort
    scored = []
    for hc in candidates:
        hc = project_to_target(hc, len(hc))
        s = official_score(hc)
        if np.isfinite(s):
            scored.append((s, hc))
    scored.sort(key=lambda x: x[0])

    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()

    # Phase 1: Subgradient + momentum on top candidates
    top_k = min(10, len(scored))
    local_results = []
    for idx in range(top_k):
        remaining = deadline - time.perf_counter()
        if remaining < 50:
            break
        s_c, h_c = scored[idx]
        n_c = len(h_c)
        t_lim = min(22.0, remaining / max(1, top_k - idx) * 0.75)
        h_loc, s_loc = subgradient_momentum(h_c, n_c, max_iters=100, time_limit=t_lim, seed=seed + idx * 13)
        local_results.append((s_loc, h_loc))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()

    local_results.sort(key=lambda x: x[0])

    # Phase 2: Coordinate swap descent on best results
    for idx in range(min(6, len(local_results))):
        remaining = deadline - time.perf_counter()
        if remaining < 40:
            break
        s_c, h_c = local_results[idx]
        n_c = len(h_c)
        t_lim = min(15.0, remaining / max(1, 6 - idx) * 0.7)
        h_swap, s_swap = coordinate_swap_descent(h_c, n_c, max_sweeps=5, time_limit=t_lim, seed=seed + 100 + idx)
        if s_swap < best_score:
            best_score = s_swap
            best_h = h_swap.copy()
        local_results.append((s_swap, h_swap))

    local_results.sort(key=lambda x: x[0])

    # Phase 3: SLSQP polish
    for idx in range(min(5, len(local_results))):
        remaining = deadline - time.perf_counter()
        if remaining < 35:
            break
        s_c, h_c = local_results[idx]
        n_c = len(h_c)
        h_ref, s_ref = slsqp_polish(h_c, n_c, maxiter=90)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
        # Follow-up subgradient
        if time.perf_counter() < deadline - 25:
            h2, s2 = subgradient_momentum(h_ref, n_c, max_iters=40, time_limit=8.0, seed=seed + 300 + idx)
            if s2 < best_score:
                best_score = s2
                best_h = h2.copy()

    # Phase 4: Discrete search (for even n)
    if time.perf_counter() < deadline - 50:
        n_even = len(best_h)
        if n_even % 2 != 0:
            n_even = n_even + 1
        h_interp = interpolate(best_h, n_even)
        t_disc = min(30.0, deadline - time.perf_counter() - 40)
        if t_disc > 5:
            h_disc, s_disc = discrete_sa(h_interp, n_even, time_limit=t_disc, seed=seed + 500)
            if s_disc < best_score:
                best_score = s_disc
                best_h = h_disc.copy()
            # Polish the discrete result
            if time.perf_counter() < deadline - 20:
                h_pol, s_pol = subgradient_momentum(h_disc, n_even, max_iters=30, time_limit=10.0, seed=seed + 600)
                if s_pol < best_score:
                    best_score = s_pol
                    best_h = h_pol.copy()

    # Phase 5: Continuous SA for diversification
    if time.perf_counter() < deadline - 40:
        n_sa = len(best_h)
        h_sa, s_sa = continuous_sa(best_h, n_sa, time_limit=15.0, seed=seed + 700)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa.copy()
        h_sa2, s_sa2 = continuous_sa(interpolate(best_h, n_sa + 2, 1), n_sa + 2, time_limit=12.0, seed=seed + 800)
        if s_sa2 < best_score:
            best_score = s_sa2
            best_h = h_sa2.copy()

    # Phase 6: Final polish
    if time.perf_counter() < deadline - 15:
        h_fin, s_fin = slsqp_polish(best_h, len(best_h), maxiter=70)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    if time.perf_counter() < deadline - 8:
        h_fin2, s_fin2 = subgradient_momentum(best_h, len(best_h), max_iters=25, time_limit=6.0, seed=seed + 999)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2.copy()

    # Final validation and return
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score <= 0:
        best_h = project_to_target(parent, n_parent)
        final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
