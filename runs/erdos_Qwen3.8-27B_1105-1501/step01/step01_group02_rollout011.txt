                continue
            h[i] = hi_val - delta
            h[j] = hj_val + delta
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_delta = delta
    
    h[i] = hi_val - best_delta
    h[j] = hj_val + best_delta
    return best_c5

def full_pair_pass(h, target_sum, n, n_coarse=7, n_fine=15):
    """One complete pass over all pairs. Returns (h, c5, num_improvements)."""
    cur_c5 = compute_c5(h)
    improved = 0
    for i in range(n):
        hi_val = h[i]
        for j in range(i + 1, n):
            hj_val = h[j]
            lo = -hj_val
            hi = hi_val
            if hi - lo < 1e-12:
                continue
            old_c5 = cur_c5
            # Coarse search
            deltas = np.linspace(lo, hi, n_coarse)
            best_c5 = old_c5
            best_delta = 0.0
            for delta in deltas:
                if abs(delta) < 1e-12:
                    continue
                h[i] = hi_val - delta
                h[j] = hj_val + delta
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_delta = delta
            if best_delta != 0.0 and best_c5 < old_c5:
                # Fine search
                span = (hi - lo) / max(1, n_coarse)
                fine_lo = max(lo, best_delta - span)
                fine_hi = min(hi, best_delta + span)
                fine_deltas = np.linspace(fine_lo, fine_hi, n_fine)
                h[i] = hi_val - best_delta
                h[j] = hj_val + best_delta
                for delta in fine_deltas:
                    h[i] = hi_val - delta
                    h[j] = hj_val + delta
                    c5 = compute_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_delta = delta
            h[i] = hi_val - best_delta
            h[j] = hj_val + best_delta
            if best_c5 < old_c5 - 1e-13:
                cur_c5 = best_c5
                improved += 1
            else:
                h[i] = hi_val
                h[j] = hj_val
    return h, cur_c5, improved

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass
    
    if h0 is not None and len(h0) > 0:
        n = len(h0)
        target_sum = n / 2.0
        h0 = project_onto_constraint(h0, target_sum)
        best_h = h0.copy()
        best_c5 = compute_c5(best_h)
    else:
        n = 70
        target_sum = n / 2.0
        best_h = make_bump_init(n, alpha=2.0)
        best_c5 = compute_c5(best_h)
    
    elapsed_limit = budget_s - 10
    
    # Phase 1: Full pair passes on the best solution
    stall_rounds = 0
    while time.time() - t0 < elapsed_limit * 0.7:
        h_work = best_h.copy()
        h_work, c5_work, improved = full_pair_pass(h_work, target_sum, n, n_coarse=7, n_fine=15)
        if c5_work < best_c5 - 1e-13:
            best_h = h_work.copy()
            best_c5 = c5_work
            stall_rounds = 0
        else:
            stall_rounds += 1
            if stall_rounds >= 3:
                # Try perturbations
                escaped = False
                for scale in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                    h_try = best_h + rng.randn(n) * scale
                    h_try = project_onto_constraint(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        # Run a pair pass from the perturbed point
                        h_try, c5_try, _ = full_pair_pass(h_try, target_sum, n, n_coarse=5, n_fine=10)
                        if c5_try < best_c5:
                            best_h = h_try.copy()
