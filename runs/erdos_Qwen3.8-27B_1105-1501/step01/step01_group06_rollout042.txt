def sa_binary(n, n_iters, t0, t1, seed):
    """Simulated annealing for binary sequence with n/2 ones."""
    rng = np.random.default_rng(seed)
    k = n // 2
    
    # Initialize with random balanced binary sequence
    h = np.zeros(n)
    ones = rng.choice(n, k, replace=False)
    h[ones] = 1.0
    
    # Compute initial cross-correlation
    def compute_corr(h):
        return np.correlate(h, 1.0 - h, mode='full')
    
    corr = compute_corr(h)
    current_energy = int(np.max(corr))
    best_energy = current_energy
    best_h = h.copy()
    
    for it in range(n_iters):
        # Pick a random 1 and a random 0 to swap
        ones_pos = np.where(h == 1.0)[0]
        zeros_pos = np.where(h == 0.0)[0]
        i = ones_pos[rng.integers(len(ones_pos))]
        j = zeros_pos[rng.integers(len(zeros_pos))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        
        # Recompute correlation (or update efficiently)
        corr = compute_corr(h)
        new_energy = int(np.max(corr))
        
        # Temperature schedule
        T = t0 * (t1 / t0) ** (it / n_iters)
        
        if new_energy <= current_energy or rng.random() < np.exp(-(new_energy - current_energy) / max(T, 1e-10)):
            current_energy = new_energy
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
        else:
            # Undo swap
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_energy

import time

def eval_c5(h):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(v, target_sum):
    # Binary search for the right threshold
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def sa_binary(n, n_iters, t_start, t_end, seed, time_limit):
    """Simulated annealing for balanced binary sequence."""
    rng = np.random.default_rng(seed)
    k = n // 2
    t0 = time.time()
    
    # Initialize
    h = np.zeros(n)
    h[rng.choice(n, k, replace=False)] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_e = int(np.max(corr))
    best_e = current_e
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        
        # Swap random 1 and 0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = ones[rng.integers(k)]
        j = zeros[rng.integers(k)]
        
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_e = int(np.max(corr))
        
        T = t_start * (t_end / t_start) ** (float(it) / n_iters)
        if new_e <= current_e or rng.random() < np.exp(-(new_e - current_e) / max(T, 1e-10)):
            current_e = new_e
            if new_e < best_e:
                best_e = new_e
                best_h = h.copy()
                # Reset temperature on improvement? (optional)
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_e

        h[j] = 1.0
        ones_arr[ri] = j
        zeros_arr[rj] = i
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_e = int(np.max(corr))

                best_h = h.copy()
                last_improve = it
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_arr[ri] = i
            zeros_arr[rj] = j

def local_search_binary(h, max_time=10.0):
    """1-opt local search: try all single swaps, keep best."""
    n = len(h)
    k = n // 2
    t0 = time.time()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_e = int(np.max(corr))
    
    while time.time() - t0 < max_time:
        improved = False
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        best_e_swap = current_e
        best_pair = None
        
        for i in ones:
            for j in zeros:
                h[int(i)] = 0.0
                h[int(j)] = 1.0
                corr2 = np.correlate(h, 1.0 - h, mode='full')
                e = int(np.max(corr2))
                h[int(i)] = 1.0
                h[int(j)] = 0.0
                if e < best_e_swap:
                    best_e_swap = e
                    best_pair = (int(i), int(j))
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            current_e = best_e_swap
            improved = True
        
        if not improved:
            break
    
    return h, current_e

def slsqp_optimize(h, n, target_sum, beta, maxiter=200, time_limit=60.0):
    """SLSQP optimization with smooth max objective."""
    t0 = time.time()
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        dx = 2.0 / n
        c = corr * dx
        c_max = np.max(c)
        shifted = np.clip(beta * (c - c_max), -500.0, 500.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / beta
    
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    return res.x
