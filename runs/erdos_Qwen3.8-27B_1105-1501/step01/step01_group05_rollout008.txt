def score(h):
    """Compute the exact C5 score."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n=None):
    """Project h to [0,1]^n with sum = n/2."""
    if n is None:

    if len(h) < n:

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    for _ in range(100):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    return h

    U = min(1.0, S)
    if U - L < 1e-10:
        return h, score(h)

    # Sample the line

def full_pair_descent(h, max_passes=20, n_samples=15, golden_iters=8, deadline=None, seed=42):
    """Run full pair-line descent until convergence."""
    rng = np.random.default_rng(seed)
    n = len(h)
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    cur_score = score(h)
    
    for p in range(max_passes):
        if deadline is not None and time.perf_counter() > deadline:
            break
        rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            if deadline is not None and time.perf_counter() > deadline:
                break
            h, s = pair_line_minimize(h, i, j, n_samples, golden_iters)
            if s < cur_score - 1e-13:
                cur_score = s
                improved = True
        if not improved:
            break
    
    return h, cur_score

def block_perturb(h, block_size, step_range, rng):
    """Zero-sum block perturbation for basin hopping."""
    n = len(h)
    hs = h.copy()
    m = min(block_size, n)
    if m < 2:
        return h
    block = rng.choice(n, size=m, replace=False)
    d = rng.standard_normal(m)
    d -= d.mean()
    # Scale to respect box constraints
    max_step = 0.0
    for idx, di in zip(block, d):
        if di > 0:
            max_step = max(max_step, di * hs[idx])  # can decrease
        elif di < 0:
            max_step = max(max_step, (-di) * (1.0 - hs[idx]))  # can increase
    if max_step < 1e-10:
        return h
    step = rng.uniform(step_range[0], step_range[1])
    lam = step / max_step
    for idx, di in zip(block, d):
        hs[idx] += lam * di
    return repair(hs, n)

def interpolate(h_src, n_target):
    """Interpolate h_src to n_target points."""
    m = len(h_src)
    if m == n_target:

    h_new = np.interp(xt, xs, h_src, left=h_src[0], right=h_src[-1])
    return repair(h_new, n_target)

import numpy as np
from scipy import optimize
import time


def score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair(h, n=None):
    if n is None:
        n = len(h)
    h = np.array(h, dtype=float).ravel()[:n]
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            if delta > 0:
                h[idx] += delta
            else:
                break
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            if delta > 0:
                h[idx] -= delta
            else:
                break
    return h


def pair_line_minimize(h, i, j, n_samples=20, golden_iters=10):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-10:
        return h, score(h)

    hs = h.copy()
    cur_s = score(h)
    best_x = h[i]
    best_s = cur_s

    xs = np.linspace(L, U, n_samples)
    for x in xs:
        if abs(x - best_x) < 1e-12:
            continue
        hs[i] = x
        hs[j] = S - x
        s = score(hs)
        if s < best_s:
            best_s = s
            best_x = x

    # Golden section in best interval (and neighbors)
    intervals = []
    for k in range(len(xs) - 1):
        mid = (xs[k] + xs[k + 1]) / 2.0
        dist = abs(mid - best_x)
        intervals.append((dist, k, xs[k], xs[k + 1]))
    intervals.sort()
    tried = set()

    for _, k_idx, a, b in intervals[:3]:
        key = (round(a, 10), round(b, 10))
        if key in tried or b - a < 1e-10:
            continue
        tried.add(key)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        hs[i] = c
        hs[j] = S - c
        fc = score(hs)
        hs[i] = d
        hs[j] = S - d
        fd = score(hs)

        for _ in range(golden_iters):
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = b - gr * (b - a)
                hs[i] = c
                hs[j] = S - c
                fc = score(hs)
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                hs[i] = d
                hs[j] = S - d
                fd = score(hs)

        if fc < fd:
            gx, gs = c, fc
        else:
            gx, gs = d, fd
        if gs < best_s:
            best_s = gs
            best_x = gx

    hs[i] = best_x
    hs[j] = S - best_x
    final_s = score(hs)
    return hs, final_s


def full_pair_descent(h, max_passes=20, n_samples=18, golden_iters=8,
                      deadline=None, seed=42, min_seg_width=0.02):
    rng = np.random.default_rng(seed)
    n = len(h)
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            S = h[i] + h[j]
            L = max(0.0, S - 1.0)
            U = min(1.0, S)
            if U - L >= min_seg_width:
                pairs.append((i, j))

    if not pairs:
        return h, score(h)

    cur_score = score(h)
    pairs_arr = np.array(pairs)

    for p in range(max_passes):
        if deadline is not None and time.perf_counter() > deadline:
            break
        order = rng.permutation(len(pairs_arr))
        improved = False
        for idx in order:
            if deadline is not None and time.perf_counter() > deadline:
                break
            i, j = pairs_arr[idx]
            S = h[i] + h[j]
            L = max(0.0, S - 1.0)
            U = min(1.0, S)
            if U - L < min_seg_width:
                continue
            h, s = pair_line_minimize(h, i, j, n_samples, golden_iters)
            if s < cur_score - 1e-12:
                cur_score = s
                improved = True
        if not improved:
            break

    final_s = score(h)
    return h, final_s


def block_perturb(h, block_size=None, step_range=(0.03, 0.12), rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.default_rng()
    if block_size is None:
        block_size = max(3, n // 8)
    block_size = min(block_size, n)
    hs = h.copy()
    block = rng.choice(n, size=block_size, replace=False)
    d = rng.standard_normal(block_size)
    d -= d.mean()
    max_scale = 1e10
    for idx, di in zip(block, d):
        if di > 0:
            max_scale = min(max_scale, hs[idx] / di)
        elif di < 0:
            max_scale = min(max_scale, (1.0 - hs[idx]) / (-di))
    if max_scale < 1e-10:
        return h
    max_step = rng.uniform(step_range[0], step_range[1])
    lam = min(max_step, max_scale)
    for idx, di in zip(block, d):
        hs[idx] += lam * di
    return repair(hs, n)


def interpolate_to(h_src, n_target, offset=0):
    m = len(h_src)
    if m == n_target:
        return repair(h_src.copy(), n_target)
    xs = (np.arange(m) + 0.5) / m
    xt = ((np.arange(n_target) + 0.5 + offset * m / n_target) % n_target) / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    xt_sorted = np.clip(xt_sorted, 0, 1)
    h_new = np.interp(xt_sorted, xs, h_src, left=h_src[0], right=h_src[-1])
    h_new = h_new[np.argsort(order)]
    return repair(h_new, n_target)
