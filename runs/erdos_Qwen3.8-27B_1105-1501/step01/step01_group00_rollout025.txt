def repair_sum(h):
    """Ensure sum(h) == len(h)/2 and 0 <= h <= 1."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    residual = target - h.sum()
    if abs(residual) < 1e-14:
        return h
    if residual > 0:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-15:
                break
    else:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if -residual < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return repair_sum(h_int)

        # Shuffle pair order
        rng.shuffle(pairs)
        improved = False

            t_lo = max(-h[a], h[b] - 1.0)

    ma2 = s <= a

    mb2 = s <= b

    gp[ma2] -= h[a - s[ma2]]

    gp[mb2] += h[b - s[mb2]]

def cd_sweep(h, V, pairs, deadline):
    """One sweep of coordinate descent. Returns (h, V, best, improved_count)."""
    n = len(h)
    best = float(V.max())
    count = 0
