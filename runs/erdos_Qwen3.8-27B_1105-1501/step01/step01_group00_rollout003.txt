    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(v.max())

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    h = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    break
                h[idx] -= sub
                residual += sub

def grad_row(h, s):
    """Gradient of the correlation at lag s with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]
    else:
        t = -s
        for i in range(n - t):
            g[i + t] += (1.0 - h[i])
            g[i] -= h[i + t]
    return g

    w = np.exp(beta * (vals[order] - vals[order].max()))
    w /= w.sum()

        s = lag_idx - (n - 1)
        d -= w[i] * grad_row(h, s)
    d -= d.mean()
    return d, vals.max()

        g += w[i] * gr

    n = len(z)
    target = n / 2.0
    h = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def grad_row(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]
    else:
        t = -s
        for i in range(n - t):
            g[i + t] += (1.0 - h[i])
            g[i] -= h[i + t]
    return g

def active_direction(h, K, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    order = np.argsort(vals)[::-1][:K]
    w = np.exp(beta * (vals[order] - vals[order].max()))
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(order)) / len(order)
    else:
        w /= ws
    d = np.zeros(n)
    for i in range(len(order)):
        s = order[i] - (n - 1)
        d -= w[i] * grad_row(h, s)
    d -= d.mean()
    return d, float(vals.max())

def pair_search(h, rng, n_pairs, n_samples, deadline):
    local_h = h.copy()
    local_score = score(local_h)
    n = len(local_h)
    pairs = 0
    while pairs < n_pairs:
        if time.time() >= deadline:
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        pairs += 1
        t_lo = max(-local_h[a], local_h[b] - 1.0)
        t_hi = min(1.0 - local_h[a], local_h[b])
        if t_hi - t_lo < 1e-14:
            continue
        ts = np.linspace(t_lo, t_hi, n_samples + 2)[1:-1]
        ts = np.concatenate([ts, [t_lo * 0.99, t_hi * 0.99, (t_lo + t_hi) / 2.0]])
        best_t = 0.0
        best_sc = local_score
        for t in ts:
            if t < t_lo - 1e-12 or t > t_hi + 1e-12:
                continue
            local_h[a] += t
            local_h[b] -= t
            if local_h[a] < -1e-12 or local_h[a] > 1.0 + 1e-12:
                local_h[a] -= t
                local_h[b] += t
                continue
            if local_h[b] < -1e-12 or local_h[b] > 1.0 + 1e-12:
                local_h[a] -= t
                local_h[b] += t
                continue
            sc = score(local_h)
            local_h[a] -= t
            local_h[b] += t
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_t = t
        if best_t != 0.0:
            local_h[a] += best_t
            local_h[b] -= best_t
            local_score = best_sc
    local_h = make_feasible(local_h)
    final_sc = score(local_h)
    if final_sc <= local_score + 1e-10:
        return local_h, final_sc
    return h, score(h)

def direction_search(h, rng, deadline, n_steps=80):
    local_h = h.copy()
    local_score = score(local_h)
    n = len(local_h)
    K = max(5, min(25, int(0.3 * (2 * n - 1))))
    for step in range(n_steps):
        if time.time() >= deadline:
            break
        d, cmax = active_direction(local_h, K, 40.0)
        dnrm = np.max(np.abs(d))
        if dnrm < 1e-14:
            break
        d = d / dnrm
        best_sc = local_score
        best_alpha = 0.0
        best_sign = 1
        alphas = [0.001, 0.003, 0.01, 0.03, 0.06, 0.1, 0.2, 0.4]
        for sgn in [1, -1]:
            for alpha in alphas:
                cand = local_h + sgn * alpha * d
                if np.any(cand < -1e-12) or np.any(cand > 1.0 + 1e-12):
                    continue
                sc = score(cand)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_alpha = alpha
                    best_sign = sgn
        if best_alpha > 0:
            local_h = local_h + best_sign * best_alpha * d
            local_score = best_sc
        else:
            break
    local_h = make_feasible(local_h)
    final_sc = score(local_h)
    if final_sc <= local_score + 1e-10:
        return local_h, final_sc
    return h, score(h)

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + (cmax + logsum / beta - cmax) + logsum / beta - cmax + cmax

def smooth_obj2(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta + np.max(shifted) / beta

def smooth_grad2(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    g = np.zeros(n)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        gr = grad_row(x, s)
        g += w[i] * gr
    return dx * g

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def slsqp_solve(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_obj2, h0, args=(beta,), jac=smooth_grad2,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def wave_init(n, f1, f2, a1, a2, p1, p2):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.sin(2.0 * np.pi * f1 * x + p1) + a2 * np.cos(2.0 * np.pi * f2 * x + p2)
    return make_feasible(h)

def composite_pair_search(h, rng, n_rounds, n_pairs_per_round, deadline):
    local_h = h.copy()
    local_score = score(local_h)
    n = len(local_h)
    for rnd in range(n_rounds):
        if time.time() >= deadline:
            break
        local_h, local_score = pair_search(local_h, rng, n_pairs_per_round, 5, deadline)
    return local_h, local_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(12.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')
    best_n = 0

    def is_dead():
        return time.time() >= deadline

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 8 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                best_n = len(parent)
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 48
        best_h = wave_init(n0, 2, 3, 0.2, 0.1, 0.5, 1.0)
        best_score = score(best_h)
        best_n = len(best_h)

    # Phase 1: Intensive pair search
    if not is_dead():
        h_new, sc = pair_search(best_h, rng, 5000, 6, deadline)
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 2: Direction search
    if not is_dead():
        h_new, sc = direction_search(best_h, rng, deadline, 100)
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 3: SLSQP polishing
    if not is_dead():
        for beta in [50, 200, 800, 3000]:
            if is_dead():
                break
            h_new, sc = slsqp_solve(best_h, beta, 120)
            if h_new is not None and sc < best_score - 1e-14:
                best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 4: More pair search after SLSQP
    if not is_dead():
        h_new, sc = pair_search(best_h, rng, 5000, 7, deadline)
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 5: Direction search round 2
    if not is_dead():
        h_new, sc = direction_search(best_h, rng, deadline, 80)
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 6: Multi-resolution
    if not is_dead():
        n_base = len(best_h)
        n_opts_list = sorted(set([max(24, n_base - 8), max(30, n_base - 4),
                                   n_base, min(70, n_base + 4), min(76, n_base + 10)]))
        for n_opt in n_opts_list:
            if is_dead():
                break
            h_int = interpolate_to(best_h, n_opt)
            sc_int = score(h_int)
            if sc_int < best_score - 1e-14:
                best_h, best_score, best_n = h_int, sc_int, len(h_int)
            if not is_dead():
                h_new, sc = pair_search(h_int, rng, 3000, 6, deadline)
                if sc < best_score - 1e-14:
                    best_h, best_score, best_n = h_new, sc, len(h_new)
            if not is_dead():
                h_new, sc = direction_search(h_int, rng, deadline, 50)
                if sc < best_score - 1e-14:
                    best_h, best_score, best_n = h_new, sc, len(h_new)
            if not is_dead():
                h_new, sc = slsqp_solve(best_h, 500, 80)
                if h_new is not None and sc < best_score - 1e-14:
                    best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 7: Random perturbations from best
    if not is_dead():
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(3, min(10, int(remaining / 12.0)))
        for i in range(n_starts):
            if is_dead():
                break
            sigma = 0.02 + 0.06 * rng.random()
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            sc_cand = score(h_cand)
            if sc_cand < best_score - 1e-14:
                best_h, best_score, best_n = h_cand, sc_cand, n
            if not is_dead():
                h_new, sc = pair_search(h_cand, rng, 2000, 5, deadline)
                if sc < best_score - 1e-14:
                    best_h, best_score, best_n = h_new, sc, len(h_new)
            if not is_dead():
                h_new, sc = slsqp_solve(best_h, 300, 80)
                if h_new is not None and sc < best_score - 1e-14:
                    best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 8: Diverse wave starts
    if not is_dead():
        n_w = len(best_h)
        wave_configs = [(1, 2, 0.18, 0.08, 0.3, 1.7),
                        (2, 4, 0.15, 0.10, 2.1, 0.8),
                        (3, 5, 0.12, 0.07, 4.2, 2.5),
                        (1, 3, 0.20, 0.10, 1.1, 3.9),
                        (2, 5, 0.14, 0.09, 5.0, 0.4)]
        for cfg in wave_configs:
            if is_dead():
                break
            h_wave = wave_init(n_w, cfg[0], cfg[1], cfg[2], cfg[3], cfg[4], cfg[5])
            sc_w = score(h_wave)
            if sc_w < best_score - 1e-14:
                best_h, best_score, best_n = h_wave, sc_w, n_w
            if sc_w < 0.42 and not is_dead():
                h_new, sc = pair_search(h_wave, rng, 2000, 5, deadline)
                if sc < best_score - 1e-14:
                    best_h, best_score, best_n = h_new, sc, len(h_new)
                if not is_dead():
                    h_new, sc = slsqp_solve(h_new, 200, 80)
                    if h_new is not None and sc < best_score - 1e-14:
                        best_h, best_score, best_n = h_new, sc, len(h_new)

    # Phase 9: Final intensive polishing
    if not is_dead():
        h_new, sc = pair_search(best_h, rng, 8000, 8, deadline)
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = h_new, sc, len(h_new)

    if not is_dead():
        h_new, sc = direction_search(best_h, rng, deadline, 120)
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = h_new, sc, len(h_new)

    if not is_dead():
        for beta in [1000, 5000, 20000]:
            if is_dead():
                break
            h_new, sc = slsqp_solve(best_h, beta, 200)
            if h_new is not None and sc < best_score - 1e-14:
                best_h, best_score, best_n = h_new, sc, len(h_new)

    if not is_dead():
        h_new, sc = pair_search(best_h, rng, 6000, 10, deadline)
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = h_new, sc, len(h_new)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)) or final_score > 0.5 or abs(final_h.sum() - final_n / 2.0) > 1e-8:
        fb_n = 48
        final_h = np.full(fb_n, 0.5)
        final_score = score(final_h)
        final_n = fb_n

    return final_h, final_score + 1e-13, final_n
