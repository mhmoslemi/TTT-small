```python
def subgradient(h, n):
    """Compute subgradient of max_s c_s(h)"""
    dx = 2.0 / n
    # Compute all c_s values
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    # Find active constraints
    active = np.where(corr >= max_c - 1e-10)[0]
    # Gradient of c_s with respect to h
    # c_s = sum_i h[i] * (1-h[i+s]) * dx
    # dc_s/dh[j] = dx * [(1-h[j+s]) if j+s valid + (-h[j-s]) if j-s valid]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        # Contribution to gradient from this shift
        if s >= 0:
            # i ranges from 0 to n-1-s
            for i in range(n - s):
                grad[i] += dx * (1.0 - h[i + s])
                grad[i + s] += dx * (-h[i])
        else:
            t = -s
            for i in range(t, n):
                grad[i] += dx * (1.0 - h[i - t])
                grad[i - t] += dx * (-h[i])
    # Normalize by number of active constraints
    grad /= len(active)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    z = np.clip(z, -10, 10)  # safety
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, active_indices=None):
    """Compute gradient of c_s for given shifts, or for all active shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if active_indices is None:
        max_c = np.max(corr)
        active_indices = np.where(corr >= max_c - 1e-10)[0]
    
    grad = np.zeros(n)
    for k in active_indices:
        s = k - (n - 1)
        if s >= 0:
            # i from 0 to n-1-s
            length = n - s
            grad[:length] += dx * (1.0 - h[s:s+length])
            grad[s:s+length] += dx * (-h[:length])
        else:
            t = -s
            # i from t to n-1
            length = n - t
            grad[t:] += dx * (1.0 - h[:length])
            grad[:length] += dx * (-h[t:])
    return grad / len(active_indices)

def subgradient_descent(h0, max_iters=5000, c_step=0.5):
    """Polya-Shor subgradient method."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_score = official_score(h0)
    
    for it in range(max_iters):
        # Compute subgradient
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        max_c = np.max(corr)
        # Use all near-active constraints for smoother gradient
        threshold = max_c - 0.01
        active = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                length = n - s
                grad[:length] += (2.0/n) * (1.0 - h[s:s+length])
                grad[s:s+length] += (2.0/n) * (-h[:length])
            else:
                t = -s
                length = n - t
                grad[t:] += (2.0/n) * (1.0 - h[:length])
                grad[:length] += (2.0/n) * (-h[t:])
        grad /= len(active)
        
        # Remove mean component (since constraint is sum=constant)
        grad = grad - np.mean(grad)
        
        # Step size
        alpha = c_step / np.sqrt(it + 1)
        
        # Projection: h_new = project(h - alpha * grad)
        h_new = project_feasible(h - alpha * grad, n)
        
        # Evaluate
        score = official_score(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        h = h_new
```

For efficiency, I need to compute the effect of a swap quickly. When I swap h[i] and h[j] (where h[i]=0, h[j]=1, or vice versa), the change in N_10(s) for each shift s can be computed.

Let's say h[i] was 0 and h[j] was 1 (so after swap, h[i]=1, h[j]=0).
For shift s:
- Before: contribution from i was 0*(1-h[i+s]) = 0, and contribution from j was 1*(1-h[j+s])
  Also: if i-s valid, contribution was h[i-s]*(1-0) = h[i-s]
  If j-s valid, contribution was h[j-s]*(1-1) = 0
- After: contribution from i was 1*(1-h[i+s]), and from j was 0
  If i-s valid: h[i-s]*(1-1) = 0
  If j-s valid: h[j-s]*(1-0) = h[j-s]

So the change in c_s (the raw sum, before multiplying by dx) is:
Δc_s = [new contribution from i, j] - [old contribution from i, j]

Let me denote:
- a = h[i+s] (if valid), b = h[i-s] (if valid), c = h[j+s] (if valid), d = h[j-s] (if valid)
- Old contribution: 0*(1-a) + b*(1-0) + 1*(1-c) + d*(1-1) = b + (1-c)

  - h[p] * (1-h[p+s]) if p+s valid (p is the "first" index)
  - h[p-s] * (1-h[p]) if p-s valid (p is the "second" index, i.e., the "-s" shift)

For position i (was 0, now 1):
- As first index (shift s): old = 0*(1-h[i+s]) = 0, new = 1*(1-h[i+s]) = 1-h[i+s]
- As second index (shift -s): old = h[i-s]*(1-0) = h[i-s], new = h[i-s]*(1-1) = 0

For position j (was 1, now 0):
- As first index (shift s): old = 1*(1-h[j+s]) = 1-h[j+s], new = 0*(1-h[j+s]) = 0
- As second index (shift -s): old = h[j-s]*(1-1) = 0, new = h[j-s]*(1-0) = h[j-s]

So for shift s:
Δc_s = (1-h[i+s]) - h[i-s] - (1-h[j+s]) + h[j-s]

     = h[j+s] + h[j-s] - h[i+s] - h[i-s]

(where each term is 0 if the index is out of range)
