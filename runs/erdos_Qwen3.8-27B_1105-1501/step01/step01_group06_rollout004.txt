def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_feasible(v, target_sum):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = float(np.sum(x)) - target_sum
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(x > 1e-14)[0]
            for i in idx:
                dec = min(residual, x[i])
                if dec > 0:
                    x[i] -= dec
                    residual -= dec
                    if abs(residual) < 1e-12:
                        break
        else:
            idx = np.where(x < 1.0 - 1e-14)[0]
            for i in idx:
                inc = min(-residual, 1.0 - x[i])
                if inc > 0:

                    residual += inc
                    if abs(residual) < 1e-12:
                        break
    return x

def subgradient_scores(h):
    n = len(h)

    c_max = float(np.max(corr))
    tol = max(1e-10, 0.08 * abs(c_max) + 1e-6)
    active_mask = corr >= (c_max - tol)
    scores = np.zeros(n)
    active_positions = np.where(active_mask)[0]
    for p in active_positions:
        k = p - (n - 1)
        for t in range(n):
            if 0 <= t - k < n:
                scores[t] += (1.0 - h[t - k])
            if 0 <= t + k < n:
                scores[t] -= h[t + k]
    return scores
