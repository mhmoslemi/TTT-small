    # v_{j-d} = v_i = 1
    # L_{-d} = (1-h_{2i-j})*[0≤2i-j<n] - 0 - (1-h_i) - h_j*1
    #        = (1-h_{2i-j})*[0≤2i-j<n] - (1-h_i) - h_j
    
    Lnd = -(1.0 - h[i]) - h[j]
    two_i_minus_j = 2 * i - j
    if 0 <= two_i_minus_j < n:
        Lnd += (1.0 - h[two_i_minus_j])
    Qnd = 1.0
    
    # Unaffected max
    mask = np.ones(2*n-1, dtype=bool)
    mask[idx_0] = False
    mask[idx_d] = False
    mask[idx_nd] = False
    M = float(np.max(F_all[mask]))
    
    # Find best s
    candidates = [s_lo, s_hi, 0.0]
    
    # Vertex of a(s) = F0 + L0*s + Q0*s²: s = -L0/(2*Q0) = -L0/(-4) = L0/4
    candidates.append(L0 / 4.0)
    # Vertex of b(s) = Fd + Ld*s + Qd*s²: s = -Ld/(2*Qd) = -Ld/2
    candidates.append(-Ld / 2.0)
    # Vertex of c(s): s = -Lnd/2
    candidates.append(-Lnd / 2.0)
    # Intersections
    # a=b: (Q0-Qd)s² + (L0-Ld)s + (F0-Fd) = 0 → -3s² + (L0-Ld)s + (F0-Fd) = 0
    # 3s² + (Ld-L0)s + (Fd-F0) = 0
    _add_roots(candidates, 3.0, Ld - L0, Fd - F0)
    # a=c: -3s² + (L0-Lnd)s + (F0-Fnd) = 0 → 3s² + (Lnd-L0)s + (Fnd-F0) = 0
    _add_roots(candidates, 3.0, Lnd - L0, Fnd - F0)
    # b=c: (Qd-Qnd)s² + (Ld-Lnd)s + (Fd-Fnd) = 0 → 0*s² + (Ld-Lnd)s + (Fd-Fnd) = 0
    if abs(Ld - Lnd) > 1e-15:
        candidates.append(-(Fd - Fnd) / (Ld - Lnd))
    
    best_s = 0.0
    best_val = float(np.max(F_all))  # current value
    
    for s in candidates:
        if s < s_lo - 1e-10 or s > s_hi + 1e-10:
            continue
        s = float(np.clip(s, s_lo, s_hi))
        a = F0 + L0*s + Q0*s*s
        b = Fd + Ld*s + Qd*s*s
        c = Fnd + Lnd*s + Qnd*s*s
        g = max(M, a, b, c)
        if g < best_val - 1e-13:
            best_val = g
            best_s = s
    
    # Fine grid as backup
    n_grid = 16
    for gi in range(n_grid + 1):
        s = s_lo + (s_hi - s_lo) * gi / n_grid
        a = F0 + L0*s + Q0*s*s
        b = Fd + Ld*s + Qd*s*s
        c = Fnd + Lnd*s + Qnd*s*s
        g = max(M, a, b, c)
        if g < best_val - 1e-13:
            best_val = g
            best_s = s
    
    return best_s

def _add_roots(candidates, a, b, c):
    if abs(a) < 1e-15:
        if abs(b) > 1e-15:
            candidates.append(-c / b)
        return
    disc = b*b - 4.0*a*c
    if disc < -1e-12:
        return
    disc = max(0.0, disc)
    sqrt_d = np.sqrt(disc)
    candidates.append((-b + sqrt_d) / (2.0 * a))
    candidates.append((-b - sqrt_d) / (2.0 * a))

import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_all_F(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _add_quadratic_roots(lst, a, b, c):
    if abs(a) < 1e-15:
        if abs(b) > 1e-15:
            lst.append(-c / b)
        return
    disc = b * b - 4.0 * a * c
    if disc < -1e-12:
        return
    disc = max(0.0, disc)
    sq = np.sqrt(disc)
    lst.append((-b + sq) / (2.0 * a))
    lst.append((-b - sq) / (2.0 * a))

def find_optimal_s(h, i, j, n, F_all):
    s_lo = max(-h[i], h[j] - 1.0)
    s_hi = min(1.0 - h[i], h[j])
    if s_hi - s_lo < 1e-14:
        return 0.0
    d = j - i
    idx0 = n - 1
    idxd = d + n - 1
    idxnd = -d + n - 1
    F0 = F_all[idx0]
    Fd = F_all[idxd]
    Fnd = F_all[idxnd]
    L0 = 2.0 * (h[j] - h[i])
    Q0 = -2.0
    Ld = (1.0 - h[j]) + h[i]
    tji = 2 * j - i
    if 0 <= tji < n:
        Ld -= (1.0 - h[tji])
    Qd = 1.0
    Lnd = -(1.0 - h[i]) - h[j]
    tij = 2 * i - j
    if 0 <= tij < n:
        Lnd += (1.0 - h[tij])
    Qnd = 1.0
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[idx0] = False
    mask[idxd] = False
    mask[idxnd] = False
    M = float(np.max(F_all[mask]))
    cands = [s_lo, s_hi, 0.0, L0 / 4.0, -Ld / 2.0, -Lnd / 2.0]
    _add_quadratic_roots(cands, 3.0, Ld - L0, Fd - F0)
    _add_quadratic_roots(cands, 3.0, Lnd - L0, Fnd - F0)
    if abs(Ld - Lnd) > 1e-15:
        cands.append(-(Fd - Fnd) / (Ld - Lnd))
    best_s = 0.0
    best_val = float(np.max(F_all))
    for s in cands:
        if s < s_lo - 1e-10 or s > s_hi + 1e-10:
            continue
        s = float(np.clip(s, s_lo, s_hi))
        a = F0 + L0 * s + Q0 * s * s
        b = Fd + Ld * s + Qd * s * s
        c = Fnd + Lnd * s + Qnd * s * s
        g = max(M, a, b, c)
        if g < best_val - 1e-13:
            best_val = g
            best_s = s
    ng = 16
    for gi in range(ng + 1):
        s = s_lo + (s_hi - s_lo) * gi / ng
        a = F0 + L0 * s + Q0 * s * s
        b = Fd + Ld * s + Qd * s * s
        c = Fnd + Lnd * s + Qnd * s * s
        g = max(M, a, b, c)
        if g < best_val - 1e-13:
            best_val = g
            best_s = s
    return best_s

def pair_descent(h, time_budget_s=10.0, seed=0, max_passes=500):
    n = len(h)
    h = h.copy()
    F_all = compute_all_F(h)
    current_best = float(np.max(F_all))
    t_start = time.time()
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    rng = np.random.default_rng(seed)
    improved_passes = 0
    for pass_num in range(max_passes):
        if time.time() - t_start > time_budget_s:
            break
        improved = False
        if pass_num > 10:
            rng.shuffle(pairs)
        for (i, j) in pairs:
            s = find_optimal_s(h, i, j, n, F_all)
            if abs(s) > 1e-10:
                h[i] += s
                h[j] -= s
                F_all = compute_all_F(h)
                new_best = float(np.max(F_all))
                if new_best < current_best - 1e-12:
                    current_best = new_best
                improved = True
        if improved:
            improved_passes = 0
        else:
            improved_passes += 1
            if improved_passes >= 5:
                break
    return make_feasible(h)

def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def dual_transform(h):
    return make_feasible(1.0 - h[::-1].copy())
