    #      sum(delta) = 0
    #      -h[j] <= delta[j] <= 1-h[j]

    c_vec[-1] = 1.0

        row[j] = 1.0
        A_rows.append(row)

        row2 = np.zeros(nv)
        row2[j] = -1.0
        A_rows.append(row2)
        b_rows.append(step_scale)

```python
def gradient_matrix(h, shifts):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((len(shifts), n))
    for idx, s in enumerate(shifts):
        s = int(s)
        if s >= 0:
            if n - s > 0:
                grads[idx, :n-s] += (1.0 - h[s:n]) * dx
                grads[idx, s:] += (-h[:n-s]) * dx
        else:
            sp = -s
            if n - sp > 0:
                grads[idx, sp:] += (1.0 - h[:n-sp]) * dx
                grads[idx, :n-sp] += (-h[sp:]) * dx
    return grads
```
