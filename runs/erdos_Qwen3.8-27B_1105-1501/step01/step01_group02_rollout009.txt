import numpy as np

def project_box_sum(x, target):
    x = np.array(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - target) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    c = (lo + hi) / 2.0
    proj = np.clip(x - c, 0.0, 1.0)
    residual = target - np.sum(proj)
    if abs(residual) > 1e-14:
        if residual > 0:
            caps = 1.0 - proj
            order = np.argsort(-caps)
            for idx in order:
                add = min(caps[idx], residual)
                proj[idx] += add
                residual -= add
                if residual < 1e-14:
                    break
        else:
            caps = proj.copy()
            order = np.argsort(-caps)
            for idx in order:
                sub = min(caps[idx], -residual)
                proj[idx] -= sub
                residual += sub
                if residual > -1e-14:

    return proj
