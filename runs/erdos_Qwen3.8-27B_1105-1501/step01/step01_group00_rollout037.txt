```python
def compute_single_grad(h, s):
    """Compute ∂overlap[s]/∂h for a single lag s."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # overlap[s] = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
        for k in range(s, n):
            g[k] += (1.0 - h[k-s])
        for k in range(0, n-s):
            g[k] -= h[k+s]
    else:
        t = -s
        # overlap[-t] = sum_{j=0}^{n-1-t} h[j]*(1-h[j+t])
        for k in range(0, n-t):
            g[k] += (1.0 - h[k+t])
        for k in range(t, n):
            g[k] -= h[k-t]
    return g

    return grads

def project_feasible(z, target_sum):
    """Project z onto {h: 0<=h<=1, sum(h)=target_sum}."""
    n = len(z)
    z = np.asarray(z, dtype=float).copy()
    # Bisection on lambda: find lambda such that sum(clip(z - lambda, 0, 1)) = target_sum
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s < target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)
```

Let me also think about whether there's a structural insight I'm missing. The problem is to minimize max_k ∫ h(x)(1-h(x+k)) dx. 

For a binary h (which is optimal at the discrete level), this becomes:
minimize max_s (number of j where h[j]=1, h[j+s]=0, weighted by 1/(n-s) or similar)

Wait, actually the formula is:
V[s] = (2/n) * sum_{j=0}^{n-1-s} h[j+s]*(1-h[j]) (for s ≥ 0)

For binary h, this counts (with weight 2/n) the number of "1 followed by 0" at distance s.

The total number of such pairs across all s is fixed (it's related to the "disorder" of the sequence). So we want to spread the transitions evenly across all lags.

This is related to the concept of a "difference set" or "Bose-Chowla" type construction in combinatorics.

For a binary sequence of length n with n/2 ones, the number of 1→0 transitions at lag s is:
T[s] = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j]) (for s ≥ 0)

We want to minimize max_s T[s] * (2/n).

The sum of T[s] over all s ≥ 0 is:
sum_s sum_j h[j+s]*(1-h[j]) = sum_j h[j] * (n/2 - 1 - something)

The point is: the optimal binary sequence should have T[s] ≈ constant for all s, which gives max_s T[s] ≈ (total transitions) / (n-1).

```python
    n = len(h)
    target = n / 2.0

    h_bin = (h >= 0.5).astype(float)

        h_test[j] = 1

            h_bin = h_test
            best_sc = sc
            best_h = h_bin.copy()
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_single(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        idx1 = np.arange(s, n)
        g[idx1] += (1.0 - h[idx1 - s])
        idx2 = np.arange(0, n - s)
        g[idx2] -= h[idx2 + s]
    else:
        t = -s
        idx1 = np.arange(0, n - t)
        g[idx1] += (1.0 - h[idx1 + t])
        idx2 = np.arange(t, n)
        g[idx2] -= h[idx2 - t]
    return g

def project_feasible(z, target_sum):
    n = len(z)
    z = np.asarray(z, dtype=float).copy()
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s < target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def subgradient_optimize(h, n_iter=400, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_sc = official_score(h)
    velocity = np.zeros(n)
    mu = 0.85
    for i in range(n_iter):
        V = compute_overlaps(h)
        vmax = V.max()
        active = V >= vmax - 0.008
        if active.sum() == 0:
            active = np.zeros(len(V), dtype=bool)
            active[np.argmax(V)] = True
        g = np.zeros(n)
        for p in np.where(active)[0]:
            s = p - (n - 1)
            g += grad_single(h, s)
        g /= active.sum()
        alpha = 0.02 / (1.0 + 0.005 * i)
        velocity = mu * velocity - alpha * g
        h_new = project_feasible(h + velocity, target)
        sc = official_score(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_sc

def lp_step(h, V, tol=0.02, max_active=20):
    n = len(h)
    target = n / 2.0
    vmax = V.max()
    active_mask = V >= vmax - tol
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > max_active:
        order = np.argsort(V[active_idx])
        active_idx = active_idx[order[-max_active:]]
    m = len(active_idx)
    if m == 0:
        return h
    G = np.zeros((m, n))
    for i, p in enumerate(active_idx):
        s = p - (n - 1)
        G[i] = grad_single(h, s)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            return h_new
    except Exception:
        pass
    return h

def lp_iterations(h, n_iter=40):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_sc = official_score(h)
    for i in range(n_iter):
        V = compute_overlaps(h)
        tol = 0.015 + 0.003 * i
        h_new = lp_step(h, V, tol=tol, max_active=15 + i)
        sc = official_score(h_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = h_new.copy()
            h = h_new
        else:
            h_new2 = lp_step(h, V, tol=0.005, max_active=10)
            sc2 = official_score(h_new2)
            if sc2 < best_sc - 1e-12:
                best_sc = sc2
                best_h = h_new2.copy()
                h = h_new2
            else:
                break
    return best_h, best_sc

def coord_descent(h, n_rounds=8):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_sc = official_score(h)
    steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for step in steps:
        for _ in range(n_rounds):
            improved = False
            for k in range(n):
                for direction in [1, -1]:
                    h_test = best_h.copy()
                    h_test[k] += direction * step
                    h_test = project_feasible(h_test, target)
                    sc = official_score(h_test)
                    if sc < best_sc - 1e-13:
                        best_h = h_test
                        best_sc = sc
                        improved = True
                        break
            if not improved:
                break
    return best_h, best_sc

def binary_local_search(h, n_swaps=5000, seed=42):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    h_bin = (h >= 0.5).astype(float)
    s = h_bin.sum()
    if s > target:
        ones = np.where(h_bin == 1)[0]
        rng.shuffle(ones)
        for idx in ones[:int(s - target)]:
            h_bin[idx] = 0
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(zeros)
        for idx in zeros[:int(target - s)]:
            h_bin[idx] = 1
    best_h = h_bin.copy()
    best_sc = official_score(h_bin)
    for _ in range(n_swaps):
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_test = h_bin.copy()
        h_test[i] = 0.0
        h_test[j] = 1.0
        sc = official_score(h_test)
        if sc < best_sc - 1e-12:
            h_bin = h_test
            best_sc = sc
            best_h = h_bin.copy()
    return best_h, best_sc

def binary_deterministic_search(h, n_sweeps=3):

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return project_feasible(h_int, n_tgt / 2.0)

def make_initial(n, seed=42):

    return project_feasible(h, n / 2.0)

        try_update(best_h, best_score)
    if not is_done():
        best_h, best_score = coord_descent(best_h, n_rounds=8)
        try_update(best_h, best_score)

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = official_score(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
```
