def get_correlations(h):
    """Return all correlation values scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_of_correlations(h):
    """
    Return gradient matrix G of shape (2n-1, n) where
    G[k, m] = d(r[k]*dx)/d(h[m])
    r[k] = sum_{i=lo}^{hi} h[i] * (1 - h[i + n - 1 - k])
    lo = max(0, k-n+1), hi = min(n-1, k)
    """
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for k in range(K):
        s = n - 1 - k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d(r[k])/d(h[m]) = (1 - h[m+s]) * I(lo<=m<=hi) - h[m-s] * I(lo<=m-s<=hi)

        m_hi2 = hi + s
        # Clamp to valid range

        m_hi2 = min(n - 1, m_hi2)
        for m in range(m_lo2, m_hi2 + 1):
            G[k, m] -= h[m - s]
    G *= dx
    return G

        self.ranges = []
        for k in range(self.K):
            s = self.n - 1 - k
            lo = max(0, k - self.n + 1)
            hi = min(self.n - 1, k)
            m2lo = max(0, lo + s)
            m2hi = min(self.n - 1, hi + s)
            self.ranges.append((s, lo, hi, m2lo, m2hi))
    
    def objective(self, x):
        return x[0]
    
    def objective_jac(self, x):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g
    
    def eq_fun(self, x):
        return np.array([np.sum(x[1:]) - self.target])
    
    def eq_jac(self, x):
        J = np.zeros((1, self.n + 1))
        J[0, 1:] = 1.0
        return J
    
    def ineq_fun(self, x):
        h = x[1:]
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return x[0] - corrs
    
    def ineq_jac(self, x):
        h = x[1:]
        J = np.zeros((self.K, self.n + 1))
        J[:, 0] = 1.0
        for k in range(self.K):
            s, lo, hi, m2lo, m2hi = self.ranges[k]
            if lo <= hi:
                J[k, 1 + lo:1 + hi + 1] -= (1.0 - h[lo + s:hi + 1 + s]) * self.dx
            if m2lo <= m2hi:
                J[k, 1 + m2lo:1 + m2hi + 1] += h[m2lo - s:m2hi + 1 - s] * self.dx
        return J


def run_slsqp(h0, max_iter=500):
    """Run SLSQP optimization on the Erdős problem."""
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0, target)
    t0 = compute_c5(h)
    x0 = np.concatenate([[t0], h])
    
    prob = ErdosSLSQP(h0)
    bounds = [(0.0, None)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac}
    ]
    
    result = minimize(
        prob.objective, x0, method='SLSQP',
        jac=prob.objective_jac,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    
    h_opt = project_feasible(result.x[1:], target)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt


def smooth_max_step(h, alpha, lr, target):
    """One step of smoothed max gradient descent."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    # Stable log-sum-exp gradient
    z = np.exp(alpha * (corr - max_c))
    w = z / np.sum(z)
    # Gradient of the smoothed max
    # d/dh[m] of sum_k w_k * c_k
    # = sum_k w_k * d(c_k)/d(h[m])
    grad = np.zeros(n)
    K = 2 * n - 1
    for k in range(K):
        s = n - 1 - k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if w[k] < 1e-15:
            continue
        if lo <= hi:
            grad[lo:hi+1] += w[k] * (1.0 - h[lo+s:hi+1+s]) * dx
        m2lo = max(0, lo + s)
        m2hi = min(n - 1, hi + s)
        if m2lo <= m2hi:
            grad[m2lo:m2hi+1] -= w[k] * h[m2lo-s:m2hi+1-s] * dx
    h_new = h - lr * grad
    return project_feasible(h_new, target)

def binary_snap(h, target):
    """Snap h to nearest binary values, then adjust for sum."""
    h_bin = (h > 0.5).astype(float)
    # h_bin has sum = number of 1s, which may not equal target
    # Adjust: if sum > target, change some 1s to 0s; if sum < target, change some 0s to 1s
    current_sum = h_bin.sum()
    if current_sum > target:
        # Need to change (current_sum - target) 1s to 0s
        n_to_change = int(np.round(current_sum - target))
        # Change the ones with smallest h values (closest to 0.5 from below)
        ones = np.where(h_bin == 1.0)[0]

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        s_mask = float(mask.sum())
        if s_mask > 0:
            result[mask] += diff / s_mask
            result = np.clip(result, 0.0, 1.0)
    return result
