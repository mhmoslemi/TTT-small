def subgradient_step(h, alpha):
    n = len(h)
    dx = 2.0 / n
    # Compute all lags
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = np.argmax(corr)
    
    # Gradient of the active lag
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        # d/dc_k h[i] where c_k = sum_j h[j]*(1-h[j+shift])
        # = (1-h[i+shift]) if i+shift is valid, plus -h[i-shift] if i-shift is valid
        if 0 <= i + shift < n:
            grad[i] += (1.0 - h[i + shift]) * dx
        if 0 <= i - shift < n:
            grad[i] += (-h[i - shift]) * dx
    
    # Projected gradient step
    h_new = h - alpha * grad
    # Project to feasible set
    h_new = make_feasible(h_new)
    return h_new, k_star

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    # Water-fill projection
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Binary search for shift
    lo, hi = -max(2.0, np.max(z)), max(2.0, -np.min(z))
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-10:
        idx = np.argsort(1.0 - h)
        for i in idx:
            add = min(residual, 1.0 - h[i])
            h[i] += add
            residual -= add
            if residual < 1e-10:
                break
    elif residual < -1e-10:
        idx = np.argsort(h)
        for i in idx:
            rem = min(-residual, h[i])
            h[i] -= rem
            residual += rem
            if residual > -1e-10:
                break
    return h

def solve_mip(n, solver='CBC'):
    """Try to find binary h of length n with sum=h*2/n that minimizes max overlap count."""
    h_var = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    constraints = [cp.sum(h_var) == n // 2]
    
    y_vars = {}
    for d in range(1, n):
        # Lag +d: sum h[i]*(1-h[i+d]) <= t
        y_pos = cp.Variable(n - d)
        y_vars[('pos', d)] = y_pos
        for i in range(n - d):
            idx_j = i + d
            constraints.append(y_pos[i] <= h_var[i])
            constraints.append(y_pos[i] <= 1 - h_var[idx_j])
            constraints.append(y_pos[i] >= h_var[i] - h_var[idx_j])
        constraints.append(cp.sum(y_pos) <= t)
        
        # Lag -d: sum h[i]*(1-h[i-d]) <= t, equivalently sum h[i+d]*(1-h[i]) <= t
        y_neg = cp.Variable(n - d)
