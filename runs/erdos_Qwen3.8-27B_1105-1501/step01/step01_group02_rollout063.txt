import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max_k corr[k] * dx."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_c5(h, temperature=0.05):
    """Compute gradient of smooth max of corr*dx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temperature)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-14:
            continue
        # Index i where h[i] is first factor: valid if 0<=i<n and 0<=i+s<n
        # Contribution to grad[i]: w * (1 - h[i+s])
        # Index i where h[i] is second factor (i+s = j, so i = j-s): contribution -w * h[i]
        a = max(0, -s)
        b = min(n - 1, n - 1 - s)
        if a <= b:
            # For i in [a, b]: first factor is h[i], second is h[i+s]
            grad[a:b+1] += w * (1.0 - h[a+s:b+s+1])
            # Second factor: h[i+s] for i in [a,b], so h[j] for j in [a+s, b+s]
            # grad[j] -= w * h[i] where j = i+s, i in [a,b]
            j_start = a + s
            j_end = b + s
            # j ranges from j_start to j_end, but we need j in [0, n-1]
            js = max(0, j_start)
            je = min(n - 1, j_end)
            if js <= je:
                # For j in [js, je], the corresponding i = j - s is in [js-s, je-s]
                grad[js:je+1] -= w * h[js-s:je-s+1]
    return grad

def convex_pl_min(corr, a, lo, hi):
    """Find d in [lo,hi] minimizing max_k (corr[k] + d*a[k])."""
    # Evaluate at endpoints
    f_lo = np.max(corr + lo * a)
    f_hi = np.max(corr + hi * a)
    
    # Check slope at lo
    active_lo = np.where(corr + lo * a == f_lo)[0]
    slope_lo = np.max(a[active_lo])
    if slope_lo >= 0:
        return lo, f_lo
    
    active_hi = np.where(corr + hi * a == f_hi)[0]
    slope_hi = np.min(a[active_hi])
    if slope_hi <= 0:
        return hi, f_hi
    
    # March from lo
    d = lo
    f = f_lo
    active = active_lo
    for _ in range(200):  # at most 139 iterations needed
        # Current slope (max a among active)
        slope = np.max(a[active])
        if slope >= 0:
            break
        # Find next breakpoint: where another line catches up to the max
        # Current max value: f (all active lines have this value at d)
        # Line k catches up when corr[k] + d'*a[k] = f
        # d' = (f - corr[k]) / a[k] for a[k] > slope (lines that are steeper)
        # We want the smallest d' > d
        candidates = a > slope
        if not np.any(candidates):
            d = hi
            break
        d_candidates = (f - corr[candidates]) / a[candidates]
        d_candidates = d_candidates[d_candidates > d + 1e-15]
        if len(d_candidates) == 0:
            d = hi
            break
        d_next = np.min(d_candidates)
        if d_next >= hi:
            d = hi
            break
        d = d_next
        f = np.max(corr + d * a)
        active = np.where(corr + d * a == f - 1e-12)[0]
        if len(active) == 0:
            active = np.where(corr + d * a >= f - 1e-10)[0]
    
    f = np.max(corr + d * a)
    return d, f
