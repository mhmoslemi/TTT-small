import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_score(h):
    n = len(h)
    return float(np.max(compute_correlate(h)) * 2.0 / n)

def compute_grads(h):
    n = len(h)
    nc = 2 * n - 1
    grads = np.zeros((nc, n))
    for p in range(nc):
        k = p - (n - 1)
        m = n - abs(k)
        if m <= 0:
            continue
        if k >= 0:
            grads[p, k:k+m] += (1.0 - h[:m])
            grads[p, :m] -= h[k:k+m]
        else:
            t = -k
            grads[p, :m] += (1.0 - h[t:t+m])
            grads[p, t:t+m] -= h[:m]
    return grads

def make_feasible(h):
    n = len(h)
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            cap = (1.0 - h[idx]) if residual > 0 else h[idx]
            if cap <= 0:
                break
            adj = min(abs(residual), cap)
            h[idx] += adj if residual > 0 else -adj
            residual -= adj if residual > 0 else -adj
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    best_h = h.copy()
    best_score = compute_score(best_h)
    for _ in range(max_iter):

        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        if len(active) > 35:
            active = active[np.argsort(scores[active])[-35:]]
        if len(active) == 1:
            grad_single = compute_grads(best_h)[active[0]]
            d = -grad_single
            d -= d.mean()
            d = np.clip(d, -1, 1)
            if np.all(np.abs(d) < 1e-10):
                break
        else:
            grads = compute_grads(best_h)
            ag = grads[active]
            n_vars = n + 1
            c_obj = np.zeros(n_vars)
            c_obj[-1] = -1.0
            A_ub = np.hstack([ag, np.ones((len(active), 1))])
            b_ub = np.zeros(len(active))
            A_eq = np.zeros((1, n_vars))
            A_eq[0, :n] = 1.0
            b_eq = np.zeros(1)
            bounds = [(-1.0, 1.0)] * n + [(0.0, None)]
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
            except Exception:
                break
            if not res.success or res.x[n] < 1e-10:
                break
            d = res.x[:n]
        pos = d > 1e-10
        neg = d < -1e-10
        a_max = 1.0
        if np.any(pos):
            a_max = min(a_max, ((1.0 - best_h)[pos] / d[pos]).min())
        if np.any(neg):

        a_max = max(a_max, 1e-8)

            sc = compute_score(h_new)
            if sc < best_score - 1e-13:
                best_h = h_new
                best_score = sc
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def _smooth_obj(x, beta, dx):
    corr = compute_correlate(x) * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def _smooth_grad(x, beta, dx):
    corr = compute_correlate(x) * dx
    m = corr.max()
    w = np.exp(beta * (corr - m))
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w = w / ws
    all_grads = compute_grads(x)
    return dx * (w @ all_grads)

def _smooth_eq(x, n):
    return np.sum(x) - n / 2.0

def _smooth_eq_jac(x, n):
    return np.ones(n)

def smooth_optimize(h, beta, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': _smooth_eq, 'jac': _smooth_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(beta, dx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            return h_new, compute_score(h_new)
    except Exception:
        pass
    return None, None
