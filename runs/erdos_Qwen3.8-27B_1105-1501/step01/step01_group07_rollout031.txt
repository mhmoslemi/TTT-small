    n = len(h)
    shift = k - (n - 1)
    g = np.zeros(n)
    i_lo = max(0, -shift)
    i_hi = min(n - 1, n - 1 - shift)
    # First term: (1-h[j+shift]) for j in [i_lo, i_hi]
    j_range = np.arange(i_lo, i_hi + 1)
    g[j_range] += (1.0 - h[j_range + shift])
    # Second term: -h[j-shift] for j-shift in [i_lo, i_hi], i.e., j in [i_lo+shift, i_hi+shift]

    j_range2 = j_range2[(j_range2 >= 0) & (j_range2 < n)]
    g[j_range2] += -h[j_range2 - shift]
    return g

    n = len(h)
    shift = k - (n - 1)
    g = np.zeros(n)
    i_lo = max(0, -shift)
    i_hi = min(n - 1, n - 1 - shift)
    j_range = np.arange(i_lo, i_hi + 1)
    g[j_range] += (1.0 - h[j_range + shift])
    j_range2 = np.arange(i_lo + shift, i_hi + shift + 1)
    valid = (j_range2 >= 0) & (j_range2 < n)
    j_range2 = j_range2[valid]
    g[j_range2] += -h[j_range2 - shift]

def lp_step(h, corr, active_lags, trust_radius):
    """Solve LP for a trust-region step."""
    n = len(h)
    # Variables: [d_0, ..., d_{n-1}, t]
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: for each active lag, grad @ d - t <= -corr[k]
    A_rows = []
    b_rows = []
    for k in active_lags:
        g = compute_lag_gradient(h, k)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corr[k])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))

    b_eq = np.zeros(1)

        lo = -min(h[j], trust_radius)
        hi = min(1.0 - h[j], trust_radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)


def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def get_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_lag_gradient(h, k):
    n = len(h)
    shift = k - (n - 1)
    g = np.zeros(n)
    i_lo = max(0, -shift)
    i_hi = min(n - 1, n - 1 - shift)
    j_range = np.arange(i_lo, i_hi + 1)
    g[j_range] += (1.0 - h[j_range + shift])
    j_lo2 = i_lo + shift
    j_hi2 = i_hi + shift
    j_lo2 = max(0, j_lo2)
    j_hi2 = min(n - 1, j_hi2)
    if j_lo2 <= j_hi2:
        j_range2 = np.arange(j_lo2, j_hi2 + 1)
        g[j_range2] += -h[j_range2 - shift]
    return g


def compute_smooth_gradient(h, T):
    n = len(h)
    corr = get_correlation(h)
    base = float(np.max(corr))
    ex = np.exp((corr - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = corr >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] > 1e-12:
            g += w[idx] * compute_lag_gradient(h, idx)
    return g, w


def lp_step(h, corr, active_lags, trust_radius):
    n = len(h)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active_lags:
        g = compute_lag_gradient(h, k)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corr[k])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = -min(h[j], trust_radius)
        hi_b = min(1.0 - h[j], trust_radius)
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def pair_search(h, i, j, cur_c, n_pts=35):
    orig_i = float(h[i])
    orig_j = float(h[j])
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None


def coord_search(h, i, cur_c, n_pts=30):
    orig = float(h[i])
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (1.0 - 2.0 * orig) / n_pts if orig > 0 else (1.0 - orig) / n_pts
        h[i] = orig + best_a
        return float(best_s)
    return None


def to_binary(h):
    n = len(h)
    target = n // 2
    if n % 2 != 0:
        target = (n - 1) // 2
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 20:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except Exception:
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: LP trust-region (main phase, 40% of budget)
    p1_end = t0 + budget_s * 0.40
    cur = best_h.copy()
    cur_c = best_c
    trust_radius = 0.12
    stall = 0
    lp_count = 0

    while time.time() < p1_end:
        corr = get_correlation(cur)
        max_corr = float(np.max(corr))
        threshold = max_corr * 0.94
        active = np.where(corr >= threshold)[0]
        if len(active) > 25:
            active = np.argsort(corr)[-25:]
        active = active.astype(int)

        d = lp_step(cur, corr, active, trust_radius)
        if d is not None and float(np.max(np.abs(d))) > 1e-9:
            improved = False
            for alpha in [1.0, 0.75, 0.5, 0.3, 0.15]:
                cand = make_feasible(cur + alpha * d)
                c_new = score_exact(cand)
                if c_new < cur_c - 1e-13:
                    cur = cand
                    cur_c = c_new
                    improved = True
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                    break
            if improved:
                stall = 0
                trust_radius = min(trust_radius * 1.15, 0.35)
            else:
                trust_radius = max(trust_radius * 0.65, 0.005)
                stall += 1
        else:
            trust_radius = max(trust_radius * 0.65, 0.005)
            stall += 1
        lp_count += 1

        if stall > 12:
            T = max(0.005, trust_radius * 0.2)
            g, _ = compute_smooth_gradient(cur, T)
            g_norm = float(np.linalg.norm(g))
            if g_norm > 1e-10:
                direction = -g / g_norm
                for alpha in [0.02, 0.01, 0.005, 0.002]:
                    cand = make_feasible(cur + alpha * direction)
                    c_new = score_exact(cand)
                    if c_new < cur_c - 1e-13:
                        cur = cand
                        cur_c = c_new
                        if cur_c < best_c:
                            best_h, best_c = cur.copy(), cur_c
                        stall = 0
                        break
            if stall > 20:
                cur = make_feasible(cur + rng.normal(0, 0.025, n))
                cur_c = score_exact(cur)
                trust_radius = 0.08
                stall = 0

    # Phase 2: Binary refinement (10% of budget)
    p2_end = t0 + budget_s * 0.50
    b = to_binary(best_h)
    b_c = score_exact(b)
    if b_c < best_c:
        best_h, best_c = b.copy(), b_c
    cur_b = best_h.copy()
    cur_b_c = best_c
    stall_b = 0
    binary_phase = (np.max(best_h) - np.min(best_h)) > 0.3

    while binary_phase and time.time() < p2_end:
        ones = np.where(cur_b > 0.5)[0]
        zeros = np.where(cur_b < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[int(rng.integers(0, len(ones)))])
        j = int(zeros[int(rng.integers(0, len(zeros)))])
        cur_b[i] = 0.0
        cur_b[j] = 1.0
        c_new = score_exact(cur_b)
        if c_new < cur_b_c - 1e-13:
            cur_b_c = c_new
            stall_b = 0
            if cur_b_c < best_c:
                best_h, best_c = cur_b.copy(), cur_b_c
        else:
            cur_b[i] = 1.0
            cur_b[j] = 0.0
            stall_b += 1
        if stall_b > 800:
            break

    # Phase 3: Fine pair/coordinate search (28% of budget)
    p3_end = t0 + budget_s * 0.78
    cur = best_h.copy()
    cur_c = best_c
    stall = 0

    while time.time() < p3_end:
        r = rng.random()
        if r < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=35)
        elif r < 0.80:
            i = int(rng.integers(0, n))
            res = coord_search(cur, i, cur_c, n_pts=30)
        else:
            i = int(rng.integers(0, n - 1))
            res = pair_search(cur, i, i + 1, cur_c, n_pts=40)

        if res is not None:
            cur_c = res
            stall = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            chain = 0
            for _ in range(5):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = pair_search(cur, i2, j2, cur_c, n_pts=25)
                if res2 is not None:
                    cur_c = res2
                    chain += 1
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            stall += 1
            if stall > 250:
                cur = make_feasible(best_h + rng.normal(0, 0.008, n))
                cur_c = score_exact(cur)
                stall = 0

    # Phase 4: Fine LP polish (10% of budget)
    p4_end = t0 + budget_s * 0.88
    cur = best_h.copy()
    cur_c = best_c
    trust_radius = 0.02
    stall = 0

    while time.time() < p4_end and trust_radius > 0.001:
        corr = get_correlation(cur)
        max_corr = float(np.max(corr))
        threshold = max_corr * 0.97
        active = np.where(corr >= threshold)[0]
        if len(active) > 30:
            active = np.argsort(corr)[-30:]
        active = active.astype(int)

        d = lp_step(cur, corr, active, trust_radius)
        if d is not None and float(np.max(np.abs(d))) > 1e-10:
            improved = False
            for alpha in [1.0, 0.5, 0.25, 0.1]:
                cand = make_feasible(cur + alpha * d)
                c_new = score_exact(cand)
                if c_new < cur_c - 1e-13:
                    cur = cand
                    cur_c = c_new
                    improved = True
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                    break
            if improved:
                stall = 0
            else:
                trust_radius *= 0.6
                stall += 1
        else:
            trust_radius *= 0.6
            stall += 1
        if stall > 8:
            break

    # Phase 5: Try alternative resolutions (5% of budget)
    p5_end = deadline
    if time.time() < p5_end:
        n_cur = len(best_h)
        for dn in [10, -10, 14, -14]:
            if time.time() > p5_end - 10:
                break
            nr = n_cur + dn
            if nr < 30 or nr > 150:
                continue
            h_r = resample_h(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(p5_end - 3, time.time() + 50)
            if time.time() < sub_dl:
                trust_r = 0.03
                stall_r = 0
                while time.time() < sub_dl and stall_r < 15:
                    corr_r = get_correlation(h_r)
                    max_r = float(np.max(corr_r))
                    thr_r = max_r * 0.95
                    act_r = np.where(corr_r >= thr_r)[0]
                    if len(act_r) > 25:
                        act_r = np.argsort(corr_r)[-25:]
                    d_r = lp_step(h_r, corr_r, act_r.astype(int), trust_r)
                    if d_r is not None and float(np.max(np.abs(d_r))) > 1e-9:
                        imp = False
                        for alpha in [1.0, 0.5, 0.25]:
                            cand_r = make_feasible(h_r + alpha * d_r)
                            c_new_r = score_exact(cand_r)
                            if c_new_r < c_r - 1e-13:
                                h_r = cand_r
                                c_r = c_new_r
                                imp = True
                                break
                        if not imp:
                            trust_r *= 0.6
                            stall_r += 1
                    else:
                        trust_r *= 0.6
                        stall_r += 1
            if c_r < best_c:
                best_h, best_c, n = h_r, c_r, len(h_r)

    # Final: short pair search + gradient polish
    if time.time() < deadline - 3:
        cur = best_h.copy()
        cur_c = best_c
        n = len(cur)
        for _ in range(200):
            if time.time() > deadline - 3:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=30)
            if res is not None:
                cur_c = res
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c

    # Final smooth gradient step
    if time.time() < deadline - 2:
        T = 0.003
        g, _ = compute_smooth_gradient(best_h, T)
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-10:
            for alpha in [0.005, 0.002, 0.001, 0.0005]:
                cand = make_feasible(best_h - alpha * g / g_norm)
                c_new = score_exact(cand)
                if c_new < best_c:
                    best_h = cand
                    best_c = c_new
                    break

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
