def compute_c5(h):
    n = len(h)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0

def smooth_max_grad(h, temperature):
    n = len(h)
    ov = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(ov)
    sh = np.clip((ov - m) / temperature, -50.0, 50.0)
    ev = np.exp(sh)
    w = ev / np.sum(ev)
    val = m + temperature * np.log(np.sum(ev))
