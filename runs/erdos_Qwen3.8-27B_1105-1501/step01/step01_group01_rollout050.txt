def coordinate_optimize(h, p, lam):
    """Find optimal value of h[p] given others fixed, using linearization."""
    n = len(h)
    dx = 2.0 / n
    # Compute current correlation values
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # For each lag k, compute a_k = dc_k/dh[p] and C_k = c_k - a_k*h[p]
    a = np.zeros(2*n - 1)
    C = np.zeros(2*n - 1)
    s0 = h[p]
    
    for k in range(2*n - 1):
        s = k - (n - 1)
        if s == 0:
            ak = 1.0 - 2.0 * s0
        else:
            ak = 0.0
            qp = p + s
            if 0 <= qp < n:
                ak += (1.0 - h[qp])
            qp2 = p - s
            if 0 <= qp2 < n:
                ak -= h[qp2]
        a[k] = ak
        C[k] = corr[k] - ak * s0
    
    # f(t) = max_k (a_k * t + C_k) * dx + lam * t
    # = max_k ((a_k + lam/dx) * t + C_k) * dx  (wait, no)
    # f(t) = [max_k (a_k * t + C_k)] * dx + lam * t
    
    # To find min over [0,1]:
    # Check t=0 and t=1
    candidates = [0.0, 1.0]
    
    # Pairwise crossings
    for k in range(2*n - 1):
        for l in range(k+1, 2*n - 1):
            da = a[l] - a[k]
            if abs(da) < 1e-14:
                continue
            t = (C[k] - C[l]) / da  # a[k]*t + C[k] = a[l]*t + C[l]
            if 0.0 < t < 1.0:
                candidates.append(t)
    
    # Evaluate f at each candidate
    best_t = candidates[0]
    best_val = np.inf
    for t in candidates:
        vals = a * t + C
        f = np.max(vals) * dx + lam * t
        if f < best_val - 1e-15:
            best_val = f
            best_t = t
    return best_t

        w = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s_val = k - (n - 1)
        if s_val == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            for p in range(n):
                c = 0.0
                q = p + s_val
                if 0 <= q < n:
                    c += 1.0 - h[q]
                q2 = p - s_val
                if 0 <= q2 < n:
                    c -= h[q2]
                grad[p] += wk * c
    return grad

def smooth_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = beta * (corr - corr.max())
    if np.max(shifted) > 500:
        return corr.max() + np.log(len(corr)) / beta
    return corr.max() + np.log(np.sum(np.exp(shifted))) / beta

def projected_gradient_opt(h0, beta, n_iter, lr_init=0.1, momentum=0.9):
    """Projected gradient with Nesterov momentum."""
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c = smooth_value(h, beta)
    
    for it in range(n_iter):
        # Nesterov step
        h_half = h + momentum * v
        g = compute_grad_smooth(h_half, beta)
        v = momentum * v - lr_init / (1 + it * 0.01) * g
        h_new = h + v
        h_new = project_feasible(h_new)
        
        c = smooth_value(h_new, beta)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    
    # Also check the exact score
    exact_c = eval_score(best_h)
    if exact_c < eval_score(h0):
        return best_h
    return h0

def coordinate_descent(h0, n_passes=10, lam=0.0):
    """Coordinate descent with exact 1D optimization."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    
    for _pass in range(n_passes):
        for p in range(n):
            # Compute a_k and C_k for all lags
            corr = np.correlate(h, 1.0 - h, mode='full')
            a = np.zeros(2 * n - 1)
            C = np.zeros(2 * n - 1)
            hp = h[p]
            
            for k in range(2 * n - 1):
                sv = k - (n - 1)
                if sv == 0:
                    ak = 1.0 - 2.0 * hp
                else:
                    ak = 0.0
                    q = p + sv
                    if 0 <= q < n:
                        ak += 1.0 - h[q]
                    q2 = p - sv
                    if 0 <= q2 < n:
                        ak -= h[q2]
                a[k] = ak
                C[k] = corr[k] - ak * hp
            
            # Find min of max_k (a_k * t + C_k) * dx + lam * t over [0,1]
            candidates = [0.0, 1.0]
            for k in range(2 * n - 1):
                for l in range(k + 1, 2 * n - 1):
                    da = a[l] - a[k]
                    if abs(da) < 1e-14:
                        continue
                    t_cross = (C[k] - C[l]) / da
                    if -0.01 <= t_cross <= 1.01:
                        candidates.append(np.clip(t_cross, 0.0, 1.0))
            
            best_t = hp
            best_val = np.inf
            for t in candidates:
                if t < -0.001 or t > 1.001:
                    continue
                t = np.clip(t, 0.0, 1.0)
                vals = a * t + C
                f = np.max(vals) * dx + lam * t
                if f < best_val - 1e-15:
                    best_val = f
                    best_t = t
            
            h[p] = best_t
        
        h = project_feasible(h)
    
    return h
