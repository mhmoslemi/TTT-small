def make_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    target = 0.5 * n
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-14:
                break

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)

def lse_obj(h, beta):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def lse_grad(h, beta):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
