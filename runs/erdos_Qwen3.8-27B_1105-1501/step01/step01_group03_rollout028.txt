    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def symmetry_transform(h):
    return make_feasible(1.0 - h[::-1])

def solve_binary_milp(n, time_limit=45):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    if n % 2 != 0 or n < 20 or n > 52:
        return None
    half = n // 2
    h_var = cp.Variable(n, boolean=True)
    t_var = cp.Variable()
    z_vars = {}
    for k in range(-(n-1), n):
        i_start = max(0, -k)
        i_end = n - 1 - abs(k)
        for i in range(i_start, i_end + 1):
            j = i + k
            if (i, j) not in z_vars:
                z_vars[(i, j)] = cp.Variable()
    constraints = [sum(h_var) == half]
    for (i, j), z in z_vars.items():
        constraints.append(z <= h_var[i])
        constraints.append(z <= h_var[j])
        constraints.append(z >= h_var[i] + h_var[j] - 1)
    for k in range(-(n-1), n):
        i_start = max(0, -k)
        i_end = n - 1 - abs(k)
        one_sum = sum(h_var[i + k] for i in range(i_start, i_end + 1))
        z_sum = sum(z_vars[(i, i + k)] for i in range(i_start, i_end + 1))
        constraints.append(one_sum - z_sum <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP, time_limit=time_limit)
        except Exception:
            try:
                prob.solve(solver=cp.SCIP, time_limit=time_limit)
            except Exception:
                return None
    if h_var.value is None:
        return None
    h_val = np.round(h_var.value).astype(float)
    if abs(np.sum(h_val) - half) > 0.5:
        return None
    return h_val

def epigraph_obj(z):
    return z[-1]

def epigraph_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def epigraph_ineq(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return z[-1] - corr * dx

def epigraph_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n + 1))
    J[:, -1] = 1.0
    for k in range(-(n-1), n):
        i_start = max(0, -k)
        i_end = n - 1 - abs(k)
        idx1 = np.arange(i_start, i_end + 1)
        idx2 = idx1 + k
        J[k + n - 1, idx1] += dx * (1.0 - h[idx2])
        J[k + n - 1, idx2] += dx * (-h[idx1])
    return J

def epigraph_eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n]) - n / 2.0)

def epigraph_eq_jac(z):
    n = len(z) - 1
    g = np.ones(n + 1)
    g[-1] = 0.0
    return g

def solve_epigraph(h0, n, maxiter=300, ftol=1e-12):
    h0 = make_feasible(h0)
    score0 = official_score(h0)
    t0 = max(score0 + 1e-4, 0.1)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.8)]
    cons = [
        {'type': 'ineq', 'fun': epigraph_ineq, 'jac': epigraph_ineq_jac},
        {'type': 'eq', 'fun': epigraph_eq, 'jac': epigraph_eq_jac}
    ]
    try:
        res = minimize(epigraph_obj, z0, jac=epigraph_obj_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x[:n])
    except Exception:
        pass
    return h0

def smooth_obj_val(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    C = corr * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(2 * n - 1) / beta, None
    return cmax + (mk + np.log(ws)) / beta, es / ws

def smooth_obj(x, beta):
    val, _ = smooth_obj_val(x, beta)
    return val

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    _, weights = smooth_obj_val(x, beta)
    if weights is None:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[k + n - 1]
        if abs(w) < 1e-15:
            continue
        i_start = max(0, -k)
        i_end = n - 1 - abs(k)
        idx1 = np.arange(i_start, i_end + 1)
        idx2 = idx1 + k
        grad[idx1] += w * (1.0 - x[idx2])
        grad[idx2] += w * (-x[idx1])
    return grad * dx

def smooth_eq(x):
    return float(np.sum(x) - len(x) / 2.0)

def smooth_eq_jac(x):
    return np.ones(len(x))

def solve_smooth(h0, n, beta, maxiter=300, ftol=1e-12):
    h0 = make_feasible(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0

def greedy_minimax(h0, n, max_iter=2000, tol=1e-10):
    h = make_feasible(h0)
    best_h = h.copy()
    best_score = official_score(h)
    dx = 2.0 / n
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx
        cmax = np.max(C)
        if cmax < best_score:
            best_score = cmax
            best_h = h.copy()
        active_thresh = max(cmax - 0.002, cmax * (1 - 0.01))
        active = np.where(C >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            i_start = max(0, -k)
            i_end = n - 1 - abs(k)
            idx1 = np.arange(i_start, i_end + 1)
            idx2 = idx1 + k
            grad[idx1] += (1.0 - h[idx2])
            grad[idx2] += (-h[idx1])
        grad /= float(len(active))
        grad *= dx
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-12:
            break
        alpha = 2.0 / gnorm
        improved = False
        for _ in range(60):
            z = h - alpha * grad
            h_new = make_feasible(z)
            new_score = official_score(h_new)
            if new_score < cmax - 1e-12:
                h = h_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            alpha = 0.5 / gnorm
            for _ in range(30):
                z = h - alpha * grad
                h_new = make_feasible(z)
                new_score = official_score(h_new)
                if new_score < cmax:
                    h = h_new
                    break
                alpha *= 0.5
            if alpha < 1e-8:
                break
    return best_h, best_score

def random_perturb(h, sigma, rng):
    noise = rng.normal(0.0, sigma, size=len(h))
    return make_feasible(h + noise)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None
    
    if parent is not None:
        parent = make_feasible(parent)
        parent_n = len(parent)
    else:
        parent_n = 60
        parent = make_feasible(np.full(parent_n, 0.5))
    
    best_h = parent.copy()
    best_score = official_score(best_h)
    
    def try_update(h_cand):
        nonlocal best_h, best_score
        if h_cand is None:
            return
        h_cand = make_feasible(h_cand)
        if not np.all(np.isfinite(h_cand)):
            return
        sc = official_score(h_cand)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc
    
    # Phase 1: Try MILP for binary solutions at even n values
    milp_candidates = []
    for n_try in [48, 50, 52]:
        if time.time() > deadline - 120:
            break
        h_bin = solve_binary_milp(n_try, time_limit=30)
        if h_bin is not None:
            sc = official_score(h_bin)
            milp_candidates.append((sc, n_try, h_bin))
            # Interpolate to parent_n and other lengths
            for tgt_n in [parent_n, parent_n + 2, parent_n - 2]:
                if 20 <= tgt_n <= 80:
                    h_interp = interp_to_length(h_bin, n_try, tgt_n)
                    try_update(h_interp)
    
    # Sort MILP candidates by score
    milp_candidates.sort(key=lambda x: x[0])
    
    # Phase 2: Continuous optimization from multiple starting points
    start_points = []
    start_points.append(('parent', parent, parent_n))
    for (sc, n_try, h_bin) in milp_candidates[:3]:
        start_points.append(('milp_{}'.format(n_try), h_bin, n_try))
    
    # Try different target lengths
    target_ns = []
    for dn in [0, -2, 2, -4, 4, -6, 6]:
        nl = parent_n + dn
        if 20 <= nl <= 80 and nl not in target_ns:
            target_ns.append(nl)
    target_ns.sort()
    
    # For each target length, optimize from multiple starts
    for n_tgt in target_ns:
        if time.time() > deadline - 60:
            break
        for (name, h_src, n_src) in start_points:
            if time.time() > deadline - 30:
                break
            h_init = interp_to_length(h_src, n_src, n_tgt)
            
            # Greedy minimax
            h_g, sc_g = greedy_minimax(h_init, n_tgt, max_iter=1500, tol=1e-11)
            try_update(h_g)
            
            # Epigraph SLSQP from greedy result
            h_e = solve_epigraph(h_g, n_tgt, maxiter=250, ftol=1e-12)
            try_update(h_e)
            
            # More greedy from epigraph result
            h_g2, sc_g2 = greedy_minimax(h_e, n_tgt, max_iter=1000, tol=1e-11)
            try_update(h_g2)
    
    # Phase 3: Log-sum-exp smoothing with increasing beta
    if time.time() < deadline - 40:
        best_n = len(best_h)
        h_cur = best_h.copy()
        for beta in [50, 200, 800, 3000]:
            if time.time() > deadline - 20:
                break
            maxiter = max(100, min(400, int(30000 / best_n)))
            h_cur = solve_smooth(h_cur, best_n, beta, maxiter=maxiter, ftol=1e-13)
            try_update(h_cur)
        # Final greedy polish
        h_g3, _ = greedy_minimax(h_cur, best_n, max_iter=2000, tol=1e-12)
        try_update(h_g3)
    
    # Phase 4: Random restarts around best solution
    if time.time() < deadline - 30:
        best_n = len(best_h)
        for sigma in [0.01, 0.03, 0.05, 0.08]:
            if time.time() > deadline - 15:
                break
            for _ in range(3):
                if time.time() > deadline - 10:
                    break
                h_pert = random_perturb(best_h, sigma, rng)
                h_g, _ = greedy_minimax(h_pert, best_n, max_iter=800, tol=1e-11)
                try_update(h_g)
                if time.time() > deadline - 10:
                    break
                h_e = solve_epigraph(h_g, best_n, maxiter=150, ftol=1e-12)
                try_update(h_e)
    
    # Phase 5: Symmetry averaging and final polish
    if time.time() < deadline - 10:
        h_sym = make_feasible(0.5 * (best_h + symmetry_transform(best_h)))
        try_update(h_sym)
        h_g4, _ = greedy_minimax(best_h, len(best_h), max_iter=3000, tol=1e-12)
        try_update(h_g4)
        h_e2 = solve_epigraph(best_h, len(best_h), maxiter=300, ftol=1e-13)
        try_update(h_e2)
    
    # Final: ensure feasibility and return
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score):
        best_h = parent.copy()
        final_score = official_score(best_h)
    
    return best_h, final_score, len(best_h)
