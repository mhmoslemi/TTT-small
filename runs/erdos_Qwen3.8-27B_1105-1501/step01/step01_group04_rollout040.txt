A practical approach:
1. Start with δ = 0
2. Find k* = argmax(c + δ*d)
3. If d[k*] < 0: the function is decreasing at δ, so increase δ. The next kink is at the intersection of line k* with the current max. Set δ = the smallest δ' > δ such that some other line exceeds line k*.
4. If d[k*] > 0: decrease δ similarly.
5. If d[k*] = 0 or the active set contains a line with d=0: we're at the minimum.

This is the "active set" method for 1D piecewise linear optimization. It converges in at most n steps (each step activates a new line). But in practice, it converges much faster.

        if s > target:
            lo = mid
        else:
            hi = mid

    n = len(h)
    m = np.arange(-(n-1), n)

    for i in range(n):

        v1 = (idx1 >= 0) & (idx1 < n)
        A[i, v1] += (1.0 - h[idx1[v1]])
        idx2 = i + m
        v2 = (idx2 >= 0) & (idx2 < n)
        A[i, v2] -= h[idx2[v2]]

        else:
            lo = d1

                if abs(s - target) > 1e-10:
                    h = project(h, target)
                continue
            else:
                # Try random sweep
                move, improve = random_sweep(h, corr, A, corr_max, 30000, seed + no_improve * 17, time_limit)
        
        if move is None or improve > -1e-12:
            # Stuck - restart
            if no_improve > 100:
                rng = np.random.RandomState(seed + no_improve * 3)
                scale = 0.02 + 0.03 * min(no_improve // 200, 3)
                h = best_h + scale * rng.randn(n)
                h = project(h, target)
                no_improve = 0
                continue
            else:
                rng = np.random.RandomState(seed + no_improve * 7 + 1)
                k = rng.randint(2, 8)
                idx = rng.choice(n, size=k, replace=False)
                h[idx] += 0.015 * rng.randn(k)
                h = project(h, target)
                continue
        
        i, j, delta = move
        h[i] += delta
        h[j] -= delta
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) > 1e-10:
            h = project(h, target)
    
    return best_h, best_c5


def anneal_binary(h0, time_budget, seed=42):
    """Simulated annealing on near-binary configurations."""
    n = len(h0)
    target = n / 2.0
    h = h0.astype(float).copy()
    # Snap to near-binary: round to {0, 0.5, 1} preserving sum
    h_rounded = np.where(h > 0.75, 1.0, np.where(h > 0.25, 0.5, 0.0))
    # Fix sum
    diff = target - np.sum(h_rounded)
    indices = np.where((h_rounded == 0.5) & (h > 0.3))[0]
    if diff > 0 and len(indices) > 0:
        idx = indices[0]
        h_rounded[idx] = min(h_rounded[idx] + diff, 1.0)
    elif diff < 0 and len(indices) > 0:
        idx = indices[0]
        h_rounded[idx] = max(h_rounded[idx] + diff, 0.0)
    h = project(h_rounded, target)
    
    best_h = h.copy()
    best_c5, _ = compute_c5_and_corr(h)
    t0 = time.time()
    time_limit = t0 + time_budget
    rng = np.random.RandomState(seed)
    
    T = 0.005
    T_min = 0.0001
    
    while time.time() < time_limit:
        c5, corr = compute_c5_and_corr(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Random swap: pick one 1-ish and one 0-ish, swap
        ones = np.where(h > 0.6)[0]
        zeros = np.where(h < 0.4)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = rng.choice(ones)
        j = rng.choice(zeros)
        delta = h[i] - h[j]  # transfer this amount
        if abs(delta) < 1e-6:
            continue
        
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = np.clip(h_new, 0.0, 1.0)
        
        c5_new, _ = compute_c5_and_corr(h_new)
        delta_c5 = c5_new - c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            h = h_new
        
        T = max(T * 0.9995, T_min)
    
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_init = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 1: Paired optimization on initial (70% budget)
    if h_init is not None:
        n0 = len(h_init)
        alloc = min(remaining() * 0.65, 700.0)
        if alloc > 30:
            h_best, c5_best = optimize_pairs(h_init, alloc, seed)
            best_h = h_best
            best_c5 = c5_best
            best_n = n0

    # Phase 2: Annealing for diversity (15% budget)
    if best_h is not None and remaining() > 40:
        alloc = min(remaining() * 0.15, 120.0)
        h_ann, c5_ann = anneal_binary(best_h, alloc, seed + 100)
        if c5_ann < best_c5:
            best_c5 = c5_ann
            best_h = h_ann

    # Phase 3: Final polish with paired optimization (remaining budget)
    if best_h is not None and remaining() > 20:
        alloc = remaining() - 5
        h_best, c5_best = optimize_pairs(best_h, alloc, seed + 500)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best

    # Phase 4: Try other n values if time
    if remaining() > 50:
        for n_trial in [199, 127]:
            if remaining() < 35:
                break
            rng = np.random.RandomState(seed + n_trial * 11)
            h_rand = np.zeros(n_trial)
            half = n_trial // 2
            if n_trial % 2 == 1:
                h_rand[half] = 0.5
            h_rand[:half] = 1.0
            h_rand = h_rand[rng.permutation(n_trial)]
            alloc = min(remaining() * 0.2, 80.0)
            if alloc < 15:
                break
            h_best, c5_best = optimize_pairs(h_rand, alloc, seed + n_trial)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best
                best_n = n_trial

    if best_h is None:
        n = 99
        best_h = np.zeros(n)
        best_h[:49] = 1.0
        best_h[49] = 0.5
        best_n = n
        best_c5, _ = compute_c5_and_corr(best_h)

    # Final verification and projection
    target = best_n / 2.0
    best_h = project(best_h, target)
    best_c5, _ = compute_c5_and_corr(best_h)

    return best_h, best_c5, best_n
