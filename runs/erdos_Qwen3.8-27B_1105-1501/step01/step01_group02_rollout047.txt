def pair_line_search(h, i, j, n_coarse=16, n_fine=40):
    """Two-stage line search for pair exchange i↔j.
    Returns (h_new, c5_improved)."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for src, dst in [(i, j), (j, i)]:

            h[dst] -= t

                best_t = t
                best_h = h.copy()

            continue

        margin = t_max / (n_coarse - 1)

        t_hi = min(t_max, best_t + margin)

            h[src] -= best_t
            h[dst] += best_t

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1.0

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass

    if h0 is not None and len(h0) > 0:
        n = len(h0)
        target = n / 2.0
        h0 = project_onto_constraint(h0, target)
        c5_0 = compute_c5(h0)
        best_c5 = c5_0
        best_h = h0.copy()
    else:
        n = 70
        target = n / 2.0
        h0 = make_h_blocks(n, 4, seed)
        c5_0 = compute_c5(h0)
        best_c5 = c5_0
        best_h = h0.copy()

    # Phase 1: Moderate grid full sweeps (40% budget)
    ph1_end = t0 + budget_s * 0.40
    h = best_h.copy()
    sweep_count = 0
    while time.time() < ph1_end:
        c5, improved = full_sweep(h, 15, 50, False, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        sweep_count += 1
        if improved == 0:
            frac = (time.time() - t0) / budget_s
            pert = 0.04 * (1.0 - 0.5 * frac)
            h = best_h + rng.randn(n) * pert
            h = project_onto_constraint(h, target)

    # Phase 2: Finer grid (30% budget)
    ph2_end = t0 + budget_s * 0.70
    h = best_h.copy()
    while time.time() < ph2_end:
        c5, improved = full_sweep(h, 25, 70, False, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if improved == 0:
            h = best_h + rng.randn(n) * 0.015
            h = project_onto_constraint(h, target)

    # Phase 3: Very fine + random order (20% budget)
    ph3_end = t0 + budget_s * 0.90
    h = best_h.copy()
    while time.time() < ph3_end:
        c5, improved = full_sweep(h, 30, 90, True, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if improved == 0:
            h = best_h + rng.randn(n) * 0.008
            h = project_onto_constraint(h, target)

    # Phase 4: Try other n values (10% budget)
    remaining = t0 + budget_s - 8.0 - time.time()
    for n_try in [50, 60, 80, 90, 100]:
        if remaining < 40:
            break
        sub_budget = min(remaining * 0.4, 60)
        target_try = n_try / 2.0
        inits = [
            make_h_blocks(n_try, 3, seed),
            make_h_blocks(n_try, 5, seed + 1),
            make_h_oscillating(n_try, 3, 7),
            make_h_random(n_try, seed + 2),
        ]
        best_local = None
        best_local_c5 = 1.0
        for h_init in inits:
            c5_init = compute_c5(h_init)
            if c5_init < best_local_c5:
                best_local_c5 = c5_init
                best_local = h_init.copy()
        if best_local is None:
            continue
        t_sub = time.time()
        h_try = best_local.copy()
        while time.time() - t_sub < sub_budget:
            c5_try, improved = full_sweep(h_try, 20, 60, False, rng)
            if c5_try < best_local_c5:
                best_local_c5 = c5_try
                best_local = h_try.copy()
            if improved == 0:
                h_try = best_local + rng.randn(n_try) * 0.02
                h_try = project_onto_constraint(h_try, target_try)
        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local.copy()
        remaining = t0 + budget_s - 8.0 - time.time()

    # Final polish
    if best_h is not None:
        n_final = len(best_h)
        target_final = n_final / 2.0
        best_h = project_onto_constraint(best_h, target_final)
        if time.time() < t0 + budget_s - 3.0:
            c5, _ = full_sweep(best_h, 30, 100, False, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = project_onto_constraint(best_h, target_final)
        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
        return best_h, best_c5, n_final

    # Fallback
    n = 70
    h = make_h_blocks(n, 4, seed)
    h = project_onto_constraint(h, n / 2.0)
    return h, compute_c5(h), n
