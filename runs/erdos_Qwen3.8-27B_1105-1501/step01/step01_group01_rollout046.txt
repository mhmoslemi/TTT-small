```python
def binary_local_search(h, max_iter, rng):
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    def compute_score(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr)
    
    current_score = compute_score(h)
    best_h = h.copy()
    best_score = current_score
    
    for it in range(max_iter):
        a = rng.choice(ones)
        b = rng.choice(zeros)
        
        # Try swap
        h[a] = 0
        h[b] = 1
        new_score = compute_score(h)
        
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[a] = 1
            h[b] = 0
    
    return best_h, best_score
```

```python
        time_for_n = max(15.0, min(120.0, (deadline - 40 - time.time()) / len(n_list)))
        max_sa_iters = int(time_for_n * 35000)  # ~35k iterations/sec estimate
        max_sa_iters = max(max_sa_iters, 20000)

        # Multiple initializations
        h_interp = interp_to_n(parent, n)
        inits = [binary_init(h_interp, n)]
        for sigma in [0.15, 0.3, 0.5]:
            h_pert = h_interp + sigma * rng.standard_normal(n)
            inits.append(binary_init(h_pert, n))
        # Random init
        idx_rand = rng.permutation(n)
        h_rand = np.zeros(n)
        h_rand[idx_rand[:n // 2]] = 1.0
        inits.append(h_rand)

        best_bin = None
        best_bin_e = np.inf

        for h_init in inits:
            if time.time() > deadline - 30:
                break
            iters = max(10000, max_sa_iters // len(inits))
            h_sa, e_sa = sa_run(h_init, rng, iters)
            if e_sa < best_bin_e:
                best_bin_e = e_sa
                best_bin = h_sa

        if best_bin is None:
            continue

        # Polish best binary
        h_pol = make_feasible(best_bin)
        s_pol = score_exact(h_pol)
        if time.time() < deadline - 15:
            h_pol, s_pol = full_polish(h_pol, deadline - 5)

        if s_pol < best_score - 1e-12:
            best_h, best_score = h_pol, s_pol

    # Final polish
    if time.time() < deadline - 3:
        h_fin, s_fin = full_polish(best_h, deadline)
        if s_fin < best_score:
            best_h, best_score = h_fin, s_fin

    best_h = make_feasible(best_h)
    final_score = score_exact(best_h)
    return (best_h, final_score, len(best_h))
```
