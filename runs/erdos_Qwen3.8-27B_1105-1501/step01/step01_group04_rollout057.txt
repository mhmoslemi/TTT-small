d(c_k)/d(h[m]) = (1-h[m-k]) * I(0<=m-k<n) - h[m+k] * I(0<=m+k<n)

And for the full gradient:
grad[m] = sum_{k=-(n-1)}^{n-1} w_k * [(1-h[m-k]) * I(0<=m-k<n) - h[m+k] * I(0<=m+k<n)]

where w_k are the softmax weights (indexed by k from -(n-1) to n-1).

This is O(n^2) to compute, which is fine for n=99.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraints(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}."""
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_value_and_grad(h, alpha):
    """Compute smoothed max of correlations and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    # Use a numerically stable softmax
    shifted = alpha * (corr - max_c)
    # Clip to avoid overflow (though max_c subtraction should handle it)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    log_sum = np.log(sum_exp)
    value = max_c + log_sum / alpha
    
    weights = exp_c / sum_exp
    
    # Compute gradient
    # grad[m] = sum_k w[k] * [(1-h[m-k]) * I(0<=m-k<n) - h[m+k] * I(0<=m+k<n)]
    # where k = idx - (n-1) for idx in 0..2n-2
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        # (1-h[m-k]) * I(0<=m-k<n): m-k in [0,n-1] means m in [k, k+n-1], intersected with [0, n-1]
        # So m ranges from max(0, k) to min(n-1, k+n-1)
        m_lo = max(0, k)
        m_hi = min(n - 1, k + n - 1)
        if m_lo <= m_hi:
            # m-k ranges from m_lo-k to m_hi-k, which is in [0, n-1]
            grad[m_lo:m_hi+1] += w * (1.0 - h[m_lo-k:m_hi-k+1])
        # -h[m+k] * I(0<=m+k<n): m+k in [0,n-1] means m in [-k, n-1-k], intersected with [0, n-1]
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            # m+k ranges from m_lo2+k to m_hi2+k, which is in [0, n-1]
            grad[m_lo2:m_hi2+1] -= w * h[m_lo2+k:m_hi2+k+1]
    
    return value, grad

def backtracking_step(h, grad, target_sum, max_lr=1.0, min_lr=1e-8, alpha_sm=50.0, max_iter=50):
    """Take a gradient descent step with backtracking line search on smoothed max."""
    n = len(h)
    # Compute current smoothed max value
    val0, _ = softmax_value_and_grad(h, alpha_sm)
    cdir = -grad / (np.linalg.norm(grad) + 1e-15)
    lr = max_lr
    for _ in range(max_iter):
        h_new = project_to_constraints(h + lr * cdir, target_sum)
        val_new, _ = softmax_value_and_grad(h_new, alpha_sm)
        # Armijo condition (simplified: just require improvement)
        if val_new < val0 - 1e-10 * lr:
            return h_new, lr
        lr *= 0.5
    return h, lr

def coordinate_descent(h, target_sum, n_steps=5000, seed=42):
    """Exact coordinate descent: try pair moves that reduce C5."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    
    deltas = [0.01, 0.02, 0.05, 0.1, 0.2]

        j = rng.randint(n)

        d = rng.choice(deltas) * (1 if rng.rand() < 0.5 else -1)
        if best_h[i] + d < -1e-10 or best_h[i] + d > 1 + 1e-10:

        h_test = best_h.copy()

        h_test[j] -= d
        h_test = np.clip(h_test, 0, 1)
        c5 = compute_c5(h_test)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_test
    
    return best_h, best_c5

def optimize_from_start(h0, budget, seed=42):

    n = len(h0)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

If I change h[i] += δ and h[j] -= δ:

  - As h[m] with m=i: contributes -δ*(1-h[i]) to c_k (the (1-h[i]) term decreases by δ)
    Wait, c_k = sum h[m+k]*(1-h[m]). If h[i] changes by δ, then:
    - The (1-h[i]) term in the product h[m+k]*(1-h[m]) with m=i changes by -δ, so the term h[i+k]*(1-h[i]) changes by -δ*h[i+k]
    - The h[i] appears in h[m+k] when m+k=i, so m=i-k. The term h[i]*(1-h[i-k]) changes by δ*(1-h[i-k])

  (with appropriate bounds checking)

    """Project onto {x: sum(x)=target_sum, 0<=x<=1} via bisection on lambda."""
    lo, hi = -1.0, 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    """Gradient of smoothed max of correlations w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)

    grad = np.zeros(n)

        if w < 1e-8:

                            i = rng.randint(n)
                            j = rng.randint(n)
            
            if i == j or i < 0 or i >= n or j < 0 or j >= n:
                continue
            src, dst = (i, j) if rng.rand() < 0.5 else (j, i)
            max_d = min(best_h[src], 1.0 - best_h[dst])
            if max_d < 1e-10:
                continue
            d = rng.uniform(0.001, 0.03) * min(1.0, max_d)
            best_h[src] -= d
            best_h[dst] += d
            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5
            else:
                best_h[src] += d
                best_h[dst] -= d
    
    return best_h, best_c5
