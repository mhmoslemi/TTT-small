import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(v, target_sum):
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def max_count(h):
    """Return the maximum correlation value (before dx scaling)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def binary_local_search(h, n, rng, t0, budget_s, sa_rounds=50, sa_temp=0.5):
    """Local search + simulated annealing on binary sequence."""
    assert h.dtype in [np.float64, float]
    assert np.all(h >= -1e-10) and np.all(h <= 1 + 1e-10)
    target = n // 2
    # Ensure exactly target ones
    h = h.copy()
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    # Fix count if needed
    num_ones = int(h.sum())
    if num_ones > target:
        ones = np.where(h == 1)[0]
        to_remove = rng.choice(ones, num_ones - target, replace=False)
        h[to_remove] = 0.0
    elif num_ones < target:
        zeros = np.where(h == 0)[0]
        to_add = rng.choice(zeros, target - num_ones, replace=False)
        h[to_add] = 1.0
    
    current_max = max_count(h)
    best_h = h.copy()
    best_max = current_max
    
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    # Phase 1: Best-improvement local search
    improved = True
    while improved:
        if time.time() - t0 > budget_s * 0.85:
            break
        improved = False
        best_swap = None
        best_new_max = current_max
        
        # Try all swaps
        for i in list(ones):
            for j in list(zeros):
                # Swap h[i] and h[j]
                h[i], h[j] = h[j], h[i]
                new_max = max_count(h)
                h[i], h[j] = h[j], h[i]
                
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            ones.discard(i)
            zeros.add(i)
            zeros.discard(j)
            ones.add(j)
            current_max = best_new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
            improved = True
    
    # Phase 2: Simulated annealing
    if time.time() - t0 < budget_s * 0.9:
        temp = sa_temp
        cooling = 0.95
        no_improve = 0
        max_no_improve = 200
        
        while temp > 0.01 and no_improve < max_no_improve:

            new_max = max_count(h)
            
            delta = new_max - current_max
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                # Accept
                current_max = new_max
                ones.discard(i)
                zeros.add(i)
                zeros.discard(j)
                ones.add(j)
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Reject
                h[i], h[j] = h[j], h[i]
                no_improve += 1
            
            temp *= cooling
    
    # Final local search from best
    h = best_h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    current_max = max_count(h)
    
    improved = True
    while improved:
        if time.time() - t0 > budget_s * 0.85:

        improved = False
        best_swap = None

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def max_raw_corr(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def make_binary(h, n, rng):
    """Convert a continuous h to a near-binary with the right sum."""
    h = h.copy().clip(0, 1)
    target = n / 2.0
    if abs(target - int(target)) < 1e-10:
        # n is even, need exactly target = n/2 ones
        # Sort by value, assign 1 to top n/2, 0 to bottom n/2
        # But this might not be optimal. Better: threshold at 0.5, fix count.
        h[h < 0.5] = 0.0
        h[h >= 0.5] = 1.0
        count = int(h.sum())
        target_int = int(target)
        if count > target_int:
            ones = np.where(h == 1)[0]
            # Remove random excess ones
            to_remove = rng.choice(ones, count - target_int, replace=False)
            h[to_remove] = 0.0
        elif count < target_int:
            zeros = np.where(h == 0)[0]
            to_add = rng.choice(zeros, target_int - count, replace=False)
            h[to_add] = 1.0
    else:
        # n is odd, need sum = (n-1)/2 + 0.5
        # Use (n-1)/2 ones, one 0.5, (n-1)/2 zeros
        h[h < 0.5] = 0.0
        h[h >= 0.5] = 1.0
        count = int(h.sum() + 0.5)  # number of 1s
        target_ones = (n - 1) // 2
        if count > target_ones:
            ones = np.where(h == 1)[0]
            to_remove = rng.choice(ones, count - target_ones - 1, replace=False)  # leave one for the 0.5
            h[to_remove] = 0.0
            # Set one of the remaining ones to 0.5
            remaining_ones = np.where(h == 1)[0]
            h[remaining_ones[0]] = 0.5
        elif count < target_ones:
            # Need more ones and a 0.5

            to_add = rng.choice(zeros, target_ones - count + 1, replace=False)
            h[to_add[:-1]] = 1.0
            h[to_add[-1]] = 0.5
        else:
            # count == target_ones, need to demote one to 0.5
            ones = np.where(h == 1)[0]
            h[ones[0]] = 0.5
    return h
