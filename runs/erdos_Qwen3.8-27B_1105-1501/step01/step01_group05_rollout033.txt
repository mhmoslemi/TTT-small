            h[idx] -= delta

        p = i + k
        g[p] += (1.0 - h[i])
        g[i] += (-h[p])

def select_pairs(v, n, max_pairs, rng):
    """Select pairs sorted by promise = |v[i] - v[j]|."""
    # Build promise matrix
    # For efficiency, compute using outer product
    v_col = v[:, np.newaxis]
    v_row = v[np.newaxis, :]
    promise = np.abs(v_col - v_row)
    np.fill_diagonal(promise, 0.0)
    
    # Get top pairs
    flat_idx = np.argsort(promise.ravel())[::-1]
    pairs = []
    seen = set()
    for fi in flat_idx:
        i = int(fi // n)
        j = int(fi % n)
        if i == j:
            continue
        key = (min(i, j), max(i, j))
        if key in seen:
            continue
        seen.add(key)
        pairs.append((i, j))
        if len(pairs) >= max_pairs:
            break
    
    # Add some random pairs for diversity
    n_random = max(5, max_pairs // 10)
    for _ in range(n_random):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            key = (min(i, j), max(i, j))
            if key not in seen:
                seen.add(key)
                pairs.append((i, j))
    
    return pairs

def slsqp_refine(h, n, time_limit=30.0):
    """SLSQP refinement with active constraints."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    
    # Select top active constraints
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    active_threshold = np.max(scaled) - 0.005
    active_idx = np.where(scaled >= active_threshold)[0]
    # Limit to top M
    if len(active_idx) > 20:
        top_vals = scaled[active_idx]
        order = np.argsort(top_vals)[::-1][:20]
        active_idx = active_idx[order]
    
    n_acts = len(active_idx)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr_x = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr_x[active_idx]
    
    def ineq_jac(x):
        h_x = x[:n]
        j = np.zeros((n_acts, n + 1))
        j[:, n] = 1.0
        for ai, ki in enumerate(active_idx):
            k = int(lags[ki])
            grad_row = corr_grad_lag(h_x, k, n)
            j[ai, :n] = -dx * grad_row
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    
    x0 = np.concatenate([h, [theta0]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)

    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            h[idx] -= delta
    return np.clip(h, 0.0, 1.0)


def corr_grad_lag(h, k, n):
    g = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start >= end:
        return g
    i_arr = np.arange(start, end)
    p_arr = i_arr + k
    g[p_arr] += (1.0 - h[i_arr])
    g[i_arr] += (-h[p_arr])
    return g


def compute_active_direction(h, score):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    T = max(0.003 + 0.015 * score, 0.002)
    theta = np.max(scaled)
    exponents = np.clip((scaled - theta) / T, -50.0, 0.0)
    weights = np.exp(exponents)
    wsum = np.sum(weights)
    if wsum < 1e-15:
        weights = np.ones_like(scaled)
        wsum = float(len(scaled))
    weights /= wsum
    lags = np.arange(-(n - 1), n)
    g = np.zeros(n)
    for idx in range(len(lags)):
        w = weights[idx]
        if w > 0.001:
            g += w * corr_grad_lag(h, int(lags[idx]), n)
    g -= np.mean(g)
    return -g


def pair_minimize(h, i, j, cur_score, n_grid, n_golden):
    n = len(h)
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-10:
        return h, cur_score

    orig_i, orig_j = h[i], h[j]
    best_x = orig_i
    best_s = cur_score

    if n_grid < 3:
        n_grid = 3
    xs = np.linspace(L, U, n_grid)
    grid_s = np.empty(n_grid)
    for gi in range(n_grid):
        h[i] = xs[gi]
        h[j] = S - xs[gi]
        grid_s[gi] = official_score(h)

    bi = int(np.argmin(grid_s))
    if grid_s[bi] < best_s - 1e-14:
        best_s = grid_s[bi]
        best_x = xs[bi]

    if n_grid > 2 and n_golden > 0:
        if bi == 0:
            a, b = L, xs[1]
        elif bi == n_grid - 1:
            a, b = xs[n_grid - 2], U
        else:
            a, b = xs[bi - 1], xs[bi + 1]

        if b - a > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            h[i] = c
            h[j] = S - c
            fc = official_score(h)
            h[i] = d
            h[j] = S - d
            fd = official_score(h)

            if fc < best_s:
                best_s = fc
                best_x = c
            if fd < best_s:
                best_s = fd
                best_x = d

            for _ in range(n_golden):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    h[i] = c
                    h[j] = S - c
                    fc = official_score(h)
                    if fc < best_s:
                        best_s = fc
                        best_x = c
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    h[i] = d
                    h[j] = S - d
                    fd = official_score(h)
                    if fd < best_s:
                        best_s = fd
                        best_x = d

    h[i] = best_x
    h[j] = S - best_x
    final_s = official_score(h)

    if final_s > cur_score + 1e-13:
        h[i] = orig_i
        h[j] = orig_j
        return h, cur_score

    return h, final_s


def select_pairs(v, n, max_pairs, rng):
    v_col = v[:, np.newaxis]
    v_row = v[np.newaxis, :]
    promise = np.abs(v_col - v_row)
    np.fill_diagonal(promise, 0.0)
    flat = promise.ravel()
    order = np.argsort(flat)[::-1]
    pairs = []
    seen = set()
    for fi in order:
        i = int(fi // n)
        j = int(fi % n)
        if i == j:
            continue
        key = (min(i, j), max(i, j))
        if key in seen:
            continue
        seen.add(key)
        pairs.append((i, j))
        if len(pairs) >= max_pairs:
            break
    n_rand = max(5, max_pairs // 15)
    for _ in range(n_rand):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            key = (min(i, j), max(i, j))
            if key not in seen:
                seen.add(key)
                pairs.append((i, j))
    return pairs


def local_pair_optimize(h, n, deadline, rng, seed, max_passes=15):
    h = h.copy()
    score = official_score(h)
    start_t = time.perf_counter()

    for p in range(max_passes):
        if time.perf_counter() > deadline:
            break

        v = compute_active_direction(h, score)
        v_norm = np.max(np.abs(v))
        if v_norm < 1e-12:
            break
        v = v / v_norm

        # Adaptive pair count: more in early passes, fewer in later
        frac = p / max(1, max_passes)
        n_pairs = int(max(60, 250 * (1.0 - 0.5 * frac)))
        n_pairs = min(n_pairs, 350)

        # Adaptive grid/golden: finer in later passes
        if p < max_passes // 3:
            n_grid = 21
            n_golden = 12
        elif p < 2 * max_passes // 3:
            n_grid = 17
            n_golden = 12
        else:
            n_grid = 13
            n_golden = 14

        pairs = select_pairs(v, n, n_pairs, rng)

        passed_score = score
        n_improved = 0

        for (pi, pj) in pairs:
            if time.perf_counter() > deadline:
                break
            h, score = pair_minimize(h, pi, pj, score, n_grid, n_golden)
            if score < passed_score - 1e-13:
                n_improved += 1

        score = official_score(h)

        if n_improved == 0:
            # Basin hopping: try several perturbations
            found_escape = False
            step_sizes = [0.03, 0.05, 0.08, 0.12, 0.02, 0.15]
            for ss in step_sizes:
                if time.perf_counter() > deadline:
                    break
                h_try = h + ss * v + rng.normal(0, 0.01, n)
                h_try = repair_vector(h_try, n)
                s_try = official_score(h_try)
                if s_try < score - 1e-12:
                    h = h_try
                    score = s_try
                    found_escape = True
                    break
            if not found_escape:
                # Try random perturbation
                for _ in range(3):
                    h_try = h + rng.normal(0, 0.04, n)
                    h_try = repair_vector(h_try, n)
                    s_try = official_score(h_try)
                    if s_try < score - 1e-12:
                        h = h_try
                        score = s_try
                        found_escape = True
                        break
            if not found_escape:
                break

    return h, official_score(h)


def slsqp_refine(h, n, time_limit=30.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)

    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    max_val = np.max(scaled)
    active_threshold = max_val - 0.008
    active_idx = np.where(scaled >= active_threshold)[0]
    if len(active_idx) > 18:
        top_vals = scaled[active_idx]
        order = np.argsort(top_vals)[::-1][:18]
        active_idx = active_idx[order]
    n_acts = len(active_idx)

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr_x = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr_x[active_idx]

    def ineq_jac(x):
        h_x = x[:n]
        j = np.zeros((n_acts, n + 1))
        j[:, n] = 1.0
        for ai, ki in enumerate(active_idx):
            k = int(lags[ki])
            grad_row = corr_grad_lag(h_x, k, n)
            j[ai, :n] = -dx * grad_row
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])

    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 80, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass

    return h, initial_score


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def gen_random_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 0:
        h[idx[m_full]] = frac
    return repair_vector(h, n)


def gen_random_continuous(n, rng):
    h = rng.uniform(0, 1, n)
    return repair_vector(h, n)


def gen_step_pattern(n, rng, n_steps):
    if n_steps >= n:
        return gen_random_continuous(n, rng)
    boundaries = sorted(rng.choice(n - 1, size=n_steps - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    vals = rng.uniform(0, 1, n_steps)
    h = np.zeros(n)
    for i in range(n_steps):
        h[boundaries[i]:boundaries[i + 1]] = vals[i]
    return repair_vector(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 12.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    n_parent = 65
    if parent is not None:
        n_parent = len(parent)
        best_h = repair_vector(parent, n_parent)
    else:
        best_h = repair_vector(np.full(65, 0.5), 65)
        n_parent = 65

    best_score = official_score(best_h)
    rng = np.random.default_rng(seed)

    # Build candidate pool
    candidates = []
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h).copy())
    candidates.append((1.0 - best_h[::-1]).copy())

    n_list = sorted(set([n_parent + dn for dn in [-5, -3, 0, 3, 5, 8]
                         if 24 <= n_parent + dn <= 110]))

    for n_try in n_list:
        for off in [0, 1, 2]:
            candidates.append(gen_interpolated(best_h, n_try, off))

    for n_try in n_list[:3]:
        for _ in range(3):
            candidates.append(gen_random_binary(n_try, rng))
        for _ in range(2):
            candidates.append(gen_random_continuous(n_try, rng))
        for _ in range(2):
            candidates.append(gen_step_pattern(n_try, rng, 4))

    # Score and sort
    scored = []
    for h_c in candidates:
        if not np.all(np.isfinite(h_c)):
            continue
        h_c = repair_vector(h_c, len(h_c))
        s = official_score(h_c)
        if np.isfinite(s) and s > 0:
            scored.append((s, h_c))

    scored.sort(key=lambda x: (x[0], len(x[1])))

    if scored and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()

    # Main optimization loop over top candidates
    n_top = min(8, len(scored))

    for ci in range(n_top):
        if time.perf_counter() > deadline - 50:
            break

        h_c = scored[ci][1].copy()
        n_c = len(h_c)

        # Time allocation weighted by rank
        remaining = deadline - time.perf_counter()
        if ci == 0:
            t_alloc = remaining * 0.35
        elif ci == 1:
            t_alloc = remaining * 0.2
        elif ci == 2:
            t_alloc = remaining * 0.13
        else:
            t_alloc = remaining * 0.08
        t_alloc = max(8.0, t_alloc)
        t_dead = time.perf_counter() + t_alloc

        h_opt, s_opt = local_pair_optimize(h_c, n_c, t_dead, rng, seed + ci * 13,
                                           max_passes=12)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()

        # SLSQP polish
        if time.perf_counter() < deadline - 35:
            h_pol, s_pol = slsqp_refine(h_opt, n_c, time_limit=min(20.0,
                max(5.0, (deadline - time.perf_counter()) * 0.15)))
            if s_pol < best_score:
                best_score = s_pol
                best_h = h_pol.copy()

            # Pair pass after SLSQP
            if time.perf_counter() < deadline - 15:
                h_opt2, s_opt2 = local_pair_optimize(h_pol, n_c,
                    time.perf_counter() + 10.0, rng, seed + 200 + ci, max_passes=8)
                if s_opt2 < best_score:
                    best_score = s_opt2
                    best_h = h_opt2.copy()

    # Second round: interpolate current best to nearby sizes
    if time.perf_counter() < deadline - 40:
        best_n = len(best_h)
        tried_ns = set([best_n])
        for dn in [0, 3, -3, 5, -5]:
            n_try = best_n + dn
            if n_try < 24 or n_try > 110 or n_try in tried_ns:
                continue
            tried_ns.add(n_try)
            if time.perf_counter() > deadline - 20:
                break
            h_int = gen_interpolated(best_h, n_try, 0)
            h_opt, s_opt = local_pair_optimize(h_int, n_try,
                time.perf_counter() + min(15.0, (deadline - time.perf_counter()) * 0.4),
                rng, seed + 600, max_passes=8)
            if s_opt < best_score:
                best_score = s_opt
                best_h = h_opt.copy()

    # Final polish
    if time.perf_counter() < deadline - 8:
        h_fin, s_fin = slsqp_refine(best_h, len(best_h), time_limit=6.0)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    if time.perf_counter() < deadline - 4:
        h_fin2, s_fin2 = local_pair_optimize(best_h, len(best_h),
            time.perf_counter() + 3.0, rng, seed + 999, max_passes=5)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2.copy()

    # Final validation
    best_h = repair_vector(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
