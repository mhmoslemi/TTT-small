```python
def min_max_affine(alpha, beta, x_low=0.0, x_high=1.0):
    # alpha, beta: arrays of length m
    # minimize max_i (alpha[i] + beta[i]*x) over x in [x_low, x_high]
    # Find all breakpoints
    candidates = [x_low, x_high]
    for i in range(len(alpha)):
        for j in range(i+1, len(alpha)):
            if abs(beta[i] - beta[j]) < 1e-15:
                continue
            x_cross = (alpha[j] - alpha[i]) / (beta[i] - beta[j])
            if x_low < x_cross < x_high:
                candidates.append(x_cross)
    candidates = np.array(candidates)
    # Evaluate g at each candidate
    best_x = candidates[0]
    best_val = np.inf
    for x in candidates:
        vals = alpha + beta * x
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_x = x
    return best_x, best_val
```

def min_max_affine_fast(alpha, beta, x_low=0.0, x_high=1.0):
    m = len(alpha)
    # Compute all pairwise crossings
    # x_cross(i,j) = (alpha[j] - alpha[i]) / (beta[i] - beta[j])
    # Use broadcasting
    diff_a = alpha[None, :] - alpha[:, None]  # m x m
    diff_b = beta[:, None] - beta[None, :]  # m x m
    # Avoid division by zero
    mask = np.abs(diff_b) > 1e-15
    diff_b_safe = np.where(mask, diff_b, 1.0)
    crossings = diff_a / diff_b_safe
    crossings = np.where(mask, crossings, np.nan)
    # Filter to (x_low, x_high)
    crossings = crossings[np.isfinite(crossings)]
    crossings = crossings[(crossings > x_low) & (crossings < x_high)]
    candidates = np.concatenate([[x_low, x_high], crossings])
    # Evaluate
    # g(x) = max_i (alpha[i] + beta[i] * x)
    # Shape: (len(candidates), m)
    vals = alpha[None, :] + beta[None, :] * candidates[:, None]
    g_vals = np.max(vals, axis=1)
    best_idx = np.argmin(g_vals)
    return candidates[best_idx], g_vals[best_idx]
```

    n = len(h)

def optimize_coordinate_dual(h, p, lam):
    """Find optimal h[p] with Lagrange multiplier for sum constraint."""
    n = len(h)
    dx = 2.0 / n
    # Get beta
    h0 = h.copy(); h0[p] = 0.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    h1 = h.copy(); h1[p] = 1.0

    c = np.correlate(h, 1.0 - h, mode='full')
    alpha = c - h[p] * beta

    # Let a_k = alpha[k]*dx, b_k = beta[k]*dx + lam
    a = alpha * dx
    b = beta * dx + lam
    x_opt = min_max_affine(a, b)

def min_max_affine_1d(alpha, beta, lo=0.0, hi=1.0):
    """Minimize max_i (alpha[i] + beta[i]*x) for x in [lo, hi]."""
    m = len(alpha)
    if m == 0:
        return (lo + hi) / 2
    # Find all pairwise crossings
    # Crossing of lines i and j: x = (alpha[j] - alpha[i]) / (beta[i] - beta[j])
    candidates = [lo, hi]
    for i in range(m):
        for j in range(i+1, m):
            db = beta[i] - beta[j]
            if abs(db) < 1e-15:
                continue
            x_cross = (alpha[j] - alpha[i]) / db
            if lo < x_cross < hi:
                candidates.append(x_cross)
    candidates = np.array(candidates)
    # Evaluate at each candidate
    # g(x) = max_i (alpha[i] + beta[i]*x)
    best_x = lo
    best_val = np.max(alpha + beta * lo)
    for x in candidates[2:]:  # skip lo and hi (already evaluated lo, now check others and hi)
        val = np.max(alpha + beta * x)
        if val < best_val:
            best_val = val
            best_x = x
    # Check hi
    val_hi = np.max(alpha + beta * hi)
    if val_hi < best_val:
        best_val = val_hi
        best_x = hi
    return best_x
```

def min_max_affine_1d_vec(alpha, beta, lo=0.0, hi=1.0):
    m = len(alpha)
    # Compute all pairwise crossings
    # Use broadcasting: (m, m) matrix of crossings
    da = alpha[None, :] - alpha[:, None]  # da[i,j] = alpha[j] - alpha[i]
    db = beta[:, None] - beta[None, :]  # db[i,j] = beta[i] - beta[j]
    mask = np.abs(db) > 1e-15
    db_safe = np.where(mask, db, 1.0)
    crossings = da / db_safe
    crossings = np.where(mask, crossings, np.nan)
    # Flatten and filter
    crossings_flat = crossings.ravel()
    crossings_flat = crossings_flat[np.isfinite(crossings_flat)]
    crossings_flat = crossings_flat[(crossings_flat > lo) & (crossings_flat < hi)]
    candidates = np.concatenate([[lo, hi], crossings_flat])

    vals = alpha[None, :] + beta[None, :] * candidates[:, None]
    g_vals = np.max(vals, axis=1)
    best_idx = np.argmin(g_vals)
    return candidates[best_idx]

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    """Project z to [0,1]^n with sum = target (default n/2)."""
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection on shift
    lo = -max(np.max(z), 0) - 1.0
    hi = 1.0 - min(np.min(z), 0) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:

            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

        return h_src.copy()

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

    """Compute coefficient of h[p] in each correlation output."""
    n = len(h)
    s_vals = np.arange(-(n-1), n)
    ps = p - s_vals
    pp = p + s_vals
    valid_ps = (ps >= 0) & (ps < n)
    valid_pp = (pp >= 0) & (pp < n)
    beta = np.zeros(2*n-1)
    nonzero_s = (s_vals != 0)
    nz_valid_ps = nonzero_s & valid_ps
    nz_valid_pp = nonzero_s & valid_pp
    beta[nz_valid_ps] += (1.0 - h[ps[nz_valid_ps]])
    beta[nz_valid_pp] += (-h[pp[nz_valid_pp]])
    beta[n-1] = 1.0 - h[p]  # s=0 case
    return beta

def min_max_affine_1d(alpha, beta, lo=0.0, hi=1.0):

    m = len(alpha)

    da = alpha[None, :] - alpha[:, None]

    """Run coordinate descent with exact 1D optimization."""

    best_score = official_score(h)
    best_h = h.copy()
    stagnant = 0
    
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        order = rng.permutation(n)
        for p in order:
            beta = get_beta(h, p)
            c = np.correlate(h, 1.0 - h, mode='full')

            x_opt = min_max_affine_1d(alpha, beta)

            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 15:
                # Plateau escape
                h = make_feasible(h + 0.1 * rng.standard_normal(n))
                stagnant = 0
    
    return best_h, best_score

def min_max_affine_1d(alpha, beta, lo=0.0, hi=1.0):

    # Quick check at boundaries
    g_lo = np.max(alpha)  # since x=lo=0
    g_hi = np.max(alpha + beta)  # since x=hi=1

    active_lo = np.abs(alpha - g_lo) < 1e-14
    if np.all(beta[active_lo] >= -1e-14):
        return lo
    active_hi = np.abs(alpha + beta - g_hi) < 1e-14
    if np.all(beta[active_hi] <= 1e-14):
        return hi
    
    # Find all pairwise crossings
    da = alpha[None, :] - alpha[:, None]
    db = beta[:, None] - beta[None, :]

    crossings_flat = crossings_flat[(crossings_flat > lo) & (crossings_flat < hi)]
    if len(crossings_flat) == 0:
        # No internal crossings, minimum is at a boundary
        return lo if g_lo <= g_hi else hi
    candidates = np.unique(crossings_flat)
    vals = alpha[None, :] + beta[None, :] * candidates[:, None]

    if best_val >= g_lo:
        return lo
    if best_val >= g_hi:
        return hi
    return candidates[best_idx]

import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -np.max(z) - 1.0
    hi = 1.0 - np.min(z) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def get_beta(h, p):
    n = len(h)
    s_vals = np.arange(-(n - 1), n)
    ps = p - s_vals
    pp = p + s_vals
    valid_ps = (ps >= 0) & (ps < n)
    valid_pp = (pp >= 0) & (pp < n)
    beta = np.zeros(2 * n - 1)
    nonzero = (s_vals != 0)
    nz_vp = nonzero & valid_ps
    nz_vpp = nonzero & valid_pp
    if np.any(nz_vp):
        beta[nz_vp] += (1.0 - h[ps[nz_vp]])
    if np.any(nz_vpp):
        beta[nz_vpp] += (-h[pp[nz_vpp]])
    beta[n - 1] = 1.0 - h[p]
    return beta

def min_max_affine_1d(alpha, beta, lo=0.0, hi=1.0):
    m = len(alpha)
    g_lo = np.max(alpha)
    g_hi = np.max(alpha + beta)
    active_lo = np.abs(alpha - g_lo) < 1e-13
    if np.all(beta[active_lo] >= -1e-13):
        return lo
    active_hi = np.abs(alpha + beta - g_hi) < 1e-13
    if np.all(beta[active_hi] <= 1e-13):
        return hi
    da = alpha[None, :] - alpha[:, None]
    db = beta[:, None] - beta[None, :]
    mask = np.abs(db) > 1e-14
    db_safe = np.where(mask, db, 1.0)
    crossings = da / db_safe
    crossings = np.where(mask, crossings, np.nan)
    flat = crossings.ravel()
    flat = flat[np.isfinite(flat)]
    flat = flat[(flat > lo) & (flat < hi)]
    if len(flat) == 0:
        return lo if g_lo <= g_hi else hi
    candidates = np.unique(flat)
    vals = alpha[None, :] + beta[None, :] * candidates[:, None]
    g_vals = np.max(vals, axis=1)
    best_idx = np.argmin(g_vals)
    best_val = g_vals[best_idx]
    if best_val >= g_lo - 1e-15:
        return lo
    if best_val >= g_hi - 1e-15:
        return hi
    return candidates[best_idx]

def coord_descent(h0, rng, max_sweeps, t_deadline, plateau_thresh=12):
    n = len(h0)
    h = h0.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_score = official_score(h)
    best_h = h.copy()
    stagnant = 0
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        order = rng.permutation(n)
        for p in order:
            beta = get_beta(h, p)
            alpha = c - h[p] * beta
            x_opt = min_max_affine_1d(alpha, beta)
            x_opt = float(np.clip(x_opt, 0.0, 1.0))
            if abs(x_opt - h[p]) > 1e-14:
                c += (x_opt - h[p]) * beta
                h[p] = x_opt
        h = make_feasible(h)
        c = np.correlate(h, 1.0 - h, mode='full')
        score = official_score(h)
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= plateau_thresh:
                h = make_feasible(h + 0.08 * rng.standard_normal(n))
                c = np.correlate(h, 1.0 - h, mode='full')
                stagnant = 0
    return best_h, best_score

def binary_pattern(n, pattern_type, rng):
    h = np.zeros(n)
    target = n // 2
    if pattern_type == 0:
        h[:target] = 1.0
    elif pattern_type == 1:
        h[::2] = 1.0
        if h.sum() > target:
            h[-2::2] = 0.0
        if h.sum() < target:
            idx = np.where(h == 0)[0]
            h[idx[:target - int(h.sum())]] = 1.0
    elif pattern_type == 2:
        idx = rng.permutation(n)
        h[idx[:target]] = 1.0
    else:
        h = rng.uniform(0, 1, n)
    return make_feasible(h)
