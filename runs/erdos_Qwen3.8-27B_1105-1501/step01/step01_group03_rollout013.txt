import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    # Bisection on shift mu: find mu such that sum(clip(z+mu, 0, 1)) = target
    lo = -1.0 - np.max(z) - 1e-10
    hi = 1.0 - np.min(z) + 1e-10
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any residual
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        # Distribute error evenly among non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:

            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)

                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_C_values(h, n):

    dx = 2.0 / n
    F = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        start = max(0, -k)
        end = min(n, n - k)
        overlap = h[start:end] * (1.0 - h[start+k:end+k])
        F[k + n - 1] = np.sum(overlap) * dx
    return F

def compute_gradient_C(h, n):
    """Compute gradients of all C_k with respect to h.
    Returns (2n-1, n) matrix where row i is ∇C_{-(n-1)+i}"""
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        idx = k + n - 1
        start = max(0, -k)
        end = min(n, n - k)
        # h_i contributes as first factor: ∂(h_i * (1-h_{i+k}))/∂h_i = (1-h_{i+k})
        for i in range(start, end):
            G[idx, i] += dx * (1.0 - h[i+k])
            # h_{i+k} contributes as second factor: ∂(h_i * (1-h_{i+k}))/∂h_{i+k} = -h_i
            G[idx, i+k] += dx * (-h[i])
    return G

def lp_tangent_step(h, n, radius, mu):
    """Solve the LP tangent step. Returns direction d or None."""
    dx = 2.0 / n
    C_vals = compute_C_values(h, n)
    G = compute_gradient_C(h, n)
    n_lags = 2 * n - 1
    
    # Variables: d_0,...,d_{n-1}, p_0,...,p_{n-1}, t
    # Total: 2n+1 variables
    n_var = 2 * n + 1
    d_idx = np.arange(n)
    p_idx = np.arange(n, 2*n)
    t_idx = 2*n
    
    # Objective: min t + mu/n * sum(p_i)
    c_obj = np.zeros(n_var)
    c_obj[p_idx] = mu / n
    c_obj[t_idx] = 1.0
    
    # Inequality constraints: G_j @ d - t <= -C_j for each lag j
    # i.e., G_j @ d + (-1)*t <= -C_j
    A_ub = np.zeros((n_lags, n_var))
    for j in range(n_lags):
        A_ub[j, d_idx] = G[j, :]
        A_ub[j, t_idx] = -1.0
    b_ub = -C_vals
    
    # Absolute value constraints: d_i <= p_i and -d_i <= p_i
    # i.e., d_i - p_i <= 0 and -d_i - p_i <= 0
    A_abs = np.zeros((2*n, n_var))
    b_abs = np.zeros(2*n)
    for i in range(n):
        A_abs[2*i, i] = 1.0      # d_i - p_i <= 0
        A_abs[2*i, n+i] = -1.0
        A_abs[2*i+1, i] = -1.0   # -d_i - p_i <= 0
        A_abs[2*i+1, n+i] = -1.0
    
    A_ub_full = np.vstack([A_ub, A_abs])
    b_ub_full = np.concatenate([b_ub, b_abs])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, d_idx] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        d_lo = max(-radius, -h[i] - 1e-10)
        d_hi = min(radius, 1.0 - h[i] + 1e-10)
        bounds.append((d_lo, d_hi))
    for i in range(n):
        bounds.append((0.0, 2*radius + 1e-10))
    bounds.append((0.0, 2.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full,

            d = res.x[:n].copy()
            # Remove mean to exactly satisfy sum constraint
            d = d - np.mean(d)
            return d
    except Exception:
        pass
    return None

from scipy.optimize import linprog
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(z, n=None):
    if n is None:
        n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    if len(z) != n:
        z = z[:n] if len(z) > n else np.pad(z, (0, n - len(z)))
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-10
    hi = 1.0 - np.min(z) + 1e-10
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            adj = err / cnt
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        err = target - np.sum(h)
        if abs(err) > 1e-14:
            order = np.argsort(np.abs(h - 0.5))
            for i in order:
                if abs(err) < 1e-15:
                    break
                if err > 0 and h[i] < 1.0:
                    d = min(err, 1.0 - h[i])
                    h[i] += d
                    err -= d
                elif err < 0 and h[i] > 0.0:
                    d = min(-err, h[i])
                    h[i] -= d
                    err += d
    return np.clip(h, 0.0, 1.0)


def compute_C_values(h, n):
    dx = 2.0 / n
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        s = max(0, -k)
        e = min(n, n - k)
        F[k + n - 1] = np.sum(h[s:e] * (1.0 - h[s + k:e + k])) * dx
    return F


def compute_all_gradients(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        idx = k + n - 1
        s = max(0, -k)
        e = min(n, n - k)
        G[idx, s:e] += dx * (1.0 - h[s + k:e + k])
        G[idx, s + k:e + k] += -dx * h[s:e]
    return G


def lp_tangent_step(h, n, radius, mu):
    dx = 2.0 / n
    C_vals = compute_C_values(h, n)
    G = compute_all_gradients(h, n)
    n_lags = 2 * n - 1
    n_var = 2 * n + 1
    c_obj = np.zeros(n_var)
    c_obj[n:2 * n] = mu / n
    c_obj[2 * n] = 1.0
    A_ub = np.zeros((n_lags + 2 * n, n_var))
    b_ub = np.zeros(n_lags + 2 * n)
    for j in range(n_lags):
        A_ub[j, 0:n] = G[j, :]
        A_ub[j, 2 * n] = -1.0
        b_ub[j] = -C_vals[j]
    for i in range(n):
        A_ub[n_lags + 2 * i, i] = 1.0
        A_ub[n_lags + 2 * i, n + i] = -1.0
        A_ub[n_lags + 2 * i + 1, i] = -1.0
        A_ub[n_lags + 2 * i + 1, n + i] = -1.0
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        d_lo = max(-radius, -h[i] - 1e-9)
        d_hi = min(radius, 1.0 - h[i] + 1e-9)
        bounds.append((d_lo, d_hi))
    for i in range(n):
        bounds.append((0.0, 2.0 * radius + 1e-9))
    bounds.append((0.0, 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[0:n].copy()
            d = d - np.mean(d)
            return d
    except Exception:
        pass
    return None

def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_feasible(h_src.copy(), n_tgt)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_tgt, n_tgt)

def binarize_threshold(h):
    n = len(h)
    target = n / 2.0
    h_bin = np.zeros(n)
    if n % 2 == 0:
        m = n // 2
        top = np.argsort(h)[::-1][:m]
        h_bin[top] = 1.0
    else:
        m = n // 2
        top = np.argsort(h)[::-1][:m]
        h_bin[top] = 1.0
        rem = np.argsort(h)[::-1][m]
        h_bin[rem] = 1.0
    return project_feasible(h_bin, n)

        h_cand = project_feasible(h_cand, len(h_cand))
        if not np.all(np.isfinite(h_cand)):
            return
        sc = official_score(h_cand)
        if not np.isfinite(sc):
            return
        if sc < best_score - 1e-15:
            best_h = h_cand.copy()
            best_score = sc
        elif abs(sc - best_score) < 1e-12 and len(h_cand) < len(best_h):
            best_h = h_cand.copy()
            best_score = sc

    # Verify gradients (once, quickly)
    max_gerr = verify_gradients(best_h, len(best_h))

    # Phase 1: LP on parent
    dl1 = min(deadline, t_start + 250)
    h1, s1 = lp_local_search(best_h, len(best_h), dl1, mu_list=[0.3, 0.1, 0.01], radius0=0.25)
    try_update(h1)

    # Phase 2: Symmetry
    if time.time() < deadline - 120:
        sym = symmetry_transform(best_h)
        dl2 = min(deadline, time.time() + 150)
        h2, s2 = lp_local_search(sym, len(sym), dl2, mu_list=[0.3, 0.1, 0.01], radius0=0.25)
        try_update(h2)

    # Phase 3: Different lengths
    if time.time() < deadline - 150:
        best_n = len(best_h)
        lengths = []
        for delta in [-6, -4, -2, 2, 4, 6, -8, 8]:
            nl = best_n + delta
            if 28 <= nl <= 95 and nl not in lengths:
                lengths.append(nl)
        lengths.sort(key=lambda x: abs(x - best_n))
        for nl in lengths[:5]:
            if time.time() > deadline - 60:
                break
            h_try = interp_to_length(best_h, nl)
            dl3 = min(deadline, time.time() + 100)
            h_r, s_r = lp_local_search(h_try, nl, dl3, mu_list=[0.2, 0.05, 0.01], radius0=0.2)
            try_update(h_r)

    # Phase 4: Random perturbations + LP
    if time.time() < deadline - 120:
        base_n = len(best_h)
        for trial in range(15):
            if time.time() > deadline - 50:
                break
            sigma = 0.05 + 0.08 * rng.random()
            h_p = random_perturb(best_h, sigma, rng)
            dl4 = min(deadline, time.time() + 60)
            h_r, s_r = lp_local_search(h_p, base_n, dl4, mu_list=[0.5, 0.1, 0.01], radius0=0.15)
            try_update(h_r)

    # Phase 5: Binarize + LP
    if time.time() < deadline - 60:
        h_bin = binarize_threshold(best_h)
        dl5 = min(deadline, time.time() + 50)
        h_r, s_r = lp_local_search(h_bin, len(h_bin), dl5, mu_list=[0.1, 0.01], radius0=0.12)
        try_update(h_r)

    # Phase 6: Cyclic shifts
    if time.time() < deadline - 50:
        bn = len(best_h)
        for shift in [1, 3, bn // 2, bn // 3]:
            if time.time() > deadline - 20:
                break
            h_c = cyclic_shift(best_h, shift)
            dl6 = min(deadline, time.time() + 35)
            h_r, s_r = lp_local_search(h_c, bn, dl6, mu_list=[0.2, 0.05], radius0=0.12)
            try_update(h_r)

    # Phase 7: Cosine-like structured inits
    if time.time() < deadline - 80:
        n_try = len(best_h)
        for period in [0.9, 1.2, 1.5, 2.0, 2.5]:
            if time.time() > deadline - 30:
                break
            for phase in [0.0, np.pi / 3, 2 * np.pi / 3, np.pi]:
                if time.time() > deadline - 15:
                    break
                h_cos = cosine_init(n_try, period, phase, rng)
                dl7 = min(deadline, time.time() + 30)
                h_r, s_r = lp_local_search(h_cos, n_try, dl7, mu_list=[0.3, 0.1], radius0=0.2)
                try_update(h_r)

    # Phase 8: Intensive final LP on best
    if time.time() < deadline - 10:
        bn = len(best_h)
        dl8 = deadline
        h_r, s_r = lp_local_search(best_h, bn, dl8,
                                   mu_list=[0.1, 0.01, 0.001, 0.0],
                                   radius0=0.1, max_iter=5000)
        try_update(h_r)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_score = official_score(best_h)

    if not np.isfinite(final_score) or final_score < 0 or final_score > 0.5:
        if parent is not None:
            best_h = project_feasible(parent.copy(), parent_n)
            final_score = official_score(best_h)
        else:
            best_h = project_feasible(np.full(parent_n, 0.5), parent_n)
            final_score = official_score(best_h)

    if final_score > best_score + 1e-10 and parent is not None:
        best_h = project_feasible(parent.copy(), parent_n)
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
