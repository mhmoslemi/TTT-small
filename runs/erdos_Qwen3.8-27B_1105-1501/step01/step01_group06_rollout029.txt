import numpy as np
import time
from scipy.optimize import linprog

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_to_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_scores_and_grads(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    scores = corr * dx
    
    grads = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        k = i - (n - 1)
        if k >= 0:
            if k == 0:
                grads[i, :] = 1.0 - 2.0 * h
            else:
                # j in [0, n-1-k]: -h[j+k]
                grads[i, 0:n-k] = -h[k:n]
                # j in [k, n-1]: +(1-h[j-k])
                grads[i, k:] += (1.0 - h[0:n-k])
        else:
            kp = -k
            # j in [0, n-1-kp]: +(1-h[j+kp])
            grads[i, 0:n-kp] = (1.0 - h[kp:n])
            # j in [kp, n-1]: -h[j-kp]
            grads[i, kp:] += -h[0:n-kp]
    return scores, grads

def compute_alpha_max(h, p):
    """Maximum alpha such that h + alpha*p stays in [0,1]."""
    alpha_max = 1e10
    for j in range(len(h)):
        if p[j] > 1e-12:
            a = (1.0 - h[j]) / p[j]
            alpha_max = min(alpha_max, a)
        elif p[j] < -1e-12:
            a = h[j] / (-p[j])
            alpha_max = min(alpha_max, a)
    return min(alpha_max, 10.0)

def line_search(h, p, alpha_max, current_c5):
    """Find best alpha in [0, alpha_max] via grid + refinement."""
    if alpha_max < 1e-10:
        return 0.0, current_c5
    
    # Coarse grid
    n_grid = 20
    alphas = np.linspace(0, alpha_max, n_grid)
    best_alpha = 0.0
    best_val = current_c5
    
    for a in alphas:
        h_new = h + a * p
        # Check bounds
        if h_new.min() < -1e-9 or h_new.max() > 1.0 + 1e-9:
            continue
        val = eval_c5(h_new)
        if val < best_val:
            best_val = val
            best_alpha = a
    
    if best_alpha == 0.0:
        return 0.0, best_val
    
    # Refine with golden section around best
    if best_alpha > alphas[0] and best_alpha < alphas[-1]:
        # Find bracket
        idx = int(best_alpha / (alpha_max / (n_grid - 1)))
        lo_a = alphas[max(0, idx - 1)]
        hi_a = alphas[min(n_grid - 1, idx + 1)]
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(30):
            c1 = hi_a - gr * (hi_a - lo_a)
            c2 = lo_a + gr * (hi_a - lo_a)
            f1 = eval_c5(h + c1 * p)
            f2 = eval_c5(h + c2 * p)
            if f1 < f2:
                hi_a = c2
            else:
                lo_a = c1
        best_alpha = (lo_a + hi_a) / 2.0
        best_val = eval_c5(h + best_alpha * p)
    
    return best_alpha, best_val

def lp_optimize(h, t0, deadline):
    """LP-based local minimax optimization."""
    n = len(h)
    target = n / 2.0
    h = project_to_sum(h, target)
    current_c5 = eval_c5(h)
    radius = 0.05
    
    scores, grads = compute_scores_and_grads(h)
    
    for iteration in range(1000):
        if time.time() > deadline:
            break
        
        # Build LP
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize tau
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = grads
        A_ub[:, -1] = -1.0
        b_ub = -scores
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        lb = np.maximum(-radius, -h)
        ub = np.minimum(radius, 1.0 - h)
        bounds = [(lb[j], ub[j]) for j in range(n)]
        bounds.append((-1.0, 1.0))  # tau
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            radius *= 0.5
            if radius < 1e-6:
                break
            continue
        
        p = res.x[:n]
        p_sum = float(np.sum(p))
        if abs(p_sum) > 1e-6:
            p -= p_sum / n  # enforce zero sum numerically
        
        if np.max(np.abs(p)) < 1e-10:
            radius *= 0.5
            if radius < 1e-6:
                break
            continue
        
        # Line search
        alpha_max = compute_alpha_max(h, p)
        alpha, best_val = line_search(h, p, alpha_max, current_c5)
        
        if alpha > 1e-10 and best_val < current_c5 - 1e-12:
            h_new = h + alpha * p
            h_new = project_to_sum(h_new, target)
            new_c5 = eval_c5(h_new)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                scores, grads = compute_scores_and_grads(h)
                if radius < 0.3:
                    radius = min(radius * 1.3, 0.3)
        else:
            radius *= 0.5
            if radius < 1e-6:
                break
    
    return h, current_c5

def interpolate_to(h, n_new):
    """Interpolate h from its current resolution to n_new points."""
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_to_sum(h_new, n_new / 2.0)
