def coord_descent_step(h, p):
    """Optimize coordinate p of h to minimize max_k S_k."""
    n = len(h)
    h = h.copy()
    old_val = h[p]
    
    # Compute all S_k with current h
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # For coordinate p, compute coefficient c_k in each S_k
    # and the "base" value (S_k without h[p]'s contribution)
    # S_k = c_k * h[p] + b_k
    
    coeffs = np.zeros(2 * n - 1)
    for i in range(2 * n - 1):
        k_lag = i - (n - 1)
        # h[p] appears as h[m+k] when m = p-k, valid if 0<=p-k<n and p-k in valid m range
        m1 = p - k_lag
        if 0 <= m1 < n:
            # m1 is valid m for this lag
            m_lo = max(0, -k_lag)
            m_hi = min(n-1, n-1-k_lag)
            if m_lo <= m1 <= m_hi:
                coeffs[i] += (1.0 - h[m1])
        # h[p] appears in (1-h[m]) when m = p, valid if p in valid m range
        m_lo = max(0, -k_lag)
        m_hi = min(n-1, n-1-k_lag)
        if m_lo <= p <= m_hi:
            # h[p+k_lag] must be valid
            p2 = p + k_lag
            if 0 <= p2 < n:
                coeffs[i] += (-h[p2])
    
    base = corr - coeffs * old_val
    
    # Now minimize max_i (coeffs[i] * x + base[i]) for x in [0,1]
    # This is a 1D convex problem
    # Use bisection on the subgradient
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = coeffs * mid + base
        max_val = np.max(vals)
        # Subgradient: sum of coeffs where vals == max_val (approximately)
        active = vals >= max_val - 1e-10
        subgrad = np.sum(coeffs[active])
        if subgrad > 0:
            hi = mid
        else:
            lo = mid
    
    h[p] = (lo + hi) / 2
    # Fix the sum constraint
    h = make_feasible(h)
    return h

def make_feasible(z, target=None):

            lo = mid
        else:
            hi = mid

    if abs(residual) > 1e-14:

    dx = 2.0 / n

        m_hi = min(n - 1, n - 1 - k)
        for m in range(m_lo, m_hi + 1):
            p1 = m + k
            p2 = m
            grad[p1] += wi * (1.0 - x[p2])
            grad[p2] += wi * (-x[p1])
    return grad

    return np.sum(x) - len(x) / 2.0

            # S_k = c_k * h[p] + b_k (w.r.t. h[p])
            old = h_cur[p]
            n2 = len(h_cur)
            # Compute coefficients
            coeffs = np.zeros(2 * n2 - 1)

                m_hi = min(n2 - 1, n2 - 1 - k)

            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')

            best_s = s_proj

def coord_descent_h(h, max_passes=3, deadline=None):
    n = len(h)
    best = h.copy()
    best_s = official_score(best)
    for _pass in range(max_passes):
        if deadline and time.time() > deadline - 10:
            break
        h_cur = best.copy()
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        for p in range(n):
            old = h_cur[p]
            if old < 1e-12 or old > 1.0 - 1e-12:
                corr_new = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                continue
            coeffs = np.zeros(2 * n - 1)
            for i in range(2 * n - 1):
                k = i - (n - 1)
                m_lo = max(0, -k)
                m_hi = min(n - 1, n - 1 - k)
                m1 = p - k
                if m_lo <= m1 <= m_hi and 0 <= m1 < n:
                    coeffs[i] += (1.0 - h_cur[m1])
                if m_lo <= p <= m_hi:
                    p2 = p + k
                    if 0 <= p2 < n:
                        coeffs[i] += (-h_cur[p2])
            base = corr - coeffs * old
            lo_x, hi_x = 0.0, 1.0
            for _ in range(35):
                mid_x = (lo_x + hi_x) / 2.0
                vals = coeffs * mid_x + base
                mv = np.max(vals)
                active = vals >= mv - 1e-13
                sg = np.sum(coeffs[active])
                if sg > 1e-14:
                    hi_x = mid_x
                else:
                    lo_x = mid_x
            h_cur[p] = (lo_x + hi_x) / 2.0
        h_proj = make_feasible(h_cur)
        s_proj = official_score(h_proj)
        if s_proj < best_s - 1e-15:
            best = h_proj
            best_s = s_proj
        else:
            break
    return best

def subgradient_polish(h, n_iter=200, deadline=None, scale=1.0):
    n = len(h)
    best = h.copy()
    best_s = official_score(best)
    h_cur = h.copy()
    step0 = scale * (2.0 / n) / n
    for it in range(n_iter):
        if deadline and time.time() > deadline - 10:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * (2.0 / n)
        mc = np.max(corr)
        gap = mc - best_s + 0.001
        active = corr >= mc - 0.005
        delta = np.max(corr) - np.min(corr)
        if delta < 1e-12:
            break
        w = (corr - np.min(corr)) / delta
        w = w / np.sum(w)
        grad = np.zeros(n)
        dx = 2.0 / n
        for i in range(2 * n - 1):
            wi = w[i]
            if wi < 1e-8:
                continue
            k = i - (n - 1)
            m_lo = max(0, -k)
            m_hi = min(n - 1, n - 1 - k)
            for m in range(m_lo, m_hi + 1):
                p1 = m + k
                p2 = m
                grad[p1] += wi * dx * (1.0 - h_cur[p2])
                grad[p2] += wi * dx * (-h_cur[p1])
        step = step0 * (1.0 / (1.0 + 0.05 * it))
        h_trial = h_cur - step * grad
        h_trial = make_feasible(h_trial)
        s_trial = official_score(h_trial)
        if s_trial < best_s:
            best = h_trial
            best_s = s_trial
            h_cur = h_trial
    return best

def push_binary(h, p=1.5):
    return make_feasible(h ** p)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = make_feasible(np.ones(48) * 0.5)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)
    parent_n = len(parent)

    # Phase 1: Multi-start LSE with diverse n and starting points
    n_list = sorted(set([
        max(24, parent_n - 10),
        max(24, parent_n - 6),
        max(24, parent_n - 4),
        max(24, parent_n - 2),
        parent_n,
        parent_n + 2,
        parent_n + 4,
        parent_n + 6,
        parent_n + 10,
    ]))
    n_list = [n for n in n_list if 24 <= n <= 72]

    beta_sched_1 = [10, 30, 90, 270, 810, 2430, 7290]
    beta_sched_quick = [30, 100, 300, 900]

    for n in n_list:
        if time.time() > deadline - 80:
            break

        h_interp = make_feasible(interp_to_n(parent, n))
        h_anti = make_feasible(1.0 - h_interp[::-1].copy())
        s_interp = official_score(h_interp)

        starts = [(h_interp, s_interp)]
        starts.append((h_anti, official_score(h_anti)))
        for sigma in [0.03, 0.08, 0.15, 0.25]:
            hs = make_feasible(h_interp + sigma * rng.standard_normal(n))
            starts.append((hs, official_score(hs)))
        for bp in [1.2, 1.5, 2.0, 3.0]:
            hs = push_binary(h_interp, bp)
            starts.append((hs, official_score(hs)))
        # Mirror + noise
        for sigma in [0.05, 0.12]:
            hs = make_feasible(0.5 * h_interp + 0.5 * h_anti + sigma * rng.standard_normal(n))
            starts.append((hs, official_score(hs)))

        # Sort by score, try best first
        starts.sort(key=lambda x: x[1])

        for h0, s0 in starts:
            if time.time() > deadline - 50:
                break
            if s0 >= best_score + 0.005:
                continue
            mi = 180 if n <= 50 else 100
            h_new = do_lse_opt(h0, beta_sched_1, mi, deadline)
            s_new = official_score(h_new)
            if s_new < best_score + 0.002:
                h	cd = coord_descent_h(h_new, max_passes=2, deadline=deadline)
                s_cd = official_score(h_cd)
                if s_cd < s_new:
                    h_new = h_cd
                    s_new = s_cd
            if s_new < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = s_new
                best_n = len(h_new)

    # Phase 2: Aggressive refinement of best
    remaining = deadline - time.time()
    if remaining > 40:
        # High-beta LSE
        h_ref = do_lse_opt(best_h, [500, 1500, 4000, 12000], 250, deadline)
        s_ref = official_score(h_ref)
        if s_ref < best_score:
            best_h = h_ref
            best_score = s_ref

        # Coord descent
        h_cd = coord_descent_h(best_h, max_passes=4, deadline=deadline)
        s_cd = official_score(h_cd)
        if s_cd < best_score:
            best_h = h_cd
            best_score = s_cd

        # Subgradient polish
        h_sg = subgradient_polish(best_h, n_iter=300, deadline=deadline, scale=0.8)
        s_sg = official_score(h_sg)
        if s_sg < best_score:
            best_h = h_sg
            best_score = s_sg

        # Direct min-max
        mi = 250 if best_n <= 50 else 150
        h_dir = do_direct_opt(best_h, maxiter=mi, deadline=deadline)
        s_dir = official_score(h_dir)
        if s_dir < best_score:
            best_h = h_dir
            best_score = s_dir

        # Another coord descent round
        h_cd2 = coord_descent_h(best_h, max_passes=3, deadline=deadline)
        s_cd2 = official_score(h_cd2)
        if s_cd2 < best_score:
            best_h = h_cd2
            best_score = s_cd2

    # Phase 3: Random restarts
    remaining = deadline - time.time()
    n_trials = min(25, int(remaining / 14))
    for _ in range(n_trials):
        if time.time() > deadline - 10:
            break
        sigma = 0.02 + 0.03 * rng.random()
        h_pert = make_feasible(best_h + sigma * rng.standard_normal(best_n))
        h_opt = do_lse_opt(h_pert, beta_sched_quick, 120, deadline)
        h_opt = coord_descent_h(h_opt, max_passes=2, deadline=deadline)
        s = official_score(h_opt)
        if s < best_score - 1e-12:
            best_h = h_opt
            best_score = s

    # Final polish
    if time.time() < deadline - 5:
        h_f = do_direct_opt(best_h, maxiter=300, deadline=deadline)
        s_f = official_score(h_f)
        if s_f < best_score:
            best_h = h_f
            best_score = s_f
        h_cd = coord_descent_h(best_h, max_passes=3, deadline=deadline)
        s_cd = official_score(h_cd)
        if s_cd < best_score:
            best_h = h_cd
            best_score = s_cd

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
