def repair(z, n):
    target = n / 2.0
    # Binary search for lambda
    lo = -np.max(z) - 2.0
    hi = 1.0 - np.min(z) + 2.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Adjust for floating point
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            # Need to add mass; add to elements with most room
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    
    return h

        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = repair(parent)
            ps = evaluate(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = np.arange(n0)
        h_init = 0.5 + 0.25 * np.sin(2 * np.pi * x / n0) + 0.08 * np.cos(4 * np.pi * x / n0)
        best_h = repair(h_init)
        best_score = evaluate(best_h)

    # Phase 1: Hard active-set optimization
    h1, f1 = active_optimize(best_h, max_iter=150, deadline=deadline)
    if f1 < best_score:
        best_h, best_score = h1, f1

    # Phase 2: Soft-weight LP optimization (handles near-active constraints)
    h2, f2 = soft_optimize(best_h, deadline)
    if f2 < best_score:
        best_h, best_score = h2, f2

    # Phase 3: Active-set again from improved point
    h3, f3 = active_optimize(best_h, max_iter=150, deadline=deadline)
    if f3 < best_score:
        best_h, best_score = h3, f3

    # Phase 4: Try different resolutions
    if time.time() < deadline - 80:
        n_base = len(best_h)
        n_options = []
        for dn in [-12, -8, -6, -4, -2, 0, 2, 4, 6, 8, 10, 14]:
            nt = n_base + dn
            if 32 <= nt <= 96 and nt not in n_options:
                n_options.append(nt)
        n_options.sort(key=lambda x: abs(x - n_base))
        for n_try in n_options[:8]:
            if time.time() > deadline - 40:
                break
            h_try = interpolate_to(best_h, n_try)
            f_try = evaluate(h_try)
            if f_try < best_score:
                best_h, best_score = h_try, f_try
            h_opt, f_opt = active_optimize(h_try, max_iter=60, deadline=deadline)
            if f_opt < best_score:
                best_h, best_score = h_opt, f_opt
            if time.time() < deadline - 20:
                h_opt2, f_opt2 = soft_optimize(best_h, deadline)
                if f_opt2 < best_score:
                    best_h, best_score = h_opt2, f_opt2

    # Phase 5: Random perturbations + re-optimize
    if time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(3, min(15, int(remaining / 25)))
        for i in range(n_starts):
            if time.time() > deadline - 20:
                break
            sigma = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12][i % 7]
            h_cand = repair(best_h + rng.normal(0, sigma, size=n))
            if time.time() > deadline - 15:
                break
            h_opt, f_opt = active_optimize(h_cand, max_iter=40, deadline=deadline)
            if f_opt < best_score:
                best_h, best_score = h_opt, f_opt

    # Phase 6: Symmetry operations
    if time.time() < deadline - 30:
        h_comp = repair(1.0 - best_h.copy())
        f_comp = evaluate(h_comp)
        if f_comp < best_score:
            best_h, best_score = h_comp, f_comp
        h_comp_opt, f_comp_opt = active_optimize(h_comp, max_iter=60, deadline=deadline)
        if f_comp_opt < best_score:
            best_h, best_score = h_comp_opt, f_comp_opt

        h_rev = repair(best_h[::-1].copy())
        f_rev = evaluate(h_rev)
        if f_rev < best_score:
            best_h, best_score = h_rev, f_rev
        h_rev_opt, f_rev_opt = active_optimize(h_rev, max_iter=60, deadline=deadline)
        if f_rev_opt < best_score:
            best_h, best_score = h_rev_opt, f_rev_opt

        h_antisym = repair(1.0 - best_h[::-1].copy())
        f_as = evaluate(h_antisym)
        if f_as < best_score:
            best_h, best_score = h_antisym, f_as
        h_as_opt, f_as_opt = active_optimize(h_antisym, max_iter=60, deadline=deadline)
        if f_as_opt < best_score:
            best_h, best_score = h_as_opt, f_as_opt

    # Phase 7: Final intensive polishing
    if time.time() < deadline:
        h7, f7 = active_optimize(best_h, max_iter=300, deadline=deadline)
        if f7 < best_score:
            best_h, best_score = h7, f7
        if time.time() < deadline - 5:
            h8, f8 = soft_optimize(best_h, deadline, betas=[100, 300, 800, 2000])
            if f8 < best_score:
                best_h, best_score = h8, f8
        if time.time() < deadline - 3:
            h9, f9 = active_optimize(best_h, max_iter=200, deadline=deadline)
            if f9 < best_score:
                best_h, best_score = h9, f9

    # Final: one more resolution sweep at very end
    if time.time() < deadline - 10:
        n_base = len(best_h)
        for dn in [-2, 2, -4, 4]:
            if time.time() > deadline - 5:
                break
            nt = n_base + dn
            if 30 <= nt <= 100:
                h_try = interpolate_to(best_h, nt)
                f_try = evaluate(h_try)
                if f_try < best_score:
                    best_h, best_score = h_try, f_try

    final_h = repair(best_h)
    final_score = evaluate(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
