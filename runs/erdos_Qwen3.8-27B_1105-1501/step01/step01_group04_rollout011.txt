def compute_c_scaled(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(v, target=None):

    if target is None:
        target = n / 2.0
    h = np.clip(v, 0.0, 1.0)

    if abs(s - target) < 1e-10:

        else:

        h_new = np.clip(h_new, 0.0, 1.0)
        # Verify with full computation
        actual = compute_c5(h_new)
        if actual < current_c5 - 1e-12:
            return h_new, actual
        # If the top-k approximation was optimistic, try without refinement
        h_new = h + ts[best_idx] * d
        h_new = np.clip(h_new, 0.0, 1.0)
        actual = compute_c5(h_new)
        if actual < current_c5 - 1e-12:
            return h_new, actual
    
    return None, current_c5

def find_worst_pairs(h, c_scaled, n_found=25, rng=None):
    """Find promising (src, dst) pairs for mass transfer."""
    n = len(h)
    top_idx = np.argsort(c_scaled)[::-1][:10]
    pairs = []
    seen = set()
    
    for k in top_idx:
        lag = k - (n - 1)
        if lag == 0:
            continue
        if lag > 0:
            contribs = h[lag:] * (1.0 - h[:-lag])
            top_m = np.argsort(contribs)[::-1][:4]
            for idx in top_m:
                src = int(idx + lag)
                dst = int(idx)
                if src != dst and (src, dst) not in seen:
                    if h[src] > 1e-6 and h[dst] < 1.0 - 1e-6:
                        seen.add((src, dst))
                        pairs.append((src, dst))
        else:
            M = -lag
            contribs = h[:-M] * (1.0 - h[M:])
            top_m = np.argsort(contribs)[::-1][:4]
            for idx in top_m:
                src = int(idx)
                dst = int(idx + M)
                if src != dst and (src, dst) not in seen:
                    if h[src] > 1e-6 and h[dst] < 1.0 - 1e-6:
                        seen.add((src, dst))
                        pairs.append((src, dst))
    
    if rng is not None:
        for _ in range(5):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j and (i, j) not in seen:
                if h[i] > 1e-6 and h[j] < 1.0 - 1e-6:
                    seen.add((i, j))
                    pairs.append((i, j))
    
    return pairs[:n_found]
