import numpy as np
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def make_spread_binary(n, seed=0):
    """Create a 'spread' binary pattern with exactly n/2 ones."""
    ones_count = n // 2
    step = n / ones_count
    positions = np.array([int(np.floor(i * step + 0.5)) % n for i in range(ones_count)])
    h = np.zeros(n)
    h[np.unique(positions)] = 1.0
    diff = ones_count - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(seed)
        h[rng.choice(zeros, diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(seed)
        h[rng.choice(ones, -diff, replace=False)] = 0.0
    return h

def pair_move(h, corr, i, j, n):

    Returns (delta, new_corr, improved)."""

    jm = j - s
    is_ = i + s
    im = i - s

    h_is = np.where((is_ >= 0) & (is_ < n), h[np.clip(is_, 0, n-1)], 0.0)
    h_im = np.where((im >= 0) & (im < n), h[np.clip(im, 0, n-1)], 0.0)
    
    g = h_js - h_jm - h_is + h_im
    
    # Bisection
    current_max = np.max(corr)
    best_delta = 0.0

    l, r = lo, hi
