def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub

                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def build_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    c = n - 1
    A = np.zeros((m, n))
    for idx in range(m):
        k = idx - c
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            A[idx, lo1:hi1+1] += (1.0 - h[lo1-k:hi1+1-k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            A[idx, lo2:hi2+1] -= h[lo2+k:hi2+1+k]
    return A

def lp_descent(h, max_alphas=10, scale=1.0):
    n = len(h)
    cur_c = score_exact(h)
    overlaps = np.correlate(h, 1.0 - h, mode='full')
    A = build_grad_matrix(h)
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A @ delta + overlaps <= t,
        cp.sum(delta) == 0,
        delta >= -h,
        delta <= 1.0 - h
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solved = False
    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
                break
        except Exception:
            continue
    
    if not solved or delta.value is None:
        return h, cur_c, False
    
    d = np.asarray(delta.value, dtype=np.float64).flatten()
    if not np.all(np.isfinite(d)):
        return h, cur_c, False
    
    norm_d = float(np.linalg.norm(d))
    if norm_d < 1e-14:
        return h, cur_c, False
    
    best_h = h.copy()
    best_c = cur_c
    improved = False
    alpha = scale
    for _ in range(max_alphas):
        h_try = make_feasible(h + alpha * d, n)
        c_try = score_exact(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try
            best_c = c_try
            improved = True
        if c_try >= cur_c:
            alpha *= 0.5
        else:
            alpha *= 0.8
        if alpha < 1e-8:
            break
    
    if not improved:
        return h, cur_c, False
    return best_h, best_c, True
