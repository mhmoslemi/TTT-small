Given the quadratic structure, C(s) = max_k (F_k + L_k*s - Q_k*s²). This is the upper envelope of quadratics.

For a minimum of C(s) (interior), we need:

- Or C(s) is at a local minimum of a single quadratic (where Q_k < 0, i.e., concave)

For two quadratics a and b:

(Q_a - Q_b)*s² + (L_a - L_b)*s + (F_a - F_b) = 0

This gives at most 2 solutions.

1. Identify the "active" lags (those with F_k near the maximum, say within some tolerance)
2. For each pair of active lags, find their intersection points

4. Evaluate C(s) at all candidate points using the official score

import numpy as np
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def compute_F(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def pair_L(h, p, q, n, one_minus_h):
    L = np.zeros(2 * n - 1)
    L[n - 1 - p: n - 1 - p + n] += one_minus_h
    L[n - 1 - q: n - 1 - q + n] -= one_minus_h
    dk = q - p
    L[n - 1] += h[q] - h[p]
    idx_dk = n - 1 + dk
    if 0 <= idx_dk < 2 * n - 1:
        L[idx_dk] += h[p]
    idx_mdk = n - 1 - dk
    if 0 <= idx_mdk < 2 * n - 1:
        L[idx_mdk] -= h[q]
    return L


def pair_Q(p, q, n):
    Q = np.zeros(2 * n - 1)
    dk = q - p
    Q[n - 1] = 2.0
    Q[n - 1 + dk] = -1.0
    Q[n - 1 - dk] = -1.0
    return Q


def line_search(F, L, Q, s_lo, s_hi, n, topK=20):
    m = 2 * n - 1
    F_sorted = np.argsort(F)[::-1]
    active_set = set(F_sorted[:topK].tolist())
    q_nz = np.where(np.abs(Q) > 1e-14)[0]
    active_set.update(q_nz.tolist())
    active = list(active_set)

    candidates = [s_lo, s_hi, 0.0]

    for idx in active:
        qi = Q[idx]
        if qi > 1e-12:
            sv = L[idx] / (2.0 * qi)
            if s_lo < sv < s_hi:
                candidates.append(sv)
        elif qi < -1e-12:
            sv = L[idx] / (2.0 * qi)
            if s_lo < sv < s_hi:
                candidates.append(sv)

    la = L[active]
    qav = Q[active]
    fav = F[active]
    for i in range(len(active)):
        for j in range(i + 1, len(active)):
            a = qav[i] - qav[j]
            b = la[i] - la[j]
            c = fav[i] - fav[j]
            if abs(a) > 1e-14:
                disc = b * b - 4.0 * a * c
                if disc >= 0:
                    sq = np.sqrt(disc)
                    s1 = (-b + sq) / (2.0 * a)
                    s2 = (-b - sq) / (2.0 * a)
                    if s_lo < s1 < s_hi:
                        candidates.append(s1)
                    if s_lo < s2 < s_hi:
                        candidates.append(s2)
            elif abs(b) > 1e-14:
                s1 = -c / b
                if s_lo < s1 < s_hi:
                    candidates.append(s1)

    best_s = 0.0
    best_val = np.inf
    for s in candidates:
        s = max(s_lo, min(s_hi, s))
        val = np.max(F + L * s - Q * s * s)
        if val < best_val:
            best_val = val
            best_s = s

    if s_lo < best_s < s_hi and best_val < np.max(F) - 1e-12:
        n_grid = 33
        grid = np.linspace(s_lo, s_hi, n_grid)
        for s in grid:
            val = np.max(F + L * s - Q * s * s)
            if val < best_val:
                best_val = val
                best_s = s

    return best_s, best_val


def local_search_pairs(h, n, max_passes=20, topK=20, deadline=None):
    h = make_feasible(h.copy())
    F = compute_F(h, n)
    one_minus_h = 1.0 - h
    current_max = float(np.max(F))
    t_start = time.time()

    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            hp = h[p]
            for q in range(p + 1, n):
                hq = h[q]
                s_lo = max(-hp, hq - 1.0)
                s_hi = min(1.0 - hp, hq)
                if s_hi - s_lo <= 1e-12:
                    continue
                L = pair_L(h, p, q, n, one_minus_h)
                Q = pair_Q(p, q, n)
                best_s, best_val = line_search(F, L, Q, s_lo, s_hi, n, topK=topK)
                if best_val < current_max - 1e-10:
                    h[p] += best_s
                    h[q] -= best_s
                    np.clip(h, 0.0, 1.0, out=h)
                    F = compute_F(h, n)
                    one_minus_h = 1.0 - h
                    current_max = float(np.max(F))
                    improved = True
        if not improved:
            break

    h = make_feasible(h)
    return h, official_score(h)


def random_multidir(h, n, rng, support_size=4):
    idx = rng.choice(n, size=support_size, replace=False)
    v = np.zeros(n)
    pos = idx[::2]
    neg = idx[1::2]
    for i in pos:
        v[i] += 1.0
    for i in neg:
        v[i] -= 1.0
    if abs(np.sum(v)) > 1e-12:
        v -= np.mean(v)
    return v

def symmetry_variants(h):
