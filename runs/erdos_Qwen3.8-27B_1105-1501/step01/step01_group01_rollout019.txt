import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Water filling: find offset theta such that sum(clip(z+theta, 0, 1)) = target
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_subgradient(h, active_weights=None):
    """Compute subgradient of max_k corr_k at h.
    active_weights: weights for each lag k (sum to 1). If None, use uniform over active set.
    """
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    if active_weights is None:
        max_c = np.max(corr)
        # Active set: lags within some tolerance of max
        tol = 1e-6
        active = corr >= max_c - tol
        if not np.any(active):
            active = np.zeros(len(corr), dtype=bool)
            active[np.argmax(corr)] = True
        active_weights = active / np.sum(active)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = active_weights[k]
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        # First term: h[p] appears as a[p] where p = i + shift, i valid
        # Valid: max(0, shift) <= p < min(n, n+shift) and 0 <= p-shift < n
        p_lo = max(0, shift)
        p_hi = min(n, n + shift)
        for p in range(p_lo, p_hi):
            j = p - shift
            if 0 <= j < n:
                g[p] += w * dx * (1.0 - h[j])
        
        # Second term: h[p] appears in b[i] = 1-h[p] where i = p
        # Valid: 0 <= p < n and 0 <= p+shift < n
        p_lo2 = max(0, -shift)
        p_hi2 = min(n, n - shift)
        for p in range(p_lo2, p_hi2):
            j = p + shift
            if 0 <= j < n:
                g[p] += w * dx * (-h[j])
    
    return g

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):

def _sep_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def _sep_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        p_lo = max(0, shift)
        p_hi = min(n, n + shift)
        for p in range(p_lo, p_hi):
            j = p - shift
            if 0 <= j < n:
                g[p] += w * (1.0 - x[j])
        p_lo2 = max(0, -shift)
        p_hi2 = min(n, n - shift)
        for p in range(p_lo2, p_hi2):
            j = p + shift
            if 0 <= j < n:
                g[p] += w * (-x[j])
    return g

    return np.sum(x) - len(x) / 2.0

    return np.ones(len(x))

            jac=_sep_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,

        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = 40.0
    deadline = t_start + budget_s - reserve
    
    rng = np.random.default_rng(seed)
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = make_feasible(np.random.default_rng(0).random(50))
    
    best_h = parent.copy()
    best_score = official_score(parent)
    best_n = len(parent)
    
    # Phase 1: Binary SA for even n values
    n_list = [48, 50, 52, 54, 56]
    
    for n in n_list:
        if time.time() > deadline - 80:
            break
        n_trials = 4
        for trial in range(n_trials):
            if time.time() > deadline - 60:
                break
            h_init = np.zeros(n)
            h_init[rng.choice(n, n // 2, replace=False)] = 1.0
            n_iter = 800000
            T0 = 0.015
            T1 = 1e-12
            h_sa, s_sa = binary_sa(h_init, n_iter, T0, T1, rng)
            h_gr, s_gr = binary_greedy(h_sa, max_rounds=300)
            if s_gr < best_score:
                best_score = s_gr
                best_h = h_gr.copy()
                best_n = n
    
    # Phase 2: Continue from best binary with more SA restarts
    if time.time() < deadline - 60:
        best_bin_n = best_n
        if best_bin_n % 2 != 0:
            best_bin_n = 50
            h_init = np.zeros(best_bin_n)
            h_init[rng.choice(best_bin_n, best_bin_n // 2, replace=False)] = 1.0
            h_sa, s_sa = binary_sa(h_init, 500000, 0.01, 1e-12, rng)
            h_gr, s_gr = binary_greedy(h_sa, max_rounds=200)
            if s_gr < best_score:
                best_score = s_gr
                best_h = h_gr.copy()
                best_bin_n = best_bin_n
        # Restart SA from best
        for restart in range(3):
            if time.time() > deadline - 40:
                break
            h_cur = best_h.astype(float).copy()
            if best_bin_n % 2 == 0 and np.allclose(h_cur, np.round(h_cur)):
                h_cur[best_h > 0.5] = 1.0
                h_cur[h_cur < 0.5] = 0.0
            h_sa2, s_sa2 = binary_sa(h_cur, 600000, 0.008, 1e-12, rng)
            h_gr2, s_gr2 = binary_greedy(h_sa2, max_rounds=200)
            if s_gr2 < best_score:
                best_score = s_gr2
                best_h = h_gr2.copy()
    
    # Phase 3: Continuous optimization from best binary and parent
    if time.time() < deadline - 30:
        cont_starts = []
        # From best binary
        cont_starts.append(best_h.astype(float).copy())
        # From parent
        cont_starts.append(parent.copy())
        # Perturbed best
        cont_starts.append(make_feasible(best_h + 0.03 * rng.standard_normal(len(best_h))))
        # Interpolated parent to best_n
        if len(parent) != best_n:
            cont_starts.append(make_feasible(interp_to_n(parent, best_n)))
        
        for h0 in cont_starts:
            if time.time() > deadline - 20:
                break
            cur_h = h0.copy()
            cur_s = official_score(cur_h)
            for beta in [50, 200, 800]:
                if time.time() > deadline - 15:
                    break
                new_h = do_smooth_opt(cur_h, beta, maxiter=150)
                new_s = official_score(new_h)
                if new_s < cur_s - 1e-12:
                    cur_h = new_h
                    cur_s = new_s
            # Direct polish
            if time.time() < deadline - 10:
                new_h = do_direct_opt(cur_h, maxiter=150)
                new_s = official_score(new_h)
                if new_s < cur_s - 1e-12:
                    cur_h = new_h
                    cur_s = new_s
            if cur_s < best_score - 1e-12:
                best_score = cur_s
                best_h = cur_h.copy()
    
    # Phase 4: Final refinement
    if time.time() < deadline - 10:
        for _ in range(5):
            if time.time() > deadline - 8:
                break
            h_pert = make_feasible(best_h + 0.015 * rng.standard_normal(len(best_h)))
            h_opt = do_smooth_opt(h_pert, 300, maxiter=120)
            h_opt = do_direct_opt(h_opt, maxiter=100)
            s = official_score(h_opt)
            if s < best_score - 1e-12:
                best_score = s
                best_h = h_opt.copy()
    
    # Final direct polish
    if time.time() < deadline - 5:
        final_h = do_direct_opt(best_h, maxiter=200)
        final_s = official_score(final_h)
        if final_s < best_score:
            best_h = final_h
            best_score = final_s
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
