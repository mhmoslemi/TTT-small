def make_feasible(v, target=None):

def active_gradient(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    
    # Softmax weights
    alpha = 30.0
    w = np.exp(alpha * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(m)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(m):
        wi = w[k]
        if wi < 1e-8:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:-lag])
            grad[:-lag] += -wi * h[lag:]
        else:
            M = -lag
            grad[:-M] += wi * (1.0 - h[M:])
            grad[M:] += -wi * h[:-M]
    
    return grad * dx
