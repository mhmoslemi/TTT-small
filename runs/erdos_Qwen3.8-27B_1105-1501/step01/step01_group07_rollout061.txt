def structured_pair_search(h, lags, n_pts=30):

    n = len(h)
    dx = 2.0 / n
    cur_c = float(np.max(lags))
    c_center = n - 1
    
    best_s = cur_c
    best_move = None
    
    # For each active lag, find the worst pair and try to fix it
    active_mask = lags >= cur_c * 0.97
    active_lags = np.where(active_mask)[0]
    
    for lag_idx in active_lags:
        k = lag_idx - c_center  # the shift
        if k == 0:
            continue
        # For this lag, the overlap is sum of h[i]*h'[i] where the shift is k
        # The "bad" pairs contribute most
        # For k > 0: overlap = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # To decrease this, we want to decrease h[i+k] or increase h[i]
        # For k < 0: overlap = sum_{i=-k}^{n-1} h[i]*(1-h[i+k])
        # To decrease this, we want to decrease h[i] or increase h[i+k]
        
        if k > 0:
            m = n - k
            # Pairs (i+k, i) for i=0..m-1
            # Contribution of pair i: h[i+k]*(1-h[i])
            contribs = h[k:] * (1 - h[:m])
            # Find the pair with largest contribution
            worst_i = int(np.argmax(contribs))
            # Try moving mass from h[worst_i+k] to h[worst_i]
            i1 = worst_i + k
            i2 = worst_i
        else:
            kk = -k
            m = n - kk
            # Pairs (i+kk, i) for i=0..m-1, contribution: h[i+kk]... 
            # Wait, for k<0: result = sum h[n+k] * (1-h[n]) where k is negative
            # = sum_{n=-k}^{n-1} h[n+k] * (1-h[n]) -- this doesn't seem right
            # Let me re-derive: for shift k<0, kk=-k>0
            # result[kk_shifted] where the index in correlate is c-kk = (n-1)-kk
            # This equals sum_n h[n-kk] * (1-h[n]) for valid n
            # Which is the same as sum_m h[m] * (1-h[m+kk]) for m=0..n-1-kk
            # To decrease: decrease h[m] or increase h[m+kk]
            contribs = h[:m] * (1 - h[kk:n])
            worst_i = int(np.argmax(contribs))
            i1 = worst_i
            i2 = worst_i + kk
        
        # Now try to move mass from i1 to i2
        orig_1 = h[i1]
        orig_2 = h[i2]
        if i1 == i2:
            continue
        
        # Range of feasible transfers: a = new h[i1] - orig_1, h[i2] = orig_2 - a
        # Need: 0 <= orig_1 + a <= 1 and 0 <= orig_2 - a <= 1
        lo = max(-orig_1, orig_2 - 1.0)
        hi = min(1.0 - orig_1, orig_2)
        if hi - lo < 1e-12:
            continue
        
        for a in np.linspace(lo, hi, n_pts):
            h[i1] = orig_1 + a
            h[i2] = orig_2 - a
            s = score_exact(h)
            if s < best_s:
                best_s = s
                best_move = (i1, i2, a)
        
        h[i1] = orig_1
        h[i2] = orig_2
    
    if best_move is not None and best_s < cur_c - 1e-14:
        i1, i2, a = best_move
        h[i1] += a
        h[i2] -= a

    return None

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()

            lo = mid
        else:
            hi = mid

import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idxs = np.argsort(z)
            for idx in idxs:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            idxs = np.argsort(-z)
            for idx in idxs:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_single_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = (1.0 - 2.0 * h)
    elif k > 0:
        j = np.arange(n)
        v1 = j >= k
        g[v1] += (1.0 - h[j[v1] - k])
        v2 = j < n - k
        g[v2] -= h[j[v2] + k]
    else:
        kk = -k
        j = np.arange(n)
        v1 = j < n - kk
        g[v1] += (1.0 - h[j[v1] + kk])
        v2 = j >= kk
        g[v2] -= h[j[v2] - kk]
    return g * dx

def active_descent(h, deadline, top_m=10, use_momentum=True, tol_frac=0.96):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    step = 0.04
    stall = 0
    vel = np.zeros(n) if use_momentum else None
    c_center = n - 1

    while time.time() < deadline:
        lags = compute_lags(cur)
        max_val = float(np.max(lags))
        active_mask = lags >= max_val * tol_frac
        active_indices = np.where(active_mask)[0]
        if len(active_indices) > top_m:
            order = np.argsort(-lags[active_indices])[:top_m]
            active_indices = active_indices[order]

        if len(active_indices) == 1:
            k = int(active_indices[0]) - c_center
            g = grad_single_lag(cur, k)
        else:
            grads = np.array([grad_single_lag(cur, int(ai) - c_center) for ai in active_indices])
            g = np.mean(grads, axis=0)
            dots = grads @ g
            if np.min(dots) < 0:
                pos_mask = dots > 0.01 * np.max(dots)
                if np.any(pos_mask):
                    g = np.sum(grads[pos_mask], axis=0)
                else:
                    g = grads[np.argmin(dots)]

        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        d = -g / gn

        if use_momentum:
            vel = 0.85 * vel + 0.15 * d
            vn = np.linalg.norm(vel)
            if vn > 1e-15:
                d = vel / vn

        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1, step * 0.05):
            cand = make_feasible(cur + s * d, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break

        if not improved:
            step *= 0.85
            stall += 1
            if use_momentum:
                vel *= 0.5
            if stall > 80:
                break
        else:
            step = min(step * 1.04, 0.08)
            stall = 0

    return best, best_c

    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def triple_search(h, deadline, rng, tol=0.95):
    """Move mass between 3 indices simultaneously"""
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    c_center = n - 1
    fail = 0

    while time.time() < deadline and fail < 200:
        lags = compute_lags(cur)
        max_val = float(np.max(lags))
        active_mask = lags >= max_val * tol
        active_indices = np.where(active_mask)[0]

        sensitive = np.zeros(n, dtype=bool)
        for ai in active_indices:
            k = int(ai) - c_center
            if k >= 0:
                sensitive[k:] = True
                sensitive[:n - k] = True
            else:
                sensitive[:n + k] = True
                sensitive[-k:] = True
        sens_idx = np.where(sensitive)[0]
        if len(sens_idx) < 3:
            sens_idx = np.arange(n)

        choices = list(rng.choice(sens_idx, size=min(3, len(sens_idx)), replace=False))
        if len(choices) < 3:
            fail += 1
            continue
        i, j, k2 = int(choices[0]), int(choices[1]), int(choices[2])
        if i == j or j == k2 or i == k2:
            fail += 1
            continue

        orig = np.array([cur[i], cur[j], cur[k2]])
        # Move a from i to j, b from i to k2 (or various patterns)
        # Pattern: decrease i by a+b, increase j by a, increase k2 by b
        # Constraints: 0<=orig[i]-a-b<=1, 0<=orig[j]+a<=1, 0<=orig[k2]+b<=1
        # a>=0, b>=0, a+b <= orig[i]
        max_ab = orig[0]
        max_a = 1.0 - orig[1]
        max_b = 1.0 - orig[2]

        n_pts = 10
        best_result = None
        for a in np.linspace(0, min(max_ab, max_a), n_pts):
            for b in np.linspace(0, min(max_ab - a, max_b), n_pts):
                if a + b < 1e-10:
                    continue
                cur[i] = orig[0] - a - b
                cur[j] = orig[1] + a
                cur[k2] = orig[2] + b
                s = score_exact(cur)
                if best_result is None or s < best_result[0]:
                    best_result = (s, a, b)

        cur[i:j] = orig[0]  # restore
        cur[i] = orig[0]
        cur[j] = orig[1]
        cur[k2] = orig[2]

        if best_result is not None and best_result[0] < cur_c - 1e-14:
            s_best, a_best, b_best = best_result
            cur[i] = orig[0] - a_best - b_best
            cur[j] = orig[1] + a_best
            cur[k2] = orig[2] + b_best
            cur_c = score_exact(cur)
            fail = 0
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
        else:
            fail += 1

    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: Active-set descent with momentum (30%)
    p1 = t0 + budget_s * 0.30
    if time.time() < p1:
        h1, c1 = active_descent(best_h, p1, top_m=10, use_momentum=True, tol_frac=0.95)
        if c1 < best_c:
            best_h, best_c = h1, c1
            n = len(best_h)

    # Phase 2: Targeted pair search (20%)
    p2 = t0 + budget_s * 0.50
    if time.time() < p2:
        h2, c2 = targeted_pair_search(best_h, p2, rng, tol=0.94)
        if c2 < best_c:
            best_h, best_c = h2, c2
            n = len(best_h)

    # Phase 3: Block moves (15%)
    p3 = t0 + budget_s * 0.65
    if time.time() < p3:
        h3, c3 = block_move_search(best_h, p3, rng, max_blk=5)
        if c3 < best_c:
            best_h, best_c = h3, c3
            n = len(best_h)

    # Phase 4: Active-set descent tight (15%)
    p4 = t0 + budget_s * 0.80
    if time.time() < p4:
        h4, c4 = active_descent(best_h, p4, top_m=20, use_momentum=False, tol_frac=0.93)
        if c4 < best_c:
            best_h, best_c = h4, c4
            n = len(best_h)

    # Phase 5: Final pair polish (10%)
    p5 = t0 + budget_s * 0.90
    if time.time() < p5:
        h5, c5 = targeted_pair_search(best_h, p5, rng, tol=0.97)
        if c5 < best_c:
            best_h, best_c = h5, c6 if False else c5
            n = len(best_h)

    # Phase 6: Resolution changes + polish (last 10%)
    p6 = deadline
    if time.time() < p6:
        for dn in [4, -4, 8, -6, 6, 10]:
            if time.time() > p6 - 5:
                break
            nr = n + dn
            if nr < 30 or nr > 150:
                continue
            h_r = resample(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub = min(p6 - 2, time.time() + 35)
            if time.time() < sub:
                h_r, c_r = active_descent(h_r, sub, top_m=12, use_momentum=True, tol_frac=0.94)
            if c_r < best_c:
                best_h, best_c, n = h_r, c_r, nr

    # Final polish
    if time.time() < deadline - 1:
        hf, cf = active_descent(best_h, deadline - 1, top_m=25, use_momentum=False, tol_frac=0.92)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
