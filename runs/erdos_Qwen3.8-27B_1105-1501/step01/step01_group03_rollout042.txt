    return make_feasible(h)

def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def golden_init(n, rng):
    h = np.zeros(n)
    k = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    for i in range(k):
        pos = int((i / float(k) + 0.5 / k) * n % n)
        if pos < n:
            positions.add(pos)
    count = 0
    for i in range(n):
        if count >= k:
            break
        if i in positions:
            h[i] = 1.0
            count += 1
    while count < k:
        h[rng.integers(n)] = 1.0
        count += 1
    if n % 2 == 1:
        h[rng.integers(n)] += 0.5
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    if parent is not None:
        parent = make_feasible(parent)
        parent_n = len(parent)
        best_h = parent.copy()
        best_score = score(best_h)
    else:
        parent_n = 60
        best_h = make_feasible(np.full(parent_n, 0.5))
        best_score = score(best_h)

    def try_update(h_cand):
        nonlocal best_h, best_score
        h_cand = make_feasible(h_cand)
        if not np.all(np.isfinite(h_cand)):
            return
        sc = score(h_cand)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc

    n_values = []
    for delta in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, 12, 14, 16, 20, -12, -16]:
        nl = parent_n + delta
        if 30 <= nl <= 100 and nl not in n_values:
            n_values.append(nl)
    for fixed in [40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80]:
        if fixed not in n_values:
            n_values.append(fixed)
    n_values.sort()

    phase1_deadline = t_start + 0.4 * budget_s
    for n in n_values:
        if time.time() > phase1_deadline:
            break
        cands = []
        c0 = interp_to_length(parent, n)
        cands.append(c0)
        cands.append(random_binary(n, rng))
        cands.append(golden_init(n, rng))
        cands.append(random_binary(n, rng))
        cands.append(make_feasible(1.0 - c0[::-1].copy()))

        seen = set()
        unique_cands = []
        for c in cands:
            c = make_feasible(c)
            key = tuple(np.round(c, 4))
            if key not in seen:
                seen.add(key)
                unique_cands.append(c)

        for c in unique_cands:
            if time.time() > phase1_deadline - 5:
                break
            cur = c.copy()
            cur = slsqp_smooth(cur, beta=100, maxiter=150)
            try_update(cur)
            cur = cd_optimize(cur, n_sweeps=30)
            try_update(cur)
            cur = slsqp_smooth(cur, beta=300, maxiter=100)
            try_update(cur)
            cur = cd_optimize(cur, n_sweeps=20)
            try_update(cur)
            hb = binary_round(cur)
            try_update(hb)

    phase2_deadline = t_start + 0.7 * budget_s
    best_n = len(best_h)
    for n in [best_n, best_n - 2, best_n + 2, best_n - 4, best_n + 4]:
        if n < 30 or n > 100:
            continue
        if time.time() > phase2_deadline:
            break
        c0 = interp_to_length(best_h, n)
        for beta in [50, 150, 500, 1000]:
            if time.time() > phase2_deadline - 5:
                break
            cur = slsqp_smooth(c0, beta=beta, maxiter=200)
            try_update(cur)
            cur = cd_optimize(cur, n_sweeps=60)
            try_update(cur)
            cur = slsqp_smooth(cur, beta=beta * 3, maxiter=150)
            try_update(cur)
            cur = cd_optimize(cur, n_sweeps=40)
            try_update(cur)
            hb = binary_round(cur)
            try_update(hb)

    phase3_deadline = deadline
    for _ in range(3):
        if time.time() > phase3_deadline - 5:
            break
        cur = best_h.copy()
        for beta in [200, 800, 2000, 5000]:
            if time.time() > phase3_deadline - 3:
                break
            cur = slsqp_smooth(cur, beta=beta, maxiter=200)
            try_update(cur)
        cur = cd_optimize(cur, n_sweeps=120)
        try_update(cur)
        cur = slsqp_smooth(cur, beta=10000, maxiter=150)
        try_update(cur)
        cur = cd_optimize(cur, n_sweeps=60)
        try_update(cur)
        hb = binary_round(cur)
        try_update(hb)

    if time.time() < deadline - 10:
        cur = best_h.copy()
        for _ in range(5):
            if time.time() > deadline - 8:
                break
            cur = cd_optimize(cur, n_sweeps=30)
            try_update(cur)

    if time.time() < deadline - 5:
        best_n = len(best_h)
        for alt_n in [best_n + 1, best_n - 1]:
            if alt_n < 30 or alt_n > 100 or time.time() > deadline - 5:
                continue
            cur = interp_to_length(best_h, alt_n)
            cur = cd_optimize(cur, n_sweeps=40)
            try_update(cur)

    best_h = make_feasible(best_h)
    final_score = score(best_h)
    if not np.isfinite(final_score) or final_score <= 0:
        if parent is not None:
            best_h = make_feasible(parent)
        else:
            best_h = make_feasible(np.full(parent_n, 0.5))
        final_score = score(best_h)

    return best_h, final_score, len(best_h)
