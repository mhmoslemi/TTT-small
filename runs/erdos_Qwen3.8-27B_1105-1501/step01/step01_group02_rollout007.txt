def exact_project(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum} exactly."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Fix any residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            # Add to elements with room
            mask = result < 1.0
            count = np.sum(mask)
            if count > 0:
                result[mask] += residual / count
                result = np.clip(result, 0.0, 1.0)
        else:
            mask = result > 0.0
            count = np.sum(mask)
            if count > 0:
                result[mask] += residual / count
                result = np.clip(result, 0.0, 1.0)
    return result

def compute_c5(h):
    """Compute the exact C5 value."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_obj(h, temp, target_sum, rho):
    """Smooth max objective with sum penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx / temp
    # Stable log-sum-exp
    m = np.max(scaled)
    log_sum = m + np.log(np.sum(np.exp(scaled - m)))
    penalty = rho * (np.sum(h) - target_sum) ** 2
    return log_sum * temp + penalty

def optimize_smooth(h_start, target_sum, temp_schedule, rho, deadline, maxiter_base=200):
    """Run scipy L-BFGS-B optimization with temperature schedule."""
    n = len(h_start)
    current = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    
    for temp in temp_schedule:
        if time.time() >= deadline:
            break
        try:
            result = minimize(
                _obj_wrapper, current, args=(temp, target_sum, rho),
                jac=_grad_wrapper, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxiter_base, 'ftol': 1e-10, 'gtol': 1e-7}
            )
            if result.x is not None:
                current = np.clip(result.x, 0.0, 1.0)
                current = exact_project(current, target_sum)
        except Exception:
            break
    
    return current

def pair_exchange_refine(h, target_sum, deadline, n_iter_per_round=500, rng=None):
    """Refine using pair exchanges with line search."""
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(n_iter_per_round):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Line search: h[i] += t, h[j] -= t
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        if t_hi - t_lo < 1e-10:
            continue
        # Sample the interval
        n_samples = 15
        ts = np.linspace(t_lo, t_hi, n_samples)
        best_t = 0.0
        best_val = best_c5
        orig_i, orig_j = h[i], h[j]
        for t in ts:
            h[i] = orig_i + t
            h[j] = orig_j - t
            c5 = compute_c5(h)
            if c5 < best_val:
                best_val = c5
                best_t = t
        h[i] = orig_i + best_t
        h[j] = orig_j - best_t
        if best_val < best_c5 - 1e-12:
            best_c5 = best_val
    
    return h, best_c5

Verify:
result[3] = h[1]*(1-h[2]) + h[2]*(1-h[1])

So the correct gradient formula is:
For each k in range(2n-1):
    s = k - (n-1)
    j_lo = max(0, s)
    j_hi = min(n-1, n-1+s)
    for j in range(j_lo, j_hi+1):
        grad[j] += w[k] * (1 - 2*h[n-1+s-j]) * dx

This can be vectorized. For a given s:
j ranges from j_lo to j_hi
k-j = n-1+s-j ranges from n-1+s-j_hi to n-1+s-j_lo

If s >= 0: j_lo=s, j_hi=n-1

  So h[k-j] = h[s:n] in reverse = h[s:n][::-1]
