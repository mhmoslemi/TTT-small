def lp_direction(h, G, c_scaled, active_idx, tol_inactive):
    """
    Find descent direction via LP.
    h: current point (n,)
    G: gradient matrix (2n-1, n)
    c_scaled: scaled correlations (2n-1,)
    active_idx: indices of active (near-max) correlations
    Returns direction d or None
    """
    n = len(h)
    L = 2 * n - 1
    # Variables: p (n), q (n), t (1) -> total 2n+1
    # d = p - q
    # Maximize t
    # s.t. G[i,:] @ d <= -t for i in active
    #      G[i,:] @ d <= c_scaled[i] + tol_inactive for i not in active (don't exceed current + margin)
    #      sum(d) = 0
    #      sum(p + q) <= 1
    #      p, q >= 0
    
    nvars = 2 * n + 1
    # c vector for objective: maximize t -> minimize -t
    c_obj = np.zeros(nvars)
    c_obj[-1] = -1.0  # minimize -t = maximize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag i:
    #   G[i,:] @ (p-q) <= rhs_i
    #   G[i,0]*p[0] - G[i,0]*q[0] + ... + G[i,n-1]*p[n-1] - G[i,n-1]*q[n-1] <= rhs_i
    
    A_rows = []
    b_rows = []
    
    for i in range(L):
        if i in active_idx:
            # G[i,:] @ d <= -t
            # G[i,:] @ p - G[i,:] @ q + t <= 0
            row = np.zeros(nvars)
            row[:n] = G[i, :]
            row[n:2*n] = -G[i, :]
            row[2*n] = 1.0
            A_rows.append(row)
            b_rows.append(0.0)
        else:
            # G[i,:] @ d <= c_scaled[i] + tol_inactive
            row = np.zeros(nvars)
            row[:n] = G[i, :]
            row[n:2*n] = -G[i, :]
            A_rows.append(row)
            b_rows.append(c_scaled[i] + tol_inactive)
    
    # sum(d) = sum(p) - sum(q) = 0 -> equality
    # sum(p + q) <= 1 -> inequality
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Add sum(p+q) <= 1
    row = np.zeros(nvars)
    row[:2*n] = 1.0
    A_ub = np.vstack([A_ub, row.reshape(1, -1)])
    b_ub = np.append(b_ub, 1.0)
    
    # Equality: sum(p) - sum(q) = 0
    A_eq_row = np.zeros(nvars)
    A_eq_row[:n] = 1.0
    A_eq_row[n:2*n] = -1.0
    A_eq = A_eq_row.reshape(1, -1)
    b_eq = np.array([0.0])
    
    bounds = [(0, None)] * (2*n) + [(None, None)]  # p,q >= 0, t free
    
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                               bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n] - res.x[n:2*n]
            t_val = res.x[2*n]
            if t_val > 1e-10:  # ensure positive reduction
                # Center d (should already be centered, but enforce)
                d = d - np.mean(d)
                norm = np.max(np.abs(d))
                if norm > 1e-12:
                    d = d / norm
                    return d
    except Exception:
        pass
    return None

import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def all_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_target(v, n_target):
    h = np.asarray(v, dtype=float).ravel().copy()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    # Bisection for shift lambda
    lo, hi = -max(1.0 - h.min(), 1.0) - 0.1, max(1.0 - h.max(), 1.0) + 0.1
    lo, hi = max(lo, -2.0), min(hi, 2.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_try) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact mass correction
    for _ in range(500):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for i in range(L):
        k = i - (n - 1)
        # Term 1: (1-h[m-k]) for valid m where 0<=m-k<n
        m1 = np.arange(max(0, k), min(n, k + n))
        if len(m1) > 0:
            G[i, m1] += dx * (1.0 - h[m1 - k])
        # Term 2: -h[m+k] for valid m where 0<=m+k<n
        m2 = np.arange(max(0, -k), min(n, n - k))
        if len(m2) > 0:
            G[i, m2] += dx * (-h[m2 + k])
    return G

def find_optimal_delta(c, slopes, lo, hi):
    """Find delta in [lo,hi] minimizing max_k(c[k] + delta*slopes[k])"""
    if hi - lo < 1e-15:
        return 0.0
    
    def f(d):
        return np.max(c + d * slopes)
    
    best_val = f(lo)
    best_d = lo
    v_hi = f(hi)
    if v_hi < best_val:
        best_val = v_hi
        best_d = hi
    if lo < 0.0 < hi:
        v_0 = f(0.0)
        if v_0 < best_val:
            best_val = v_0
            best_d = 0.0
    
    # Check if function is monotone
    vals_lo = c + lo * slopes
    active_lo = vals_lo >= np.max(vals_lo) - 1e-10
    if np.all(slopes[active_lo] >= -1e-12):
        return lo
    vals_hi = c + hi * slopes
    active_hi = vals_hi >= np.max(vals_hi) - 1e-10
    if np.all(slopes[active_hi] <= 1e-12):
        return hi
    
    # Binary search for subgradient zero
    a, b = lo, hi
    for _ in range(60):
        mid = (a + b) / 2.0
        vals = c + mid * slopes
        max_v = np.max(vals)
        active = vals >= max_v - 1e-10
        sl = slopes[active]
        if np.min(sl) >= -1e-12:
            b = mid
        elif np.max(sl) <= 1e-12:
            a = mid
        else:
            break
    
    best_d = (a + b) / 2.0
    best_val = f(best_d)
    if f(lo) < best_val:
        best_d = lo
    if f(hi) < best_val:
        best_d = hi
    
    return best_d

        G = grad_matrix(h)
        c = all_correlations(h)
        improved = False
        
        for i in range(n):
            for j in range(i + 1, n):
                if time.perf_counter() - start > time_limit:
                    break
                slopes = G[:, i] - G[:, j]
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if lo >= hi - 1e-14:
                    continue
                delta = find_optimal_delta(c, slopes, lo, hi)
                if abs(delta) < 1e-11:
                    continue
                h[i] += delta
                h[j] -= delta
                s_trial = official_score(h)
                if s_trial < score - 1e-13:
                    score = s_trial
                    improved = True
                    G = grad_matrix(h)
                    c = all_correlations(h)
                else:
                    h[i] -= delta
                    h[j] += delta
        
        if not improved:
            break
    
    return h, score

def lp_local_descent(h, max_iters=30, time_limit=30.0):
    """LP-based local descent using active set constraints"""
    n = len(h)
    L = 2 * n - 1
    start = time.perf_counter()
    score = official_score(h)
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        c = all_correlations(h)
        G = grad_matrix(h)
        S = np.max(c)
        
        # Active set: correlations within tolerance of max
        tol = max(1e-6, 0.01 * S)
        active = np.where(c >= S - tol)[0]
        
        # Build LP
        # Variables: p(n), q(n), t(1) = 2n+1
        nvars = 2 * n + 1
        c_obj = np.zeros(nvars)
        c_obj[-1] = -1.0  # maximize t
        
        A_rows = []
        b_rows = []
        
        margin_inactive = 0.01 * S + 1e-4
        
        for i in range(L):
            row = np.zeros(nvars)
            row[:n] = G[i, :]
            row[n:2*n] = -G[i, :]
            if i in active:
                # G[i,:] @ d + t <= (S - c[i])  =>  G[i,:] @ d + t <= -(c[i] - S) = S - c[i]
                # We want new value c[i] + G[i,:]@d <= S - t
                # => G[i,:]@d + t <= S - c[i]
                row[2*n] = 1.0
                A_rows.append(row)
                b_rows.append(S - c[i])
            else:
                # Don't let inactive exceed S + margin
                # G[i,:]@d <= S + margin - c[i]
                A_rows.append(row)
                b_rows.append(S + margin_inactive - c[i])
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        # sum(p+q) <= 1
        row = np.zeros(nvars)
        row[:2*n] = 1.0
        A_ub = np.vstack([A_ub, row.reshape(1, -1)])
        b_ub = np.append(b_ub, 1.0)
        
        # Equality: sum(p) - sum(q) = 0
        A_eq = np.zeros((1, nvars))
        A_eq[0, :n] = 1.0
        A_eq[0, n:2*n] = -1.0
        b_eq = np.array([0.0])
        
        bounds = [(0, None)] * (2*n) + [(0, None)]
        
        try:
            res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                                    b_eq=b_eq, bounds=bounds, method='highs')
            if res.success and res.x is not None:
                d = res.x[:n] - res.x[n:2*n]
                t_val = res.x[2*n]
                if t_val > 1e-10:
                    # Line search
                    norm_d = np.max(np.abs(d))
                    if norm_d > 1e-12:
                        d = d / norm_d
                        best_alpha = 0.0
                        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
                            h_try = project_to_target(h + alpha * d, n)
                            s_try = official_score(h_try)
                            if s_try < score - 1e-12:
                                best_alpha = alpha
                        if best_alpha > 0:
                            h = project_to_target(h + best_alpha * d, n)
                            score = official_score(h)
                            continue
        except Exception:
            pass
        
        break
    
    return h, score
