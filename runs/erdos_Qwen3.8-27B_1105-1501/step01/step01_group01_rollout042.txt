def binary_refine(h, max_sweeps=1000, verbose=False):
    """Refine a near-binary h using pair-swaps."""
    n = len(h)
    h = h.copy()
    
    # We work with the exact S values
    S = np.correlate(h, 1.0 - h, mode='full').copy()
    best_max = np.max(S)
    
    for sweep in range(max_sweeps):
        improved = False
        # For each 0-position and 1-position, try swapping
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        
        for p in zeros:
            for q in ones:
                # Compute new max S
                # Linear effect of h[p] += 1
                # and h[q] -= 1
                new_S = S.copy()
                
                # Effect of h[p] += 1 (p goes 0→1)
                # Case 1: p is first index, k in [n-1-p, 2n-2-p]
                k_lo1 = n - 1 - p
                k_hi1 = 2 * n - 2 - p
                for k in range(max(0, k_lo1), min(2*n-1, k_hi1 + 1)):
                    j = p + k - (n - 1)
                    new_S[k] += (1.0 - h[j])
                
                # Case 2: p is second index, k in [p, n-1+p]
                k_lo2 = p
                k_hi2 = n - 1 + p
                for k in range(max(0, k_lo2), min(2*n-1, k_hi2 + 1)):
                    i = p + k - (n - 1)
                    new_S[k] += -h[i]
                
                # Effect of h[q] -= 1 (q goes 1→0)
                # Case 1: q is first index, k in [n-1-q, 2n-2-q]
                k_lo1 = n - 1 - q
                k_hi1 = 2 * n - 2 - q
                for k in range(max(0, k_lo1), min(2*n-1, k_hi1 + 1)):
                    j = q + k - (n - 1)
                    new_S[k] += -(1.0 - h[j])
                
                # Case 2: q is second index, k in [q, n-1+q]
                k_lo2 = q
                k_hi2 = n - 1 + q
                for k in range(max(0, k_lo2), min(2*n-1, k_hi2 + 1)):
                    i = q + k - (n - 1)
                    new_S[k] += h[i]
                
                # Cross terms
                # (i,j)=(p,q): k = (n-1)+(q-p)
                k_cross1 = (n - 1) + (q - p)
                if 0 <= k_cross1 < 2*n-1:
                    new_S[k_cross1] += 1.0
                
                # (i,j)=(q,p): k = (n-1)+(p-q)
                k_cross2 = (n - 1) + (p - q)
                if 0 <= k_cross2 < 2*n-1:
                    new_S[k_cross2] += 1.0
                
                new_max = np.max(new_S)
                if new_max < best_max - 1e-10:
                    # Accept
                    h[p] = 1.0
                    h[q] = 0.0
                    S = new_S
                    best_max = new_max
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return h

                # delta_S = inc[p] - inc[q] + cross
                delta_S = inc[p] - inc[q]
                # Cross terms
                k1 = (n - 1) + (q - p)
                k2 = (n - 1) + (p - q)
                delta_S[k1] += 1.0
                delta_S[k2] += 1.0
                
                new_S = S + delta_S
                new_max = float(np.max(new_S))
                
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_pair = (p, q)
                    best_delta_S = delta_S
                    break
            if best_pair is not None:
                break
        
        if best_pair is None:
            break
        
        p, q = best_pair
        h[p] = 1.0
        h[q] = 0.0
        S = S + best_delta_S
        improved = True
    
    return h
