import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        # Distribute the difference
        if diff > 0:
            order = np.argsort(z)  # most room first
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(-z)  # most to remove first
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_overlaps_and_grads(h):
    """Compute overlap values and gradient matrix."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx  # shape (2n-1,)
    
    # Gradient: grad[k, m] = d(overlaps[k])/dh[m]
    # overlaps[k] = dx * sum_i h[i] * (1-h[i+d]) where d = k-(n-1)
    # d(overlaps[k])/dh[m] = dx * [(1-h[m+d]) if 0<=m+d<n] + dx * [-h[m-d]] if 0<=m-d<n
    
    d_vals = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # For each k and each m, compute gradient
    # Use broadcasting: for lag index k (with offset d_k), 
    # grad[k, m] = dx * (1-h[m+d_k]) * [0<=m+d_k<n] + dx * (-h[m-d_k]) * [0<=m-d_k<n]
    
    # More efficiently:
    # First term: for offset d, contribution to grad[k, m] is dx*(1-h[m+d]) for m in [0, n-1-d] if d>=0, or m in [-d, n-1] if d<0
    # Second term: for offset d, contribution is -dx*h[m-d] for m in [d, n-1] if d>=0, or m in [0, n-1+d] if d<0
    
    # Let me compute this more carefully using the structure.
    # For k corresponding to offset d:
    # overlaps[k] = dx * sum_{i=max(0,-d)}^{min(n-1, n-1-d)} h[i] * (1-h[i+d])
    
    # grad[k, m] = dx * (1-h[m+d]) * I(0<=m<n and 0<=m+d<n)
    #            + dx * (-h[m-d]) * I(0<=m<n and 0<=m-d<n)
    
    # For the first term: 0<=m<n and 0<=m+d<n means max(0,-d) <= m <= min(n-1, n-1-d)
    # For the second term: 0<=m<n and 0<=m-d<n means max(0,d) <= m <= min(n-1, n-1+d)
    
    # Let me build the gradient as a (2n-1) x n matrix
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n - 1)
        # First term: (1-h[m+d]) for valid m
        m_low = max(0, -d)
        m_high = min(n-1, n-1-d)
        if m_low <= m_high:
            grad[k, m_low:m_high+1] += dx * (1.0 - h[m_low+d:m_high+d+1])
        # Second term: -h[m-d] for valid m
        m_low2 = max(0, d)
        m_high2 = min(n-1, n-1+d)
        if m_low2 <= m_high2:
            grad[k, m_low2:m_high2+1] += dx * (-h[m_low2-d:m_high2-d+1])
    
    return overlaps, grad

def lp_solve(h, n, focus_lags=None):
    """Solve the LP linearization step."""
    dx = 2.0 / n
    overlaps, grad = compute_overlaps_and_grads(h)
    
    # LP: min t
    # s.t. t - sum_m grad[k,m]*h'[m] >= overlaps[k] - sum_m grad[k,m]*h[m]
    #   => t - sum_m grad[k,m]*h'[m] >= overlaps[k] - overlap_contrib
    #   where overlap_contrib = sum_m grad[k,m]*h[m] = overlaps[k] (by Euler's theorem for homogeneous? No, not homogeneous)
    #   
    # Wait: overlaps[k] = sum_i h[i]*(1-h[i+d])*dx
    # grad[k,m] = d(overlaps[k])/dh[m]
    # sum_m grad[k,m]*h[m] = d(overlaps[k])/dh[m] * h[m] summed... this is not simply overlaps[k]
    # because overlaps is not homogeneous.
    
    # Let me just compute the RHS properly:
    # Linearization: overlaps[k](h') ≈ overlaps[k](h) + sum_m grad[k,m] * (h'[m] - h[m])
    # We want: t >= overlaps[k](h) + sum_m grad[k,m] * (h'[m] - h[m])
    #         t - sum_m grad[k,m] * h'[m] >= overlaps[k](h) - sum_m grad[k,m] * h[m]
    
    rhs_base = overlaps - grad @ h  # shape (2n-1,)
    
    if focus_lags is not None:
        k_sel = focus_lags
    else:
        k_sel = np.arange(2*n-1)
    
    n_cons = len(k_sel)
    
    # c = [0, 0, ..., 0, 1] (minimize t, which is the last variable)
    # Variables: h'[0], ..., h'[n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # t - sum_m grad[k,m]*h'[m] >= rhs_base[k]
    # => -t + sum_m grad[k,m]*h'[m] <= -rhs_base[k]
    A_ub = np.zeros((n_cons, n+1))
    for i, k in enumerate(k_sel):
        A_ub[i, :n] = grad[k, :]
        A_ub[i, -1] = -1.0
    b_ub = -rhs_base[k_sel]
    
    # Equality: sum(h') = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds: 0 <= h'[m] <= 1, t >= 0 (actually t can be anything, but let's bound it)
    bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]
    return None, None

    best_c = cur_c
    alpha = 1.0

            cur_c = c_try

            if cur_c < best_c:

                best_c = cur_c

                    cur_c = c_frac
                    alpha = max(0.1, a * 0.8)
                    stagnant = 0
                    if cur_c < best_c:

                        best_c = cur_c
                    found = True
                    break
            if not found:

                    alpha = 1.0
                    stagnant = 0
    
    return best, best_c

    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = grad[k_sel]
    A_ub[:, -1] = -1.0
    b_ub = -rhs_base
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(-0.5, 0.5)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1], res
    return None, None, None

def lp_refine(h, deadline, max_iters=500, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.default_rng(0)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    alpha = 1.0
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_lp, t_lp, lres = lp_step(cur)
        if h_lp is None:
            break
        h_try = make_feasible(h_lp, n)
        c_try = score_exact(h_try)
        if c_try < cur_c - 1e-12:
            cur = h_try
            cur_c = c_try
            alpha = min(1.0, alpha * 1.08)
            stagnant = 0
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
        else:
            found = False
            for a in (alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05):
                if a < 0.01:
                    break
                h_frac = make_feasible((1 - a) * cur + a * h_lp, n)
                c_frac = score_exact(h_frac)
                if c_frac < cur_c - 1e-12:
                    cur = h_frac
                    cur_c = c_frac
                    alpha = max(0.05, a * 0.7)
                    stagnant = 0
                    if cur_c < best_c:
                        best = cur.copy()
                        best_c = cur_c
                    found = True
                    break
            if not found:
                alpha *= 0.6
                stagnant += 1
                if stagnant > 50:
                    cur = make_feasible(best + rng.normal(0, 0.01, n), n)
                    cur_c = score_exact(cur)
                    alpha = 1.0
                    stagnant = 0
    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=40):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(20):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        s = score_exact(h)
        if s < cur_c:
            return float(s)
        h[i] = orig_i
        h[j] = orig_j
    return None

def coord_search(h, i, cur_c, n_pts=30):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(score_exact(h))
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def smooth_grad_step(h, T, n):
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c0 = n - 1
    g += w[c0] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c0 + d] * dx
        wn = w[c0 - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g

def grad_descent(h, deadline, T_start, T_end, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        g = smooth_grad_step(cur, T, n)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.85
            stall += 1
            if stall > 50:
                cur = make_feasible(cur + rng.normal(0, max(0.005, T * 0.3), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.005)
                stall = 0
        else:
            step = min(step * 1.05, T * 2.0)
            stall = 0
        T = max(T_end, T * 0.9995)
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: LP refinement (35% of time)
    p1_end = t0 + budget_s * 0.35
    h1, c1 = lp_refine(best_h, p1_end, max_iters=1000, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1
    n = len(best_h)

    # Phase 2: Local search with pair and coordinate moves (15% of time)
    p2_end = t0 + budget_s * 0.50
    cur = best_h.copy()
    cur_c = best_c
    consecutive_fail = 0
    while time.time() < p2_end:
        mode = rng.random()
        if mode < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=35)
        elif mode < 0.75:
            i = int(rng.integers(0, n))
            res = coord_search(cur, i, cur_c, n_pts=30)
        else:
            i = int(rng.integers(0, n - 1))
            res = pair_search(cur, i, i + 1, cur_c, n_pts=40)
        if res is not None:
            cur_c = res
            consecutive_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
        else:
            consecutive_fail += 1
            if consecutive_fail > 400:
                cur = make_feasible(best_h + rng.normal(0, 0.006, n), n)
                cur_c = score_exact(cur)
                consecutive_fail = 0

    # Phase 3: Second LP round (15% of time)
    p3_end = t0 + budget_s * 0.65
    h3, c3 = lp_refine(best_h, p3_end, max_iters=800, rng=rng)
    if c3 < best_c:
        best_h, best_c = h3, c3
    n = len(best_h)

    # Phase 4: Gradient descent polish (10% of time)
    p4_end = t0 + budget_s * 0.75
    h4, c4 = grad_descent(best_h, p4_end, 0.005, 0.0001, 0.002, rng)
    if c4 < best_c:
        best_h, best_c = h4, c4
    n = len(best_h)

    # Phase 5: Systematic adjacent pair sweep (10% of time)
    p5_end = t0 + budget_s * 0.85
    cur = best_h.copy()
    cur_c = best_c
    sweeps = 0
    while time.time() < p5_end:
        improved = False
        for i in range(n - 1):
            if time.time() > p5_end:
                break
            res = pair_search(cur, i, i + 1, cur_c, n_pts=30)
            if res is not None:
                cur_c = res
                improved = True
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c
        if not improved:
            for _ in range(40):
                if time.time() > p5_end:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                res = pair_search(cur, i, j, cur_c, n_pts=25)
                if res is not None:
                    cur_c = res
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
        sweeps += 1

    # Phase 6: Multi-resolution exploration (7% of time)
    p6_end = deadline - 8
    if time.time() < p6_end:
        n_cur = len(best_h)
        candidates = []
        for dn in (-6, -4, -2, 2, 4, 6):
            nr = n_cur + dn
            if 20 <= nr <= 200:
                candidates.append(nr)
        for nr in candidates:
            if time.time() > p6_end - 5:
                break
            h_r = resample(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(p6_end - 3, time.time() + 20)
            h_r, c_r = lp_refine(h_r, sub_dl, max_iters=80, rng=rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    # Final LP polish
    if time.time() < deadline - 2:
        h_f, c_f = lp_refine(best_h, deadline - 2, max_iters=200, rng=rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final feasibility check
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
