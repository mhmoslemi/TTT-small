import numpy as np
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_all_lags(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return np.max(compute_all_lags(h))

def solve_1d_max_lines(c, g, lo, hi):

        elif np.all(g_act < -1e-12):

    val_hi = np.max(c + hi * g)
    if val_lo < val:
        delta, val = lo, val_lo
    if val_hi < val:
        delta, val = hi, val_hi
    
    return delta, val

def pair_sweep(h, s_vals):
    """One full sweep of pair exchanges."""
    N = len(h)
    c = compute_all_lags(h)
    current_max = c.max()
    
    for i in range(N):
        for j in range(N):
            if i == j:
                continue
            lo = max(h[i] - 1.0, -h[j])
            hi = min(h[i], 1.0 - h[j])
            if hi - lo < 1e-10:
                continue
            
            # Compute g vector
            mask_ims = (i - s_vals >= 0) & (i - s_vals < N)
            mask_jms = (j - s_vals >= 0) & (j - s_vals < N)
            mask_ips = (i + s_vals >= 0) & (i + s_vals < N)
            mask_jps = (j + s_vals >= 0) & (j + s_vals < N)
            
            g = np.zeros(2 * N - 1)
            g[mask_ims] -= (1.0 - h[i - s_vals[mask_ims]])
            g[mask_jms] += (1.0 - h[j - s_vals[mask_jms]])
            g[mask_ips] += h[i + s_vals[mask_ips]]
            g[mask_jps] -= h[j + s_vals[mask_jps]]
            
            delta, new_val = solve_1d_max_lines(c, g, lo, hi)
            
            if new_val < current_max - 1e-12:
                h[i] -= delta
                h[j] += delta
                if h[i] < 0: h[i] = 0
                if h[j] > 1: h[j] = 1
                c += delta * g
                current_max = c.max()
    
    return h, current_max
