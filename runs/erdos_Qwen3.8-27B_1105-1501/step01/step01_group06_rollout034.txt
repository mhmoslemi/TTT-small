import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def get_max_corr_binary(h):
    """For binary h, get the maximum correlation value (times dx is C5)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def binary_sa(n, n_iters, t_start, t_end, seed=42):
    rng = np.random.default_rng(seed)
    # Initialize
    pos = np.arange(n)
    rng.shuffle(pos)
    h = (pos < n // 2).astype(float)  # n/2 ones, n/2 zeros
    
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    energy = float(np.max(corr0))
    best_energy = energy
    best_h = h.copy()
    
    for it in range(n_iters):
        T = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
        if T < 1e-10:
            break
            
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_energy = float(np.max(corr_new))
        delta = new_energy - energy
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            energy = new_energy
            if energy < best_energy:
                best_energy = energy
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_energy

def local_search_binary(h, n, max_passes=100):
    """Greedy local search: try all swaps, take the best improvement."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    energy = float(np.max(corr))
    best_h = h.copy()
    best_energy = energy
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _pass in range(max_passes):
        improved = False
        best_delta = 0
        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                new_energy = float(np.max(corr_new))
                delta = energy - new_energy  # positive = improvement
                h[i], h[j] = h[j], h[i]
                
                if delta > best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
        
        if best_delta > 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            energy = float(np.max(corr))
            if energy < best_energy:
                best_energy = energy
                best_h = h.copy()
            improved = True
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        
        if not improved:
            break
    
    return best_h, best_energy

```python
def directed_local_search(h, n, max_passes=100):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    energy = float(np.max(corr))
    best_h = h.copy()
    best_energy = energy
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for _pass in range(max_passes):
        # Find worst lag
        idx_max = np.argmax(corr)
        worst_lag = idx_max - (n - 1)  # actual lag value
        
        # Find all (1,0) pairs at this lag
        if worst_lag > 0:
            # c(e) = sum_{i=0}^{n-1-e} h[i]*(1-h[i+e])
            # Pairs: (i, i+e) where h[i]=1, h[i+e]=0
            e = worst_lag
            bad_pairs = [(i, i+e) for i in range(n-e) if h[i] == 1 and h[i+e] == 0]
        elif worst_lag < 0:
            e = -worst_lag
            # For negative lag -e: c(-e) = sum_{i=0}^{n-1-e} h[i+e]*(1-h[i])
            # Pairs: (i+e, i) where h[i+e]=1, h[i]=0
            # Equivalent: for i where h[i]=-e position has 1 and i has 0
            bad_pairs = [(i+e, i) for i in range(n-e) if h[i+e] == 1 and h[i] == 0]
        else:
            break  # Lag 0, can't improve for binary
        
        if not bad_pairs:
            break
        
        # Try to break the worst pair
        best_delta = 0
        best_swap = None
        for (a, b) in bad_pairs:
            # h[a] = 1, h[b] = 0
            # Swap a with some zero position z (z != b)
            for z in zeros:
                if z == b:
                    continue
                # Swap h[a] (was 1) with h[z] (was 0)
                h[a] = 0
                h[z] = 1
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                new_energy = float(np.max(corr_new))
                delta = energy - new_energy
                h[a] = 1
                h[z] = 0
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (a, z)
            # Also try swapping b with some one position o (o != a)
            for o in ones:
                if o == a:
                    continue
                h[o] = 0
                h[b] = 1
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                new_energy = float(np.max(corr_new))
                delta = energy - new_energy
                h[o] = 1
                h[b] = 0
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (b, o)
        
        if best_swap is None or best_delta <= 0:
            break
        
        a, b = best_swap
        h[a], h[b] = h[b], h[a]
        corr = np.correlate(h, 1.0 - h, mode='full')
        energy = float(np.max(corr))
        if energy < best_energy:
            best_energy = energy
            best_h = h.copy()
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
    
    return best_h, best_energy
```

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def max_corr_binary(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(np.max(corr))

def sa_binary(n, n_iters, t_init, t_final, seed):
    rng = np.random.default_rng(seed)
    # Random initialization with n//2 ones
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:n//2]] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    energy = int(np.max(corr))
    best_energy = energy
    best_h = h.copy()
    
    for it in range(n_iters):
        T = t_init * (t_final / t_init) ** (it / max(n_iters - 1, 1))
        if T < 1e-6:
            break
            
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or h[i] == h[j]:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_energy = int(np.max(corr))
        delta = new_energy - energy
        
        if delta <= 0:
            energy = new_energy
            if energy < best_energy:
                best_energy = energy
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-float(delta) / T):
                energy = new_energy
            else:
                h[i] = 1.0
                h[j] = 0.0
    
    return best_h, best_energy

Let me optimize: instead of recomputing the full correlation each time, I can just compute the new lag counts that are affected by adding a 1 at position pos.

When I set h[pos] = 1 (from 0):
- For lag d > 0: the new contribution at position pos is h[pos]*(1-h[pos+d]) = 1*(1-h[pos+d]).
  So if h[pos+d] = 0, the count at lag d increases by 1.
- For lag d < 0: the new contribution is that h[pos] is now the "target" of a lag.
  For lag -d (d>0): h[pos-d]*(1-h[pos]) = h[pos-d]*1.
  So if h[pos-d] = 1, the count at lag -d increases by 1.

So the change in counts is:
- For d = 1 to n-1-pos: if h[pos+d] == 0, count[d] increases by 1.
- For d = 1 to pos: if h[pos-d] == 1, count[-d] increases by 1.

This is O(n) per position evaluation, and O(n²) total per step. For n/2 steps: O(n³). For n=100: 1M operations. Fast!
