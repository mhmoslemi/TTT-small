    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
    else:
        m = -s
        if m == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] += (-h[:n-m])
    return grad

def subgradient_optimize(h0, max_iter=3000, time_limit=None, t_start=None):
    """Subgradient descent with projection for the max correlation."""
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    alpha = 0.05
    patience = 0
    max_patience = 500
    
    for it in range(max_iter):
        if time_limit is not None:
            if t_start is not None and time.time() - t_start > time_limit:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        # Active set: lags within 2% of max
        threshold = max_val - 0.02 * max_val
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        # Average gradient over active lags
        grad = np.zeros(n)
        for k in active:
            grad += compute_grad_k(h, k)
        grad /= len(active)
        # Step
        h_new = h - alpha * grad
        h_new = make_feasible(h_new)
        new_score = official_score(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            h = h_new
            patience = 0
            alpha = min(alpha * 1.02, 0.8)
        else:
            h = h_new
            patience += 1
            alpha = max(alpha * 0.99, 0.005)
        if patience > max_patience:
            break
    return best_h, best_score

def simulated_annealing(h0, rng, n_swaps=20000, t_start=None, time_limit=None):
    """Simulated annealing with swap moves on near-binary solution."""
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    T = 0.02
    T_end = 0.001
    n_iters = n_swaps
    cool_rate = (T_end / T) ** (1.0 / max(n_iters, 1))
    
    # Identify one-positions and zero-positions for swap
    # For non-binary, use threshold
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        # Pick random i (to decrease) and j (to increase)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Swap amount: move from larger to smaller
        if h[i] >= h[j]:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        delta = min(h[donor], 1.0 - h[receiver])
        if delta < 0.01:
            continue
        # Try different swap amounts
        swap = delta * rng.uniform(0.3, 1.0)
        h[donor] -= swap
        h[receiver] += swap
        new_score = official_score(h)
        delta_obj = new_score - best_score
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
        else:
            if rng.random() > np.exp(-delta_obj / T):
                # Reject: undo
                h[donor] += swap
                h[receiver] -= swap
        T *= cool_rate
    return best_h, best_score

def binary_local_search(h0, rng, max_rounds=500, time_limit=None, t_start=None):
    """Greedy local search: try all swaps, accept best improvement."""
    h = h0.copy()
    n = len(h)
    best_score = official_score(h)
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        # Find one-vals and zero-vals (or near)
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = 0
        best_swap = None
        # Sample swaps if too many combinations
        if len(ones) * len(zeros) > 500:
            candidates = rng.choice(len(ones)*len(zeros), min(200, len(ones)*len(zeros)), replace=False)
        else:
            candidates = range(len(ones)*len(zeros))
        
        idx = 0
        for ci in candidates:
            ii = ci // len(zeros)
            jj = ci % len(zeros)
            i = ones[ii]
            j = zeros[jj]
            if i == j:
                continue
            delta = min(h[i], 1.0 - h[j])
            if delta < 0.05:
                continue
            # Try swapping
            save_i, save_j = h[i], h[j]
            h[i] = save_i - delta
            h[j] = save_j + delta
            s = official_score(h)
            improve = best_score - s
            if improve > best_improve:
                best_improve = improve
                best_swap = (i, j, delta)
            h[i], h[j] = save_i, save_j
        
        if best_swap is not None:
            i, j, delta = best_swap
            h[i] -= delta
            h[j] += delta
            h = make_feasible(h)
            best_score = official_score(h)
        else:
            break
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)
    
    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    
    rng = np.random.default_rng(seed)
    
    # n values to try
    n_options = [48, 50, 52, 54, 56, 58, 60, 62, 64]
    if len(parent) in n_options:
        pass
    else:
        n_options.append(len(parent))
    n_options = sorted(set(n_options))
    n_options = [n for n in n_options if 24 <= n <= 76]
    
    for n in n_options:
        if time.time() > deadline - 30:
            break
        
        # Generate starting points for this n
        starts = []
        h_interp = make_feasible(interp_to_n(parent, n))
        starts.append(h_interp)
        
        # Binary push versions
        for p in [1.3, 2.0, 4.0]:
            starts.append(push_binary(h_interp, p))
        
        # Random binary
        if n % 2 == 0:
            n_ones = n // 2
            rb = np.zeros(n)
            rb[rng.choice(n, n_ones, replace=False)] = 1.0
            starts.append(rb)
        else:
            n_ones = n // 2
            rb = np.zeros(n)
            idx = rng.choice(n, n_ones, replace=False)
            rb[idx] = 1.0
            rb[rng.integers(n)] = 0.5  # fractional to hit target n/2
            starts.append(make_feasible(rb))
        
        # Random perturbations
        for sigma in [0.05, 0.15, 0.3]:
            starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))
        
        # Anti-symmetric variant
        starts.append(make_feasible(1.0 - h_interp[::-1].copy()))
        
        for h0 in starts:
            if time.time() > deadline - 15:
                break
            cur_h = h0.copy()
            cur_score = official_score(cur_h)
            
            # Phase 1: Simulated annealing (fast, good for binary structures)
            if time.time() < deadline - 20:
                n_swaps = 15000 if n <= 56 else 8000
                sa_h, sa_score = simulated_annealing(
                    cur_h, rng, n_swaps=n_swaps,
                    time_limit=60, t_start=time.time()
                )
                if sa_score < cur_score - 1e-10:
                    cur_h = sa_h
                    cur_score = sa_score
            
            # Phase 2: Binary local search
            if time.time() < deadline - 15:
                bs_h, bs_score = binary_local_search(
                    cur_h, rng, max_rounds=200,
                    time_limit=40, t_start=time.time()
                )
                if bs_score < cur_score - 1e-10:
                    cur_h = bs_h
                    cur_score = bs_score
            
            # Phase 3: Smooth LSE with SLSQP (multiple betas)
            for beta in [50, 200, 800]:
                if time.time() > deadline - 10:
                    break
                mi = 150 if n <= 56 else 80
                new_h = do_slsqp_smooth(cur_h, beta, maxiter=mi)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-12:
                    cur_h = new_h
                    cur_score = new_score
            
            # Phase 4: Subgradient refinement
            if time.time() < deadline - 10:
                sg_h, sg_score = subgradient_optimize(
                    cur_h, max_iter=2000,
                    time_limit=30, t_start=time.time()
                )
                if sg_score < cur_score - 1e-10:
                    cur_h = sg_h
                    cur_score = sg_score
            
            # Phase 5: Direct min-max SLSQP polish
            if time.time() < deadline - 10:
                dc_h = do_slsqp_direct(cur_h, maxiter=100)
                dc_score = official_score(dc_h)
                if dc_score < cur_score - 1e-12:
                    cur_h = dc_h
                    cur_score = dc_score
            
            # Phase 6: Another SA round on the smoothed result
            if time.time() < deadline - 10:
                sa_h, sa_score = simulated_annealing(
                    cur_h, rng, n_swaps=10000,
                    time_limit=30, t_start=time.time()
                )
                if sa_score < cur_score - 1e-10:
                    cur_h = sa_h
                    cur_score = sa_score
            
            # Phase 7: Final SLSQP
            if time.time() < deadline - 5:
                new_h = do_slsqp_smooth(cur_h, 200, maxiter=100)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-12:
                    cur_h = new_h
                    cur_score = new_score
            
            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
    
    # Final refinement loop on the best solution found
    remaining = deadline - time.time()
    if remaining > 20:
        n_trials = min(20, int(remaining / 15))
        for trial in range(n_trials):
            if time.time() > deadline - 10:
                break
            h_pert = make_feasible(best_h + 0.03 * rng.standard_normal(best_n))
            # SA
            sa_h, sa_s = simulated_annealing(h_pert, rng, n_swaps=8000,
                                             time_limit=15, t_start=time.time())
            # SLSQP
            sl_h = do_slsqp_smooth(sa_h, 300, maxiter=100)
            sl_s = official_score(sl_h)
            if sl_s < best_score - 1e-12:
                best_h = sl_h.copy()
                best_score = sl_s
    
    # Final direct polish
    if time.time() < deadline - 5:
        final_h = do_slsqp_direct(best_h, maxiter=150)
        final_s = official_score(final_h)
        if final_s < best_score:
            best_h = final_h
            best_score = final_s
        
        # One more smooth pass
        final_h2 = do_slsqp_smooth(best_h, 500, maxiter=100)
        final_s2 = official_score(final_h2)
        if final_s2 < best_score:
            best_h = final_h2
            best_score = final_s2
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
