            val += (1.0 - h[jt])
        jm = j - tau
        if 0 <= jm < n:
            val -= h[jm]

        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for idx in active_idx:
            tau = int(idx) - (n - 1)
            grad += weights[idx] * grad_lag(h, tau)
        grad *= (2.0 / n)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            stall += 1
            if stall > stall_limit:
                break
            h = project_feasible(best_h + 0.05 * rng.randn(n), target)
            stall = 0
            continue
        direction = -grad / gnorm
        h_new, c5_new = line_search_multi(h, direction, target)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            h_new2, c5_new2 = line_search_multi(h, direction * 0.3, target)
            if c5_new2 < best_c5 - 1e-14:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                stall = 0
                h = h_new2
            else:
                h = h_new
        temp = max(temp * temp_decay, temp_min)
        iter_count += 1
        if stall > stall_limit:
            amp = min(0.15, 0.03 * (stall / stall_limit))
            h = project_feasible(best_h + amp * rng.randn(n), target)
            stall = 0
            temp = max(temp, 0.3)
    return best_h, best_c5


def refine_pairs(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = h0.astype(np.float64).copy()
    c5_cur = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deltas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]
    while time.time() - t0 < time_budget:
        if time.time() - t0 > time_budget:
            break
        for _ in range(20000):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] < 1e-9 or h[j] < 1e-9:
                continue
            delta = rng.choice(deltas)
            sign = rng.choice([-1, 1])
            di = sign * delta
            if h[i] + di < -1e-9 or h[i] + di > 1.0 + 1e-9:
                continue
            if h[j] - di < -1e-9 or h[j] - di > 1.0 + 1e-9:
                continue
            h[i] += di
            h[j] -= di
            c5_test = compute_c5(h)
            if c5_test < c5_cur - 1e-13:
                c5_cur = c5_test
            else:
                h[i] -= di
                h[j] += di
    return h, c5_cur


def resample_to(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old + 1)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_interp = np.interp(x_new, x_old[:-1] + (x_old[1] - x_old[0]) / 2, h_old)
    target = float(n_new) / 2.0
    h_interp = project_feasible(h_interp, target)
    return h_interp


def random_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    h = h[rng.permutation(n)]
    return h


def structured_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    frac = n % 2
    pattern = np.zeros(n)
    step = n / half
    for i in range(half):
        pos = int(round(i * step)) % n
        pattern[pos] = 1.0
    if frac == 1:
        zeros = np.where(pattern == 0)[0]
        if len(zeros) > 0:
            pattern[zeros[0]] = 0.5
    h = pattern[rng.permutation(n)]
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=np.float64)
    except (NameError, TypeError):
        pass

    n_list = [99, 97, 101, 63, 81, 127, 75, 111, 55, 105]

    if h_init is not None:
        n_parent = len(h_init)
        h_parent = project_feasible(h_init.copy(), float(n_parent) / 2.0)
        c5_parent = compute_c5(h_parent)
        best_h = h_parent.copy()
        best_c5 = c5_parent
        best_n = n_parent

        alloc1 = min(remaining() * 0.35, 350.0)
        if alloc1 > 30:
            h_opt, c5_opt = optimize_gradient(h_parent, alloc1, seed)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = len(h_opt)

    n_done = set()
    if h_init is not None:
        n_done.add(len(h_init))

    for n in n_list:
        if n in n_done:
            continue
        if remaining() < 50:
            break
        n_done.add(n)
        alloc = min(remaining() * 0.15, 120.0)
        if alloc < 25:
            break

        for init_type in range(3):
            if remaining() < 20:
                break
            if init_type == 0:
                h_new = random_init(n, rng)
            elif init_type == 1:
                h_new = structured_init(n, rng)
            else:
                if best_h is not None:
                    h_new = resample_to(best_h, n)
                else:
                    h_new = random_init(n, rng)
            h_new = project_feasible(h_new, float(n) / 2.0)
            sub_alloc = min(alloc / 3.0, remaining() - 15.0)
            if sub_alloc < 10:
                break
            h_opt, c5_opt = optimize_gradient(h_new, sub_alloc, seed + n * 10 + init_type)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = len(h_opt)

    if best_h is not None and remaining() > 30:
        alloc_ref = min(remaining() * 0.25, 200.0)
        h_ref, c5_ref = refine_pairs(best_h, alloc_ref, seed + 7777)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = len(h_ref)

    if best_h is not None and remaining() > 20:
        n_refine = len(best_h)
        for n_try in [n_refine - 2, n_refine + 2, n_refine - 4, n_refine + 4]:
            if remaining() < 15:
                break
            if n_try < 10:
                continue
            h_rs = resample_to(best_h, n_try)
            sub = min(60.0, remaining() - 10.0)
            if sub < 10:
                break
            h_opt, c5_opt = optimize_gradient(h_rs, sub, seed + n_try * 3)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    if best_h is not None and remaining() > 10:
        target = float(best_n) / 2.0
        best_h = project_feasible(best_h, target)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        n = 63
        best_h = random_init(n, rng)
        best_h = project_feasible(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
