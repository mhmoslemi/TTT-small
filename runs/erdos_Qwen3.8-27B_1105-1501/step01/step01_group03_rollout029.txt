import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_all_F(h, n):
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        if start < end:
            F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F


def compute_lag_gradient(h, k, n):
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start < end:
        grad[start:end] += (1.0 - h[start + k:end + k])
    p_start = max(0, start + k)
    p_end = min(n, end + k)
    if p_start < p_end:
        grad[p_start:p_end] += (-h[p_start - k:p_end - k])
    return grad


def safe_sigmoid(x):
    out = np.zeros_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def logit_to_h(y, n):
    target = n / 2.0
    lo, hi = -50.0, 50.0
    s_lo = np.sum(safe_sigmoid(y + lo))
    s_hi = np.sum(safe_sigmoid(y + hi))
    if not (s_lo <= target <= s_hi):
        return make_feasible(safe_sigmoid(y))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(safe_sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    c = (lo + hi) * 0.5
    h = safe_sigmoid(y + c)
    return make_feasible(h)


def h_to_logit(h, n):
    eps = 1e-12
    h_clipped = np.clip(h, eps, 1.0 - eps)
    return np.log(h_clipped / (1.0 - h_clipped))


def smooth_obj_y(y, n, beta):
    h = logit_to_h(y, n)
    dx = 2.0 / n
    F = compute_all_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(2 * n - 1) / beta
    return cmax + (mk + np.log(ws)) / beta


def smooth_grad_y(y, n, beta):
    h = logit_to_h(y, n)
    dx = 2.0 / n
    F = compute_all_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    G = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[k + n - 1]
        if abs(w) < 1e-14:
            continue
        G += w * compute_lag_gradient(h, k, n)
    G *= dx
    hw = h * (1.0 - h)
    W = np.sum(hw)
    if W < 1e-300:
        return np.zeros(n)
    V = np.dot(G, hw)
    return hw * (G - V / W)


def eq_sum(x):
    return float(np.sum(x) - len(x) / 2.0)


def eq_sum_jac(x):
    return np.ones(len(x))


def direct_obj(z):
    return float(z[-1])


def direct_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def direct_ineq(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    F = compute_all_F(h, n)
    return z[-1] - dx * F


def direct_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n + 1))
    J[:, -1] = 1.0
    for k in range(-(n - 1), n):
        J[k + n - 1, :n] = -dx * compute_lag_gradient(h, k, n)
    return J


def direct_eq(z):
    return float(np.sum(z[:-1]) - (len(z) - 1) / 2.0)


def direct_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g


def slls_solve(h0, n, maxiter=200, ftol=1e-11):
    h0 = make_feasible(h0)
    bounds = [(0.0, 1.0)] * n

def direct_solve(h0, n, maxiter=150, ftol=1e-11):
    h0 = make_feasible(h0)
    score0 = official_score(h0)
    theta0 = score0 + 0.01
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.8)]
    cons = [
        {'type': 'ineq', 'fun': direct_ineq, 'jac': direct_ineq_jac},
        {'type': 'eq', 'fun': direct_eq, 'jac': direct_eq_jac}
    ]
    try:
        res = minimize(direct_obj, z0, jac=direct_obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            return h_new
    except Exception:
        pass
    return h0


def binary_harden(h, n, rng):
    if n % 2 == 0:
        target = n // 2
    else:
        target = (n - 1) // 2
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    if n % 2 == 1:
        b[sorted_idx[target]] = 0.5
    return b
