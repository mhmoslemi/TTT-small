def compute_G(h):
    n = len(h)
    G = np.zeros((n, 2 * n - 1))
    for s in range(n):
        k = n - 1 + s
        G[s:, k] += (1.0 - h[:n - s])
        G[:n - s, k] -= h[s:]
    for t in range(1, n):
        k = n - 1 - t
        G[:n - t, k] += (1.0 - h[t:])
        G[t:, k] -= h[:n - t]
    return G

def lse_smooth(c, beta):
    cmax = c.max()
    shifted = beta * (c - cmax)
    max_shifted = shifted.max()
    exp_s = np.exp(shifted - max_shifted)
    log_sum = max_shifted + np.log(exp_s.sum())
    return cmax + log_sum / beta

def lse_grad(c, h, beta):
    cmax = c.max()
    shifted = beta * (c - cmax)
    max_shifted = shifted.max()
    exp_s = np.exp(shifted - max_shifted)
    w = exp_s / exp_s.sum()
    n = len(h)
    dx = 2.0 / n
    G = compute_G(h)
    return dx * (w @ G)

def projected_grad_opt(h0, beta, n_iter, deadline):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    lr = 0.05
    prev_h = h.copy()
    prev_g = None
    best_local = official_score(h)
    best_h_local = h.copy()
    stall_count = 0

    for it in range(n_iter):
        if time.time() >= deadline:
            break
        c = compute_corr(h)
        grad = lse_grad(c, h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        if prev_g is not None:
            s_vec = h - prev_h
            y_vec = grad - prev_g
            sy = s_vec @ y_vec
            ss = s_vec @ s_vec
            yy = y_vec @ y_vec
            if yy > 1e-16 and ss > 1e-16:
                if abs(sy) > 1e-14:
                    lr_bb1 = ss / max(sy, 1e-8)
                    lr_bb2 = max(sy, 1e-8) / yy
                    lr = 0.5 * (lr_bb1 + lr_bb2)
                lr = max(1e-8, min(lr, 2.0 / (beta + 1)))

        prev_h = h.copy()
        prev_g = grad.copy()

        for trial in range(15):
            new_h = make_feasible(h - lr * grad)
            new_c = compute_corr(new_h)
            new_lse = lse_smooth(new_c, beta)
            cur_lse = lse_smooth(c, beta)
            if new_lse <= cur_lse + 1e-12:
                h = new_h
                break
            lr *= 0.5

        cur_score = official_score(h)
        if cur_score < best_local - 1e-12:
            best_local = cur_score
            best_h_local = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                lr *= 0.8
                stall_count = 0

    return best_h_local

def try_slsqp_direct(h0, maxiter, deadline):
    n = len(h0)
    global _N_SLSQP, _DX_SLSQP
    _N_SLSQP = n
    _DX_SLSQP = 2.0 / n
    theta0 = official_score(h0) + 1e-6
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def mm_obj(z):
        return z[-1]

    def mm_obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def mm_ineq_fun(z):
        hh = z[:_N_SLSQP]
        th = z[-1]
        vals = np.correlate(hh, 1.0 - hh, mode='full') * _DX_SLSQP
        return th - vals

    def mm_ineq_jac(z):
        hh = z[:_N_SLSQP]
        G = compute_G(hh)
        J = -_DX_SLSQP * G
        J = np.hstack([J, np.ones((J.shape[0], 1))])
        return J

    def mm_eq_fun(z):
        return np.sum(z[:_N_SLSQP]) - _N_SLSQP / 2.0

    def mm_eq_jac(z):
        g = np.zeros(_N_SLSQP + 1)
        g[:_N_SLSQP] = 1.0
        return g

    cons = [
        {'type': 'ineq', 'fun': mm_ineq_fun, 'jac': mm_ineq_jac},
        {'type': 'eq', 'fun': mm_eq_fun, 'jac': mm_eq_jac}
    ]
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def try_slsqp_smooth(h0, beta, maxiter, deadline):
    n = len(h0)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def sl_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return lse_smooth(c, beta)

    def sl_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return lse_grad(c, x, beta)

    def sl_eq(x):
        return np.sum(x) - n / 2.0

    def sl_eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sl_eq, 'jac': sl_eq_jac}]
    try:
        res = minimize(sl_obj, h0, jac=sl_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym(h):
    h_new = 1.0 - h[::-1].copy()
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def is_deadline():
        return time.time() >= deadline

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-15:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = np.arange(n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * x / n0) + 0.1 * np.cos(4 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    n_base = len(best_h)

    # Phase 1: Projected gradient with increasing beta
    for beta in [20, 80, 300, 1000, 4000, 15000, 50000]:
        if is_deadline():
            break
        cur = best_h.copy()
        n_iter = max(200, int((deadline - time.time()) * 8))
        n_iter = min(n_iter, 3000)
        h_new = projected_grad_opt(cur, beta, n_iter, deadline)
        sc = official_score(h_new)
        if sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 2: SLSQP direct minimax polish
    if not is_deadline():
        h_new, sc = try_slsqp_direct(best_h, maxiter=300, deadline=deadline)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 3: SLSQP smooth polish
    for beta in [500, 3000, 20000]:
        if is_deadline():
            break
        h_new, sc = try_slsqp_smooth(best_h, beta, maxiter=200, deadline=deadline)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 4: More projected gradient from improved point
    for beta in [10000, 50000, 200000]:
        if is_deadline():
            break
        cur = best_h.copy()
        n_iter = max(300, int((deadline - time.time()) * 10))
        n_iter = min(n_iter, 5000)
        h_new = projected_grad_opt(cur, beta, n_iter, deadline)
        sc = official_score(h_new)
        if sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 5: Random restarts
    if not is_deadline():
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(3, min(15, int(remaining / 20)))
        sigmas = [0.02, 0.04, 0.08, 0.12, 0.05, 0.03, 0.1, 0.06, 0.015, 0.07]
        for i in range(n_starts):
            if is_deadline():
                break
            sigma = sigmas[i % len(sigmas)]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            sc_cand = official_score(h_cand)
            if sc_cand < best_score - 1e-14:
                update(h_cand, sc_cand)
            # Quick projected gradient from this start
            h_new = projected_grad_opt(h_cand, 2000, 500, deadline)
            sc_new = official_score(h_new)
            if sc_new < best_score - 1e-14:
                update(h_new, sc_new)

    # Phase 6: Different resolutions
    if not is_deadline():
        n_opts = sorted(set([
            max(30, n_base - 15),
            max(35, n_base - 10),
            n_base,
            min(80, n_base + 10),
            min(85, n_base + 15),
            min(90, n_base + 20),
        ]))
        for n_opt in n_opts:
            if is_deadline():
                break
            if n_opt == len(best_h):
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-14:
                update(h_int, sc_int)
            # Optimize at this resolution
            h_new = projected_grad_opt(h_int, 3000, 800, deadline)
            sc_new = official_score(h_new)
            if sc_new < best_score - 1e-14:
                update(h_new, sc_new)
            if not is_deadline():
                h_slsqp, sc_sl = try_slsqp_direct(h_new, maxiter=200, deadline=deadline)
                if h_slsqp is not None and sc_sl < best_score - 1e-14:
                    update(h_slsqp, sc_sl)

    # Phase 7: Anti-symmetric transform
    if not is_deadline():
        h_sym = anti_sym(best_h)
        sc_sym = official_score(h_sym)
        if sc_sym < best_score - 1e-14:
            update(h_sym, sc_sym)
        h_new = projected_grad_opt(h_sym, 5000, 500, deadline)
        sc_new = official_score(h_new)
        if sc_new < best_score - 1e-14:
            update(h_new, sc_new)

    # Phase 8: Final SLSQP polish
    if not is_deadline():
        h_new, sc = try_slsqp_direct(best_h, maxiter=400, deadline=deadline)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 9: Final projected gradient with very high beta
    if not is_deadline():
        for beta in [100000, 500000]:
            if is_deadline():
                break
            h_new = projected_grad_opt(best_h, beta, 1000, deadline)
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                update(h_new, sc)

    # Phase 10: Last chance SLSQP
    if not is_deadline():
        h_new, sc = try_slsqp_direct(best_h, maxiter=200, deadline=deadline)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-13
    return final_h, final_bound, final_n
