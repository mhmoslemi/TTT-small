```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_target(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    
    # Water-filling via bisection
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    
    # Exact repair
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-10:
        # Find coordinates with room
        if residual > 0:
            # Need to add
            rooms = 1.0 - x
            idx = np.where(rooms > 1e-12)[0]
        else:
            # Need to subtract
            rooms = x
            idx = np.where(rooms > 1e-12)[0]
        
        for i in idx:
            if abs(residual) < 1e-12:
                break
            amt = min(abs(residual), rooms[i])
            if residual > 0:
                x[i] += amt
            else:
                x[i] -= amt
            residual -= np.sign(residual) * amt
    
    return x

def grad_for_scores(h, n):
    """Compute gradients of all correlation scores w.r.t. h.
    Returns array of shape (2n-1, n) where row j is gradient of c_j."""
    num_scores = 2 * n - 1
    grads = np.zeros((num_scores, n))
    
    for j in range(num_scores):
        k = j - (n - 1)  # actual lag
        if k >= 0:
            # c_j = sum_{i=0}^{n-1-k} h[i+k](1-h[i])
            # grad[m] = (1-h[m-k]) for k<=m<=n-1, minus h[m+k] for 0<=m<=n-1-k
            for m in range(k, n):
                grads[j, m] += (1.0 - h[m - k])
            for m in range(0, n - k):
                grads[j, m] -= h[m + k]
        else:
            g = -k
            # c_j = sum_{i=g}^{n-1} h[i-g](1-h[i])
            # grad[m] = (1-h[m+g]) for 0<=m<=n-1-g, minus h[m-g] for g<=m<=n-1
            for m in range(0, n - g):
                grads[j, m] += (1.0 - h[m + g])
            for m in range(g, n):
                grads[j, m] -= h[m - g]
    
    return grads
```

```python
def lp_descent_direction(h, c_vals, grads, active_idx, radius, lam=5.0):
    n = len(h)
    num_active = len(active_idx)
    
    # Variables: p (n), r (1), tau (1) -> total n+2
    # Minimize: tau + lam * r
    # Subject to:
    #   For each active j: c_j + grad_j · p <= tau  =>  grad_j · p - tau <= -c_j
    #   p_i - r <= 0 for all i
    #   -p_i - r <= 0 for all i
    #   sum(p) = 0
    #   -min(h_i, radius) <= p_i <= min(1-h_i, radius)
    #   0 <= r <= radius
    #   tau >= -inf (free, but bounded implicitly)
    
    c_obj = np.zeros(n + 2)
    c_obj[n] = lam  # r
    c_obj[n+1] = 1.0  # tau
    
    # Inequality constraints: A_ub @ x <= b_ub
    rows = []
    rhs = []
    
    # Active constraint rows
    for idx in active_idx:
        row = np.zeros(n + 2)
        row[:n] = grads[idx]
        row[n+1] = -1.0  # -tau
        rows.append(row)
        rhs.append(-c_vals[idx])
    
    # r constraints: p_i - r <= 0
    for i in range(n):
        row = np.zeros(n + 2)
        row[i] = 1.0
        row[n] = -1.0
        rows.append(row)
        rhs.append(0.0)
    
    # -r constraints: -p_i - r <= 0
    for i in range(n):
        row = np.zeros(n + 2)
        row[i] = -1.0
        row[n] = -1.0
        rows.append(row)
        rhs.append(0.0)
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    lower = np.zeros(n + 2)
    upper = np.zeros(n + 2)
    for i in range(n):
        lower[i] = -min(h[i], radius)
        upper[i] = min(1.0 - h[i], radius)
    lower[n] = 0.0
    upper[n] = radius
    lower[n+1] = -np.inf
    upper[n+1] = np.inf
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lower, upper)), method='highs')
    
    if not res.success:
        return None
    
    p = res.x[:n]
    return p
```

Wait, there's an issue with the LP. The objective is to minimize tau + lam*r. But if we can always make tau very negative by making p very large (within the radius), the LP will push p to the boundary. The lam*r term regularizes this, but we need to make sure the regularization is strong enough.

Actually, the issue is more subtle. If we set p=0, then tau must be at least max(c_j for active j) = max_c. So the minimum possible tau (with p=0) is max_c. If we can find p that reduces the predicted max, tau can go below max_c. The LP will find the p that minimizes tau + lam*r, which balances the reduction in tau against the step size.

The step size is then determined by line search on the actual objective. The LP just gives a direction.

Let me think about the line search. Given direction p (zero-sum, bounded), we want to find alpha in [0, alpha_max] that minimizes C5(h + alpha*p), where alpha_max is the largest alpha such that h + alpha*p is still in [0,1].

```python
def line_search(h, p, max_iter=50):
    n = len(h)
    # Find alpha_max
    alpha_max = np.inf
    for i in range(n):
        if p[i] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-12:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    alpha_max = min(alpha_max, 1.0)  # cap at 1 for safety
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = 0.0
    b = alpha_max
    
    def f(alpha):
        h_new = h + alpha * p
        return exact_c5(h_new)
    
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    
    c5_0 = f(0.0)
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    alpha_star = (a + b) / 2.0
    c5_star = f(alpha_star)
    
    if c5_star >= c5_0:
        return 0.0, c5_0
    
    return alpha_star, c5_star
```

```python
def weighted_subgradient(c_vals, grads, temperature=0.01):
    """Compute weighted gradient of the max."""
    max_c = np.max(c_vals)
    weights = np.exp(-(c_vals - max_c) / temperature)
    weights = weights / weights.sum()
    g = weights @ grads  # shape (n,)
    # Project to zero-sum
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_target(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - x)[::-1]  # most room first
        else:
            idx = np.argsort(x)  # largest first (most to remove)
        for i in idx:
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                amt = min(1.0 - x[i], residual)
                x[i] += amt
                residual -= amt
            else:
                amt = min(x[i], -residual)
                x[i] -= amt
                residual += amt
    return x

def compute_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_grads(h, n):
    """Compute gradients of all 2n-1 correlation scores."""
    num = 2 * n - 1
    grads = np.zeros((num, n))
    for j in range(num):
        k = j - (n - 1)
        if k >= 0:
            m1 = np.arange(k, n)
            grads[j, m1] += (1.0 - h[m1 - k])
            m2 = np.arange(0, n - k)
            grads[j, m2] -= h[m2 + k]
        else:
            g = -k
            m1 = np.arange(0, n - g)
            grads[j, m1] += (1.0 - h[m1 + g])
            m2 = np.arange(g, n)
            grads[j, m2] -= h[m2 - g]
    return grads

def lp_direction(h, scores, grads, active, radius):
    n = len(h)
    c_obj = np.zeros(n + 2)
    c_obj[n] = 5.0    # regularization on r
    c_obj[n+1] = 1.0  # tau
    
    rows = []
    rhs = []
    
    for idx in active:
        row = np.zeros(n + 2)
        row[:n] = grads[idx]
        row[n+1] = -1.0
        rows.append(row)
        rhs.append(-scores[idx])
    
    for i in range(n):
        row = np.zeros(n + 2)
        row[i] = 1.0
        row[n] = -1.0
        rows.append(row)
        rhs.append(0.0)
    
    for i in range(n):
        row = np.zeros(n + 2)
        row[i] = -1.0
        row[n] = -1.0
        rows.append(row)
        rhs.append(0.0)
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    lower = np.zeros(n + 2)
    upper = np.zeros(n + 2)
    for i in range(n):
        lower[i] = -min(h[i], radius)
        upper[i] = min(1.0 - h[i], radius)
    lower[n] = 0.0
    upper[n] = radius
    lower[n+1] = -np.inf
    upper[n+1] = np.inf
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lower, upper)), method='highs',
                  options={'presolve': True})
    
    if not res.success:
        return None
    p = res.x[:n]
    # Clean up tiny values
    p[np.abs(p) < 1e-12] = 0.0
    return p

def line_search(h, p):

                best_alpha = alpha
```

```python
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
        else:  # cyclic
            i = it % n
            j = (it + 1) % n

        hi = h[i]
        hj = h[j]

        # Valid range: max(0, hi+hj-1) <= x <= min(1, hi+hj)
        xlo = max(0.0, hi + hj - 1.0)
```

    n = len(h)

    return float(np.max(corr)) * dx


def project_to_target(v, target):
    n = len(v)

    v[~np.isfinite(v)] = 0.5
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - x)[::-1]
        else:
            order = np.argsort(x)
        for i in order:
            if abs(residual) < 1e-14:
                break
            if residual > 0:
                amt = min(1.0 - x[i], residual)
                x[i] += amt
                residual -= amt
            else:
                amt = min(x[i], -residual)
                x[i] -= amt
                residual += amt
    return x

def compute_grads(h, n):
    num = 2 * n - 1
    grads = np.zeros((num, n))
    for j in range(num):
        k = j - (n - 1)
        if k >= 0:
            m1 = np.arange(k, n)
            grads[j, m1] += (1.0 - h[m1 - k])
            m2 = np.arange(0, n - k)
            grads[j, m2] -= h[m2 + k]
        else:
            g = -k
            m1 = np.arange(0, n - g)
            grads[j, m1] += (1.0 - h[m1 + g])
            m2 = np.arange(g, n)
            grads[j, m2] -= h[m2 - g]
    return grads
