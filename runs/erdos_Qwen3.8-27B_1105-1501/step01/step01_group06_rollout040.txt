        dk = (h_jk - h_ik) + (h_i_k - h_j_k)
        delta[k + n - 1] = dx * dk
    return delta


def coordinate_descent(h, n, target_sum, max_rounds=5, rng=None):
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    c = compute_c(h)
    if rng is None:
        rng = np.random.default_rng(0)

    for _round in range(max_rounds):
        improved = False
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
        for i, j in pairs:
            d_min = max(-h[i], -(1.0 - h[j]))
            d_max = min(1.0 - h[i], h[j])
            if d_min >= d_max:
                continue
            delta = swap_delta(h, i, j, n)
            # Try candidate deltas
            candidates = set()
            candidates.add(0.0)
            candidates.add(d_min)
            candidates.add(d_max)
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                candidates.add(d_min + frac * (d_max - d_min))
            best_d = 0.0
            best_c5_try = best_c5
            for d in candidates:
                if d < d_min - 1e-12 or d > d_max + 1e-12:
                    continue
                new_c = c + d * delta
                c5_try = float(np.max(new_c))
                if c5_try < best_c5_try:
                    best_c5_try = c5_try
                    best_d = d
            if best_d != 0.0:
                h[i] += best_d
                h[j] -= best_d
                c = compute_c(h)
                if best_c5_try < best_c5 - 1e-12:
                    best_c5 = best_c5_try
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def target_subgradient_h(h, n, target_sum, top_k, steps, lr0):
    """Subgradient descent targeting top k lags."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    for step in range(steps):
        c_vals = compute_c(h)
        c_max = float(np.max(c_vals))
        # Pick top k lags
        order = np.argsort(c_vals)[::-1]
        active = order[:top_k]
        # Compute gradient using only active lags
        g = np.zeros(n)
        dx = 2.0 / n
        center = n - 1
        for idx in active:
            d = idx - center
            if d >= 0:
                for j in range(n - d):
                    g[j] += dx * (1.0 - h[j + d])
                for j in range(d, n):
                    g[j] -= dx * h[j - d]
            else:
                dd = -d
                for j in range(dd, n):
                    g[j] += dx * (1.0 - h[j - dd])
                for j in range(n - dd):
                    g[j] -= dx * h[j + dd]
        # Projected gradient step
        h_new = project(h - lr0 * g, target_sum)
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr0 *= 0.998
    return best_h, best_c5


def gen_sine_init(n, freq=3, amplitude=0.4, phase=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amplitude * np.sin(np.pi * freq * x + phase)
    return project(h, n / 2.0)


def gen_block_init(n):
    block_size = max(1, n // 6)
    h = np.zeros(n)
    for start in range(0, n, 2 * block_size):
        end = min(start + block_size, n)
        h[start:end] = 1.0
    return project(h, n / 2.0)


def gen_random_binary_init(n, seed):
    rng = np.random.default_rng(seed)
    h = rng.randint(0, 2, size=n).astype(float)
    return project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def time_left(frac=0.0):
        return budget_s * (1.0 - frac) - elapsed()

    # Start from parent
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target_sum = n / 2.0
    h = project(h0, target_sum)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    best_n = n

    # Phase 1: Initial SLSQP with increasing beta (use 25% of time)
    t1 = 0.25
    betas = [5, 10, 20, 50, 100, 200, 500, 1000, 3000, 5000]
    for beta in betas:
        if elapsed() > budget_s * t1:
            break
        h_opt = slsqp_opt(h, n, target_sum, beta, 300)
        h_opt = project(h_opt, target_sum)
        c5 = eval_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 2: Momentum descent with high beta (use 15% of time)
    t2 = 0.40
    for beta in [5000, 10000, 20000, 50000]:
        if elapsed() > budget_s * t2:
            break
        h_best, c5_best = momentum_descent(h, n, target_sum, beta, 2000, 0.3, mom=0.9)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            h = h_best.copy()

    # Phase 3: Target subgradient (use 10% of time)
    t3 = 0.50
    for top_k in [1, 2, 3, 5]:
        if elapsed() > budget_s * t3:
            break
        h_best, c5_best = target_subgradient_h(best_h, n, target_sum, top_k, 3000, 0.1)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            h = h_best.copy()

    # Phase 4: Coordinate descent (swap-based) - use 20% of time
    t4 = 0.70
    for _ in range(30):
        if elapsed() > budget_s * t4:
            break
        h_best, c5_best = coordinate_descent(best_h, n, target_sum, max_rounds=3, rng=rng)
        if c5_best < best_c5 - 1e-12:
            best_c5 = c5_best
            best_h = h_best.copy()
            h = h_best.copy()
        else:
            # Try with a perturbed start
            h_pert = best_h + rng.normal(0, 0.02, n)
            h_pert = project(h_pert, target_sum)
            h_best, c5_best = coordinate_descent(h_pert, n, target_sum, max_rounds=3, rng=rng)
            if c5_best < best_c5 - 1e-12:
                best_c5 = c5_best
                best_h = h_best.copy()
                h = h_best.copy()

    # Phase 5: SLSQP polish after coordinate descent
    if elapsed() < budget_s * 0.75:
        for beta in [10000, 50000, 100000]:
            if elapsed() > budget_s * 0.75:
                break
            h_opt = slsqp_opt(best_h, n, target_sum, beta, 300)
            h_opt = project(h_opt, target_sum)
            c5 = eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 6: Local perturbations + SLSQP (use remaining time until 97%)
    t6 = 0.97
    while elapsed() < budget_s * t6:
        # Random perturbation
        ps = rng.choice([0.005, 0.01, 0.02, 0.05, 0.1])
        mask = rng.random(n) < 0.5
        h_pert = best_h.copy()
        if np.any(mask):
            h_pert[mask] += rng.normal(0, ps, int(np.sum(mask)))
        h_pert = project(h_pert, target_sum)
        c5_p = eval_c5(h_pert)

        # Always try SLSQP from the perturbed point if it's close
        if c5_p < best_c5 + 0.01:
            beta = 20000 if c5_p < best_c5 else 10000
            maxiter = 150 if c5_p < best_c5 else 100
            h_opt = slsqp_opt(h_pert, n, target_sum, beta, maxiter)
            h_opt = project(h_opt, target_sum)
            c5 = eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                h = h_opt.copy()

        # Also try coordinate descent from perturbation
        if elapsed() < budget_s * t6:
            h_best, c5_best = coordinate_descent(h_pert, n, target_sum, max_rounds=2, rng=rng)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
                h = h_best.copy()

        # Sometimes do a full SLSQP from the best
        if rng.random() < 0.1 and elapsed() < budget_s * t6:
            h_opt = slsqp_opt(best_h, n, target_sum, 50000, 200)
            h_opt = project(h_opt, target_sum)
            c5 = eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 7: Try different n values with algorithmic initializations
    if elapsed() < budget_s * 0.98:
        for n_try in [64, 65, 66, 70]:
            if elapsed() > budget_s * 0.98:
                break
            n_ts = n_try
            ts = n_ts / 2.0
            for init_fn_name in ['sine3', 'sine5', 'block', 'randbin']:
                if elapsed() > budget_s * 0.98:
                    break
                if init_fn_name == 'sine3':
                    h_try = gen_sine_init(n_ts, freq=3, amplitude=0.4)
                elif init_fn_name == 'sine5':
                    h_try = gen_sine_init(n_ts, freq=5, amplitude=0.35)
                elif init_fn_name == 'block':
                    h_try = gen_block_init(n_ts)
                else:
                    h_try = gen_random_binary_init(n_ts, seed + n_ts)

                # Quick optimization
                h_try = slsqp_opt(h_try, n_ts, ts, 100, 100)
                h_try = project(h_try, n_ts, ts) if False else project(h_try, ts)
                c5_try = eval_c5(h_try)
                if c5_try < best_c5:
                    h_try = slsqp_opt(h_try, n_ts, ts, 500, 200)
                    h_try = project(h_try, ts)
                    c5_try = eval_c5(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        best_n = n_ts

    # Final: one more polish on the overall best
    if elapsed() < budget_s * 0.995:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        h_opt = slsqp_opt(best_h, n_final, ts_final, 100000, 200)
        h_opt = project(h_opt, ts_final)
        c5 = eval_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n_final

    # Final coordinate descent polish
    if elapsed() < budget_s * 0.998:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        h_best, c5_best = coordinate_descent(best_h, n_final, ts_final, max_rounds=5, rng=rng)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            best_n = n_final

    # Verify and return
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
