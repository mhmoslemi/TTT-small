def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_F(h, n):
    """Compute F_k = sum h[j]*(1-h[j+k]) for all valid shifts k."""
    F = np.zeros(2 * n - 1)
    mid = n - 1
    for k in range(-(n-1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + mid] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F

def project_to_constraint(h, n):
    """Project h to {sum = n/2, 0 <= h <= 1}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Binary search for shift
    lo = -max(h) - 1e-6
    hi = 1.0 - min(h) + 1e-6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    err = target - np.sum(h_proj)
    if abs(err) > 1e-14:
        # Distribute to coordinates with room
        if err > 0:
            order = np.argsort(-h_proj)  # most room first (closest to 1)
            # Actually, we want to add to those with room, so those < 1
            order = np.argsort(h_proj)  # least first, most room
