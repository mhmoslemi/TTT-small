def smooth_obj_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # For numerical stability, subtract max
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    obj = max_c + np.log(exp_sum) / beta
    weights = exp_vals / exp_sum
    # Gradient: sum_k weights[k] * grad_k
    # Use the fast gradient computation
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15:
            continue
        s = k - (n-1)
        if s < 0:
            ss = -s
            grad[:n-ss] += w * (1.0 - h[ss:n])
            grad[ss:] += w * (-h[:n-ss])
        elif s > 0:
            grad[ss:] += w * (1.0 - h[:n-ss])
            grad[:n-ss] += w * (-h[ss:n])
        else:
            grad += w * (1.0 - 2.0 * h)
    return obj, grad
