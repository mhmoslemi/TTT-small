def compute_c5_and_shifts(h):

    return c5, corr * dx

def project_onto_box_sum(h, target, lo=0.0, hi=1.0):

        s = np.sum(np.clip(h - mid, lo, hi))
        if s > target:
            lo_t = mid
        else:
            hi_t = mid
    threshold = (lo_t + hi_t) / 2.0
    return np.clip(h - threshold, lo, hi)

                h_try[j] -= delta
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
        
        if improved:
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 50 == 0:
                # Perturb to escape local optimum
                step = max(step * 0.95, min_step)
                if stall_count > 200:
                    h_try = best_h + step * 0.5 * rng.randn(n)
                    h_try = project_onto_box_sum(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_h = h_try
                        best_c5 = c5_try
                    stall_count = 0
    
    return best_h, best_c5


def fine_pair_search(h, target, time_budget, seed):
    """Exhaustive fine-grained pair search with tiny steps."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    deltas = [0.005, 0.002, 0.001, 0.0005]
    
    while time.time() - t0 < time_budget:
        improved = False
        # Try pairs related to current max
        corr_raw = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_idx = np.argmax(corr_raw)
        shift = max_idx - (n - 1)
        
        # Focus on pairs for this shift
        for delta in deltas:
            if time.time() - t0 >= time_budget:
                return best_h, best_c5
            if shift >= 0:
                for i in range(n - shift):
                    j = i + shift
                    if best_h[j] > 1e-8 and best_h[i] < 1.0 - 1e-8:
                        h_try = best_h.copy()
                        h_try[i] += delta
                        h_try[j] -= delta
                        h_try = project_onto_box_sum(h_try, target)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-12:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
                            break
            else:
                ns = -shift
                for i in range(ns, n):
                    j = i - ns
                    if best_h[i] > 1e-8 and best_h[j] < 1.0 - 1e-8:
                        h_try = best_h.copy()
                        h_try[j] += delta
                        h_try[i] -= delta
                        h_try = project_onto_box_sum(h_try, target)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-12:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
                            break
            if improved:
                break
        
        # Also try random pairs with small deltas
        for _ in range(2000):
            if time.time() - t0 >= time_budget:
                return best_h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for delta in [0.003, 0.001]:
                for sign in [-1, 1]:
                    d = delta * sign
                    if 0 <= best_h[i] + d and best_h[i] + d <= 1:
                        if 0 <= best_h[j] - d and best_h[j] - d <= 1:
                            h_try = best_h.copy()
                            h_try[i] += d
                            h_try[j] -= d
                            c5_try = compute_c5(h_try)
                            if c5_try < best_c5 - 1e-12:
                                best_c5 = c5_try
                                best_h = h_try
                                improved = True
                                break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Perturb slightly
            h_try = best_h + 0.005 * rng.randn(n)
            h_try = project_onto_box_sum(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
            else:
                # Accept anyway to explore, but remember best
                best_h = h_try
    
    return best_h, best_c5


def multi_shift_gradient_step(h, target, n, alpha=5.0):
    """Compute a gradient-like direction that pushes down all near-max shifts."""
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr_raw)
    # Soft weights
    weights = np.exp(alpha * (corr_raw - max_c))
    weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    for k_idx in range(len(corr_raw)):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        shift = k_idx - (n - 1)
        if shift >= 0:
            # C = sum_{i=0}^{n-1-shift} h[i+shift]*(1-h[i])
            # dC/dh[j] for j in [shift, n-1]: += (1-h[j-shift])
            # dC/dh[j] for j in [0, n-1-shift]: -= h[j+shift]
            for j in range(shift, n):
                grad[j] += w * (1.0 - h[j - shift])
            for j in range(0, n - shift):
                grad[j] -= w * h[j + shift]
        else:
            ns = -shift
            # C = sum_{i=ns}^{n-1} h[i]*(1-h[i-ns])
            # dC/dh[j] for j in [ns, n-1]: += (1-h[j-ns])
            # dC/dh[j] for j in [0, n-1-ns]: -= h[j+ns]
            for j in range(ns, n):
                grad[j] += w * (1.0 - h[j - ns])
            for j in range(0, n - ns):
                grad[j] -= w * h[j + ns]
    
    return grad * (2.0 / n)


def gradient_descent_optimize(h, target, time_budget, seed):
    """Optimize using multi-shift gradient with adaptive learning rate."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    lr = 0.01
    alpha = 3.0
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    
    while time.time() - t0 < time_budget:
        grad = multi_shift_gradient_step(h, target, n, alpha)
        v = momentum * v - lr * grad
        h_try = project_onto_box_sum(h + v, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            stall = 0
        else:
            h = best_h.copy()
            stall += 1
            if stall > 100:
                lr = max(lr * 0.9, 0.0005)
                v *= 0.5
                # Perturb
                h = best_h + 0.02 * rng.randn(n)
                h = project_onto_box_sum(h, target)
                v = np.zeros(n)
                stall = 0
        
        alpha = min(alpha * 1.05, 50.0)
        lr = max(lr * 0.995, 0.0003)
    
    return best_h, best_c5


def try_alt_n(n, time_budget, seed):
    """Try a different number of points."""
    target = n / 2.0
    rng = np.random.RandomState(seed)
    
    # Initialize: evenly spread ones
    h = np.zeros(n)
    ones_needed = int(target)
    frac = target - ones_needed
    # Place ones evenly
    step = n / ones_needed if ones_needed > 0 else n
    for i in range(ones_needed):
        pos = int(round(i * step)) % n
        h[pos] = 1.0
    if frac > 0.01:
        # Place fractional value at a zero position
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = frac
    
    h = project_onto_box_sum(h, target)
    h, c5 = focused_optimize(h, target, time_budget, seed)
    return h, c5, n


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        n = 99
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h[n // 2] = 0.5
    
    n = len(h)
    target = n / 2.0
    h = project_onto_box_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - t_start
    
    # Phase 1: Gradient-based optimization (25% of budget)
    phase1_budget = budget_s * 0.25
    h1, c5_1 = gradient_descent_optimize(best_h, target, phase1_budget, seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    elif elapsed() > budget_s * 0.8:
        best_h = project_onto_box_sum(best_h, target)
        return best_h, best_c5, n
    
    # Phase 2: Focused active-set optimization (35% of budget)
    phase2_budget = min(budget_s * 0.35, budget_s - elapsed() - budget_s * 0.3)
    if phase2_budget > 10:
        h2, c5_2 = focused_optimize(best_h, target, phase2_budget, seed + 1)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
    
    # Phase 3: Fine pair search (30% of budget)
    phase3_budget = min(budget_s * 0.30, budget_s - elapsed() - 10)
    if phase3_budget > 10:
        h3, c5_3 = fine_pair_search(best_h, target, phase3_budget, seed + 2)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
    
    # Phase 4: Try alternative n values if time remains
    if elapsed() < budget_s - 30:
        remaining = budget_s - elapsed() - 5
        # Try n=97
        if remaining > 40:
            t_alt = min(remaining * 0.4, 60.0)
            h_alt, c5_alt, n_alt = try_alt_n(97, t_alt, seed + 100)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt
                n = n_alt
                target = n / 2.0
    
    # Final: one more fine pass if time remains
    if elapsed() < budget_s - 5:
        final_budget = budget_s - elapsed() - 2
        if final_budget > 5:
            h_final, c5_final = fine_pair_search(best_h, target, final_budget, seed + 999)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
    
    # Ensure feasibility
    target = len(best_h) / 2.0
    best_h = project_onto_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
