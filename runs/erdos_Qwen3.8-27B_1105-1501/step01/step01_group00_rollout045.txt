import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_grads(h, s_list):
    """Compute gradients of V_s for s in s_list.
    Returns array of shape (len(s_list), n)
    """
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((len(s_list), n))
    for idx, s in enumerate(s_list):
        if s >= 0:
            L = n - s
            grads[idx, s:] += dx * (1.0 - h[:L])
            grads[idx, :L] -= dx * h[s:]
        else:
            t = -s
            L = n - t
            grads[idx, :L] += dx * (1.0 - h[t:])
            grads[idx, t:] -= dx * h[:L]
    return grads

def active_set_lp(h, top_k=15, delta=0.1):
    """Solve the active-set LP to find descent direction.
    Returns (direction, t_value) or (None, None)
    """
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0-h, mode='full') * dx
    L = len(vals)  # 2n-1
    
    # Select active set: top_k values
    active_idx = np.argsort(vals)[-top_k:][::-1]  # descending
    active_s = active_idx - (n-1)
    
    # Compute gradients for active set
    J = compute_grads(h, active_s)  # shape (top_k, n)
    v_active = vals[active_idx]  # shape (top_k,)
    
    # LP: minimize t
    # subject to: J @ d - t <= -v_active (i.e., v_active + J@p <= t)
    #             sum(d) = 0
    #             -delta <= d_i <= delta
    #             t <= v_active.max() (upper bound on t to ensure descent)
    
    # Variables: [d_0, ..., d_{n-1}, t]  (n+1 vars)
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active p: J[p] @ d - t <= -v_active[p]
    A_parts = []
    b_parts = []
    
    # Active constraint rows
    A_active = np.hstack([J, -np.ones((top_k, 1))])
    b_active = -v_active
    A_parts.append(A_active)
    b_parts.append(b_active)

    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break

def grads_for_shifts(h, shifts):
    """Compute gradients of V_s for given shifts. Returns (len(shifts), n) array."""
    n = len(h)
    dx = 2.0 / n
    out = np.zeros((len(shifts), n))
    for idx, s in enumerate(shifts):
        if s >= 0:
            L = n - s
            out[idx, s:] += dx * (1.0 - h[:L])
            out[idx, :L] -= dx * h[s:]
        else:
            t = -s
            L = n - t
            out[idx, :L] += dx * (1.0 - h[t:])
            out[idx, t:] -= dx * h[:L]
    return out

def solve_lp_direction(h, vals, top_k, delta, rng):
    """Solve active-set LP. Returns (direction, t_opt) or (None, None)."""
    n = len(h)
    L = len(vals)  # 2n-1
    top_k = min(top_k, L)
    
    # Get indices of top_k largest vals
    active_idx = np.argsort(vals)[-top_k:][::-1]
    active_s = active_idx - (n - 1)
    v_act = vals[active_idx]
    
    J = grads_for_shifts(h, active_s)  # (top_k, n)
    v_max = v_act[0]  # largest
    
    # LP: minimize t
    # s.t. v_act + J @ d <= t  (for each active)
    #      sum(d) = 0
    #      -delta <= d_i <= delta
    #      t <= v_max - 1e-10 (ensure strict improvement)

    c[-1] = 1.0
    
    # Add small random perturbation to break degeneracy
    c[:n] += rng.uniform(-1e-6, 1e-6, n)
    
    A_parts = []
    b_parts = []
    
    # Active constraints: J @ d - t <= -v_act
    A_parts.append(np.hstack([J, -np.ones((top_k, 1))]))
    b_parts.append(-v_act)
    
    # Box: d_i <= delta
    I = np.eye(n)
    Z = np.zeros((n, 1))
    A_parts.append(np.hstack([I, Z]))
    b_parts.append(np.full(n, delta))
    # -d_i <= delta
    A_parts.append(np.hstack([-I, Z]))
    b_parts.append(np.full(n, delta))
    
    # t <= v_max - eps
    A_parts.append(np.array([np.zeros(n) + [1.0]]))
    b_parts.append(np.array([v_max - 1e-10]))
    
    A_ub = np.vstack(A_parts)
    b_ub = np.concatenate(b_parts)
    
    A_eq = np.array([[1.0] * n + [0.0]])
    b_eq = np.array([0.0])
    
    bounds = [(-delta, delta)] * n + [(None, v_max - 1e-10)]

            t = res.x[-1]
            if t < v_max - 1e-12:
                return d, t
    except Exception:
        pass
    return None, None
