    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        w, g = smooth_max_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.85
            stall += 1
            if stall > 50:
                cur = make_feasible(cur + rng.normal(0, max(0.003, T * 0.3), n), n)
                cur_c = score_exact(cur)
                step = max(0.003, step)
                stall = 0
        else:
            step = min(step * 1.03, 0.05)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c


def pair_search(h, i, j, cur_c, n_pts=40):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                best_a = 0.0
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None


def coord_search(h, i, cur_c, n_pts=30):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 20:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)


def generate_starts(h_parent, n, rng):
    starts = [h_parent.copy()]
    starts.append(make_feasible(np.sort(h_parent)))
    starts.append(make_feasible(np.sort(h_parent)[::-1]))
    starts.append(make_feasible(h_parent[::-1].copy()))
    h_bin = np.where(h_parent > 0.5, 1.0, 0.0)
    starts.append(make_feasible(h_bin))
    for _ in range(3):
        h_r = make_feasible(h_parent + rng.normal(0, 0.04, n))
        starts.append(h_r)
    # Truncated-sine perturbation
    h_sine = h_parent + 0.03 * np.sin(2.0 * np.pi * np.arange(n) / n * 3.0)
    starts.append(make_feasible(h_sine))
    return starts


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        best_h = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, Exception):
        best_h = np.full(70, 0.5)
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: SLSQP active-set from parent (25% of time)
    p1_end = t0 + budget_s * 0.25
    h1, c1 = slsqp_active_iter(best_h, p1_end, k_active=20, lam=0.0, max_rounds=6)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP from multiple diverse starts (25% of time)
    p2_end = t0 + budget_s * 0.50
    starts = generate_starts(best_h, n, rng)
    for s_h in starts:
        if time.time() > p2_end:
            break
        s_c = score_exact(s_h)
        if s_c < best_c + 0.03:
            sub_dl = min(p2_end, time.time() + budget_s * 0.035)
            if time.time() < sub_dl:
                h_s, c_s = slsqp_active_iter(s_h, sub_dl, k_active=18, lam=0.0, max_rounds=4)
                if c_s < best_c:
                    best_h, best_c = h_s, c_s

    # Phase 3: Smooth gradient descent (15% of time)
    p3_end = t0 + budget_s * 0.65
    if time.time() < p3_end:
        h3, c3 = smooth_grad_descent(best_h, p3_end, 0.005, 0.0002, 0.999, 0.005, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: SLSQP refinement after gradient descent (10% of time)
    p4_end = t0 + budget_s * 0.75
    if time.time() < p4_end:
        h4, c4 = slsqp_active_iter(best_h, p4_end, k_active=15, lam=0.0, max_rounds=4)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Pair and coordinate polishing (15% of time)
    p5_end = t0 + budget_s * 0.90
    cur = best_h.copy()
    cur_c = best_c
    n = len(cur)
    consecutive_fail = 0
    while time.time() < p5_end:
        mode = rng.random()
        if mode < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=35)
        elif mode < 0.75:
            i = int(rng.integers(0, n - 1))
            res = pair_search(cur, i, i + 1, cur_c, n_pts=40)
        elif mode < 0.90:
            i = int(rng.integers(0, n))
            res = coord_search(cur, i, cur_c, n_pts=30)
        else:
            # Nearby pair from active overlap structure
            active = get_top_k_indices(cur, 5)
            k_idx = active[int(rng.integers(0, len(active)))]
            k = k_idx - (n - 1)
            if k >= 0:
                ii = int(rng.integers(0, max(1, n - k)))
            else:
                p = -k
                ii = int(rng.integers(p, n))
            jj = (ii + k) % n if k >= 0 else (ii + k) % n
            if 0 <= jj < n and jj != ii:
                res = pair_search(cur, ii, jj, cur_c, n_pts=30)
            else:
                res = None
        if res is not None:
            cur_c = res
            consecutive_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            for _ in range(3):
                if time.time() > p5_end:
                    break
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = pair_search(cur, i2, j2, cur_c, n_pts=25)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            consecutive_fail += 1
            if consecutive_fail > 400:
                cur = make_feasible(best_h + rng.normal(0, 0.005, n), n)
                cur_c = score_exact(cur)
                consecutive_fail = 0

    # Phase 6: Resolution changes (5% of time)
    p6_end = deadline
    n_cur = len(best_h)
    for dn in (-4, 4, -6, 6):
        if time.time() > p6_end - 10:
            break
        nr = n_cur + dn
        if nr < 20 or nr > 200:
            continue
        h_r = resample(best_h, nr)
        if h_r is None:
            continue
        c_r = score_exact(h_r)
        if c_r < best_c + 0.005:
            sub_dl = min(p6_end - 5, time.time() + 30)
            if time.time() < sub_dl:
                h_r, c_r = slsqp_active_iter(h_r, sub_dl, k_active=15, lam=0.0, max_rounds=3)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
                n_cur = nr

    # Final SLSQP polish
    if time.time() < deadline - 3:
        active = get_top_k_indices(best_h, 10)
        solver = EpigraphSolver(best_h, active, best_h, 0.0)
        h_f, c_f = solver.solve(best_c)
        if h_f is not None and c_f < best_c - 1e-10:
            best_h, best_c = h_f, c_f

    # Ensure feasibility
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
