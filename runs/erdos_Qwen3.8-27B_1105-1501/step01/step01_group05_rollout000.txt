    h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_tgt, offset=0.0):
    m = len(h_src)
    if m == 0:
        return repair(np.full(n_tgt, 0.5), n_tgt)
    xs = (np.arange(m) + 0.5) / m
    xt = ((np.arange(n_tgt) + 0.5 + offset) / n_tgt) % 1.0
    xt_sorted = np.sort(xt)
    vals = np.interp(xt_sorted, xs, h_src, left=h_src[0], right=h_src[-1])
    idx = np.argsort(xt)
    h_out = np.empty(n_tgt)
    h_out[idx] = vals
    return repair(h_out, n_tgt)


def pair_line_minimize(h, i, j, n_golden=25):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-12:
        return h
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h_c = h.copy()
    h_c[i] = c
    h_c[j] = S - c
    fc = objective(h_c)
    h_d = h.copy()
    h_d[i] = d
    h_d[j] = S - d
    fd = objective(h_d)
    for _ in range(n_golden):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h_c = h.copy()
            h_c[i] = c
            h_c[j] = S - c
            fc = objective(h_c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h_d = h.copy()
            h_d[i] = d
            h_d[j] = S - d
            fd = objective(h_d)
    best_x = (a + b) / 2.0
    h_best = h.copy()
    h_best[i] = best_x
    h_best[j] = S - best_x
    h_best = np.clip(h_best, 0.0, 1.0)
    return h_best


def full_pair_pass(h, pass_order, n_golden=25, deadline=None):
    n = len(h)
    cur_score = objective(h)
    improved = False
    for pair_idx in pass_order:
        if deadline is not None and time.perf_counter() > deadline:
            break
        i, j = pair_idx
        h_new = pair_line_minimize(h, i, j, n_golden)
        s_new = objective(h_new)
        if s_new < cur_score - 1e-13:
            h = h_new
            cur_score = s_new
            improved = True
    return h, cur_score, improved


def local_search(h, deadline, max_passes=200, n_golden=25, seed=42):
    n = len(h)
    rng = np.random.default_rng(seed)
    cur_score = objective(h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    no_improve_count = 0
    for p in range(max_passes):
        if time.perf_counter() > deadline:
            break
        order = list(pairs)
        rng.shuffle(order)
        h, cur_score, improved = full_pair_pass(h, order, n_golden, deadline)
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 2:
                break
        else:
            no_improve_count = 0
    return h, objective(h)


def block_perturbation(h, rng, block_size=8, step_max=0.15):
    n = len(h)
    bs = min(block_size, n)
    block = rng.choice(n, size=bs, replace=False)
    d = rng.normal(0, 1, bs)
    d -= np.mean(d)
    norm = np.max(np.abs(d))
    if norm < 1e-15:
        return h.copy()
    d /= norm
    scale = rng.uniform(0.02, step_max)
    h_new = h.copy()
    for k in range(bs):
        idx = int(block[k])
        h_new[idx] = h[idx] + scale * d[k]
    return h_new


def gen_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return repair(h, n)


def gen_binary_pattern(n, rng):
    period = rng.choice([3, 4, 5, 6, 7, 8, 9, 10, 12])
    fill = rng.uniform(0.4, 0.6)
    base = np.zeros(period)
    n_ones = int(round(fill * period))
    base[:n_ones] = 1.0
    rng.shuffle(base)
    repeats = n // period + 2
    h = np.tile(base, repeats)[:n]
    return repair(h, n)


def gen_smooth(n, rng):
    freq = rng.uniform(2, 8)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 1, n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
    h += rng.normal(0, 0.03, n)
    return repair(h, n)


def gen_blended(h_a, h_b, rng, n):
    wa = interpolate_to(h_a, n, rng.uniform(-0.1, 0.1))
    wb = interpolate_to(h_b, n, rng.uniform(-0.1, 0.1))
    w = rng.beta(2, 2)
    h = (1 - w) * wa + w * wb
    return repair(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if pv.size >= 8 and pv.size <= 200 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = repair(parent, n_parent)
    else:
        n_parent = 64
        best_h = repair(np.full(n_parent, 0.5), n_parent)

    best_score = objective(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    n_candidates = []
    for dn in [-15, -10, -5, -2, 0, 2, 5, 10, 15, 20]:
        nt = n_parent + dn
        if 24 <= nt <= 100:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))
    if not n_candidates:
        n_candidates = [32, 48, 64, 80]

    scored_candidates = []

    best_h_repaired = repair(best_h, best_n)
    scored_candidates.append((objective(best_h_repaired), best_h_repaired.copy()))
    scored_candidates.append((objective(repair(best_h[::-1], best_n)), repair(best_h[::-1], best_n).copy()))
    comp = repair(1.0 - best_h, best_n)
    scored_candidates.append((objective(comp), comp.copy()))

    for nt in n_candidates:
        if time.perf_counter() > deadline - 100:
            break
        for off in [0.0, 0.5, 1.0]:
            h_c = interpolate_to(best_h, nt, off)
            s_c = objective(h_c)
            scored_candidates.append((s_c, h_c.copy()))

    for nt in n_candidates[:6]:
        if time.perf_counter() > deadline - 100:
            break
        for _ in range(3):
            h_c = gen_binary(nt, rng)
            scored_candidates.append((objective(h_c), h_c.copy()))
        for _ in range(3):
            h_c = gen_binary_pattern(nt, rng)
            scored_candidates.append((objective(h_c), h_c.copy()))
        for _ in range(3):
            h_c = gen_smooth(nt, rng)
            scored_candidates.append((objective(h_c), h_c.copy()))

    scored_candidates.sort(key=lambda x: x[0])
    seen_n = set()
    unique_candidates = []
    for s, h in scored_candidates:
        n = len(h)
        key = (n, round(s, 6))
        if key not in seen_n:
            seen_n.add(key)
            unique_candidates.append((s, h))
    scored_candidates = unique_candidates[:30]

    local_results = []
    n_local = min(12, len(scored_candidates))
    for idx in range(n_local):
        if time.perf_counter() > deadline - 200:
            break
        s_c, h_c = scored_candidates[idx]
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_alloc = min(remaining * 0.6 / max(1, n_local - idx), 80.0)
        t_alloc = max(t_alloc, 5.0)
        dl = time.perf_counter() + t_alloc
        h_loc, s_loc = local_search(h_c, dl, max_passes=100, n_golden=22, seed=seed + idx * 31)
        local_results.append((s_loc, h_loc))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c

    local_results.sort(key=lambda x: x[0])

    basin_results = []
    n_basin = min(8, len(local_results))
    for idx in range(n_basin):
        if time.perf_counter() > deadline - 150:
            break
        s_c, h_c = local_results[idx]
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_alloc = min(remaining * 0.5 / max(1, n_basin - idx), 60.0)
        t_alloc = max(t_alloc, 5.0)
        dl_basin = time.perf_counter() + t_alloc * 0.4
        dl_local = time.perf_counter() + t_alloc

        for hop in range(3):
            if time.perf_counter() > dl_basin:
                break
            h_mut = block_perturbation(h_c, rng, block_size=rng.integers(4, max(5, n_c // 4)), step_max=0.12)
            h_mut = repair(h_mut, n_c)
            s_mut = objective(h_mut)
            if s_mut < s_c + 0.002:
                h_loc, s_loc = local_search(h_mut, dl_local, max_passes=50, n_golden=22, seed=seed + 1000 + idx * 13 + hop)
                basin_results.append((s_loc, h_loc))
                if s_loc < best_score:
                    best_score = s_loc
                    best_h = h_loc.copy()
                    best_n = n_c

    all_results = local_results + basin_results
    all_results.sort(key=lambda x: x[0])

    n_final = min(6, len(all_results))
    for idx in range(n_final):
        if time.perf_counter() > deadline - 50:
            break
        s_c, h_c = all_results[idx]
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_alloc = min(remaining * 0.5, 40.0)
        t_alloc = max(t_alloc, 5.0)
        dl = time.perf_counter() + t_alloc
        h_pol, s_pol = local_search(h_c, dl, max_passes=100, n_golden=30, seed=seed + 5000 + idx)
        if s_pol < best_score:
            best_score = s_pol
            best_h = h_pol.copy()
            best_n = n_c

    if time.perf_counter() < deadline - 20:
        for nt in [best_n, best_n + 1, best_n - 1, best_n + 3, best_n - 3]:
            if nt < 24 or nt > 100:
                continue
            if time.perf_counter() > deadline - 15:
                break
            h_interp = interpolate_to(best_h, nt)
            h_loc, s_loc = local_search(h_interp, deadline - 10, max_passes=50, n_golden=25, seed=seed + 9000)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()
                best_n = nt

    if time.perf_counter() < deadline - 10:
        h_fin, s_fin = local_search(best_h, deadline - 5, max_passes=50, n_golden=30, seed=seed + 12345)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    best_h = repair(best_h, len(best_h))
    final_score = objective(best_h)
    if final_score < best_score:
        best_score = final_score

    return (best_h, float(best_score), int(len(best_h)))
