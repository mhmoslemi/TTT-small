import time

from scipy.signal import fftconvolve

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(h, alpha, beta=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_scaled = np.max(scaled)
    if max_scaled > 500:
        return float('inf')
    val = (1.0 / alpha) * np.log(np.sum(np.exp(scaled - max_scaled))) + max_scaled / alpha
    if beta > 0:
        sum_h = np.sum(h)
        target = n / 2.0
        val += beta * (sum_h - target) ** 2
    return val

def smooth_gradient(h, alpha, beta=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    
    # w has length 2n-1, corresponding to lags -(n-1) to n-1
    # w_nonneg corresponds to lags 0 to n-1, which are indices n-1 to 2n-2
    w_nonneg = w[n-1:]  # length n
    
    # First term: convolve w_nonneg with (1-h), take first n elements
    first_term = np.convolve(w_nonneg, 1.0 - h, mode='full')[:n]
    
    # Second term: correlate h with w_nonneg, take first n elements
    second_term = np.correlate(h, w_nonneg, mode='full')[:n]
    
    grad = dx * (first_term - second_term)
    
    if beta > 0:
        target = n / 2.0
        grad += 2.0 * beta * (np.sum(h) - target) * np.ones(n)
    
    return grad

def smooth_objective_and_grad(h, alpha, beta=1e4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    
    # Objective: (1/alpha) * log(sum_exp) + max_scaled/alpha + beta * (sum(h) - n/2)^2
    obj = (1.0 / alpha) * np.log(sum_exp) + max_scaled / alpha
    target = n / 2.0
    sum_h = np.sum(h)
    obj += beta * (sum_h - target) ** 2
    
    # Gradient
    w = exp_scaled / sum_exp  # normalized weights, length 2n-1
    w_nonneg = w[n-1:]  # length n, corresponds to lags 0 to n-1
    
    first_term = np.convolve(w_nonneg, 1.0 - h, mode='full')[:n]
    second_term = np.correlate(h, w_nonneg, mode='full')[:n]
    
    grad = dx * (first_term - second_term)
    grad += 2.0 * beta * (sum_h - target) * np.ones(n)
    
    return obj, grad

def project_sum(h, target):
    """Project h onto {x: 0 <= x <= 1, sum(x) = target} using bisection."""
    lo = -1.0
    hi = 2.0
    for _ in range(60):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_and_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    s = np.sum(exp_s)

    target = n * 0.5
    sh = np.sum(h)

    t2 = np.correlate(h, w_pos, mode='full')[:n]

    return obj, g

    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
