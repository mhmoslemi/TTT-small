import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Water-filling projection
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):

    if n_src == n_dst:

    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

from scipy.optimize import minimize

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def subgrad_descent(h, max_iter=500, alpha0=0.1, decay=0.98):
    """Subgradient descent for max of correlations."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = official_score(h)
    alpha = alpha0
    
    for it in range(max_iter):
        h = make_feasible(h)  # ensure feasibility
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        score = max_val * dx
        
        if score < best_score - 1e-10:
            best_score = score
            best_h = h.copy()
        
        # Find active lags (within tolerance of max)
        tol = 0.05 * max_val + 1e-10
        active = np.where(corr >= max_val - tol)[0]
        
        # Compute subgradient: average gradient of active lags
        subgrad = np.zeros(n)
        for m in active:
            lag = m - (n - 1)
            # Gradient of corr[m] w.r.t. h
            if lag >= 0:
                # corr[m] = sum_{i=0}^{n-lag-1} h[i+lag]*(1-h[i])
                # d/dh[j] = (1-h[j-lag]) if j>=lag - h[j+lag] if j < n-lag
                valid_first = np.arange(lag, n)  # j where h[j] appears as h[i+lag], i=j-lag
                for j in valid_first:
                    subgrad[j] += (1.0 - h[j - lag])
                valid_second = np.arange(0, n - lag)  # j where h[j] appears as h[i], i=j
                for j in valid_second:
                    subgrad[j] += (-h[j + lag])
            else:
                # corr[m] = sum_{i=-lag}^{n-1} h[i+lag]*(1-h[i])
                # Let l = -lag > 0
                # corr[m] = sum_{i=l}^{n-1} h[i-l]*(1-h[i])
                l = -lag
                valid_first = np.arange(0, n - l)  # j where h[j] = h[i-l], i=j+l
                for j in valid_first:
                    subgrad[j] += (1.0 - h[j + l])
                valid_second = np.arange(l, n)  # j where h[j] = h[i], i=j
                for j in valid_second:
                    subgrad[j] += (-h[j - l])
        
        subgrad /= len(active)
        
        # Take step
        h = h - alpha * subgrad
        alpha *= decay
    
    return best_h, best_score

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):

def subgrad_descent(h, max_iter=300, alpha0=0.05, decay=0.995, tol_frac=0.01):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()

        max_val = np.max(corr)

            best_score = score
            best_h = h.copy()

        # Compute weighted subgradient
        subgrad = np.zeros(n)

            lag = m - (n - 1)
            if lag == 0:
                subgrad += (1.0 - 2.0 * h)
            elif lag > 0:
                subgrad[lag:] += (1.0 - h[:n - lag])
                subgrad[:n - lag] += (-h[lag:])
            else:
                l = -lag
                subgrad[:n - l] += (1.0 - h[l:])
                subgrad[l:] += (-h[:n - l])

def coord_descent(h, sweeps=20, gs_iters=45):
    n = len(h)
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_score = official_score(h)

        improved = False

def _smooth_fun(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c * dx
    return (max_c + np.log(exp_sum) / beta) * dx

def _smooth_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:] += w * (1.0 - x[:n - lag])
            grad[:n - lag] += w * (-x[lag:])
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - x[l:])
            grad[l:] += w * (-x[:n - l])
    return grad * dx

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def do_smooth_opt(h0, beta, maxiter=150):
    n = len(h0)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            _smooth_fun, h0, args=(beta,),
            jac=_smooth_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0
