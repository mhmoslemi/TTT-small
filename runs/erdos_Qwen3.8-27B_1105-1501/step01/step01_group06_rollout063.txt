    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)

import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    """Exact C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def get_corr_vals(h):
    """Get all correlation values (unnormalized)."""
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(v, target):
    """Project onto {x: 0<=x<=1, sum(x)=target}."""
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - result.sum()
    if abs(res) > 1e-10:
        if res > 0:
            order = np.argsort(1.0 - result)[::-1]
        else:
            order = np.argsort(result)[::-1]
        for i in order:
            if res > 0:
                a = min(1.0 - result[i], res)
                result[i] += a
                res -= a
            else:
                a = min(result[i], -res)
                result[i] -= a
                res += a
            if abs(res) < 1e-12:
                break
    return result

def lag_grad(h, k_idx, n):
    """Gradient of correlation at index k_idx w.r.t. h."""
    dx = 2.0 / n
    d = (n - 1) - k_idx
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[:n-d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n-d]
    else:
        g_ = -d
        g[g_:] += dx * (1.0 - h[:n-g_])
        g[:n-g_] -= dx * h[g_:]
    return g

def binary_local_search(h_bin, n, max_sweeps=2000, time_limit=None, t0=None):
    """Local search on binary sequence via swaps."""
    h = h_bin.copy()
    c5 = eval_c5(h)
    dx = 2.0 / n
    corr = get_corr_vals(h)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        best_delta = 0
        best_pair = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                corr_new = get_corr_vals(h)
                max_new = np.max(corr_new) * dx
                if max_new < c5 - 1e-12:
                    delta = c5 - max_new
                    if delta > best_delta:
                        best_delta = delta
                        best_pair = (i, j, max_new)
                h[i], h[j] = 1, 0
        if best_pair is None:
            break
        i, j, c5 = best_pair
        h[i], h[j] = 0, 1
        corr = get_corr_vals(h)
    
    return h, c5

def eval_c5(h):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(v, target):
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - result.sum()
    if abs(res) > 1e-10:
        if res > 0:
            order = np.argsort(1.0 - result)[::-1]
        else:
            order = np.argsort(result)[::-1]
        for i in order:
            if res > 0:
                a = min(1.0 - result[i], res)
                result[i] += a
                res -= a
            else:
                a = min(result[i], -res)
                result[i] -= a
                res += a
            if abs(res) < 1e-12:
                break
    return result

def lag_grad(h, k_idx, n):
    dx = 2.0 / n
    d = (n - 1) - k_idx
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[:n - d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n - d]
    else:
        g_ = -d
        g[g_:] += dx * (1.0 - h[:n - g_])
        g[:n - g_] -= dx * h[g_:]
    return g

def lp_step(h, n, rho, lam=0.05, max_active=25, tol_frac=0.02):
    dx = 2.0 / n
    corr = get_corr(h)
    c_vals = corr * dx
    c_max = c_vals.max()
    thresh = c_max - tol_frac * max(c_max, 1e-8)
    active = np.where(c_vals >= thresh)[0]
    if len(active) > max_active:
        order = np.argsort(c_vals[active])[::-1][:max_active]
        active = active[order]
    if len(active) == 0:
        active = np.array([np.argmax(c_vals)])
    G = np.array([lag_grad(h, k, n) for k in active])
    f_act = c_vals[active]
    nv = n + 2
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0
    c_obj[n + 1] = lam
    n_ineq = len(active) + 2 * n
    A_ub = np.zeros((n_ineq, nv))
    b_ub = np.zeros(n_ineq)
    A_ub[:len(active), :n] = G
    A_ub[:len(active), n] = -1.0
    b_ub[:len(active)] = -f_act
    A_ub[len(active):len(active) + n, :n] = np.eye(n)
    A_ub[len(active):len(active) + n, n + 1] = -1.0
    A_ub[len(active) + n:len(active) + 2 * n, :n] = -np.eye(n)
    A_ub[len(active) + n:len(active) + 2 * n, n + 1] = -1.0

            continue
        n_restarts = 4
        time_per_restart = (binary_deadline - time.time()) / (len(n_binary) * n_restarts)
        time_per_restart = max(time_per_restart, 2.0)
        time_per_restart = min(time_per_restart, 15.0)
        for r in range(n_restarts):
            if time.time() > binary_deadline or remaining() < 60:
                break
            dl = time.time() + time_per_restart
            h_bin = np.zeros(n)
            pos = rng.choice(n, size=int(target), replace=False)
            h_bin[pos] = 1.0
            h_ls, c5_ls = binary_local_search(h_bin, n, max_sweeps=800, deadline=dl)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n
            # Also try continuous refinement from binary
            if c5_ls < best_c5 + 0.01 and remaining() > 40:
                h_cr, c5_cr = continuous_refine(h_ls, n,
                    deadline=min(time.time() + 15.0, binary_deadline), rho0=0.1, lam=0.03)
                if c5_cr < best_c5:
                    best_c5 = c5_cr
                    best_h = h_cr.copy()
                    best_n = n

    # Phase 3: Interpolation to different n + refine
    interp_deadline = t0 + budget_s * 0.75
    for n in n_binary:
        if time.time() > interp_deadline or remaining() < 40:
            break
        h_interp = interpolate_h(best_h, best_n, n)
        c5_interp = eval_c5(h_interp)
        if c5_interp < best_c5 + 0.005:
            h_cr, c5_cr = continuous_refine(h_interp, n,
                deadline=min(time.time() + 20.0, interp_deadline), rho0=0.15, lam=0.03)
            if c5_cr < best_c5:
                best_c5 = c5_cr
                best_h = h_cr.copy()
                best_n = n

    # Phase 4: Multi-start perturbations
    multi_deadline = t0 + budget_s * 0.90
    pert_sizes = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.25]
    att = 0
    while time.time() < multi_deadline and remaining() > 25:
        ps = pert_sizes[att % len(pert_sizes)]
        att += 1
        h_pert = best_h + rng.normal(0, ps, best_n)
        h_pert = project_sum(h_pert, best_n / 2.0)
        c5_pert = eval_c5(h_pert)
        if c5_pert < best_c5:
            h_cr, c5_cr = continuous_refine(h_pert, best_n,
                deadline=min(time.time() + 10.0, multi_deadline), rho0=0.1, lam=0.03)
            if c5_cr < best_c5:
                best_c5 = c5_cr
                best_h = h_cr.copy()
        elif c5_pert < best_c5 + 0.003 and remaining() > 30:
            h_cr, c5_cr = continuous_refine(h_pert, best_n,
                deadline=min(time.time() + 8.0, multi_deadline), rho0=0.08, lam=0.05)
            if c5_cr < best_c5:
                best_c5 = c5_cr
                best_h = h_cr.copy()

    # Phase 5: Final polish
    if remaining() > 15:
        h_final, c5_final = continuous_refine(best_h.copy(), best_n,
            deadline=min(time.time() + remaining() * 0.6, t0 + budget_s - 5.0),
            rho0=0.05, lam=0.01)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_sum(best_h, best_n / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, best_n
