Let me think about the gradient computation more carefully.

For lag k (where k ranges from -(n-1) to n-1):
F_k(h) = sum_{i=start}^{end-1} h[i] * (1 - h[i+k])

where start = max(0, -k), end = min(n, n-k)

∂F_k/∂h[i] = (1 - h[i+k]) for i in overlap region (i+k in range)
∂F_k/∂h[i+k] = -h[i] for i in overlap region

So for a given lag k:
- For i from start to end-1:
  - grad[i] += (1 - h[i+k])
  - grad[i+k] += (-h[i])

The scaled value is C_k = dx * F_k, so the gradient of C_k is dx * grad.

Now for the LP formulation:
- Variables: d (n), p (n), t (1), total 2n+1
- Objective: minimize t + (mu/n) * sum(p_i)
- Constraints:
  - For each lag j: G[j,:] @ d - t <= -C[j]  (i.e., C[j] + G[j,:] @ d <= t)
  - sum(d) = 0 (equality, mass preservation)
  - d_i - p_i <= 0 (p_i >= d_i)
  - -d_i - p_i <= 0 (p_i >= -d_i)
  - Bounds: d_i in [lo_i, hi_i], p_i in [0, 2*rad+1e-10], t in [0, 2]

Where lo_i = max(-rad, -h[i]), hi_i = min(rad, 1-h[i])

```python
def compute_F(h):
    n = len(h)
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F

def compute_grad_rows(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        idx = k + n - 1
        start = max(0, -k)
        end = min(n, n - k)
        G[idx, start:end] += (1.0 - h[start + k:end + k])
        G[idx, start + k:end + k] += (-h[start:end])
    return G
```

```python
def lp_step(h, rad, mu, n_lags=None):
    n = len(h)
    dx = 2.0 / n
    F = compute_F(h)
    C = F * dx
    G = compute_grad_rows(h)
    cmax = np.max(C)
    
    # Select which lags to include (all or top)
    if n_lags is None or n_lags >= 2 * n - 1:
        all_idx = np.arange(2 * n - 1)
    else:
        order = np.argsort(-C)
        all_idx = order[:n_lags]
    
    n_constraints = len(all_idx)
    n_vars = 2 * n + 1  # d (n), p (n), t (1)
    
    # Objective: minimize t + (mu/n) * sum(p_i)
    c_obj = np.zeros(n_vars)
    c_obj[n:2*n] = mu / n
    c_obj[2*n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag j: G[j,:] @ d - t <= -C[j]
    # i.e., G[j,0] * d_0 + ... + G[j,n-1] * d_{n-1} + 0*p + (-1)*t <= -C[j]
    A_parts = []
    b_parts = []
    
    for idx in all_idx:
        row = np.zeros(n_vars)
        row[0:n] = G[idx, :] * dx  # gradient of C (scaled)
        row[2*n] = -1.0
        A_parts.append(row)
        b_parts.append(-C[idx])
    
    # p_i >= d_i => d_i - p_i <= 0
    for i in range(n):
        row = np.zeros(n_vars)
        row[i] = 1.0
        row[n+i] = -1.0
        A_parts.append(row)
        b_parts.append(1e-10)
    
    # p_i >= -d_i => -d_i - p_i <= 0
    for i in range(n):
        row = np.zeros(n_vars)
        row[i] = -1.0
        row[n+i] = -1.0
        A_parts.append(row)
        b_parts.append(1e-10)
    
    A_ub = np.array(A_parts)
    b_ub = np.array(b_parts)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_i = max(-rad, -h[i] + 1e-12)
        hi_i = min(rad, 1.0 - h[i] - 1e-12)
        bounds.append((lo_i, hi_i))
    for i in range(n):
        bounds.append((0.0, 2.0 * rad + 1e-10))
    bounds.append((0.0, 2.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            d = res.x[0:n].copy()
            # Ensure sum(d) = 0 exactly
            d -= np.mean(d)
            return d
    except Exception:
        pass
    return None
```

For each target bin j covering [j*dxt, (j+1)*dxt):
- The integral of h_src over this interval = (cum_src at right edge - cum_src at left edge)

```python
def area_transfer(h_src, n_tgt):
    n_src = len(h_src)
    dxs = 2.0 / n_src
    dxt = 2.0 / n_tgt
    # Build cumulative integral of step function (left-continuous)
    # At position x, the integral from 0 to x is:
    # sum of h[i] * (min(x, (i+1)*dxs) - i*dxs) for all i
    # For a step function, this is piecewise linear
    # cum[i] = integral from 0 to i*dxs = dxs * sum(h[0:i])
    cum = np.zeros(n_src + 1)
    cum[1:] = np.cumsum(h_src) * dxs
    # For target bin j: integral from j*dxt to (j+1)*dxt
    # Linear interpolation of the cumulative
    x_left = np.arange(n_tgt) * dxt
    x_right = (np.arange(n_tgt) + 1) * dxt
    x_cum = np.arange(n_src + 1) * dxs
    cum_left = np.interp(x_left, x_cum, cum)
    cum_right = np.interp(x_right, x_cum, cum)
    h_raw = (cum_right - cum_left) / dxt
    return project_exact(h_raw)
```
