import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_simplex(h, target):
    """Project h to {0 <= x <= 1, sum(x) = target}"""
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any remaining sum error
    diff = float(np.sum(result)) - target
    if abs(diff) > 1e-12:
        # Adjust one element
        if diff > 0:
            idx = np.argmax(result)
            result[idx] -= diff
        else:
            idx = np.argmin(result)
            result[idx] -= diff
        result = np.clip(result, 0.0, 1.0)
    return result

def exact_line_search(h, d, n_evals=30):
    """Exact 1D line search along h + t*d for t in feasible interval"""
    n = len(h)
    # Find feasible interval
    t_min = 0.0
    t_max = 0.0
    for i in range(n):
        if abs(d[i]) > 1e-15:
            if d[i] > 0:
                t_min = max(t_min, -h[i] / d[i])
                t_max = min(t_max, (1.0 - h[i]) / d[i])
            else:
                t_min = max(t_min, (1.0 - h[i]) / d[i])
                t_max = min(t_max, -h[i] / d[i])
    if t_min >= t_max:
        return h.copy(), compute_c5(h), 0.0
    
    # Shrink slightly to avoid boundary issues
    margin = (t_max - t_min) * 0.01
    t_min += margin
    t_max -= margin
    if t_min >= t_max:
        return h.copy(), compute_c5(h), 0.0
    
    # Sample objective
    best_t = 0.0
    best_val = compute_c5(h)
    
    # Coarse grid
    ts = np.linspace(t_min, t_max, n_evals)
    vals = np.array([compute_c5(h + t * d) for t in ts])
    best_idx = np.argmin(vals)
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Refine around best
    if 0 < best_idx < n_evals - 1:
        t_lo = ts[max(0, best_idx - 1)]
        t_hi = ts[min(n_evals - 1, best_idx + 1)]
        for _ in range(15):  # Golden section
            gr = (np.sqrt(5) - 1) / 2
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)
            v1 = compute_c5(h + t1 * d)
            v2 = compute_c5(h + t2 * d)
            if v1 < v2:
                t_hi = t2
            else:
                t_lo = t1
            if abs(t_hi - t_lo) < 1e-8:
                break
        best_t = (t_lo + t_hi) / 2
        best_val = compute_c5(h + best_t * d)
    
    # Include t=0 as candidate
    v0 = compute_c5(h)
    if v0 < best_val:
        best_t = 0.0
        best_val = v0
    
    return h + best_t * d, best_val, best_t

    return grad

def feasible_interval(h, d):
    """Find [t_min, t_max] such that h + t*d is in [0,1]^n"""
    n = len(h)
    t_min = 0.0
    t_max = 0.0
    nonzero = np.abs(d) > 1e-15
    if not np.any(nonzero):
        return 0.0, 0.0
    for i in np.where(nonzero)[0]:
        di = d[i]
        if di > 0:
            t_min = max(t_min, -h[i] / di)
            t_max = min(t_max, (1.0 - h[i]) / di)
        else:
            t_min = max(t_min, (1.0 - h[i]) / di)
            t_max = min(t_max, -h[i] / di)
    return t_min, t_max

def exact_line_search(h, d, coarse_evals=25, refine_iters=12):
    """Exact 1D line search along h + t*d"""
    n = len(h)
    t_min, t_max = feasible_interval(h, d)
    if t_max - t_min < 1e-12:
        return h.copy(), compute_c5(h)
    
    # Shrink to avoid exact boundaries
    margin = (t_max - t_min) * 0.005
    t_min += margin
    t_max -= margin
    if t_min >= t_max:
        return h.copy(), compute_c5(h)
    
    current_val = compute_c5(h)
    best_t = 0.0
    best_val = current_val
    
    # Coarse grid search
    ts = np.linspace(t_min, t_max, coarse_evals)
    for t in ts:
        candidate = h + t * d
        v = compute_c5(candidate)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Golden section refinement
    if best_t != 0.0:
        # Find neighborhood of best_t
        idx_best = int(np.argmin(np.abs(ts - best_t)))
        t_lo = ts[max(0, idx_best - 1)]
        t_hi = ts[min(coarse_evals - 1, idx_best + 1)]
        if t_lo < t_hi:
            for _ in range(refine_iters):
                gr = (np.sqrt(5) - 1) / 2
                t1 = t_hi - gr * (t_hi - t_lo)
                t2 = t_lo + gr * (t_hi - t_lo)
                v1 = compute_c5(h + t1 * d)
                v2 = compute_c5(h + t2 * d)
                if v1 < v2:
                    t_hi = t2
                else:
                    t_lo = t1
                if t_hi - t_lo < 1e-10:
                    break
            mid_t = (t_lo + t_hi) / 2
            mid_v = compute_c5(h + mid_t * d)
            if mid_v < best_val:
                best_val = mid_v
                best_t = mid_t
    
    return h + best_t * d, best_val

def make_pair_direction(h, i, j):
    """Create direction e_i - e_j"""
    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def make_random_direction(n, rng):
    """Create random zero-sum direction"""
    d = rng.randn(n)
    d -= d.mean()
    return d

def make_smooth_direction(n, rng, mode=None):
    """Create smooth sinusoidal direction"""
    if mode is None:
        mode = rng.randint(1, 5)
    x = np.arange(n) / n
    d = np.sin(2 * np.pi * mode * x + rng.uniform(0, 2*np.pi))
    d -= d.mean()
    return d

def resample(h_old, n_new):
    """Resample h from old grid to new grid using piecewise constant integration"""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    # Build cumulative integral of h_old
    cum_old = np.concatenate([[0], np.cumsum(h_old * dx_old)])
    
    # For each new cell, compute the integral over that cell
    # New cell i covers [2*i/n_new, 2*(i+1)/n_new)
    new_h = np.zeros(n_new)
    for i in range(n_new):
        x_left = 2.0 * i / n_new
        x_right = 2.0 * (i + 1) / n_new
        # Find which old cells overlap
        # Old cell j covers [2*j/n_old, 2*(j+1)/n_old)
        # Integral over [x_left, x_right] of h(x) dx
        total = 0.0
        for j in range(n_old):
            old_left = 2.0 * j / n_old
            old_right = 2.0 * (j + 1) / n_old
            overlap_left = max(x_left, old_left)
            overlap_right = min(x_right, old_right)
            if overlap_right > overlap_left:
                total += h_old[j] * (overlap_right - overlap_left)
        new_h[i] = total / dx_new
    
    # Project to exact sum constraint
    target_new = float(n_new) / 2.0
    return project_to_simplex(new_h, target_new)
