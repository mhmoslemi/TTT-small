def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(v, target):
    """Project v to [0,1]^n with sum = target"""
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    # Water-filling to get exact sum
    s = np.sum(v)
    if abs(s - target) < 1e-10:
        return v
    # Bisection for mu
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] += per
            result = np.clip(result, 0.0, 1.0)
    return result

def active_subgradient(h, top_k=8):
    """Compute subgradient of max_k c_k(h) where c_k are scaled correlations"""
    n = len(h)
    corr = compute_corr_scaled(h)
    m = len(corr)  # 2*n-1
    c_max = np.max(corr)
    
    # Select top-k correlations
    if m <= top_k:
        weights = np.ones(m) / m
    else:
        # Use soft-max weighting over top entries
        sorted_idx = np.argsort(corr)[::-1]
        top_idx = sorted_idx[:top_k]
        # Weights: 1 for top, exponential decay for rest
        tau = max(1e-4, 0.001 * c_max)
        weights = np.exp((corr - c_max) / tau)
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            weights = np.zeros(m)
            weights[top_idx] = 1.0
            w_sum = top_k
        weights /= w_sum
    
    # Compute gradient of weighted sum of correlations
    # corr[j] where lag = j - (n-1)
    grad = np.zeros(n)
    n_minus_1 = n - 1
    
    for j in range(m):
        w = weights[j]
        if w < 1e-12:
            continue
        lag = j - n_minus_1
        if lag == 0:
            # c_0 = sum_i h[i](1-h[i])
            grad += w * (1.0 - 2.0 * h)

                grad[i + lag] += w * (1.0 - h[i])
                grad[i] += w * (-h[i + lag])
        else:

            # c_{-M} = sum_{i=0}^{n-1-M} h[i]*(1-h[i+M])
            for i in range(n - M):
                grad[i] += w * (1.0 - h[i + M])
                grad[i + M] += w * (-h[i])
    
    return grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(v, target):
    n = len(v)
    v = np.clip(v.astype(float).copy(), 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:
        interior = (result > 1e-10) & (result < 1.0 - 1e-10)
        idx = np.where(interior)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
            # Re-fix
            residual2 = target - np.sum(result)
            if abs(residual2) > 1e-14:
                interior2 = (result > 0) & (result < 1.0)
                idx2 = np.where(interior2)[0]
                if len(idx2) > 0:
                    result[idx2[0]] += residual2
                    result = np.clip(result, 0.0, 1.0)
    return result

def line_coefficients(h, d):
    """Compute A_k, B_k, C_k for each correlation along the line h+t*d"""
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    n_minus_1 = n - 1
    
    C = np.zeros(m)
    B = np.zeros(m)
    A = np.zeros(m)
    
    # Lag 0
    C[n_minus_1] = np.sum(h * (1 - h))
    B[n_minus_1] = np.sum(d * (1 - 2 * h))
    A[n_minus_1] = -np.sum(d * d)
    
    # Positive lags L = 1..n-1
    for L in range(1, n):
        idx_c = n_minus_1 + L
        hc = h[L:]
        hl = h[:n-L]
        dc = d[L:]
        dl = d[:n-L]
        C[idx_c] = np.sum(hc * (1 - hl))
        B[idx_c] = np.sum(dc * (1 - hl) - hc * dl)
        A[idx_c] = -np.sum(dc * dl)
    
    # Negative lags M = 1..n-1
    for M in range(1, n):
        idx_c = n_minus_1 - M
        hc = h[:n-M]
        hl = h[M:]
        dc = d[:n-M]
        dl = d[M:]
        C[idx_c] = np.sum(hc * (1 - hl))
        B[idx_c] = np.sum(dc * (1 - hl) - hc * dl)
        A[idx_c] = -np.sum(dc * dl)
    
    # Scale by dx
    C *= dx
    B *= dx
    A *= dx
    
    return A, B, C

def exact_line_search(h, d, t_lo, t_hi, max_candidates=500):
    """Find t in [t_lo, t_hi] that minimizes max_k (A_k*t^2 + B_k*t + C_k)"""
    A, B, C = line_coefficients(h, d)
    m = len(A)
    
    # Build candidate list
    candidates = set()
    candidates.add(t_lo)
    candidates.add(t_hi)
    candidates.add(0.0)
    
    # Vertices of individual quadratics
    for k in range(m):
        if A[k] > 1e-15:
            tv = -B[k] / (2.0 * A[k])
            if t_lo <= tv <= t_hi:
                candidates.add(tv)
    
    # Pairwise intersections - only for most important pairs
    # To keep it manageable, use top-k correlations by C value
    top_k = min(30, m)
    top_idx = np.argsort(C)[::-1][:top_k]
    
    for i in range(len(top_idx)):
        for j in range(i+1, len(top_idx)):
            ki, kj = top_idx[i], top_idx[j]
            da = A[ki] - A[kj]
            db = B[ki] - B[kj]
            dc = C[ki] - C[kj]
            if abs(da) < 1e-15:
                # Linear intersection
                if abs(db) > 1e-15:
                    t_int = -dc / db
                    if t_lo <= t_int <= t_hi:
                        candidates.add(t_int)
            else:
                disc = db*db - 4.0*da*dc
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-db + sqrt_disc) / (2.0 * da)
                    t2 = (-db - sqrt_disc) / (2.0 * da)
                    if t_lo <= t1 <= t_hi:
                        candidates.add(t1)
                    if t_lo <= t2 <= t_hi:
                        candidates.add(t2)
    
    # Dense grid fallback
    n_grid = 50
    grid = np.linspace(t_lo, t_hi, n_grid)
    candidates = list(candidates) + list(grid)
    
    # Evaluate phi(t) at all candidates
    best_t = 0.0
    best_val = np.inf
    t_arr = np.array(candidates)
    # Clip to interval
    t_arr = np.clip(t_arr, t_lo, t_hi)
    
    # Vectorized evaluation
    # phi(t) = max_k A[k]*t^2 + B[k]*t + C[k]
    # Shape: (len(t_arr), m)
    vals = A[None, :] * t_arr[:, None]**2 + B[None, :] * t_arr[:, None] + C[None, :]
    phi_vals = np.max(vals, axis=1)
    
    best_idx = np.argmin(phi_vals)
    best_t = t_arr[best_idx]
    best_val = phi_vals[best_idx]
    
    return best_t, best_val

def compute_c5(h):

def make_feasible(v, target):
    n = len(v)
    v = np.clip(v.astype(float).copy(), 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s_mid > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:

def line_coefficients(h, d):

    dx = 2.0 / n
    n1 = n - 1
    C = np.zeros(m)
    B = np.zeros(m)

    B[n1] = np.sum(d * (1 - 2 * h))
    A[n1] = -np.sum(d * d)
    
    # Positive lags
    for L in range(1, n):
        idx = n1 + L
        hc = h[L:]
        hl = h[:n - L]
        dc = d[L:]
        dl = d[:n - L]
        C[idx] = np.sum(hc * (1 - hl))
        B[idx] = np.sum(dc * (1 - hl) - hc * dl)
        A[idx] = -np.sum(dc * dl)

        hc = h[:n - M]
        hl = h[M:]

        C[idx] = np.sum(hc * (1 - hl))
        B[idx] = np.sum(dc * (1 - hl) - hc * dl)
        A[idx] = -np.sum(dc * dl)

    B *= dx
    A *= dx
    return A, B, C

def active_subgrad(h, top_k=16):
    n = len(h)
    corr = compute_corr_scaled(h)
    m = len(corr)
    n1 = n - 1
    
    # Weights

    tau = max(1e-4, 0.002 * c_max)

    ws = np.sum(w)

        idx = np.argmax(corr)
        w = np.zeros(m)
        w[idx] = 1.0
    else:
        w /= ws

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_feasible(v, target):
    n = len(v)
    v = np.clip(v.astype(float).copy(), 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:
        interior = (result > 1e-10) & (result < 1.0 - 1e-10)
        idx = np.where(interior)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result


def line_coefficients(h, d):
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    n1 = n - 1
    C = np.zeros(m)
    B = np.zeros(m)
    A = np.zeros(m)
    C[n1] = np.sum(h * (1.0 - h))
    B[n1] = np.sum(d * (1.0 - 2.0 * h))
    A[n1] = -np.sum(d * d)
    for L in range(1, n):
        idx = n1 + L
        hc = h[L:]
        hl = h[:n - L]
        dc = d[L:]
        dl = d[:n - L]
        C[idx] = np.sum(hc * (1.0 - hl))
        B[idx] = np.sum(dc * (1.0 - hl) - hc * dl)
        A[idx] = -np.sum(dc * dl)
    for M in range(1, n):
        idx = n1 - M
        hc = h[:n - M]
        hl = h[M:]
        dc = d[:n - M]
        dl = d[M:]
        C[idx] = np.sum(hc * (1.0 - hl))
        B[idx] = np.sum(dc * (1.0 - hl) - hc * dl)
        A[idx] = -np.sum(dc * dl)
    C *= dx
    B *= dx
    A *= dx
    return A, B, C


def feasible_interval(h, d):
    t_lo = -np.inf
    t_hi = np.inf
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    t_lo = max(t_lo, -5.0)
    t_hi = min(t_hi, 5.0)
    if t_lo >= t_hi - 1e-12:
        return None
    return t_lo, t_hi


def exact_line_search(h, d, t_lo, t_hi):
    m = 2 * len(h) - 1
    A, B, C = line_coefficients(h, d)
    candidates = [t_lo, t_hi, 0.0]
    mask_v = A > 1e-15
    if np.any(mask_v):
        tv = -B[mask_v] / (2.0 * A[mask_v])
        valid = (tv > t_lo) & (tv < t_hi)
        candidates.extend(tv[valid].tolist())
    K = min(35, m)
    top = np.argsort(C)[::-1][:K]
    At = A[top]
    Bt = B[top]
    Ct = C[top]
    Ad = At[:, None] - At[None, :]
    Bd = Bt[:, None] - Bt[None, :]
    Cd = Ct[:, None] - Ct[None, :]
    tri = np.triu(np.ones((K, K), dtype=bool), k=1)
    da = Ad[tri]
    db = Bd[tri]
    dc = Cd[tri]
    lin_mask = np.abs(da) < 1e-15
    lin_valid = lin_mask & (np.abs(db) > 1e-15)
    if np.any(lin_valid):
        roots_lin = -dc[lin_valid] / db[lin_valid]
        valid_r = (roots_lin > t_lo) & (roots_lin < t_hi)
        candidates.extend(roots_lin[valid_r].tolist())
    quad_mask = ~lin_mask
    if np.any(quad_mask):
        disc = db[quad_mask] ** 2 - 4.0 * da[quad_mask] * dc[quad_mask]
        valid_q = disc >= 0
        if np.any(valid_q):
            sq = np.sqrt(disc[valid_q])
            idx_q = np.where(quad_mask)[0][valid_q]
            r1 = (-db[idx_q] + sq) / (2.0 * da[idx_q])
            r2 = (-db[idx_q] - sq) / (2.0 * da[idx_q])
            v1 = (r1 > t_lo) & (r1 < t_hi)
            v2 = (r2 > t_lo) & (r2 < t_hi)
            candidates.extend(r1[v1].tolist())
            candidates.extend(r2[v2].tolist())
    n_grid = 50
    candidates.extend(np.linspace(t_lo, t_hi, n_grid).tolist())
    t_arr = np.array(candidates)
    t_arr = np.clip(t_arr, t_lo, t_hi)
    vals = A[None, :] * t_arr[:, None] ** 2 + B[None, :] * t_arr[:, None] + C[None, :]
    phi = np.max(vals, axis=1)
    best_idx = np.argmin(phi)
    return float(t_arr[best_idx]), float(phi[best_idx])


def active_subgrad(h, top_k=12):
    n = len(h)
    corr = compute_corr_scaled(h)
    m = len(corr)
    n1 = n - 1
    c_max = np.max(corr)
    tau = max(1e-4, 0.003 * c_max)
    w = np.exp((corr - c_max) / tau)
    ws = np.sum(w)
    if ws < 1e-15:
        w = np.zeros(m)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for j in range(m):
        wj = w[j]
        if wj < 1e-11:
            continue
        lag = j - n1
        if lag == 0:
            grad += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wj * (1.0 - h[:n - lag])
            grad[:n - lag] -= wj * h[lag:]
        else:
            M = -lag
            grad[:n - M] += wj * (1.0 - h[M:])
            grad[M:] -= wj * h[:n - M]
    return grad


def resample(h_src, n_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src, left=0.0, right=0.0)


def optimize_with_line_search(h0, timeout, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = make_feasible(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    stall_limit = 30
    alpha = 0.003
    n_pairs = 5
    n_rands = 3

    while time.time() - t_start < timeout:
        directions = []
        g = active_subgrad(h, top_k) if (top_k := min(12, n // 5 + 4)) > 0 else active_subgrad(h)
        d = -g
        d -= np.mean(d)
        sc = np.max(np.abs(d))
        if sc > 1e-14:
            directions.append(d / sc)
        sorted_g = np.argsort(g)
        for _ in range(n_pairs):
            i = sorted_g[rng.randint(0, n // 2)]
            j = sorted_g[n - 1 - rng.randint(0, n // 2)]
            if h[i] >= 1.0 - 1e-8 or h[j] <= 1e-8:
                continue
            if i == j:
                continue
            d_pair = np.zeros(n)
            d_pair[i] = 1.0
            d_pair[j] = -1.0
            directions.append(d_pair)
        for _ in range(n_rands):
            d_r = rng.randn(n)
            d_r -= np.mean(d_r)
            sc_r = np.max(np.abs(d_r))
            if sc_r > 1e-14:
                directions.append(d_r / sc_r)
        improved = False
        for d_dir in directions:
            interval = feasible_interval(h, d_dir)
            if interval is None:
                continue
            t_lo, t_hi = interval
            t_opt, val_opt = exact_line_search(h, d_dir, t_lo, t_hi)
            h_new = h + t_opt * d_dir
            h_new = make_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                improved = True
            h = h_new

        else:

            h = best_h + beta * rng.randn(n)
            h = make_feasible(h, target)
            stall = 0
            n_pairs = max(3, n_pairs - 1)
            n_rands = min(6, n_rands + 1)

    return best_h, best_c5


def coordinate_refine(h, timeout, seed=42):

    target = float(n) / 2.0
    h = h.copy()
    best_h = h.copy()

        np.random.seed(seed + int((time.time() - t_start) * 1000) % 10000)

        order = list(rng_perm.permutation(n))
        improved = False
        for i in order:
            if time.time() - t_start >= timeout:
                break
            if h[i] <= 1e-12 or h[i] >= 1.0 - 1e-12:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            avg = 1.0 / (n - 1)

            others[i] = 0.0
            d = d - avg * others
            d[i] = 1.0 - avg * 0
            d[i] = 1.0
            d[i] -= avg
            others2 = np.ones(n)

            d[i] = n - 1
