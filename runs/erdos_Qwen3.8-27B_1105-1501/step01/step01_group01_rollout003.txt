import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_simplex(z, target):
    """Project z onto {0<=h<=1, sum(h)=target}"""
    n = len(z)
    # Find lambda such that sum(clip(z - lambda, 0, 1)) = target
    # clip(z - lambda, 0, 1) = min(max(z - lambda, 0), 1)
    # This is a decreasing function of lambda
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    # Actually we need: if z_i - lambda > 1, clip to 1; if < 0, clip to 0
    # So the function f(lambda) = sum(clip(z_i - lambda, 0, 1)) is decreasing
    # f(lo) >= n (if lo is small enough), f(hi) <= 0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z - lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if add == 0:
                break
            h[i] += add
            residual -= add
    return h

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            # c = sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i])
            # grad[i] += w * (1-h[i-shift]) for i in [shift, n-1]
            # grad[i] -= w * h[i+shift] for i in [0, n-1-shift]
            ell = n - shift
            grad[shift:n] += w * (1.0 - h[0:ell])
            grad[0:ell] -= w * h[shift:n]
        else:
            t = -shift
            ell = n - t
            # c = sum_{i=0}^{n-1-t} h[i] * (1-h[i+t])
            # grad[i] += w * (1-h[i+t]) for i in [0, n-1-t]
            # grad[i] -= w * h[i-t] for i in [t, n-1]
            grad[0:ell] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:ell]
    return grad

def plsed(h0, betas, max_iters_total, deadline):
    """Projected Log-Sum-Exp Descent"""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = official_score(h)
    alpha = 0.1
    stagnant = 0
    
    for beta in betas:
        if time.time() > deadline:
            break
        max_iters = max(30, max_iters_total // len(betas))
        for it in range(max_iters):
            if time.time() > deadline:
                break
            phi = smooth_objective(h, beta)
            grad = smooth_gradient(h, beta)
            
            # Backtracking line search
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            
            for ls in range(20):
                step = alpha * h / (grad_norm + 1e-10)  # scale step
                # Actually, simpler: fixed alpha times gradient
                h_new = project_box_simplex(h - alpha * grad, target)
                phi_new = smooth_objective(h_new, beta)
                
                if phi_new < phi - 1e-12 * max(1, abs(phi)):
                    # Success
                    h = h_new
                    phi = phi_new
                    alpha = min(alpha * 1.5, 2.0)
                    stagnant = 0
                    # Update best
                    score = official_score(h)
                    if score < best_score - 1e-14:
                        best_score = score
                        best_h = h.copy()
                    break
                else:
                    alpha = alpha * 0.5
            
            if alpha < 1e-8:
                alpha = 0.01
                # Try a random perturbation
                stagnation = phi
        # End beta loop
    
    return best_h, best_score

- Actually, that's O(n^2) moves to try. For n=50, that's ~625 moves.
- For each move, recompute C5 in O(n^2) time.
- Total per iteration: O(n^4) which is too slow for n=50.

Better approach: for each candidate move, compute the delta in C5 efficiently.

When I flip position a from 1 to 0:
- For shift s ≥ 0: c_s changes by:
  - If a is in [s, n-1] (i.e., a-s is a valid index in [0, n-1-s]): the term h[a]*(1-h[a-s]) becomes 0, so c_s decreases by (1-h[a-s]) = 1 (since h[a-s] is 0 or 1, and if it were 1, the original term was 0 anyway... wait no.
  
  Original term at position n=a-s (for shift s): h[n+s]*(1-h[n]) = h[a]*(1-h[a-s])
  Since h[a]=1, this is (1-h[a-s]).
  After flip, h[a]=0, so this becomes 0.
  Change: -(1-h[a-s]) = -(1 if h[a-s]=0 else 0) = -1 if h[a-s]=0, else 0.
  
  - If a is in [0, n-1-s] (i.e., a+s is valid): the term h[a+s]*(1-h[a]) becomes h[a+s]*(1-0) = h[a+s].
  Original: h[a+s]*(1-1) = 0.
  Change: +h[a+s].

Similarly for flipping position b from 0 to 1:
- For shift s ≥ 0:
  - If b is in [s, n-1]: term h[b]*(1-h[b-s]) becomes (1-h[b-s]). Original was 0. Change: +(1-h[b-s]).
  - If b is in [0, n-1-s]: term h[b+s]*(1-h[b]) becomes h[b+s]*0 = 0. Original was h[b+s]. Change: -h[b+s].

For shift s < 0 (t = -s):
c_{-t} = count of i in [0, n-1-t] where h[i]=1 and h[i+t]=0 (using the formula c = Σ h[i]*(1-h[i+t]))

Wait, I used a different convention. Let me stick with:
For shift s (can be negative), the correlation contribution is:
c_s = Σ_{valid n} h[n+s] * (1-h[n])

For s ≥ 0: n ranges from 0 to n-1-s
For s < 0, t=-s: n ranges from t to n-1, and c_{-t} = Σ_{n=t}^{n-1} h[n-t]*(1-h[n])

OK this is getting complicated for the binary case. Let me just use the numpy correlate for the binary case too since n is small.

For the binary local search, I'll use a simpler heuristic:
1. Start with a random balanced binary vector
2. At each step, try K random swaps (flip a random 1 to 0 and a random 0 to 1)
3. Accept if C5 decreases (or with some probability for simulated annealing)
4. After many iterations, return the best found

This is O(K * n^2) per iteration (K numpy correlates of length n), which for n=50, K=20 is about 20 * 2500 = 50,000 operations per iteration. With 10000 iterations, that's 500M operations. In numpy, each correlate is fast (C implementation), so 20 correlates of length 50 is maybe 0.01ms each, total 0.2ms per iteration. 10000 iterations = 2 seconds. That's fine.

import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_simplex(z, target):
    n = len(z)
    lo = float(np.min(z)) - 2.0
    hi = float(np.max(z)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z - lam, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if add == 0:
                break
            h[i] += add
            residual -= add
    return h

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lse_objective(h, beta):
    corr = compute_corr_scaled(h)
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            ell = n - s
            grad[s:n] += w * (1.0 - h[0:ell])
            grad[0:ell] -= w * h[s:n]
        else:
            t = -s
            ell = n - t
            grad[0:ell] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:ell]
    return grad

def plsed(h0, betas, iters_per_beta, deadline):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = official_score(h)
    for beta in betas:
        if time.time() > deadline:
            break
        alpha = 0.05
        stall_count = 0
        for it in range(iters_per_beta):
            if time.time() > deadline:
                break
            grad = lse_gradient(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            phi_old = lse_objective(h, beta)
            accepted = False
            for ls in range(25):
                h_new = project_box_simplex(h - alpha * grad, target)
                phi_new = lse_objective(h_new, beta)
                sc_new = official_score(h_new)
                if phi_new < phi_old - 1e-14 or sc_new < best_score - 1e-14:
                    h = h_new
                    if sc_new < best_score:
                        best_score = sc_new
                        best_h = h.copy()
                    alpha = min(alpha * 1.4, 5.0)
                    stall_count = 0
                    accepted = True
                    break
                else:
                    alpha *= 0.5
            if not accepted:
                stall_count += 1
                if stall_count > 15:
                    break
    return best_h, best_score

def binary_c5(bits, n):
    h = ((bits & ((1 << n) - 1)) > 0).astype(float) if isinstance(bits, int) else bits.astype(float)
    return official_score(h)

def bits_to_array(bits, n):
    return np.array([1.0 if (bits >> i) & 1 else 0.0 for i in range(n)])

def array_to_bits(h):
    bits = 0
    for i in range(len(h)):
        if h[i] > 0.5:
            bits |= (1 << i)
    return bits

def binary_local_search(bits, n, max_iters, deadline):
    target = n // 2
    cur_bits = bits
    cur_score = binary_c5(bits, n)
    best_bits = bits
    best_score = cur_score
    one_positions = [i for i in range(n) if (bits >> i) & 1]
    zero_positions = [i for i in range(n) if not (bits >> i) & 1]
    rng = np.random.default_rng(12345)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(one_positions) == 0 or len(zero_positions) == 0:
            break
        a = one_positions[rng.integers(len(one_positions))]
        b = zero_positions[rng.integers(len(zero_positions))]
        new_bits = cur_bits ^ (1 << a) ^ (1 << b)
        new_score = binary_c5(new_bits, n)
        if new_score < cur_score - 1e-15:
            cur_bits = new_bits
            cur_score = new_score
            one_positions[a] = b
            zero_positions[zero_positions.index(b)] = a
            if cur_score < best_score:
                best_score = cur_score
                best_bits = cur_bits
        else:
            if rng.random() < 0.1:
                cur_bits = new_bits
                cur_score = new_score
                one_positions[a] = b
                zero_positions[zero_positions.index(b)] = a
    return best_bits, best_score

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def threshold_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    bits = np.zeros(n)
    bits[idx] = 1.0
    return bits

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    return project_box_simplex(z, target)
