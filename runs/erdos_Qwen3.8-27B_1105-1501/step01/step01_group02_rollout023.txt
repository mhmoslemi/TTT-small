    """Find best t for the move h[i]+=t, h[j]-=t preserving sum."""
    Lo = max(-h[i], h[j] - 1.0)
    Hi = min(1.0 - h[i], h[j])
    if Hi - Lo < 1e-12:
        return 0.0
    # C5(t) is convex in t, find minimum
    # Use golden section search (works for convex functions)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = Lo, Hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h_save_i = h[i]
    h_save_j = h[j]

        h[i] = h_save_i + t
        h[j] = h_save_j - t

            fc = fd

            break
    h[i] = h_save_i
    h[j] = h_save_j

def smooth_max_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(corr)
    exp_c = np.exp((corr - mx) / temp)
    z = np.sum(exp_c)
    val = mx + temp * np.log(z)
    w = exp_c / z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        a1 = max(0, s)
        a2 = min(n - 1, n - 1 + s)
        if a1 <= a2:
            idx = np.arange(a1, a2 + 1)
            grad[idx] += wk * (1.0 - h[idx - s])
        b1 = max(0, -s)
        b2 = min(n - 1, n - 1 - s)
        if b1 <= b2:
            idx = np.arange(b1, b2 + 1)
            grad[idx] -= wk * h[idx + s]
    return val, grad

            h[i] = hi_s + d_val
            h[j] = hj_s - d_val
            fd = c5_value(h)
    t_best = (a + b) / 2.0
    h[i] = hi_s + t_best
    h[j] = hj_s - t_best
    return t_best

def pair_search_h(h, i, j, iters=22):
    """Try pair move. Returns (new_c5, improved_flag, old_i, old_j)."""
    n = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return c5_value(h), False, h[i], h[j]
    c5_before = c5_value(h)
    old_i = h[i]
    old_j = h[j]
    t = pair_golden_search(h, i, j, iters)
    c5_after = c5_value(h)
    if c5_after >= c5_before - 1e-14:
        h[i] = old_i
        h[j] = old_j
        return c5_before, False, old_i, old_j
    return c5_after, True, old_i, old_j


def coord_manifold_sweep(a, m, layout, iters=20):
    improved = False
    for p in range(m):
        a_s = a[p]
        a[p] = 0.0
        h0 = build_map(a, m, layout)
        f0 = c5_value(h0)
        a[p] = 1.0
        h1 = build_map(a, m, layout)
        f1 = c5_value(h1)
        # Golden section on [0,1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        ga, gb = 0.0, 1.0
        gc = gb - gr * (gb - ga)
        gd = ga + gr * (gb - ga)
        a[p] = gc
        hc = build_map(a, m, layout)
        fgc = c5_value(hc)
        a[p] = gd
        hd = build_map(a, m, layout)
        fgd = c5_value(hd)
        for _ in range(iters):
            if gb - ga < 1e-9:
                break
            if fgc <= fgd:
                gb = gd
                gd = gc
                fgd = fgc
                gc = gb - gr * (gb - ga)
                a[p] = gc
                hc = build_map(a, m, layout)
                fgc = c5_value(hc)
            else:
                ga = gc
                gc = gd
                fgc = fgd
                gd = ga + gr * (gb - ga)
                a[p] = gd
                hd = build_map(a, m, layout)
                fgd = c5_value(hd)
        t_best = (ga + gb) / 2.0
        a[p] = t_best
        hn = build_map(a, m, layout)
        fn = c5_value(hn)
        if fn < min(f0, f1) - 1e-14 and fn < c5_value(build_map(np.concatenate([a[:p], [a_s], a[p:1] if p+1>m else a[p:]]) if False else np.insert(np.delete(a, p), p, a_s), m, layout)) - 1e-14:
            improved = True
        elif abs(t_best - a_s) < 1e-8:
            pass
    return a, improved


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def time_left():
        return budget_s - 8.0 - (time.time() - t0)

    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass

    if h_parent is None or len(h_parent) == 0 or not np.all(np.isfinite(h_parent)):
        n = 70
        h_parent = np.clip(0.5 + 0.3 * np.sin(2.0 * np.pi * np.arange(n) / n * 3.0), 0.0, 1.0)
        h_parent = project_box_sum(h_parent, n / 2.0)

    n = len(h_parent)
    target = n / 2.0
    h_parent = project_box_sum(h_parent, target)

    best_h = h_parent.copy()
    best_c5 = c5_value(best_h)

    m = n // 2

    # Try reversed parent
    h_rev = h_parent[::-1].copy()
    h_rev = project_box_sum(h_rev, target)
    c5_rev = c5_value(h_rev)
    if c5_rev < best_c5:
        best_c5 = c5_rev
        best_h = h_rev.copy()

    # Try shifted parents
    for shift in [1, 2, n // 2, n // 3]:
        h_sh = np.roll(best_h, shift)
        c5_sh = c5_value(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()

    if time_left() < 5:
        return best_h, best_c5, n

    # Phase 1: Manifold L-BFGS-B optimization
    for layout in [0, 1]:
        if time_left() < 60:
            break

        # Project parent onto manifold
        if layout == 0:
            a0 = np.clip(0.5 * (best_h[:m] + 1.0 - best_h[m:]), 0.0, 1.0)
        else:
            a0 = np.empty(m)
            for p in range(m):
                a0[p] = 0.5 * (best_h[p] + 1.0 - best_h[n - 1 - p])
            a0 = np.clip(a0, 0.0, 1.0)

        # Evaluate initial manifold projection
        h_check = build_map(a0, m, layout)
        h_check = project_box_sum(h_check, target)
        c5_check = c5_value(h_check)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_check.copy()

        # L-BFGS-B with temperature annealing
        temps = [0.06, 0.035, 0.02, 0.01, 0.005]
        deadline_hit = [False]

        def dl_check():
            if time_left() < 30:
                deadline_hit[0] = True
                return True
            return False

        a_opt = run_lgbfgs(a0, m, layout, temps, 150, dl_check)
        h_opt = build_map(a_opt, m, layout)
        h_opt = project_box_sum(h_opt, target)
        c5_opt = c5_value(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # Coord refinement on manifold
        if time_left() > 30:
            a_ref = a_opt.copy()
            for sweep in range(3):
                if time_left() < 15:
                    break
                a_ref, imp = coord_manifold_sweep(a_ref, m, layout, iters=18)
                if not imp:
                    break
                h_ref = build_map(a_ref, m, layout)
                c5_ref = c5_value(h_ref)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()

        # Noisy restarts on manifold
        if time_left() > 60:
            for ns in [0.15, 0.3, 0.5]:
                if time_left() < 20:
                    break
                a_noisy = np.clip(a_opt + rng.randn(m) * ns, 0.0, 1.0)
                deadline_hit2 = [False]

                def dl_check2():
                    if time_left() < 15:
                        deadline_hit2[0] = True
                        return True
                    return False

                a_noisy_opt = run_lgbfgs(a_noisy, m, layout, [0.03, 0.01, 0.005], 100, dl_check2)
                h_noisy = build_map(a_noisy_opt, m, layout)
                h_noisy = project_box_sum(h_noisy, target)
                c5_noisy = c5_value(h_noisy)
                if c5_noisy < best_c5:
                    best_c5 = c5_noisy
                    best_h = h_noisy.copy()

    if time_left() < 5:
        return best_h, best_c5, n

    # Phase 2: Full pair search (systematic, all pairs)
    h_work = best_h.copy()
    max_sweeps = 30
    for sweep in range(max_sweeps):
        if time_left() < 30:
            break
        any_imp = False
        n_pairs = n * (n - 1) // 2
        # Alternate: forward and reverse order each sweep
        for i in range(n):
            if time_left() < 10:
                break
            for j in range(i + 1, n):
                if time_left() < 5:
                    break
                lo = max(-h_work[i], h_work[j] - 1.0)
                hi = min(1.0 - h_work[i], h_work[j])
                if hi - lo < 1e-10:
                    continue
                c5_b = c5_value(h_work)
                old_i = h_work[i]
                old_j = h_work[j]
                pair_golden_search(h_work, i, j, 20)
                c5_a = c5_value(h_work)
                if c5_a < c5_b - 1e-13:
                    any_imp = True
                    if c5_a < best_c5:
                        best_c5 = c5_a
                        best_h = h_work.copy()
                else:
                    h_work[i] = old_i
                    h_work[j] = old_j
        if not any_imp:
            break

    if time_left() < 5:
        return best_h, best_c5, n

    # Phase 3: Random restarts + quick optimization
    for restart in range(8):
        if time_left() < 40:
            break
        noise_scale = 0.05 + 0.05 * restart
        h_try = np.clip(best_h + rng.randn(n) * noise_scale, 0.0, 1.0)
        h_try = project_box_sum(h_try, target)
        c5_try = c5_value(h_try)

        # Quick manifold projection + L-BFGS-B
        for layout in [0, 1]:
            if time_left() < 20:
                break
            if layout == 0:
                a_try = np.clip(0.5 * (h_try[:m] + 1.0 - h_try[m:]), 0.0, 1.0)
            else:
                a_try = np.empty(m)
                for p in range(m):
                    a_try[p] = 0.5 * (h_try[p] + 1.0 - h_try[n - 1 - p])
                a_try = np.clip(a_try, 0.0, 1.0)

            deadline_hit3 = [False]

            def dl_check3():
                if time_left() < 15:
                    deadline_hit3[0] = True
                    return True
                return False

            a_try_opt = run_lgbfgs(a_try, m, layout, [0.02, 0.008], 80, dl_check3)
            h_try2 = build_map(a_try_opt, m, layout)
            h_try2 = project_box_sum(h_try2, target)
            c5_try2 = c5_value(h_try2)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

        # Also try direct full pair sweep from h_try
        if time_left() > 30 and c5_try < best_c5 + 0.01:
            h_try = project_box_sum(h_try, target)
            for i in range(n):
                if time_left() < 10:
                    break
                for j in range(i + 1, n):
                    if time_left() < 5:
                        break
                    lo = max(-h_try[i], h_try[j] - 1.0)
                    hi = min(1.0 - h_try[i], h_try[j])
                    if hi - lo < 1e-10:
                        continue
                    c5_b = c5_value(h_try)
                    old_i = h_try[i]
                    old_j = h_try[j]
                    pair_golden_search(h_try, i, j, 15)
                    c5_a = c5_value(h_try)
                    if c5_a < c5_b - 1e-13:
                        if c5_a < best_c5:
                            best_c5 = c5_a
                            best_h = h_try.copy()
                    else:
                        h_try[i] = old_i
                        h_try[j] = old_j

    if time_left() < 5:
        return best_h, best_c5, n

    # Phase 4: Final fine polishing on best_h
    h_final = best_h.copy()
    for pass_num in range(5):
        if time_left() < 15:
            break
        for i in range(n):
            if time_left() < 5:
                break
            for j in range(i + 1, n):
                if time_left() < 3:
                    break
                lo = max(-h_final[i], h_final[j] - 1.0)
                hi = min(1.0 - h_final[i], h_final[j])
                if hi - lo < 1e-10:
                    continue
                c5_b = c5_value(h_final)
                old_i = h_final[i]
                old_j = h_final[j]
                pair_golden_search(h_final, i, j, 25)
                c5_a = c5_value(h_final)
                if c5_a < best_c5:
                    best_c5 = c5_a
                    best_h = h_final.copy()
                elif c5_a >= c5_b - 1e_14 if False else c5_a >= c5_b - 1e-14:
                    h_final[i] = old_i
                    h_final[j] = old_j

    # Final feasibility
    best_h = project_box_sum(best_h, target)
    best_c5 = c5_value(best_h)

    return best_h, best_c5, n
