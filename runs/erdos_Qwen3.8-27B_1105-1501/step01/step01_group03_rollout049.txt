    """Compute score for a binary sequence represented as a 0/1 array."""
    h = bits.astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def flip_pair(bits, i, j, n):
    """Flip bits at positions i and j (one 1->0, one 0->1) to maintain sum."""
    new_bits = bits.copy()
    new_bits[i] = 1 - new_bits[i]
    new_bits[j] = 1 - new_bits[j]
    return new_bits

def simulated_annealing_binary(n, initial_bits, max_iter=50000, t_start=1.0, t_end=0.01, rng=None):
    """Simulated annealing on binary sequences with exactly n/2 ones."""
    if rng is None:
        rng = np.random.default_rng(0)
    n_half = n // 2
    bits = initial_bits.copy()
    # Ensure exactly n/2 ones
    ones_idx = np.where(bits == 1)[0]
    zeros_idx = np.where(bits == 0)[0]
    if len(ones_idx) > n_half:
        # Need to flip some ones to zeros
        excess = len(ones_idx) - n_half
        to_flip = rng.choice(ones_idx, size=excess, replace=False)
        for idx in to_flip:
            bits[idx] = 0
    elif len(ones_idx) < n_half:
        deficit = n_half - len(ones_idx)
        to_flip = rng.choice(zeros_idx, size=deficit, replace=False)
        for idx in to_flip:
            bits[idx] = 1
    
    curr_score = binary_score(bits, n)
    best_bits = bits.copy()
    best_score = curr_score
    
    for it in range(max_iter):
        t = t_start * (t_end / t_start) ** (it / max_iter)
        # Choose a random pair to flip (one 1 and one 0)
        ones_idx = np.where(bits == 1)[0]
        zeros_idx = np.where(bits == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        new_bits = bits.copy()
        new_bits[i] = 0
        new_bits[j] = 1
        new_score = binary_score(new_bits, n)
        delta = new_score - curr_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            bits = new_bits
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best_bits = bits.copy()
    return best_bits, best_score

    """Score for binary sequence. bits is a 0/1 array."""
    n = len(bits)
    h = bits.astype(np.float64)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sa_binary(n, initial_bits, max_iter, t_start, t_end, rng):
    """Simulated annealing for binary sequences with n/2 ones."""
    n_half = n // 2
    bits = initial_bits.copy()
    
    # Ensure exactly n/2 ones
    ones_count = int(np.sum(bits))
    if ones_count != n_half:
        ones_idx = np.where(bits == 1)[0]
        zeros_idx = np.where(bits == 0)[0]
        if ones_count > n_half:
            excess = ones_count - n_half
            to_zero = rng.choice(ones_idx, size=excess, replace=False)
            bits[to_zero] = 0
        else:
            deficit = n_half - ones_count
            to_one = rng.choice(zeros_idx, size=deficit, replace=False)
            bits[to_one] = 1
    
    curr_h = bits.astype(np.float64)
    curr_score = binary_score_fast(bits)
    best_bits = bits.copy()
    best_score = curr_score
    
    for it in range(max_iter):
        t = t_start * (t_end / t_start) ** (it / float(max_iter))
        
        # Choose a 1 and a 0 to swap
        ones_idx = np.where(bits == 1)[0]
        zeros_idx = np.where(bits == 0)[0]
        i = ones_idx[rng.integers(0, len(ones_idx))]
        j = zeros_idx[rng.integers(0, len(zeros_idx))]
        
        # Swap
        bits[i] = 0
        bits[j] = 1
        curr_h[i] = 0.0
        curr_h[j] = 1.0
        new_score = binary_score_fast(bits)
        
        delta = new_score - curr_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best_bits = bits.copy()
        else:
            # Revert
            bits[i] = 1
            bits[j] = 0
            curr_h[i] = 1.0
            curr_h[j] = 0.0
    
    return best_bits, best_score

    h0 = make_feasible(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0


def epig_solve(h0, maxiter=200):
    n = len(h0)
    h0 = make_feasible(h0)
    dx = 2.0 / n
    s0 = official_score(h0)
    z0 = np.concatenate([h0, [s0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': epig_ineq, 'jac': epig_ineq_jac},
        {'type': 'eq', 'fun': epig_eq, 'jac': epig_eq_jac}
    ]
    try:
        res = minimize(epig_obj, z0, jac=epig_obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x[:n])
    except Exception:
        pass
    return h0


def interp_h(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    xs = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    ht = np.interp(xt, xs, h_src)
    return make_feasible(ht)


def round_to_binary(h, n):
    nh = n // 2
    si = np.argsort(-h)
    bits = np.zeros(n)
    bits[si[:nh]] = 1
    return bits


def make_block_bits(n, num_blocks):
    nh = n // 2
    bits = np.zeros(n)
    total_ones = nh
    per_block = total_ones // num_blocks
    rem = total_ones % num_blocks
    pos = 0
    for b in range(num_blocks):
        sz = per_block + (1 if b < rem else 0)
        start = pos
        end = start + sz
        bits[start:end] = 1
        pos = end + (n - total_ones) // num_blocks if b < num_blocks - 1 else max(end, pos)
        gap = (n - 2 * total_ones) // num_blocks
        pos = end + gap
    return bits


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 8.0
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    best_h = None
    best_score = 1.0

    if parent is not None:
        parent = make_feasible(parent)
        best_h = parent.copy()
        best_score = official_score(best_h)

    def try_update(hc):
        nonlocal best_h, best_score
        hc = make_feasible(hc)
        if not np.all(np.isfinite(hc)):
            return
        sc = official_score(hc)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = hc.copy()
            best_score = sc

    # Phase 1: Binary SA for even n
    even_ns = [60, 58, 62, 56, 64, 66, 68, 70]
    for n in even_ns:
        if time.time() > deadline - 40:
            break
        nh = n // 2
        inits = []

        if parent is not None:
            hp = interp_h(parent, n)
            inits.append(round_to_binary(hp, n))

        for nb in [2, 3, 4, 5, 6]:
            b = np.zeros(n)
            seg = n // (2 * nb)
            for i in range(nb):
                s = i * 2 * seg
                b[s:s + seg] = 1
            oc = int(np.sum(b))
            if oc > nh:
                oi = np.where(b == 1)[0]
                b[rng.choice(oi, size=oc - nh, replace=False)] = 0
            elif oc < nh:
                zi = np.where(b == 0)[0]
                b[rng.choice(zi, size=nh - oc, replace=False)] = 1
            inits.append(b.copy())

        for _ in range(3):
            b = np.zeros(n)
            b[rng.choice(n, size=nh, replace=False)] = 1
            inits.append(b)

        for ini in inits[:6]:
            if time.time() > deadline - 20:
                break
            mi = 60000 if n <= 60 else 40000
            bb, bs = sa_binary(n, ini, mi, 0.25, 0.001, rng)
            try_update(bb.astype(np.float64))

        if time.time() > deadline - 20:
            break

        if best_score < official_score(interp_h(parent, n)) if parent is not None else True:
            cur_n_best = make_feasible(interp_h(best_h, n))
            for beta in [100, 400]:
                if time.time() > deadline - 15:
                    break
                cur_n_best = smooth_solve(cur_n_best, beta, maxiter=150)
                try_update(cur_n_best)

    # Phase 2: Polish best with epigraph SLSQP
    if best_h is not None and time.time() < deadline - 30:
        n_best = len(best_h)
        cur = best_h.copy()
        for mt in [300, 200]:
            if time.time() > deadline - 20:
                break
            cur = epig_solve(cur, maxiter=mt)
            try_update(cur)

    # Phase 3: Try smooth solve from best with various betas
    if best_h is not None and time.time() < deadline - 25:
        n_best = len(best_h)
        for beta in [50, 200, 800, 2000]:
            if time.time() > deadline - 15:
                break
            cand = best_h.copy()
            cand = smooth_solve(cand, beta, maxiter=200)
            try_update(cand)

    # Phase 4: Refine from best binary with small continuous perturbations
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        for sigma in [0.01, 0.03, 0.05, 0.08]:
            if time.time() > deadline - 10:
                break
            for _ in range(3):
                if time.time() > deadline - 8:
                    break
                c = make_feasible(best_h + rng.normal(0, sigma, n_best))
                c = smooth_solve(c, 500, maxiter=100)
                try_update(c)

    # Phase 5: Final epigraph polish
    if best_h is not None and time.time() < deadline - 10:
        cur = epig_solve(best_h, maxiter=300)
        try_update(cur)

    if best_h is None:
        best_h = make_feasible(np.full(60, 0.5))
        best_score = official_score(best_h)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score < 0:
        best_h = make_feasible(np.full(60, 0.5))
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
