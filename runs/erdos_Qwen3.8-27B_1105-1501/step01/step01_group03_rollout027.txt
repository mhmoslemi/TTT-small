```python
def compute_G(h, n):
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        ki = k + n - 1
        # First term: (1-h[p+k]) for valid p
        p_start = max(0, -k)
        p_end = min(n, n - k)
        G[ki, p_start:p_end] += (1 - h[p_start+k:p_end+k])
        # Second term: h[p-k] for valid p
        p_start2 = max(0, k)
        p_end2 = min(n, n + k)
        G[ki, p_start2:p_end2] += h[p_start2-k:p_end2-k]
    return G
```

```python
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_G(h, n):
    """Compute gradient matrix G[k,p] = dF_k/dh_p"""
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        ki = k + n - 1
        p1_start = max(0, -k)
        p1_end = min(n, n - k)
        G[ki, p1_start:p1_end] += (1.0 - h[p1_start+k:p1_end+k])
        p2_start = max(0, k)
        p2_end = min(n, n + k)
        G[ki, p2_start:p2_end] += h[p2_start-k:p2_end-k]
    return G

def solve_1d_minimax(C, L, dlo, dhi, max_iter=50):
    """
    Solve min_{delta in [dlo, dhi]} max_k (C[k] + L[k]*delta)
    Returns (best_delta, best_value)
    """
    if dlo >= dhi - 1e-15:
        d = dlo
        return d, float(np.max(C + L * d))
    
    # Value at boundaries and at 0 (if in range)
    candidates = [dlo, dhi]
    if dlo < 0 < dhi:
        candidates.append(0.0)
    
    best_val = float(np.min([np.max(C + L * d) for d in candidates]))
    best_delta = candidates[0]
    for d in candidates[1:]:
        if np.max(C + L * d) < best_val:
            best_val = float(np.max(C + L * d))
            best_delta = d
    
    # Binary search on c
    c_lo = 0.0  # safe lower bound
    c_hi = best_val
    
    # Tighten c_lo
    # At the optimal point, c* >= min_k C_k (trivially)
    c_lo = float(np.min(C))
    if c_lo >= c_hi - 1e-15:
        return best_delta, best_val
    
    pos_mask = L > 1e-14
    neg_mask = L < -1e-14
    zero_mask = ~(pos_mask | neg_mask)
    
    for _ in range(max_iter):
        c_mid = (c_lo + c_hi) * 0.5
        if c_hi - c_lo < 1e-13 * max(1.0, abs(c_hi)):
            break
        
        # Check feasibility
        if np.any(zero_mask):
            if np.max(C[zero_mask]) > c_mid:
                c_lo = c_mid
                continue
        
        upper = dhi
        if np.any(pos_mask):
            upper = min(upper, float(np.min((c_mid - C[pos_mask]) / L[pos_mask])))
        
        lower = dlo
        if np.any(neg_mask):
            lower = max(lower, float(np.max((c_mid - C[neg_mask]) / L[neg_mask])))
        
        if lower <= upper + 1e-15:
            c_hi = c_mid
        else:
            c_lo = c_mid
    
    # Find delta at c_hi
    upper = dhi
    if np.any(pos_mask):
        upper = min(upper, float(np.min((c_hi - C[pos_mask]) / L[pos_mask])))
    lower = dlo
    if np.any(neg_mask):
        lower = max(lower, float(np.max((c_hi - C[neg_mask]) / L[neg_mask])))
    
    if lower > upper:
        # Numerical issue, use best from candidates
        return best_delta, best_val
    
    delta_star = (lower + upper) * 0.5
    val = float(np.max(C + L * delta_star))
    
    if val < best_val:
        best_val = val
        best_delta = delta_star
    
    return best_delta, best_val

def compute_F(h, n):
    """Compute all F_k values (raw, unscaled by dx)"""
    F = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        ki = k + n - 1
        p_start = max(0, -k)
        p_end = min(n, n - k)
        F[ki] = np.sum(h[p_start:p_end] * (1.0 - h[p_start+k:p_end+k]))
    return F

def pair_exchange_refine(h, n, max_passes=50, time_limit=None, t_start=0):

    best_score = official_score(h)
    h = h.copy()

    C = compute_F(h, n)  # raw F_k values
    
    passes_no_improve = 0
    total_improvements = 0

            break

                pairs.append((i, j))
        rng_local = np.random.default_rng(p + 12345)
        rng_local.shuffle(pairs)

            if time_limit is not None and time.time() - t_start > time_limit:

            dlo = -min(h[j], 1.0 - h[i])
            dhi = min(h[i], 1.0 - h[j])
            if dhi - dlo < 1e-15:
                continue
            
            # L_k = G[k,j] - G[k,i]
            L = G[:, j] - G[:, i]

                h = np.clip(h, 0.0, 1.0)
                # Recompute C and G

                improved_this_pass += 1
                total_improvements += 1

            if passes_no_improve >= 3:
                break
        else:
            passes_no_improve = 0
```
