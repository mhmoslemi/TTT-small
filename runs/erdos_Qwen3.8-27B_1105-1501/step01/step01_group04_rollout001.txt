    best_c5 = compute_c5(best_h)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    temperature = 0.003
    min_temp = 1e-5
    no_improve = 0
    max_no_improve = 80000
    iter_count = 0

    while time.time() - t0 < time_budget:
        for _ in range(20000):
            if time.time() - t0 >= time_budget:
                return best_h, best_c5
            iter_count += 1
            # Random pair move
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if best_h[i] < 1e-10:
                # Try adding to i from j
                if best_h[j] < 1e-10:
                    continue
                # Actually swap direction
                src, dst = j, i
            else:
                if best_h[j] < 1e-10:
                    src, dst = i, j
                else:
                    if rng.random() < 0.5:
                        src, dst = i, j
                    else:
                        src, dst = j, i

            max_delta = min(best_h[src], 1.0 - best_h[dst])
            if max_delta < 1e-8:
                continue
            # Adaptive delta: smaller when close to optimum
            scale = 0.1 + 0.4 * min(1.0, temperature / 0.003)
            delta = rng.random() * max_delta * scale * 2.0
            if delta < 1e-9:
                continue

            h_new = best_h.copy()
            h_new[src] -= delta
            h_new[dst] += delta
            c5_new = compute_c5(h_new)

            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                no_improve = 0
            else:
                no_improve += 1
                # Metropolis acceptance
                if c5_new - best_c5 < temperature and rng.random() < np.exp(-(c5_new - best_c5) / max(temperature, 1e-10)):
                    best_h = h_new

        # Cooldown
        if no_improve > max_no_improve:
            # Perturb and restart
            perturb = (temperature * 0.5 + 0.01) * rng.randn(n)
            h_pert = project_to_constraint(best_h + perturb, target)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert
            no_improve = 0
            temperature = max(temperature * 0.8, min_temp)
        else:
            temperature = max(temperature * 0.995, min_temp)

    return best_h, best_c5


def greedy_polish(h0, time_budget, seed=42):
    """Greedy coordinate descent: try small changes at each position."""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deltas = [0.005, 0.01, 0.02, 0.05, 0.1]

    while time.time() - t0 < time_budget:
        improved = False
        idx = rng.permutation(n)
        for i in idx:
            if time.time() - t0 >= time_budget:
                return best_h, best_c5
            if best_h[i] < 1e-10 or best_h[i] > 1.0 - 1e-10:
                continue
            # Try moving from i to a random j
            for _ in range(5):
                j = rng.randint(n)
                if i == j:
                    continue
                if best_h[j] > 1.0 - 1e-10:
                    continue
                for d in deltas:
                    for sign in [1.0, -1.0]:
                        delta = d * sign
                        if best_h[i] + delta < -1e-10 or best_h[i] + delta > 1.0 + 1e-10:
                            continue
                        if best_h[j] - delta < -1e-10 or best_h[j] - delta > 1.0 + 1e-10:
                            continue
                        h_new = best_h.copy()
                        h_new[i] += delta
                        h_new[j] -= delta
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5 - 1e-12:
                            best_c5 = c5_new
                            best_h = h_new
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                improved = False

    return best_h, best_c5


def random_binary_init(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h[rng.permutation(n)]


def structured_init(n, rng):
    """Create a spread-out binary pattern."""
    h = np.zeros(n)
    k = n // 2
    # Place ones at roughly equally spaced positions with jitter
    base = np.arange(k) * n / k
    jitter = rng.uniform(-0.3, 0.3, size=k)
    positions = np.round(base + jitter).astype(int) % n
    # Deduplicate
    seen = set()
    valid = []
    for p in positions:
        if p not in seen:
            seen.add(p)
            valid.append(p)
    for i, p in enumerate(valid):
        h[p] = 1.0
    # Fill remaining if needed
    need = k - len(valid)
    if need > 0:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for i in range(min(need, len(zeros))):
            h[zeros[i]] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial if available
    h_init = None
    n_init = 0
    try:
        if 'initial_h_values' in dir() or True:
            h_init_arr = np.array(initial_h_values, dtype=float)
            if len(h_init_arr) > 0:
                h_init = h_init_arr
                n_init = len(h_init_arr)
    except (NameError, TypeError):
        pass

    # Phase 1: Intensive local search on initial
    if h_init is not None:
        alloc1 = min(remaining() * 0.45, 450.0)
        if alloc1 > 30:
            h_best, c5_best = local_search_sa(h_init, alloc1, seed=seed)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best
                best_n = n_init
            # Also try greedy polish
            if remaining() > 30:
                alloc1b = min(remaining() * 0.1, 80.0)
                h_best2, c5_best2 = greedy_polish(h_best, alloc1b, seed=seed + 1)
                if c5_best2 < best_c5:
                    best_c5 = c5_best2
                    best_h = h_best2
                    best_n = best_n

    # Phase 2: Try other n values
    n_candidates = [n_init] if n_init > 0 else []
    for n_try in [47, 63, 81, 99, 119, 127, 31, 41, 51, 71, 101, 131, 143, 161]:
        if n_try not in n_candidates and n_try != n_init:
            n_candidates.append(n_try)
    # Sort by proximity to n_init for better time allocation
    if n_init > 0:
        n_candidates.sort(key=lambda x: abs(x - n_init))

    for n_try in n_candidates:
        if remaining() < 40:
            break
        if n_try == n_init and best_h is not None:
            continue  # Already optimized

        alloc = min(remaining() * 0.1, 100.0)
        if alloc < 25:
            break

        # Try a couple of initializations
        for init_seed in [seed + n_try, seed + n_try + 7]:
            if remaining() < 20:
                break
            h_try = structured_init(n_try, np.random.RandomState(init_seed))
            sub_alloc = min(alloc * 0.45, remaining() - 15.0)
            if sub_alloc < 10:
                break
            h_best, c5_best = local_search_sa(h_try, sub_alloc, seed=init_seed)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best
                best_n = n_try

    # Phase 3: If we have a good solution, do final polishing
    if best_h is not None and remaining() > 30:
        # Re-run local search with remaining time
        alloc3 = min(remaining() * 0.6, 300.0)
        h_best, c5_best = local_search_sa(best_h, alloc3, seed=seed + 9999)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best

        # Greedy polish
        if remaining() > 20:
            alloc3b = min(remaining() * 0.4, 100.0)
            h_best2, c5_best2 = greedy_polish(best_h, alloc3b, seed=seed + 8888)
            if c5_best2 < best_c5:
                best_c5 = c5_best2
                best_h = h_best2

    # Fallback
    if best_h is None:
        n_fb = 99
        h_fb = structured_init(n_fb, rng)
        best_h = h_fb
        best_c5 = compute_c5(h_fb)
        best_n = n_fb

    # Ensure constraint
    target = float(best_n) / 2.0
    best_h = project_to_constraint(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
