def project_box_sum(x, target):
    """Project x onto {y : 0 <= y_i <= 1, sum(y) = target}."""
    x = np.array(x, dtype=float).copy()
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    cur_sum = np.sum(x)
    if abs(cur_sum - target) < 1e-10:
        return x
    # Binary search for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):

            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

    residual = target - np.sum(x)
    if residual > 0:

from scipy.optimize import minimize

def project_box_sum(x, target):
    x = np.array(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    cur_sum = np.sum(x)
    if abs(cur_sum - target) < 1e-10:
        return x
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            for i in range(len(x)):
                room = 1.0 - x[i]
                if room > 1e-12:
                    add = min(room, residual)
                    x[i] += add
                    residual -= add
                    if residual <= 1e-12:
                        break
        else:
            for i in range(len(x)):
                if x[i] > 1e-12:
                    sub = min(x[i], -residual)
                    x[i] -= sub
                    residual += sub
                    if residual >= -1e-12:
                        break
    return np.clip(x, 0.0, 1.0)

def smooth_max_obj_sc(a, n, T):

    h = np.empty(n)
    h[:m] = a
    h[m:] = 1.0 - a
    dx = 2.0 / n

    F = corr * dx
    M = np.max(F)
    shifted = (F - M) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    obj = M + T * np.log(Z)
    
    weights = exp_vals / Z
    grad_h = np.zeros(n)
    for r in range(len(F)):
        w = weights[r]
        if w < 1e-15:
            continue
        s = r - (n - 1)
        # Term 1: (1-h[j-s]) for j in [max(s,0), min(n-1+s, n-1)]
        j_lo = max(s, 0)
        j_hi = min(n - 1 + s, n - 1)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            grad_h[js] += (1.0 - h[js - s]) * w
        # Term 2: -h[j+s] for j in [max(0,-s), min(n-1, n-1-s)]
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            js2 = np.arange(j_lo2, j_hi2 + 1)
            grad_h[js2] -= h[js2 + s] * w
    
    grad_h *= dx
    grad_a = grad_h[:m] - grad_h[m:]
    
    return obj, grad_a

    h = np.empty(n)
    for p in range(m):
        h[p] = a[p]
        h[n - 1 - p] = 1.0 - a[p]

    F = corr * dx
    M = np.max(F)
    shifted = (F - M) / T
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    obj = M + T * np.log(Z)
    
    weights = exp_vals / Z
    grad_h = np.zeros(n)
    for r in range(len(F)):
        w = weights[r]
        if w < 1e-15:

        j_hi = min(n - 1 + s, n - 1)

2. Exhaustive pair search on the best solution (try ALL pairs, ~2 seconds per pass, do multiple passes)
3. Random pair search with more aggressive moves (fill remaining time)

import numpy as np
import time
from scipy.optimize import minimize

def project_box_sum(x, target):
    x = np.array(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    cur_sum = np.sum(x)
    if abs(cur_sum - target) < 1e-10:
        return x
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            for i in range(len(x)):
                if x[i] < 1.0:
                    add = min(1.0 - x[i], residual)
                    x[i] += add
                    residual -= add
                    if residual <= 1e-12:
                        break
        else:
            for i in range(len(x)):
                if x[i] > 0.0:
                    sub = min(x[i], -residual)
                    x[i] -= sub
                    residual += sub
                    if residual >= -1e-12:
                        break
    return np.clip(x, 0.0, 1.0)

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def build_h_from_a(a, n, layout):
    m = n // 2
    h = np.empty(n)
    if layout == 'SC':
        h[:m] = a
        h[m:] = 1.0 - a
    else:
        for p in range(m):
            h[p] = a[p]
            h[n - 1 - p] = 1.0 - a[p]
    return np.clip(h, 0.0, 1.0)

def transfer_to_manifold(h_parent, n, layout):
    m = n // 2
    if len(h_parent) != n:
        x_old = np.linspace(0.0, 2.0, len(h_parent))
        x_new = np.linspace(0.0, 2.0, n)
        h_res = np.interp(x_new, x_old, h_parent)
    else:
        h_res = h_parent.copy()
    h_res = project_box_sum(h_res, n / 2.0)
    a = np.empty(m)
    if layout == 'SC':
        a[:] = 0.5 * (h_res[:m] + 1.0 - h_res[m:])
    else:
        for p in range(m):
            a[p] = 0.5 * (h_res[p] + 1.0 - h_res[n - 1 - p])
    return np.clip(a, 0.0, 1.0)

def smooth_obj_sc(a, n, T):
    m = n // 2
    h = np.empty(n)
    h[:m] = a
    h[m:] = 1.0 - a
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    M = np.max(F)
    sh = np.clip((F - M) / T, -500.0, 0.0)
    ev = np.exp(sh)
    Z = np.sum(ev)
    obj = M + T * np.log(Z)
    w = ev / Z
    grad_h = np.zeros(n)
    for r in range(len(F)):
        wr = w[r]
        if wr < 1e-15:
            continue
        s = r - (n - 1)
        j1_lo = max(s, 0)
        j1_hi = min(n - 1 + s, n - 1)
        if j1_lo <= j1_hi:
            js = np.arange(j1_lo, j1_hi + 1)
            grad_h[js] += (1.0 - h[js - s]) * wr
        j2_lo = max(0, -s)
        j2_hi = min(n - 1, n - 1 - s)
        if j2_lo <= j2_hi:
            js2 = np.arange(j2_lo, j2_hi + 1)
            grad_h[js2] -= h[js2 + s] * wr
    grad_h *= dx
    grad_a = grad_h[:m] - grad_h[m:]
    return obj, grad_a

def smooth_obj_rc(a, n, T):
    m = n // 2
    h = np.empty(n)
    for p in range(m):
        h[p] = a[p]
        h[n - 1 - p] = 1.0 - a[p]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    M = np.max(F)
    sh = np.clip((F - M) / T, -500.0, 0.0)
    ev = np.exp(sh)
    Z = np.sum(ev)
    obj = M + T * np.log(Z)
    w = ev / Z
    grad_h = np.zeros(n)
    for r in range(len(F)):
        wr = w[r]
        if wr < 1e-15:
            continue
        s = r - (n - 1)
        j1_lo = max(s, 0)
        j1_hi = min(n - 1 + s, n - 1)
        if j1_lo <= j1_hi:
            js = np.arange(j1_lo, j1_hi + 1)
            grad_h[js] += (1.0 - h[js - s]) * wr
        j2_lo = max(0, -s)
        j2_hi = min(n - 1, n - 1 - s)
        if j2_lo <= j2_hi:
            js2 = np.arange(j2_lo, j2_hi + 1)
            grad_h[js2] -= h[js2 + s] * wr
    grad_h *= dx
    idx = np.arange(m)
    grad_a = grad_h[idx] - grad_h[n - 1 - idx]
    return obj, grad_a

def optimize_manifold(a_init, n, layout, deadline):
    m = len(a_init)
    best_a = a_init.copy()
    h_cur = build_h_from_a(best_a, n, layout)
    h_cur = project_box_sum(h_cur, n / 2.0)
    best_exact = exact_c5(h_cur)
    
    temps = [0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003]
    
    for T in temps:
        if time.time() > deadline:
            break
        func = smooth_obj_sc if layout == 'SC' else smooth_obj_rc
        try:
            res = minimize(lambda a, T=T, n=n, f=func: (f(a, n, T)),
                         best_a, jac=True, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * m,
                         options={'maxiter': 200, 'maxfun': 2000, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = np.clip(res.x, 0.0, 1.0)
                h_new = build_h_from_a(a_new, n, layout)
                h_new = project_box_sum(h_new, n / 2.0)
                c5_new = exact_c5(h_new)
                if c5_new < best_exact:
                    best_exact = c5_new
                    best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_exact

                    c5 = exact_c5(h)

                        best_val = val

                best_exact = best_c5
                improved = True
        if not improved:
            break
    
    h_final = build_h_from_a(a, n, layout)
    h_final = project_box_sum(h_final, n / 2.0)
    return a, h_final, best_exact

def exhaustive_pair_search(h, deadline, max_passes=20):
    """Try all pairs with line search."""
    n = len(h)
    best_c5 = exact_c5(h)
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi_val = h[i]
                hj_val = h[j]
                t_lo = max(-hi_val, hj_val - 1.0)
                t_hi = min(1.0 - hi_val, hj_val)
                if t_hi - t_lo < 1e-10:
                    continue
                # Grid search for optimal t
                test_ts = np.linspace(t_lo, t_hi, 13)
                best_t = 0.0
                best_c5_local = best_c5
                for t in test_ts:
                    h[i] = hi_val + t
                    h[j] = hj_val - t
                    c5 = exact_c5(h)
                    if c5 < best_c5_local:
                        best_c5_local = c5
                        best_t = t
                if best_t != 0.0:
                    h[i] = hi_val + best_t
                    h[j] = hj_val - best_t
                    best_c5 = best_c5_local
                    improved = True
                else:
                    h[i] = hi_val
                    h[j] = hj_val
        if not improved:
            break
    
    return h, best_c5
