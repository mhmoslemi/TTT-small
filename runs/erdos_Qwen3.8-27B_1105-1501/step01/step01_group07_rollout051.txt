import numpy as np
import time
import cvxpy as cp

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def score_exact(h):
    return float(np.max(compute_lags(h)))

def lag_gradient(h, p):
    n = len(h)
    c = n - 1
    d = p - c
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        for j in range(min(m, n)):
            grad[j] += (1.0 - h[j + d])
        for j in range(d, n):
            grad[j] += (-h[j - d])
    else:
        dp = -d
        m = n - dp
        for j in range(dp, n):
            grad[j] += (1.0 - h[j - dp])
        for j in range(min(m, n)):
            grad[j] += (-h[j + dp])
    return grad

def slp_step(h, max_active=20, active_width=0.08, r=0.2):
    n = len(h)
    lags = compute_lags(h)
    max_lag = float(np.max(lags))
    threshold = max_lag - active_width * max_lag
    active = np.where(lags >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(lags[active])[::-1]][:max_active]
    if len(active) == 0:
        return None, None
    
    A = np.zeros((len(active), n))
    b = np.zeros(len(active))
    for k, p in enumerate(active):
        g = lag_gradient(h, p)
        A[k] = g
        b[k] = lags[p] - np.dot(g, h)
    
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A @ x + b <= t,
        cp.sum(x) == n / 2,
        x >= 0, x <= 1,
        x - h <= r, h - x <= r,
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False, time_limit=5)
        except Exception:
            return None, None
    
    if x.value is None:
        return None, None
    
    x_lp = np.array(x.value).flatten()
    cur_c = score_exact(h)
    best_x = None
    best_c = cur_c
    for alpha in (1.0, 0.7, 0.5, 0.3, 0.15):
        x_try = make_feasible((1.0 - alpha) * h + alpha * x_lp, n)
        c_try = score_exact(x_try)
        if c_try < best_c - 1e-13:
            best_x = x_try
            best_c = c_try
            break
    if best_x is None:
        return None, None
    return best_x, best_c

def smooth_grad_step(h, T, step, n_iter, deadline):
    n = len(h)
    cur_c = score_exact(h)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        lags = compute_lags(h)
        max_lag = float(np.max(lags))
        ex = np.exp((lags - max_lag) / T)
        S = float(np.sum(ex))
        if S < 1e-300:
            w = np.zeros(len(lags))
            w[int(np.argmax(lags))] = 1.0
        else:
            w = ex / S
        g = np.zeros(n)
        for p in range(len(lags)):
            if w[p] < 1e-12:
                continue
            g += w[p] * lag_gradient(h, p)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25):
            cand = make_feasible(h + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                h = cand
                cur_c = c_new
                improved = True
                break
        if not improved:
            step *= 0.8
            if step < 1e-7:
                break
        else:
            step = min(step * 1.1, 0.1)
    return h, cur_c

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None
