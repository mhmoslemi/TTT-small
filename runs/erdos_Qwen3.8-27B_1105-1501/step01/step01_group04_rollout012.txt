def exact_line_search(h, d, n_eval=20):
    """Find best t in feasible interval minimizing C5(h + t*d)"""
    interval = feasible_interval(h, d)
    if interval is None:
        return 0.0, compute_c5(h)
    lo, hi = interval
    if hi - lo < 1e-12:
        return 0.0, compute_c5(h)
    
    best_t = 0.0
    best_c5 = compute_c5(h)
    
    # Generate candidate t values
    candidates = np.concatenate([
        [0.0, lo, hi, (lo+hi)/2],
        np.linspace(lo, hi, n_eval)
    ])
    
    for t in candidates:
        h_t = h + t * d
        if h_t.min() < -1e-8 or h_t.max() > 1.0 + 1e-8:
            continue
        h_t = np.clip(h_t, 0.0, 1.0)
        c5 = compute_c5(h_t)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    # Refine around best
    if best_t != 0.0 and best_t > lo and best_t < hi:
        step = (hi - lo) / n_eval
        for frac in [0.25, 0.5, 0.75]:
            for sign in [-1, 1]:
                t2 = best_t + sign * step * frac
                if lo < t2 < hi:
                    h_t = np.clip(h + t2 * d, 0.0, 1.0)
                    c5 = compute_c5(h_t)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_t = t2
    
    return best_t, best_c5

def resample_to(h_old, n_new):
    """Resample step function from n_old grid to n_new grid"""
    n_old = len(h_old)
    old_dx = 2.0 / n_old
    new_dx = 2.0 / n_new
    
    # Using the formula: new_h[j] = (1/new_dx) * sum_i h_old[i] * overlap(old_cell_i, new_cell_j)
    # For efficiency, use the fact that overlap can be computed from cell boundaries
    
    old_left = np.arange(n_old) * old_dx
    old_right = old_left + old_dx
    new_left = np.arange(n_new) * new_dx
    new_right = new_left + new_dx
    
    new_h = np.zeros(n_new)
    
    # For each new cell, compute overlap with all old cells
    # Vectorized approach: for each new cell j, find which old cells overlap
    # Old cell i overlaps new cell j if old_left[i] < new_right[j] and old_right[i] > new_left[j]
    
    # More efficient: use cumulative sums
    # overlap(i, j) = max(0, min(old_right[i], new_right[j]) - max(old_left[i], new_left[j]))
    
    # This is O(n_old * n_new) which is fine for our sizes
    for j in range(n_new):
        nl, nr = new_left[j], new_right[j]
        # Old cells that overlap [nl, nr]
        i_start = max(0, int(np.floor(nl / old_dx)))
        i_end = min(n_old - 1, int(np.ceil(nr / old_dx)) - 1)
        for i in range(i_start, i_end + 1):
            ol, orr = old_left[i], old_right[i]
            overlap = min(orr, nr) - max(ol, nl)
            if overlap > 0:
                new_h[j] += h_old[i] * overlap / new_dx
    
    return np.clip(new_h, 0.0, 1.0)

                d[i] = 1.0 - 1.0 / (n - 1)
            else:
                # Random pair of blocks
                b1 = rng.randint(0, n // 2)
                b2 = rng.randint(n // 2, n)
                bsz = rng.randint(3, n // 4 + 1)
                d = np.zeros(n)
                d[b1:min(b1+bsz, n)] = 1.0
                d[b2:min(b2+bsz, n)] = -1.0
                d -= d.mean()
            
            norm = np.max(np.abs(d))
            if norm < 1e-10:
                continue
            d /= norm
            
            bt, c5n = exact_line_search(h, d, n_eval)
            if c5n < best_c5 - 1e-14:
                h_new = np.clip(h + bt * d, 0.0, 1.0)
                h_new = project_to_simplex(h_new, target)
                c5c = compute_c5(h_new)
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
        
        if improved:
            stall = 0
            alpha *= 1.005
        else:
            stall += 1
            if stall > 100:
                h = project_to_simplex(best_h + 0.02 * rng.randn(n), target)
                stall = 0
                alpha = max(alpha * 0.6, 5.0)
            else:
                alpha *= 1.02
        
        if stall > 200:
            n_eval = max(10, n_eval - 2)
    
    return best_h, best_c5
