import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def eval_c5_raw(h):
    """Return the raw max correlation (without dx scaling)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_feasible(v, target_sum):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=target_sum}."""
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def make_binary_seq(n, rng):
    """Random binary sequence with n/2 ones."""
    h = np.zeros(n)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def greedy_improve_binary(h, n, max_rounds=200):
    """Greedy: try all swaps, take the best."""
    h = h.copy()
    best_raw = eval_c5_raw(h)
    
    for _ in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0
        best_pair = None
        best_new_max = best_raw
        
        for i in ones:
            hi_val = h[i]  # should be 1
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_max = eval_c5_raw(h)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_pair = (int(i), int(j))
                h[i] = hi_val
                h[j] = 0.0
        
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        best_raw = best_new_max
    
    return h, best_raw * 2.0 / n

def simulated_annealing_binary(h, n, rng, t0, deadline, t0_temp=0.05):
    """SA on binary sequences with swap moves."""
    n = len(h)
    current = h.copy()
    current_raw = eval_c5_raw(current)
    best = current.copy()
    best_raw = current_raw
    
    ones = set(np.where(current == 1.0)[0].tolist())
    zeros = set(np.where(current == 0.0)[0].tolist())
    
    n_iters = 0
    no_improve = 0
    
    while time.time() < deadline:
        t = (time.time() - t0) / (deadline - t0)
        temp = t0_temp * (0.01) ** t  # exponential cooling
        
        # Pick random i from ones, j from zeros
        i_list = list(ones)
        j_list = list(zeros)
        i = i_list[rng.integers(len(i_list))]
        j = j_list[rng.integers(len(j_list))]
        
        # Make swap
        current[i] = 0.0
        current[j] = 1.0
        new_raw = eval_c5_raw(current)
        delta = new_raw - current_raw
        
        if delta < 0 or rng.random() < np.exp(-delta / (temp * 2.0 / n)):
            # Accept
            current_raw = new_raw
            ones.discard(i)
            zeros.discard(j)
            ones.add(j)
            zeros.add(i)
            
            if new_raw < best_raw:
                best_raw = new_raw
                best = current.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Reject: undo
            current[i] = 1.0
            current[j] = 0.0
            no_improve += 1
        
        n_iters += 1
        
        # Periodic greedy polish
        if no_improve > 5000:
            current, polished_raw = greedy_improve_binary(current, n, max_rounds=50)
            current_raw = polished_raw
            ones = set(np.where(current == 1.0)[0].tolist())
            zeros = set(np.where(current == 0.0)[0].tolist())
            if current_raw < best_raw:
                best_raw = current_raw
                best = current.copy()
            no_improve = 0
    
    return best, best_raw * 2.0 / n
