def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_official(h):
    n = len(h)
    dx = 2.0 / float(n)
    return float(np.max(raw_corr(h)) * dx)

def project_to_target(v, target):
    v = np.asarray(v, dtype=float).copy()
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Exact sum fix
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add mass; add to elements with most room
            room = 1.0 - result
            idx = np.argmin(result)  # element with most room to grow
            result[idx] += residual
            if result[idx] > 1.0:
                # Distribute
                order = np.argsort(result)
                total_needed = residual
                for idx2 in order:
                    space = 1.0 - result[idx2]
                    if space > 0:
                        add = min(space, total_needed)
                        result[idx2] += add
                        total_needed -= add
                        if total_needed < 1e-15:
                            break
        else:
            # Need to remove mass
            order = np.argsort(-result)
            total_needed = -residual
            for idx2 in order:

                    rem = min(result[idx2], total_needed)
                    result[idx2] -= rem
                    total_needed -= rem

def compute_Q(h, weights):
    """Compute Q[t] = sum_k w_k (corr(e_t, g)[k] - corr(h, e_t)[k])"""
    n = len(h)
    g = 1.0 - h
    L = 2 * n - 1
    Q = np.zeros(n)

        # corr(e_t, g) = g shifted appropriately
        # corr(h, e_t) = h shifted appropriately
        # Actually let's just compute directly for small n
        p = np.zeros(n)
        p[t] = 1.0
        A_t = np.correlate(p, g, mode='full') - np.correlate(h, p, mode='full')
        Q[t] = float(np.dot(weights, A_t))
    return Q

def line_search(h, i, j, state_raw, n_grid=17):
    """Find best delta for transfer from j to i, minimizing max correlation."""
    T = min(1.0 - h[i], h[j])
    if T <= 1e-12:
        return 0.0, 0.0
    
    # Build p = e_i - e_j
    p = np.zeros(len(h))
    p[i] = 1.0
    p[j] = -1.0
    
    g = 1.0 - h
    A = np.correlate(p, g, mode='full') - np.correlate(h, p, mode='full')
    R = np.correlate(p, p, mode='full')
    
    base = state_raw  # current raw correlation
    
    def eval_delta(s):
        return float(np.max(base + s * A - s * s * R))
    
    # Grid search
    deltas = np.linspace(0, T, n_grid)
    vals = np.array([eval_delta(d) for d in deltas])
    best_idx = int(np.argmin(vals))
    best_delta = deltas[best_idx]
    best_val = vals[best_idx]
    
    # Golden section refinement around best
    if 0 < best_idx < n_grid - 1:
        lo = deltas[max(0, best_idx - 1)]
        hi = deltas[min(n_grid - 1, best_idx + 1)]
    else:
        lo, hi = 0.0, T
    
    # Golden section minimization
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_delta(c)
    fd = eval_delta(d)
    for _ in range(30):
        if b - a < 1e-10 * max(1.0, T):
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = eval_delta(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = eval_delta(d)
    
    mid = (a + b) / 2.0
    fm = eval_delta(mid)
    if fm < best_val:
        best_delta = mid
        best_val = fm
    
    return best_delta, best_val

    if best_delta == 0.0 or best_delta >= T - 1e-14:
        return best_delta
    # Golden section refinement
    idx = int(np.argmin([f(d) for d in deltas]))
    a = deltas[max(0, idx - 1)]
    b = deltas[min(n_grid - 1, idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = f(c), f(d)
    for _ in range(55):
        if b - a < 1e-13:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = f(d)
    mid = (a + b) * 0.5
    fm = f(mid)
    if fm < best_val:
        best_delta, best_val = mid, fm
    return best_delta

def compute_scores(h, raw_cur):
    n = len(h)
    raw_max = float(np.max(raw_cur))
    spread = max(raw_max - float(np.median(raw_cur)), 1e-12)
    beta = 20.0 / spread
    shifted = np.clip(beta * (raw_cur - raw_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    if wsum < 1e-30:
        k_top = min(5, len(raw_cur))
        weights = np.zeros(len(raw_cur))
        weights[np.argpartition(raw_cur, -k_top)[-k_top:]] = 1.0 / k_top
    else:
        weights /= wsum
    g = 1.0 - h
    scores = np.zeros(n)
    for t in range(n):
        p = np.zeros(n)
        p[t] = 1.0
        A_t = np.correlate(p, g, mode='full') - np.correlate(h, p, mode='full')
        scores[t] = float(np.dot(weights, A_t))
    return scores

def local_search(h, deadline, max_passes=500, rng=None, verbose=False):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_official(best_h)
    best_raw = raw_corr(best_h)
    if rng is None:
        rng = np.random.default_rng(777)
    stagnation = 0
    perturb_count = 0

    for _ in range(max_passes):
        if time.time() >= deadline:
            break
        h_cur = best_h
        raw_cur = best_raw
        c5_cur = best_c5

        scores = compute_scores(h_cur, raw_cur)
        n_cand = min(35, n)
        source_order = np.argsort(-scores)[:n_cand]
        target_order = np.argsort(scores)[:n_cand]

        candidates = []
        for si in source_order:
            for ti in target_order:
                if si != ti:
                    candidates.append((ti, si, scores[si] - scores[ti]))
        # Add random
        for _ in range(min(15, n)):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                candidates.append((i, j, scores[j] - scores[i] + rng.normal(0, 0.05)))
        candidates.sort(key=lambda x: -x[2])

        improved = False
        seen = set()
        for (i, j, sv) in candidates[:50]:
            if time.time() >= deadline:
                break
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            T = min(1.0 - h_cur[i], h_cur[j])
            if T <= 1e-12:
                continue
            best_delta = line_search_exact(h_cur, i, j, raw_cur)
            if best_delta > 1e-12:
                h_new = h_cur.copy()
                h_new[i] = h_cur[i] + best_delta
                h_new[j] = h_cur[j] - best_delta
                if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                    h_new = project_to_target(h_new, n / 2.0)
                c5_new = c5_official(h_new)
                if c5_new < c5_cur - 1e-14:
                    best_h = h_new
                    best_c5 = c5_new
                    best_raw = raw_corr(h_new)
                    c5_cur = c5_new
                    improved = True
                    break

        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 15:
                # Random perturbation to escape
                perturb_count += 1
                h_pert = best_h.copy()
                mask = rng.random(n) < 0.3
                h_pert[mask] += rng.normal(0, 0.01 * perturb_count, np.sum(mask))
                h_pert = project_to_target(h_pert, n / 2.0)
                c5_pert = c5_official(h_pert)
                if c5_pert < best_c5 + 0.002:
                    best_h = h_pert
                    best_c5 = c5_pert
                    best_raw = raw_corr(h_pert)
                    stagnation = 0
                if stagnation >= 40:
                    break
    return best_h, best_c5

def symmetrize(h):
    n = len(h)
    h_rev = h[::-1].copy()
    target = n / 2.0
    best = h.copy()
    best_c5 = c5_official(best)
    for alpha in np.linspace(0.0, 1.0, 11):
        h_sym = (1.0 - alpha) * h + alpha * h_rev
        h_sym = project_to_target(h_sym, target)
        c5 = c5_official(h_sym)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h_sym.copy()
    return best, best_c5

def resample_to(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, new_n)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h0 = None

    if h0 is None or len(h0) == 0 or not np.all(np.isfinite(h0)):
        n0 = 64
        t = np.linspace(0.0, 2.0, n0)
        h0 = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * t / 2.0)
        h0 = project_to_target(h0, n0 / 2.0)

    h0 = project_to_target(h0, len(h0) / 2.0)
    best_h = h0.copy()
    best_c5 = c5_official(h0)
    n = len(h0)

    # Phase 1: Quick warm-up (4%)
    best_h, best_c5 = local_search(best_h, t0 + budget_s * 0.04, max_passes=20, rng=rng)

    # Phase 2: Symmetrize (2%)
    h_sym, c5_sym = symmetrize(best_h)
    if c5_sym < best_c5:
        best_h, best_c5 = h_sym, c5_sym

    # Phase 3: Main search (40%)
    best_h, best_c5 = local_search(best_h, t0 + budget_s * 0.48, max_passes=600, rng=rng)

    # Phase 4: Symmetrize + search (12%)
    h_sym, c5_sym = symmetrize(best_h)
    if c5_sym < best_c5:
        best_h, best_c5 = h_sym, c5_sym
    best_h, best_c5 = local_search(best_h, t0 + budget_s * 0.62, max_passes=300, rng=rng)

    # Phase 5: Resolution changes (22%)
    parent_n = len(best_h)
    test_lengths = sorted(set([parent_n - 3, parent_n - 2, parent_n - 1,
                                parent_n + 1, parent_n + 2, parent_n + 3,
                                parent_n - 5, parent_n + 5, parent_n + 8,
                                48, 56, 60, 63, 64, 65, 68, 70, 72, 80, 96]))
    test_lengths = [ln for ln in test_lengths if 20 <= ln <= max(200, parent_n + 20)]

    for new_n in test_lengths:
        if time.time() >= t0 + budget_s * 0.84:
            break
        if new_n == len(best_h):
            continue
        h_res = resample_to(best_h, new_n)
        h_res = project_to_target(h_res, new_n / 2.0)
        child_dl = min(t0 + budget_s * 0.84, time.time() + budget_s * 0.012)
        if child_dl - time.time() > 0.3:
            h_opt, c5_opt = local_search(h_res, child_dl, max_passes=50, rng=rng)
        else:
            h_opt, c5_opt = h_res, c5_official(h_res)
        if c5_opt < best_c5 - 1e-10:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Final polish (12%)
    h_sym, c5_sym = symmetrize(best_h)
    if c5_sym < best_c5:
        best_h, best_c5 = h_sym, c5_sym
    best_h, best_c5 = local_search(best_h, t0 + budget_s * 0.97, max_passes=200, rng=rng)

    # Final: ensure feasibility
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = c5_official(best_h)

    if not np.all(np.isfinite(best_h)) or len(best_h) < 4:
        n_fb = 64
        best_h = project_to_target(np.full(n_fb, 0.5, dtype=float), n_fb / 2.0)
        final_c5 = c5_official(best_h)

    return best_h, final_c5, int(len(best_h))
