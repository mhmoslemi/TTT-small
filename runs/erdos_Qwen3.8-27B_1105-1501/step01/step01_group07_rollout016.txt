    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(z)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(-z)
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return np.clip(z, 0.0, 1.0)

def make_feasible_pal(free, n):
    """Make palindromic free variables feasible: sum(free) = n/4, 0 <= free <= 1."""
    m = n // 2
    free = np.asarray(free, dtype=np.float64).copy()[:m]
    target = n / 4.0
    lo = float(np.min(free)) - 1.0
    hi = float(np.max(free)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(free - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(free - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(z)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(-z)
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return np.clip(z, 0.0, 1.0)

def full_from_pal(free, n):
    m = n // 2
    full = np.empty(n)
    full[:m] = free
    full[m:] = free[::-1]
    return full

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def scores_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def pal_score(free, n):
    full = full_from_pal(free, n)
    return score_exact(full)

def smooth_grad_pal(free, n, T):
    full = full_from_pal(free, n)
    s = scores_all(full)
    base = s.max()
    ex = np.exp((s - base) / T)
    S = ex.sum()
    if S < 1e-300:
        mask = s >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    dx = 2.0 / n
    m = n // 2
    g_full = np.zeros(n)
    c = n - 1
    for d in range(n):
        if d == 0:
            g_full += w[c] * dx * (1.0 - 2.0 * full)
        else:
            mm = n - d
            wp = w[c + d] * dx
            wn = w[c - d] * dx
            g_full[:mm] += wn * (1.0 - full[d:]) - wp * full[d:]
            g_full[mm:] += wp
