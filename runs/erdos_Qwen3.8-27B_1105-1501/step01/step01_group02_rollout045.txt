    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target_sum}."""

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temperature)
    weights = exp_vals / np.sum(exp_vals)

        w = weights[k] * dx

            grad[indices2] -= w * h[indices2 - s]

            grad[indices2] -= w * h[indices2 - s]  # indices2 - s = indices2 + t, range [t, n-1]
    return grad
