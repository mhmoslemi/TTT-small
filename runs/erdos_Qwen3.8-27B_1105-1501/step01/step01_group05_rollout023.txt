def weighted_grad(h, w, dx, n):
    """Compute dx * sum_r w[r] * grad_r(h) where w has length 2n-1."""
    n = len(h)
    lags = np.arange(-(n-1), n)
    g = np.zeros(n)
    for r in range(2*n-1):
        s = lags[r]
        wr = w[r]
        if abs(wr) < 1e-15:
            continue
        # First term: (1-h[m-s]) for valid m-s
        if s >= 0:
            # m ranges from s to n-1
            g[s:] += wr * (1.0 - h[:n-s])
        else:
            a = -s
            # m ranges from 0 to n-1-a
            g[:n-a] += wr * (1.0 - h[a:])
        # Second term: -h[m+s] for valid m+s
        if s >= 0:
            # m ranges from 0 to n-1-s
            g[:n-s] += wr * (-h[s:])
        else:
            a = -s
            # m ranges from a to n-1
            g[a:] += wr * (-h[:n-a])
    g *= dx
    return g

import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n_target):
    h = np.array(v, dtype=float).ravel()[:n_target].copy()
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            idx = np.argmax(1.0 - h)
            delta = min(residual, 1.0 - h[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    return np.clip(h, 0.0, 1.0)


def interpolate_to(h, n_target, shift=0):
    n = len(h)
    if n == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xt = ((np.arange(n_target) + 0.5 + shift) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_s = xt[order]
    vals = np.interp(xt_s, xs, h, left=h[0], right=h[-1])
    return project_to_target(vals[np.argsort(order)], n_target)


def weighted_grad(h, w, n):
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if abs(wr) < 1e-15:
            continue
        s = lags[r]
        if s >= 0:
            if n - s > 0:
                g[s:] += wr * (1.0 - h[:n - s])
                g[:n - s] += wr * (-h[s:])
        else:
            a = -s
            if n - a > 0:
                g[:n - a] += wr * (1.0 - h[a:])
                g[a:] += wr * (-h[:n - a])
    return g * dx


def soft_max_local(h, n, time_limit=30.0):
    dx = 2.0 / n
    target = n / 2.0
    h = project_to_target(h, n)
    start_time = time.perf_counter()
    best_h = h.copy()
    best_score = official_score(h)

    for tau_frac in [0.04, 0.025, 0.015, 0.01]:
        if time.perf_counter() - start_time > time_limit:
            break
        tau = tau_frac

        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full") * dx
            mx = np.max(c)
            shifted = (c - mx) / tau
            shifted -= np.max(shifted)
            return mx + tau * np.log(np.sum(np.exp(shifted))) + 500.0 * (np.sum(x) - target) ** 2

        def grad(x):
            c = np.correlate(x, 1.0 - x, mode="full") * dx
            mx = np.max(c)
            shifted = (c - mx) / tau
            shifted -= np.max(shifted)
            ew = np.exp(shifted)
            ew /= np.sum(ew)
            g = weighted_grad(x, ew, n)
            g += 1000.0 * (np.sum(x) - target) * np.ones(n)
            return g

def slsqp_polish(h, n, time_limit=20.0):
    dx = 2.0 / n
    target = n / 2.0
    lags = np.arange(-(n - 1), n)
    L = 2 * n - 1

    def corr_vec(x):
        return np.correlate(x, 1.0 - x, mode="full") * dx

    initial_score = official_score(h)
    theta0 = initial_score + 1e-6
    x0 = np.concatenate([h, [theta0]])

    def objective(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        return x[n] - corr_vec(x[:n])

    def ineq_jac(x):
        hx = x[:n]
        jj = np.zeros((L, n + 1))
        jj[:, n] = 1.0
        for r in range(L):
            s = lags[r]
            if s >= 0:
                if n - s > 0:
                    jj[r, s:] += -(1.0 - hx[:n - s])
                    jj[r, :n - s] += hx[s:]
            else:
                a = -s
                if n - a > 0:
                    jj[r, :n - a] += -(1.0 - hx[a:])
                    jj[r, a:] += hx[:n - a]
        jj[:, :n] *= dx
        return jj

    def eq_fun(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    start = time.perf_counter()
    try:
        res = optimize.minimize(
            objective, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_new = project_to_target(res.x[:n], n)
            s_new = official_score(h_new)
            if s_new < initial_score - 1e-12:
                return h_new, s_new
    except Exception:
        pass
    return h, initial_score


def gen_random_cont(n, rng):
    h = rng.uniform(0.1, 0.9, n)
    return project_to_target(h, n)


def gen_random_discrete(n, rng):
    target = n / 2.0
    m = int(target)
    frac = target - m
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m]] = 1.0
    if frac > 0.5:
        h[idx[m]] = frac
    elif frac > 0:
        h[idx[m]] = frac
    return project_to_target(h, n)


def gen_plateau(n, rng, n_pl):
    h = np.zeros(n)
    if n_pl >= n:
        return gen_random_cont(n, rng)
    bounds = np.sort(rng.choice(range(1, n), size=n_pl - 1, replace=False))
    bounds = np.concatenate([[0], bounds, [n]])
    vals = rng.choice([0.0, 0.3, 0.5, 0.7, 1.0], size=n_pl)
    for i in range(n_pl):
        h[bounds[i]:bounds[i + 1]] = vals[i]
    return project_to_target(h, n)


def subgradient_momentum(h, n, time_limit=20.0):
    dx = 2.0 / n
    target = n / 2.0
    lags = np.arange(-(n - 1), n)
    h = project_to_target(h, n).copy()
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    momentum = np.zeros(n)
    lr = 0.3
    start = time.perf_counter()
    tau = 0.02

    while time.perf_counter() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        mx = np.max(corr)
        shifted = (corr - mx) / tau
        shifted -= np.max(shifted)
        w = np.exp(shifted)
        w /= np.sum(w)

        g = weighted_grad(h, w, n)
        g -= np.mean(g)
        momentum = 0.8 * momentum + 0.2 * g
        direction = -momentum
        direction -= np.mean(direction)
        norm = np.max(np.abs(direction))
        if norm < 1e-14:
            break
        direction /= norm

        best_alpha = 0.0
        best_s = score
        for sign in [1.0, -1.0]:
            for a in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
                h_try = project_to_target(h + sign * a * direction, n)
                s_try = official_score(h_try)
                if s_try < best_s - 1e-14:
                    best_s = s_try
                    best_alpha = sign * a
        if best_alpha != 0.0:
            h = project_to_target(h + best_alpha * direction, n)
            score = official_score(h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
            lr = min(lr * 1.1, 0.8)
        else:
            lr = max(lr * 0.7, 0.01)
            if lr < 0.01:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 5.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = project_to_target(parent, n_parent)
    else:
        n_parent = 65
        best_h = project_to_target(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    candidates = []
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h).copy())
    candidates.append((1.0 - best_h[::-1]).copy())
    h_avg = (best_h + best_h[::-1]) / 2.0
    candidates.append(project_to_target(h_avg, n_parent))

    n_options = sorted(set([
        n_parent, n_parent - 5, n_parent - 3, n_parent - 1,
        n_parent + 1, n_parent + 3, n_parent + 5, n_parent + 8
    ]))
    n_options = [x for x in n_options if x >= 24]

    for n_try in n_options:
        candidates.append(interpolate_to(best_h, n_try))
        candidates.append(interpolate_to(best_h, n_try, shift=1))
        candidates.append(interpolate_to(best_h, n_try, shift=2))

    for n_try in n_options[:5]:
        for _ in range(3):
            candidates.append(gen_random_cont(n_try, rng))
        for _ in range(2):
            candidates.append(gen_random_discrete(n_try, rng))
        for _ in range(2):
            candidates.append(gen_plateau(n_try, rng, 3))
        for _ in range(2):
            candidates.append(gen_plateau(n_try, rng, 4))

    scored = []
    for c in candidates:
        if not np.all(np.isfinite(c)):
            continue
        c = project_to_target(c, len(c))
        s = official_score(c)
        if np.isfinite(s):
            scored.append((s, c))

    scored.sort(key=lambda x: x[0])

    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    top_k = min(12, len(scored))
    local_results = []

    for idx in range(top_k):
        if time.perf_counter() > deadline - 80:
            break
        s_c, h_c = scored[idx]
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_per = min(35.0, max(4.0, remaining * 0.5 / max(1, top_k - idx)))

        # Phase A: coordinate search (fast, exact)
        h_a, s_a = coordinate_search(h_c, n_c, time_limit=t_per * 0.4)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()
            best_n = n_c

        # Phase B: soft-max L-BFGS-B
        if time.perf_counter() < deadline - 60:
            h_b, s_b = soft_max_local(h_a, n_c, time_limit=t_per * 0.4)
            if s_b < best_score:
                best_score = s_b
                best_h = h_b.copy()
                best_n = n_c
                h_a = h_b

        # Phase C: subgradient + momentum
        if time.perf_counter() < deadline - 50:
            h_c2, s_c2 = subgradient_momentum(h_a, n_c, time_limit=t_per * 0.2)
            if s_c2 < best_score:
                best_score = s_c2
                best_h = h_c2.copy()
                best_n = n_c
                h_a = h_c2

        # Phase D: another coordinate pass on best
        if time.perf_counter() < deadline - 45:
            h_d, s_d = coordinate_search(h_a, n_c, time_limit=min(10.0, t_per * 0.2))
            if s_d < best_score:
                best_score = s_d
                best_h = h_d.copy()
                best_n = n_c

        local_results.append((best_score if s_d < best_score else min(s_a, s_b, s_c2), h_a.copy()))

    local_results.sort(key=lambda x: x[0])

    # SLSQP polish on top results
    to_polish = local_results[:min(4, len(local_results))]
    if not any(np.array_equal(h_r, best_h) for _, h_r in to_polish):
        to_polish.append((best_score, best_h.copy()))

    for idx, (s_r, h_r) in enumerate(to_polish):
        if time.perf_counter() > deadline - 30:
            break
        n_r = len(h_r)
        h_p, s_p = slsqp_polish(h_r, n_r, time_limit=min(15.0, (deadline - time.perf_counter()) * 0.4))
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()
            best_n = n_r
        # Final coordinate pass after SLSQP
        if time.perf_counter() < deadline - 15:
            h_f, s_f = coordinate_search(h_p, n_r, time_limit=min(8.0, (deadline - time.perf_counter()) * 0.3))
            if s_f < best_score:
                best_score = s_f
                best_h = h_f.copy()
                best_n = n_r

    # Cross-resolution: interpolate best to nearby n and re-optimize
    if time.perf_counter() < deadline - 25:
        for dn in [-3, 3, -5, 5]:
            if time.perf_counter() > deadline - 15:
                break
            n_try = best_n + dn
            if n_try < 24:
                continue
            h_int = interpolate_to(best_h, n_try)
            h_opt, s_opt = coordinate_search(h_int, n_try, time_limit=8.0)
            if s_opt < best_score:
                best_score = s_opt
                best_h = h_opt.copy()
                best_n = n_try
            if time.perf_counter() < deadline - 15:
                h_opt2, s_opt2 = soft_max_local(h_opt, n_try, time_limit=8.0)
                if s_opt2 < best_score:
                    best_score = s_opt2
                    best_h = h_opt2.copy()
                    best_n = n_try

    # Final polish
    if time.perf_counter() < deadline - 8:
        h_fin, s_fin = coordinate_search(best_h, best_n, time_limit=min(5.0, deadline - time.perf_counter() - 3))
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    # Final validation
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
