    #        h[i] - δ <= 1 → δ >= h[i] - 1 (always true since h[i] <= 1)

    """Minimize max(C + δ*g) for δ in [a, b]. Function is convex in δ."""

        f2 = np.max(C + m2 * g)

        else:

Actually, I realize there's one more optimization I can make. The ternary search can be replaced with a more efficient method for convex piecewise linear functions.

For f(δ) = max_s (C[s] + δ * g[s]), the minimum over [a,b] is at:
- The point where the "active" set changes (a breakpoint), or
- An endpoint

The breakpoints are at δ = (C[s1] - C[s2]) / (g[s2] - g[s1]) for pairs (s1, s2) where g[s1] ≠ g[s2].

There are O(n²) breakpoints in the worst case, but typically the number of "active" lags (those near the max) is small, so the number of relevant breakpoints is small.

A practical approach: evaluate f at a, b, and at the "natural" breakpoint where the current max is tied with the second max. This gives a very good approximation.

But for simplicity and robustness, I'll stick with ternary search with 40-50 iterations.

import numpy as np
import time


def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    """Compute C5 = max_k corr_k * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_corr(h):
    """Compute full correlation array."""
    return np.correlate(h, 1.0 - h, mode="full")


def precompute_P_full(h, n):
    """Precompute P matrix: P[i, s_idx] = -v[i+s] + h[i-s]."""
    v = 1.0 - h
    P = np.zeros((n, 2 * n - 1))
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        # Vectorized computation for each i
        # For -v[i+s]: valid i range is max(0, -s) to min(n-1, n-1-s)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo < i_hi:
            P[i_lo:i_hi, idx] -= v[i_lo + s:i_hi + s]
        # For h[i-s]: valid i range is max(0, s) to min(n-1, n-1+s)
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            P[i_lo2:i_hi2, idx] += h[i_lo2 - s:i_hi2 - s]
    return P


def precompute_P_row(h, v, i, n):
    """Compute a single row of P."""
    row = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        if 0 <= i + s < n:
            row[idx] -= v[i + s]
        if 0 <= i - s < n:
            row[idx] += h[i - s]
    return row


def ternary_search(C, g, a, b, iters=45):
    """Minimize max(C + delta * g) over delta in [a, b]."""
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(C + m1 * g)
        f2 = np.max(C + m2 * g)
        if f1 <= f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

    """One full sweep over all pairs with line search.
    Returns (h, C, P, n_improved, n_accepted_worse)."""
    v = 1.0 - h
    n_imp = 0
    n_worse = 0
    order = list(range(n))
    if accept_worse:
        rng.shuffle(order)
    
    for i in order:

import numpy as np
import time


def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    """Compute C5 = max_k corr_k * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_corr(h):
    """Compute full correlation array."""
    return np.correlate(h, 1.0 - h, mode="full")


def precompute_P_full(h, n):
    """Precompute P matrix where P[i, idx] = -v[i+s] + h[i-s], s=idx-(n-1)."""
    v = 1.0 - h
    P = np.zeros((n, 2 * n - 1))
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo < i_hi:
            P[i_lo:i_hi, idx] -= v[i_lo + s:i_hi + s]
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            P[i_lo2:i_hi2, idx] += h[i_lo2 - s:i_hi2 - s]
    return P


def precompute_P_row(h, v, i, n):
    """Compute a single row of P for index i."""
    row = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        if 0 <= i + s < n:
            row[idx] -= v[i + s]
        if 0 <= i - s < n:
            row[idx] += h[i - s]
    return row


def ternary_search(C, g, a, b, iters=50):
    """Minimize max(C + delta * g) over delta in [a, b]. Convex in delta."""
    for _ in range(iters):
        if b - a < 1e-14:
            break
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(C + m1 * g)
        f2 = np.max(C + m2 * g)
        if f1 <= f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_sweep(h, C, P, n, rng, use_sa, temp, deadline):
    """One full sweep over all pairs with exact line search.
    Returns (h, C, P, n_improved)."""
    v = 1.0 - h
    n_imp = 0
    n_pairs = 0
    for i in range(n):
        for j in range(i + 1, n):
            if use_sa and n_pairs % 200 == 0 and time.time() > deadline:
                return h, C, P, n_imp
            n_pairs += 1
            dlo = -min(h[j], 1.0 - h[i])
            dhi = min(h[i], 1.0 - h[j])
            if dhi - dlo < 1e-13:
                continue
            g = P[i] - P[j]
            # Quick check: if g is all zeros, no effect
            if np.max(np.abs(g)) < 1e-15:
                continue
            old_max = np.max(C)
            delta = ternary_search(C, g, dlo, dhi, iters=50)
            if abs(delta) < 1e-13:
                continue
            C_new = C + delta * g
            new_max = np.max(C_new)
            improve = old_max - new_max
            if improve > 1e-13:
                h[i] -= delta
                h[j] += delta
                C = C_new
                v = 1.0 - h
                P[i] = precompute_P_row(h, v, i, n)
                P[j] = precompute_P_row(h, v, j, n)
                n_imp += 1
            elif use_sa and improve > -temp:

                    v = 1.0 - h
                    P[i] = precompute_P_row(h, v, i, n)
                    P[j] = precompute_P_row(h, v, j, n)

                best_h = h.copy()
    return best_h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8

    # Get parent if available
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass

    if parent is not None and len(parent) > 0:
        n = len(parent)
    else:
        n = 70

    target_sum = n / 2.0
    dx = 2.0 / n

    # Build initial candidates
    candidates = []
    if parent is not None and len(parent) > 0:
        candidates.append(parent.copy())
    candidates.append(np.random.RandomState(seed + 1).rand(n))
    candidates.append(make_block_init(n, rng))
    candidates.append(make_alt_init(n, rng))
    candidates.append(make_sin_init(n, rng))

    # Pick best initial
    best_h = None
    best_c5 = float('inf')
    best_C = None
    for cand in candidates:
        cand = project_onto_constraint(cand, target_sum)
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()
            best_C = compute_corr(cand).copy()

    # Main optimization: iterative pair sweeps
    total_rounds = 0
    stall_rounds = 0

    while time.time() < deadline:
        h = best_h.copy()
        C = best_C.copy()
        P = precompute_P_full(h, n)

        # Sub-phase 1: Pure greedy sweeps (no SA)
        for _ in range(30):
            if time.time() > deadline:
                break
            h, C, P, n_imp = pair_sweep(h, C, P, n, rng, use_sa=False, temp=0.0,
                                         deadline=deadline)
            c5 = np.max(C) * dx
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                best_C = C.copy()
                stall_rounds = 0
            if n_imp == 0:
                break

        if time.time() >= deadline:
            break

        # Sub-phase 2: SA sweeps to escape local minima
        temp = 0.003
        for sa_iter in range(10):
            if time.time() > deadline:
                break
            h2 = best_h.copy()
            C2 = best_C.copy()
            P2 = precompute_P_full(h2, n)
            h2, C2, P2, _ = pair_sweep(h2, C2, P2, n, rng, use_sa=True, temp=temp,
                                        deadline=deadline)
            # Cool
            temp *= 0.6
            c5_2 = np.max(C2) * dx
            if c5_2 < best_c5 - 1e-13:
                best_c5 = c5_2
                best_h = h2.copy()
                best_C = C2.copy()
                stall_rounds = 0

        total_rounds += 1

        # Check for stall
        c5_now = np.max(best_C) * dx
        if c5_now >= best_c5 - 1e-15:
            stall_rounds += 1

        # If stalled for too long, do a bigger perturbation
        if stall_rounds >= 3:
            rng2 = np.random.RandomState(seed + 100 + total_rounds)
            for scale in [0.05, 0.1, 0.2, 0.3]:
                h_p = best_h + rng2.randn(n) * scale
                h_p = project_onto_constraint(h_p, target_sum)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5 + 0.01:
                    best_h = h_p
                    best_C = compute_corr(h_p)
                    stall_rounds = 0
                    break
            if stall_rounds >= 5:
                # Try a completely fresh random start
                h_fresh = rng.rand(n)
                h_fresh = project_onto_constraint(h_fresh, target_sum)
                c5_f = compute_c5(h_fresh)
                if c5_f < best_c5 + 0.02:
                    best_h = h_fresh
                    best_C = compute_corr(h_fresh)
                stall_rounds = 0

    # Final refinement phase
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        C = best_C.copy()
        P = precompute_P_full(h, n)
        for _ in range(50):
            if time.time() > t0 + budget_s - 3:
                break
            h, C, P, n_imp = pair_sweep(h, C, P, n, rng, use_sa=False, temp=0.0,
                                         deadline=t0 + budget_s - 3)
            c5 = np.max(C) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_C = C.copy()
            if n_imp == 0:
                break

    # Final feasibility check
    best_h = project_onto_constraint(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, final_c5, n
