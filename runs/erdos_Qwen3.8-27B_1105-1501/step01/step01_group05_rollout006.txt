def pair_minimize(h, i, j, grid_points=15, golden_iters=8):

    for idx, x in enumerate(xs):

        scores[idx] = s
        if s < best_val:

    # Golden section on the best interval

    best_idx = np.argmin(scores)

def select_pairs(h, grad, n, count, rng):
    """Select promising pairs based on gradient differences"""
    # For a pair (i,j), the effect on the objective along the pair direction
    # is approximately (grad[i] - grad[j]) * delta (for small delta)
    # We want large |grad[i] - grad[j]|
    
    # Get magnitude of gradient
    g_mag = np.abs(grad)
    
    # Score all pairs by |grad[i] - grad[j]|
    # This is n*(n-1)/2 which for n=65 is ~2000
    # We can compute this efficiently
    
    # For efficiency, just use random + gradient-guided
    pairs = []
    seen = set()
    
    # Top gradient pairs
    top_indices = np.argsort(g_mag)[::-1]
    for idx_i in range(min(30, n)):
        i = top_indices[idx_i]
        for idx_j in range(min(30, n)):
            if idx_j == idx_i:
                continue
            j = top_indices[idx_j]
            key = (min(i,j), max(i,j))
            if key not in seen:
                seen.add(key)
                pairs.append((int(i), int(j)))
                if len(pairs) >= count:
                    return pairs
    
    # Fill with random
    while len(pairs) < count:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        key = (min(i,j), max(i,j))
        if key not in seen:
            seen.add(key)
            pairs.append((key[0], key[1]))
    
    return pairs[:count]

def slsqp_refine(h, n, time_limit=15.0):

    dx = 2.0 / n
    target_sum = n / 2.0

    scaled = corr * dx
    top_m = min(12, 2*n-1)
    active_indices = np.argsort(scaled)[::-1][:top_m]
    current_theta = float(np.max(scaled)) + 1e-5
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = np.clip(x[:n], 0, 1)
        theta = x[n]
        c = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
        return theta - c[active_indices]
    
    def ineq_jac(x):
        h_x = np.clip(x[:n], 0, 1)
        j = np.zeros((len(active_indices), n + 1))
        j[:, n] = 1.0
        for row, r in enumerate(active_indices):
            q = lags[r]
            start = max(0, -q)
            end = min(n, n - q)
            if start < end:
                i_arr = np.arange(start, end)
                p_arr = i_arr + q
                j[row, p_arr] += dx * (1.0 - h_x[i_arr])
                j[row, i_arr] -= dx * h_x[p_arr]
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    x0 = np.concatenate([h, [current_theta]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},

        pass

def gen_interpolated(parent, n_out, offset=0):

     sum(h) = n/2
     0 <= h <= 1

This is a convex optimization problem! Wait, is it? C_r(h) = dx * sum h_{i+r}(1-h_i) = dx * (sum h_{i+r} - sum h_{i+r}*h_i). The term sum h_{i+r}*h_i is bilinear (not convex). So it's NOT convex in general.

However, if we fix the values at certain positions, it becomes linear in the remaining. The pair move approach exploits this: fixing all except h_i and h_j, and h_i+h_j=S, we have h_j = S-h_i, so the problem becomes 1D in h_i with a piecewise-linear (actually piecewise-quadratic) objective.

Let me think about whether there's a better approach...

Actually, for the bilinear form, if we think of it as: for a given pair (i,j), the objective as a function of x=h_i is:
F(x) = max_r C_r(x) where C_r(x) is a quadratic in x (at most).

The max of quadratics is convex! So the 1D minimization has a unique minimum (or a flat region). This means golden section will work perfectly.

Wait, is max of quadratics always convex? No! The max of convex functions is convex, but quadratics can be non-convex (negative leading coefficient). Let me check: C_r(h) = dx * sum h_p(1-h_i). If h_i = x and h_j = S-x, then for a lag r that involves both i and j, we get a term like (S-x)(1-x) = S - Sx - x + x^2, which has positive x^2 coefficient. For a lag involving only i: x(1-h_k) for some k≠j, which is linear in x. For a lag involving only j: (S-x)(1-h_k) which is linear in x.

Wait, actually let me be more careful. C_r(h) = dx * sum over valid (i,p) pairs of h_p * (1-h_i). If we set h_i = x, h_j = S-x:

For a term h_p * (1-h_i):
- If p=j and i=i: (S-x)(1-x) = S - Sx - x + x^2 → quadratic with +x^2
- If p=i and i=j (i.e., the pair is reversed): x(1-(S-x)) = x(S-x) = Sx-x^2 → quadratic with -x^2
- If only one of i,j is involved: linear
- If neither: constant

So some terms are convex quadratic (+x^2) and some are concave quadratic (-x^2). The max of these is NOT necessarily convex. So golden section might not find the global minimum in the 1D problem.

However, in practice, the grid + golden section approach should still work well because:
1. The function is piecewise smooth
2. The grid catches the general shape
3. Golden section refines locally

The grid is essential to avoid getting trapped in a local minimum of the 1D problem.

Let me think about what else could help. One idea: after the pair descent converges to a local minimum, try a "reheating" step where we add noise and then cool down. This is essentially simulated annealing / basin hopping.

Another idea: try different n values more aggressively. The optimal n might not be 65.

Let me also think about the structure of good solutions. For the Erdős overlap problem, good constructions often have a periodic or quasi-periodic structure. The parent construction with n=65 likely has some such structure. Interpolating or modifying it slightly might help.

import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            h[idx] += min(residual, room[idx])
        else:
            idx = np.argmax(h)
            h[idx] -= min(-residual, h[idx])
    return h

def compute_active_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    theta = float(np.max(scaled))
    T = max(0.003 + 0.012 * theta, 0.002)
    weights = np.exp((scaled - theta) / T)
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        weights = np.zeros_like(scaled)
        weights[np.argmax(scaled)] = 1.0
        w_sum = 1.0
    weights /= w_sum
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for r in range(len(lags)):
        if weights[r] < 1e-7:
            continue
        q = lags[r]
        start = max(0, -q)
        end = min(n, n - q)
        if start >= end:
            continue
        i_arr = np.arange(start, end)
        p_arr = i_arr + q
        w = weights[r] * dx
        grad[p_arr] += w * (1.0 - h[i_arr])
        grad[i_arr] -= w * h[p_arr]
    grad -= np.mean(grad)
    return grad, scaled, theta

def pair_minimize(h, i, j, grid_points=15, golden_iters=8):
    n = len(h)
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-10:
        return h.copy(), official_score(h)
    xs = np.linspace(L, U, grid_points)
    best_val = np.inf
    best_x = h[i]
    scores = np.empty(grid_points)
    h_tmp = h.copy()
    for idx, x in enumerate(xs):
        h_tmp[i] = x
        h_tmp[j] = S - x
        s = official_score(h_tmp)
        scores[idx] = s
        if s < best_val:
            best_val = s
            best_x = x
    best_idx = int(np.argmin(scores))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_idx > 0:
        a = xs[best_idx - 1]
    else:
        a = L
    if best_idx < grid_points - 1:
        b = xs[best_idx + 1]
    else:
        b = U
    if b - a < 1e-12:
        h_out = h.copy()
        h_out[i] = best_x
        h_out[j] = S - best_x
        return h_out, official_score(h_out)
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h_tmp[i] = c
    h_tmp[j] = S - c
    fc = official_score(h_tmp)
    h_tmp[i] = d
    h_tmp[j] = S - d
    fd = official_score(h_tmp)
    for _ in range(golden_iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h_tmp[i] = c
            h_tmp[j] = S - c
            fc = official_score(h_tmp)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h_tmp[i] = d
            h_tmp[j] = S - d
            fd = official_score(h_tmp)
    if fc < best_val:
        best_val = fc
        best_x = c
    if fd < best_val:
        best_val = fd
        best_x = d
    h_out = h.copy()
    h_out[i] = best_x
    h_out[j] = S - best_x
    h_out = np.clip(h_out, 0.0, 1.0)
    final_s = official_score(h_out)
    return h_out, final_s

def select_pairs(h, grad, n, count, rng):
    g_diff = np.abs(grad)
    top_idx = np.argsort(g_diff)[::-1][:min(25, n)]
    pairs = []
    seen = set()
    for ii in range(len(top_idx)):
        i = int(top_idx[ii])
        for jj in range(len(top_idx)):
            if jj == ii:
                continue
            j = int(top_idx[jj])
            key = (min(i, j), max(i, j))
            if key not in seen:
                seen.add(key)
                pairs.append(key)
                if len(pairs) >= count:
                    return pairs
    attempts = 0
    while len(pairs) < count and attempts < count * 10:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            attempts += 1
            continue
        key = (min(i, j), max(i, j))
        if key not in seen:
            seen.add(key)
            pairs.append(key)
        attempts += 1
    return pairs[:count]

def local_pair_descent(h, n, time_limit, seed, max_passes=50):
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n)
    score = official_score(h)
    start_time = time.perf_counter()
    stagnation = 0
    for pass_num in range(max_passes):
        if time.perf_counter() - start_time > time_limit:
            break
        grad, scaled, theta = compute_active_grad(h)
        n_pairs = min(120, max(40, 5000 // (n // 10 + 1)))
        pairs = select_pairs(h, grad, n, n_pairs, rng)
        pass_improved = False
        for (i, j) in pairs:
            if time.perf_counter() - start_time > time_limit:
                break
            new_h, new_s = pair_minimize(h, i, j, grid_points=13, golden_iters=7)
            if new_s < score - 1e-13:
                h = new_h
                score = new_s
                pass_improved = True
        if not pass_improved:
            stagnation += 1
            if np.linalg.norm(grad) > 1e-10:
                direction = -grad
                direction -= np.mean(direction)
                norm = np.max(np.abs(direction))
                if norm > 1e-12:
                    direction /= norm
                    best_new_s = score
                    best_new_h = h
                    for alpha in [0.01, 0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                        for sign in [1, -1]:
                            h_try = repair_vector(h + sign * alpha * direction, n)
                            s_try = official_score(h_try)
                            if s_try < best_new_s - 1e-13:
                                best_new_s = s_try
                                best_new_h = h_try.copy()
                    if best_new_s < score:
                        h = best_new_h
                        score = best_new_s
                        pass_improved = True
                        stagnation = 0
            if stagnation >= 4:
                noise_scale = 0.03 + 0.02 * stagnation
                noise = rng.normal(0, noise_scale, n)
                noise -= np.mean(noise)
                h_noisy = repair_vector(h + noise, n)
                s_noisy = official_score(h_noisy)
                if s_noisy < score + 0.005:
                    h = h_noisy
                    score = s_noisy
                stagnation = 0
        else:
            stagnation = 0
    return h, official_score(h)

def slsqp_refine(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    top_m = min(15, 2 * n - 1)
    active_indices = np.argsort(scaled)[::-1][:top_m]
    current_theta = float(np.max(scaled)) + 1e-5

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = np.clip(x[:n], 0.0, 1.0)
        c = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
        return x[n] - c[active_indices]

    def ineq_jac(x):
        h_x = np.clip(x[:n], 0.0, 1.0)
        j = np.zeros((len(active_indices), n + 1))
        j[:, n] = 1.0
        for row, r in enumerate(active_indices):
            q = int(lags[r])
            start = max(0, -q)
            end = min(n, n - q)
            if start < end:
                i_arr = np.arange(start, end)
                p_arr = i_arr + q
                j[row, p_arr] += dx * (1.0 - h_x[i_arr])
                j[row, i_arr] -= dx * h_x[p_arr]
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h, [current_theta]])
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    initial_score = official_score(h)
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 70, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-12:
                return h_out, s
    except Exception:
        pass
    return h, initial_score

def gen_interpolated(parent, n_out, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n_out, 0.5), n_out)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_out) + 0.5 + offset) % n_out) * 2.0 / n_out
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n_out)
