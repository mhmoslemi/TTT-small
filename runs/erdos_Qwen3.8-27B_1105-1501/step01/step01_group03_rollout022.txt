    # Using correlation

    return corr  # length 2n-1, centered at n-1

    """Compute weighted gradient for active shifts.
    k_offset is the index in the correlation array (0 to 2n-2).
    The actual shift is k_offset - (n-1).
    w is the weight for this shift."""
    k = k_offset - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # h[i] * (1 - h[i+k]) for i = 0..n-1-k
        # d/dh[i] = (1 - h[i+k]) for i = 0..n-1-k
        # d/dh[i+k] = -h[i] for i = 0..n-1-k, i.e., for j = k..n-1, d/dh[j] = -h[j-k]
        grad[:n-k] += (1.0 - h[k:n])
        grad[k:] += (-h[:n-k])
    else:

        grad[k_neg:] += (1.0 - h[:n-k_neg])
        grad[:n-k_neg] += (-h[k_neg:])

Contribution: -h[j-k] = -(1 - (1-h[j-k]))... no, just -h[j-k].

from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    lo = -1.0 - np.max(h) - 1e-9
    hi = 1.0 - np.min(h) + 1e-9
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(err) < 1e-14:
                break
            if err > 0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            else:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_for_shift(h, n, k):
    """Gradient of F_k(h) where k is the actual shift (not correlation index)."""
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:k+m])
            g[k:k+m] += (-h[:m])
    else:
        kk = -k
        m = n - kk
        if m > 0:
            g[kk:kk+m] += (1.0 - h[:m])
            g[:m] += (-h[kk:kk+m])
    return g

def soft_max_gradient(h, n, beta):
    """Compute (smoothed max value, gradient) using log-sum-exp."""
    dx = 2.0 / n
    corr = compute_correlation(h)  # length 2n-1
    C = corr * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        obj = cmax + np.log(len(C)) / beta
        weights = np.ones(len(C)) / len(C)
    else:
        obj = cmax + (mk + np.log(ws)) / beta
        weights = es / ws
    grad = np.zeros(n)
    for idx in range(len(C)):
        w = weights[idx]
        if w > 1e-12:
            k = idx - (n - 1)
            grad += w * gradient_for_shift(h, n, k)
    return obj, grad * dx

def projected_gradient_descent(h0, n, beta, max_iter=5000, alpha0=0.01, 
                                 momentum0=0.9, anneal=True, verbose=False):
    """Projected gradient descent with momentum on smooth max objective."""
    h = project_feasible(h0, n)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_obj = official_score(h)
    alpha = alpha0
    momentum = momentum0
    
    for it in range(max_iter):
        obj, grad = soft_max_gradient(h, n, beta)
        
        # Momentum update
        v = momentum * v - alpha * grad
        h_new = project_feasible(h + v, n)
        
        # Check improvement
        new_score = official_score(h_new)
        if new_score < best_obj:
            best_h = h_new.copy()
            best_obj = new_score
        
        # Adaptive step size: if objective increased, reduce alpha
        if new_score > obj * (1 + 1e-6) + 1e-8:
            alpha *= 0.9
            h_new = h  # undo
        else:
            if it % 100 == 0 and alpha < alpha0:
                alpha *= 1.01  # slowly increase back
            
        h = h_new
        
        # Anneal momentum
        if anneal and it % 500 == 499:
            momentum *= 0.95
            v *= 0.5
        
        if verbose and it % 1000 == 0:
            print(f"  iter {it}, score={best_obj:.6f}, obj={obj:.6f}")
    
    return best_h, best_obj
