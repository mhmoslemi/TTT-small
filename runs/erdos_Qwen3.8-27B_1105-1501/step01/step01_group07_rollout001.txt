def fourier_cem(h_init, deadline, K, n_particles, elite_frac, rng):

    x = (np.arange(n) + 0.5) * (2.0 / n)
    dim = 2 * K

    mean = np.zeros(dim)
    std = np.ones(dim) * 0.15
    
    best_h = h_init.copy()
    best_c = score_exact(best_h)
    
    while time.time() < deadline:
        # Generate particles
        particles = rng.normal(mean, std, size=(n_particles, dim))
        scores = np.zeros(n_particles)
        for p in range(n_particles):
            h_p = 0.5 + np.zeros(n)
            for k in range(1, K+1):
                h_p += particles[p, 2*(k-1)] * np.cos(k * np.pi * x)
                h_p += particles[p, 2*(k-1)+1] * np.sin(k * np.pi * x)
            h_p = make_feasible(h_p, n)
            scores[p] = score_exact(h_p)
        
        # Update best
        min_idx = int(np.argmin(scores))
        if scores[min_idx] < best_c:
            h_p = 0.5 + np.zeros(n)
            for k in range(1, K+1):
                h_p += particles[min_idx, 2*(k-1)] * np.cos(k * np.pi * x)
                h_p += particles[min_idx, 2*(k-1)+1] * np.sin(k * np.pi * x)
            h_p = make_feasible(h_p, n)
            best_h = h_p
            best_c = scores[min_idx]
        
        # Elite selection
        n_elite = max(1, int(n_particles * elite_frac))
        elite_idx = np.argsort(scores)[:n_elite]
        mean = particles[elite_idx].mean(axis=0)
        std = np.maximum(particles[elite_idx].std(axis=0), 0.01)
        std *= 0.9  # shrink
    
    return best_h, best_c

score_d = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d]) * dx (for positive lags)

To reduce score_d, we want to either:

- Increase h[i+d] where h[i] = 1

The pair (i, i+d) is a "conflict" if h[i]=1 and h[i+d]=0 or vice versa.

For the golden section search on pair (i, i+d):
- We're looking for the optimal a such that h[i] → h[i]+a and h[i+d] → h[i+d]-a
- This maintains the sum constraint
- The effect on score_d: if h[i] > h[i+d], increasing h[i+d] (a > 0) helps reduce the conflict

1. Find the top K worst lags
