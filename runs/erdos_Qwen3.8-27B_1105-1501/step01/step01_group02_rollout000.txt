    """Chain rule from h-space to manifold variable space."""
    n = 2 * m
    if layout == 'SC':
        return grad_h[:m] - grad_h[m:]
    else:  # RC

    n = 2 * m

    if layout == 'SC':

        h[m:] = 1.0 - a
