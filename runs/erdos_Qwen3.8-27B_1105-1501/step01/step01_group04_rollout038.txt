def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_full(h):

def smooth_max_gradient(h, alpha):
    """Compute gradient of the smooth max (log-sum-exp) of scaled correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    # Shift to prevent overflow
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)

    grad = dx * (T1 - T2)
    return grad, np.max(corr)

def project_box_sum(h, target):

        else:

def active_set_step(h, rho, top_m=10):
    """Find a step direction using active set LP."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top m lags
    top_idx = np.argsort(corr)[-top_m:][::-1]
    max_c = corr[top_idx[0]]
    
    # Compute gradients of the top m constraints
    grads = np.zeros((top_m, n))
    for ii, k in enumerate(top_idx):
        ell = k - (n - 1)
        b = 1.0 - h
        for m in range(n):
            g = 0.0
            im1 = m - ell
            if 0 <= im1 < n:
                g += b[im1]
            im2 = m + ell
            if 0 <= im2 < n:
                g -= h[im2]
            grads[ii, m] = dx * g
    
    # Solve LP: min eta s.t. grads @ delta <= eta, sum(delta)=0, |delta|<=rho
    # Variables: [delta_0, ..., delta_{n-1}, eta]
    # c = [0, ..., 0, 1] (minimize eta)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active constraint i: grads[i] @ delta - eta <= 0
    A_ub = np.zeros((top_m, n + 1))
    A_ub[:top_m, :n] = grads
    A_ub[:top_m, n] = -1.0  # -eta
    b_ub = np.zeros(top_m)
    
    # Bounds: -rho <= delta_i <= rho, eta unbounded (or bounded)
    bounds = [(-rho, rho)] * n + [(None, None)]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.status == 0:
        delta = result.x[:n]
        eta = result.x[n]
        return delta, eta
    return None, None

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = [(-rho, rho)] * n + [(None, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x[n] < -1e-6:
            delta = res.x[:n]
            # Line search: try full step, then halve
            found = False
            for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
                h_try = project_box_sum(h + t * delta, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    stag = 0
                    found = True
                    break
            if not found:
                stag += 1
        else:
            stag += 1
        if stag > 50:
            rng = np.random.RandomState(seed + stag)
            h = best_h + 0.01 * rng.randn(n)
            h = project_box_sum(h, target)
            stag = 0
        rho = max(rho * rho_decay, 0.005)
    return best_h, best_c5

def local_refine(h, time_budget, seed=42):
    """Random coordinate pair perturbation for final polishing."""
    n = len(h)
    target = float(n) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    while time.time() - t0 < time_budget:
        for _ in range(30000):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if best_h[i] <= 1e-10 or best_h[j] >= 1.0 - 1e-10:
                continue
            d = min(best_h[j], 1.0 - best_h[i]) * rng.uniform(0.05, 0.6)
            if rng.random() < 0.5:
                d = -d
            h_test = best_h.copy()
            h_test[i] += d
            h_test[j] -= d
            if h_test[i] < -1e-10 or h_test[i] > 1.0 + 1e-10:
                continue
            if h_test[j] < -1e-10 or h_test[j] > 1.0 + 1e-10:
                continue
            h_test = np.clip(h_test, 0.0, 1.0)
            c5 = compute_c5(h_test)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h_test.copy()
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    """Interpolate from old grid to new grid and re-project."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_interp = np.interp(x_new, x_old, h_old)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return project_box_sum(h_interp, float(n_new) / 2.0)

def random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    idx = rng.permutation(n)
    return h[idx]

def structured_init(n, seed):
    """Create a spread-out binary pattern."""
    rng = np.random.RandomState(seed)
    half = n // 2
    h = np.zeros(n)
    # Spread ones evenly
    positions = np.linspace(0, n, half + 1, dtype=int)[:-1]
    positions = np.sort(rng.permutation(positions))
    for p in positions:
        h[p % n] = 1.0
    if h.sum() > half:
        excess = h.sum() - half
        ones_idx = np.where(h == 1.0)[0]
        for i in range(int(excess)):
            h[ones_idx[i]] = 0.0
    elif h.sum() < half:
        deficit = half - h.sum()
        zeros_idx = np.where(h == 0.0)[0]
        for i in range(int(deficit)):
            h[zeros_idx[i]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial values
    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError, ValueError):
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = project_box_sum(h_init, float(n0) / 2.0)
        c5_init = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
        best_n = n0

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Smoothed gradient at initial dimension
    if h_init is not None and time_left() > 80:
        alloc = min(time_left() * 0.25, 280.0)
        h_best, c5_best = optimize_smoothed(h_init, alloc, seed)
        if c5_best < best_c5:
            best_h = h_best
            best_c5 = c5_best
            best_n = n0

    # Phase 2: Active set LP at initial dimension
    if best_h is not None and time_left() > 60:
        alloc = min(time_left() * 0.20, 180.0)
        h_best, c5_best = optimize_active_set(best_h, alloc, seed + 100)
        if c5_best < best_c5:
            best_h = h_best
            best_c5 = c5_best
            best_n = len(h_best)

    # Phase 3: More smoothed gradient with different params
    if best_h is not None and time_left() > 50:
        alloc = min(time_left() * 0.20, 160.0)
        h_best, c5_best = optimize_smoothed(best_h, alloc, seed + 200)
        if c5_best < best_c5:
            best_h = h_best
            best_c5 = c5_best
            best_n = len(h_best)

    # Phase 4: Try nearby dimensions
    if best_h is not None and time_left() > 40:
        n0 = best_n
        dim_list = []
        for dn in [-3, -2, -1, 1, 2, 3, 4, -5, 5, 6, -4]:
            nn = n0 + dn
            if 32 <= nn <= 200:
                dim_list.append(nn)
        # Deduplicate
        seen = set()
        dim_list = [d for d in dim_list if not (d in seen or seen.add(d))]
        dim_list.sort(key=lambda x: abs(x - n0))
        for n_try in dim_list[:6]:
            if time_left() < 25:
                break
            h_try = interpolate_h(best_h, n_try)
            alloc = min(time_left() * 0.08, 70.0)
            if alloc < 12:
                continue
            h_best, c5_best = optimize_smoothed(h_try, alloc, seed + n_try * 7)
            if c5_best < best_c5:
                best_h = h_best
                best_c5 = c5_best
                best_n = n_try

    # Phase 5: Active set on best if still time
    if best_h is not None and time_left() > 40:
        alloc = min(time_left() * 0.20, 150.0)
        h_best, c5_best = optimize_active_set(best_h, alloc, seed + 300)
        if c5_best < best_c5:
            best_h = h_best
            best_c5 = c5_best
            best_n = len(h_best)

    # Phase 6: Local refine
    if best_h is not None and time_left() > 25:
        alloc = min(time_left() * 0.25, 180.0)
        h_best, c5_best = local_refine(best_h, alloc, seed + 500)
        if c5_best < best_c5:
            best_h = h_best
            best_c5 = c5_best
            best_n = len(h_best)

    # Phase 7: Final smoothed pass
    if best_h is not None and time_left() > 20:
        alloc = min(time_left() * 0.3, 150.0)
        h_best, c5_best = optimize_smoothed(best_h, alloc, seed + 700)
        if c5_best < best_c5:
            best_h = h_best
            best_c5 = c5_best
            best_n = len(h_best)

    # Final feasibility check
    if best_h is not None:
        best_h = project_box_sum(best_h, float(best_n) / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        n = 63
        best_h = project_box_sum(random_init(n, seed), float(n) / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
