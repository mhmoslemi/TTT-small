def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    h = np.asarray(v, dtype=float).copy().ravel()[:n]
    if len(h) < n:
        pad = np.full(n - len(h), 0.5)
        h = np.concatenate([h, pad])
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection for shift
    lo = -(np.max(h) + 1.0)
    hi = (1.0 - np.min(h)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact mass correction
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            if delta > 0:
                h[idx] += delta
            else:
                break
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            if delta > 0:
                h[idx] -= delta
            else:
                break
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, s, n, dx):
    """Gradient of c_s * dx w.r.t. h, where s is the offset (-(n-1) to n-1)."""
    g = np.zeros(n)
    if s >= 0:

        # h[i] appears as second factor (in 1-h) when i-s is valid (i goes s to n-1)
        # Contribution: -h[i-s]

The optimal t tells us: if t < 0, we have a descent direction d. The actual step will be alpha * d where alpha is chosen by line search.

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    h = np.asarray(v, dtype=float).ravel()[:n].copy()
    if len(h) < n:
        h = np.concatenate([h, np.full(n - len(h), 0.5)])
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection for uniform shift
    lo = -(np.max(h) + 0.5)
    hi = (1.0 - np.min(h)) + 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact mass
    for _ in range(500):
        r = target - np.sum(h)
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
    return h

def grad_for_lag(h, s, dx):
    """Gradient of c_s * dx with respect to h."""
    n = len(h)
    if s == 0:
        return dx * (1.0 - 2.0 * h)
    a = abs(s)
    g = np.zeros(n)
    g[:n-a] += (1.0 - h[a:])
    g[a:] += (-h[:n-a])
    return dx * g

def grad_matrix(h, dx):
    """Full gradient matrix, shape (2n-1, n). Row k corresponds to lag s = k-(n-1)."""
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for s in range(-(n-1), n):
        k = s + n - 1
        G[k] = grad_for_lag(h, s, dx)
    return G

def lp_direction(h, G, c_scaled):
    """Find descent direction via LP. Returns (d, t) or (None, t)."""
    n = len(h)
    L = 2 * n - 1
    # Variables: d_0,...,d_{n-1}, t  => n+1 variables
    # Objective: minimize t => c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: G_s * d - t <= 0 for all s
    # i.e., G_s * d + (-1)*t <= 0
    A_ub = np.hstack([G, -np.ones((L, 1))])
    b_ub = np.zeros(L)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -U <= d_i <= U, t free-ish (-2 to 2)
    U = 1.0
    bounds = [(-U, U)] * n + [(-2.0, 2.0)]
    
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                               bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            t = res.x[n]
            # Center d (should already be ~0 sum, but clean up)
            d -= np.mean(d)
            return d, t
        else:
            return None, float('inf')
    except Exception:
        return None, float('inf')

def line_search(h, d, dx, n, max_alpha=2.0):

    best_alpha = 0.0
    best_score = official_score(h)
    alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]

            best_alpha = -a
    return best_alpha, best_score

def discrete_sa(h_init, n, time_limit=30.0, seed=42):
    """Simulated annealing on binary (or near-binary) states."""
    rng = np.random.default_rng(seed)
    dx = 2.0 / n
    target = n / 2.0
    start = time.perf_counter()
    
    # Initialize: threshold the continuous h
    h = np.zeros(n)
    sorted_idx = np.argsort(h_init)[::-1]
    for i in range(int(target)):
        h[sorted_idx[i]] = 1.0
    if target != int(target):
        h[sorted_idx[int(target)]] = target - int(target)
    h = project_to_target(h, n)
    
    best_h = h.copy()
    best_score = official_score(h)
    cur_score = best_score
    
    # Temperature schedule
    T_start = 0.1
    T_end = 0.001
    n_iters = 5000
    
    for it in range(n_iters):
        if time.perf_counter() - start > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / n_iters)
        
        # Random swap: pick one 1 and one 0, swap them
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        new_score = official_score(h)
        delta = new_score - cur_score
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # Revert
    
    return best_h, best_score

def slsqp_polish(h, n, time_limit=15.0):
    """SLSQP minimax polish."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        J = np.zeros((n_lags, n + 1))
        J[:, n] = 1.0
        G = grad_matrix(h_x, dx)
        J[:, :n] = -G
        return J
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score

def interpolate_to_n(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)

import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n):
    h = np.asarray(v, dtype=float).ravel()[:n].copy()
    if len(h) < n:
        h = np.concatenate([h, np.full(n - len(h), 0.5)])
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo = -(np.max(h) + 0.5)
    hi = (1.0 - np.min(h)) + 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(500):
        r = target - np.sum(h)
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
    return np.clip(h, 0.0, 1.0)


def grad_for_lag(h, s, dx):
    n = len(h)
    if s == 0:
        return dx * (1.0 - 2.0 * h)
    a = abs(s)
    g = np.zeros(n)
    g[:n - a] += (1.0 - h[a:])
    g[a:] += (-h[:n - a])
    return dx * g


def grad_matrix(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = s + n - 1
        G[k] = grad_for_lag(h, s, dx)
    return G


def lp_direction(G, c_scaled, top_k=None):
    n = G.shape[1]
    L = G.shape[0]
    if top_k is not None and top_k < L:
        margins = np.max(c_scaled) - c_scaled
        active = np.argsort(margins)[:top_k]
        G_use = G[active]
    else:
        G_use = G
    L_use = len(G_use)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([G_use, -np.ones((L_use, 1))])
    b_ub = np.zeros(L_use)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    U = 1.0
    bounds = [(-U, U)] * n + [(-2.0, 2.0)]
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                               bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n].copy()
            t = float(res.x[n])
            d -= np.mean(d)
            return d, t
    except Exception:
        pass
    return None, float('inf')


def continuous_local_search(h, n, time_limit=40.0, max_iters=80, seed=42):
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    dx = 2.0 / n
    start = time.perf_counter()
    stuck_count = 0

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break

        G = grad_matrix(h, dx)
        c_scaled = np.correlate(h, 1.0 - h, mode="full") * dx

        moved = False
        d, t = lp_direction(G, c_scaled)
        if d is None or t > -1e-7:
            d, t = lp_direction(G, c_scaled, top_k=12)
        if d is None or t > -1e-8:
            d, t = lp_direction(G, c_scaled, top_k=5)

        if d is not None and t < -1e-8:
            dn = np.max(np.abs(d))
            if dn > 1e-12:
                d = d / dn
                for scale in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                    h_try = project_to_target(h + scale * d, n)
                    s_try = official_score(h_try)
                    if s_try < score - 1e-13:
                        score = s_try
                        h = h_try
                        moved = True
                        break
                if not moved:
                    for scale in [0.5, 0.3, 0.1, 0.05]:
                        h_try = project_to_target(h - scale * d, n)
                        s_try = official_score(h_try)
                        if s_try < score - 1e-13:
                            score = s_try
                            h = h_try
                            moved = True
                            break

        if not moved:
            r = rng.normal(0, 1, n)
            r -= np.mean(r)
            rn = np.max(np.abs(r))
            if rn > 1e-12:
                r /= rn
            for scale in [0.05, 0.1, 0.15, 0.2, 0.3]:
                h_try = project_to_target(h + scale * r, n)
                s_try = official_score(h_try)
                if s_try < score - 1e-13:
                    score = s_try
                    h = h_try
                    moved = True
                    break

        if not moved:
            improved = False
            for _ in range(80):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for delta in [0.03, 0.05, 0.08, 0.12, 0.18]:
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    h_new = project_to_target(h_try, n)
                    s_try = official_score(h_new)
                    if s_try < score - 1e-13:
                        h = h_new
                        score = s_try
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                stuck_count += 1
                if stuck_count > 8:
                    break
        else:
            stuck_count = 0

    return h, official_score(h)


def discrete_sa(h_init, n, time_limit=40.0, seed=42):
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    h = np.zeros(n)
    sorted_idx = np.argsort(h_init)[::-1]
    n_ones = n // 2
    for i in range(n_ones):
        h[sorted_idx[i]] = 1.0
    h = project_to_target(h, n)

    best_h = h.copy()
    best_score = official_score(h)
    cur_score = best_score

    T_start = 0.08
    T_end = 0.001
    max_iters = 20000

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)

        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_score = official_score(h)
        delta = new_score - cur_score

        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    return best_h, best_score


def slsqp_polish(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        J = np.zeros((n_lags, n + 1))
        J[:, n] = 1.0
        J[:, :n] = -grad_matrix(h_x, dx)
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def interpolate_to_n(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)
