import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_max_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    T1 = np.convolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2

def active_set_gradient(h, epsilon):
    """Gradient of sum of active correlations (those within epsilon of max)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    active = corr > max_c - epsilon
    w = np.zeros(len(corr))
    w[active] = 1.0
    # Weight by how close to max
    w[active] = (corr[active] - max_c + epsilon) / epsilon
    T1 = np.convolve(1.0 - h, w, mode='full')[n-1:2*n-1]
    T2 = np.correlate(w, h, mode='valid')[::-1]
    return T1 - T2

def optimize_primary(h0, time_budget, seed=42):
    """Main optimization using smoothed max with adaptive parameters"""
    n = len(h0)
    target = float(n) / 2.0
    h = project_to_sum(np.clip(h0.copy(), 0.0, 1.0), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

    lr = 0.01
    v = np.zeros(n)

            v_new = 0.9 * v - lr * grad
            h_nesterov = h + prev_v + v_new  # look-ahead
            h_nesterov = project_to_sum(h_nesterov, target)
            grad = smoothed_max_gradient(h_nesterov, alpha_ls)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_sum(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smoothed_max_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    s = np.sum(exp_c)
    p = exp_c / s
    T1 = np.convolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2


def smooth_penalty_gradient(h, tau):
    """Gradient of sum(max(0, dx*corr[k] - tau)^2)"""
    n = len(h)

    viol = np.maximum(0.0, corr_vals - tau)

        return np.zeros(n)
    w = 2.0 * viol
    T1 = np.convolve(1.0 - h, w, mode='full')[n-1:2*n-1]
    T2 = np.correlate(w, h, mode='valid')[::-1]
    return T1 - T2
