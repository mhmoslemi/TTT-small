**Multi-active-set projected subgradient:**

2. Find the top m active lags (m = 3-7, adaptive).
3. Compute the direction d that minimizes max_{k∈A} ∇F_k · d (the "Chebyshev direction").

import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).ravel().copy()
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(y, 0.0, 1.0)
    target = n / 2.0
    smin = float(-y.min())
    smax = float(1.0 - y.max())
    for _ in range(60):
        mid = (smin + smax) / 2.0
        z = np.clip(y + mid, 0.0, 1.0)
        if z.sum() > target:
            smax = mid
        else:
            smin = mid
    z = np.clip(y + (smin + smax) / 2.0, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(z)
            remain = diff
            for i in order:
                if remain <= 1e-14:
                    break
                add = min(remain, 1.0 - z[i])
                z[i] += add
                remain -= add
        else:
            order = np.argsort(-z)
            remain = -diff
            for i in order:
                if remain <= 1e-14:
                    break
                sub = min(remain, z[i])
                z[i] -= sub
                remain -= sub
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_F_k(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:k+m])
            g[k:k+m] -= h[:m]
    else:
        p = -k
        m = n - p
        if m > 0:
            g[p:p+m] += (1.0 - h[:m])
            g[:m] -= h[p:p+m]
    return g

def epigraph_solve(h0, deadline, lam=0.0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    nvars = n + 1
    ncon = 2 * n - 1
    lags = list(range(-(n - 1), n))

    def ep_obj(z):
        if lam > 0:
            return float(z[-1] + lam * np.sum((z[:-1] - h0) ** 2))
        return float(z[-1])

    def ep_obj_grad(z):
        g = np.zeros(nvars)
        g[-1] = 1.0
        if lam > 0:
            g[:-1] = 2.0 * lam * (z[:-1] - h0)
        return g

    def ep_ineq(z):
        h = z[:-1]
        t = z[-1]
        Fv = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - Fv

    def ep_ineq_jac(z):
        h = z[:-1]
        J = np.zeros((ncon, nvars))
        for idx, k in enumerate(lags):
            J[idx, :-1] = -grad_F_k(h, k, n) * dx
            J[idx, -1] = 1.0
        return J

    def ep_eq(z):
        return float(np.sum(z[:-1]) - n / 2.0)

    def ep_eq_jac(z):
        g = np.zeros(nvars)
        g[:-1] = 1.0
        return g

    z0 = np.append(h0.copy(), score_exact(h0) + 0.005)
    bounds = [(0.0, 1.0)] * n + [(score_exact(h0) * 0.5, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ep_ineq, 'jac': ep_ineq_jac},
        {'type': 'eq', 'fun': ep_eq, 'jac': ep_eq_jac}
    ]
    try:
        result = minimize(ep_obj, z0, jac=ep_obj_grad, bounds=bounds,
                          constraints=cons, method='SLSQP',
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        h_sol = make_feasible(result.x[:-1], n)
        c_sol = score_exact(h_sol)
        if c_sol < score_exact(h0):
            return h_sol, c_sol
    except Exception:
        pass
    return h0.copy(), score_exact(h0)

def subgradient_minimax(h0, deadline, T0=0.05, T_min=0.001, T_decay=0.999,
                        step0=0.02, m_active=5, rng=None):
    n = len(h0)
    dx = 2.0 / n
    cur = h0.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T0
    step = step0
    stall = 0
    if rng is None:
        rng = np.random.default_rng(0)

    while time.time() < deadline:
        Fv = np.correlate(cur, 1.0 - h = 1.0, mode='full') * dx
        Fv = np.correlate(cur, 1.0 - cur, mode='full') * dx
        Fmax = float(np.max(Fv))
        # Select top m_active lags
        top_idx = np.argsort(Fv)[-m_active:][::-1]
        # Compute direction: negative mean gradient of active lags
        g_dir = np.zeros(n)
        for idx in top_idx:
            k = int(idx - (n - 1))
            g_dir += grad_F_k(cur, k, n)
        g_dir *= dx
        gn = np.linalg.norm(g_dir)
        if gn < 1e-15:
            T = max(T_min, T * 0.5)
            if T < T_min * 0.5:
                break
            continue
        direction = -g_dir / gn
        improved = False
        for frac in (1.0, 0.5, 0.25, 0.125, 0.05):
            s = step * frac
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 50:
                # Perturb to escape
                cur = make_feasible(cur + rng.normal(0, max(0.003, T * 0.5), n), n)
                cur_c = score_exact(cur)
                step = max(0.005, T)
                stall = 0
        else:
            step = min(step * 1.03, 0.1)
            stall = 0
        T = max(T_min, T * T_decay)

    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=35):
    n = len(h)
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                h[i] = orig_i
                h[j] = orig_j
                return None
    return float(best_s) if best_s < cur_c - 1e-14 else None

def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10 or n_new % 2 != 0:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def generate_starts(h_parent, n, rng):
    """Generate diverse starting points."""
    starts = [h_parent.copy() if len(h_parent) == n else make_feasible(h_parent, n)]
    # Sorted (ascending)
    s = np.sort(starts[0])
    starts.append(make_feasible(s, n))
    # Sorted (descending)
    s = np.sort(starts[0])[::-1]
    starts.append(make_feasible(s, n))
    # Reversed
    starts.append(make_feasible(starts[0][::-1].copy(), n))
    # Cyclic shift by n//4
    sh = n // 4
    starts.append(make_feasible(np.roll(starts[0], sh).copy(), n))
    # Binary quantization
    b = (starts[0] > 0.5).astype(float)
    b = make_feasible(b, n)
    starts.append(b)
    # Anti-binary (swap 0s and 1s)
    b2 = 1.0 - b
    starts.append(make_feasible(b2, n))
    return starts
