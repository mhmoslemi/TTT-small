def dca_lp_step(h0, max_iters=200, step_sizes=None):
    """One DCA iteration using LP linearization."""
    n = len(h0)
    dx = 2.0 / n
    corr = np.correlate(h0, 1-h0, mode='full') * dx
    current_max = np.max(corr)
    
    # Compute gradients
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        shift = k - (n-1)
        for i in range(n):
            g = 0.0
            j = i - shift
            if 0 <= j < n:
                g += (1 - h0[j])
            j2 = i + shift
            if 0 <= j2 < n:
                g -= h0[j2]
            grads[k, i] = g * dx
    
    # LP: minimize t
    # s.t. grads[k,:] @ h - t <= -corr[k] for all k  (L_k(h) = corr[k] + grads[k,:]@(h-h0) <= t)
    #      0 <= h[i] <= 1
    #      sum(h) = n/2
    #
    # L_k(h) = corr[k] + grads[k,:] @ (h - h0) <= t
    #        = corr[k] - grads[k,:] @ h0 + grads[k,:] @ h <= t
    # const_k = corr[k] - grads[k,:] @ h0
    const = corr - grads @ h0
    
    # Variables: x = [h(0..n-1), t], length n+1
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # Row from correlation: grads[k,:] @ h - t <= -const[k]
    A_ub_corr = np.hstack([grads, -np.ones((2*n-1, 1))])
    b_ub_corr = -const
    
    # h[i] <= 1
    A_ub_h1 = np.hstack([np.eye(n), np.zeros((n,1))])
    b_ub_h1 = np.ones(n)
    
    # -h[i] <= 0
    A_ub_h0 = np.hstack([-np.eye(n), np.zeros((n,1))])
    b_ub_h0 = np.zeros(n)
    
    A_ub = np.vstack([A_ub_corr, A_ub_h1, A_ub_h0])
    b_ub = np.concatenate([b_ub_corr, b_ub_h1, b_ub_h0])
    
    # Equality: sum(h) = n/2
    A_eq = np.hstack([np.ones(n), np.zeros(1)])
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]  # h in [0,1], t >= 0
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        h_new = res.x[:n]
        return h_new
    return None

def make_feasible(z, target=None):

    z = np.asarray(z, dtype=float).ravel().copy()

    """Compute gradients of all correlation lags with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))

        for i in range(n):
            g = 0.0
            j = i - shift
            if 0 <= j < n:
                g += (1.0 - h[j])
            j2 = i + shift
            if 0 <= j2 < n:
                g -= h[j2]
            grads[k, i] = g * dx
    return grads

def dca_lp_step(h0):
    """One DCA iteration: solve LP with linearized correlations."""
    n = len(h0)
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full') * dx
    grads = compute_grads(h0)
    
    # Linearization: L_k(h) = corr[k] + grads[k,:] @ (h - h0) = (corr[k] - grads[k,:]@h0) + grads[k,:]@h
    const = corr - grads @ h0
    
    # LP: min t s.t. grads[k,:]@h + const[k] <= t, 0<=h<=1, sum(h)=n/2
    # Variables: x = [h, t], size n+1
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequalities: grads[k,:]@h - t <= -const[k]
    A_ub = np.zeros((2*n-1+n+n, n+1))
    b_ub = np.zeros(2*n-1+n+n)
    
    # Correlation constraints
    A_ub[:2*n-1, :n] = grads
    A_ub[:2*n-1, -1] = -1.0
    b_ub[:2*n-1] = -const
    
    # h[i] <= 1
    idx = 2*n-1
    A_ub[idx:idx+n, :n] = np.eye(n)
    b_ub[idx:idx+n] = 1.0
    
    # -h[i] <= 0
    idx2 = 2*n-1+n
    A_ub[idx2:idx2+n, :n] = -np.eye(n)
    b_ub[idx2:idx2+n] = 0.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:

        return make_feasible(h_new)
    return None
