    target = n / 2.0

    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5

        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()

    for _ in range(200):
        diff = target - np.sum(y)
        if abs(diff) < 1e-13:

        if diff > 0:
            mask = y < 1.0
            if not np.any(mask):
                break
            y[mask] = np.minimum(y[mask] + diff / mask.sum(), 1.0)
        else:

            if not np.any(mask):
                break
            y[mask] = np.maximum(y[mask] + diff / mask.sum(), 0.0)

    diff = target - np.sum(y)
    if abs(diff) > 1e-14:

        y[idx] = float(np.clip(y[idx] + diff, 0.0, 1.0))

    result[n - 1] = np.dot(h, 1.0 - h)

        result[n - 1 + k] = np.dot(h[k:], 1.0 - h[:m])
        result[n - 1 - k] = np.dot(h[:m], 1.0 - h[k:])

        return t + lam * np.dot(h - h_ref, h - h_ref)
