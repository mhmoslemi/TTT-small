def lp_step(h, active_indices, max_iter=50):
    """
    active_indices: indices into the full correlation array (c = n-1 is center)
    Returns a new h after LP step and line search.
    """
    n = len(h)
    scores = all_scores(h)
    c = n - 1
    
    # Get active lags (offsets from center)
    lags = active_indices - c
    
    # Compute gradients
    grads = compute_grad_active(h, lags)  # shape (k, n)
    
    # Current values of active scores
    s_vals = scores[active_indices]
    
    # LP: min t
    # s.t. grads @ u <= t - s_vals  (for each active lag)
    #      -h[j] <= u[j] <= 1-h[j]
    #      sum(u) = 0
    
    k = len(active_indices)
    # Variables: u[0], ..., u[n-1], t → total n+1 variables
    # Objective: minimize t → c_obj = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag i: grads[i] @ u - t <= -s_vals[i]
    # i.e., [grads[i], -1] @ [u, t] <= -s_vals[i]
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -s_vals
    
    # Bound constraints on u: -h[j] <= u[j] <= 1-h[j]
    # t is unbounded (but will be bounded by other constraints)
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)] + [(None, None)]
    
    # Equality constraint: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return h, score_exact(h)
    
    u_star = res.x[:n]
    
    # Line search
    best_h = h.copy()
    best_c = score_exact(h)
    for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
        cand = h + alpha * u_star
        cand = make_feasible(cand, n)
        c_new = score_exact(cand)
        if c_new < best_c:
            best_h = cand
            best_c = c_new
    
    return best_h, best_c

import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(z)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(z)[::-1]
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def all_gradients(h):

    c = n - 1
    G = np.zeros((2 * n - 1, n))

        G[c + d, d:] += (1.0 - h[:m]) * dx
        G[c - d, :m] += (1.0 - h[d:]) * dx

    n = len(h)
    scores = all_scores(h)
    G = all_gradients(h)
    k = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)] + [(None, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, float(np.max(scores))
    if res.status != 0:
        return h, float(np.max(scores))
    u_star = res.x[:n]
    if np.max(np.abs(u_star)) < 1e-12:
        return h, float(np.max(scores))
    best_h = h.copy()
    best_c = float(np.max(scores))

                no_improve = 0
                lp_count = 0
            else:
                # Local search burst
                for _ in range(30):
                    if time.time() > p1_end:
                        break
                    mode = rng.random()
                    if mode < 0.6:
                        i = int(rng.integers(0, n))
                        j = int(rng.integers(0, n))
                        if i == j:
                            continue
                        res = pair_search(cur, i, j, cur_c, n_pts=25)
                    elif mode < 0.8:
                        i = int(rng.integers(0, n))
                        res = coord_search(cur, i, cur_c, n_pts=20)
                    else:
                        i = int(rng.integers(0, n - 1))
                        res = pair_search(cur, i, i + 1, cur_c, n_pts=30)
                    if res is not None:
                        cur_c = res
                        no_improve = 0
                        if cur_c < best_c:
                            best_h, best_c = cur.copy(), cur_c
                        break
                else:
                    no_improve += 1

    # Phase 2: Systematic local search (25% of time)
    p2_end = t0 + budget_s * 0.80
    cur = best_h.copy()
    cur_c = best_c
    stall = 0
    while time.time() < p2_end:
        improved = False
        # Sweep adjacent pairs
        for i in range(n - 1):
            if time.time() > p2_end:
                break
            res = pair_search(cur, i, i + 1, cur_c, n_pts=35)
            if res is not None:
                cur_c = res
                improved = True
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c
        if not improved:
            # Random pairs
            for _ in range(40):
                if time.time() > p2_end:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                res = pair_search(cur, i, j, cur_c, n_pts=30)
                if res is not None:
                    cur_c = res
                    improved = True
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                    break
        if not improved:
            stall += 1
            if stall > 10:
                # Try LP again
                new_h, new_c = lp_step(cur)
                if new_c < cur_c - 1e-12:
                    cur = new_h
                    cur_c = new_c
                    stall = 0
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    # Perturb and restart
                    cur = make_feasible(best_h + rng.normal(0, 0.015, n), n)
                    cur_c = score_exact(cur)
                    stall = 0
        else:
            stall = 0

    # Phase 3: Try alternative initializations (10% of time)
    p3_end = t0 + budget_s * 0.92
    if time.time() < p3_end:
        # Try a few binary and smooth patterns
        attempts = 0
        while time.time() < p3_end and attempts < 6:
            attempts += 1
            if attempts == 1:
                cand = make_feasible(binary_init(n, rng), n)
            elif attempts == 2:
                cand = make_feasible(smooth_pattern(n, 1.0, rng.random() * 2), n)
            elif attempts == 3:
                cand = make_feasible(smooth_pattern(n, 0.5, rng.random() * 4), n)
            elif attempts == 4:
                cand = make_feasible(best_h + rng.normal(0, 0.05, n), n)
            elif attempts == 5:
                # Antipodal: flip best_h
                cand = make_feasible(1.0 - best_h, n)
            else:
                x = np.linspace(0, 2, n)
                freq = int(rng.integers(3, 8))
                cand_h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / 2.0 + rng.random() * 2)
                cand = make_feasible(cand_h, n)

            c_cand = score_exact(cand)
            if c_cand < best_c:
                # This initialization is already better!
                best_h, best_c = cand, c_cand
            # Do a few LP steps from this start
            sub_dl = min(p3_end - 2, time.time() + 60)
            for _ in range(10):
                if time.time() > sub_dl:
                    break
                new_h, new_c = lp_step(cand)
                if new_c < c_cand - 1e-12:
                    cand = new_h
                    c_cand = new_c
                    if c_cand < best_c:
                        best_h, best_c = cand.copy(), c_cand
                        break

    # Phase 4: Final polish (8% of time)
    p4_end = deadline
    if time.time() < p4_end:
        cur = best_h.copy()
        cur_c = best_c
        # Alternating LP and local search
        for iteration in range(50):
            if time.time() > p4_end - 1:
                break
            new_h, new_c = lp_step(cur)
            if new_c < cur_c - 1e-12:
                cur = new_h
                cur_c = new_c
            else:
                for _ in range(20):
                    if time.time() > p4_end - 1:
                        break
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
                    res = pair_search(cur, i, j, cur_c, n_pts=30)
                    if res is not None:
                        cur_c = res
                        break
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c

    # Final: try subgradient steps with the best solution
    if time.time() < deadline - 1:
        cur = best_h.copy()
        cur_c = best_c
        step_size = 0.02
        for _ in range(50):
            if time.time() > deadline - 1:
                break
            scores = all_scores(cur)
            G = all_gradients(cur)
            new_h, new_c = subgrad_step(cur, scores, G, step_size)
            if new_c < cur_c - 1e-12:
                cur = new_h
                cur_c = new_c
                step_size = min(step_size * 1.1, 0.05)
            else:
                step_size *= 0.8
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c

    # Ensure final feasibility
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
