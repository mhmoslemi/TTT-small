def project_feasible(v, n):
    h = np.asarray(v, dtype=float).ravel().copy()
    if len(h) < n:
        h = np.concatenate([h, np.full(n - len(h), 0.5)])
    h = h[:n]
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        r = target - h.sum()
        if abs(r) < 1e-15:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + r)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + r)
    h = np.clip(h, 0.0, 1.0)
    r = target - h.sum()
    if abs(r) > 1e-14:
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += r
        else:
            idx = np.argmax(h)
            h[idx] += r
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_lag(h, q):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    p = np.arange(n)
    m1 = (p - q) >= 0
    m1 &= (p - q) < n
    g[m1] += (1.0 - h[p[m1] - q])
    m2 = (p + q) >= 0
    m2 &= (p + q) < n
    g[m2] += (-h[p[m2] + q])
    return dx * g


def pair_minimize(h, i, j, n, grid_res, golden_iters):

    L = max(0.0, S - 1.0)

def active_weights_from_corr(corr_vals):
    theta = np.max(corr_vals)
    T = max(0.003 + 0.01 * theta, 0.002)
    w = np.exp((corr_vals - theta) / T)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones_like(corr_vals) / len(corr_vals)
    else:
        w = w / wsum
    return w


def composite_grad(h, n, lags, weights):
    G = np.zeros(n)
    p = np.arange(n)
    for idx in range(len(lags)):
        w = weights[idx]
        if w < 1e-12:
            continue
        q = lags[idx]
        m1 = (p - q) >= 0
        m1 &= (p - q) < n
        G[m1] += w * (1.0 - h[p[m1] - q])
        m2 = (p + q) >= 0
        m2 &= (p + q) < n
        G[m2] += w * (-h[p[m2] + q])
    return G

    score = official_score(h)
    best_h = h.copy()
    best_score = score
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    corr = corr_scaled(best_h)
    weights = active_weights_from_corr(corr)
    G = dx * composite_grad(best_h, n, lags, weights)

    pairs_set = set()
    for i in range(n - 1):
        pairs_set.add((i, i + 1))
    for dist in [2, 3, 4]:
        for i in range(n - dist):
            pairs_set.add((i, i + dist))
    for _ in range(num_pairs):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j:
            pairs_set.add((min(i, j), max(i, j)))

    pairs_list = list(pairs_set)
    pair_scores = []
    for (i, j) in pairs_list:
        diff = G[j] - G[i]
        pair_scores.append((diff, i, j))
    pair_scores.sort(key=lambda x: -x[0])

    improved = False
    for (diff, i, j) in pair_scores[:num_pairs]:

            corr = corr_scaled(best_h)

            G = dx * composite_grad(best_h, n, lags, weights)

def slsqp_polish(h, n, time_limit=30.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    initial_score = official_score(h)
    corr = corr_scaled(h)
    top_k = min(12, len(corr))
    top_idx = np.argsort(corr)[::-1][:top_k]
    top_lags = lags[top_idx]

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        th = x[n]
        c = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
        return th - c[top_idx]

    def ineq_jac(x):
        h_x = x[:n]
        p = np.arange(n)
        j = np.zeros((len(top_idx), n + 1))
        j[:, n] = 1.0
        for ii, ki in enumerate(top_idx):
            q = lags[ki]
            m1 = (p - q) >= 0
            m1 &= (p - q) < n
            j[ii, p[m1]] += -dx * (1.0 - h_x[p[m1] - q])
            m2 = (p + q) >= 0
            m2 &= (p + q) < n
            j[ii, p[m2]] += -dx * (-h_x[p[m2] + q])
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h, [initial_score + 1e-4]])
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            obj, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12}

        if result.x is not None and np.all(np.isfinite(result.x[:n])):

            if s < initial_score - 1e-12:

    return h, initial_score


def interpolate_to(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_feasible(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_feasible(h, n_target)

    scored.sort(key=lambda x: (x[0], len(x[1])))

    if len(scored) > 0 and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()

    n_top = min(12, len(scored))
    local_results = []
    for idx in range(n_top):
        if time.perf_counter() > deadline - 120:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_alloc = min(80.0, max(10.0, remaining / max(1, n_top - idx) * 0.8))
        t_budget = time.perf_counter() + t_alloc
        h_loc, s_loc = run_pair_engine(h_c, n_c, rng, t_budget, num_rounds=8)
        local_results.append((s_loc, h_loc))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()

    local_results.sort(key=lambda x: x[0])

    to_refine = local_results[:min(6, len(local_results))]
    found_best = False
    for s_r, h_r in to_refine:
        if np.array_equal(h_r, best_h):
            found_best = True
            break
    if not found_best and best_score < 1e8:
        to_refine.append((best_score, best_h.copy()))

    for idx, (s_c, h_c) in enumerate(to_refine):
        if time.perf_counter() > deadline - 50:
            break
        n_c = len(h_c)
        h_ref, s_ref = slsqp_polish(h_c, n_c, time_limit=20.0)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
        if time.perf_counter() > deadline - 35:
            break
        h_2, s_2 = run_pair_engine(h_ref, n_c, rng,
                                  time.perf_counter() + min(15.0, deadline - time.perf_counter() - 10),
                                  num_rounds=5)
        if s_2 < best_score:
            best_score = s_2
            best_h = h_2.copy()

    if time.perf_counter() < deadline - 20:
        for n_try in [len(best_h), len(best_h) + 8, len(best_h) - 5]:
            if time.perf_counter() > deadline - 15:
                break
            if n_try < 20:
                continue
            h_interp = interpolate_to(best_h, n_try)
            h_loc, s_loc = run_pair_engine(h_interp, n_try, rng,
                                          time.perf_counter() + min(15.0, deadline - time.perf_counter() - 8),
                                          num_rounds=6)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()

    if time.perf_counter() < deadline - 10:
        h_fin, s_fin = slsqp_polish(best_h, len(best_h), time_limit=8.0)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    if time.perf_counter() < deadline - 5:
        h_fin2, s_fin2 = run_pair_engine(best_h, len(best_h), rng,
                                         time.perf_counter() + min(5.0, deadline - time.perf_counter() - 2),
                                         num_rounds=3)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2.copy()

    best_h = project_feasible(best_h, len(best_h))
    final_score = official_score(best_h)
    c5_bound = float(final_score)
    if c5_bound < 0.0:
        c5_bound = 0.0

    return (best_h, c5_bound, int(len(best_h)))
