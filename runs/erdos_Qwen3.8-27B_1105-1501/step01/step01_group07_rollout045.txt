import time
import cvxpy as cp

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def lp_trust_region(h0, active_d, trust_eps):
    """Solve LP to minimize max linear approximation of active lags,
    with trust region |h - h0| <= trust_eps."""
    n = len(h0)
    dx = 2.0 / n
    c = n - 1
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h0 - trust_eps <= h_var,
        h_var <= h0 + trust_eps,
        cp.sum(h_var) == n / 2.0
    ]
    
    for d in active_d:
        k = d - c
        f_k = 0.0
        grad = np.zeros(n)
        # Compute f_k and gradient
        if k >= 0:
            for i in range(n - k):
                f_k += h0[i] * (1.0 - h0[i + k])
                grad[i] += dx * (1.0 - h0[i + k])
                grad[i + k] -= dx * h0[i]
        else:
            kk = -k
            for i in range(kk, n):
                f_k += h0[i] * (1.0 - h0[i - kk])
                grad[i] += dx * (1.0 - h0[i - kk])
                grad[i - kk] -= dx * h0[i]
        f_k *= dx
        constraints.append(f_k + grad @ (h_var - h0) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=20000, abs_acc=1e-12, rel_acc=1e-12)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None
    
    if h_var.value is None or t_var.value is None:
        return None
    
    return np.array(h_var.value).flatten()

def pair_search(h, i, j, cur_c, n_pts=40):
    n = len(h)
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi_val = min(1.0 - orig_i, orig_j)
    if hi_val - lo < 1e-14:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi_val, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-12 and best_s < cur_c - 1e-14:
        # Golden section refinement
        span = (hi_val - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi_val, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-12:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=30):
    n = len(h)
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-12:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 20:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def all_lags(h):
    n = len(h)

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))

        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))

def lp_step(h0, active_d, trust_eps, solver='ECOS'):
    n = len(h0)
    dx = 2.0 / n
    c = n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        h0 - trust_eps <= h_var,
        h_var <= h0 + trust_eps,
        cp.sum(h_var) == n / 2.0
    ]
    for d in active_d:
        k = d - c
        f_k = 0.0
        grad = np.zeros(n)
        if k >= 0:
            idx = np.arange(n - k)
            f_k = float(np.sum(h0[idx] * (1.0 - h0[idx + k]))) * dx
            grad[idx] += dx * (1.0 - h0[idx + k])
            grad[idx + k] -= dx * h0[idx]
        else:
            kk = -k
            idx = np.arange(kk, n)
            f_k = float(np.sum(h0[idx] * (1.0 - h0[idx - kk]))) * dx
            grad[idx] += dx * (1.0 - h0[idx - kk])
            grad[idx - kk] -= dx * h0[idx]
        constraints.append(f_k + grad @ (h_var - h0) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        if solver == 'ECOS':
            prob.solve(solver=cp.ECOS, max_iter=30000, abs_acc=1e-12, rel_acc=1e-12)
        else:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10)
        except Exception:
            return None
    if h_var.value is None or t_var.value is None:
        return None
    return np.array(h_var.value).flatten()

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

            if c_r < best_c:
                best_h, best_c = h_r, c_r
                n = len(best_h)

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
