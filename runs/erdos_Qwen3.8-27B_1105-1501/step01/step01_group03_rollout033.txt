import numpy as np
import time


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        idx = np.argsort(h)
        for ii in idx:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[ii] < 1.0:
                d = min(err, 1.0 - h[ii])
                h[ii] += d
                err -= d
            elif err < 0 and h[ii] > 0.0:
                d = min(-err, h[ii])
                h[ii] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def line_search_1d(F, L, Q, s_lo, s_hi, grid_n=80):
    if s_hi - s_lo < 1e-12:
        return 0.0, F.max()
    g0 = F.max()
    s_grid = np.linspace(s_lo, s_hi, grid_n)
    sv = s_grid[:, None]
    vals = F[None, :] + L[None, :] * sv + Q[None, :] * (sv * sv)
    g_vals = vals.max(axis=1)
    best_idx = np.argmin(g_vals)
    best_g = g_vals[best_idx]
    if best_g >= g0 - 1e-13:
        return 0.0, g0
    lo = s_grid[max(0, best_idx - 1)]
    hi = s_grid[min(grid_n - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(F + L * c + Q * c * c)
    fd = np.max(F + L * d + Q * d * d)
    for _ in range(50):
        if b - a < 1e-11:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = np.max(F + L * c + Q * c * c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = np.max(F + L * d + Q * d * d)
    best_s = (a + b) * 0.5
    best_g = np.max(F + L * best_s + Q * best_s * best_s)
    if best_g >= g0 - 1e-12:
        return 0.0, g0
    best_s = max(s_lo, min(s_hi, best_s))
    return best_s, best_g


def compute_pair_LQ(h, i, j):
    n = len(h)
    nL = 2 * n - 1
    L = np.zeros(nL)
    Q = np.zeros(nL)
    hi_val = h[i]
    hj_val = h[j]
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        idx = k + n - 1
        lk = 0.0
        qk = 0.0
        if start <= i < end:
            ik = i + k
            if 0 <= ik < n:
                vi = 1.0 if ik == i else (-1.0 if ik == j else 0.0)
                lk += (1.0 - h[ik]) - hi_val * vi
                qk -= vi
        if start <= j < end:
            jk = j + k
            if 0 <= jk < n:
                vj = 1.0 if jk == i else (-1.0 if jk == j else 0.0)
                lk += -(1.0 - h[jk]) - hj_val * vj
                qk += vj
        L[idx] = lk
        Q[idx] = qk
    return L, Q


def compute_dir_LQ(h, v):
    n = len(h)
    nL = 2 * n - 1
    L = np.zeros(nL)
    Q = np.zeros(nL)
    nz = np.nonzero(np.abs(v) > 1e-15)[0]
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        idx = k + n - 1
        lk = 0.0
        qk = 0.0
        for t in nz:
            if start <= t < end:
                tk = t + k
                if 0 <= tk < n:
                    lk += v[t] * (1.0 - h[tk]) - h[t] * v[tk]
                    qk -= v[t] * v[tk]
        L[idx] = lk
        Q[idx] = qk
    return L, Q


def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def pair_pass(h, F, deadline):
    n = len(h)
    improved = False
    for i in range(n):
        if time.time() > deadline:
            break
        for j in range(i + 1, n):
            if time.time() > deadline:
                break
            hi_val = h[i]
            hj_val = h[j]
            s_lo = max(-hi_val, hj_val - 1.0)
            s_hi = min(1.0 - hi_val, hj_val)
            if s_hi - s_lo < 1e-10:
                continue
            L, Q = compute_pair_LQ(h, i, j)
            best_s, best_g = line_search_1d(F, L, Q, s_lo, s_hi)
            if best_s != 0.0 and abs(best_s) > 1e-12:
                h_new = h.copy()
                h_new[i] = np.clip(hi_val + best_s, 0.0, 1.0)
                h_new[j] = np.clip(hj_val - best_s, 0.0, 1.0)
                sc_orig = F.max()
                sc_new = best_g
                if sc_new < sc_orig - 1e-11:
                    h = h_new
                    F = np.correlate(h, 1.0 - h, mode='full')
                    improved = True
    return h, F, improved


def multi_coord_pass(h, F, rng, n_dirs, deadline):
    n = len(h)
    improved = False
    for _ in range(n_dirs):
        if time.time() > deadline:
            break
        r = rng.integers(2, 6)
        idx = rng.choice(n, size=r, replace=False)
        weights = rng.standard_normal(r)
        weights -= weights.mean()
        norm = np.sum(weights * weights)
        if norm < 1e-10:
            continue
        weights /= np.sqrt(norm)
        v = np.zeros(n)
        for ii, w in zip(idx, weights):
            v[ii] = w
        s_lo = -np.inf
        s_hi = np.inf
        for ii in range(n):
            if v[ii] > 1e-12:
                s_lo = max(s_lo, -h[ii] / v[ii])
                s_hi = min(s_hi, (1.0 - h[ii]) / v[ii])
            elif v[ii] < -1e-12:
                s_lo = max(s_lo, (1.0 - h[ii]) / v[ii])
                s_hi = min(s_hi, -h[ii] / v[ii])
        if s_hi - s_lo < 1e-10:
            continue
        L, Q = compute_dir_LQ(h, v)
        best_s, best_g = line_search_1d(F, L, Q, s_lo, s_hi)
        if best_s != 0.0 and abs(best_s) > 1e-12:
            h_new = h + best_s * v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = make_feasible(h_new)
            sc_new = F.max()
            if best_g < sc_new - 1e-11:
                h = h_new
                F = np.correlate(h, 1.0 - h, mode='full')
                improved = True
    return h, F, improved
