Finding the minimum of a convex piecewise linear function with 139 pieces can be done in O(n) using a linear-time algorithm for the upper envelope.

But in practice, for Python, even O(n^2) = O(10000) per pair is fine if we're doing it for a subset of pairs.

My new approach should focus on:

2. **Systematic sweep**: Go through all pairs, applying improvements

Let me also think about the structure of the problem. The objective is:
C5 = max_k sum_i h[i] * (1-h[i+k]) * dx

This is the maximum cross-correlation between h and (1-h) at any lag.

For a binary sequence (h ∈ {0,1}), this is related to the maximum of:
- Number of 1s in h that align with 0s in the shifted (1-h)
- Which equals: number of positions where h[i]=1 and h[i+k]=0

This is asymmetric in general (the correlation at lag k and lag -k are different).

For the continuous case, h can take values in [0,1], giving more flexibility.

Let me think about what a good solution looks like. If h were uniformly 1/2 everywhere, then:
corr_k = sum (1/2)(1/2) = n/4 for all k (in the interior)
C5 = (n/4) * (2/n) = 0.5

import numpy as np
import time


def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    """Compute C5 = max_k corr_k * dx."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_corr(h):
    """Full correlation vector."""
    return np.correlate(h, 1.0 - h, mode="full")


def pair_gradient(h, a, b):
    """
    Compute A[k] = d(corr_k)/d(d) for move h[a]+=d, h[b]-=d.
    corr_k uses np.correlate(h, 1-h, 'full') indexing.
    """
    n = len(h)
    A = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        L = max(0, s)
        U = min(n - 1, n - 1 - s)
        ga = 0.0
        if L <= a <= U:
            as_idx = a + s
            if 0 <= as_idx < n:
                ga += (1.0 - h[as_idx])
        a_ms = a - s
        if L <= a_ms <= U:
            ga -= h[a_ms]
        gb = 0.0
        if L <= b <= U:
            bs_idx = b + s
            if 0 <= bs_idx < n:
                gb += (1.0 - h[bs_idx])
        b_ms = b - s
        if L <= b_ms <= U:
            gb -= h[b_ms]
        A[k] = ga - gb
    return A


def find_optimal_d(corr, A, d_low, d_high):
    """
    Minimize max(corr + d*A) over d in [d_low, d_high].
    Uses iterative active-set refinement.
    """
    if d_high <= d_low:
        return d_low
    if np.max(A) <= 1e-14:
        return d_high
    if np.min(A) >= -1e-14:
        return d_low

    # Iterative method: restrict to active set, recompute min, repeat
    n_lags = len(corr)
    idx = np.arange(n_lags)

    for _ in range(80):
        vals = corr[idx] + d_low * A[idx]
        m = np.max(vals)
        active = idx[vals >= m - 1e-12]
        amin = np.min(A[active])
        amax = np.max(A[active])
        if amin < -1e-12 and amax > 1e-12:
            d_new = d_low
            break
        elif amin >= -1e-12:
            d_new = d_low
            break
        else:
            # All active have negative slope; need a line with positive slope
            pos_mask = A > 1e-12
            if not np.any(pos_mask):
                d_new = d_high
                break
            pos_idx = np.where(pos_mask)[0]
            # Find the positive-slope line that hits the current level earliest
            # At d_low, the level is m. Line i reaches m at:
            # d_i = (m - corr[i]) / A[i]
            m_low = np.max(corr[idx] + d_low * A[idx])
            d_cross = (m_low - corr[pos_idx]) / A[pos_idx]
            valid = d_cross >= d_low - 1e-12
            if not np.any(valid):
                d_new = d_high
                break
            d_new = np.max(d_cross[valid])
            if d_new > d_high:
                d_new = d_high
            idx = np.concatenate([idx, pos_idx])
            idx = np.unique(idx)
            continue

    # Refine locally around d_new
    left = max(d_low, d_new - 0.05)
    right = min(d_high, d_new + 0.05)
    best_d = d_new
    best_f = np.max(corr + d_new * A)
    for _ in range(200):
        if right - left < 1e-10:
            break
        m1 = left + (right - left) / 3.0
        m2 = right - (right - left) / 3.0
        f1 = np.max(corr + m1 * A)
        f2 = np.max(corr + m2 * A)
        if f1 < f2:
            right = m2
        else:
            left = m1
        mid = (left + right) / 2.0
        fm = np.max(corr + mid * A)
        if fm < best_f:
            best_f = fm
            best_d = mid

    # Also check boundaries
    f_lo = np.max(corr + d_low * A)
    f_hi = np.max(corr + d_high * A)
    if f_lo < best_f:
        best_d = d_low
    if f_hi < best_f:
        best_d = d_high

    return best_d


def try_pair(h, a, b, corr, dx):
    """Try the optimal pair move h[a]+=d, h[b]-=d. Return (improved_h, new_c5, new_corr) or None."""
    A = pair_gradient(h, a, b)
    d_low = max(-h[a], h[b] - 1.0)
    d_high = min(1.0 - h[a], h[b])
    if d_high - d_low < 1e-12:
        return None
    d_opt = find_optimal_d(corr, A, d_low, d_high)
    if abs(d_opt) < 1e-12:
        return None
    h_new = h.copy()
    h_new[a] = np.clip(h[a] + d_opt, 0.0, 1.0)
    h_new[b] = np.clip(h[b] - d_opt, 0.0, 1.0)
    # Verify sum constraint
    if abs(np.sum(h_new) - np.sum(h)) > 1e-6:
        return None
    new_corr = compute_corr(h_new)
    new_c5 = np.max(new_corr) * dx
    return h_new, new_c5, new_corr


def make_binary_initial(n, rng):
    """Create a structured binary initial pattern."""
    h = np.zeros(n)
    h[::2] = 1.0
    if np.sum(h) != n // 2:
        h[-1] = n / 2.0 - np.sum(h[:-1])
    return h

                a = indices[ii]

                    if c5_new < best_c5 - 1e-12:

                        best_h = h_new.copy()
                    h0 = h_new
                    corr = corr_new
                    improved += 1

        if no_improve_count >= 3:

            corr = compute_corr(h0)
            no_improve_count = 0

    ph2_end = t0 + budget_s * 0.82
    perturb_scales = [0.03, 0.05, 0.08, 0.12, 0.04, 0.06]

        h0 = best_h + rng.randn(n) * scale
        h0 = project_onto_constraint(h0, target_sum)
        corr = compute_corr(h0)
        cur_c5 = np.max(corr) * dx

                        h0 = h_new
