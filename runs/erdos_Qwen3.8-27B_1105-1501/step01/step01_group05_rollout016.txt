import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            idx = np.argmax(1.0 - h)
            delta = min(residual, 1.0 - h[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        idx = np.argmax(1.0 - h) if residual > 0 else np.argmax(h)
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_gradient(h, beta):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    
    # Numerically stable log-sum-exp
    m = np.max(scaled)
    log_sum_exp = m + np.log(np.sum(np.exp(beta * (scaled - m)))) / beta
    
    # Weights for each lag
    shifts = beta * (scaled - m)
    weights = np.exp(shifts)
    weights /= np.sum(weights)
    
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)
    
    grad = np.zeros(n)
    for ki in range(n_lags):
        w = weights[ki]
        if w < 1e-12:
            continue
        k = abs(lags[ki])
        # ∂f_k/∂h[j] = dx * [(1-h[j+k]) * I(0<=j<=n-1-k) - h[j-k] * I(k<=j<=n-1)]
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        else:
            part1_idx = slice(0, n - k)
            part2_idx = slice(k, n)
            grad[part1_idx] += w * dx * (1.0 - h[part1_idx + k])
            grad[part2_idx] -= w * dx * h[part2_idx - k]
    
    return log_sum_exp, grad

def lse_descent(h, n, beta_start=10, beta_end=200, n_steps=50, 
                lr_start=0.05, lr_end=0.005, time_limit=30.0, seed=42):

    h = repair_vector(h, n)
    score = official_score(h)
    start_time = time.perf_counter()
    
    for step in range(n_steps):
        if time.perf_counter() - start_time > time_limit:
            break
        
        progress = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        _, grad = lse_gradient(h, beta)

        grad -= np.mean(grad)

import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(300):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            idx = np.argmax(1.0 - h)
            delta = min(residual, 1.0 - h[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] -= residual
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    shifts = beta * (scaled - m)
    weights = np.exp(shifts - np.max(shifts))
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros_like(weights)
        weights[np.argmin(-(scaled - m))] = 1.0
        wsum = 1.0
    weights /= wsum
    n_lags = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ki in range(n_lags):
        w = weights[ki]
        if w < 1e-12:
            continue
        k = lags[ki]
        ak = abs(k)
        if ak == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        else:
            part1 = slice(0, n - ak)
            part2 = slice(ak, n)
            grad[part1] += w * dx * (1.0 - h[part1 + ak])
            grad[part2] -= w * dx * h[part2 - ak]
    return grad


def lse_descent(h, n, time_limit=30.0, seed=42):
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n)
    score = official_score(h)
    start_time = time.perf_counter()
    beta = 5.0
    lr = 0.1
    it = 0
    stagnation = 0
    while time.perf_counter() - start_time < time_limit:
        it += 1
        if it > 200:
            break
        grad = lse_gradient(h, beta)
        grad -= np.mean(grad)
        noise_scale = 0.02 * max(0.1, 1.0 - it / 100.0)
        noise = rng.normal(0, noise_scale, n)
        noise -= np.mean(noise)
        direction = -grad + noise
        norm = np.linalg.norm(direction)
        if norm < 1e-13:
            break
        direction /= norm
        improved = False
        for a in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 2.0]:
            h_try = repair_vector(h + a * direction, n)
            s = official_score(h_try)
            if s < score - 1e-13:
                h = h_try
                score = s
                improved = True
                break
        if not improved:
            for a in [lr * 0.5, lr * 0.1]:
                h_try = repair_vector(h - a * direction, n)
                s = official_score(h_try)
                if s < score - 1e-13:
                    h = h_try
                    score = s
                    improved = True
                    break
        if not improved:
            stagnation += 1
            if stagnation > 8:
                # Try random coordinate moves
                moved = False
                for _ in range(30):
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i == j:
                        continue
                    d = rng.choice([0.02, 0.05, 0.1, 0.15, 0.2])
                    h_try = h.copy()
                    h_try[i] += d
                    h_try[j] -= d
                    h_new = repair_vector(h_try, n)
                    s = official_score(h_new)
                    if s < score - 1e-12:
                        h = h_new
                        score = s
                        moved = True
                        break
                if not moved:
                    beta *= 1.5
                    lr *= 0.7
                    if beta > 500.0:
                        break
            beta *= 1.1
            lr *= 0.95
        else:
            stagnation = 0
    return h, official_score(h)

def slsqp_refine(h, n, max_iter=60, time_limit=30.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki in range(n_lags):
            k = lags[ki]
            ak = abs(k)
            if ak == 0:
                j[ki, :n] = -dx * (1.0 - 2.0 * h_x)
            else:
                row = np.zeros(n)
                if 0 <= n - 1 - ak:
                    row[0:n - ak] = (1.0 - h_x[ak:n])
                row[ak:n] -= h_x[0:n - ak]
                j[ki, :n] = -dx * row
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    initial_score = official_score(h)
    theta0 = initial_score + 1e-6
    x0 = np.concatenate([h, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
            constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return repair_vector(h, n)

def gen_step_pattern(n, rng, n_steps):
    boundaries = sorted(rng.choice(n - 1, size=n_steps - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]

    # Generate initial candidates
    candidates = []
    candidates.append((best_h.copy(), best_n))
    candidates.append((best_h[::-1].copy(), best_n))
    for n_try in res_list:
        if n_try == best_n:
            continue
        candidates.append((gen_interpolated(best_h, n_try), n_try))
        candidates.append((gen_interpolated(best_h, n_try, 1), n_try))

    # Add diverse random candidates at a few sizes
    for n_try in [best_n, max(16, best_n - 10), best_n + 10]:
        if time.perf_counter() > deadline - 200:
            break
        for _ in range(3):
            candidates.append((gen_perturbed(best_h[:n_try] if n_try <= len(best_h)
                              else gen_interpolated(best_h, n_try), rng, 0.15), n_try))
        for _ in range(2):
            candidates.append((gen_step_pattern(n_try, rng, 3), n_try))
        for _ in range(2):
            candidates.append((gen_step_pattern(n_try, rng, 4), n_try))

    # Score all candidates
    scored = []
    for h_c, n_c in candidates:
        h_c = repair_vector(h_c, n_c)
        s = official_score(h_c)
        if np.isfinite(s):
            scored.append((s, h_c, n_c))
    scored.sort(key=lambda x: x[0])

    if len(scored) > 0 and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = scored[0][2]

    # Main optimization loop: alternate between LSE descent and coordinate descent
    top_k = min(6, len(scored))
    for idx in range(top_k):
        if time.perf_counter() > deadline - 60:
            break
        s_c, h_c, n_c = scored[idx]
        remaining = deadline - time.perf_counter()
        t_alloc = min(120.0, max(20.0, remaining * 0.6 / max(1, top_k - idx)))

        # Phase 1: LSE gradient descent
        t1 = t_alloc * 0.35
        h1, s1 = lse_descent(h_c, n_c, time_limit=t1, seed=seed + idx * 13)
        if s1 < best_score:
            best_score = s1
            best_h = h1.copy()
            best_n = n_c

        # Phase 2: Coordinate descent on result of phase 1
        t2 = t_alloc * 0.30
        h2, s2 = coordinate_descent_fast(h1, n_c, max_sweeps=12, time_limit=t2)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()
            best_n = n_c

        # Phase 3: SLSQP refinement
        t3 = t_alloc * 0.25
        h3, s3 = slsqp_refine(h2, n_c, max_iter=50, time_limit=t3)
        if s3 < best_score:
            best_score = s3
            best_h = h3.copy()
            best_n = n_c

        # Phase 4: Coordinate descent on SLSQP result
        t4 = max(5.0, t_alloc * 0.10)
        h4, s4 = coordinate_descent_fast(h3, n_c, max_sweeps=8, time_limit=t4)
        if s4 < best_score:
            best_score = s4
            best_h = h4.copy()
            best_n = n_c

    # Second round: interpolate best to different resolutions and optimize
    if time.perf_counter() < deadline - 80:
        for n_try in [best_n, best_n + 5, best_n + 10, max(16, best_n - 5)]:
            if time.perf_counter() > deadline - 40:
                break
            if n_try == best_n:
                h_start = gen_perturbed(best_h, rng, 0.05)
            else:
                h_start = gen_interpolated(best_h, n_try)
            remaining = deadline - time.perf_counter()
            t_alloc = min(60.0, max(10.0, remaining * 0.4))
            h_a, s_a = lse_descent(h_start, n_try, time_limit=t_alloc * 0.4, seed=seed + 500)
            h_b, s_b = coordinate_descent_fast(h_a, n_try, max_sweeps=10, time_limit=t_alloc * 0.3)
            h_c2, s_c2 = slsqp_refine(h_b, n_try, max_iter=40, time_limit=t_alloc * 0.25)
            h_d, s_d = coordinate_descent_fast(h_c2, n_try, max_sweeps=8, time_limit=t_alloc * 0.15)
            for hh, ss in [(h_a, s_a), (h_b, s_b), (h_c2, s_c2), (h_d, s_d)]:
                if ss < best_score:
                    best_score = ss
                    best_h = hh.copy()
                    best_n = len(hh)

    # Final polish: coordinate descent + SLSQP on current best
    if time.perf_counter() < deadline - 20:
        h_p, s_p = coordinate_descent_fast(best_h, best_n, max_sweeps=10,
                                           time_limit=min(30.0, (deadline - time.perf_counter()) * 0.4))
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()
    if time.perf_counter() < deadline - 10:
        h_s, s_s = slsqp_refine(best_h, best_n, max_iter=40,
                                time_limit=min(20.0, (deadline - time.perf_counter()) * 0.5))
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()
    if time.perf_counter() < deadline - 5:
        h_f, s_f = coordinate_descent_fast(best_h, best_n, max_sweeps=5, time_limit=5.0)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f.copy()

    # Final validation
    best_h = repair_vector(best_h, len(best_h))
    final_score = official_score(best_h)
    return (best_h, float(final_score), int(len(best_h)))
