Let me also think about what makes the problem difficult. The objective is:
f(h) = max_k c_k(h)

where c_k(h) are smooth (bilinear) functions. The max of smooth functions is convex (if each c_k is convex) or just smooth (if they're not convex). In our case, c_k is bilinear (not convex), so f is not necessarily convex. This means we can get stuck in local minima.

import numpy as np
import time

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx_order = np.argsort(z)
            for idx in idx_order:
                add = min(diff, 1.0 - z[idx])
                if add > 0:
                    z[idx] += add
                    diff -= add
                    if abs(diff) < 1e-13:
                        break
        else:
            idx_order = np.argsort(-z)
            for idx in idx_order:
                sub = min(-diff, z[idx])
                if sub > 0:
                    z[idx] -= sub
                    diff += sub
                    if abs(diff) < 1e-13:
                        break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def overlap_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_grad(h, T):
    n = len(h)
    scores = overlap_scores(h)
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[c + d]
        wn = w[c - d]
        if wp < 1e-15 and wn < 1e-15:
            continue
        m = n - d
        g[d:] += wp * (1.0 - h[:m])
        g[:m] -= wp * h[d:]
        g[:m] += wn * (1.0 - h[d:])
        g[d:] -= wn * h[:m]
    return g, base

def golden_line_search(h, direction, cur_score, max_iter=25):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    hi = 0.3
    cand_hi = make_feasible(h + hi * direction)
    score_hi = score_exact(cand_hi)
    if score_hi >= cur_score:
        hi = 0.1
        cand_hi = make_feasible(h + hi * direction)
        score_hi = score_exact(cand_hi)
        if score_hi >= cur_score:
            hi = 0.05
            cand_hi = make_feasible(h + hi * direction)
            score_hi = score_exact(cand_hi)
            if score_hi >= cur_score:
                return None, cur_score
    lo = 0.0
    a = lo + (1.0 - gr) * (hi - lo)
    b = lo + gr * (hi - lo)
    cand_a = make_feasible(h + a * direction)
    score_a = score_exact(cand_a)
    cand_b = make_feasible(h + b * direction)
    score_b = score_exact(cand_b)
    for _ in range(max_iter):
        if score_a < score_b:
            hi = b
            b = a
            score_b = score_a
            a = lo + (1.0 - gr) * (hi - lo)
            cand_a = make_feasible(h + a * direction)
            score_a = score_exact(cand_a)
        else:
            lo = a
            a = b
            score_a = score_b
            b = lo + gr * (hi - lo)
            cand_b = make_feasible(h + b * direction)
            score_b = score_exact(cand_b)
    s = (lo + hi) * 0.5
    best_cand = make_feasible(h + s * direction)
    best_score = score_exact(best_cand)
    if best_score < cur_score:
        return best_cand, best_score
    return None, cur_score

def fista_optimize(h, deadline, T_start, T_end, T_decay, max_iters=None):
    n = len(h)
    cur = h.copy()
    prev = cur.copy()
    cur_score = score_exact(cur)
    best = cur.copy()
    best_score = cur_score
    T = T_start
    t_k = 1.0
    momentum = 0.8
    stall = 0
    it = 0
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break
        it += 1
        g, _ = smooth_max_grad(cur, T)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            T = max(T_end, T * 0.9)
            continue
        direction = -g / g_norm
        # Nesterov extrapolation
        theta = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = cur + (t_k - 1.0) / theta * (cur - prev)
        y = np.clip(y, 0.0, 1.0)
        g_y, _ = smooth_max_grad(y, T)
        direction = -g_y / max(float(np.linalg.norm(g_y)), 1e-14)
        new_h, new_score = golden_line_search(y, direction, score_exact(y), max_iter=18)
        if new_h is not None:
            if new_score < cur_score - 1e-14:
                prev = cur
                cur = new_h
                cur_score = new_score
                stall = 0
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score
            else:
                prev = cur
                cur = new_h
                cur_score = score_exact(cur)
                stall += 1
        else:
            # Try smaller fixed step
            for s in (0.02, 0.01, 0.005):
                cand = make_feasible(cur - s * direction)
                cs = score_exact(cand)
                if cs < cur_score - 1e-14:
                    prev = cur
                    cur = cand
                    cur_score = cs
                    stall = 0
                    if cur_score < best_score:
                        best = cur.copy()
                        best_score = cur_score
                    break
            else:
                stall += 1
                if stall > 50:
                    cur = make_feasible(best + np.random.default_rng(it).normal(0, 0.01, n))
                    cur_score = score_exact(cur)
                    prev = cur.copy()
                    stall = 0
        t_k = theta
        T = max(T_end, T * T_decay)
    return best, best_score

def pair_exact_search(h, i, j, cur_score, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_score
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_score - 1e-14 and abs(best_a) > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo2 = max(lo, best_a - 2.0 * (hi - lo) / n_pts)
        hi2 = min(hi, best_a + 2.0 * (hi - lo) / n_pts)
        if hi2 - lo2 > 1e-12:
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s < cur_score - 1e-14:
                return float(best_s)
    return None

def coord_exact_search(h, i, cur_score, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_score
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        zz = make_feasible(h)
        s = score_exact(zz)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_score - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        hh = make_feasible(h)
        return score_exact(hh)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10 or n_new > 200:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)
