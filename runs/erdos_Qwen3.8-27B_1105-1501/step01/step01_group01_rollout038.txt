import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, n):
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = n / 2.0
    lo = -float(np.max(z)) - 2.0
    hi = 2.0 - float(np.min(z))
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            a = min(residual, 1.0 - h[i]) if residual > 0 else max(residual, -h[i])
            if abs(a) < 1e-15:
                break
            h[i] += a
            residual -= a
    return h

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lse_weights(corr, beta):
    mc = np.max(corr)
    sh = np.clip(beta * (corr - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return ev / es

def lse_score(h, beta):
    corr = corr_values(h)
    mc = np.max(corr)
    sh = np.clip(beta * (corr - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def compute_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            ns = n - s
            grad[:ns] += wk * dx * (1.0 - h[s:])
            grad[s:] -= wk * dx * h[:ns]
        else:
            sp = -s
            ns = n - sp
            grad[sp:] += wk * dx * (1.0 - h[:ns])
            grad[:ns] -= wk * dx * h[sp:]
    return grad

def sum_sq_score(h):
    corr = corr_values(h)
    return float(np.sum(corr * corr))

def sum_sq_grad(h):
    n = len(h)
    corr = corr_values(h)
    # grad = 2 * sum_k c_k * dc_k/dh
    # We can compute this by: for each k, add 2*c_k * dc_k/dh
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        ck = corr[k]
        if abs(ck) < 1e-14:
            continue
        s = k - (n - 1)
        coeff = 2.0 * ck * dx
        if s >= 0:
            ns = n - s
            grad[:ns] += coeff * (1.0 - h[s:])
            grad[s:] -= coeff * h[:ns]
        else:
            sp = -s
            ns = n - sp
            grad[sp:] += coeff * (1.0 - h[:ns])
            grad[:ns] -= coeff * h[sp:]
    return grad

def pgd_smooth(h0, betas, iters_per_beta, alpha_scale=2.0, mu=0.85, deadline=None):
    h = h0.copy()
    n = len(h)
    v = np.zeros(n)
    best_h = h.copy()
    best_actual = official_score(h)

    for beta in betas:
        alpha = alpha_scale / (beta * 0.1 + 1.0) / n
        cur_lse = lse_score(h, beta)
        stall = 0
        for it in range(iters_per_beta):
            if deadline is not None and time.time() > deadline:
                return best_h, best_actual
            corr = corr_values(h)
            w = lse_weights(corr, beta)
            g = compute_grad(h, w)
            v = mu * v + g
            h_try = project_feasible(h - alpha * v, n)
            new_lse = lse_score(h_try, beta)
            if new_lse <= cur_lse + 1e-13:
                h = h_try
                cur_lse = new_lse
                sc = official_score(h)
                if sc < best_actual - 1e-14:
                    best_actual = sc
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                alpha *= 0.5
                v *= 0.5
                if alpha < 1e-14:
                    break
            if stall > iters_per_beta // 2:
                break
    return best_h, best_actual

def pgd_sum_sq(h0, iters, alpha=0.001, deadline=None):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_actual = official_score(h)
    cur_sq = sum_sq_score(h)

            cur_sq = new_sq

def binary_init(n, rng):
    m = n // 2
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m]] = 1.0
    if n % 2 == 1:
