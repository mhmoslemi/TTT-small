def pair_line_search(h, i, j, corr, dx, n):
    """Find optimal delta for pair exchange h[i]-=delta, h[j]+=delta.
    Returns (best_delta, best_c5)."""
    # Feasible range for delta
    delta_lo = -min(h[j], 1.0 - h[i])
    delta_hi = min(h[i], 1.0 - h[j])
    if delta_hi - delta_lo < 1e-12:
        return 0.0, compute_c5(h)
    
    # Precompute c_s for all lags s = -(n-1) to n-1
    # corr index k corresponds to lag s = k - (n-1)
    c = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        s = k - (n - 1)
        # c_s = h[i-s] + h[i+s] - h[j-s] - h[j+s] (with 0 for out of bounds)
        c[k] = 0.0
        if 0 <= i - s < n:
            c[k] += h[i - s]
        if 0 <= i + s < n:
            c[k] += h[i + s]
        if 0 <= j - s < n:
            c[k] -= h[j - s]
        if 0 <= j + s < n:
            c[k] -= h[j + s]
    
    # Ternary search for min of max(corr + delta*c)
    def obj(delta):
        return np.max(corr + delta * c) * dx
    
    # Ternary search
    lo, hi = delta_lo, delta_hi
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    best_delta = (lo + hi) / 2.0
    best_val = obj(best_delta)
    
    # Also check the current value (delta=0) and boundaries
    current_val = np.max(corr) * dx
    if obj(0.0) < best_val:
        best_val = obj(0.0)
        best_delta = 0.0
    if obj(delta_lo) < best_val:
        best_val = obj(delta_lo)
        best_delta = delta_lo
    if obj(delta_hi) < best_val:
        best_val = obj(delta_hi)
        best_delta = delta_hi
    
    return best_delta, best_val

Actually, I want to reconsider one thing. The ternary search I'm doing is on the function f(δ) = max_k (corr_k + δ * c_k). This is a convex piecewise-linear function. The minimum of such a function over [lo, hi] can be found more efficiently than ternary search:

1. The minimum is either at a "kink" (where the argmax changes) or at a boundary.
2. The kinks occur at δ values where two linear functions are equal: corr_k + δ*c_k = corr_m + δ*c_m, i.e., δ = (corr_m - corr_k) / (c_k - c_m).

But finding all kinks is O(n²) which is more expensive than the O(n * 40) ternary search for small n.

Alternatively, I can use the fact that the subgradient of f at δ is the convex hull of {c_k : k ∈ argmax_k (corr_k + δ*c_k)}. The minimum is where 0 is in the subgradient.

A bisection-like approach:
- At a given δ, compute the set S(δ) = argmax_k (corr_k + δ*c_k)
- If min_{k∈S} c_k > 0: f is decreasing, so minimum is to the left
- If max_{k∈S} c_k < 0: f is increasing, so minimum is to the right
- Otherwise: δ is a minimum (or at a kink)

But this requires evaluating the argmax at each step, which is O(n), and we need O(log(1/ε)) steps. Same complexity as ternary search.

For convex functions, golden section search is slightly better than ternary search (it reuses one evaluation). But the difference is minor.

Let me just use ternary search with 40 iterations. It's simple and effective.

import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)


def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    phi = 0.6180339887
    positions = np.array([int((i * phi * n) % n) for i in range(n)])
    positions = np.unique(positions)
    count = n // 2
    if len(positions) >= count:
        h[positions[:count]] = 1.0
    else:
        h[::2] = 1.0
    return project_sum(h, n / 2.0)


def compute_c_vec(h, i, j, s_vals, n):
    sz = 2 * n - 1
    ip = i + s_vals
    im = i - s_vals
    sig_i = np.zeros(sz)
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    sig_i[vp] += h[ip[vp]]
    sig_i[vm] += h[im[vm]]
    jp = j + s_vals
    jm = j - s_vals
    sig_j = np.zeros(sz)
    vj = (jp >= 0) & (jp < n)
    vmj = (jm >= 0) & (jm < n)
    sig_j[vj] += h[jp[vj]]
    sig_j[vmj] += h[jm[vmj]]
    return sig_i - sig_j


def find_best_delta(corr, c, d_lo, d_hi):
    """Find delta in [d_lo, d_hi] minimizing max(corr + delta*c)."""
    lo, hi = d_lo, d_hi
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * c) <= np.max(corr + m2 * c):
            hi = m2
        else:
            lo = m1
    best_d = (lo + hi) / 2.0
    best_v = np.max(corr + best_d * c)
    # Check boundaries and 0
    v0 = np.max(corr)
    if v0 < best_v:

    v_lo = np.max(corr + d_lo * c)
    if v_lo < best_v:
        best_d, best_v = d_lo, v_lo
    v_hi = np.max(corr + d_hi * c)
    if v_hi < best_v:
        best_d, best_v = d_hi, v_hi
    return best_d, best_v
