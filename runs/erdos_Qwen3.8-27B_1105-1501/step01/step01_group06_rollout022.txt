def binary_energy(h):
    """Compute max lag count for binary h"""
    n = len(h)
    max_count = 0
    for d in range(1, n):
        # h[i] * (1 - h[i+d]) for i = 0 to n-d-1
        count = np.sum(h[:n-d] * (1 - h[d:]))
        if count > max_count:

    return max_count

    return float(np.max(corr * dx))

    h = h.copy()
    best_h = h.copy()
    best_e = float(binary_energy(h))
    current_e = float(best_e)
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h < 0.5)[0].tolist())
    
    for step in range(n_steps):

        # Pick random swap
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        h[i] = 0; h[j] = 1
        ones.discard(i); ones.add(j)
        zeros.discard(j); zeros.add(i)
        new_e = float(binary_energy(h))

            if new_e < best_e:

                best_h = h.copy()

            # Revert
            h[i] = 1; h[j] = 0
            ones.discard(j); ones.add(i)
            zeros.discard(i); zeros.add(j)

        if w < 1e-15:

            grad[d:] += w * (1.0 - h[:n-d])

            grad[d2:] -= w * h[:n-d2]
    return grad
