def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corrs(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    # Water filling
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2
    result = np.clip(h - mu, 0, 1)
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual among interior points
        interior = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            n_int = np.sum(interior)
            result[interior] += residual / n_int
            result = np.clip(result, 0, 1)
        else:
            # Just adjust one point
            idx = np.argmax(1.0 - np.abs(result - 0.5))
            result[idx] += residual
            result = np.clip(result, 0, 1)
    return result

def compute_subgradient(h, n_active=8):
    """Compute approximate subgradient of max_k c_k(h)."""
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corrs)
    
    # Get top-k active
    top_idx = np.argsort(corrs)[-n_active:]
    # Weights: 1 for top, decay for rest
    weights = np.zeros(len(corrs))
    for idx in top_idx:
        weights[idx] = 1.0
    # Add some weight to near-active
    near_mask = corrs >= c_max * 0.95
    weights[near_mask] += 0.5
    weights = weights / np.sum(weights)
    
    # Compute gradient of weighted sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        w = weights[k] * dx
        if lag > 0:
            # c = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # dc/dh[j] for j in [lag, n-1]: (1-h[j-lag])
            # dc/dh[j] for j in [0, n-1-lag]: -h[j+lag]
            for i in range(n - lag):
                grad[i + lag] += w * (1 - h[i])
                grad[i] += -w * h[i + lag]
        elif lag < 0:
            # c = sum_{i=0}^{n-1+lag} h[i]*(1-h[i-lag])
            L = -lag
            for i in range(n - L):
                grad[i] += w * (1 - h[i + L])
                grad[i + L] += -w * h[i]
        else:
            # c = sum h[i]*(1-h[i])
            for i in range(n):
                grad[i] += w * (1 - 2 * h[i])
    
    return grad

If h is binary (0 or 1), then h(x)(1-h(x+k)) = 1 when h(x)=1 and h(x+k)=0. So we're counting the number of (1,0) pairs at each lag.

The constraint ∫h = 1 with h ∈ [0,1] and interval [0,2] means the "mass" of h is 1, which is half the interval length. So on average, h = 1/2.

For a binary h with exactly half the positions being 1:
- Total (1,0) pairs at lag k = c_k (unscaled)
- We want to minimize max_k c_k * 2/n

The sum over all k of c_k = ∑_k ∑_i h[i](1-h[i+k]) = ∑_i h[i] * (number of j ≠ i with h[j]=0 in range) 

This is getting complicated. Let me just focus on the algorithm.

One key observation: for the binary case, c_k = number of i such that h[i]=1 and h[i+k]=0 (for k>0). Also c_{-k} = number of i such that h[i]=1 and h[i-k]=0 = number of i such that h[i+k]=1 and h[i]=0.

Note that c_k + c_{-k} = number of (different) pairs at distance k = number of i where {h[i], h[i+k]} = {0,1}.

So c_k + c_{-k} = (n-k) - (number of same pairs at distance k) = (n-k) - (number of 00 pairs) - (number of 11 pairs).

For a binary string with n/2 ones and n/2 zeros:
Total pairs at distance k: n-k
Number of 11 pairs + number of 00 pairs + number of different pairs = n-k
Number of different pairs at distance k = c_k + c_{-k}

We want to minimize max(c_k, c_{-k}) for all k. Since c_k + c_{-k} is determined by the arrangement, we want to make c_k ≈ c_{-k} for each k (symmetry), and also make the "different pairs" count as small as possible for the worst k.

The minimum possible value of max_k (c_k + c_{-k}) / 2... this is related to the "discrepancy" of the binary string.

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    lo, hi = -0.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(h - mu, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-13:
        interior = np.where((result > 1e-8) & (result < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            result[interior] += residual / len(interior)
        else:
            idx = np.argmin(np.abs(result - 0.5))
            result[idx] += residual
        result = np.clip(result, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:
        interior = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            result[interior[0]] += residual
            result = np.clip(result, 0.0, 1.0)
    return result


def compute_subgradient(h, n_active=10):
    n = len(h)
    dx = 2.0 / n
    corrs_raw = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corrs_raw)
    sorted_idx = np.argsort(corrs_raw)[::-1]
    active_count = min(n_active, len(corrs_raw))
    active = sorted_idx[:active_count]
    threshold = c_max * 0.92
    for idx in range(len(corrs_raw)):
        if corrs_raw[idx] >= threshold:
            if idx not in active[:active_count]:
                active = np.append(active, idx)
    grad = np.zeros(n)
    for k in active:
        lag = int(k) - (n - 1)
        w = dx
        if lag > 0:
            L = lag
            if L < n:
                grad[L:] += w * (1.0 - h[:n - L])
                grad[:n - L] += -w * h[L:]
        elif lag < 0:
            M = -lag
            if M < n:
                grad[:n - M] += w * (1.0 - h[M:])
                grad[M:] += -w * h[:n - M]
        else:
            grad += w * (1.0 - 2.0 * h)
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-14:
        grad /= gnorm
    return grad

def line_search(h, d, current_c5, n_grid=100):
    n = len(h)
    t_lo, t_hi = feasible_interval(h, d)
    if t_hi - t_lo < 1e-12:
        return 0.0, current_c5
    best_t = 0.0
    best_val = current_c5
    ts = np.linspace(t_lo, t_hi, n_grid)
    for t in ts:
        h_t = h + t * d
        if np.any(h_t < -1e-9) or np.any(h_t > 1.0 + 1e-9):
            continue
        c5_t = compute_c5(h_t)
        if c5_t < best_val:
            best_val = c5_t
            best_t = t
    if best_val < current_c5 - 1e-10:
        margin = (t_hi - t_lo) / n_grid * 2.0
        t_ref_lo = max(t_lo, best_t - margin)
        t_ref_hi = min(t_hi, best_t + margin)
        ts2 = np.linspace(t_ref_lo, t_ref_hi, 50)
        for t in ts2:
            h_t = h + t * d
            if np.any(h_t < -1e-9) or np.any(h_t > 1.0 + 1e-9):
                continue
            c5_t = compute_c5(h_t)
            if c5_t < best_val:
                best_val = c5_t
                best_t = t
    return best_t, best_val


def random_direction(n, rng):
    d = rng.randn(n)
    d -= np.mean(d)
    mx = np.max(np.abs(d))
    if mx < 1e-14:
        d = np.zeros(n)
        d[0] = 1.0
        d[1] = -1.0
        return d
    return d / mx


def generate_directions(h, grad, rng, n, n_pairs=5):
    directions = []
    directions.append(-grad)
    sorted_idx = np.argsort(grad)
    low_grp = sorted_idx[:max(1, n // 4)]
    high_grp = sorted_idx[-max(1, n // 4):]
    used = set()
    for _ in range(n_pairs * 3):
        i = int(rng.choice(low_grp))
        j = int(rng.choice(high_grp))
        if i == j or (i, j) in used:
            continue
        if h[i] >= 1.0 - 1e-6 or h[j] <= 1e-6:
            continue
        used.add((i, j))
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        directions.append(d)
        if len(used) >= n_pairs:
            break
    directions.append(random_direction(n, rng))
    if len(used) < n_pairs:
        for _ in range(n_pairs - len(used)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j and h[i] < 1.0 - 1e-6 and h[j] > 1e-6:
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                directions.append(d)
    return directions


def optimize_main(h0, time_budget, seed=42):
    n = len(h0)
    h = make_feasible(h0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    momentum_v = np.zeros(n)
    beta = 0.7
    lr = 1.0
    stall = 0
    stall_limit = 150
    n_active = 10
    n_grid = 100
    n_pairs = 5

    while time.time() - t0 < time_budget:
        if time.time() - t0 > time_budget:
            break

        cur_c5 = compute_c5(h)
        grad = compute_subgradient(h, n_active)
        momentum_v = beta * momentum_v - lr * grad

        directions = generate_directions(h, momentum_v, rng, n, n_pairs)
        # Also add -grad directly
        directions.append(-grad)

        best_step_t = 0.0
        best_step_val = cur_c5
        best_dir = None

        for d in directions:
            if time.time() - t0 > time_budget:
                break
            d_clean = d - np.mean(d)
            mx = np.max(np.abs(d_clean))
            if mx < 1e-14:
                continue
            d_clean = d_clean / mx
            t, val = line_search(h, d_clean, cur_c5, n_grid)
            if val < best_step_val:
                best_step_val = val
                best_step_t = t
                best_dir = d_clean

        if best_dir is not None and best_step_val < best_c5 - 1e-12:
            h_new = h + best_step_t * best_dir
            h_new = make_feasible(h_new)
            c5_new = compute_c5(h_new)
            if c5_new <= best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > stall_limit:
            noise = 0.03 + 0.03 * min(stall // stall_limit, 5)
            h = best_h + noise * rng.randn(n)
            h = make_feasible(h)
            momentum_v = np.zeros(n)
            stall = 0
            beta = min(beta + 0.05, 0.95)
            n_active = max(5, n_active - 2)
            if n_active < 5:
                n_active = 10
        elif stall > stall_limit // 2:
            beta = min(beta + 0.02, 0.95)

    return best_h, best_c5


def polish_pairs(h, time_budget, seed=42):
    n = len(h)
    h = make_feasible(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    interior = np.where((h > 1e-4) & (h < 1.0 - 1e-4))[0]
    if len(interior) < 2:
        return best_h, best_c5

    delta_options = np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.03, 0.07, 0.15])

    sweep = 0
    while time.time() - t0 < time_budget:
        improved = False
        idx_perm = rng.permutation(len(interior))
        for ii in idx_perm:
            if time.time() - t0 > time_budget:
                break
            i = int(interior[ii])
            for jj in rng.randint(0, len(interior), 3):
                j = int(interior[jj])
                if i == j:
                    continue
                for delta in delta_options[:4]:
                    for sign in [1, -1]:
                        h_test = best_h.copy()
                        h_test[i] += sign * delta
                        h_test[j] -= sign * delta
                        if h_test[i] < 0 or h_test[i] > 1 or h_test[j] < 0 or h_test[j] > 1:
                            continue
                        c5_test = compute_c5(h_test)
                        if c5_test < best_c5 - 1e-13:
                            best_c5 = c5_test
                            best_h = h_test.copy()
                            h = h_test.copy()
                            improved = True
        sweep += 1
        if not improved and sweep > 3:
            break

    return best_h, best_c5


def resample_to(h_src, n_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src, dtype=float) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst, dtype=float) + 0.5) * 2.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst
