    grad = np.zeros(n)
    if k >= 0:
        # First part: j in [0, n-k-1], grad[j] += (1 - h[j+k])
        if n - k > 0:

        # Second part: j in [0, n-kp-1], grad[j] += -h[j-k] = -h[j+kp]
        grad[0:n-kp] += -h[kp:n]

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = np.min(z) - 2.0
    hi = np.max(z) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-14:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_F(h, n):
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F


def compute_lag_grad(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            grad[0:n - k] += (1.0 - h[k:n])
        if k < n:
            grad[k:n] += -h[0:n - k]
    else:
        kp = -k
        grad[kp:n] += (1.0 - h[0:n - kp])
        grad[0:n - kp] += -h[kp:n]
    return grad


def _safe_sigmoid(x):
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def logit_map(y):
    n = len(y)
    target = n / 2.0
    lo = -60.0
    hi = 60.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = np.sum(_safe_sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    c = 0.5 * (lo + hi)
    return _safe_sigmoid(y + c)


def logit_obj(y, beta):
    h = logit_map(y)
    n = len(h)
    dx = 2.0 / n
    F = compute_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(len(C)) / beta
    return cmax + (mk + np.log(ws)) / beta


def logit_grad(y, beta):
    h = logit_map(y)
    n = len(h)
    dx = 2.0 / n
    F = compute_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    g_h = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        g_h += w * compute_lag_grad(h, k, n)
    g_h *= dx
    w_arr = h * (1.0 - h)
    W = np.sum(w_arr)
    if W < 1e-300:
        return np.zeros(n)
    g_dot_w = np.dot(g_h, w_arr)
    return w_arr * (g_h - g_dot_w / W)


def optimize_logit(h0, beta_list, maxiter=80, deadline=None):
    n = len(h0)
    y0 = _safe_logit(h0)
    best_y = y0.copy()
    best_h = make_feasible(h0.copy())
    best_sc = official_score(best_h)
    for beta in beta_list:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(logit_obj, best_y, args=(beta,), jac=logit_grad,
                           method='L-BFGS-B',
                           options={'maxiter': maxiter, 'gtol': 1e-10,
                                    'ftol': 1e-15, 'maxls': 30})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = make_feasible(logit_map(res.x))
                sc = official_score(h_new)
                if np.isfinite(sc) and sc < best_sc - 1e-15:
                    best_sc = sc
                    best_h = h_new.copy()
                    best_y = res.x.copy()
        except Exception:
            pass
    return best_h, best_sc


def _safe_logit(h):
    h = np.clip(h, 1e-10, 1.0 - 1e-10)
    return np.log(h / (1.0 - h))


def symmetry_transform(h):
    return make_feasible(1.0 - h[::-1].copy())


def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def fourier_basis(n, max_freq=5):
    x = (np.arange(n) + 0.5) * 2.0 / n
    cols = []
    for m in range(1, max_freq + 1):
        cols.append(np.cos(np.pi * m * x))
        cols.append(np.sin(np.pi * m * x))
    B = np.column_stack(cols)
    B -= B.mean(axis=0, keepdims=True)
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-10] = 1.0
    B /= norms
    return B


def active_set_pgd(h, n_iters=200, top_k=8, deadline=None):
    n = len(h)
    dx = 2.0 / n
    h = make_feasible(h.copy())
    best = official_score(h)
    step_base = 0.05
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        F = compute_F(h, n)
        C = F * dx
        cmax = np.max(C)
        if abs(cmax - best) < 1e-14:
            best = cmax
        # Select top lags
        idx_sorted = np.argsort(C)[::-1]
        top_idx = idx_sorted[:top_k]
        # Weighted gradient
        weights = np.zeros(2 * n - 1)
        top_C = C[top_idx]
        w_sum = np.sum(top_C)
        if w_sum < 1e-30:
            break
        for idx in top_idx:
            weights[idx] = top_C[idx] / w_sum
        g = np.zeros(n)
        for idx in top_idx:
            w = weights[idx]
            if w < 1e-14:
                continue
            k = idx - (n - 1)
            g += w * compute_lag_grad(h, k, n)
        g *= dx
        # Projected gradient: find d = clip(-g + lam, -h, 1-h) with sum(d)=0
        lo_l = -10.0
        hi_l = 10.0
        for _ in range(60):
            mid_l = 0.5 * (lo_l + hi_l)
            d = np.clip(-g + mid_l, -h, 1.0 - h)
            s = np.sum(d)
            if s > 0:
                hi_l = mid_l
            else:
                lo_l = mid_l
        lam = 0.5 * (lo_l + hi_l)
        d = np.clip(-g + lam, -h, 1.0 - h)
        # Line search
        best_alpha = 0.0
        best_val = best
        for trial in range(20):
            alpha = step_base * (0.7 ** trial)
            h_trial = make_feasible(h + alpha * d)
            sc = official_score(h_trial)
            if sc < best_val - 1e-15:
                best_val = sc
                best_alpha = alpha
                h = h_trial.copy()
                break
            if alpha < 1e-6:
                break
        if best_alpha == 0.0:
            step_base *= 0.5
            if step_base < 1e-6:
                break
        else:
            step_base = min(step_base * 1.2, 0.1)
            if best_val < best - 1e-14:
                best = best_val
    h = make_feasible(h)
    final_sc = official_score(h)
    if final_sc < best:
        best = final_sc
    return h, best


def direct_solve(h0, maxiter=120, deadline=None):
    n = len(h0)
    dx = 2.0 / n
    score0 = official_score(h0)
    theta0 = score0 + 1e-4
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def d_obj(z):
        return z[-1]

    def d_obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def d_ineq(z):
        h = z[:n]
        F = compute_F(h, n)
        return z[-1] - dx * F

    def d_ineq_jac(z):
        h = z[:n]
        nn = len(h)
        J = np.zeros((2 * nn - 1, nn + 1))
        J[:, -1] = 1.0
        for k in range(-(nn - 1), nn):
            J[k + nn - 1, :nn] = -dx * compute_lag_grad(h, k, nn)
        return J

    def d_eq(z):
        return float(np.sum(z[:n]) - n / 2.0)

    def d_eq_jac(z):
        g = np.ones(n + 1)
        g[-1] = 0.0
        return g

    cons = [
        {'type': 'ineq', 'fun': d_ineq, 'jac': d_ineq_jac},
        {'type': 'eq', 'fun': d_eq, 'jac': d_eq_jac}
    ]
    try:
        res = minimize(d_obj, z0, jac=d_obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            return h_new
    except Exception:
        pass
    return h0
