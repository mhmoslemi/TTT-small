def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def eval_c5_all(h):
    """Return all correlation values (before multiplying by dx) and the C5 value."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c, float(np.max(c)) * dx

def project_feasible(v, target_sum):
    """Project v to [0,1]^n with sum = target_sum using bisection on threshold."""
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_worst_grad(h):
    """Compute worst-case gradient: for each coordinate, the max partial derivative over active lags."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    # Active lags: those within tolerance of max
    tol = max(c_max * 1e-6, 1e-12)
    active = c >= c_max - tol
    
    # For each active lag, compute gradient
    # c[p] = sum_n h[n+m]*(1-h[n]) where m = p-(n-1)
    # dc[p]/dh[i] = (1-h[i-m])*I(0<=i-m<n) - h[i+m]*I(0<=i+m<n)
    worst_grad = np.full(n, -np.inf)
    for p in range(2*n-1):
        if not active[p]:
            continue
        m = p - (n-1)
        for i in range(n):
            grad = 0.0
            # Term 1: h[i] appears as h[n+m] when n = i-m, needs 0 <= i-m < n
            if 0 <= i - m < n:
                grad += (1.0 - h[i - m])
            # Term 2: h[i] appears in (1-h[n]) when n = i, needs 0 <= i+m < n
            if 0 <= i + m < n:
                grad -= h[i + m]
            worst_grad[i] = max(worst_grad[i], grad)
    
    # Fill in -inf with 0 (for coordinates not affected by any active lag)
    worst_grad = np.where(np.isfinite(worst_grad), worst_grad, 0.0)
    return worst_grad, c_max

    δ_max = min(1.0 - h[i], h[j])

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def eval_c5_raw(h):

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_feasible(v, target_sum):
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def pair_directional_deriv(h, i, j):
    """
    Upper bound on directional derivative of C5 in direction (h[j]+=δ, h[i]-=δ).
    Returns max over active lags of (grad_j - grad_i).
    """
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(c))
    tol = max(c_max * 1e-4, 1e-10)
    active = c >= c_max - tol

    for p in range(2 * n - 1):
        if not active[p]:
            continue

        # grad_j = dc[p]/dh[j]
        gj = 0.0

            gj += (1.0 - h[j - m])
        if 0 <= j + m < n:
            gj -= h[j + m]
        # grad_i = dc[p]/dh[i]
        gi = 0.0
        if 0 <= i - m < n:
            gi += (1.0 - h[i - m])
        if 0 <= i + m < n:
            gi -= h[i + m]
        
        dd = gj - gi
        if dd > best_dd:
            best_dd = dd
    
    return best_dd if np.isfinite(best_dd) else 0.0

    """
    Golden section search for optimal δ in: h[j] += δ, h[i] -= δ.
    Returns (new_h, new_c5, improved).
    """

    # h[i] - δ >= 0 => δ <= h[i]
    # h[i] - δ <= 1 => δ >= h[i] - 1
    a = max(-h[j], h[i] - 1.0)
    b = min(1.0 - h[j], h[i])
    
    if b - a < 1e-12:
        return h, eval_c5(h), False
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f(delta):
        h_temp = h.copy()
        h_temp[j] += delta
        h_temp[i] -= delta
        return eval_c5(h_temp)
    
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)
    f2 = f(c2)
    
    best_delta = c1 if f1 < f2 else c2
    best_f = min(f1, f2)
    
    for _ in range(35):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
        
        if f1 < best_f:
            best_f = f1
            best_delta = c1
        if f2 < best_f:
            best_f = f2
            best_delta = c2
        
        if abs(b - a) < 1e-11:
            break
    
    h_new = h.copy()
    h_new[j] += best_delta
    h_new[i] -= best_delta
    h_new[j] = np.clip(h_new[j], 0.0, 1.0)
    h_new[i] = np.clip(h_new[i], 0.0, 1.0)
    
    # Verify

    c5_old = eval_c5(h)
    improved = c5_new < c5_old - 1e-12
    
    return h_new, c5_new, improved

- If n+m = j (Δh = δ) and n = j (Δh = δ): product = δ², requires m = 0
- If n+m = i (Δh = -δ) and n = i (Δh = -δ): product = δ², requires m = 0

-δ² * I(m = j-i) + 2δ² * I(m=0) - δ² * I(m = i-j)

c_new[p] = c[p] + δ*A[p] + δ²*B[p]

B[p] = 2*I(m=0) - I(m=j-i) - I(m=i-j)

(These B values are in {-1, 0, 1, 2}.)

Now, C5_new(δ) = max_p (c[p] + δ*A[p] + δ²*B[p]) * dx

This is the max of a set of quadratic functions, which is a convex function of δ. I can find its minimum by ternary search or by finding the point where the "active" quadratic changes.

Actually, since it's convex, golden section search works perfectly. And I can evaluate C5_new(δ) in O(n) time (just compute all c[p] + δ*A[p] + δ²*B[p] and take the max), avoiding the O(n²) correlation computation!

def compute_pair_coefficients(h, i, j, c):
    """
    Precompute A and B arrays for pair transfer (h[j]+=δ, h[i]-=δ).
    c is the current correlation array.
    Returns (A, B) arrays of length 2n-1.
    """
    n = len(h)
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    
    for p in range(2*n-1):
        m = p - (n-1)
        a_val = 0.0
        if 0 <= j - m < n:
            a_val += (1.0 - h[j - m])
        if 0 <= i - m < n:
            a_val -= (1.0 - h[i - m])
        if 0 <= j + m < n:
            a_val -= h[j + m]
        if 0 <= i + m < n:
            a_val += h[i + m]
        A[p] = a_val
        
        b_val = 0.0
        if m == 0:
            b_val = 2.0
        if m == j - i:
            b_val -= 1.0
        if m == i - j:
            b_val -= 1.0
        B[p] = b_val
    
    return A, B
