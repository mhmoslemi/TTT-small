import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - float(np.max(z))
    hi = 2.0 - float(np.min(z))
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(z + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)


def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)


def coord_descent(h, max_sweeps=200, n_grid=300, deadline=None):
    """Coordinate descent with exact 1D subproblem solution."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    n_shifts = 2 * n - 1
    si_arr = np.arange(n_shifts)
    s_arr = si_arr - (n - 1)
    
    best_score = float(np.max(corr) * dx)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        improved = False
        
        for i in range(n):
            old_val = h[i]
            
            # Compute a_s for this coordinate
            m1 = i - s_arr
            m2 = i + s_arr
            valid1 = (m1 >= 0) & (m1 < n)
            valid2 = (m2 >= 0) & (m2 < n)
            
            a_s = np.zeros(n_shifts)
            if np.any(valid1):
                a_s[valid1] = (1.0 - h[m1[valid1]])
            if np.any(valid2):
                a_s[valid2] += (-h[m2[valid2]])
            
            # b_s (constant part for each shift)
            # For s=0: b_0 = corr[n-1] - old_val*(1-old_val)
            b_0 = corr[n-1] - old_val * (1.0 - old_val)
            # For s≠0: b_s = corr[si] - a_s[si] * old_val
            b_s = corr - a_s * old_val
            b_s[n-1] = b_0
            
            # Grid evaluation
            t_grid = np.linspace(0.0, 1.0, n_grid)
            # Linear max (excluding s=0)
            lin = a_s[None, :] * t_grid[:, None] + b_s[None, :]
            lin[:, n-1] = -np.inf  # exclude quadratic shift from linear max
            lin_max = np.max(lin, axis=1)
            # Quadratic term
            quad_val = -t_grid**2 + t_grid + b_0
            f_vals = np.maximum(lin_max, quad_val) * dx
            
            best_idx = int(np.argmin(f_vals))
            t_new = t_grid[best_idx]
            
            if abs(t_new - old_val) > 1e-10:
                delta = t_new - old_val
                h[i] = t_new
                # Update corr
                corr[n-1] += (t_new - t_new**2) - (old_val - old_val**2)
                mask = np.ones(n_shifts, dtype=bool)
                mask[n-1] = False
                corr[mask] += a_s[mask] * delta
                improved = True
        
        # Project to feasible set
        deficit = target - float(np.sum(h))
        if abs(deficit) > 1e-8:
            h = make_feasible(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        score = float(np.max(corr) * dx)
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_score


def _lse_obj(x, beta, n):
    dx = 2.0 / n
    corr = np.correlate(np.clip(x, 0, 1), 1.0 - np.clip(x, 0, 1), mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return float(max_c + np.log(exp_sum) / beta)
