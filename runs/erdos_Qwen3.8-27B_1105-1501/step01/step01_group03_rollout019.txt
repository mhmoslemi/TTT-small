def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    # Project onto {0 ≤ h ≤ 1, sum(h) = n/2}
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.clip(z, -2, 2)  # reasonable range
    # Bisection on shift to achieve exact sum
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(z + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    # Fix rounding
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        idx = np.argsort(np.abs(h - 0.5))[::-1]  # adjust values furthest from 0.5
        for i in idx:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1:
                d = min(err, 1 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0, 1)

def solve_milp(n, solver='CBC', timeout=120):

    h = cp.Variable(n, boolean=True)

        for i in range(n - m):

            z_vars.append(z)
            z_indices.append((i, m))

def compute_all_g(h, n):
    """Compute all g_k(h) values and their gradients."""
    dx = 2.0 / n
    g = np.zeros(2*n - 1)
    # k ranges from -(n-1) to (n-1), indexed as k + n - 1
    for k in range(-(n-1), n):
        if k >= 0:
            # g_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            overlap = h[k:] * (1-h[:n-k])
        else:

            # g_{-m} = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            overlap = h[:n-m] * (1-h[m:])
        g[k + n - 1] = np.sum(overlap) * dx
    return g

def compute_subgrad(h, n, active_indices):

    dx = 2.0 / n

    """Gradient of g_k with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:

        grad[:] = (1 - 2*h) * dx

Actually, for this problem, the log-sum-exp approximation is:

As β → ∞, f_smooth → max_k g_k(h). The gradient is:

This is a weighted average of the gradients, with more weight on the active constraints.

The issue with large β is that the function becomes very peaked (close to the max), making optimization hard. The issue with small β is that the approximation is poor.

A good strategy:
1. Start with small β (smooth function, easy to optimize)
2. Solve to convergence
3. Increase β
4. Solve again (warm start)
5. Repeat until β is large enough

This is the "homotopy" or "continuation" method.

Actually, for L-BFGS-B, I can handle the sum constraint by:
- After each step, project back to the feasible set (clip to [0,1] and adjust sum)
- This is like a "projected L-BFGS" method

Or better: use the fact that the feasible set is {h : 0 ≤ h ≤ 1, 1^T h = n/2}. The projection onto this set is:
1. Start with h
2. Find α such that sum(clip(h + α, 0, 1)) = n/2 (bisection)
3. Return clip(h + α, 0, 1)

This is exactly the `make_feasible` function! So I can use a projected gradient / projected L-BFGS method.

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -z.max() - 1e-6
    hi = 1.0 - z.min() + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        # Distribute error
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_all_g(h):
    """Compute all correlation values (unscaled by dx)."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def grad_gk(h, k):
    """Gradient of g_k (unscaled) with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1 - 2*h) * dx
    elif k > 0:
        m = k
        if m < n:
            grad[m:] += (1 - h[:n-m]) * dx
            grad[:n-m] += (-h[m:]) * dx
    else:
        m = -k
        if m < n:
            grad[:n-m] += (1 - h[m:]) * dx
            grad[m:] += (-h[:n-m]) * dx
    return grad

def smooth_fn_and_grad(h, beta):
    """Log-sum-exp smoothed objective and gradient."""
    g = compute_all_g(h)
    cmax = np.max(g)
    shifted = beta * (g - cmax)
    shifted = shifted - np.max(shifted)  # numerical stability
    exps = np.exp(shifted)
    ws = np.sum(exps)
    if ws < 1e-300:
        obj = cmax + np.log(len(g)) / beta
        weights = np.ones(len(g)) / len(g)
    else:
        obj = cmax + (np.max(shifted) + np.log(ws)) / beta
        weights = exps / ws
    grad = np.zeros(len(h))
    n = len(h)
    for idx, w in enumerate(weights):
        if w < 1e-15:
            continue
        k = idx - (n - 1)
        grad += w * grad_gk(h, k)
    return obj, grad

def slsqp_solve(h0, beta, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_fn_and_grad(x, beta)[0]
    
    def grad(x):
        return smooth_fn_and_grad(x, beta)[1]
    
    try:
        res = minimize(obj, h0, jac=grad, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except:
        pass
    return h0

def continuation_solve(h0, betas, maxiter_per_beta=200):
    h = h0.copy()
    for beta in betas:
        h = slsqp_solve(h, beta, maxiter=maxiter_per_beta)
    return h

def subgradient_method(h0, n_iters=500, step_init=0.1):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    
    for t in range(n_iters):
        g = compute_all_g(h) * (2.0/n)
        cmax = np.max(g)
        # Active set: constraints within 10% of max
        tol = 0.1 * cmax + 1e-10
        active = np.where(g > cmax - tol)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += grad_gk(h, k)
        grad /= max(len(active), 1)
        
        step = step_init / np.sqrt(t + 1)
        h_new = h - step * grad
        h_new = make_feasible(h_new)
        
        sc = official_score(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score

def milp_binary(n, timeout=60):
    """Try to find a good binary pattern using MILP."""
    if n % 2 != 0:
        return None
    
    dx = 2.0 / n
    h_var = cp.Variable(n, boolean=True)
    t_var = cp.Variable()
    
    cons = [cp.sum(h_var) == n // 2]
    
    # For each shift m = 1..n-1, add constraint t >= c[m]*dx
    # c[m] = sum_{i=m}^{n-1} h[i] - sum_{i=0}^{n-1-m} h[i]*h[i+m]
    # For binary, h[i]*h[i+m] needs auxiliary vars
    
    # To keep it manageable, only use a subset of shifts
    # Actually, let's use all shifts but with z variables
    z_vars = {}
    for m in range(1, n):
        for i in range(n - m):
            z = cp.Variable()
            z_vars[(i, m)] = z
            cons.append(z <= h_var[i])
            cons.append(z <= h_var[i + m])
            cons.append(z >= h_var[i] + h_var[i + m] - 1)
            cons.append(z >= 0)
    
    for m in range(1, n):
        # c[m] = sum_{i=m}^{n-1} h[i] - sum_{i=0}^{n-1-m} z[i,m]
        z_sum_m = sum(z_vars[(i, m)] for i in range(n - m))
        c_m = cp.sum(h_var[m:]) - z_sum_m
        cons.append(t_var >= c_m * dx)
        
        # c[-m] = sum_{i=0}^{n-1-m} h[i] - sum_{i=0}^{n-1-m} z[i,m]
        c_neg_m = cp.sum(h_var[:n-m]) - z_sum_m
        cons.append(t_var >= c_neg_m * dx)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=timeout)
        if h_var.value is not None:
            h_val = np.round(h_var.value)
            # Verify
            return make_feasible(h_val)
    except:
        pass
    
    try:
        prob.solve(solver=cp.SCIP, time_limit=timeout)
        if h_var.value is not None:
            h_val = np.round(h_var.value)
            return make_feasible(h_val)
    except:
        pass
    
    return None
