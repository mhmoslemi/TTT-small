def project_feasible(y, target):

    n = len(y)

            h[i] += add

import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(y, target):
    n = len(y)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if residual > 0:
                d = min(residual, 1.0 - h[idx])
            else:
                d = max(residual, -h[idx])
            h[idx] += d
            residual -= d
            if abs(residual) < 1e-14:
                break
    return h

def compute_lse_grad(h, beta):
    """Compute (phi, grad) for log-sum-exp objective."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    max_c = float(np.max(c))
    shifted = np.clip(beta * (c - max_c), -400.0, 0.0)
    ev = np.exp(shifted)
    es = float(np.sum(ev))
    if es < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
        phi = max_c
    else:
        w = ev / es
        phi = max_c + np.log(es) / beta
    # Gradient
    grad = np.zeros(n)
    wdx = w * dx
    n1 = n - 1
    for ki in range(2*n-1):
        wk = wdx[ki]
        if wk < 1e-15:
            continue
        s = ki - n1
        if s >= 0:
            ell = n - s
            grad[s:] += wk * (1.0 - h[:ell])
            grad[:ell] -= wk * h[s:]
        else:
            t = -s
            ell = n - t
            grad[:ell] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:ell]
    return phi, grad

        if alpha < 1e-12:

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(y, target):
    n = len(y)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        if res > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if res > 0:
                d = min(res, 1.0 - h[idx])
            else:
                d = max(res, -h[idx])
            h[idx] += d
            res -= d
            if abs(res) < 1e-14:
                break
    return h

def lse_obj_grad(h, beta, topk=None):
    n = len(h)
    dx = 2.0 / n

        idx = np.argpartition(c, -topk)[-topk:]
        c_use = c[idx]
        max_c = float(np.max(c_use))
        shifted = np.clip(beta * (c_use - max_c), -400.0, 0.0)
        ev = np.exp(shifted)
