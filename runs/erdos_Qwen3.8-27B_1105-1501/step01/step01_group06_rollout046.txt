```python
def compute_lag_gradients(h, n):
    grads = np.zeros((2*n-1, n))
    for k_idx in range(2*n-1):
        d = k_idx - (n-1)
        if d == 0:
            grads[k_idx, :] = 1.0 - 2.0*h
        elif d > 0:
            # (1-h[j-d]) for j in [d, n-1]
            grads[k_idx, d:] += (1.0 - h[:-d])

            grads[k_idx, m:] -= h[:n-m]
```

```python
def lp_direction(h, n, radius, lam, top_k=None):
    """Find LP trust-region direction.
    Minimizes tau + lam*r subject to:
      grad_l · p - tau <= -score_l for selected lags
      |p_i| <= r
      |p_i| <= radius
      sum(p) = 0
      h_i + p_i in [0, 1]
    """
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx  # these are the actual C5 values per lag
    
    # Select top-k lags (active constraints)
    if top_k is not None and top_k < 2*n-1:
        # Select top-k by score value
        thresh_idx = sorted(range(2*n-1), key=lambda i: scores[i], reverse=True)[:top_k]
    else:
        thresh_idx = list(range(2*n-1))
    
    n_constraints = len(thresh_idx)
    # Variables: p (n), r (1), tau (1) → total n+2
    # Objective: minimize tau + lam*r
    c = np.zeros(n+2)
    c[n] = lam  # r
    c[n+1] = 1.0  # tau
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each selected lag l: grad_l · p - tau <= -score_l
    # i.e., [grad_l, 0, -1] @ [p, r, tau] <= -score_l
    # Also: p_i - r <= 0 and -p_i - r <= 0 for each i
    # Also: p_i <= radius and -p_i <= radius
    
    rows = []
    rhs = []
    G = compute_lag_gradients(h, n)
    
    for l in thresh_idx:
        row = np.zeros(n+2)
        row[:n] = G[l, :]
        row[n+1] = -1.0
        rows.append(row)
        rhs.append(-scores[l])
    
    # p_i - r <= 0
    for i in range(n):
        row = np.zeros(n+2)
        row[i] = 1.0
        row[n] = -1.0
        rows.append(row)
        rhs.append(0.0)
        # -p_i - r <= 0
        row2 = np.zeros(n+2)
        row2[i] = -1.0
        row2[n] = -1.0
        rows.append(row2)
        rhs.append(0.0)
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n+2))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i])
        hi_i = min(radius, 1.0 - h[i])
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, radius))  # r
    bounds.append((None, None))  # tau is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        p = res.x[:n]
        return p
    return None
```

```python
            radius *= 0.7
            if radius < min_radius:

        else:

            if radius < min_radius:
```
