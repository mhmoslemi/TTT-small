import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_target(v, target):
    v = np.asarray(v, dtype=float)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fine-tune for exact sum
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                amt = residual / cnt
                x[mask] = np.minimum(x[mask] + amt, 1.0)
        else:
            mask = x > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                amt = (-residual) / cnt
                x[mask] = np.maximum(x[mask] - amt, 0.0)
    return x

def compute_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_all(h):
    """Return array of shape (2n-1, n) with gradients of each score w.r.t. h."""
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        idx = k + (n - 1)
        g = abs(k)
        if g == 0:
            grads[idx] = 1.0 - 2.0 * h
        elif k > 0:
            # S_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[j]: (1-h[j-k]) if k<=j<n, and -h[j+k] if j<n-k
            j_range1 = np.arange(k, n)
            grads[idx, j_range1] += (1.0 - h[j_range1 - k])
            j_range2 = np.arange(n - k)
            grads[idx, j_range2] += (-h[j_range2 + k])
        else:
            # S_{-g} = sum_{i=0}^{n-1-g} h[i]*(1-h[i+g])
            # d/dh[j]: (1-h[j+g]) if j<n-g, and -h[j-g] if j>=g
            j_range1 = np.arange(n - g)
            grads[idx, j_range1] += (1.0 - h[j_range1 + g])
            j_range2 = np.arange(g, n)
            grads[idx, j_range2] += (-h[j_range2 - g])
    return grads

def solve_lp(h, target, radius, all_constraints=True):
    n = len(h)
    scores = compute_scores(h)
    grads = compute_grad_all(h)
    
    n_lags = 2 * n - 1
    if all_constraints:
        indices = np.arange(n_lags)
    else:
        # Use top-K active
        sorted_idx = np.argsort(-scores)
        indices = sorted_idx[:min(n_lags, 30)]
    
    # LP: min tau subject to grad_l @ p - tau <= -score_l, sum(p)=0, bounds on p
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(indices), n + 1))
    b_ub = np.zeros(len(indices))
    for i, l in enumerate(indices):
        A_ub[i, :n] = grads[l]
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[l]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    lb = np.maximum(-radius, -h)
    ub = np.minimum(radius, 1.0 - h)
    bounds = list(zip(lb, ub)) + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], scores
        return None, scores
    except Exception:
        return None, scores

def subgrad_direction(h, radius):
    """Subgradient of max as fallback direction."""
    n = len(h)
    scores = compute_scores(h)
    grads = compute_grad_all(h)
    s_max = np.max(scores)
    # Active set: scores within 1e-6 of max
    active = np.where(scores >= s_max - 1e-6)[0]
    g = np.mean(grads[active], axis=0)
    p = -g - np.mean(-g)  # descent direction with zero sum
    p_norm = np.max(np.abs(p))

    p = p / p_norm * radius
