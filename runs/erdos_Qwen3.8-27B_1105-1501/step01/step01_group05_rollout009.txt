def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(1000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h

def gen_interpolated(parent, n, offset=0):

    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])

def compute_active_direction(h, n, dx, top_k=8, temperature=None):
    """Compute weighted gradient from top active constraints."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    theta = np.max(corr)
    if temperature is None:
        temperature = max(0.003 + 0.015 * theta, 0.002)
    
    weights = np.exp((corr - theta) / temperature)
    total_w = np.sum(weights)
    if total_w < 1e-15:
        return np.zeros(n)
    weights /= total_w
    
    # Select top constraints
    top_indices = np.argsort(corr)[-top_k:]
    
    g = np.zeros(n)
    for ki in top_indices:
        lag = ki - (n - 1)
        w = weights[ki]
        # Gradient of C_lag with respect to h
        # C_lag = dx * sum_i h[i+lag] * (1-h[i]) over valid i
        start = max(0, -lag)
        end = min(n, n - lag)
        for i in range(start, end):
            p = i + lag
            g[p] += w * (1.0 - h[i])
            g[i] += w * (-h[p])
    g *= dx
    return g

            d = c
            fd = fc
            c = b - gr * (b - a)

            h[j] = S - c

            fc = fd

    """SLSQP refinement with only top active constraints."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1
    
    # Determine active constraints
    corr = np.correlate(h_init, 1.0 - h_init, mode="full") * dx
    top_mask = np.argsort(corr)[-max_active:]
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def eq_fun(x):
        return np.array([np.sum(x[:n]) - target_sum])
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    # Inequality constraints for top active lags
    for ki in top_mask:
        lag = lags[ki]
        start = max(0, -lag)
        end = min(n, n - lag)
        
        def ineq_fun(x, ki=ki, lag=lag, start=start, end=end):
            h_x = x[:n]
            theta = x[n]
            val = 0.0
            for i in range(start, end):
                p = i + lag
                val += h_x[p] * (1.0 - h_x[i])
            return np.array([theta - dx * val])
        
        def ineq_jac(x, ki=ki, lag=lag, start=start, end=end):
            h_x = x[:n]
            j = np.zeros(n + 1)
            j[n] = 1.0
            for i in range(start, end):
                p = i + lag
                j[p] += -dx * (1.0 - h_x[i])
                j[i] += -dx * h_x[p]
            return j
        
        constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
    
    initial_score = official_score(h_init)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h_init, [theta0]])
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-9}

            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h_init.copy(), initial_score

import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(2000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    return np.clip(h, 0.0, 1.0)


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def gen_random_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 0:
        h[idx[m_full]] = frac
    return repair_vector(h, n)


def gen_random_continuous(n, rng):
    h = rng.uniform(0, 1, n)
    return repair_vector(h, n)


def gen_step_family(n, rng, num_steps):
    if num_steps >= n:
        h = rng.uniform(0, 1, n)
        return repair_vector(h, n)
    boundaries = sorted(rng.choice(n - 1, size=num_steps - 1, replace=False))
    bl = [0] + list(boundaries) + [n]
    vals = rng.uniform(0, 1, num_steps)
    h = np.zeros(n)
    for i in range(num_steps):
        h[bl[i]:bl[i + 1]] = vals[i]
    return repair_vector(h, n)


def compute_weighted_grad(h, n, dx, top_k=10, temperature=0.005):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    theta = np.max(corr)
    T = max(0.002, temperature + 0.01 * theta)
    weights = np.exp((corr - theta) / T)
    tw = np.sum(weights)
    if tw < 1e-15:
        return np.zeros(n), theta
    weights /= tw
    top_idx = np.argsort(corr)[-top_k:]
    g = np.zeros(n)
    for ki in top_idx:
        lag = ki - (n - 1)
        w = weights[ki]
        start = max(0, -lag)
        end = min(n, n - lag)
        for i in range(start, end):
            p = i + lag
            g[p] += w * (1.0 - h[i])
            g[i] += w * (-h[p])
    g *= dx
    return g, theta


def pair_minimize_1d(h, i, j, n, dx, n_grid=13, n_golden=10):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-12:
        return h[i], official_score(h)
    cur_score = official_score(h)
    best_x = h[i]
    best_s = cur_score
    xs = np.linspace(L, U, n_grid)
    for x in xs:
        h[i] = x
        h[j] = S - x
        s = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
        if s < best_s:
            best_s = s
            best_x = x
    idx_best = int(np.argmin(np.abs(xs - best_x)))
    if 0 < idx_best < n_grid - 1:
        ga, gb = xs[idx_best - 1], xs[idx_best + 1]
    elif idx_best == 0:
        ga, gb = L, xs[min(2, n_grid - 1)]
    else:
        ga, gb = xs[max(0, n_grid - 3)], U
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = ga, gb
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] = c
    h[j] = S - c
    fc = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    h[i] = d
    h[j] = S - d
    fd = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    for _ in range(n_golden):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[i] = c
            h[j] = S - c
            fc = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[i] = d
            h[j] = S - d
            fd = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    for xc in [a, b, c, d, best_x, 0.5 * (L + U)]:
        xc = float(np.clip(xc, L, U))
        h[i] = xc
        h[j] = S - xc
        s = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
        if s < best_s:
            best_s = s
            best_x = xc
    h[i] = best_x
    h[j] = S - best_x
    return best_x, best_s


def local_pair_descent(h_init, n, time_limit=60.0, seed=42, verbose=False):
    rng = np.random.default_rng(seed)
    h = h_init.copy()
    dx = 2.0 / n
    score = official_score(h)
    start_time = time.perf_counter()
    no_improve = 0
    max_no_improve = 40
    pass_count = 0

    while time.perf_counter() - start_time < time_limit:
        pass_count += 1
        no_improve += 1
        if no_improve > max_no_improve:
            break
        g, theta = compute_weighted_grad(h, n, dx, top_k=8, temperature=0.004)
        pairs = set()
        pos_idx = np.where(g > 0.005)[0]
        neg_idx = np.where(g < -0.005)[0]
        for pi in pos_idx[:12]:
            for ni in neg_idx[:12]:
                if pi != ni:
                    pairs.add((int(pi), int(ni)))
        for i in range(n):
            pairs.add((i, (i + 1) % n))
        for _ in range(40):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((i, j))
        pair_list = list(pairs)
        rng.shuffle(pair_list)
        max_pairs = min(len(pair_list), 80)
        for k in range(max_pairs):
            if time.perf_counter() - start_time > time_limit:
                break
            i, j = pair_list[k]
            h_saved = h.copy()
            _, s_new = pair_minimize_1d(h, i, j, n, dx, n_grid=11, n_golden=8)
            if s_new < score - 1e-13:
                score = s_new
                no_improve = 0
            else:
                h[:] = h_saved
        if no_improve >= 3 and no_improve < max_no_improve:
            g2, _ = compute_weighted_grad(h, n, dx, top_k=12, temperature=0.003)
            direction = -g2
            direction -= np.mean(direction)
            nrm = np.linalg.norm(direction)
            if nrm > 1e-10:
                direction /= nrm
                for alpha in [0.08, 0.15, 0.25, 0.4, -0.08, -0.15]:
                    h_try = repair_vector(h + alpha * direction * 0.1, n)
                    s_try = official_score(h_try)
                    if s_try < score - 1e-12:
                        h = h_try
                        score = s_try
                        no_improve = 0
                        break
    return h, official_score(h)


def slsqp_refine(h_init, n, time_limit=20.0, max_active=12):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    corr = np.correlate(h_init, 1.0 - h_init, mode="full") * dx
    top_mask = np.argsort(corr)[-max_active:]

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def eq_fun(x):
        return np.array([np.sum(x[:n]) - target_sum])

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    for ki in top_mask:
        lag = int(lags[ki])
        start = max(0, -lag)
        end = min(n, n - lag)

        def make_ineq(lag_v, st, en):
            def fun(x):
                hx = x[:n]
                theta = x[n]
                val = 0.0
                for m in range(st, en):
                    p = m + lag_v
                    val += hx[p] * (1.0 - hx[m])
                return np.array([theta - dx * val])
            def jac(x):
                hx = x[:n]
                j = np.zeros(n + 1)
                j[n] = 1.0
                for m in range(st, en):
                    p = m + lag_v
                    j[p] += -dx * (1.0 - hx[m])
                    j[m] += -dx * hx[p]
                return j
            return fun, jac

        f_i, j_i = make_ineq(lag, start, end)
        constraints.append({'type': 'ineq', 'fun': f_i, 'jac': j_i})

    initial_score = official_score(h_init)
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h_init, [theta0]])

    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': 60, 'ftol': 1e-9}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h_init.copy(), initial_score
