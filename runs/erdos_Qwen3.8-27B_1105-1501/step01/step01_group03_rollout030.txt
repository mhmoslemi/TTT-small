            p_end = min(n, end + k)
            if p_start < p_end:

        # Direction: -grad, but we need to maintain sum(h) = n/2 and 0<=h<=1
        # Use a simple step with projection
        step = 0.1 / (np.max(np.abs(grad)) + 1e-10)
        h_new = h - step * grad
        h_new = make_feasible(h_new)
        
        sc_new = official_score(h_new)
        sc_old = official_score(h)
        if sc_new < sc_old:
            h = h_new
        else:
            # Try smaller step
            step *= 0.5
            h_new = h - step * grad
            h_new = make_feasible(h_new)
            sc_new = official_score(h_new)
            if sc_new < sc_old:
                h = h_new
            else:
                break
    
    return h

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def safe_sigmoid(x):
    out = np.zeros_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def safe_logit(p):
    p = np.asarray(p, dtype=float)
    p = np.clip(p, 1e-14, 1.0 - 1e-14)
    return np.log(p / (1.0 - p))

def find_c(y, target):
    n = len(y)
    lo, hi = -60.0, 60.0
    while safe_sigmoid(y + lo).sum() > target and lo > -200:
        lo -= 30.0

def compute_all_F(h, n):
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F

def smooth_max_obj_grad_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = compute_all_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        obj = cmax
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        obj = cmax + (mk + np.log(ws)) / beta
        weights = es / ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[k + n - 1]
        if w < 1e-14:

def logit_obj_grad(y, beta):
    n = len(y)
    target = n / 2.0
    c = find_c(y, target)
    h = safe_sigmoid(y + c)
    h = np.clip(h, 1e-12, 1.0 - 1e-12)

    g_w = np.dot(g_h, w)

    """Momentum-based optimization on logit manifold."""
