```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def exact_local_search(h, time_budget, seed=42):
    """Local search via pair swaps with exact C5 evaluation."""
    n = len(h)
    target = float(n) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    pairs = np.array([(i, j) for i in range(n) for j in range(i+1, n)])
    n_pairs = len(pairs)
    
    # Try a set of delta values
    deltas = np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.5])
    
    stall = 0
    max_stall = 50
    
    while time.time() - t0 < time_budget:
        # Shuffle pair order
        order = rng.permutation(n_pairs)
        improved = False
        
        for idx in order:
            if time.time() - t0 > time_budget:
                break
            i, j = pairs[idx]
            
            # Try moving weight from j to i and vice versa
            for sign in [1, -1]:
                for d in deltas:
                    delta = sign * d
                    new_hi = best_h[i] + delta
                    new_hj = best_h[j] - delta
                    if new_hi < -0.01 or new_hi > 1.01 or new_hj < -0.01 or new_hj > 1.01:
                        continue
                    new_h = best_h.copy()
                    new_h[i] = np.clip(new_hi, 0, 1)
                    new_h[j] = np.clip(new_hj, 0, 1)
                    # Check if sum is still valid (clipping might break it)
                    if abs(np.sum(new_h) - target) > 1e-6:

                    if c5 < best_c5 - 1e-12:
```

```python
    c = np.correlate(h, 1-h, mode='full'), length 2n-1.
    c[k] = sum_{m in V_k} h[m] * (1-h[m+s_k]) where s_k = k-(n-1)

    for k in range(2*n-1):
        s = k - (n - 1)
        # Check if i is in V_k
        if 0 <= i < n and 0 <= i+s < n:
            A[k] += (1.0 - h[i+s])
        # Check if i-s is in V_k (i.e., the term where m+s = i, so m = i-s)
        m1 = i - s
        if 0 <= m1 < n and 0 <= m1+s < n:
            A[k] -= h[m1]
        # Check if j is in V_k
        if 0 <= j < n and 0 <= j+s < n:
            A[k] -= (1.0 - h[j+s])
        # Check if j-s is in V_k
        m2 = j - s
        if 0 <= m2 < n and 0 <= m2+s < n:
            A[k] += h[m2]
    return A
```

This is O(n) per pair. With ~5000 pairs and n=99, that's about 500K operations per full sweep, which is fast.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    T1 = np.convolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2

def thue_morse_init(n):
    """Generate Thue-Morse based initialization."""
    t = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Scale to have sum = n/2
    # Current sum is approximately n/2 already for large n
    # But let's project to be safe
    return project_to_simplex(t, float(n)/2.0)

def random_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    if n % 2 == 1:
        h[idx[n//2]] = 0.5
    return h

def gradient_descent(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = project_to_simplex(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    alpha = 10.0
    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    
    while time.time() - t0 < time_budget:
        for _ in range(100):
            if time.time() - t0 >= time_budget:
                break
            grad = smooth_max_grad(h, alpha)
            v = momentum * v - lr * grad
            h_new = project_to_simplex(h + v, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            h = h_new
        
        if stall > 300:
            rng = np.random.RandomState(seed + stall)
            h = best_h + 0.03 * rng.randn(n)
            h = project_to_simplex(h, target)
            v = np.zeros(n)
            lr = max(lr * 0.8, 0.001)
            stall = 0
        
        alpha = min(alpha * 1.03, 200.0)
        lr = max(lr * 0.995, 0.0001)
    
    return best_h, best_c5

def compute_A(h, i, j, n):
    """Compute A[k] = dc[k]/dδ for h[i]+=δ, h[j]-=δ."""
    A = np.zeros(2*n-1)
    for k in range(2*n-1):
        s = k - (n-1)
        # h[i] appears as h[m] when m=i is in V_k: need 0<=i<n, 0<=i+s<n
        if 0 <= i < n and 0 <= i+s < n:
            A[k] += (1.0 - h[i+s])
        # h[i] appears as (1-h[m+s]) when m+s=i, m=i-s: need 0<=i-s<n, 0<=i<n
        m1 = i - s
        if 0 <= m1 < n and 0 <= i < n:
            A[k] -= h[m1]
        # h[j] appears as h[m] when m=j is in V_k
        if 0 <= j < n and 0 <= j+s < n:
            A[k] -= (1.0 - h[j+s])
        # h[j] appears as (1-h[m+s]) when m+s=j, m=j-s
        m2 = j - s
        if 0 <= m2 < n and 0 <= j < n:
            A[k] += h[m2]
    return A
