def compute_all_gradients(h, n):
    """Compute gradient of each lag score w.r.t. h. Returns array of shape (2n-1, n)."""
    grads = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        if d >= 0:
            # score = sum_{k=0}^{n-1-d} h[k]*(1-h[k+d])
            # grad[j] = (1-h[j+d]) for j in [0, n-1-d], minus h[j-d] for j in [d, n-1]
            if d == 0:
                grads[r] = 1.0 - 2.0 * h
            else:
                # First term: (1-h[j+d]) for j = 0, ..., n-1-d
                idx1 = np.arange(0, n-d)
                grads[r, idx1] += (1.0 - h[idx1 + d])
                # Second term: -h[j-d] for j = d, ..., n-1
                idx2 = np.arange(d, n)
                grads[r, idx2] -= h[idx2 - d]
        else:
            g = -d
            # score = sum_{k=g}^{n-1} h[k]*(1-h[k-g])
            # grad[j] = (1-h[j-g]) for j = g, ..., n-1, minus h[j+g] for j = 0, ..., n-1-g
            idx1 = np.arange(g, n)
            grads[r, idx1] += (1.0 - h[idx1 - g])
            idx2 = np.arange(0, n-g)
            grads[r, idx2] -= h[idx2 + g]
    return grads

def lp_direction(h, n, score, grads_top, radius, lam):
    """
    Solve LP to find direction p that reduces the max score.
    score: current scores (2n-1,)
    grads_top: gradients of top active lags, shape (K, n)
    Returns direction p or None if LP fails.
    """
    K = grads_top.shape[0]
    # Variables: [p(0..n-1), r, tau] -> total n+2
    c = np.zeros(n + 2)
    c[n] = lam      # coefficient of r
    c[n+1] = 1.0    # coefficient of tau
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag: grad_k · p - tau <= -score_k
    # For each j: p_j - r <= 0, -p_j - r <= 0
    rows = []
    rhs = []
    for k in range(K):
        row = np.zeros(n + 2)
        row[:n] = grads_top[k]
        row[n+1] = -1.0
        rows.append(row)
        rhs.append(-score[k])
    
    for j in range(n):
        row = np.zeros(n + 2)
        row[j] = 1.0
        row[n] = -1.0
        rows.append(row)
        rhs.append(0.0)
        
        row2 = np.zeros(n + 2)
        row2[j] = -1.0
        row2[n] = -1.0
        rows.append(row2)
        rhs.append(-0.0)  # -p_j - r <= 0
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, radius))  # r in [0, radius]
    bounds.append((None, None))   # tau unbounded
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    p = res.x[:n]
    # Ensure sum(p) = 0 exactly
    p -= np.mean(p)
    # Clip to bounds
    for j in range(n):
        p[j] = max(-min(radius, h[j]), min(min(radius, 1.0-h[j]), p[j]))
    # Re-fix sum
    p -= np.mean(p)
    
    return p

import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project(v, target):
    n = len(v)
    v = np.nan_to_num(np.asarray(v, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.clip(v - theta, 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_grads(h, n):
    grads = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        if d >= 0:
            if d == 0:
                grads[r] = 1.0 - 2.0*h
            else:
                idx1 = np.arange(0, n-d)
                grads[r, idx1] += 1.0 - h[idx1 + d]
                idx2 = np.arange(d, n)
                grads[r, idx2] -= h[idx2 - d]
        else:
            g = -d
            idx1 = np.arange(g, n)
            grads[r, idx1] += 1.0 - h[idx1 - g]
            idx2 = np.arange(0, n-g)
            grads[r, idx2] -= h[idx2 + g]
    return grads

def lp_direction(h, n, scores, grads, radius, lam, top_k):
    # Select top-k active lags
    idx = np.argsort(scores)[::-1][:top_k]
    scores_top = scores[idx]
    grads_top = grads[idx]
    K = len(idx)
    
    # Variables: [p(0..n-1), r, tau] -> n+2 variables
    c = np.zeros(n + 2)
    c[n] = lam
    c[n+1] = 1.0
    
    A_ub_rows = []
    b_ub = []
    
    # Active lag constraints: grad_k . p - tau <= -score_k
    for k in range(K):
        row = np.zeros(n + 2)
        row[:n] = grads_top[k]
        row[n+1] = -1.0
        A_ub_rows.append(row)
        b_ub.append(-scores_top[k])
    
    # p_j - r <= 0
    for j in range(n):
        row = np.zeros(n + 2)
        row[j] = 1.0
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub.append(0.0)
        row2 = np.zeros(n + 2)
        row2[j] = -1.0
        row2[n] = -1.0
        A_ub_rows.append(row2)
        b_ub.append(-0.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    
    A_eq = np.zeros((1, n+2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0-h[j])))
    bounds.append((0.0, radius))
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if not res.success:
        return None
    
    p = res.x[:n].copy()
    p -= np.mean(p)
    # Re-clip
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0-h[j])
        p[j] = np.clip(p[j], lo, hi)
    p -= np.mean(p)
    
    if np.max(np.abs(p)) < 1e-10:
        return None
    return p

def line_search(h, p, n):
    alpha_max = np.inf
    pos = p > 1e-12
    neg = p < -1e-12
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-p[neg]))))
    if not np.isfinite(alpha_max) or alpha_max < 1e-14:
        return 0.0, eval_c5(h)
    alpha_max = min(alpha_max, 2.0)
    
    # Grid search
    n_grid = 25
    alphas = np.linspace(0.0, alpha_max, n_grid)
    best_alpha = 0.0
    best_c5 = eval_c5(h)
    for a in alphas[1:]:
        h_try = h + a * p
        c5_try = eval_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_alpha = a
    
    # Golden section if interior
    if 0 < best_alpha < alpha_max:
        idx = int(round(best_alpha / alpha_max * (n_grid-1)))
        lo_a = alphas[max(0, idx-2)]
        hi_a = alphas[min(n_grid-1, idx+2)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(15):
            a1 = hi_a - gr*(hi_a - lo_a)
            a2 = lo_a + gr*(hi_a - lo_a)
            c1 = eval_c5(h + a1*p)
            c2 = eval_c5(h + a2*p)
            if c1 <= c2:
                hi_a = a2
            else:
                lo_a = a1
        mid = (lo_a + hi_a) / 2.0
        c_mid = eval_c5(h + mid*p)
        if c_mid < best_c5:
            best_c5 = c_mid
            best_alpha = mid
    
    return best_alpha, best_c5

def lp_optimize(h, n, target, max_iter=200, t_start=None, deadline_frac=1.0, budget=1100.0):
    radius = 0.15
    lam = 1.0
    top_k = 2*n - 1  # use all lags initially
    best_h = h.copy()
    best_c5 = eval_c5(h)
    consecutive_fails = 0
    
    for it in range(max_iter):
        if t_start is not None:
            if time.time() > t_start + deadline_frac * budget:
                break
        scores = compute_scores(h)
        grads = compute_all_grads(h, n)
        p = lp_direction(h, n, scores, grads, radius, lam, top_k)
        if p is None:
            consecutive_fails += 1
            radius *= 0.7
            if consecutive_falls > 8 or radius < 1e-6:
                break
            continue
        
        alpha, c5_new = line_search(h, p, n)
        if c5_new < best_c5 - 1e-12:
            h = h + alpha * p
            h = project(h, target)
            best_h = h.copy()
            best_c5 = eval_c5(h)
            consecutive_fails = 0
            # Restorse radius or slightly increase
            radius = min(radius * 1.1, 0.3)
        else:
            consecutive_fails += 1
            radius *= 0.7
            if consecutive_fails > 10 or radius < 1e-6:
                break
    
    return best_h, best_c5

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def perturb(h, rng, scale):
    h_p = h.copy()
    h_p += rng.normal(0, scale, len(h))
    return project(h_p, len(h) / 2.0)
