    """Compute gradient matrix G where G[k, j] = d(corr[k])/d(h[j])"""
    n = len(h)

        h_pert = best_h + rng.normal(0, ps, best_n)
        # Zero-sum perturbation
        h_pert -= np.mean(h_pert - best_h)  # ensure same mean
        h_pert = project_to_sum(h_pert, target0)
        
        deadline_sub = min(time.time() + budget_s * 0.02, deadline2)
        h_new, c5_new = lp_optimize(h_pert, target0, 50, 0.03, 2.0, deadline_sub)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 3: Try different n values
    deadline3 = t0 + budget_s * 0.95
    for delta_n in [1, -1, 2, -2, 3, -3]:
        if time.time() > deadline3:
            break
        n_try = best_n + delta_n
        if n_try < 20 or n_try > 128:
            continue
        target_try = n_try / 2.0
        h_try = interpolate_to_n(best_h, n_try)
        deadline_sub = min(time.time() + budget_s * 0.03, deadline3)
        h_opt, c5_opt = lp_optimize(h_try, target_try, 100, 0.04, 2.0, deadline_sub)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
    
    # Final
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_c5 = exact_c5(best_h)
    return best_h, best_c5, best_n
