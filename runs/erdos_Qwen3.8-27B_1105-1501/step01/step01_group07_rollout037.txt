    g = S * dx

    ex = np.exp(g_shifted / T)

```python
def pair_optimize_fast(h, i, j, n_grid=30):
    """Find optimal transfer from j to i using fast per-lag updates."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # raw correlation (no dx)
    
    # Compute v_d = change in c_d per unit transfer from j to i
    # v_d = (1-h[i-d])*I - h[i+d]*I - (1-h[j-d])*I + h[j+d]*I
    v = np.zeros(2*n-1)
    for d in range(-(n-1), n):
        val = 0.0
        id_ = i - d
        idp = i + d
        jd_ = j - d
        jdp = j + d
        if 0 <= id_ <= n-1:
            val += (1.0 - h[id_])
        if 0 <= idp <= n-1:
            val -= h[idp]
        if 0 <= jd_ <= n-1:
            val -= (1.0 - h[jd_])
        if 0 <= jdp <= n-1:
            val += h[jdp]
        v[n-1+d] = val
    
    # Valid range for a: h[i]+a in [0,1] and h[j]-a in [0,1]
    a_lo = max(-h[i], h[j] - 1.0)
    a_hi = min(1.0 - h[i], h[j])
    
    if a_hi - a_lo < 1e-14:
        return 0.0, score_exact(h)
    
    def obj(a):
        return float(np.max(c + a * v) * dx)
    
    # Grid search
    best_a = 0.0
    best_val = obj(0.0)  # current value
    for a in np.linspace(a_lo, a_hi, n_grid):
        val = obj(a)
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    if abs(best_a) > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = max(a_lo, best_a - 2.0*(a_hi-a_lo)/n_grid)
        hi = min(a_hi, best_a + 2.0*(a_hi-a_lo)/n_grid)
        for _ in range(30):
            a1 = lo + (1.0-gr)*(hi-lo)
            a2 = lo + gr*(hi-lo)
            if obj(a1) < obj(a2
