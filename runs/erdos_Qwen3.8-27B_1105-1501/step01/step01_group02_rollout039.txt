```python
def fractional_local_search(h, max_time=50.0, n_samples=30):
    n = len(h)
    target_sum = n / 2.0
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    while time.time() - t0 < max_time:
        improved = False
        for i in range(n):
            if h[i] < 1e-10:
                continue
            for j in range(n):
                if i == j or h[j] > 1.0 - 1e-10:
                    continue
                # Try moving d from i to j
                d_max = min(h[i], 1.0 - h[j])
                if d_max < 1e-10:
                    continue
                ds = np.linspace(0, d_max, n_samples)
                best_d = 0.0
                best_c5_trial = compute_c5(best_h)
                for d in ds:
                    if d == 0:
                        continue
                    best_h[i] -= d
                    best_h[j] += d
                    c5 = compute_c5(best_h)
                    if c5 < best_c5_trial:
                        best_c5_trial = c5
                        best_d = d
                if best_d > 0:
                    best_h[i] -= best_d
                    best_h[j] += best_d
                    best_c5 = best_c5_trial
                    improved = True
                else:
                    best_h[i] += best_d  # restore (best_d=0 means no change)
                    best_h[j] -= best_d
        if not improved:
            break
    
    return best_h, best_c5
```
