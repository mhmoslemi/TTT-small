def stochastic_sa(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = project_sum(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    temp = 0.005

    while time.time() - t0 < time_budget:
        for _ in range(300):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(-0.15, 0.15)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_sum(h_new, target)
            c5_new = compute_c5(h_new)
            diff = c5_new - best_c5
            if diff < 0 or rng.random() < np.exp(-diff / temp):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        temp *= 0.97

    return best_h, best_c5


def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = n // 2
    h[:ones] = 1.0
    if n % 2 == 1:
        h[ones] = 0.5
    idx = rng.permutation(n)
    return h[idx]


def block_init(n, rng):
    h = np.zeros(n)
    ones = n // 2
    blocks = rng.randint(3, 8)
    block_size = max(1, ones // blocks)
    positions = sorted(rng.choice(n, size=blocks, replace=False))
    for pos in positions:
        start = pos
        for d in range(block_size):
            h[(start + d) % n] = 1.0
    if n % 2 == 1 and np.sum(h) == ones:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = 0.5
    return h


def low_freq_init(n, rng):
    h = np.zeros(n)
    freq = rng.uniform(0.3, 1.5)
    phase = rng.uniform(0, 2 * np.pi)
    for i in range(n):
        h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / n + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_sum(h, float(n) / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    # Phase 1: Gradient opt on initial_h_values (low alpha for broad search)
    if h_init is not None and remaining() > 60:
        n0 = len(h_init)
        alloc = min(remaining() * 0.25, 250.0)
        h_best, c5_best = optimize_gradient(h_init, alloc, seed, alpha0=2.0, alpha_growth=1.06, lr0=0.02, lr_decay=0.98)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            best_n = n0

    # Phase 2: High alpha sharpening on best
    if best_h is not None and remaining() > 40:
        alloc = min(remaining() * 0.15, 150.0)
        h_best, c5_best = optimize_gradient(best_h, alloc, seed + 1, alpha0=30.0, alpha_growth=1.03, lr0=0.008, lr_decay=0.99, mom0=0.9, mom_growth=0.001)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()

    # Phase 3: Penalty-based refinement
    if best_h is not None and remaining() > 40:
        alloc = min(remaining() * 0.12, 120.0)
        h_best, c5_best = optimize_penalty(best_h, alloc, seed + 2)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()

    # Phase 4: Stochastic annealing
    if best_h is not None and remaining() > 40:
        alloc = min(remaining() * 0.1, 100.0)
        h_best, c5_best = stochastic_sa(best_h, alloc, seed + 3)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()

    # Phase 5: Try other n values
    n_candidates = [31, 47, 63, 81, 99, 127]
    for n in n_candidates:
        if remaining() < 25:
            break
        alloc = min(remaining() * 0.07, 60.0)
        if alloc < 12:
            break

        for init_type in range(3):
            if remaining() < 12:
                break
            s = seed * 1000 + n * 10 + init_type
            local_rng = np.random.RandomState(s)
            if init_type == 0:
                h_new = random_binary_init(n, local_rng)
            elif init_type == 1:
                h_new = block_init(n, local_rng)
            else:
                h_new = low_freq_init(n, local_rng)
            h_new = project_sum(h_new, float(n) / 2.0)
            sub_alloc = min(alloc / 3.0, remaining() - 8.0)
            if sub_alloc < 6:
                break
            h_best, c5_best = optimize_gradient(h_new, sub_alloc, s, alpha0=4.0, alpha_growth=1.05)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
                best_n = n

    # Phase 6: Local search refinement on global best
    if best_h is not None and remaining() > 30:
        alloc = min(remaining() * 0.15, 150.0)
        h_best, c5_best = local_search(best_h, alloc, seed + 4)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()

    # Phase 7: Final gradient polish with very sharp max
    if best_h is not None and remaining() > 20:
        alloc = min(remaining() * 0.1, 80.0)
        h_best, c5_best = optimize_gradient(best_h, alloc, seed + 5, alpha0=100.0, alpha_growth=1.01, lr0=0.003, lr_decay=0.995, mom0=0.92)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()

    # Final projection and evaluation
    if best_h is not None:
        best_h = project_sum(best_h, float(best_n) / 2.0)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        n = 99
        best_h = random_binary_init(n, rng)
        best_h = project_sum(best_h, float(n) / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
