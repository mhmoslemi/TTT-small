    for a_idx in range(len(active)):
        a = active[a_idx]
        for b_idx in range(a_idx+1, len(active)):
            b = active[b_idx]
            da = Q[a] - Q[b]
            db = L[a] - L[b]
            dc = C[a] - C[b]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    s = -dc / db

                        candidates.append(float(s))
            else:

                        if s_lo < s < s_hi:

    # Individual minima

            if s_lo < s_min < s_hi:

    if len(candidates) == 0:

    scores = F.max(axis=1) * dx
    best_idx = scores.argmin()
    
    return float(candidates[best_idx]), float(scores[best_idx])

import numpy as np
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        idx = np.argsort(h) if err > 0 else np.argsort(-h)
        for i in idx:
            if abs(err) < 1e-14:
                break
            if err > 0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            else:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def interp_to_len(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def compute_LQ(i, j, n, h):
    k_range = np.arange(-(n - 1), n)
    idx_ip = i + k_range
    idx_im = i - k_range
    idx_jp = j + k_range
    idx_jm = j - k_range
    valid_ip = (idx_ip >= 0) & (idx_ip < n)
    valid_im = (idx_im >= 0) & (idx_im < n)
    valid_jp = (idx_jp >= 0) & (idx_jp < n)
    valid_jm = (idx_jm >= 0) & (idx_jm < n)
    L = np.zeros(2 * n - 1)
    L[valid_ip] += (1.0 - h[idx_ip[valid_ip]])
    L[valid_im] -= h[idx_im[valid_im]]
    L[valid_jp] -= (1.0 - h[idx_jp[valid_jp]])

    d = j - i
    Q[n - 1 + d] += 1.0
    Q[n - 1 - d] += 1.0
    return L, Q


def line_search(h, C, L, Q, s_lo, s_hi, n, K=20, n_grid=15):
    dx = 2.0 / n
    if s_hi - s_lo < 1e-14:
        return 0.0, np.max(C) * dx
    active = np.argsort(C)[-K:]
    candidates = [s_lo, s_hi, 0.0]
    grid = np.linspace(s_lo, s_hi, n_grid)
    candidates.extend(grid.tolist())
    for a_idx in range(len(active)):
        a = active[a_idx]
        for b_idx in range(a_idx + 1, len(active)):
            b = active[b_idx]
            da = Q[a] - Q[b]
            db = L[a] - L[b]
            dc = C[a] - C[b]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    s = -dc / db
                    if s_lo < s < s_hi:
                        candidates.append(float(s))
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= 0.0:
                    sd = np.sqrt(disc)
                    for s in [(-db + sd) / (2.0 * da), (-db - sd) / (2.0 * da)]:
                        if s_lo < s < s_hi:
                            candidates.append(float(s))
    for k in active:
        if Q[k] > 1e-14:
            s_min = -L[k] / (2.0 * Q[k])
            if s_lo < s_min < s_hi:
                candidates.append(float(s_min))
    candidates = np.array(candidates)
    mask = (candidates >= s_lo - 1e-12) & (candidates <= s_hi + 1e-12)
    candidates = candidates[mask]
    if len(candidates) == 0:
        candidates = np.array([0.0])
    F = C[None, :] + L[None, :] * candidates[:, None] + Q[None, :] * candidates[:, None] ** 2
    scores = F.max(axis=1) * dx
    best_idx = scores.argmin()
    return float(candidates[best_idx]), float(scores[best_idx])


def pair_pass(h, n, max_pairs=None, prefer_steep=True):
    C = np.correlate(h, 1.0 - h, mode='full')
    current_score = float(np.max(C)) * (2.0 / n)
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            if h[i] < 1e-10 and h[j] < 1e-10:
                continue
            if h[i] > 1 - 1e-10 and h[j] > 1 - 1e-10:
                continue
            pairs.append((i, j))
    if prefer_steep and len(pairs) > 0:
        diffs = np.abs(h[:, None] - h[None, :])
        scored = []
        for (i, j) in pairs:
            scored.append((diffs[i, j], i, j))
        scored.sort(key=lambda x: -x[0])
        pairs = [(i, j) for _, i, j in scored]
    if max_pairs is not None and len(pairs) > max_pairs:
        pairs = pairs[:max_pairs]
    improved = False
    for (i, j) in pairs:
        s_lo = max(-h[i], h[j] - 1.0)
        s_hi = min(1.0 - h[i], h[j])
        if s_hi - s_lo < 1e-14:
            continue
        L, Q = compute_LQ(i, j, n, h)
        best_s, best_sc = line_search(h, C, L, Q, s_lo, s_hi, n)
        if best_s == 0.0:
            continue
        if best_sc < current_score - 1e-13:
            h[i] += best_s
            h[j] -= best_s
            h[i] = min(1.0, max(0.0, h[i]))
            h[j] = min(1.0, max(0.0, h[j]))
            current_score = best_sc
            C = np.correlate(h, 1.0 - h, mode='full')
            improved = True
    return h, current_score, improved


def multi_dir_pass(h, n, num_dirs, rng, max_time=None, t0=None):

    current_score = float(np.max(C)) * (2.0 / n)

        v = np.zeros(n)
        v[idx] = signs.astype(float)
        v_sum = v.sum()
        if abs(v_sum) < 1e-14:

        else:
            v -= v_sum / n
        s_lo = 0.0
        s_hi = 0.0
        for m in range(n):

            elif v[m] < -1e-14:

        if s_hi - s_lo < 1e-14:
            s_lo, s_hi = -s_hi, -s_lo
            if s_hi - s_lo < 1e-14:
                continue
        h_try = h + s_lo * v
        s_test = (s_hi - s_lo) * 0.5 + s_lo
        h_test = h + s_test * v
        sc_test = float(np.max(np.correlate(np.clip(h_test, 0, 1), 1.0 - np.clip(h_test, 0, 1), mode='full'))) * (2.0 / n)
        if sc_test < current_score - 1e-13:
