def official_score(h):

def make_feasible(z, target=None):

def compute_corr(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def lse_obj_and_grad(x, beta):
    """Compute the smooth max (LSE) objective and gradient, vectorized."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        obj = max_c
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        obj = max_c + np.log(exp_sum) / beta
        weights = np.exp(shifted) / exp_sum
    
    # Vectorized gradient
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n-1)
        if s >= 0:
            grad[s:] += w * (1.0 - x[:n-s])
            grad[:n-s] -= w * x[s:]
        else:
            grad[:n+s] += w * (1.0 - x[-s:])
            grad[-s:] -= w * x[:n+s]
    
    return obj, grad

def lse_only_x(x, beta):
    return lse_obj_and_grad(x, beta)[0]

def lse_grad_only_x(x, beta):
    return lse_obj_and_grad(x, beta)[1]

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def smooth_optimize(h0, beta, maxiter=500, ftol=1e-16):
    """Optimize using SLSQP with vectorized LSE gradient."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lse_only_x, h0, args=(beta,),
            jac=lse_grad_only_x,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0.copy()

def direct_minmax(h0, maxiter=300):
    """Direct min-max: minimize t subject to t >= S_k for all k."""
    n = len(h0)
    dx = 2.0 / n
    
    def obj(z):
        return z[-1]
    
    def obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def ineq(z):
        h = z[:-1]
        t = z[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    
    def ineq_jac(z):
        h = z[:-1]
        n_local = len(h)
        dx_local = 2.0 / n_local
        # Build Jacobian efficiently
        # d(ink)/dz = [d(ink)/dh, 1]
        # d(ink)/dh[p] = dx * (contribution of h[p] to -S_k)
        # Actually ineq[k] = t - S_k, so d(ineq[k])/dt = 1, d(ineq[k])/dh = -dS_k/dh
        jac = np.zeros((2*n_local-1, n_local+1))
        jac[:, -1] = 1.0
        for k in range(2*n_local-1):
            s = k - (n_local-1)
            if s >= 0:
                # dS_k/dh[p] for p in [s, n-1]: += (1-h[p-s])
                # dS_k/dh[p] for p in [0, n-1-s]: -= h[p+s]
                jac[k, s:] -= dx_local * (1.0 - h[:n_local-s])
                jac[k, :n_local-s] -= dx_local * h[s:]
            else:
                jac[k, :n_local+s] -= dx_local * (1.0 - h[-s:])
                jac[k, -s:] -= dx_local * h[:n_local+s]
        return jac
    
    def eq_z(z):
        return np.sum(z[:-1]) - len(z[:-1]) / 2.0
    
    def eq_z_jac(z):
        n_local = len(z) - 1
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    t0 = official_score(h0) + 1e-4
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    try:
        res = minimize(
            obj, z0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
                {'type': 'eq', 'fun': eq_z, 'jac': eq_z_jac}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0.copy()

def push_binary(h, p):
    """Push h towards binary by raising to power p and re-projecting."""
    return make_feasible(h ** p)

def local_polish(h, n_iter=2000, step=0.01):
    """Simple coordinate/local search polish."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = official_score(h)
    rng = np.random.default_rng(42)
    
    for it in range(n_iter):
        if it % 100 == 0 and step < 0.001:
            break
        # Try random coordinate perturbations
        improved = False
        for _ in range(3):
            i = rng.integers(n)
            for sign in [1, -1]:
                h_new = h.copy()
                delta = sign * step * (0.3 + 0.7 * rng.random())
                h_new[i] = np.clip(h_new[i] + delta, 0, 1)
                # Fix feasibility
                h_new = make_feasible(h_new)
                s = official_score(h_new)
                if s < best - 1e-14:
                    h = h_new
                    best = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            step *= 0.8
            if step < 0.0005:
                break
    
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def lse_grad(x, beta):
    """Vectorized gradient of the LSE-smoothed max correlation."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = exp_vals / exp_sum
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n-1)
        if s >= 0:
            grad[s:] += w * (1.0 - x[:n-s])
            grad[:n-s] -= w * x[s:]
        else:
            grad[:n+s] += w * (1.0 - x[-s:])
            grad[-s:] -= w * x[:n+s]
    return grad

def lse_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def projected_gradient(h0, beta, max_iter=5000, lr_init=0.01):
    """Projected gradient method with backtracking line search."""
    h = h0.copy()
    n = len(h)
    best = official_score(h)
    lr = lr_init
    
    for it in range(max_iter):
        g = lse_grad(h, beta)
        # Center the gradient (remove mean to respect sum constraint approximately)
        g_centered = g - np.mean(g)
        
        # Line search
        for _ in range(30):
            h_trial = make_feasible(h - lr * g_centered)
            obj = lse_obj(h_trial, beta)
            if obj < lse_obj(h, beta) - 1e-4 * lr * np.dot(g, g):
                break
            lr *= 0.5
        else:
            lr *= 0.5
            continue

        if s < best:
            best = s
            lr = min(lr * 1.1, 0.1)  # slightly increase if progress
        else:
            lr *= 0.7  # decrease if no progress
        
        if lr < 1e-7:
            break
    
    return h, best

def direct_minmax(h0, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    def obj(z): return z[-1]
    def obj_j(z):
        g = np.zeros(len(z)); g[-1] = 1.0; return g
    def ineq(z):
        h = z[:-1]
        return z[-1] - np.correlate(h, 1.0-h, mode='full')*dx
    def ineq_j(z):
        h = z[:-1]
        jl = len(h)
        dx_l = 2.0/jl
        jac = np.zeros((2*jl-1, jl+1))
        jac[:,-1] = 1.0
        for k in range(2*jl-1):
            s = k-(jl-1)
            if s >= 0:
                jac[k, s:] -= dx_l*(1.0-h[:jl-s])
                jac[k, :jl-s] -= dx_l*h[s:]
            else:
                jac[k, :jl+s] -= dx_l*(1.0-h[-s:])
                jac[k, -s:] -= dx_l*h[:jl+s]
        return jac
    def eq_z(z): return np.sum(z[:-1])-len(z[:-1])/2.0
    def eq_z_j(z):
        nl=len(z)-1; g=np.zeros(len(z)); g[:-1]=1.0; return g
    
    t0 = official_score(h0)+1e-4
    z0 = np.concatenate([h0,[t0]])
    bounds = [(0.0,1.0)]*n+[(0.0,1.0)]
    try:
        res = minimize(obj, z0, jac=obj_j, method='SLSQP', bounds=bounds,
                       constraints=[{'type':'ineq','fun':ineq,'jac':ineq_j},
                                    {'type':'eq','fun':eq_z,'jac':eq_z_j}],
                       options={'maxiter':maxiter,'ftol':1e-16})
        if hasattr(res,'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0.copy()
