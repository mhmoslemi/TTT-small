    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum} via bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):

        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    """Smooth max objective + penalty for sum constraint."""
    n = len(h)
    dx = 2.0 / n
    proj = np.clip(h, 0.0, 1.0)

    # Stable softmax
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / t)
    smax = m + t * np.log(np.sum(exp_vals))

_g_t = 0.1
_g_rho = 50.0  
_g_target = 35.0

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / _g_t)
    smax = m + _g_t * np.log(np.sum(exp_vals))
    penalty = _g_rho * (np.sum(x) - _g_target) ** 2

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / _g_t)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)

        s = k - (n - 1)

        if w < 1e-15:
            continue

        if p1s < p1e:
            grad[p1s + s : p1e + s] += w * (1.0 - x[p1s:p1e])
        p2s = max(0, s)

            grad[p2s - s : p2e - s] -= w * x[p2s:p2e]
    grad += 2.0 * _g_rho * (np.sum(x) - _g_target)
    return grad

    h = project_box_sum(h0, target_sum)
    bounds = [(0.0, 1.0)] * n
    _g_target = target_sum
    _g_rho = rho
    
    for t in t_list:
        if time.time() > deadline:
            break
        _g_t = t
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
        except Exception:
            break
        c5_old = compute_c5(h)
        h_candidate = project_box_sum(res.x, target_sum)
        c5_new = compute_c5(h_candidate)
        if c5_new < c5_old:
            h = h_candidate
    return h

def pair_exchange_refine(h, target_sum, deadline, n_sweeps_max=500):
    """Iterative pair exchange to reduce C5."""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(12345)
    
    for sweep in range(n_sweeps_max):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Feasible range for t: h_i + t in [0,1], h_j - t in [0,1]
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-10:
                continue
            # Coarse search
            best_t = 0.0
            best_val = best_c5
            for t in np.linspace(t_lo, t_hi, 15):
                h[i] += t
                h[j] -= t
                c5 = compute_c5(h)
                h[i] -= t
                h[j] += t
                if c5 < best_val:
                    best_val = c5
                    best_t = t
            if best_val < best_c5 - 1e-12:

                span = (t_hi - t_lo) / 15
                fine_lo = max(t_lo, best_t - span)
                fine_hi = min(t_hi, best_t + span)

                    h[i] += t
                    h[j] -= t
                    c5 = compute_c5(h)
                    h[i] -= t
                    h[j] += t
                    if c5 < best_val:
                        best_val = c5
                        best_t = t
                h[i] += best_t
                h[j] -= best_t
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5
