import numpy as np
from scipy.optimize import minimize
import time

# --- Core functions ---

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def get_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, s):
    """Gradient of the s-th lag overlap (times dx)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        # vectorized
        g[:m] += (1.0 - h[s:s+m])
        g[s:s+m] += (-h[:m])
    else:
        t = -s
        m = n - t
        g[t:t+m] += (1.0 - h[:m])
        g[:m] += (-h[t:t+m])
    return dx * g

def do_project(h):
    n = len(h)
    tgt = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < tgt: lo = mid
        else: hi = mid

    r = tgt - h.sum()
    if abs(r) > 1e-13:
        if r > 0:
            idx = np.argsort(1.0 - h)[::-1]

                a = min(r, 1.0 - h[i])
                if a <= 0: break
                h[i] += a; r -= a

                if a <= 0: break

    """Projected subgradient with adaptive step size."""
    n = len(h0)
    h = h0.copy()
    score0 = eval_score(h)
    if not np.isfinite(score0):
        return h, score0
    
    best_h = h.copy()
    best_score = score0
    alpha = alpha0
    last_h = h.copy()
    last_g = None
    
    for it in range(max_iters):
        vals = get_vals(h)
        cmax = vals.max()
        
        # Active set: lags within 5% of max (or top-5)
        thresh = cmax - 0.05 * cmax
        active_mask = vals >= thresh
        if active_mask.sum() < 3:
            top_idx = np.argsort(vals)[::-1][:5]
            active_mask = np.zeros(len(vals), dtype=bool)
            active_mask[top_idx] = True
        if active_mask.sum() > 20:
            top_idx = np.argsort(vals)[::-1][:20]
            active_mask = np.zeros(len(vals), dtype=bool)
            active_mask[top_idx] = True
        
        # Compute subgradient (average over active)
        g = np.zeros(n)
        n_active = 0
        for p in range(len(vals)):
            if active_mask[p]:
                s = p - (n - 1)
                g += lag_grad(h, s)
                n_active += 1
        g /= max(n_active, 1)
        
        # Barzilai-Borwein step size
        if last_g is not None:
            s_bb = h - last_h
            y_bb = g - last_g
            sy = np.dot(s_bb, y_bb)
            yy = np.dot(y_bb, y_bb)
            ss = np.dot(s_bb, s_bb)
            if yy > 1e-16 and sy > 0:
                alpha = sy / yy  # BB1
                alpha = min(alpha, 0.1)
                alpha = max(alpha, 1e-6)
            elif ss > 1e-16 and sy > 0:
                alpha = sy / ss  # BB2
                alpha = min(alpha, 0.1)
                alpha = max(alpha, 1e-6)
        
        # Take step
        h_new = do_project(h - alpha * g)
        sc_new = eval_score(h_new)
        
        if sc_new < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc_new
        if sc_new < eval_score(h) - 1e-14:
            h = h_new
        else:
            alpha *= 0.5  # reduce step
            h_new = do_project(h - alpha * g)
            sc_new2 = eval_score(h_new)
            if sc_new2 < eval_score(h) - 1e-14:
                h = h_new
            else:
                alpha *= 0.5
        
        last_h = h.copy()
        last_g = g.copy()
        
        # Convergence check
        if alpha < 1e-8:
            alpha = alpha0 * 0.1  # restart with smaller
        if it % 500 == 0 and alpha < 1e-6:
            break
        
        if time.time() > t_start + 300:  # time limit for this phase
            break
    
    return best_h, best_score
