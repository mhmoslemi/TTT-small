    grad = grad_weighted(h, w) * dx
    obj = cmax + np.log(ws) / alpha
    return obj, grad

def eq_sum(x):
    return np.sum(x) - len(x) / 2.0

def eq_sum_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, alpha, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]

    def objfun(x):
        o, _ = smooth_max_obj_grad(x, alpha)
        return o

    def objjac(x):
        _, g = smooth_max_obj_grad(x, alpha)
        return g

    try:
        res = minimize(objfun, h0, jac=objjac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def newton_active_set(h0, n_active_max=12, maxiter=40, tol=1e-14):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()

    for _ in range(maxiter):
        vals = all_correlate(h) * dx
        C = vals.max()

        threshold = C - 0.12 * C
        active_idx = np.where(vals >= threshold)[0]
        if len(active_idx) > n_active_max:
            active_idx = np.argsort(vals)[::-1][:n_active_max]
        elif len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(vals))])

        grad = np.zeros(n)
        H = np.zeros((n, n))
        for p in active_idx:
            s = int(p) - (n - 1)
            if s >= 0:
                m = n - s
                if s < n:
                    grad[s:] += (1.0 - h[:m])
                if m > 0:
                    grad[:m] -= h[s:]
                for j in range(m):
                    H[j, j + s] -= 1.0
                    H[j + s, j] -= 1.0
            else:
                t = -s
                m = n - t
                if n - t > 0:
                    grad[:n - t] += (1.0 - h[t:])
                if t < n:
                    grad[t:] -= h[:n - t]
                for j in range(m):
                    H[j, j + t] -= 1.0
                    H[j + t, j] -= 1.0

        n_act = len(active_idx)
        grad /= n_act
        H /= n_act

        A = np.zeros((n + 1, n + 1))
        A[:n, :n] = H
        A[:n, n] = 1.0
        A[n, :n] = 1.0
        b = np.zeros(n + 1)
        b[:n] = -grad

        try:
            sol = np.linalg.solve(A + 1e-8 * np.eye(n + 1), b)
        except np.linalg.LinAlgError:
            break

        delta = sol[:n]
        alpha_s = 1.0
        improved = False
        for _ in range(30):
            h_try = np.clip(h + alpha_s * delta, 0.0, 1.0)
            h_try = make_feasible(h_try)
            new_C = official_score(h_try)
            if new_C < C - 1e-15:
                h = h_try
                improved = True
                if C - new_C < tol:
                    break
                break
            alpha_s *= 0.5
        if not improved:
            break

    return h, official_score(h)

def binary_local_search(h0, max_passes=80):
    n = len(h0)
    h = np.round(h0).astype(float)
    target = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target:
        ones = np.where(h == 1)[0]
        order = np.argsort(h0[ones])
        for idx in order[:num_ones - target]:
            h[ones[idx]] = 0
    elif num_ones < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-h0[zeros])
        for idx in order[:target - num_ones]:
            h[zeros[idx]] = 1

    current_score = official_score(h)

    for _ in range(max_passes):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_swap = None
        best_new_score = current_score
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_score = official_score(h)
                h[i] = 1.0
                h[j] = 0.0
                if new_score < best_new_score - 1e-14:
                    best_new_score = new_score
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            current_score = best_new_score
            improved = True
        if not improved:
            break
    return h, current_score

def simulated_annealing_binary(h0, n_iter=2000, t_start=0.02, t_end=0.001):
    n = len(h0)
    h = np.round(h0).astype(float)
    target = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target:
        ones = np.where(h == 1)[0]
        order = np.argsort(h0[ones])
        for idx in order[:num_ones - target]:
            h[ones[idx]] = 0
    elif num_ones < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-h0[zeros])
        for idx in order[:target - num_ones]:
            h[zeros[idx]] = 1

    current_score = official_score(h)
    best_h = h.copy()
    best_score = current_score
    rng = np.random.default_rng(12345)

    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (it / max(n_iter - 1, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_score = official_score(h)
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0

    return best_h, best_score

def mm_obj(z):
    return z[-1]

def mm_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def mm_ineq_fun(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    theta = z[-1]
    vals = all_correlate(h) * dx
    return theta - vals

def mm_eq_fun(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0

def mm_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def mm_ineq_jac(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    grads = grad_weighted(h, np.ones(2 * n - 1))
    J = -dx * grads
    J = np.hstack([J, np.ones((J.shape[0], 1))])
    return J

def try_direct_minimax(h0, maxiter=300):
    n = len(h0)
    theta0 = official_score(h0) + 1e-8
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': mm_ineq_fun, 'jac': mm_ineq_jac},
        {'type': 'eq', 'fun': mm_eq_fun, 'jac': mm_eq_jac}
    ]
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(8.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')
    best_n = 0

    def is_deadline():
        return time.time() >= deadline

    def update(h, sc):
        nonlocal best_h, best_score, best_n
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)
        elif abs(sc - best_score) <= 1e-14 and len(h) < best_n:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                best_n = len(parent)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = np.arange(n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)
        best_n = n0

    update(best_h, best_score)

    # Phase 1: Binary local search
    if not is_deadline():
        h_bin, sc_bin = binary_local_search(best_h, max_passes=60)
        update(h_bin, sc_bin)
        # Polish the binary solution continuously
        h_pol, sc_pol = slsqp_smooth(h_bin, 500, maxiter=200)
        if h_pol is not None:
            update(h_pol, sc_pol)
        h_new, sc = newton_active_set(h_bin, maxiter=30)
        update(h_new, sc)

    # Phase 2: Simulated annealing on binary space
    if not is_deadline():
        h_sa, sc_sa = simulated_annealing_binary(best_h, n_iter=3000, t_start=0.03, t_end=0.001)
        update(h_sa, sc_sa)
        # Polish SA result
        h_pol, sc_pol = slsqp_smooth(h_sa, 500, maxiter=200)
        if h_pol is not None:
            update(h_pol, sc_pol)
        h_new, sc = newton_active_set(h_sa, maxiter=30)
        update(h_new, sc)

    # Phase 3: Smooth optimization ladder from best
    if not is_deadline():
        cur = best_h.copy()
        for beta in [50, 200, 800, 3000, 10000]:
            if is_deadline():
                break
            h_new, sc = slsqp_smooth(cur, beta, maxiter=200)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
                cur = h_new

    # Phase 4: Newton active set
    if not is_deadline():
        h_new, sc = newton_active_set(best_h, n_active_max=12, maxiter=40)
        update(h_new, sc)
        h_new, sc = newton_active_set(best_h, n_active_max=6, maxiter=40)
        update(h_new, sc)

    # Phase 5: Direct minimax
    if not is_deadline():
        h_new, sc = try_direct_minimax(best_h, maxiter=300)
        if h_new is not None:
            update(h_new, sc)

    # Phase 6: Multi-resolution
    if not is_deadline():
        n_base = len(best_h)
        n_opts = sorted(set([
            max(20, n_base - 10),
            n_base,
            min(80, n_base + 10),
            min(100, n_base + 20),
        ]))
        for n_opt in n_opts:
            if is_deadline():
                break
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            h_new, sc = slsqp_smooth(h_int, 300, maxiter=150)
            if h_new is not None:
                update(h_new, sc)
            h_new, sc = newton_active_set(h_int, maxiter=25)
            update(h_new, sc)

    # Phase 7: Random perturbations + polish
    if not is_deadline():
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(3, min(15, int(remaining / 15)))
        for i in range(n_starts):
            if is_deadline():
                break
            sigma = [0.03, 0.06, 0.1, 0.15, 0.2, 0.3][i % 6]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_new, sc = slsqp_smooth(h_cand, 500, maxiter=150)
            if h_new is not None:
                update(h_new, sc)
                h_cand = h_new
            h_new, sc = newton_active_set(h_cand, maxiter=25)
            update(h_new, sc)

    # Phase 8: Final intensive polishing
    if not is_deadline():
        for beta in [100, 500, 2000, 8000, 20000]:
            if is_deadline():
                break
            h_new, sc = slsqp_smooth(best_h, beta, maxiter=300)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
        if not is_deadline():
            h_new, sc = newton_active_set(best_h, n_active_max=15, maxiter=50)
            update(h_new, sc)
        if not is_deadline():
            h_new, sc = try_direct_minimax(best_h, maxiter=400)
            if h_new is not None:
                update(h_new, sc)

    # Phase 9: Last chance
    if not is_deadline():
        h_new, sc = newton_active_set(best_h, n_active_max=8, maxiter=50)
        update(h_new, sc)
        if not is_deadline():
            h_new, sc = try_direct_minimax(best_h, maxiter=200)
            if h_new is not None:
                update(h_new, sc)

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-13

    return final_h, final_bound, final_n
