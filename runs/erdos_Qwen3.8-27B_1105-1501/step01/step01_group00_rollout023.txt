                if a - shift == b and 0 <= b <= max_j:
                    quad = 1.0
                # Quadratic: b-shift == a (b = a+shift)
                elif b - shift == a and 0 <= a <= max_j:
                    quad = 1.0
        else:
            shift = half - p  # positive
            max_j = n - 1 - shift
            # F = sum_{j=0}^{max_j} h[j]*(1-h[j+shift])
            # a in first factor: j=a
            if 0 <= a <= max_j:
                coeff += (1.0 - h[a + shift])
            # a in second factor: j+shift=a, j=a-shift
            if 0 <= a - shift <= max_j:
                coeff -= h[a - shift]
            # b in first factor: j=b
            if 0 <= b <= max_j:
                coeff -= (1.0 - h[b + shift])
            # b in second factor: j+shift=b, j=b-shift
            if 0 <= b - shift <= max_j:
                coeff += h[b - shift]
            # Quadratic: a+shift == b (b = a+shift), i.e., j=a term
            if a + shift == b and 0 <= a <= max_j:
                quad = 1.0
            # Quadratic: b+shift == a (a = b+shift), i.e., j=b term
            elif b + shift == a and 0 <= b <= max_j:
                quad = 1.0
        
        c[p] = coeff
        q[p] = quad
    
    return c, q

def eval_at_delta(base_corr, c, q, delta, dx):
    vals = base_corr + c * delta + q * delta * delta
    return float(np.max(vals)) * dx

def golden_section_search(base_corr, c, q, delta_lo, delta_hi, dx, curr_score):
    """Find delta that minimizes C(delta) = max_p(F_p + c_p*delta + q_p*delta^2) * dx"""
    if delta_hi - delta_lo < 1e-15:
        return 0.0, curr_score
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = delta_lo
    b = delta_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = eval_at_delta(base_corr, c, q, c_pt, dx)
    fd = eval_at_delta(base_corr, c, q, d_pt, dx)
    
    for _ in range(50):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = eval_at_delta(base_corr, c, q, c_pt, dx)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = eval_at_delta(base_corr, c, q, d_pt, dx)
    
    delta_opt = (a + b) / 2.0
    score_opt = eval_at_delta(base_corr, c, q, delta_opt, dx)
    return delta_opt, score_opt

def greedy_pair_pass(h, base_corr, score, pair_order, dx, tol):
    """One pass of greedy pair moves. Returns (h, base_corr, score, n_accepted)."""
    n = len(h)
    n_accepted = 0
    
    for (a, b) in pair_order:
        delta_lo = max(-h[a], h[b] - 1.0)
        delta_hi = min(1.0 - h[a], h[b])
        
        if delta_hi - delta_lo < 1e-14:
            continue
        
        c, q = pair_coefficients(h, a, b)
        delta_opt, score_opt = golden_section_search(base_corr, c, q, delta_lo, delta_hi, dx, score)
        
        if score_opt < score - tol:
            # Accept the move
            h[a] += delta_opt
            h[b] -= delta_opt
            h[a] = np.clip(h[a], 0.0, 1.0)
            h[b] = np.clip(h[b], 0.0, 1.0)
            score = score_opt
            base_corr = compute_corr(h)
            n_accepted += 1
    
    return h, base_corr, score, n_accepted

def generate_pairs(n, rng):
    """Generate all pairs (a,b) with a < b, shuffled."""
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    rng.shuffle(pairs)
    return pairs
