def official_score(h):

    if n == 0:

def project_to_target(v, n):
    h = np.clip(np.asarray(v, dtype=float).ravel(), 0.0, 1.0)
    if len(h) != n:
        if len(h) < n:
            h = np.pad(h, (0, n - len(h)), constant_values=0.5)
        else:
            h = h[:n]
        h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection for L2-ish projection
    lo, hi = -1.0 - h.max(), 2.0 - h.min()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact mass correction
    for _ in range(100):
        residual = target - h.sum()
        if abs(residual) < 1e-14:

        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta

def grad_row(h, s):

    n = len(h)
    L = len(active_mask)

        return None

    #      sum(d) = 0
    #      -1 <= d_i <= 1

        pass
    return None

def local_search_lp(h, max_time=30.0, max_iters=200, seed=42):
    """LP-based local search for minimax."""
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    
    no_improve = 0
    step_scale = 0.1
    
    for it in range(max_iters):
        if time.perf_counter() - start > max_time:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = corr * dx
        theta = np.max(scaled)
        
        # Active set
        tol = max(0.005, 0.05 * theta)
        active_mask = scaled >= theta - tol
        
        # Get LP direction
        G = compute_all_gradients(h)
        result = lp_direction(h, G, active_mask, max_norm=step_scale)
        
        d = None
        if result is not None:
            d, t = result
            if t >= 0:
                # No descent direction from LP, use negative average gradient
                active_idx = np.where(active_mask)[0]
                d = -G[active_idx].mean(axis=0)
                d -= d.mean()  # preserve mass
                norm = np.linalg.norm(d, 1)
                if norm > 1e-14:
                    d = d / norm * step_scale
                else:
                    d = None
        else:
            # Fallback: negative gradient of max
            active_idx = np.where(active_mask)[0]
            d = -G[active_idx].mean(axis=0)
            d -= d.mean()
            norm = np.linalg.norm(d, 1)
            if norm > 1e-14:
                d = d / norm * step_scale
            else:
                d = None
        
        if d is None:
            # Random perturbation
            d = rng.normal(0, 0.1, n)
            d -= d.mean()
        
        # Line search
        best_alpha = 0.0
        best_score_local = score
        for sign in [1, -1]:
            for alpha_mult in [0.1, 0.2, 0.5, 1.0, 2.0]:
                alpha = sign * alpha_mult
                h_new = project_to_target(h + alpha * d, n)
                s_new = official_score(h_new)
                if s_new < best_score_local - 1e-12:
                    best_score_local = s_new
                    best_alpha = alpha
        
        if best_alpha != 0.0:
            h = project_to_target(h + best_alpha * d, n)
            score = best_score_local
            no_improve = 0
            # Reduce step scale slightly
            step_scale = max(0.01, step_scale * 0.95)
        else:
            no_improve += 1
            # Try random direction
            if no_improve >= 5:
                d_rand = rng.normal(0, 0.2, n)
                d_rand -= d_rand.mean()
                for alpha in [0.05, 0.1, 0.2, 0.3]:
                    h_new = project_to_target(h + alpha * d_rand, n)
                    s_new = official_score(h_new)
                    if s_new < score - 1e-12:
                        h = h_new
