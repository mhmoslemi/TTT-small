def score_exact(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

    lo = -2.0 - np.max(z)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:

            else:

            if abs(add) < 1e-15:

            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

    for it in range(max_iter):
        grad, _ = lse_gradient(h_cur, beta)
        v = mu * v - lr * grad
        h_trial = h_cur + v
        h_new = make_feasible(h_trial)
        s = score_exact(h_new)
        if s < best:
            best = s
            h_cur = h_new
        else:
            v *= 0.5
        lr *= decay
    return h_cur, best

def frank_wolfe_method(h0, max_iter=200, beta=80.0):
    n = len(h0)
    h_cur = h0.copy()
    best = score_exact(h_cur)
    target = n / 2.0
    for it in range(max_iter):
        grad, _ = lse_gradient(h_cur, beta)
        order = np.argsort(grad)
        v = np.zeros(n)
        n_full = int(target)
        frac = target - n_full
        v[order[:n_full]] = 1.0
        if frac > 1e-10:
            v[order[n_full]] = frac
        t = 2.0 / (it + 2.0)
        h_trial = (1.0 - t) * h_cur + t * v
        h_new = make_feasible(h_trial)
        s = score_exact(h_new)
        if s < best:
            best = s
            h_cur = h_new
        elif s < best + 0.0005:
            h_cur = h_new
    return h_cur, best

def pair_swap_anneal(h0, max_iter=300, t0=0.1, t1=0.001, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h0)
    h_cur = h0.copy()
    best = score_exact(h_cur)
    for it in range(max_iter):
        T = t0 * (t1 / t0) ** (it / max_iter)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(0.02, 0.15)
        h_trial = h_cur.copy()
        transfer = min(delta, 1.0 - h_trial[i], h_trial[j])
        if transfer < 1e-8:
            continue
        h_trial[i] += transfer
        h_trial[j] -= transfer
        s = score_exact(h_trial)
        delta_s = s - best
        if delta_s < 0 or rng.random() < np.exp(-delta_s / max(T, 1e-10)):
            h_cur = h_trial
            if s < best:
                best = s
    return h_cur, best

def push_binary(h, p=1.5):
    return make_feasible(h ** p)

def coarse_to_fine_slsqp(h0, betas, maxiter_each=100):
    h_cur = h0.copy()
    for beta in betas:
        h_cur = slsqp_lse(h_cur, beta, maxiter=maxiter_each)
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(48) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = score_exact(best_h)
    rng = np.random.default_rng(seed)
    pn = len(parent)

    # Phase 1: Multi-n candidates with quick optimization
    n_list = sorted(set([
        max(28, pn - 10), max(28, pn - 6), max(28, pn - 4),
        max(28, pn - 2), pn, pn + 2, pn + 4, pn + 6, pn + 10
    ]))
    n_list = [n for n in n_list if 28 <= n <= 76]

    candidates = []
    for n in n_list:
        if time.time() > deadline - 80:
            break
        base = make_feasible(interp_to_n(parent, n))
        candidates.append((n, base))
        for sigma in [0.04, 0.08, 0.15]:
            p = make_feasible(base + sigma * rng.standard_normal(n))
            candidates.append((n, p))
        for bp in [1.2, 1.6, 2.2]:
            candidates.append((n, push_binary(base, bp)))

    # Quick coarse-to-fine on all candidates
    quick_betas = [5, 20, 80]
    for n, h0 in candidates:
        if time.time() > deadline - 50:
            break
        h_opt = coarse_to_fine_slsqp(h0, quick_betas, maxiter_each=60)
        s = score_exact(h_opt)
        if s < best_score:
            best_score = s
            best_h = h_opt.copy()

    # Phase 2: Intensive optimization on best
    intensive_n_list = sorted(set([
        max(30, len(best_h) - 4), len(best_h) - 2, len(best_h),
        len(best_h) + 2, len(best_h) + 4
    ]))
    intensive_n_list = [n for n in intensive_n_list if 28 <= n <= 76]

    for n in intensive_n_list:
        if time.time() > deadline - 40:
            break

        h0 = make_feasible(interp_to_n(best_h, n))

        # Coarse-to-fine SLSQP
        if time.time() < deadline - 25:
            h_cf = coarse_to_fine_slsqp(h0, [5, 15, 50, 150, 400], maxiter_each=80)
            s = score_exact(h_cf)
            if s < best_score:
                best_score = s
                best_h = h_cf.copy()

        # Subgradient with momentum
        if time.time() < deadline - 20:
            h_sg, _ = subgrad_momentum(best_h, max_iter=300, beta=25.0, lr0=0.04)
            s = score_exact(h_sg)
            if s < best_score:
                best_score = s
                best_h = h_sg.copy()

        # Frank-Wolfe
        if time.time() < deadline - 15:
            h_fw, _ = frank_wolfe_method(best_h, max_iter=150, beta=60.0)
            s = score_exact(h_fw)
            if s < best_score:
                best_score = s
                best_h = h_fw.copy()

        # Final SLSQP polish
        if time.time() < deadline - 10:
            h_polish = slsqp_lse(best_h, beta=500, maxiter=200)
            s = score_exact(h_polish)
            if s < best_score:
                best_score = s
                best_h = h_polish.copy()

    # Phase 3: Pair-swap annealing to explore
    if time.time() < deadline - 20:
        h_anneal, _ = pair_swap_anneal(best_h, max_iter=200, t0=0.05, t1=0.005, rng=rng)
        h_anneal = slsqp_lse(h_anneal, beta=200, maxiter=150)
        s = score_exact(h_anneal)
        if s < best_score:
            best_score = s
            best_h = h_anneal.copy()

    # Phase 4: Fine-tuning with small perturbations
    remaining = deadline - time.time()
    if remaining > 15:
        n_trials = min(15, int(remaining / 12))
        for trial in range(n_trials):
            if time.time() > deadline - 8:
                break
            improved = False
            for sigma in [0.005, 0.015, 0.03]:
                if time.time() > deadline - 5:
                    break
                h_pert = make_feasible(best_h + sigma * rng.standard_normal(len(best_h)))
                h_opt = slsqp_lse(h_pert, beta=300, maxiter=120)
                s = score_exact(h_opt)
                if s < best_score:
                    best_score = s
                    best_h = h_opt.copy()
                    improved = True

    # Phase 5: High-precision final polish
    if time.time() < deadline - 3:
        for beta in [200, 500, 1000]:
            if time.time() > deadline - 2:
                break
            h_final = slsqp_lse(best_h, beta=beta, maxiter=200)
            s = score_exact(h_final)
            if s < best_score:
                best_score = s
                best_h = h_final.copy()

    best_h = make_feasible(best_h)
    best_score = score_exact(best_h)

    return (best_h, best_score, len(best_h))
