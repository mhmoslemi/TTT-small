            # Simple approach: ternary search (since f is convex in u)
            lo_u, hi_u = 0.0, 1.0
            for _ in range(50):
                mid1 = lo_u + (hi_u - lo_u) / 3
                mid2 = hi_u - (hi_u - lo_u) / 3
                f_mid1 = np.max(a * mid1 + b)
                f_mid2 = np.max(a * mid2 + b)
                if f_mid1 < f_mid2:
                    hi_u = mid2
                else:
                    lo_u = mid1
            
            u_opt = (lo_u + hi_u) / 2
            # Also check 0 and 1
            f_0 = np.max(b)
            f_1 = np.max(a + b)
            f_u = np.max(a * u_opt + b)
            
            if f_0 <= f_u and f_0 <= f_1:
                u_opt = 0.0
            elif f_1 <= f_u and f_1 <= f_0:
                u_opt = 1.0
            
            if abs(u_opt - h[k]) > 1e-10:
                h[k] = u_opt
                improved = True
        
        if not improved:
            break
        sc = official_score(h)
        if sc < best_score:
            best_score = sc
    
    return h, best_score

    f1 = np.max(a + b)

    active1 = np.where(a + b == f1)[0]
    if np.all(a[active1] < 1e-12):
        return 1.0

        elif max_slope < -1e-10:
            # f is decreasing at mid, minimum is to the right
            lo = mid
        else:

import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)

def optimize_1d(a, b):
    """Find u in [0,1] minimizing max(a*u + b)."""
    f0 = np.max(b)
    f1 = np.max(a + b)
    
    # Check if minimum is at boundary
    tol = 1e-10
    active0 = b >= f0 - tol
    if np.all(a[active0] > -tol):
        return 0.0
    active1 = (a + b) >= f1 - tol
    if np.all(a[active1] < tol):
        return 1.0
    
    # Bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        fmid = vals.max()
        active = vals >= fmid - tol
        if not np.any(active):
            active = np.zeros(len(a), dtype=bool)
            active[np.argmax(vals)] = True
        min_slope = a[active].min()
        max_slope = a[active].max()
        if min_slope > tol:
            hi = mid
        elif max_slope < -tol:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

def coord_descent(h, max_sweeps=30, tol=1e-12):
    """Exact coordinate descent."""
    n = len(h)
    h = h.copy()
    prev_score = official_score(h)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for k in range(n):
            # Compute a and b vectors
            h0 = h.copy(); h0[k] = 0.0
            h1 = h.copy(); h1[k] = 1.0
            s0 = np.correlate(h0, 1.0 - h0, mode='full')
            s1 = np.correlate(h1, 1.0 - h1, mode='full')
            a = s1 - s0
            b = s0
            u_opt = optimize_1d(a, b)
            change = abs(u_opt - h[k])
            if change > max_change:
                max_change = change
            h[k] = u_opt
        
        # Project to satisfy sum constraint
        h = make_feasible(h)
        
        if max_change < tol:
            break
    
    return h

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    w /= ws
    # Gradient of each val w.r.t. x
    # This is more complex; let's use the manual approach
    grads = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        s = p - (n-1)
        if s >= 0:
            # val = sum_{i=0}^{n-1-s} x[i+s](1-x[i])
            for j in range(n):
                g = 0.0
                if j >= s:  # j = i+s, i = j-s, need i in [0, n-1-s] i.e. j in [s, n-1]
                    if j <= n-1:
                        g += (1.0 - x[j-s])
                if j < n-s:  # j = i, need j in [0, n-1-s]
                    g -= x[j+s]
                grads[p, j] = g
        else:
            t = -s
            # val = sum_{i=0}^{n-1-t} x[i](1-x[i+t])
            for j in range(n):
                g = 0.0
                if j < n-t:  # j = i, need j in [0, n-1-t]
                    g += (1.0 - x[j+t])
                if j >= t and j <= n-1:  # j = i+t, i = j-t, need i in [0, n-1-t] i.e. j in [t, n-1]
                    g -= x[j-t]
                grads[p, j] = g
    return dx * (w @ grads)

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def try_slsqp(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            return h_new, official_score(h_new)
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def subgradient_method(h0, max_iter=10000):

    h_prev = h.copy()
    g_prev = None

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def optimize_1d(a, b):
    """Find u in [0,1] minimizing max(a*u + b). a,b are 1D arrays."""
    tol = 1e-11
    f0 = np.max(b)
    f1 = np.max(a + b)
    active0 = b >= f0 - tol
    if np.all(a[active0] > -tol):
        return 0.0
    active1 = (a + b) >= f1 - tol
    if np.all(a[active1] < tol):
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        fmid = vals.max()
        active = vals >= fmid - tol
        if not np.any(active):
            active = np.zeros(len(a), dtype=bool)
            active[np.argmax(vals)] = True
        min_slope = a[active].min()
        max_slope = a[active].max()
        if min_slope > tol:
            hi = mid
        elif max_slope < -tol:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

def compute_a_vector(h, k):
    """Compute slope vector a[p] for coordinate k."""
    n = len(h)
    dx = 2.0 / n
    a = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        s = p - (n - 1)
        val = 0.0
        if s >= 0:
            if s <= k <= n - 1:
                val += (1.0 - h[k - s])
            if k <= n - 1 - s:
                val -= h[k + s]
        else:
            t = -s
            if k <= n - 1 - t:
                val += (1.0 - h[k + t])
            if t <= k <= n - 1:

        a[p] = val * dx
    return a

def coord_descent(h, max_sweeps=30, tol=1e-14):
    """Exact coordinate descent exploiting linearity in each coordinate."""
    n = len(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        g = np.correlate(h, 1.0 - h, mode='full')
        max_change = 0.0
        for k in range(n):
            a = compute_a_vector(h, k)
            b = g - a * h[k]
            u_opt = optimize_1d(a, b)

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def compute_grads(h):
    """Compute gradient of each correlate value w.r.t. h. Returns (2n-1, n) array."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            m = np.arange(n - s)
            grads[p, m + s] += (1.0 - h[m]) * dx
            grads[p, m] -= h[m + s] * dx
        else:
            t = -s
            m = np.arange(n - t)
            grads[p, m] += (1.0 - h[m + t]) * dx
            grads[p, m + t] -= h[m] * dx
    return grads

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    grads = compute_grads(x)
    return w @ grads

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def try_slsqp(h0, beta, maxiter=200):

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        pass
    return None, None

def subgradient_step(h, step_size):

                    sc = official_score(h_new)
                    update(h_new, sc)
                    if not is_deadline():
                        h_cd = coord_descent(h_new, max_sweeps=8)
                        update(h_cd, official_score(h_cd))
            # Interpolate back to best_h resolution
            if not is_deadline():
                h_back = interpolate_to(best_h, len(best_h))
                sc_back = official_score(h_back)
                update(h_back, sc_back)
    
    # Phase 6: 0-1 rounding attempts
    if not is_deadline():
        n = len(best_h)
        h_bin = round_to_binary(best_h)
        sc_bin = official_score(h_bin)
        update(h_bin, sc_bin)
        # Refine the binary solution
        if not is_deadline():
            h_ref = make_feasible(h_bin + rng.normal(0, 0.05, size=n))
            h_cd = coord_descent(h_ref, max_sweeps=15)
            update(h_cd, official_score(h_cd))
            if not is_deadline():
                h_new, sc = try_slsqp(h_cd, 3000, maxiter=100)
                if h_new is not None:
                    update(h_new, sc)
    
    # Phase 7: Multi-start with diverse perturbations
    if not is_deadline():
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(4, min(15, int(remaining / 20)))
        for i in range(n_starts):
            if is_deadline():
                break
            sigma = 0.02 + 0.12 * rng.random()
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            sc_cand = official_score(h_cand)
            update(h_cand, sc_cand)
            if not is_deadline():
                h_cd = coord_descent(h_cand, max_sweeps=12)
                update(h_cd, official_score(h_cd))
            if not is_deadline():
                h_new, sc = try_slsqp(best_h, 5000, maxiter=80)
                if h_new is not None:
                    update(h_new, sc)
    
    # Phase 8: Structured initializations
    if not is_deadline():
        n = len(best_h)
        # Try different structured patterns
        patterns = [
            'two_block', 'three_block', 'alternating'
        ]
        for pat_name in patterns:
            if is_deadline():
                break
            if pat_name == 'two_block':
                # Two blocks of 1s separated by 0s
                frac1 = 0.25 + 0.1 * rng.random()
                h_struct = np.zeros(n)
                h_struct[:int(frac1 * n)] = 1.0
                gap_start = int(frac1 * n) + int(0.2 * n)
                h_struct[gap_start:gap_start + int(n - gap_start - int(frac1 * n))] = 1.0
            elif pat_name == 'three_block':
                h_struct = np.zeros(n)
                third = n // 3
                h_struct[:third // 2] = 1.0
                h_struct[third : third + third // 2] = 1.0
                h_struct[2*third : 2*third + third // 2] = 1.0
            else:
                # Alternating blocks
                block_size = max(1, n // 12)
                h_struct = np.zeros(n)
                for b in range(0, n, 2 * block_size):
                    h_struct[b:b + block_size] = 1.0
            h_struct = make_feasible(h_struct)
            sc_struct = official_score(h_struct)
            update(h_struct, sc_struct)
            if not is_deadline():
                h_cd = coord_descent(h_struct, max_sweeps=15)
                update(h_cd, official_score(h_cd))
    
    # Phase 9: Final intensive polish
    if not is_deadline():
        h_cd = coord_descent(best_h, max_sweeps=30)
        update(h_cd, official_score(h_cd))
        for beta in [5000, 20000, 80000]:
            if is_deadline():
                break
            h_new, sc = try_slsqp(best_h, beta, maxiter=200)
            if h_new is not None:
                update(h_new, sc)
                if not is_deadline():
                    h_cd = coord_descent(best_h, max_sweeps=10)
                    update(h_cd, official_score(h_cd))
    
    # Phase 10: Final CD sweep
    if not is_deadline():
        h_final_cd = coord_descent(best_h, max_sweeps=40, tol=1e-15)
        update(h_final_cd, official_score(h_final_cd))
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
