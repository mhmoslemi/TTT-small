def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        if s < target:

        else:

    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)

            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_corr_and_grads(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    R = 2 * n - 1
    grad = np.zeros((R, n))
    for k in range(R):
        L = k - (n - 1)
        if L >= 0:
            for i in range(n - L):
                f = i + L
                s = i
                grad[k, f] += (1.0 - h[s])
                grad[k, s] += (-h[f])
        else:
            aL = -L
            for i in range(aL, n):
                f = i
                s = i - aL
                grad[k, f] += (1.0 - h[s])
                grad[k, s] += (-h[f])
    return corr, grad

def majorization_step(h0, rho, top_m=20, reg=0.005, solver=None):
    n = len(h0)
    corr, grad = compute_corr_and_grads(h0, n)
    R = 2 * n - 1
    top_indices = np.argsort(corr)[::-1][:min(top_m, R)]
    delta = cp.Variable(n)
    t = cp.Variable()
    sq = cp.sum_squares(delta)
    constraints = [
        cp.sum(delta) == 0,
        delta >= -h0 - 1e-8,
        delta <= 1.0 - h0 + 1e-8,
    ]
    if rho is not None:
        constraints.append(sq <= rho * rho)
    for k in top_indices:
        constraints.append(corr[k] + grad[k] @ delta + sq <= t)
    objective = cp.Minimize(t + reg * sq)
    prob = cp.Problem(objective, constraints)
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if delta.value is None:
        return None
    d = np.asarray(delta.value).ravel()
    if not np.all(np.isfinite(d)):
        return None
    h_new = make_feasible(h0 + d)
    return h_new

def smooth_minimize(h0, maxiter=100):
    n = len(h0)
    dx = 2.0 / n

    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = np.max(c)
        beta = 200.0
        shifted = np.clip(beta * (c - mc), -500, 0)
        es = np.sum(np.exp(shifted))
        return mc + np.log(es) / beta if es > 1e-300 else mc

    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = np.max(c)
        beta = 200.0
        shifted = np.clip(beta * (c - mc), -500, 0)
        ev = np.exp(shifted)
        es = np.sum(ev)
        w = ev / es if es > 1e-300 else np.ones(2 * n - 1) / (2 * n - 1)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wi = w[k] * dx
            if wi < 1e-15:
                continue
            L = k - (n - 1)
            if L >= 0:
                for i in range(n - L):
                    f, s = i + L, i
                    g[f] += wi * (1.0 - x[s])
                    g[s] += wi * (-x[f])
            else:
                aL = -L
                for i in range(aL, n):
                    f, s = i, i - aL
                    g[f] += wi * (1.0 - x[s])
                    g[s] += wi * (-x[f])
        return g

    try:
        res = minimize(
            obj_fn, h0, jac=obj_jac, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(40) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)

    n_list = sorted(set([max(28, parent_n - 6), max(28, parent_n - 3), parent_n,
                         parent_n + 3, parent_n + 6]))
    n_list = [n for n in n_list if 28 <= n <= 68]
    solvers_to_try = ['CLARABEL', 'ECOS', 'SCS']

    for n in n_list:
        if time.time() > deadline - 80:
            break
        h_base = make_feasible(interp_to_n(parent, n))
        starts = [h_base]
        for sigma in [0.02, 0.06, 0.12]:
            starts.append(make_feasible(h_base + sigma * rng.standard_normal(n)))
        starts.append(make_feasible(h_base ** 1.4))
        starts.append(make_feasible(h_base ** 0.75))
        starts.append(make_feasible(1.0 - h_base[::-1].copy()))

        for h0 in starts:
            if time.time() > deadline - 30:
                break
            cur_h = h0.copy()
            cur_score = official_score(cur_h)
            if cur_score < best_score - 1e-10:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = n

            for rho in [0.25, 0.15, 0.10, 0.07, 0.05, 0.03, 0.02]:
                if time.time() > deadline - 25:
                    break
                improved = False
                for solver in solvers_to_try:
                    new_h = majorization_step(cur_h, rho, top_m=20, reg=0.005, solver=solver)
                    if new_h is not None:
                        s = official_score(new_h)
                        if s < cur_score - 1e-12:
                            cur_h = new_h
                            cur_score = s
                            improved = True
                            if cur_score < best_score - 1e-10:
                                best_h = cur_h.copy()
                                best_score = cur_score
                                best_n = n
                            break
                if not improved:
                    continue

            if time.time() < deadline - 15:
                sh = smooth_minimize(cur_h, maxiter=80)
                ss = official_score(sh)
                if ss < cur_score - 1e-12:
                    cur_h = sh
                    cur_score = ss
                    if cur_score < best_score - 1e-10:
                        best_h = cur_h.copy()
                        best_score = cur_score
                        best_n = len(cur_h)

    if time.time() < deadline - 20:
        for trial in range(15):
            if time.time() > deadline - 10:
                break
            for rho in [0.08, 0.04]:
                h_pert = make_feasible(best_h + 0.015 * rng.standard_normal(best_n))
                new_h = majorization_step(h_pert, rho, top_m=20, reg=0.005, solver='CLARABEL')
                if new_h is not None:
                    s = official_score(new_h)
                    if s < best_score - 1e-12:
                        best_h = new_h
                        best_score = s
                        break
        if time.time() < deadline - 10:
            fh = smooth_minimize(best_h, maxiter=120)
            fs = official_score(fh)
            if fs < best_score - 1e-12:
                best_h = fh
                best_score = fs

    if time.time() < deadline - 5:
        for rho in [0.03, 0.02, 0.01]:
            new_h = majorization_step(best_h, rho, top_m=25, reg=0.003, solver='CLARABEL')
            if new_h is not None:
                s = official_score(new_h)
                if s < best_score - 1e-12:
                    best_h = new_h
                    best_score = s

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
