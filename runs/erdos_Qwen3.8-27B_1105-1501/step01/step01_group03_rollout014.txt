import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_all_F(h, n):
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F


def compute_lag_gradient(h, k, n):
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start < end:
        grad[start:end] += (1.0 - h[start + k:end + k])
    p_start = max(0, start + k)
    p_end = min(n, end + k)
    if p_start < p_end:
        grad[p_start:p_end] += (-h[p_start - k:p_end - k])
    return grad


def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    F = compute_all_F(x, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax
    return cmax + (mk + np.log(ws)) / beta


def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    F = compute_all_F(x, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[k + n - 1]
        if abs(w) < 1e-15:
            continue
        grad += w * compute_lag_gradient(x, k, n)
    return grad * dx


def eq_sum(x):
    return float(np.sum(x) - len(x) / 2.0)


def eq_sum_jac(x):
    return np.ones(len(x))


def smooth_solve(h0, n, beta, maxiter=200, ftol=1e-12):
    h0 = make_feasible(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]
    try:

                       bounds=bounds, constraints=cons, method='SLSQP',

            return make_feasible(res.x)
    except Exception:
        pass
    return h0

        {'type': 'eq', 'fun': direct_eq, 'jac': direct_eq_jac}

def milp_binary(n, time_limit=60):
    """Solve MILP for optimal binary sequence of length n."""
    try:
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()
        constraints = [cp.sum(h) == n // 2]
        if n % 2 == 1:
            constraints = [cp.sum(h) == (n // 2)]
            # For odd n, we can't have exactly n/2 ones in a binary sequence
            # So MILP won't work directly for odd n
            # Skip odd n
            if n % 2 != 0:
                return None
        
        y_constraints = []
        shift_sums = []
        
        for k in range(1, n):
            y_pos = cp.Variable(n - k)
            y_neg = cp.Variable(n - k)
            
            for i in range(n - k):
                y_constraints += [
                    y_pos[i] <= h[i],
                    y_pos[i] <= 1 - h[i + k],
                    y_pos[i] >= h[i] - h[i + k],
                    y_pos[i] >= 0,
                    y_neg[i] <= h[i + k],
                    y_neg[i] <= 1 - h[i],
                    y_neg[i] >= h[i + k] - h[i],
                    y_neg[i] >= 0
                ]
            
            shift_sums.append(cp.sum(y_pos))
            shift_sums.append(cp.sum(y_neg))
        
        constraints += y_constraints
        for s in shift_sums:
            constraints.append(s <= t)
        constraints.append(t >= 0)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        
        if h.value is not None:
            h_val = np.round(np.array(h.value)).astype(int).astype(float)
            # Verify and fix
            s = int(np.sum(h_val))
            target = n // 2
            if s > target:
                idx = np.where(h_val == 1)[0]
                for i in range(s - target):

            elif s < target:
                idx = np.where(h_val == 0)[0]

            h[best_pair[1]] = 1

            no_improve = 0
        else:
