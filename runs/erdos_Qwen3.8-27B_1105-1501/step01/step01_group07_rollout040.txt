    c = n - 1
    total = 2 * n - 1
    dx = 2.0 / n
    
    # Compute overlaps and gradients
    overlaps = np.correlate(h_star, 1.0 - h_star, mode='full') * dx
    G = compute_grad_matrix(h_star)
    
    if active_indices is None:

    n_act = len(active_indices)

        bounds.append((lo, hi))

    if res.success:

        return h_new

import numpy as np

from scipy.optimize import linprog

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_grad_matrix(h):
    n = len(h)
    total = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((total, n))
    for d in range(0, n):
        m_pos = n - 1 + d
        if d == 0:
            G[m_pos, :] = dx * (1.0 - 2.0 * h)
        else:
            G[m_pos, d:] += dx * (1.0 - h[:n - d])
            G[m_pos, :n - d] -= dx * h[d:]
        if d > 0:
            m_neg = n - 1 - d
            G[m_neg, :n - d] += dx * (1.0 - h[d:])
            G[m_neg, d:] -= dx * h[:n - d]
    return G

def solve_lp(h_star, r):
    n = len(h_star)
    total = 2 * n - 1
    dx = 2.0 / n
    overlaps = np.correlate(h_star, 1.0 - h_star, mode='full') * dx
    G = compute_grad_matrix(h_star)
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((total, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -overlaps + G @ h_star
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo_b = max(0.0, h_star[i] - r)
        hi_b = min(1.0, h_star[i] + r)
        if lo_b > hi_b - 1e-15:
            lo_b = hi_b = min(1.0, max(0.0, h_star[i]))
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.success:
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        return h_new
    return None

def solve_lp_penalty(h_star, lam):

    dx = 2.0 / n
    overlaps = np.correlate(h_star, 1.0 - h_star, mode='full') * dx

    c_vec = np.zeros(n + 2)
    c_vec[n] = 1.0
    c_vec[n + 1] = lam
    n_ineq = total + 2 * n
    A_ub = np.zeros((n_ineq, n + 2))
    A_ub[:total, :n] = G
    A_ub[:total, n] = -1.0
    b_ub = np.empty(n_ineq)
    b_ub[:total] = -overlaps + G @ h_star
    for i in range(n):
        A_ub[total + i, i] = 1.0
        A_ub[total + i, n + 1] = -1.0
        b_ub[total + i] = 0.0
        A_ub[total + n + i, i] = -1.0
        A_ub[total + n + i, n + 1] = -1.0
        b_ub[total + n + i] = 0.0
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
    bounds.append((None, None))
    bounds.append((0.0, None))
    try:

    if res.success:

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(20):
                if hi2 - lo2 < 1e-10:
                    break
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            bs = score_exact(h)
            if bs < cur_c:
                best_s = bs
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    G = compute_grad_matrix(h)
    g = G.T @ w
    return w, g

def grad_step(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        w, g = smooth_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 30:
                cur = make_feasible(cur + rng.normal(0, max(0.005, T * 0.3), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.005)
                stall = 0
        else:
            step = min(step * 1.05, T * 2.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

                if cur_c < best_c:
                    best_h = cur.copy()
                    best_c = cur_c

    # Phase 3: Fine LP (12% time)
    p3_end = t0 + budget_s * 0.92
    cur = best_h.copy()
    cur_c = best_c
    r = 0.008
    while time.time() < p3_end:
        h_new = solve_lp(cur, r)
        if h_new is not None:
            h_new = make_feasible(h_new, n)
            c_new = score_exact(h_new)
            if c_new < cur_c - 1e-14:
                cur = h_new
                cur_c = c_new
                if cur_c < best_c:
                    best_h = cur.copy()
                    best_c = cur_c
            else:
                r *= 0.7
                if r < 0.0005:
                    cur = best_h.copy()
                    cur_c = best_c
                    r = 0.008
        else:
            r *= 0.5
            if r < 0.0003:
                break

    # Phase 4: Gradient descent fallback + final polish (8% time)
    if time.time() < deadline - 3:
        sub_dl = deadline - 3
        h_g, c_g = grad_step(best_h, sub_dl, 0.005, 0.0001, 0.999, 0.003, rng)
        if c_g < best_c:
            best_h, best_c = h_g, c_g
        n = len(best_h)

    # Final very fine pair search on active overlap pairs
    if time.time() < deadline - 2:
        cur = best_h.copy()
        cur_c = best_c
        c_idx = n - 1
        overlaps = np.correlate(cur, 1.0 - cur, mode='full') * (2.0 / n)
        threshold = best_c * 0.92
        active = np.where(overlaps > threshold)[0]
        for m in active:
            if time.time() > deadline - 2:
                break
            d = abs(m - c_idx)
            if d == 0:
                continue
            for i in range(n - d):
                if time.time() > deadline - 2:
                    break
                j = i + d
                if j >= n:
                    break
                res = pair_search(cur, i, j, cur_c, n_pts=40)
                if res is not None:
                    cur_c = res
                    if cur_c < best_c:
                        best_h = cur.copy()
                        best_c = cur_c

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
