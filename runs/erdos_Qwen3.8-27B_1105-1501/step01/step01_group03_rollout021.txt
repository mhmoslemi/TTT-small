def safe_sigmoid(x):
    """Numerically stable sigmoid."""
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    exp_x = np.exp(x[~pos])
    out[~pos] = exp_x / (1.0 + exp_x)
    return out

def active_obj(z, h_act, n):
    return z[-1]


def active_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def active_ineq(z, active_lags, n):
    h = z[:n]
    dx = 2.0 / n
    F = compute_all_F(h, n) * dx
    return z[-1] - F[active_lags]


def active_ineq_jac(z, active_lags, n):
    h = z[:n]
    dx = 2.0 / n
    m = len(active_lags)
    J = np.zeros((m, n + 1))
    J[:, -1] = 1.0
    for j, k in enumerate(active_lags):
        J[j, :n] = -dx * lag_grad(h, k, n)
    return J


def active_eq(z):
    return float(np.sum(z[:-1]) - len(z[:-1]) / 2.0)


def active_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g


def active_set_solve(h0, n, top_m=6, maxiter=200):
    dx = 2.0 / n
    F = compute_all_F(h0, n) * dx
    idx = np.argsort(F)[::-1]
    active_lags = (idx[:top_m] - (n - 1)).astype(int)
    score0 = float(np.max(F))
    theta0 = score0 + 0.005
    z0 = np.concatenate([h0.copy(), [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.8)]
    cons = [
        {'type': 'ineq', 'fun': lambda z: active_ineq(z, active_lags, n),
         'jac': lambda z: active_ineq_jac(z, active_lags, n)},
        {'type': 'eq', 'fun': active_eq, 'jac': active_eq_jac}
    ]
    try:
        res = minimize(active_obj, z0, jac=active_obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x[:n])
    except Exception:
        pass
    return make_feasible(h0)


def fourier_basis(n, q_max):
    x = (np.arange(n) + 0.5) * 2.0 / n
    cols = [np.ones(n) / np.sqrt(n)]
    for m in range(1, q_max + 1):
        cols.append(np.cos(np.pi * m * x) * np.sqrt(2.0 / n))
        cols.append(np.sin(np.pi * m * x) * np.sqrt(2.0 / n))
    B = np.column_stack(cols)
    for j in range(B.shape[1]):
        B[:, j] -= np.mean(B[:, j])
    B[:, 0] = 0.0
    return B


def fourier_optimize(h0, n, q, rng, maxiter=100):
    B = fourier_basis(n, q)
    dim = B.shape[1]
    # Project h0 onto basis
    target = n / 2.0
    h_centered = h0 - target / n
    coeffs = B.T @ h_centered

    def fo(c):
        h = target / n + B @ c
        h = make_feasible(h)
        return official_score(h)

    best_c = coeffs.copy()
    best_f = fo(best_c)
    res = minimize(fo, coeffs, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-6, 'fatol': 1e-7})
    if res.fun < best_f:
        best_c = res.x
        best_f = res.fun
    h_best = make_feasible(target / n + B @ best_c)
    return h_best


def interp_area(h_src, n_src, n_tgt):
    """Area-preserving interpolation."""
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    x_src = (np.arange(n_src) + 0.5) * dx_src
    x_tgt = (np.arange(n_tgt) + 0.5) * dx_tgt
    # Linear interpolation
    h_lin = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_lin)


def symmetry_transform(h):
    return make_feasible(1.0 - h[::-1].copy())


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if np.all(np.isfinite(pv)) and len(pv) >= 10:
            parent = make_feasible(pv)
    except (NameError, AttributeError, TypeError):
        parent = None

    if parent is not None:
        best_h = parent.copy()
        best_score = official_score(best_h)
        best_n = len(parent)
    else:
        best_n = 40
        best_h = make_feasible(np.full(best_n, 0.5))
        best_score = official_score(best_h)

    def try_update(h_cand):
        nonlocal best_h, best_score
        if h_cand is None or not np.all(np.isfinite(h_cand)):
            return
        h_cand = make_feasible(h_cand)
        sc = official_score(h_cand)
        if np.isfinite(sc) and (sc < best_score - 1e-13 or
                                (abs(sc - best_score) < 1e-12 and len(h_cand) < len(best_h))):
            best_h = h_cand.copy()
            best_score = sc

    def time_ok(remaining=10.0):
        return time.time() < deadline - remaining

    # Stage 1: Logit optimization with increasing beta
    if time_ok(20):
        n = best_n
        y0 = np.clip(np.log(best_h / (1.0 - best_h)), -30, 30)
        cur_y = y0.copy()
        betas = [30, 80, 200, 600, 1500, 4000]
        for beta in betas:
            if not time_ok(15):
                break
            maxiter = max(30, min(150, int(8000.0 / max(n, 10))))
            try:
                res = minimize(smooth_obj_l, cur_y, args=(beta, n),
                               jac=lambda y: smooth_grad_l(y, beta, n),
                               method='L-BFGS-B',
                               options={'maxiter': maxiter, 'gtol': 1e-8})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    cur_y = res.x
                    h_new = make_feasible(logit_h(cur_y, n))
                    try_update(h_new)
            except Exception:
                pass

    # Stage 2: Active set refinement on best
    if time_ok(30):
        n = len(best_h)
        for top_m in [3, 5, 8, 12]:
            if not time_ok(15):
                break
            h_new = active_set_solve(best_h, n, top_m=top_m, maxiter=150)
            try_update(h_new)
            if time_ok(10):
                h_new2 = active_set_solve(h_new, n, top_m=top_m, maxiter=100)
                try_update(h_new2)

    # Stage 3: Symmetry + perturbation + re-optimize
    if time_ok(30):
        sym_h = symmetry_transform(best_h)
        try_update(sym_h)
        for sigma in [0.01, 0.03, 0.05, 0.08]:
            if not time_ok(10):
                break
            for _ in range(3):
                if not time_ok(5):
                    break
                perm = make_feasible(best_h + rng.normal(0, sigma, len(best_h)))
                n_p = len(perm)
                # Quick logit polish
                y_p = np.clip(np.log(perm / (1.0 - perm)), -25, 25)
                try:
                    res = minimize(smooth_obj_l, y_p, args=(300, n_p),
                                   jac=lambda y: smooth_grad_l(y, 300, n_p),
                                   method='L-BFGS-B', options={'maxiter': 60, 'gtol': 1e-6})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        try_update(make_feasible(logit_h(res.x, n_p)))
                except Exception:
                    pass
                # Active set on perturbed
                if time_ok(5):
                    try_update(active_set_solve(perm, n_p, top_m=5, maxiter=80))

    # Stage 4: Fourier basis search
    if time_ok(40):
        n = len(best_h)
        for q in [3, 5, 7, 10]:
            if not time_ok(15):
                break
            h_f = fourier_optimize(best_h, n, q, rng, maxiter=80)
            try_update(h_f)
            # Polish fourier result with logit
            if time_ok(10):
                y_f = np.clip(np.log(h_f / (1.0 - h_f)), -25, 25)
                try:
                    res = minimize(smooth_obj_l, y_f, args=(500, n),
                                   jac=lambda y: smooth_grad_l(y, 500, n),
                                   method='L-BFGS-B', options={'maxiter': 50, 'gtol': 1e-7})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        try_update(make_feasible(logit_h(res.x, n)))
                except Exception:
                    pass

    # Stage 5: Resolution transfer (try smaller n)
    if time_ok(60):
        n_cur = len(best_h)
        for dn in [-4, -6, -8, -12, -16]:
            if not time_ok(20):
                break
            n_tgt = n_cur + dn
            if n_tgt < 20:
                continue
            h_small = interp_area(best_h, n_cur, n_tgt)
            try_update(h_small)
            # Polish at this resolution
            if time_ok(15):
                y_s = np.clip(np.log(h_small / (1.0 - h_small)), -25, 25)
                for beta in [100, 500]:
                    if not time_ok(8):
                        break
                    try:
                        res = minimize(smooth_obj_l, y_s, args=(beta, n_tgt),
                                       jac=lambda y: smooth_grad_l(y, beta, n_tgt),
                                       method='L-BFGS-B', options={'maxiter': 60, 'gtol': 1e-7})
                        if res.x is not None and np.all(np.isfinite(res.x)):
                            y_s = res.x
                            try_update(make_feasible(logit_h(res.x, n_tgt)))
                    except Exception:
                        pass
                if time_ok(10):
                    try_update(active_set_solve(h_small, n_tgt, top_m=5, maxiter=100))

    # Stage 6: Final intensive polishing on best
    if time_ok(30):
        n = len(best_h)
        # Multiple active set passes with varying top-m
        for _ in range(3):
            if not time_ok(10):
                break
            for tm in [2, 4, 6, 10, 15]:
                if not time_ok(5):
                    break
                h_new = active_set_solve(best_h, n, top_m=tm, maxiter=120)
                try_update(h_new)

    # Stage 7: One more logit pass at very high beta
    if time_ok(20):
        n = len(best_h)
        y_h = np.clip(np.log(best_h / (1.0 - best_h)), -25, 25)
        for beta in [2000, 5000]:
            if not time_ok(8):
                break
            try:
                res = minimize(smooth_obj_l, y_h, args=(beta, n),
                               jac=lambda y: smooth_grad_l(y, beta, n),
                               method='L-BFGS-B', options={'maxiter': 80, 'gtol': 1e-9})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    y_h = res.x
                    try_update(make_feasible(logit_h(res.x, n)))
            except Exception:
                pass

    # Final validation
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score < 0:
        if parent is not None:
            best_h = parent.copy()
        else:
            best_h = make_feasible(np.full(40, 0.5))
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
