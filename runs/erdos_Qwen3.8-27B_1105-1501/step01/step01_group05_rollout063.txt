        g = np.zeros(n + 1)
        g[n] = 1.0

        g[:n] = 1.0
        return g

def interpolate_to_n(parent, n_target, offset=0):
    """Interpolate parent array to n_target points."""
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_s = xt[order]
    h = np.interp(xt_s, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)

        pass
    return h, official_score(h)


def interpolate_to_n(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_s = xt[order]
    h = np.interp(xt_s, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)


def gen_random_continuous(n, rng):
    h = 0.5 + 0.3 * rng.uniform(-1, 1, n)
    return project_to_target(h, n)


def gen_sine_pattern(n, rng):
    t = np.linspace(0, 1, n, endpoint=False)
    freq = rng.uniform(1.0, 6.0)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * t + phase)
    return project_to_target(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 5.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = project_to_target(parent, n_parent)
    else:
        n_parent = 64
        best_h = project_to_target(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    # Phase 1: Build candidate pool
    candidates = []
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h).copy())
    candidates.append((1.0 - best_h)[::-1].copy())

    n_set = sorted(set([n_parent, n_parent - 5, n_parent - 10,
                        n_parent + 5, n_parent + 10, n_parent + 15,
                        32, 40, 48, 55, 60, 65, 70, 72, 80]))
    n_set = [n for n in n_set if 24 <= n <= 120]

    for n_try in n_set:
        if n_try != n_parent:
            candidates.append(interpolate_to_n(best_h, n_try))
            candidates.append(interpolate_to_n(best_h, n_try, offset=1))

    for n_try in n_set[:6]:
        candidates.append(gen_random_continuous(n_try, rng))
        candidates.append(gen_sine_pattern(n_try, rng))

    scored = []
    for h_c in candidates:
        n_c = len(h_c)
        h_c = project_to_target(h_c, n_c)
        s = official_score(h_c)
        if np.isfinite(s):
            scored.append((s, h_c, n_c))

    scored.sort(key=lambda x: (x[0], x[2]))

    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = scored[0][2]

    # Phase 2: LP descent on top candidates
    n_top = min(10, len(scored))
    for i in range(n_top):
        if time.perf_counter() > deadline - 80:
            break
        s_c, h_c, n_c = scored[i]
        remaining = deadline - time.perf_counter()
        t_limit = min(remaining * 0.5 / max(1, n_top - i), 100.0)
        if t_limit < 5:
            break
        h_loc, s_loc = local_lp_descent(h_c, n_c, max_iters=200,
                                        time_limit=t_limit, seed=seed + i * 17)
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c

    # Phase 3: SLSQP polish
    if time.perf_counter() < deadline - 50:
        t_rem = deadline - time.perf_counter() - 40
        h_pol, s_pol = slsqp_polish(best_h, best_n, time_limit=min(60, t_rem), maxiter=60)
        if s_pol < best_score:
            best_score = s_pol
            best_h = h_pol.copy()

    # Phase 4: Interpolate best to new resolutions
    if time.perf_counter() < deadline - 60:
        for dn in [0, 5, -5, 10, -10]:
            if time.perf_counter() > deadline - 50:
                break
            n_try = best_n + dn
            if 24 <= n_try <= 120:
                h_interp = interpolate_to_n(best_h, n_try)
                remaining = deadline - time.perf_counter() - 40
                h_loc, s_loc = local_lp_descent(h_interp, n_try, max_iters=100,
                                                time_limit=min(50, remaining),
                                                seed=seed + 300 + abs(dn))
                if s_loc < best_score:
                    best_score = s_loc
                    best_h = h_loc.copy()
                    best_n = n_try
                if time.perf_counter() < deadline - 35:
                    h_pol, s_pol = slsqp_polish(h_loc, n_try,
                                                 time_limit=min(20, deadline - time.perf_counter() - 20),
                                                 maxiter=50)
                    if s_pol < best_score:
                        best_score = s_pol
                        best_h = h_pol.copy()
                        best_n = n_try

    # Phase 5: Random restarts near best
    if time.perf_counter() < deadline - 40:
        for trial in range(3):
            if time.perf_counter() > deadline - 35:
                break
            pert = rng.normal(0, 0.02, best_n)
            pert -= pert.mean()
            h_pert = project_to_target(best_h + pert, best_n)
            remaining = deadline - time.perf_counter() - 30
            h_loc, s_loc = local_lp_descent(h_pert, best_n, max_iters=80,
                                            time_limit=min(30, remaining),
                                            seed=seed + 500 + trial)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()

    # Phase 6: Final polish
    if time.perf_counter() < deadline - 10:
        h_fin, s_fin = slsqp_polish(best_h, best_n,
                                     time_limit=min(15, deadline - time.perf_counter() - 5),
                                     maxiter=50)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
