    A_eq = np.zeros((1, nvar))

        hi = min(1.0, h[j] + delta)
        bounds.append((lo, hi))

                h[j] += best_delta
                best_c5 = best_val
                improved += 1

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def get_corr_values(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def grad_lag(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + s < n:
            grad[j] += (1.0 - h[j + s]) * dx
        if 0 <= j - s < n:
            grad[j] -= h[j - s] * dx
    return grad


def solve_lp_step(h, active_indices, delta):
    n = len(h)
    corr_vals = get_corr_values(h)
    nvar = n + 1

    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0

    A_rows = []
    b_vals = []
    for ki in active_indices:
        s = ki - (n - 1)
        g = grad_lag(h, s)
        f_k = corr_vals[ki]
        rhs = -f_k + np.dot(g, h)
        row = np.zeros(nvar)
        row[:n] = -g
        row[n] = 1.0
        A_rows.append(row)
        b_vals.append(rhs)

    if not A_rows:
        return None

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)

    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = []
    for j in range(n):
        bounds.append((max(0.0, h[j] - delta), min(1.0, h[j] + delta)))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def line_search_step(h0, h1, target_sum, n_evals=25):
    best_alpha = 0.0
    best_c5 = compute_c5(h0)
    for a in np.linspace(0.01, 1.0, n_evals):
        h_try = h0 + a * (h1 - h0)
        h_try = project_onto_constraint(h_try, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_alpha = a
    h_new = h0 + best_alpha * (h1 - h0)
    h_new = project_onto_constraint(h_new, target_sum)
    return h_new, best_c5


def pair_sweep(h, target, n_grid=15):
    n = len(h)
    improved = 0
    cur_c5 = compute_c5(h)

    for i in range(n):
        for j in range(i + 1, n):
            lo = max(-h[i], h[j] - 1.0)
            hi = min(h[i], 1.0 - h[j])
            if hi - lo < 1e-10:
                continue

            best_d = 0.0
            best_v = cur_c5
            for d in np.linspace(lo, hi, n_grid):
                if abs(d) < 1e-12:
                    continue
                h[i] -= d
                h[j] += d
                v = compute_c5(h)
                if v < best_v:
                    best_v = v
                    best_d = d
                h[i] += d
                h[j] -= d

            if best_d != 0.0:
                h[i] -= best_d
                h[j] += best_d
                cur_c5 = best_v
                improved += 1

    return h, improved, cur_c5

import numpy as np
import time
from scipy.optimize import linprog


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def get_corr_values(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def grad_lag(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + s < n:
            grad[j] += (1.0 - h[j + s]) * dx
        if 0 <= j - s < n:
            grad[j] -= h[j - s] * dx
    return grad


def solve_lp_step(h, active_indices, delta):
    n = len(h)
    corr_vals = get_corr_values(h)
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    A_rows = []
    b_vals = []
    for ki in active_indices:
        s = ki - (n - 1)
        g = grad_lag(h, s)
        f_k = corr_vals[ki]
        rhs = np.dot(g, h) - f_k
        row = np.zeros(nvar)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(rhs)
    if not A_rows:
        return None
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for j in range(n):
        bounds.append((max(0.0, h[j] - delta), min(1.0, h[j] + delta)))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def line_search_step(h0, h1, target, n_evals=25):
    best_alpha = 0.0
    best_c5 = compute_c5(h0)
    for a in np.linspace(0.02, 1.0, n_evals):
        h_try = h0 + a * (h1 - h0)
        h_try = project_onto_constraint(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_alpha = a
    if best_alpha == 0.0:
        return h0, best_c5
    h_new = h0 + best_alpha * (h1 - h0)
    h_new = project_onto_constraint(h_new, target)
    return h_new, best_c5


def pair_sweep(h, target, n_grid=20):
    n = len(h)
    improved = 0
    cur_c5 = compute_c5(h)
    for i in range(n):
        for j in range(i + 1, n):
            lo = max(-h[i], h[j] - 1.0)
            hi = min(h[i], 1.0 - h[j])
            if hi - lo < 1e-10:
                continue
            best_d = 0.0
            best_v = cur_c5
            for d in np.linspace(lo, hi, n_grid):
                if abs(d) < 1e-12:
                    continue
                h[i] -= d
                h[j] += d
                v = compute_c5(h)
                if v < best_v:
                    best_v = v
                    best_d = d
                h[i] += d
                h[j] -= d
            if best_d != 0.0:
                h[i] -= best_d
                h[j] += best_d
                cur_c5 = best_v
                improved += 1
    return h, improved, cur_c5


def optimize_from(h_start, target, sub_budget, rng, t_global):
    n = len(h_start)
    t_start = time.time()

        if len(active) > 25:

        h_new = solve_lp_step(h_lp, list(active), delta)
        if h_new is not None:

            if c5_new < best_c5 - 1e-9:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_lp = h_new.copy()
                delta = min(delta * 1.05, 0.2)
            elif c5_new < compute_c5(h_lp) - 1e-9:
                h_lp = h_new.copy()
            else:
                delta *= 0.7
        else:
            delta *= 0.7
        if delta < 0.008:
            break

    # Pair exchange phase
    h_polish = best_h.copy()
    plateau = 0
    while time.time() < t_global - 5:

        h_polish, improved, c5_p = pair_sweep(h_polish, target, ng)

            best_c5 = c5_p

        if plateau >= 4:

                h_polish = h_try.copy()
            plateau = 0
