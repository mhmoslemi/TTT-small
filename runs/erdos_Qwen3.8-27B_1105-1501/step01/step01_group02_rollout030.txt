    t = np.zeros(n, dtype=np.int64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = 1.0 - t.astype(float)  # t=0 -> h=1, t=1 -> h=0
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        # Flip some 1s to 0s
        idx_1 = np.where(h == 1)[0]
        rng_flip = np.random.RandomState(0)
        rng_flip.shuffle(idx_1)
        for idx in idx_1[:n_ones - n // 2]:
            h[idx] = 0.0
    elif n_ones < n // 2:
        idx_0 = np.where(h == 0)[0]
        rng_flip = np.random.RandomState(0)
        rng_flip.shuffle(idx_0)
        for idx in idx_0[:n // 2 - n_ones]:
            h[idx] = 1.0
    return h

def make_vdc_h(n):
    """Van der Corput based initial condition."""
    vdc = np.zeros(n)
    for i in range(n):
        v = 0.0
        d = 1.0
        x = i + 1
        while x > 0:
            v += (x % 2) * d
            x //= 2
            d /= 2.0
        vdc[i] = v
    sorted_idx = np.argsort(vdc)
    h = np.zeros(n)
    h[sorted_idx[:n // 2]] = 1.0
    return h

def pair_grad(h, i, j):
    """Compute g[k] for pair swap h[i]-δ, h[j]+δ."""
    n = len(h)
    g = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        v = 0.0
        a = i + s
        if 0 <= a < n:
            v += h[a]
        a = j + s
        if 0 <= a < n:
            v -= h[a]
        a = i - s
        if 0 <= a < n:
            v += h[a]
        a = j - s
        if 0 <= a < n:
            v -= h[a]
        g[k] = v
    return g

def best_delta(corr, g, lo, hi):
    """Find δ in [lo,hi] minimizing max(corr + δ*g). Returns (δ, val)."""
    if lo >= hi:
        return lo, float(np.max(corr))
    
    def f(d):
        return float(np.max(corr + d * g))
    
    # Coarse search
    m = 40
    ds = np.linspace(lo, hi, m)
    vs = np.empty(m)
    for idx in range(m):
        vs[idx] = f(ds[idx])
    bi = int(np.argmin(vs))
    
    if bi == 0:
        return lo, vs[0]
    if bi == m - 1:
        return hi, vs[-1]
    
    # Golden section
    a, b = ds[bi - 1], ds[bi + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(30):
        m1 = a + (1.0 - gr) * (b - a)
        m2 = a + gr * (b - a)
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    mid = (a + b) / 2.0
    return mid, f(mid)

def local_opt(h, corr, target_sum, dur, rng, max_step=1.0):
    """Local optimization via random pair swaps."""
    n = len(h)
    dx = 2.0 / n
    end = time.time() + dur
    bh = h.copy()
    bc = float(np.max(corr) * dx)
    
    while time.time() < end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[j], -max_step)
        dhi = min(h[i], max_step)
        if dhi - dlo < 1e-10:
            continue
        g = pair_grad(h, i, j)
        bd, bv = best_delta(corr, g, dlo, dhi)
        if bv < float(np.max(corr)) - 1e-10:
            h[i] = max(0.0, min(1.0, h[i] - bd))
            h[j] = max(0.0, min(1.0, h[j] + bd))
            corr = compute_corr(h)
            c = float(np.max(corr) * dx)
            if c < bc:
                bc = c
                bh = h.copy()
    return bh, bc
