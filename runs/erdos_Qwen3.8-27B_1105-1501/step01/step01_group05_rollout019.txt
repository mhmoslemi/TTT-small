import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n_target):
    h = np.array(v, dtype=float).ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    lo = target - n_target - 1.0
    hi = n_target - target + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(500):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta <= 0:

            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta <= 0:

            h[idx] -= delta
    return h


def scaled_corr(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode="full")


def lse_value(h, beta):
    c = scaled_corr(h)
    c_max = np.max(c)
    shifted = np.clip(beta * (c - c_max), -500, 500)
    return c_max + np.log(np.sum(np.exp(shifted))) / beta


def lse_grad(h, beta):
    c = scaled_corr(h)
    c_max = np.max(c)
    shifted = np.clip(beta * (c - c_max), -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    w = np.exp(shifted - log_sum)
    n = len(h)
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    idx = np.arange(n)
    grad = np.zeros(n)
    for ki in range(len(lags)):
        if w[ki] < 1e-15:
            continue
        k = lags[ki]
        wk = w[ki] * dx
        m1 = (idx + k >= 0) & (idx + k < n)
        if np.any(m1):
            grad[m1] += wk * (1.0 - h[idx[m1] + k])
        m2 = (idx - k >= 0) & (idx - k < n)
        if np.any(m2):
            grad[m2] -= wk * h[idx[m2] - k])
    return grad

    target_sum = n / 2.0

    def obj(x):
        return lse_value(x, beta)

    def obj_grad(x):
        return lse_grad(x, beta)

    def eq_fun(x):
        return np.sum(x) - target_sum

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    init_s = official_score(h)
    try:

            obj, h, jac=obj_grad, method='SLSQP',

def slsqp_minimax(h, maxiter=40, time_limit=25.0):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)

    idx = np.arange(n)
    init_score = official_score(h)

        g[n] = 1.0

        h_x = x[:n]
        theta = x[n]
        c = dx * np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - c

    def ineq_jac(x):

        J = np.zeros((n_lags, n + 1))
        J[:, n] = 1.0
        for ki, k in enumerate(lags):

                row[m2] -= h_x[idx[m2] - k]
            J[ki, :n] = -dx * row
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

        g = np.zeros(n + 1)
        g[:n] = 1.0

                    h_new[j] += d
                    h_new[l] -= d
                    if h_new[j] < -1e-10 or h_new[j] > 1.0 + 1e-10:
                        continue
                    if h_new[l] < -1e-10 or h_new[l] > 1.0 + 1e-10:
                        continue
                    s = official_score(h_new)
                    if s < best_new_score - 1e-12:
                        best_new_score = s
                        best_delta = d
                        best_l = l

            if best_l is not None:
                h[j] += best_delta
                h[best_l] -= best_delta
                score = best_new_score
                improved = True
        if not improved:
            break

    h = project_to_target(h, n)
    return h, official_score(h)


def interp_parent(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)


def gen_structured(n, rng):
    t = np.linspace(0, 1, n)
    freq = rng.uniform(2, 6)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.03, n)
    return project_to_target(h, n)


def gen_binary_balanced(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    k = n // 2
    h[idx[:k]] = 1.0
    if n % 2 == 1:
        h[idx[k]] = 1.0
    return project_to_target(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    deadline = t_start + budget_s - 5.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = project_to_target(parent, n_parent)
    else:
        n_parent = 65
        best_h = project_to_target(np.full(65, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    # Build diverse candidate pool
    candidates = [best_h.copy(), best_h[::-1].copy(),
                  project_to_target(1.0 - best_h, best_n),
                  project_to_target(1.0 - best_h[::-1], best_n)]

    n_list = sorted(set([best_n, max(40, best_n - 8), best_n + 8,
                         best_n + 16, max(40, best_n - 16), best_n + 24]))
    for n_try in n_list:
        if time.perf_counter() > deadline - 120:
            break
        candidates.append(interp_parent(best_h, n_try))
        candidates.append(interp_parent(best_h, n_try, offset=1))
        candidates.append(interp_parent(best_h, n_try, offset=2))

    for n_try in [best_n, max(40, best_n - 8), best_n + 8]:
        for _ in range(3):
            candidates.append(gen_structured(n_try, rng))
        for _ in range(2):
            candidates.append(gen_binary_balanced(n_try, rng))
        h_rand = rng.uniform(0, 1, n_try)
        candidates.append(project_to_target(h_rand, n_try))

    # Score and sort
    scored = []
    for h_c in candidates:
        if not np.all(np.isfinite(h_c)):
            continue
        h_c = project_to_target(h_c, len(h_c))
        s = official_score(h_c)
        if np.isfinite(s) and s < best_score + 0.01:
            scored.append((s, h_c))
    scored.sort(key=lambda x: (x[0], len(x[1])))

    if scored and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    # Main optimization on top candidates
    top_k = min(6, len(scored))
    for idx in range(top_k):
        if time.perf_counter() > deadline - 150:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        cur_score = official_score(h_c)

        # Log-sum-exp SLSQP with increasing beta
        for beta in [30, 80, 200, 500]:
            if time.perf_counter() > deadline - 100:
                break
            remaining = deadline - time.perf_counter()
            t_lim = min(18.0, max(4.0, remaining * 0.12))
            h_c, cur_score = slsqp_lse(h_c, beta, maxiter=35, time_limit=t_lim)

        # Coordinate descent
        if time.perf_counter() < deadline - 80:
            remaining = deadline - time.perf_counter()
            t_lim = min(12.0, max(3.0, remaining * 0.1))
            h_c, cur_score = coord_descent(h_c, max_rounds=35, time_limit=t_lim)

        # Minimax SLSQP
        if time.perf_counter() < deadline - 60:
            remaining = deadline - time.perf_counter()
            t_lim = min(14.0, max(4.0, remaining * 0.12))
            h_c, cur_score = slsqp_minimax(h_c, maxiter=30, time_limit=t_lim)

        # More coordinate descent
        if time.perf_counter() < deadline - 40:
            h_c, cur_score = coord_descent(h_c, max_rounds=25, time_limit=8.0)

        # Second LSE pass with higher beta
        if time.perf_counter() < deadline - 30:
            h_c, cur_score = slsqp_lse(h_c, 800, maxiter=20, time_limit=10.0)

        if cur_score < best_score:
            best_score = cur_score
            best_h = h_c.copy()
            best_n = n_c

    # Cross-resolution: interpolate best to new n and optimize
    if time.perf_counter() < deadline - 80:
        n_explore = [best_n, best_n + 8, best_n + 16, max(40, best_n - 8)]
        for n_try in n_explore:
            if time.perf_counter() > deadline - 50:
                break
            h_i = interp_parent(best_h, n_try)
            s_i = official_score(h_i)
            for beta in [50, 200]:
                if time.perf_counter() > deadline - 45:
                    break
                h_i, s_i = slsqp_lse(h_i, beta, maxiter=25, time_limit=8.0)
            h_i, s_i = coord_descent(h_i, max_rounds=20, time_limit=6.0)
            if time.perf_counter() < deadline - 30:
                h_i, s_i = slsqp_minimax(h_i, maxiter=20, time_limit=8.0)
            if s_i < best_score:
                best_score = s_i
                best_h = h_i.copy()
                best_n = n_try

    # Perturbation-based restart from best
    if time.perf_counter() < deadline - 40:
        for attempt in range(3):
            if time.perf_counter() > deadline - 35:
                break
            h_p = best_h + rng.normal(0, 0.05, best_n)
            h_p = project_to_target(h_p, best_n)
            for beta in [30, 100]:
                if time.perf_counter() > deadline - 30:
                    break
                h_p, s_p = slsqp_lse(h_p, beta, maxiter=20, time_limit=6.0)
            h_p, s_p = coord_descent(h_p, max_rounds=15, time_limit=4.0)
            if s_p < best_score:
                best_score = s_p
                best_h = h_p.copy()

    # Final polish
    if time.perf_counter() < deadline - 15:
        best_h, best_score = slsqp_minimax(best_h, maxiter=35, time_limit=8.0)
    if time.perf_counter() < deadline - 8:
        best_h, best_score = coord_descent(best_h, max_rounds=20, time_limit=4.0)
    if time.perf_counter() < deadline - 5:
        best_h, best_score = slsqp_lse(best_h, 500, maxiter=15, time_limit=2.0)

    # Final validation
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
