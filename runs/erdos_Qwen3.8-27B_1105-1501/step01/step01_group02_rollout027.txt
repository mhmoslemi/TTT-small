    improved_total = 0
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-12:
                    continue
                # Ternary search for optimal t in [lo, hi]
                a, b = lo, hi
                f0 = compute_c5(h)
                # Quick check: evaluate at midpoint
                h[i] += (a + b) / 2.0
                h[j] -= (a + b) / 2.0
                fm = compute_c5(h)
                h[i] -= (a + b) / 2.0
                h[j] += (a + b) / 2.0
                if fm >= f0 - 1e-13:
                    # Try endpoints quickly
                    best_t = 0.0
                    best_f = f0
                    h[i] += a
                    h[j] -= a
                    fa = compute_c5(h)
                    h[i] -= a
                    h[j] += a
                    if fa < best_f:
                        best_f = fa
                        best_t = a
                    h[i] += b
                    h[j] -= b
                    fb = compute_c5(h)
                    h[i] -= b
                    h[j] += b
                    if fb < best_f:
                        best_f = fb
                        best_t = b
                    if best_f < f0 - 1e-12:
                        h[i] += best_t
                        h[j] -= best_t
                        h = project_onto_constraint(h, target_sum)
                        improved = True
                        improved_total += 1
                    continue
                # Ternary search
                for _ter in range(30):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    h[i] += m1
                    h[j] -= m1
                    f1 = compute_c5(h)
                    h[i] -= m1
                    h[j] += m1
                    h[i] += m2
                    h[j] -= m2
                    f2 = compute_c5(h)
                    h[i] -= m2
                    h[j] += m2
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                t_opt = (a + b) / 2.0
                # Evaluate at t_opt
                h[i] += t_opt
                h[j] -= t_opt
                # Also check endpoints one more time
                h[i] -= t_opt
                h[j] += t_opt
                best_t = 0.0
                best_f = f0
                for t_try in [a, t_opt, b]:
                    h[i] += t_try
                    h[j] -= t_try
                    ft = compute_c5(h)
                    h[i] -= t_try
                    h[j] += t_try
                    if ft < best_f:
                        best_f = ft
                        best_t = t_try
                if best_f < f0 - 1e-12:
                    h[i] += best_t
                    h[j] -= best_t
                    h = project_onto_constraint(h, target_sum)
                    improved = True
                    improved_total += 1
        if not improved:
            break
    return h, improved_total


def l_bfgs_optimize(h, target_sum, T_list, rho=50.0, deadline=None):
    """Run L-BFGS-B at multiple temperature levels."""
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    for T in T_list:
        if deadline is not None and time.time() > deadline:
            break
        h_cur = project_onto_constraint(h_cur, target_sum)
        sp_minimize(
            smooth_max_obj, h_cur,
            args=(T, rho, n),
            jac=smooth_max_grad,
            args_grad=None,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-7, 'disp': False}
        ) if False else None
        # Use proper args for both fun and jac
        res = sp_minimize(
            lambda x: smooth_max_obj(x, T, rho, n),
            h_cur,
            jac=lambda x: smooth_max_grad(x, T, rho, n),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-7, 'disp': False}
        )
        if res.x is not None:
            h_new = project_onto_constraint(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
    return h_cur


def make_binary(h, target_sum):
    """Convert to nearest binary solution with exact sum."""
    n = len(h)
    nb = int(round(target_sum))
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n)
    for idx in sorted_idx[-nb:]:
        h_bin[idx] = 1.0
    return h_bin


def upsample(h, factor):
    """Upsample by interpolation."""
    n = len(h)
    n_new = n * factor
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    target = n_new / 2.0
    h_new = project_onto_constraint(h_new, target)
    return h_new


def downsample(h, factor):
    """Downsample by averaging."""
    n = len(h)
    n_new = n // factor
    if n_new * factor != n:
        return None
    h_reshaped = h.reshape(n_new, factor)
    h_new = h_reshaped.mean(axis=1)
    h_new[0] += 2.0 * n_new / 2.0 - h_new.sum()  # quick fix
    target = n_new / 2.0
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_onto_constraint(h_new, target)
    return h_new


def _obj_wrapper(x, T, rho, n):
    return smooth_max_obj(x, T, rho, n)


def _grad_wrapper(x, T, rho, n):
    return smooth_max_grad(x, T, rho, n)


def l_bfgs_no_lambda(h, target_sum, T_list, rho=50.0, deadline=None):
    """L-BFGS using global helpers to avoid lambdas."""
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    for T in T_list:
        if deadline is not None and time.time() > deadline:
            break
        h_cur = project_onto_constraint(h_cur, target_sum)
        res = sp_minimize(
            _obj_wrapper, h_cur,
            args=(T, rho, n),
            jac=_grad_wrapper,
            args_grad=None,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-7, 'disp': False}
        ) if False else None
        # scipy minimize: fun and jac can share args via args parameter
        res = sp_minimize(
            _obj_wrapper, h_cur,
            args=(T, rho, n),
            method='L-BFGS-B',
            bounds=bounds,
            jac=True,
            options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-7, 'disp': False}
        )
        if res.x is not None:
            h_new = project_onto_constraint(res.x, target_sum)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h_cur)
            if c5_new < c5_old:
                h_cur = h_new
    return h_cur

def l_bfgs_state(h, target_sum, T_list, rho=50.0, deadline=None):
    n = len(h)
    global _g_T, _g_rho, _g_n
    _g_rho = rho
    _g_n = n
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    for T in T_list:
        if deadline is not None and time.time() > deadline:
            break
        _g_T = T
        h_cur = project_onto_constraint(h_cur, target_sum)
        res = sp_minimize(
            _state_obj, h_cur,
            jac=_state_grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-7, 'disp': False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_onto_constraint(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
    return h_cur


def focused_pair_search(h, target_sum, deadline, max_rounds=100000):
    """Iterative pair search with focus on active lags."""
    n = len(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c5 = compute_c5(h)
        cv = corr_vec(h) * (2.0 / n)
        max_val = cv.max()
        # Find active lags (within 2% of max)
        active = np.where(cv > max_val * 0.98)[0]
        # Build set of active shift magnitudes
        active_shifts = set()
        for k in active:
            s = abs(k - (n - 1))
            active_shifts.add(s)
        # Try pairs that are at active shift distances first
        tried = set()
        best_improvement = 0.0
        best_move = None
        for i in range(n):
            for s in active_shifts:
                j = i + s
                if j < n and (min(i, j), max(i, j)) not in tried:
                    tried.add((min(i, j), max(i, j)))
                    lo = max(-h[i], h[j] - 1.0)
                    hi = min(1.0 - h[i], h[j])
                    if hi - lo < 1e-12:
                        continue
                    # Quick ternary search (fewer iterations for speed)
                    a, b = lo, hi
                    for _ in range(15):
                        m1 = a + (b - a) / 3.0
                        m2 = b - (b - a) / 3.0
                        h[i] += m1; h[j] -= m1
                        f1 = compute_c5(h)
                        h[i] -= m1; h[j] += m1
                        h[i] += m2; h[j] -= m2
                        f2 = compute_c5(h)
                        h[i] -= m2; h[j] += m2
                        if f1 < f2:
                            b = m2
                        else:
                            a = m1
                    t_opt = (a + b) / 2.0
                    h[i] += t_opt; h[j] -= t_opt
                    # Also check 0
                    best_t = 0.0
                    best_f = c5
                    f_opt = compute_c5(h)
                    if f_opt < best_f:
                        best_f = f_opt
                        best_t = t_opt
                    h[i] -= t_opt; h[j] += t_opt
                    if best_f < c5 - 1e-12 and c5 - best_f > best_improvement:
                        best_improvement = c5 - best_f
                        best_move = (i, j, best_t)
        if best_move is None:
            # Try all pairs
            for i in range(n):
                for j in range(i + 1, n):
                    lo = max(-h[i], h[j] - 1.0)
                    hi = min(1.0 - h[i], h[j])
                    if hi - lo < 1e-12:
                        continue
                    a, b = lo, hi
                    for _ in range(12):
                        m1 = a + (b - a) / 3.0
                        m2 = b - (b - a) / 3.0
                        h[i] += m1; h[j] -= m1
                        f1 = compute_c5(h)
                        h[i] -= m1; h[j] += m1
                        h[i] += m2; h[j] -= m2
                        f2 = compute_c5(h)
                        h[i] -= m2; h[j] += m2
                        if f1 < f2:
                            b = m2
                        else:
                            a = m1
                    t_opt = (a + b) / 2.0
                    h[i] += t_opt; h[j] -= t_opt
                    f_opt = compute_c5(h)
                    h[i] -= t_opt; h[j] += t_opt
                    if f_opt < c5 - best_improvement - 1e-12:
                        best_improvement = c5 - f_opt
                        best_move = (i, j, t_opt)
        if best_move is not None:
            i, j, t = best_move
            h[i] += t
            h[j] -= t
            h = project_onto_constraint(h, target_sum)
        else:
            break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    global _g_T, _g_rho, _g_n
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0

    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass

    if h0 is None or len(h0) == 0:
        n_init = 70
        h0 = np.zeros(n_init)
        h0[::2] = 1.0
        if n_init % 2 == 1:
            h0[-1] = 0.0
        h0 = project_onto_constraint(h0, n_init / 2.0)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS smooth optimization
    ph1_deadline = t0 + budget_s * 0.20
    T_list = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    h_smooth = l_bfgs_state(best_h, target_sum, T_list, rho=100.0, deadline=ph1_deadline)
    c5_smooth = compute_c5(h_smooth)
    if c5_smooth < best_c5:
        best_c5 = c5_smooth
        best_h = h_smooth.copy()

    # Also try from a slightly perturbed start
    if time.time() < ph1_deadline:
        h_pert = best_h + rng.randn(n) * 0.05
        h_pert = project_onto_constraint(h_pert, target_sum)
        h_pert_opt = l_bfgs_state(h_pert, target_sum, [0.05, 0.02, 0.01], rho=100.0, deadline=ph1_deadline)
        c5_pert = compute_c5(h_pert_opt)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert_opt.copy()

    # Phase 2: Focused pair search (main optimization)
    ph2_deadline = t0 + budget_s * 0.65
    h_work = best_h.copy()
    h_work = focused_pair_search(h_work, target_sum, ph2_deadline)
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()

    # Phase 3: Multiple rounds of pair search with perturbation
    remaining_deadline = t0 + budget_s * 0.90
    for restart in range(10):
        if time.time() > remaining_deadline:
            break
        h_try = best_h.copy()
        # Small perturbation
        n_pert = rng.randint(3, 8)
        idx = rng.choice(n, size=n_pert, replace=False)
        for ip in idx:
            h_try[ip] += rng.uniform(-0.05, 0.05)
        h_try = project_onto_constraint(h_try, target_sum)
        # Focus search
        slot_deadline = min(remaining_deadline, time.time() + (remaining_deadline - time.time()) / (10 - restart) - 1.0)
        h_try = focused_pair_search(h_try, target_sum, max(slot_deadline, time.time() + 2.0))
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Try binary rounding
    h_bin = make_binary(best_h, int(target_sum))
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        c5_bin2 = compute_c5(h_bin)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin.copy()
        # Also do pair search on binary solution
        if time.time() < t0 + budget_s - 10:
            h_bin_opt = focused_pair_search(h_bin.copy(), target_sum, t0 + budget_s - 5)
            c5_bin_opt = compute_c5(h_bin_opt)
            if c5_bin_opt < best_c5:
                best_c5 = c5_bin_opt
                best_h = h_bin_opt.copy()

    # Phase 5: Try upsampled version if time
    if time.time() < t0 + budget_s - 60 and n <= 100:
        h_up = upsample(best_h, 2)
        n_up = len(h_up)
        target_up = n_up / 2.0
        h_up = project_onto_constraint(h_up, target_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.005:  # only if competitive
            h_up = focused_pair_search(h_up, target_up, min(t0 + budget_s - 10, time.time() + 40))
            c5_up = compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()

    # Final: try mirror
    h_mirror = best_h[::-1].copy()
    c5_mirror = compute_c5(h_mirror)
    if c5_mirror < best_c5:
        best_c5 = c5_mirror
        best_h = h_mirror.copy()

    # Final polish
    if time.time() < t0 + budget_s - 3:
        h_final = focused_pair_search(best_h.copy(), target_sum, t0 + budget_s - 1)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_onto_constraint(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        # Recompute is just for safety
        pass
    best_c5 = final_c5

    return best_h, best_c5, len(best_h)
