def _smObj(x):
    h = np.clip(x, 0.0, 1.0)
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    T = _g_T
    m = np.max(vals)
    e = np.exp((vals - m) / T)
    s = np.sum(e)
    smooth = m + T * np.log(s)
    res = np.sum(h) - _g_target
    return smooth + _g_rho * res * res

def _smGrad(x):
    h = np.clip(x, 0.0, 1.0)
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    T = _g_T
    m = np.max(vals)
    e = np.exp((vals - m) / T)
    w = e / np.sum(e)

    for idx in range(2 * n - 1):
