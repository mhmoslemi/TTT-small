def make_feasible(z):
    n = len(z)
    target = n / 2.0
    h = np.clip(np.asarray(z, dtype=float), 0.0, 1.0).copy()
    residual = target - h.sum()
    if abs(residual) < 1e-14:
        return h
    if residual > 0:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual <= 1e-15:
                break

    cur_score = official_score(cur)
    all_p = all_pairs(n)
    n_rand_passes = min(10, max_passes // 3)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        if p < n_rand_passes:
            pl = random_pairs(n, min(len(all_p), 2000), rng)
        else:
            pl = all_p
            rng.shuffle(pl)
        cur, new_score = pair_descent_pass(cur, cur_score, pl, n_samples=10, deadline=deadline)
        if new_score >= cur_score - 1e-15:
            # No improvement, try with more samples
            cur2, sc2 = pair_descent_pass(cur, cur_score, pl, n_samples=14, deadline=deadline)
            if sc2 >= cur_score - 1e-15:
                # Try once more with different random pairs
                pl2 = random_pairs(n, min(len(all_p), 1500), rng)
                cur2, sc2 = pair_descent_pass(cur, cur_score, pl2, n_samples=10, deadline=deadline)
                if sc2 >= cur_score - 1e-15:
                    break
            cur, cur_score = cur2, sc2
        else:
            cur_score = new_score
    return cur, cur_score

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def full_optimize(h0, deadline, betas=None, cd_passes=40):
    if betas is None:
        betas = [100, 500, 2000, 8000, 30000, 100000]
    cur = h0.copy()
    cur_score = official_score(cur)
    for beta in betas:
        if time.time() > deadline:
            break
        h_new, sc = run_slsqp(cur, beta, maxiter=150)
        if h_new is not None and sc < cur_score - 1e-14:
            cur, cur_score = h_new, sc
        elif time.time() > deadline:
            break
    cur, cur_score = coordinate_descent(cur, deadline, max_passes=cd_passes,
                                        rng=np.random.default_rng(777))
    return cur, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 6 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError, Exception):
        pass

    if best_h is None:
        n0 = 60
        k = np.arange(n0)
        h_init = 0.5 + 0.25 * np.sin(2 * np.pi * k / n0) + 0.15 * np.cos(4 * np.pi * k / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    # Phase 1: SLSQP + CD on parent
    if time.time() < deadline:
        h_new, sc = full_optimize(best_h, deadline,
                                  betas=[200, 1000, 4000, 15000, 60000, 200000],
                                  cd_passes=50)
        if sc < best_score - 1e-14:
            best_h, best_score = h_new, sc

    # Phase 2: Multi-resolution
    n_base = len(best_h)
    n_opts = sorted(set([
        max(24, n_base - 15), max(24, n_base - 10), max(24, n_base - 5),
        n_base + 5 if n_base + 5 <= 80 else n_base - 5,
        n_base + 10 if n_base + 10 <= 80 else n_base + 5,
        n_base + 15 if n_base + 15 <= 80 else n_base + 10,
    ]))
    n_opts = [x for x in n_opts if x != n_base and 16 <= x <= 80]
    for n_opt in n_opts:
        if time.time() > deadline - 30:
            break
        h_int = interpolate_to(best_h, n_opt)
        sc_int = official_score(h_int)
        h_new, sc = full_optimize(h_int, deadline,
                                  betas=[200, 1000, 5000, 20000],
                                  cd_passes=30)
        if sc < best_score - 1e-14:
            best_h, best_score = h_new, sc

    # Phase 3: Random perturbations from best
    n = len(best_h)
    remaining = deadline - time.time()
    n_starts = max(3, min(12, int(remaining / 20)))
    sigmas = [0.02, 0.04, 0.06, 0.1, 0.03, 0.08, 0.05, 0.12, 0.025, 0.07, 0.04, 0.09]
    for i in range(n_starts):
        if time.time() > deadline - 10:
            break
        sigma = sigmas[i % len(sigmas)]
        h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
        h_new, sc = full_optimize(h_cand, deadline,
                                  betas=[500, 2000, 10000],
                                  cd_passes=20)
        if sc < best_score - 1e-14:
            best_h, best_score = h_new, sc

    # Phase 4: Long coordinate descent on best
    if time.time() < deadline - 5:
        rem = deadline - time.time()
        max_p = max(20, min(200, int(rem / 2)))
        best_h, best_score = coordinate_descent(best_h, deadline,
                                                max_passes=max_p,
                                                rng=np.random.default_rng(seed + 99))

    # Phase 5: Final SLSQP polish at very high beta
    if time.time() < deadline - 3:
        for beta in [100000, 300000]:
            if time.time() > deadline - 2:
                break
            h_new, sc = run_slsqp(best_h, beta, maxiter=100)
            if h_new is not None and sc < best_score - 1e-14:
                best_h, best_score = h_new, sc
                best_h, best_score = coordinate_descent(best_h, deadline, max_passes=15,
                                                        rng=np.random.default_rng(seed + 200))

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Safety fallback
    if not np.all(np.isfinite(final_h)) or not np.all(final_h >= -1e-12) or not np.all(final_h <= 1 + 1e-12):
        final_h = np.full(max(8, final_n), 0.5)
        final_score = official_score(final_h)
        final_n = len(final_h)

    final_bound = final_score + 1e-13
    return final_h, final_bound, final_n
