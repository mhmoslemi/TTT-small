import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        order = np.arange(n)
        for i in order:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def subgradient_step(h, step_size, n):
    """One step of projected subgradient method."""
    dx = 2.0 / n
    # Compute all F(k)
    F = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    
    C = F * dx
    cmax = np.max(C)
    
    # Find active set (shifts achieving the max)
    active = C >= cmax - 1e-10
    
    # Compute subgradient
    grad = np.zeros(n)
    for ki in np.where(active)[0]:
        k = ki - n + 1
        start = max(0, -k)
        end = min(n, n - k)
        for j in range(start, end):
            # Gradient contribution from shift k at position j
            # h[j] appears as h[i] with i=j: contributes (1-h[j+k])
            if j + k < n and j + k >= 0:
                grad[j] += (1.0 - h[j + k])
            # h[j] appears as h[i+k] with i=j-k: contributes -h[j-k]
            if j - k >= 0 and j - k < n:
                grad[j] += (-h[j - k])
    grad *= dx / np.sum(active)
    
    # Update
    h_new = h - step_size * grad
    # Project onto feasible set
    h_new = make_feasible(h_new)

def compute_all_gradients(h, n):
    """Compute gradients of all g_k. Returns array of shape (2n-1, n)."""
    grads = np.zeros((2*n-1, n))
    for idx, k in enumerate(range(-(n-1), n)):
        start = max(0, -k)
        end = min(n, n - k)
        # First part: (1 - h[j+k]) for j in [start, end)

        else:

        j_end = min(n, end + k)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        s = max(0, -k)
        e = min(n, n - k)
        F[k + n - 1] = np.dot(h[s:e], 1.0 - h[s + k:e + k])
    return F

    dx = 2.0 / n

    for it in range(max_iter):
