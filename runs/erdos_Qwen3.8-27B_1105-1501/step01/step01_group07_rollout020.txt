```python
import numpy as np
import time
import cvxpy as cp

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def active_gradients(h, scores, tol=0.01):
    """Compute gradients for active lags and their average."""
    n = len(h)
    max_s = np.max(scores)
    active = np.where(scores >= max_s - tol)[0]
    
    # Compute gradient for each active lag
    # scores has length 2n-1, index j corresponds to lag k = j - (n-1)
    grads = []
    for j in active:
        k = j - (n - 1)  # lag
        if k == 0:
            # r[n-1] = sum h[i]*(1-h[i])
            # d/dh[j] = 1 - 2*h[j]
            g = 1.0 - 2.0 * h
        elif k > 0:
            # r[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[j] = (1-h[j-k]) if j >= k, minus h[j+k] if j <= n-1-k
            g = np.zeros(n)
            if k <= n - 1:
                g[k:] += (1.0 - h[:n-k])  # (1-h[j-k]) for j = k, ..., n-1
                g[:n-k] -= h[k:]  # -h[j+k] for j = 0, ..., n-1-k
        else:
            # k < 0, let d = -k
            d = -k
            # r[n-1-d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # d/dh[j] = (1-h[j+d]) if j <= n-1-d, minus h[j-d] if j >= d
            g = np.zeros(n)
            g[:n-d] += (1.0 - h[d:])  # (1-h[j+d]) for j = 0, ..., n-1-d
            g[d:] -= h[:n-d]  # -h[j-d] for j = d, ..., n-1
        grads.append(g)
    
    avg_grad = np.mean(grads, axis=0)
    return avg_grad, active

def lp_step(h, scores, rho, active_tol=0.01):
    """Solve the trust-region LP to find a step."""
    n = len(h)
    max_s = np.max(scores)
    active = np.where(scores >= max_s - active_tol)[0]
    
    # Compute gradients for active lags
    grad_list = []
    for j in active:
        k = j - (n - 1)
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            d = -k
            g = np.zeros(n)
            g[:n-d] += (1.0 - h[d:])
            g[d:] -= h[:n-d]
        grad_list.append(g)
    
    G = np.array(grad_list)  # shape (|K|, n)
    
    # LP: minimize t
    # s.t. G @ delta <= t - scores[active]
    #      max(-h[i], -rho) <= delta[i] <= min(1-h[i], rho)
    #      sum(delta) = 0
    
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [G @ delta <= t - scores[active]]
    lo = np.maximum(-h, -rho)
    hi = np.minimum(1.0 - h, rho)
    constraints.append(delta >= lo)
    constraints.append(delta <= hi)
    constraints.append(cp.sum(delta) == 0)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
    except:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except:
            return None, None
    
    if delta.value is None or t.value is None:
        return None, None
    
    if t.value >= max_s - 1e-10:
        return None, None  # No improvement possible in this trust region
    
    return delta.value, t.value
```

            if delta.value is not None:
                solved = True
                break
        except Exception:
            continue
    
    if not solved or delta.value is None:
        return None, None
    
    dv = np.asarray(delta.value, dtype=np.float64)
    tv = float(t.value)
    if tv >= max_s - 1e-12:
        return None, None
    return dv, tv

def project_subgradient_step(h, g, alpha, n):
    """Take a projected step: h + alpha*(-g), projected onto feasible set."""
    cand = make_feasible(h - alpha * g, n)
    return cand

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(20):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                best_a = 0.0
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

def grad_descent(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        w, g = smooth_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.3:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.85
            stall += 1
            if stall > 40:
                cur = make_feasible(cur + rng.normal(0, max(0.003, T * 0.3), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.003)
                stall = 0
        else:
            step = min(step * 1.05, T * 2.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 20:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)
