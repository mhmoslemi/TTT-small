def lp_step(h, dx, n, active_lags, delta=None):
    # h: current point, shape (n,)
    # active_lags: list of lag offsets (can be negative or positive)
    # Returns: new_h, new_score
    
    # Compute gradients for each active lag
    # For lag d: f_d(h) = dx * sum_i h[i] * (1-h[i+d])
    # df_d/dh[i] = dx * [(1-h[i+d]) if i+d valid, 0 otherwise] - dx * [h[i-d] if i-d valid, 0 otherwise]
    
    # Build the LP
    # Minimize t
    # Subject to:
    #   For each active lag d: linearized_f_d(h) ≤ t
    #   sum(h) = n/2
    #   0 ≤ h_i ≤ 1
    
    # c = [0, 0, ..., 0, 1] (minimize t, which is the last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active lag d:
    #   dx * sum_i g_i^d * h_i - t ≤ f_d(h) - dx * sum_i g_i^d * h_i^*  ... wait
    #   
    # Linearized: f_d(h) ≈ f_d(h*) + sum_i g_i^d * (h_i - h_i^*)
    # where g_i^d = dx * [(1-h_{i+d}^*) * δ(i+d valid) - h_{i-d}^* * δ(i-d valid)]
    #
    # Constraint: f_d(h*) + sum_i g_i^d * h_i - sum_i g_i^d * h_i^* ≤ t
    #   sum_i g_i^d * h_i - t ≤ sum_i g_i^d * h_i^* - f_d(h*)
    
    A_rows = []
    b_rows = []
    
    scores = np.correlate(h, 1-h, mode='full') * dx
    c_idx = n - 1
    
    for d in active_lags:
        idx = c_idx + d
        if idx < 0 or idx >= 2*n - 1:
            continue
        # Compute gradient of overlap at lag d
        g = np.zeros(n)
        if d >= 0:
            # f = dx * sum_{i=0}^{n-1-d} h[i] * (1 - h[i+d])
            for i in range(n - d):
                g[i] += dx * (1 - h[i+d])
                g[i+d] += -dx * h[i]
        else:
            # f = dx * sum_{i=-d}^{n-1} h[i] * (1 - h[i+d])  where i+d >= 0
            dd = -d
            for i in range(dd, n):
                g[i] += dx * (1 - h[i - dd])
                g[i - dd] += -dx * h[i]
        
        rhs = float(np.dot(g, h)) - scores[idx]
        # sum_i g_i * h_i - t ≤ rhs
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 ≤ h_i ≤ 1, -inf ≤ t ≤ inf (or t ≥ 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    # Solve
    from scipy.optimize import linprog
    res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        new_h = res.x[:n]
        new_h = np.clip(new_h, 0, 1)
        # Fix sum
        diff = n/2 - np.sum(new_h)

            new_h = make_feasible(new_h, n)
        return new_h, score_exact(new_h)
    return None, None

def make_feasible(y, n=None):
    if n is None:

            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
