def smooth_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    m = np.max(F)
    scaled = (F - m) / temp
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        a = max(0, s)
        b = min(n - 1, n - 1 + s)
        if a <= b:
            grad[a:b+1] += wk * (1.0 - h[a-s:b-s+1])
        a2 = max(0, -s)
        b2 = min(n - 1, n - 1 - s)
        if a2 <= b2:
            grad[a2:b2+1] -= wk * h[a2+s:b2+s+1]
    return grad, m

def pair_search(h, i, j, n_samples):

    lo = max(-hi, hj - 1.0)

        return exact_c5(h)
    t_vals = np.linspace(lo, hi_b, n_samples)
    best_c5 = exact_c5(h)
    best_t = 0.0
    for t in t_vals:
        h[i] = hi + t
        h[j] = hj - t
        c = exact_c5(h)
        if c < best_c5:
            best_c5 = c
            best_t = t
    h[i] = hi + best_t
    h[j] = hj - best_t
    return best_c5
