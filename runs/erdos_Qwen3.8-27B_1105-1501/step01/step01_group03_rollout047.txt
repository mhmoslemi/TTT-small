from scipy.optimize import minimize

def official_score(h):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        order = np.arange(n)
        for i in order:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def project_to_binary(h, target_n=None):
    h = np.asarray(h, dtype=float).ravel()
    if target_n is None:
        target_n = len(h)
    if len(h) != target_n:
        x_src = (np.arange(len(h)) + 0.5) * 2.0 / len(h)
        x_tgt = (np.arange(target_n) + 0.5) * 2.0 / target_n
        h = np.interp(x_tgt, x_src, h)
    target = target_n // 2
    if target * 2 != target_n:
        return make_feasible(h)
    sorted_idx = np.argsort(h)[::-1]
    result = np.zeros(target_n)
    result[sorted_idx[:target]] = 1.0
    return result


def local_search_binary(h_bin, time_limit=30.0, t_start=None, max_rounds=2000):

    if n % 2 != 0:
        return h_bin.copy(), official_score(h_bin)
    h = h_bin.copy().astype(float)
    score = official_score(h)
    if t_start is None:
        t_start = time.time()
    round_num = 0
    while round_num < max_rounds:
        round_num += 1
        if time.time() - t_start > time_limit:

        best_delta = 1e-15
        best_swap = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

def smooth_obj_and_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    C = corr * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        val = cmax + np.log(len(C)) / beta
        weights = np.ones(len(C)) / len(C)
    else:
        val = cmax + (mk + np.log(ws)) / beta
        weights = es / ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[k + n - 1]
        if abs(w) < 1e-15:
            continue
        start = max(0, -k)
        end = min(n, n - k)
        g1 = np.zeros(n)
        g1[start:end] = (1.0 - x[start + k:end + k])
        p_start = max(0, start + k)
        p_end = min(n, end + k)
        g1[p_start:p_end] += (-x[p_start - k:p_end - k])
        grad += w * g1
    return val, grad * dx

    n = len(h0)
    h0 = make_feasible(h0)
    obj_func = lambda x: smooth_obj_and_grad(x, beta)[0]
    jac_func = lambda x: smooth_obj_and_grad(x, beta)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_func, h0, jac=jac_func, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0


def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    if parent is not None:
        parent = make_feasible(parent)
        parent_n = len(parent)
        best_h = parent.copy()
        best_score = official_score(best_h)
    else:
        parent_n = 60
        best_h = np.full(parent_n, 0.5)
        best_score = official_score(best_h)

    # Determine candidate even lengths
    base_n = parent_n if parent_n % 2 == 0 else parent_n - 1
    lengths = [base_n]
    for d in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, 12, -12]:
        nl = base_n + d
        if 20 <= nl <= 80 and nl % 2 == 0 and nl not in lengths:
            lengths.append(nl)
    lengths.sort()

    # Phase 1: Binary local search at multiple lengths
    for n in lengths:
        if time.time() > deadline - 80:
            break

        starts = []
        if parent is not None:
            starts.append(project_to_binary(parent, n))
            # Symmetry transform
            starts.append(project_to_binary(1.0 - parent[::-1], n))

        # Random starts
        for _ in range(12):
            idx = rng.choice(n, size=n // 2, replace=False)
            h = np.zeros(n)
            h[idx] = 1.0
            starts.append(h)

        # Modular constructions
        for step in range(2, n):
            g = int(np.gcd(step, n))
            num_cycles = n // g
            if num_cycles == 2 and g == n // 2:
                pos = set()
                for i in range(n // 2):
                    pos.add((i * step) % n)
                if len(pos) == n // 2:
                    h = np.zeros(n)
                    for p in pos:
                        h[p] = 1.0
                    starts.append(h)
                    break
        # "Interleaved" pattern
        h_alt = np.zeros(n)
        h_alt[::3] = 1.0
        if np.sum(h_alt) < n // 2:
            idx = np.where(h_alt == 0)[0]
            extra = n // 2 - int(np.sum(h_alt))
            if extra > 0 and len(idx) >= extra:
                h_alt[idx[:extra]] = 1.0
        elif np.sum(h_alt) > n // 2:
            idx = np.where(h_alt == 1)[0]
            extra = int(np.sum(h_alt)) - n // 2
            h_alt[idx[-extra:]] = 0.0
        if int(np.sum(h_alt)) == n // 2:
            starts.append(h_alt)

        # Score and sort starting points
        scored = []
        seen_set = set()
        for s in starts:
            sc = official_score(s)
            key = tuple(s.astype(int).tolist())
            if key in seen_set:
                continue
            seen_set.add(key)
            scored.append((sc, s))
        scored.sort(key=lambda x: x[0])

        # Local search on top candidates
        top_k = min(6, len(scored))
        for i in range(top_k):
            if time.time() > deadline - 70:
                break
            h0 = scored[i][1].copy()
            h_ls, ls_score = local_search_binary(h0, time_limit=25.0, t_start=t_start)
            if ls_score < best_score - 1e-14:
                best_score = ls_score
                best_h = h_ls.copy()

        # Kick-and-search on the best binary at this length
        if time.time() > deadline - 60:
            break
        if scored:
            h_kick, kick_score = kick_and_search(
                scored[0][1], n_kicks=5, kick_size=3,
                time_limit=40.0, t_start=t_start, rng=rng)
            if kick_score < best_score - 1e-14:
                best_score = kick_score
                best_h = h_kick.copy()

    # Phase 2: If we found a good binary solution, do continuous refinement
    if time.time() < deadline - 40:
        best_n = len(best_h)
        h_cont = make_feasible(best_h.astype(float))
        for beta in [30, 100, 500, 2000]:
            if time.time() > deadline - 20:
                break
            h_cont = smooth_solve(h_cont, beta, maxiter=150)
            sc = official_score(h_cont)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_cont.copy()

    # Phase 3: Try alternative lengths for continuous optimization
    if time.time() < deadline - 30:
        for dn in [-2, 2, -4, 4]:
            if time.time() > deadline - 15:
                break
            alt_n = len(best_h) + dn
            if alt_n < 20 or alt_n > 80:
                continue
            alt_h = interp_to_length(best_h, alt_n)
            for beta in [100, 500]:
                if time.time() > deadline - 10:
                    break
                alt_h = smooth_solve(alt_h, beta, maxiter=120)
                sc = official_score(alt_h)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = alt_h.copy()

    # Phase 4: Final binary refinement on best continuous solution
    if time.time() < deadline - 20:
        bn = len(best_h)
        if bn % 2 == 0:
            h_bin = project_to_binary(best_h, bn)
            h_ls, ls_score = local_search_binary(h_bin, time_limit=20.0, t_start=t_start)
            if ls_score < best_score - 1e-14:
                best_score = ls_score
                best_h = h_ls.copy()
            # Also try continuous from this binary
            h_c2 = make_feasible(h_ls.copy())
            h_c2 = smooth_solve(h_c2, 500, maxiter=100)
            sc2 = official_score(h_c2)
            if sc2 < best_score - 1e-14:
                best_score = sc2
                best_h = h_c2.copy()

    # Final validation
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score < 0:
        if parent is not None:
            best_h = parent.copy()
            final_score = official_score(best_h)
        else:
            best_h = make_feasible(np.full(60, 0.5))
            final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
