import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_to_target(v, target):
    v = np.array(v, dtype=float)
    v[~np.isfinite(v)] = 0.5
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_all_grads(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for j_out in range(2 * n - 1):
        k = j_out - (n - 1)
        g = abs(k)
        mask1 = np.zeros(n)
        mask2 = np.zeros(n)
        if g == 0:
            mask1[:] = 1.0
            mask2[:] = 1.0
            grads[j_out] = (1.0 - h) - h
        else:
            mask1[g:] = 1.0
            shifted1 = np.zeros(n)
            shifted1[g:] = h[:n - g]
            mask2[:n - g] = 1.0
            shifted2 = np.zeros(n)
            shifted2[:n - g] = h[g:]
            grads[j_out] = mask1 * (1.0 - shifted1) - mask2 * shifted2
    return grads

def solve_lp(h, scores, grads, active_idx, radius, mu):
    n = len(h)
    # Variables: [p (n), r (1), tau (1)]
    # Objective: minimize mu * r + tau
    c = np.zeros(n + 2)
    c[n] = mu      # coefficient of r
    c[n + 1] = 1.0  # coefficient of tau
    
    # Inequality constraints
    A_rows = []
    b_rows = []
    
    # Lag constraints: grad_l · p - tau <= -s_l
    for idx in active_idx:
        row = np.zeros(n + 2)
        row[:n] = grads[idx]
        row[n + 1] = -1.0
        A_rows.append(row)
        b_rows.append(-scores[idx])
    
    # p_i - r <= 0
    for i in range(n):
        row = np.zeros(n + 2)
        row[i] = 1.0
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # -p_i - r <= 0
    for i in range(n):
        row = np.zeros(n + 2)
        row[i] = -1.0
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, radius))  # r
    bounds.append((-np.inf, np.inf))  # tau
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    else:
        return None

def slsqp_beta(h, n, target_sum, beta, maxiter):
    # Smooth objective using log-sum-exp
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        dx = 2.0 / n
        c_vals = corr * dx
        c_max = np.max(c_vals)
        shifted = np.clip(beta * (c_vals - c_max), -500, 500)
        return c_max + np.log(np.sum(np.exp(shifted))) / beta
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        dx = 2.0 / n
        c_vals = corr * dx
        c_max = np.max(c_vals)
        shifted = np.clip(beta * (c_vals - c_max), -500, 500)
        exp_s = np.exp(shifted)
        weights = exp_s / np.sum(exp_s)
        # Gradient computation (same as before)
        g = np.zeros(n)
        m = np.arange(n)
        for d in range(-(n - 1), n):
            w = weights[d + n - 1] * dx
            if abs(w) < 1e-15:
                continue
            if d >= 0:
                idx1 = m[m >= d]
                if len(idx1) > 0:
                    g[idx1] += w * (1.0 - x[idx1 - d])
                idx2 = m[m + d < n]
                if len(idx2) > 0:
                    g[idx2] -= w * x[idx2 + d]
            else:
                d2 = -d
                idx1 = m[m < n - d2]
                if len(idx1) > 0:
                    g[idx1] += w * (1.0 - x[idx1 + d2])
                idx2 = m[m >= d2]
                if len(idx2) > 0:
                    g[idx2] -= w * x[idx2 - d2]
        return g
    
    def constraint_eq(x):
        return np.sum(x) - target_sum
    
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': constraint_eq, 'args': ()}]
    
    res = minimize(obj, h, method='SLSQP', jac=grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    return res.x

def binary_local_search(h_binary, n, target_sum, max_iters, rng):
    # Local search on binary sequences
    h = h_binary.copy()
    best_c5, _ = exact_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for _ in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        c5_new, _ = exact_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
    return h, best_c5
