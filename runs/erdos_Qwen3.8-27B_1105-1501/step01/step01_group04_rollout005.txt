```python
def compute_all_corr_grads(h):
    """Compute gradient of each corr[k] w.r.t. h. Returns array of shape (2n-1, n)."""
    n = len(h)
    L = 2 * n - 1
    grads = np.zeros((L, n))
    for k in range(L):
        # grad[k][m] = (1-h[m-k+n-1]) * [0<=m-k+n-1<n] - h[m+k-n+1] * [0<=m+k-n+1<n]
        m = np.arange(n)
        idx1 = m - k + n - 1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = m + k - n + 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        g = np.zeros(n)
        g[valid1] = (1 - h[idx1[valid1]])
        g[valid2] -= h[idx2[valid2]]
        grads[k] = g
    return grads
```

_slqp_n = 0
_slqp_dx = 0.0
_slqp_target = 0.0
_slqp_active = None

def _slqp_obj(y):
    return y[0]

def _slqp_obj_grad(y):
    g = np.zeros(_slqp_n + 1)
    g[0] = 1.0
    return g

def _slqp_ineq_fun(y):
    h = y[1:]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return y[0] - corr[_slqp_active] * _slqp_dx

def _slqp_ineq_jac(y):
    h = y[1:]
    n = _slqp_n
    active = _slqp_active
    m = len(active)
    jac = np.zeros((m, n + 1))
    m_arr = np.arange(n)
    for i, k in enumerate(active):
        jac[i, 0] = 1.0
        idx1 = n - 1 - k + m_arr
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = k - n + 1 + m_arr
        valid2 = (idx2 >= 0) & (idx2 < n)
        gk = np.zeros(n)
        gk[valid1] = 1.0 - h[idx1[valid1]]
        gk[valid2] -= h[idx2[valid2]]
        jac[i, 1:] = -gk * _slqp_dx
    return jac

def _slqp_eq_fun(y):
    return np.array([np.sum(y[1:]) - _slqp_target])

def _slqp_eq_jac(y):
    J = np.zeros((1, _slqp_n + 1))
    J[0, 1:] = 1.0
    return J

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, target):
    """Project h to [0,1]^n with sum = target using bisection."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    shift = (lo + hi) / 2.0
    result = np.clip(h - shift, 0.0, 1.0)
    # Exact adjustment for remaining error
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        interior = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(interior):
            adj = diff / np.sum(interior)
            result[interior] += adj
            result = np.clip(result, 0.0, 1.0)
        else:
            # Force adjustment on the element closest to 0.5
            idx = np.argmin(np.abs(result - 0.5))
            result[idx] += diff
            result = np.clip(result, 0.0, 1.0)
    return result
