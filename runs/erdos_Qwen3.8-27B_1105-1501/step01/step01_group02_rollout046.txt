            best_delta = d
    return best_delta


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h = np.zeros(70)
        h[::2] = 1.0

    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_onto_constraint(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth max gradient descent (35% budget)
    ph1_end = t0 + budget_s * 0.35
    temp = 0.06
    lr = 0.04
    stall = 0

    while time.time() < ph1_end:
        grad = smooth_max_gradient(h, temp)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old - 1e-12:
            h = h_new
            stall = 0
            lr = min(lr * 1.015, 0.12)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            stall += 1
            lr *= 0.94
            if stall > 60:
                temp = max(temp * 0.65, 0.003)
                lr = max(lr, 0.015)
            if stall > 250:
                h = best_h + rng.randn(n) * 0.06
                h = project_onto_constraint(h, target_sum)
                temp = 0.05
                lr = 0.03
                stall = 0

    # Phase 2: Pair line search (45% budget)
    ph2_end = t0 + budget_s * 0.80
    h = best_h.copy()
    shifts_val = compute_shifts(h)
    no_improve = 0

    while time.time() < ph2_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] < 0.005 or h[j] > 0.995:
            continue
        delta = find_best_pair_delta(h, i, j, shifts_val, dx, n, rng)
        if abs(delta) > 1e-9:
            h[i] -= delta
            h[j] += delta
            shifts_val = compute_shifts(h)
            c5_cur = np.max(shifts_val)
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 300:
            h = best_h.copy()
            n_pert = max(3, n // 15)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h[pert_idx] += rng.randn(n_pert) * 0.08
            h = project_onto_constraint(h, target_sum)
            shifts_val = compute_shifts(h)
            no_improve = 0

    # Phase 3: Fine polishing (20% budget)
    h = best_h.copy()
    shifts_val = compute_shifts(h)

    while time.time() < t0 + budget_s - 5:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = find_best_pair_delta(h, i, j, shifts_val, dx, n, rng)
        if abs(delta) > 1e-10:
            h[i] -= delta
            h[j] += delta
            shifts_val = compute_shifts(h)
            c5_cur = np.max(shifts_val)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()

    # Final feasibility
    best_h = project_onto_constraint(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
