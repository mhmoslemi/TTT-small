def lp_gradient(h, p=2.0):
    """Gradient of sum_k C(k)^p"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx  # shape (2n-1,)
    
    # w[m] = p * C[m]^(p-1) * dx
    # Handle C[m] < 0: use |C[m]|^(p-1) * sign or just C[m]^(p-1)
    # Since C(k) can be negative, for p not integer we need care
    # Use |C|^(p-1) * C = C * |C|^(p-2) to handle signs
    # Actually for even p, C^p is always positive, and d(C^p)/dC = p*C^(p-1)
    # For odd p, same formula works
    # Let's just use C^(p-1) which handles signs correctly for integer p
    if p == 1.0:
        w = np.ones_like(C)
    elif p == 2.0:
        w = 2.0 * C
    else:
        # For general p, use C * |C|^(p-2) = C^(p-1) for C>=0
        # For C<0 and odd p: C^(p-1) = |C|^(p-1) * sign(C)^(p-1)
        # For C<0 and even p: C^(p-1) = -|C|^(p-1)
        # Simplest: just use np.sign(C) * np.abs(C)**(p-1)
        w = p * np.sign(C) * np.abs(C) ** (p - 1)

        d_min = max(-best_h[i], best_h[j] - 1.0)

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def get_C_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_box_sum(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def _correlation_grad(h, w):
    """Compute sum_k w[k] * dC(k)/dh for all h simultaneously.
    w has shape (2n-1,) corresponding to lags -(n-1)..(n-1)."""
    n = len(h)
    lags = np.arange(-(n - 1), n)
    J = np.arange(n)[:, None]
    L = lags[None, :]
    jp = J + L
    jm = J - L
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    jp_safe = np.clip(jp, 0, n - 1)
    jm_safe = np.clip(jm, 0, n - 1)
    term_p = valid_p.astype(float) * (1.0 - h[jp_safe])
    term_m = valid_m.astype(float) * h[jm_safe]
    return (term_p - term_m) @ w


def lp_gradient(h, p):
    n = len(h)
    dx = 2.0 / n
    C = get_C_values(h)
    if p == 1.0:
        w = np.ones_like(C) * dx
    elif p == 2.0:
        w = 2.0 * C * dx
    else:
        w = p * np.sign(C) * np.abs(C) ** (p - 1) * dx
    return _correlation_grad(h, w)


def smooth_max_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    C = get_C_values(h)
    max_C = np.max(C)
    exp_C = np.exp(alpha * (C - max_C))
    p = exp_C / np.sum(exp_C)
    w = p * dx
    return _correlation_grad(h, w)


def threshold_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    C = get_C_values(h)
    violations = np.maximum(0.0, C - tau)
    w = 2.0 * violations * dx
    return _correlation_grad(h, w)


def gradient_solve(h0, target, grad_fn, time_limit, lr_init, momentum_init, max_steps=5000):
    """Run projected gradient descent with momentum."""
    n = len(h0)
    h = h0.astype(float).copy()
    v = np.zeros(n)
    lr = lr_init
    momentum = momentum_init
    t0 = time.time()
    
    for step in range(max_steps):
        if time.time() - t0 > time_limit:
            break
        grad = grad_fn(h)
        v = momentum * v - lr * grad
        h = project_box_sum(h + v, target)
        lr = max(lr * 0.9997, lr_init * 0.05)
        momentum = min(momentum + 0.00003, 0.995)
    return h


def refine_targeted(h, target, time_limit, seed=42):
    """Targeted coordinate pair moves focusing on bottleneck lags."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    while time.time() - t0 < time_limit:
        C = get_C_values(best_h)
        # Get top lags
        top_k = np.argsort(C)[-5:][::-1]
        progressed = False
        
        for lag_idx in top_k:
            lag = int(lag_idx - (n - 1))
            if lag == 0:
                continue
            # Find contributing pairs where h[i]*(1-h[i+lag]) is large
            if lag > 0:
                valid_i = np.arange(0, n - lag)
            else:
                valid_i = np.arange(0, n + lag)
            
            if len(valid_i) == 0:
                continue
            
            contrib = best_h[valid_i] * (1.0 - best_h[valid_i + lag])
            # Try to fix the worst contributing pair
            if np.max(contrib) < 0.01:
                continue
            worst_local = np.argmax(contrib)
            i = valid_i[worst_local]
            j = i + lag
            
            # Try moving mass between i and j
            d_min = max(-best_h[i], best_h[j] - 1.0)
            d_max = min(1.0 - best_h[i], best_h[j])
            if d_max - d_min < 1e-6:
                # Try moving with a third point
                k = rng.randint(n)
                if k == i or k == j:
                    continue
                d_min2 = max(-best_h[i], best_h[k] - 1.0)
                d_max2 = min(1.0 - best_h[i], best_h[k])
                if d_max2 - d_min2 < 1e-6:
                    continue
                for frac in [0.3, 0.5, 0.7, -0.3, -0.5, -0.7]:
                    delta = frac * (d_max2 - d_min2) + d_min2
                    if delta <= d_min2 + 1e-10 or delta >= d_max2 - 1e-10:
                        continue
                    h_test = best_h.copy()
                    h_test[i] += delta
                    h_test[k] -= delta
                    if h_test[i] < -1e-10 or h_test[i] > 1 + 1e-10:
                        continue
                    if h_test[k] < -1e-10 or h_test[k] > 1 + 1e-10:
                        continue
                    c5 = compute_c5(h_test)
                    if c5 < best_c5 - 1e-12:
                        best_c5 = c5
                        best_h = h_test.copy()
                        progressed = True
                        break
            else:
                for frac in [0.3, 0.5, 0.7, -0.3, -0.5, -0.7]:
                    delta = frac * (d_max - d_min) + d_min
                    if delta <= d_min + 1e-10 or delta >= d_max - 1e-10:
                        continue
                    h_test = best_h.copy()
                    h_test[i] += delta
                    h_test[j] -= delta
                    if h_test[i] < -1e-10 or h_test[i] > 1 + 1e-10:
                        continue
                    if h_test[j] < -1e-10 or h_test[j] > 1 + 1e-10:
                        continue
                    c5 = compute_c5(h_test)
                    if c5 < best_c5 - 1e-12:
                        best_c5 = c5
                        best_h = h_test.copy()
                        progressed = True
                        break
    
        if not progressed:
            # Random pair moves as fallback
            for _ in range(200):
                if time.time() - t0 > time_limit:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d_min = max(-best_h[i], best_h[j] - 1.0)
                d_max = min(1.0 - best_h[i], best_h[j])
                if d_max - d_min < 1e-6:
                    continue
                delta = rng.uniform(0.2, 0.8) * (d_max - d_min) + d_min
                if delta <= d_min + 1e-10 or delta >= d_max - 1e-10:
                    continue
                h_test = best_h.copy()
                h_test[i] += delta
                h_test[j] -= delta
                if h_test[i] < -1e-10 or h_test[i] > 1 + 1e-10:
                    continue
                if h_test[j] < -1e-10 or h_test[j] > 1 + 1e-10:
                    continue
                c5 = compute_c5(h_test)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h_test.copy()
    
    return best_h, best_c5

    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: L_p gradient descent with increasing p (55% budget)
    p_schedule = [2, 3, 4, 6, 8, 12, 16]
    phase1_budget = min(budget_s * 0.55, 550.0)
    per_p = phase1_budget / len(p_schedule)

    for p in p_schedule:
        if remaining() < 20:
            break
        alloc = min(per_p, remaining() - 10)
        if alloc < 5:
            break
        h_cur = best_h.copy()
        h_new = gradient_solve(h_cur, target, lambda h, p=p: lp_gradient(h, int(p)), alloc,
                               lr_init=0.02, momentum_init=0.88, max_steps=8000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        else:
            # Try with current best as start
            h_new = gradient_solve(best_h, target, lambda h, p=p: lp_gradient(h, int(p)), alloc,
                                   lr_init=0.015, momentum_init=0.9, max_steps=6000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

        # Quick targeted refinement between phases
        if remaining() > 15:
            h_r, c5_r = refine_targeted(best_h, target, min(remaining() * 0.03, 10.0), seed + p)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 2: Smooth max gradient descent (20% budget)
    if remaining() > 20:
        alpha_schedule = [5, 10, 20, 40, 80]
        phase2_budget = min(budget_s * 0.20, 200.0)
        per_alpha = phase2_budget / len(alpha_schedule)

        for alpha in alpha_schedule:
            if remaining() < 15:
                break
            alloc = min(per_alpha, remaining() - 10)
            if alloc < 5:
                break
            h_new = gradient_solve(best_h, target, lambda h, a=alpha: smooth_max_gradient(h, float(a)),
                                   alloc, lr_init=0.01, momentum_init=0.92, max_steps=6000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

            # Targeted refine
            if remaining() > 10:
                h_r, c5_r = refine_targeted(best_h, target, min(remaining() * 0.02, 8.0), seed + 500 + alpha)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()

    # Phase 3: Threshold penalty (15% budget)
    if remaining() > 15:
        c5_start = best_c5
        tau_deltas = [0.0004, 0.0003, 0.0002, 0.00015, 0.0001, 0.00005, 0.00002]
        phase3_budget = min(budget_s * 0.15, 120.0)
        per_tau = phase3_budget / len(tau_deltas)

        for dt in tau_deltas:
            if remaining() < 10:
                break
            tau = c5_start - dt
            if tau < 0.30:
                break
            alloc = min(per_tau, remaining() - 5)
            if alloc < 3:
                break
            h_new = gradient_solve(best_h, target, lambda h, t=tau: threshold_gradient(h, float(t)),
                                   alloc, lr_init=0.008, momentum_init=0.93, max_steps=4000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 4: Final refinement and restarts (10% budget)
    if remaining() > 15:
        alloc = min(remaining() * 0.5, 60.0)
        h_r, c5_r = refine_targeted(best_h, target, alloc, seed + 999)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Restart with perturbation if time remains
    if remaining() > 20:
        for trial in range(3):
            if remaining() < 15:
                break
            rng = np.random.RandomState(seed + 5000 + trial * 13)
            h_pert = best_h + rng.randn(n) * 0.015
            h_pert = project_box_sum(h_pert, target)
            # Quick L_8 + smooth max
            h_new = gradient_solve(h_pert, target, lambda h: lp_gradient(h, 8), min(remaining() * 0.2, 30.0),
                                   lr_init=0.01, momentum_init=0.9, max_steps=3000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Final targeted refinement
    if remaining() > 10:
        h_r, c5_r = refine_targeted(best_h, target, min(remaining() * 0.4, 40.0), seed + 7777)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Ensure constraint satisfaction
    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
