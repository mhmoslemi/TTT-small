def lp_direction(h, radius):
    n = len(h)
    scores = compute_raw_scores(h)
    grad = compute_grad_matrix(h)
    
    # Variables: p_0, ..., p_{n-1}, tau (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    # Inequality: grad[k] @ p - tau <= -scores[k] for each k
    A_ub = np.hstack([grad, -np.ones((2*n-1, 1))])
    b_ub = -scores
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    p_lo = -np.minimum(radius, h)
    p_hi = np.minimum(radius, 1.0 - h)
    bounds = list(zip(p_lo, p_hi)) + [(-100.0, 100.0)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n].copy()
    return None

def line_search(h, p, c5_current, dx, n):
    # Find alpha_max such that h + alpha*p is in [0,1]^n
    alpha_max = 1.0  # default (trust region already limits p)
    for i in range(n):
        if p[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-15:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    
    if alpha_max <= 0:
        return 0.0, c5_current
    
    # Grid search
    n_grid = 30
    alphas = np.linspace(0, alpha_max, n_grid)
    best_alpha = 0.0
    best_c5 = c5_current
    for a in alphas[1:]:  # skip 0 (already have c5_current)
        h_new = h + a * p
        # Check bounds (should be satisfied by construction)
        c5_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_alpha = a
    
    # Refine with golden section around best_alpha
    if best_alpha > 0 and best_alpha < alpha_max:
        # Find the interval
        idx = int(np.argmin(np.abs(alphas - best_alpha)))
        a_lo = alphas[max(0, idx-1)]
        a_hi = alphas[min(n_grid-1, idx+1)]
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        a1 = a_hi - gr * (a_hi - a_lo)
        a2 = a_lo + gr * (a_hi - a_lo)
        f1 = float(np.max(np.correlate(h + a1*p, 1.0 - (h + a1*p), mode='full')) * dx)
        f2 = float(np.max(np.correlate(h + a2*p, 1.0 - (h + a2*p), mode='full')) * dx)
        for _ in range(30):
            if f1 < f2:
                a_hi = a2; a2 = a1; f2 = f1
                a1 = a_hi - gr * (a_hi - a_lo)
                f1 = float(np.max(np.correlate(h + a1*p, 1.0 - (h + a1*p), mode='full')) * dx)
            else:
                a_lo = a1; a1 = a2; f1 = f2
                a2 = a_lo + gr * (a_hi - a_lo)
                f2 = float(np.max(np.correlate(h + a2*p, 1.0 - (h + a2*p), mode='full')) * dx)
        best_alpha = (a_lo + a_hi) / 2
        best_c5 = float(np.max(np.correlate(h + best_alpha*p, 1.0 - (h + best_alpha*p), mode='full')) * dx)
    
    return best_alpha, best_c5
