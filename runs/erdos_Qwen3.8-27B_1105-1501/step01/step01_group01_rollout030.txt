import numpy as np
from scipy.optimize import minimize
import time

def score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

    delta_max = min(1.0 - h[i], h[j])

def paired_cd(h, n_full_passes=100, max_inner=40):
    """Paired coordinate descent."""
    n = len(h)
    h = h.copy()
    
    for p in range(n_full_passes):
        for i in range(n):
            for j in range(i+1, n):
                # Skip if too close
                if abs(h[i] - 0.5) < 1e-8 and abs(h[j] - 0.5) < 1e-8:
                    continue
                
                delta_min = max(-h[i], h[j] - 1.0)
                delta_max = min(1.0 - h[i], h[j])
                if delta_max - delta_min < 1e-12:
                    continue
                
                # Compute A, B coefficients
                A_k = np.zeros(2*n-1)
                B_k = np.zeros(2*n-1)
                for k in range(2*n-1):
                    s = k - (n-1)
                    a = 0
                    b = 0
                    # Cross: i+s=j
                    if 0 <= i+s < n and i+s == j:
                        a += 1
                        b += (1 - h[j] + h[i])
                    # Cross: j+s=i
                    elif 0 <= j+s < n and j+s == i:
                        a += 1
                        b += (-1 + h[i] - h[j])
                    # h[i] first factor (non-cross)
                    if 0 <= i+s < n and i+s != j:
                        b += (1 - h[i+s])
                    # h[i] second factor (non-cross)
                    if 0 <= i-s < n and i-s != j:
                        b += (-h[i-s])
                    # h[j] first factor (non-cross)
                    if 0 <= j+s < n and j+s != i:
                        b += (-(1 - h[j+s]))
                    # h[j] second factor (non-cross)
                    if 0 <= j-s < n and j-s != i:
                        b += (+h[j-s])
                    A_k[k] = a
                    B_k[k] = b
                
                # Current corr values
                corr = np.correlate(h, 1.0 - h, mode='full')
                
                # Objective at delta: max_k (corr[k] + A_k[k]*d² + B_k[k]*d)
                def obj(d):
                    new_vals = corr + A_k * d * d + B_k * d
                    return np.max(new_vals)
                
                # Golden section
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a_g = delta_min
                b_g = delta_max
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)
                fc = obj(c_g)
                fd = obj(d_g)
                for _ in range(max_inner):
                    if b_g - a_g < 1e-10:
                        break
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        fc = obj(c_g)
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        fd = obj(d_g)
                
                best_d = (a_g + b_g) / 2.0
                h[i] += best_d
                h[j] -= best_d
        
        # Project
        h = make_feasible(h)
    
    return h
