def line_search(h, i, j, C, T):
    """Find best s in [0, T] for transfer from j to i. Returns (best_s, best_c5_raw)."""
    n = len(h)
    # Compute A and R
    p = np.zeros(n)
    p[i] = 1.0
    p[j] = -1.0
    g_vec = 1.0 - h
    A = np.correlate(p, g_vec, 'full') - np.correlate(h, p, 'full')
    R = np.correlate(p, p, 'full')
    
    # Objective: max_k(C[k] + s*A[k] - s^2*R[k])
    # Grid search
    n_grid = 25
    s_vals = np.linspace(0.0, T, n_grid + 1)
    best_s = 0.0
    best_val = float(np.max(C))
    for s in s_vals[1:]:
        vals = C + s * A - s*s * R
        v = float(np.max(vals))
        if v < best_val:
            best_val = v
            best_s = s
    # Golden section refinement
    if 0 < best_s < T:
        lo = max(0.0, best_s - T/n_grid)
        hi = min(T, best_s + T/n_grid)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(40):
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1 = float(np.max(C + m1*A - m1*m1*R))
            f2 = float(np.max(C + m2*A - m2*m2*R))
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        s_gold = (lo + hi) / 2.0
        f_gold = float(np.max(C + s_gold*A - s_gold*s_gold*R))
        if f_gold < best_val:
            best_s = s_gold
            best_val = f_gold
    return best_s, best_val

            m2 = lo + gr * (hi - lo)
            f1 = float(np.max(C + m1*A - m1*m1*R))
            f2 = float(np.max(C + m2*A - m2*m2*R))
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        s_g = (lo + hi) / 2.0

        if f_g < val_best:
            s_best = s_g
            val_best = f_g
    
    return s_best, val_best

def interpolate_h(h, new_n):
    """Interpolate h to new length."""
    n = len(h)
    if n == new_n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

import numpy as np
import time


def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_official(h):
    n = len(h)
    return float(np.max(raw_corr(h)) * 2.0 / n)


def project_feasible(v, target_sum):
    v = np.asarray(v, dtype=float).copy()
    v = np.where(np.isfinite(v), v, 0.5)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(x < 1.0)[0]
            if len(idx) > 0:
                x[idx] += residual / len(idx)
                x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.where(x > 0.0)[0]
            if len(idx) > 0:
                x[idx] += residual / len(idx)
                x = np.clip(x, 0.0, 1.0)
    return x


def precompute_unit_corrs(h):
    n = len(h)
    L = 2 * n - 1
    g = 1.0 - h
    V = np.zeros((n, L))
    W = np.zeros((n, L))
    for i in range(n):
        e_i = np.zeros(n)
        e_i[i] = 1.0
        V[i] = np.correlate(e_i, g, 'full')
        W[i] = np.correlate(h, e_i, 'full')
    return V, W


def compute_subgradient(h, active_idx):
    n = len(h)
    g = np.zeros(n)
    for idx in active_idx:
        lag = idx - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            ll = -lag
            g[ll:] += (1.0 - h[:n - ll])
            g[:n - ll] -= h[ll:]
    return g


def line_search_fast(h, i, j, C, V, W, T):
    n = len(h)
    if T < 1e-14:
        return 0.0, float(np.max(C))
    A = V[i] - V[j] - W[i] + W[j]
    R = np.zeros(2 * n - 1)
    R[n - 1] = 2.0
    d_idx_j = j - i + n - 1
    d_idx_i = i - j + n - 1
    if 0 <= d_idx_j < len(R):
        R[d_idx_j] -= 1.0
    if 0 <= d_idx_i < len(R):
        R[d_idx_i] -= 1.0
    n_grid = 25
    s_best = 0.0
    val_best = float(np.max(C))
    for k in range(1, n_grid + 1):
        s = k * T / n_grid
        vals = C + s * A - s * s * R
        v = float(np.max(vals))
        if v < val_best:
            val_best = v
            s_best = s
    if 0 < s_best < T and val_best < float(np.max(C)) - 1e-12:
        step = T / n_grid
        lo = max(0.0, s_best - step)
        hi = min(T, s_best + step)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(50):
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1 = float(np.max(C + m1 * A - m1 * m1 * R))
            f2 = float(np.max(C + m2 * A - m2 * m2 * R))
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        s_g = (lo + hi) / 2.0
        f_g = float(np.max(C + s_g * A - s_g * s_g * R))
        if f_g < val_best:
            s_best = s_g
            val_best = f_g
    return s_best, val_best


def pair_exchange_optimize(h_init, t0, deadline_time, rng, max_passes):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = c5_official(h)
    C = raw_corr(h)
    raw_max = float(np.max(C))
    stagnant = 0
    passes_done = 0

    for p in range(max_passes):
        if time.time() > deadline_time:
            break
        passes_done += 1

        V, W = precompute_unit_corrs(h)

        thr = raw_max - max(1e-10, 0.008 * raw_max)
        active = np.where(C >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        grad = compute_subgradient(h, active)

        order = np.argsort(grad)
        n_sel = min(15, max(8, n // 5))
        targets = order[:n_sel]
        sources = order[n_sel:]

        best_s = 0.0
        best_i = -1
        best_j = -1
        best_val = raw_max

        n_try = min(300, n * n)
        tried = set()
        attempts = 0

        for i in targets[:min(n_sel, 20)]:
            for j in sources[:min(n_sel, 20)]:

                if ii == jj or (ii, jj) in tried:
                    continue
                tried.add((ii, jj))
                T = min(1.0 - h[ii], h[jj])
                if T < 1e-12:
                    continue
                s, val = line_search_fast(h, ii, jj, C, V, W, T)
                if val < best_val - 1e-12:
                    best_val = val
                    best_s = s
                    best_i = ii
                    best_j = jj
                attempts += 1
                if attempts >= n_try // 2:
                    break
            if attempts >= n_try // 2:
                break

        for _ in range(n_try // 2):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii == jj:
                continue
            T = min(1.0 - h[ii], h[jj])
            if T < 1e-12:
                continue
            s, val = line_search_fast(h, ii, jj, C, V, W, T)
            if val < best_val - 1e-12:
                best_val = val
                best_s = s
                best_i = ii
                best_j = jj

        if best_i >= 0:
            h[best_i] += best_s
            h[best_j] -= best_s
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            C = raw_corr(h)
            new_rm = float(np.max(C))
            new_c5 = c5_official(h)
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                raw_max = new_rm
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant >= 15:
            h = best_h.copy()
            mask = rng.random(n) < 0.4
            if np.any(mask):
                h[mask] += rng.normal(0, 0.04, int(np.sum(mask)))
            h = project_feasible(h, target)
            C = raw_corr(h)
            raw_max = float(np.max(C))
            stagnant = 0

    return best_h, best_c5


def try_symmetrize(h):
    target = len(h) / 2.0
    best = h.copy()
    best_c5 = c5_official(h)
    for alpha in [0.1, 0.25, 0.5, 0.75, 0.9]:
        h_s = (1.0 - alpha) * h + alpha * h[::-1]
        h_s = project_feasible(h_s, target)
        c5 = c5_official(h_s)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h_s.copy()
    return best, best_c5


def try_sharpen(h, scale):
    n = len(h)
    target = n / 2.0
    h_new = np.where(h < 0.5, h / scale, 1.0 - (1.0 - h) / scale)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    return h_new


def resample_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    h_new = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    h_new = project_feasible(h_new, target)
    return h_new
