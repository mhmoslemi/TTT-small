Let me think about the LP approach more carefully:

At a point h, the correlation at lag d is:
c(d) = dx * sum_i h[i] * (1 - h[i+d]) for valid i

For a small perturbation δ (with sum(δ) = 0, and 0 ≤ h+δ ≤ 1), the change in c(d) is approximately:
Δc(d) ≈ dx * sum_i [δ[i]*(1-h[i+d]) + h[i]*(-δ[i+d])]

So if we want to minimize max_d c(d), we can set up an LP:
minimize t
subject to: c(d) + Δc(d) ≤ t for all active d
sum(δ) = 0
-h[i] ≤ δ[i] ≤ 1-h[i]

This is a valid LP! The objective is linear in δ and t, and all constraints are linear.

Let me think about the size: if we have n points, we have n+1 lags (0 to 2n-1 in the full correlation, but we only care about lags 0 to n-1 for the meaningful range... actually we care about all lags in the "full" mode).

Wait, the correlation is computed as np.correlate(h, 1-h, mode="full"), which gives 2n-1 values. After multiplying by dx, we take the max.

For the LP, I'd need to express each correlation term linearly in δ. Let me think about this:

c_full[m] = dx * sum_i h[i] * (1-h[i+m-n+1]) for valid indices (in the full correlation)

Actually, let me be more careful. np.correlate(h, 1-h, mode='full') computes:
result[k] = sum_i h[i] * (1-h)[i+k-n+1] for k = 0, 1, ..., 2n-2

For a perturbation δ with sum(δ)=0:
result[k] changes by: dx * sum_i [δ[i] * (1-h[i+k-n+1]) + h[i] * (-δ[i+k-n+1])]

This is linear in δ. So the LP is:
minimize t
subject to: t ≥ c_full[k] + dx * sum_i [δ[i] * (1-h[i+k-n+1]) + h[i] * (-δ[i+k-n+1])] for all k
sum(δ) = 0
-h[i] ≤ δ[i] ≤ 1-h[i]

The number of variables is n+1 (n for δ, 1 for t).
The number of constraints is (2n-1) + 1 + 2n = 4n (approximately).

For n=70, that's about 280 constraints and 71 variables. This is very tractable for an LP solver!

We can then use this LP step iteratively:
1. Solve the LP to get δ*
2. Take a step h_new = h + α*δ* (with α ≤ 1 for safety)
3. Accept if the exact score improves
4. Repeat

    # Let j = i + k - (n-1), so i = j - k + (n-1)

    #        = dx * sum_i [delta[i] * (1-h[i+off]) - h[i] * delta[i+off]]

    # So dc[k] = dx * sum_i [delta[i] * (1-h[i+off]) - h[i-off] * delta[i]]  (where valid)

            val = 0.0

                val -= h[i_minus]

        constraints.append(t >= c[k] + coeffs @ delta)
    
    prob = cp.Problem(objective, constraints)
    
    try:

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c_idx = n - 1
    g += w[c_idx] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c_idx + d] * dx
        wn = w[c_idx - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return w, g

For lag k (0 to 2n-2), offset = k - (n-1):
- If offset >= 0: the correlation is sum_{i=0}^{n-1-offset} h[i] * (1-h[i+offset])
- If offset < 0: the correlation is sum_{i=-offset}^{n-1} h[i] * (1-h[i+offset]) (where i+offset >= 0)

The derivative with respect to h[j]:
- For offset >= 0:
  - h[j] appears in h[j]*(1-h[j+offset]) for j < n-offset, derivative: (1-h[j+offset])
  - h[j] appears in h[j-offset]*(1-h[j]) for j >= offset, derivative: -h[j-offset]
  - Total: (1-h[j+offset])*[j < n-offset] - h[j-offset]*[j >= offset]
  
- For offset < 0 (let m = -offset > 0):
  - h[j] appears in h[j+m]*(1-h[j]) for j < n-m, derivative: -h[j+m]
  - h[j] appears in h[j]*(1-h[j-m]) for j >= m, derivative: (1-h[j-m])
  - Total: -h[j+m]*[j < n-m] + (1-h[j-m])*[j >= m]

So the coefficient for delta[j] in c[k] is:
dx * [(1-h[j+offset])*[0<=j+offset<n] - h[j-offset]*[0<=j-offset<n]]

This can be computed efficiently:

def compute_coeff_matrix(h, active_lags):

    for idx, k in enumerate(active_lags):

        c = np.zeros(n)
        if off >= 0:
            c[:n-off] += (1.0 - h[off:])  # (1-h[j+off]) for j in [0, n-off-1]
            c[off:] -= h[:n-off]           # -h[j-off] for j in [off, n-1]
        else:
            m = -off
            c[m:] += (1.0 - h[:n-m])       # (1-h[j-m]) for j in [m, n-1]
            c[:n-m] -= h[m:]                # -h[j+m] for j in [0, n-m-1]
        coeffs[idx] = c * dx
    return coeffs

import numpy as np

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmin(np.where(mask, z, 2.0)))
                z[idx2] += min(rem, 1.0 - z[idx2])
        else:
            idx = int(np.argmax(z))
            sub = min(-diff, z[idx])
            z[idx] -= sub
            rem = -diff - sub
            if abs(rem) > 1e-13:
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                idx2 = int(np.argmax(np.where(mask, z, -1.0)))
                z[idx2] -= min(rem, z[idx2])
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def correlation_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_coeff_matrix(h):
    """Compute the (2n-1) x n matrix where row k = dx * d(c[k])/d(h)."""
    n = len(h)
    dx = 2.0 / n
    coeffs = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        off = k - (n - 1)
        c = np.zeros(n)
        if off >= 0:
            c[:n - off] += (1.0 - h[off:])
            c[off:] -= h[:n - off]
        else:
            m = -off
            c[m:] += (1.0 - h[:n - m])
            c[:n - m] -= h[m:]
        coeffs[k] = c * dx
    return coeffs

def lp_step(h, active_indices=None):
    """Solve LP to find direction minimizing max linearized correlation."""
    n = len(h)
    c = correlation_vector(h)
    base = float(np.max(c))
    
    if active_indices is None:
        # Use lags within 3% of max (at least 3 lags)
        threshold = base - 0.03 * max(base, 0.01)
        active_indices = np.where(c >= threshold)[0]
        if len(active_indices) < 3:
            active_indices = np.argsort(c)[-max(3, n // 10):]
    
    A = compute_coeff_matrix(h)
    A_active = A[active_indices]  # shape (m, n)
    c_active = c[active_indices]  # shape (m,)
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(delta) == 0,
        delta >= -h,
        delta <= 1.0 - h
    ]
    # t >= c_active + A_active @ delta
    constraints.append(t >= c_active + A_active @ delta)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    for solver in (cp.ECOS, cp.SCS, cp.CVXOPT):
        try:
            prob.solve(solver=solver, max_iters=1000)
            break
        except:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, base
    
    d = delta.value
    if d is None:
        return None, None, base
    d = np.asarray(d, dtype=np.float64)
    if not np.all(np.isfinite(d)):
        return None, None, base
    
    # Line search
    best_h = h.copy()
    best_c = base
    for alpha in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):
        cand = make_feasible(h + alpha * d, n)
        c_new = score_exact(cand)
        if c_new < best_c - 1e-15:
            best_h = cand
            best_c = c_new
    
    if best_c >= base - 1e-14:
        return None, None, base
    return best_h, best_c, base

def pair_search(h, i, j, cur_c, n_pts=40):

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    c_idx = n - 1
    g += w[c_idx] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c_idx + d] * dx
        wn = w[c_idx - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g

        T = max(T_end, T * T_decay)
    return best, best_c
