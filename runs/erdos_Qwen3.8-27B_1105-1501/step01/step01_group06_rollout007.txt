    n = len(v)

    hi = float(v.max()) + 2.0

def line_search_transfer(h, i, j, n_grid=25):
    """Find best δ for transfer from j to i, returning best δ and best h."""
    n = len(h)
    dx = 2.0 / n
    T = min(h[j], 1.0 - h[i])
    if T < 1e-12:
        return 0.0, h
    
    base_corr = np.correlate(h, 1.0 - h, mode='full')
    base_max = float(np.max(base_corr))
    
    # Sample δ values
    deltas = np.linspace(0, T, n_grid)
    best_delta = 0.0
    best_obj = base_max
    
    for delta in deltas:
        if delta < 1e-15:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # Clip for safety
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        obj = float(np.max(corr))
        if obj < best_obj:
            best_obj = obj
            best_delta = delta
    
    # Refine around best
    if 0 < best_delta < T:
        lo = max(0, best_delta - 2*T/n_grid)
        hi = min(T, best_delta + 2*T/n_grid)
        deltas2 = np.linspace(lo, hi, n_grid)
        for delta in deltas2:

            h_new[i] = np.clip(h[i] + delta, 0, 1)
            h_new[j] = np.clip(h[j] - delta, 0, 1)

    for d in range(-(n-1), n):
        idx = d + n - 1
        l = 0.0
        if 0 <= i + d < n:
            l += g[i + d]
        if 0 <= j + d < n:
            l -= g[j + d]
        if 0 <= i - d < n:
            l -= h[i - d]
        if 0 <= j - d < n:
            l += h[j - d]
        q = 0.0
        if d == 0:
            q = -2.0
        if d == j - i and d != 0:
            q += 1.0
        if d == i - j and d != 0:
            q += 1.0
        lin[idx] = l
        quad[idx] = q
    return lin, quad

def find_best_delta(C, lin, quad, T, n_grid=30):
    """Find δ in [0, T] minimizing max(C + δ*lin + δ²*quad)."""
    best_obj = float(np.max(C))
    best_delta = 0.0
    
    deltas = np.linspace(0, T, n_grid)
    for delta in deltas[1:]:  # skip 0
        vals = C + delta * lin + delta * delta * quad
        obj = float(np.max(vals))
        if obj < best_obj:
            best_obj = obj
            best_delta = delta
    
    # Refine
    if 0 < best_delta < T:
        step = T / n_grid
        lo = max(0.0, best_delta - step)
        hi = min(T, best_delta + step)
        deltas2 = np.linspace(lo, hi, 30)
        for delta in deltas2:
            vals = C + delta * lin + delta * delta * quad
            obj = float(np.max(vals))
            if obj < best_obj:
                best_obj = obj
                best_delta = delta
    
    return best_delta, best_obj
