```python
def make_feasible(h, target=None):
    """Project h to [0,1]^n with sum = n/2 using water-filling."""
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    out = np.clip(h - mu, 0.0, 1.0)
    # Fix any tiny residual
    residual = target - np.sum(out)
    if abs(residual) > 1e-12:

        interior = np.where((out > 1e-10) & (out < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            idx = interior[0]
            out[idx] += residual
            out[idx] = np.clip(out[idx], 0.0, 1.0)
```

```python
def pair_line_search(h, G, cs, i, j):
    """Find optimal δ for transfer from i to j."""
    # Feasible range for δ
    delta_lo = max(h[i] - 1.0, -h[j])  # h[i]-δ >= 0 and h[j]+δ >= 0
    delta_hi = min(h[i], 1.0 - h[j])    # h[i]-δ <= 1 and h[j]+δ <= 1
    
    if delta_lo >= delta_hi:
        return 0.0, compute_c5(h)
    
    d = G[j, :] - G[i, :]
    
    # Evaluate C5(δ) = max_k(cs_k + δ * d_k)
    def f(delta):
        return np.max(cs + delta * d)
    
    # Convex minimization using ternary search
    best_val = f(0.0)
    best_delta = 0.0
    
    lo, hi = delta_lo, delta_hi
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0

        f1, f2 = f(m1), f(m2)
        if f1 < f2:
            hi = m2
        else:

        # Check current best

        if fmid < best_val:

            best_delta = (lo + hi) / 2.0

        ft = f(dtest)
        if ft < best_val:
            best_val = ft
            best_delta = dtest
    
    return best_delta, best_val
```

- For each k: G[:,k]^T @ d - t <= -c_k → [G[:,k], -1] @ x <= -c_k
- d_i + h_i >= 0 → -d_i <= h_i → [-e_i^T, 0] @ x <= h_i

Equality constraints:

Bounds: -h_i <= d_i <= 1-h_i, -inf <= t <= inf

```python
import numpy as np
import time
from scipy.optimize import linprog

def compute_c_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_c_scaled(h)))

def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - out.sum()
    if abs(res) > 1e-14:
        slack = out * (1.0 - out)
        idx = int(np.argmax(slack))
        out[idx] = float(np.clip(out[idx] + res, 0.0, 1.0))
    return out

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    G = np.zeros((n, n_corr))
    for k_idx in range(n_corr):
        k = k_idx - (n - 1)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if k >= 0:
            t1[k:n] = 1.0 - h[0:n-k]
            t2[0:n-k] = -h[k:n]
        else:
            mk = -k
            t1[0:n-mk] = 1.0 - h[mk:n]
            t2[mk:n] = -h[0:n-mk]
        G[:, k_idx] = dx * (t1 + t2)
    return G

def solve_lp_step(h, cs, G):
    n = len(h)
    n_corr = G.shape[1]
    
    # Variables: [d_0, ..., d_{n-1}, t]
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: G[:,k]^T d - t <= -c[k]
    A_corr = np.hstack([G, -np.ones((n_corr, 1))])
    b_corr = -cs
    
    # d_i >= -h[i] → -d_i <= h[i]
    A_lb = np.zeros((n, n + 1))
    for i in range(n):
        A_lb[i, i] = -1.0
    b_lb = h.copy()
    
    # d_i <= 1-h[i] → d_i <= 1-h[i]
    A_ub_box = np.zeros((n, n + 1))
    for i in range(n):
        A_ub_box[i, i] = 1.0
    b_ub_box = 1.0 - h
    
    A_ub = np.vstack([A_corr, A_lb, A_ub_box])
    b_ub = np.concatenate([b_corr, b_lb, b_ub_box])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except:
        return None
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    t_val = res.x[n]
    
    # Check improvement: t_val should be < max(cs) = c5
    if t_val >= cs.max() - 1e-10:
        return None
    
    return d

def optimize_lp(h0, time_budget, seed=42):
    n = len(h0)
    h = make_feasible(h0.astype(float).copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    
    while time.time() - t0 < time_budget:
        cs = compute_c_scaled(h)
        G = grad_matrix(h)
        d = solve_lp_step(h, cs, G)
        
        if d is None:
            # No LP improvement; try perturbation
            stall += 1
            if stall > 5:
                h = make_feasible(best_h + 0.05 * rng.randn(n))
                stall = 0
            continue
        
        # Backtracking line search
        accepted = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = h + frac * d
            h_new = make_feasible(h_new)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                accepted = True
                stall = 0
                break
        
        if not accepted:
            stall += 1
    
    return best_h, best_c5
```
