import time

def compute_c_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_c_scaled(h)))

def make_feasible(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.clip(v.astype(float), 0.0, 1.0)
    # Water-filling to achieve exact target
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    out = np.clip(v - mu, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(out)
    if abs(residual) > 1e-12:
        # Distribute residual among interior points
        interior = np.where((out > 1e-8) & (out < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            per = residual / len(interior)
            for idx in interior:
                out[idx] += per
                out[idx] = min(max(out[idx], 0.0), 1.0)
    return out

def feasible_interval(h, d, tol=1e-10):
    """Compute the feasible interval [t_lo, t_hi] for h + t*d"""
    pos = d > tol
    neg = d < -tol
    t_hi = np.inf
    t_lo = -np.inf
    if np.any(pos):
        t_hi = np.min((1.0 - h[pos]) / d[pos])
    if np.any(neg):
        t_lo = np.max(h[neg] / d[neg])  # d[neg] < 0, so this gives negative values
    return t_lo, t_hi

def exact_line_search(h, d, n_grid=200, top_k=30):
    """Exact 1D minimization of C5 along feasible line h + t*d"""
    n = len(h)
    t_lo, t_hi = feasible_interval(h, d)
    
    if t_lo >= t_hi or not np.isfinite(t_lo) or not np.isfinite(t_hi):
        return h, compute_c5(h)
    
    # Cap the interval to avoid extreme values
    mid = (t_lo + t_hi) / 2.0
    half_width = (t_hi - t_lo) / 2.0
    if half_width > 5.0:
        t_lo = mid - 5.0
        t_hi = mid + 5.0
        # Re-clip to feasibility
        pos = d > 1e-10
        neg = d < -1e-10
        if np.any(pos):
            t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            t_lo = max(t_lo, np.max(h[neg] / d[neg]))
    
    if t_lo >= t_hi:
        return h, compute_c5(h)
    
    # Compute quadratic coefficients
    A, B, C = compute_quadratic_coeffs(h, d)
    
    # Candidate t values
    candidates = [t_lo, t_hi, 0.0]
    
    # Add vertices of active quadratics
    active_mask = np.zeros(len(A), dtype=bool)
    c_current = C  # current correlation values

            bb = B[ki] - B[kj]
            cc = C[ki] - C[kj]

def make_pair_direction(h, i, j):

    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0

def gradient_based_pairs(h, g, limit=10):
    """Generate pair directions based on gradient: increase low-g coords, decrease high-g coords"""
    n = len(h)
    # Sort by gradient
    order = np.argsort(g)
    # Low gradient = favorable to increase
    # High gradient = favorable to decrease
    low_coords = [order[i] for i in range(min(20, n)) if h[order[i]] < 1.0 - 1e-8]
    high_coords = [order[n-1-i] for i in range(min(20, n)) if h[order[n-1-i]] > 1e-8]
    
    directions = []
    rng = np.random.RandomState(hash(h.tobytes()) % 2**31)
    
    for i in range(limit):
        if not low_coords or not high_coords:
            break
        a = rng.choice(low_coords)
        b = rng.choice(high_coords)
        if a != b and h[a] < 1.0 - 1e-8 and h[b] > 1e-8:
            d = np.zeros(n)
            d[a] = 1.0
            d[b] = -1.0
            directions.append(d)
    
    return directions

def random_direction(n, rng):

import numpy as np
import time
import math


def compute_c_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h):
    return float(np.max(compute_c_scaled(h)))


def make_feasible(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = v.astype(float).copy()
    v = np.clip(v, 0.0, 1.0)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    out = np.clip(v - mu, 0.0, 1.0)
    residual = target - np.sum(out)
    if abs(residual) > 1e-12:
        interior = np.where((out > 1e-8) & (out < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            per = residual / len(interior)
            out[interior] += per
            out = np.clip(out, 0.0, 1.0)
    return out


def quad_coeffs(h, d):
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    A = np.zeros(m)
    B = np.zeros(m)
    # Lag 0
    idx = n - 1
    B[idx] = np.dot(d, 1.0 - 2.0 * h) * dx
    A[idx] = -np.dot(d, d) * dx
    # Positive and negative lags
    for L in range(1, n):
        ip = n - 1 + L
        ineg = n - 1 - L
        B[ip] = (np.dot(d[L:], 1.0 - h[:-L]) - np.dot(d[:-L], h[L:])) * dx
        A[ip] = -np.dot(d[L:], d[:-L]) * dx
        B[ineg] = (np.dot(d[:-L], 1.0 - h[L:]) - np.dot(d[:-L], h[L:])) * dx
        A[ineg] = -np.dot(d[:-L], d[L:]) * dx
    return A, B, C


def feasible_interval(h, d):
    tol = 1e-12
    pos = d > tol
    neg = d < -tol
    t_hi = np.inf
    t_lo = -np.inf
    if np.any(pos):
        t_hi = np.min((1.0 - h[pos]) / d[pos])
    if np.any(neg):
        t_lo = np.max(h[neg] / d[neg])
    return t_lo, t_hi


def exact_line_search(h, d, n_grid=150, top_k=25):
    n = len(h)
    t_lo, t_hi = feasible_interval(h, d)
    if t_lo >= t_hi or not np.isfinite(t_lo) or not np.isfinite(t_hi):
        return h, compute_c5(h)
    mid_t = (t_lo + t_hi) / 2.0
    half_w = (t_hi - t_lo) / 2.0
    if half_w > 3.0:
        t_lo = mid_t - 3.0
        t_hi = mid_t + 3.0
        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            t_lo = max(t_lo, np.max(h[neg] / d[neg]))
    if t_lo >= t_hi:
        return h, compute_c5(h)

    A, B, C = quad_coeffs(h, d)
    current_c5 = float(np.max(C))

    candidates = [t_lo, t_hi, 0.0]

    # Vertices of top quadratics
    c_order = np.argsort(C)[-top_k:]
    for k in c_order:
        if A[k] > 1e-14:
            t_v = -B[k] / (2.0 * A[k])
            if t_lo < t_v < t_hi:
                candidates.append(t_v)

    # Pairwise intersections of top quadratics
    for i in range(len(c_order)):
        for j in range(i + 1, len(c_order)):
            ki, kj = c_order[i], c_order[j]
            aa = A[ki] - A[kj]
            bb = B[ki] - B[kj]
            cc = C[ki] - C[kj]
            if abs(aa) < 1e-14:
                if abs(bb) > 1e-14:
                    t_int = -cc / bb
                    if t_lo < t_int < t_hi:
                        candidates.append(t_int)
            else:
                disc = bb * bb - 4.0 * aa * cc
                if disc >= 0:
                    sq = math.sqrt(disc)
                    t1 = (-bb + sq) / (2.0 * aa)
                    t2 = (-bb - sq) / (2.0 * aa)
                    if t_lo < t1 < t_hi:
                        candidates.append(t1)
                    if t_lo < t2 < t_hi:
                        candidates.append(t2)

    # Dense grid
    grid = np.linspace(t_lo, t_hi, n_grid)
    candidates.extend(grid.tolist())

    # Evaluate model phi(t) = max_k(A_k*t^2 + B_k*t + C_k)
    best_t = 0.0
    best_model = current_c5
    for t in candidates:
        if t < t_lo - 1e-10 or t > t_hi + 1e-10:
            continue
        tc = min(max(t, t_lo), t_hi)
        vals = A * tc * tc + B * tc + C
        phi_t = np.max(vals)
        if phi_t < best_model - 1e-13:
            best_model = phi_t
            best_t = tc

    if best_t == 0.0:
        return h, current_c5

    # Verify with exact evaluation
    h_new = h + best_t * d
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    res = target - np.sum(h_new)
    if abs(res) > 1e-10:
        h_new = make_feasible(h + best_t * d, target)
    c5_exact = compute_c5(h_new)
    if c5_exact <= current_c5 + 1e-12:
        return h_new, c5_exact
    return h, current_c5


def gradient_direction(h, top_r=20):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_r:]
    g = np.zeros(n)
    w = 1.0 / top_r
    for k in top_idx:
        L = k - (n - 1)
        if L == 0:
            g += w * (1.0 - 2.0 * h) * dx
        elif L > 0:
            if L < n:
                g[L:] += w * (1.0 - h[:-L]) * dx
                g[:-L] -= w * h[L:] * dx
        else:
            M = -L
            if M < n:
                g[:-M] += w * (1.0 - h[M:]) * dx
                g[M:] -= w * h[:-M] * dx
    return g


def gen_pair_directions(h, g, limit, rng):
    n = len(h)
    directions = []
    order = np.argsort(g)
    low_c = [order[i] for i in range(min(30, n)) if h[order[i]] < 1.0 - 1e-8]
    high_c = [order[n - 1 - i] for i in range(min(30, n)) if h[order[n - 1 - i]] > 1e-8]
    for _ in range(limit):
        if not low_c or not high_c:
            break
        a = int(rng.randint(len(low_c)))
        b = int(rng.randint(len(high_c)))
        a = low_c[a]
        b = high_c[b]
        if a == b:
            continue
        if h[a] >= 1.0 - 1e-8 or h[b] <= 1e-8:
            continue
        d = np.zeros(n)
        d[a] = 1.0
        d[b] = -1.0
        directions.append(d)
    return directions


def random_dir(n, rng):
    d = rng.randn(n)
    d -= d.mean()
    s = np.max(np.abs(d))
    if s < 1e-14:
        return None
    return d / s


def optimize_exact(h0, time_budget, seed=42):
    n = len(h0)
    h = make_feasible(h0.astype(float).copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    stall_limit = 80
    pair_limit = 8

    while time.time() - t0 < time_budget:
        g = gradient_direction(h)
        directions = []

        # Gradient descent direction
        dg = -g
        dg -= dg.mean()
        gs = np.max(np.abs(dg))
        if gs > 1e-14:
            directions.append(dg / gs)

        # Pair directions
        pairs = gen_pair_directions(h, g, pair_limit, rng)
        directions.extend(pairs)

        # Random directions
        for _ in range(3):
            dr = random_dir(n, rng)
            if dr is not None:
                directions.append(dr)

        # Try each direction
        improved = False
        for d in directions:
            if time.time() - t0 >= time_budget:
                break
            d = d - d.mean()
            sc = np.max(np.abs(d))
            if sc < 1e-14:
                continue
            d = d / sc
            h_new, c5_new = exact_line_search(h, d)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= stall_limit:
                # Perturb
                amp = 0.01 + 0.02 * (stall / (2.0 * stall_limit))
                h = best_h + amp * rng.randn(n)
                h = make_feasible(h)
                stall = 0
                # If still stuck, bigger perturbation
                if stall == 0 and time.time() - t0 > time_budget * 0.7:
                    h = best_h + 0.05 * rng.randn(n)
                    h = make_feasible(h)

    return best_h, best_c5


def resample_h(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_target) + 0.5) * 2.0 / n_target
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt, n_target / 2.0)


def random_binary_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    h = h[rng.permutation(n)]
    return make_feasible(h)


def multi_coordinate_direction(h, g, rng, n_coords=4):
    """Move n_coords coordinates at once based on gradient"""
    n = len(h)
    order_inc = np.argsort(g)[:min(n_coords, n)]
    order_dec = np.argsort(g)[-min(n_coords, n):]
    d = np.zeros(n)
    valid_inc = [i for i in order_inc if h[i] < 1.0 - 1e-8]
    valid_dec = [i for i in order_dec if h[i] > 1e-8]
    if not valid_inc or not valid_dec:
        return None
    for i in valid_inc:
        d[i] = 1.0
    for i in valid_dec:
        d[i] = -1.0
    d -= d.mean()
    s = np.max(np.abs(d))
    if s < 1e-14:
        return None
    return d / s


def optimize_exact_advanced(h0, time_budget, seed=42):
    """Enhanced version with multi-coordinate moves and adaptive parameters"""
    n = len(h0)
    h = make_feasible(h0.astype(float).copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    stall_limit = 60
    iteration = 0

    while time.time() - t0 < time_budget:
        iteration += 1
        g = gradient_direction(h, top_r=max(10, min(30, n // 3)))
        directions = []

        # Gradient direction
        dg = -g
        dg -= dg.mean()
        gs = np.max(np.abs(dg))
        if gs > 1e-14:
            directions.append(dg / gs)

        # Multi-coordinate direction
        mc = multi_coordinate_direction(h, g, rng, n_coords=4)
        if mc is not None:
            directions.append(mc)

        # Pair directions (more in early stages)
        n_pairs = 10 if iteration < 50 else 6
        pairs = gen_pair_directions(h, g, n_pairs, rng)
        directions.extend(pairs)

        # Random directions
        n_rand = 4 if iteration < 100 else 3
        for _ in range(n_rand):
            dr = random_dir(n, rng)
            if dr is not None:
                directions.append(dr)

        # Try all directions
        improved = False
        for d in directions:
            if time.time() - t0 >= time_budget:
                break
            d = d - d.mean()
            sc = np.max(np.abs(d))
            if sc < 1e-14:
                continue
            d = d / sc
            h_new, c5_new = exact_line_search(h, d, n_grid=120, top_k=20)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= stall_limit:
                # Escalating perturbation
                level = min(stall // stall_limit, 5)
                amp = 0.02 * (level + 1)
                h = best_h + amp * rng.randn(n)
                h = make_feasible(h)
                stall = 0
                # Occasionally try a full restart from best
                if level >= 3:
                    h = resample_h(best_h, n)  # Slight perturbation via resampling
                    h = best_h + 0.03 * rng.randn(n)
                    h = make_feasible(h)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining_time():
        return budget_s - (time.time() - t_start)

    # Load initial values
    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    if h_init is not None:
        h_init = make_feasible(h_init)
        n_init = len(h_init)
        c5_init = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
        best_n = n_init
    else:
        n_init = 99
        rng_f = np.random.RandomState(seed)
        h_init = random_binary_init(n_init, rng_f)
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)
        best_n = n_init

    # Phase 1: Main optimization on initial dimension
    if remaining_time() > 80:
        alloc = min(remaining_time() * 0.45, 450.0)
        if alloc > 30:
            h1, c5_1 = optimize_exact_advanced(h_init, alloc, seed)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1
                best_n = len(h1)

    # Phase 2: Try nearby dimensions
    rng = np.random.RandomState(seed + 777)
    n_cur = best_n
    dims = []
    for dn in [3, 5, 7, -3, -5, 11, -11, 13, -7, 9]:
        nd = n_cur + dn
        if 31 <= nd <= 200 and nd not in dims:
            dims.append(nd)

    for n_try in dims:
        if remaining_time() < 25:
            break
        alloc = min(remaining_time() * 0.12, 100.0)
        if alloc < 15:
            break

        # Resample from best
        h_res = resample_h(best_h, n_try)
        c5_res = compute_c5(h_res)

        # Random init
        h_rand = random_binary_init(n_try, rng)
        c5_rand = compute_c5(h_rand)

        # Pick better starting point
        if c5_res <= c5_rand:
            h_start = h_res
        else:
            h_start = h_rand

        h_opt, c5_opt = optimize_exact_advanced(h_start, alloc, seed + n_try * 13)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(h_opt)

    # Phase 3: Final refinement
    if remaining_time() > 25:
        alloc = min(remaining_time() * 0.35, 180.0)
        if alloc > 15:
            h_fin, c5_fin = optimize_exact_advanced(best_h, alloc, seed + 55555)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin
                best_n = len(h_fin)

    # Final: verify and ensure feasibility
    best_h = make_feasible(best_h)
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    return best_h, best_c5, best_n
