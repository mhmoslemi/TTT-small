For a piecewise linear convex function, the minimum is at a "kink" or a flat region. Ternary search will find the kink (or flat region) to high precision.

But here's a subtlety: if the function is flat over an interval [u1, u2], any point in that interval is optimal. Different optimal u values might lead to different subsequent updates in the coordinate descent, potentially affecting convergence. In practice, this shouldn't be a big issue.

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        else:

def solve_1d_minmax(a, d):

    # f(u) = max_k(a_k + d_k*u) is convex (max of affine functions)

        else:

    return candidates[idx], vals[idx]

def coord_descent(h, n, max_passes=5000, tol=1e-13):
    """Coordinate descent for min max_k F[k] with sum(h)=n/2."""
    h = make_feasible(h.copy())
    best_score = official_score(h)
    k_vals = np.arange(-(n-1), n)
    m = 2 * n - 1

        max_change = 0.0

            vf = (ik >= 0) & (ik < n)
            ik2 = i - k_vals
            vb = (ik2 >= 0) & (ik2 < n)
            
            c = np.zeros(m)
            c[vf] = 1.0 - h[ik[vf]]
            c[vb] -= h[ik2[vb]]
            
            # Skip if c is all zeros (coordinate doesn't affect any F[k])
            if np.max(np.abs(c)) < 1e-15:
                continue

                F += (u_opt - h[i]) * c
                h[i] = u_opt

            best_score = sc
        
        if max_change < tol:

def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def slsqp_polish(h0, maxiter=200):
    """SLSQP refinement using smooth max approximation."""
    n = len(h0)
    dx = 2.0 / n
    h0 = make_feasible(h0.copy())
    
    def obj(x):
        F = np.correlate(x, 1.0 - x, mode='full')
        C = F * dx
        return float(np.max(C))
    
    def grad(x):
        F = np.correlate(x, 1.0 - x, mode='full')
        C = F * dx
        idx = np.argmax(C)
        k = idx - (n - 1)
        g = np.zeros(n)
        start = max(0, -k)
        end = min(n, n - k)
        g[start:end] += (1.0 - x[start + k:end + k])
        ps = max(0, start + k)
        pe = min(n, end + k)
        if ps < pe:
            g[ps:pe] += (-x[ps-k:pe-k])
        return g * dx
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h0, jac=grad, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except:
        pass
    return h0
