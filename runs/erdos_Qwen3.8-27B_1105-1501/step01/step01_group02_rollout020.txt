```python
def do_pair_sweep(h, target_sum, n_grid=100, max_rounds=5):
    n = len(h)
    total_improvement = 0.0
    c5_current = compute_c5(h)
    
    for round in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Find best d to move from i to j
                best_d, best_c5 = optimize_pair(h, i, j, n_grid)
                if best_c5 < c5_current - 1e-12:
                    d = best_d
                    h[i] -= d
                    h[j] += d
                    c5_current = best_c5
                    total_improvement += (c5_current - best_c5)  # negative
                    improved = True
        if not improved:
            break
    
    return c5_current  # or total_improvement
```

```python
def compute_pair_coefficients(h, i, j):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    g = np.zeros(2*n-1)
    q = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        s = k - (n-1)
        delta_g = 0.0
        delta_q = 0.0
        
        # Pair (i, i+s): valid if 0 <= i+s < n
        if 0 <= i+s < n:
            if i+s == j:  # s = j-i
                delta_q += 1.0
                delta_g += -(h[i] + 1.0 - h[j])
            else:
                delta_g += -(1.0 - h[i+s])
        
        # Pair (i-s, i): valid if 0 <= i-s < n
        a = i - s
        if 0 <= a < n:
            if a == j:  # s = i-j
                delta_q += 1.0
                delta_g += (h[j] + 1.0 - h[i])
            else:
                delta_g += h[a]
        
        # Pair (j, j+s): valid if 0 <= j+s < n
        if 0 <= j+s < n:
            if j+s == i:  # s = i-j, same pair as (i-s, i) above
                # Already counted! Skip.
                pass
            else:
                delta_g += (1.0 - h[j+s])
        
        # Pair (j-s, j): valid if 0 <= j-s < n
        a2 = j - s
        if 0 <= a2 < n:
            if a2 == i:  # s = j-i, same pair as (i, i+s) above
                # Already counted! Skip.
                pass
            else:
                delta_g += -h[a2]
        
        g[k] = delta_g
        q[k] = delta_q
    
    return corr, g, q
```
