        return t - vals[active_idx]  # should be >= 0
    
    def cons_active_grad(z):
        h = z[:n]
        G = grad_all(h)
        J = G[active_idx, :]  # shape (len(active_idx), n)
        J = -J  # d(t - V_p)/dh = -dV_p/dh
        J_t = np.ones((len(active_idx), 1))
        return np.hstack([J, J_t])
    
    def cons_sum(z):
        return np.sum(z[:n]) - n/2.0
    
    def cons_sum_grad(z):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    t0 = official_score(h0) + 1e-8
    z0 = np.concatenate([h0, [t0]])
    
    bounds = [(0.0, 1.0)]*n + [(0.0, 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': cons_active, 'jac': cons_active_grad},
        {'type': 'eq', 'fun': cons_sum, 'jac': cons_sum_grad}
    ]
    
    res = minimize(objective, z0, jac=objective_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    if res.success or (res.x is not None and np.all(np.isfinite(res.x))):
        h_new = make_feasible(res.x[:n])
        return h_new, official_score(h_new)
    return None, None

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_all(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] -= dx * h[s:]
            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K]
    return G

def smooth_obj_grad(h, beta):
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_all(h)
    grad = w @ G
    return obj, grad

def pg_solve(h0, beta, time_limit=30.0, max_iter=20000):
    n = len(h0)
    h = h0.copy()
    f, g = smooth_obj_grad(h, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-28:
        return h
    alpha = 1.0 / (gnorm2 + 1e-12)
    alpha = min(alpha, 2.0 / (np.max(g**2) + 1e-12))
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = smooth_obj_grad(h, beta)[1]
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-28:
            break
        step = alpha
        for _ in range(30):
            y = make_feasible(h - step * g)
            f_trial = smooth_obj_grad(y, beta)[0]
            if f_trial <= f - 1e-4 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-12:
            break
        f_new = smooth_obj_grad(y, beta)[0]
        if not np.isfinite(f_new):
            break
        h, f = y, f_new
        if f_new < f:
            alpha = min(alpha * 1.8, 5.0 / (gnorm2 + 1e-12))
    return h

def direct_minimax(h0, active_idx, max_iter=500):
    n = len(h0)

    def obj_func(z):
        return z[-1]

    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0])

    active.update(top_idx.tolist())
    return sorted(active)

    n_src = len(h_src)

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

    h_sym = 1.0 - h[::-1].copy()
    h_avg = 0.5 * (h + h_sym)
