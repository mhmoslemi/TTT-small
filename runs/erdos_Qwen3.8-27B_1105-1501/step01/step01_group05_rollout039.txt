    best_s = current_score
    best_alpha = 0.0

    if norm_d < 1e-15:
        return best_h, best_s, best_alpha
    direction = direction / norm_d
    
    alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

    max_val = np.max(c_scaled)

    active_idx = np.where(margins <= 0.01 * max_val + 0.005)[0]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c_scaled)])
    
    # Weight by inverse margin (more weight to more active)
    weights = 1.0 / (margins[active_idx] + 1e-6)
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for i, r in enumerate(active_idx):
        k = lags[r]
        g += weights[i] * grad_for_shift(h, k)
    
    # Subtract mean to maintain (approximately) mass
    g -= np.mean(g)
    
    # Add small random perturbation
    g += rng.normal(0, 0.05, n)
    g -= np.mean(g)
    
    return g

        if alpha != 0.0 and s_new < score - 1e-14:
            h = h_new
            score = s_new
            c_scaled, lags = score_all(h)
            no_improve_count = 0
        else:

            # Try random direction
            d_rand = rng.normal(0, 1, n)
            d_rand -= np.mean(d_rand)
            h_rand, s_rand, _ = line_search(h, d_rand, score, n)
            if s_rand < score - 1e-14:
                h = h_rand
                score = s_rand
                c_scaled, lags = score_all(h)
                no_improve_count = 0

def coordinate_descent(h, n, max_iters=200, time_limit=15.0, seed=42):
    """Simple coordinate descent with pair moves."""
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    start_time = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start_time > time_limit:
            break
        
        improved = False
        # Try random pair moves
        for _ in range(50):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.02, 0.05, 0.1, 0.15, 0.2]:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                h_try[j] = np.clip(h_try[j] - delta, 0, 1)
                # Fix sum
                h_new = project_to_target(h_try, n)
                s = official_score(h_new)
                if s < score - 1e-13:
                    h = h_new
                    score = s
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try swapping large values
            i = rng.choice(np.argsort(h)[-10:])
            j = rng.choice(np.argsort(h)[:10:])
            if h[i] > h[j] + 0.05:
                h_try = h.copy()
                h_try[i] -= 0.05
                h_try[j] += 0.05
                h_new = project_to_target(h_try, n)
                s = official_score(h_new)
                if s < score - 1e-13:
                    h = h_new
                    score = s
                    improved = True
        
        if not improved and it > 30:
            break
    
    return h, official_score(h)

def discrete_sa(h_init, n, time_limit=30.0, seed=42):
    """Simulated annealing on 0/1 patterns (for even n)."""
    if n % 2 != 0:
        return None, None
    
    rng = np.random.default_rng(seed)
    target = n // 2
    
    # Initialize from h_init by thresholding
    order = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[order[:target]] = 1.0
    
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    
    T = 0.5
    T_min = 0.01
    cooling = 0.98
    start_time = time.perf_counter()
    it = 0
    
    while time.perf_counter() - start_time < time_limit and T > T_min:
        # Try a random swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        
        s_new = official_score(h_new)
        delta = s_new - score
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            score = s_new
            if s_new < best_score:
                best_score = s_new
                best_h = h.copy()
        
        T *= cooling
        it += 1
    
    return best_h, best_score

def slsqp_polish(h, n, time_limit=15.0):
    """SLSQP minimax polish."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    initial_score = official_score(h)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        L = 2 * n - 1
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        for r in range(L):
            k = lags[r]
            J[r, :n] = -grad_for_shift(h_x, k)
        return J
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-4
    
    x0 = np.concatenate([h, [theta0]])
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def interpolate_to(parent, n_target, offset=0):
    """Interpolate parent to n_target points."""
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    if m == n_target:
        return project_to_target(parent, n_target)
    
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    
    # Handle wrap-around for periodic interpolation
    h = np.interp(xt, xs, np.concatenate([parent, parent]), left=parent[0], right=parent[-1])
    # Actually, for non-periodic, just use standard interp
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return project_to_target(h, n_target)

def gen_candidates(parent, n_list, rng):
    """Generate diverse candidate initializations."""
    candidates = []
    
    # Canonical transformations
    candidates.append(parent.copy())
    candidates.append(parent[::-1].copy())
    candidates.append((1.0 - parent).copy())
    candidates.append((1.0 - parent[::-1]).copy())
    
    # Symmetric averages
    candidates.append(project_to_target((parent + parent[::-1]) / 2.0, len(parent)))
    candidates.append(project_to_target((parent + (1.0 - parent[::-1])) / 2.0, len(parent)))
    
    # Interpolations to different resolutions
    for n_t in n_list:
        if n_t != len(parent):
            for off in [0, 1, 2]:
                candidates.append(interpolate_to(parent, n_t, off))

    m = len(parent)

        for _ in range(3):
            pert = h_base + rng.normal(0, 0.1, n_t)
            candidates.append(project_to_target(pert, n_t))
    
    return candidates

This is the log-sum-exp approximation, which converges to the max as β → ∞. We can optimize this with gradient methods, using a large but finite β.

The gradient of f_β is:
∇f_β = (1/∑_k exp(β c_k dx)) * ∑_k β * exp(β c_k dx) * ∇(c_k dx)
= ∑_k w_k * ∇(c_k dx)

where w_k = exp(β c_k dx) / ∑_j exp(β c_j dx) are softmax weights.

This is a convex combination of the individual gradient rows, weighted by how close each c_k is to the maximum. For large β, this concentrates on the active constraints.

import numpy as np
from scipy import optimize
from scipy.optimize import linprog
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n=None):
    if n is None:
        n = len(v)
    h = np.array(v, dtype=float).copy()
    if len(h) != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo = -max(h) - 1.0
    hi = 1.0 - min(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(300):
        resid = target - np.sum(h)
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            idx = np.argmax(1.0 - h)
            delta = min(resid, 1.0 - h[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-resid, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    return np.clip(h, 0.0, 1.0)


def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def get_lags(n):
    return np.arange(-(n - 1), n)


def grad_one_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    idx = np.arange(n)
    g = np.zeros(n)
    ok1 = (idx - k >= 0) & (idx - k < n)
    g[ok1] += (1.0 - h[idx[ok1] - k]) * dx
    ok2 = (idx + k >= 0) & (idx + k < n)
    g[ok2] += (-h[idx[ok2] + k]) * dx
    return g


def lse_gradient(h, beta=30.0):
    n = len(h)
    c_s = compute_corr_scaled(h)
    max_c = np.max(c_s)
    # Softmax weights (shifted to avoid overflow)
    log_w = beta * (c_s - max_c)
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(len(c_s))
        w[np.argmax(c_s)] = 1.0
        w_sum = 1.0
    else:
        w /= w_sum
    lags = get_lags(n)
    g = np.zeros(n)
    # Only compute for non-negligible weights
    active = w > 1e-10
    for i in np.where(active)[0]:
        g += w[i] * grad_one_shift(h, lags[i])
    g -= np.mean(g)
    return g, c_s, max_c


def line_search(h, direction, current_score, n):
    h_arr = h.copy()
    best_s = current_score
    best_alpha = 0.0
    norm_d = np.max(np.abs(direction))
    if norm_d < 1e-15:
        return h_arr, best_s, best_alpha
    direction = direction / norm_d
    alphas = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35, 0.5, 0.7, 1.0]
    for sign in [1, -1]:
        for a in alphas:
            alpha = sign * a
            h_new = project_to_target(h + alpha * direction, n)
            s = official_score(h_new)
            if s < best_s - 1e-14:
                best_s = s
                h_arr = h_new
                best_alpha = alpha
    return h_arr, best_s, best_alpha


def lp_direction(h, c_s, lags, active_tol=0.008, max_active=35):
    n = len(h)
    max_val = np.max(c_s)
    margins = max_val - c_s
    active_idx = np.where(margins <= active_tol)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c_s)])
    elif len(active_idx) > max_active:
        order = np.argsort(margins[active_idx])[:max_active]
        active_idx = active_idx[order]
    n_vars = 2 * n + 1
    rho_idx = 2 * n
    c_obj = np.zeros(n_vars)
    c_obj[rho_idx] = -1.0
    n_a = len(active_idx)
    A_ub = np.zeros((n_a + 1, n_vars))
    b_ub = np.zeros(n_a + 1)
    for i, r in enumerate(active_idx):
        g = grad_one_shift(h, lags[r])
        A_ub[i, :n] = g
        A_ub[i, n:2 * n] = -g
        A_ub[i, rho_idx] = -1.0
    A_ub[n_a, :2 * n] = 1.0
    b_ub[n_a] = 1.0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2 * n] = -1.0
    b_eq = np.zeros(1)
    bounds = [(0, max(0, 1 - h[i])) for i in range(n)] + \
             [(0, max(0, h[i])) for i in range(n)] + [(0, 10.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 5.0})
        if res.status == 0 and res.x is not None:
            d = res.x[:n] - res.x[n:2 * n]
            rho = res.x[rho_idx]
            if rho > 1e-10 or np.max(np.abs(d)) > 1e-8:
                return d, rho
    except Exception:
        pass
    return None, 0.0


def local_search_combined(h, n, time_limit=25.0, seed=42):
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    c_s = compute_corr_scaled(h)
    lags = get_lags(n)
    start_time = time.perf_counter()
    no_improve = 0
    beta = 25.0
    best_local = score

    for it in range(120):
        if time.perf_counter() - start_time > time_limit:
            break

        # Primary: LSE gradient
        g, c_s, max_c = lse_gradient(h, beta=beta)
        d = -g
        if np.max(np.abs(d)) > 1e-14:
            h_new, s_new, alpha = line_search(h, d, score, n)
            if alpha != 0.0 and s_new < score - 1e-14:
                h = h_new
                score = s_new
                c_s = compute_corr_scaled(h)
                no_improve = 0
                if score < best_local:
                    best_local = score
                continue

        # Secondary: LP direction
        d_lp, rho = lp_direction(h, c_s, lags, active_tol=0.006)
        if d_lp is not None and np.max(np.abs(d_lp)) > 1e-12:
            h_new, s_new, alpha = line_search(h, d_lp, score, n)
            if alpha != 0.0 and s_new < score - 1e-14:
                h = h_new
                score = s_new
                c_s = compute_corr_scaled(h)
                no_improve = 0
                continue

        # Tertiary: random perturbations
        improved_rand = False
        for _ in range(10):
            d_rand = rng.normal(0, 1, n)
            d_rand -= np.mean(d_rand)
            h_new, s_new, _ = line_search(h, d_rand, score, n)
            if s_new < score - 1e-14:
                h = h_new
                score = s_new
                c_s = compute_corr_scaled(h)
                no_improve = 0
                improved_rand = True
                break

        if not improved_rand:
            no_improve += 1
            # Increase beta to focus more on active constraints
            beta = min(beta * 1.3, 200.0)
            if no_improve > 12:
                break

    return h, official_score(h)


def discrete_local_search(h_init, n, time_limit=20.0, seed=42):
    if n % 2 != 0:
        return None, None
    rng = np.random.default_rng(seed)
    target = n // 2
    order = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[order[:target]] = 1.0
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    start_time = time.perf_counter()
    T = 0.3
    T_min = 0.02
    cooling = 0.985
    no_improve = 0

    while time.perf_counter() - start_time < time_limit and T > T_min:
        if time.perf_counter() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        s_new = official_score(h_new)
        delta = s_new - score
        if delta < -1e-14 or rng.random() < np.exp(-max(0, delta) / T):
            h = h_new
            score = s_new
            if s_new < best_score - 1e-14:
                best_score = s_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cooling
        if no_improve > 500:
            no_improve = 0
            T = max(T, 0.05)
    return best_h, best_score


def slsqp_polish(h, n, time_limit=12.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = get_lags(n)
    initial_score = official_score(h)

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        L = 2 * n - 1
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        for r in range(L):
            J[r, :n] = -grad_one_shift(h_x, lags[r])
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def interpolate_to(parent, n_target, offset=0):
    m = len(parent)
    if m == n_target:
        return project_to_target(parent, n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    xt_clipped = np.minimum(xt, 2.0 - 1e-10)
    h = np.interp(xt_clipped, xs, parent, left=parent[0], right=parent[-1])
    return project_to_target(h, n_target)


def gen_structured_candidates(parent, n_list, rng):
    candidates = []
    candidates.append(parent.copy())
    candidates.append(parent[::-1].copy())
    candidates.append((1.0 - parent).copy())
    candidates.append((1.0 - parent[::-1]).copy())
    m = len(parent)
    candidates.append(project_to_target((parent + parent[::-1]) / 2.0, m))
    for n_t in n_list:
        if n_t != m:
            for off in [0, 1]:
                candidates.append(interpolate_to(parent, n_t, off))
        t = np.linspace(0, 1, n_t, endpoint=False)
        for freq in [2, 3, 4, 5, 6]:
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.35, 0.5)
            h_sin = 0.5 + amp * np.sin(2 * np.pi * freq * t + phase)
            candidates.append(project_to_target(h_sin, n_t))
        for _ in range(2):
            h_rand = rng.uniform(0, 1, n_t)
            candidates.append(project_to_target(h_rand, n_t))
        if n_t % 2 == 0:
            target = n_t // 2
            idx = rng.permutation(n_t)
            h_bin = np.zeros(n_t)
            h_bin[idx[:target]] = 1.0
            candidates.append(h_bin.copy())
        # Block pattern
        n_blocks = rng.integers(4, 10)
        h_block = np.zeros(n_t)
        for b in range(n_blocks):
            start = int(b * n_t / n_blocks)
            end = int((b + 1) * n_t / n_blocks)
            h_block[start:end] = rng.uniform(0, 1)
        candidates.append(project_to_target(h_block, n_t))
    return candidates
