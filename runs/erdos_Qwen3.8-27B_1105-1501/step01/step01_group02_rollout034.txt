def project_box_sum(x, target):

        x = np.where(np.isfinite(x), x, 0.0)

    if abs(s - target) < 1e-12:
        return x

def pair_line_search(h, i, j, n_coarse=50, n_fine=20):
    """Optimize t in h[i] += t, h[j] -= t."""
    hi_val = h[i]
    hj_val = h[j]
    # Feasible range for t
    lo = max(-hi_val, hj_val - 1.0)
    hi = min(1.0 - hi_val, hj_val)
    if hi - lo < 1e-12:
        return h, compute_c5(h)
    
    # Coarse grid search
    best_t = 0.0
    best_c5 = compute_c5(h)
    for t in np.linspace(lo, hi, n_coarse):
        h[i] = hi_val + t
        h[j] = hj_val - t
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    # Fine refinement around best_t
    span = (hi - lo) / n_coarse
    for t in np.linspace(best_t - span, best_t + span, n_fine):
        if t < lo or t > hi:
            continue
        h[i] = hi_val + t
        h[j] = hj_val - t
        c5 = compute_c5(h)
        if c5 < best_c5:

            a = np.clip(a + rng.randn(m) * 0.08, 0.0, 1.0)
            build_fn = build_sc if layout == 0 else build_rc

            temps = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
            for T in temps:
                if time.time() > deadline - 15:
                    break
                maxiter = 200
                a = manifold_lbfgs(a, T, layout, maxiter)

            a, c5_mf = coord_descent_a(a, layout, max_sweeps=35, n_test=22, deadline=deadline - 5)
            h_cand = build_fn(a)
            h_cand = project_box_sum(h_cand, n / 2.0)
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()

            for restart in range(3):
                if time.time() > deadline - 10:
                    break
                a_r = np.clip(a + rng.randn(m) * 0.25, 0.0, 1.0)
                for T in [0.05, 0.025, 0.01]:
                    if time.time() > deadline - 8:
                        break
                    a_r = manifold_lbfgs(a_r, T, layout, 120)
                if time.time() > deadline - 6:
                    break
                a_r, _ = coord_descent_a(a_r, layout, max_sweeps=15, n_test=18, deadline=deadline - 3)
                h_r = build_fn(a_r)
                h_r = project_box_sum(h_r, n / 2.0)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()

    if best_h is not None:
        if time.time() < deadline - 10:
            h = best_h.copy()
            h, c5_ref = full_pair_sweep(h, rng, deadline=deadline)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h.copy()

    if best_h is None:
        n_fb = 64
        m_fb = n_fb // 2
        a_fb = np.where(np.arange(m_fb) % 2 == 0, 1.0, 0.0)
        best_h = build_sc(a_fb)
        best_h = project_box_sum(best_h, n_fb / 2.0)
        best_c5 = compute_c5(best_h)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
