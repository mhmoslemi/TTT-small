def project_feasible(v, target_sum):
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum} using binary search on threshold."""
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    off = n - 1

    A = E[i] - E[j]
    return float(np.max(A[active_indices]))

def golden_search(f, a, b, tol=1e-10, max_iter=100):
    """Minimize convex f on [a, b] using golden section search."""
    if b - a < tol:
        return (a + b) / 2
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ≈ 0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0

    vals = base_corr + delta * A + delta * delta * B
    return float(np.max(vals) * dx)

        return h, False

    B = np.zeros(2*n - 1)
    B[j - i + off] = 1.0
    B[i - j + off] = 1.0
    
    def f(delta):
        return eval_pair_delta(base_corr, A, B, delta, n)
    
    # Check endpoints
    f0 = f(0.0)
    fT = f(T)
    
    # Golden section search on [0, T]
    delta_star = golden_search(f, 0.0, T, tol=1e-12, max_iter=60)
    f_star = f(delta_star)
    
    if f_star < f0 - 1e-14:
        h[i] += delta_star
        h[j] -= delta_star
        return h, True
    return h, False
