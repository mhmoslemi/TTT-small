                constraints.append(cp.sum(lin) * dx <= t)

                j_range = np.arange(n - t_shift)
                lin = h[j_range]*(1-h0[j_range+t_shift]) + h0[j_range]*(1-h[j_range+t_shift]) - h0[j_range]*(1-h0[j_range+t_shift])
                constraints.append(cp.sum(lin) * dx <= t)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS)
            if h.value is not None and t.value is not None:
                h_new = np.clip(h.value, 0, 1)
                h_new = make_feasible(h_new)
                score = official_score(h_new)
                if score < official_score(h0) - 1e-10:
                    h0 = h_new
                else:
                    break
            else:
                break
        except:
            break
    
    return h0, official_score(h0)

def make_feasible(z):

import numpy as np
import time
from scipy.optimize import minimize, linprog

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def smooth_obj_pen(x, beta, penalty):
    n = len(x)
    dx = 2.0 / n
    target = n / 2.0
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted - shifted.max()))) + shifted.max()
    f = cmax + logsum / beta
    viol = np.sum(x) - target
    f += penalty * viol * viol
    return f

def smooth_grad_pen(x, beta, penalty):
    n = len(x)
    dx = 2.0 / n
    target = n / 2.0
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    g = np.zeros(n)
    for p in range(len(vals)):
        wp = w[p]
        if wp < 1e-14:
            continue
        s = p - (n - 1)
        if s >= 0:
            g[s:] += wp * dx * (1.0 - x[:n - s])
            g[:n - s] -= wp * dx * x[s:]
        else:
            t = -s
            g[:n - t] += wp * dx * (1.0 - x[t:])
            g[t:] -= wp * dx * x[:n - t]
    viol = np.sum(x) - target
    g += 2.0 * penalty * viol
    return g

def lbfgs_optimize(h0, beta=2000, penalty=100, maxiter=400):
    n = len(h0)
    try:
        res = minimize(smooth_obj_pen, h0, args=(beta, penalty),
                       jac=smooth_grad_pen, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        h_new = make_feasible(res.x)
        score = official_score(h_new)
        if np.isfinite(score):
            return h_new, score
    except Exception:
        pass
    return None, None

def lp_linearize(h0, max_iter=40, time_limit=None):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    current = h0.copy()
    current_score = official_score(current)
    t_start = time.time()
    n_shifts = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    for it in range(max_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        A_rows = []
        b_rows = []
        for p in range(n_shifts):
            s = p - (n - 1)
            if s >= 0:
                jr = np.arange(n - s)
                a = np.zeros(n)
                a[jr + s] += (1.0 - current[jr])
                a[jr] -= current[jr + s]
                const = np.sum(current[jr + s] * (1.0 - current[jr]))
            else:
                tv = -s
                jr = np.arange(n - tv)
                a = np.zeros(n)
                a[jr] += (1.0 - current[jr + tv])
                a[jr + tv] -= current[jr]
                const = np.sum(current[jr] * (1.0 - current[jr + tv]))
            A_rows.append(np.append(dx * a, -1.0))
            b_rows.append(-dx * const)
        A = np.array(A_rows)
        b = np.array(b_rows)
        try:

                          bounds=bounds, method='highs')
        except Exception:

                res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='revised simplex')
            except Exception:
                break
        if not res.success or res.x is None:
            break
        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        h_lp = make_feasible(h_lp)
        score_lp = official_score(h_lp)
        if score_lp < current_score - 1e-12:
            current = h_lp
            current_score = score_lp

                    current = h_d
                    current_score = sc_d
                    improved = True

    return current, current_score

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def slsqp_polish(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    beta = 5000.0
    def obj(x):
        vals = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        return cmax + np.log(np.sum(np.exp(shifted))) / beta
    def grad(x):
        vals = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        w = np.exp(shifted - shifted.max())
        w /= w.sum()
        g = np.zeros(n)
        for p in range(len(vals)):
            wp = w[p]
            if wp < 1e-14:
                continue
            s = p - (n - 1)
            if s >= 0:
                g[s:] += wp * dx * (1.0 - x[:n - s])
                g[:n - s] -= wp * dx * x[s:]
            else:
                t = -s
                g[:n - t] += wp * dx * (1.0 - x[t:])
                g[t:] -= wp * dx * x[:n - t]
        return g
    def eq_fun(x):
        return np.sum(x) - target
    def eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None
