    shifted = np.clip(beta * (c_vals - c_max), -500.0, 500.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / beta

def smooth_max_grad(c_vals, h, beta, n):

    exp_s = np.exp(shifted)

            if upper >= 0:
                grad[0:upper+1] += w * (1.0 - h[L:upper+1+L])
                grad[L:upper+1+L] -= w * h[0:upper+1]
    return grad

            best_h = h_new.copy()
        h = h_new
        if c5 - best_c5 > 0.01:  # if we're far from best, recover
            h = best_h.copy()
    return best_h, best_c5

        return smooth_max_val(c_vals, beta)

        c_vals = compute_c_vals(x)
        return smooth_max_grad(c_vals, x, beta, n)

             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_start, method='SLSQP', jac=grad,
                   bounds=bounds, constraints=cons,

def project_feasible(v, target_sum):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
