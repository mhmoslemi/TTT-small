import numpy as np

from scipy.optimize import minimize

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - z)
            for idx in order:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(z)
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def score_and_weights(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(scores))
    ex = np.exp((scores - m) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= m - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    return m, w

def compute_gradient(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m_ = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m_] += wp * (1.0 - h[d:])
        g[d:] -= wp * h[:m_]
        g[d:] += wn * (1.0 - h[:m_])
        g[:m_] -= wn * h[d:]
    return g

def smooth_obj_and_grad(h, T):
    m, w = score_and_weights(h, T)
    S = float(np.sum(w))
    val = m + T * np.log(S)
    g = compute_gradient(h, w)
    return val, g

def optimize_lbfgsb(h, deadline, T_start, T_end, T_decay, max_iter_per_T=100):
    n = len(h)
    h0 = h.copy()
    best_h = h0.copy()
    best_c = score_exact(h0)
    T = T_start
    bounds = [(0.0, 1.0)] * n
    while time.time() < deadline and T > T_end:
        def fun(x):
            v, _ = smooth_obj_and_grad(x, T)
            return v
        def gfun(x):
            _, g = smooth_obj_and_grad(x, T)
            return g
        try:
            res = minimize(fun, h0, jac=gfun, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iter_per_T, 'gtol': 1e-9, 'maxls': 30})
            h_new = make_feasible(res.x, n)
            c_new = score_exact(h_new)
            if c_new < best_c:
                best_h = h_new
                best_c = c_new
                h0 = h_new
        except Exception:
            pass
        T *= T_decay
    return best_h, best_c

def pair_search(h, i, j, cur_c, n_pts=50, refine=True):
    orig_i = h[i]
    orig_j = h[j]
    lo = -orig_j
    hi = 1.0 - orig_i
    if hi - lo < 1e-14:
        return None

    best_s = cur_c
    xs = np.linspace(lo, hi, n_pts)
    for a in xs:

        h[j] = orig_j - a

            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        if refine:
            span = (hi - lo) / n_pts
            lo2 = max(lo, best_a - 2.0 * span)
            hi2 = min(hi, best_a + 2.0 * span)
            if hi2 - lo2 > 1e-12:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                a2 = lo2 + gr * (hi2 - lo2)
                for _ in range(35):
                    h[i] = orig_i + a1
                    h[j] = orig_j - a1

                    h[i] = orig_i + a2
                    h[j] = orig_j - a2
                    s2 = score_exact(h)
                    if s1 < s2:
                        if s1 < best_s:
                            best_s = s1
                            best_a = a1
                        hi2 = a2
                        a2 = a1
                        a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                    else:
                        if s2 < best_s:
                            best_s = s2
                            best_a = a2
                        lo2 = a1
                        a1 = a2
                        a2 = lo2 + gr * (hi2 - lo2)
                best_a = (lo2 + hi2) * 0.5
                h[i] = orig_i + best_a
                h[j] = orig_j - best_a
                best_s = score_exact(h)
                if best_s >= cur_c:
                    h[i] = orig_i
                    h[j] = orig_j
                    return None
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    h[i] = orig_i
    h[j] = orig_j
    return None

def grad_descent_momentum(h, deadline, T_start, T_end, T_decay, step0, momentum, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    vel = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        m, w = score_and_weights(cur, T)
        g = compute_gradient(cur, w)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            T = max(T_end * 0.5, T * 0.9)
            continue
        vel = momentum * vel - step * g / gn
        cand = make_feasible(cur + vel, n)

        if c_new < cur_c - 1e-14:
            cur = cand
            cur_c = c_new
            if cur_c < best_c:

                best_c = cur_c

            stall = 0

                    cur_c = c2

                    cur = make_feasible(cur + rng.normal(0, max(0.003, T * 0.2), n), n)
                    cur_c = score_exact(cur)
                    vel = np.zeros(n)
                    step = max(T, 0.003)
                    stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def make_binary_seed(n, rng, pattern='random'):
    if pattern == 'random':
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    elif pattern == 'alternating':
        h = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    elif pattern == 'clustered':
        h = np.zeros(n)
        h[:n//2] = 1.0
    else:
        h = np.full(n, 0.5)
    return make_feasible(h, n)
