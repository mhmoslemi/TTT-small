import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    """Project v onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(np.array(v, dtype=float).ravel()[:n], 0.0, 1.0)
    target = n / 2.0
    # Bisection on additive shift
    lo = target - n  # enough to make all zero
    hi = target  # enough to make all one
    # Actually: sum(clamp(v + lambda, 0, 1)) is monotone in lambda
    # At lambda = -min(v), at least one is 0
    # At lambda = 1 - max(v), at least one is 1
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    # Exact mass correction
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0, 1)
    return h

def compute_grad_row(h, s):
    """Gradient of correlation at lag s (signed) with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # h[j+s] * (1-h[j]) for j=0..n-s-1

        g[idx1] += (1.0 - h[idx1 - s])
        idx2 = np.arange(0, n - s)
        g[idx2] += (-h[idx2 + s])
    else:
        a = -s
        # h[j] * (1-h[j-a]) for j=a..n-1  (equivalently h[j+a] * (1-h[j]) for j=0..n-a-1)
        # Wait, for s < 0, lag is s = -(n-1) + r...

        # c(s) = sum over valid j of h[j+s] * (1-h[j])
        # If s < 0, let a = -s > 0
        # c(-a) = sum_{j=a}^{n-1} h[j-a] * (1-h[j])
        # d/dh[i]: 
        #   i appears in h[j-a] when j=i+a, valid if i+a < n, i>=0: contribution (1-h[i+a])
        #   i appears in (1-h[j]) when j=i, valid if i>=a: contribution -h[i-a]
        idx1 = np.arange(0, n - a)
        g[idx1] += (1.0 - h[idx1 + a])
        idx2 = np.arange(a, n)
        g[idx2] += (-h[idx2 - a])
    return g

def softmax_logsumexp(c, beta):
    """Compute (1/beta)*log(sum(exp(beta*c))) and its gradient weights"""
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    exp_vals = np.exp(shifted)
    logsum = c_max + np.log(np.sum(exp_vals)) / beta
    weights = exp_vals / np.sum(exp_vals)
    return logsum, weights

def lse_objective_and_grad(h, beta, penalty):
    """Log-sum-exp objective with sum penalty"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_scaled = corr * dx
    lse_val, weights = softmax_logsumexp(c_scaled, beta)
    
    # Sum penalty
    sum_h = np.sum(h)
    target = n / 2.0
    penalty_val = penalty * (sum_h - target) ** 2
    
    total = lse_val + penalty_val
    
    # Gradient
    grad = np.zeros(n)
    for r, w in enumerate(weights):
        if w < 1e-10:
            continue
        s = r - (n - 1)
        grad += w * dx * compute_grad_row(h, s)
    grad += 2 * penalty * (sum_h - target)
    
    return total, grad

def optimize_lse(h_init, beta, penalty, maxiter=200, ftol=1e-10):
    """L-BFGS-B optimization of log-sum-exp + penalty"""
    n = len(h_init)
    
    def func(x):
        val, grad = lse_objective_and_grad(x, beta, penalty)
        return val, grad
    
    x0 = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    
    result = optimize.minimize(
        lambda x: func(x)[0],
        x0,
        jac=lambda x: func(x)[1],
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxiter * 2}
    )
    return result.x

def slsqp_refine(h, n, maxiter=100):
    """SLSQP minimax refinement"""
    dx = 2.0 / n
    target = n / 2.0
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        hx = x[:n]
        return x[n] - dx * np.correlate(hx, 1.0 - hx, mode="full")
    
    def ineq_jac(x):
        hx = x[:n]
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, s in enumerate(lags):
            j[ki, :n] = -dx * compute_grad_row(hx, s)
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    h0 = project_to_target(h, n)
    s0 = official_score(h0)
    theta0 = s0 + 1e-6
    
    x0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None:
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < s0 - 1e-10:
                return h_out, s
    except:
        pass
    return h0, s0

def interpolate_parent(parent, target_n, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(target_n, 0.5), target_n)
    xs = (np.arange(m) + 0.5) * (2.0 / m)
    xt = ((np.arange(target_n) + 0.5 + offset) % target_n) * (2.0 / target_n)
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return project_to_target(h, target_n)

For coordinate i:
- For each lag s, the coefficient of h[i] in c(s) is:
  - If i-s >= 0 and i-s < n: h[i] appears as h[j+s] where j = i-s, coefficient is (1-h[i-s])
  - If i+s >= 0 and i+s < n: h[i] appears in (1-h[j]) where j = i, coefficient is -h[i+s]
  - So a_s = (1-h[i-s])*[valid] - h[i+s]*[valid]
  - And b_s = c(s) - a_s * h[i] (the current value minus the h[i] contribution)

Then we want to minimize max_s (a_s * x + b_s) over x ∈ [0,1].

The minimum of a piecewise-linear convex function occurs where two lines cross, or at a boundary. We can find it by:
1. Evaluating at x=0 and x=1
2. Finding all pairwise intersection points in [0,1]
3. Checking which gives the minimum of the max

For n=65, there are 129 lags, so 129*129/2 ≈ 8300 intersection points to check. This is fast enough.

Actually, a simpler approach: the minimum of max of linear functions over [0,1] can be found by bisection on t:
- For a given t, check if max_s (a_s * x + b_s) ≤ t has a feasible x ∈ [0,1]
- This gives a system of linear inequalities: a_s * x ≤ t - b_s for all s
- Which means: x ≥ (t - b_s) / a_s for a_s > 0, x ≤ (t - b_s) / a_s for a_s < 0
- So feasible iff max_{a_s>0} (t-b_s)/a_s ≤ min_{a_s<0} (t-b_s)/a_s, and both in [0,1]

Bisection on t is clean and O(n) per iteration.

def coordinate_update(h, i, n):
    """Find optimal h[i] given all other coordinates fixed"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Current value
    current = h[i]
    
    # For each lag, compute a_s and b_s
    # c_s = sum h[j+s]*(1-h[j]) * dx
    # h[i] appears with coefficient:
    #   In h[j+s] term: j = i-s, valid if 0<=i-s<n, coefficient = (1-h[i-s])
    #   In (1-h[j]) term: j = i, valid if 0<=i+s<n, coefficient = -h[i+s]
    lags = np.arange(-(n-1), n)
    n_lags = 2*n - 1
    
    a = np.zeros(n_lags)
    for r, s in enumerate(lags):
        coeff = 0.0
        if 0 <= i - s < n:
            coeff += (1.0 - h[i - s])
        if 0 <= i + s < n:
            coeff += (-h[i + s])
        a[r] = coeff * dx
    
    b = corr - a * current
    
    # Minimize max_r (a[r]*x + b[r]) for x in [0,1]
    # Bisection on t
    lo = 0.0
    hi = np.max(np.abs(a)) + np.max(np.abs(b)) + 1.0
    
    best_x = current
    best_t = np.max(np.abs(a) + np.abs(b))
    
    for _ in range(80):

        # Check if exists x in [0,1] such that a[r]*x + b[r] <= t for all r
        # => a[r]*x <= t - b[r] for all r
        lower = 0.0
        upper = 1.0
        feasible = True
        for r in range(n_lags):
            if a[r] > 1e-15:
                x_min = (t - b[r]) / a[r]
                lower = max(lower, x_min)
            elif a[r] < -1e-15:
                x_max = (t - b[r)) / a[r]  # note: dividing by negative flips inequality

                    feasible = False
                    break
        if not feasible or upper < lower:
            lo = t
        else:
            hi = t
            best_x = (lower + upper) / 2
            best_t = t
    
    return best_x

        if w < 1e-12:

        s = lags[r]

def slsqp_refine(h, n, maxiter=80):
    dx = 2.0 / n
    target = n / 2.0
    n_lags = 2 * n - 1
    lags = np.arange(-(n - 1), n)

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:n]
        return x[n] - dx * np.correlate(hx, 1.0 - hx, mode="full")

    def ineq_jac(x):
        hx = x[:n]
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, s in enumerate(lags):
            j[ki, :n] = -dx * compute_grad_row(hx, s)
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    h0 = project_to_target(h, n)
    s0 = official_score(h0)
    theta0 = s0 + 1e-5
    x0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < s0 - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h0, s0

        s = official_score(h_c)
        if np.isfinite(s) and s > 0:
            scored.append((s, h_c))

    scored.sort(key=lambda x: (x[0], len(x[1])))
    best_score = scored[0][0] if scored else 1.0
    best_h = scored[0][1].copy() if scored else np.full(64, 0.5)

    # Phase 1: L-BFGS-B with log-sum-exp
    n_lsq = min(6, len(scored))
    for idx in range(n_lsq):
        remaining = deadline - time.perf_counter()
        if remaining < 60:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        t_per = max(5.0, min(40.0, remaining / max(1, n_lsq - idx) * 0.5))

        import signal
        for beta, pen in [(40, 20), (80, 50), (150, 100)]:
            if time.perf_counter() > deadline - 60:
                break
            h_opt = lse_optimize(h_c, beta, pen, maxiter=int(min(120, t_per * 5)))
            h_opt = project_to_target(h_opt, n_c)
            s_opt = official_score(h_opt)
            if s_opt < best_score - 1e-12:
                best_score = s_opt
                best_h = h_opt.copy()

    # Phase 2: Coordinate descent
    if time.perf_counter() < deadline - 100:
        n_best = len(best_h)
        order = list(range(n_best))
        rng2 = np.random.default_rng(seed + 100)
        for _ in range(3):
            if time.perf_counter() > deadline - 100:
                break
            rng2.shuffle(order)
            remaining_t = min(25.0, (deadline - time.perf_counter()) * 0.15)
            h_cd, s_cd = coordinate_optimize(best_h, n_best, order, max_rounds=20, time_limit=remaining_t)
            if s_cd < best_score - 1e-12:
                best_score = s_cd
                best_h = h_cd.copy()

    # Phase 3: Local descent on pool
    if time.perf_counter() < deadline - 120:
        pool = [(best_score, best_h.copy())]
        for s, h_c in scored[:5]:
            if not np.array_equal(h_c, best_h):
                pool.append((s, h_c.copy()))
        for dn in [-2, 2, 4]:
            n_try = len(best_h) + dn
            if 20 <= n_try <= 140:
                h_i = interpolate_parent(best_h, n_try)
                pool.append((official_score(h_i), h_i))

        pool.sort(key=lambda x: (x[0], len(x[1])))
        n_pool = min(5, len(pool))
        for idx in range(n_pool):
            remaining = deadline - time.perf_counter()
            if remaining < 40:
                break
            h_c = pool[idx][1].copy()
            n_c = len(h_c)
            t_lim = min(20.0, max(5.0, remaining / max(1, n_pool - idx) * 0.5))
            h_loc, s_loc = local_descent_refined(h_c, n_c, max_iters=35, time_limit=t_lim, seed=seed + 200 + idx * 7)
            if s_loc < best_score - 1e-12:
                best_score = s_loc
                best_h = h_loc.copy()

    # Phase 4: Coordinate descent again on best
    if time.perf_counter() < deadline - 60:
        n_best = len(best_h)
        order = list(range(n_best))
        rng.shuffle(order)
        remaining_t = min(20.0, (deadline - time.perf_counter()) * 0.2)
        h_cd, s_cd = coordinate_optimize(best_h, n_best, order, max_rounds=25, time_limit=remaining_t)
        if s_cd < best_score - 1e-12:
            best_score = s_cd
            best_h = h_cd.copy()

    # Phase 5: SLSQP polish
    for _ in range(2):
        if time.perf_counter() > deadline - 25:
            break
        h0 = best_h.copy()
        n0 = len(h0)
        h_ref, s_ref = slsqp_refine(h0, n0, maxiter=60)
        if s_ref < best_score - 1e-12:
            best_score = s_ref
            best_h = h_ref.copy()
        if time.perf_counter() < deadline - 15:
            h_loc, s_loc = local_descent_refined(best_h, len(best_h), max_iters=20, time_limit=8.0, seed=seed + 800)
            if s_loc < best_score - 1e-12:
                best_score = s_loc
                best_h = h_loc.copy()

    # Phase 6: Final L-BFGS on best
    if time.perf_counter() < deadline - 20:
        n_b = len(best_h)
        for beta, pen in [(60, 30), (100, 80)]:
            if time.perf_counter() > deadline - 10:
                break
            h_opt = lse_optimize(best_h, beta, pen, maxiter=60)
            h_opt = project_to_target(h_opt, n_b)
            s_opt = official_score(h_opt)
            if s_opt < best_score - 1e-12:
                best_score = s_opt
                best_h = h_opt.copy()

    # Final: coordinate descent + SLSQP
    if time.perf_counter() < deadline - 8:
        n_f = len(best_h)
        order = list(range(n_f))
        rng.shuffle(order)
        h_cd, s_cd = coordinate_optimize(best_h, n_f, order, max_rounds=15, time_limit=5.0)
        if s_cd < best_score - 1e-12:
            best_score = s_cd
            best_h = h_cd.copy()

    if time.perf_counter() < deadline - 5:
        h_fin, s_fin = slsqp_refine(best_h, len(best_h), maxiter=40)
        if s_fin < best_score - 1e-12:
            best_score = s_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
