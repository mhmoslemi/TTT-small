import time

def eval_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

    lo = -2.0
    hi = 3.0

- sum_n δ_{n+k,j}δ_{n,j}: n must satisfy n+k=j AND n=j, so k=0. Valid if 0≤j<n. = I(k=0)

So Q_k = -(I(k=0) - I(k=i-j) - I(k=j-i) + I(k=0))
= -2*I(k=0) + I(k=i-j) + I(k=j-i)

In terms of output index p = k + (n-1):
Q[p] = -2 if p = n-1, +1 if p = n-1+(i-j), +1 if p = n-1+(j-i), 0 otherwise.

Great, so the formula is:
c[p](δ) = a[p] + B[p]*δ + Q[p]*δ²

And I need to find δ ∈ [0, Tmax] that minimizes max_p c[p](δ).

This is a min-max problem over a set of (at most 3) quadratics and the rest linear. I can solve this exactly or approximately.

For an exact solution, the minimum of max occurs either:
1. At a point where two (or more) curves cross and are both maximal
2. At a local minimum of one of the curves that is the maximum there
3. At a boundary (δ=0 or δ=Tmax)

For a practical solution, golden section search on the (convex-ish) function max_p c[p](δ) should work well.

Actually, is the function convex? Max of linear functions is convex. Max of a convex quadratic and linear functions is convex. Max of a concave quadratic and other stuff is not necessarily convex.

The Q[p] values are: -2 at center, +1 at two other positions. So we have:
- c_center(δ) = a_center + B_center*δ - 2δ² (concave)

- c_{j-i}(δ) = a_{j-i} + B_{j-i}*δ + δ² (convex)

The max of these is not necessarily convex because of the concave center term. However, the max is still a piecewise function that I can search over.

- The first-order change in the active lags is: Δc_k ≈ δ * (B_k(i) - B_k(j))
- We want this to be negative for all active k
- So we want B_k(i) < B_k(j) for all active k

The best pair minimizes max_k (B_k(i) - B_k(j)) over active k, subject to B_k(i) < B_k(j) for the most important k.

import time

    """Get indices of lags that are near the maximum"""
    cmax = float(np.max(c))
    tol = max(cmax * tol_frac, 1e-10)
    active = np.where(c >= cmax - tol)[0]
    # If too many, take the top ones
    if len(active) > max_count:
        c_sorted = np.argsort(c)[::-1]
        active = c_sorted[:max_count]
    return active

    """B_k[i] = (1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)"""
    i_arr = np.arange(n)
    B = np.zeros(n)
    idx1 = i_arr - k
    valid1 = (idx1 >= 0) & (idx1 < n)
    B[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = i_arr + k
    valid2 = (idx2 >= 0) & (idx2 < n)
    B[valid2] -= h[idx2[valid2]]
    return B

def subgradient(h, n, active_lag_indices):
    """Compute subgradient g[i] = sum_{k in active} B_k[i]"""
    center = n - 1
    g = np.zeros(n)
    for p in active_lag_indices:
        k = p - center
        g += compute_B_for_lag(h, n, k)
    return g

    Find optimal delta for h_i += delta, h_j -= delta.

    if Tmax < 1e-14:

    B = np.zeros(2 * n - 1)

    # B[p] = B_k(i) - B_k(j)
    # B_k(i) = (1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)
    # B_k(j) = (1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)

        bi = 0.0
        if 0 <= i - k < n:
            bi += (1.0 - h[i - k])
        if 0 <= i + k < n:
            bi -= h[i + k]
        
        bj = 0.0
        if 0 <= j - k < n:
            bj += (1.0 - h[j - k])
        if 0 <= j + k < n:
            bj -= h[j + k]

import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(v, target_sum):
    lo = -3.0
    hi = 4.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_B_pair(h, n, i, j):
    """Compute B[p] for all output indices p, for transfer h_i+=d, h_j-=d"""
    B = np.zeros(2 * n - 1)
    center = n - 1
    ks = np.arange(-(n - 1), n)
    
    # For h_i increase: contribution to c_k is (1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)
    idx_a = i - ks
    va = (idx_a >= 0) & (idx_a < n)
    B[va] += (1.0 - h[idx_a[va]])
    idx_b = i + ks
    vb = (idx_b >= 0) & (idx_b < n)
    B[vb] -= h[idx_b[vb]]
    
    # For h_j decrease: negative of the above with j
    idx_c = j - ks
    vc = (idx_c >= 0) & (idx_c < n)
    B[vc] -= (1.0 - h[idx_c[vc]])
    idx_d = j + ks
    vd = (idx_d >= 0) & (idx_d < n)
    B[vd] += h[idx_d[vd]]
    
    return B

def compute_Q_pair(n, i, j):

    Q = np.zeros(2 * n - 1)
    center = n - 1
    Q[center] = -2.0
    p1 = center + (i - j)
    p2 = center + (j - i)
    Q[p1] += 1.0
    Q[p2] += 1.0
    return Q

    base_max = float(np.max(a))
    
    if Tmax < 1e-14:
        return 0.0, None
    
    def obj(d):
        return float(np.max(a + B * d + Q * d * d))
    
    # Coarse grid
    ng = 21
    grid = np.linspace(0.0, Tmax, ng)
    vals = np.array([obj(d) for d in grid])
    best_idx = int(np.argmin(vals))
    
    if vals[best_idx] >= base_max - 1e-13:
        return 0.0, None
    
    # Refine

        lo, hi = 0.0, grid[1]
    elif best_idx == ng - 1:
        lo, hi = grid[ng - 2], Tmax
    else:
        lo, hi = grid[best_idx - 1], grid[best_idx + 1]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1, f2 = obj(x1), obj(x2)
    
    for _ in range(40):
        if f1 < f2:
            hi = x2; x2 = x1; f2 = f1
            x1 = hi - gr * (hi - lo); f1 = obj(x1)
        else:
            lo = x1; x1 = x2; f1 = f2
            x2 = lo + gr * (hi - lo); f2 = obj(x2)
        if hi - lo < 1e-14:
            break
    
    d_opt = (lo + hi) / 2.0
    v_opt = obj(d_opt)
    if v_opt < base_max - 1e-13:
        return d_opt, v_opt
    return 0.0, None
