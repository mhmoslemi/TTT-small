        passes += 1
        improved = False
        idxs = np.arange(len(pairs))
        rng.shuffle(idxs)
        for pi in idxs:
            if time.time() - t0 >= max_time:
                break
            a, b = pairs[int(pi)]
            t_lo = max(-h[a], h[b] - 1.0)
            t_hi = min(1.0 - h[a], h[b])
            if t_hi - t_lo < 1e-12:
                continue
            try:
                res = minimize_scalar(two_point_obj, args=(h, a, b),
                                      bounds=(t_lo, t_hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 30})
                if res.fun < sc - 1e-13:
                    h[a] += res.x
                    h[b] -= res.x
                    h = np.clip(h, 0.0, 1.0)
                    h = repair(h)
                    sc = official_score(h)
                    improved = True
            except Exception:
                pass
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 2:
                sigma = 0.008 + 0.012 * rng.random()
                h_cand = repair(h + rng.normal(0, sigma, size=n))
                sc_cand = official_score(h_cand)
                if sc_cand < sc:
                    h = h_cand
                    sc = sc_cand
                h_sym = repair((h + (1.0 - h[::-1])) / 2.0)
                sc_sym = official_score(h_sym)
                if sc_sym < sc:
                    h = h_sym
                    sc = sc_sym
                no_improve = 0
                if passes > 200:
                    break
    return h, sc

def subgrad_descent(h0, max_time, rng):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    sc = official_score(h)
    t0 = time.time()
    alpha = 0.008
    stall = 0
    while time.time() - t0 < max_time:
        vals = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = vals.max()
        tol = 0.004 * dx
        active = np.where(vals >= cmax - tol)[0]
        g = np.zeros(n)
        for p in active:
            s = int(p) - (n - 1)
            g += grad_lag(h, s)
        g /= len(active)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        h_new = repair(h - alpha * g)
        sc_new = official_score(h_new)
        if sc_new < sc - 1e-14:
            h = h_new
            sc = sc_new
            stall = 0
        else:
            stall += 1
            alpha *= 0.92
            if alpha < 1e-10:
                break
            if stall > 80:
                h_cand = repair(h + rng.normal(0, 0.01, size=n))
                sc_cand = official_score(h_cand)
                if sc_cand < sc:
                    h = h_cand
                    sc = sc_cand
                alpha = 0.004
                stall = 0
    return h, sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.07 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')
    best_n = 0

    def can_update(h, sc):
        nonlocal best_h, best_score, best_n
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)
        elif sc <= best_score + 1e-14 and len(h) < best_n:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)

    def tleft():
        return deadline - time.time()

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = repair(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                best_n = len(parent)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        best_h = make_wave(48)
        best_score = official_score(best_h)
        best_n = len(best_h)
    can_update(best_h, best_score)

    # Phase 1: Main 2-point exact descent (longest phase)
    if tleft() > 30:
        h1, s1 = two_point_descent(best_h, min(400, 0.45 * tleft()), rng)
        can_update(h1, s1)

    # Phase 2: Subgradient on active lags
    if tleft() > 25:
        h2, s2 = subgrad_descent(best_h, min(120, 0.2 * tleft()), rng)
        can_update(h2, s2)

    # Phase 3: Another 2-point pass from improved solution
    if tleft() > 25:
        h3, s3 = two_point_descent(best_h, min(250, 0.3 * tleft()), rng)
        can_update(h3, s3)

    # Phase 4: Resolution adaptation
    if tleft() > 30:
        n_base = len(best_h)
        n_opts = sorted(set([
            max(24, n_base - 12),
            max(28, n_base - 5),
            n_base,
            min(80, n_base + 5),
            min(80, n_base + 12),
        ]))
        for no in n_opts:
            if tleft() < 20:
                break
            hi = interpolate_to(best_h, no)
            si = official_score(hi)
            can_update(hi, si)
            if tleft() > 25:
                hn, sn = two_point_descent(hi, min(80, 0.25 * tleft()), rng)
                can_update(hn, sn)

    # Phase 5: Multi-start with random perturbations
    if tleft() > 50:
        n = len(best_h)
        n_starts = max(2, min(5, int(tleft() / 60)))
        for i in range(n_starts):
            if tleft() < 30:
                break
            sigma = 0.02 + 0.06 * rng.random()
            hc = repair(best_h + rng.normal(0, sigma, size=n))
            sc_c = official_score(hc)
            if sc_c < best_score:
                hc, sc_c = two_point_descent(hc, min(100, 0.4 * tleft()), rng)
            can_update(hc, sc_c)

    # Phase 6: Anti-symmetric transform + 2-point
    if tleft() > 25:
        h_sym = repair((best_h + (1.0 - best_h[::-1])) / 2.0)
        sc_sym = official_score(h_sym)
        can_update(h_sym, sc_sym)
        if tleft() > 30:
            hs, ss = two_point_descent(h_sym, min(100, 0.3 * tleft()), rng)
            can_update(hs, ss)

    # Phase 7: Final intensive polish
    if tleft() > 20:
        hf, sf = two_point_descent(best_h, min(300, tleft() * 0.8), rng)
        can_update(hf, sf)

    # Final validation
    final_h = repair(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    if not np.isfinite(final_score) or final_h.size < 4:
        final_h = np.full(8, 0.5)
        final_score = official_score(final_h)
        final_n = 8

    return final_h, final_score, final_n
