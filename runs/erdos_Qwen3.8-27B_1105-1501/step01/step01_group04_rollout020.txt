```python
import numpy as np
import time
from scipy.optimize import linprog, minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(k, h, n, dx):
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            m_start, m_end = k, n
            g[m_start:m_end] += (1.0 - h[0:m_end-m_start])
            g[0:m_end-m_start] -= h[k:n]
    else:
        kk = -k
        if kk < n:
            g[0:n-kk] += (1.0 - h[kk:n])
            g[kk:n] -= h[0:n-kk]
    return dx * g

def grad_smoothed(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = corr_scaled(h)
    max_c = np.max(c)
    # Avoid overflow
    z = np.exp(alpha * (c - max_c))
    p = z / np.sum(z)
    # Weighted sum of gradients
    g = np.zeros(n)
    for k in range(2*n-1):
        if p[k] > 1e-15:
            g += p[k] * grad_lag(k, h, n, dx)
    return g

def project_box_sum(h, target, n):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}."""
    h = np.clip(h, 0.0, 1.0)
    # Need sum = target
    # Binary search for the offset
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    offset = (lo + hi) / 2.0
    return np.clip(h - offset, 0.0, 1.0)

def lp_direction(h, active_set, rho):
    """Solve the LP to find a descent direction for the active constraints."""
    n = len(h)
    dx = 2.0 / n
    m = len(active_set)
    
    # Variables: delta (n) + eta (1)
    # c = [0, ..., 0, 1] (minimize eta)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: grad_k · delta - eta <= 0 for each active k
    # => -grad_k · delta + eta >= 0
    # For linprog: A_ub @ x <= b_ub
    # We want: grad_k · delta - eta <= 0 for each k
    # A_ub has rows: [grad_k, -1] for each active k
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, k in enumerate(active_set):
        gk = grad_lag(k, h, n, dx)
        A_ub[i, :n] = gk
        A_ub[i, n] = -1.0
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -rho <= delta_i <= rho, eta unbounded (or bounded)
    bounds = [(-rho, rho)] * n + [(-10.0, 10.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

def slsqp_polish(h0, time_budget):
    """Polish using SLSQP with explicit min-max constraints."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    t_end = time.time() + time_budget
    
    def objective(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def constraint_eq(x):
        return np.array([np.sum(x[1:]) - target])
    
    def constraint_eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, 1:] = 1.0
        return J
    
    def constraint_ineq(x):
        h = x[1:]
        c = corr_scaled(h)
        return x[0] - c  # >= 0 means x[0] >= c[k] for all k
    
    def constraint_ineq_jac(x):
        h = x[1:]
        n_local = len(h)
        dx_local = 2.0 / n_local
        L = 2 * n_local - 1
        J = np.zeros((L, n_local + 1))
        for k in range(L):
            J[k, 0] = 1.0  # derivative of t
            gk = grad_lag(k, h, n_local, dx_local)
            J[k, 1:] = -gk  # derivative of -(c[k])
        return J
    
    x0 = np.zeros(n + 1)
    x0[0] = compute_c5(h0) + 0.01
    x0[1:] = h0
    
    bounds_slsqp = [(None, None)] + [(0.0, 1.0)] * n
    
    result = minimize(objective, x0, method='SLSQP',
                      jac=obj_grad,
                      bounds=bounds_slsqp,
                      options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
    
    if result.success and result.x[0] < compute_c5(h0):
        h_new = np.clip(result.x[1:], 0.0, 1.0)
        h_new = project_box_sum(h_new, target, n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    return h0, compute_c5(h0)

def smooth_optimize(h0, time_budget, seed=42):
    """Smoothed gradient optimization with Adam."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    t_end = t0 + time_budget
    
    alpha = 2.0
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    step = 0
    stall = 0
    
    while time.time() < t_end:
        grad = grad_smoothed(h, alpha)
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad**2
        m_hat = m_buf / (1 - beta1**step)
        v_hat = v_buf / (1 - beta2**step)
        update = m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h - lr * update, target, n)
        
        c5_new = compute_c5(h_new)
```

```python
    z = np.exp(alpha * (c - max_c))
    p = z / np.sum(z)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-14:
            continue
        lag = idx - (n - 1)
        g += w * grad_lag(lag, h, n, dx)
    return g


def project_box_sum(h, target):
    h = np.array(h, dtype=float)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def lp_direction(h, active_idx, rho, n, dx):
    m = len(active_idx)
    if m == 0:
        return None, None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, idx in enumerate(active_idx):
        lag = idx - (n - 1)
        gk = grad_lag(lag, h, n, dx)
        A_ub[i, :n] = gk
        A_ub[i, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds_lp = [(-rho, rho)] * n + [(-10.0, 10.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_optimize(h0, time_budget, seed):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = project_box_sum(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    t_end = t_start + time_budget

    rho = 0.15
    tau = 2e-3
    fail_count = 0
    iter_count = 0

    while time.time() < t_end:
        c = corr_scaled(h)
        C = float(np.max(c))

        active = np.where(c >= C - tau)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 25:
            order = np.argsort(c[active])[::-1][:25]
            active = active[order]

        delta, eta = lp_direction(h, active, rho, n, dx)

        if delta is None or eta >= -1e-10:
            fail_count += 1
            if fail_count > 30:
                tau = max(tau * 0.6, 1e-4)
                rho = max(rho * 0.7, 0.02)
                fail_count = 0
                if rho < 0.01:
                    break
            continue

        improved = False
        for scale in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
            h_new = project_box_sum(h + scale * delta, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
            elif c5_new < C - 1e-10:
                h = h_new
                improved = True
                break

        if improved:
            fail_count = 0
            iter_count += 1
            if iter_count % 5 == 0:
                rho = min(rho * 1.08, 0.25)
            if best_c5 < 0.3815:
                tau = min(tau * 0.92, 3e-5)
        else:
            fail_count += 1
            if fail_count > 40:
                rho *= 0.7
                tau = max(tau * 0.5, 5e-5)
                fail_count = 0

    return best_h, best_c5


def smooth_optimize(h0, time_budget, seed):
    n = len(h0)
    target = n / 2.0
    h = project_box_sum(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    t_end = t_start + time_budget

    alpha = 3.0
    lr = 0.008
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    step = 0
    stall = 0

    while time.time() < t_end:
        grad = grad_smoothed_fast(h, alpha)
        step += 1
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        update = m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h - lr * update, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                alpha *= 2.0
                lr *= 0.75
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                step = 0
                stall = 0
                rng = np.random.RandomState(seed + step)
                h = project_box_sum(best_h + 0.02 * rng.randn(n), target)
        if step % 50 == 0:
            alpha = min(alpha * 1.04, 80.0)
            lr = max(lr * 0.997, 0.0003)

    return best_h, best_c5


def slsqp_polish(h0, time_budget):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1

    def obj_func(x):
        return x[0]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def con_eq(x):
        return np.array([np.sum(x[1:]) - target])

    def con_eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, 1:] = 1.0
        return J

    def con_ineq(x):
        hh = x[1:]
        cc = np.correlate(hh, 1.0 - hh, mode='full') * dx
        return x[0] - cc

    def con_ineq_jac(x):
        hh = x[1:]
        J = np.zeros((L, n + 1))
        for idx in range(L):
            lag = idx - (n - 1)
            gk = grad_lag(lag, hh, n, dx)
            J[idx, 0] = 1.0
            J[idx, 1:] = -gk
        return J

    x0 = np.zeros(n + 1)
    x0[0] = compute_c5(h0) + 0.02
    x0[1:] = project_box_sum(h0, target)
    bounds_slsqp = [(None, None)] + [(0.0, 1.0)] * n

    try:
        result = minimize(obj_func, x0, method='SLSQP', jac=obj_jac,
                          constraints=[
                              {'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac},
                              {'type': 'ineq', 'fun': con_ineq, 'jac': con_ineq_jac}
                          ],
                          bounds=bounds_slsqp,
                          options={'maxiter': 300, 'ftol': 1e-14, 'disp': False})
        if result.success:
            h_new = project_box_sum(result.x[1:], target)
            c5_new = compute_c5(h_new)
            c5_orig = compute_c5(h0)
            if c5_new < c5_orig:
                return h_new, c5_new
    except Exception:
        pass
    return h0, compute_c5(h0)


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return np.clip(h_new, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = project_box_sum(h_init, n0 / 2.0)
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)

    if best_h is None:
        n0 = 99
        rng = np.random.RandomState(seed)
        h_r = np.zeros(n0)
        perm = rng.permutation(n0)
        h_r[perm[:n0 // 2]] = 1.0
        h_r[perm[n0 // 2]] = 0.5
        best_h = project_box_sum(h_r, n0 / 2.0)
        best_c5 = compute_c5(best_h)

    # Phase 1: LP on initial
    if remaining() > 80:
        alloc = min(remaining() * 0.25, 250.0)
        h_b, c5_b = lp_optimize(best_h, alloc, seed)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 2: Smoothed Adam
    if remaining() > 60:
        alloc = min(remaining() * 0.2, 180.0)
        h_b, c5_b = smooth_optimize(best_h, alloc, seed + 100)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 3: LP again
    if remaining() > 50:
        alloc = min(remaining() * 0.2, 200.0)
        h_b, c5_b = lp_optimize(best_h, alloc, seed + 200)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 4: Dimension exploration
    if remaining() > 50:
        n_cur = len(best_h)
        dims_try = []
        for d in [-3, -2, -1, 1, 2, 3, -5, 5, -7, 7]:
            nt = n_cur + d
            if 40 <= nt <= 200 and nt != n_cur:
                dims_try.append(nt)
        for nt in dims_try[:6]:
            if remaining() < 35:
                break
            h_interp = interpolate_h(best_h, nt)
            h_interp = project_box_sum(h_interp, nt / 2.0)
            c5_i = compute_c5(h_interp)
            if c5_i < best_c5:
                best_c5 = c5_i
                best_h = h_interp
            alloc = min(remaining() * 0.15, 100.0)
            if alloc < 15:
                continue
            h_b, c5_b = lp_optimize(h_interp, alloc, seed + nt)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b

    # Phase 5: SLSQP polish
    if remaining() > 45:
        alloc = min(remaining() * 0.3, 280.0)
        h_b, c5_b = slsqp_polish(best_h, alloc)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 6: Final LP
    if remaining() > 25:
        alloc = min(remaining() * 0.4, 180.0)
        h_b, c5_b = lp_optimize(best_h, alloc, seed + 500)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 7: Final smooth
    if remaining() > 15:
        alloc = min(remaining() * 0.3, 80.0)
        h_b, c5_b = smooth_optimize(best_h, alloc, seed + 700)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
```
