def official_score(h):

    if n == 0:

    dx = 2.0 / n

    return float(np.max(corr) * dx)

        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)

        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)

            h[idx] -= delta

    return h

    for _ in range(n_swaps):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            h_new[i], h_new[j] = h_new[j], h_new[i]
    return repair_vector(h_new, n)

def interpolate_to_n(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n_target, 0.5), n_target)
    if m == n_target:
        return repair_vector(parent.copy(), n_target)
    xs = (np.arange(m) + 0.5) / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) / n_target
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n_target)

def gen_candidate(n, rng, mode='uniform'):
    if mode == 'uniform':
        h = rng.uniform(0, 1, n)
    elif mode == 'binary':
        target = n // 2
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:target]] = 1.0
        if n % 2 == 1:
            h[idx[target]] = 0.5
    elif mode == 'sine':
        t = np.linspace(0, 1, n)
        freq = float(rng.uniform(2, 7))
        phase = float(rng.uniform(0, 2 * np.pi))
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t + phase)
    elif mode == 'block':
        n_blocks = int(rng.integers(2, 6))
        if n_blocks - 1 < n - 1:
            boundaries = np.sort(rng.integers(1, n, size=n_blocks - 1))
        else:
            boundaries = np.arange(1, n_blocks, dtype=int)
        boundaries = np.concatenate([[0], boundaries, [n]])
        vals = rng.uniform(0.0, 1.0, n_blocks)
        h = np.zeros(n)
        for i in range(n_blocks):
            h[boundaries[i]:boundaries[i + 1]] = vals[i]
    elif mode == 'triangular':
        t = np.linspace(0, 1, n)
        h = 1.0 - 2.0 * np.abs(t - 0.5)
    else:
        h = rng.uniform(0, 1, n)
    return repair_vector(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 12.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) >= 8 and len(pv) <= 200 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    rng = np.random.default_rng(seed)

    if parent is not None:
        best_n = len(parent)
        best_h = repair_vector(parent.copy(), best_n)
    else:
        best_n = 65
        best_h = repair_vector(np.full(65, 0.5), 65)

    best_score = official_score(best_h)

    # Initial full descent
    h_w = best_h.copy()
    h_w, s_w = pair_descent(h_w, n_grid=14, n_golden=8, max_passes=5,
                            deadline=t0 + 15.0, rng=rng)
    if s_w < best_score:
        best_score = s_w
        best_h = h_w.copy()

    # Generate diverse candidates
    candidates = []
    candidates.append(best_h.copy())
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h).copy())
    candidates.append((1.0 - best_h[::-1]).copy())

    n_list = sorted(set([best_n, best_n - 3, best_n - 1, best_n + 1,
                         best_n + 3, best_n + 6, best_n - 6, best_n + 10]))
    n_list = [x for x in n_list if 24 <= x <= 120]

    for n_try in n_list:
        if time.perf_counter() > deadline - 80:
            break
        for off in [0, 1, 2]:
            h_c = interpolate_to_n(best_h, n_try, offset=off)
            candidates.append(h_c)
        for mode in ['sine', 'binary', 'block']:
            h_c = gen_candidate(n_try, rng, mode=mode)
            candidates.append(h_c)

    # Score and sort candidates
    scored = []
    for h_c in candidates:
        if not np.all(np.isfinite(h_c)):
            continue
        h_c = repair_vector(h_c, len(h_c))
        s_c = official_score(h_c)
        scored.append((s_c, h_c))
    scored.sort(key=lambda x: (x[0], len(x[1])))

    # Update best from scored
    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    # Descend on top candidates
    n_top = min(6, len(scored))
    for idx in range(n_top):
        if time.perf_counter() > deadline - 120:
            break
        s_c, h_c = scored[idx]
        remaining = deadline - time.perf_counter()
        if remaining < 20:
            break
        t_lim = min(remaining * 0.35, 25.0)
        h_d, s_d = pair_descent(h_c.copy(), n_grid=14, n_golden=8,
                                max_passes=6, deadline=t0 + t_lim, rng=rng)
        if s_d < best_score - 1e-14:
            best_score = s_d
            best_h = h_d.copy()
            best_n = len(h_d)

    # Basin hopping loop
    hop_count = 0
    stagnant = 0
    best_stagnant = stagnant

    while time.perf_counter() < deadline - 60:
        hop_count += 1

        # Adaptive perturbation
        base_scale = 0.08 + 0.06 * min(float(hop_count) / 40.0, 1.0)
        if stagnant > 15:
            base_scale += 0.08
            stagnant = 0

        remaining = deadline - time.perf_counter()
        if remaining < 15:
            break

        # Choose mutation type
        mu_type = rng.random()
        if mu_type < 0.4:
            h_mut = block_perturb(best_h, rng, scale=base_scale)
        elif mu_type < 0.7:
            h_mut = swap_perturb(best_h, rng, n_swaps=int(rng.integers(2, 5)))
        else:
            h_mut = block_perturb(best_h, rng, scale=base_scale * 0.5)
            h_mut = swap_perturb(h_mut, rng, n_swaps=1)

        t_lim = min(remaining * 0.25, 18.0)
        h_d, s_d = pair_descent(h_mut, n_grid=12, n_golden=6,
                                max_passes=4, deadline=t0 + t_lim, rng=rng)

        if s_d < best_score - 1e-13:
            best_score = s_d
            best_h = h_d.copy()
            best_n = len(h_d)
            stagnant = 0
        else:
            stagnant += 1

        # Periodic: try blended random candidate
        if hop_count % 8 == 0 and time.perf_counter() < deadline - 80:
            h_rnd = gen_candidate(best_n, rng, mode='sine')
            blend = 0.3 + 0.3 * rng.random()
            h_blend = (1.0 - blend) * best_h + blend * h_rnd
            h_blend = repair_vector(h_blend, best_n)
            t_lim2 = min(remaining * 0.2, 12.0)
            h_b2, s_b2 = pair_descent(h_blend, n_grid=12, n_golden=6,
                                      max_passes=4, deadline=t0 + t_lim2, rng=rng)
            if s_b2 < best_score - 1e-13:
                best_score = s_b2
                best_h = h_b2.copy()
                best_n = len(h_b2)
                stagnant = 0

        # Try different dimension occasionally
        if hop_count % 20 == 0 and time.perf_counter() < deadline - 100:
            n_alt = best_n + int(rng.integers(-5, 8))
            if 24 <= n_alt <= 120 and n_alt != best_n:
                h_alt = interpolate_to_n(best_h, n_alt, offset=int(rng.integers(0, 3)))
                t_lim3 = min(remaining * 0.2, 14.0)
                h_a2, s_a2 = pair_descent(h_alt, n_grid=12, n_golden=6,
                                          max_passes=5, deadline=t0 + t_lim3, rng=rng)
                if s_a2 < best_score - 1e-13:
                    best_score = s_a2
                    best_h = h_a2.copy()
                    best_n = len(h_alt)

    # Final fine-grained polish
    if time.perf_counter() < deadline - 20:
        remaining = deadline - time.perf_counter()
        t_lim = min(remaining * 0.7, 30.0)
        h_fin, s_fin = pair_descent(best_h.copy(), n_grid=20, n_golden=10,
                                    max_passes=8, deadline=t0 + t_lim, rng=rng)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    # Ultra-fine final pass
    if time.perf_counter() < deadline - 8:
        remaining = deadline - time.perf_counter()
        t_lim = min(remaining * 0.9, 15.0)
        h_ultra, s_ultra = pair_descent(best_h.copy(), n_grid=24, n_golden=12,
                                        max_passes=10, deadline=t0 + t_lim, rng=rng)
        if s_ultra < best_score:
            best_score = s_ultra
            best_h = h_ultra.copy()

    # Final validation
    best_h = repair_vector(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
