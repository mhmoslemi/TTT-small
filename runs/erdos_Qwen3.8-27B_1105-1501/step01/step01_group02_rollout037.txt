    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:

    for i, j in pairs:
        key = (min(i, j), max(i, j))
        if key in seen:
            continue
        seen.add(key)
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta


def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_golden_structure(n, rng):
    h = np.zeros(n)
    phi = 0.6180339887
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 - frac
    h = project_onto_constraint(h, n / 2.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    n_base = 70
    candidates = []

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) == n_base:
            candidates.append(h0)
    except (NameError, Exception):
        pass

    if not candidates:
        candidates.append(np.full(n_base, 0.5))

    for _ in range(8):
        candidates.append(make_random_binary(n_base, rng))
    candidates.append(make_golden_structure(n_base, rng))

    best_h = None
    best_c5 = float('inf')
    for h_c in candidates:
        h_c = project_onto_constraint(h_c.copy(), n_base / 2.0)
        c5 = compute_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    n = n_base

    # Phase 1: Full pair sweeps
    phase1_end = t0 + budget_s * 0.50
    stall = 0
    while time.time() < phase1_end:
        h = best_h.copy()
        full_pair_sweep(h, rng)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                h = best_h + rng.randn(n) * 0.04
                h = project_onto_constraint(h, n / 2.0)
                for _ in range(400):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    d = pair_optimal_delta(h, i, j)
                    if abs(d) > 1e-12:
                        h[i] -= d
                        h[j] += d
                cp = compute_c5(h)
                if cp < best_c5:
                    best_c5 = cp
                    best_h = h.copy()
                stall = 0

    # Phase 2: Gradient-guided sweeps
    phase2_end = t0 + budget_s * 0.80
    stall = 0
    while time.time() < phase2_end:
        h = best_h.copy()
        gradient_guided_sweep(h, rng)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 6:
                scale = 0.01 + 0.02 * (stall % 5)
                h = best_h + rng.randn(n) * scale
                h = project_onto_constraint(h, n / 2.0)
                for _ in range(500):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    d = pair_optimal_delta(h, i, j)
                    if abs(d) > 1e-12:
                        h[i] -= d
                        h[j] += d
                cp = compute_c5(h)
                if cp < best_c5:
                    best_c5 = cp
                    best_h = h.copy()
                stall = 0

    # Phase 3: Fine random pair search
    h = best_h.copy()
    while time.time() < t0 + budget_s - 3:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = pair_optimal_delta(h, i, j)
        if abs(d) > 1e-12:
            h[i] -= d
            h[j] += d
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    best_h = project_onto_constraint(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
