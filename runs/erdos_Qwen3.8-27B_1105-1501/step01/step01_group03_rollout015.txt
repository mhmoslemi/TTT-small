```python
    start = max(0, -k)
    end = min(n, n - k)
    if start < end:

    p_start = max(0, start + k)

def symmetry_transform(h):
    return make_feasible(1.0 - h[::-1])


def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def h_to_logit(h, n):
    """Convert h to logit coordinates (y) assuming c=0 initially."""
    h_clipped = np.clip(h, 1e-10, 1.0 - 1e-10)
    y = np.log(h_clipped / (1.0 - h_clipped))
    return y
```

```python
def lag_gradient(h, k, n):
    """Gradient of F_k = sum_i h[i+k]*(1-h[i]) with respect to h.
    where i ranges over valid indices."""
    grad = np.zeros(n)
    # Case 1: j = i+k, so i = j-k. Contribution: (1-h[i]) = (1-h[j-k])
    # Valid when: max(0,-k) <= j-k <= min(n-1, n-1-k)
    # i.e., max(k, 0) <= j <= min(n-1+k, n-1)
    j_lo1 = max(k, 0)
    j_hi1 = min(n - 1 + k, n - 1) + 1  # exclusive
    if j_lo1 < j_hi1:
        grad[j_lo1:j_hi1] += (1.0 - h[j_lo1 - k:j_hi1 - k])
    
    # Case 2: j = i. Contribution: -h[i+k] = -h[j+k]
    # Valid when: max(0,-k) <= j <= min(n-1, n-1-k)
    j_lo2 = max(0, -k)
    j_hi2 = min(n - 1, n - 1 - k) + 1  # exclusive
    if j_lo2 < j_hi2:
        grad[j_lo2:j_hi2] += (-h[j_lo2 + k:j_hi2 + k])
    
    return grad
```
