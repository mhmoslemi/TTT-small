def make_feasible(z):

        if r > 0:

        else:

def smax_grad(h, beta):
    """Return (objective, gradient) of smooth max."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    sh = beta * (vals - cmax)
    w = np.exp(sh)
    ws = w.sum()
    w /= ws
    obj = cmax + np.log(ws) / beta
    g = np.zeros(n)
    omh = 1.0 - h

        wp = w[p]
        if wp < 1e-12: continue

            g[s:] += wp * omh[:n-s]
            g[:n-s] -= wp * h[s:]
        elif s < 0:
            t = -s
            if t < n:
                g[:n-t] += wp * omh[t:]
                g[t:] -= wp * h[:n-t]
    g *= dx
    return obj, g

  - Second factor contribution: $h_{i-s} \cdot (1-h_i) \cdot dx$ → coefficient of $h_i$ is $-h_{i-s} \cdot dx$

  So total coefficient of $h_i$ in $f_s$:
  $a_{s,i} = dx \cdot [(1-h_{i+s}) \cdot \mathbf{1}[0 \leq i+s < n] - h_{i-s} \cdot \mathbf{1}[0 \leq i-s < n]]$

  And the constant term (independent of $h_i$) in $f_s$:
  $b_{s} = f_s - h_i \cdot a_{s,i}$

  But I actually don't need $b_s$ explicitly. I just need to compute:
  $f_s(h_i) = a_{s,i} \cdot h_i + b_s$

  where $b_s = f_s - a_{s,i} \cdot h_i^{(current)}$

  For the 1D minimization, I need:
  $S(h_i) = C(h_i) + \frac{1}{\beta} \log \sum_s \exp(\beta(f_s(h_i) - C(h_i)))$

  where $f_s(h_i) = a_s \cdot h_i + b_s$ and $C(h_i) = \max_s (a_s \cdot h_i + b_s)$.

  Since each $f_s$ is linear in $h_i$ and $S$ is the smooth max of linear functions, $S$ is convex in $h_i$. I can find the minimum by bisection on $S'(h_i)$.

  $S'(h_i) = \sum_s w_s \cdot a_s$ where $w_s$ are the softmax weights.

  $S'(h_i) = 0$ can be found by bisection on $[0, 1]$.

def coord_descent_smax(h0, beta, max_sweeps=100, tol=1e-12):

                js2 = i - s

                    ai -= h[js2]

            b = vals - a * h[i]

        # Re-project
        h = make_feasible(h)

def minimize_1d_convex(a, b, beta, mu, sum_dev, lo=0.0, hi=1.0):
    """Find t in [lo, hi] minimizing smooth max of (a*t + b) plus sum penalty."""
    # S'(t) = sum_s w_s * a_s + mu * sum_dev
    # where w_s = softmax(beta * (a_s*t + b_s - C(t)))
    
    def deriv(t):
        ft = a * t + b
        C = ft.max()
        shifted = beta * (ft - C)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            return 0.0
        w /= ws
        return np.dot(w, a) + mu * sum_dev
    
    # Bisection
    flo, fhi = deriv(lo), deriv(hi)
    if flo > 0: return lo
    if fhi < 0: return hi
    for _ in range(60):
        mid = (lo + hi) / 2.0
        fm = deriv(mid)
        if fm > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float), -2.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - h.sum()
    if abs(r) > 1e-13:
        if r > 0:
            for idx in np.argsort(1.0 - h):
                a = min(r, 1.0 - h[idx])
                h[idx] += a; r -= a
                if abs(r) < 1e-15: break
        else:
            for idx in np.argsort(h):
                a = min(-r, h[idx])
                h[idx] -= a; r += a
                if abs(r) < 1e-15: break
    return h

def smax_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    sh = beta * (vals - cmax)
    w = np.exp(sh)
    ws = w.sum()
    w /= ws
    obj = cmax + np.log(ws) / beta
    g = np.zeros(n)
    omh = 1.0 - h
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-12:
            continue
        s = p - (n - 1)
        if s >= 0 and s < n:
            g[s:] += wp * omh[:n - s]
            g[:n - s] -= wp * h[s:]
        elif s < 0:
            t = -s
            if t < n:
                g[:n - t] += wp * omh[t:]
                g[t:] -= wp * h[:n - t]
    g *= dx
    return obj, g

def _deriv_1d(a, b, beta, mu, sum_dev, t):
    ft = a * t + b
    C = ft.max()
    shifted = beta * (ft - C)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        return 0.0
    w /= ws
    return np.dot(w, a) + mu * sum_dev

        fm = _deriv_1d(a, b, beta, mu, sum_dev, mid)

def coord_descent(h0, beta, mu, n_sweeps=100):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()

    for sweep in range(n_sweeps):
        h_old = h.copy()
        sum_h = h.sum()

            a = np.zeros(2 * n - 1)

                s = p - (n - 1)
                ai = 0.0
                js = i + s
                if 0 <= js < n:

def slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)

    def obj(x):
        return smax_obj_grad(x, beta)[0]

    def grad(x):
        return smax_obj_grad(x, beta)[1]

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def lbfgsb_smooth(h0, beta, penalty=5e5, maxiter=300):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v = smax_obj_grad(x, beta)[0]
        s = x.sum() - target
        return v + penalty * s * s

    def grad(x):
        g = smax_obj_grad(x, beta)[1]
        s = x.sum() - target
        g = g + 2.0 * penalty * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:

def pg_momentum(h0, beta, alpha0=0.05, mu=0.85, maxiter=500):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()

    alpha = alpha0
    for it in range(maxiter):

            best_h = h_new.copy()
        h = h_new
        alpha = alpha0 / (1.0 + it * 0.02)
