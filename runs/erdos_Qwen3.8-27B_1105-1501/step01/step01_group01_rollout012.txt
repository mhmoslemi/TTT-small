    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, target=None):
    z = np.asarray(z, dtype=float)
    n = len(z)
    if target is None:
        target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - z.max()
    hi = 2.0 - z.min()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if res > 0:
                d = min(res, 1.0 - h[i])
            else:
                d = max(res, -h[i])
            h[i] += d
            res -= d
            if abs(res) < 1e-15:
                break
    return h

def lse_grad(h, beta):
    """Compute gradient of LSE(beta * corr) and the LSE value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = exp_vals / exp_sum
    # lse_value = max_c + log(exp_sum) / beta
    lse_val = max_c + np.log(exp_sum) / beta
    
    # Gradient
    g = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n-1)
        if s >= 0:
            g[s:n] += w * (1.0 - h[0:n-s])
            if n-s > 0:
                g[0:n-s] += w * (-h[s:n])
        else:
            s2 = -s
            g[0:n-s2] += w * (1.0 - h[s2:n])
            g[s2:n] += w * (-h[0:n-s2])
    return lse_val, g

def subgrad_topm(h, m=5):
    """Compute subgradient averaged over top-m lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-m:]
    g = np.zeros(n)
    for k in top_idx:
        s = k - (n-1)
        if s >= 0:
            g[s:n] += (1.0 - h[0:n-s])
            if n-s > 0:
                g[0:n-s] += (-h[s:n])
        else:
            s2 = -s
            if n-s2 > 0:
                g[0:n-s2] += (1.0 - h[s2:n])
            g[s2:n] += (-h[0:n-s2])
    return (g / m) * dx  # scale by dx to match objective scaling

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def lse_obj(x, beta):
    val, _ = lse_grad(x, beta)
    return val

def lse_jac(x, beta):
    _, g = lse_grad(x, beta)
    return g

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

        except Exception:
            pass
    
    best_h = project_feasible(best_h)
    final_score = score(best_h)
    return (best_h, final_score, len(best_h))
