def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z):
    # Project z onto {h : 0≤h≤1, sum(h)=n/2}
    n = len(z)
    target = n / 2.0
    z = np.clip(np.nan_to_num(z), 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(60):

        h = np.clip(z + mid, 0, 1)
        if h.sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    # Fix residual
    err = target - h.sum()
    if abs(err) > 1e-13:
        idx = np.argsort(-np.abs(h - 0.5))[::-1]  # change values furthest from 0.5 first
        for i in idx:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1:
                d = min(err, 1 - h[i])
                h[i] += d; err -= d
            elif err < 0 and h[i] > 0:
                d = min(-err, h[i])
                h[i] -= d; err += d
    return h

    """Gradient of f_d = sum_{valid i} h[i](1-h[i+d]) with respect to h."""
    g = np.zeros(n)
    if d >= 0:
        # f_d = sum_{i=0}^{n-1-d} h[i](1-h[i+d])
        # df/dh[i] = (1-h[i+d]) for i < n-d
        # df/dh[i+d] = -h[i] for i+d >= d, i.e., for j >= d where j = i+d
        # So: g[i] += (1-h[i+d]) for i=0..n-1-d
        #     g[i] += -h[i-d] for i=d..n-1
        for i in range(n - d):
            g[i] += (1.0 - h[i + d])
        for i in range(d, n):
            g[i] += (-h[i - d])
    else:

        for i in range(dd, n):
            g[i] += (1.0 - h[i - dd])
        for i in range(n - dd):
            g[i] += (-h[i + dd])
    return g

    if d >= 0:
        return np.sum(h[:n-d] * (1.0 - h[d:]))
    else:
        dd = -d
        return np.sum(h[dd:] * (1.0 - h[:n-dd]))

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # length 2n-1, index n-1+d is f_d

    n = len(h)
    dx = 2.0 / n

    best = h.copy()
    best_score = evaluate_c5(h)
    
    alpha = alpha0

        F = all_f(h, n)
        c5 = np.max(F) * dx
        if c5 < best_score:
            best_score = c5
            best = h.copy()
        
        # Get top K shifts

        weights = np.zeros(len(F))

        maxv = vals[0]
        beta = 50.0

        ws /= ws.sum()

        for ki in range(K):

        if gnorm < 1e-12:

        # Adaptive step: Barzilai-Borwein style

            best = h_new.copy()
            h = h_new
            # Increase step slightly
            alpha = min(alpha * 1.05, 0.5)
        else:
            h = h_new  # still move, but don't update best
            alpha *= cool
        
        if t % 1000 == 0:

    return best, best_score
