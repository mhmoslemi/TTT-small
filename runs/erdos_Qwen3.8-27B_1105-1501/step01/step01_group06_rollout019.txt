import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * dx
    return c5, corr * dx

def project_to_target(v, target):
    """Project v to [0,1]^n with sum = target using water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    
    # Fix residual
    residual = target - float(x.sum())
    if abs(residual) < 1e-12:
        return x
    
    # Sequential fix
    if residual > 0:
        # Need to add residual, prioritize coordinates with most room
        order = np.argsort(1.0 - x)  # most room first
        for idx in order:
            room = 1.0 - x[idx]
            add = min(room, residual)
            if add <= 1e-15:
                continue
            x[idx] += add
            residual -= add
            if abs(residual) < 1e-12:
                break
    else:
        residual = -residual
        order = np.argsort(x)  # least value first (most to remove)
        for idx in order:
            avail = x[idx]
            rem = min(avail, residual)
            if rem <= 1e-15:
                continue
            x[idx] -= rem
            residual -= rem
            if abs(residual) < 1e-12:
                break
    
    return x

def compute_lag_gradients(h, n):
    """
    Compute gradient of each lag score w.r.t. h.
    Returns array of shape (2n-1, n) where grad[k, j] = d(s_k)/dh[j]
    """
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        if d >= 0:
            # S_d = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # dS/dh[j] = (1-h[j-d])*1_{j>=d} - h[j+d]*1_{j<=n-1-d}

from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_to_target(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    residual = target - float(x.sum())
    if abs(residual) < 1e-12:
        return x
    if residual > 0:
        order = np.argsort(1.0 - x)
        for idx in order:
            room = 1.0 - x[idx]
            if room <= 1e-15:
                continue
            add = min(room, residual)
            x[idx] += add
            residual -= add
            if abs(residual) < 1e-13:
                break
    else:
        residual = -residual
        order = np.argsort(x)
        for idx in order:
            avail = x[idx]
            if avail <= 1e-15:
                continue
            rem = min(avail, residual)
            x[idx] -= rem
            residual -= rem
            if abs(residual) < 1e-13:
                break
    return x

def compute_all_grads(h, n):
    """Compute gradient of each lag score. Shape (2n-1, n)."""
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        if d >= 0:
            # dS/dh[j] = (1-h[j-d]) for j in [d, n-1] minus h[j+d] for j in [0, n-1-d]
            js = np.arange(d, n)
            grads[k, js] += (1.0 - h[js - d])
            js2 = np.arange(0, n - d)
            grads[k, js2] -= h[js2 + d]
        else:
            g = -d
            js = np.arange(0, n - g)
            grads[k, js] += (1.0 - h[js + g])
            js2 = np.arange(g, n)
            grads[k, js2] -= h[js2 - g]
    return grads

def line_search_exact(h, p, s_current, n, max_iter=30):
    """Exact line search: find best alpha in [0, alpha_max] such that h+alpha*p is feasible and C5 decreases."""
    # Compute max feasible alpha
    alpha_max = 1e10
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-p[neg]))))
    if alpha_max < 1e-12:
        return 0.0, None
    
    alpha_max = min(alpha_max, 1.0)
    
    # Grid search
    best_alpha = 0.0
    best_c5 = s_current
    n_grid = 21
    alphas = np.linspace(0, alpha_max, n_grid)
    for a in alphas[1:]:  # skip 0
        h_test = h + a * p
        c5_test = exact_c5(h_test)[0]
        if c5_test < best_c5:
            best_c5 = c5_test
            best_alpha = a
    
    if best_alpha == 0.0:
        return 0.0, None
    
    # Refine with golden section around best
    # Find bracket
    idx_best = int((best_alpha / alpha_max) * (n_grid - 1))
    lo_idx = max(0, idx_best - 1)
    hi_idx = min(n_grid - 1, idx_best + 1)
    lo_a = alphas[lo_idx]
    hi_a = alphas[hi_idx]
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    a1 = hi_a - gr * (hi_a - lo_a)
    a2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + a1 * p)[0]
    f2 = exact_c5(h + a2 * p)[0]
    for _ in range(max_iter):
        if f1 < f2:

def solve_lp_direction(h, n, grads, scores, radius, lam=5.0, time_limit=None):
    """
    Solve the minimax LP to find direction p.
    Variables: [p (n), r (1), tau (1)]
    """
    n_vars = n + 2
    p_idx = slice(0, n)
    r_idx = n
    tau_idx = n + 1
    
    # Objective: minimize lam * r + tau
    c = np.zeros(n_vars)
    c[r_idx] = lam
    c[tau_idx] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    A_rows = []
    b_vals = []
    
    # For each lag k: s_k + grad_k @ p - tau <= 0
    # => grad_k @ p + 0*r - 1*tau <= -s_k
    n_lags = 2 * n - 1
    A_lag = np.zeros((n_lags, n_vars))
    A_lag[:, p_idx] = grads
    A_lag[:, tau_idx] = -1.0
    b_lag = -scores
    
    # r constraints: p_j - r <= 0 and -p_j - r <= 0
    A_r = np.zeros((2 * n, n_vars))
    for j in range(n):
        A_r[j, j] = 1.0
        A_r[j, r_idx] = -1.0
        A_r[n + j, j] = -1.0
        A_r[n + j, r_idx] = -1.0
    b_r = np.zeros(2 * n)
    
    A_ub = np.vstack([A_lag, A_r])
    b_ub = np.concatenate([b_lag, b_r])
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, p_idx] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-12)
        hi = min(radius, 1.0 - h[j] - 1e-12)
        if lo >= hi:
            lo = 0.0
            hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, radius))  # r
    bounds.append((None, None))    # tau free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    p = res.x[p_idx]
    return p

def interpolate_to_n(h, n_new):
    """Interpolate h to new resolution n_new."""

    if n_old == n_new:
        return h.copy()

    new_x = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(new_x, old_x, h)
    return project_to_target(h_new, n_new / 2.0)

    for _ in range(25):
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)[0]
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)[0]
    mid_a = (lo_a + hi_a) / 2.0
    f_mid = exact_c5(h + mid_a * p)[0]
    candidates = [(c1, f1), (c2, f2), (mid_a, f_mid)]
    best_cand = min(candidates, key=lambda t: t[1])
    if best_cand[1] < best_c5:
        best_alpha = best_cand[0]
        best_c5 = best_cand[1]
    return best_alpha, best_c5

def solve_lp_direction(h, n, grads, scores, radius, lam):
    n_vars = n + 2
    A_rows = []
    b_vals = []
    n_lags = 2 * n - 1
    A_lag = np.zeros((n_lags, n_vars))
    A_lag[:, 0:n] = grads
    A_lag[:, n + 1] = -1.0
    b_lag = -scores
    A_rows.append(A_lag)
    b_vals.append(b_lag)
    A_r = np.zeros((2 * n + 1, n_vars))
    for j in range(n):
        A_r[j, j] = 1.0
        A_r[j, n] = -1.0
        A_r[n + j, j] = -1.0
        A_r[n + j, n] = -1.0
    A_r[2 * n, n] = 1.0
    b_r = np.zeros(2 * n + 1)
    b_r[2 * n] = radius
    A_rows.append(A_r)
    b_vals.append(b_r)
    A_ub = np.vstack(A_rows)
    b_ub = np.concatenate(b_vals)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-12)
        hi = min(radius, 1.0 - h[j] - 1e-12)
        if lo >= hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, radius))
    bounds.append((None, None))
    c_obj = np.zeros(n_vars)
    c_obj[n] = lam
    c_obj[n + 1] = 1.0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[0:n]

def lp_local_optimize(h, n, max_iters, time_budget, init_radius=0.15, lam=5.0):
    h = project_to_target(h.copy(), n / 2.0)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)[0]
    radius = init_radius
    stagnation = 0
    for it in range(max_iters):
        if time.time() > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr
        grads = compute_all_grads(h, n)
        p = solve_lp_direction(h, n, grads, scores, radius, lam)
        if p is None:
            radius *= 0.5
            stagnation += 1
            if stagnation > 10:
                break
            continue
        if np.max(np.abs(p)) < 1e-12:
            radius *= 0.5
            stagnation += 1
            if stagnation > 10:
                break
            continue
        alpha, c5_new = line_search_exact(h, p, best_c5, n)
        if alpha > 1e-10 and c5_new < best_c5 - 1e-12:
            h_new = h + alpha * p
            h_new = project_to_target(h_new, target)
            c5_actual = exact_c5(h_new)[0]
            if c5_actual < best_c5:
                best_c5 = c5_actual
                best_h = h_new.copy()
                h = h_new.copy()
                radius = min(radius * 1.3, 0.5)
                stagnation = 0
            else:
                radius *= 0.6
                stagnation += 1
        else:
            radius *= 0.6
            stagnation += 1
        if stagnation > 15:
            break
    return best_h, best_c5

def interpolate_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_to_target(h.copy(), n_new / 2.0)
    old_x = (np.arange(n_old) + 0.5) / n_old
    new_x = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(new_x, old_x, h)
    return project_to_target(h_new, n_new / 2.0)

def random_perturb_feasible(h, sigma, rng):
    n = len(h)
    noise = rng.normal(0, sigma, n)
    noise -= noise.mean()
    return project_to_target(h + noise, n / 2.0)

def complement_feasible(h):
    n = len(h)
    return project_to_target(1.0 - h, n / 2.0)

def reverse_feasible(h):
    return project_to_target(h[::-1].copy(), len(h) / 2.0)

def subgradient_step(h, n, k_active, grad_w):
    g = np.zeros(n)
    for idx, w in zip(k_active, grad_w):
        g += w * compute_all_grads(h, n)[idx]
    d = -g
    d -= d.mean()
    dmax = np.max(np.abs(d))
    if dmax < 1e-14:
        return None
    d = d / dmax
    return d

def subgrad_local_optimize(h, n, max_iters, time_budget):
    h = project_to_target(h.copy(), n / 2.0)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)[0]
    radius = 0.1
    for it in range(max_iters):
        if time.time() > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr
        s_max = float(np.max(scores))
        k_active = np.argsort(scores)[-min(10, 2 * n - 1):]
        active_scores = scores[k_active]
        weights = np.maximum(0, active_scores - s_max + np.std(active_scores))
        if np.sum(weights) < 1e-12:
            weights = np.ones(len(k_active))
        weights = weights / np.sum(weights)
        d = subgradient_step(h, n, k_active, weights)
        if d is None:
            break
        alpha, c5_new = line_search_exact(h, d * radius, best_c5, n)
        if alpha > 1e-10 and c5_new < best_c5 - 1e-12:
            h_new = project_to_target(h + alpha * d * radius, target)
            c5_actual = exact_c5(h_new)[0]
            if c5_actual < best_c5:
                best_c5 = c5_actual
                best_h = h_new.copy()
                h = h_new.copy()
            radius = min(radius * 1.2, 0.5)
        else:
            radius *= 0.7
            if radius < 1e-8:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is None:
        n0 = 64
        h0 = np.ones(n0) * 0.5
        for j in range(n0):
            h0[j] = 0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * j / n0)
        h0 = project_to_target(h0, n0 / 2.0)

    n = len(h0)
    h = project_to_target(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)[0]
    best_n = n

    def elapsed():
        return time.time() - t0

    def update(h_new, n_new, c5_new):
        nonlocal best_h, best_c5, best_n
        if c5_new < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = n_new

    # Phase 1: LP local opt on starting point
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = lp_local_optimize(h, n, max_iters=400, time_budget=dl1, init_radius=0.2, lam=3.0)
    update(h1, n, c5_1)
    # Second pass with different params
    if elapsed() < budget_s * 0.30:
        h1b, c5_1b = lp_local_optimize(h1, n, max_iters=300, time_budget=dl1, init_radius=0.08, lam=10.0)
        update(h1b, n, c5_1b)

    # Phase 2: Try different resolutions
    dl2 = t0 + budget_s * 0.60
    n_candidates = sorted(set(list(range(max(16, n - 10), min(128, n + 11))) + [48, 52, 54, 56, 58, 64, 66, 68, 70, 72, 76, 80]),
                          key=lambda x: abs(x - n))
    for nc in n_candidates:
        if time.time() > dl2:
            break
        if nc == best_n:
            continue
        h_interp = interpolate_to_n(best_h, nc)
        per_budget = min(dl2 - time.time(), budget_s * 0.025)
        if per_budget < 3:
            continue
        h_nc, c5_nc = lp_local_optimize(h_interp, nc, max_iters=100, time_budget=time.time() + per_budget,
                                        init_radius=0.15, lam=5.0)
        update(h_nc, nc, c5_nc)

    # Phase 3: Random restarts with LP
    dl3 = t0 + budget_s * 0.85
    sigs = [0.05, 0.1, 0.02, 0.15, 0.08, 0.2, 0.03, 0.12]
    si = 0
    while elapsed() < budget_s * 0.85:
        sig = sigs[si % len(sigs)]
        si += 1
        h_pert = random_perturb_feasible(best_h, sig, rng)
        per_budget = min(budget_s * 0.015, max(2.0, dl3 - time.time()))
        if time.time() + per_budget > dl3:
            per_budget = max(2.0, dl3 - time.time())
        if per_budget < 2:
            break
        h_opt, c5_opt = lp_local_optimize(h_pert, best_n, max_iters=50, time_budget=time.time() + per_budget,
                                          init_radius=0.1, lam=5.0)
        update(h_opt, best_n, c5_opt)

    # Phase 4: Transform exploration
    dl4 = t0 + budget_s * 0.93
    if elapsed() < budget_s * 0.86:
        h_comp = complement_feasible(best_h)
        h_rev = reverse_feasible(best_h)
        for h_t in [h_comp, h_rev]:
            if time.time() > dl4:
                break
            per_b = min(budget_s * 0.015, dl4 - time.time())
            if per_b < 2:
                break
            h_opt2, c5_2 = lp_local_optimize(h_t, best_n, max_iters=40, time_budget=time.time() + per_b,
                                             init_radius=0.12, lam=5.0)
            update(h_opt2, best_n, c5_2)
        # Blend with complement
        for am in [0.3, 0.5, 0.7]:
            if time.time() > dl4:
                break
            h_blend = am * best_h + (1.0 - am) * (1.0 - best_h)
            h_blend = project_to_target(h_blend, best_n / 2.0)
            per_b = min(budget_s * 0.01, dl4 - time.time())
            if per_b < 2:
                break
            h_opt3, c5_3 = lp_local_optimize(h_blend, best_n, max_iters=30, time_budget=time.time() + per_b,
                                             init_radius=0.1, lam=5.0)
            update(h_opt3, best_n, c5_3)

    # Phase 5: Subgradient polish on best
    dl5 = t0 + budget_s * 0.97
    if elapsed() < budget_s * 0.93:
        h_sg, c5_sg = subgrad_local_optimize(best_h, best_n, max_iters=200, time_budget=dl5)
        update(h_sg, best_n, c5_sg)

    # Phase 6: Final LP polish
    dl6 = t0 + budget_s * 0.995
    h_final, c5_final = lp_local_optimize(best_h, best_n, max_iters=300, time_budget=dl6,
                                          init_radius=0.05, lam=20.0)
    update(h_final, best_n, c5_final)

    # Ensure feasibility
    best_h = project_to_target(best_h, best_n / 2.0)
    best_c5 = exact_c5(best_h)[0]

    return best_h, best_c5, best_n
