from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad_row(s, h):
    """Vectorized gradient of V_s w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        if L <= 0:
            return g
        g[s:] += (1.0 - h[:L])
        g[:L] -= h[s:]
    else:
        t = -s
        L = n - t
        if L <= 0:
            return g
        g[:L] += (1.0 - h[t:])
        g[t:] -= h[:L]
    return g * dx

def make_feasible(z):
    """Project z onto {0 <= h <= 1, sum(h) = n/2}."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -2.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
            else:
                add = max(residual, -h[idx])
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def weighted_subgradient(h, beta=50):
    """Compute weighted subgradient of max V_s."""
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    w /= ws
    # Compute weighted gradient
    n = len(h)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] > 1e-10:
            s = p - (n - 1)
            grad += w[p] * compute_grad_row(s, h)
    return grad, cmax

def lp_direction(h, active_idx, delta=0.01):
    """Solve LP to find descent direction using active constraints."""
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    
    # Build LP
    # Variables: h[0], ..., h[n-1], theta
    # Minimize theta
    # Constraints:
    #   V_s*(h) + grad_s · (h - h*) <= theta  for s in active
    #   0 <= h[i] <= 1
    #   sum(h) = n/2
    
    active = [p for p in active_idx if vals[p] >= cmax - delta]
    if not active:
        active = [int(np.argmax(vals))]
    
    A_eq_rows = []
    b_eq = []
    A_ub_rows = []
    b_ub = []
    
    for p in active:
        s = p - (n - 1)
        g = compute_grad_row(s, h)
        # V_p + g · (h - h*) <= theta
        # => g · h - theta <= V_p - g · h*
        row = np.concatenate([g, [-1.0]])
        A_ub_rows.append(row)
        b_ub.append(vals[p] - g @ h)
    
    # Equality: sum(h) = n/2
    eq_row = np.zeros(n + 1)
    eq_row[:n] = 1.0
    A_eq_rows.append(eq_row)
    b_eq.append(n / 2.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    A_eq = np.array(A_eq_rows)
    b_eq = np.array(b_eq)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize theta
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        theta_new = res.x[-1]
        return h_new, theta_new
    return None, None

def _lp_direction(h, top_k=5, delta_frac=0.005):
    """Find a descent direction using LP on active constraints."""
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    
    # Select active lags
    threshold = cmax - delta_frac
    active = np.where(vals >= threshold)[0]
    if len(active) > top_k:
        # Keep the top_k
        active_order = np.argsort(vals[active])[::-1][:top_k]
        active = active[active_order]
    
    # Build LP
    # min theta
    # s.t. V_p + grad_p · (h - h*) <= theta  for p in active
    #      0 <= h <= 1
    #      sum(h) = n/2
    G = all_grads(h)
    n_vars = n + 1  # h[0..n-1], theta
    
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(active), n_vars))
    b_ub = np.zeros(len(active))
    for i, p in enumerate(active):
        g = G[p]
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = vals[p] - g @ h
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:

def all_grads(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                G[p, s:] += (1.0 - h[:L]) * dx
                G[p, :L] -= h[s:] * dx
        else:
            t = -s
            L = n - t
            if L > 0:
                G[p, :L] += (1.0 - h[t:]) * dx
                G[p, t:] -= h[:L] * dx
    return G
