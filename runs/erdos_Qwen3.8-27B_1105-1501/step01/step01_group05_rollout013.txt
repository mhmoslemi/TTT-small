import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_box_simplex(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_lag_gradients(h):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    grads = np.zeros((2 * n - 1, n))
    for ki, k in enumerate(lags):
        if k == 0:
            grads[ki] = 1.0 - 2.0 * h
        elif k > 0:
            grads[ki, :n - k] = 1.0 - h[k:]
            grads[ki, k:] = -h[:n - k]
        else:
            kp = -k
            grads[ki, kp:] = 1.0 - h[:n - kp]
            grads[ki, :n - kp] = -h[kp:]
    return grads


def smooth_grads_and_score(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx * beta
    smax = np.max(scaled)
    exp_s = np.exp(scaled - smax)
    exp_sum = exp_s.sum()
    score = (smax + np.log(exp_sum)) / beta
    p = exp_s / exp_sum
    grads = compute_lag_gradients(h)
    grad = (p * dx) @ grads
    return score, grad, p, corr * dx


def interpolate_to_n(parent, n):
    m = len(parent)
    if m == n:
        return project_onto_box_simplex(parent.copy(), n / 2.0)
    xs = (np.arange(m) + 0.5) * (2.0 / m)
    xt = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(xt, xs, parent)
    return project_onto_box_simplex(h, n / 2.0)

    rng = np.random.default_rng(seed)

    momentum = np.zeros(n)
    beta_schedule = [60, 40, 30, 20, 15, 12, 10, 8, 7, 6, 5, 4]
    iters_per_beta = 25
    for beta in beta_schedule:
        if time.perf_counter() - start > time_limit:
            break

            if time.perf_counter() - start > time_limit:

            gnorm = np.linalg.norm(grad)

                break
            momentum = 0.7 * momentum + 0.3 * grad

def slsqp_refine(h_init, time_limit, seed):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    initial_score = official_score(h_init)

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        return x[n] - dx * np.correlate(h_x, 1.0 - h_x, mode="full")

    def ineq_jac(x):
        h_x = x[:n]
        grads = compute_lag_gradients(h_x)
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        j[:, :n] = -dx * grads
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h_init, [theta0]])
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    start = time.perf_counter()
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 60, 'ftol': 1e-12, 'disp': False}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_onto_box_simplex(result.x[:n], target_sum)
            s = official_score(h_out)
            if s < initial_score:
                return h_out, s
    except Exception:
        pass
    return h_init, initial_score


def gen_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 1e-10:
        h[idx[m_full]] = frac
    return project_onto_box_simplex(h, target)


def gen_continuous(n, rng):
    h = rng.uniform(0, 1, n)
    return project_onto_box_simplex(h, n / 2.0)


def gen_mixed(n, rng):
    h = np.zeros(n)

    m_full = int(target) - rng.integers(0, 4)

    return project_onto_box_simplex(h, target)

        h_sp, s_sp = structured_pair_search(h_gd, t2, seed + idx *  200)
        if s_sp < best_score:
            best_score, best_h, best_n = s_sp, h_sp.copy(), n_c

        h_pe, s_pe = pair_exchange_search(h_sp if s_sp < s_gd else h_gd, t3, seed + idx * 300)
        if s_pe < best_score:
            best_score, best_h, best_n = s_pe, h_pe.copy(), n_c

        current_best = h_pe if s_pe <= s_sp else (h_sp if s_sp <= s_gd else h_gd)
        current_score = min(s_pe, s_sp, s_gd)
        h_sl, s_sl = slsqp_refine(current_best, t4, seed + idx * 400)
        if s_sl < best_score:
            best_score, best_h, best_n = s_sl, h_sl.copy(), n_c

    if time.perf_counter() < deadline - 80:
        h_i2 = interpolate_to_n(best_h, best_n + 3)
        t_rem = min(60.0, deadline - time.perf_counter() - 40)
        h_gd2, s_gd2 = smooth_gradient_descent(h_i2, t_rem * 0.4, seed + 1000)
        h_pe2, s_pe2 = pair_exchange_search(h_gd2, t_rem * 0.3, seed + 1100)
        h_sl2, s_sl2 = slsqp_refine(h_pe2, t_rem * 0.3, seed + 1200)
        if s_sl2 < best_score:
            best_score, best_h, best_n = s_sl2, h_sl2.copy(), best_n + 3

    if time.perf_counter() < deadline - 30:
        h_pol, s_pol = slsqp_refine(best_h, min(20.0, deadline - time.perf_counter() - 10), seed + 2000)
        if s_pol < best_score:
            best_score, best_h = s_pol, h_pol.copy()

    best_h = project_onto_box_simplex(best_h, len(best_h) / 2.0)
    final_score = official_score(best_h)
    best_score = final_score
    if best_score < 0:
        best_score = 0.0
    return (best_h, float(best_score), int(len(best_h)))
