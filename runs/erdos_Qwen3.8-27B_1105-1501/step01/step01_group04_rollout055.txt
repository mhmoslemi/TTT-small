import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    T1 = np.convolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2

def project_to_simplex(h, target):
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def optimize_smoothed(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = project_to_simplex(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    alpha = 20.0
    lr = 0.02
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    best_lr = lr
    
    while time.time() - t0 < time_budget:
        for _ in range(1000):
            if time.time() - t0 >= time_budget:
                break
            grad = smooth_max_grad(h, alpha)
            # Normalize gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-8:
                v = momentum * v - lr * grad / grad_norm
            h_new = project_to_simplex(h + v, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            h = h_new
        
        if stall > 200:
            rng = np.random.RandomState(seed + stall)
            h = best_h + 0.03 * rng.randn(n)
            h = project_to_simplex(h, target)
            v = np.zeros(n)
            lr = max(lr * 0.5, 0.0001)
            alpha = min(alpha * 2.0, 200.0)
            stall = 0
        
        alpha = min(alpha * 1.05, 200.0)
        lr = max(lr * 0.998, 0.00005)
        momentum = min(momentum + 0.0001, 0.95)
    
    return best_h, best_c5

def make_initial(n, rng):

    # Create a spread-out pattern
    h = np.zeros(n)
    # Use golden ratio spacing
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.zeros(int(target), dtype=int)

    seen = set()

        if p not in seen:
            seen.add(p)
            unique_positions.append(p)
    # Fill in
    for p in unique_positions:
        h[p] = 1.0
    # If we need a 0.5 (for odd n)

                    h[z] = 1.0
                    if abs(h.sum() - target) < 1e-10:
                        break
    # Ensure exact sum
    h = project_to_simplex(h, target)
    return h

Actually, let me reconsider the gradient computation. The objective is:
f(h) = max_k sum_i h[i] * (1 - h[i - d_k])

where d_k is the lag corresponding to correlation index k in the full correlation.

For the smooth max approximation:
f_smooth(h) = (1/alpha) * log(sum_k exp(alpha * corr_k))

The gradient is:
df_smooth/dh[j] = sum_k p_k * d(corr_k)/dh[j]

where p_k = exp(alpha * corr_k) / sum_m exp(alpha * corr_m)

Now, corr_k = sum_i h[i] * (1 - h[i - d_k]) where d_k = k - (n-1)

The derivative d(corr_k)/dh[j]:
- h[j] appears in corr_k when i = j: contributes (1 - h[j - d_k]) if j - d_k is valid
- h[j] appears in corr_k when i - d_k = j, i.e., i = j + d_k: contributes -h[j + d_k] if j + d_k is valid

So: d(corr_k)/dh[j] = (1 - h[j - d_k]) * [j-d_k valid] - h[j + d_k] * [j+d_k valid]
