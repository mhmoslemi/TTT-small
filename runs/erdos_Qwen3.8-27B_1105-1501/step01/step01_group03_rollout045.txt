def official_score(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, -2.0, 2.0)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        # Distribute error to non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += err / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix

            if abs(err) < 1e-14:
                break
            if err > 0:

                h[i] += d
                err -= d
            else:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

    """Compute log-sum-exp smooth max of C."""
    cmax = np.max(C)
    if beta < 1e-10:

    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    s = np.sum(es)
    if s < 1e-300:
        return cmax
    return cmax + (mk + np.log(s)) / beta

def smooth_max_grad_weights(C, beta):

    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)

    s = np.sum(es)

    return es / s

    grad = np.zeros(n)

            p = start + k  # positions i+k for i in [start, end)

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, -3.0, 3.0)
    lo = -3.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        order = np.argsort(np.abs(h - 0.5))
        for i in order:
            if abs(err) < 1e-14:
                break
            if err > 0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            else:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_C(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def smooth_max_val(C, beta):
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    s = np.sum(es)
    if s < 1e-300:
        return cmax
    return cmax + (mk + np.log(s)) / beta


def smooth_max_weights(C, beta):
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    s = np.sum(es)
    if s < 1e-300:
        return np.ones(len(C)) / len(C)
    return es / s


def grad_overlap_from_weights(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    offset = n - 1
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        k = idx - offset
        start = max(0, -k)
        end = min(n, n - k)
        if start >= end:
            continue
        grad[start:end] += w * (1.0 - h[start + k:end + k])
        p_start = start + k
        p_end = end + k
        if p_start < p_end:
            grad[p_start:p_end] += w * (-h[start:end])
    return grad * dx


def nesterov_opt(h0, n, beta, n_iters, lr_scale=1.0):
    h = h0.copy()
    y = h.copy()
    dx = 2.0 / n
    L = max(1.0, beta * 2.0)
    lr = lr_scale / L
    for t in range(n_iters):
        mt = 2.0 / (t + 2.0)
        y_new = (1.0 - mt) * y + mt * h
        C = compute_C(y_new)
        w = smooth_max_weights(C, beta)
        g = grad_overlap_from_weights(y_new, w, n)
        h_new = y_new - lr * g
        h_new = make_feasible(h_new)
        h = h_new
        y = h_new
    return h


def nesterov_adaptive_beta(h0, n, beta_schedule, iters_per_phase, lr_scale=1.0):
    h = h0.copy()
    for beta in beta_schedule:
        h = nesterov_opt(h, n, beta, iters_per_phase, lr_scale)
    return h


def obj_lse(x, beta):
    C = compute_C(x)
    return smooth_max_val(C, beta)


def jac_lse(x, beta):
    C = compute_C(x)
    w = smooth_max_weights(C, beta)
    return grad_overlap_from_weights(x, w, len(x))


def eq_sum(x):
    return float(np.sum(x) - len(x) / 2.0)


def eq_sum_jac(x):
    return np.ones(len(x))


def slsqp_lse(h0, beta, maxiter=200):
    n = len(h0)
    h0 = make_feasible(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]
    try:
        res = minimize(obj_lse, h0, args=(beta,), jac=jac_lse,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0


def interp_to_n(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)


def random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return make_feasible(h + 0.3 * rng.uniform(-1, 1, n))


def structured_init(n, rng):
    style = rng.integers(0, 4)
    h = np.zeros(n)
    if style == 0:
        p = rng.integers(3, max(4, n // 3))
        q = rng.integers(1, p)
        for i in range(n):
            h[i] = 1.0 if (i % p) < q else 0.0
    elif style == 1:
        k = rng.integers(1, max(2, n // 6))
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * k * np.arange(n) / n + phase)
    elif style == 2:
        m = rng.integers(n // 4, 3 * n // 4)
        h[:m] = 1.0
    else:
        h = rng.uniform(0.2, 0.8, n)
    return make_feasible(h + 0.1 * rng.normal(0, 1, n))


def constraint_cycling(h0, n, max_rounds=30, top_k=3):
    h = h0.copy()
    for _ in range(max_rounds):
        C = compute_C(h)
        cmax = np.max(C)
        active = np.argsort(C)[-top_k:]
        dx = 2.0 / n

        def obj_cc(x):
            Cx = compute_C(x)
            return float(np.max(Cx[active]))

        def jac_cc(x):
            Cx = compute_C(x)
            cm = np.max(Cx[active])
            am = Cx[active] == cm
            if np.sum(am) == 0:
                am = np.ones(len(active), dtype=bool)
            w = am / np.sum(am)
            grad = np.zeros(n)
            for ii, idx in enumerate(active):
                if w[ii] == 0:
                    continue
                k = idx - (n - 1)
                start = max(0, -k)
                end = min(n, n - k)
                if start >= end:
                    continue
                grad[start:end] += w[ii] * (1.0 - h[start + k:end + k])
                p_start = start + k
                p_end = end + k
                if p_start < p_end:
                    grad[p_start:p_end] += w[ii] * (-h[start:end])
            return grad * dx

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]
        try:
            res = minimize(obj_cc, h, jac=jac_cc, bounds=bounds,
                           constraints=cons, method='SLSQP',
                           options={'maxiter': 50, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = make_feasible(res.x)
                if official_score(h_new) < official_score(h) - 1e-14:
                    h = h_new
        except Exception:
            pass
    return h
