import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    """Project z to feasible set: 0<=h<=1, sum(h)=n/2"""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, -2.0, 2.0)
    
    # Binary search for shift that makes sum(clip(z+shift)) = target
    lo = -2.0 - np.max(z) - 1e-6
    hi = 2.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    
    # Exact correction
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        # Distribute error proportionally
        idx = np.argsort(h)
        for i in idx:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def compute_all_F(h, n):
    """Compute all overlap values F[s] for s = -(n-1) to n-1"""
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F

def compute_lag_gradient(h, k, n):
    """Gradient of F[k] with respect to h"""
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start < end:
        # h[i] contributes (1-h[i+k])
        grad[start:end] += (1.0 - h[start + k:end + k])
        # h[i+k] contributes -h[i]
        p = start + k
        grad[p:end + k] += (-h[start:end])
    return grad

def compute_all_gradients(h, n):
    """Compute gradient of each F[k]"""
    grads = []
    for k in range(-(n - 1), n):
        grads.append(compute_lag_gradient(h, k, n))
    return grads

def safe_sigmoid(x):
    """Numerically stable sigmoid"""
    x = np.asarray(x)
    out = np.zeros_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def safe_logit(h):
    """Numerically stable logit (inverse sigmoid)"""
    h = np.clip(h, 1e-12, 1.0 - 1e-12)
    return np.log(h / (1.0 - h))

def logit_to_h(y, c):
    """Convert logit coordinates y with shift c to h values"""
    return safe_sigmoid(y + c)

def solve_logit_shift(y):
    """Find c such that sum(sigmoid(y+c)) = n/2"""
    n = len(y)
    target = n / 2.0
    
    def sum_fn(c):
        return np.sum(safe_sigmoid(y + c))
    
    # Bracket
    lo, hi = -50.0, 50.0
    s_lo, s_hi = sum_fn(lo), sum_fn(hi)
    if s_lo > target:
        lo, hi = lo - 50, lo
        s_lo = sum_fn(lo)
    if s_hi < target:
        lo, hi = hi, hi + 50
        s_hi = sum_fn(hi)
    
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = sum_fn(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) * 0.5

def logit_objective_and_grad(y, beta):
    """
    Objective and gradient in logit space.
    y: unconstrained parameters
    Returns (objective_value, gradient)
    """
    n = len(y)
    c = solve_logit_shift(y)
    h = logit_to_h(y, c)
    dx = 2.0 / n
    
    F = compute_all_F(h, n)
    C = F * dx
    cmax = np.max(C)
    
    # Log-sum-exp
    shifted = beta * (C - cmax)
    m = np.max(shifted)
    if m < -700:
        es = np.ones_like(shifted) / len(shifted)
        ws = 1.0
        obj = cmax + np.log(len(C)) / beta
    else:
        es = np.exp(shifted - m)
        ws = np.sum(es)
        if ws < 1e-300:
            es = np.ones_like(shifted) / len(shifted)
            ws = 1.0
        obj = cmax + (m + np.log(ws)) / beta
    
    # Weights
    weights = es / ws
    
    # Gradient in h-space: sum_k weights[k] * grad_F[k] * dx
    g_h = np.zeros(n)
    for idx in range(len(C)):
        w = weights[idx]
        if abs(w) < 1e-12:
            continue
        k = idx - (n - 1)
        g_h += w * compute_lag_gradient(h, k, n)
    g_h *= dx
    
    # Chain rule: ∂Obj/∂y_j = h_j(1-h_j) * (g_h[j] - sum(g_h*w)/W)
    w_vec = h * (1.0 - h)
    W = np.sum(w_vec)
    if W < 1e-15:
        return obj, np.zeros(n)
    
    g_dot_w = np.dot(g_h, w_vec)
    g_y = w_vec * (g_h - g_dot_w / W)
    
    return obj, g_y

def smooth_solve_logit(y_init, beta, maxiter=200):
    """Solve logit-space optimization for given beta"""
    n = len(y_init)
    
    def obj(y):
        val, _ = logit_objective_and_grad(y, beta)
        return val
    
    def grad(y):
        _, g = logit_objective_and_grad(y, beta)
        return g
    
    try:
        res = minimize(obj, y_init, jac=grad, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            c = solve_logit_shift(res.x)
            h = logit_to_h(res.x, c)
            return make_feasible(h)
    except Exception:
        pass
    return None

def interp_to_length(h_src, n_src, n_tgt):
    """Interpolate h from n_src to n_tgt samples, preserving area"""
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def symmetry_transform(h):
    n = len(h)
    return make_feasible(1.0 - h[::-1])

def random_perturb(h, sigma, rng):
    n = len(h)
    noise = rng.normal(0.0, sigma, size=n)
    return make_feasible(h + noise)

def active_set_refine(h, n, n_top=8, max_iter=50, step=0.03):
    """
    Active set gradient descent: minimize weighted sum of top lags.
    """
    h = make_feasible(h.copy())
    dx = 2.0 / n
    
    for it in range(max_iter):
        F = compute_all_F(h, n)
        C = F * dx
        cmax = np.max(C)
        
        # Select top n_top lags
        sorted_idx = np.argsort(C)[::-1]
        top_idx = sorted_idx[:min(n_top, len(C))]
        
        # Weight: proportional to how close to max
        top_C = C[top_idx]
        gap = cmax - top_C
        weights = np.exp(-gap * 50.0)
        weights /= weights.sum()
        
        # Compute weighted gradient
        g = np.zeros(n)
        for i, idx in enumerate(top_idx):
            k = idx - (n - 1)
            g += weights[i] * compute_lag_gradient(h, k, n)
        g *= dx
        
        # Project gradient to satisfy sum constraint (subtract mean)
        g = g - np.mean(g)
        
        # Line search
        best_step = 0.0
        best_score = official_score(h)
        for trial_step in [step, step*0.5, step*0.25, step*0.1, step*2.0]:
            h_trial = make_feasible(h - trial_step * g)
            sc = official_score(h_trial)
            if sc < best_score - 1e-12:
                best_score = sc
                best_step = trial_step
        
        if best_step == 0.0:
            break
        
        h = make_feasible(h - best_step * g)
    
    return h
