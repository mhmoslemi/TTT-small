    dx = 2.0 / n
    S = h[i] + h[j]
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    
    if delta_max - delta_min < 1e-12:
        return h, None
    
    # Compute current correlations
    corr = np.correlate(h, 1.0 - h, mode="full")
    lags = np.arange(-(n-1), n)

    # For lag k (index = k + n - 1):
    # b_k = -h[i+k]*I(0<=i+k<n) + (1-h[i-k])*I(0<=i-k<n) 
    #       + h[j+k]*I(0<=j+k<n) - (1-h[j-k])*I(0<=j-k<n)
