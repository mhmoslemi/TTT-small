def score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, s):
    """Gradient of V_s with respect to h.
    V_s = dx * sum_{m} h[m+s]*(1-h[m]) for s >= 0
    V_{-t} = dx * sum_{m} h[m]*(1-h[m+t]) for t > 0 (s = -t)
    """
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        # V_s = dx * sum_{m=0}^{n-1-s} h[m+s]*(1-h[m])
        # dV/dh[i]: 
        #   if i appears as m+s: coefficient is dx*(1-h[m]) = dx*(1-h[i-s]), for i in [s, n-1]
        #   if i appears as m: coefficient is dx*(-h[m+s]) = dx*(-h[i+s]), for i in [0, n-1-s]
        L = n - s
        if L > 0:
            g[s:] += dx * (1.0 - h[:L])  # from h[m+s] term, i = m+s, m = i-s
            g[:L] -= dx * h[s:]           # from (1-h[m]) term, i = m
    else:
        t = -s
        L = n - t
        if L > 0:
            # V_{-t} = dx * sum_{m=0}^{L-1} h[m]*(1-h[m+t])
            # dV/dh[i]:
            #   if i appears as m: coefficient is dx*(1-h[m+t]) = dx*(1-h[i+t]), for i in [0, L-1]
            #   if i appears as m+t: coefficient is dx*(-h[m]) = dx*(-h[i-t]), for i in [t, n-1]
            g[:L] += dx * (1.0 - h[t:])   # from h[m] term
            g[t:] -= dx * h[:L]           # from (1-h[m+t]) term, i = m+t
    return g

def all_grads(h):
    """Gradients for all lags. Returns array of shape (2n-1, n)."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for p in range(L):
        s = p - (n - 1)
        G[p] = grad_lag(h, s)
    return G

        return None
    
    d = res.x[:n]
    T_new = res.x[-1]
    return d, T_new
