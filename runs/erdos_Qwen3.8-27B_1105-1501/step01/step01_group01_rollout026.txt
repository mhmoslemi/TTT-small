import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    """Project z onto {h: 0<=h<=1, sum(h)=target} using water-filling."""
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection on lambda
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_smoothed_grad(h, beta):
    """Compute gradient of log-sum-exp smoothed max of correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)  # length 2n-1, unscaled
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = exp_vals / exp_sum
    
    # grad[i] = dx * sum_k w[k] * [(1-h[i-k]) * I(0<=i-k<n) - h[i+k] * I(0<=i+k<n)]
    # where k = idx - (n-1)
    A = w * dx
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        k = idx - (n - 1)
        a = A[idx]
        if abs(a) < 1e-15:
            continue
        # First part: a * (1 - h[i-k]) for valid i (0 <= i-k < n)
        # i ranges from max(0,k) to min(n-1, n-1+k)
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += a * (1.0 - h[i_lo-k:i_hi-k+1])
        # Second part: -a * h[i+k] for valid i (0 <= i+k < n)
        # i ranges from max(0,-k) to min(n-1, n-1-k)
        i_lo2 = max(0, -k)
        i_hi2 = min(n - 1, n - 1 - k)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_lo2+(i_hi2-i_lo2+1)] -= a * h[i_lo2+k:i_hi2+k+1]
    
    return grad

def compute_smoothed_obj(h, beta):

    dx = 2.0 / n
    corr = compute_corr(h) * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    return max_c + np.log(np.sum(np.exp(shifted))) / beta

def projected_gradient(h0, beta_schedule, max_iters_per_beta=500, lr_init=0.05, momentum=0.9):
    """Projected gradient descent with momentum and adaptive beta."""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)  # momentum
    best_local = h.copy()
    best_local_score = official_score(h)
    
    for beta in beta_schedule:
        lr = lr_init
        for it in range(max_iters_per_beta):
            g = compute_smoothed_grad(h, beta)
            v = momentum * v - lr * g
            h_new = make_feasible(h + v)
            obj_new = compute_smoothed_obj(h_new, beta)
            obj_old = compute_smoothed_obj(h, beta)
            
            if obj_new > obj_old:
                # Backtrack
                lr = lr * 0.5
                v = 0.5 * v
                h_new = make_feasible(h + v)
                if compute_smoothed_obj(h_new, beta) > obj_old:
                    lr = max(lr, 1e-6)
                    v = -lr * g
                    h_new = make_feasible(h + v)
            
            h = h_new
            score = official_score(h)
            if score < best_local_score:
                best_local_score = score
                best_local = h.copy()
        
        # Decay momentum at beta transitions
        v *= 0.5
    
    return best_local, best_local_score
