    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = tau * dx * corr
    max_val = np.max(vals)
    shifted = vals - max_val
    exp_shifted = np.exp(shifted)

def pair_anneal(h, n, steps, T_start=0.01, T_end=0.001, seed=42):

    rng = np.random.default_rng(seed)
    h = repair_vector(h.copy(), n)

    best_h = h.copy()
    best_score = score
    
    for step in range(steps):

        i = rng.integers(0, n)

        delta = rng.uniform(0.02, 0.15)

    vals = tau * dx * corr

    w = es / ses

        else:
            mp = -m
            grad[mp:] += wi * (1.0 - h[:n-mp])
            grad[:n-mp] -= wi * h[mp:]
    grad *= dx
    return obj, grad

import numpy as np

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(500):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(g, n):
    target = n / 2.0
    lo, hi = g.min() - 1.5, g.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(g - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        for _ in range(200):
            if abs(target - h.sum()) < 1e-14:
                break
            r = target - h.sum()
            if r > 0:
                idx = np.argmax(1.0 - h)
                h[idx] = min(1.0, h[idx] + r)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + r)
    return h


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = tau * dx * corr
    mv = np.max(vals)
    sh = vals - mv
    es = np.exp(sh)
    ses = es.sum()
    obj = (mv + np.log(ses)) / tau
    w = es / ses
    grad = np.zeros(n)
    for ki in range(len(corr)):
        wi = w[ki]
        if wi < 1e-10:
            continue
        m = ki - (n - 1)
        if m >= 0:
            grad[:n - m] += wi * (1.0 - h[m:])
            grad[m:] -= wi * h[:n - m]
        else:
            mp = -m
            grad[mp:] += wi * (1.0 - h[:n - mp])
            grad[:n - mp] -= wi * h[mp:]
    grad *= dx
    return obj, grad


def lse_continue(h, n, tau_list, steps_per_phase, lr_base=1.5, seed=0):

    h = repair_vector(h.copy(), n)
    best_h = h.copy()
    best_s = official_score(h)
    cur_s = best_s
    for tau in tau_list:
        for st in range(steps_per_phase):
            _, grad = lse_obj_grad(h, tau)
            grad -= np.mean(grad)
            improved = False
            for lr in [lr_base, lr_base * 0.5, lr_base * 0.25, lr_base * 0.1, lr_base * 0.05]:

def pair_anneal(h, n, num_steps, T_init=0.004, T_final=0.0004, seed=42):
    rng = np.random.default_rng(seed)
    h = repair_vector(h.copy(), n)
    score = official_score(h)
    best_h = h.copy()
    best_s = score
    for st in range(num_steps):
        frac = st / max(1, num_steps - 1)
        T = T_init * (T_final / T_init) ** frac

def gen_interp(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def gen_binary(n, rng):

    m_f = int(target)
    frac = target - m_f
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_f]] = 1.0
    if frac > 0:
        h[idx[m_f]] = frac
    return repair_vector(h, n)


def gen_steps(n, rng, ns):
    if ns >= n:
        return repair_vector(rng.uniform(0, 1, n), n)
    bounds = sorted(rng.choice(n - 1, size=ns - 1, replace=False))
    bounds = [0] + list(bounds) + [n]
    vals = rng.uniform(0, 1, ns)
    h = np.zeros(n)
    for i in range(ns):
        h[bounds[i]:bounds[i + 1]] = vals[i]
    return repair_vector(h, n)


def slsqp_run(h, n, time_limit=15.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    initial_score = official_score(h)

    def obj_fn(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fn(x):
        hx = x[:n]
        theta = x[n]
        c = np.correlate(hx, 1.0 - hx, mode="full")
        return theta - dx * c

    def ineq_jac(x):
        hx = x[:n]
        nlags = 2 * n - 1
        j = np.zeros((nlags, n + 1))
        j[:, n] = 1.0
        for ki, m in enumerate(lags):
            if m >= 0:
                j[ki, :n - m] -= dx * (1.0 - hx[m:])
                j[ki, m:] += dx * hx[:n - m]
            else:
                mp = -m
                j[ki, mp:] -= dx * (1.0 - hx[:n - mp])
                j[ki, :n - mp] += dx * hx[mp:]
        return j

    def eq_fn(x):
        return np.sum(x[:n]) - target_sum

    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    x0 = np.concatenate([h, [initial_score + 1e-6]])
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac}
    ]
    try:
        t0 = time.perf_counter()
        result = optimize.minimize(
            obj_fn, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-12}
        )
        if time.perf_counter() - t0 < time_limit and result.x is not None:
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    deadline = t_start + budget_s - 8.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    rng = np.random.default_rng(seed)

    if parent is not None:
        n_par = len(parent)
        best_h = repair_vector(parent, n_par)
    else:
        n_par = 50
        best_h = repair_vector(np.full(50, 0.5), n_par)

    best_score = official_score(best_h)
    best_n = len(best_h)

    n_list = sorted(set([
        max(28, n_par - 12), max(28, n_par - 6), n_par,
        n_par + 4, n_par + 8, n_par + 12, n_par + 16,
        35, 40, 45, 50, 55, 60, 65, 70, 75
    ]) & set(range(25, 85)))

    candidates = []
    for n_try in n_list:
        candidates.append(gen_interp(best_h, n_try))
        candidates.append(gen_interp(best_h, n_try, offset=1))
        candidates.append(gen_interp(best_h, n_try, offset=2))

    for n_try in n_list[:6]:
        for _ in range(2):
            candidates.append(gen_binary(n_try, rng))
        for ns in [4, 6, 8):
            candidates.append(gen_steps(n_try, rng, ns))
        h_rand = rng.uniform(0, 1, n_try)
        candidates.append(repair_vector(h_rand, n_try))

    scored = []
    for h_c in candidates:
        if not np.all(np.isfinite(h_c)):
            continue
        s = official_score(h_c)
        if np.isfinite(s):
            scored.append((s, h_c))

    scored.sort(key=lambda x: (x[0], len(x[1])))

    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    # Phase 2: LSE continuation on diverse top candidates
    tau_sched = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0]
    top_k = min(10, len(scored))
    seen_hashes = set()

    for idx in range(top_k):
        if time.perf_counter() > deadline - 150:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        h_key = tuple((h_c * 100).astype(int))
        if h_key in seen_hashes:
            continue
        seen_hashes.add(h_key)
        spc = 20
        h_lse, s_lse = lse_continue(h_c, n_c, tau_sched, spc, lr_base=1.2, seed=seed + idx * 7)
        if s_lse < best_score:
            best_score = s_lse
            best_h = h_lse.copy()
            best_n = n_c

    # Phase 3: Exact descent on top candidates
    pool = [(best_score, best_h.copy())]
    for idx in range(min(8, len(scored))):
        if len(pool[0][1]) != len(scored[idx][1]) or not np.allclose(pool[0][1], scored[idx][1]):
            pool.append((scored[idx][0], scored[idx][1].copy()))

    for idx, (s_p, h_p) in enumerate(pool[:7]):
        if time.perf_counter() > deadline - 120:
            break
        n_p = len(h_p)
        remaining = max(15, (deadline - time.perf_counter()) / max(1, len(pool[:7]) - idx))
        t_lim = min(25.0, remaining * 0.7)
        h_loc, s_loc = exact_descent(h_p, n_p, max_iters=70, time_limit=t_lim, seed=seed + idx * 13 + 5)
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_p

    # Phase 4: SLSQP polish
    if time.perf_counter() < deadline - 40:
        h_sp, s_sp = slsqp_run(best_h, best_n, time_limit=12.0)
        if s_sp < best_score:
            best_score = s_sp
            best_h = h_sp.copy()
        if time.perf_counter() < deadline - 20:
            h_l2, s_l2 = exact_descent(h_sp, best_n, max_iters=40, time_limit=8.0, seed=seed + 500)
            if s_l2 < best_score:
                best_score = s_l2
                best_h = h_l2.copy()

    # Phase 5: Annealing
    if time.perf_counter() < deadline - 50:
        h_ann, s_ann = pair_anneal(best_h, best_n, num_steps=5000, T_init=0.003, T_final=0.0003, seed=seed + 77)
        if s_ann < best_score:
            best_score = s_ann
            best_h = h_ann.copy()
        # Descent after annealing
        if time.perf_counter() < deadline - 30:
            h_a2, s_a2 = exact_descent(h_ann, best_n, max_iters=50, time_limit=15.0, seed=seed + 88)
            if s_a2 < best_score:
                best_score = s_a2
                best_h = h_a2.copy()

    # Phase 6: Try resampling best to nearby resolutions
    if time.perf_counter() < deadline - 40:
        for n_try in [best_n + 5, best_n - 5, best_n + 10, best_n - 10]:
            if n_try < 25 or n_try > 85:
                continue
            if time.perf_counter() > deadline - 15:
                break
            h_int = gen_interp(best_h, n_try)
            h_d, s_d = exact_descent(h_int, n_try, max_iters=50, time_limit=10.0, seed=seed + 400 + n_try)
            if s_d < best_score:
                best_score = s_d
                best_h = h_d.copy()
                best_n = n_try

    # Phase 7: Second LSE pass from best
    if time.perf_counter() < deadline - 30:
        h_lse3, s_lse3 = lse_continue(best_h, best_n, [50.0, 150.0, 400.0], 20, lr_base=0.8, seed=seed + 600)
        if s_lse3 < best_score:
            best_score = s_lse3
            best_h = h_lse3.copy()
        if time.perf_counter() < deadline - 10:
            h_d2, s_d2 = exact_descent(h_lse3, best_n, max_iters=40, time_limit=10.0, seed=seed + 700)
            if s_d2 < best_score:
                best_score = s_d2
                best_h = h_d2.copy()

    # Phase 8: Final polish
    if time.perf_counter() < deadline - 12:
        h_fin, s_fin = exact_descent(best_h, best_n, max_iters=50, time_limit=8.0, seed=seed + 999)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    if time.perf_counter() < deadline - 6:
        h_fin2, s_fin2 = slsqp_run(best_h, best_n, time_limit=5.0)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2.copy()

    if time.perf_counter() < deadline - 3:
        h_fin3, s_fin3 = exact_descent(best_h, best_n, max_iters=15, time_limit=3.0, seed=seed + 1111)
        if s_fin3 < best_score:
            best_score = s_fin3
            best_h = h_fin3.copy()

    best_h = repair_vector(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(max(0.0, final_score)), int(len(best_h)))
