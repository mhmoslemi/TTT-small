    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    h = np.clip(h, 0.0, 1.0)
    return h

    """Round h to a 0-1 vector with exactly n/2 ones."""
    target = n // 2
    h_sorted_idx = np.argsort(-h)
    result = np.zeros(n)
    result[h_sorted_idx[:target]] = 1.0
    return result

def local_search_01(h0, max_iter=1000):
    """Local search on 0-1 configurations using swap moves."""
    n = len(h0)
    h = h0.copy()
    curr = official_score(h)
    
    for it in range(max_iter):
        improved = False
        vals = np.correlate(h, 1.0 - h, mode='full')
        max_val = vals.max()
        max_idx = vals.argmax()
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_improve = 0
        best_swap = None
        
        for i in ones:
            for j in zeros:
                # Try swapping: h[i] = 0, h[j] = 1
                h[i] = 0.0
                h[j] = 1.0
                new_max = np.correlate(h, 1.0 - h, mode='full').max()
                h[i] = 1.0
                h[j] = 0.0
                
                if new_max < max_val - 1e-12:
                    improve = max_val - new_max
                    if improve > best_improve:
                        best_improve = improve
                        best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            curr = official_score(h)
            improved = True
        
        if not improved:
            break
    
    return h, official_score(h)

Wait, this is O(n² * n) per iteration (n² swaps, each requiring O(n) correlation computation). For n=60, that's 60² * 60 = 216,000 per iteration. With 1000 iterations, that's 216 million operations. In numpy, each correlation is vectorized, so it should be fast. But 1000 * 3600 correlations might be too slow.

Let me optimize: instead of computing the full correlation for each swap, I can compute only the changed entries.

When we swap h[i] from 1 to 0 and h[j] from 0 to 1:
- The correlation at shift s changes by:
  - Contribution from position i: was h[i]*(1-h[i-s]) or h[i+s]*(1-h[i]) depending on direction
  - Contribution from position j: similar

Actually, this is getting complex. Let me just use the full correlation but limit the number of iterations.

For n=60, np.correlate takes about 1-2 microseconds. So 3600 swaps * 2μs = 7.2ms per iteration. With 1000 iterations, that's 7.2 seconds. That's reasonable!

But wait, we also need to consider that we're doing this in Python with a loop over 3600 pairs. Each iteration of the Python loop has overhead. Let me estimate: each np.correlate call is ~1μs, plus Python loop overhead ~1μs, so ~2μs per swap. 3600 * 2μs = 7.2ms per full sweep. 1000 sweeps = 7.2s. That's fine.

Actually, I realize I should be smarter. Let me only try a subset of swaps:
- Focus on swaps that are likely to reduce the max correlation
- The max correlation is at some shift s*. Changes at positions far from each other won't affect this shift much.

Or even better: use a "first improvement" strategy instead of "best improvement" to speed things up.

Let me also add a "tabu" list to avoid cycling.

Actually, for the budget of 1100 seconds, I have plenty of time. Let me implement:
1. 0-1 local search with best-improvement (limited to 200 iterations or until convergence)
2. Continuous polishing from the 0-1 solution
3. Multiple restarts with different 0-1 initializations
4. Interpolation to different resolutions

Let me also think about a different angle: **the problem has a symmetry**. If h is optimal, then 1-h[::−1] (reversed and complemented) is also optimal. And the "average" (h + 1-h[::−1])/2 might be better.

Actually wait: if h is a solution, then g(x) = 1 - h(2-x) is also a solution (by change of variables x → 2-x). Let me verify:
∫ g(x)(1-g(x+k)) dx = ∫ (1-h(2-x))(1-(1-h(2-x+k))) dx
= ∫ (1-h(2-x)) * h(2-x+k) dx
Let u = 2-x: = ∫ (1-h(u)) * h(u+k) du (with limits flipped, but the integral is over [0,2] both ways)
= ∫ h(u+k)(1-h(u)) du

And the original is ∫ h(x)(1-h(x+k)) dx. Let u = x+k: = ∫ h(u-k)(1-h(u)) du.

These are different! So the symmetry doesn't directly apply. But the max over k of one equals the max over k of the other (since k ranges over all reals/shifts). So yes, g is also optimal.

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add; residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub; residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def round_to_01(h, n):
    target = n // 2
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def local_search_01(h0, max_sweeps=500, first_improve=False):
    n = len(h0)
    h = h0.copy()
    curr = official_score(h)
    for _ in range(max_sweeps):
        vals = np.correlate(h, 1.0 - h, mode='full')
        max_val = vals.max()
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_improve = 0.0
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                nm = np.correlate(h, 1.0 - h, mode='full').max()
                h[i] = 1.0; h[j] = 0.0
                if nm < max_val - 1e-12:
                    imp = max_val - nm
                    if imp > best_improve + 1e-15:
                        best_improve = imp
                        best_swap = (i, j)
                    if first_improve:
                        break
            if first_improve and best_swap is not None:
                break
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0; h[j] = 1.0
            curr = official_score(h)
        else:
            break
    return h.copy(), official_score(h)

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_obj_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    # Gradient of sum_k w_k * c_k * dx where c_k = sum_i h[i+a]*(1-h[i+b])
    # dc_k/dh[j] = sum over pairs involving j
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            # c_k = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # d/dh[j]: if j = i+s => (1-h[i]) = (1-h[j-s]) for i=j-s
            #          if j = i => -h[i+s] = -h[j+s]
            for i in range(n - s):
                if i + s == j:
                    grad[j] += w[k] * (1.0 - x[i])
                if i == j:
                    grad[j] += w[k] * (-x[i + s])
        else:
            t = -s
            # c_k = sum_{i=0}^{n-1-t} h[i-t]*(1-h[i])
            for i in range(t, n):
                if i - t == j:
                    grad[j] += w[k] * (1.0 - x[i])
                if i == j:
                    grad[j] += w[k] * (-x[i - t])
    return dx * grad

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def mm_obj(z):
    return z[-1]

def mm_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def mm_ineq_fun(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    theta = z[-1]
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return theta - vals

def mm_ineq_jac(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    J = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                J[k, i + s] += (1.0 - h[i])
                J[k, i] += (-h[i + s])
        else:
            t = -s
            for i in range(t, n):
                J[k, i - t] += (1.0 - h[i])
                J[k, i] += (-h[i - t])
        J[k, n] = 1.0
    return -dx * J

def mm_eq_fun(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0

def mm_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def try_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def try_direct(h0, maxiter=300):
    n = len(h0)
    theta0 = official_score(h0) + 1e-6
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': mm_ineq_fun, 'jac': mm_ineq_jac},
        {'type': 'eq', 'fun': mm_eq_fun, 'jac': mm_eq_jac}
    ]
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def structured_init(n, mode=0):
    """Generate diverse structured initial 0-1 configurations."""
    h = np.zeros(n)
    half = n // 2
    if mode == 0:
        # Alternating
        h[::2] = 1.0
        if sum(h) > half:
            idx = np.where(h == 1.0)[0]
            h[idx[-1:]] = 0.0
        elif sum(h) < half:
            idx = np.where(h == 0.0)[0]
            h[idx[0]] = 1.0
    elif mode == 1:
        # Quadratic residue-like pattern
        p = n + 1 if (n + 1) ** 0.5 != (n + 1) ** 0.5 else n + 1
        # Use a simple quadratic pattern
        x = np.arange(n)
        scores = (x * x) % (n + 1)
        idx = np.argsort(-scores)[:half]
        h[idx] = 1.0
    elif mode == 2:
        # Sine-based thresholding
        x = np.linspace(0, 2 * np.pi, n, endpoint=False)
        scores = 0.5 + 0.3 * np.sin(3 * x) + 0.2 * np.cos(7 * x + 1.0)
        idx = np.argsort(-scores)[:half]
        h[idx] = 1.0
    elif mode == 3:
        # Striped (blocks of 2)
        for i in range(0, n - 1, 4):
            h[i:i+2] = 1.0
        cnt = int(h.sum())
        if cnt > half:
            idx = np.where(h == 1.0)[0]
            for i in idx[-(cnt - half):]:
                h[i] = 0.0
        elif cnt < half:
            idx = np.where(h == 0.0)[0]
            for i in idx[:half - cnt]:
                h[i] = 1.0
    elif mode == 4:
        # Golden ratio based
        x = np.arange(n)
        scores = np.sin(2 * np.pi * 0.6180339887 * x)
        idx = np.argsort(-scores)[:half]
        h[idx] = 1.0
    return h
