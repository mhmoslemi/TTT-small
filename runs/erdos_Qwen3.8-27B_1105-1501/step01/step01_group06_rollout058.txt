import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(v, target_sum):
    v = np.asarray(v, dtype=float)
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(150):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix any residual
    residual = target_sum - float(np.sum(x))
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        count = np.sum(mask)
        if count > 0:
            x[mask] += residual / count
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_gradients(h):
    n = len(h)
    num_lags = 2 * n - 1
    grads = np.zeros((num_lags, n))
    for k in range(num_lags):
        d = k - (n - 1)
        if d >= 0:
            # j in [0, d-1]: -h[j+d]
            if d > 0 and d < n:
                idx = np.arange(0, min(d, n))
                grads[k, idx] = -h[idx + d]
            # j in [d, n-d-1]: (1-h[j-d]) - h[j+d]
            lo = d
            hi = n - d
            if lo < hi:
                idx = np.arange(lo, hi)
                grads[k, idx] = (1.0 - h[idx - d]) - h[idx + d]
            # j in [n-d, n-1]: (1-h[j-d])
            lo = n - d
            if lo < n:
                idx = np.arange(lo, n)
                grads[k, idx] = 1.0 - h[idx - d]
        else:
            g = -d
            # j in [0, g-1]: (1-h[j+g])
            if g > 0 and g < n:
                idx = np.arange(0, min(g, n))
                grads[k, idx] = 1.0 - h[idx + g]
            # j in [g, n-g-1]: (1-h[j+g]) - h[j-g]
            lo = g
            hi = n - g
            if lo < hi:

            lo = n - g
            if lo < n:

                grads[k, idx] = -h[idx - g]

def solve_lp_direction(h, corr, grads, radius):
    n = len(h)
    num_lags = 2 * n - 1
    
    # Variables: p (n) + tau (1) = n+1 variables
    # Objective: minimize tau
    c = np.zeros(n + 1)
    c[n] = 1.0  # coefficient of tau
    
    # Inequality constraints: grad_k · p - tau <= -corr_k
    # A_ub has shape (num_lags, n+1)
    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = grads  # grad_k · p
    A_ub[:, n] = -1.0    # -tau
    b_ub = -corr  # -corr_k
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: max(-radius, -h_i) <= p_i <= min(radius, 1-h_i)
    p_lo = np.maximum(-radius, -h + 1e-12)
    p_hi = np.minimum(radius, 1.0 - h - 1e-12)
    bounds = list(zip(p_lo, p_hi))
    # tau is free (well, bounded by the constraints)
    bounds.append((-np.inf, np.inf))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def line_search(h, p, current_c5, max_evals=50):
    n = len(h)
    # Find max alpha keeping h + alpha*p in [0,1]
    alpha_max = 1.0
    pos_mask = p > 1e-12
    neg_mask = p < -1e-12
    if np.any(pos_mask):
        alpha_max = min(alpha_max, ((1.0 - h[pos_mask]) / p[pos_mask]).min())
    if np.any(neg_mask):
        alpha_max = min(alpha_max, (h[neg_mask] / (-p[neg_mask])).min())
    alpha_max = max(alpha_max, 1e-10)
    
    # Grid search + refinement
    alphas = np.concatenate([np.linspace(0, alpha_max, 15), [alpha_max]])
    best_alpha = 0.0
    best_c5 = current_c5
    
    for a in alphas:
        h_trial = h + a * p
        c5_trial = exact_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_alpha = a
    
    # Golden section refinement around best_alpha

def solve_lp(h, corr, grads, radius, lam_reg):
    n = len(h)
    num_lags = 2 * n - 1
    # Variables: p(n), tau(1) -> n+1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -corr
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    eps = 1e-12
    p_lo = np.maximum(-radius, -(h - eps))
    p_hi = np.minimum(radius, (1.0 - h) - eps)
    # Ensure p_lo < p_hi
    for i in range(n):
        if p_lo[i] >= p_hi[i]:
            mid = min(max(h[i], 0.0), max(1.0 - h[i], 0.0))
            p_lo[i] = mid - max(radius, 1e-8)
            p_hi[i] = mid + max(radius, 1e-8)
    p_lo = np.maximum(p_lo, -1.0)
    p_hi = np.minimum(p_hi, 1.0)
    bounds = list(zip(p_lo, p_hi))
    bounds.append((-np.inf, np.inf))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        p = res.x[:n]
        # Add small regularization effect by scaling
        norm_p = float(np.max(np.abs(p)))
        if norm_p > 0:
            p = p / norm_p * min(norm_p, radius)
        return p
    return None

def line_search(h, p, current_c5, max_grid=30):
    n = len(h)
    alpha_max = 1.0
    pos_mask = p > 1e-12
    neg_mask = p < -1e-12
    if np.any(pos_mask):
        alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / p[pos_mask]).min()))
    if np.any(neg_mask):
        alpha_max = min(alpha_max, float((h[neg_mask] / (-p[neg_mask])).min()))
    alpha_max = max(alpha_max, 1e-10) * 0.999
    # Dense grid
    alphas = np.linspace(0.0, alpha_max, max_grid)
    best_alpha = 0.0
    best_c5 = current_c5
    for a in alphas:
        h_trial = h + a * p
        c5_t = exact_c5(h_trial)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_alpha = a
    # Refine around best if it's not at boundary
    if 0 < best_alpha < alpha_max * 0.95:
        step = alpha_max / max_grid
        left = max(0.0, best_alpha - step)
        right = min(alpha_max, best_alpha + step)
        for _ in range(25):
            m1 = left + (right - left) / 3.0
            m2 = right - (right - left) / 3.0
            f1 = exact_c5(h + m1 * p)
            f2 = exact_c5(h + m2 * p)
            if f1 < f2:
                right = m2
            else:
                left = m1
            if right - left < 1e-12:
                break
        mid = (left + right) / 2.0
        f_mid = exact_c5(h + mid * p)
        if f_mid < best_c5:
            best_c5 = f_mid
            best_alpha = mid
    return best_alpha, best_c5

def lp_optimize(h, target_sum, time_limit, initial_radius=0.3):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    radius = initial_radius
    stuck = 0
    stuck_limit = 40
    # Gradient verification
    corr = np.correlate(h, 1.0 - h, mode='full')
    grads = compute_gradients(h)
    
    while time_limit > 1.0:
        if stuck >= stuck_limit:
            if radius < 0.005:
                # Perturb to escape
                pert = np.random.default_rng(int(time.time()*1000) % 10000).normal(0, 0.03, n)
                h_new = project_feasible(h + pert, target_sum)
                c5_new = exact_c5(h_new)
                if c5_new < best_c5 + 0.01:
                    h = h_new
                radius = 0.05
                stuck = 0
            else:
                radius *= 0.5
                stuck = 0
                continue
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        grads = compute_gradients(h)
        p = solve_lp(h, corr, grads, radius, 0.0)
        
        if p is not None and np.max(np.abs(p)) > 1e-12:
            alpha, c5_new = line_search(h, p, best_c5)
            if alpha > 1e-10 and c5_new < best_c5 - 1e-14:
                h = h + alpha * p
                h = project_feasible(h, target_sum)
                c5_actual = exact_c5(h)
                if c5_actual <= best_c5 + 1e-12:
                    best_c5 = c5_actual
                    best_h = h.copy()
                    stuck = 0
                    radius = min(radius * 1.15, 0.8)
                else:
                    h = best_h.copy()
                    stuck += 2
                    radius *= 0.8
            else:
                stuck += 1
                radius *= 0.85
        else:
            stuck += 2
            radius *= 0.75
        
        # Periodically try complement
        if stuck % 10 == 5 and radius < 0.02:
            h_comp = 1.0 - h
            h_comp = project_feasible(h_comp, target_sum)
            c5_comp = exact_c5(h_comp)
            if c5_comp < best_c5:
                best_c5 = c5_comp
                best_h = h_comp.copy()
                h = h_comp.copy()
                radius = 0.2
                stuck = 0
        time_limit = max(0.0, time_limit - (time.time() - (time.time()-time_limit)))
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    old_x = (np.arange(n_old) + 0.5) / n_old
    new_x = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(new_x, old_x, h_old)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    # Initialize from parent
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None
    if h0 is None or len(h0) < 4:
        n = 63
        h0 = np.full(n, 0.5)
        h0[:n//2] = 1.0
        h0 = project_feasible(h0, n / 2.0)
    
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0, target_sum)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    def elapsed():
        return time.time() - t0
    
    def remaining(frac=1.0):
        return max(0.0, budget_s * frac - elapsed())
    
    # Phase 1: Main LP optimization (60% of budget)
    t_alloc = budget_s * 0.55
    best_h, best_c5 = lp_optimize(h, target_sum, t_alloc, initial_radius=0.3)
    
    # Phase 2: Try complement and perturbations (15% of budget)
    if remaining(0.85) > 3.0:
        # Complement
        h_comp = project_feasible(1.0 - best_h, target_sum)
        c5_comp = exact_c5(h_comp)
        
        # Try complement with small perturbations
        for trial in range(5):
            if remaining(0.85) < 2.0:
                break
            h_try = h_comp + rng.normal(0, 0.02, n)
            h_try = project_feasible(h_try, target_sum)
            h_try, c5_try = lp_optimize(h_try, target_sum, min(remaining(0.85)*0.15, 20.0), 0.2)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        
        # Random perturbations of best
        for trial in range(8):
            if remaining(0.85) < 2.0:
                break
            scale = [0.01, 0.02, 0.03, 0.05, 0.03, 0.02, 0.01, 0.08][trial]
            h_try = best_h + rng.normal(0, scale, n)
            h_try = project_feasible(h_try, target_sum)
            h_try, c5_try = lp_optimize(h_try, target_sum, min(remaining(0.85)*0.1, 15.0), 0.15)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Phase 3: Try different resolutions (15% of budget)
    if remaining(0.97) > 3.0:
        for dn in [-3, -2, -1, 1, 2, 3, 5]:
            if remaining(0.97) < 2.0:
                break
            n_new = n + dn
            if n_new < 20 or n_new > 100:
                continue
            h_interp = interpolate_h(best_h, n_new)
            h_interp = project_feasible(h_interp, n_new / 2.0)
            t_for_this = min(remaining(0.97) * 0.08, 15.0)
            h_opt, c5_opt = lp_optimize(h_interp, n_new / 2.0, t_for_this, 0.2)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                n = n_new
                target_sum = n / 2.0
    
    # Phase 4: Final polish with very small steps (10% of budget)
    if remaining(0.99) > 3.0:
        n = len(best_h)
        target_sum = n / 2.0
        best_h, best_c5 = lp_optimize(best_h, target_sum, remaining(0.99) * 0.5, 0.05)
    
    # Final: verify and return
    n = len(best_h)
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = exact_c5(best_h)
    
    return best_h, best_c5, n
