```python
def lp_step(h, delta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    scores = scores_exact(h)
    
    # Identify active lags
    if top_k is not None and top_k < len(scores):
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(len(scores))
    
    # Build linear coefficients for each active lag
    # For lag index k in full correlation, shift = k - (n-1)
    dh = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [sum(dh) == 0]  # Note: need to use cp.sum(dh)
    for k in idx:
        shift = k - (n - 1)
        # Compute linear coefficients
        coeffs = np.zeros(n)
        if shift >= 0:
            for i in range(n - shift):
                coeffs[i] += (1.0 - h[i + shift])
                coeffs[i + shift] -= h[i]
        else:
            s = -shift
            for i in range(s, n):
                coeffs[i] += (1.0 - h[i - s])
                coeffs[i - s] -= h[i]
        constraints.append(cp.sum(coeffs * dx * dh) <= t - scores[k])
    
    # Trust region
    constraints.append(dh >= -delta)
    constraints.append(dh <= delta)
    # Bounds
    constraints.append(dh >= -h)
    constraints.append(dh <= 1.0 - h)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return None
    
    if dh.value is None:
        return None
    
    return dh.value, t.value
```

```python
def lp_trust_region(h, delta, deadline, solver):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_scores = scores_exact(cur)
    cur_c = float(np.max(cur_scores))
    best = cur.copy()
    best_c = cur_c
    iters = 0
    
    while time.time() < deadline:
        # Build LP
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            shift = k - (n-1)
            if shift >= 0:
                for i in range(n - shift):
                    A[k, i] += (1.0 - cur[i+shift])
                    A[k, i+shift] -= cur[i]
            else:
                s = -shift
                for i in range(s, n):
                    A[k, i] += (1.0 - cur[i-s])
                    A[k, i-s] -= cur[i]
        
        dh = cp.Variable(n)
        t = cp.Variable()
        
        constraints = [
            cp.sum(dh) == 0,
            A @ (dx * dh) <= t - cur_scores,
            dh >= -delta,
            dh <= delta,
            dh >= -cur,
            dh <= 1.0 - cur,
        ]
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            delta *= 0.5
            if delta < 1e-8:
                break
            continue
        
        if dh.value is None:
            delta *= 0.5
            if delta < 1e-8:
                break
            continue
        
        # Check if the LP found a meaningful direction
        if np.max(np.abs(dh.value)) < 1e-10:
            delta *= 0.7
            if delta < 1e-8:
                break
            continue
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            cand = cur + alpha * dh.value
            cand = np.clip(cand, 0.0, 1.0)
            # Fix the sum constraint
            cand = make_feasible(cand, n)
            cand_scores = scores_exact(cand)
            cand_c = float(np.max(cand_scores))
            if cand_c < cur_c - 1e-12:
                cur = cand
                cur_scores = cand_scores
                cur_c = cand_c
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                improved = True
                delta = max(delta, min(0.1, cur_c))  # adaptively set delta
                break
        
        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                break
            # Try a random perturbation
            if iters % 50 == 0:
                cur = make_feasible(cur + np.random.default_rng(iters).normal(0, 0.01, n), n)
                cur_scores = scores_exact(cur)
                cur_c = float(np.max(cur_scores))
        else:
            delta = min(delta * 1.1, 0.3)
        
        iters += 1
    
    return best, best_c
```

3. Do exhaustive swap search (try all possible swaps)
4. Return the best binary solution

The exhaustive swap search is:
- Repeat until no improvement:
  - For each pair (i, j) where h[i]=1, h[j]=0:
    - Swap: h[i]=0, h[j]=1
    - If score improves, keep the swap; else revert
- Return the result

```python
            m = n - s
```

```python
from scipy.optimize import linprog

def lp_step_scipy(h, delta):
    n = len(h)
    dx = 2.0 / n
    cur_scores = scores_exact(h)
    A = build_A(h)
    Adx = A * dx  # (2n-1, n)
    
    # x = [dh (n), t (1)]
    # minimize t
    # subject to: Adx @ dh - t <= -cur_scores  (2n-1 constraints)
    #             sum(dh) = 0  (1 equality)
    #             lo <= dh <= hi  (bounds)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: Adx @ dh - t <= -cur_scores
    # [Adx | -1] @ [dh, t] <= -cur_scores
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = Adx
    A_ub[:, n] = -1.0
    b_ub = -cur_scores
    
    # Equality: sum(dh) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    lo = np.maximum(-delta, -h)
    hi = np.minimum(delta, 1.0 - h)
    bounds = list(zip(lo, hi)) + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, res.cost if hasattr(res, 'cost') else None
    return res.x[:n], res.x[-1]
```
