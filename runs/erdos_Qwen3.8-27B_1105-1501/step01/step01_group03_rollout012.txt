    for ki in top_lags:
        k = ki - (n - 1)
        g = compute_grad_for_lag(h, k, n)
        order = np.argsort(g)
        for lp in range(3):
            i_min = order[lp]
            i_max = order[-1 - lp]
            if i_min != i_max:
                v_pair = np.zeros(n)
                v_pair[i_min] = 1.0
                v_pair[i_max] = -1.0
                directions.append(v_pair)
    
    for _ in range(n_rand):
        r = rng.integers(3, min(12, n))
        idx = rng.choice(n, size=r, replace=False)
        v_r = np.zeros(n)
        v_r[idx] = rng.normal(0, 1, r)
        v_r -= np.mean(v_r)
        nn = np.linalg.norm(v_r)
        if nn > 1e-10:
            directions.append(v_r / nn)
    
    seen = set()
    for v in directions:
        key = tuple(np.round(v, 4))
        if key in seen:
            continue
        seen.add(key)
        L, Q = compute_LQ(h, v)
        lo, hi = get_interval(h, v)
        if hi - lo < 1e-14:
            continue
        s, Fmax_new = line_search_eval(F, L, Q, lo, hi, K=K_active)
        if abs(s) < 1e-14:
            continue
        h_new = make_feasible(h + s * v)
        score_new = official_score(h_new)
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(8.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve
    
    rng = np.random.default_rng(seed)
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if len(parent) < 2 or not np.all(np.isfinite(parent)):
            parent = None
    except Exception:
        parent = None
    
    if parent is not None:
        best_h = make_feasible(parent)
    else:
        best_h = make_feasible(np.full(61, 0.5))
    
    best_score = official_score(best_h)
    F = compute_F(best_h)
    Fmax = F.max()
    
    K_active = 25
    no_improve = 0
    
    while time.time() < deadline:
        F = compute_F(best_h)
        Fmax = F.max()
        
        h_cand, sc_cand = do_line_search_iteration(
            best_h, F, K_active, rng, n_pairs=6, n_rand=4)
        
        if sc_cand < best_score - 1e-15:
            best_h = h_cand
            best_score = sc_cand
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                sigma = 0.03 / (1.0 + 0.05 * no_improve)
                for attempt in range(3):
                    h_p = make_feasible(best_h + rng.normal(0, sigma, len(best_h)))
                    sc_p = official_score(h_p)
                    if sc_p < best_score:
                        best_h = h_p
                        best_score = sc_p
                        break
                no_improve = 0
    
    if time.time() < deadline - 5.0:
        for delta in [-4, 4, -6, 6]:
            if time.time() > deadline - 3.0:
                break
            n_try = len(best_h) + delta
            if n_try < 30 or n_try > 90:
                continue
            h_try = interp_h(best_h, n_try)
            for _ in range(15):
                if time.time() > deadline - 3.0:
                    break
                F_t = compute_F(h_try)
                h_try, sc_try = do_line_search_iteration(h_try, F_t, 20, rng, n_pairs=4, n_rand=3)
            sc_try = official_score(h_try)
            if sc_try < best_score - 1e-15:
                best_h = h_try
                best_score = sc_try
    
    if time.time() < deadline - 3.0:
        for _ in range(20):
            if time.time() > deadline - 2.0:
                break
            F = compute_F(best_h)
            h_c, sc_c = do_line_search_iteration(best_h, F, 30, rng, n_pairs=4, n_rand=2)
            if sc_c < best_score - 1e-15:
                best_h = h_c
                best_score = sc_c
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score >= 0.5:
        if parent is not None:
            best_h = np.asarray(parent, dtype=float).ravel()
            best_h = make_feasible(best_h)
            final_score = official_score(best_h)
        else:
            best_h = make_feasible(np.full(61, 0.5))
            final_score = official_score(best_h)
    
    return best_h, final_score, len(best_h)
