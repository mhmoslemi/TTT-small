def project_box_sum(x, target):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    c = (lo + hi) / 2.0
    x = np.clip(x - c, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - x)[::-1]
            for i in order:
                add = min(1.0 - x[i], residual)
                if add > 0:
                    x[i] += add
                    residual -= add
                if abs(residual) < 1e-10:
                    break
        else:
            order = np.argsort(x)
            for i in order:
                sub = min(x[i], -residual)
                if sub > 0:
                    x[i] -= sub
                    residual += sub
                if abs(residual) < 1e-10:

        b2 = min(n - 1, n - 1 - s)
        if a2 <= b2:
            grad[a2:b2 + 1] -= w * h[a2 + s:b2 + s + 1]
    return grad
