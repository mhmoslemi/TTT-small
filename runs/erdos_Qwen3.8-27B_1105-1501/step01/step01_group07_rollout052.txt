def gradient_matrix_vectorized(h, lags):
    n = len(h)
    K = len(lags)
    G = np.zeros((K, n))
    for idx, k in enumerate(lags):
        d = k - (n - 1)
        if d >= 0:
            # (1-h[m+d]) for m in [0, n-d-1]
            if n - d > 0:
                G[idx, :n-d] += (1.0 - h[d:])
            # -h[m-d] for m in [d, n-1]
            if n - d > 0:
                G[idx, d:] -= h[:n-d]
        else:
            dp = -d
            # (1-h[m-dp]) for m in [dp, n-1]
            if n - dp > 0:
                G[idx, dp:] += (1.0 - h[:n-dp])
            # -h[m+dp] for m in [0, n-dp-1]
            if n - dp > 0:
                G[idx, :n-dp] -= h[dp:]
    return G

            m = n - dp
            if m > 0:
                G[idx, dp:] += (1.0 - h[:m])
                G[idx, :m] -= h[dp:]
    return G

def active_set_lp(h, K, delta=0.0):
    n = len(h)
    dx = 2.0 / n
    overlaps = overlap_array(h)
    max_val = np.max(overlaps)
    if delta > 0:
        threshold = max_val - delta
        top_idx = np.where(overlaps >= threshold)[0]
    else:
        top_idx = np.argsort(overlaps)[-K:][::-1]
    top_idx = top_idx[np.argsort(overlaps[top_idx])[::-1]]
    K_eff = len(top_idx)
    if K_eff < 2:
        return None
    G = gradient_matrix_vec(h, top_idx)
    c_vals = overlaps[top_idx]
    Gh = G @ h
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K_eff, n + 1))
    A_ub[:, :n] = dx * G
    A_ub[:, n] = -1.0
    b_ub = -c_vals + dx * Gh
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    h_lp = np.clip(res.x[:n], 0.0, 1.0)
    dir_vec = h_lp - h
    nv = float(np.linalg.norm(dir_vec))
    if nv < 1e-11:
        return None
    cur_c = score_exact(h)
    best_h = h.copy()
    best_c = cur_c
    alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]
    for alpha in alphas:
        cand = h + alpha * dir_vec
        cand = make_feasible(cand, n)
        c_new = score_exact(cand)
        if c_new < best_c:
            best_c = c_new
            best_h = cand
    if best_c < cur_c - 1e-15:
        return best_h, best_c
    return None

def pair_search(h, i, j, cur_c, n_pts=30):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig_i
    h[j] = orig_j
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / max(n_pts - 1, 1)
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            h[i] = orig_i + (lo2 + hi2) * 0.5
            h[j] = orig_j - (lo2 + hi2) * 0.5
            final_s = score_exact(h)
            if final_s < cur_c - 1e-14:
                return float(final_s)
    h[i] = orig_i
    h[j] = orig_j
    return None

def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        lo2 = -orig + (0.0 - 0.0)
        span = (1.0) / max(n_pts - 1, 1)
        lo2 = max(-orig, best_a - 1.0 * span)
        hi2 = min(1.0 - orig, best_a + 1.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(25):
                h[i] = orig + a1
                s1 = score_exact(h)
                h[i] = orig + a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            h[i] = orig + (lo2 + hi2) * 0.5
            final_s = score_exact(h)
            if final_s < cur_c - 1e-14:
                return float(final_s)
    h[i] = orig
    return None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new <= 10 or n_new == n_old:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h_old, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return make_feasible(h_new, n_new)

def smooth_softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= float(w.sum())
    else:
        w = ex / S
    g = np.zeros(n)
    c = n - 1
    g += w[c] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[c + d] * dx
        wn = w[c - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g

def grad_descent_refine(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        g = smooth_softmax_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            stall += 1
            if stall > 30:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.85
            stall += 1
            if stall > 50:
                cur = make_feasible(cur + rng.normal(0, max(0.004, T * 0.2), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.004)
                stall = 0
        else:
            step = min(step * 1.03, T * 2.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)
    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: Active-set LP (35% time)
    p1_end = t0 + budget_s * 0.35
    cur = best_h.copy()
    cur_c = best_c
    stall = 0
    K_list = [3, 5, 8, 12, 16, 20, 25, 30]
    ki = 0
    while time.time() < p1_end:
        K = K_list[min(ki, len(K_list) - 1)]
        res = active_set_lp(cur, K)
        if res is not None:
            cur, cur_c = res
            stall = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
        else:
            stall += 1
            if stall > 30:
                if ki < len(K_list) - 1:
                    ki += 1
                else:
                    cur = make_feasible(cur + rng.normal(0, 0.015, n), n)
                    cur_c = score_exact(cur)
                stall = 0
        if time.time() < p1_end and stall % 10 == 0:
            h2, c2 = active_set_lp(cur, 40, delta=0.002)
            if h2 is not None and c2 < cur_c:
                cur, cur_c = h2, c2
                stall = 0
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c

    # Phase 2: Gradient descent refinement (15% time)
    p2_end = t0 + budget_s * 0.50
    if time.time() < p2_end:
        h2, c2 = grad_descent_refine(best_h, p2_end, 0.01, 0.0001, 0.999, 0.005, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2
        n = len(best_h)

    # Phase 3: Pair/coordinate search (30% time)
    p3_end = t0 + budget_s * 0.80
    cur = best_h.copy()
    cur_c = best_c
    consec_fail = 0
    while time.time() < p3_end:
        mode = rng.random()
        if mode < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            r = pair_search(cur, i, j, cur_c, n_pts=30)
        elif mode < 0.75:
            i = int(rng.integers(0, n))
            r = coord_search(cur, i, cur_c, n_pts=25)
        elif mode < 0.90:
            i = int(rng.integers(0, n - 1))
            r = pair_search(cur, i, i + 1, cur_c, n_pts=40)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            r = pair_search(cur, i, j, cur_c, n_pts=20)
        if r is not None:
            cur_c = r
            consec_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            for _ in range(3):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    break
                r2 = pair_search(cur, i2, j2, cur_c, n_pts=20)
                if r2 is not None:
                    cur_c = r2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            consec_fail += 1
            if consec_fail > 250:
                cur = make_feasible(best_h + rng.normal(0, 0.006, n), n)
                cur_c = score_exact(cur)
                consec_fail = 0

    # Phase 4: Final LP + polish (15% time)
    p4_end = deadline
    if time.time() < p4_end:
        cur = best_h.copy()
        cur_c = best_c
        stall = 0
        while time.time() < p4_end - 10:
            res = active_set_lp(cur, 35)
            if res is not None:
                cur, cur_c = res
                stall = 0
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c
            else:
                stall += 1
                if stall > 60:
                    break
        if time.time() < p4_end - 3:
            for i in range(n - 1):
                if time.time() > p4_end:
                    break
                r = pair_search(best_h, i, i + 1, best_c, n_pts=35)
                if r is not None:
                    best_h[i] += 0
                    best_c = r
        if time.time() < p4_end - 2:
            h_f, c_f = grad_descent_refine(best_h, p4_end, 0.002, 0.00003, 0.999, 0.001, rng)
            if c_f < best_c:
                best_h, best_c = h_f, c_f

    # Try alternative resolutions (if time remains)
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        for dn in [-6, -3, 3, 6, 9]:
            if time.time() > deadline - 5:
                break
            nr = n_cur + dn
            if nr < 20 or nr > 120:
                continue
            h_r = resample_h(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(deadline - 3, time.time() + 25)
            if time.time() < sub_dl:
                h_r, c_r = grad_descent_refine(h_r, sub_dl, 0.005, 0.0002, 0.999, 0.003, rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
