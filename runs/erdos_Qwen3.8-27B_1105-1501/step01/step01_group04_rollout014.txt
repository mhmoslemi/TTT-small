def resample_area(h_old, n_new):

    dx_old = 2.0 / n_old
    new_dx = 2.0 / n_new
    h_new = np.zeros(n_new)
    for i in range(n_new):
        x_left = i * new_dx
        x_right = (i+1) * new_dx
        # Find overlap with each old cell
        for j in range(n_old):
            old_left = j * dx_old
            old_right = (j+1) * dx_old
            overlap = max(0, min(x_right, old_right) - max(x_left, old_left))
            if overlap > 0:
                h_new[i] += h_old[j] * overlap
        h_new[i] /= new_dx  # Convert from integral to average value
    # Clip and project
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_simplex(h_new, float(n_new) / 2.0)
    return h_new

def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Exact fix
    r = target - np.sum(result)
    if r > 1e-14:
        # Add r to the coordinate with most slack
        idx = np.argmax(1.0 - result)
        result[idx] = min(1.0, result[idx] + r)
    elif r < -1e-14:
        # Subtract |r| from the coordinate with most mass
        idx = np.argmax(result)
        result[idx] = max(0.0, result[idx] + r)
    return result

def local_search_exact(h_init, time_budget, seed=42):
    n = len(h_init)
    target = float(n) / 2.0
    h = project_to_simplex(np.array(h_init, dtype=float), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    
    while time.time() - t0 < time_budget and stall < 200:
        any_improve = False
        
        # Try each pair (i,j) with i < j
        # Use random order for diversity
        order_i = rng.permutation(n)
        for ii in order_i:
            i = int(ii)
            if time.time() - t0 >= time_budget:
                break
            if h[i] < 1e-10:
                continue
            order_j = rng.permutation(n)
            for jj in order_j:
                j = int(jj)
                if j == i:
                    continue
                if h[j] >= 1.0 - 1e-10:
                    continue
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-10:
                    continue
                
                # Try full move and half move
                for frac in (1.0, 0.5):
                    d = max_d * frac
                    h[i] -= d
                    h[j] += d
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-15:
                        best_c5 = c5
                        best_h = h.copy()
                        any_improve = True
                        break
                    h[i] += d  # revert
                    h[j] -= d
        
        if any_improve:
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            amp = 0.02 + 0.01 * min(stall, 10)
            h = best_h + amp * rng.randn(n)
            h = project_to_simplex(h, target)
    
    return best_h, best_c5

def resample_preserving(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    new_dx = 2.0 / n_new
    h_new = np.zeros(n_new)
    for i in range(n_new):
        xl = i * new_dx
        xr = (i + 1) * new_dx
        j0 = max(0, int(xl // dx_old))
        j1 = min(n_old, int(np.ceil(xr // dx_old)))
        area = 0.0
        for j in range(j0, j1):
            ol = j * dx_old
            orr = (j + 1) * dx_old
            ov = min(xr, orr) - max(xl, ol)
            if ov > 0:
                area += h_old[j] * ov
        h_new[i] = area / new_dx
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_to_simplex(h_new, float(n_new) / 2.0)
