from scipy.optimize import minimize
import time

def score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asfarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Binary search for shift
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def antisym_project(h):

        h_new[i] = (h[i] + 1.0 - h[n - 1 - i]) / 2.0
        h_new[n - 1 - i] = 1.0 - h_new[i]

import numpy as np
from scipy.optimize import minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def antisym_project(h):
    n = len(h)
    h_new = np.zeros(n)
    for i in range((n + 1) // 2):
        h_new[i] = (h[i] + 1.0 - h[n - 1 - i]) / 2.0
        h_new[n - 1 - i] = 1.0 - h_new[i]
    return h_new

def coord_step(h, i, n_samples=100):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    p0 = n - 1
    old_hi = h[i]
    h[i] = 0.0
    base_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = old_hi
    A = np.zeros(n_lags)
    A[p0] += dx
    B = -dx
    for s in range(1, n - i):
        A[n - 1 + s] += dx * (1.0 - h[i + s])
    for s in range(1, i + 1):
        A[n - 1 + s] += -dx * h[i - s]
    for t in range(1, i + 1):
        A[n - 1 - t] += dx * (1.0 - h[i - t])
    for t in range(1, n - i):
        A[n - 1 - t] += -dx * h[i + t]
    xs = np.linspace(0.0, 1.0, n_samples)
    vals = base_vals[None, :] + A[None, :] * xs[:, None]
    vals[:, p0] += B * xs * xs
    f_vals = vals.max(axis=1)
    best_idx = f_vals.argmin()
    best_x = float(xs[best_idx])
    if best_idx > 0 and best_idx < n_samples - 1:
        lo_x, hi_x = float(xs[best_idx - 1]), float(xs[best_idx + 1])
    elif best_idx == 0:
        lo_x, hi_x = 0.0, float(xs[1])
    else:
        lo_x, hi_x = float(xs[-2]), 1.0
    xs2 = np.linspace(lo_x, hi_x, 50)
    vals2 = base_vals[None, :] + A[None, :] * xs2[:, None]
    vals2[:, p0] += B * xs2 * xs2
    f_vals2 = vals2.max(axis=1)
    best_x = float(xs2[f_vals2.argmin()])
    h[i] = best_x
    return h

def coord_descent(h, max_passes=80, tol=1e-12, time_limit=None, t_start=None, seed=42):
    n = len(h)
    h = h.copy()
    current = eval_score(h)
    rng = np.random.default_rng(seed)
    for p in range(max_passes):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        order = rng.permutation(n)
        improved = False
        for i in order:
            h = coord_step(h, i, n_samples=80)
            new_score = eval_score(h)
            if new_score < current - tol:
                current = new_score
                improved = True
        if not improved:
            break
    return h, current

def binary_greedy(h, max_passes=30, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy()
    current = eval_score(h)
    for p in range(max_passes):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        for i in range(n):
            for val in (0.0, 1.0):
                if abs(h[i] - val) < 1e-10:
                    continue
                old = h[i]
                h[i] = val
                sc = eval_score(h)
                if sc < current - 1e-12:
                    current = sc
                    improved = True
                else:
                    h[i] = old
        if not improved:
            break
    return h, current

def pair_optimize(h, i, j, n_grid=20):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    p0 = n - 1
    old_hi, old_hj = h[i], h[j]
    h[i] = 0.0
    h[j] = 0.0
    base_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = old_hi
    h[j] = old_hj
    A_i = np.zeros(n_lags)
    A_j = np.zeros(n_lags)
    A_ij = np.zeros(n_lags)
    B_ii = 0.0
    B_jj = 0.0
    B_ij = 0.0
    # s=0: x(1-x) + y(1-y) = x - x^2 + y - y^2
    A_i[p0] += dx
    A_j[p0] += dx
    B_ii = -dx
    B_jj = -dx
    for s in range(1, n - i):
        A_i[n - 1 + s] += dx * (1.0 - h[i + s])
    for s in range(1, i + 1):
        A_i[n - 1 + s] += -dx * (h[i - s] if (i - s) != j else 1.0)
    for t in range(1, i + 1):
        A_i[n - 1 - t] += dx * (1.0 - h[i - t] if (i - t) != j else 1.0)
    for t in range(1, n - i):
        A_i[n - 1 - t] += -dx * (h[i + t] if (i + t) != j else 0.0)
    for s in range(1, n - j):
        A_j[n - 1 + s] += dx * (1.0 - h[j + s] if (j + s) != i else 1.0)
    for s in range(1, j + 1):
        A_j[n - 1 + s] += -dx * (h[j - s] if (j - s) != i else 0.0)
    for t in range(1, j + 1):
        A_j[n - 1 - t] += dx * (1.0 - h[j - t])
    for t in range(1, n - j):
        A_j[n - 1 - t] += -dx * (h[j + t])
    # Bilinear terms: where h[i] and h[j] both appear in same lag
    # s>0: if |i-j|==s, then term h[i]*(1-h[j]) or h[j]*(1-h[i]) appears

        A_i[n - 1 + d] += 0.0  # h[i]*(1-h[j]): the (1-h[j]) part
        # Actually: the term is h[i]*(1-h[j]) = h[i] - h[i]*h[j]

        # h[i]*(1-h[j]) = h[i] - h[i]*h[j]
        # = A_i_part * x + A_ij_part * x * y
        # A_i_part = dx (from the h[i] part)
        # A_ij_part = -dx (from the -h[i]*h[j] part)
        # But I already added dx*(1-h[i+s]) to A_i for s=d, which is dx*(1-0)=dx since h[j]=0 was used
        # Actually no - I computed A_i with h at original values (after restoring). Let me redo this more carefully.
    # This is getting complex. Let me use a simpler approach: direct evaluation on grid.
    h[i] = old_hi
    h[j] = old_hj
    # Direct 2D grid search
    xs = np.linspace(0, 1, n_grid)
    ys = np.linspace(0, 1, n_grid)
    best_score = eval_score(h)
    best_x, best_y = old_hi, old_hj
    for x in xs:
        h[i] = x
        for y in ys:
            h[j] = y
            sc = eval_score(h)
            if sc < best_score - 1e-13:
                best_score = sc
                best_x, best_y = x, y
    h[i] = best_x
    h[j] = best_y
    return h, best_score

def pair_coord_descent(h, max_passes=10, time_limit=None, t_start=None, seed=7):
    n = len(h)
    h = h.copy()
    current = eval_score(h)
    rng = np.random.default_rng(seed)
    for p in range(max_passes):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        # Find active lags
        dx = 2.0 / n
        vals = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = vals.max()
        active = np.where(vals > cmax - 0.01)[0]
        # Collect active coordinates
        active_coords = set()
        for p_idx in active:
            s = p_idx - (n - 1)
            if s >= 0:
                for j in range(n - s):
                    active_coords.add(j)
                    active_coords.add(j + s)
            else:
                t = -s
                for j in range(n - t):
                    active_coords.add(j)
                    active_coords.add(j + t)
        active_list = list(active_coords)
        if len(active_list) < 2:
            break
        # Pick random pairs from active coords
        npairs = min(50, len(active_list) * (len(active_list) - 1) // 2)
        indices = rng.choice(len(active_list), size=(npairs, 2), replace=False)
        improved = False
        for pair_idx in indices:
            if time_limit is not None and t_start is not None:
                if time.time() - t_start > time_limit:
                    break
            i = active_list[pair_idx[0]]
            j = active_list[pair_idx[1]]
            if i == j:
                continue
            h, sc = pair_optimize(h, i, j, n_grid=12)
            if sc < current - 1e-12:
                current = sc
                improved = True
        if not improved:
            # Fallback: 1D coordinate descent
            h, current = coord_descent(h, max_passes=10, time_limit=time_limit, t_start=t_start)
            break
    return h, current

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_wave(n, freq1, freq2, amp1, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def make_blocks(n, n_blocks, block_vals):
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        h[start:end] = block_vals[b % len(block_vals)]
    return make_feasible(h)

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    w /= w.sum()
    # Gradient of V[p] with respect to x[j]
    n_lags = 2 * n - 1
    g = np.zeros(n)
    for p_idx in range(n_lags):
        s = p_idx - (n - 1)
        if s >= 0:
            for j in range(n - s):
                # dV[p]/dx[j] = dx*(1-x[j+s]) from first factor
                g[j] += w[p_idx] * dx * (1.0 - x[j + s])
                if j + s < n:
                    # dV[p]/dx[j+s] = -dx*x[j] from (1-x[j+s])
                    g[j + s] += w[p_idx] * (-dx) * x[j]
        else:
            t = -s
            for j in range(n - t):
                # V = sum x[j+t]*(1-x[j])
                g[j + t] += w[p_idx] * dx * (1.0 - x[j])
                g[j] += w[p_idx] * (-dx) * x[j + t]
    return g

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def try_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = eval_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None
