    lo = delta_min
    hi = delta_max
    for _ in range(n_iter):
        mid = (lo + hi) / 2
        vals = corr_full + d * mid
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        if len(active) == 0:
            active = np.where(vals == max_val)[0]
        subgrad_min = np.min(d[active])
        subgrad_max = np.max(d[active])
        if subgrad_min <= 0 <= subgrad_max:
            mid is the answer (or very close)
            break
        elif subgrad_min > 0:
            # Function is increasing at mid, minimum is to the left
            hi = mid
        else:  # subgrad_max < 0
            # Function is decreasing at mid, minimum is to the right
            lo = mid
    return (lo + hi) / 2

    h[::2] = 1.0
    h = 0.5 + 0.25 * np.sin(2 * np.pi * 3 * x / 2.0) + 0.1 * np.cos(2 * np.pi * 7 * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_constraint(h, n / 2.0)
    return h


def do_full_sweep(h, corr_full, s_arr, n, n_grid):
    raw_c5 = np.max(corr_full)
    improved = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_min = max(-h[j], h[i] - 1.0)
            d_max = min(h[i], 1.0 - h[j])
            if d_max - d_min < 1e-12:
                continue
            d = g_vec(j, s_arr, h) - g_vec(i, s_arr, h)
            best_delta, best_val = line_search_1d(corr_full, d, d_min, d_max, n_grid)
            if best_val < raw_c5 - 1e-10:
                h[i] -= best_delta
                h[j] += best_delta
                raw_c5 = best_val
                corr_full = compute_corr_full(h)
                improved += 1
    return h, corr_full, raw_c5, improved


def do_random_sweep(h, corr_full, s_arr, n, rng, n_pairs, n_grid):
    raw_c5 = np.max(corr_full)
    improved = 0
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_min = max(-h[j], h[i] - 1.0)
        d_max = min(h[i], 1.0 - h[j])
        if d_max - d_min < 1e-12:
            continue
        d = g_vec(j, s_arr, h) - g_vec(i, s_arr, h)
        best_delta, best_val = line_search_1d(corr_full, d, d_min, d_max, n_grid)
        if best_val < raw_c5 - 1e-10:
            h[i] -= best_delta
            h[j] += best_delta
            raw_c5 = best_val
            corr_full = compute_corr_full(h)
            improved += 1
    return h, corr_full, raw_c5, improved
