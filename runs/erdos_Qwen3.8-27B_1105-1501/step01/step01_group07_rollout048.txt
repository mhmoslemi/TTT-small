import numpy as np
import time
from scipy.optimize import linprog

def make_feasible(y, n=None):
    """Project y onto {0<=h<=1, sum(h)=n/2} using bisection on threshold."""
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix small residual
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        # Distribute to entries with most room
        if diff > 0:
            rooms = 1.0 - z
            order = np.argsort(-rooms)
            for idx in order:
                add = min(diff, rooms[idx])
                z[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(z)
            for idx in order:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return np.clip(z, 0.0, 1.0)

def compute_scores(h):
    """Compute all shift scores S_k = correlate(h, 1-h) * dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def score_exact(h):
    return float(np.max(compute_scores(h)))

def compute_subgradient(h, scores, tol=1e-8):
    """Compute subgradient of max_k S_k at h.
    Returns (gradient_vector, active_set)."""
    n = len(h)
    dx = 2.0 / n
    max_s = float(np.max(scores))
    # Active set: all k where S_k >= max_s - tol
    active = np.where(scores >= max_s - tol)[0]
    
    # For each active shift, compute gradient
    # Shift index in correlate 'full' mode: index c corresponds to shift d = c - (n-1)
    # S_d = dx * sum_{valid i} h[i] * (1 - h[i+d])
    
    grad = np.zeros(n)
    for c_idx in active:
        d = c_idx - (n - 1)
        if d >= 0:
            # S_d = dx * sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
            # ∂/∂h[j]: 
            #   +dx*(1-h[j+d]) if j+d < n (i.e., j < n-d)
            #   -dx*h[j-d] if j >= d
            m = n - d  # number of terms
            # j appears as h[i] for i=j: valid if j < m, contribution dx*(1-h[j+d])
            if m > 0:
                grad[:m] += dx * (1.0 - h[d:])  # h[j+d] for j in [0, m-1] is h[d:m]
            # j appears as h[i+d] for i=j-d: valid if j-d >= 0 and j-d < m, i.e., j >= d and j < n
            if d > 0:
                grad[d:] -= dx * h[:m]  # h[j-d] for j in [d, n-1] is h[0:n-d-1] = h[:m]
        else:
            # d < 0, let dd = -d = |d|
            dd = -d
            # S_{-dd} = dx * sum_{i=dd}^{n-1} h[i] * (1-h[i-dd])
            m = n - dd  # number of terms
            # j appears as h[i] for i=j: valid if j >= dd, contribution dx*(1-h[j-dd])
            if m > 0:
                grad[dd:] += dx * (1.0 - h[:m])  # h[j-dd] for j in [dd, n-1] is h[0:n-dd-1]
            # j appears as h[i-dd] for i=j+dd: valid if j+dd < n and j+dd >= dd (always), i.e., j < n-dd
            if dd > 0:
                grad[:m] -= dx * h[dd:]  # h[j+dd] for j in [0, n-dd-1] is h[dd:]
    
    # Normalize: subgradient is average of active gradients
    # Actually for min of max, the subdifferential is conv of {∇S_k : k in A}
    # A natural choice is the average
    grad /= len(active)
    return grad, active

def lp_step(h, scores, grad_dict, active_idx, max_iterations=1):
    """Solve LP to find a descent direction.
    Returns delta (n-vector) and t value, or None if infeasible."""
    n = len(h)
    num_active = len(active_idx)
    if num_active == 0:
        return None
    if num_active > 30:
        # Too many constraints, use only the top ones
        order = np.argsort(-scores)
        active_idx = order[:30]
        grad_dict = {i: grad_dict[i] for i in active_idx if i in grad_dict}
        num_active = len(active_idx)
    
    # Variables: [delta_0, ..., delta_{n-1}, t] (n+1 vars)
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: grad_k · delta - t <= -S_k
    A_ub = np.zeros((num_active, n + 1))
    b_ub = np.zeros(num_active)
    for i, c_idx in enumerate(active_idx):
        A_ub[i, :n] = grad_dict[c_idx]
        A_ub[i, -1] = -1.0
        b_ub[i] = -scores[c_idx]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i] <= delta[i] <= 1-h[i], t >= 0
    bounds = []
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 10000})

            return None
        delta = res.x[:n].copy()

    except Exception:

def resample(h_old, n_new):

    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0

        base = float(np.max(scores))

        S = float(np.sum(ex))
        if S < 1e-300:
            mask = scores >= base - 1e-10

        else:
            w = ex / S
        g = np.zeros(n)
        c_idx = n - 1
        g += w[c_idx] * dx * (1.0 - 2.0 * cur)
        for d in range(1, n):
            m = n - d
            wp = w[c_idx + d] * dx
            wn = w[c_idx - d] * dx
            g[:m] += wn * (1.0 - cur[d:]) - wp * cur[d:]
            g[d:] += wp * (1.0 - cur[:m]) - wn * cur[:m]

            T = max(T_end, T * 0.5)

    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(35):
                h[i] = oi + a1
                h[j] = oj - a1
                s1 = score_exact(h)
                h[i] = oi + a2
                h[j] = oj - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
        best_a = (lo2 + hi2) * 0.5
        h[i] = oi + best_a
        h[j] = oj - best_a
        best_s = score_exact(h)
        if best_s >= cur_c:
            best_a = 0.0
    h[i] = oi
    h[j] = oj
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = oi + best_a
        h[j] = oj - best_a
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=30):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def systematic_pair_pass(cur, cur_c, n):
    """Do a full pass over adjacent pairs."""
    improved_any = False
    for i in range(n - 1):
        res = pair_search(cur, i, i + 1, cur_c, n_pts=25)
        if res is not None:
            cur_c = res
            improved_any = True
    return cur_c, improved_any

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: LP-based descent
    p1_end = t0 + budget_s * 0.28
    if time.time() < p1_end:
        h1, c1 = lp_descent(best_h, p1_end, tol=1e-5, max_steps=400)
        if c1 < best_c:
            best_h, best_c = h1, c1
            n = len(best_h)

    # Phase 2: Gradient descent
    p2_end = t0 + budget_s * 0.45
    if time.time() < p2_end:
        h2, c2 = grad_descent(best_h, p2_end, 0.08, 0.0005, 0.999, 0.02, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2
            n = len(best_h)

    # Phase 3: Intensive pair/coord search with chaining
    p3_end = t0 + budget_s * 0.72
    cur = best_h.copy()
    cur_c = best_c
    consec_fail = 0
    while time.time() < p3_end:
        mode = rng.random()
        if mode < 0.60:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=40)
        elif mode < 0.75:
            i = int(rng.integers(0, n))
            res = coord_search(cur, i, cur_c, n_pts=35)
        elif mode < 0.90:
            i = int(rng.integers(0, n - 1))
            res = pair_search(cur, i, i + 1, cur_c, n_pts=50)
        else:
            # Triple move: move mass from i to j and from j to k
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) < 3:
                continue
            oi, oj, ok = cur[i], cur[j], cur[k]
            best_s = cur_c
            best_env = (0.0, 0.0)
            for a in np.linspace(max(-oi, -1.0), min(1.0 - oi, 1.0), 8):
                for b in np.linspace(max(-oj, -1.0 - a), min(1.0 - oj, 1.0 - a), 8):
                    c = -(a + b)
                    if c < -ok - 1e-9 or c > 1.0 - ok + 1e-9:
                        continue
                    cur[i], cur[j], cur[k] = oi + a, oj + b, ok + c
                    s = score_exact(cur)
                    if s < best_s:
                        best_s = s
                        best_env = (a, b)
            cur[i], cur[j], cur[k] = oi, oj, ok
            if best_env != (0.0, 0.0) and best_s < cur_c - 1e-13:
                cur[i], cur[j], cur[k] = oi + best_env[0], oj + best_env[1], ok + best_env[0] + best_env[1]
                res = score_exact(cur)
            else:
                res = None
        if res is not None:
            cur_c = res
            consec_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            for _ in range(6):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = pair_search(cur, i2, j2, cur_c, n_pts=30)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            consec_fail += 1
            if consec_fail > 250:
                cur = make_feasible(best_h + rng.normal(0, 0.006, n), n)
                cur_c = score_exact(cur)
                consec_fail = 0

    # Phase 4: Second LP round
    p4_end = t0 + budget_s * 0.85
    if time.time() < p4_end:
        # Systematic adjacent pass first
        cur = best_h.copy()
        cur_c = best_c
        sub_dl = min(p4_end - 10, time.time() + 60)
        while time.time() < sub_dl:
            cur_c, imp = systematic_pair_pass(cur, cur_c, n)
            if not imp:
                break
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
        h4, c4 = lp_descent(best_h, p4_end, tol=1e-6, max_steps=300)
        if c4 < best_c:
            best_h, best_c = h4, c4
            n = len(best_h)

    # Phase 5: Try alternate resolutions
    p5_end = deadline - 10
    if time.time() < p5_end:
        n_cur = len(best_h)
        for dn in (-4, 4, -6, 6, -8, 8):
            if time.time() > p5_end - 15:
                break
            nr = n_cur + dn
            if nr < 30 or nr > 100:
                continue
            h_r = resample(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sd = min(p5_end - 5, time.time() + 40)
            if time.time() < sd:
                h_r, c_r = lp_descent(h_r, sd, tol=1e-5, max_steps=80)
            if time.time() < p5_end - 5:
                h_r, c_r = grad_descent(h_r, min(p5_end - 5, time.time() + 25),
                                        0.01, 0.0003, 0.999, 0.005, rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
                n = len(best_h)

    # Final polish
    if time.time() < deadline - 2:
        h_f, c_f = grad_descent(best_h, deadline - 2, 0.004, 0.0001, 0.999, 0.003, rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        # One more LP on very small scale
        if time.time() < deadline - 1:
            h_f2, c_f2 = lp_descent(best_h, deadline - 1, tol=1e-6, max_steps=50)
            if c_f2 < best_c:
                best_h, best_c = h_f2, c_f2

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
