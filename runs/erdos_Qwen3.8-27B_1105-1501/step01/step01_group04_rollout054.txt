    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        # Pick a random pair (i, j)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Find optimal delta
        # Compute s(k) for each k
        corr = np.correlate(h, 1.0-h, mode='full')

        # s(k) = g[i0+c] - g[j0+c] - h[i0-c] + h[j0-c]
        # where g = 1-h
        
        g = 1.0 - h
        c = np.arange(-(n-1), n)  # length 2n-1

        s[mask1] += g[i + c[mask1]]
        
        mask2 = (j + c >= 0) & (j + c < n)
        s[mask2] -= g[j + c[mask2]]
        
        mask3 = (i - c >= 0) & (i - c < n)
        s[mask3] -= h[i - c[mask3]]
        
        mask4 = (j - c >= 0) & (j - c < n)
        s[mask4] += h[j - c[mask4]]
        
        # f(delta) = max_k (corr[k] + delta * s[k]) * dx
        # Minimize over delta in [lo, hi]
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        
        # Convex 1D minimization: use the fact that the min is at a breakpoint or endpoint
        # Simple approach: ternary search (works for convex functions)
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max((corr + m1 * s)) * dx
            f2 = np.max((corr + m2 * s)) * dx
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        
        delta = (lo + hi) / 2.0
        
        # Apply
        h[i] += delta
        h[j] -= delta
        h = np.clip(h, 0, 1)
        
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_polytope(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target}"""
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    c = np.arange(-(n-1), n)
    
    s = np.zeros(2*n-1)
    mask1 = (i + c >= 0) & (i + c < n)
