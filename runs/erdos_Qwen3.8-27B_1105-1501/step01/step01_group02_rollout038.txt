```python
def compute_grad_matrix(h, n):
    """Compute the gradient matrix G where G[j, s_idx] = d f_s / d h[j]."""
    # s ranges from -(n-1) to (n-1), s_idx = s + (n-1)
    G = np.zeros((n, 2*n - 1))
    for s in range(-(n-1), n):
        s_idx = s + n - 1
        lo = max(0, s)
        hi = min(n-1, n-1+s)
        # First index range: j in [lo, hi], contribution: 1 - h[j-s]
        for j in range(lo, hi+1):
            G[j, s_idx] += (1.0 - h[j-s])
        # Second index range: j in [max(-s,0), min(n-1-s, n-1)], contribution: -h[j+s]
        lo2 = max(-s, 0)
        hi2 = min(n-1-s, n-1)
        for j in range(lo2, hi2+1):
            G[j, s_idx] -= h[j+s]
    return G
```

```python
def compute_overlaps(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h):
    return float(np.max(compute_overlaps(h)))


def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((n, 2 * n - 1))
    for s in range(-(n - 1), n):
        si = s + n - 1
        lo1, hi1 = max(0, s), min(n - 1, n - 1 + s)
        lo2, hi2 = max(-s, 0), min(n - 1 - s, n - 1)
        if lo1 <= hi1:
            G[lo1:hi1 + 1, si] += (1.0 - h[lo1 - s:hi1 - s + 1])
        if lo2 <= hi2:
            G[lo2:hi2 + 1, si] -= h[lo2 + s:hi2 + s + 1]
    return G


def solve_slp_direction(h, delta):
    n = len(h)
    nsh = 2 * n - 1
    G = compute_grad_matrix(h)
    f = compute_overlaps(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((nsh, n + 1))
    A_ub[:, :n] = G.T
    A_ub[:, n] = -1.0
    b_ub = -f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo_d = np.maximum(-h, -delta)
    hi_d = np.minimum(1.0 - h, delta)
    bounds = [(float(lo_d[i]), float(hi_d[i])) for i in range(n)] + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        d = res.x[:n].astype(float).copy()
        d -= d.mean()
        d = np.clip(d, lo_d, hi_d)
        d -= d.mean()
        d = np.clip(d, lo_d, hi_d)
        norm_d = np.max(np.abs(d))
        if norm_d < 1e-13:
            return None
        return d
    return None


def golden_line_search(h, d, n_iter=45):
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    dp = a + gr * (b - a)
    fc = compute_c5(h + c * d)
    fd = compute_c5(h + dp * d)
    for _ in range(n_iter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b, dp, fd = dp, c, fc
            c = b - gr * (b - a)
            fc = compute_c5(h + c * d)
        else:
            a, c, fc = c, dp, fd
            dp = a + gr * (b - a)
            fd = compute_c5(h + dp * d)
    best_a, best_v = 0.0, compute_c5(h)
    for al in (0.0, 1.0, 0.5 * (a + b), c, dp):
        v = compute_c5(h + al * d)
        if v < best_v:
            best_v, best_a = v, al
    return best_a, best_v
```
