def target_penalty_grad(h, tau, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * (dx * corr - tau)
    x = np.clip(x, -50, 50)
    sigma = 1.0 / (1.0 + np.exp(-x))
    
    # Now compute grad[i] = alpha * sum_k sigma[k] * d(corr[k])/d(h[i])
    # d(corr[k])/d(h[i]) where k = n-1+s:
    # = (1-h[i-s])*1[0≤i-s<n] - h[i+s]*1[0≤i+s<n]
    
    grad = np.zeros(n)
    # For efficiency, only iterate over k where sigma[k] is non-negligible
    for s in range(-(n-1), n):
        w = sigma[n-1+s]
        if w < 1e-10:
            continue
        # (1-h[i-s]) for valid i-s
        lo_s = max(0, s)  # i-s >= 0 => i >= s
        hi_s = min(n, n+s)  # i-s < n => i < n+s
        if lo_s < hi_s:
            grad[lo_s:hi_s] += w * (1.0 - h[np.arange(lo_s, hi_s) - s])
        
        # -h[i+s] for valid i+s
        lo_p = max(0, -s)  # i+s >= 0 => i >= -s
        hi_p = min(n, n-s)  # i+s < n => i < n-s
        if lo_p < hi_p:
            grad[lo_p:hi_p] -= w * h[np.arange(lo_p, hi_p) + s]
    
    return alpha * grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def target_penalty_grad(h, tau, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * (dx * corr - tau)
    x = np.clip(x, -50, 50)
    sigma = 1.0 / (1.0 + np.exp(-x))
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = sigma[n-1+s]
        if w < 1e-12:
            continue
        lo_s = max(0, s)
        hi_s = min(n, n + s)
        if lo_s < hi_s:
            idx = np.arange(lo_s, hi_s) - s
            grad[lo_s:hi_s] += w * (1.0 - h[idx])
        lo_p = max(0, -s)
        hi_p = min(n, n - s)
        if lo_p < hi_p:
            idx = np.arange(lo_p, hi_p) + s
            grad[lo_p:hi_p] -= w * h[idx]
    
    return alpha * grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    """Gradient of smoothed max of dx*corr, for the parent approach."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    # grad[i] = dx * sum_k p[k] * d(corr[k])/d(h[i])
    # d(corr[n-1+s])/d(h[i]) = (1-h[i-s])*1[valid] - h[i+s]*1[valid]
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = p[n-1+s]
        if w < 1e-14:
            continue
        # (1-h[i-s]) term
        lo = max(0, s)
        hi = min(n, n + s)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[np.arange(lo, hi) - s])
        # -h[i+s] term
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[np.arange(lo2, hi2) + s]
    return dx * grad

def target_grad(h, tau, alpha, dx):
    """Gradient of sum softplus(alpha*(dx*corr[k]-tau))."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * (dx * corr - tau)
    x = np.clip(x, -50, 50)
    sigma = 1.0 / (1.0 + np.exp(-x))
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = sigma[n-1+s]
        if w < 1e-12:
            continue
        lo = max(0, s)
        hi = min(n, n + s)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[np.arange(lo, hi) - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[np.arange(lo2, hi2) + s]
    return alpha * dx * grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = p[n - 1 + s]
        if w < 1e-14:
            continue
        lo = max(0, s)
        hi = min(n, n + s)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[np.arange(lo, hi) - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[np.arange(lo2, hi2) + s]
    return dx * grad


def target_grad(h, tau, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * (dx * corr - tau)
    x = np.clip(x, -50, 50)
    sigma = 1.0 / (1.0 + np.exp(-x))
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = sigma[n - 1 + s]
        if w < 1e-12:
            continue
        lo = max(0, s)
        hi = min(n, n + s)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[np.arange(lo, hi) - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[np.arange(lo2, hi2) + s]
    return alpha * dx * grad


def optimize_smoothed(h0, time_budget, seed=42):
    n = len(h0)
    target = n / 2.0
    h = project_to_simplex(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    alpha = 15.0
    lr = 0.04
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    while time.time() - t0 < time_budget:
        for _ in range(80):
            if time.time() - t0 >= time_budget:
                return best_h, best_c5
            grad = smooth_max_grad(h, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            v = momentum * v - lr * (grad / gnorm)
            h_new = project_to_simplex(h + v, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            h = h_new
        if stall > 200:
            rng = np.random.RandomState(seed + int(time.time() * 1000) % 10000)
            h = best_h + 0.02 * rng.randn(n)
            h = project_to_simplex(h, target)
            v = np.zeros(n)
            stall = 0
            lr = max(lr * 0.75, 0.001)
        alpha = alpha * 1.04 + 0.3
        lr = max(lr * 0.996, 0.0005)
        momentum = min(momentum + 0.0005, 0.99)
    return best_h, best_c5


def target_feasibility_opt(h0, tau_start, tau_end, n_steps, alpha, lr, time_budget, seed=42):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = project_to_simplex(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    tau_seq = np.linspace(tau_start, tau_end, n_steps)
    momentum = 0.82
    v = np.zeros(n)
    for tau in tau_seq:
        if time.time() - t0 >= time_budget:
            break
        lr_local = lr
        for it in range(500):
            if time.time() - t0 >= time_budget:
                break
            grad = target_grad(h, tau, alpha, dx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            v = momentum * v - lr_local * (grad / gnorm)
            h_new = project_to_simplex(h + v, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
            lr_local *= 0.999
        v = np.zeros(n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def pair_refine(h0, time_budget, seed=42):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    no_improve = 0
    while time.time() - t0 < time_budget:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_k = np.argmax(corr)
        s = max_k - (n - 1)
        candidates = []
        if s >= 0:
            for i in range(n - s):
                if best_h[i] > 0.15 and best_h[i + s] < 0.85:
                    candidates.append((i, i + s))
        else:
            sa = -s
            for i in range(n - sa):
                if best_h[i + sa] > 0.15 and best_h[i] < 0.85:
                    candidates.append((i, i + sa))
        if not candidates:
            for _ in range(30):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j and 0.01 < best_h[i] < 0.99:
                    candidates.append((i, j))
        rng.shuffle(candidates)
        improved = False
        for i, j in candidates[:20]:
            if time.time() - t0 >= time_budget:
                break
            if i == j:
                continue
            c = best_h[i] + best_h[j]
            lo_v = max(0.0, c - 1.0)
            hi_v = min(1.0, c)
            if hi_v - lo_v < 1e-7:
                continue
            n_splits = 8
            for k in range(n_splits):
                v = lo_v + (hi_v - lo_v) * k / (n_splits - 1)
                h_test = best_h.copy()
                h_test[i] = v
                h_test[j] = c - v
                c5_t = compute_c5(h_test)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_test.copy()
                    improved = True
                    break
                v2 = lo_v + (hi_v - lo_v) * (k + 0.5) / (n_splits - 1)
                h_test2 = best_h.copy()
                h_test2[i] = v2
                h_test2[j] = c - v2
                c5_t2 = compute_c5(h_test2)
                if c5_t2 < best_c5 - 1e-13:
                    best_c5 = c5_t2
                    best_h = h_test2.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve % 5 == 0:
                h_test = best_h + 0.02 * rng.randn(n)
                h_test = project_to_simplex(h_test, target)
                c5_t = compute_c5(h_test)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_test.copy()
                    no_improve = 0
    return best_h, best_c5


def random_init(n, rng):
    h = np.zeros(n)
    ones = n // 2
    h[:ones] = 1.0
    if n % 2 == 1:
        h[ones] = 0.5
    idx = rng.permutation(n)
    return h[idx]


def structured_init(n, rng):
    h = np.zeros(n)
    center = n // 2
    h[center] = 1.0 if n % 2 == 0 else 0.5
    total = n / 2.0
    remaining = total - h.sum()
    spread = np.arange(1, (n // 2) + 1)
    for d in spread:
        if remaining <= 0:
            break
        pos1 = (center + d) % n
        pos2 = (center - d) % n
        for pos in [pos1, pos2]:
            if h[pos] < 1.0 - 1e-9 and remaining > 0:
                val = min(1.0, remaining)
                h[pos] = val
                remaining -= val
        if remaining <= 0:
            break
    frac = n % 2
    h = project_to_simplex(h, n / 2.0)
    idx = rng.permutation(n)
    return h[idx]


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    if h_init is not None:
        n0 = len(h_init)
        best_n = n0
        best_h = h_init.copy()
        best_c5 = compute_c5(best_h)

        # Phase 1: Short smoothed-max to settle
        alloc1 = min(remaining() * 0.12, 100.0)
        if alloc1 > 20:
            h1, c5_1 = optimize_smoothed(best_h, alloc1, seed)
            if c5_1 < best_c5:
                best_h, best_c5 = h1, c5_1

    # Phase 2: Target feasibility (main new method)
    if best_h is not None and remaining() > 30:
        n = best_n
        # Coarse tau sweep
        for alpha_try in [60, 120, 250]:
            if remaining() < 25:
                break
            tau_start = best_c5 - 0.0002
            tau_end = best_c5 - 0.0015
            alloc = min(remaining() * 0.15, 150.0)
            if alloc < 15:
                break
            h2, c5_2 = target_feasibility_opt(
                best_h, tau_start, tau_end, 6, alpha_try, 0.12, alloc, seed + alpha_try
            )
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2

        # Fine tau sweep around new best
        if remaining() > 30:
            tau_start = best_c5 - 0.0001
            tau_end = best_c5 - 0.0008
            alloc = min(remaining() * 0.2, 180.0)
            h2, c5_2 = target_feasibility_opt(
                best_h, tau_start, tau_end, 8, 350, 0.09, alloc, seed + 500
            )
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2

    # Phase 3: Pair polishing
    if best_h is not None and remaining() > 25:
        alloc3 = min(remaining() * 0.2, 180.0)
        h3, c5_3 = pair_refine(best_h, alloc3, seed + 777)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Another round of target feasibility with higher alpha
    if best_h is not None and remaining() > 25:
        tau_start = best_c5 - 0.00005
        tau_end = max(tau_start - 0.0006, 0.365)
        alloc4 = min(remaining() * 0.25, 200.0)
        h4, c5_4 = target_feasibility_opt(
            best_h, tau_start, tau_end, 10, 500, 0.06, alloc4, seed + 999
        )
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Phase 5: Short smoothed max to fine-tune
    if best_h is not None and remaining() > 20:
        alloc5 = min(remaining() * 0.15, 100.0)
        h5, c5_5 = optimize_smoothed(best_h, alloc5, seed + 333)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    # Phase 6: Final pair polish
    if best_h is not None and remaining() > 15:
        h6, c5_6 = pair_refine(best_h, remaining() * 0.4, seed + 1234)
        if c5_6 < best_c5:
            best_h, best_c5 = h6, c5_6

    # Try a different n if time allows
    if remaining() > 60:
        n_try = 127
        h_try = random_init(n_try, rng)
        h_try = project_to_simplex(h_try, n_try / 2.0)
        alloc_try = min(remaining() * 0.3, 180.0)
        h_t, c5_t = optimize_smoothed(h_try, alloc_try * 0.5, seed + 55)
        if c5_t < best_c5 and remaining() > 20:
            h_t2, c5_t2 = pair_refine(h_t, remaining() * 0.3, seed + 66)
            if c5_t2 < best_c5:
                best_h, best_c5, best_n = h_t2, c5_t2, n_try

    # Fallback
    if best_h is None:
        n_fb = 99
        best_h = random_init(n_fb, rng)
        best_h = project_to_simplex(best_h, n_fb / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n_fb

    target = best_n / 2.0
    best_h = project_to_simplex(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
