```python
    # Simple approach: try all intersection points of pairs of lines,
    # plus the boundaries.
    candidates = [t_min, t_max]
    
    for i in range(n_lines):
        for j in range(i+1, n_lines):
            if abs(a[i] - a[j]) > 1e-15:
                t_cross = (b[j] - b[i]) / (a[i] - a[j])
                if t_min <= t_cross <= t_max:
                    candidates.append(t_cross)
    
    best_t = t_min
    best_val = np.inf
    for t in candidates:
        val = np.max(a * t + b)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t
```

This is O(n²) in the number of lines (which is 2n-1 ≈ 121), so about 7000 pair comparisons. That's fine.

```python
def solve_1d_max_of_lines(a, b, t_min=0.0, t_max=1.0):
    n_lines = len(a)
    # Compute all pairwise crossings
    # t_ij = (b[i] - b[j]) / (a[i] - a[j])
    # Vectorize: for each pair (i,j), compute crossing
    i_idx, j_idx = np.triu_indices(n_lines, k=1)
    da = a[i_idx] - a[j_idx]
    # Filter out pairs with nearly equal slopes
    mask = np.abs(da) > 1e-15
    if np.any(mask):
        t_cross = (b[i_idx[mask]] - b[j_idx[mask]]) / da[mask]
        t_cross = t_cross[(t_cross >= t_min) & (t_cross <= t_max)]
    else:
        t_cross = np.array([])
    
    candidates = np.concatenate([[t_min, t_max], t_cross])
    # Evaluate f at all candidates
    # f(t) = max_k (a[k]*t + b[k])
    # Vectorize: compute a*t + b for all (t, k) pairs
    # This is a matrix of size (len(candidates) × n_lines)
    vals = a[None, :] * candidates[:, None] + b[None, :]
    f_vals = vals.max(axis=1)
    best_idx = np.argmin(f_vals)
    return candidates[best_idx]
```

This is O(n_lines²) memory and compute, but vectorized. For n_lines=121: 121² = 14641 values, which is tiny. The pairwise computation is 7260 values. All very fast in numpy.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        order = np.arange(n)
        for i in order:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def pnorm_obj_grad(h, p, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full')
    C = F * dx
    C_safe = np.maximum(C, 1e-30)
    total = np.sum(C_safe ** p)
    obj = total ** (1.0 / p)
    total_pow = total ** (1.0 / p - 1.0)
    weight = total_pow * p * C_safe ** (p - 1.0)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weight[k_idx]
        if abs(w) < 1e-25:
            continue
        k = k_idx - (n - 1)
        start = max(0, -k)
        end = min(n, n - k)
        if start >= end:
            continue
        idx1 = np.arange(start, end)
        idx2 = idx1 + k
        grad[idx1] += w * (1.0 - h[idx2])
        grad[idx2] += w * (-h[idx1])
    grad *= dx
    if lam > 0:
        sum_viol = np.sum(h) - n / 2.0
        grad += 2.0 * lam * sum_viol
    return obj, grad

def solve_1d_max_of_lines(a, b, t_min=0.0, t_max=1.0):
    n_lines = len(a)
    i_idx, j_idx = np.triu_indices(n_lines, k=1)
    da = a[i_idx] - a[j_idx]
    mask = np.abs(da) > 1e-14
    if np.any(mask):
        t_cross = (b[i_idx[mask]] - b[j_idx[mask]]) / da[mask]
        t_cross = t_cross[(t_cross >= t_min - 1e-12) & (t_cross <= t_max + 1e-12)]
        t_cross = np.clip(t_cross, t_min, t_max)
    else:
        t_cross = np.array([])
    candidates = np.concatenate([[t_min, t_max], t_cross])
    vals = a[None, :] * candidates[:, None] + b[None, :]
    f_vals = vals.max(axis=1)
    best_idx = np.argmin(f_vals)
    return float(candidates[best_idx])

def lbgfs_optimize(h0, n, p, lam, maxiter=200, gtol=1e-10):
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        obj, _ = pnorm_obj_grad(x, p, lam)
        return obj
    
    def jac(x):
        _, g = pnorm_obj_grad(x, p, lam)
        return g
    
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0
