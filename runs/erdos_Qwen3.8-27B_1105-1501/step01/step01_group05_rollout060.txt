def lp_local_descent(h, max_iters=100, time_limit=30.0, seed=42):
    n = len(h)
    h = project_to_target(h, n).copy()
    score = official_score(h)
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        c = corr_scaled(h)
        theta = c.max()
        if abs(theta - score) > 1e-10:
            score = theta
        
        # Active set
        tol = max(0.005, 0.02 * theta)
        active = np.where(c >= theta - tol)[0]
        
        # Limit active set size for LP speed
        if len(active) > 40:
            # Keep the top 40
            idx = np.argsort(c[active])[-40:]
            active = active[idx]
        
        G = grad_matrix(h)
        
        # Build LP
        # Variables: p(n), q(n), rho(1) → 2n+1 vars
        nvar = 2 * n + 1
        rho_idx = 2 * n
        
        # Objective: minimize -rho
        c_obj = np.zeros(nvar)
        c_obj[rho_idx] = -1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        A_rows = []
        b_vals = []
        
        # Active constraints: G[r,:] @ (p-q) + rho <= 0
        for r in active:
            row = np.zeros(nvar)
            row[:n] = G[r, :]  # p part
            row[n:2*n] = -G[r, :]  # q part
            row[rho_idx] = 1.0
            A_rows.append(row)
            b_vals.append(0.0)
        
        # Inactive constraints: G[r,:] @ (p-q) <= margin_r
        inactive = np.setdiff1d(np.arange(2*n-1), active)
        for r in inactive:
            row = np.zeros(nvar)
            row[:n] = G[r, :]
            row[n:2*n] = -G[r, :]
            A_rows.append(row)
            b_vals.append(theta - c[r])
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        
        # Equality constraints: sum(p) - sum(q) = 0
        A_eq = np.zeros((1, nvar))
        A_eq[0, :n] = 1.0
        A_eq[0, n:2*n] = -1.0
        b_eq = np.zeros(1)
        
        # L1 constraint: sum(p) + sum(q) <= 1
        # Add to A_ub
        l1_row = np.zeros(nvar)
        l1_row[:n] = 1.0
        l1_row[n:2*n] = 1.0
        A_ub = np.vstack([A_ub, l1_row])
        b_ub = np.append(b_ub, 1.0)
        
        # Bounds
        bounds = [(0, None)] * n + [(0, None)] * n + [(-1.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if res is None or not res.success or res.x is None:
            # Try fallback: negative gradient of active set
            break
        
        d = res.x[:n] - res.x[n:2*n]
        rho_val = res.x[rho_idx]
        
        if rho_val <= 1e-10:
            # No improvement direction found, try a perturbation
            d = -np.mean(G[active], axis=0)
            d -= np.mean(d)
            d_norm = np.linalg.norm(d, 1)
            if d_norm < 1e-15:
                break
            d /= d_norm
        
        # Line search
        best_alpha = 0.0
        best_score = score
        # Compute max feasible alpha
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        max_alpha = 1.0
        if np.any(pos_mask):
            max_alpha = min(max_alpha, np.min(h[pos_mask] / d[pos_mask]))
        if np.any(neg_mask):
            max_alpha = min(max_alpha, np.min(-h[neg_mask] / d[neg_mask]))
        max_alpha = max(max_alpha, 0.01)
        
        for alpha in [0.1, 0.25, 0.5, 0.75, 1.0]:
            a = alpha * max_alpha
            if a <= 0:
                continue
            h_try = project_to_target(h + a * d, n)
            s_try = official_score(h_try)
            if s_try < best_score - 1e-12:
                best_score = s_try
                best_alpha = a
        
        if best_alpha > 0:
            h = project_to_target(h + best_alpha * d, n)
            score = best_score
        else:
            # Try random perturbation
            d_rand = rng.normal(0, 0.1, n)
            d_rand -= np.mean(d_rand)
            h_try = project_to_target(h + 0.1 * d_rand, n)
            s_try = official_score(h_try)
            if s_try < score - 1e-12:
                h = h_try
                score = s_try
            else:
                break
    
    return h, official_score(h)

def interpolate_to(parent, n_target, offset=0):
    m = len(parent)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    # For interpolation, we need sorted xt
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)

def gen_balanced_binary(n, rng):
    """Generate a random balanced binary sequence of length n (n must be even)."""
    if n % 2 != 0:
        n += 1
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return project_to_target(h, n)

def slsqp_polish(h, time_limit=20.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        G = grad_matrix(h_x)
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        j[:, :n] = -G
        return j
    
    def eq_fun(x):
        return np.array([np.sum(x[:n]) - target_sum])
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-4
    
    x0 = np.concatenate([h, [theta0]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    start = time.perf_counter()
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if time.perf_counter() - start > time_limit:
            return h, initial_score
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n):
    h = np.asarray(v, dtype=float).ravel()[:n].copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)

    vmin, vmax = h.min(), h.max()
    lo = max(-vmax, -1.0) - 0.05
    hi = min(1.0 - vmin, 1.0) + 0.05
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += min(residual, 1.0 - h[idx])
        else:
            idx = np.argmax(h)
            h[idx] -= min(-residual, h[idx])
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n - 1), n):
        r = s + n - 1
        if s >= 0:
            G[r, s:] += dx * (1.0 - h[:n - s])
            G[r, :n - s] += dx * (-h[s:])
        else:
            a = -s
            G[r, a:] += dx * (1.0 - h[:n - a])
            G[r, :n - a] += dx * (-h[a:])
    return G


def interpolate_to(parent, n_target, offset=0):
    m = len(parent)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)


def gen_balanced_binary(n, rng):
    if n % 2 != 0:
        n += 1
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return project_to_target(h, n)


def gen_step_pattern(n, rng, n_blocks):
    if n_blocks >= n:
        return project_to_target(rng.uniform(0, 1, n), n)
    sizes = np.full(n_blocks, n // n_blocks)
    sizes[:n % n_blocks] += 1
    rng.shuffle(sizes)
    vals = rng.uniform(0.05, 0.95, n_blocks)
    h = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        h[pos:pos + sizes[i]] = vals[i]
        pos += sizes[i]
    return project_to_target(h, n)


def lp_local_descent(h, max_iters=150, time_limit=30.0, seed=42):
    n = len(h)
    h = project_to_target(h, n).copy()
    score = official_score(h)
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    best_h = h.copy()
    best_score = score

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break

        c = corr_scaled(h)
        theta = c.max()

        tol = max(0.008, 0.03 * theta)
        active = np.where(c >= theta - tol)[0]
        if len(active) > 35:
            idx = np.argsort(c[active])[-35:]
            active = active[idx]

        G = grad_matrix(h)

        nvar = 2 * n + 1
        rho_idx = 2 * n

        c_obj = np.zeros(nvar)
        c_obj[rho_idx] = -1.0

        A_rows = []
        b_vals = []

        for r in active:
            row = np.zeros(nvar)
            row[:n] = G[r, :]
            row[n:2 * n] = -G[r, :]
            row[rho_idx] = 1.0
            A_rows.append(row)
            b_vals.append(0.0)

        inactive = np.setdiff1d(np.arange(2 * n - 1), active)
        for r in inactive:
            row = np.zeros(nvar)
            row[:n] = G[r, :]
            row[n:2 * n] = -G[r, :]
            A_rows.append(row)
            b_vals.append(theta - c[r])

        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)

        l1_row = np.zeros(nvar)
        l1_row[:n] = 1.0
        l1_row[n:2 * n] = 1.0
        A_ub = np.vstack([A_ub, l1_row])
        b_ub = np.append(b_ub, 1.0)

        A_eq = np.zeros((1, nvar))
        A_eq[0, :n] = 1.0
        A_eq[0, n:2 * n] = -1.0

        bounds = [(0, None)] * n + [(0, None)] * n + [(-1.0, 1.0)]

        d = None
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if res is not None and res.success and res.x is not None:
                d = res.x[:n] - res.x[n:2 * n]
                rho_val = res.x[rho_idx]
                if rho_val <= 1e-12:
                    d = None
        except Exception:
            d = None

        if d is None:
            if len(active) > 0:
                d = -np.mean(G[active], axis=0)
            else:
                d = -G[0]
            d -= np.mean(d)
            dn = np.linalg.norm(d, 1)
            if dn < 1e-14:
                break
            d /= dn

        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        max_alpha = 1.0
        if np.any(pos_mask):
            max_alpha = min(max_alpha, 0.95 * np.min(h[pos_mask] / d[pos_mask]))
        if np.any(neg_mask):
            max_alpha = min(max_alpha, 0.95 * np.min(-h[neg_mask] / d[neg_mask]))
        max_alpha = max(max_alpha, 0.01)

        improved = False
        for alpha_frac in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]:
            a = alpha_frac * max_alpha
            if a <= 0:
                continue
            h_try = project_to_target(h + a * d, n)
            s_try = official_score(h_try)
            if s_try < score - 1e-13:
                h = h_try
                score = s_try
                improved = True
                break

        if not improved:
            for _ in range(20):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for delta in [0.05, 0.1, 0.15, 0.03]:
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    h_try = project_to_target(h_try, n)
                    s_try = official_score(h_try)
                    if s_try < score - 1e-12:
                        h = h_try
                        score = s_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

        if score < best_score:
            best_score = score
            best_h = h.copy()

    return best_h, best_score


def slsqp_polish(h, time_limit=25.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    initial_score = official_score(h)

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        G = grad_matrix(h_x)
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        j[:, :n] = -G
        return j

    def eq_fun(x):
        return np.array([np.sum(x[:n]) - target_sum])

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])

    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    start = time.perf_counter()
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 80, 'ftol': 1e-11}
        )
        if time.perf_counter() - start > time_limit:
            return h, initial_score
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass

    return h, initial_score


def binary_anneal(n, rng, time_limit=20.0):
    if n % 2 != 0:
        n += 1
    h = gen_balanced_binary(n, rng)
    score = official_score(h)
    start = time.perf_counter()
    T = 0.15

    while time.perf_counter() - start < time_limit:
        T *= 0.995
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or h[i] == h[j]:
            continue
        h_try = h.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        s_try = official_score(h_try)
        delta = s_try - score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            h = h_try
            score = s_try

    return project_to_target(h, n), official_score(h)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 5.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = project_to_target(parent, n_parent)
    else:
        n_parent = 64
        best_h = project_to_target(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = n_parent

    rng = np.random.default_rng(seed)

    candidates = [(best_score, best_h.copy())]

    for variant in [best_h[::-1].copy(), (1.0 - best_h).copy(),
                    (1.0 - best_h[::-1]).copy(), (best_h + best_h[::-1]) / 2.0]:
        v = project_to_target(variant, n_parent)
        s = official_score(v)
        if s < best_score:
            best_score = s
            best_h = v.copy()
            best_n = n_parent
        candidates.append((s, v))

    n_list = sorted(set([n_parent, max(32, n_parent - 10), n_parent + 8,
                         n_parent + 16, max(32, n_parent - 5), n_parent + 5,
                         max(32, n_parent - 16), n_parent + 12]))
    n_list = [n for n in n_list if 32 <= n <= 130]

    for n_try in n_list:
        if time.perf_counter() > deadline - 60:
            break
        for offset in [0, 1, 2, n_try // 4, n_try // 3]:
            h_interp = interpolate_to(best_h, n_try, offset)
            s = official_score(h_interp)
            candidates.append((s, h_interp))

    for n_try in n_list[:5]:
        if time.perf_counter() > deadline - 60:
            break
        for _ in range(3):
            h_rand = project_to_target(rng.uniform(0.15, 0.85, n_try), n_try)
            candidates.append((official_score(h_rand), h_rand))
        for _ in range(2):
            h_step = gen_step_pattern(n_try, rng, rng.integers(3, 8))
            candidates.append((official_score(h_step), h_step))
        if n_try % 2 == 0:
            for _ in range(2):
                h_bin = gen_balanced_binary(n_try, rng)
                candidates.append((official_score(h_bin), h_bin))
        for freq in [2, 3, 4, 5, 6]:
            t = np.linspace(0, 2, n_try)
            h_s = 0.5 + 0.25 * np.sin(np.pi * freq * t + rng.uniform(0, 2 * np.pi))
            h_s = project_to_target(h_s, n_try)
            candidates.append((official_score(h_s), h_s))

    candidates.sort(key=lambda x: (x[0], len(x[1])))
    if candidates[0][0] < best_score:
        best_score = candidates[0][0]
        best_h = candidates[0][1].copy()
        best_n = len(best_h)

    # Run LP descent on top candidates
    top_k = min(12, len(candidates))
    for idx in range(top_k):
        if time.perf_counter() > deadline - 100:
            break
        s_c, h_c = candidates[idx]
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        tlim = min(30.0, max(5.0, remaining * 0.25 / max(1, top_k - idx)))
        h_new, s_new = lp_local_descent(h_c, max_iters=120, time_limit=tlim, seed=seed + idx * 17)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()
            best_n = n_c
        candidates.append((s_new, h_new))

    candidates.sort(key=lambda x: (x[0], len(x[1])))

    # SLSQP polish on top results
    to_refine = candidates[:min(6, len(candidates))]
    for idx, (s_c, h_c) in enumerate(to_refine):
        if time.perf_counter() > deadline - 60:
            break
        n_c = len(h_c)
        tlim = min(30.0, max(5.0, (deadline - time.perf_counter() - 30) / max(1, len(to_refine) - idx)))
        h_ref, s_ref = slsqp_polish(h_c, time_limit=tlim)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
            best_n = n_c
        # LP after SLSQP
        if time.perf_counter() < deadline - 30:
            h_lp, s_lp = lp_local_descent(h_ref, max_iters=50, time_limit=10.0, seed=seed + 300 + idx)
            if s_lp < best_score:
                best_score = s_lp
                best_h = h_lp.copy()
                best_n = n_c

    # Binary annealing for even sizes
    if time.perf_counter() < deadline - 60:
        for n_bin in [n_parent if n_parent % 2 == 0 else n_parent + 1,
                      n_parent + 2 if (n_parent + 2) % 2 == 0 else n_parent + 3]:
            if time.perf_counter() > deadline - 30:
                break
            if n_bin < 32 or n_bin > 130:
                continue
            h_b, s_b = binary_anneal(n_bin, rng, time_limit=min(20.0, deadline - time.perf_counter() - 20))
            if s_b < best_score:
                best_score = s_b
                best_h = h_b.copy()
                best_n = n_bin
            # Polish binary result continuously
            if time.perf_counter() < deadline - 40:
                h_new, s_new = lp_local_descent(h_b, max_iters=60, time_limit=15.0, seed=seed + 600)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new.copy()
                    best_n = n_bin

    # Multi-resolution refinement with best
    if time.perf_counter() < deadline - 40:
        for n_try in [best_n, best_n + 4, best_n + 8, max(32, best_n - 4), best_n + 12]:
            if time.perf_counter() > deadline - 20:
                break
            if n_try < 32 or n_try > 130:
                continue
            for offset in [0, n_try // 5]:
                if time.perf_counter() > deadline - 15:
                    break
                h_interp = interpolate_to(best_h, n_try, offset)
                h_new, s_new = lp_local_descent(h_interp, max_iters=80, time_limit=
                                                min(15.0, deadline - time.perf_counter() - 10),
                                                  seed=seed + 800)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new.copy()
                    best_n = n_try
                if time.perf_counter() < deadline - 15:
                    h_pol, s_pol = slsqp_polish(h_new, time_limit=
                                                min(12.0, deadline - time.perf_counter() - 5))
                    if s_pol < best_score:
                        best_score = s_pol
                        best_h = h_pol.copy()
                        best_n = n_try

    # Final polish
    if time.perf_counter() < deadline - 10:
        h_final, s_final = slsqp_polish(best_h, time_limit=min(10.0, deadline - time.perf_counter() - 4))
        if s_final < best_score:
            best_score = s_final
            best_h = h_final.copy()
        if time.perf_counter() < deadline - 5:
            h_fin2, s_fin2 = lp_local_descent(best_h, max_iters=30, time_limit=
                                               min(5.0, deadline - time.perf_counter() - 2),
                                                seed=seed + 999)
            if
