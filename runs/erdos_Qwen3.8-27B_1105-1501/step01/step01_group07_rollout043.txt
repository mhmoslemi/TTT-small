In the continuous formulation, if h(x) = 1/2 + f(x) where f has zero mean, then:

def smooth_max_grad(h, T, c_vals):
    """Compute gradient of smooth max of c_vals w.r.t. h, using weights from softmax."""
    n = len(h)
    base = float(np.max(c_vals))
    ex = np.exp((c_vals - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        w = np.zeros(len(c_vals))
        mask = c_vals >= base - 1e-10
        w[mask] = 1.0
        w /= w.sum()
    else:
        w = ex / S
    
    d_vals = np.arange(-(n - 1), n)
    g = np.zeros(n)
    for idx, d in enumerate(d_vals):
        if abs(w[idx]) < 1e-10:
            continue
        coeff = compute_coeff(h, 0, np.array([0]))  # not used, we do it directly
        # Actually, gradient of smooth max = sum_k w_k * grad c_k
        # grad c_k w.r.t. h[j] = coeff(j, d_k)
        # This would be O(n * 2n) = O(n^2) per call which is fine for n=70
        pass
    
    # More efficient: use the correlation structure
    # g[j] = sum_k w_k * coeff(j, d_k)
    # coeff(j, d) for d >= 0: (1-h[j-d])*[j>=d] - h[j+d]*[j+d<n]
    # coeff(j, d) for d < 0, e=-d: (1-h[j+e])*[j<n-e] - h[j-e]*[j>=e]
    
    g = np.zeros(n)
    for j in range(n):
        for idx, d in enumerate(d_vals):
            if abs(w[idx]) < 1e-10:
                continue
            if d >= 0:
                c = 0.0
                if j >= d:
                    c += (1.0 - h[j - d])
                if j + d < n:
                    c -= h[j + d]
            else:
                e = -d
                c = 0.0
                if j < n - e:
                    c += (1.0 - h[j + e])
                if j >= e:
                    c -= h[j - e]
            g[j] += w[idx] * c
    return g, w

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    n = 70
    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(n, 0.5))
    best_c = score_exact(best_h)
    
    d_vals = np.arange(-(n - 1), n)
    K_active = 40
    
    # Get active set
    c_all = np.correlate(best_h, 1.0 - h_arr if False else best_h, mode='full')
    # Actually need c values for (h, 1-h) correlation
    c_all = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    def get_active(c_arr, K):
        return np.argsort(c_arr)[-K:]
    
    active_idx = get_active(c_all, K_active)
    active_d = d_vals[active_idx]
    active_c = c_all[active_idx].copy()
    
    # Phase 1: Fast pair optimization (70% of time)
    p1_end = t0 + budget_s * 0.70
    h_cur = best_h.copy()
    c
    _cur = best_c
    c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    stall = 0
    iter_count = 0
    
    while time.time() < p1_end:
        iter_count += 1
        
        # Pick pair
        mode = rng.random()
        if mode < 0.6:
            i0 = int(rng.integers(0, n))
            j0 = int(rng.integers(0, n))
        elif mode < 0.8:
            i0 = int(rng.integers(0, n - 1))
            j0 = i0 + 1
        else:
            i0 = int(rng.integers(0, n))
            j0 = int(rng.integers(0, n))
        
        if i0 == j0:
            continue
        
        orig_i = h_cur[i0]
        orig_j = h_cur[j0]
        t_lo = max(-orig_i, orig_j - 1.0)
        t_hi = min(1.0 - orig_i, orig_j)
        if t_hi - t_lo < 1e-12:
            continue
        
        # Compute A = coeff(i0, d) - coeff(j0, d) for active d's
        A_i = compute_coeff(h_cur, i0, active_d)
        A_j = compute_coeff(h_cur, j0, active_d)
        A = A_i - A_j
        
        # Solve 1D
        t_opt = solve_1d_max(active_c, A, t_lo, t_hi)
        
        if abs(t_opt) < 1e-12:
            stall += 1
            continue
        
        # Apply and evaluate
        h_cur[i0] = orig_i + t_opt
        h_cur[j0] = orig_j - t_opt
        
        # Update active c values
        active_c_new = active_c + t_opt * A
        
        # Also compute the true max (including non-active lags) for acceptance
        new_max = float(np.max(active_c_new))
        # Check if non-active lags might now exceed
        if iter_count % 50 == 0 or stall > 50:
            c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            true_max = float(np.max(c_all_cur))
            if true_max > new_max + 1e-8:
                # Re-identify active set
                active_idx = get_active(c_all_cur, K_active)
                active_d = d_vals[active_idx]
                active_c = c_all_cur[active_idx].copy()
                new_max = true_max
        
        if new_max < c_cur - 1e-12:
            c_cur = new_max
            stall = 0
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            # Revert
            h_cur[i0] = orig_i
            h_cur[j0] = orig_j
            stall += 1
        
        # Periodic active set update
        if iter_count % 200 == 0:
            c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            active_idx = get_active(c_all_cur, K_active)
            active_d = d_vals[active_idx]
            active_c = c_all_cur[active_idx].copy()
        
        # Stall escape: gradient step
        if stall > 200:
            T = max(0.005, 0.1 * np.exp(-stall / 100.0))
            g, _ = smooth_max_grad(h_cur, T, c_all_cur)
            gn = float(np.linalg.norm(g))
            if gn > 1e-14:
                step = 0.02
                for _ in range(10):
                    cand = make_feasible(h_cur - step * g / gn, n)
                    sc = score_exact(cand)
                    if sc < c_cur:
                        h_cur = cand
                        c_cur = sc
                        c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                        if c_cur < best_c:
                            best_c = c_cur
                            best_h = h_cur.copy()
                        break
                    step *= 0.5
            stall = 0
    
    # Phase 2: Systematic sweep (15% of time)
    p2_end = t0 + budget_s * 0.85
    h_cur = best_h.copy()
    c_cur = best_c
    c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    active_idx = get_active(c_all_cur, K_active)
    active_d = d_vals[active_idx]
    active_c = c_all_cur[active_idx].copy()
    sweep = 0
    
    while time.time() < p2_end:
        improved_sweep = False
        for i0 in range(n - 1):
            if time.time() > p2_end:
                break
            j0 = i0 + 1
            orig_i = h_cur[i0]
            orig_j = h_cur[j0]
            t_lo = max(-orig_i, orig_j - 1.0)
            t_hi = min(1.0 - orig_i, orig_j)
            if t_hi - t_lo < 1e-12:
                continue
            A_i = compute_coeff(h_cur, i0, active_d)
            A_j = compute_coeff(h_cur, j0, active_d)
            A = A_i - A_j
            t_opt = solve_1d_max(active_c, A, t_lo, t_hi)
            if abs(t_opt) > 1e-12:
                h_cur[i0] = orig_i + t_opt
                h_cur[j0] = orig_j - t_opt
                active_c = active_c + t_opt * A
                new_max = float(np.max(active_c))
                if new_max < c_cur - 1e-12:
                    c_cur = new_max
                    improved_sweep = True
                    if c_cur < best_c:
                        best_c = c_cur
                        best_h = h_cur.copy()
                else:
                    h_cur[i0] = orig_i
                    h_cur[j0] = orig_j
        # Update active set after sweep
        c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        active_idx = get_active(c_all_cur, K_active)
        active_d = d_vals[active_idx]
        active_c = c_all_cur[active_idx].copy()
        c_cur = float(np.max(c_all_cur))
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
        sweep += 1
        if not improved_sweep and sweep > 5:
            break
    
    # Phase 3: Fine gradient + pair (10% of time)
    p3_end = deadline
    h_cur = best_h.copy()
    c_cur = best_c
    c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    active_idx = get_active(c_all_cur, K_active)
    active_d = d_vals[active_idx]
    active_c = c_all_cur[active_idx].copy()
    
    while time.time() < p3_end:
        # Interleave gradient and pair
        if rng.random() < 0.3:
            T = 0.003
            g, _ = smooth_max_grad(h_cur, T, c_all_cur)
            gn = float(np.linalg.norm(g))
            if gn > 1e-14:
                for step in (0.01, 0.005, 0.002, 0.001):
                    cand = make_feasible(h_cur - step * g / gn, n)
                    sc = score_exact(cand)
                    if sc < c_cur:
                        h_cur = cand
                        c_cur = sc
                        c_all_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                        if c_cur < best_c:
                            best_c = c_cur
                            best_h = h_cur.copy()
                        break
        else:
            i0 = int(rng.integers(0, n))
            j0 = int(rng.integers(0, n))
            if i0 == j0:
                continue
            orig_i = h_cur[i0]
            orig_j = h_cur[j0]
            t_lo = max(-orig_i, orig_j - 1.0)
            t_hi = min(1.0 - orig_i, orig_j)
            if t_hi - t_lo < 1e-12:
                continue
            A_i = compute_coeff(h_cur, i0, active_d)
            A_j = compute_coeff(h_cur, j0, active_d)
            A = A_i - A_j
            t_opt = solve_1d_max(active_c, A, t_lo, t_hi)
            if abs(t_opt) > 1e-12:
                h_cur[i0] = orig_i + t_opt
                h_cur[j0] = orig_j - t_opt
                active_c = active_c + t_opt * A
                new_max = float(np.max(active_c))
                if new_max < c_cur - 1e-13:
                    c_cur = new_max
                    if c_cur < best_c:
                        best_c = c_cur
                        best_h = h_cur.copy()
                else:
                    h_cur[i0] = orig_i
                    h_cur[j0] = orig_j
    
    # Phase 4: Try other n values (last 5% of time)
    p4_end = deadline
    if time.time() < p4_end:
        for dn in [5, -5, 10, -10]:
            if time.time() > p4_end - 10:
                break
            nr = n + dn
            if nr < 20 or nr > 200:
                continue
            h_r = resample_h(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            # Quick pair optimization on the resampled
            sub_end = min(p4_end - 2, time.time() + 30)
            d_r = np.arange(-(nr - 1), nr)
            c_r_all = np.correlate(h_r, 1.0 - h_r, mode='full')
            act_r = np.argsort(c_r_all)[-30:]
            act_r_d = d_r[act_r]
            act_r_c = c_r_all[act_r].copy()
            while time.time() < sub_end:
                i0 = int(rng.integers(0, nr))
                j0 = int(rng.integers(0, nr))
                if i0 == j0:
                    continue
                oi = h_r[i0]
                oj = h_r[j0]
                tlo = max(-oi, oj - 1.0)
                thi = min(1.0 - oi, oj)
                if thi - tlo < 1e-12:
                    continue
                Ai = compute_coeff(h_r, i0, act_r_d)
                Aj = compute_coeff(h_r, j0, act_r_d)
                tt = solve_1d_max(act_r_c, Ai - Aj, tlo, thi)
                if abs(tt) > 1e-12:
                    h_r[i0] = oi + tt
                    h_r[j0] = oj - tt
                    act_r_c = act_r_c + tt * (Ai - Aj)
                    nm = float(np.max(act_r_c))
                    if nm < c_r - 1e-12:
                        c_r = nm
                        if c_r < best_c:
                            best_c = c_r
                            best_h = h_r.copy()
                            n = nr
                    else:
                        h_r[i0] = oi
                        h_r[j0] = oj
    
    # Final check
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
