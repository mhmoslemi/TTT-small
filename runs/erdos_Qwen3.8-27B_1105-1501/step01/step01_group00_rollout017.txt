    s.t. t - J_a @ d >= V_a for a in active

         sum(d) = 0
         -delta <= d[i] <= delta

    A_ub = np.hstack([G_a, -np.ones((len(active_indices), 1))])

    A_eq[0, :n] = 1.0

from scipy.optimize import minimize, linprog

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grads(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grads = np.zeros((L, n))
    for p in range(L):
        s = p - (n - 1)
        if s >= 0:
            Ls = n - s
            g = np.zeros(n)
            g[s:] += (1.0 - h[:Ls])
            g[:Ls] -= h[s:]
            grads[p] = g
        else:
            t = -s
            Lt = n - t
            g = np.zeros(n)
            g[:Lt] += (1.0 - h[t:])
            g[t:] -= h[:Lt]
            grads[p] = g
    return grads * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -3.0, 3.0)
    lo = -4.0
    hi = 4.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-13:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0:
                break
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-15:

    elif residual < -1e-13:

        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 0:
                break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-15:

    return np.clip(h, 0.0, 1.0)

    L = 2 * n - 1
    V = compute_values(h)
    G = compute_grads(h)
    sorted_idx = np.argsort(V)[::-1]
    active = sorted_idx[:min(top_k, L)]
    K = len(active)
    G_a = G[active]
    V_a = V[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G_a, -np.ones((K, 1))])
    b_ub = -V_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lb = np.maximum(-h, -delta)
    ub = np.minimum(1.0 - h, delta)
    bounds = list(zip(lb, ub)) + [(-1.0, 1.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    d = res.x[:n]
    if np.max(np.abs(d)) < 1e-12:
        return None, None
    return d, res.x[-1]

def smooth_grad_dir(h, beta):
    n = len(h)
    V = compute_values(h)
    G = compute_grads(h)
    cmax = V.max()
    shifted = beta * (V - cmax)
    shifted = np.clip(shifted, -500, 0)

    else:
        w /= ws
    g = w @ G
    gn = np.linalg.norm(g)
    if gn < 1e-14:

    return -g / gn

    return best_h, best_s

                        break

    # Phase 4: Explore different resolutions
    if not get_deadline():
        n_base = len(best_h)
        n_opts = sorted(set([max(30, n_base - 8), max(24, n_base - 12),
                              min(72, n_base + 8), min(64, n_base + 4)]))
        for n_opt in n_opts:
            if get_deadline():
                break
            if n_opt == n_base:
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-14:
                best_h = h_int.copy()
                best_score = sc_int
            # Optimize at this resolution
            cur_h = h_int
            cur_score = sc_int
            for _ in range(8):
                if get_deadline():
                    break
                d, _ = lp_direction(cur_h, 12, 0.06)
                if d is not None:
                    h_new, s_new = line_search(cur_h, d, cur_score, max_alpha=0.6, n_trials=10)
                    if h_new is not None and s_new < cur_score - 1e-14:
                        cur_h = h_new
                        cur_score = s_new
                        if cur_score < best_score - 1e-14:
                            best_h = cur_h.copy()
                            best_score = cur_score
                    else:
                        break
                else:
                    break

    # Phase 5: Random restarts with local optimization
    if not get_deadline():
        n = len(best_h)
        remaining = deadline - time.time()
        n_starts = max(2, min(12, int(remaining / 15)))
        for i in range(n_starts):
            if get_deadline():
                break
            sigma = [0.02, 0.04, 0.08, 0.12, 0.05, 0.03][i % 6]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, n))
            sc_cand = official_score(h_cand)
            cur_h = h_cand
            cur_score = sc_cand
            for _ in range(10):
                if get_deadline():
                    break
                d, _ = lp_direction(cur_h, 10, 0.05)
                if d is not None:
                    h_new, s_new = line_search(cur_h, d, cur_score, max_alpha=0.5, n_trials=8)
                    if h_new is not None and s_new < cur_score - 1e-14:
                        cur_h = h_new
                        cur_score = s_new
                        if cur_score < best_score - 1e-14:
                            best_h = cur_h.copy()
                            best_score = cur_score
                    else:
                        break
                else:
                    break

    # Phase 6: Anti-symmetric exploration
    if not get_deadline():
        h_sym = make_feasible(1.0 - best_h[::-1].copy())
        sc_sym = official_score(h_sym)
        if sc_sym < best_score - 1e-14:
            best_h = h_sym.copy()
            best_score = sc_sym
        else:
            # Blend
            for alpha in [0.3, 0.5, 0.7]:
                h_blend = make_feasible(alpha * best_h + (1 - alpha) * h_sym)
                sc_blend = official_score(h_blend)
                if sc_blend < best_score - 1e-14:
                    best_h = h_blend.copy()
                    best_score = sc_blend

    # Phase 7: Final intensive polish
    if not get_deadline():
        cur_h = best_h.copy()
        cur_score = best_score
        for beta in [100, 500, 2000, 8000, 20000]:
            if get_deadline():
                break
            h_new, sc = smooth_optimize(cur_h, beta, maxiter=60)
            if h_new is not None and sc < cur_score - 1e-14:
                cur_h = h_new
                cur_score = sc
                if sc < best_score - 1e-14:
                    best_h = cur_h.copy()
                    best_score = sc
        # Final LP polish
        for _ in range(20):
            if get_deadline():
                break
            d, _ = lp_direction(cur_h, 20, 0.02)
