def repair_exact(z):
    """Make z feasible: clip to [0,1] and adjust sum to n/2."""
    n = len(z)
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)

    lo, hi = -2.0, 2.0

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def pair_move_score(h, a, b, deltas):
    """Evaluate score for multiple delta values of pair move (a,b)."""
    h_work = h.copy()
    scores = np.empty(len(deltas))
    for k, d in enumerate(deltas):
        h_work[a] = h[a] + d
        h_work[b] = h[b] - d
        scores[k] = official_score(h_work)
    h_work[a] = h[a]
    h_work[b] = h[b]
    return scores

def pair_greedy_pass(h, n_samples=20, time_limit=None, t_start=None):
    """One full pass of pair greedy. Returns improved h and score."""
    n = len(h)
    cur_score = official_score(h)
    improved = True
    passes = 0
    max_passes = 50
    
    while improved and passes < max_passes:
        improved = False
        passes += 1
        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        
        # Try all pairs in random order? Or systematic?
        # Systematic for determinism, but we'll do random for diversity
        for a in range(n):
            for b in range(a+1, n):
                if time_limit is not None and time.time() - t_start > time_limit:
                    break
                # Feasible interval
                lo = max(-h[a], h[b] - 1.0)
                hi = min(1.0 - h[a], h[b])
                if hi - lo < 1e-12:

                # Sample deltas
                deltas = np.linspace(lo, hi, n_samples)
                scores = pair_move_score(h, a, b, deltas)
                best_k = np.argmin(scores)
                if scores[best_k] < cur_score - 1e-14:
                    d_best = deltas[best_k]
                    h[a] += d_best
                    h[b] -= d_best
                    cur_score = scores[best_k]
                    improved = True
    return h, cur_score

            h[b] -= d
            h[a] = np.clip(h[a], 0.0, 1.0)
            h[b] = np.clip(h[b], 0.0, 1.0)
            new_score = official_score(h)
            delta = new_score - cur_score
            if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_score = new_score
                if cur_score < best_local_score - 1e-14:
                    best_local_score = cur_score
                    best_local = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[a] -= d
                h[b] += d
        T *= 0.96
        if no_improve > max_no_improve:
            h = best_local.copy()
            cur_score = best_local_score
            no_improve = 0
            T = max(T, 0.005)
        if no_improve > 500:
            break
    return best_local, best_local_score

def try_direction_polish(h, corr, dx, time_limit, t_start):
    n = len(h)
    cur_score = official_score(h)
    while time.time() - t_start < time_limit:
        direction = gradient_direction(h, corr)
        if direction is None:
            break
        new_h, new_score, new_corr = direction_line_search(h, corr, dx, direction)
        if new_score < cur_score - 1e-14:
            h = new_h
            corr = new_corr
            cur_score = new_score
        else:
            desc = -direction
            new_h2, new_score2, new_corr2 = direction_line_search(h, corr, dx, desc)
            if new_score2 < cur_score - 1e-14:
                h = new_h2
                corr = new_corr2
                cur_score = new_score2
            else:
                break
    return h, cur_score, corr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')
    best_n = 0

    def update_global(h_cand, sc):
        nonlocal best_h, best_score, best_n
        if h_cand is None or not np.isfinite(sc):
            return
        if sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc
            best_n = len(h_cand)
        elif abs(sc - best_score) <= 1e-14:
            if best_h is None or len(h_cand) < best_n:
                best_h = h_cand.copy()
                best_score = sc
                best_n = len(h_cand)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = repair_exact(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                best_n = len(parent)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        best_h = gen_wave(n0, 3, 0.25, 0)
        best_score = official_score(best_h)
        best_n = n0

    n = len(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')

    # Phase 1: Intensive pair greedy
    if time.time() < deadline:
        t_limit = min(200.0, (deadline - time.time()) * 0.4)
        best_h, best_score = greedy_pair_pass(
            best_h, corr, dx, n_grid=30, max_passes=80, time_limit=t_limit, t_start=t0)

    # Phase 2: Direction polish
    if time.time() < deadline:
        t_limit = min(40.0, (deadline - time.time()) * 0.15)
        best_h, best_score, corr = try_direction_polish(
            best_h, np.correlate(best_h, 1.0 - best_h, mode='full'),
            2.0 / len(best_h), t_limit, time.time())

    # Phase 3: Anti-symmetric check
    h_as = anti_sym(best_h)
    sc_as = official_score(h_as)
    if sc_as < best_score - 1e-14:
        best_h = h_as
        best_score = sc_as
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        dx = 2.0 / len(best_h)
        if time.time() < deadline:
            t_limit = min(60.0, (deadline - time.time()) * 0.15)
            best_h, best_score = greedy_pair_pass(
                best_h, corr, dx, n_grid=30, max_passes=40, time_limit=t_limit, t_start=time.time())

    # Phase 4: SA exploration from best
    if time.time() < deadline:
        t_limit = min(100.0, (deadline - time.time()) * 0.2)
        sa_h, sa_score = sa_search(best_h.copy(), None, dx, t_limit, time.time(), seed + 1)
        if sa_score < best_score - 1e-14:
            best_h = sa_h
            best_score = sa_score
            if time.time() < deadline:
                n2 = len(best_h)
                dx2 = 2.0 / n2
                corr2 = np.correlate(best_h, 1.0 - best_h, mode='full')
                t_lim = min(60.0, (deadline - time.time()) * 0.15)
                best_h, best_score = greedy_pair_pass(
                    best_h, corr2, dx2, n_grid=30, max_passes=30, time_limit=t_lim, t_start=time.time())

    # Phase 5: Resolution sweep
    if time.time() < deadline:
        n_base = len(best_h)
        target_ns = sorted(set([
            max(20, n_base - 10),
            max(24, n_base - 5),
            n_base,
            min(90, n_base + 5),
            min(100, n_base + 10),
        ]))
        remaining = deadline - time.time()
        per_n = max(5.0, remaining / (len(target_ns) + 1))
        for n_t in target_ns:
            if time.time() > deadline - 5:
                break
            h_t = interp_to(best_h, n_t)
            sc_t = official_score(h_t)
            if sc_t < best_score - 1e-14:
                best_h = h_t
                best_score = sc_t
            dx_t = 2.0 / n_t
            corr_t = np.correlate(best_h, 1.0 - best_h, mode='full')
            t_lim = min(per_n, deadline - time.time() - 3)
            if t_lim > 3:
                best_h, best_score = greedy_pair_pass(
                    best_h, corr_t, dx_t, n_grid=25, max_passes=20,
                    time_limit=t_lim, t_start=time.time())
            if best_score < sc_t - 1e-14:
                pass  # already updated
            sc_check = official_score(best_h)
            if sc_check < best_score - 1e-14:
                best_score = sc_check

    # Phase 6: More structured starts + optimization
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_best = len(best_h)
        n_try = max(20, min(80, n_best))
        starts = []
        for freq in [2, 3, 4, 5, 6]:
            for amp in [0.15, 0.25, 0.35, 0.45]:
                starts.append(gen_wave(n_try, freq, amp, freq))
        starts.append(np.full(n_try, 0.5))
        n_starts = len(starts)
        per_start = max(3.0, remaining / (n_starts + 2))
        for i, h_s in enumerate(starts):
            if time.time() > deadline - 5:
                break
            sc_s = official_score(h_s)
            t_lim = min(per_start * 0.6, deadline - time.time() - 3)
            if t_lim > 2:
                dx_s = 2.0 / n_try
                corr_s = np.correlate(h_s, 1.0 - h_s, mode='full')
                h_opt, sc_opt = greedy_pair_pass(
                    h_s.copy(), corr_s, dx_s, n_grid=20, max_passes=15,
                    time_limit=t_lim, t_start=time.time())
                if sc_opt < best_score - 1e-14:
                    best_h = h_opt
                    best_score = sc_opt

    # Phase 7: SA from structured best with different seed
    if time.time() < deadline:
        t_limit = min(80.0, (deadline - time.time()) * 0.25)
        sa_h2, sa_score2 = sa_search(best_h.copy(), None, 2.0 / len(best_h),
                                     t_limit, time.time(), seed + 100)
        if sa_score2 < best_score - 1e-14:
            best_h = sa_h2
            best_score = sa_score2
            if time.time() < deadline:
                n3 = len(best_h)
                dx3 = 2.0 / n3
                corr3 = np.correlate(best_h, 1.0 - best_h, mode='full')
                t_lim = min(50.0, deadline - time.time() - 3)
                best_h, best_score = greedy_pair_pass(
                    best_h, corr3, dx3, n_grid=30, max_passes=30,
                    time_limit=t_lim, t_start=time.time())

    # Phase 8: Final intensive polish
    if time.time() < deadline:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        corr_f = np.correlate(best_h, 1.0 - best_h, mode='full')
        t_lim = deadline - time.time() - 3
        if t_lim > 5:
            best_h, best_score = greedy_pair_pass(
                best_h, corr_f, dx_f, n_grid=40, max_passes=100,
                time_limit=t_lim * 0.6, t_start=time.time())
        if time.time() < deadline - 3:
            nb = len(best_h)
            dx_b = 2.0 / nb
            corr_b = np.correlate(best_h, 1.0 - nb * 0 + best_h * 0 + (1.0 - best_h), mode='full')
            t_lim2 = min(20.0, deadline - time.time() - 2)
            if t_lim2 > 2:
                best_h, best_score, _ = try_direction_polish(
                    best_h, np.correlate(best_h, 1.0 - best_h, mode='full'),
                    2.0 / len(best_h), t_lim2, time.time())

    # Final validation
    final_h = repair_exact(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Verify feasibility
    assert len(final_h) == final_n
    assert np.all(final_h >= -1e-14) and np.all(final_h <= 1.0 + 1e-14)
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-10
    final_h = np.clip(final_h, 0.0, 1.0)
    final_score = official_score(final_h)

    return final_h, final_score + 1e-13, final_n
