Let me think about the epigraph formulation more carefully:

For n points, the correlation has 2n-1 values (lags -(n-1) to (n-1)).
- For lag k ≥ 0: F(k) = sum_{j=0}^{n-k-1} h[j] * (1-h[j+k])

For SLSQP:
- Variables: h[0..n-1], t (total n+1)
- Objective: t
- Constraints: F(k)*dx - t ≤ 0 for each k (2n-1 constraints)
- Equality: sum(h) = n/2
- Bounds: 0 ≤ h[i] ≤ 1, t ≥ 0

The gradient of F(k) with respect to h[i]:
For k ≥ 0, m = n-k:
- ∂F(k)/∂h[j] = (1-h[j+k]) for j = 0..m-1
- ∂F(k)/∂h[j+k] = -h[j] for j = 0..m-1

For k < 0, p = -k, m = n-p:
- ∂F(k)/∂h[j+p] = (1-h[j]) for j = 0..m-1

    n = len(h)
    dx = 2.0 / n
    F = compute_all_F(h) * dx
    # Get top K lags
    top_idx = np.argsort(F)[-K:]
    active_F = F[top_idx]
    base = active_F.max()
    # Soft-max weights
    T = 0.005
    w = np.exp((active_F - base) / T)
    w /= w.sum()
    # Gradient of weighted sum
    g = np.zeros(n)
    for idx in top_idx:
        k = idx - (n-1)  # convert to lag
        mk = n - abs(k)
        if k >= 0:
            g[:mk] += w[idx - top_idx[0]] * dx * (-(1.0 - h[k:]))
            g[k:] += w[idx - top_idx[0]] * dx * h[:mk]
        else:

            g[p:] += w[idx - top_idx[0]] * dx * (-(1.0 - h[:mk]))
            g[:mk] += w[idx - top_idx[0]] * dx * h[p:]
    # Project onto sum constraint: subtract mean
    g -= g.mean()
    # Normalize
    gn = np.linalg.norm(g)
    if gn < 1e-15:

    direction = -g / gn

    cur_c = score_exact(h)

        c = score_exact(cand)
        if c < best_c:
            best_c = c
            best_h = cand
            break

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()

            lo = mid
        else:
            hi = mid

def grad_descent_quick(h, deadline, T_start=0.01, T_end=0.0005, step0=0.01, rng=None):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    if rng is None:
        rng = np.random.default_rng(0)
    while time.time() < deadline:
        F = compute_all_F(cur) * dx
        base = F.max()
        ex = np.exp((F - base) / T)
        S = ex.sum()
        if S < 1e-300:
            mask = F >= base - 1e-10
            w = mask.astype(float)
            w /= w.sum()
        else:
            w = ex / S
        g = np.zeros(n)
        c = n - 1
        g += w[c] * dx * (1.0 - 2.0 * cur)
        for d in range(1, n):
            m = n - d
            wp = w[c + d] * dx
            wn = w[c - d] * dx
            g[:m] += wn * (1.0 - cur[d:]) - wp * cur[d:]
            g[d:] += wp * (1.0 - cur[:m]) - wn * cur[:m]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        direction = -g / gn
        improved = False

                break
        else:
