def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - np.sum(z)
    if abs(residual) < 1e-15:
        return z
    if residual > 0:
        order = np.argsort(1.0 - z)
        for idx in order:
            add = min(residual, 1.0 - z[idx])
            if add <= 0:
                break
            z[idx] += add
            residual -= add
            if abs(residual) < 1e-15:
                break
    else:
        order = np.argsort(z)
        for idx in order:
            sub = min(-residual, z[idx])
            if sub <= 0:
                break
            z[idx] -= sub
            residual += sub
            if abs(residual) < 1e-15:
                break
    return np.clip(z, 0.0, 1.0)

def two_coord_update(h, m, mp, n_grid=300):
    """
    Find the best t such that h[m] += t, h[mp] -= t minimizes the max overlap.
    Returns (new_h, score).
    """
    n = len(h)
    h_m = h[m]
    h_mp = h[mp]
    
    t_min = max(-h_m, h_mp - 1.0)
    t_max = min(1.0 - h_m, h_mp)
    
    if t_max - t_min < 1e-14:
        return h, official_score(h)
    
    # Base correlate
    base = np.correlate(h, 1.0 - h, mode='full')
    
    # Determine affected k values
    # m affects k in [m, m+n-1] ∩ [0, 2n-2]
    # mp affects k in [mp, mp+n-1] ∩ [0, 2n-2]
    k_lo_m = m
    k_hi_m = min(2*n-2, m+n-1)
    k_lo_mp = mp
    k_hi_mp = min(2*n-2, mp+n-1)
    
    aff_lo = min(k_lo_m, k_lo_mp)
    aff_hi = max(k_hi_m, k_hi_mp)
    
    # For each k in [aff_lo, aff_hi], compute a_k, b_k
    # g_k(t) = base[k] + a_k*t + b_k*t^2
    k_vals = np.arange(aff_lo, aff_hi + 1)
    a = np.zeros(len(k_vals))
    b = np.zeros(len(k_vals))
    
    for i, k in enumerate(k_vals):
        in_m = (k_lo_m <= k <= k_hi_m)
        in_mp = (k_lo_mp <= k <= k_hi_mp)
        
        if not in_m and not in_mp:
            continue
        
        if in_m and in_mp:
            # Both m and mp are in valid range
            if k == 2*m and k == 2*mp:
                # Impossible since m ≠ mp
                pass
            elif k == 2*m:
                # m is diagonal
                a[i] += (1.0 - 2.0*h_m)
                b[i] += -1.0
                # mp is non-diagonal (k-mp = 2m-mp ≠ m since m≠mp)
                kmp = k - mp
                a[i] += -(1.0 - 2.0*h[kmp])
            elif k == 2*mp:
                # mp is diagonal
                a[i] += -(1.0 - 2.0*h_mp)
                b[i] += -1.0
                # m is non-diagonal (k-m = 2mp-m ≠ mp since m≠mp)
                km = k - m
                a[i] += (1.0 - 2.0*h[km])
            elif k == m + mp:
                # Cross case
                a[i] += 2.0*(h_m - h_mp)
                b[i] += 2.0
            else:
                # Non-diagonal, non-cross
                km = k - m
                kmp = k - mp
                a[i] += (1.0 - 2.0*h[km])
                a[i] += -(1.0 - 2.0*h[kmp])
        elif in_m:
            if k == 2*m:
                a[i] += (1.0 - 2.0*h_m)
                b[i] += -1.0
            else:
                km = k - m
                a[i] += (1.0 - 2.0*h[km])
        else:  # in_mp only
            if k == 2*mp:
                a[i] += -(1.0 - 2.0*h_mp)
                b[i] += -1.0
            else:
                kmp = k - mp
                a[i] += -(1.0 - 2.0*h[kmp])
    
    # Non-affected max
    mask = np.ones(2*n-1, dtype=bool)
    mask[aff_lo:aff_hi+1] = False
    if mask.any():
        G_const = base[mask].max()
    else:
        G_const = -np.inf
    
    # Grid search
    ts = np.linspace(t_min, t_max, n_grid)
    aff_base = base[aff_lo:aff_hi+1]
    
    # Compute all values: shape (n_grid, n_aff)
    vals = aff_base[None, :] + a[None, :] * ts[:, None] + b[None, :] * ts[:, None]**2
    max_vals = np.max(vals, axis=1)
    final_vals = np.maximum(max_vals, G_const)
    
    best_idx = np.argmin(final_vals)
    best_t = ts[best_idx]
    best_val = final_vals[best_idx]
    
    # Refine with more grid points around best
    if 0 < best_idx < n_grid - 1:
        lo = ts[best_idx - 1]
        hi = ts[best_idx + 1]
        ts2 = np.linspace(lo, hi, 200)
        vals2 = aff_base[None, :] + a[None, :] * ts2[:, None] + b[None, :] * ts2[:, None]**2
        max_vals2 = np.max(vals2, axis=1)
        final_vals2 = np.maximum(max_vals2, G_const)
        best_idx2 = np.argmin(final_vals2)
        best_t = ts2[best_idx2]
        best_val = final_vals2[best_idx2]
    
    best_t = np.clip(best_t, t_min, t_max)
    
    h_new = h.copy()
    h_new[m] += best_t
    h_new[mp] -= best_t
    
    return h_new, best_val * (2.0/n)  # This is the score (times dx)

        def smooth_obj(x, beta):
            vals = np.correlate(x, 1.0-x, mode='full') * (2.0/len(x))
            cmax = vals.max()
            shifted = beta * (vals - cmax)
            return cmax + np.log(np.sum(np.exp(shifted))) / beta

                if sc < best_score:
                    best_h = h_new
                    best_score = sc

            pass
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    
    return final_h, final_score, final_n

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - np.sum(z)
    if abs(residual) < 1e-15:
        return z
    if residual > 0:
        order = np.argsort(1.0 - z)
        for idx in order:
            add = min(residual, 1.0 - z[idx])
            if add <= 0:
                break
            z[idx] += add
            residual -= add
            if abs(residual) < 1e-15:
                break
    else:
        order = np.argsort(z)
        for idx in order:
            sub = min(-residual, z[idx])
            if sub <= 0:
                break
            z[idx] -= sub
            residual += sub
            if abs(residual) < 1e-15:
                break
    return np.clip(z, 0.0, 1.0)

def two_coord_update(h, m, mp, n_grid=200):

    h_m = h[m]
    h_mp = h[mp]
    t_min = max(-h_m, h_mp - 1.0)
    t_max = min(1.0 - h_m, h_mp)

        return h, None
    base = np.correlate(h, 1.0 - h, mode='full')

    k_hi_mp = min(2 * n - 2, mp + n - 1)

    aff_hi = max(k_hi_m, k_hi_mp)

    a = np.zeros(n_aff)
    b = np.zeros(n_aff)
    for i in range(n_aff):
        k = aff_lo + i
        in_m = (k_lo_m <= k <= k_hi_m)
        in_mp = (k_lo_mp <= k <= k_hi_mp)
        if not in_m and not in_mp:

        if in_m and in_mp:

                a[i] += -(1.0 - 2.0 * h_mp)
                b[i] += -1.0
                km = k - m

                b[i] += 2.0
            else:
                km = k - m
                kmp = k - mp
                if 0 <= km <= n - 1:
                    a[i] += (1.0 - 2.0 * h[km])
                if 0 <= kmp <= n - 1:
                    a[i] += -(1.0 - 2.0 * h[kmp])
        elif in_m:
            if k == 2 * m:
                a[i] += (1.0 - 2.0 * h_m)
                b[i] += -1.0
            else:
                km = k - m
                if 0 <= km <= n - 1:

def steepest_descent_step(h, n_grid=200):
    n = len(h)
    best_improvement = -np.inf
    best_result = None
    current_score = official_score(h)
    for m in range(n):
        for mp in range(m + 1, n):
            h_new, val = two_coord_update(h, m, mp, n_grid=n_grid)
            if val is not None and val < current_score:
                improvement = current_score - val
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_result = (h_new, val)
    return best_result, best_improvement

def random_sweep(h, rng, n_grid=200, max_pairs=None):
    n = len(h)
    pairs = [(m, mp) for m in range(n) for mp in range(m + 1, n)]
    rng.shuffle(pairs)
    if max_pairs is not None:
        pairs = pairs[:max_pairs]
    for (m, mp) in pairs:
        h_new, val = two_coord_update(h, m, mp, n_grid=n_grid)
        if val is not None:
            cur_sc = official_score(h)
            if val < cur_sc - 1e-15:
                h = h_new
    return h

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def slsqp_polish(h0, beta=500.0, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None
