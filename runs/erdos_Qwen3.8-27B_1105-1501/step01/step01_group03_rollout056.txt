import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible_bin(z):
    """Round to binary with exact sum = n/2 (requires n even)."""
    n = len(z)
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5)
    z = np.clip(z, 0, 1)
    h = (z >= 0.5).astype(float)
    target = n / 2.0
    err = target - h.sum()
    if err > 0:
        idx = np.where(h == 0)[0]
        order = np.argsort(-z[idx])
        for j in range(int(err)):
            if j < len(order):
                h[idx[order[j]]] = 1.0
    elif err < 0:
        idx = np.where(h == 1)[0]
        order = np.argsort(z[idx])
        for j in range(int(-err)):
            if j < len(order):
                h[idx[order[j]]] = 0.0
    return h

def make_feasible_cont(z):
    """Project z to [0,1]^n with sum = n/2."""
    n = len(z)
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5)
    z = np.clip(z, 0, 1)
    target = n / 2.0
    # Bisection on additive shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(z + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    # Final correction
    err = target - h.sum()
    if abs(err) > 1e-10:
        if err > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            h[mask] = np.clip(h[mask] + err / mask.sum(), 0, 1)
    return h

    """Compute F[d] for d = -(n-1) to n-1 (indexed 0 to 2n-2)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # Length 2n-1, index d+n-1 corresponds to shift d

def binary_hill_climb(h, time_limit=50.0, t0=None):
    """All-swap hill climbing on binary sequence."""
    n = len(h)
    if n % 2 == 1:
        return h, official_score(h)
    dx = 2.0 / n
    h = h.copy()
    F = compute_F(h)
    best_score = np.max(F) * dx
    best_h = h.copy()
    
    if t0 is None:
        t0 = time.time()
    
    for sweep in range(500):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 0  # improvement in max F (positive = improvement)
        best_pair = None
        
        # For efficiency, compute effect of each position flip
        # effect[i, d_idx] = change in F[d_idx] when h[i] is flipped
        effects = np.zeros((n, 2*n-1))
        for i in range(n):
            for d in range(-(n-1), n):
                di = d + n - 1
                delta = 0.0
                # h[i] as first factor in F[d]: valid if 0 <= i+d < n
                if 0 <= i + d < n:
                    if h[i] == 1:
                        delta -= (1 - h[i+d])  # 1->0: lose 1*(1-h[i+d])
                    else:
                        delta += (1 - h[i+d])  # 0->1: gain 1*(1-h[i+d])
                # h[i] as second factor in F[d]: j+d=i, j=i-d, valid if 0<=i-d<n
                j = i - d
                if 0 <= j < n:
                    if h[i] == 1:
                        delta += h[j]  # 1->0: h[j]*(1-1)=0 -> h[j]*(1-0)=h[j]
                    else:
                        delta -= h[j]  # 0->1: h[j]*(1-0)=h[j] -> h[j]*(1-1)=0
                effects[i, di] = delta
        
        # For each (p, q) swap: new_F = F + effects[p] + effects[q]
        cur_max = np.max(F)
        for p in ones:
            fp = F + effects[p]
            for q in zeros:
                new_max = np.max(fp + effects[q])
                improve = cur_max - new_max
                if improve > best_improve + 1e-15:
                    best_improve = improve
                    best_pair = (p, q)
        
        if best_pair is not None:
            p, q = best_pair
            h[p], h[q] = h[q], h[p]
            F = compute_F(h)
            sc = np.max(F) * dx
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h.copy()
        else:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible_bin(z):
    n = len(z)
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5)
    z = np.clip(z, 0, 1)
    h = (z >= 0.5).astype(float)
    target = n / 2.0
    err = target - h.sum()
    if err > 0:
        idx = np.where(h == 0)[0]
        order = np.argsort(-z[idx])
        for j in range(min(int(err), len(order))):
            h[idx[order[j]]] = 1.0
    elif err < 0:
        idx = np.where(h == 1)[0]
        order = np.argsort(z[idx])
        for j in range(min(int(-err), len(order))):
            h[idx[order[j]]] = 0.0
    return h


def make_feasible_cont(z):
    n = len(z)
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5)
    z = np.clip(z, 0, 1)
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0, 1)
    err = target - h.sum()
    if abs(err) > 1e-10:
        if err > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            h[mask] = np.clip(h[mask] + err / mask.sum(), 0, 1)
    return h


def compute_F(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_lag_grad(h, d, n):
    grad = np.zeros(n)
    start = max(0, -d)
    end = min(n, n - d)
    if start < end:
        grad[start:end] += (1.0 - h[start + d:end + d])
    ps = max(0, start + d)
    pe = min(n, end + d)
    if ps < pe:
        grad[ps:pe] += (-h[ps - d:pe - d])
    return grad


def binary_hill_climb(h, time_limit=30.0, t0=None):
    n = len(h)
    if n % 2 == 1:
        return h.copy(), official_score(h)
    dx = 2.0 / n
    h = h.copy()
    F = compute_F(h)
    best_score = float(np.max(F) * dx)
    best_h = h.copy()
    if t0 is None:
        t0 = time.time()
    for _ in range(1000):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        effects = np.zeros((n, 2 * n - 1))
        for i in range(n):
            for d in range(-(n - 1), n):
                di = d + n - 1
                delta = 0.0
                if 0 <= i + d < n:
                    if h[i] == 1:
                        delta -= (1.0 - h[i + d])
                    else:
                        delta += (1.0 - h[i + d])
                j = i - d
                if 0 <= j < n:
                    if h[i] == 1:
                        delta += h[j]
                    else:
                        delta -= h[j]
                effects[i, di] = delta
        cur_max = np.max(F)
        best_improve = 0.0
        best_pair = None
        for p in ones:
            fp = F + effects[p]
            new_maxs = np.max(fp[None, :] + effects[zeros, :], axis=1)
            bi = int(np.argmin(new_maxs))
            improve = cur_max - new_maxs[bi]
            if improve > best_improve + 1e-15:
                best_improve = improve
                best_pair = (p, int(zeros[bi]))
        if best_pair is not None:
            p, q = best_pair
            h[p], h[q] = h[q], h[p]
            F = compute_F(h)
            sc = float(np.max(F) * dx)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h.copy()
        else:
            break
    return best_h, best_score


def binary_sa(h, time_limit=20.0, t0=None, T0=0.05, cooling=0.95):
    n = len(h)
    if n % 2 == 1:
        return h.copy(), official_score(h)
    dx = 2.0 / n
    h = h.copy()

    best_h = h.copy()
    if t0 is None:
        t0 = time.time()
    T = T0
    rng = np.random.default_rng(12345)
    for it in range(20000):

            break
        T *= cooling
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        q = int(rng.choice(zeros))
        old_max = float(np.max(F))
        h[p], h[q] = h[q], h[p]
        F_new = compute_F(h)
        new_max = float(np.max(F_new))
        delta = (new_max - old_max) * dx
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            F = F_new
            sc = new_max * dx
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h.copy()
        else:
            h[p], h[q] = h[q], h[p]
    return best_h, best_score


def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    F = compute_F(x)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax
    return cmax + (mk + np.log(ws)) / beta


def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    F = compute_F(x)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        w = weights[d + n - 1]
        if abs(w) < 1e-15:
            continue
        grad += w * compute_lag_grad(x, d, n)
    return grad * dx


def eq_sum(x):
    return float(np.sum(x) - len(x) / 2.0)


def eq_sum_jac(x):
    return np.ones(len(x))


def smooth_optimize(h0, beta, maxiter=200, ftol=1e-12):
    n = len(h0)
    h0 = make_feasible_cont(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible_cont(res.x)
    except Exception:
        pass
    return h0


def direct_ineq(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    F = compute_F(h)
    return z[-1] - dx * F


def direct_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n + 1))
    J[:, -1] = 1.0
    for d in range(-(n - 1), n):
        J[d + n - 1, :n] = -dx * compute_lag_grad(h, d, n)
    return J


def direct_eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n]) - n / 2.0)


def direct_eq_jac(z):
    n = len(z) - 1
    g = np.ones(n + 1)
    g[-1] = 0.0
    return g


def direct_obj(z):
    return float(z[-1])


def direct_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def direct_optimize(h0, maxiter=150, ftol=1e-12):

    h0 = make_feasible_cont(h0)
    score0 = official_score(h0)
    t0 = score0 + 1e-4
    z0 = np.concatenate([h0, [t0]])

        {'type': 'ineq', 'fun': direct_ineq, 'jac': direct_ineq_jac},

                       constraints=cons, method='SLSQP',

    except Exception:
        pass
    return h0


def interp_to(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible_cont(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible_cont(h_tgt)


def coord_descent(h, n_iter=50, t0=None, time_limit=30.0):
    """Coordinate descent exploiting linearity of F[k] in each variable."""
    n = len(h)
    h = make_feasible_cont(h.copy())
    if t0 is None:
        t0 = time.time()
    lam = 0.0
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            save = h[i]
            # Compute F values as function of h[i]
            # F[d] = const_d + b_d * h[i] for d != 0 (where i is involved)
            # F[0] has h[i] - h[i]^2
            # We'll just do a 1D scan
            best_t = np.inf
            best_val = save
            for trial in np.linspace(0, 1, 50):
                h[i] = trial
                # Enforce sum: adjust another variable
                err = n / 2.0 - h.sum()
                if abs(err) > 1e-10:
                    # Find a variable to adjust
                    if err > 0:
                        mask = np.arange(n) != i
                        mask_h = h[mask] < 1.0
                        if mask_h.sum() > 0:
                            idxs = np.where(mask)[0][mask_h]
                            h[i] = trial
                            err2 = n / 2.0 - h.sum()
                            h[idxs[0]] = np.clip(h[idxs[0]] + err2, 0, 1)
                    else:
                        mask = np.arange(n) != i
                        mask_h = h[mask] > 0.0
                        if mask_h.sum() > 0:
                            idxs = np.where(mask)[0][mask_h]
                            h[i] = trial
                            err2 = n / 2.0 - h.sum()
                            h[idxs[0]] = np.clip(h[idxs[0]] + err2, 0, 1)
                if h[i] != trial or (err2 if abs(err) > 1e-10 else 0) != 0:
                    # Skip complex cases, just evaluate
                    pass
                sc = official_score(h)
                if sc < best_t:
                    best_t = sc
                    best_val = h.copy()
            h = best_val.copy()
            if abs(h[i] - save) > 1e-8:
                improved = True
        if not improved and it > 5:
            break
