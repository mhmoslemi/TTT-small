        used.add(p)
        h[p] = 1.0
        count += 1
    if count < ones:
        zeros = np.where(h == 0)[0]
        for k in range(ones - count):
            if k < len(zeros):
                h[zeros[k]] = 1.0
    return h


def zc_init(n, rng):
    """Zadoff-Chu inspired initialization."""
    h_base = np.arange(n, dtype=float)
    phase = np.pi * h_base * (h_base + 1.0) / n
    h = 0.5 + 0.5 * np.cos(phase)
    h += 0.03 * rng.randn(n)
    return project_to_box_sum(h, float(n) // 2 + (0.5 if n % 2 else 0))


def swap_local_search(h0, time_budget, seed=42):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    step_sizes = [0.01, 0.02, 0.04, 0.08, 0.12, 0.2, 0.35, 0.5]

    while time.time() - t0 < time_budget:
        for _ in range(4000):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = step_sizes[rng.randint(len(step_sizes))]
            if rng.rand() < 0.5:
                d = -d
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i] = np.clip(ni, 0, 1)
            h[j] = np.clip(nj, 0, 1)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c5


def simulated_annealing_binary(h0, time_budget, seed=42):
    """SA on the continuous h, biased toward binary values."""
    n = len(h0)
    target = float(n) / 2.0
    h = project_to_box_sum(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    temp = 0.005
    cool = 0.997

    while time.time() - t0 < time_budget and temp > 1e-6:
        for _ in range(2000):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = 0.05 * rng.randn()
            ni = h[i] + d
            nj = h[j] - d
            if ni < -0.1 or ni > 1.1 or nj < -0.1 or nj > 1.1:
                continue
            h[i] = np.clip(ni, 0, 1)
            h[j] = np.clip(nj, 0, 1)
            # Renormalize
            h = project_to_box_sum(h, target)
            c5_new = compute_c5(h)
            delta = c5_new - best_c5
            if delta < 0 or rng.rand() < np.exp(-delta / temp):
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                h = best_h.copy()
        temp *= cool
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
        if len(h_init) == 0:
            h_init = None
    except (NameError, TypeError):
        h_init = None

    # Phase 1: Optimize initial construction
    if h_init is not None:
        n0 = len(h_init)
        best_c5 = compute_c5(h_init)
        best_h = h_init.copy()
        best_n = n0

        # 1a: Smoothed max - aggressive
        alloc = min(remaining() * 0.28, 280.0)
        if alloc > 30:
            h_new, c5_new = optimize_smoothed_max(best_h, alloc, seed=seed,
                                                   alpha0=15.0, lr0=0.014)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

        # 1b: Threshold feasibility push
        if remaining() > 80:
            tau = best_c5 - 0.0004
            alloc = min(remaining() * 0.18, 160.0)
            h_new, c5_new = optimize_threshold(best_h, tau, alloc,
                                                seed=seed + 100, alpha=30.0, lr=0.008)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

        # 1c: Simulated annealing for escape
        if remaining() > 60:
            alloc = min(remaining() * 0.12, 100.0)
            h_new, c5_new = simulated_annealing_binary(best_h, alloc, seed=seed + 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

        # 1d: Another smoothed max round
        if remaining() > 50:
            alloc = min(remaining() * 0.12, 100.0)
            h_new, c5_new = optimize_smoothed_max(best_h, alloc, seed=seed + 500,
                                                   alpha0=40.0, lr0=0.004)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 2: Try other n values
    if best_h is None:
        n = 99
        best_h = binary_spread_init(n, rng)
        best_c5 = compute_c5(best_h)
        best_n = n

    n_candidates = [63, 47, 81, 127, 161, 199]
    n_candidates = [x for x in n_candidates if x != best_n]

    for n in n_candidates:
        if remaining() < 45:
            break
        alloc_n = min(remaining() * 0.1, 70.0)
        if alloc_n < 15:
            break

        for init_i in range(2):
            if remaining() < 15:
                break
            if init_i == 0:
                h_new = binary_spread_init(n, np.random.RandomState(seed + n * 7))
                h_new = h_new + 0.01 * np.random.RandomState(seed + n * 11).randn(n)
                h_new = project_to_box_sum(h_new, float(n) / 2.0)
            else:
                h_new = zc_init(n, np.random.RandomState(seed + n * 13))

            sub_alloc = min(alloc_n / 2.0, remaining() - 10)
            if sub_alloc < 8:
                break
            h_opt, c5_opt = optimize_smoothed_max(h_new, sub_alloc,
                                                   seed=seed + n * 10 + init_i,
                                                   alpha0=12.0, lr0=0.01)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n

    # Phase 3: Local refinement
    if best_h is not None and remaining() > 30:
        alloc = min(remaining() * 0.25, 180.0)
        h_ref, c5_ref = swap_local_search(best_h, alloc, seed=seed + 77777)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Phase 4: Final threshold push
    if best_h is not None and remaining() > 30:
        tau = best_c5 - 0.00015
        alloc = min(remaining() * 0.2, 90.0)
        h_final, c5_final = optimize_threshold(best_h, tau, alloc,
                                                seed=seed + 99999, alpha=50.0, lr=0.003)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Phase 5: Final local refinement
    if best_h is not None and remaining() > 15:
        alloc = min(remaining() * 0.35, 80.0)
        h_ref, c5_ref = swap_local_search(best_h, alloc, seed=seed + 111111)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Ensure constraint
    target = float(best_n) / 2.0
    best_h = project_to_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
