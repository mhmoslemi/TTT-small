from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project_to_target(v, target):
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add more; add to entries closest to 1
            idx = np.argsort(-x)[:max(1, int(abs(residual)) + 1)]
            for i in idx:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            # Need to subtract; subtract from entries closest to 0
            idx = np.argsort(x)[:max(1, int(abs(residual)) + 1)]
            for i in idx:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return x

def compute_gradients(h, n):
    """Compute gradient of each lag score w.r.t. h.
    Returns array of shape (2n-1, n) where grads[r, i] = ds_r/dh_i.
    """
    grads = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        if k >= 0:
            # grad[i] = (1-h[i-k]) for i in [k, n-1]
            #          - h[i+k]   for i in [0, n-k-1]
            i1 = np.arange(k, n)
            grads[r, i1] += (1.0 - h[i1 - k])
            i2 = np.arange(0, n - k)
            grads[r, i2] -= h[i2 + k]
        else:
            g = -k
            # grad[i] = (1-h[i+g]) for i in [0, n-g-1]
            #          - h[i-g]   for i in [g, n-1]
            i1 = np.arange(0, n - g)
            grads[r, i1] += (1.0 - h[i1 + g])
            i2 = np.arange(g, n)
            grads[r, i2] -= h[i2 - g]
    return grads

def lp_direction(h, scores, grads, radius):
    """Find LP direction to reduce max score.
    Returns (p, predicted_tau) or (None, None) if no improvement.
    """
    n = len(h)
    L = len(scores)
    num_vars = n + 1  # p (n) + tau (1)
    
    # Objective: minimize tau
    c = np.zeros(num_vars)
    c[-1] = 1.0
    
    # Inequality constraints: scores[r] + grads[r] @ p <= tau
    # => grads[r] @ p - tau <= -scores[r]
    A_ub = np.zeros((L, num_vars))
    b_ub = np.zeros(L)
    for r in range(L):
        A_ub[r, :n] = grads[r]
        A_ub[r, -1] = -1.0
        b_ub[r] = -scores[r]
    
    # Equality: sum p = 0
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: p_i in [max(-radius, -h_i), min(radius, 1-h_i)], tau unbounded below (but >= min scores)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))  # tau free
    # Actually tau should be bounded: tau >= min(scores) - something
    # Let's keep it free or set a wide bound
    bounds[-1] = (min(scores) - radius * (n + 1), max(scores) + radius * (n + 1))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    p = res.x[:n]
    tau = res.x[-1]
    return p, tau

def line_search(h, p, current_c5, radius):
    """1D line search along h + alpha*p. Returns (best_alpha, best_h, best_c5)."""
    n = len(h)
    # Find alpha_max

    for i in range(n):

        elif p[i] < -1e-15:

    alpha_max = min(alpha_max, radius) * 0.999
    
    if alpha_max <= 0:
        return 0.0, h.copy(), current_c5
    
    # Grid search
    alphas = np.linspace(0, alpha_max, 30)
    best_alpha = 0.0
    best_c5 = current_c5
    best_h = h.copy()
    
    for a in alphas:

        h_new = h + a * p
        h_new = project_to_target(h_new, n / 2.0)
        c5_new, _ = exact_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_alpha = a
            best_h = h_new.copy()
    
    # Golden section refinement around best_alpha
    if best_alpha > 0 and best_alpha < alpha_max:
        # Find bracket
        left = max(0, best_alpha - alpha_max / 30 * 2)
        right = min(alpha_max, best_alpha + alpha_max / 30 * 2)
        gr = (np.sqrt(5) - 1) / 2
        a1 = right - gr * (right - left)
        a2 = left + gr * (right - left)
        f1 = exact_c5(project_to_target(h + a1 * p, n/2.0))[0]
        f2 = exact_c5(project_to_target(h + a2 * p, n/2.0))[0]
        for _ in range(30):

                a2 = a1
                f2 = f1
                a1 = right - gr * (right - left)
                f1 = exact_c5(project_to_target(h + a1 * p, n/2.0))[0]
            else:
                left = a1
                a1 = a2
                f1 = f2
                a2 = left + gr * (right - left)
                f2 = exact_c5(project_to_target(h + a2 * p, n/2.0))[0]
        a_best = (left + right) / 2
        h_gold = project_to_target(h + a_best * p, n/2.0)
        c5_gold = exact_c5(h_gold)[0]
        if c5_gold < best_c5:

            best_alpha = a_best

import numpy as np
from scipy.optimize import linprog
import time


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    return float(np.max(scores)), scores


def project_to_target(v, target):
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.argsort(-x)
            for i in idx:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            idx = np.argsort(x)
            for i in idx:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return x


def compute_gradients(h, n):
    grads = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        if k >= 0:
            i1 = np.arange(k, n)
            grads[r, i1] += (1.0 - h[i1 - k])
            i2 = np.arange(0, n - k)
            grads[r, i2] -= h[i2 + k]
        else:
            g = -k
            i1 = np.arange(0, n - g)
            grads[r, i1] += (1.0 - h[i1 + g])
            i2 = np.arange(g, n)
            grads[r, i2] -= h[i2 - g]
    return grads


def lp_direction(h, scores_arr, grads, radius):
    n = len(h)
    L = len(scores_arr)
    num_vars = n + 1
    c = np.zeros(num_vars)
    c[-1] = 1.0
    A_ub = np.zeros((L, num_vars))
    b_ub = np.zeros(L)
    for r in range(L):
        A_ub[r, :n] = grads[r]
        A_ub[r, -1] = -1.0
        b_ub[r] = -scores_arr[r]
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-10)
        hi = min(radius, 1.0 - h[i] - 1e-10)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    s_min = float(np.min(scores_arr))
    s_max = float(np.max(scores_arr))
    bounds.append((s_min - radius * (n + 1), s_max + radius * (n + 1)))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    p = res.x[:n]
    tau = res.x[-1]
    return p, tau


def line_search(h, p, n, current_c5, max_radius):
    alpha_max = max_radius
    for i in range(n):
        if p[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    alpha_max *= 0.9995
    if alpha_max <= 1e-12:
        return h.copy(), current_c5
    target_sum = n / 2.0
    alphas = np.linspace(0, alpha_max, 25)
    best_alpha = 0.0
    best_c5 = current_c5
    best_h = h.copy()
    for a in alphas:
        if a <= 1e-12:
            continue
        h_new = project_to_target(h + a * p, target_sum)
        c5_new, _ = exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_alpha = a
            best_h = h_new.copy()
    if best_alpha > 0 and best_alpha < alpha_max:
        span = alpha_max / 25.0
        left = max(0.0, best_alpha - 2 * span)
        right = min(alpha_max, best_alpha + 2 * span)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1 = right - gr * (right - left)
        a2 = left + gr * (right - left)
        f1 = exact_c5(project_to_target(h + a1 * p, target_sum))[0]
        f2 = exact_c5(project_to_target(h + a2 * p, target_sum))[0]
        for _ in range(25):
            if f1 < f2:
                right = a2
                a2 = a1
                f2 = f1
                a1 = right - gr * (right - left)
                f1 = exact_c5(project_to_target(h + a1 * p, target_sum))[0]
            else:
                left = a1
                a1 = a2
                f1 = f2
                a2 = left + gr * (right - left)
                f2 = exact_c5(project_to_target(h + a2 * p, target_sum))[0]
        a_best = (left + right) / 2.0
        h_gold = project_to_target(h + a_best * p, target_sum)
        c5_gold, _ = exact_c5(h_gold)
        if c5_gold < best_c5 - 1e-13:
            best_c5 = c5_gold
            best_h = h_gold
    return best_h, best_c5


def lp_local_optimize(h, deadline, initial_radius=0.02, max_iters=2000):
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5, _ = exact_c5(h)
    radius = initial_radius
    stall_count = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        _, scores_arr = exact_c5(best_h)
        current_max = float(np.max(scores_arr))
        grads = compute_gradients(best_h, n)
        p, tau = lp_direction(best_h, scores_arr, grads, radius)
        if p is None:
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        if tau >= current_max - 1e-9:
            stall_count += 1
            radius *= 0.7
            if radius < 1e-7:
                if stall_count > 20:
                    break
                p_rand = np.zeros(n)
                mask = np.random.default_rng(hash((n, stall_count)) % (2**31 - 1)).random(n) < 0.3
                p_rand[mask] = np.sign(np.random.default_rng(stall_count * 7).random(np.sum(mask)) - 0.5)
                p_rand -= np.mean(p_rand)
                p_rand *= radius
                h_try = project_to_target(best_h + p_rand, target_sum)
                c5_try, _ = exact_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    stall_count = 0
                continue
        h_new, c5_new = line_search(best_h, p, n, current_max, radius)
        if c5_new < best_c5 - 1e-11:
            best_c5 = c5_new
            best_h = h_new
            stall_count = 0
            radius = min(radius * 1.2, 0.08)
        else:
            stall_count += 1
            radius *= 0.6
            if radius < 1e-7:
                break
    return best_h, best_c5


def interpolate_to_n(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):

    t_start = time.time()
