```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, target=None):
    """Project z onto {0<=h<=1, sum(h)=n/2}"""
    n = len(z)
    z = np.clip(z, -10, 10)  # avoid overflow in bisection
    if target is None:
        target = n / 2.0
    # Bisection for lambda
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0, 1)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return h

def compute_smooth_obj_and_grad(h, beta):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
        obj = max_c
    else:
        weights = exp_vals / exp_sum
        obj = max_c + np.log(exp_sum) / beta
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        t = abs(shift)
        if t == 0:

        else:

            grad[t:] += w * (1.0 - h[:n-t])
            # -h[j+t] for j in [0, n-1-t]
            grad[:n-t] += w * (-h[t:])
    
    return obj, grad

def plsed(h0, betas, max_iters_per_beta, time_limit=None, t_start=None):
    """Projected Log-Sum-Exp Descent."""
    best_h = h0.copy()
    best_score = official_score(h0)
    
    for beta in betas:
        h = best_h.copy()
        alpha = 0.1
        for it in range(max_iters_per_beta):
            if time_limit is not None and time.time() - t_start > time_limit:
                break
            
            obj, grad = compute_smooth_obj_and_grad(h, beta)
            
            # Gradient step
            for _ in range(10):  # Line search
                h_new = project_feasible(h - alpha * grad)
                obj_new, _ = compute_smooth_obj_and_grad(h_new, beta)
                score_new = official_score(h_new)
                if obj_new < obj - 1e-10 or score_new < best_score:
                    h = h_new
                    if score_new < best_score:
                        best_score = score_new
                        best_h = h_new.copy()
                    break
                alpha *= 0.5
            else:
                # No progress, try reducing alpha further or break
                if alpha < 1e-10:
                    break
            
            # Adaptive alpha
            if obj_new < obj - 1e-10:
                alpha = min(alpha * 1.2, 1.0)
    
    return best_h, best_score

def bits_to_score(bits, n):
    """Compute C5 for binary h represented as integer bits."""
    max_count = 0
    for t in range(1, n):
        mask = (1 << (n - t)) - 1
        # Count (0,1) pairs: h[i]=0, h[i+t]=1
        first = bits & mask
        second = (bits >> t) & mask
        count_01 = (second & ~first & mask).bit_count()
        # Count (1,0) pairs: h[i]=1, h[i+t]=0
        count_10 = (first & ~second & mask).bit_count()
        max_count = max(max_count, count_01, count_10)
    dx = 2.0 / n
    return max_count * dx

        # Pick a random 1 and a random 0 to swap
        ones = [i for i in range(n) if (bits >> i) & 1]
        zeros = [i for i in range(n) if not ((bits >> i) & 1)]
        if not ones or not zeros:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        new_bits = bits ^ (1 << i) ^ (1 << j)
        new_score = bits_to_score(new_bits, n)
        
        if new_score < current_score - 1e-15:
            current_score = new_score
            bits = new_bits
            if new_score < best_score:
                best_score = new_score
                best_bits = new_bits
        # else: reject (simple hill climbing)
    
    return best_bits, best_score
```

∂c(-1)/∂h[0] = (1-h[1]) = 1-b (only first term)
∂c(-1)/∂h[1] = (1-h[2]) - h[0] = 1-c-a (both terms)
∂c(-1)/∂h[2] = (1-h[3]) - h[1] = 1-d-b (both terms)
∂c(-1)/∂h[3] = -h[2] = -c (only second term)

My numpy formula for -t (t=1, n=4):
grad[:n-t] += w- * (1 - h[t:])  → grad[0:3] += w- * (1-h[1:4]) = w- * (1-b, 1-c, 1-d)
if n-2*t+1 > 0: n-2+1 = 3 > 0
  grad[t:n-t+1] -= w- * h[:n-2*t+1]  → grad[1:4] -= w- * h[0:3] = w- * (a, b, c)

Result:
grad[0] = w-*(1-b) ✓
grad[1] = w-*(1-c) - w-*a = w-*(1-c-a) ✓
grad[2] = w-*(1-d) - w-*b = w-*(1-d-b) ✓
grad[3] = -w-*c ✓

```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, target=None):
    n = len(z)
    z = np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return h

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
        obj = max_c
    else:
        weights = exp_vals / exp_sum
        obj = max_c + np.log(exp_sum) / beta
    grad = np.zeros(n)
    w0 = weights[n - 1] * dx
    grad += w0 * (1.0 - 2.0 * h)
    for t in range(1, n):
        wp = weights[n - 1 + t] * dx
        wm = weights[n - 1 - t] * dx
        grad[t:] += wp * (1.0 - h[:n - t])
        grad[:n - t] -= wp * h[t:]
        grad[:n - t] += wm * (1.0 - h[t:])
        if n - 2 * t + 1 > 0:
            grad[t:n - t + 1] -= wm * h[:n - 2 * t + 1]
    return obj, grad

def plsed(h0, betas, max_iter, deadline, best_score_ref):
    best_h = h0.copy()
    best_score = official_score(h0)
    h = h0.copy()
    for beta in betas:
        alpha = 0.5
        stagnation = 0
        for it in range(max_iter):
            if time.time() > deadline:
                break
            obj, grad = smooth_obj_grad(h, beta)
            moved = False
            for _ls in range(12):
                h_try = project_feasible(h - alpha * grad)
                score_try = official_score(h_try)
                if score_try < best_score - 1e-12:
                    best_score = score_try
                    best_h = h_try.copy()
                    h = h_try
                    moved = True
                    break
                obj_try, _ = smooth_obj_grad(h_try, beta)
                if obj_try < obj - 1e-10:
                    h = h_try
                    moved = True
                    break
                alpha *= 0.4
            if moved:
                alpha = min(alpha * 1.3, 2.0)
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 30:
                    break
    return best_h, best_score

def bits_score(bits, n):
    max_c = 0
    for t in range(1, n):
        mask = (1 << (n - t)) - 1
        first = bits & mask
        second = (bits >> t) & mask
        c01 = (second & (~first) & mask).bit_count()
        c10 = (first & (~second) & mask).bit_count()
        if c01 > max_c:
            max_c = c01
        if c10 > max_c:
            max_c = c10
    return max_c * (2.0 / n)

def bits_to_h(bits, n):
    h = np.zeros(n)
    for i in range(n):
        if (bits >> i) & 1:
            h[i] = 1.0
    return h

def h_to_bits(h):
    bits = 0
    n = len(h)
    for i in range(n):
        if h[i] > 0.5:
            bits |= (1 << i)
    return bits

def binary_sa(bits, n, rng, max_iters, deadline):
    target = n // 2
    current_score = bits_score(bits, n)
    best_bits = bits
    best_score = current_score
    temp = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 500 == 0 and it > 0:
            temp *= 0.95
        ones = []
        zeros = []
        for i in range(n):
            if (bits >> i) & 1:
                ones.append(i)
            else:
                zeros.append(i)
        if not ones or not zeros:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        new_bits = bits ^ (1 << i) ^ (1 << j)
        new_score = bits_score(new_bits, n)
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            current_score = new_score
            bits = new_bits
            if new_score < best_score:
                best_score = new_score
                best_bits = new_bits
    return best_bits, best_score

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def make_feasible(z):
    return project_feasible(z)
```
