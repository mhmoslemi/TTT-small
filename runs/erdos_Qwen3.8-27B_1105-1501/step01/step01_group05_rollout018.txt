import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(v, n):
    h = np.array(v, dtype=float).copy().ravel()
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    h = h[:n]
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-13:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            if delta < 1e-15:
                break
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            if delta < 1e-15:
                break
            h[idx] -= delta
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] -= residual
        h = np.clip(h, 0.0, 1.0)
    return h

def gen_interpolated(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_feasible(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_feasible(h, n_target)

def gen_random_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 1e-10:
        h[idx[m_full]] = frac
    return project_feasible(h, n)

def gen_random_continuous(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def gen_step_function(n, rng, num_levels=2):
    # Create a step function with num_levels distinct values
    if num_levels >= n:
        return gen_random_continuous(n, rng)
    # Random boundaries
    boundaries = sorted(rng.choice(n - 1, size=min(num_levels - 1, n - 2), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    # Random values (try to include 0 and 1)
    vals = np.zeros(num_levels)
    vals[0] = 1.0
    vals[-1] = 0.0
    for k in range(1, num_levels - 1):
        vals[k] = rng.uniform(0, 1)
    h = np.zeros(n)
    for i in range(len(boundaries) - 1):
        h[boundaries[i]:boundaries[i + 1]] = vals[min(i, num_levels - 1)]
    return project_feasible(h, n)

def compute_active_gradient(h, n, top_k=None):
    """Compute weighted gradient of active constraints."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    theta = np.max(corr)
    # Weight by proximity to maximum
    T = max(0.003 + 0.01 * theta, 0.002)
    weights = np.exp((corr - theta) / T)
    wsum = np.sum(weights)
    if wsum < 1e-15:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    
    # Compute gradient
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ki, k in enumerate(lags):
        w = weights[ki]
        if w < 1e-8:
            continue
        # Gradient of C_k = dx * sum_i h_{i+k}(1-h_i)
        start = max(0, -k)
        end = min(n, n - k)
        for i in range(start, end):
            p = i + k
            grad[p] += w * (1.0 - h[i])
            grad[i] -= w * h[p]
    grad *= dx
    grad -= np.mean(grad)  # Project out equality constraint
    return grad, theta, corr

def pair_descent(h, n, max_passes=20, time_limit=60.0, seed=42):

    rng = np.random.default_rng(seed)
    h = h.copy()
    score = official_score(h)
    start_time = time.perf_counter()
    
    n_pairs = n * (n - 1) // 2
    
    for p in range(max_passes):
        if time.perf_counter() - start_time > time_limit:
            break
        
        # Compute active gradient for pair selection
        grad, theta, corr = compute_active_gradient(h, n)
        
        # Score each pair by expected improvement
        # Pair (i,j): expected change ~ -grad[i]*(delta_i) - grad[j]*(delta_j)
        # For pair move, delta_i = x - h[i], delta_j = -(x - h[i])
        # So change ~ -grad[i]*(x-h[i]) - grad[j]*(h[i]-x) = (grad[j]-grad[i])*(x-h[i])
        # Best if grad[j] > grad[i] and we increase x, or grad[j] < grad[i] and decrease x
        
        # Generate pair list: sorted by |grad[i] - grad[j]| descending
        # But computing all pairs is O(n^2) which is fine for n~65
        pair_scores = []
        for i in range(n):
            for j in range(i + 1, n):
                S = h[i] + h[j]
                L = max(0.0, S - 1.0)
                U = min(1.0, S)
                if U - L < 1e-8:
                    continue
                # Expected improvement heuristic
                diff = grad[j] - grad[i]
                # We can move x from h[i] to [L,U]
                # If diff > 0, increasing x helps (move h[i] up, h[j] down)
                # If diff < 0, decreasing x helps
                max_move = max(h[i] - L, U - h[i])
                pair_score = abs(diff) * max_move
                pair_scores.append((pair_score, i, j))
        
        # Sort and take top pairs
        pair_scores.sort(key=lambda x: -x[0])
        top_pairs = pair_scores[:min(200, len(pair_scores))]

def slsqp_refine(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    initial_score = official_score(h)
    
    # Select top active constraints
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    top_lags = np.argsort(-corr)[:min(8, 2 * n - 1)]
    
    def corr_q(hx, q):
        start = max(0, -q)
        end = min(len(hx), len(hx) - q)
        idx = np.arange(start, end)
        p = idx + q
        return dx * np.sum(hx[p] * (1.0 - hx[idx]))
    
    def corr_q_grad(hx, q):
        start = max(0, -q)
        end = min(len(hx), len(hx) - q)
        idx = np.arange(start, end)
        p = idx + q
        g = np.zeros(len(hx))
        g[p] += (1.0 - hx[idx])
        g[idx] -= hx[p]
        g *= dx
        return g
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    # Inequality constraints: theta >= C_q(h) for top lags
    n_ineq = len(top_lags)
    
    def ineq_fun(x):
        hx = x[:n]
        theta = x[n]
        vals = np.zeros(n_ineq)
        for m, lag in enumerate(top_lags):
            vals[m] = theta - corr_q(hx, int(lag))
        return vals
    
    def ineq_jac(x):
        hx = x[:n]
        n_ineq = len(top_lags)
        J = np.zeros((n_ineq, n + 1))
        for m, lag in enumerate(top_lags):
            g = corr_q_grad(hx, int(lag))
            J[m, :n] = -g
            J[m, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-6
    x0 = np.concatenate([h, [theta0]])
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        t0 = time.perf_counter()
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if time.perf_counter() - t0 < time_limit and result.x is not None:
            h_out = project_feasible(result.x[:n], n)
            s = official_score(h_out)
            if np.isfinite(s) and s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def compute_active_gradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    theta = float(np.max(corr))
    T = max(0.003 + 0.01 * theta, 0.002)
    weights = np.exp((corr - theta) / T)
    wsum = float(np.sum(weights))
    if wsum < 1e-15:
        weights = np.zeros_like(corr)
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum

        p = idx + k

        grad[idx] -= w * h[p]

        g[p] += (1.0 - hx[idx])

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-6
    x0 = np.concatenate([h, [theta0]])

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        t0 = time.perf_counter()
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-10}
        )
        if time.perf_counter() - t0 < time_limit and result.x is not None:
            h_out = project_feasible(result.x[:n], n)
            s = official_score(h_out)
            if np.isfinite(s) and s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass

    return h, initial_score


def smooth_direction_step(h, n, rng, alpha_list=None):
    """Try a smooth gradient-based step for diversification."""
    grad, theta = compute_active_gradient(h, n)
    norm = np.max(np.abs(grad))
    if norm < 1e-12:
        return h, official_score(h)
    direction = -grad / norm
    direction -= np.mean(direction)

    if alpha_list is None:
        alpha_list = [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]

    score = official_score(h)
    best_h = h.copy()
    best_s = score
    for a in alpha_list:
        for sign in [1, -1]:
            h_try = project_feasible(h + sign * a * direction, n)
            s = official_score(h_try)
            if s < best_s - 1e-12:
                best_s = s
                best_h = h_try.copy()
    return best_h, best_s


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 10.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = project_feasible(parent, n_parent)
    else:
        n_parent = 64
        best_h = project_feasible(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    # Generate diverse candidates
    candidates = [best_h.copy()]
    candidates.append(best_h[::-1].copy())
    candidates.append((1.0 - best_h[::-1]).copy())
    candidates.append((1.0 - best_h).copy())

    n_list = sorted(set([
        best_n, max(40, best_n - 5), best_n + 5, best_n + 10,
        max(40, best_n - 10), best_n + 15
    ]))

    for n_try in n_list[:6]:
        candidates.append(gen_interpolated(best_h, n_try))
        candidates.append(gen_interpolated(best_h, n_try, offset=1))
        candidates.append(gen_interpolated(best_h, n_try, offset=2))
        candidates.append(gen_random_binary(n_try, rng))
        candidates.append(gen_random_binary(n_try, rng))
        candidates.append(gen_random_continuous(n_try, rng))
        candidates.append(gen_step_function(n_try, rng, num_levels=3))
        candidates.append(gen_step_function(n_try, rng, num_levels=4))
        candidates.append(gen_step_function(n_try, rng, num_levels=5))

    # Score all candidates
    scored = []
    for hc in candidates:
        if not np.all(np.isfinite(hc)):
            continue
        hc = project_feasible(hc, len(hc))
        s = official_score(hc)
        if np.isfinite(s) and s > 0:
            scored.append((s, hc))

    scored.sort(key=lambda x: (x[0], len(x[1])))

    if len(scored) > 0 and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    # Run pair descent on top candidates
    n_candidates = min(10, len(scored))
    remaining = deadline - time.perf_counter()

    for idx in range(n_candidates):
        if time.perf_counter() > deadline - 40:
            break
        hc = scored[idx][1].copy()
        nc = len(hc)
        remaining = deadline - time.perf_counter()
        frac = 0.8 / max(1, n_candidates - idx)
        t_limit = min(80.0, max(5.0, remaining * frac * 0.8))

        h_loc, s_loc = pair_descent(hc, nc, max_passes=50, time_limit=t_limit, seed=seed + idx * 17)
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = nc

        # Smooth direction diversification + re-optimize
        if time.perf_counter() < deadline - 25:
            h_div, s_div = smooth_direction_step(h_loc, nc, rng)
            if s_div < s_loc:
                h_loc, s_loc = h_div, s_div
                if s_loc < best_score:
                    best_score = s_loc
                    best_h = h_loc.copy()
                    best_n = nc
                h_loc2, s_loc2 = pair_descent(h_loc, nc, max_passes=20, time_limit=min(15.0, max(3.0, (deadline - time.perf_counter()) * 0.3)), seed=seed + 700 + idx)
                if s_loc2 < best_score:
                    best_score = s_loc2
                    best_h = h_loc2.copy()
                    best_n = nc

        # SLSQP polish
        if time.perf_counter() < deadline - 20:
            h_ref, s_ref = slsqp_refine(h_loc, nc, time_limit=min(20.0, deadline - time.perf_counter() - 15))
            if s_ref < best_score:
                best_score = s_ref
                best_h = h_ref.copy()
                best_n = nc
            # Pair descent after SLSQP
            if time.perf_counter() < deadline - 15:
                h_2, s_2 = pair_descent(h_ref, nc, max_passes=20, time_limit=min(12.0, deadline - time.perf_counter() - 10), seed=seed + 900 + idx * 11)
                if s_2 < best_score:
                    best_score = s_2
                    best_h = h_2.copy()
                    best_n = nc

    # Second round: interpolate current best to nearby n and optimize
    if time.perf_counter() < deadline - 30:
        for n_try in [best_n, best_n + 5, best_n - 5, best_n + 10]:
            if n_try < 36 or time.perf_counter() > deadline - 20:
                continue
            h_interp = gen_interpolated(best_h, n_try)
            h_loc, s_loc = pair_descent(h_interp, n_try, max_passes=30, time_limit=min(15.0, deadline - time.perf_counter() - 15), seed=seed + 1500)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()
                best_n = n_try
            if time.perf_counter() < deadline - 15:
                h_ref, s_ref = slsqp_refine(h_loc, n_try, time_limit=min(10.0, deadline - time.perf_counter() - 10))
                if s_ref < best_score:
                    best_score = s_ref
                    best_h = h_ref.copy()
                    best_n = n_try

    # Final polish
    if time.perf_counter() < deadline - 10:
        h_fin, s_fin = slsqp_refine(best_h, best_n, time_limit=min(12.0, deadline - time.perf_counter() - 5))
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    if time.perf_counter() < deadline - 5:
        h_fin2, s_fin2 = pair_descent(best_h, best_n, max_passes=15, time_limit=min(5.0, deadline - time.perf_counter() - 2), seed=seed + 9999)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2.copy()

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
