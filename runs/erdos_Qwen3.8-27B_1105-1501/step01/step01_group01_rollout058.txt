def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:

            else:

            if abs(add) < 1e-16:

            h[i] += add
            residual -= add

    if n_src == n_dst:

    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def subgradient_step(h, step_size, n_active=1):
    """Take one subgradient step on the max of correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top n_active lags
    sorted_idx = np.argsort(corr)[::-1][:n_active]
    # Compute subgradient as average over active lags
    grad = np.zeros(n)
    for k in sorted_idx:
        shift = k - (n - 1)
        # dS(shift)/dh[p] = (1-h[p-shift]) * I(0<=p-shift<n) - h[p+shift] * I(0<=p+shift<n)
        for p in range(n):
            # First term: p = i+shift, so i = p-shift
            if 0 <= p - shift < n:
                grad[p] += (1.0 - h[p - shift])
            # Second term: p = i, so j = i+shift = p+shift
            if 0 <= p + shift < n:
                grad[p] += (-h[p + shift])
    grad /= n_active
    return grad

    grad = subgradient_step(h, step_size, n_active)
    h_new = make_feasible(h - step_size * grad)
    return h_new

        else:

def active_grad(h):
    """Compute subgradient of max correlation term."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    shift = k_max - (n - 1)
    grad = np.zeros(n)

def multi_active_grad(h, n_active=3):
    """Compute averaged subgradient over top n_active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[::-1][:n_active]

            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]

def binary_push(h, p=2.0):
    """Push h toward binary by raising to power p."""
    # Center around 0.5, push, recenter
    h_centered = h - 0.5
    # Scale to avoid overflow
    scale = np.max(np.abs(h_centered))
    if scale < 1e-10:
        return h.copy()
    h_centered = h_centered / scale
    h_pushed = 0.5 + scale * np.sign(h_centered) * np.abs(h_centered) ** (1.0/p)
    # Actually for pushing toward binary, we want values near 0.5 to move toward 0 or 1
    # h^p for p>1 pushes toward 0, (1-h)^p for the complement
    # Better: h_new = h^p / (h^p + (1-h)^p) which is a sigmoid-like push
    h_new = h ** p / (h ** p + (1 - h) ** p + 1e-30)
    return make_feasible(h_new)

def lse_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    return max_c + np.log(np.sum(np.exp(shifted))) / beta

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            grad[shift:] += w * (1.0 - h[:n - shift])
            grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    return grad

        else:
            s = -shift
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]

    best_s = score(h)
    
    for it in range(max_iter):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        s = score(h)
        if s >= best_s:
            h[i] = 1
            h[j] = 0
        else:
            best_s = s
    
    return h, best_s

import numpy as np
from scipy.optimize import minimize
import time

def score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def active_subgrad(h, n_active=1):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k in top_k:
        shift = k - (n - 1)
        if shift >= 0:
            grad[shift:] += (1.0 - h[:n - shift])
            grad[:n - shift] -= h[shift:]
        else:
            s = -shift
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    grad /= n_active
    return grad

def lse_obj_function(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    sh = np.clip(beta * (corr - mc), -500, 0)
    return mc + np.log(np.sum(np.exp(sh))) / beta

def lse_grad_function(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    sh = np.clip(beta * (corr - mc), -500, 0)
    ev = np.exp(sh)
    w = ev / (np.sum(ev) + 1e-300)
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            grad[shift:] += wk * (1.0 - h[:n - shift])
            grad[:n - shift] -= wk * h[shift:]
        else:
            s = -shift
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
    return grad

def _eq_fn(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac_fn(x):
    return np.ones(len(x))

def _box_bounds_fn(n):
    return [(0.0, 1.0)] * n

def lse_opt(h0, beta, maxiter=200):
    n = len(h0)
    try:
        res = minimize(
            lse_obj_function, h0, args=(beta,), jac=lse_grad_function,
            method='SLSQP',
            bounds=_box_bounds_fn(n),
            constraints=[{'type': 'eq', 'fun': _eq_fn, 'jac': _eq_jac_fn}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0

def sg_descent(h0, max_iter=3000, step0=0.05, decay=0.999, momentum=0.9, n_active=1, patience=500):
    h = h0.copy()
    best_h = h.copy()
    best_s = score(h)
    cur_s = best_s
    v = np.zeros_like(h)
    step = step0
    no_improve = 0
    for it in range(max_iter):
        g = active_subgrad(h, n_active)
        v = momentum * v - step * g
        h_new = make_feasible(h + v)
        s_new = score(h_new)
        if s_new < best_s:
            best_h = h_new.copy()
            best_s = s_new
            no_improve = 0
        else:
            no_improve += 1
        if s_new <= cur_s:
            h = h_new
            cur_s = s_new
        else:
            step *= decay
            v *= 0.5
        if step < 1e-9:
            break
        if no_improve > patience:
            break
    return best_h, best_s

def binary_push(h, p=2.0):
    h_safe = np.clip(h, 1e-15, 1.0 - 1e-15)
    h_new = h_safe ** p / (h_safe ** p + (1.0 - h_safe) ** p + 1e-30)
    return make_feasible(h_new)

def random_binary(n, rng):
    h = np.zeros(n)
    target = n // 2
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    if n % 2 == 1:
        h[idx[target]] = 0.5
    return make_feasible(h)
