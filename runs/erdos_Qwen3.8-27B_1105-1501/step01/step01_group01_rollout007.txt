    """Take a subgradient step on the max overlap, then project."""
    n = len(h)

def project(z, target=None):
    """Project z onto {0≤h≤1, sum(h)=target}."""
    n = len(z)

def pair_swap_descent(h, n_iter=50):

    n = len(h)
    best = h.copy()
