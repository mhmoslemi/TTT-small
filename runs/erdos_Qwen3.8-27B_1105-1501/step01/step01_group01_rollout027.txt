def project_feasible(z):
    """Project z onto {0<=h<=1, sum(h)=n/2}"""
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    target = n / 2.0
    lo = np.min(z) - 2.0
    hi = np.max(z) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s < target:
            hi = mid
        else:
            lo = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual

    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)
        else:

            else:

    n = len(h)

        upper = (bits >> s) & mask
        lower = bits & mask
        count = (upper & ~lower & mask).bit_count()
        if count > max_count:

    """Compute smooth LSE objective and gradient"""
    n = len(h)
    dx = 2.0 / n

    if exp_sum < 1e-300:

    else:
        weights = exp_vals / exp_sum

    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n - 1)

            ell = n - s
            if ell <= 0:

            grad[:ell] -= w * h[s:]

            grad[:ell] -= w * h[t:]
    return phi, grad

def project_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

def bit_score(bits, n):
    dx = 2.0 / n
    max_count = 0
    for s in range(1, n):
        mask = (1 << (n - s)) - 1
        upper = (bits >> s) & mask
        lower = bits & mask
        count = (upper & ~lower & mask).bit_count()
        if count > max_count:
            max_count = count
    return dx * max_count
