            order = np.argsort(z)

        h[i+1] = max(0.0, min(1.0, oj + a - b))
        h[i+2] = max(0.0, min(1.0, ok + b))
        c_new = score_exact(make_feasible(h))
        if c_new < cur_c - 1e-14:
            return float(c_new)
    h[i] = oi; h[i+1] = oj; h[i+2] = ok
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: Focused gradient descent (22% time)
    p1_end = t0 + budget_s * 0.22
    if time.time() < p1_end:
        h1, c1 = grad_desc(best_h, p1_end, 0.12, 0.002, 8, 0.025, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
        n = len(best_h)

    # Phase 2: Intensive local search with golden section (30% time)
    p2_end = t0 + budget_s * 0.52
    cur = best_h.copy()
    cur_c = best_c
    fails = 0
    while time.time() < p2_end:
        r = rng.random()
        if r < 0.55:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = golden_pair(cur, i, j, cur_c, n_iter=26)
        elif r < 0.75:
            i = int(rng.integers(0, n - 1))
            res = golden_pair(cur, i, i + 1, cur_c, n_iter=28)
        elif r < 0.85:
            i = int(rng.integers(0, max(1, n - 2)))
            res = block3_search(cur, i, cur_c)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = golden_pair(cur, i, j, cur_c, n_iter=24)
        if res is not None:
            cur_c = res
            fails = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            for _ in range(3):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = golden_pair(cur, i2, j2, cur_c, n_iter=22)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            fails += 1
            if fails > 300:
                cur = make_feasible(best_h + rng.normal(0, 0.008, n), n)
                cur_c = score_exact(cur)
                fails = 0

    # Phase 3: Fine gradient descent (15% time)
    p3_end = t0 + budget_s * 0.67
    if time.time() < p3_end:
        h3, c3 = grad_desc(best_h, p3_end, 0.006, 0.0003, 5, 0.003, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3
        n = len(best_h)

    # Phase 4: Systematic adjacent pair sweep + random (13% time)
    p4_end = t0 + budget_s * 0.80
    cur = best_h.copy()
    cur_c = best_c
    pass_count = 0
    while time.time() < p4_end:
        improved = False
        for i in range(n - 1):
            if time.time() > p4_end:
                break
            res = golden_pair(cur, i, i + 1, cur_c, n_iter=25)
            if res is not None:
                cur_c = res
                improved = True
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c
        if not improved:
            for _ in range(30):
                if time.time() > p4_end:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                res = golden_pair(cur, i, j, cur_c, n_iter=25)
                if res is not None:
                    cur_c = res
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
        pass_count += 1

    # Phase 5: Multi-resolution (12% time)
    p5_end = t0 + budget_s * 0.92
    n_cur = len(best_h)
    tried_ns = set()
    for dn in (0, -5, 5, -10, 10, -15, 15, 20):
        if time.time() > p5_end - 5:
            break
        nr = n_cur + dn
        if nr < 20 or nr > 150 or nr in tried_ns or nr == n_cur:
            continue
        tried_ns.add(nr)
        h_r = resample(best_h, nr)
        if h_r is None:
            continue
        c_r = score_exact(h_r)
        sub_dl = min(p5_end - 3, time.time() + 35)
        if time.time() < sub_dl:
            h_r, c_r = grad_desc(h_r, sub_dl, 0.008, 0.0005, 6, 0.004, rng)
            # Quick local polish
            if time.time() < p5_end - 3:
                nr_n = len(h_r)
                for _ in range(50):
                    if time.time() > p5_end - 3:
                        break
                    i = int(rng.integers(0, nr_n))
                    j = int(rng.integers(0, nr_n))
                    if i == j:
                        continue
                    res = golden_pair(h_r, i, j, c_r, n_iter=20)
                    if res is not None:
                        c_r = res
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 6: Final gradient polish (8% time)
    if time.time() < deadline - 2:
        n = len(best_h)
        h_f, c_f = grad_desc(best_h, deadline - 3, 0.003, 0.0001, 5, 0.002, rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        # Final local polish
        n = len(best_h)
        for _ in range(200):
            if time.time() > deadline - 2:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = golden_pair(best_h, i, j, best_c, n_iter=22)
            if res is not None:
                best_c = res

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
