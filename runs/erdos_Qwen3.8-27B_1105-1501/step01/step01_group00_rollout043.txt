def repair_exact(x):

    n = x.size
    if n == 0:
        return x
    x[~np.isfinite(x)] = 0.5
    target = 0.5 * n
    lo = -float(np.max(x)) - 1.0
    hi = 1.0 - float(np.min(x)) + 1.0
    # ensure bounds
    for _ in range(10):
        if float(np.sum(np.clip(x + lo, 0.0, 1.0))) > target:
            lo -= 1.0
        else:
            break
    for _ in range(10):
        if float(np.sum(np.clip(x + hi, 0.0, 1.0))) < target:
            hi += 1.0
        else:
            break
    for _ in range(90):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    residual = target - float(np.sum(h))

            if cap <= 1e-15:

import numpy as np

def repair_exact(x):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    x[~np.isfinite(x)] = 0.5
    target = 0.5 * n

    lo = -float(np.max(x)) - 1.0
    hi = 1.0 - float(np.min(x)) + 1.0

    for _ in range(12):
        if float(np.sum(np.clip(x + lo, 0.0, 1.0))) > target:
            lo -= 1.0
        else:
            break
    for _ in range(12):
        if float(np.sum(np.clip(x + hi, 0.0, 1.0))) < target:
            hi += 1.0
        else:
            break

    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - float(np.sum(h))
    tol = 1e-12

    if residual > tol:
        order = np.argsort(1.0 - h)
        for idx in order:
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(residual, cap)
            h[idx] += add
            residual -= add
            if residual <= 1e-14:
                break
    elif residual < -tol:
        order = np.argsort(h)
        for idx in order:
            cap = h[idx]
            if cap <= 1e-15:
                continue
            sub = min(-residual, cap)
            h[idx] -= sub
            residual += sub
            if residual >= -1e-14:
                break

    return np.clip(h, 0.0, 1.0)


def score_h(h):
    n = len(h)
    if n == 0:
        return float("inf")
    v = np.correlate(h, 1.0 - h, mode="full")
    if v.size == 0:
        return float("inf")
    return float(v.max() * (2.0 / n))


def compute_sensitivities(h):
    n = int(len(h))
    m = 2 * n - 1
    S = np.zeros((n, m), dtype=float)
    S[:, n - 1] = 1.0 - 2.0 * h
    for s in range(1, n):
        L = n - s
        pp = n - 1 + s
        pn = n - 1 - s
        S[:L, pp] -= h[s:s + L]
        S[s:, pp] += 1.0 - h[:L]
        S[:L, pn] += 1.0 - h[s:s + L]
        S[s:, pn] -= h[:L]
    return S


def apply_pair(h, i, j, delta):
    h2 = h.copy()
    s = float(h[i]) + float(h[j])
    ah = float(h[i]) + float(delta)
    if ah < 0.0:
        ah = 0.0
    elif ah > 1.0:
        ah = 1.0
    bh = s - ah
    if bh < 0.0:
        bh = 0.0
        ah = s
    elif bh > 1.0:
        bh = 1.0
        ah = s - 1.0
    h2[i] = ah
    h2[j] = bh
    return h2


def feasible_bounds(h, d):
    lower = -np.inf
    upper = np.inf
    pos = d > 1e-15
    if np.any(pos):
        lower = max(lower, float(np.max(-h[pos] / d[pos])))
        upper = min(upper, float(np.min((1.0 - h[pos]) / d[pos])))
    neg = d < -1e-15
    if np.any(neg):
        lower = max(lower, float(np.max((1.0 - h[neg]) / d[neg])))
        upper = min(upper, float(np.min(h[neg] / (-d[neg]))))
    if not (np.isfinite(lower) and np.isfinite(upper)):
        return 0.0, 0.0
    if lower < 0.0:
        lower += 1e-12
    if upper > 0.0:
        upper -= 1e-12
    if upper < lower:
        mid = 0.5 * (upper + lower)
        lower = mid
        upper = mid
    return float(lower), float(upper)


def pair_eval_scores(base, L, dvals, p0, p1, p2, dx):
    d2 = dvals * dvals
    V = base + np.outer(dvals, L)
    m = V.shape[1]
    if 0 <= p0 < m:
        V[:, p0] -= 2.0 * d2
    if 0 <= p1 < m:
        V[:, p1] += d2
    if 0 <= p2 < m:
        V[:, p2] += d2
    return np.max(V, axis=1) * dx


def pair_score_at_delta(h, S, base, i, j, delta, dx):
    L = S[i] - S[j]
    n = len(h)
    delta = float(delta)
    d2 = delta * delta
    p0 = n - 1
    dabs = abs(i - j)
    p1 = n - 1 - dabs
    p2 = n - 1 + dabs
    V = base + delta * L
    m = len(base)
    if 0 <= p0 < m:
        V[p0] -= 2.0 * d2
    if 0 <= p1 < m:
        V[p1] += d2
    if 0 <= p2 < m:
        V[p2] += d2
    return float(V.max() * dx)


def pair_line_search(h, S, base, i, j, c, dx, coarse=81, refine=17, levels=2, improve_tol=1e-13):
    a = float(h[i])
    b = float(h[j])
    lo = max(-a, b - 1.0)
    hi = min(1.0 - a, b)
    if hi - lo < 1e-11:
        return None
    lo += 1e-12
    hi -= 1e-12
    if hi - lo < 1e-11:
        return None

    L = S[i] - S[j]
    n = len(h)
    p0 = n - 1
    dabs = abs(i - j)
    p1 = n - 1 - dabs
    p2 = n - 1 + dabs

    dvals = np.linspace(lo, hi, int(coarse))
    scores = pair_eval_scores(base, L, dvals, p0, p1, p2, dx)
    if not np.all(np.isfinite(scores)):
        return None

    idx = int(np.argmin(scores))
    best_score = float(scores[idx])
    best_d = float(dvals[idx])
    if best_score >= c - improve_tol:
        return None

    if 0 < idx < len(dvals) - 1:
        left = float(dvals[idx - 1])
        right = float(dvals[idx + 1])
    else:
        left = best_d
        right = best_d

    for _ in range(int(levels)):
        if right - left < 1e-14:
            break
        rvals = np.linspace(left, right, int(refine))
        rs = pair_eval_scores(base, L, rvals, p0, p1, p2, dx)
        if not np.all(np.isfinite(rs)):
            break
        ridx = int(np.argmin(rs))

        if rscore < best_score - 1e-15:
            best_score = rscore
            best_d = float(rvals[ridx])
            if 0 < ridx < len(rvals) - 1:

                right = float(rvals[ridx + 1])
            else:
                break
        else:
            if rscore < best_score:
                best_score = rscore
                best_d = float(rvals[ridx])
            break

    if best_score < c - improve_tol:
        return best_d, best_score
    return None


def direction_eval_scores(base, L, Q, alphas, dx):
    V = base + np.outer(alphas, L) - np.outer(alphas * alphas, Q)
    return np.max(V, axis=1) * dx


def line_search_generic(h, S, base, d, c, dx, line_coarse=129, refine=19, levels=2, improve_tol=1e-13):

    best_a = float(alphas[idx])

        return None

        left = float(alphas[idx - 1])
        right = float(alphas[idx + 1])
    else:
        left = best_a
        right = best_a

        if right - left < 1e-14:
            break

def pair_pass_once(h, S, base, c, dx, pairs, deadline, coarse=81, refine=17, levels=2, improve_tol=1e-13):

            break
        res = pair_line_search(h, S, base, i, j, c, dx,
                               coarse=coarse, refine=refine, levels=levels,
                               improve_tol=improve_tol)
        if res is None:

        delta = res[0]
        h2 = apply_pair(h, i, j, delta)
        if (float(np.min(h2)) < -1e-8 or float(np.max(h2)) > 1.0 + 1e-8 or

def anneal_search(h0, budget_s, rng, deadline_abs=None, sample_deltas=5):
    tcur = time.time()
    if deadline_abs is None:
        deadline = tcur + max(0.0, budget_s)
    else:
        deadline = min(tcur + max(0.0, budget_s), deadline_abs)

    h = repair_exact(np.asarray(h0, dtype=float).copy())
    n = len(h)
    if n == 0:
        return h, score_h(h), n
    dx = 2.0 / n
    base = np.correlate(h, 1.0 - h, mode="full")
    S = compute_sensitivities(h)
    c = float(base.max() * dx)

    best_h = h.copy()
    best_c = c

    T0 = 0.008
    TE = 1e-5
    max_cool_steps = 30000
    steps = 0

    while time.time() < deadline:
        steps += 1
        T = max(TE, T0 * (TE / T0) ** (steps / float(max_cool_steps)))

        if n < 2:
            break

        i = int(rng.integers(n))
        j = int(rng.integers(n - 1))
        if j >= i:
            j += 1

        a = float(h[i])
        b = float(h[j])
        lo = max(-a, b - 1.0)
        hi = min(1.0 - a, b)
        if hi - lo < 1e-12:
            continue

        deltas = rng.uniform(lo, hi, size=int(sample_deltas))
        best_delta = None
        best_sc = float("inf")
        for delta in deltas:
            sc = pair_score_at_delta(h, S, base, i, j, float(delta), dx)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_delta = float(delta)

        if best_delta is None or not np.isfinite(best_sc):
            continue

        accept = False
        if best_sc < c - 1e-14:
            accept = True
        else:
            diff = best_sc - c
            if diff < 1e-3:
                if rng.random() < float(np.exp(-diff / T)):
                    accept = True

        if accept:
            h2 = apply_pair(h, i, j, best_delta)
            if (float(np.min(h2)) < -1e-8 or float(np.max(h2)) > 1.0 + 1e-8 or
                    abs(float(np.sum(h2)) - 0.5 * n) > 1e-7):
                h2 = repair_exact(h2)
            sc = float(np.correlate(h2, 1.0 - h2, mode="full").max() * dx)
            if not np.isfinite(sc):
                continue
            h = h2
            c = sc
            base = np.correlate(h, 1.0 - h, mode="full")
            S = compute_sensitivities(h)
            if c < best_c - 1e-13:
                best_c = c
                best_h = h.copy()

    return best_h, best_c, n


def optimize_local(h0, budget_s, rng, deadline_abs=None,
                   pair_coarse=81, pair_refine=17, pair_levels=2,
                   line_coarse=129, max_outer=5000, random_dirs=8):
    tcur = time.time()
    if deadline_abs is None:
        deadline = tcur + max(0.0, budget_s)
    else:
        deadline = min(tcur + max(0.0, budget_s), deadline_abs)

    h = repair_exact(np.asarray(h0, dtype=float).copy())
    n = len(h)
    if deadline <= tcur + 0.2 or n == 0:
        return h, score_h(h), n

    dx = 2.0 / n
    base = np.correlate(h, 1.0 - h, mode="full")
    S = compute_sensitivities(h)
    c = float(base.max() * dx)

    pairs = []
    if n <= 80:
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
    else:
        all_i = np.repeat(np.arange(n), n)
        all_j = np.tile(np.arange(n), n)
        mask = all_i < all_j
        ii = all_i[mask]
        jj = all_j[mask]
        max_pairs = 3000
        if len(ii) > max_pairs:
            idx = rng.choice(len(ii), size=max_pairs, replace=False)
            ii = ii[idx]
            jj = jj[idx]
        pairs = list(zip(ii.tolist(), jj.tolist()))

    stagnant = 0
    outer = 0

    while time.time() < deadline and outer < max_outer:
        improved = False

        if len(pairs) <= 2600 and time.time() + 2.0 < deadline:
            h, S, base, c, imp = pair_pass_once(
                h, S, base, c, dx, pairs, deadline,
                coarse=pair_coarse, refine=pair_refine, levels=pair_levels
            )
            if imp:
                improved = True
        elif len(pairs) > 2600 and time.time() + 1.0 < deadline:
            if len(pairs) <= 1500:
                sub = pairs
            else:
                sub = [pairs[int(rng.integers(0, len(pairs)))] for _ in range(1500)]
            h, S, base, c, imp = pair_pass_once(
                h, S, base, c, dx, sub, deadline,
                coarse=pair_coarse, refine=pair_refine, levels=pair_levels
            )
            if imp:
                improved = True

        if time.time() + 0.5 < deadline:
            h, S, base, c, imp = global_round_once(
                h, S, base, c, dx, deadline, rng,
                n_random=random_dirs, line_coarse=line_coarse
            )
            if imp:
                improved = True

        if not improved:
            stagnant += 1
            if stagnant >= 2:
                break
        else:
            stagnant = 0
        outer += 1

    return h, c, n


def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == 0:
        return repair_exact(np.zeros(int(n_tgt)))
    src_x = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_x = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_tgt = np.interp(tgt_x, src_x, h_src)
    return repair_exact(h_tgt)


def random_wave_start(n, rng, amplitude=0.6):
    x = np.arange(n) + 0.5
    h = 0.5 + np.zeros(n, dtype=float)
    for f in range(1, 5):
        amp = amplitude / (f + 1.0)
        amp *= 0.5 + float(rng.random())
        phase = 2.0 * np.pi * float(rng.random())
        h = h + amp * np.sin(2.0 * np.pi * f * x / n + phase)
    return repair_exact(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
        if h0.size == 0:
            raise ValueError("empty parent")
        h0 = repair_exact(h0)
    except Exception:
        h0 = random_wave_start(60, rng, amplitude=0.7)

    c0 = score_h(h0)
    if np.isfinite(c0):
        best_h = h0.copy()
        best_c = c0
        best_n = len(h0)

    if best_h is None:
        best_h = random_wave_start(60, rng, amplitude=0.7)
        best_c = score_h(best_h)
        best_n = len(best_h)

    nb = best_n
    for s in range(1, nb):
        hs = np.roll(best_h, s)
        cs = score_h(hs)
        if np.isfinite(cs) and (cs < best_c - 1e-12 or (abs(cs - best_c) <= 1e-12 and best_n > nb)):
            best_h = hs.copy()
            best_c = cs
            best_n = nb

    if time.time() + 3.0 < deadline:
        b = min(0.50 * budget_s, deadline - time.time())
        if b > 5.0:
            h1, c1, n1 = optimize_local(
                best_h, b, rng, deadline_abs=deadline,
                pair_coarse=97, pair_refine=17, pair_levels=2,
                line_coarse=161, random_dirs=10
            )
            if np.isfinite(c1) and (c1 < best_c - 1e-12 or (abs(c1 - best_c) <= 1e-12 and n1 < best_n)):
                best_h = h1.copy()
                best_c = c1
                best_n = n1

    if time.time() + 15.0 < deadline:
        b = min(0.14 * budget_s, deadline - time.time())
        if b > 15.0:
            h1, c1, n1 = anneal_search(best_h, b, rng, deadline_abs=deadline, sample_deltas=5)
            if np.isfinite(c1) and (c1 < best_c - 1e-12 or (abs(c1 - best_c) <= 1e-12 and n1 < best_n)):
                best_h = h1.copy()
                best_c = c1
                best_n = n1

    if time.time() + 5.0 < deadline:
        b = min(0.15 * budget_s, deadline - time.time())
        if b > 5.0:
            h1, c1, n1 = optimize_local(
                best_h, b, rng, deadline_abs=deadline,
                pair_coarse=97, pair_refine=17, pair_levels=2,
                line_coarse=161, random_dirs=8
            )
            if np.isfinite(c1) and (c1 < best_c - 1e-12 or (abs(c1 - best_c) <= 1e-12 and n1 < best_n)):
                best_h = h1.copy()
                best_c = c1
                best_n = n1

    target_list = []
    for dt in (-10, -6, -4, -2, 2, 4, 6, 10):
        t = best_n + dt
        if 24 <= t <= 80 and t != best_n:
            target_list.append(t)
    for t in (32, 40, 48, 54, 60, 64, 70, 76):
        if 24 <= t <= 80 and t != best_n:
            target_list.append(t)

    seen = []
    for t in target_list:
        if t not in seen:
            seen.append(t)
    target_list = seen
    target_list = target_list[:7]

    remaining_targets = len(target_list)
    for t in target_list:
        if time.time() >= deadline - 15.0:
            break
        remaining_targets -= 1
        h_int = interpolate_to(best_h, t)
        c_int = score_h(h_int)
        if np.isfinite(c_int) and (c_int < best_c - 1e-12 or (abs(c_int - best_c) <= 1e-12 and t < best_n)):
            best_h = h_int.copy()
            best_c = c_int
            best_n = t

        time_left = deadline - time.time()
        b = min(90.0, max(0.0, time_left) / max(1, remaining_targets + 1))
        if b < 12.0:
            break
        h1, c1, n1 = optimize_local(
            h_int, b, rng, deadline_abs=deadline,
            pair_coarse=73, pair_refine=15, pair_levels=2,
            line_coarse=129, random_dirs=5
        )
        if np.isfinite(c1) and (c1 < best_c - 1e-12 or (abs(c1 - best_c) <= 1e-12 and n1 < best_n)):
            best_h = h1.copy()
            best_c = c1
            best_n = n1

    sigma_list = (0.02, 0.04, 0.07, 0.11, 0.17, 0.26, 0.38)
    restarts = 0
    while time.time() < deadline - 25.0:
        if restarts >= 8:
            break
        sigma = sigma_list[restarts % len(sigma_list)]
        nr = int(best_n)
        if nr < 2:
            break
        if restarts == 3:
            h_r = random_wave_start(nr, rng, amplitude=0.8)
        else:
            h_r = repair_exact(best_h + rng.normal(0.0, sigma, size=nr))
            if rng.random() < 0.25:
                h_r = np.roll(h_r, int(rng.integers(0, nr)))

        c_r = score_h(h_r)
        if np.isfinite(c_r) and (c_r < best_c - 1e-12 or (abs(c_r - best_c) <= 1e-12 and nr < best_n)):
            best_h = h_r.copy()
            best_c = c_r
            best_n = nr

        time_left = deadline - time.time()
        b = min(55.0, max(0.0, time_left) / max(1, (8 - restarts)))
        if b < 18.0:
            break
        h1, c1, n1 = optimize_local(
            h_r, b, rng, deadline_abs=deadline,
            pair_coarse=65, pair_refine=13, pair_levels=2,
            line_coarse=129, random_dirs=6
        )
        if np.isfinite(c1) and (c1 < best_c - 1e-12 or (abs(c1 - best_c) <= 1e-12 and n1 < best_n)):
            best_h = h1.copy()
            best_c = c1
            best_n = n1
        restarts += 1

    if time.time() + 3.0 < deadline:
        b = deadline - 0.5 - time.time()
        if b > 3.0:
            h1, c1, n1 = optimize_local(
                best_h, b, rng, deadline_abs=deadline - 0.5,
                pair_coarse=121, pair_refine=19, pair_levels=2,
                line_coarse=201, random_dirs=4
            )
            if np.isfinite(c1) and (c1 < best_c - 1e-12 or (abs(c1 - best_c) <= 1e-12 and n1 < best_n)):
                best_h = h1.copy()
                best_c = c1
                best_n = n1

    final_h = repair_exact(np.asarray(best_h, dtype=float).copy())
    final_c = score_h(final_h)
    final_n = len(final_h)

    if not np.isfinite(final_c):
        final_h = repair_exact(0.5 * np.ones(final_n))
        final_c = score_h(final_h)

    return final_h, final_c, final_n
