def exact_c5(h):
    n = len(h)

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_to_target(v, target):
    """Project v to [0,1] with sum = target using water-filling."""
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    n = len(v)

        return np.zeros(n)

        return np.ones(n)

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def lp_direction(h, scores, grads, radius, lambda_reg, n_lags_to_use):
    """Find a direction p using LP minimax linearization."""
    n = len(h)
    # Select top n_lags_to_use scores
    order = np.argsort(scores)[::-1]
    selected = order[:n_lags_to_use]
    
    # Build LP:
    # Variables: [p_0, ..., p_{n-1}, tau, r_var] = n+2 vars
    # Objective: minimize tau + lambda_reg * r_var
    c = np.zeros(n + 2)
    c[n] = 1.0       # tau
    c[n+1] = lambda_reg  # r_var
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1. For each selected lag r: grad_r · p - tau <= -score_r
    n_constraints = len(selected) + 2 * n
    
    A_ub = np.zeros((n_constraints, n + 2))
    b_ub = np.zeros(n_constraints)
    
    for i, r in enumerate(selected):
        A_ub[i, :n] = grads[r]
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[r]
    
    # 2. p[j] - r_var <= 0
    base = len(selected)
    for j in range(n):
        A_ub[base + j, j] = 1.0
        A_ub[base + j, n+1] = -1.0
        b_ub[base + j] = 0.0
    
    # 3. -p[j] - r_var <= 0
    base2 = len(selected) + n
    for j in range(n):
        A_ub[base2 + j, j] = -1.0
        A_ub[base2 + j, n+1] = -1.0
        b_ub[base2 + j] = 0.0
    
    # Bounds
    bounds = []
    for j in range(n):
        lo_p = max(-radius, -h[j])
        hi_p = min(radius, 1.0 - h[j])
        bounds.append((lo_p, hi_p))
    bounds.append((None, None))  # tau
    bounds.append((0.0, radius))  # r_var
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        p = res.x[:n]
        return p
    return np.zeros(n)
