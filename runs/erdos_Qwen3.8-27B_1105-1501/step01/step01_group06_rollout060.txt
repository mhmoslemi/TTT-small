```python
def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_target(v, target):
    """Project v to [0,1] with sum = target using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for water-filling threshold
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            mask = result < 1.0
            count = int(np.sum(mask))
            if count > 0:
                per = residual / count
                idx = np.where(mask)[0]
                result[idx] = np.minimum(result[idx] + per, 1.0)
        else:
            mask = result > 0.0
            count = int(np.sum(mask))
            if count > 0:
                per = residual / count
                idx = np.where(mask)[0]
                result[idx] = np.maximum(result[idx] + per, 0.0)
    return result

def compute_scores(h):
    """Compute all correlation scores (without dx factor)."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_grads(h, n):
    """Compute gradient for each of the 2n-1 lags.
    Returns array of shape (2n-1, n)."""
    n2 = 2 * n - 1
    grads = np.zeros((n2, n))
    for k in range(n2):
        d = k - (n - 1)
        if d >= 0:
            # Valid i: [0, n-1-d]
            i_range = range(0, n - d)
            for i in i_range:
                grads[k, i] += (1.0 - h[i + d])
            # Second term: j in [d, n-1]
            for j in range(d, n):
                grads[k, j] -= h[j - d]
        elif d < 0:

            # Valid i: [g, n-1]
            for i in range(g, n):
                grads[k, i] += (1.0 - h[i - g])
            # Second term: j in [0, n-1-g]
            for j in range(0, n - g):
                grads[k, j] -= h[j + g]
        else:
            # d = 0
            for j in range(n):
                grads[k, j] = 1.0 - 2.0 * h[j]
    return grads
```

```python
def lp_direction(h, scores, grads, radius, top_k=None, mu=0.0):
    """Solve LP to find a descent direction.
    Minimize tau (+ mu * r if mu > 0)
    subject to: grad_k · p - tau <= -scores[k] for selected k
                max(-radius, -h_i) <= p_i <= min(radius, 1-h_i)
                sum p_i = 0
    """
    n = len(h)
    n2 = 2 * n - 1
    
    # Select top lags
    if top_k is None or top_k >= n2:
        sel = np.arange(n2)
    else:
        # Sort by score descending
        order = np.argsort(scores)[::-1]
        sel = order[:top_k]
    
    L = len(sel)
    
    # Variables: p (n) + tau (1) = n+1 variables
    # If mu > 0, add r variable: n+2 variables
    
    if mu > 0:
        # x = [p_0, ..., p_{n-1}, tau, r]
        nv = n + 2
        c = np.zeros(nv)
        c[n] = 1.0      # tau
        c[n+1] = mu     # r
    else:
        nv = n + 1
        c = np.zeros(nv)
        c[n] = 1.0      # tau
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each selected lag k: grad_k · p - tau <= -scores[k]
    # grad_k · p - tau + 0*r <= -scores[k]
    
    A_ub_rows = []
    b_ub_list = []
    for k in sel:
        row = np.zeros(nv)
        row[:n] = grads[k]
        row[n] = -1.0  # -tau
        if mu > 0:
            # r constraints: -r <= p_i <= r
            # p_i - r <= 0
            # -p_i - r <= 0
            pass  # add below
        b_ub_list.append(-scores[k])
        A_ub_rows.append(row)
    
    if mu > 0:
        # Add r constraints: p_i - r <= 0 and -p_i - r <= 0
        for i in range(n):
            row1 = np.zeros(nv)
            row1[i] = 1.0
            row1[n+1] = -1.0
            A_ub_rows.append(row1)
            b_ub_list.append(0.0)
            
            row2 = np.zeros(nv)
            row2[i] = -1.0
            row2[n+1] = -1.0
            A_ub_rows.append(row2)
            b_ub_list.append(0.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_list)
    
    # Equality: sum p_i = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))  # tau free
    if mu > 0:
        bounds.append((0.0, radius))  # r bounded
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='revised_simplex')
        except Exception:
            return None
    
    if res.success:
        p = res.x[:n]
        tau = res.x[n]
        return p, tau
    return None
```

```python
def line_search(h, p, current_raw_max, max_evals=50):
    """Exact line search along h + alpha * p for alpha in [0, alpha_max].
    Returns (best_alpha, best_raw_max) or (0, current_raw_max) if no improvement."""
    n = len(h)
    
    # Compute alpha_max to keep h + alpha*p in [0,1]
    alpha_max = 1.0
    for i in range(n):
        if p[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-15:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    
    if alpha_max < 1e-12:
        return 0.0, current_raw_max
    
    # Grid search
```

```python
def project_to_target(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                result[mask] += residual / cnt
                result = np.clip(result, 0.0, 1.0)
        else:
            mask = result > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                result[mask] += residual / cnt
                result = np.clip(result, 0.0, 1.0)

    n2 = 2 * n - 1

            j2 = np.arange(0, n - g)
```

```python
def lp_direction(h, scores, grads, radius):
    n = len(h)
    n2 = 2 * n - 1
    # Variables: [p_0, ..., p_{n-1}, tau]
    nv = n + 1
    c = np.zeros(nv)
    c[n] = 1.0  # minimize tau
    
    # Constraints: grad_k @ p - tau <= -scores[k] for all k
    A_ub = np.zeros((n2, nv))
    for k in range(n2):
        A_ub[k, :n] = grads[k]
        A_ub[k, n] = -1.0
    b_ub = -scores
    
    # Equality: sum p = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))  # tau
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None, None
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_target(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                result[mask] += residual / cnt
                result = np.minimum(result, 1.0)
        else:
            mask = result > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                result[mask] += residual / cnt
                result = np.maximum(result, 0.0)
    return result

def compute_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_grads(h, n):
    n2 = 2 * n - 1
    grads = np.zeros((n2, n))
    for k in range(n2):
        d = k - (n - 1)
        if d == 0:
            grads[k] = 1.0 - 2.0 * h
        elif d > 0:
            grads[k, :n - d] += (1.0 - h[d:])
            grads[k, d:] -= h[:n - d]
        else:
            g = -d
            grads[k, g:] += (1.0 - h[:n - g])
            grads[k, :n - g] -= h[g:]
    return grads

def lp_direction(h, scores, grads, radius):
    n = len(h)
    n2 = 2 * n - 1
    nv = n + 1
    c = np.zeros(nv)
    c[n] = 1.0
    A_ub = np.zeros((n2, nv))
    for k in range(n2):
        A_ub[k, :n] = grads[k]
        A_ub[k, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i])
        hi_i = min(radius, 1.0 - h[i])
        if lo_i >= hi_i:
            return None, None
        bounds.append((lo_i, hi_i))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.success:
        return res.x[:n].copy(), float(res.x[n])
    return None, None

def line_search(h, p, current_raw, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):

    if alpha_max < 1e-10:
        return 0.0, current_raw
    n_grid = 25
    alphas = np.linspace(0.0, alpha_max, n_grid)
    best_alpha = 0.0
    best_raw = current_raw
    for idx in range(1, n_grid):
        a = alphas[idx]
        h_new = h + a * p
        raw = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')))
        if raw < best_raw:
            best_raw = raw
            best_alpha = a
    if best_alpha <= 0:
        return 0.0, current_raw
    best_idx = int(np.argmin(alphas - best_alpha))
    lo = alphas[max(0, best_idx - 1)]
    hi = alphas[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = float(np.max(np.correlate(h + a1 * p, 1.0 - (h + a1 * p), mode='full')))
    f2 = float(np.max(np.correlate(h + a2 * p, 1.0 - (h + a2 * p), mode='full')))
    for _ in range(30):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = float(np.max(np.correlate(h + a1 * p, 1.0 - (h + a1 * p), mode='full')))
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = float(np.max(np.correlate(h + a2 * p, 1.0 - (h + a2 * p), mode='full')))
    best_a = 0.5 * (lo + hi)
    f_best = float(np.max(np.correlate(h + best_a * p, 1.0 - (h + best_a * p), mode='full')))
    if f_best < best_raw:
        best_raw = f_best
        best_alpha = best_a
    if best_raw < current_raw:

def transfer_resolution(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    old_x = (np.arange(n_old) + 0.5) / n_old
    new_x = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(new_x, old_x, h_old)
    return project_to_target(h_new, n_new / 2.0)

def random_perturb(h, rng, strength):
    n = len(h)
    h_p = h.copy()
    mask = rng.random(n) < 0.5
    if np.any(mask):
        h_p[mask] += rng.normal(0, strength, int(np.sum(mask)))
    return project_to_target(h_p, n / 2.0)
```
