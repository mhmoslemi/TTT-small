import numpy as np
from scipy import optimize
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    """Project v to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(np.asarray(v, dtype=float).ravel(), 0.0, 1.0)
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    h = h[:n].copy()
    target = n / 2.0
    # Bisection on uniform shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        q = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(q)
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact mass correction
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h):
    """Return G of shape (2n-1, n) where G[r, i] = d(c_r*dx)/dh_i"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    idx = np.arange(n)
    for r in range(L):
        s = r - (n - 1)
        # For each i: contribution is dx * [(1-h_{i-s}) * I(0<=i-s<n) + (-h_{i+s}) * I(0<=i+s<n)]
        # Part 1: (1 - h_{i-s}) where 0 <= i-s < n
        valid1 = (idx >= s) & (idx - s < n) if s >= 0 else (idx - s >= 0) & (idx < n)
        # Actually: need 0 <= i-s < n
        valid1 = (idx - s >= 0) & (idx - s < n)
        G[r, valid1] += dx * (1.0 - h[idx[valid1] - s])
        # Part 2: -h_{i+s} where 0 <= i+s < n
        valid2 = (idx + s >= 0) & (idx + s < n)
        G[r, valid2] += dx * (-h[idx[valid2] + s])
    return G

def lp_minimax_direction(h, G, c_scaled, max_active=40):
    """LP to find direction that minimizes predicted max correlation.
    Returns direction d or None."""
    n = len(h)
    L = 2 * n - 1
    score = np.max(c_scaled)
    
    # Active set: lags with correlation within tolerance of max
    tol = max(1e-6, 0.02 * score)
    active = np.where(c_scaled >= score - tol)[0]
    
    # Limit active set size
    if len(active) > max_active:
        order = np.argsort(c_scaled[active])[::-1]
        active = active[order[:max_active]]
    
    if len(active) == 0:
        return None
    
    # LP: minimize t
    # s.t. G[r,:] @ d - t <= -c_scaled[r] for active r
    #      sum(d) = 0
    #      -1 <= d_i <= 1
    # Variables: d_0,...,d_{n-1}, t (index n)
    
    # Objective: [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: G[r,:] @ d - t <= -c_scaled[r] → G[r,:] @ d + [0,...,0,-1] @ t <= -c_scaled[r]
    A_ub = np.zeros((len(active), n + 1))
    for i, r in enumerate(active):
        A_ub[i, :n] = G[r, :]
        A_ub[i, n] = -1.0
    b_ub = -c_scaled[active]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -1 <= d_i <= 1, t unbounded below (or t >= -1 to help solver)
    bounds = [(-1.0, 1.0)] * n + [(-1.0, 1.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            d = res.x[:n]
            # Check that this is actually a descent direction
            predicted_max = np.max(c_scaled[active] + G[active, :] @ d)
            if predicted_max < score - 1e-10:
                return d
            # Even if predicted doesn't show improvement in LP, try the direction
            # (nonlinearity might help)
            if np.max(np.abs(d)) > 1e-8:
                return d
    except Exception:
        pass
    return None

def gradient_direction(h, G, c_scaled, max_active=20):
    """Fallback: negative gradient of active correlations, centered."""
    n = len(h)
    score = np.max(c_scaled)
    tol = max(1e-6, 0.05 * score)
    active = np.where(c_scaled >= score - tol)[0]
    if len(active) > max_active:
        order = np.argsort(c_scaled[active])[::-1]
        active = active[order[:max_active]]
    if len(active) == 0:
        active = np.array([np.argmax(c_scaled)])
    
    g = G[active, :].mean(axis=0)
    g -= np.mean(g)  # preserve sum
    if np.max(np.abs(g)) < 1e-12:
        return None
    return g / np.max(np.abs(g))

def line_search(h, d, score, n, max_steps=10):
    """Try steps along direction d, return best (h_new, score_new, alpha)."""
    best_h = h.copy()
    best_score = score
    best_alpha = 0.0
    alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    norm = np.max(np.abs(d))
    if norm < 1e-12:
        return best_h, best_score, best_alpha
    d = d / norm
    for sign in [1, -1]:
        for a in alphas:
            alpha = sign * a
            h_new = project_to_target(h + alpha * d, n)
            s = official_score(h_new)
            if s < best_score - 1e-13:
                best_score = s
                best_h = h_new
                best_alpha = alpha
    return best_h, best_score, best_alpha

def local_lp_descent(h, n, max_iters=60, time_limit=30.0, seed=42):
    """Main LP-based local descent."""
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    start = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        
        dx = 2.0 / n
        c_scaled = dx * np.correlate(h, 1.0 - h, mode="full")
        G = compute_grad_matrix(h)
        
        # Try LP direction
        d = lp_minimax_direction(h, G, c_scaled)
        
        # Fallback to gradient
        if d is None:
            d = gradient_direction(h, G, c_scaled)
        
        if d is None:
            # Random perturbation
            d = rng.normal(0, 1, n)
            d -= np.mean(d)
        
        h_new, s_new, alpha = line_search(h, d, score, n)
        
        if s_new < score - 1e-13:
            h = h_new
            score = s_new
        else:
            # Try gradient direction as backup
            d2 = gradient_direction(h, G, c_scaled)
            if d2 is not None:
                h2, s2, a2 = line_search(h, d2, score, n)
                if s2 < score - 1e-13:
                    h = h2
                    score = s2
                else:
                    break
            else:
                break
    
    return h, official_score(h)

def pairwise_exchange(h, n, max_rounds=50, time_limit=15.0, seed=42):
    """Local search via pairwise mass transfer."""
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    start = time.perf_counter()
    
    for _ in range(max_rounds):
        if time.perf_counter() - start > time_limit:
            break
        improved = False
        for _ in range(50):
            if time.perf_counter() - start > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for delta in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                for sign in [1, -1]:
                    h_try = h.copy()
                    h_try[i] += sign * delta
                    h_try[j] -= sign * delta
                    h_new = project_to_target(h_try, n)
                    s = official_score(h_new)
                    if s < score - 1e-13:
                        h = h_new
                        score = s
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, official_score(h)

def slsqp_polish(h, n, time_limit=20.0):
    """SLSQP minimax polish."""
    dx = 2.0 / n
    target_sum = n / 2.0
    L = 2 * n - 1
    score0 = official_score(h)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        hx = x[:n]
        corr = dx * np.correlate(hx, 1.0 - hx, mode="full")
        return x[n] - corr
    
    def ineq_jac(x):
        hx = x[:n]
        G = compute_grad_matrix(hx)
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        J[:, :n] = -G
        return J
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = score0 + 1e-6
    x0 = np.concatenate([project_to_target(h, n), [theta0]])
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = optimize.minimize(objective, x0, jac=objective_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 80, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x[:n])):
            h_out = project_to_target(res.x[:n], n)
            s = official_score(h_out)
            if s < score0 - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, score0

def discrete_sa(h_init, n, time_limit=30.0, seed=42):
    """Discrete (binary) simulated annealing for even n."""
    if n % 2 != 0:
        return None, None
    rng = np.random.default_rng(seed)
    target = n // 2
    
    # Initialize from h_init by thresholding
    h = np.zeros(n)
    sorted_idx = np.argsort(h_init)[::-1]
    h[sorted_idx[:target]] = 1.0
    
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    
    start = time.perf_counter()
    T = 0.5
    T_min = 0.01
    
    while time.perf_counter() - start < time_limit:
        # Propose move: swap a 1 with a 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[int(rng.integers(0, len(ones)))]
        j = zeros[int(rng.integers(0, len(zeros)))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_score = official_score(h)
        
        delta = new_score - score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            score = new_score
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

def interpolate(parent, target_n, offset=0):
    """Interpolate parent to target_n samples."""
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(target_n, 0.5), target_n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(target_n) + 0.5 + offset) % target_n) * 2.0 / target_n
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, target_n)

            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + r)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + r)
    return h


def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    idx = np.arange(n)
    for r in range(L):
        s = r - (n - 1)
        v1 = (idx - s >= 0) & (idx - s < n)
        if np.any(v1):
            G[r, v1] += dx * (1.0 - h[idx[v1] - s])
        v2 = (idx + s >= 0) & (idx + s < n)
        if np.any(v2):
            G[r, v2] += dx * (-h[idx[v2] + s])
    return G
