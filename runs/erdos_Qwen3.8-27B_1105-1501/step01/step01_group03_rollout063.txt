        if n - k > 0:

        if n - ka > 0:
            grad[:n-ka] += -h[ka:]
    return grad

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)

def solve_lp(n, time_limit=30):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    u = cp.Variable((n, n))
    
    constraints = [
        cp.sum(h_var) == n / 2,
        h_var >= 0, h_var <= 1,
        t_var >= 0,
        u <= h_var[:, None],
        u <= h_var[None, :],
        u >= h_var[:, None] + h_var[None, :] - 1,
        u >= 0,
    ]
    
    for k in range(-(n-1), n):
        valid_i = np.array([i for i in range(n) if 0 <= i+k < n])
        S = cp.sum(h_var[valid_i])
        j_idx = valid_i + k
        P = cp.sum(u[valid_i, j_idx])
        constraints.append((S - P) * dx <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    t0 = time.time()
    for solver in [cp.GUROBI, cp.MOSEK, cp.CBC, cp.GLPK, cp.ECOS]:
        if time.time() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate') and h_var.value is not None:
                return np.asarray(h_var.value).flatten().astype(float), float(t_var.value)
        except Exception:
            continue
    return None, None

def compute_F(h, n):
    """Compute F[k] for all k (without dx factor)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # index k+n-1 corresponds to shift k

def grad_smooth_max(h, n, beta):
    """Gradient of the smooth max of F[k]*dx."""
    dx = 2.0 / n
    F = compute_F(h, n)  # length 2n-1
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = es.sum()
    if ws < 1e-300:
        w = np.ones(2*n - 1) / (2*n - 1)
    else:
        w = es / ws
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if abs(wk) < 1e-15:
            continue
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += wk * (1.0 - h[k:n])
            if k < n:
                grad[k:] += wk * (-h[:n-k])
        else:
            ka = -k
            if n - ka > 0:
                grad[ka:] += wk * (1.0 - h[:n-ka])
            if n - ka > 0:
                grad[:n-ka] += wk * (-h[ka:])
    
    return grad * dx

def projected_gradient(h0, n, betas, max_iters, time_limit):
    """Projected gradient method with smooth max."""
    t0 = time.time()
    h = make_feasible(h0.copy())
    best_h = h.copy()
    best_score = official_score(h)
    
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        lr = 1.0 / (beta + n)  # initial learning rate
        for it in range(max_iters):
            if time.time() - t0 > time_limit:
                break
            g = grad_smooth_max(h, n, beta)
            # Backtracking line search
            for attempt in range(30):
                h_new = make_feasible(h - lr * g)
                sc = official_score(h_new)
                if sc < best_score or sc < official_score(h):
                    h = h_new
                    if sc < best_score:
                        best_score = sc
                        best_h = h.copy()
                    break
                lr *= 0.5
            if lr < 1e-15:
                break
    
    return best_h, best_score

def binary_rounding(h, n):
    """Round h to a binary sequence with exactly n/2 ones."""
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def binary_local_search(h_init, n, max_sweeps=50, time_limit=15.0):
    t0 = time.time()
    h = binary_rounding(h_init, n)
    best_score = official_score(h)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 1e-12
        best_swap = None
        
        for a in ones:
            for b in zeros:
                if time.time() - t0 > time_limit:
                    break
                h[a] = 0.0
                h[b] = 1.0
                sc = official_score(h)
                h[a] = 1.0
                h[b] = 0.0
                delta = best_score - sc
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (a, b)
        
        if best_swap is not None:
            a, b = best_swap
            h[a] = 0.0
            h[b] = 1.0
            best_score = official_score(h)
            best_h = h.copy()
            improved = True
        if not improved:
            break
    
    return best_h, best_score

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_F(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_smooth_max(h, n, beta):
    dx = 2.0 / n
    F = compute_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = es.sum()
    if ws < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = es / ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if abs(wk) < 1e-15:
            continue
        if k >= 0:
            if n - k > 0:
                grad[:n - k] += wk * (1.0 - h[k:n])
            if k < n:
                grad[k:] += wk * (-h[:n - k])
        else:
            ka = -k
            if n - ka > 0:
                grad[ka:] += wk * (1.0 - h[:n - ka])
            if n - ka > 0:
                grad[:n - ka] += wk * (-h[ka:])
    return grad * dx


def smooth_max_val(h, n, beta):
    dx = 2.0 / n
    F = compute_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = es.sum()
    if ws < 1e-300:
        return cmax
    return cmax + (mk + np.log(ws)) / beta


def solve_lp_reduced(n, K, time_limit=40):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2,
        h_var >= 0,
        h_var <= 1,
        t_var >= 0,
    ]
    for k in range(-K, K + 1):
        vi = np.array([i for i in range(n) if 0 <= i + k < n])
        if len(vi) == 0:
            continue
        vj = vi + k
        u_k = cp.Variable(len(vi))
        S = cp.sum(h_var[vi])
        P = cp.sum(u_k)
        constraints.append((S - P) * dx <= t_var)
        constraints.append(u_k <= h_var[vi])
        constraints.append(u_k <= h_var[vj])
        constraints.append(u_k >= h_var[vi] + h_var[vj] - 1)
        constraints.append(u_k >= 0)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    t0 = time.time()
    for solver in [cp.GUROBI, cp.MOSEK, cp.CBC, cp.GLPK, cp.ECOS]:
        if time.time() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate') and h_var.value is not None:
                return np.asarray(h_var.value).flatten().astype(float), float(t_var.value)
        except Exception:
            continue
    return None, None


def binary_rounding(h, n):
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b


def binary_local_search(h_init, n, max_sweeps=40, time_limit=15.0):
    t0 = time.time()
    h = binary_rounding(h_init, n)
    best_score = official_score(h)
    best_h = h.copy()
    for _ in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-12
        best_swap = None
        for a in ones:
            for b in zeros:
                if time.time() - t0 > time_limit:
                    break
                h[a] = 0.0
                h[b] = 1.0
                sc = official_score(h)
                h[a] = 1.0
                h[b] = 0.0
                delta = best_score - sc
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (a, b)
        if best_swap is not None:
            a, b = best_swap
            h[a] = 0.0
            h[b] = 1.0
            best_score = official_score(h)
            best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_score


def projected_gradient(h0, n, betas, max_iters, time_limit):
    t0 = time.time()
    h = make_feasible(h0.copy())
    best_h = h.copy()
    best_score = official_score(h)
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        lr = 2.0 / (beta * (2.0 / n) + 0.5)
        for it in range(max_iters):
            if time.time() - t0 > time_limit:
                break
            g = grad_smooth_max(h, n, beta)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            for attempt in range(40):
                h_new = make_feasible(h - lr * g)
                sc = official_score(h_new)
                if sc <= best_score:
                    h = h_new
                    if sc < best_score - 1e-15:
                        best_score = sc
                        best_h = h.copy()
                    break
                lr *= 0.5
            if lr < 1e-18:
                break
    return best_h, best_score


def eq_sum_fun(x):
    return float(np.sum(x) - len(x) / 2.0)


def eq_sum_jac(x):
    return np.ones(len(x))


def slsqp_smooth(x, beta, n):
    return smooth_max_val(x, n, beta)


def slsqp_smooth_grad(x, beta, n):
    return grad_smooth_max(x, n, beta)


def slsqp_refine(h0, n, betas, maxiter=150, time_limit=60):
    t0 = time.time()
    h = make_feasible(h0.copy())
    best_h = h.copy()
    best_score = official_score(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum_fun, 'jac': eq_sum_jac}]
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(slsqp_smooth, h, args=(beta, n),
                           jac=slsqp_smooth_grad, bounds=bounds,
                           constraints=cons, method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = make_feasible(res.x)
                sc = official_score(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
                    h = h_new.copy()
        except Exception:
            pass
    return best_h, best_score


def interp_to_length(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)
