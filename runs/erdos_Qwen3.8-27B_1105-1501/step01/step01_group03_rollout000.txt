def lp_local_step(h, radius, lam, max_iters=3):
    """Tangent LP local step with exact acceptance. Returns improved h or None."""
    n = len(h)
    C_vals = compute_all_C(h)
    G = compute_all_gradients(h)
    cur_score = float(np.max(C_vals))
    m = 2 * n - 1
    
    # Build LP: vars = [d(n), p(n), t(1)]
    # min t + (lam/n) * sum(p_i)
    # s.t. G@d - t <= -C (m rows)
    #      d_i - p_i <= 0 (n rows)
    #      -d_i - p_i <= 0 (n rows)
    #      sum(d) = 0 (equality)
    #      bounds on d, p, t
    
    c_obj = np.zeros(2 * n + 1)
    c_obj[n:2*n] = lam / n
    c_obj[2*n] = 1.0
    
    A_ub = np.zeros((m + 2 * n, 2 * n + 1))
    b_ub = np.zeros(m + 2 * n)
    
    # Linearization: G@d - t <= -C
    A_ub[:m, :n] = G
    A_ub[:m, 2*n] = -1.0
    b_ub[:m] = -C_vals
    
    # d_i - p_i <= 0
    for i in range(n):
        A_ub[m + i, i] = 1.0
        A_ub[m + i, n + i] = -1.0
        b_ub[m + i] = 0.0
    
    # -d_i - p_i <= 0
    for i in range(n):
        A_ub[m + n + i, i] = -1.0
        A_ub[m + n + i, n + i] = -1.0
        b_ub[m + n + i] = 0.0
    
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lo = np.maximum(-radius, -h - 1e-10)
    hi = np.minimum(radius, 1.0 - h + 1e-10)
    bounds = list(zip(lo, hi)) + [(0.0, radius + 1e-10)] * n + [(0.0, 2.0)]
    
    best_h = h.copy()
    best_sc = cur_score
    
    for it in range(max_iters):
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if res is None or not res.success:
            break
        if res.x is None:
            break
        
        d = res.x[:n]
        # Re-center to ensure sum is exactly 0
        d = d - np.mean(d)
        
        # Try multiple step sizes
        for alpha in [1.0, 0.6, 0.35, 0.15]:
            h_try = project_feasible(h + alpha * d)
            if not np.all(np.isfinite(h_try)):
                continue
            sc = official_score(h_try)
            if np.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best_h = h_try.copy()
                break
        if best_sc < cur_score - 1e-14:
            # Shrink radius for next iteration (model is now tighter)
            radius *= 0.6
            # Update C_vals and G for next iteration
            break
        else:
            radius *= 0.4
            break
    
    if official_score(best_h) < official_score(h) - 1e-15:
        return best_h
    return None


def smooth_solve(h0, beta, maxiter=150):
    """Smooth max SLSQP fallback."""
    n = len(h0)
    dx = 2.0 / n
    
    def obj(x):
        F = compute_all_C(x)
        cmax = np.max(F)
        shifted = beta * (F - cmax)
        mk = np.max(shifted)
        es = np.exp(shifted - mk)
        ws = np.sum(es)
        if ws < 1e-300:
            return cmax + np.log(2 * n - 1) / beta
        return cmax + (mk + np.log(ws)) / beta
    
    h0f = project_feasible(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0f, jac=None, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x)
    except Exception:
        pass
    return h0f


def area_transfer(h_src, n_src, n_tgt):
    """Area-preserving dimension transfer."""
    if n_src == n_tgt:
        return project_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    # Build piecewise constant and integrate
    h_step = np.empty(n_tgt)
    for j in range(n_tgt):
        x_lo = j * 2.0 / n_tgt
        x_hi = (j + 1) * 2.0 / n_tgt
        # Find source points in this interval
        mask = (x_src >= x_lo) & (x_src < x_hi)
        if np.any(mask):
            # Weighted average of source values in interval
            total_weight = np.sum(2.0 / n_src)
            h_step[j] = np.sum(h_src[mask] * (2.0 / n_src)) / max(x_hi - x_lo, 1e-15)
        else:
            # Fall back to interpolation
            h_step[j] = np.interp(x_tgt[j], x_src, h_src)
    return project_feasible(h_step)


def binary_from_h(h, rng):
    """Convert to binary (or near-binary) feasible vector."""
    n = len(h)
    target = n / 2.0
    if abs(target - int(target)) < 1e-10:
        # Even n: exactly n/2 ones
        k = int(target)
        idx = np.zeros(n)
        top_k = np.argsort(h)[::-1][:k]
        idx[top_k] = 1.0
        return idx
    else:
        # Odd n: n//2 ones, one at 0.5
        k = int(target)  # floor
        idx = np.zeros(n)
        top_k = np.argsort(h)[::-1][:k]
        idx[top_k] = 1.0
        remaining = np.where(idx == 0.0)[0]
        idx[remaining[0]] = 1.0 - (k - np.floor(target))  # fractional
        return idx


def sa_binary(h_bin, rng, steps=20000, t_start=0.01, t_end=0.001):
    """Simulated annealing on binary vectors."""
    n = len(h_bin)
    ones = np.where(h_bin >= 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    cur = h_bin.copy()
    cur_sc = official_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    cooling = (t_end / t_start) ** (1.0 / max(steps, 1))
    
    for s in range(steps):
        T = t_start * (cooling ** s)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.integers(0, len(ones))
        j = rng.integers(0, len(zeros))
        oi = ones[i]
        oz = zeros[j]
        # Propose swap
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_sc = official_score(cur)
        if new_sc < cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / max(T, 1e-10)):
            cur_sc = new_sc
            ones[i] = oz
            zeros[j] = oi
            if new_sc < best_sc - 1e-15:
                best = cur.copy()
                best_sc = new_sc
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
    return best


def random_perturb(h, sigma, rng):

    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(2 * n - 1) / beta
    return cmax + (mk + np.log(ws)) / beta


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(30.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    if parent is not None:
        parent = project_feasible(parent)
        parent_n = len(parent)
        best_h = parent.copy()
        best_score = official_score(best_h)
    else:
        parent_n = 60
        best_h = project_feasible(np.full(parent_n, 0.5))
        best_score = official_score(best_h)

    def try_update(h_cand):
        nonlocal best_h, best_score
        h_cand = project_feasible(h_cand)
        if not np.all(np.isfinite(h_cand)):
            return
        sc = official_score(h_cand)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc

    def elapsed():
        return time.time() - t_start

    def time_left():
        return deadline - time.time()

    # Phase 1: Immediate LP refinement of parent at its length
    if time_left() > 60:
        for radius in [0.3, 0.15, 0.08, 0.04]:
            if time_left() < 30:
                break
            h_try = lp_local_step(best_h, radius, lam=0.05, max_iters=5)
            if h_try is not None:
                try_update(h_try)

    # Phase 2: Dimension sweep with LP
    lengths = sorted(set([
        max(24, parent_n - 8), max(24, parent_n - 4), max(24, parent_n - 2),
        parent_n, parent_n + 2, parent_n + 4, parent_n + 8
    ]))
    lengths = [l for l in lengths if 24 <= l <= 80]

    for n in lengths:
        if time_left() < 45:
            break
        h_n = area_transfer(best_h, len(best_h), n)
        try_update(h_n)
        
        # LP refinement at this length
        for radius in [0.25, 0.1]:
            if time_left() < 30:
                break
            h_try = lp_local_step(h_n, radius, lam=0.03, max_iters=4)
            if h_try is not None:
                h_n = h_try
                try_update(h_n)
        
        # Smooth fallback
        if time_left() > 20:
            h_s = smooth_solve(h_n, beta=200, maxiter=100)
            try_update(h_s)
            # LP again after smooth
            h_try = lp_local_step(h_s, 0.1, lam=0.03, max_iters=3)
            if h_try is not None:
                try_update(h_try)

    # Phase 3: Binary SA exploration
    if time_left() > 80:
        for n in lengths[:4]:
            if time_left() < 40:
                break
            h_n = area_transfer(best_h, len(best_h), n)
            h_bin = binary_from_h(h_n, rng)
            # Run SA
            h_sa = sa_binary(h_bin, rng, steps=min(15000, int(time_left() * 200)),
                            t_start=0.008, t_end=0.0005)
            try_update(h_sa)
            # LP from binary
            if time_left() > 20:
                h_try = lp_local_step(h_sa, 0.2, lam=0.02, max_iters=5)
                if h_try is not None:
                    try_update(h_try)

    # Phase 4: Random perturbations + LP
    if time_left() > 60:
        for sigma in [0.03, 0.08, 0.15, 0.25]:
            if time_left() < 30:
                break
            for _ in range(3):
                if time_left() < 20:
                    break
                h_p = random_perturb(best_h, sigma, rng)
                # LP from perturbed
                h_try = lp_local_step(h_p, 0.15, lam=0.03, max_iters=4)
                if h_try is not None:
                    try_update(h_try)
                try_update(h_p)

    # Phase 5: Intensive local refinement
    if time_left() > 50:
        n_best = len(best_h)
        h_cur = best_h.copy()
        for radius in [0.15, 0.08, 0.04, 0.02, 0.01]:
            if time_left() < 20:
                break
            for _ in range(3):
                if time_left() < 15:
                    break
                h_try = lp_local_step(h_cur, radius, lam=0.02, max_iters=5)
                if h_try is not None:
                    h_cur = h_try
                    try_update(h_cur)
                else:
                    break

    # Phase 6: More dimensions after refinement
    if time_left() > 40:
        for dl in [-3, -1, 1, 3, 5, 7]:
            if time_left() < 20:
                break
            n_try = len(best_h) + dl
            if 24 <= n_try <= 80:
                h_t = area_transfer(best_h, len(best_h), n_try)
                try_update(h_t)
                h_try = lp_local_step(h_t, 0.12, lam=0.02, max_iters=4)
                if h_try is not None:
                    try_update(h_try)

    # Phase 7: Final deep local search
    if time_left() > 30:
        for radius in [0.06, 0.03, 0.015, 0.005]:
            if time_left() < 15:
                break
            h_try = lp_local_step(best_h, radius, lam=0.01, max_iters=8)
            if h_try is not None:
                try_update(h_try)

    # Final validation
    best_h = project_feasible(best_h)
    final_score = official_score(best_h)
    
    # Ensure we don't return worse than parent
    if parent is not None:
        parent_score = official_score(parent)
        if final_score > parent_score + 1e-12:
            best_h = parent.copy()
            final_score = parent_score

    return best_h, final_score, len(best_h)
