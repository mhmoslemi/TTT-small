def compute_swap_delta(h, a, b, n):
    """Compute delta in overlap array when swapping h[a]=0->1 and h[b]=1->0."""
    delta = np.zeros(2*n - 1)
    
    # For position a (0->1):
    # a+L valid: L from -a to n-1-a
    L_start_a = -a
    L_end_a = n - 1 - a
    if L_start_a <= L_end_a:
        idx_start = L_start_a + n - 1
        idx_end = L_end_a + n - 1
        # For these L values, a+L ranges from 0 to n-1
        # 1-h[a+L] for a+L in [0, n-1]
        delta[idx_start:idx_end+1] += (1.0 - h[a + np.arange(L_start_a, L_end_a+1)])
    
    # a-L valid: L from 1-n+a to a
    L_start_a2 = 1 - n + a
    L_end_a2 = a
    if L_start_a2 <= L_end_a2:
        idx_start = L_start_a2 + n - 1
        idx_end = L_end_a2 + n - 1
        # a-L ranges from a to n-1 (when L goes from a to 1-n+a)
        # Actually when L = L_start_a2 = 1-n+a, a-L = a-(1-n+a) = n-1
        # When L = L_end_a2 = a, a-L = 0
        # So a-L ranges from n-1 down to 0 as L goes from L_start_a2 to L_end_a2
        a_minus_L = a - np.arange(L_start_a2, L_end_a2+1)
        delta[idx_start:idx_end+1] -= h[a_minus_L]
    
    # For position b (1->0):
    L_start_b = -b
    L_end_b = n - 1 - b
    if L_start_b <= L_end_b:
        idx_start = L_start_b + n - 1
        idx_end = L_end_b + n - 1
        delta[idx_start:idx_end+1] -= (1.0 - h[b + np.arange(L_start_b, L_end_b+1)])
    
    L_start_b2 = 1 - n + b
    L_end_b2 = b
    if L_start_b2 <= L_end_b2:
        idx_start = L_start_b2 + n - 1
        idx_end = L_end_b2 + n - 1
        b_minus_L = b - np.arange(L_start_b2, L_end_b2+1)
        delta[idx_start:idx_end+1] += h[b_minus_L]
    
    return delta

    current_max = float(np.max(overlap))
    best_max = current_max
    best_h = h.copy()
    
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for it in range(iters):
        T = T0 * (T1/T0) ** (it / max(iters-1, 1))
        a = zeros[rng.integers(len(zeros))]
        b = ones[rng.integers(len(ones))]
        
        delta = compute_swap_delta(h, a, b, n)
        new_max = float(np.max(overlap + delta))
        
        if new_max < current_max or rng.random() < np.exp(-(new_max - current_max) / max(T, 1e-10)):
            h[a] = 1.0
            h[b] = 0.0

            current_max = new_max
            
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
            
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
    
    return best_h, best_max
