def make_feasible(y, n=None):
    if n is None:
        n = len(y)

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(z))
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            rem = diff - add
            if abs(rem) > 1e-13:

                    if i == idx:
                        continue
                    a = min(rem, 1.0 - z[i])
                    z[i] += a
                    rem -= a

    obj[n] = 1.0

        g = lag_gradient(h, m, n)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -scores[m]

        bounds.append((lo, hi))

            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            if not improved:
                break

    # Phase 5: Try alternative resolutions (remaining time)
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        for dn in [-5, -4, 4, 5]:
            if time.time() > deadline - 10:
                break
            nr = n_cur + dn
            if nr < 20 or nr > 150:
                continue
            h_r = resample_h(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_deadline = min(deadline - 5, time.time() + 30)
            # Quick LP refinement on new resolution
            radius_r = 0.1
            for _ in range(200):
                if time.time() > sub_deadline:
                    break
                scores_r = compute_scores(h_r)
                na = min(8, len(scores_r))
                top_r = np.argsort(scores_r)[::-1][:na]
                delta_r = lp_trust_region(h_r, scores_r, top_r, radius_r, nr)
                if delta_r is not None:
                    cand_r, cand_rc, _ = backtrack_line_search(h_r, delta_r, c_r, nr)
                    if cand_r is not None:
                        h_r, c_r = cand_r, cand_rc
                else:
                    break
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    # Final: ensure feasibility and compute exact score
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
