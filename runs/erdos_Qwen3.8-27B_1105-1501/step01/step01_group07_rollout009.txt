import numpy as np
import time

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    # Bisection on threshold
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(z)
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(z)
    if abs(diff) > 1e-14:
        if diff > 0:
            # Add to smallest values
            order = np.argsort(z)
            for idx in order:
                a = min(diff, 1.0 - z[idx])
                z[idx] += a
                diff -= a
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(-z)
            for idx in order:
                a = min(-diff, z[idx])
                z[idx] -= a
                diff += a
                if abs(diff) < 1e-14:
                    break
    return np.clip(z, 0.0, 1.0)

    S = float(np.sum(ex))

def golden_section(f, lo, hi, tol=1e-10, max_iter=60):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0, f((a + b) / 2.0)

def pair_search(h, i, j, cur_c):
    orig_i, orig_j = h[i], h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    h[i] = orig_i
    h[j] = orig_j
    
    def f(a):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        h[i] = orig_i
        h[j] = orig_j
        return s
    
    a_star, f_star = golden_section(f, lo, hi, tol=1e-9, max_iter=50)
    if f_star < cur_c - 1e-14 and abs(a_star) > 1e-12:
        h[i] = orig_i + a_star
        h[j] = orig_j - a_star
        return float(f_star)
    return None

def coord_search(h, i, cur_c):
    orig = h[i]
    lo, hi = -orig, 1.0 - orig
    if hi - lo < 1e-12:
        return None
    
    def f(a):
        h[i] = orig + a
        s = score_exact(h)
        h[i] = orig
        return s
    
    a_star, f_star = golden_section(f, lo, hi, tol=1e-9, max_iter=50)
    if f_star < cur_c - 1e-14 and abs(a_star) > 1e-12:
        h[i] = orig + a_star
        return float(f_star)
    return None

def multi_coord_search(h, indices, cur_c, n_iters=5):
    """Cyclic coordinate search over multiple indices."""
    cur_c = cur_c
    for _ in range(n_iters):
        improved = False
        for i in indices:
            res = coord_search(h, i, cur_c)
            if res is not None:
                cur_c = res
                improved = True
        if not improved:

    return cur_c

def make_feasible(y, n=None):
    if n is None:

    y = np.asarray(y, dtype=np.float64).copy()

    lo = float(np.min(y)) - 1.0

        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(z)
            for idx in order:
                a = min(diff, 1.0 - z[idx])
                z[idx] += a
                diff -= a
                if abs(diff) < 1e-14:

        else:

            fc = fd
