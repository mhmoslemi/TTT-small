    n = len(h)

Let me also think about the Nesterov acceleration. The standard Nesterov method for minimizing a convex function f with subgradient ∂f is:

t_{k+1} = (1 + sqrt(1 + 4*t_k²)) / 2

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - z.max()
    hi = 2.0 - z.min()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
    elif shift > 0:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] += -h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] += -h[:n - s]
    return g

def compute_all_grads(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for shift in range(-(n - 1), n):
        k = shift + n - 1
        grads[k] = dx * grad_for_shift(h, shift)
    return grads

def subgradient_opt(h0, deadline, max_iter=3000, verbose=False):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = official_score(h)
    alpha = 0.02
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = raw_corr(h)
        max_c = np.max(corr)
        if max_c * dx < best_score - 1e-14:
            best_score = max_c * dx
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                alpha *= 0.7
                stall = 0
                if alpha < 1e-6:
                    break

        tol_active = 0.15 * dx
        active_mask = corr >= max_c - tol_active
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])

        active_corr = corr[active_idx]
        w = np.exp(50.0 * (active_corr - max_c))
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(len(active_idx)) / len(active_idx)
        else:
            w /= w_sum

        g = np.zeros(n)
        for idx, k in enumerate(active_idx):
            g += w[idx] * dx * grad_for_shift(h, k - (n - 1))

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break

        a_try = alpha
        improved = False
        for _ in range(15):
            h_new = project_feasible(h - a_try * g)
            s_new = official_score(h_new)
            if s_new < max_c * dx - 1e-14:
                h = h_new
                alpha = min(a_try * 1.1, 0.1)
                improved = True
                break
            a_try *= 0.5

        if not improved and stall > 500:
            break

    return best_h, best_score

def cvx_active_qp(h0, deadline, top_k=15, trust_radius=0.15):
    n = len(h0)
    dx = 2.0 / n
    corr = raw_corr(h0)
    max_c = np.max(corr)

    order = np.argsort(corr)[::-1]
    active = order[:min(top_k, 2 * n - 1)]

    ref_vals = corr[active].copy()
    grads_a = np.zeros((len(active), n))
    for i, k in enumerate(active):
        grads_a[i] = dx * grad_for_shift(h0, k - (n - 1))

    t0_val = max_c + 0.01
    delta = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        cp.sum(delta) == 0,
        delta >= -h0,
        delta <= 1.0 - h0,
        cp.sum_squares(delta) <= trust_radius * trust_radius,
    ]

    for i in range(len(active)):
        constraints.append(
            ref_vals[i] + grads_a[i] @ delta + 0.5 * cp.sum_squares(delta) <= t_var
        )

    obj = cp.Minimize(t_var + 0.001 * cp.sum_squares(delta))
    prob = cp.Problem(obj, constraints)

    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            return h0.copy()

    if delta.value is None or not np.all(np.isfinite(delta.value)):
        return h0.copy()

    h_new = project_feasible(h0 + delta.value)
    s_new = official_score(h_new)
    if s_new < official_score(h0):
        return h_new
    return h0.copy()

def pair_descent_polish(h0, deadline, n_sweeps=50):

            x0 = h[i]
