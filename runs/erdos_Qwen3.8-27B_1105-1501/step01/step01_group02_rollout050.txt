def pair_line_search(h, i, j, n_evals=7):
    """For pair (i,j), find best δ minimizing C5.
    new_h[i] = h[i] - δ, new_h[j] = h[j] + δ
    Returns (best_h, best_c5, best_delta)

    lo = max(h[i] - 1.0, -h[j])

    h_copy = h.copy()

import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def score_pairs(h, top_k=5):
    """Score all pairs based on their effect on top-k active lags.
    Returns array of scores, shape (n, n, 2) where [i,j,0] is score for
    decreasing i and increasing j, [i,j,1] is score for decreasing j and increasing i.
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top-k lag indices
    top_indices = np.argsort(corr)[-top_k:]
    
    scores = np.zeros((n, n, 2))
    for s_idx in top_indices:
        s = s_idx - (n - 1)  # shift
        # For shifting, compute the effect on each index
        # Effect of changing h[idx] by δ on corr[s_idx]:
        # δ * (1 - h[idx+s]) - h[idx-s] * δ (with boundary handling)
        effect = np.zeros(n)
        for idx in range(n):
            # Term from h[idx] in the product: h[idx] * (1-h[idx+s])
            # If we increase h[idx] by δ: effect = δ*(1-h[idx+s]) if idx+s valid
            # Term from (1-h[idx]) : h[idx-s] * (1-h[idx])
            # If we increase h[idx] by δ: effect = -h[idx-s]*δ if idx-s valid
            e = 0.0
            if 0 <= idx + s < n:
                e += (1.0 - h[idx + s])
            if 0 <= idx - s < n:
                e -= h[idx - s]
            effect[idx] = e
        
        # For pair (i,j), decreasing i by δ and increasing j by δ:
        # Change in corr = δ * (effect[j] - effect[i])
        # We want this to be negative, so we want effect[j] < effect[i]
        # Score for direction (i→j): effect[i] - effect[j] (higher is better)
        # Score for direction (j→i): effect[j] - effect[i]
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                scores[i, j, 0] += effect[i] - effect[j]
                scores[i, j, 1] += effect[j] - effect[i]
    
    return scores

def make_effect_array(h, s):
    n = len(h)
    eff = np.zeros(n)
    for idx in range(n):
        e = 0.0
        if 0 <= idx + s < n:
            e += (1.0 - h[idx + s])
        if 0 <= idx - s < n:
            e -= h[idx - s]
        eff[idx] = e
    return eff


def try_pair(h, i, j, cur_c5, n_delta=12):
    """Try decreasing h[i] by d and increasing h[j] by d. Returns (best_h, best_c5, best_d)."""
    n = len(h)

    d_hi = min(h[i], 1.0 - j >= 0 and 1.0 - h[j] or 1.0 - h[j], 1.0)
    d_hi = min(h[i], 1.0 - h[j])

    best_c5 = cur_c5
