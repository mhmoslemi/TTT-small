For Phase 3, the QP formulation:
minimize z
subject to: c_k(h) ≤ z for k in active set
           sum(h) = n/2
           0 ≤ h ≤ 1

where c_k(h) = sum_i h[i]*(1-h[i+offset_k]) is quadratic in h.

This is a convex QP if the quadratic form is convex. But c_k is NOT convex in general (it has both positive and negative quadratic terms). So this is a non-convex QP.

However, if we linearize c_k around the current point h₀:
c_k(h) ≈ c_k(h₀) + ∇c_k(h₀) · (h - h₀)

Then the problem becomes a linear program:
minimize z
subject to: ∇c_k(h₀) · h ≤ z - c_k(h₀) + ∇c_k(h₀) · h₀ for k in active set
           sum(h) = n/2
           0 ≤ h ≤ 1

This LP can be solved efficiently. Then we check if the solution satisfies the original (nonlinear) constraints, and if so, accept it. If not, update the active set and re-solve.

This is essentially a sequential linearization approach (like the Gauss-Newton method but for constraints).

def _grad_soft(h, T, top_k=30):

    dx = 2.0 / n
    c = _corr(h)
    top_idx = np.argsort(c)[::-1][:top_k]

def _lp_refine(h, n, active_lags, max_iter=20):
    """LP-based active set refinement."""
    h_cur = h.copy()
    for it in range(max_iter):
        c = _corr(h_cur)
        # Check if solution is valid (all corrections within tolerance)
        # Build LP:
        # Variables: h[0],...,h[n-1], z (n+1 variables)
        # Minimize: z
        # Subject to:
        #   grad_c_k · h - z <= -c_k(h_cur) + grad_c_k · h_cur  for each active k
        #   sum(h) = n/2
        #   0 <= h <= 1
        #   z >= 0 (implicit from constraints)
        
        m = len(active_lags)
        # Inequality constraints: A_ub @ x <= b_ub
        A_rows = []
        b_rows = []
        for k in active_lags:
            gk = _grad_corr(h_cur, k)
            ck = c[k]
            # Constraint: gk · h - z <= ck - gk · h_cur  (linearization of c_k(h) <= z)
            # gk · (h - h_cur) + ck <= z
            # gk · h - z <= -ck + gk · h_cur
            row = np.zeros(n + 1)
            row[:n] = gk
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-ck + np.dot(gk, h_cur))
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 <= h <= 1, z >= 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        # Objective: minimize z
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        z_new = res.x[n]
        
        # Check if the LP solution actually satisfies the original (nonlinear) constraints
        c_new = _corr(h_new)
        max_active_new = max(c_new[k] for k in active_lags)
        
        # Also check non-active lags
        max_all_new = np.max(c_new)
        
        if max_active_new <= z_new + 1e-8:
            # Linearization is accurate; accept
            h_cur = h_new
            # Update active set
            top_idx = np.argsort(c_new)[::-1][:max(m, 10)]
            active_lags = list(top_idx)
            # Check convergence
            if max_all_new <= z_new * 1.001 + 1e-10:
                break
        else:
            # Linearization is not accurate; take a step toward the LP solution
            alpha = 0.5
            h_try = _project((1 - alpha) * h_cur + alpha * h_new, n)
            if _score(h_try) < _score(h_cur):

                c_new = _corr(h_cur)
                top_idx = np.argsort(c_new)[::-1][:max(m, 10)]
                active_lags = list(top_idx)
            else:
                break
    
    return h_cur

def _grad_descent(h, deadline, T_start=0.1, T_end=0.001, T_decay=0.9997,
                  step0=0.02, top_k=30, rng=None):
    """Smoothed gradient descent."""
    n = len(h)
    cur = h.copy()
    cur_c = _score(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    
    while time.time() < deadline:
        g = _grad_soft(cur, T, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        
        direction = -g / gn
        improved = False
        for s in (step, step*0.5, step*0.25, step*0.12, step*0.05):
            cand = _project(cur + s * direction, n)
            c_new = _score(cand)
            if c_new < cur_c - 1e-15:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 50:
                if rng is not None:
                    cur = _project(cur + rng.normal(0, max(0.003, T*0.5), n), n)
                    cur_c = _score(cur)
                step = max(T, 0.003)
                stall = 0
        else:
            step = min(step * 1.03, 0.1)
            stall = 0
        
        T = max(T_end, T * T_decay)
    
    return best, best_c

def _fourier_init(n, a1=0.3, a2=0.15, a3=0.05):
    """Fourier-based initialization."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.cos(2 * np.pi * x) + a2 * np.cos(4 * np.pi * x + np.pi/3) + a3 * np.cos(6 * np.pi * x)
    return _project(h, n)

def _binary_local_search(h_init, deadline, max_iters=5000, rng=None):
    """Local search on binary sequences."""
    n = len(h_init)
    # Start from binary projection
    h_bin = (h_init >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.sum(h_bin)
    if ones > n/2:
        idx = np.where(h_bin == 1)[0]
        rng.shuffle(idx)
        for i in idx[:int(ones - n/2)]:
            h_bin[i] = 0
    elif ones < n/2:
        idx = np.where(h_bin == 0)[0]
        rng.shuffle(idx)
        for i in idx[:int(n/2 - ones)]:
            h_bin[i] = 1
    
    cur_c = _score(h_bin)
    best = h_bin.copy()
    best_c = cur_c
    
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # Try random pair swap
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c_new = _score(h_bin)
        if c_new < cur_c - 1e-12:
            cur_c = c_new
            if c_new < best_c:
                best = h_bin.copy()
                best_c = c_new
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    
    return best, best_c
