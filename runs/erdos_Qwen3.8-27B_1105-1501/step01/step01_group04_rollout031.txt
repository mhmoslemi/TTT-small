    hi = 2.0

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, target):
    lo = float(z.min()) - 1.5
    hi = float(z.max()) + 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    y = np.clip(z - alpha, 0.0, 1.0)
    # Fix residual by adjusting one coordinate
    res = target - np.sum(y)
    if abs(res) > 1e-12:
        if res > 0:
            slack = 1.0 - y
            order = np.argsort(-slack)
            for idx in order:
                add = min(res, slack[idx])
                if add <= 0: break
                y[idx] += add
                res -= add
                if res < 1e-12: break
        else:
            order = np.argsort(-y)
            for idx in order:
                rem = min(-res, y[idx])
                if rem <= 0: break
                y[idx] -= rem
                res += rem
                if res > -1e-12: break
    return y

def pair_slopes(h, i, j):
    n = len(h)
    d = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        L = p - (n - 1)
        val = 0.0
        if 0 <= i - L < n:
            val += (1.0 - h[i - L])
        if 0 <= i + L < n:
            val -= h[i + L]
        if 0 <= j - L < n:
            val -= (1.0 - h[j - L])
        if 0 <= j + L < n:
            val += h[j + L]
        d[p] = val
    return d

def exact_line_min(c0, d, t_lo, t_hi):
    """Minimize max_k(c0[k] + t*d[k]) over t in [t_lo, t_hi]. Returns (best_t, best_val)."""
    # Convex piecewise linear: evaluate at breakpoints
    # Quick: coarse grid + refine
    n_pts = 80
    ts = np.linspace(t_lo, t_hi, n_pts)
    vals = np.max(c0[None, :] + ts[:, None] * d[None, :], axis=1)
    best_idx = np.argmin(vals)
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    # Refine
    if 0 < best_idx < n_pts - 1:
        ts2 = np.linspace(ts[best_idx - 1], ts[best_idx + 1], 40)
        vals2 = np.max(c0[None, :] + ts2[:, None] * d[None, :], axis=1)
        idx2 = np.argmin(vals2)
        best_t = ts2[idx2]
        best_val = vals2[idx2]
    # Check endpoints and t=0
    val_0 = np.max(c0)
    if t_lo <= 0 <= t_hi and val_0 < best_val:
        best_t = 0.0
        best_val = val_0
    return best_t, best_val

def resample(h_old, n_new):
    """Resample h_old (n_old points on [0,2]) to n_new points on [0,2]."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    # h_old represents a step function: h_old[i] is the value on [i*dx_old, (i+1)*dx_old)
    # Sample at the centers of the new cells
    centers_new = (np.arange(n_new) + 0.5) * dx_new
    # Convert to old cell indices
    old_idx = np.floor(centers_new / dx_old).astype(int)
    old_idx = np.clip(old_idx, 0, n_old - 1)
    h_new = h_old[old_idx].copy()
    
    # Fix sum
    target = float(n_new) / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

                grad[:n-Lp] += oh[Lp:]    # (1-h[i-L]) = (1-h[i+Lp]) for i in [0, n-Lp)
                grad[Lp:] -= h[:n-Lp]     # -h[i+L] = -h[i-Lp] for i in [Lp, n)
    return grad / max(len(active), 1)

For L >= 0, i in [0, n-L): i+L is in [L, n), so h[i+L] = h[L+i]. The array h[L:] has elements h[L], h[L+1], ..., h[n-1]. Placed at positions 0, 1, ..., n-L-1. So grad[:n-L] -= h[L:]. ✓

Wait, one more thought on the line search. For a convex piecewise-linear function with M pieces, the minimum is at a breakpoint. The breakpoints occur where two of the M linear functions cross. There are at most M*(M-1)/2 breakpoints, but most will be outside the feasible interval.

A more efficient approach: since the function is convex, I can use a binary search on the slope. The slope at t is:
- If there's a unique argmax k*, the slope is d[k*]
- If there are multiple argmax, the subgradient is the convex hull of their d values

So I can:
1. Evaluate at t=0, find argmax, get slope
2. If slope < 0, move right (increase t)
3. If slope > 0, move left
4. Binary search to find where slope crosses 0

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(z, target):
    lo = float(z.min()) - 1.5
    hi = float(z.max()) + 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    y = np.clip(z - alpha, 0.0, 1.0)
    res = target - np.sum(y)
    if abs(res) > 1e-12:
        if res > 0:
            slack = 1.0 - y
            order = np.argsort(-slack)
            for idx in order:
                add = min(res, slack[idx])
                if add <= 0:
                    break
                y[idx] += add
                res -= add
                if res < 1e-12:
                    break
        else:
            order = np.argsort(-y)
            for idx in order:
                rem = min(-res, y[idx])
                if rem <= 0:
                    break
                y[idx] -= rem
                res += rem
                if res > -1e-12:
                    break
    return y


def compute_grad_v(h, active):
    n = len(h)
    grad = np.zeros(n)
    oh = 1.0 - h
    for p in active:
        L = p - (n - 1)
        if L >= 0:
            if n - L > 0:
                grad[L:] += oh[:n - L]
                grad[:n - L] -= h[L:]
        else:
            Lp = -L
            if n - Lp > 0:
                grad[:n - Lp] += oh[Lp:]
                grad[Lp:] -= h[:n - Lp]
    return grad / max(len(active), 1)


def pair_slopes(h, i, j):
    n = len(h)
    d = np.zeros(2 * n - 1)
    oh = 1.0 - h
    for p in range(2 * n - 1):
        L = p - (n - 1)
        val = 0.0
        if 0 <= i - L < n:
            val += oh[i - L]
        if 0 <= i + L < n:
            val -= h[i + L]
        if 0 <= j - L < n:
            val -= oh[j - L]
        if 0 <= j + L < n:
            val += h[j + L]
        d[p] = val
    return d


def exact_line_min(c0, d, t_lo, t_hi):
    n_pts = 80
    ts = np.linspace(t_lo, t_hi, n_pts)
    vals = np.max(c0[None, :] + ts[:, None] * d[None, :], axis=1)
    best_idx = np.argmin(vals)
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    if 0 < best_idx < n_pts - 1:
        ts2 = np.linspace(ts[best_idx - 1], ts[best_idx + 1], 40)
        vals2 = np.max(c0[None, :] + ts2[:, None] * d[None, :], axis=1)
        idx2 = np.argmin(vals2)
        if vals2[idx2] < best_val:
            best_t = ts2[idx2]
            best_val = vals2[idx2]
    val0 = np.max(c0)
    if t_lo <= 0.0 <= t_hi and val0 < best_val:
        best_t = 0.0
        best_val = val0
    return best_t, best_val


def propose_pairs(h, c0, active, rng, n):
    pairs = []
    grad = compute_grad_v(h, active)
    k_sel = max(5, n // 6)
    sources = np.argsort(grad)[-k_sel:]
    targets = np.argsort(grad)[:k_sel]
    for s in sources:
        for t in targets:
            if s != t and h[s] > 0.01 and h[t] < 0.99:
                pairs.append((t, s))
    for p in active[:6]:
        L = p - (n - 1)
        if L >= 0 and n - L > 2:
            contrib = h[L:] * (1.0 - h[:n - L])
            top_m = np.argsort(contrib)[-4:]
            for m in top_m:
                src = m + L
                dst = m
                if src != dst and h[src] > 0.05 and h[dst] < 0.95:
                    pairs.append((dst, src))
        elif L < 0:
            Lp = -L
            if n - Lp > 2:
                contrib = h[:n - Lp] * (1.0 - h[Lp:])
                top_m = np.argsort(contrib)[-4:]
                for m in top_m:
                    src = m
                    dst = m + Lp
                    if src != dst and h[src] > 0.05 and h[dst] < 0.95:
                        pairs.append((dst, src))
    for _ in range(25):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            pairs.append((j, i))
    for i in range(n - 1):
        if abs(h[i] - h[i + 1]) > 0.15:
            pairs.append((i, i + 1))
            pairs.append((i + 1, i))
    return pairs


def optimize_pairs(h, time_budget, seed=42):
    n = len(h)
    target = float(n) / 2.0
    dx = 2.0 / n
    h = project_feasible(h.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    no_improve = 0
    escape_amp = 0.08

    while time.time() - t0 < time_budget - 5.0:
        c0 = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(c0)
        current_c5 = max_c * dx

        if current_c5 < best_c5 - 1e-14:
            best_c5 = current_c5
            best_h = h.copy()

        rel_tol = 0.04
        thresh = max_c * (1.0 - rel_tol)
        active = np.where(c0 >= thresh)[0]

        proposed = propose_pairs(h, c0, active, rng, n)
        if len(proposed) > 120:
            rng2 = np.random.RandomState(int(time.time() * 1000) % 100000)
            idx = rng2.choice(len(proposed), 120, replace=False)
            proposed = [proposed[i] for i in idx]

        best_improvement = 0.0
        best_pair = None
        best_t = 0.0

        for (i, j) in proposed:
            if i == j:
                continue
            t_lo = max(-h[i] + 1e-10, h[j] - 1.0 + 1e-10)
            t_hi = min(1.0 - h[i] - 1e-10, h[j] - 1e-10)
            if t_lo >= t_hi:
                continue
            d = pair_slopes(h, i, j)
            t_opt, val_opt = exact_line_min(c0, d, t_lo, t_hi)
            opt_c5 = val_opt * dx
            improvement = current_c5 - opt_c5
            if improvement > best_improvement:
                best_improvement = improvement
                best_pair = (i, j)
                best_t = t_opt

        if best_pair is not None and best_improvement > 1e-12:
            i, j = best_pair
            h[i] += best_t
            h[j] -= best_t
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                noise = rng.randn(n)
                noise -= np.mean(noise)
                h_try = h + escape_amp * noise
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    h = h_try.copy()
                    best_h = h.copy()
                    best_c5 = c5_try
                escape_amp = min(escape_amp * 1.2, 0.4)
                no_improve = 0

    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_h = h.copy()
        best_c5 = final_c5

    return best_h, best_c5


def resample_linear(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    target = float(n_new) / 2.0
    h_new = project_feasible(h_new, target)
    return h_new


def random_feasible(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    h = h[rng.permutation(n)]
    return project_feasible(h, float(n) / 2.0)


def optimize_multi_init(h_init, n, time_budget, seed):
