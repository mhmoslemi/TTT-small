        self._cache_val = None
    
    def value_and_grad(self, z):
        key = z.tobytes()
        if key == self._cache_key:
            return self._cache_val
        val, grad = soft_obj_grad_sym(z, self.n, self.beta)
        self._cache_key = key
        self._cache_val = (val, grad)
        return self._cache_val
    
    def fun(self, z):
        return self.value_and_grad(z)[0]
    
    def jac(self, z):
        return self.value_and_grad(z)[1]

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -11.0, 11.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        L = n - d
        G[n-1+d, :L] -= dx * h[d:]
        G[n-1+d, d:] += dx * (1.0 - h[:L])
        G[n-1-d, :L] += dx * (1.0 - h[d:])
        G[n-1-d, d:] -= dx * h[:L]
    return G

def sym_project(h):
    return 0.5 * (h + 1.0 - h[::-1].copy())

def z_to_h_sym(z, n):
    M = n // 2
    left = 1.0 / (1.0 + np.exp(-np.clip(z, -50.0, 50.0)))
    h = np.empty(n)
    h[:M] = left
    h[M:] = 1.0 - left[::-1]
    return h

def h_to_z_sym(h, n):
    M = n // 2
    left = np.clip(h[:M], 1e-12, 1.0 - 1e-12)
    return np.log(left / (1.0 - left))

def soft_obj_grad_sym(z, n, beta):
    M = n // 2
    h = z_to_h_sym(z, n)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    s_max = shifted.max()
    logsum = s_max + np.log(np.sum(np.exp(shifted - s_max)))
    obj = cmax + logsum / beta
    w = np.exp(shifted - s_max)
    w /= w.sum()
    G = compute_grad_matrix(h)
    sig = h[:M] * (1.0 - h[:M])
    Gz_diff = G[:, :M] - G[:, n-1:M-1:-1]
    grad = (w @ Gz_diff) * sig
    return obj, grad

class SoftObj:
    def __init__(self, n, beta):
        self.n = n
        self.M = n // 2
        self.beta = beta
        self._key = None
        self._val = None

    def vg(self, z):
        k = z.tobytes()
        if k == self._key:
            return self._val
        v = soft_obj_grad_sym(z, self.n, self.beta)
        self._key = k
        self._val = v
        return v

    def fun(self, z):
        return self.vg(z)[0]

    def jac(self, z):
        return self.vg(z)[1]

class ActiveNLP:
    def __init__(self, n, active_idx):
        self.n = n
        self.M = n // 2
        self.active = active_idx
        self._key = None
        self._G = None
        self._h = None

    def _compute(self, x):
        z = x[:self.M]
        k = z.tobytes()
        if k == self._key:
            return self._h, self._G
        h = z_to_h_sym(z, self.n)
        G = compute_grad_matrix(h)
        self._key = k
        self._h = h
        self._G = G
        return h, G

    def fun(self, x):
        return x[-1]

    def jac(self, x):
        g = np.zeros(self.M + 1)
        g[-1] = 1.0
        return g

    def con_fun(self, x):
        h, _ = self._compute(x)
        vals = compute_vals(h)
        return x[-1] - vals[self.active]

    def con_jac(self, x):
        h, G = self._compute(x)
        sig = h[:self.M] * (1.0 - h[:self.M])
        H = (G[:, :self.M] - G[:, self.n-1:self.n-1-self.M:-1]) * sig
        J = -H[self.active, :]
        J = np.hstack([J, np.ones((len(self.active), 1))])
        return J
