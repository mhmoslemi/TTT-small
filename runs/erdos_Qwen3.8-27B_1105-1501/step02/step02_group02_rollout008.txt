def _project(z):
    """Project z onto {x: 0<=x<=1, sum(x)=n/2}"""
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]

            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)

            h[i] += add
            res -= add

def _make_binary_pattern(n, pattern_type, rng):
    """Create structured binary patterns"""
    if n % 2 != 0:
        n -= 1
    h = np.zeros(n)
    target = n // 2
    if pattern_type == 0:
        # Alternating
        h[::2] = 1.0
        if h.sum() != target:
            # Adjust
            excess = h.sum() - target
            if excess > 0:
                idx = np.where(h == 1)[0]
                for i in idx[:excess]:
                    h[i] = 0
                    # Find a zero to set to one
                    zeros = np.where(h == 0)[0]
                    if len(zeros) > 0:
                        h[zeros[-1]] = 1
            elif excess < 0:

                for i in idx[:abs(excess)]:
                    h[i] = 1

                    if len(ones) > 0:
                        h[ones[-1]] = 0
    elif pattern_type == 1:

        shift = rng.integers(0, n)
        for i in range(n):
            if i % n < target:
                h[(i + shift) % n] = 1.0
    else:
        # Golden ratio placement
        phi = (1 + np.sqrt(5)) / 2
        pos = np.floor(np.arange(target) * phi * n / target) % n
        h[np.unique(pos)] = 1.0
        # Fix count
        if h.sum() < target:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, size=target - h.sum(), replace=False)] = 1.0
        elif h.sum() > target:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, size=h.sum() - target, replace=False)] = 0.0
    # Ensure exact count
    if h.sum() != target:

            idx = np.where(h == 1)[0]

        else:
            idx = np.where(h == 0)[0]
            h[rng.choice(idx, size=int(target - h.sum()), replace=False)] = 1.0

def _binary_sa(h0, n_iters, rng, t_start=0.1, t_end=0.001):
    """Simulated annealing on binary vectors preserving exact count"""
    n = len(h0)
    target = n // 2
    cur = h0.copy()
    cur_s = _score(cur)
    best = cur.copy()
    best_s = cur_s
    
    for it in range(n_iters):
        T = t_start * (t_end / t_start) ** (it / max(1, n_iters - 1))
        # Propose move: swap a 1 with a 0
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0
        cur[oz] = 1
        new_s = _score(cur)
        delta = new_s - cur_s
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_s = new_s
            if new_s < best_s:
                best_s = new_s
                best = cur.copy()
        else:
            cur[oi] = 1
            cur[oz] = 0
    
    return best, best_s

def _active_slsqp(h0, maxiter=200, top_m=12):
    """Active-set SLSQP: minimize t s.t. corr_k <= t for top lags, sum(h)=n/2, 0<=h<=1"""
    n = len(h0)
    cv0 = _corr(h0)
    
    # Find active lags
    idx_sorted = np.argsort(cv0)[::-1][:top_m]
    t0 = cv0.max() + 0.01
    
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def _fun(z):
        return z[-1]
    
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr(h)
        return t - cv[idx_sorted]
    
    def _eq(z):
        return z[:-1].sum() - n / 2.0
    
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            h_new = _project(res['x'][:-1])
            return h_new
    except Exception:
        pass
    return h0

def _make_binary(n, rng, ptype):

def _active_slsqp_iterative(h0, max_rounds=5, maxiter=120):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)
    for r in range(max_rounds):
        cv = _corr(cur)
        t_cur = cv.max()
        thr = t_cur - 0.002
        active_idx = np.where(cv >= thr)[0]
        if len(active_idx) > 25:
            active_idx = np.argsort(cv)[::-1][:25]
        z0 = np.concatenate([cur, [t_cur + 0.005]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        def _fun(z):
            return z[-1]
        def _fun_jac(z):
            g = np.zeros(len(z))
            g[-1] = 1.0
            return g
        def _ieq(z):
            h = z[:-1]
            t = z[-1]
            cv2 = _corr(h)
            return t - cv2[active_idx]
        def _eq(z):
            return z[:-1].sum() - n / 2.0
        def _eq_jac(z):
            g = np.zeros(len(z))
            g[:-1] = 1.0
            return g
        try:
            res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'ineq', 'fun': _ieq},
                                        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                h_new = _project(res['x'][:-1])
                s_new = _score(h_new)
                if s_new < cur_s - 1e-14:
                    cur = h_new
                    cur_s = s_new
                else:
                    break
        except Exception:
            break
    return cur, cur_s

def _lse_slsqp(h0, beta, maxiter=120):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def _lo(x):

        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta
    def _lo_grad(x):
        cv = _corr(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            j0 = max(0, s)
            j1 = min(n, n + s)
            for p in range(j0, j1):
                g[p] += wk * (1.0 - x[p - s])
            p0 = max(0, -s)
            p1 = min(n, n - s)
            for p in range(p0, p1):
                g[p] += wk * (-x[p + s])
        return g
    def _eq(x):
        return x.sum() - n / 2.0
    def _eq_jac(x):
        return np.ones(n)
    try:
        res = minimize(_lo, h0, jac=_lo_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):

    nrest = min(8, int(rem / 35))
    for _ in range(nrest):
        if time.time() > deadline - 12:
            break
        hp = _project(best_h + 0.025 * rng.standard_normal(len(best_h)))
        cur, cs = _chain(hp, deadline, rng)
        if cs < best_score - 1e-12:
            best_h = cur.copy(); best_score = cs; best_n = len(cur)
    # Final
    best_h = _project(best_h)
    final_score = _score(best_h)
    return (best_h, final_score, len(best_h))
