def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def subgradient_polish(h0, max_iter=5000, time_limit=None, t_start=None, tol=0.005):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    alpha = 0.02
    for it in range(max_iter):
        if time_limit is not None and time.time() > t_start + time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        threshold = m - tol * m
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for k in active:
            s2 = k - (n-1)
            if s2 >= 0:
                if s2 == 0:
                    grad += (1.0 - 2.0 * h)
                else:
                    grad[s2:] += (1.0 - h[:n-s2])
                    grad[:n-s2] += (-h[s2:])
            else:
                m2 = -s2
                if m2 == 0:
                    grad += (1.0 - 2.0 * h)
                else:
                    grad[:n-m2] += (1.0 - h[m2:])
                    grad[m2:] += (-h[:n-m2])
        grad /= len(active)
        h_new = make_feasible(h - alpha * grad)
        new_score = official_score(h_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.01, 0.5)
        else:
            h = h_new
            alpha = max(alpha * 0.995, 0.001)
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def project_anti_symmetric(h):
    n = len(h)
    half = n // 2
    g = (h[:half] - h[half:] + 1.0) / 2.0
    g = np.clip(g, 0.0, 1.0)
    out = np.empty(n)
    out[:half] = g
    out[half:] = 1.0 - g
    return out

def lse_anti_obj_grad(x, beta):
    half = len(x)
    n = 2 * half
    h = np.empty(n)
    h[:half] = x
    h[half:] = 1.0 - x
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0.0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        obj = m
        weights = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        obj = m + np.log(s) / beta
        weights = e / s
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad_h += (1.0 - 2.0 * h)
            else:
                grad_h[s2:] += (1.0 - h[:n - s2])
                grad_h[:n - s2] += (-h[s2:])
        else:

            if m2 == 0:

            else:
                grad_h[:n - m2] += (1.0 - h[m2:])
                grad_h[m2:] += (-h[:n - m2])
    grad_h *= dx
    grad = grad_h[:half] - grad_h[half:]
    return obj, grad

    return np.sum(x) - len(x) / 2.0

def binary_anti_local_search(g0, max_rounds=500, time_budget=15.0, t_ref=None):
    half = len(g0)
    n = 2 * half
    g = (g0 > 0.5).astype(float)
    n_ones_target = half // 2
    cur_ones = int(np.sum(g == 1))
    if cur_ones > n_ones_target:
        ones_idx = np.where(g == 1)[0]
        order = np.argsort(g0[ones_idx])[::-1]
        g[ones_idx[order[n_ones_target:]]] = 0
    elif cur_ones < n_ones_target:
        zeros_idx = np.where(g == 0)[0]
        order = np.argsort(-g0[zeros_idx])
        g[zeros_idx[order[:n_ones_target - cur_ones]]] = 1
    
    h = np.empty(n)
    h[:half] = g
    h[half:] = 1.0 - g
    best = official_score(h)
    
    ones_idx = np.where(g == 1)[0]
    zeros_idx = np.where(g == 0)[0]
    
    for rnd in range(max_rounds):
        if t_ref is not None and time.time() - t_ref > time_budget:
            break
        best_improve = 0.0
        best_swap = None
        for i in ones_idx:
            for j in zeros_idx:
                g[i] = 0.0
                g[j] = 1.0
                h[:half] = g
                h[half:] = 1.0 - g
                s = official_score(h)
                imp = best - s
                if imp > best_improve:

                    best_swap = (i, j)

        i, j = best_swap
        g[i] = 0.0
        g[j] = 1.0

        h[half:] = 1.0 - g
        best = official_score(h)
        ones_idx = np.where(g == 1)[0]
        zeros_idx = np.where(g == 0)[0]

    best_h = h.copy()

        m = np.max(corr)
