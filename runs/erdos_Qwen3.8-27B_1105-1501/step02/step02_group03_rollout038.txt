def official_score(h):

    dx = 2.0 / n

def make_feasible(z):

def compute_corr_values(h):

def build_grad_matrix(h):
    """Build the (2n-1) x n gradient matrix for all correlation values."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s == 0:
            G[p] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            L = n - s
            # C = dx * sum_{j=0}^{L-1} h[j+s]*(1-h[j])
            # d/dh[j+s] = dx*(1-h[j]), d/dh[j] = -dx*h[j+s]
            G[p, s:s+L] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[s:s+L]

            G[p, :L] += dx * (1.0 - h[t:t+L])
            G[p, t:t+L] -= dx * h[:L]
    return G

    """Reconstruct anti-symmetric h from z (left half via logistic)."""
    M = n // 2
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))
    h = np.empty(n)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[M] = 0.5
    return h

def z_from_h(h, n):
    """Extract z from anti-symmetric h."""
    M = n // 2
    u = np.clip(h[:M], 1e-12, 1 - 1e-12)
    z = np.log(u / (1.0 - u))
    return z

def sym_grad_z(h, G):
    """For anti-symmetric h, compute dC_s/dz_j for all s and j.
    G is (2n-1) x n, returns (2n-1) x M."""
    n = len(h)
    M = n // 2
    u = h[:M]
    du = u * (1.0 - u)
    # dC_s/dz_j = du_j * (G[s, j] - G[s, n-1-j])
    Gz = du[np.newaxis, :] * (G[:, :M] - G[:, n-1-np.arange(M)])
    return Gz

def smooth_minimax_sym(z, beta, n):
    """Log-sum-exp smoothed minimax over anti-symmetric family."""
    h = sym_from_z(z, n)
    vals = compute_corr_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)

    shifted = np.clip(shifted, -700, 700)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted - logsum)
    G = build_grad_matrix(h)
    Gz = sym_grad_z(h, G)
    grad = w @ Gz
    return obj, grad

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)


def compute_corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s == 0:
            G[p] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            L = n - s
            G[p, s:s + L] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[s:s + L]
        else:
            t = -s
            L = n - t
            G[p, :L] += dx * (1.0 - h[t:t + L])
            G[p, t:t + L] -= dx * h[:L]
    return G


def sym_from_z(z, n):
    M = n // 2
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -50.0, 50.0)))
    h = np.empty(n)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[M] = 0.5
    return np.clip(h, 0.0, 1.0)


def z_from_h(h, n):
    M = n // 2
    u = np.clip(h[:M], 1e-12, 1.0 - 1e-12)
    return np.log(u / (1.0 - u))


def smooth_minimax_sym(z, beta, n):
    h = sym_from_z(z, n)
    vals = compute_corr_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -700.0, 700.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted - logsum)
    G = build_grad_matrix(h)
    M = n // 2
    u = h[:M]
    du = u * (1.0 - u)
    Gz = du[np.newaxis, :] * (G[:, :M] - G[:, n - 1 - np.arange(M)])
    grad = w @ Gz
    return obj, grad


def find_active(vals, frac=0.3, min_count=10):
    cmax = vals.max()
    threshold = cmax - max(0.008 * cmax, 1e-5)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)


def interpolate_h(h_src, n_tgt):
    n_src = len(h_src)
    src_x = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_x = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_x, src_x, h_src)
    return make_feasible(h_int)


def smooth_sym_lu(p):
    n = p['n']
    beta = p['beta']
    z = p['z']
    return smooth_minimax_sym(z, beta, n)


def smooth_sym_obj_factory(n, beta):
    def obj(z):
        return smooth_minimax_sym(z, beta, n)[0]
    def jac(z):
        return smooth_minimax_sym(z, beta, n)[1]
    return obj, jac


def active_set_sym_slsqp(z, n, active_idx, t0, max_iter=300):
    M = len(z)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def con_fun(x):
        h = sym_from_z(x[:-1], n)
        vals = compute_corr_values(h)
        return x[-1] - vals[active_idx]

    def con_jac(x):
        h = sym_from_z(x[:-1], n)
        G = build_grad_matrix(h)
        u = h[:M]
        du = u * (1.0 - u)
        Gz = du[np.newaxis, :] * (G[:, :M] - G[:, n - 1 - np.arange(M)])
        J = -Gz[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])

    x0 = np.concatenate([z, [t0]])
    bounds = [(-20.0, 20.0)] * M + [(0.0, 3.0)]
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            z_new = res.x[:-1]
            h_new = sym_from_z(z_new, n)
            sc = official_score(h_new)
            return z_new, h_new, sc
    except Exception:
        pass
    return None, None, None


def active_set_full_slsqp(h, active_idx, t0, max_iter=300):
    n = len(h)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def con_fun(x):
        vals = compute_corr_values(x[:-1])
        return x[-1] - vals[active_idx]

    def con_jac(x):
        hh = x[:-1]
        G = build_grad_matrix(hh)
        J = -G[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])

    def eq_fun(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    cons = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:-1])
            sc = official_score(h_new)
            return h_new, sc
    except Exception:
        pass
    return None, None


def coord_polish_sym(z, n, max_sweeps=8, tol=1e-12):
    M = n // 2
    z_cur = z.copy()
    for sweep in range(max_sweeps):
        improved = False
        h = sym_from_z(z_cur, n)
        vals = compute_corr_values(h)
        cmax = vals.max()
        for j in range(M):
            G = build_grad_matrix(h)
            a = G[:, j] - G[:, n - 1 - j]
            u_old = h[j]
            b = vals - a * u_old

            def fd(u_val):
                return np.max(a * u_val + b)

            try:
                res = minimize_scalar(fd, bounds=(1e-10, 1.0 - 1e-10),
                                      method='bounded', options={'xatol': 1e-12})
                if res.success and np.isfinite(res.x):
                    u_new = res.x
                    if abs(u_new - u_old) > tol:
                        z_cur[j] = np.log(max(u_new, 1e-12) / max(1.0 - u_new, 1e-12))
                        improved = True
                        h = sym_from_z(z_cur, n)
                        vals = compute_corr_values(h)
                        cmax = vals.max()
            except Exception:
                pass
        if not improved:
            break
    return z_cur
