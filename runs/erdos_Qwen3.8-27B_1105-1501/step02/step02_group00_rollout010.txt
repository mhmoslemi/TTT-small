    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute gradient matrix G where G[r, i] = d(score_r)/d(h_i)"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _smooth_grad_step(h, deadline, beta=50, max_iters=10000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step_size = 0.01
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = _scores(h)
        # Softmax weights
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        w /= np.sum(w)
        
        G = _grad_matrix(h)
        # Direction = -weighted sum of gradients
        d = -(G.T @ w)
        
        # Project direction to maintain sum constraint
        d -= np.mean(d)
        
        # Normalize
        norm = np.max(np.abs(d))
        if norm < 1e-15:
            break
        d /= norm
        
        # Line search
        alpha_max = 0.5
        for j in range(n):
            if d[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
            elif d[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-d[j]))
        alpha_max = max(alpha_max, 1e-8)
        
        # Try several step sizes
        best_a = 0
        best_c5_trial = cur_c5
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            a = alpha_max * frac
            h_try = h + a * d
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_trial:
                best_c5_trial = c5_try
                best_a = a
        
        if best_a == 0:
            step_size *= 0.5
            if step_size < 1e-6:
                break
            continue
        
        h = h + best_a * d
        h = _project_box_sum(h, target)
        cur_c5 = _exact_c5(h)

            best_c5 = cur_c5
            best_h = h.copy()

            step_size *= 0.7

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 500})

            return res.x[:n]

        pass
    return None

        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP polish
    deadline4 = t0 + budget_s * 0.65
    h4, c5_4 = _lp_loop(best_h, deadline4, radius_init=0.05, max_iters=3000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary local search
    deadline5 = t0 + budget_s * 0.75
    hb, c5b = _binary_local_search(best_h, deadline5, max_iters=80000, rng=rng)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()

    # Phase 6: Refine binary result fractionally
    if c5b < best_c5:
        deadline6 = t0 + budget_s * 0.80
        h6, c5_6 = _continuation_opt(best_h, deadline6, beta_start=5.0, beta_end=300.0, n_beta_steps=8)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        h6b, c5_6b = _lp_loop(best_h, deadline6, radius_init=0.1, max_iters=2000)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Multi-resolution search
    deadline7 = t0 + budget_s * 0.92
    n_tries = [best_n - 8, best_n - 4, best_n + 4, best_n + 8, best_n + 16,
               best_n - 16, 48, 56, 72, 80, 96, 112, 128]
    n_tries = sorted(set(x for x in n_tries if 24 <= x <= 160))
    for n_try in n_tries:
        if time.time() > deadline7:
            break
        per_n_time = budget_s * 0.03
        h_r = _resample(best_h, n_try)
        deadline_r = min(deadline7, time.time() + per_n_time)
        h_r, c5_r = _continuation_opt(h_r, deadline_r, beta_start=5.0, beta_end=200.0, n_beta_steps=8)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Quick LP polish on the new best
            if time.time() < deadline7:
                h_rp, c5_rp = _lp_loop(best_h, min(deadline7, time.time() + per_n_time * 0.5),
                                        radius_init=0.08, max_iters=1000)
                if c5_rp < best_c5:
                    best_c5 = c5_rp
                    best_h = h_rp.copy()

    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.96:
        deadline8 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, deadline8, max_passes=30)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 9: Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.995, radius_init=0.03, max_iters=2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Final projection
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
