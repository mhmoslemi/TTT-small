import numpy as np
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_corr(h, k):
    """Gradient of correlation at lag k (numpy.correlate index offset from center).
    k is the shift: c[k_center + k] where k_center = n-1.
    """
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        t = -k
        if n - t > 0:
            g[:n-t] += (1.0 - h[t:])
            g[t:] -= h[:n-t]
    return g

def project_feasible(z):
    """Project z onto {h: 0 <= h <= 1, sum(h) = n/2}."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return project_feasible(h_int)

def lp_direction(h, k_active):
    """Find a descent direction using LP.
    Returns direction d (satisfying sum(d)=0, -h <= d <= 1-h)
    that tries to decrease the top-k_active correlations.
    """
    n = len(h)
    dx = 2.0 / n
    corr_vals = compute_correlations(h)
    # Get top k_active lag indices (0-indexed in the full correlation array)
    top_idx = np.argsort(corr_vals)[-k_active:]
    
    # Build gradient matrix for these lags
    # Each lag p in top_idx corresponds to shift k = p - (n-1)
    A_list = []
    b_list = []
    for p in top_idx:
        k = p - (n - 1)
        g = grad_corr(h, k) * dx
        A_list.append(g)  # We want g @ d <= -epsilon (decrease)
        b_list.append(-corr_vals[p])  # Actually, we want corr_vals[p] + g @ d <= t
    
    # LP: minimize t
    # subject to: corr_vals[p] + A_p @ d <= t for each p in top
    #             sum(d) = 0
    #             -h <= d <= 1-h
    #
    # Variables: d_0, ..., d_{n-1}, t
    # c = [0, ..., 0, 1] (minimize t)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = []
    b_ub = []
    for i, p in enumerate(top_idx):
        k = p - (n - 1)
        g = grad_corr(h, k) * dx
        # corr_vals[p] + g @ d - t <= 0
        row = np.zeros(n + 1)
        row[:-1] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-corr_vals[p])
    
    # Box constraints: -h <= d <= 1-h
    for j in range(n):
        # d_j <= 1 - h[j]
        row = np.zeros(n + 1)
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - h[j])
        # -d_j <= h[j]  i.e., d_j >= -h[j]
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_ub.append(row)
        b_ub.append(h[j])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Solve LP
    try:
        res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=b_eq, method='highs')
        if res.status == 0:
            d = res.x[:n]
            t_val = res.x[-1]
            return d, t_val
    except Exception:
        pass
    return None, None

def line_search(h, d, max_alpha=1.0, n_trials=20):
    """Find best alpha in [0, max_alpha] for direction d."""
    best_alpha = 0.0
    best_score = official_score(h)
    
    # Compute max feasible alpha
    alpha_max = max_alpha
    for j in range(len(h)):
        if d[j] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-12:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    
    if alpha_max <= 1e-12:
        return 0.0, best_score, h
    
    alphas = np.linspace(0.01, alpha_max, n_trials)
    for alpha in alphas:
        h_try = project_feasible(h + alpha * d)
        sc = official_score(h_try)
        if sc < best_score - 1e-14:
            best_score = sc
            best_alpha = alpha
    
    return best_alpha, best_score, project_feasible(h + best_alpha * d)

def subgradient_step(h, k_active, eta):
    """Take a subgradient step using the average of top-k gradients."""
    n = len(h)
    dx = 2.0 / n
    corr_vals = compute_correlations(h)
    top_idx = np.argsort(corr_vals)[-k_active:]
    
    # Average gradient of top-k correlations
    g_avg = np.zeros(n)
    for p in top_idx:
        k = p - (n - 1)
        g_avg += grad_corr(h, k)
    g_avg *= dx / len(top_idx)
    
    # Step and project
    h_new = project_feasible(h - eta * g_avg)
    return h_new

def optimize_lp(h0, max_iter=100, k_active=10, eta_start=0.5, eta_end=0.01):

    n = len(h0)
    h = h0.copy()
    best_score = official_score(h)
    stall = 0

            h_new = subgradient_step(h, k_active, eta)

        else:

                h_new = subgradient_step(h, k_active, eta)

            g = grad_corr(h, k) * dx
            constraints.append(corr_vals[p] + g @ (h_var - h) <= t_var)
        
        obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h))
        prob = cp.Problem(obj, constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved or h_var.value is None or not np.all(np.isfinite(h_var.value)):
            break
        
        h_new = project_feasible(h_var.value)
        sc = official_score(h_new)
        if sc < best_score - 1e-14:
            h = h_new
            best_score = sc
            stall = 0
        else:
            improved = False
            for alpha in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best_score - 1e-14:
                    h = h_try
                    best_score = sc_try
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall >= 8:
                    break
    
    return h, best_score

def paired_coordinate_descent(h0, max_sweeps=30, n_pairs=50):
    """Try paired coordinate moves: change h[j] by +d and h[i] by -d."""
    n = len(h0)
    rng = np.random.default_rng(123)
    h = h0.copy()
    best_score = official_score(h)
    
    for sweep in range(max_sweeps):
        improved_this_sweep = False
        for _ in range(n_pairs):
            j = rng.integers(n)
            i = rng.integers(n)
            if i == j:
                continue
            # Try moving mass from j to i and vice versa
            for sign in [1, -1]:
                d_max = min(h[j], 1.0 - h[i]) if sign > 0 else min(h[i], 1.0 - h[j])
                if d_max < 1e-6:
                    continue
                # Try a few step sizes
                for frac in [0.1, 0.3, 0.5, 0.8, 1.0]:
                    d = frac * d_max
                    h_try = h.copy()
                    if sign > 0:
                        h_try[j] -= d
                        h_try[i] += d
                    else:
                        h_try[j] += d
                        h_try[i] -= d
                    h_try = np.clip(h_try, 0.0, 1.0)
                    sc = official_score(h_try)
                    if sc < best_score - 1e-14:
                        h = h_try
                        best_score = sc
                        improved_this_sweep = True
                        break
                if improved_this_sweep:
                    break
            if improved_this_sweep:
                break
        if not improved_this_sweep:
            break
    
    return h, best_score

def anti_sym(h):
    return project_feasible(1.0 - h[::-1].copy())
