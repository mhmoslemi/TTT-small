import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    """Project z onto the feasible set: h in [0,1], sum(h) = n/2."""
    n = len(z)
    target = n / 2.0
    z = np.clip(z, -5.0, 5.0)
    # Bisection on the shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return h

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def subgradient_step(h, step_size, n_active=10, tol=1e-6):
    """One subgradient step for the minimax problem."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    # Find active constraints (within tol of max)
    active = np.where(vals >= cmax - tol)[0]
    if len(active) == 0:
        active = np.argsort(vals)[-n_active:]
    elif len(active) > n_active:
        # Take top n_active
        top_local = np.argsort(vals[active])[-n_active:]
        active = active[top_local]
    
    # Compute subgradient: average of gradients of active constraints
    g = np.zeros(n)
    for p in active:
        off = n - 1 - p
        # F_p = sum_{i: 0<=i<n, 0<=i+off<n} h[i] * (1-h[i+off])
        # dF_p/dh[i] = (1-h[i+off]) if i+off valid, minus h[i-off] if i-off valid
        if off >= 0:
            # i+off valid when i < n-off
            for i in range(n - off):
                g[i] += (1.0 - h[i + off])
            # i-off valid when i >= off, i.e., i = j+off for j in [0, n-1-off]
            # F_p = sum_j h[j] * (1-h[j+off])
            # dF_p/dh[i] for i = j+off: -h[j] = -h[i-off]
            for i in range(off, n):
                g[i] -= h[i - off]
        else:
            # off < 0, let t = -off > 0
            t = -off
            # F_p = sum_{i: 0<=i<n, 0<=i-off<n} h[i] * (1-h[i+off])
            # i+off = i-t, valid when i >= t
            # So F_p = sum_{i=t}^{n-1} h[i] * (1-h[i-t])
            for i in range(t, n):
                g[i] += (1.0 - h[i - t])
                g[i - t] -= h[i]
    
    g /= len(active)
    # Take step
    h_new = h - step_size * g
    h_new = make_feasible(h_new)
    return h_new

def subgradient_descent(h0, max_iter=200, step0=0.01, decay=0.999, n_active=5):
    """Subgradient descent with geometric decay step size."""
    h = h0.copy()
    best = official_score(h)
    step = step0
    for it in range(max_iter):
        h_new = subgradient_step(h, step, n_active=n_active)
        sc = official_score(h_new)
        if sc < best:
            best = sc
            h = h_new
        step *= decay
    return h, best

def coordinate_descent(h0, max_sweeps=10, tol=1e-10):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Compute C(t) = max_p (a_p * t + b_p) for t in [0,1]
            # where F_p(h) = a_p * h[i] + b_p
            # For each lag p, find the coefficient of h[i]
            n_corr = 2 * n - 1
            a_p = np.zeros(n_corr)
            b_p = np.zeros(n_corr)
            
            for p in range(n_corr):
                off = n - 1 - p
                # F_p = sum_{valid j} h[j] * (1 - h[j+off])
                # Terms involving h[i]:
                # 1. j = i: h[i] * (1-h[i+off]) if i+off valid
                # 2. j such that j+off = i, i.e., j = i-off: h[i-off] * (1-h[i]) if i-off valid
                a = 0.0
                b = 0.0
                for j in range(n):
                    j_plus = j + off
                    if 0 <= j_plus < n:
                        if j == i:
                            # h[i] * (1-h[i+off])
                            a += (1.0 - h[j_plus])
                            # b is 0 for this term (it's all in a*h[i])
                        elif j_plus == i:
                            # h[j] * (1-h[i]) = h[j] - h[j]*h[i]
                            a -= h[j]
                            b += h[j]
                        else:
                            b += h[j] * (1.0 - h[j_plus])
                a_p[p] = a
                b_p[p] = b
            # C(t) = max_p (a_p[p] * t + b_p[p]) * dx
            # Find min over t in [0,1]
            # This is a convex PL function. Min at breakpoint or boundary.
            # Collect all breakpoints
            breakpoints = set([0.0, 1.0])
            for p1 in range(n_corr):
                for p2 in range(p1+1, n_corr):
                    da = a_p[p1] - a_p[p2]
                    if abs(da) > 1e-14:
                        tb = (b_p[p2] - b_p[p1]) / da
                        if 0.0 <= tb <= 1.0:
                            breakpoints.add(tb)
            # Also check where individual terms are minimized (vertex of each line)
            # Actually for max of lines, the min is at a vertex (breakpoint) or boundary
            best_t = old_val
            best_c = float('inf')
            for t in breakpoints:
                t = max(0.0, min(1.0, t))
                c = max(a_p * t + b_p)
                if c < best_c:
                    best_c = c
                    best_t = t
            
            if best_t != old_val:
                # Update h[i] and compensate
                delta = old_val - best_t
                h[i] = best_t
                # Compensate in other coordinates proportionally
                if abs(delta) > 1e-14:
                    others = np.delete(np.arange(n), i)
                    # Add delta/n (approximately) to all others, then project
                    h[others] += delta / (n - 1)
                    h = make_feasible(h)
                    sc = official_score(h)
                    if sc < best - 1e-14:
                        best = sc
                        improved = True
                    else:
                        h[i] = old_val  # revert
                        h = make_feasible(h)
        
        if not improved:
            break
    
    return h, best

def active_set_qp(h0, k_active=5, reg=0.01):
    """One active-set QP step."""
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        off = n - 1 - p
        # Compute gradient of F_p at h0
        g_p = np.zeros(n)
        if off >= 0:
            for i in range(n - off):
                g_p[i] += (1.0 - h0[i + off])
                g_p[i + off] -= h0[i]
        else:
            t = -off
            for i in range(t, n):
                g_p[i] += (1.0 - h0[i - t])
                g_p[i - t] -= h0[i]
        
        constraints.append(vals[p] + dx * cp.multiply(g_p, h_var - h0) <= t_var)
    
    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    
    for solver in ['CLARABEL', 'ECOS', 'CVXOPT']:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)

def qa_tractive_set_optimize(h0, max_iter=30, k_active=5, reg_start=5.0, reg_end=0.005):

    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    stall = 0
    for it in range(max_iter):
        reg = reg_start * (reg_end / reg_start) ** (it / max(1, max_iter - 1))

            break
        sc = official_score(h_new)
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False

                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 6:
                    break
    return h, best

        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        h_init = 0.5 + 0.35 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)
    update(best_h, best_score)

    # Phase 1: Try symmetry transforms of parent
    if time.time() < deadline - 30:
        for transform in [reflect_h, shift_h, complement_shift]:
            if time.time() > deadline - 15:
                break
            h_t = transform(best_h)
            sc_t = official_score(h_t)
            update(h_t, sc_t)

    # Phase 2: Subgradient descent from best
    if time.time() < deadline - 60:
        h_sg, sc_sg = subgradient_descent(best_h, max_iter=400, step0=0.008, decay=0.997,
                                          tol=1e-5, max_active=5, deadline=deadline - 30)
        update(h_sg, sc_sg)

    # Phase 3: Active-set QP
    if time.time() < deadline - 60:
        h_qp, sc_qp = active_set_optimize(best_h, max_iter=25, k_active=5,
                                         reg_start=3.0, reg_end=0.005, deadline=deadline - 20)
        update(h_qp, sc_qp)

    # Phase 4: SLSQP smooth max
    if time.time() < deadline - 40:
        for beta in [1000, 5000, 20000]:
            if time.time() > deadline - 15:
                break
            h_new, sc = try_slsqp(best_h, beta=beta, maxiter=80, deadline=deadline - 10)
            if h_new is not None:
                update(h_new, sc)

    # Phase 5: Coordinate descent
    if time.time() < deadline - 40:
        h_cd, sc_cd = coordinate_descent(best_h, max_sweeps=8, deadline=deadline - 15)
        update(h_cd, sc_cd)

    # Phase 6: Resolution sweep (key: try different n)
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        n_list = sorted(set([30, 36, 40, 44, 48, 50, 52, 54, 56, 58, 60, 62, 64, 68, 72]))
        for n_opt in n_list:
            if time.time() > deadline - 30:
                break
            if n_opt < 20:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            # QP at this resolution
            if time.time() < deadline - 30:
                h_qp, sc_qp = active_set_optimize(h_int, max_iter=20, k_active=5,
                                                reg_start=2.0, reg_end=0.01, deadline=deadline - 15)
                update(h_qp, sc_qp)
            # Subgradient at this resolution
            if time.time() < deadline - 30:
                h_sg, sc_sg = subgradient_descent(h_int, max_iter=200, step0=0.005,
                                                decay=0.997, max_active=5, deadline=deadline - 15)
                update(h_sg, sc_sg)
            # Coordinate descent at this resolution
            if time.time() < deadline - 20:
                h_cd, sc_cd = coordinate_descent(h_int, max_sweeps=3, deadline=deadline - 10)
                update(h_cd, sc_cd)

    # Phase 7: QP with different hyperparameters
    if time.time() < deadline - 50:
        for k_a in [3, 4, 6, 8]:
            if time.time() > deadline - 15:
                break
            h_qp, sc_qp = active_set_optimize(best_h, max_iter=20, k_active=k_a,
                                             reg_start=1.5, reg_end=0.003, deadline=deadline - 10)
            update(h_qp, sc_qp)

    # Phase 8: Alternating CD + QP
    if time.time() < deadline - 50:
        h = best_h.copy()
        for it in range(4):
            if time.time() > deadline - 20:
                break
            h, _ = coordinate_descent(h, max_sweeps=3, deadline=deadline - 15)
            h, _ = active_set_optimize(h, max_iter=10, k_active=5,
                                       reg_start=0.5, reg_end=0.005, deadline=deadline - 15)
            update(h, official_score(h))

    # Phase 9: More subgradient with different params
    if time.time() < deadline - 40:
        for step0 in [0.003, 0.012, 0.02]:
            if time.time() > deadline - 15:
                break
            h_sg, sc_sg = subgradient_descent(best_h, max_iter=300, step0=step0,
                                             decay=0.996, max_active=3, deadline=deadline - 10)
            update(h_sg, sc_sg)

    # Phase 10: Random restarts + optimize
    if time.time() < deadline - 60:
        for i in range(8):
            if time.time() > deadline - 20:
                break
            sigma = 0.01 + 0.04 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            # Optimise the perturbed point
            if time.time() < deadline - 20:
                h_qp, sc_qp = active_set_optimize(h_cand, max_iter=15, k_active=5,
                                                reg_start=1.0, reg_end=0.01, deadline=deadline - 15)
                update(h_qp, sc_qp)
            if time.time() < deadline - 20:
                h_cd, sc_cd = coordinate_descent(h_cand, max_sweeps=3, deadline=deadline - 10)
                update(h_cd, sc_cd)

    # Phase 11: SLSQP at high beta from best
    if time.time() < deadline - 20:
        h_new, sc = try_slsqp(best_h, beta=50000, maxiter=60, deadline=deadline - 5)
        if h_new is not None:
            update(h_new, sc)

    # Phase 12: Final intensive polish
    if time.time() < deadline - 15:
        h_qp, sc = active_set_optimize(best_h, max_iter=15, k_active=4,
                                       reg_start=0.3, reg_end=0.001, deadline=deadline - 5)
        update(h_qp, sc)
        h_cd, sc = coordinate_descent(best_h, max_sweeps=3, deadline=deadline - 3)
        update(h_cd, sc)

    # Final
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score + 1e-12, len(final_h)
