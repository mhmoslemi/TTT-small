import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def build_sym_h(u):
    m = len(u)
    h = np.empty(2 * m)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def compute_grad_full(h, w):
    n = len(h)
    dx = 2.0 / n
    g_full = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-30:
            continue
        s = p - (n - 1)
        wp = w[p] * dx
        if s >= 0:
            L = n - s
            g_full[:L] += wp * (-h[s:])
            g_full[s:] += wp * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            g_full[:K] += wp * (1.0 - h[t:])
            g_full[t:] += wp * (-h[:K])
    return g_full

def grad_u_from_weights(u, w):
    m = len(u)
    h = build_sym_h(u)
    g_full = compute_grad_full(h, w)
    return g_full[:m] - g_full[m:][::-1]

def sym_smooth_obj_grad(u, beta):
    m = len(u)
    h = build_sym_h(u)
    n = 2 * m
    vals = corr_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    grad = grad_u_from_weights(u, w)
    return obj, grad

def pg_solve_sym(u0, beta, time_limit=10.0):
    m = len(u0)
    u = np.clip(u0.copy(), 0.0, 1.0)
    f, g = sym_smooth_obj_grad(u, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-28:
        return u
    alpha = min(1.0 / (gnorm2 + 1e-12), 2.0 / (np.max(g**2) + 1e-12))
    alpha = min(alpha, 0.5)
    t0 = time.time()
    for it in range(50000):
        if time.time() - t0 > time_limit:

        g = sym_smooth_obj_grad(u, beta)[1]

            f_trial = sym_smooth_obj_grad(u_trial, beta)[0]
            if f_trial <= f - 1e-4 * step * gnorm2:

            step *= 0.4

            break
        u, f = u_trial, f_surrogate

def find_active(vals, frac=0.3, min_count=8):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())

    h0 = build_sym_h(u0)
    vals0 = corr_vals(h0)
    t0_init = vals0.max() + 1e-6

    def obj_func(z):
        return z[-1]

    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(z):
        u = z[:m]
        t = z[-1]
        h = build_sym_h(u)
        vals = corr_vals(h)
        return t - vals[active_idx]

    def ineq_jac(z):
        u = np.clip(z[:m], 0.0, 1.0)
        h = build_sym_h(u)
        w = np.zeros(2 * len(h) - 1)
        for k, idx in enumerate(active_idx):
            w[idx] = 1.0
        g_full = compute_grad_full(h, w)
        jac_rows = (g_full[:m] - g_full[m:][::-1])  # for each active constraint
        # Actually we need it per row. Let me recompute.
        # For active constraint at index idx:
        #   d(t - vals[idx])/du = -d(vals[idx])/du = -(g_full[:m] - g_full[m:][::-1]) where g_full is for w being indicator of {idx}
        # This is expensive to do per row. Let me use a different approach.
        # For each active index, compute the gradient row.
        J = np.zeros((len(active_idx), m + 1))
        for k, idx in enumerate(active_idx):
            w_single = np.zeros(2 * len(h) - 1)
            w_single[idx] = 1.0
            g_f = compute_grad_full(h, w_single)
            J[k, :m] = -(g_f[:m] - g_f[m:][::-1])
            J[k, m] = 1.0
        return J

    def eq_fun(z):
        return 0.0  # No equality constraint needed (auto-satisfied)

    z_start = np.concatenate([np.clip(u0, 0.0, 1.0), [t0_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj_func, z_start, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            u_new = np.clip(res.x[:m], 0.0, 1.0)
            h_new = build_sym_h(u_new)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return u_new, sc
    except Exception:
        pass
    return None, None

def interpolate_u(u_src, m_tgt):
    n_src = len(u_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / (2 * n_src))
    tgt_c = (np.arange(m_tgt) + 0.5) * (2.0 / (2 * m_tgt))
    u_int = np.interp(tgt_c, src_c, u_src)
    return np.clip(u_int, 0.0, 1.0)

def extract_u_from_h(h):
    n = len(h)
    m = n // 2
    u = (h[:m] + (1.0 - h[m:][::-1])) / 2.0
    return np.clip(u, 0.0, 1.0)

def pg_solve_general(h0, beta, time_limit=10.0):
    n = len(h0)
    h = h0.copy()
    f, g = None, None

    def obj_grad(z):
        z = make_feasible(z)
        vals = corr_vals(z)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        logsum = np.log(np.sum(np.exp(shifted)))
        obj = cmax + logsum / beta
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(vals)) / len(vals)
        else:
            w /= ws
        g_f = compute_grad_full(z, w)
        return obj, g_f

    f, g = obj_grad(h)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-28:
        return h
    alpha = min(1.0 / (gnorm2 + 1e-12), 0.5)
    t0 = time.time()
    for it in range(30000):
        if time.time() - t0 > time_limit:
            break
        _, g = obj_grad(h)
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-26:
            break
        step = alpha
        for _ in range(30):
            h_trial = make_feasible(h - step * g)
            f_trial = obj_grad(h_trial)[0]
            if f_trial <= f - 1e-4 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-14:
            break
        f_new = obj_grad(h_trial)[0]
        if not np.isfinite(f_new):
            break
        h, f = h_trial, f_new
        alpha = min(alpha * 1.6, 0.5 / (gnorm2 + 1e-12))
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    def check_deadline():
        return time.time() >= deadline

    def update(h_cand, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc
        elif abs(sc - best_score) <= 1e-14 and best_h is not None and len(h_cand) < len(best_h):
            best_h = h_cand.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        m0 = n0 // 2
        x = np.arange(m0)
        u_init = 0.5 + 0.3 * np.sin(2 * np.pi * x / m0) + 0.1 * np.cos(4 * np.pi * x / m0)
        best_h = build_sym_h(np.clip(u_init, 0, 1))
        best_score = official_score(best_h)

    update(best_h, best_score)

    # Phase 1: Symmetric optimization on parent size
    n_par = len(best_h)
    m_par = n_par // 2
    u_par = extract_u_from_h(best_h)
    h_sym_par = build_sym_h(u_par)
    sc_sym_par = official_score(h_sym_par)
    if sc_sym_par < best_score - 1e-14:
        update(h_sym_par, sc_sym_par)

    if time_left() > 40:
        for beta in [50, 100, 300, 1000, 3000, 10000, 30000]:
            if check_deadline():
                break
            tl = min(time_left() * 0.15, 15.0)
            if tl < 2:
                break
            u_par = pg_solve_sym(u_par, beta, time_limit=tl)
            h_cand = build_sym_h(u_par)
            sc = official_score(h_cand)
            if sc < best_score - 1e-14:
                update(h_cand, sc)

    # Phase 2: SLSQP active-set on best symmetric solution
    if time_left() > 30:
        u_cur = extract_u_from_h(best_h)
        h_cur = build_sym_h(u_cur)
        vals = corr_vals(h_cur)
        for frac in [0.25, 0.4, 0.6]:
            if check_deadline():
                break
            active = find_active(vals, frac=frac, min_count=10)
            u_new, sc = slsqp_solve_sym(u_cur, active, max_iter=400)
            if u_new is not None and sc < best_score - 1e-14:
                update(build_sym_h(u_new), sc)
                u_cur = u_new
            vals = corr_vals(build_sym_h(u_cur))

    # Phase 3: Try multiple n values
    if time_left() > 60:
        n_cur = len(best_h)
        n_options = sorted(set([
            max(40, n_cur - 8), max(44, n_cur - 6), max(48, n_cur - 4),
            n_cur, min(80, n_cur + 4), min(84, n_cur + 6), min(88, n_cur + 8),
        ]))
        for n_opt in n_options:
            if check_deadline():
                break
            if n_opt % 2 != 0:
                n_opt += 1
            m_opt = n_opt // 2
            # Interpolate from current best
            u_src = extract_u_from_h(best_h)
            u_opt = interpolate_u(u_src, m_opt)
            h_opt = build_sym_h(u_opt)
            sc_opt = official_score(h_opt)
            # Optimize
            for beta in [200, 1000, 5000]:
                if check_deadline():
                    break
                tl = min(time_left() * 0.1, 10.0)
                if tl < 2:
                    break
                u_opt = pg_solve_sym(u_opt, beta, time_limit=tl)
                h_opt = build_sym_h(u_opt)
                sc_opt = official_score(h_opt)
                if sc_opt < best_score - 1e-14:
                    update(h_opt, sc_opt)
            # SLSQP refinement
            if time_left() > 10:
                vals_opt = corr_vals(build_sym_h(u_opt))
                active_opt = find_active(vals_opt, frac=0.35, min_count=10)
                u_new, sc_new = slsqp_solve_sym(u_opt, active_opt, max_iter=300)
                if u_new is not None and sc_new < best_score - 1e-14:
                    update(build_sym_h(u_new), sc_new)

    # Phase 4: Non-symmetric optimization (in case symmetric is not optimal)
    if time_left() > 60:
        h_gen = best_h.copy()
        for beta in [500, 2000, 10000]:
            if check_deadline():
                break
            tl = min(time_left() * 0.15, 15.0)
            if tl < 3:
                break
            h_gen = pg_solve_general(h_gen, beta, time_limit=tl)
            sc = official_score(h_gen)
            if sc < best_score - 1e-14:
                update(h_gen, sc)

    # Phase 5: Random restarts on symmetric subspace
    if time_left() > 50:
        n_cur = len(best_h)
        m_cur = n_cur // 2
        n_starts = max(2, min(8, int(time_left() / 20)))
        for i in range(n_starts):
            if check_deadline():
                break
            sigma = 0.03 * (1 + 0.3 * i)
            u_base = extract_u_from_h(best_h)
            u_rand = np.clip(u_base + rng.normal(0, sigma, size=m_cur), 0.0, 1.0)
            h_rand = build_sym_h(u_rand)
            sc_rand = official_score(h_rand)
            if sc_rand < best_score - 1e-14:
                update(h_rand, sc_rand)
            for beta in [1000, 5000]:
                if check_deadline():
                    break
                tl = min(time_left() * 0.1, 8.0)
                if tl < 2:
                    break
                u_rand = pg_solve_sym(u_rand, beta, time_limit=tl)
                h_rand = build_sym_h(u_rand)
                sc_rand = official_score(h_rand)
                if sc_rand < best_score - 1e-14:
                    update(h_rand, sc_rand)

    # Phase 6: Intensive SLSQP refinement
    if time_left() > 30:
        u_fin = extract_u_from_h(best_h)
        h_fin = build_sym_h(u_fin)
        vals_fin = corr_vals(h_fin)
        # Multiple active set selections
        for frac in [0.2, 0.3, 0.5, 0.7]:
            if check_deadline():
                break
            active = find_active(vals_fin, frac=frac, min_count=8)
            u_new, sc = slsqp_solve_sym(u_fin, active, max_iter=500)
            if u_new is not None and sc < best_score - 1e-14:
                update(build_sym_h(u_new), sc)
                u_fin = u_new
            vals_fin = corr_vals(build_sym_h(u_fin))

    # Phase 7: Final polish with very high beta
    if time_left() > 20:
        u_pol = extract_u_from_h(best_h)
        for beta in [20000, 50000, 100000]:
            if check_deadline():
                break
            tl = min(time_left() * 0.2, 10.0)
            if tl < 2:
                break
            u_pol = pg_solve_sym(u_pol, beta, time_limit=tl)
            h_pol = build_sym_h(u_pol)
            sc = official_score(h_pol)
            if sc < best_score - 1e-14:
                update(h_pol, sc)

    # Phase 8: Final SLSQP sweep
    if time_left() > 15:
        u_last = extract_u_from_h(best_h)
        h_last = build_sym_h(u_last)
        vals_last = corr_vals(h_last)
        # Try all elements as active (top few)
        top_n = min(15, len(vals_last))
        top_idx = np.argsort(vals_last)[::-1][:top_n]
        for size in [5, 8, 12, top_n]:
            if check_deadline():
                break
            active = top_idx[:size].tolist()
            active.sort()
            u_new, sc = slsqp_solve_sym(u_last, active, max_iter=400)
            if u_new is not None and sc < best_score - 1e-14:
                update(build_sym_h(u_new), sc)

    # Final: ensure best_h is valid
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0.0, 1.0)
    # Ensure sum constraint
    n_fin = len(final_h)
    target = n_fin / 2.0
    residual = target - np.sum(final_h)
    if abs(residual) > 1e-12:
        final_h = make_feasible(final_h)

    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(final_h >= -1e-15), "Negative values in final h"
    assert np.all(final_h <= 1.0 + 1e-15), "Values > 1 in final h"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-10, "Sum constraint violated"

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
