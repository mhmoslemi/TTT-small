Let me also add a "subgradient descent" approach as an alternative to the LP:

At each step:
1. Compute all f(d) values
2. Compute the average gradient over the top-k active shifts (not just the single max)
3. Take a step in the negative gradient direction
4. Project onto feasible set

Using multiple active shifts (not just the single max) gives a more robust direction and can escape the "stuck at a single active constraint" problem.

```python
def _subgrad_step(h, k_active=3, step_size=0.1):
    """Take a subgradient step using top-k active constraints"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Top-k active
    top_k = np.argsort(-corr)[:k_active]
    # Average gradient
    grad = np.zeros(n)
    for r in top_k:
        d = r - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + d < n:
                g += (1.0 - h[j + d])
            if 0 <= j - d < n:
                g += -h[j - d]
            grad[j] += g
    grad /= len(top_k)
    # Center to maintain sum
    grad -= np.mean(grad)
    # Normalize
    norm = np.max(np.abs(grad))
    if norm > 1e-15:
        grad /= norm
    return grad
```

```python
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    x = np.clip(v - thresh, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_shift(h, d):
    """Gradient of f(d) = sum_i h[i]*(1-h[i+d]) with respect to h"""
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        jp = j + d
        if 0 <= jp < n:
            val += (1.0 - h[jp])
        jm = j - d
        if 0 <= jm < n:
            val += -h[jm]
        g[j] = val
    return g

def _lp_optimize(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Build gradient matrix for active constraints
        # Use top few active shifts for direction finding
        top_k = np.argsort(-corr)
        
        # Try LP with all constraints
        G = np.zeros((2*n-1, n))
        for d in range(-(n-1), n):
            r = n - 1 + d
            g = _grad_for_shift(h, d)
            G[r] = g
        
        # LP: find direction p that decreases all f(d)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo_b = max(-radius, -h[j])
            hi_b = min(radius, 1.0 - h[j])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        
        p = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            p = None
        
        if p is None or np.max(np.abs(p)) < 1e-12:
            # Fallback: use subgradient of worst constraint
            d_worst = top_k[0] - (n-1)
            g = _grad_for_shift(h, d_worst)
            g -= np.mean(g)
            norm = np.max(np.abs(g))
            if norm > 1e-15:
                p = -g / norm
            else:
                stall += 1
                radius *= 0.7
                if radius < 1e-6:
                    break
                continue
        
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)
        
        # Try several step sizes

        best_new_c5 = cur_c5
        for frac in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a = alpha_max * frac
            h_new = _project(h + a * p, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_new_c5:
                best_new_c5 = c5_new
                best_alpha = a
        
        if best_alpha > 0 and best_new_c5 < cur_c5 - 1e-12:
            h = _project(h + best_alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.7
            if stall > 20 and stall % 5 == 0:
                radius = max(radius, 0.05)

    return best_h, best_c5
```
