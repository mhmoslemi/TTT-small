    bounds.append((0.0, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            return h_new, sc
    except Exception:
        pass
    return None, None

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -600.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def fast_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(weights)):
        wv = weights[idx]
        if abs(wv) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        coeff = wv * dx
        if s >= 0:
            grad[s:s + L] += coeff * (1.0 - h[0:L])
            grad[0:L] -= coeff * h[s:s + L]
        else:
            t = -s
            grad[0:L] += coeff * (1.0 - h[t:n])
            grad[t:n] -= coeff * h[0:L]
    return grad

def pg_step(h, beta, alpha_max=0.5, max_ls=14):
    n = len(h)
    V = compute_all_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = fast_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new.copy()
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def random_binary_start(n, rng):
    """Create a random 0/1 start with exactly n/2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    # Add small noise to help escape symmetry
    h = h + rng.normal(0, 0.05, n)
    return make_feasible(h)

def structured_start(n, rng, pattern='random'):
    """Create various structured starting points."""
    if pattern == 'random':
        return random_binary_start(n, rng)
    elif pattern == 'interleaved':
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
        # Ensure exactly n/2 ones
        total = h.sum()
        if total > n / 2:
            excess = int(total - n / 2)
            ones = np.where(h > 0.5)[0]
            remove = rng.choice(ones, size=excess, replace=False)
            h[remove] = 0.0
        elif total < n / 2:
            deficit = int(n / 2 - total)
            zeros = np.where(h < 0.5)[0]
            add = rng.choice(zeros, size=deficit, replace=False)
            h[add] = 1.0
        h = h + rng.normal(0, 0.03, n)
        return make_feasible(h)
    elif pattern == 'blocks':
        # Random block structure
        h = np.zeros(n)
        pos = 0
        while pos < n:
            block_len = rng.integers(2, max(3, n // 4))
            if rng.random() < 0.5:
                h[pos:pos + block_len] = 1.0
            pos += block_len
        total = h.sum()
        if total != n / 2:
            if total > n / 2:
                excess = int(total - n / 2)
                ones = np.where(h > 0.5)[0]
                remove = rng.choice(ones, size=excess, replace=False)
                h[remove] = 0.0
            else:
                deficit = int(n / 2 - total)
                zeros = np.where(h < 0.5)[0]
                add = rng.choice(zeros, size=deficit, replace=False)
                h[add] = 1.0
        h = h + rng.normal(0, 0.03, n)
        return make_feasible(h)
    else:
        return random_binary_start(n, rng)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(10.0, 0.03 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    # Get initial values
    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        h_init = make_feasible(h_init)
    except (NameError, AttributeError):
        n = 60
        h_init = random_binary_start(n, rng)

    best_h = h_init.copy()
    best_score = official_score(h_init)

    # Phase 1: SCA optimization from initial point
    h_cur = h_init.copy()
    trust = 0.4
    stall = 0
    while time_left() > 300:
        h_new, sc = sca_lp_step(h_cur, trust=trust)
        if h_new is not None:
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = h_new.copy()
                stall = 0
                h_cur = h_new
                trust = min(trust * 1.15, 0.6)
            else:
                stall += 1
                # Try without trust region occasionally
                if stall % 10 == 0:
                    h_full, sc_full = sca_lp_step(h_cur, trust=None)
                    if h_full is not None and sc_full < best_score - 1e-12:
                        best_score = sc_full
                        best_h = h_full.copy()
                        h_cur = h_full
                        trust = 0.3
                        stall = 0
                trust = max(trust * 0.85, 0.05)
        else:
            stall += 5

        # Occasionally add PG step
        if stall > 15 or (stall > 5 and rng.random() < 0.3):
            h_pg, sc_pg = pg_step(h_cur, beta=80.0, alpha_max=0.3, max_ls=10)
            if sc_pg < best_score - 1e-12:
                best_score = sc_pg
                best_h = h_pg.copy()
            h_cur = h_pg if sc_pg < official_score(h_cur) else h_cur
            trust = 0.2
            stall = 0

        # Restart if stuck too long
        if stall > 60:
            h_cur = make_feasible(best_h + rng.normal(0, 0.08, len(best_h)))
            trust = 0.4
            stall = 0

    # Phase 2: Try multiple n values with SCA
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-10, -5, -3, 0, 3, 5, 10, 15]:
        nc = n_base + delta
        if 20 <= nc <= 100:
            n_candidates.add(nc)

    for nc in sorted(n_candidates):
        if time_left() < 120:
            break
        if nc == n_base:
            continue
        h_int = interpolate_to(best_h, nc)
        sc_int = official_score(h_int)
        h_cur = h_int.copy()
        if sc_int < best_score:
            best_score = sc_int
            best_h = h_int.copy()

        # SCA iterations
        trust = 0.3
        stall = 0
        while time_left() > 120 and stall < 40:
            h_new, sc = sca_lp_step(h_cur, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = h_new.copy()
                    stall = 0
                    trust = min(trust * 1.1, 0.5)
                else:
                    stall += 1
                    trust = max(trust * 0.85, 0.03)
            else:
                stall += 3
            if stall > 20:
                h_pg, sc_pg = pg_step(h_cur, beta=100.0, alpha_max=0.2, max_ls=10)
                if sc_pg < best_score - 1e-12:
                    best_score = sc_pg
                    best_h = h_pg.copy()
                h_cur = h_pg
                trust = 0.15
                stall = 0

    # Phase 3: Random restarts with different structures
    if time_left() > 100:
        n_try = len(best_h)
        for restart_iter in range(10):
            if time_left() < 80:
                break
            patterns = ['random', 'interleaved', 'blocks']
            pat = patterns[restart_iter % len(patterns)]
            h_rand = structured_start(n_try, rng, pattern=pat)
            sc_rand = official_score(h_rand)
            if sc_rand < best_score:
                best_score = sc_rand
                best_h = h_rand.copy()

            # SCA from this start
            h_cur = h_rand.copy()
            trust = 0.5
            stall = 0
            local_best = sc_rand
            local_best_h = h_cur.copy()
            while time_left() > 80 and stall < 50:
                h_new, sc = sca_lp_step(h_cur, trust=trust)
                if h_new is not None:
                    if sc < local_best - 1e-12:
                        local_best = sc
                        local_best_h = h_new.copy()
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_h = h_new.copy()
                        stall = 0
                        trust = min(trust * 1.1, 0.6)
                    else:
                        stall += 1
                        trust = max(trust * 0.85, 0.05)
                else:
                    stall += 3
                if stall > 25:
                    h_pg, sc_pg = pg_step(h_cur, beta=60.0, alpha_max=0.3, max_ls=10)
                    if sc_pg < local_best - 1e-12:
                        local_best = sc_pg
                        local_best_h = h_pg.copy()
                        if sc_pg < best_score - 1e-12:
                            best_score = sc_pg
                            best_h = h_pg.copy()
                    h_cur = h_pg if sc_pg < official_score(h_cur) else h_cur
                    trust = 0.2
                    stall = 0
            # Also try continuing from global best
            h_cur = best_h.copy()

    # Phase 4: Fine SCA polish on best solution
    h_cur = best_h.copy()
    trust = 0.15
    for _ in range(200):
        if time_left() < 40:
            break
        h_new, sc = sca_lp_step(h_cur, trust=trust)
        if h_new is not None:
            if sc < best_score - 1e-13:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
            else:
                trust = max(trust * 0.9, 0.02)
        else:
            trust = max(trust * 0.8, 0.02)

    # Phase 5: PG fine-tuning
    h_cur = best_h.copy()
    beta = 150.0
    stall = 0
    while time_left() > 25:
        h_new, sc = pg_step(h_cur, beta=beta, alpha_max=0.2, max_ls=12)
        if sc < best_score - 1e-13:
            best_score = sc
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h_cur = h_new
        beta = min(beta * 1.02, 600.0)
        if stall > 30:
            break

    # Phase 6: Final SCA with very tight trust
    h_cur = best_h.copy()
    for trust_val in [0.08, 0.04, 0.02, 0.01]:
        if time_left() < 15:
            break
        for _ in range(50):
            if time_left() < 15:
                break
            h_new, sc = sca_lp_step(h_cur, trust=trust_val)
            if h_new is not None:
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h_new.copy()
                h_cur = h_new
            else:
                break

    # Phase 7: Final PG micro-steps
    h_cur = best_h.copy()
    for _ in range(200):
        if time_left() < 5:
            break
        h_new, sc = pg_step(h_cur, beta=400.0, alpha_max=0.08, max_ls=8)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    return final_h, final_score, final_n
