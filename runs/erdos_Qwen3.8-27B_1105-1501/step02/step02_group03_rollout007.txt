    """Log-sum-exp smooth approximation to max correlation."""
    h = make_antisym(y, n)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + log_sum / beta
    # Gradient
    G = corr_grad_h(h)
    w = np.exp(shifted)
    w_sum = w.sum()

    m = n // 2

def compute_vals(h):

def make_feasible(h):

    lo, hi = -2.0, 2.0

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])

                residual += sub

                    break

def corr_grad_h(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] += dx * (1.0 - h[s:s+L])
            G[p, s:s+L] -= dx * h[:L]
        else:
            a = -s
            L = n - a
            G[p, a:a+L] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[a:a+L]
    return G

    h = make_antisym(y, n)
    vals = compute_vals(h)
    cmax = vals.max()

    shifted = np.clip(shifted, -700, 0)

    y = y0.copy()
