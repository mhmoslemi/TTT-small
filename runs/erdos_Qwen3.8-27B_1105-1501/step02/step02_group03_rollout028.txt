        vals = compute_vals(h_cur)
        active = find_active(vals, frac=0.25, min_count=10)
        u_new, sc = solve_minimax_slsqp(u, active, max_iter=500)
        if u_new is not None and sc < best_score - 1e-14:
            update(build_h_from_u(u_new))
    
    # Phase 4: Iterate active set refinement
    if time_left() > 30:
        u = project_to_antisym(best_h)
        for _ in range(4):
            if check_deadline():
                break
            h_cur = build_h_from_u(u)
            vals = compute_vals(h_cur)
            for frac in [0.2, 0.35, 0.5]:
                if check_deadline():
                    break
                active = find_active(vals, frac=frac, min_count=8)
                u_new, sc = solve_minimax_slsqp(u, active, max_iter=300)
                if u_new is not None and sc < best_score - 1e-14:
                    update(build_h_from_u(u_new))
                    u = u_new.copy()
                    vals = compute_vals(build_h_from_u(u))
    
    # Phase 5: Multi-resolution
    if time_left() > 50:
        m_cur = len(project_to_antisym(best_h))
        n_opts = sorted(set([
            max(48, 2 * (m_cur - 3)),
            max(52, 2 * (m_cur - 2)),
            2 * m_cur,
            min(76, 2 * (m_cur + 2)),
            min(80, 2 * (m_cur + 3)),
        ]))
        for n_opt in n_opts:
            if check_deadline() or time_left() < 15:
                break
            if n_opt == len(best_h):
                continue
            m_opt = n_opt // 2
            u_opt = interpolate_u(project_to_antisym(best_h), m_opt)
            h_int = build_h_from_u(u_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-14:
                update(h_int)
            # Optimize at this resolution
            for beta in [500, 2000, 8000]:
                if check_deadline():
                    break
                tl = min(time_left() * 0.15, 10.0)
                if tl < 3:
                    break
                u_opt = solve_lbfgsb(u_opt, beta, time_limit=tl)
                h_cand = build_h_from_u(u_opt)
                sc = official_score(h_cand)
                if sc < best_score - 1e-14:
                    update(h_cand)
            # SLSQP polish
            if time_left() > 10:
                h_cand = build_h_from_u(u_opt)
                vals_c = compute_vals(h_cand)
                act_c = find_active(vals_c, frac=0.3, min_count=8)
                u_new, sc = solve_minimax_slsqp(u_opt, act_c, max_iter=250)
                if u_new is not None and sc < best_score - 1e-14:
                    update(build_h_from_u(u_new))
    
    # Phase 6: Random restarts in anti-symmetric subspace
    if time_left() > 40:
        m_cur = len(project_to_antisym(best_h))
        u_best = project_to_antisym(best_h)
        n_starts = max(4, min(12, int(time_left() / 20)))
        sigmas = [0.02, 0.05, 0.08, 0.12, 0.18, 0.25]
        for i in range(n_starts):
            if check_deadline() or time_left() < 10:
                break
            sigma = sigmas[i % len(sigmas)]
            u_cand = np.clip(u_best + rng.normal(0, sigma, size=m_cur), 0, 1)
            h_cand = build_h_from_u(u_cand)
            sc_cand = official_score(h_cand)
            if sc_cand < best_score - 1e-14:
                update(h_cand)
            # Short L-BFGS-B polish
            if time_left() > 8:
                u_cand = solve_lbfgsb(u_cand, 3000, time_limit=min(time_left() * 0.15, 10.0))
                h_cand = build_h_from_u(u_cand)
                sc = official_score(h_cand)
                if sc < best_score - 1e-14:
                    update(h_cand)
                    u_cand = u_cand  # keep for next iteration
                if time_left() > 8:
                    h_cand = build_h_from_u(u_cand)
                    vals_c = compute_vals(h_cand)
                    act_c = find_active(vals_c, frac=0.25, min_count=8)
                    u_new, sc = solve_minimax_slsqp(u_cand, act_c, max_iter=200)
                    if u_new is not None and sc < best_score - 1e-14:
                        update(build_h_from_u(u_new))
    
    # Phase 7: Structured random patterns
    if time_left() > 30:
        m_cur = len(project_to_antisym(best_h))
        for trial in range(6):
            if check_deadline() or time_left() < 8:
                break
            # Create structured patterns
            x = np.linspace(0, 1, m_cur)
            if trial == 0:
                u_cand = 0.5 + 0.3 * np.sin(2 * np.pi * (1 + trial) * x)
            elif trial == 1:
                u_cand = 0.5 + 0.4 * np.cos(4 * np.pi * x) * 0.5
            elif trial == 2:
                u_cand = np.linspace(0.1, 0.9, m_cur)
            elif trial == 3:
                u_cand = 0.5 + 0.3 * np.sign(x - 0.5)
            elif trial == 4:
                u_cand = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * 2 * x), 0, 1)
            else:
                u_cand = np.clip(0.5 + rng.normal(0, 0.15, size=m_cur), 0, 1)
            u_cand = np.clip(u_cand, 0, 1)
            h_cand = build_h_from_u(u_cand)
            sc = official_score(h_cand)
            if sc < best_score - 1e-14:
                update(h_cand)
            if time_left() > 8:
                u_cand = solve_lbfgsb(u_cand, 2000, time_limit=min(time_left() * 0.2, 10.0))
                h_cand = build_h_from_u(u_cand)
                sc = official_score(h_cand)
                if sc < best_score - 1e-14:
                    update(h_cand)
    
    # Phase 8: Final intensive polish on best solution
    if time_left() > 20:
        u_best = project_to_antisym(best_h)
        # Very high beta L-BFGS-B
        for beta in [10000, 30000, 80000]:
            if check_deadline() or time_left() < 5:
                break
            tl = min(time_left() * 0.3, 12.0)
            u_best = solve_lbfgsb(u_best, beta, time_limit=tl)
            h_cand = build_h_from_u(u_best)
            sc = official_score(h_cand)
            if sc < best_score - 1e-14:
                update(h_cand)
        
        # Final SLSQP with broad active set
        if time_left() > 10:
            h_cand = build_h_from_u(u_best)
            vals = compute_vals(h_cand)
            active = find_active(vals, frac=0.4, min_count=12)
            u_new, sc = solve_minimax_slsqp(u_best, active, max_iter=600)
            if u_new is not None and sc < best_score - 1e-14:
                update(build_h_from_u(u_new))
        
        # One more round with tighter active set
        if time_left() > 10:
            u_best = project_to_antisym(best_h)
            h_cand = build_h_from_u(u_best)
            vals = compute_vals(h_cand)
            active = find_active(vals, frac=0.15, min_count=6)
            u_new, sc = solve_minimax_slsqp(u_best, active, max_iter=500)
            if u_new is not None and sc < best_score - 1e-14:
                update(build_h_from_u(u_new))
    
    # Phase 9: Try full (non-antisymmetric) optimization as cross-check
    if time_left() > 25:
        n_full = len(best_h)
        # Use L-BFGS-B on full space with penalty for sum constraint
        h_full = best_h.copy()
        
        def full_obj(u_full):
            h = np.clip(u_full, 0, 1)
            penalty = 1000.0 * (np.sum(h) - n_full / 2.0) ** 2
            return official_score(h) + penalty
        
        def full_grad(u_full):
            h = np.clip(u_full, 0, 1)
            # Numerical gradient for the penalty part
            penalty = 1000.0 * (np.sum(h) - n_full / 2.0) ** 2
            g = 2 * 1000.0 * (np.sum(h) - n_full / 2.0) * np.ones(n_full)
            # Approximate gradient of max via subgradient
            vals = compute_vals(h)
            imax = vals.argmax()
            dx = 2.0 / n_full
            s = imax - (n_full - 1)
            gv = np.zeros(n_full)
            if s >= 0:
                L = n_full - s
                gv[:L] += dx * (1.0 - h[:L])
                gv[s:] -= dx * h[:L]
            else:
                t = -s
                K = n_full - t
                gv[:K] += dx * (1.0 - h[t:])
                gv[t:] -= dx * h[:K]
            return gv + g
        
        try:
            res = minimize(full_obj, h_full, jac=full_grad, method='L-BFGS-B',
                           bounds=[(0, 1)] * n_full,
                           options={'maxiter': 200, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = np.clip(res.x, 0, 1)
                # Fix sum
                res_sum = n_full / 2.0 - np.sum(h_new)
                if abs(res_sum) > 1e-6:
                    if res_sum > 0:
                        order = np.argsort(1.0 - h_new)
                        for idx in order:
                            add = min(res_sum, 1.0 - h_new[idx])
                            h_new[idx] += add
                            res_sum -= add
                            if res_sum < 1e-12:
                                break
                    else:
                        order = np.argsort(h_new)
                        for idx in order:
                            sub = min(-res_sum, h_new[idx])
                            h_new[idx] -= sub
                            res_sum += sub
                            if res_sum > -1e-12:
                                break
                sc = official_score(h_new)
                if sc < best_score - 1e-14:
                    update(h_new)
        except Exception:
            pass
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    n_fin = len(final_h)
    
    # Ensure exact sum
    target = n_fin / 2.0
    res = target - np.sum(final_h)
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(1.0 - final_h)
            for idx in order:
                add = min(res, 1.0 - final_h[idx])
                final_h[idx] += add
                res -= add
                if res < 1e-14:
                    break
        else:
            order = np.argsort(final_h)
            for idx in order:
                sub = min(-res, final_h[idx])
                final_h[idx] -= sub
                res += sub
                if res > -1e-14:
                    break
    
    final_h = np.clip(final_h, 0, 1)
    final_score = official_score(final_h)
    
    # Safety: if something went wrong, use a simple feasible fallback
    if not np.isfinite(final_score) or final_score > 0.5:
        m_fb = 33
        u_fb = np.full(m_fb, 0.5)
        final_h = build_h_from_u(u_fb)
        final_score = official_score(final_h)
    
    final_bound = final_score + 1e-12
    return final_h, final_bound, n_fin
