    m = len(u0)
    n = 2 * m
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(z):
        u = z[:m]
        t = z[-1]
        u = clip_u(u)
        h = build_sym_h(u)
        vals = compute_vals(h)
        return t - vals[active_idx]
    
    def ineq_jac(z):
        u = z[:m]
        u = clip_u(u)
        G = grad_u_all(u)
        J = -G[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])
    
    h0 = build_sym_h(clip_u(u0))
    v0 = compute_vals(h0)
    t0 = v0.max() + 1e-8
    z0 = np.concatenate([u0, [t0]])
    bounds = [(0.0, 1.0)] * m + [(-1.0, 5.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            u_new = clip_u(res.x[:m])
            h_new = build_sym_h(u_new)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return u_new, sc
    except Exception:
        pass
    return None, None

def pg_solve_u(u0, beta, time_limit=20.0, max_iter=5000):
    m = len(u0)
    u = clip_u(u0.copy())
    f, g = smooth_obj_grad_u(u, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-28:
        return u
    alpha = 1.0 / (gnorm2 + 1e-12)
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = smooth_obj_grad_u(u, beta)[1]
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-28:
            break
        step = alpha
        for _ in range(30):
            u_trial = clip_u(u - step * g)
            f_trial = smooth_obj_grad_u(u_trial, beta)[0]
            if f_trial <= f - 1e-5 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-14:
            break
        f_new = f_trial
        if not np.isfinite(f_new):
            break
        u, f = u_trial, f_new
        if f_new < f - 1e-15:
            alpha = min(alpha * 1.5, 5.0 / (gnorm2 + 1e-12))
        else:
            alpha *= 0.6
    return u

def smooth_obj_grad_u(u, beta):
    m = len(u)
    n = 2 * m
    h = build_sym_h(u)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_u_all(u)
    grad = w @ G
    return obj, grad

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo, hi = -12.0, 12.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add

                if sub <= 0: break
                h[idx] -= sub
                residual += sub

    m = len(u)
    h = build_sym_h(u)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()

        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_u_all(u)

        return u
    alpha = min(1.0 / (gnorm2 + 1e-12), 2.0 / (np.max(g**2) + 1e-12))
    t0 = time.time()

        G = grad_u_all(u)
        J = -G[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])
    
    h0 = build_sym_h(clip_u(u0))
    v0 = compute_vals(h0)
    t0 = v0.max() + 1e-8
    z0 = np.concatenate([clip_u(u0), [t0]])
    bounds = [(0.0, 1.0)] * m + [(-1.0, 5.0)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            u_new = clip_u(res.x[:m])
            h_new = build_sym_h(u_new)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return u_new, sc
    except Exception:
        pass
    return None, None

def find_active(vals, frac=0.25, min_count=8):
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def project_to_anti(h_full, n_tgt):
    """Project h_full to anti-symmetric subspace of dimension n_tgt."""
    m = n_tgt // 2
    # Interpolate h_full to n_tgt
    n_src = len(h_full)
    src_x = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src
    tgt_x = np.linspace(0, 2, n_tgt, endpoint=False) + 1.0/n_tgt
    h_int = np.interp(tgt_x, src_x, h_full)
    # Project to anti-symmetric
    u = 0.5 * (h_int[:m] + 1.0 - h_int[n_tgt-1:m-1:-1])
    return clip_u(u)
