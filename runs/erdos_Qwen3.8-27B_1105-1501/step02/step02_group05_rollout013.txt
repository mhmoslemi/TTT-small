import numpy as np
from scipy.optimize import linprog
import time

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def score_exact(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def grad_for_lag(h, p):
    """Compute gradient of V[p] = dx * correlate(h, 1-h)[p] with respect to h."""
    n = len(h)
    dx = 2.0 / n
    k = p - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        # Term 1: (1 - h[j-k]) for j in [k, n-1]
        g[k:] += (1.0 - h[:n-k])
        # Term 2: -h[j+k] for j in [0, n-1-k]
        g[:n-k] -= h[k:]
    else:
        t = -k
        # Term 1: (1 - h[j+t]) for j in [0, n-1-t]
        g[:n-t] += (1.0 - h[t:])
        # Term 2: -h[j-t] for j in [t, n-1]
        g[t:] -= h[:n-t]
    return dx * g

        if s < target:

        else:

    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:

        else:

            else:

def grad_for_lag(h, p):
    n = len(h)
    dx = 2.0 / n
    k = p - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        t = -k
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return dx * g

    if not is_dead():
        h_new, sc_new = lp_refine_multi(best_h, is_dead, total_iters=200)
        if sc_new < best_score - 1e-14:
            best_h = h_new
            best_score = sc_new

    # Phase 2: Binary SA from multiple starts
    if not is_dead():
        binary_starts = [balance_binary(best_h, rng)]
        # Add a deterministic pattern
        n_b = n
        pattern = np.zeros(n_b)
        for i in range(n_b):
            pattern[i] = 1.0 if (i * 2 // n_b) < 1 else 0.0
        # Make it balanced
        pattern = balance_binary(pattern, rng)
        binary_starts.append(pattern)
        # Random balanced
        perm = rng.permutation(n_b)
        rand_b = np.zeros(n_b)
        rand_b[perm[:n_b//2]] = 1.0
        binary_starts.append(rand_b)

        for bs in binary_starts:
            if is_dead():
                break
            b_best, b_sc = binary_sa(bs, 3000, rng, is_dead)
            # Score binary directly
            if b_sc < best_score - 1e-14:
                best_h = b_best.copy()
                best_score = b_sc
            # Convert to continuous and refine
            h_cont = make_feasible(b_best)
            h_new, sc_new = lp_refine_multi(h_cont, is_dead, total_iters=100)
            if sc_new < best_score - 1e-14:
                best_h = h_new
                best_score = sc_new

    # Phase 3: Main intensive LP on best
    if not is_dead():
        h_new, sc_new = lp_refine_multi(best_h, is_dead, total_iters=300)
        if sc_new < best_score - 1e-14:
            best_h = h_new
            best_score = sc_new

    # Phase 4: Two-point polish
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 4))
        best_h, best_score = two_point_simple(best_h, n_tries, rng, is_dead)

    # Phase 5: More LP after two-point
    if not is_dead():
        h_new, sc_new = lp_refine_multi(best_h, is_dead, total_iters=200)
        if sc_new < best_score - 1e-14:
            best_h = h_new
            best_score = sc_new

    # Phase 6: Resolution scanning
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set([n_base - 4, n_base - 8, n_base + 4, n_base + 8,
                             40, 44, 48, 52, 56, 64, 72]))
        n_opts = [x for x in n_opts if 24 <= x <= 100 and x != n_base]
        for n_opt in n_opts:
            if is_dead():
                break
            h_int = interp_to(best_h, n_opt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-14:
                best_h = h_int
                best_score = sc_int
            # Quick LP on this resolution
            h_new, sc_new = lp_refine_multi(h_int, is_dead, total_iters=80)
            if sc_new < best_score - 1e-14:
                best_h = h_new
                best_score = sc_new

    # Phase 7: Final intensive two-point polish
    if not is_dead():
        n_tries = max(3000, int((deadline - time.time()) * 5))
        best_h, best_score = two_point_simple(best_h, n_tries, rng, is_dead)

    # Phase 8: Final LP polish
    if not is_dead():
        h_new, sc_new = lp_refine_multi(best_h, is_dead, total_iters=150)
        if sc_new < best_score - 1e-14:
            best_h = h_new
            best_score = sc_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    # Safety check
    if not np.all(np.isfinite(final_h)):
        final_h = np.full(60, 0.5)
        final_score = score_exact(final_h)
        final_n = 60

    if final_h.min() < -1e-10 or final_h.max() > 1.0 + 1e-10:
        final_h = np.clip(final_h, 0.0, 1.0)
        final_h = make_feasible(final_h)
        final_score = score_exact(final_h)

    if abs(final_h.sum() - final_n / 2.0) > 1e-9:
        final_h = make_feasible(final_h)
        final_score = score_exact(final_h)

    return final_h, final_score, final_n
