def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(res) < 1e-15:
                break

                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if add == 0:
                continue
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _lse_grad(h, beta, ax_lags=None, lag_grads=None):

def _slsqp_smooth(h0, beta, ax_lags=None, maxiter=150, maxtime=30):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()

    def _obj(x):
        return _lse_obj(x, beta, ax_lags)

    def _obj_jac(x):
        return _lse_grad(x, beta, ax_lags)

    try:
        res = minimize(_obj, h0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            out = _project(res['x'])
            if _eq_fun(out) > -1e-9 and _eq_fun(out) < 1e-9:
                return out
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_direct(h0, maxiter=120):
    n = len(h0)
    cv0 = _corr_vec(h0)
    t0 = float(cv0.max() + 0.003)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _dobj(z):
        return z[-1]

    def _dobj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _dineq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv

    def _deq(z):
        return z[:-1].sum() - n / 2.0

    def _deq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_dobj, z0, jac=_dobj_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _dineq},
                           {'type': 'eq', 'fun': _deq, 'jac': _deq_jac}

                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):

    except Exception:
        pass
    return h0

    y = h0.copy()

    t_val = 1.0
    for it in range(n_iter):
        grad = _lse_grad(y, beta, ax_lags)
        h_new = _project(y - alpha_init * grad + mu * (h_cur - h_prev))
        vel = h_cur - h_new
        h_prev = h_cur
        h_cur = h_new
        t_prev = t_val
        t_val = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = _project(h_cur + (t_prev - 1.0) / t_val * (h_cur - h_prev))
        h_avg += (h_cur - h_avg) / (it + 1)
        if it % 200 == 0:

            if s < best_s:

                best_h = h_cur.copy()

            ax = np.argsort(cv)[::-1][:12]
            new = _slsqp_smooth(best_h, beta=beta, ax_lags=ax, maxiter=120)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h = new
                best_score = ns

    # Phase 5: Direct min-max final
    if time.time() < deadline - 15:
        new = _slsqp_direct(best_h, maxiter=120)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h = new
            best_score = ns

    # Phase 6: Random restarts around best
    remaining = deadline - time.time()
    n_restarts = min(8, int(remaining / 30))
    for _ in range(n_restarts):
        if time.time() > deadline - 10:
            break
        sigma = 0.02 + 0.02 * rng.random()
        h_pert = _project(best_h + sigma * rng.standard_normal(len(best_h)))
        cur, cur_s = _chain_optimize(h_pert, deadline, rng)
        if cur_s < best_score - 1e-12:
            best_h = cur.copy()
            best_score = cur_s

    # Final FISTA pass
    if time.time() < deadline - 5:
        best_h, best_score = _fista_subgrad(best_h, beta=1500, n_iter=3000,
                                            alpha_init=0.002, mu=0.93)

    # Final projection and scoring
    best_h = _project(best_h)
    final_score = _score(best_h)

    return (best_h, final_score, len(best_h))
