def lp_direction(h, corr, D, trust):
    n = len(h)
    m = len(corr)  # 2n-1
    
    # Variables: [d_0, ..., d_{n-1}, t]
    c = np.zeros(n + 1)

    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 - ((i * phi) % 1.0)
    h = project_onto_constraint(h, n / 2.0)
    return h


def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_sinusoidal_h(n, rng):
    h = np.zeros(n)
    freq = rng.uniform(1.0, 4.0)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(freq * x + phase)
    h = project_onto_constraint(h, n / 2.0)
    return h


def gradient_guided_pairs(h, rng, n_pairs):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, "full")
    m = np.max(corr)
    temp = 0.05
    weights = np.exp((corr - m) / temp)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = weights[n - 1 + s]
        if w < 1e-14:
            continue
        i_lo1 = max(0, s)
        i_hi1 = min(n - 1, s + n - 1)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            grad[idx] += w * (1.0 - h[idx - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            grad[idx] -= w * h[idx + s]
    grad *= dx
    sources = [i for i in range(n) if grad[i] > 0.003]
    sinks = [i for i in range(n) if grad[i] < -0.003]
    for _ in range(n_pairs):
        if sources and sinks:
            i = sources[rng.randint(len(sources))]
            j = sinks[rng.randint(len(sinks))]
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_deadline = t0 + budget_s - 5.0

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        h0 = project_onto_constraint(h0, len(h0) / 2.0)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c0
        best_n = len(h0)
    except (NameError, Exception):
        h0 = np.full(70, 0.5)
        best_h = h0.copy()
        best_c5 = compute_c5(h0)
        best_n = len(h0)

    n_parent = len(h0)

    # Phase 1: LP optimization of parent
    remaining = hard_deadline - time.time()
    t1 = min(remaining * 0.30, 300.0)
    if t1 > 15:
        h_lp, _ = lp_optimize(best_h.copy(), t1, rng)
        c_lp = compute_c5(h_lp)
        if c_lp < best_c5:
            best_h = h_lp.copy()
            best_c5 = c_lp
    # Pair polish
    if time.time() < hard_deadline - 20:
        hp = best_h.copy()
        pair_sweep(hp, rng, n_passes=2)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_h = hp.copy()
            best_c5 = cp

    # Phase 2: Multi-size exploration
    size_list = []
    for delta in [4, 8, 12, 16, 20, 24, 28, 32, 36, 40]:
        s = n_parent - delta
        if 16 <= s < n_parent:
            size_list.append(s)
    for s in [50, 46, 42, 38, 34, 30, 26, 22, 18]:
        if s not in size_list and 16 <= s < n_parent:
            size_list.append(s)
    size_list = sorted(set(size_list), reverse=True)

    remaining = hard_deadline - time.time()
    if remaining > 30:
        n_sizes = len(size_list)
        for idx, s in enumerate(size_list):
            remaining = hard_deadline - time.time()
            if remaining < 15:
                break
            # Time allocation: proportional to remaining slots
            slots_left = n_sizes - idx
            alloc = remaining * 0.85 / max(slots_left, 1)
            alloc = min(alloc, 120.0)
            alloc = max(alloc, 10.0)

            # Starting points for this size
            starts = []
            if best_h is not None:
                if len(best_h) == s:
                    starts.append(best_h.copy())
                else:
                    starts.append(interpolate_h(best_h, s))
            starts.append(make_structural_h(s, rng))
            if remaining > 40:
                starts.append(make_random_binary(s, rng))
                starts.append(make_sinusoidal_h(s, rng))

            local_best_h = None
            local_best_c5 = float('inf')

            for si, start_h in enumerate(starts):
                time_left = hard_deadline - time.time()
                if time_left < 8:
                    break
                per_start = min(time_left * 0.4, alloc / max(len(starts) - si, 1))
                per_start = max(per_start, 5.0)

                h_try, _ = lp_optimize(start_h.copy(), per_start, rng)
                ct = compute_c5(h_try)
                if ct < local_best_c5:
                    local_best_c5 = ct
                    local_best_h = h_try.copy()

            if local_best_h is not None and local_best_c5 < best_c5 - 1e-6:
                if time.time() < hard_deadline - 10:
                    random_pair_polish(local_best_h, rng, best_n * 3)
                    ct2 = compute_c5(local_best_h)
                    if ct2 < local_best_c5:
                        local_best_c5 = ct2
                if local_best_c5 < best_c5:
                    best_h = local_best_h.copy()
                    best_c5 = local_best_c5
                    best_n = s

    # Phase 3: Refine best with gradient-guided + pair
    remaining = hard_deadline - time.time()
    if remaining > 20:
        h_final = best_h.copy()
        n_f = len(h_final)
        if remaining > 50:
            h_final, _ = lp_optimize(h_final, min(remaining * 0.35, 60.0), rng)
        if time.time() < hard_deadline - 10:
            gradient_guided_pairs(h_final, rng, n_f * n_f // 4)
        if time.time() < hard_deadline - 10:
            pair_sweep(h_final, rng, n_passes=1)
        cf = compute_c5(h_final)
        if cf < best_c5:
            best_h = h_final.copy()
            best_c5 = cf
            best_n = n_f

    # Phase 4: Final micro-polish
    remaining = hard_deadline - time.time()
    if remaining > 5:
        h_m = best_h.copy()
        n_m = len(h_m)
        random_pair_polish(h_m, rng, min(n_m * 4, int(remaining * 20)))
        cm = compute_c5(h_m)
        if cm < best_c5:
            best_h = h_m
            best_c5 = cm
            best_n = n_m

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
