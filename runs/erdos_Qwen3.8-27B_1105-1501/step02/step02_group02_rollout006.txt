def _score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_s(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    j0, j1 = max(0, s), min(n, n + s)
    if j1 > j0:
        g[j0:j1] += (1.0 - h[j0 - s:j1 - s]) * dx
    p0, p1 = max(0, -s), min(n, n - s)
    if p1 > p0:
        g[p0:p1] -= h[p0 + s:p1 + s] * dx
    return g

def _grad_top(h, m=6):
    n = len(h)
    cv = _corr_vec(h)
    idx = np.argsort(cv)[::-1][:m]
    vals = cv[idx]
    beta_w = 80.0
    w = np.exp(beta_w * (vals - vals[0]))
    s = w.sum()
    if s < 1e-300:
        w = np.ones(m) / m
    else:
        w /= s
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        s_shift = k - (n - 1)
        g += w[ii] * _grad_s(h, s_shift)
    return g

def _nesterov_subgrad(h0, n_iter, m=6, alpha0=0.012, rho=0.9):
    n = len(h0)
    x = h0.copy()
    best_h = h0.copy()

    alpha = alpha0
    t = 1.0
    y = h0.copy()
    for it in range(n_iter):
        g = _grad_top(y, m)
        y_new = _project(y - alpha * g)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        x_new = _project(x + beta_mom * (y_new - x))
        x = x_new
        t = t_new
        y = x + beta_mom * (y_new - x)
        y = _project(y)
        # Adaptive alpha
        if it % 100 == 0 or it < 20:
            s_cur = _score(x)
            if s_cur < best_s - 1e-14:

def _binary_anneal(n, n_iter, rng, h_init=None):
    """Simulated annealing on balanced binary vectors."""
    target = n // 2
    if h_init is not None:
        h = (h_init > 0.5).astype(float)
        # Fix count
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) != target:
            h = np.zeros(n)
            h[:target] = 1.0
    else:
        h = np.zeros(n)
        pos = rng.choice(n, size=target, replace=False)
        h[pos] = 1.0
    
    cur_score = _binary_score(h)
    best_h = h.copy()
    best_score = cur_score
    T = 0.003
    T_min = 0.0001
    
    for it in range(n_iter):
        # Pick a swap: one 1 and one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0

            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * 0.9995, T_min)
    
    return best_h, best_score

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_to_continuous(h_bin, n_cont, kernel_width=3):
    """Convert binary to continuous by smoothing and projecting."""
    if len(h_bin) == n_cont:
        h = h_bin.astype(float).copy()
    else:
        h = _interp(h_bin, n_cont)
    # Smooth
    k = np.exp(-0.5 * (np.arange(-kernel_width, kernel_width + 1) / (kernel_width * 0.5)) ** 2)
    k /= k.sum()
    h_smooth = np.convolve(h, k, mode='same')
    return _project(h_smooth)
