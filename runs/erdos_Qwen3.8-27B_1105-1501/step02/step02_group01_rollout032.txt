        else:
            pairs = [(rng.integers(n), rng.integers(n)) for _ in range(n*n//3)]
        
        for i, j in pairs:
            if i == j:
                continue
            # Line search: h[i] += t, h[j] -= t
            # Bounds: 0 <= h[i]+t <= 1, 0 <= h[j]-t <= 1
            t_lo = max(0.0, h[i] - 1.0 + 0.0, 0.0 - (h[j]))  # h[i]+t >= 0, h[j]-t >= 0
            t_hi = min(1.0 - h[i], h[j], 1.0)  # h[i]+t <= 1, h[j]-t <= 1
            t_lo = max(t_lo, -h[j])  # h[j]-t >= 0 means t <= h[j]
            t_hi = min(t_hi, h[j])    # h[j]-t >= 0
            t_lo = max(t_lo, -(h[i])) # h[i]+t >= 0 means t >= -h[i]
            t_hi = min(t_hi, 1.0 - h[i])  # h[i]+t <= 1
            
            t_lo = max(-h[i], -(h[j]))
            t_hi = min(1.0 - h[i], h[j])
            
            if t_lo >= t_hi:
                continue
            
            # Golden section search on [t_lo, t_hi]
            best_t = 0.0
            best_f = best_score + 1  # we want to find any improvement
            φ = (np.sqrt(5) - 1) / 2
            a, b = t_lo, t_hi
            c = b - φ * (b - a)
            d = a + φ * (b - a)
            f_c = _pair_score(h, i, j, c)
            f_d = _pair_score(h, i, j, d)
            for _ in range(30):
                if f_c < f_d:
                    b, d, f_d = d, c, f_c
                    d = b - φ * (b - a)
                    if f_d > best_f:
                        best_f = f_d
                    f_d = _pair_score(h, i, j, d)
                else:
                    a, c, f_c = c, d, f_d
                    c = a + φ * (b - a)
                    if f_c > best_f:
                        best_f = f_c
                    f_c = _pair_score(h, i, j, c)
                if b - a < 1e-10:
                    break
            
            mid = (a + b) / 2
            f_mid = _pair_score(h, i, j, mid)
            candidates = [0.0, c, d, mid]
            best_t, best_f = 0.0, best_score
            for t in candidates:
                t = np.clip(t, t_lo, t_hi)
                f = _pair_score(h, i, j, t)
                if f < best_f:
                    best_f = f
                    best_t = t

                h[j] -= best_t
                h = make_feasible(h)
                best_score = official_score(h)

    n = len(h)
