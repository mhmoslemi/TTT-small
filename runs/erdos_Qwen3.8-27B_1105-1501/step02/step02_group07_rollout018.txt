            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def fourier_to_h(params, n, K):
    """Convert Fourier parameters to h values.
    params: array of length 2*K (a_1, b_1, a_2, b_2, ..., a_K, b_K)
    h_i = 0.5 + sum_{k=1}^{K} (a_k cos(2πk i/n) + b_k sin(2πk i/n))
    Then clip and adjust sum.
    """
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = params[2*(k-1)]
        b = params[2*(k-1) + 1]
        h += a * np.cos(2 * np.pi * k * i / n) + b * np.sin(2 * np.pi * k * i / n)
    return make_feasible(h)

def h_to_fourier(h, K):
    """Extract first K Fourier coefficients from h."""
    n = len(h)
    i = np.arange(n)
    params = np.zeros(2 * K)
    for k in range(1, K + 1):
        params[2*(k-1)] = (2.0 / n) * np.sum((h - 0.5) * np.cos(2 * np.pi * k * i / n))
        params[2*(k-1) + 1] = (2.0 / n) * np.sum((h - 0.5) * np.sin(2 * np.pi * k * i / n))
    return params

def pairwise_step(h, i, j, n_grid=100):
    """Optimize the pair (h[i], h[j]) keeping their sum constant."""
    c = h[i] + h[j]
    # h[i] = t, h[j] = c - t
    # t must satisfy: 0 <= t <= 1 and 0 <= c-t <= 1
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    if t_hi - t_lo < 1e-12:
        return h, official_score(h)
    
    h_cur = h.copy()
    h_cur[i] = 0.5 * c  # start at midpoint
    h_cur[j] = 0.5 * c
    
    best_t = 0.5 * c
    best_f = official_score(h_cur)
    
    # Dense grid search
    ts = np.linspace(t_lo + 1e-10, t_hi - 1e-10, n_grid)
    for t in ts:
        h_cur[i] = t
        h_cur[j] = c - t
        f = official_score(h_cur)
        if f < best_f:
            best_f = f
            best_t = t
    
    # Refine around best
    if n_grid > 1:
        step = ts[1] - ts[0]
        t_lo2 = max(t_lo, best_t - 3 * step)
        t_hi2 = min(t_hi, best_t + 3 * step)
        ts2 = np.linspace(t_lo2, t_hi2, 50)
        for t in ts2:
            h_cur[i] = t
            h_cur[j] = c - t
            f = official_score(h_cur)
            if f < best_f:
                best_f = f
                best_t = t
    
    h[i] = best_t
    h[j] = c - best_t
    return h, best_f

def pg_step(h, beta, alpha_max=0.8, max_ls=15):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    # Softmax weights
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new.copy()
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.15):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None
