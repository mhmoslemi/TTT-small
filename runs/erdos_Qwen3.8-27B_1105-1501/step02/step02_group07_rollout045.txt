import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_lp(h, trust, use_all=True, threshold=None):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    
    if use_all:
        active = np.arange(len(V))
    else:
        active = np.where(V >= V_max - threshold)[0]
    
    if len(active) == 0:
        return None, None
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        g = np.zeros(n)
        if s >= 0:
            g[s:s+L] += (1.0 - h[0:L])
            g[0:L] -= h[s:s+L]
        else:
            t = -s
            g[0:L] += (1.0 - h[t:t+L])
            g[t:t+L] -= h[0:L]
        g *= dx
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo_d = max(-trust, -h[i])
        hi_d = min(trust, 1.0 - h[i])
        bounds_list.append((lo_d, hi_d))
    bounds_list.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def pg_step(h, beta, alpha_max=0.6, max_ls=16):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:s+L] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:s+L]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:t+L])
            grad[t:t+L] -= wi * h[0:L]
    
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-7:
            break
    return best_h, best_f

def binary_round(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin
