import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _pnorm_obj(x, p):
    cv = _corr_vec(x)
    if cv.max() < 1e-15:
        return 0.0
    m = cv.max()
    shifted = cv / m
    s = np.sum(shifted ** p)
    return m * (s + 1e-300) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    cv = _corr_vec(x)
    m = cv.max()
    if m < 1e-15:
        return np.zeros(n)
    shifted = cv / m
    w = (shifted ** p) / (np.sum(shifted ** p) + 1e-300)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for pidx in range(j0, j1):
            g[pidx] += wk * (1.0 - x[pidx - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for pidx in range(p0, p1):
            g[pidx] += wk * (-x[pidx + s])
    return g

def _slsqp_pnorm(h0, p, maxiter=120):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_pnorm_obj, h0, args=(p,), jac=_pnorm_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _direct_minimax(h0, maxiter=150):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = cv.max() + 0.003
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv

    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_push(h, p=1.5):
    return _project(h ** p)

def _mirror(h):
    return _project(h[::-1].copy())

def _de_mutation(base, pop, F, CR, rng):
    n = len(base)
    a, b, c = rng.choice(len(pop), size=3, replace=False)
    mutant = pop[a] + F * (pop[b] - pop[c])
    # Blend with base
    mask = rng.random(n) < CR
    trial = base.copy()
    trial[mask] = mutant[mask]
    return _project(trial)

def _de_stage(h_best, n, budget_frac, rng):
    """Differential evolution global search."""
    pop_size = max(24, min(50, int(budget_frac * 40)))
    pop = [h_best.copy()]
    pop.append(_mirror(h_best))
    for sigma in [0.1, 0.2, 0.35]:
        pop.append(_project(h_best + sigma * rng.standard_normal(n)))
    for p in [1.3, 2.0, 4.0]:
        pop.append(_binary_push(h_best, p))
    # Structured: block patterns
    for blk in [4, 7, 11]:
        pattern = np.tile([1.0, 0.0], blk)
        pat = np.repeat(pattern, n // blk + 1)[:n]
        pop.append(_project(0.5 * h_best + 0.5 * pat))
    while len(pop) < pop_size:
        pop.append(_project(h_best + 0.25 * rng.standard_normal(n)))

    scores = [_score(p) for p in pop]
    best_idx = int(np.argmin(scores))
    best_val = scores[best_idx]
    best_h = pop[best_idx].copy()

    gen_limit = max(30, int(budget_frac * 200))
    for gen in range(gen_limit):
        improved = False
        for i in range(pop_size):
            F = 0.5 + 0.5 * rng.random()
            CR = 0.6 + 0.3 * rng.random()
            trial = _de_mutation(pop[i], pop, F, CR, rng)
            ts = _score(trial)
            if ts < scores[i]:
                pop[i] = trial
                scores[i] = ts
                improved = True
                if ts < best_val:
                    best_val = ts
                    best_h = trial.copy()
                    best_idx = i
        if not improved and gen > 10:
            break
        # Replace worst with perturbation of best
        worst = int(np.argmax(scores))
        pop[worst] = _project(pop[best_idx] + 0.05 * rng.standard_normal(n))
        scores[worst] = _score(pop[worst])

    return best_h, best_val

def _coord_descent(h, n_iter=200, deadline=None):
    """Active-set coordinate descent: repeatedly minimize the max correlation."""
    n = len(h)
    cur = h.copy()
    cur_s = _score(cur)
    best_h = cur.copy()
    best_s = cur_s
    alpha = 0.1
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        cv = _corr_vec(cur)
        k_max = int(np.argmax(cv))
        s = k_max - (n - 1)
        # Gradient of F_{k_max}
        g = np.zeros(n)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for pidx in range(j0, j1):
            g[pidx] += (1.0 - cur[pidx - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for pidx in range(p0, p1):
            g[pidx] += (-cur[pidx + s])
        # Projected gradient step
        new = _project(cur - alpha * g)
        new_s = _score(new)
        if new_s < best_s - 1e-14:
            best_s = new_s
            best_h = new.copy()
            cur = new
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.7
            if alpha < 1e-6:
                break
    return best_h, best_s

def _fista_pegasus(h, beta, n_iter, alpha_init=0.005, mu=0.88):
    """FISTA with log-sum-exp smooth max."""
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    y = h.copy()
    h_avg = h.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_s = _score(h)
    t = 1.0
    for it in range(n_iter):
        cv = _corr_vec(y)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            j0 = max(0, s)
            j1 = min(n, n + s)
            for pidx in range(j0, j1):
                grad[pidx] += wk * (1.0 - y[pidx - s])
            p0 = max(0, -s)
            p1 = min(n, n - s)
            for pidx in range(p0, p1):
                grad[pidx] += wk * (-y[pidx + s])
        h_new = y - alpha_init * (1.0 - mu) * vel + mu * mu * (h_cur - h_prev)
        h_new = _project(h_new)
        t += 1.0
        vel = h_cur - h_new
        h_prev = h_cur
        h_cur = h_new
        t_prev = t - 1.0
        t_new = 1.5 * t
        y = h_cur + (t_prev / t_new) * (h_cur - h_prev)
        y = _project(y)
        h_avg += (h_cur - h_avg) / t
        if it % 200 == 0:
            s_val = _score(h_cur)
            if s_val < best_s:
                best_s = s_val
                best_h = h_cur.copy()
        if it % 500 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s_val = _score(h_cur)
    if s_val < best_s:
        best_s = s_val
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s
