import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, r, n):
    """Gradient of score[r] w.r.t. h. Returns array of length n."""
    tau = r - (n - 1)
    g = np.zeros(n)
    if tau >= 0:
        m = n - tau
        if m > 0:
            if tau > 0:
                g[tau:] = 1.0 - h[:m]
                g[:m] -= h[tau:]
            else:
                g[:] = 1.0 - 2.0 * h
    else:
        g_ = -tau
        m = n - g_
        if m > 0:
            g[:m] = 1.0 - h[g_:]
            g[g_:] -= h[:m]
    return g

def _smooth_max_grad(h, beta):
    """Gradient of the smooth max (log-sum-exp) of scores."""
    n = len(h)
    scores = _scores(h)
    # Softmax weights
    shifted = scores - np.max(scores)
    w = np.exp(beta * shifted)
    w /= np.sum(w)
    # Gradient = G.T @ w
    # G[r, j] = d(score[r])/d(h[j])
    # So (G.T @ w)[j] = sum_r w[r] * G[r, j]
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] > 1e-15:
            grad += w[r] * _score_grad(h, r, n)
    return grad, scores, w

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, p, c5_cur):
    """Golden section line search along h + alpha*p."""
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    # Coarse grid
    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.01 * alpha_max, alpha_max, 20)
    for a in alphas:
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    idx = int(best_a / alpha_max * 19)
    lo_a = max(0.0, alphas[max(0, idx-2)])
    hi_a = min(alpha_max, alphas[min(19, idx+2)])
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _smooth_descend(h, deadline, beta_init=20.0, maxiters=3000):
    """Main optimization: smooth-max gradient descent with adaptive beta."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_init

        p = -p  # descent direction

            continue

        p_norm = np.max(np.abs(p))
        if p_norm > 1.0:

            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Adapt beta: if improvement was good, keep or slightly increase
            beta = min(beta * 1.02, 100.0)

            # Try different beta

                beta = max(5.0, beta * 0.7)
            if stall > 30:
                beta = min(100.0, beta * 1.5)
    
    return best_h, best_c5

def _multi_lp_direction(h, scores, G, top_k, radius):
    """Find direction p that reduces top-K scores simultaneously.
    max t s.t. G[r_k, :] @ p <= -t for each top-k row r_k
                 sum(p) = 0
                 -radius <= p[j] - h[j] <= radius (box relative to h)
    Actually, bounds on p directly: |p[j]| <= radius
    """
    n = len(h)
    # Sort scores, get top K indices
    order = np.argsort(scores)[::-1][:top_k]
    
    # Variables: [p_0, ..., p_{n-1}, t]  (n+1 variables)
    # Objective: maximize t => minimize -t
    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -t = maximize t
    
    # Inequality constraints: G[r_k, :] @ p <= -t
    # Rewritten: G[r_k, :] @ p + t <= 0
    A_ub = np.zeros((top_k, n + 1))
    for i, r in enumerate(order):
        A_ub[i, :n] = G[r, :]
        A_ub[i, n] = 1.0
    b_ub = np.zeros(top_k)
    
    # Also: each individual score should not increase too much
    # Actually the constraint G[r_k] @ p <= -t already ensures reduction
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: |p[j]| <= radius
    bounds = [(-radius, radius)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None
