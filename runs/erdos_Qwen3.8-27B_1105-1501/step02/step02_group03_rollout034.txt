def soft_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    vmax = vals.max()
    shifted = beta * (vals - vmax)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    logsum = max_shifted + np.log(exp_shifted.sum())
    obj = vmax + logsum / beta
    w = exp_shifted / exp_shifted.sum()
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-12:
            continue
        k = p - (n - 1)
        if k >= 0:
            L = n - k
            if L <= 0:
                continue
            grad[k:k+L] += wp * dx * (1.0 - h[:L])
            grad[:L] -= wp * dx * h[k:k+L]
        else:
            a = -k
            L = n - a
            if L <= 0:
                continue
            grad[:L] += wp * dx * (1.0 - h[a:a+L])
            grad[a:a+L] -= wp * dx * h[:L]
    
    return obj, grad

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):

    lo, hi = -11.0, 11.0

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break

                residual -= add

                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def sym_project(h):

def sym_to_full(u):
    m = len(u)
    h = np.zeros(2 * m)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

def full_to_sym(h):

            grad_h[a:a+L] -= wp * dx * h[:L]

def find_active(h, frac=0.3, min_count=10):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    threshold = cmax - max(0.015 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0])
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)
