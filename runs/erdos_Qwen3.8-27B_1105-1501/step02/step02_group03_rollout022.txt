def interpolate_to_n(h_src, n_tgt):

    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    # Make sum exact
    target = n_tgt / 2.0
    residual = target - np.sum(h_int)
    if abs(residual) > 1e-14:
        # Distribute uniformly
        h_int += residual / n_tgt
        h_int = np.clip(h_int, 0.0, 1.0)
        # Fix residual again
        residual2 = target - np.sum(h_int)
        if abs(residual2) > 1e-14:
            # Add to non-saturated
            if residual2 > 0:
                order = np.argsort(1.0 - h_int)
            else:
                order = np.argsort(h_int)
            for idx in order:
                if residual2 > 0:
                    add = min(residual2, 1.0 - h_int[idx])
                    h_int[idx] += add
                    residual2 -= add
                else:
                    sub = min(-residual2, h_int[idx])
                    h_int[idx] -= sub
                    residual2 += sub
                if abs(residual2) < 1e-14:
                    break
    return h_int

def official_score(h):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def build_anti_sym(y, n):
    if n % 2 == 0:
        m = n // 2
        h = np.empty(n)
        h[:m] = y
        h[m:] = 1.0 - y[::-1]
    else:
        m = (n - 1) // 2
        h = np.empty(n)
        h[:m] = y
        h[m] = 0.5
        h[m+1:] = 1.0 - y[::-1]
    return h

def smooth_obj_grad_ys(y, n, beta):

    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -300, 300)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    m = n // 2 if n % 2 == 0 else (n - 1) // 2

    dh *= dx
    idx = np.arange(m)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            best_h = parent.copy()
            best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        x = np.arange(n0)
        h_init = 0.5 + 0.15 * np.sin(2 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    n_parent = len(best_h)

    # Try different n values
    n_list = sorted(set([n - 2, n for n in [n_parent, n_parent - 2, n_parent + 2, n_parent + 4] if 40 <= n <= 80]))

    for n in n_list:
        if time_left() < 25:
            break

        # Interpolate/prepare h at this n
        if n == n_parent:
            h_n = best_h.copy()
        else:
            h_n = interpolate_to_n(best_h, n)

        # Anti-symmetric projection
        y = project_anti_sym(h_n)
        h_asy = build_anti_sym(y, n)
        sc_asy = official_score(h_asy)
        if sc_asy < best_score - 1e-14:
            best_h = h_asy.copy()
            best_score = sc_asy

        # PG with log-sum-exp in anti-symmetric space
        if time_left() > 20:
            betas = [20, 50, 100, 200, 500, 1000, 3000]
            tl = min(time_left() * 0.2, 50.0)
            if tl > 3:
                y = pg_solve_ys(y, n, betas, time_limit=tl)
                h_new = build_anti_sym(y, n)
                sc = official_score(h_new)
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc

        # SLSQP polish in anti-symmetric space
        if time_left() > 15:
            h_cur = build_anti_sym(y, n)
            vals = compute_vals(h_cur)
            active = find_active_lags(vals, top_k=15)
            y_pol, sc_pol, h_pol = slsqp_polish_ys(y, n, active, max_iter=300)
            if y_pol is not None and sc_pol < best_score - 1e-14:
                best_h = h_pol.copy()
                best_score = sc_pol
                y = y_pol

            # Second SLSQP with more constraints
            if time_left() > 15:
                h_cur = build_anti_sym(y, n)
                vals = compute_vals(h_cur)
                active2 = find_active_lags(vals, top_k=25)
                y_pol2, sc_pol2, h_pol2 = slsqp_polish_ys(y, n, active2, max_iter=400)
                if y_pol2 is not None and sc_pol2 < best_score - 1e-14:
                    best_h = h_pol2.copy()
                    best_score = sc_pol2
                    y = y_pol2

    # Full-space SLSQP on best
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        vals_b = compute_vals(best_h)
        active_b = find_active_lags(vals_b, top_k=18)
        h_new, sc_new = slsqp_full(best_h, active_b, max_iter=500)
        if h_new is not None and sc_new < best_score - 1e-14:
            best_h = h_new
            best_score = sc_new

        # Second round with updated active set
        if time_left() > 20:
            vals_b2 = compute_vals(best_h)
            active_b2 = find_active_lags(vals_b2, top_k=25)
            h_new2, sc_new2 = slsqp_full(best_h, active_b2, max_iter=400)
            if h_new2 is not None and sc_new2 < best_score - 1e-14:
                best_h = h_new2
                best_score = sc_new2

    # Try symmetric break around best
    if time_left() > 20 and best_h is not None:
        h_br, sc_br = try_symmetric_break(best_h, len(best_h), rng, n_trials=5)
        if sc_br < best_score - 1e-14:
            best_h = h_br
            best_score = sc_br
            # Polish the broken-symmetry solution
            if time_left() > 15:
                vals_br = compute_vals(best_h)
                active_br = find_active_lags(vals_br, top_k=20)
                h_new3, sc_new3 = slsqp_full(best_h, active_br, max_iter=400)
                if h_new3 is not None and sc_new3 < best_score - 1e-14:
                    best_h = h_new3
                    best_score = sc_new3

    # Random restarts in anti-symmetric space
    if time_left() > 40:
        n_cur = len(best_h)
        n_trials = max(2, min(6, int(time_left() / 15)))
        for trial in range(n_trials):
            if time_left() < 15:
                break
            sigma = [0.03, 0.06, 0.1, 0.15, 0.2, 0.3][trial % 6]
            h_rand = make_feasible(best_h + rng.normal(0, sigma, size=n_cur))
            sc_rand = official_score(h_rand)
            if sc_rand < best_score - 1e-14:
                best_h = h_rand.copy()
                best_score = sc_rand
            # Project and optimize
            y_r = project_anti_sym(h_rand)
            y_r = pg_solve_ys(y_r, n_cur, [500, 2000], time_limit=min(time_left() * 0.15, 12))
            h_r = build_anti_sym(y_r, n_cur)
            sc_r = official_score(h_r)
            if sc_r < best_score - 1e-14:
                best_h = h_r.copy()
                best_score = sc_r
            if time_left() > 12:
                vals_r = compute_vals(h_r)
                act_r = find_active_lags(vals_r, top_k=20)
                yp, scp, hp = slsqp_polish_ys(y_r, n_cur, act_r, max_iter=300)
                if yp is not None and scp < best_score - 1e-14:
                    best_h = hp.copy()
                    best_score = scp

    # Final high-beta PG polish
    if time_left() > 20 and best_h is not None:
        n_fin = len(best_h)
        y_fin = project_anti_sym(best_h)
        for beta in [5000, 15000]:
            if time_left() < 10:
                break
            tl = min(time_left() * 0.3, 15.0)
            if tl < 3:
                break
            y_fin = pg_solve_ys(y_fin, n_fin, [beta], time_limit=tl)
            h_fin = build_anti_sym(y_fin, n_fin)
            sc_fin = official_score(h_fin)
            if sc_fin < best_score - 1e-14:
                best_h = h_fin.copy()
                best_score = sc_fin

    # Final SLSQP
    if time_left() > 15 and best_h is not None:
        vals_fin = compute_vals(best_h)
        active_fin = find_active_lags(vals_fin, top_k=30)
        h_new, sc_new = slsqp_full(best_h, active_fin, max_iter=600)
        if h_new is not None and sc_new < best_score - 1e-14:
            best_h = h_new
            best_score = sc_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Safety check
    if not np.all(final_h >= -1e-15) or not np.all(final_h <= 1.0 + 1e-15):
        final_h = np.clip(final_h, 0.0, 1.0)
        final_h = make_feasible(final_h)
        final_score = official_score(final_h)
        final_n = len(final_h)

    return final_h, final_score + 1e-12, final_n
