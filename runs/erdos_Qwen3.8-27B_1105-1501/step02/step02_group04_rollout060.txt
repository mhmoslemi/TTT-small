def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Find a coordinate with slack
        if residual > 0:
            candidates = np.where(result < 1.0)[0]
        else:
            candidates = np.where(result > 0.0)[0]
        if len(candidates) > 0:
            c = candidates[0]
            amend = min(residual, 1.0 - result[c]) if residual > 0 else max(residual, -result[c])
            result[c] += amend
            residual -= amend
        # If still residual, spread evenly
        if abs(residual) > 1e-12:
            if residual > 0:
                candidates = np.where(result < 1.0)[0]
            else:
                candidates = np.where(result > 0.0)[0]
            if len(candidates) > 0:
                amt = residual / len(candidates)
                for c in candidates:
                    result[c] = np.clip(result[c] + amt, 0.0, 1.0)
    return result

    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")

    exp_vals = np.exp(shifted)
    S = np.sum(exp_vals)
    weights = exp_vals / S
    value = dx * (m + T * np.log(S))
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = weights[n - 1 + s]
        if w < 1e-18:
            continue
        p_lo = max(0, s)
        p_hi = min(n - 1, s + n - 1)
        if p_lo <= p_hi:
            p_idx = np.arange(p_lo, p_hi + 1)
            grad[p_idx] += w * (1.0 - h[p_idx - s])
        p_lo2 = max(0, -s)
        p_hi2 = min(n - 1, n - 1 - s)
        if p_lo2 <= p_hi2:
            p_idx2 = np.arange(p_lo2, p_hi2 + 1)
            grad[p_idx2] -= w * h[p_idx2 + s]
    grad *= dx
    return value, grad


def _slsqp_obj(x, T):
    val, _ = soft_value_grad(x, T)
    return val


def _slsqp_jac(x, T):
    _, g = soft_value_grad(x, T)
    return g


def _cons_fun(x, n):
    return float(np.sum(x) - n / 2.0)


def _cons_jac(x, n):
    return np.ones(n)


def run_slsqp_from(h_start, T, n, maxiter=80, ftol=1e-8):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        result = minimize(
            _slsqp_obj, h_start.copy(), args=(T,),
            jac=_slsqp_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return result.x
    except Exception:
        return h_start.copy()


def projected_gradient_step(h, T, n, step_size):
    _, grad = soft_value_grad(h, T)
    h_new = h - step_size * grad
    h_new = project_onto_constraint(h_new, n / 2.0)
    return h_new


def pair_line_search(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hi_v, hj_v - 1.0)
    hi = min(1.0 - hi_v, hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(35):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v + m1
        h[j] = hj_v - m1
        f1 = compute_c5(h)
        h[i] = hi_v + m2
        h[j] = hj_v - m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    new_hi = np.clip(hi_v + t_opt, 0.0, 1.0)
    new_hj = np.clip(hj_v - t_opt, 0.0, 1.0)
    if abs(new_hi + new_hj - hi_v - hj_v) < 1e-9:
        return t_opt
    return 0.0


def full_pair_sweep(h, rng, n):
    perm = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            if i > j:
                i, j = j, i
            delta = pair_line_search(h, i, j)
            if abs(delta) > 1e-12:
                h[i] += delta
                h[j] -= delta


def random_pair_sweep(h, rng, n, num_pairs):
    for _ in range(num_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if i > j:
            i, j = j, i
        delta = pair_line_search(h, i, j)
        if abs(delta) > 1e-12:
            h[i] += delta
            h[j] -= delta


def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] += 0.5
    return h


def make_sine_start(n, rng):
    x = np.arange(n) / n
    h = np.zeros(n)
    for _ in range(3):
        freq = rng.randint(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h += np.sin(2 * np.pi * freq * x + phase)
    h = 0.5 + 0.3 * h / (np.max(np.abs(h)) + 1e-10)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_constraint(h, n / 2.0)


def slsqp_anneal(h_start, n, temps, deadline, maxiter_per_t=60):
    h_cur = h_start.copy()
    for T in temps:
        if time.time() > deadline:
            break
        h_prop = run_slsqp_from(h_cur, T, n, maxiter=maxiter_per_t, ftol=1e-8)
        h_prop = project_onto_constraint(h_prop, n / 2.0)
        if not np.all(np.isfinite(h_prop)):
            continue
        h_cur = h_prop
    return h_cur


def basin_hop_slsqp(h_base, n, deadline, rng):
    best_local = h_base.copy()
    best_local_c5 = compute_c5(best_local)
    scales = [0.06, 0.04, 0.03, 0.02, 0.015, 0.01, 0.007]
    short_temps = [0.8, 0.3, 0.1, 0.04, 0.015]
    attempts = 0
    while time.time() < deadline:
        scale = scales[attempts % len(scales)]
        h_pert = best_local + rng.randn(n) * scale
        h_pert = project_onto_constraint(h_pert, n / 2.0)
        h_prop = slsqp_anneal(h_pert, n, short_temps, deadline, maxiter_per_t=40)
        h_prop = project_onto_constraint(h_prop, n / 2.0)
        if not np.all(np.isfinite(h_prop)):
            attempts += 1
            continue
        c_prop = compute_c5(h_prop)
        if c_prop < best_local_c5 - 1e-10:
            best_local_c5 = c_prop
            best_local = h_prop.copy()
        attempts += 1
    return best_local, best_local_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_deadline = t0 + budget_s - 3.0

    n = 70
    target_sum = n / 2.0
    best_h = None
    best_c5 = float('inf')

    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) == n and np.all(np.isfinite(h0)):
            h0 = project_onto_constraint(h0, target_sum)
            c = compute_c5(h0)
            if c < best_c5:
                best_c5 = c
                best_h = h0.copy()
    except (NameError, Exception):
        pass

    # Fallback starts if no parent
    if best_h is None:
        for _ in range(15):
            h_c = project_onto_constraint(make_random_binary(n, rng), target_sum)
            c = compute_c5(h_c)
            if c < best_c5:
                best_c5 = c
                best_h = h_c.copy()
        for _ in range(5):
            h_c = make_sine_start(n, rng)
            c = compute_c5(h_c)
            if c < best_c5:
                best_c5 = c
                best_h = h_c.copy()
        if best_h is None:
            best_h = np.full(n, 0.5)
            best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing from best start (25% budget)
    phase1_end = t0 + budget_s * 0.25
    main_temps = [2.0, 1.2, 0.7, 0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006]
    h_annealed = slsqp_anneal(best_h, n, main_temps, phase1_end, maxiter_per_t=80)
    h_annealed = project_onto_constraint(h_annealed, target_sum)
    c_annealed = compute_c5(h_annealed)
    if c_annealed < best_c5:
        best_c5 = c_annealed
        best_h = h_annealed.copy()

    # Additional annealing rounds with slightly different starts
    while time.time() < phase1_end:
        h_pert = best_h + rng.randn(n) * 0.02
        h_pert = project_onto_constraint(h_pert, target_sum)
        h_prop = slsqp_anneal(h_pert, n, [0.5, 0.2, 0.08, 0.03, 0.01], phase1_end, maxiter_per_t=50)
        h_prop = project_onto_constraint(h_prop, target_sum)
        c_prop = compute_c5(h_prop)
        if c_prop < best_c5:
            best_c5 = c_prop
            best_h = h_prop.copy()

    # Phase 2: Basin hopping (35% budget)
    phase2_end = t0 + budget_s * 0.60
    h_hop, c_hop = basin_hop_slsqp(best_h, n, phase2_end, rng)
    if c_hop < best_c5:
        best_c5 = c_hop
        best_h = h_hop.copy()

    # Phase 2b: More targeted SLSQP with fine temps around current best
    while time.time() < phase2_end:
        fine_temps = [0.05, 0.02, 0.008, 0.003, 0.001]
        h_prop = slsqp_anneal(best_h, n, fine_temps, phase2_end, maxiter_per_t=60)
        h_prop = project_onto_constraint(h_prop, target_sum)
        c_prop = compute_c5(h_prop)
        if c_prop < best_c5 - 1e-10:
            best_c5 = c_prop
            best_h = h_prop.copy()
        else:
            # Small perturbation then fine SLSQP
            h_pert = best_h + rng.randn(n) * 0.005
            h_pert = project_onto_constraint(h_pert, target_sum)
            h_prop = slsqp_anneal(h_pert, n, [0.03, 0.01, 0.003], phase2_end, maxiter_per_t=50)
            h_prop = project_onto_constraint(h_prop, target_sum)
            c_prop = compute_c5(h_prop)
            if c_prop < best_c5 - 1e-10:
                best_c5 = c_prop
                best_h = h_prop.copy()

    # Phase 3: Pair polishing (30% budget)
    phase3_end = t0 + budget_s * 0.90
    h_work = best_h.copy()
    sweeps_done = 0
    while time.time() < phase3_end:
        h_sweep = h_work.copy()
        full_pair_sweep(h_sweep, rng, n)
        c_sweep = compute_c5(h_sweep)
        if c_sweep < best_c5 - 1e-12:
            best_c5 = c_sweep
            best_h = h_sweep.copy()
            h_work = h_sweep.copy()
            sweeps_done = 0
        else:
            sweeps_done += 1
            h_work = h_sweep.copy()
            if sweeps_done >= 3:
                # Perturb and do random pair moves
                h_work = best_h + rng.randn(n) * 0.008
                h_work = project_onto_constraint(h_work, target_sum)
                random_pair_sweep(h_work, rng, n, n * n)
                c_work = compute_c5(h_work)
                if c_work < best_c5:
                    best_c5 = c_work
                    best_h = h_work.copy()
                sweeps_done = 0

    # Phase 4: Final fine pair optimization (10% budget)
    h_work = best_h.copy()
    while time.time() < hard_deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if i > j:
            i, j = j, i
        delta = pair_line_search(h_work, i, j)
        if abs(delta) > 1e-12:
            h_work[i] += delta
            h_work[j] -= delta
            c_wk = compute_c5(h_work)
            if c_wk < best_c5:
                best_c5 = c_wk
                best_h = h_work.copy()

    # Final projection and evaluation
    best_h = project_onto_constraint(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
