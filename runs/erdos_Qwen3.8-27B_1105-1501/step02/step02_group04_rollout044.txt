import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} using bisection on threshold."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment for exact sum
    residual = target_sum - np.sum(proj)
    if abs(residual) > 1e-10:
        # Find elements not at boundary
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        n_free = np.sum(free)
        if n_free > 0:
            adj = residual / n_free
            proj[free] += adj
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def smooth_objective_and_grad(h, T):
    """Compute the log-sum-exp smooth approximation of max(corr)*dx and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    # Softmax weights (stable)
    z = np.exp((corr - cmax) / T)
    S = np.sum(z)
    w = z / S  # softmax weights
    value = (cmax + T * np.log(S)) * dx
    
    # Gradient using the formula derived above
    offset = n - 1
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        L = s - offset  # lag from -(n-1) to (n-1)
        ws = w[s]
        if ws < 1e-15:
            continue
        # First term: w[s] * (1-h[i-L]) for valid i-L
        i_lo1 = max(0, L)
        i_hi1 = min(n - 1, n - 1 + L)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            grad[idx] += ws * (1.0 - h[idx - L])
        # Second term: -w[s] * h[i+L] for valid i+L
        i_lo2 = max(0, -L)
        i_hi2 = min(n - 1, n - 1 - L)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            grad[idx] -= ws * h[idx + L]
    grad *= dx
    return value, grad


def smooth_wrap(h, T):
    val, _ = smooth_objective_and_grad(h, T)
    return val


def smooth_jac_wrap(h, T):
    _, g = smooth_objective_and_grad(h, T)
    return g


def run_slsqp_smooth(h0, n, T, maxiter=100):
    """Run SLSQP on the smooth objective."""
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x, ns=n/2.0: np.sum(x) - ns}]
    
    res = minimize(
        smooth_wrap, h0, args=(T,),
        jac=lambda x, T_val=T: smooth_jac_wrap(x, T_val),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-10}
    )
    if res.success or res.x is not None:
        return np.clip(res.x, 0.0, 1.0)

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(proj)
    if abs(residual) > 1e-10:
        free = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        n_free = int(np.sum(free))
        if n_free > 0:
            proj[free] += residual / n_free
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def smooth_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    z = np.exp((corr - cmax) / T)
    S = np.sum(z)
    w = z / S
    value = (cmax + T * np.log(S)) * dx
    offset = n - 1
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        L = s - offset
        ws = w[s]
        if ws < 1e-14:
            continue
        i_lo1 = max(0, L)
        i_hi1 = min(n - 1, n - 1 + L)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            grad[idx] += ws * (1.0 - h[idx - L])
        i_lo2 = max(0, -L)
        i_hi2 = min(n - 1, n - 1 - L)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            grad[idx] -= ws * h[idx + L]
    grad *= dx
    return value, grad


def _obj_fun(x, T):
    v, _ = smooth_val_grad(x, T)
    return v


def _obj_jac(x, T):
    _, g = smooth_val_grad(x, T)
    return g


def _sum_eq(x):
    return np.sum(x) - x.size / 2.0


def run_slsqp(h0, T, maxiter=80):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_fun, h0, args=(T,), jac=_obj_jac,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sum_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h0


def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_sweep(h, rng, deadline, max_pairs=None):
    n = len(h)
    count = 0
    maxp = max_pairs if max_pairs else n * n
    while time.time() < deadline and count < maxp:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if i > j:
            i, j = j, i
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-13:
            h[i] -= delta
            h[j] += delta
        count += 1


def full_pair_sweep(h, rng):
    n = len(h)
    perm = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-13:
                h[i] -= delta
                h[j] += delta

def anneal_slsqp(h0, n, deadline, base_temps=None):

    if base_temps is None:
        dx = 2.0 / n
        base_temps = [6.0 * dx, 3.5 * dx, 1.8 * dx, 0.9 * dx, 0.45 * dx,

    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c_best = c_cur
    for T in base_temps:
        if time.time() > deadline:
            break

        h_new = project_feasible(h_new, n / 2.0)
        c_new = compute_c5(h_new)

            h_cur = h_new.copy()

    return h_best, c_best


def basin_hoop(h_best, c_best, n, rng, deadline):
    """Basin hopping: perturb, polish, accept if improves global."""
    while time.time() < deadline:
        scale = 0.03 + 0.04 * rng.rand()
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n / 2.0)
        # Quick SLSQP at moderate temperature

        T = 1.5 * dx
        h_new = run_slsqp(h_pert, T, maxiter=30)

        if c_new < c_best - 1e-10:
            c_best = c_new
            h_best = h_new.copy()
