from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_of_scores(h):
    """Returns (2n-1, n) matrix where row d is grad of score d."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(2 * n - 1):
        offset = d - (n - 1)
        if offset >= 0:
            m = n - offset
            G[d, :m] = (1.0 - h[offset:offset + m])
            G[d, offset:] += (-h[:m])
        else:
            g = -offset
            m = n - g
            G[d, g:] = (1.0 - h[:m])
            G[d, :m] += (-h[g:])
    return G

def _smoothed_grad(h, beta):
    """Gradient of (1/beta)*log(sum exp(beta*S_k)) = weighted avg of grad S_k."""
    s = _scores(h)
    max_s = s.max()
    w = np.exp(beta * (s - max_s))
    w /= w.sum()
    G = _grad_of_scores(h)
    return G.T @ w  # shape (n,)

def _top_k_grad(h, k=5):
    """Weighted gradient targeting top-k active shifts."""
    s = _scores(h)
    top_idx = np.argsort(s)[-k:]
    top_s = s[top_idx]
    # Weight by proximity to max (tanh-like)
    w = np.exp(50 * (top_s - top_s.max()))
    w /= w.sum()
    G = _grad_of_scores(h)
    return G[top_idx].T @ w

def _line_search_1d(h, p, c5_cur, alpha_max):
    """Find best alpha in [0, alpha_max] to minimize C5(h + alpha*p)."""
    # Coarse grid
    alphas = np.linspace(0, alpha_max, 21)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        if a < 1e-10:
            continue
        h_t = h + a * p
        c5_t = max(0.0, min(1.0, h_t).max() - 0.0)  # quick check
        if h_t.min() < -0.01 or h_t.max() > 1.01:
            continue
        c5_t = _exact_c5(np.clip(h_t, 0, 1))
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    
    if best_a < 1e-8:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.1)
    hi = min(alpha_max, best_a + alpha_max * 0.1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-10 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _grad_descent(h, deadline, beta_start=20.0, beta_end=200.0, max_iters=500, lr=0.01):
    """Smoothed gradient descent with adaptive beta and momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    mu = 0.8  # momentum coefficient
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        # Adaptive beta
        frac = min(1.0, it / (max_iters * 0.6))
        beta = beta_start + (beta_end - beta_start) * frac
        
        g = _smoothed_grad(h, beta)
        
        # Keep sum zero (for feasibility)
        g -= np.mean(g)
        
        # Momentum
        momentum = mu * momentum - lr * g
        p = momentum / max(1e-10, abs(momentum).max())
        
        # Compute max step
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        alpha, new_c5 = _line_search_1d(h, p, cur_c5, alpha_max)
        
        if new_c5 < cur_c5 - 1e-12:
            h = np.clip(h + alpha * p, 0, 1)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.1, 0.1)
        else:
            stall += 1
            lr *= 0.8
            momentum *= 0.5

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(2 * n - 1):
        offset = d - (n - 1)
        if offset >= 0:
            m = n - offset
            G[d, :m] = (1.0 - h[offset:offset + m])
            G[d, offset:] += (-h[:m])
        else:
            g = -offset
            m = n - g
            G[d, g:] = (1.0 - h[:m])
            G[d, :m] += (-h[g:])
    return G

    max_s = s.max()

    d = G.T @ w
    d -= np.mean(d)  # ensure sum zero for feasibility
    norm = np.abs(d).max()
    if norm < 1e-15:
        return None
    return -d / norm  # negate for descent
