        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta

        mc = cv.max()

        return x.sum() - n / 2.0

    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        t_ratio = float(it) / float(max(it + 1, 1))
        h_new = y - alpha0 * grad
        h_new = h_new + mu * (h_cur - h_prev)
        h_new = _project(h_new)
        h_prev = h_cur.copy()
        h_cur = h_new
        t_new = 1.5 * (it + 1)
        y = h_cur + (t_ratio * h_cur - t_ratio * h_prev)
        y = _project(y)
        h_avg += (h_cur - h_avg) / (it + 1)
        if it % 200 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 1000 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s = _score(h_cur)
    if s < best_s:
        best_h = h_cur.copy()
        best_s = s
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_h = ha.copy()
        best_s = sa
    return best_h, best_s

def _chain_optimize(h0, deadline, rng):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)
    # CD phase 1: all lags
    if time.time() < deadline - 20:
        n_sw = min(200, max(50, int((deadline - time.time()) / 0.05)))
        cur, cur_s = _coord_descent(cur, n_sw, top_m=None, lam_lr=0.03, rng=rng, deadline=deadline)
    # CD phase 2: top 15
    if time.time() < deadline - 15:
        cur, cur_s = _coord_descent(cur, 300, top_m=15, lam_lr=0.02, rng=rng, deadline=deadline)
    # SLSQP smooth
    if time.time() < deadline - 10:
        new = _slsqp_smooth(cur, beta=300, maxiter=120)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns
    # CD phase 3: top 8
    if time.time() < deadline - 10:
        cur, cur_s = _coord_descent(cur, 400, top_m=8, lam_lr=0.01, rng=rng, deadline=deadline)
    # SLSQP direct
    if time.time() < deadline - 8:
        new = _slsqp_direct(cur, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns
    # Fista
    if time.time() < deadline - 5:
        cur, cur_s = _fista_subgrad(cur, beta=500, n_iter=3000, alpha0=0.003, mu=0.88)
    # Final CD
    if time.time() < deadline - 3:
        cur, cur_s = _coord_descent(cur, 200, top_m=6, lam_lr=0.005, rng=rng, deadline=deadline)
    return cur, cur_s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(40) * 0.5)

    rng = np.random.default_rng(seed)
    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    pn = len(parent)

    # Phase 1: Screen multiple n values with quick CD
    n_candidates = sorted(set([pn - 6, pn - 4, pn - 2, pn, pn + 2, pn + 4, pn + 6, pn + 8]))
    n_candidates = [n for n in n_candidates if 24 <= n <= 80]

    phase1_best = {}
    for n in n_candidates:
        if time.time() > deadline - 120:
            break
        h_init = _interp(parent, n)
        s0 = _score(h_init)
        # Quick CD
        h_cd, s_cd = _coord_descent(h_init, 150, top_m=20, lam_lr=0.05, rng=rng, deadline=deadline)
        if s_cd < s0:
            h_init, s0 = h_cd, s_cd
        phase1_best[n] = (s0, h_init.copy())
        if s0 < best_score - 1e-12:
            best_h = h_init.copy()
            best_score = s0
            best_n = n

    # Phase 2: Focused optimization on top candidates
    sorted_ns = sorted(phase1_best.keys(), key=lambda n: phase1_best[n][0])
    top_ns = sorted_ns[:min(4, len(sorted_ns))]

    for n in top_ns:
        if time.time() > deadline - 60:
            break
        h_base = phase1_best[n][1].copy()
        # Generate diverse starts
        starts = [h_base.copy(), _mirror(h_base), _binary_push(h_base, 1.5), _binary_push(h_base, 2.5)]
        for sigma in [0.05, 0.12]:
            starts.append(_project(h_base + sigma * rng.standard_normal(n)))

        for h0 in starts:
            if time.time() > deadline - 30:
                break
            cur, cur_s = _chain_optimize(h0, deadline, rng)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)

    # Phase 3: Intensive refinement on global best
    if time.time() < deadline - 40:
        # CD with very focused active set
        best_h, best_score = _coord_descent(best_h, 500, top_m=6, lam_lr=0.008, rng=rng, deadline=deadline)
        # SLSQP multi-beta
        for beta in [200, 600, 1200]:
            if time.time() > deadline - 15:
                break
            new = _slsqp_smooth(best_h, beta=beta, maxiter=120)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h = new
                best_score = ns
        # Fista
        if time.time() < deadline - 10:
            best_h, best_score = _fista_subgrad(best_h, beta=800, n_iter=5000, alpha0=0.002, mu=0.9)
        # Final CD
        if time.time() < deadline - 5:
            best_h, best_score = _coord_descent(best_h, 300, top_m=5, lam_lr=0.004, rng=rng, deadline=deadline)

    # Phase 4: Random restarts
    remaining = deadline - time.time()
    n_restarts = min(8, int(remaining / 40))
    for _ in range(n_restarts):
        if time.time() > deadline - 10:
            break
        sigma = 0.02 + 0.03 * rng.random()
        h_pert = _project(best_h + sigma * rng.standard_normal(best_n))
        cur, cur_s = _chain_optimize(h_pert, deadline, rng)
        if cur_s < best_score - 1e-12:
            best_h = cur.copy()
            best_score = cur_s
            best_n = len(cur)

    # Final polish
    if time.time() < deadline - 5:
        new = _slsqp_direct(best_h, maxiter=100)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h = new
            best_score = ns

    best_h = _project(best_h)
    best_score = _score(best_h)
    return (best_h, best_score, len(best_h))
