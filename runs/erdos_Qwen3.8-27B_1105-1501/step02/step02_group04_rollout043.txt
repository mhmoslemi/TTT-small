    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} using bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_L(h, i, j, n):
    """Compute linear coefficient vector L for transfer h[i]+=t, h[j]-=t."""
    offset = n - 1
    L = np.zeros(2 * n - 1)
    
    # Term 1: (1-h[i+k-offset]) for valid k
    # Valid when 0 <= i+k-offset < n, i.e., offset-i <= k <= offset-i+n-1
    k_lo1 = max(0, offset - i)
    k_hi1 = min(2*n-2, offset - i + n - 1)
    if k_lo1 <= k_hi1:
        idx = np.arange(k_lo1, k_hi1 + 1)
        src_idx = i + idx - offset  # = i + k - offset, ranges from 0 to n-1
        L[idx] += (1.0 - h[src_idx])
    
    # Term 2: -(1-h[j+k-offset]) for valid k
    k_lo2 = max(0, offset - j)
    k_hi2 = min(2*n-2, offset - j + n - 1)
    if k_lo2 <= k_hi2:
        idx = np.arange(k_lo2, k_hi2 + 1)
        src_idx = j + idx - offset
        L[idx] -= (1.0 - h[src_idx])
    
    # Term 3: -h[i-k+offset] for valid k
    # Valid when 0 <= i-k+offset < n, i.e., i-n+1+offset... 
    # i-k+offset >= 0 → k <= i+offset = i+n-1
    # i-k+offset < n → k > i-n → k >= i-n+1
    k_lo3 = max(0, i - n + 1)

    if k_lo3 <= k_hi3:

        L[idx] -= h[src_idx]

        idx = np.arange(k_lo4, k_hi4 + 1)
        src_idx = j - idx + offset
        L[idx] += h[src_idx]
    
    return L

    Q = np.zeros(2 * n - 1)
    Q[offset] += -2.0
    k1 = j - i + offset
    if 0 <= k1 <= 2*n-2 and k1 != offset:
        Q[k1] += 1.0
    k2 = i - j + offset
    if 0 <= k2 <= 2*n-2 and k2 != offset:
        Q[k2] += 1.0
    return Q

def line_search(corr, L, Q, t_lo, t_hi, n_steps=25):
    """Ternary search for t in [t_lo, t_hi] minimizing max(corr + t*L + t^2*Q)."""
    a, b = t_lo, t_hi
    for _ in range(n_steps):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + m1 * L + m1*m1 * Q)
        f2 = np.max(corr + m2 * L + m2*m2 * Q)
        if f1 < f2:
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    return t_opt, np.max(corr + t_opt * L + t_opt*t_opt * Q)

                t_lo = max(-h[i], -(1.0 - h[j]))  # t >= -h[i] (h[i]+t>=0) and t >= h[j]-1 (h[j]-t<=1 → t>=h[j]-1)

                    improved = True

    A[i, k] = 1 - h[i+k-offset] for valid k, else 0
    B[i, k] = h[i-k+offset] for valid k, else 0

import numpy as np
import time
import math


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    err = target_sum - np.sum(result)
    if abs(err) > 1e-12:
        # Distribute error over non-boundary elements
        slack = np.where(result > 0 & (result < 1), 1.0, 0.0)
        if np.sum(slack) > 0:
            adj = err / np.sum(slack)
            result += adj * slack
            np.clip(result, 0.0, 1.0, out=result)
    return result


def precompute_AB(h, n):
    offset = n - 1
    m = 2 * n - 1
    A = np.zeros((n, m))
    B = np.zeros((n, m))
    for i in range(n):
        k_lo = max(0, offset - i)
        k_hi = min(m - 1, offset - i + n - 1)
        if k_lo <= k_hi:
            src = np.arange(k_lo, k_hi + 1) + i - offset
            A[i, k_lo:k_hi + 1] = 1.0 - h[src]
        k_lo = max(0, i - n + 1)
        k_hi = min(m - 1, i + n - 1)
        if k_lo <= k_hi:
            src = i - np.arange(k_lo, k_hi + 1) + offset
            B[i, k_lo:k_hi + 1] = h[src]
    return A, B


def get_quad_info(n, i, j):
    """Return (positions, values) for the quadratic part."""
    offset = n - 1
    positions = [offset]
    values = [-2.0]
    k1 = j - i + offset
    if 0 <= k1 < 2 * n - 1 and k1 != offset:
        positions.append(k1)
        values.append(1.0)
    k2 = i - j + offset
    if 0 <= k2 < 2 * n - 1 and k2 != offset:
        positions.append(k2)
        values.append(1.0)
    return positions, values


def line_search_min(corr, L, qpos, qval, t_lo, t_hi, steps=22):
    a, b = t_lo, t_hi
    if b - a < 1e-13:
        return 0.0, float(np.max(corr))
    for _ in range(steps):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        v1 = corr + m1 * L
        for p, v in zip(qpos, qval):
            v1[p] += m1 * m1 * v
        f1 = float(np.max(v1))
        v2 = corr + m2 * L
        for p, v in zip(qpos, qval):
            v2[p] += m2 * m2 * v
        f2 = float(np.max(v2))
        if f1 < f2:
            b = m2
        else:
            a = m1
    t = 0.5 * (a + b)
    v = corr + t * L
    for p, val in zip(qpos, qval):
        v[p] += t * t * val
    return t, float(np.max(v))
