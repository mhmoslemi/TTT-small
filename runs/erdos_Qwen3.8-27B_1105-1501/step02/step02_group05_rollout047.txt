        lo_b = max(-h[i], -rho)
        hi_b = min(1.0 - h[i], rho)
        bounds.append((lo_b, hi_b))
    bounds.append((-1.0, C))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='interior-point')
        except Exception:
            return None, None
    if res.status not in (0, 3):
        return None, None
    if not np.all(np.isfinite(res.x)):
        return None, None
    u = res.x[:n]
    if np.max(np.abs(u)) < 1e-10:
        return None, None
    t_pred = res.x[n]
    return u, t_pred

def lp_refine(h, max_iters=200, time_limit=None, t_start=None):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    rho_start = 0.25
    rho = rho_start
    alpha_list = [1.0, 0.75, 0.5, 0.35, 0.22, 0.12, 0.06, 0.03]
    K_max = 36
    eta_frac = 0.004
    stall = 0
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        V = all_lag_values(best)
        C = float(V.max())
        if abs(C - best_sc) > 1e-10:
            best_sc = C
        u, t_pred = lp_step(best, V, C, rho, active_p=None, K_max=K_max, eta_frac=eta_frac)
        if u is None:
            rho *= 0.7
            if rho < 0.003:
                rho = rho_start
                eta_frac = max(eta_frac * 0.5, 0.0005)
                K_max = min(K_max + 8, 60)
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0
        improved = False
        for alpha in alpha_list:
            trial = make_feasible(best + alpha * u)
            sc = score_exact(trial)
            if sc < best_sc - 1e-12:
                best = trial
                best_sc = sc
                improved = True
                break
        if not improved:
            rho *= 0.8
            if rho < 0.003:
                rho = rho_start * 0.6
                eta_frac = max(eta_frac * 1.5, 0.02)
                K_max = max(20, K_max - 8)
    return best, best_sc

def binary_from_continuous(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    b = (h > 0.5).astype(float)
    if np.sum(b) != target:
        if np.sum(b) > target:
            diffs = np.where(b == 1, 0.5 - h, -1.0)
            order = np.argsort(diffs)[::-1]
            excess = int(np.sum(b)) - target
            for i in range(excess):
                b[order[i]] = 0.0
        else:
            diffs = np.where(b == 0, h - 0.5, -1.0)
            order = np.argsort(diffs)[::-1]
            deficit = target - int(np.sum(b))
            for i in range(deficit):
                b[order[i]] = 1.0
    if np.sum(b) != target:
        return None
    return b

def binary_local_search(b, n_steps=5000, time_limit=None, rng=None):
    n = len(b)
    target = n // 2
    if rng is None:
        rng = np.random.default_rng(0)
    cur = b.copy()
    cur_sc = score_exact(cur)
    best = cur.copy()
    best_sc = cur_sc
    pos1 = np.where(cur == 1)[0]
    pos0 = np.where(cur == 0)[0]
    T = 5e-4
    for it in range(n_steps):
        if time_limit is not None and time.time() > time_limit:
            break
        i = pos1[rng.integers(len(pos1))]
        j = pos0[rng.integers(len(pos0))]
        cur[i] = 0.0
        cur[j] = 1.0
        pos1 = np.sort(np.delete(pos1, np.where(pos1 == i)[0][0]))
        pos0 = np.sort(np.delete(pos0, np.where(pos0 == j)[0][0]))
        pos1 = np.insert(pos1, np.searchsorted(pos1, j), j)
        pos0 = np.insert(pos0, np.searchsorted(pos0, i), i)
        new_sc = score_exact(cur)
        delta = new_sc - cur_sc
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            pos1 = np.sort(np.delete(pos1, np.where(pos1 == j)[0][0]))
            pos0 = np.sort(np.delete(pos0, np.where(pos0 == i)[0][0]))
            pos1 = np.insert(pos1, np.searchsorted(pos1, i), i)
            pos0 = np.insert(pos0, np.searchsorted(pos0, j), j)
        T *= 0.9995
    return best, best_sc

def anti_sym(h):
    h_rev = h[::-1].copy()
    h_new = 0.5 * (h + 1.0 - h_rev)
    return make_feasible(h_new)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_wave(n, f1, a1, f2, a2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + phase) + a2 * np.cos(2 * np.pi * f2 * x + phase * 0.5)
    return make_feasible(h)

def two_point_refine(h, n_tries, rng, time_limit=None):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if time_limit is not None and time.time() > time_limit:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        for t in [t_lo + 0.5 * span, t_lo + 0.25 * span, t_lo + 0.75 * span,
                  t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            best[a] = va
            best[b] = vb
            sc = score_exact(best)
            if sc < best_sc:
                best_sc = sc
                break
            else:
                best[a] -= t
                best[b] += t
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_ok():
        return time.time() < deadline

    best_h = None
    best_score = float('inf')

    def try_update(h_cand):
        nonlocal best_h, best_score
        if h_cand is None:
            return
        h_valid = make_feasible(h_cand)
        if not np.all(np.isfinite(h_valid)):
            return
        sc = score_exact(h_valid)
        if not np.isfinite(sc):
            return
        if sc < best_score - 1e-12:
            best_h = h_valid.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.1)
        best_score = score_exact(best_h)

    # Try anti-symmetric
    try_update(anti_sym(best_h))

    # Phase 1: LP refinement on parent
    if time_ok():
        n = len(best_h)
        remaining = deadline - time.time()
        t_lim = min(deadline, time.time() + 0.4 * remaining)
        h_ref, sc_ref = lp_refine(best_h, max_iters=300, time_limit=t_lim)
        try_update(h_ref)

    # Phase 2: Binary search
    if time_ok() and len(best_h) % 2 == 0:
        b = binary_from_continuous(best_h)
        if b is not None:
            b_best, b_sc = binary_local_search(b, n_steps=4000, rng=rng, time_limit=min(deadline, time.time() + 60))
            try_update(b_best)
            # LP on binary
            if time_ok():
                h_bin = make_feasible(b_best)
                t_lim = min(deadline, time.time() + 80)
                h_ref2, sc2 = lp_refine(h_bin, max_iters=200, time_limit=t_lim)
                try_update(h_ref2)

    # Phase 3: More LP refinement
    if time_ok():
        remaining = deadline - time.time()
        t_lim = min(deadline, time.time() + 0.35 * remaining)
        n = len(best_h)
        h_ref, sc_ref = lp_refine(best_h, max_iters=300, time_limit=t_lim)
        try_update(h_ref)

    # Phase 4: 2-point exchange polish
    if time_ok():
        remaining = deadline - time.time()
        n_tries = int(0.3 * remaining * 200)
        n_tries = max(2000, min(n_tries, 30000))
        h_tp, sc_tp = two_point_refine(best_h, n_tries, rng, time_limit=min(deadline, time.time() + 0.25 * remaining))
        try_update(h_tp)

    # Phase 5: LP again after 2-point
    if time_ok():
        remaining = deadline - time.time()
        t_lim = min(deadline, time.time() + 0.3 * remaining)
        h_ref, sc_ref = lp_refine(best_h, max_iters=200, time_limit=t_lim)
        try_update(h_ref)

    # Phase 6: Resolution scanning
    if time_ok():
        n_base = len(best_h)
        n_opts = sorted(set([n_base - 8, n_base - 4, n_base + 4, n_base + 8,
                             n_base + 12, n_base - 12, 48, 56, 64, 72, 80]))
        n_opts = [x for x in n_opts if 32 <= x <= 100]
        for no in n_opts:
            if not time_ok():
                break
            if no == len(best_h):
                continue
            h_int = interp_to(best_h, no)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int
                best_score = sc_int
            if time_ok():
                t_lim = min(deadline, time.time() + 30)
                h_ref, sc_ref = lp_refine(best_h, max_iters=100, time_limit=t_lim)
                try_update(h_ref)

    # Phase 7: Final LP polish
    if time_ok():
        remaining = deadline - time.time()
        t_lim = min(deadline, time.time() + 0.4 * remaining)
        h_ref, sc_ref = lp_refine(best_h, max_iters=300, time_limit=t_lim)
        try_update(h_ref)

    # Phase 8: Final 2-point
    if time_ok():
        remaining = max(10.0, deadline - time.time())
        n_tries = int(remaining * 100)
        n_tries = max(1000, min(n_tries, 15000))
        h_tp, sc_tp = two_point_refine(best_h, n_tries, rng, time_limit=deadline - 5)
        try_update(h_tp)

    # Phase 9: Final LP
    if time_ok() and time.time() < deadline - 5:
        h_ref, sc_ref = lp_refine(best_h, max_iters=100, time_limit=deadline - 5)
        try_update(h_ref)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8

    return final_h, final_score, final_n
