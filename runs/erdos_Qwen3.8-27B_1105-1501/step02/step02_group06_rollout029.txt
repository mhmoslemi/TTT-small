        else:
            accepted = False
            for alpha in [0.6, 0.3, 0.15]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-13:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 6:
                    break
    return h, best

def smooth_obj_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = compute_corr(h) * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))

def try_slsqp(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj_h, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def random_feasible(n, rng):
    z = rng.random(n)
    return make_feasible(z)

def step_pattern(n, center=0.5, width=0.25, steepness=10.0):
    x = (np.arange(n) + 0.5) / n
    h = 1.0 / (1.0 + np.exp(-steepness * (center - x)))
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_score = float('inf')
    best_n = 0
    
    def update(h, sc):
        nonlocal best_h, best_score, best_n
        if h is not None and np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)
    
    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                update(parent, ps)
    except (NameError, AttributeError):
        pass
    
    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(np.pi * x))
        best_score = official_score(best_h)
        best_n = len(best_h)
    
    # Phase 1: Aggressive pairwise coordinate descent
    if time.time() < deadline - 30:
        h_opt, sc_opt = pairwise_coord_descent(best_h, max_passes=80, tol=1e-12, deadline=deadline - 30)
        update(h_opt, sc_opt)
    
    # Phase 2: Active-set QP with multiple k_active
    cur = best_h.copy()
    for k_a in [5, 8, 3]:
        if time.time() > deadline - 60:
            break
        h_opt, sc_opt = run_active_set(cur, max_iter=25, k_active=k_a,
                                       reg_start=2.0, reg_end=0.002,
                                       deadline=deadline - 60)
        if sc_opt < best_score - 1e-14:
            update(h_opt, sc_opt)
            cur = h_opt
    
    # Phase 3: SLSQP smooth
    if time.time() < deadline - 40:
        for beta in [500, 2000, 10000]:
            if time.time() > deadline - 40:
                break
            h_new, sc = try_slsqp(cur, beta, maxiter=80)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
                cur = h_new
    
    # Phase 4: Another round of pairwise descent
    if time.time() < deadline - 40:
        h_opt, sc_opt = pairwise_coord_descent(best_h, max_passes=60, tol=1e-13, deadline=deadline - 40)
        update(h_opt, sc_opt)
    
    # Phase 5: Resolution sweep
    if time.time() < deadline - 80:
        n_base = len(best_h)
        n_opts = sorted(set([
            max(30, n_base - 10), max(30, n_base - 5), n_base,
            min(90, n_base + 5), min(90, n_base + 10)
        ]))
        for n_opt in n_opts:
            if time.time() > deadline - 80:
                break
            if n_opt < 24 or n_opt > 100:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            if time.time() < deadline - 40:
                h_opt, sc_opt = pairwise_coord_descent(h_int, max_passes=40, tol=1e-12,
                                                        deadline=deadline - 40)
                update(h_opt, sc_opt)
            if time.time() < deadline - 30:
                h_opt, sc_opt = run_active_set(h_opt if h_opt is not None else h_int,
                                               max_iter=15, k_active=5,
                                               reg_start=1.5, reg_end=0.005,
                                               deadline=deadline - 30)
                update(h_opt, sc_opt)
    
    # Phase 6: Random perturbations + optimize
    if time.time() < deadline - 60:
        for i in range(20):
            if time.time() > deadline - 60:
                break
            sigma = 0.03 + 0.07 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = pairwise_coord_descent(h_cand, max_passes=20, tol=1e-11,
                                                    deadline=deadline - 60)
            update(h_opt, sc_opt)
            if time.time() < deadline - 40:
                h_opt2, sc_opt2 = run_active_set(h_opt, max_iter=10, k_active=5,
                                                  reg_start=1.0, reg_end=0.01,
                                                  deadline=deadline - 40)
                update(h_opt2, sc_opt2)
    
    # Phase 7: Try complement
    if time.time() < deadline - 30:
        h_comp = make_feasible(1.0 - best_h)
        sc_comp = official_score(h_comp)
        update(h_comp, sc_comp)
        h_opt, sc_opt = pairwise_coord_descent(h_comp, max_passes=20, tol=1e-12,
                                                deadline=deadline - 30)
        update(h_opt, sc_opt)
    
    # Phase 8: Final intensive polish
    if time.time() < deadline - 20:
        for k_a in [3, 5, 7]:
            if time.time() > deadline - 20:
                break
            h_opt, sc_opt = run_active_set(best_h, max_iter=20, k_active=k_a,
                                           reg_start=0.5, reg_end=0.001,
                                           deadline=deadline - 20)
            update(h_opt, sc_opt)
        if time.time() < deadline - 15:
            h_new, sc = try_slsqp(best_h, 20000, maxiter=60)
            if h_new is not None:
                update(h_new, sc)
    
    # Phase 9: One more pairwise pass on absolute best
    if time.time() < deadline - 10:
        h_opt, sc_opt = pairwise_coord_descent(best_h, max_passes=30, tol=1e-13,
                                                deadline=deadline - 10)
        update(h_opt, sc_opt)
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = final_score + 1e-13
    return final_h, final_bound, len(final_h)
