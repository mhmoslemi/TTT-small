import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target_sum):
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(v)
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix floating point error
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binary_anneal(n, seed, max_iters=50000, deadline=None):
    """Simulated annealing on binary sequences with n/2 ones."""
    rng = np.random.default_rng(seed)
    # Start with random binary sequence
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    
    dx = 2.0 / n
    scores = _corr_scores(h)
    cur_max = float(scores.max())
    best_h = h.copy()
    best_max = cur_max
    
    T = 5.0
    cooling = 0.997
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        # Pick a random 1 and a random 0, swap them
        ones_pos = np.where(h == 1.0)[0]
        zeros_pos = np.where(h == 0.0)[0]
        i = ones_pos[rng.integers(len(ones_pos))]
        j = zeros_pos[rng.integers(len(zeros_pos))]
        
        # Try swap
        h[i] = 0.0
        h[j] = 1.0
        new_scores = _corr_scores(h)
        new_max = float(new_scores.max())
        
        delta = new_max - cur_max
        if delta < 0 or (T > 0.01 and rng.random() < np.exp(-delta / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # Undo
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cooling
    
    return best_h, best_max * dx

def _smooth_max_optimize(h, deadline, n_inner=2000, beta_init=50.0):
    """Minimize log-sum-exp of correlation scores using projected gradient."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    beta = beta_init
    lr = 0.05
    
    for it in range(n_inner):
        if time.time() > deadline:
            break
        
        scores = _corr_scores(h)
        # Soft max
        s_max = scores.max()
        w = np.exp(beta * (scores - s_max))
        w /= w.sum()
        
        # Gradient of log-sum-exp w.r.t. h
        # d/dh_i [sum_r w_r * sum_j h[j](1-h[j+r-idx])]
        # = sum_r w_r * G[r, i] where G is the gradient matrix
        # For correlation: c[r] = sum_i h[i](1-h[i+d]) where d = r-(n-1)
        # dc[r]/dh[i] = (1-h[i+d]) - h[i-d] (with appropriate indexing)
        
        # Let me compute the gradient directly
        grad = np.zeros(n)
        for r in range(2*n-1):
            if w[r] < 1e-10:
                continue
            d = r - (n-1)
            if d >= 0:
                # c[r] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
                # dc/dh[i] for i in [0, n-1-d]: (1-h[i+d])
                # dc/dh[i] for i in [d, n-1]: -h[i-d]
                i_range = np.arange(n-d)
                grad[i_range] += w[r] * (1.0 - h[i_range + d])
                grad[i_range + d] += w[r] * (-h[i_range])
            else:
                g = -d
                # c[r] = sum_{i=g}^{n-1} h[i]*(1-h[i-g])
                # dc/dh[i] for i in [g, n-1]: (1-h[i-g])
                # dc/dh[i] for i in [0, n-1-g]: -h[i+g]
                i_range = np.arange(g, n)
                grad[i_range] += w[r] * (1.0 - h[i_range - g])
                grad[i_range - g] += w[r] * (-h[i_range])
        
        # Subtract mean to respect the sum constraint
        grad -= grad.mean()
        
        # Step
        h_new = h - lr * grad
        h_new = _project_box_sum(h_new, target)
        
        c5 = _exact_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive learning rate
        if c5 > best_c5 + 0.001:
            lr *= 0.7
        elif it % 100 == 0:
            beta *= 1.5
            lr = min(lr * 1.1, 0.5)
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    x = np.clip(v - thresh, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_smooth_max(h, scores, beta):
    """Compute gradient of log-sum-exp(beta * scores) w.r.t. h."""
    n = len(h)
    smax = scores.max()
    w = np.exp(beta * (scores - smax))
    w /= w.sum()
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        if d == 0:
            continue  # c[0] = sum h[i]*(1-h[i]) which for binary is 0
        if d > 0:
            # c[r] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # d/dh[i] = (1-h[i+d]) for 0<=i<=n-1-d
            # d/dh[i] = -h[i-d] for d<=i<=n-1
            idx1 = np.arange(n - d)
            grad[idx1] += w[r] * (1.0 - h[idx1 + d])
            idx2 = np.arange(d, n)
            grad[idx2] += w[r] * (-h[idx2 - d])
        else:
            g = -d
            # c[r] = sum_{i=g}^{n-1} h[i]*(1-h[i-g])
            # d/dh[i] = (1-h[i-g]) for g<=i<=n-1
            # d/dh[i] = -h[i+g] for 0<=i<=n-1-g
            idx1 = np.arange(g, n)
            grad[idx1] += w[r] * (1.0 - h[idx1 - g])
            idx2 = np.arange(n - g)
            grad[idx2] += w[r] * (-h[idx2 + g])
    
    return grad

def _softmax_grad_descent(h, deadline, max_iters=5000, beta_init=20.0):
    """Minimize smooth max of correlation using projected gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    beta = beta_init
    lr = 0.02
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = _corr(h)
        grad = _grad_smooth_max(h, scores, beta)
        grad -= grad.mean()  # respect sum constraint
        
        # Backtracking line search
        alpha = lr
        h_new = h - alpha * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        # Simple: try a few step sizes
        best_alpha = 0.0
        best_c5_trial = c5_new
        for trial_alpha in [alpha, alpha*0.5, alpha*2, alpha*0.25]:
            ht = _project(h - trial_alpha * grad, target)
            ct = _exact_c5(ht)
            if ct < best_c5_trial:
                best_c5_trial = ct
                best_alpha = trial_alpha
        
        if best_alpha > 0:
            h = _project(h - best_alpha * grad, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                beta *= 1.3
                no_improve = 0
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 100:
                break
    
    return best_h, best_c5

def _binary_anneal(n, seed, deadline):
    """Simulated annealing on binary sequences."""
    rng = np.random.default_rng(seed)
    dx = 2.0 / n
    
    # Random initial
    h = np.zeros(n)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    
    scores = _corr(h)
    cur_max = scores.max()
    best_h = h.copy()
    best_max = cur_max
    
    T = 3.0
    cool = 0.9995
    iter_count = 0
    
    while time.time() < deadline and T > 0.05:
        # Pick random 1 and 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        if i == j:
            continue
        
        # Proposed swap
        h[i] = 0.0
        h[j] = 1.0
        new_scores = _corr(h)
        new_max = new_scores.max()
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cool
        iter_count += 1
    
    return best_h, best_max * dx

def _greedy_binary_refine(h, deadline):
    """Greedy: try all single swaps and take the best improving one."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    best_max = _corr(h).max()
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0
        best_swap = None
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_max = _corr(h).max()
                delta = new_max - best_max
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_max = _corr(h).max()
            improved = True
            if time.time() > deadline:
                break
    
    return h, best_max * dx

def _pair_continuous(h, deadline):
    """Try all pairs (i,j) and adjust h[i], h[j] to minimize C5."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    
    for _pass in range(10):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                # Try moving mass from i to j and vice versa
                for direction in [(1,-1), (-1,1)]:
                    ai, aj = direction
                    # Max step
                    if ai > 0:
                        a_max = min(1.0 - h[i], h[j])
                    else:
                        a_max = min(h[i], 1.0 - h[j])
                    if a_max < 1e-8:
                        continue
                    # Try a few step sizes
                    best_a = 0
                    best_c5_trial = best_c5
                    for frac in [0.25, 0.5, 0.75, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[i] += ai * a
                        h_t[j] += aj * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5_trial:
                            best_c5_trial = c5_t
                            best_a = a
                    
                    if best_a > 0:
                        h[i] += ai * best_a
                        h[j] += aj * best_a
                        best_c5 = _exact_c5(h)
                        best_h = h.copy()
                        improved = True
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
