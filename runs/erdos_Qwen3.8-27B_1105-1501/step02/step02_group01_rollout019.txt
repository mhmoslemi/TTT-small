def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)

- At each iteration, linearize f2 at the current point: f2(h) ≈ f2(h_k) + (h-h_k)^T ∇f2(h_k)
- Solve the convex subproblem: min f1(h) - f2(h_k) - (h-h_k)^T ∇f2(h_k)

At the optimum, the KKT conditions say:
- There exist λ_s ≥ 0 (for the active constraints C_s = C_max) such that:
  ∑_s λ_s ∂C_s/∂h[j] + μ * 1[j at bound] - ν = 0 for all j

where ν is the Lagrange multiplier for the equality constraint sum h = n/2.

The gradient ∂C_s/∂h[j] is exactly w_s[j] (the weight I computed above). So at the optimum:
∑_s λ_s w_s[j] = ν for all j not at bound, and appropriate inequality for j at bound.

This means that at the optimum, the weighted sum of the weight vectors (over active constraints) should be constant (equal to ν) for all free coordinates.

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
