    """Find optimal δ for changing h[i]+=δ, h[j]-=δ."""
    n = len(h)
    dx = 2.0 / n
    # Feasible range
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-12:
        return 0.0
    
    # Current values
    vals = compute_all_vals(h)  # O(n^2) but fast with numpy
    
    # Gradient difference d_s = ∂R(s)/∂h[i] - ∂R(s)/∂h[j] (times dx)
    gi = grad_for_coord(h, i)  # O(n)
    gj = grad_for_coord(h, j)  # O(n)
    d = gi - gj  # O(n)
    
    # f(δ) ≈ max_s (vals[s] + δ * d[s])
    # Find minimum over [d_lo, d_hi]
    
    # Quick check: if all d ≥ 0, min at d_lo; if all d ≤ 0, min at d_hi
    if np.min(d) >= 0:
        return d_lo
    if np.max(d) <= 0:
        return d_hi
    
    # Binary search for the point where subgradient contains 0
    lo, hi = d_lo, d_hi
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        # Evaluate f at mid: which s are active?
        f_vals = vals + mid * d
        f_max = f_vals.max()
        active = f_vals >= f_max - 1e-15
        # Subgradient at mid is {d[s] for active s}
        if np.min(d[active]) > 1e-10:
            hi = mid  # f is increasing here, min is to the left
        elif np.max(d[active]) < -1e-10:
            lo = mid  # f is decreasing here, min is to the right
        else:
            break  # 0 is in subgradient, we're at the minimum
    
    return 0.5 * (lo + hi)
