from scipy.optimize import minimize
import time

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _score(h):
    return float(np.max(_corr(h)))

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(res) < 1e-15:
                break
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_lag(h, s):
    """Gradient of F_s(h) = dx * sum_{i} h[i]*(1-h[i+s]) for valid i."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    i0 = max(0, -s)
    i1 = min(n, n - s)
    for i in range(i0, i1):
        g[i] += (1.0 - h[i + s])
        g[i + s] += (-h[i])
    return g * dx

def _pnorm_obj(h, p):
    cv = _corr(h)
    cv = np.maximum(cv, 1e-15)
    return np.sum(cv ** p) ** (1.0 / p)

def _pnorm_grad(h, p):
    cv = _corr(h)
    cv_safe = np.maximum(cv, 1e-15)
    S = np.sum(cv_safe ** p)
    f = S ** (1.0 / p)
    # df/dh_j = f^(1-p) * sum_k cv_k^(p-1) * dF_k/dh_j
    # = S^(1/p - 1) * sum_k cv_k^(p-1) * dF_k/dh_j
    coeff = S ** (1.0 / p - 1.0)
    g = np.zeros(len(h))
    n = len(h)
    for k in range(2 * n - 1):
        w = cv_safe[k] ** (p - 1)
        if w < 1e-15:
            continue
        s = k - (n - 1)
        g += w * _grad_lag(h, s)
    return g * coeff

def _slsqp_pnorm(h0, p, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def _efun(x):
        return x.sum() - n / 2.0
    
    def _ejac(x):
        return np.ones(n)
    
    try:
        res = minimize(_pnorm_obj, h0, args=(p,), jac=_pnorm_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_direct(h0, maxiter=200):
    n = len(h0)
    cv = _corr(h0)
    t0 = cv.max() + 0.002
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def _fun(z):
        return z[-1]
    
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        return t - _corr(h)
    
    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0
    
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,

                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}

        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):

    except Exception:

    return h0

def _coord_descent(h, n_steps=5000, rng=None):
    """Sum-preserving coordinate descent."""
    n = len(h)
    if rng is None:
        rng = np.random.default_rng(0)
    h = h.copy()
    best_s = _score(h)
    for step in range(n_steps):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Try moving mass from i to j and vice versa
        for direction in [1, -1]:
            for delta in [0.01, 0.005, 0.001]:
                if direction == 1:
                    # Move from i to j
                    if h[i] < delta or h[j] > 1.0 - delta:
                        continue
                    h[i] -= delta
                    h[j] += delta
                else:
                    # Move from j to i
                    if h[j] < delta or h[i] > 1.0 - delta:
                        continue
                    h[j] -= delta
                    h[i] += delta
                s = _score(h)
                if s < best_s:
                    best_s = s
                else:
                    # Revert
                    if direction == 1:
                        h[i] += delta
                        h[j] -= delta
                    else:
                        h[j] += delta
                        h[i] -= delta
                    break
    return h, best_s

def _de_global(h_best, n_gen=40, pop_size=25, rng=None):
    """Differential evolution for global search."""
    n = len(h_best)
    if rng is None:
        rng = np.random.default_rng(42)
    
    s_best = _score(h_best)
    # Initialize population
    pop = [h_best.copy()]
    # Add diverse members
    for _ in range(pop_size - 1):
        h = h_best + 0.2 * rng.standard_normal(n)
        pop.append(_project(h))
    
    for gen in range(n_gen):
        new_pop = [pop[0].copy()]
        for idx in range(1, pop_size):
            # DE/rand/1
            a, b, c = rng.choice(pop_size, 3, replace=False)
            while a == b or a == c or b == c:
                a, b, c = rng.choice(pop_size, 3, replace=False)
            F = rng.uniform(0.5, 1.0)
            CR = rng.uniform(0.3, 0.9)
            v = pop[a] + F * (pop[b] - pop[c])
            v = _project(v)
            # Cross
            trial = pop[idx].copy()
            for j in range(n):
                if rng.random() < CR:
                    trial[j] = v[j]
            trial = _project(trial)
            # Select
            if _score(trial) < _score(pop[idx]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[idx].copy())
        pop = new_pop
        # Track best
        for h in pop:
            s = _score(h)
            if s < s_best:
                s_best = s
                pop[0] = h.copy()
    
    return pop[0], s_best

def _active_set_slsqp(h0, max_rounds=10, beta_range=(50, 500)):
    """Iterative active-set: minimize weighted sum of top lags."""
    n = len(h0)
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    
    def _efun(x):
        return x.sum() - n / 2.0
    
    def _ejac(x):
        return np.ones(n)
    
    beta = beta_range[0]
    for rnd in range(max_rounds):
        cv = _corr(h)
        Fmax = cv.max()
        # Weight: softmax-like on top lags
        w = np.exp(beta * (cv - Fmax))
        w = w / w.sum()
        
        def _obj(x):
            cvx = _corr(x)
            return float(np.sum(w * cvx))
        
        def _obj_jac(x):
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-15:
                    continue
                s = k - (n - 1)
                g += w[k] * _grad_lag(x, s)
            return g
        
        try:
            res = minimize(_obj, h, jac=_obj_jac, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                           options={'maxiter': 80, 'ftol': 1e-16})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                h_new = _project(res['x'])
                if _score(h_new) < _score(h):
                    h = h_new
        except Exception:
            pass
        beta = min(beta * 1.5, beta_range[1])
    
    return h

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

One idea: **alternating minimization**. Fix all but one coordinate, optimize that coordinate. For coordinate h[i], the objective is:

For a fixed set of other coordinates, F_k as a function of h[i] = x is:

F_k(x) = b_k (for lags k where i is not involved)

where a_k = dx * (1 - h[i+s_k]) for the "i as first" contribution, or a_k = dx * (-h[i-s_k]) for the "i as second" contribution, or both.

Actually, h[i] appears in F_k for:
- Lag s where i+s < n: h[i] appears as h[i]*(1-h[i+s]), so coefficient of h[i] is (1-h[i+s])*dx
- Lag -s where i-s >= 0: h[i] appears as h[i-s]*(1-h[i]), so coefficient of h[i] is -h[i-s]*dx

def _coord_optimize(h, n_sweeps=5):
    """Exact coordinate descent: for each coordinate, find optimal value."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for sweep in range(n_sweeps):
        F = _corr(h)  # shape (2n-1,)
        
        for i in range(n):
            # Compute c_k(i) = (1-h[i+s])*1{i+s<n} - h[i-s]*1{i-s>=0} for each k
            # and d_k = F[k] - F[k] (we'll work with F_k + dx*(x-x_i)*c_k)
            # New F_k = F[k] + dx*(x - h[i]) * c_k
            # = dx*c_k*x + (F[k] - dx*h[i]*c_k)
            
            a = np.zeros(2 * n - 1)  # slopes
            b = F.copy()  # intercepts (initially F_k)
            
            for k in range(2 * n - 1):
                s = k - (n - 1)
                c = 0.0
                if i + s < n and i + s >= 0:
                    c += (1.0 - h[i + s])
                if i - s >= 0 and i - s < n:
                    c += (-h[i - s])
                a[k] = dx * c
                b[k] = F[k] - dx * h[i] * c
            
            # Minimize max_k (a[k]*x + b[k]) over x in [0,1]
            # Find all intersection points
            candidates = [0.0, 1.0]
            m = 2 * n - 1
            for p in range(m):
                for q in range(p + 1, m):
                    if abs(a[p] - a[q]) < 1e-15:
                        continue
                    x_int = (b[q] - b[p]) / (a[p] - a[q])
                    if 0.0 < x_int < 1.0:
                        candidates.append(x_int)
            
            # Evaluate at all candidates
            best_x = 0.0
            best_val = np.inf
            for x in candidates:
                val = np.max(a * x + b)
                if val < best_val:
                    best_val = val
                    best_x = x
            
            if best_x != h[i]:
                h[i] = best_x
    
        # Re-project to maintain exact sum
        h = _project(h)
    
    return h

def _min_max_lines(a, b, x_lo=0.0, x_hi=1.0):
    """Minimize max_k (a[k]*x + b[k]) for x in [x_lo, x_hi]."""
    # Coarse grid
    x_grid = np.linspace(x_lo, x_hi, 501)
    vals = np.max(a[:, None] * x_grid[None, :] + b[:, None], axis=0)
    idx = np.argmin(vals)
    
    # Refine with golden section
    if idx == 0:
        return x_grid[0]
    if idx == len(x_grid) - 1:
        return x_grid[-1]
    
    lo = x_grid[idx - 1]
    hi = x_grid[idx + 1]
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = np.max(a * c + b)
    fd = np.max(a * d + b)
    
    for _ in range(60):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = np.max(a * c + b)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = np.max(a * d + b)
    
    return 0.5 * (lo + hi)

def _project(z):

    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        if s < target:

        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:

        else:
            idx = np.argsort(h)[::-1]
        for i in idx:

            if res > 0:

            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _pn_obj(x, p):
    cv = _corr(x)
    cv = np.maximum(cv, 1e-15)
    return float(np.sum(cv ** p) ** (1.0 / p))


def _pn_jac(x, p):
    n = len(x)
    cv = _corr(x)
    cv_safe = np.maximum(cv, 1e-15)
    S = np.sum(cv_safe ** p)
    if S < 1e-300:
        return np.zeros(n)
    coeff = S ** (1.0 / p - 1.0)
    g = np.zeros(n)
    n2m1 = 2 * n - 1
    for k in range(n2m1):
        w = cv_safe[k] ** (p - 1)
        if w < 1e-15:
            continue
        s = k - (n - 1)
        i0 = max(0, -s)
        i1 = min(n, n - s)
        for j in range(i0, i1):
            g[j] += w * (1.0 - x[j + s])
            g[j + s] += w * (-x[j])
    return g * (2.0 / n) * coeff

        # Precompute grad matrix
        gm = _grad_matrix(h)

        ws = w.copy()
        gx = w @ gm  # shape (n,) -- this is gradient of sum w_k * F_k

        try:
            res = minimize(_ws_obj, h, args=(ws, gx, 1.0), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac}],
                           options={'maxiter': 60, 'ftol': 1e-16})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                h_new = _project(res['x'])
                if _score(h_new) < _score(h):
                    h = h_new
        except Exception:
            pass
        beta = min(beta * 1.4, beta_end)
    return h


def _ws_obj(x, w, old_grad, dummy):
    cv = _corr(x)
    return float(np.sum(w * cv))


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _de_global(h_best, n_gen=30, pop_size=20, rng=None):
    n = len(h_best)
    if rng is None:
        rng = np.random.default_rng(123)
    s_best = _score(h_best)
    pop = [h_best.copy()]
    for _ in range(pop_size - 1):
        h = h_best + 0.15 * rng.standard_normal(n)
        pop.append(_project(h))
    scores = [_score(h) for h in pop]
    for gen in range(n_gen):
        for idx in range(1, pop_size):
            candidates = [r for r in range(pop_size) if r != idx]
            a, b, c = rng.choice(candidates, 3, replace=False)
            F = rng.uniform(0.5, 1.0)
            v = pop[a] + F * (pop[b] - pop[c])
            v = _project(v)
            trial = v
            s_trial = _score(trial)
            if s_trial < scores[idx]:
                pop[idx] = trial
                scores[idx] = s_trial
                if s_trial < s_best:
                    s_best = s_trial
                    pop[0] = trial.copy()
                    scores[0] = s_trial
    return pop[0], s_best


def _stochastic_cd(h, n_steps=5000, rng=None):
    """Stochastic coordinate descent with sum-preserving pair moves."""
    n = len(h)
    if rng is None:
        rng = np.random.default_rng(77)
    h = h.copy()
    F = _corr(h)
    best_s = float(np.max(F))
    for step in range(n_steps):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        for delta in [0.02, 0.01, 0.005, 0.002]:
            old_i, old_j = h[i], h[j]
            h[i] = old_i - delta
            h[j] = old_j + delta
            if h[i] < 0 or h[j] > 1:
                h[i], h[j] = old_i, old_j
                continue
            F_new = _corr(h)
            s_new = float(np.max(F_new))
            if s_new < best_s - 1e-14:
                best_s = s_new
                F = F_new
                break
            h[i], h[j] = old_i, old_j
    return h, best_s


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(40) * 0.5)

    best_h = parent.copy()
    best_s = _score(best_h)
    best_n = len(best_h)

    # Phase 1: Screen nearby n values
    parent_n = len(parent)
    n_candidates = sorted(set(
        list(range(max(20, parent_n - 8), min(75, parent_n + 13), 2)) + [parent_n]
    ))
    phase1 = {}
    for n in n_candidates:
        if time.time() > deadline - 100:
            break
        hi = _interp(parent, n)
        s = _score(hi)
        phase1[n] = (s, hi)
        if s < best_s:
            best_h = hi.copy()
            best_s = s
            best_n = n

    # Phase 2: Coordinate descent (key new method)
    if time.time() < deadline - 80:
        h_cd = _coord_optimize_v2(best_h, n_sweeps=40, verify_every=5)
        s_cd = _score(h_cd)
        if s_cd < best_s:
            best_h = h_cd
            best_s = s_cd

    # Phase 3: P-norm SLSQP with increasing p
    for p in [8, 16, 32, 64]:
        if time.time() > deadline - 60:
            break
        h_new = _slsqp_pnorm(best_h, p=p, maxiter=120)
        s_new = _score(h_new)
        if s_new < best_s - 1e-14:
            best_h = h_new
            best_s = s_new
            best_n = len(best_h)

    # Phase 4: DE global search (if budget allows)
    remaining = deadline - time.time()
    if remaining > 150:
        de_budget = int(remaining * 0.2)
        n_gen = min(35, max(10, de_budget // 3))
        h_de, s_de = _de_global(best_h, n_gen=n_gen, pop_size=18, rng=rng)
        if s_de < best_s:
            best_h = h_de
            best_s = s_de
            # Polish the DE result
            if time.time() < deadline - 40:
                h_cd = _coord_optimize_v2(best_h, n_sweeps=20, verify_every=5)
                s_cd = _score(h_cd)
                if s_cd < best_s:
                    best_h = h_cd
                    best_s = s_cd

    # Phase 5: Active-set SLSQP
    if time.time() < deadline - 50:
        h_as = _active_set_slsqp(best_h, max_rounds=8, beta_start=40, beta_end=400)
        s_as = _score(h_as)
        if s_as < best_s:
            best_h = h_as
            best_s = s_as

    # Phase 6: More coordinate descent
    if time.time() < deadline - 40:
        h_cd = _coord_optimize_v2(best_h, n_sweeps=40, verify_every=5)
        s_cd = _score(h_cd)
        if s_cd < best_s:
            best_h = h_cd
            best_s = s_cd

    # Phase 7: Direct minimax
    if time.time() < deadline - 40:
        h_dm = _slsqp_direct(best_h, maxiter=120)
        s_dm = _score(h_dm)
        if s_dm < best_s - 1e-14:
            best_h = h_dm
            best_s = s_dm

    # Phase 8: Final p-norm + coord descent
    if time.time() < deadline - 30:
        for p in [32, 128]:
            if time.time() > deadline - 15:
                break
            h_new = _slsqp_pnorm(best_h, p=p, maxiter=100)
            s_new = _score(h_new)
            if s_new < best_s - 1e-14:
                best_h = h_new
                best_s = s_new

    if time.time() < deadline - 20:
        h_cd = _coord_optimize_v2(best_h, n_sweeps=50, verify_every=5)
        s_cd = _score(h_cd)
        if s_cd < best_s:
            best_h = h_cd
            best_s = s_cd

    # Phase 9: Stochastic CD for fine polish
    if time.time() < deadline - 10:
        n_steps = min(10000, int((deadline - time.time() - 5) * 3000))
        h_st, s_st = _stochastic_cd(best_h, n_steps=max(n_steps, 2000), rng=rng)
        if s_st < best_s:
            best_h = h_st
            best_s = s_st

    # Phase 10: Final coordinate descent polish
    if time.time() < deadline - 5:
        h_final = _coord_optimize_v2(best_h, n_sweeps=30, verify_every=5)
        s_final = _score(h_final)
        if s_final < best_s:
            best_h = h_final
            best_s = s_final

    # Final feasibility check
    best_h = _project(best_h)
    final_s = _score(best_h)

    return (best_h, final_s, len(best_h))
