            G[p] = 0
            G[p, :L] = dx * (-(1.0 - h[s:]) + h[:L])
            G[p, s:] = dx * (-(1.0 - h[s:]))
        else:
            t = -s
            K = n - t
            G[p, :K] = dx * (-h[t:] + (1.0 - h[:K]))
            G[p, t:] = dx * (-(1.0 - h[:K]))
    return G

def grad_correct(h, max_lag=None):
    n = len(h)
    dx = 2.0 / n
    if max_lag is None:
        max_lag = n - 1
    G = np.zeros((2 * n - 1, n))
    for s in range(-max_lag, max_lag + 1):
        p = s + (n - 1)
        if s == 0:
            G[p, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            L = n - s
            G[p, :L] = dx * ((1.0 - h[s:]) + h[:L])
            G[p, s:] = dx * ((1.0 - h[s:]) - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] = dx * ((1.0 - h[t:]) - h[:K])
            G[p, t:] = dx * ((1.0 - h[t:]) + h[t:])
    return G

def verify_gradient(h, num_grad):
    n = len(h)
    vals = compute_vals(h)
    fg = num_grad(h)
    fd = np.zeros(2 * n - 1)
    eps = 1e-7
    for m in range(n):
        h_plus = h.copy()
        h_plus[m] = min(1.0, h[m] + eps)
        h_minus = h.copy()
        h_minus[m] = max(0.0, h[m] - eps)
        d_plus = (compute_vals(h_plus) - vals) / eps
        d_minus = (vals - compute_vals(h_minus)) / eps
        if h[m] >= 1.0 - 1e-12:
            fd[m] = d_minus
        elif h[m] <= 1e-12:
            fd[m] = d_plus
        else:
            fd[m] = (d_plus + d_minus) / 2.0
    max_diff = np.max(np.abs(fg - fd))
    return max_diff

def smooth_obj(h, beta, max_lag=None):
    n = len(h)
    vals = compute_vals(h)[:n]
    cmax = vals.max()
    shifted = beta * (vals[:n] - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(n) / n
    else:
        w /= ws
    return obj, w, cmax

def smooth_obj_grad(h, beta, max_lag=None):
    n = len(h)
    vals = compute_vals(h)[:n]
    cmax = vals.max()
    shifted = beta * (vals[:n] - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(n) / n
    else:
        w /= ws
    G_full = grad_correct(h)
    grad = w @ G_full[:n]
    return obj, grad

def solve_minimax_slsqp(h0, max_lag, tol_gap=1e-5):
    n = len(h0)
    dx = 2.0 / n
    T_init = official_score(h0)
    z = np.empty(n + 1)
    z_prev = None
    c_cur = T_init
    for it in range(20):
        z[:] = 0
        z[:n] = h0
        z[n] = c_cur
        def obj_func(v):
            return v[n]
        def obj_grad(v):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        def ineq_fun(v):
            hc = v[:n]
            t = v[n]
            return t - compute_vals(hc)[:n]
        def ineq_jac(v):
            hc = v[:n]
            G = -grad_correct(hc)[:n]
            J = np.zeros((n, n + 1))
            J[:, :n] = G
            J[:, n] = 1.0
            return J
        def eq_fun(v):
            return np.sum(v[:n]) - n / 2.0
        def eq_jac(v):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
        ]
        try:
            res = minimize(obj_func, z.copy(), jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = make_feasible(res.x[:n])
                sc = official_score(h_new)
                if np.isfinite(sc) and sc < c_cur - 1e-9:
                    if z_prev is not None and np.linalg.norm(res.x[:n] - z_prev[:n]) < 1e-5:
                        update_report = (h_new, sc, res.x[n], T_init, it + 1, sc - res.x[n], n)
                        return update_report
                    h0 = h_new
                    c_cur = sc
                    z_prev = res.x.copy()
                else:
                    update_report = (h_new, official_score(h_new), res.x[n], T_init, it + 1, 0.0, n)
                    return update_report
            else:
                break
        except Exception:
            break
        z_prev = z.copy()
    update_report = (make_feasible(h0), official_score(make_feasible(h0)), c_cur, T_init, 20, 0.0, n)
    return update_report

def active_set_slsqp(h0, threshold=0.9, max_iter=300, max_active=None):
    n = len(h0)
    dx = 2.0 / n
    cmax = official_score(h0)
    active_idx = list(range(n))
    t0 = cmax + 1e-5
    for it in range(max_iter):
        def obj_func(z):
            return z[-1]
        def obj_grad(z):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        da = len(active_idx)
        def ineq_fun(z):
            h = z[:n]
            t = z[-1]
            vals = compute_vals(h)
            return t - vals[active_idx]
        def ineq_jac(z):
            h = z[:n]
            G = -grad_correct(h)
            J = np.zeros((da, n + 1))
            for i, ai in enumerate(active_idx):
                J[i, :n] = G[ai, :]
                J[i, n] = 1.0
            return J
        def eq_fun(z):
            return np.sum(z[:n]) - n / 2.0
        def eq_jac(z):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        z0 = np.empty(n + 1)
        z0[:n] = h0
        z0[n] = t0
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
        ]
        try:
            res = minimize(obj_func, z0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-14})
        except Exception:
            break
        if res.x is None or not np.all(np.isfinite(res.x)):
            break
        h_new = make_feasible(res.x[:n])
        sc = official_score(h_new)
        if sc < cmax - 1e-9:
            cmax = sc
            h0 = h_new
            t0 = cmax + 1e-5
            vals = compute_vals(h0)
            thr = max(cmax - 0.005, 0.5 * cmax * threshold)
            active_mask = vals[:n] >= thr
            active_idx = list(np.where(active_mask)[0])
            if max_active is not None and len(active_idx) > max_active:
                top = np.argsort(vals[:n])[::-1][:max_active]
                active_idx = sorted(top.tolist())
            if len(active_idx) < 2:
                active_idx = list(range(n))
        else:
            break
        if cmax - t0 < -1e-6:
            t0 = cmax + 1e-5
    return h0, official_score(h0)

def pg_solve(h0, beta, time_limit=20.0, max_iter=10000, max_lag=None):
    n = len(h0)
    h = h0.copy()
    f0, g0 = smooth_obj_grad(h, beta, max_lag)
    f = f0
    g = g0
    t0 = time.time()
    alpha = min(1.0 / (max(np.max(g**2), 1e-12) + 1e-12), 2.0)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-26:
            break
        step = alpha
        for _ in range(20):
            y = make_feasible(h - step * g)
            f_trial = smooth_obj(y, beta, max_lag)[0]
            if f_trial <= f - 1e-5 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-13:
            break
        f_new = smooth_obj(y, beta, max_lag)[0]
        if not np.isfinite(f_new):
            break
        if f_new < f:
            h = y
            f = f_new
            g = smooth_obj_grad(y, beta, max_lag)[1]
            alpha = min(alpha * 1.8, 2.0 / (max(np.max(g**2), 1e-12) + 1e-12))
        else:
            alpha *= 0.5
            if alpha < 1e-14:
                break
    return h

def symmetrize(h):
    h_sym = 1.0 - h[::-1].copy()
    h_avg = 0.5 * (h + h_sym)
    return make_feasible(h_avg)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def generate_wave_h(n, freqs, amps, phases):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.zeros(n)
    for f, a, ph in zip(freqs, amps, phases):
        h += a * np.sin(2 * np.pi * f * x + ph)
    h = 0.5 + 0.3 * h
    return make_feasible(h)
