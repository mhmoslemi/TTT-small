    shifted = np.clip(beta * (V - V.max()), -500.0, 0.0)
    w = np.exp(shifted)
    s = w.sum()
    return w / s if s > 1e-300 else np.full(len(V), 1.0/len(V))

def grad_w(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(w)):
        if w[i] < 1e-14: continue
        s = i - (n-1)
        L = n - abs(s)
        if L <= 0: continue
        if s >= 0:
            g[s:] += w[i]*dx*(1-h[:L]); g[:L] -= w[i]*dx*h[s:]
        else:
            t = -s
            g[:L] += w[i]*dx*(1-h[t:]); g[t:] -= w[i]*dx*h[:L]
    return g
