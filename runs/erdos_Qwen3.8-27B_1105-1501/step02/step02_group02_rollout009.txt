def _lag_gradient(h, s):
    """Gradient of F_s(h) = (2/n) * np.correlate(h, 1-h, 'full')[s + n - 1] with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # F_s = (2/n) * sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
        # d/dh[j] = (2/n) * (-h[j+s]) for j in [0, n-1-s]
        #         + (2/n) * (1-h[j-s]) for j in [s, n-1]
        g[:n-s] += -h[s:]  # -h[j+s] for j=0..n-1-s
        g[s:] += 1.0 - h[:-s]  # (1-h[j-s]) for j=s..n-1
    else:
        sp = -s
        # F_s = (2/n) * sum_{i=sp}^{n-1} h[i-sp]*(1-h[i])
        # d/dh[j] = (2/n) * (-h[j-sp]) for j in [sp, n-1]
        #         + (2/n) * (1-h[j+sp]) for j in [0, n-sp-1]
        g[sp:] += -h[:-sp]  # -h[j-sp] for j=sp..n-1
        g[:n-sp] += 1.0 - h[sp:]  # (1-h[j+sp]) for j=0..n-sp-1
    return (2.0 / n) * g

import numpy as np
from scipy.optimize import minimize
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _lag_gradient(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += -h[s:]
        g[s:] += 1.0 - h[:n - s]
    else:
        sp = -s
        g[sp:] += -h[:n - sp]
        g[:n - sp] += 1.0 - h[sp:]
    return (2.0 / n) * g


def _max_subgradient(h, tol=1e-6):
    n = len(h)
    cv = _corr_vec(h)
    t = cv.max()
    active_mask = cv > t - tol
    active_idx = np.where(active_mask)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(cv)])
    g = np.zeros(n)
    for k in active_idx:
        s = k - (n - 1)
        g += _lag_gradient(h, s)
    g /= len(active_idx)
    return g, t, active_idx


def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta


def _lse_grad(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        g += wk * _lag_gradient(x, s)
    return g


def _fista_subgrad(h0, n_iter, alpha_init=1.0, mu=0.9, use_lse=False, beta=500):
    n = len(h0)
    y = h0.copy()
    h_cur = h0.copy()
    h_prev = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1.0
    for it in range(n_iter):
        if use_lse:
            g = _lse_grad(y, beta)
        else:
            g, _, _ = _max_subgradient(y, tol=1e-5)
        h_new = _project(y - alpha_init * g + mu * (h_cur - h_prev))
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = h_new + (t_prev / t - 1.0) * (h_new - h_cur)
        y = _project(y)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_cur - h_avg) / (it + 1)
        if it % 200 == 0 or it == n_iter - 1:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 500 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s


def _fista_backtrack(h0, n_iter, alpha_init=2.0, mu=0.85, tol=1e-5):
    n = len(h0)
    y = h0.copy()
    h_cur = h0.copy()
    h_prev = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1.0
    alpha = alpha_init
    for it in range(n_iter):
        g, t_val, _ = _max_subgradient(y, tol=tol)
        # Backtracking line search
        h_candidate = _project(y - alpha * g)
        if _score(h_candidate) >= t_val + 1e-10 and alpha > 1e-6:
            alpha *= 0.5
            h_candidate = _project(y - alpha * g)
        if alpha < 1e-8:
            alpha = alpha_init * 0.1
            h_candidate = _project(y - alpha * g)
        else:
            if _score(h_candidate) < best_s - 1e-12:
                pass  # good step
            else:
                alpha *= 0.5
                h_candidate = _project(y - alpha * g)
        h_new = h_candidate
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = h_new + (t_prev / t - 1.0) * (h_new - h_cur)
        y = _project(y)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_cur - h_avg) / (it + 1)
        if it % 200 == 0 or it == n_iter - 1:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 500 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s


def _slsqp_eigraph(h0, maxiter=100):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = cv.max() + 0.002
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.01, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv

    def _eq(z):
        return z[:-1].sum() - n / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            h_new = _project(res['x'][:-1])
            if _score(h_new) <= _score(h0) + 1e-10:
                return h_new
    except Exception:
        pass
    return h0


def _slsqp_lse(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _coord_local_search(h0, max_rounds=200, deadline=None):
    n = len(h0)
    h = h0.copy()
    s = _score(h)
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        rounds += 1
        cv = _corr_vec(h)
        t = cv.max()
        idx = np.arange(n)
        for i in idx:
            if deadline is not None and time.time() > deadline:
                break
            for j in idx:
                if j <= i:
                    continue
                if h[i] <= h[j]:
                    continue
                # Try swapping
                h2 = h.copy()
                h2[i], h2[j] = h[j], h2[i]
                s2 = _score(h2)
                if s2 < s - 1e-12:
                    h = h2
                    s = s2
                    improved = True
                    break
                # Try transferring mass from i to j
                if h[i] > 0.05 and h[j] < 0.95:
                    for frac in [0.3, 0.5, 0.7]:
                        amt = frac * h[i]
                        h2 = h.copy()
                        h2[i] -= amt
                        h2[j] += amt
                        h2 = _project(h2)
                        s2 = _score(h2)
                        if s2 < s - 1e-12:
                            h = h2
                            s = s2
                            improved = True
                            break
                    if improved:
                        break
    return h, s


def _binary_local_search(h0, max_rounds=100, n_trials=40, deadline=None):
    n = len(h0)
    h = h0.copy()
    s = _score(h)
    rng = np.random.default_rng(12345)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        found = False
        for _ in range(n_trials):
            if deadline is not None and time.time() > deadline:
                break
            # Random pair swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            h2 = h.copy()
            h2[i], h2[j] = h2[j], h2[i]
            s2 = _score(h2)
            if s2 < s - 1e-14:
                h = h2
                s = s2
                found = True
            # Random perturbation
            h2 = _project(h + 0.02 * rng.standard_normal(n))
            s2 = _score(h2)
            if s2 < s - 1e-14:
                h = h2
                s = s2
                found = True
        if not found:
            break
    return h, s


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _deep_optimize(h0, deadline):

                phase1.append((_score(h_int), n, h_int))
        else:
            phase1.append((_score(h_int), n, h_int))
        sc = phase1[-1][0]
        if sc < best_score - 1e-12:
            best_h = phase1[-1][2].copy()
            best_score = sc
            best_n = n

    # Update best from parent if still better
    ps = _score(parent)
    if ps < best_score:
        best_h = parent.copy()
        best_score = ps
        best_n = parent_n

    # Phase 2: Deep optimization on top dimensions
    phase1.sort(key=lambda x: x[0])
    top_results = []
    seen_ns = set()
    for s, n, h in phase1:
        if n not in seen_ns:

            top_results.append((s, n, h))

    for s, n, h_base in top_results:

            cur2, cur_s2 = _deep_optimize(h_pert, deadline - 20)
            if cur_s2 < best_score - 1e-12:

    # Final feasibility

    final_score = _score(best_h)

    return (best_h, final_score, len(best_h))
