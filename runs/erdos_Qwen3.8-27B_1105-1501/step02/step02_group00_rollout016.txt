def get_scores(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(v, target):
    n = len(v)
    target = float(target)
    if target <= 1e-15:
        return np.zeros(n)
    if target >= n - 1e-15:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            adj = res / cnt
            x[mask] = np.clip(x[mask] + adj, 0.0, 1.0)
    return x

def smooth_obj_and_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    s = np.correlate(h, 1.0 - h, mode='full') * dx
    bs = beta * s
    m = float(np.max(bs))
    exp_bs = np.exp(bs - m)
    total = float(np.sum(exp_bs))
    w = exp_bs / total
    obj = (m + np.log(total)) / beta
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        k = r - (n - 1)
        js = max(0, k)
        je = min(n, n + k)
        if je > js:
            g[js:je] += w[r] * dx * (1.0 - h[js-k:je-k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if je2 > js2:
            g[js2:je2] += w[r] * dx * (-h[js2+k:je2+k])
    if penalty > 0:
        sh = float(np.sum(h))
        tgt = n / 2.0
        obj += penalty * (sh - tgt) ** 2
        g += 2.0 * penalty * (sh - tgt)
    return obj, g

def run_lsm(h, beta, penalty, max_iter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        o, _ = smooth_obj_and_grad(x, beta, penalty)
        return o
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, beta, penalty)
        return g
    
    try:
        res = minimize(fun, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': max_iter, 
                               'maxfun': max_iter * 2,
                               'maxls': 50})
        if time.time() < deadline:
            return res.x
    except Exception:
        pass
    return h

def active_grad(h, top_k=3):
    """Compute subgradient using only top-k active lags."""
    n = len(h)
    dx = 2.0 / n
    s = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top-k lags
    idx = np.argsort(s)[-top_k:]
    max_s = s[idx[0]]
    # Weights: uniform over active
    w = np.zeros(2 * n - 1)
    for r in idx:
        w[r] = 1.0 / len(idx)
    
    g = np.zeros(n)
    for r in idx:
        wr = w[r]
        k = r - (n - 1)
        js = max(0, k)
        je = min(n, n + k)
        if je > js:
            g[js:je] += wr * dx * (1.0 - h[js-k:je-k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if je2 > js2:
            g[js2:je2] += wr * dx * (-h[js2+k:je2+k])
    return g

def subgradient_step(h, target, step, deadline, max_iters=5000, top_k=3):

    n = len(h)
    h = h.copy()
    best_h = h.copy()

    alpha = step
    stall = 0

            else:

def pair_descent(h, target, deadline, max_passes=20):
    """Local pair search."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_exact(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:

                continue
            for k in range(j + 1, n, 2):
                if h[k] <= 1e-6 or h[k] >= 1.0 - 1e-6:
                    continue
