            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def compute_lags(h):
    n = len(h)

def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        m = k - (n - 1)
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if m >= 0:
            term1[:n-m] = 1.0 - h[m:]
            term2[m:] = h[:n-m]
        else:
            p = -m
            term1[p:] = 1.0 - h[:n-p]
            term2[:n-p] = h[p:]
        G[k] = dx * (term1 - term2)
    return G

def solve_lp_minimax(h):
    n = len(h)
    lags = compute_lags(h)
    c_max = lags.max()
    G = compute_gradient_matrix(h)
    
    d = cp.Variable(n)
    eps = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        d >= -h,
        d <= 1.0 - h
    ]
    # For each lag k: G[k] @ d + eps <= c_max - lags[k]
    for k in range(len(lags)):
        constraints.append(G[k, :] @ d + eps <= c_max - lags[k])
    
    prob = cp.Problem(cp.Maximize(eps), constraints)
    
    last_err = None
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.GLOP]:
        try:
            prob.solve(solver=solver, max_iter=10000)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return np.array(d.value).flatten(), float(eps.value)
        except Exception as e:
            last_err = e
            continue
    return None, 0.0

                best = h_trial
                best_sc = sc
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_sc

        cmax = lags.max()
        tol = 0.03
        active = np.where(lags >= cmax - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(lags)])
        w = np.exp(80.0 * (lags[active] - cmax))
        w = w / w.sum()
        G = compute_gradient_matrix(best)
        grad = G[active].T @ w
        d = -grad
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25, alpha * 4.0]:
            h_trial = make_feasible(best + mult * d)
            sc = official_score(h_trial)
            if sc < best_sc - 1e-14:
                best = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:
            alpha = min(alpha * 1.1, 0.1)
    return best, best_sc

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.15)
        best_score = official_score(best_h)

    # Phase 1: LP iterations at parent resolution
    if not is_dead():
        h_new, sc = lp_iteration(best_h, max_iters=500, is_dead=is_dead)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Phase 2: Subgradient polish
    if not is_dead():
        h_new, sc = subgradient_polish(best_h, 100, is_dead)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Phase 3: LP again after polish
    if not is_dead():
        h_new, sc = lp_iteration(best_h, max_iters=300, is_dead=is_dead)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Phase 4: Try different resolutions with LP
    n_base = len(best_h)
    n_opts = set()
    for dn in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 10, 14, 16, 20, 24]:
        nt = n_base + dn
        if 30 <= nt <= 140:
            n_opts.add(nt)
    for nc in [36, 40, 44, 48, 52, 56, 64, 72, 80, 90, 100, 110, 120]:
        if 30 <= nc <= 140:
            n_opts.add(nc)
    n_opts = sorted(n_opts)

    for n_opt in n_opts:
        if is_dead():
            break
        if n_opt == n_base:
            continue
        h_int = interp_to(best_h, n_opt)
        sc_int = official_score(h_int)
        if sc_int < best_score:
            best_h = h_int
            best_score = sc_int
        # Run LP on this resolution
        h_new, sc = lp_iteration(h_int, max_iters=150, is_dead=is_dead)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Phase 5: 2-point polish on best
    if not is_dead():
        n_tries = max(3000, int((deadline - time.time()) * 4))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 6: LP on polished solution
    if not is_dead():
        h_new, sc = lp_iteration(best_h, max_iters=300, is_dead=is_dead)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Phase 7: Interpolate back to best resolution and LP again
    if not is_dead():
        n_best = len(best_h)
        for n_try in [n_best - 4, n_best + 4, n_best - 8, n_best + 8]:
            if is_dead() or n_try < 30 or n_try > 140:
                continue
            h_int = interp_to(best_h, n_try)
            sc_int = official_score(h_int)
            if sc_int < best_score:
                best_h = h_int
                best_score = sc_int
            h_new, sc = lp_iteration(h_int, max_iters=200, is_dead=is_dead)
            if sc < best_score:
                best_h = h_new
                best_score = sc

    # Phase 8: Random perturbations + LP
    if not is_dead():
        for sigma in [0.03, 0.06, 0.1, 0.15, 0.25]:
            if is_dead():
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=len(best_h)))
            h_new, sc = lp_iteration(h_pert, max_iters=200, is_dead=is_dead)
            if sc < best_score:
                best_h = h_new
                best_score = sc
            if not is_dead():
                h_new2, sc2 = two_point_search(h_pert, 800, rng, is_dead)
                if sc2 < best_score:
                    best_h = h_new2
                    best_score = sc2

    # Phase 9: Anti-symmetric exploration
    if not is_dead():
        h_rev = 1.0 - best_h[::-1].copy()
        h_rev = make_feasible(h_rev)
        sc_rev = official_score(h_rev)
        h_sym = make_feasible(0.5 * (best_h + h_rev))
        sc_sym = official_score(h_sym)
        if sc_sym < best_score:
            best_h = h_sym
            best_score = sc_sym
        if not is_dead():
            h_new, sc = lp_iteration(best_h, max_iters=200, is_dead=is_dead)
            if sc < best_score:
                best_h = h_new
                best_score = sc

    # Phase 10: Final intensive 2-point + LP
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 3))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    if not is_dead():
        h_new, sc = lp_iteration(best_h, max_iters=200, is_dead=is_dead)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-10

    return final_h, final_score, final_n
