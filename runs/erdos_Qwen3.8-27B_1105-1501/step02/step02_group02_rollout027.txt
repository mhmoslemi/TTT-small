def _score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0/n))

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0/n)

def _corr_subset(h, k_indices):
    n = len(h)
    dx = 2.0 / n
    result = np.zeros(len(k_indices))
    for i, k in enumerate(k_indices):
        s = k - (n - 1)
        if s >= 0:
            result[i] = dx * np.sum(h[0:n-s] * (1.0 - h[s:n]))
        else:
            r = -s
            result[i] = dx * np.sum(h[r:n] * (1.0 - h[0:n-r]))
    return result

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - float(np.max(z))
    hi = 2.0 - float(np.min(z))
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(res) < 1e-14:
                break
            d = min(res, 1.0 - h[i]) if res > 0 else max(res, h[i] - 1.0)
            h[i] += d
            res -= d
    return np.clip(h, 0.0, 1.0)

def _interp(h_src, n_dst):
    if len(h_src) == n_dst:
        return h_src.copy()
    xs = (np.arange(len(h_src)) + 0.5) * (2.0 / len(h_src))
    xd = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(xd, xs, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _binary_push(h, p=2.0):
    return _project(h ** p)

def _active_solve(h0, active_idx, center, rho, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    cv0 = _corr(h0)
    t0 = float(cv0[active_idx].max()) + 0.003
    m = len(active_idx)
    nv = n + 1
    
    def obj(z):
        return z[-1] + 0.5 * rho * np.sum((z[:n] - center)**2)
    
    def obj_grad(z):
        g = np.zeros(nv)
        g[-1] = 1.0
        if rho > 0:
            g[:n] = rho * (z[:n] - center)
        return g
    
    def con_ineq(z):
        h = z[:n]
        return z[-1] - _corr_subset(h, active_idx)
    
    def con_ineq_jac(z):
        h = z[:n]
        J = np.zeros((m, nv))
        for i, k in enumerate(active_idx):
            J[i, -1] = 1.0
            s = k - (n - 1)
            if s >= 0:
                J[i, 0:n-s] -= dx * (1.0 - h[s:n])
                J[i, s:n] += dx * h[0:n-s]
            else:
                r = -s
                J[i, r:n] -= dx * (1.0 - h[0:n-r])
                J[i, 0:n-r] += dx * h[r:n]
        return J
    
    def con_eq(z):
        return z[:n].sum() - n / 2.0
    
    def con_eq_jac(z):
        g = np.zeros(nv)
        g[:n] = 1.0
        return g
    
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * nv
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[
                          {'type': 'ineq', 'fun': con_ineq, 'jac': con_ineq_jac},
                          {'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}
                      ],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res['x'])):
            return _project(res['x'][:n])
    except Exception:
        pass
    return h0.copy()

def _lse_smooth(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    
    def lse_obj(x):
        cv = _corr(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta
    
    def lse_grad(x):
        cv = _corr(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2*n-1, 1.0/(2*n-1))
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n-1)
            c = dx * wk
            if s >= 0:
                g[0:n-s] += c * (1.0 - x[s:n])
                g[s:n] -= c * x[0:n-s]
            else:
                r = -s
                g[r:n] += c * (1.0 - x[0:n-r])
                g[0:n-r] -= c * x[r:n]
        return g
    
    bounds = [(0.0, 1.0)] * n
    def eq_fun(x): return x.sum() - n/2.0
    def eq_jac(x): return np.ones(n)
    
    try:
        res = minimize(lse_obj, h0, jac=lse_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0.copy()

def _fista_subgrad(h0, beta, n_iter, alpha=0.005, mu=0.9):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t_counter = 1.0
    
    for it in range(n_iter):
        cv = _corr(y)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2*n-1, 1.0/(2*n-1))
        else:
            w = ev / es
        
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-15: continue
            s = k - (n-1)
            c = dx * wk
            if s >= 0:
                g[0:n-s] += c * (1.0 - y[s:n])
                g[s:n] -= c * y[0:n-s]
            else:
                r = -s
                g[r:n] += c * (1.0 - y[0:n-r])
                g[0:n-r] -= c * y[r:n]
        
        h_new = y - alpha * g
        h_new = _project(h_new)
        
        t_new = 1.5 * t_counter
        y = h_new + (t_counter / t_new) * (h_new - h_prev)
        y = _project(y)
        
        h_prev = h_cur
        h_cur = h_new
        t_counter = t_new
        
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
    
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    
    return best_h, best_s

import numpy as np
from scipy.optimize import minimize
import time

_ACTIVE_IDX = None
_CENTER = None
_RHO = None
_N_ACT = None
_BETA_LSE = 100
_N_LSE = 57

def _score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _corr_subset(h, k_indices):
    n = len(h)
    dx = 2.0 / n
    result = np.zeros(len(k_indices))
    for i, k in enumerate(k_indices):
        s = k - (n - 1)
        if s >= 0:
            result[i] = dx * np.sum(h[0:n-s] * (1.0 - h[s:n]))
        else:
            r = -s
            result[i] = dx * np.sum(h[r:n] * (1.0 - h[0:n-r]))
    return result

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - float(np.max(z))
    hi = 2.0 - float(np.min(z))
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(res) < 1e-14:
                break
            d = min(res, 1.0 - h[i]) if res > 0 else max(res, h[i] - 1.0)
            h[i] += d
            res -= d
    return np.clip(h, 0.0, 1.0)

def _interp(h_src, n_dst):
    if len(h_src) == n_dst:
        return h_src.copy()
    xs = (np.arange(len(h_src)) + 0.5) * (2.0 / len(h_src))
    xd = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(xd, xs, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _binary_push(h, p):
    return _project(h ** p)

def _blur(h, kernel=(1, 3, 3, 1)):
    k = np.array(kernel, dtype=float)
    k = k / k.sum()
    h_pad = np.concatenate([h[:1], h, h[-1:]])
    result = np.convolve(h_pad, k, mode='valid')
    return _project(result)

def _active_obj(z):
    n = _N_ACT
    return z[-1] + 0.5 * _RHO * np.sum((z[:n] - _CENTER) ** 2)

def _active_obj_grad(z):
    n = _N_ACT
    g = np.zeros(len(z))
    g[-1] = 1.0
    if _RHO > 0:
        g[:n] = _RHO * (z[:n] - _CENTER)
    return g

def _active_ineq(z):
    n = _N_ACT
    return z[-1] - _corr_subset(z[:n], _ACTIVE_IDX)

def _active_ineq_jac(z):
    n = _N_ACT
    dx = 2.0 / n
    h = z[:n]
    m = len(_ACTIVE_IDX)
    J = np.zeros((m, n + 1))
    for i, k in enumerate(_ACTIVE_IDX):
        J[i, -1] = 1.0
        s = k - (n - 1)
        if s >= 0:
            J[i, 0:n-s] -= dx * (1.0 - h[s:n])
            J[i, s:n] += dx * h[0:n-s]
        else:
            r = -s
            J[i, r:n] -= dx * (1.0 - h[0:n-r])
            J[i, 0:n-r] += dx * h[r:n]
    return J

def _active_eq(z):
    return z[:_N_ACT].sum() - _N_ACT / 2.0

def _active_eq_jac(z):
    g = np.zeros(len(z))
    g[:_N_ACT] = 1.0
    return g

def _active_solve(h0, active_idx, center, rho, maxiter):
    global _ACTIVE_IDX, _CENTER, _RHO, _N_ACT
    n = len(h0)
    cv0 = _corr(h0)
    t0 = float(cv0[active_idx].max()) + 0.003
    _ACTIVE_IDX = np.asarray(active_idx)
    _CENTER = center.copy()
    _RHO = rho
    _N_ACT = n
    z0 = np.concatenate([h0, [min(t0, 0.99)]])
    bounds = [(0.0, 1.0)] * (n + 1)
    try:
        res = minimize(_active_obj, z0, jac=_active_obj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[
                          {'type': 'ineq', 'fun': _active_ineq, 'jac': _active_ineq_jac},
                          {'type': 'eq', 'fun': _active_eq, 'jac': _active_eq_jac}
                      ],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res['x'])):
            return _project(res['x'][:n])
    except Exception:
        pass
    return h0.copy()

def _lse_obj_func(z):
    n = _N_LSE
    cv = _corr(z)
    mc = cv.max()
    sh = np.clip(_BETA_LSE * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / _BETA_LSE

def _lse_obj_grad(z):
    n = _N_LSE
    dx = 2.0 / n
    cv = _corr(z)
    mc = cv.max()
    sh = np.clip(_BETA_LSE * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        c = dx * wk
        if s >= 0:
            g[0:n-s] += c * (1.0 - z[s:n])
            g[s:n] -= c * z[0:n-s]
        else:
            r = -s
            g[r:n] += c * (1.0 - z[0:n-r])
            g[0:n-r] -= c * z[r:n]
    return g

def _lse_eq(z):
    return z[:_N_LSE].sum() - _N_LSE / 2.0

def _lse_eq_jac(z):
    g = np.zeros(len(z))
    g[:_N_LSE] = 1.0
    return g

def _lse_smooth(h0, beta, maxiter):
    global _N_LSE, _BETA_LSE
    n = len(h0)
    _N_LSE = n
    _BETA_LSE = beta
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj_func, h0, jac=_lse_obj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _lse_eq, 'jac': _lse_eq_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0.copy()

def _fista_subgrad(h0, beta, n_iter, alpha, mu):
    n = len(h0)
    dx = 2.0 / n
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t_counter = 1.0
    for it in range(n_iter):
        cv = _corr(y)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            c = dx * wk
            if s >= 0:
                g[0:n-s] += c * (1.0 - y[s:n])
                g[s:n] -= c * y[0:n-s]
            else:
                r = -s
                g[r:n] += c * (1.0 - y[0:n-r])
                g[0:n-r] -= c * y[r:n]
        h_new = _project(y - alpha * g)
        t_new = 1.5 * t_counter
        y = _project(h_new + (t_counter / t_new) * (h_new - h_prev))
        h_prev = h_cur
        h_cur = h_new
        t_counter = t_new
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    return best_h, best_s

def _active_rounds(h, n_rounds, maxiter, deadline):
    n = len(h)
    best = h.copy()
    best_s = _score(best)
    for rnd in range(n_rounds):
        if time.time() > deadline - 5:
            break
        cv = _corr(best)
        cv_max = cv.max()
        threshold = cv_max - 0.002
        active = np.where(cv >= threshold)[0]
        top_k = np.argsort(cv)[::-1][:min(8, len(cv))]
        active = np.unique(np.concatenate([active, top_k]))
        improved = False
        for rho in [5e-4, 1e-4, 5e-5, 0.0]:
            candidate = _active_solve(best, active, best, rho, maxiter)
            s = _score(candidate)
            if s < best_s - 1e-14:
                best = candidate
                best_s = s
                improved = True
                break
    return best, best_s

def _deep_optimize(h0, deadline, rng):
    n = len(h0)
    best = h0.copy()
    best_s = _score(best)

    # LSE warm-up
    if time.time() < deadline - 40:
        candidate = _lse_smooth(best, 100, 50)
        s = _score(candidate)
        if s < best_s:
            best, best_s = candidate, s

    # Active-set iteration phase 1 (broader)
    best, best_s = _active_rounds(best, 6, 120, deadline)

    # LSE mid polish
    if time.time() < deadline - 20:
        candidate = _lse_smooth(best, 400, 60)

    # Final active polish

    return best, best_s

def _gen_starts(base, n, rng):
    starts = [base.copy(), _mirror(base)]
    for p in [1.5, 2.0, 2.5, 3.0, 4.0]:
