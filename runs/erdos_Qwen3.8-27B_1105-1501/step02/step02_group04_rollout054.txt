    dx = 2.0 / n

        if wj < 1e-15:

def project_feasible(x, target_sum):
    """Project x onto {0<=x_i<=1, sum(x)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        # Find free coordinates
        free = (result > 1e-9) & (result < 1.0 - 1e-9)
        if np.any(free):
            n_free = np.sum(free)
            adj = residual / n_free
            result[free] = np.clip(result[free] + adj, 0.0, 1.0)
        else:

            if residual > 0:

                idx = np.argsort(result)[:int(abs(residual)) + 1]

                        result[i] += add

                        result[i] -= sub

    for _ in range(steps):
        grad = smooth_gradient_raw(h, T)
        # Project gradient onto feasible direction (sum(grad)=0, respect bounds)
        grad = grad - np.mean(grad)
        # Scale to step size
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        direction = grad / norm * step_size
        h_new = h + direction
        h_new = project_feasible(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _current_T
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_deadline = t0 + budget_s - 3.0

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 10:
            parent = None
    except (NameError, Exception):
        pass

    if parent is not None:
        n_main = len(parent)
        parent = project_feasible(parent)
    else:
        n_main = 70
        parent = project_feasible(np.full(n_main, 0.5))

    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = n_main

    # Temperature schedule (raw correlation units)
    T_coarse = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25]
    T_fine = [0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.002, 0.001]
    T_very_fine = [0.0008, 0.0005, 0.0003, 0.00015, 0.00008]

    # Phase 1: Anneal parent (30% of time)
    phase1_end = t0 + budget_s * 0.25
    h_cur = parent.copy()
    for T in T_coarse + T_fine:
        if time.time() > phase1_end:
            break
        h_cur = slsqp_single_step(h_cur, T, maxiter=200)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = len(h_cur)

    # Phase 2: Basin hopping at main n (35% of time)
    phase2_end = t0 + budget_s * 0.60
    n_hop = best_n
    hop_count = 0
    while time.time() < phase2_end:
        hop_count += 1
        scale = 0.03 + 0.02 * (hop_count % 5)
        h_pert = best_h + rng.randn(len(best_h)) * scale
        h_pert = project_feasible(h_pert)
        h_try = h_pert.copy()
        # Quick anneal with fewer T values
        T_hop = T_fine[-6:] + T_very_fine
        for T in T_hop:
            if time.time() > phase2_end:
                break
            h_try = slsqp_single_step(h_try, T, maxiter=100)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)

    # Phase 3: Try different n values (20% of time)
    phase3_end = t0 + budget_s * 0.80
    n_candidates = sorted(set([30, 40, 50, 60, 70, 80, 90, 100, 120, 140]),
                          key=lambda x: abs(x - best_n))
    for n_try in n_candidates[:6]:
        if time.time() > phase3_end:
            break
        if n_try == best_n:
            continue
        # Transfer best to new n by interpolation
        x_old = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp)
        # Also try random starts
        starts = [h_interp, make_random_binary(n_try, rng), make_sine_pattern(n_try, rng), make_golden_pattern(n_try)]
        for h_start in starts:
            if time.time() > phase3_end:
                break
            h_try = h_start.copy()
            T_try = T_fine[-4:] + T_very_fine
            for T in T_try:
                if time.time() > phase3_end:
                    break
                h_try = slsqp_single_step(h_try, T, maxiter=100)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = len(h_try)

    # Phase 4: Fine polish on best (15% of time)
    phase4_end = t0 + budget_s * 0.93
    h_cur = best_h.copy()
    for T in T_very_fine + [0.00004, 0.00002, 0.00001]:
        if time.time() > phase4_end:
            break
        h_cur = slsqp_single_step(h_cur, T, maxiter=300)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = len(h_cur)

    # Gradient direction polish
    if time.time() < phase4_end:
        h_grad, c5_grad = gradient_direction_polish(best_h, T=0.005, steps=80, step_size=0.005)
        if c5_grad < best_c5 - 1e-12:
            best_c5 = c5_grad
            best_h = h_grad.copy()

    # Phase 5: Pair polishing (remaining time)
    h_final = best_h.copy()
    pair_deadline = t0 + budget_s - 5.0
    while time.time() < pair_deadline:
        n_f = len(h_final)
        perm = rng.permutation(n_f)
        improved = False
        for a in range(n_f):
            if time.time() > pair_deadline:
                break
            for b in range(a + 1, n_f):
                i, j = perm[a], perm[b]
                if i > j:
                    i, j = j, i
                d = pair_line_search(h_final, i, j)
                if abs(d) > 1e-12:
                    h_final[i] -= d
                    h_final[j] += d
                    c5_f = compute_c5(h_final)
                    if c5_f < best_c5 - 1e-12:
                        best_c5 = c5_f
                        best_h = h_final.copy()
                        best_n = len(h_final)
                        improved = True
        if not improved:
            # Try a few random pairs
            for _ in range(200):
                if time.time() > pair_deadline:
                    break
                i = rng.randint(n_f)
                j = rng.randint(n_f)
                if i == j:
                    continue
                d = pair_line_search(h_final, i, j)
                if abs(d) > 1e-12:
                    h_final[i] -= d
                    h_final[j] += d
                    c5_f = compute_c5(h_final)
                    if c5_f < best_c5 - 1e-12:
                        best_c5 = c5_f
                        best_h = h_final.copy()
                        best_n = len(h_final)
                        improved = True
            if not improved:
                break

    # Final projection and validation
    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        # Shouldn't happen but safety
        final_c5 = best_c5

    return best_h, float(final_c5), int(best_n)
