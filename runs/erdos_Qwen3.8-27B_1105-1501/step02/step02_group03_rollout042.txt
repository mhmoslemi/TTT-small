import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo = -np.max(np.abs(z)) - 2.0
    hi = np.max(np.abs(z)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def sym_to_h(z):
    M = len(z)
    h = np.empty(2 * M)
    h[:M] = z
    h[M:] = 1.0 - z[::-1]
    return h

def h_to_sym(h):
    n = len(h)
    M = n // 2
    if n % 2 == 0:
        return 0.5 * (h[:M] + 1.0 - h[M:][::-1])
    else:
        return 0.5 * (h[:M] + 1.0 - h[M+1:][::-1])

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_full(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] -= dx * h[s:]
            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K]
    return G

def soft_obj_grad_sym(z, beta):
    n = 2 * len(z)
    h = sym_to_h(z)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    logsum = max_s + np.log(sum_exp)
    obj = cmax + logsum / beta
    w = exp_s / sum_exp
    G = grad_full(h)
    M = len(z)
    v = w @ G
    grad = v[:M] - v[n - 1 - np.arange(M)]
    return obj, grad
