def _exact_c5(h):

def _scores(h):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))

        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:

            m = n - g

def _lp_direction(h, scores, G, radius, top_k=None):

    n = len(h)
    n_lags = len(scores)

        G_k = G[idx]
    else:

        G_k = G
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # maximize t
    
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    except Exception:
        pass
    return None

    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.6
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    # Dense initial grid
    n_samp = 30
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = np.argmin(vals)
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.15, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 15 and radius < 0.03:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_optimize(h, deadline, n_steps=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 5.0 * (200.0 / 5.0) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        alpha = 0.05
        for _ in range(40):
            h_new = _project(h + alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=12):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Aggressive LP with large radius
    dl = t0 + budget_s * 0.20
    h, c5 = _lp_optimize(h, dl, top_k_list=[None, None, 20], radius_init=0.3, max_iters=3000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Smooth gradient with increasing beta
    if time.time() < t0 + budget_s * 0.35:
        dl = t0 + budget_s * 0.38
        h, c5 = _smooth_optimize(best_h, dl, n_steps=250)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Refined LP with small top_k
    if time.time() < t0 + budget_s * 0.50:
        dl = t0 + budget_s * 0.55
        h, c5 = _lp_optimize(best_h, dl, top_k_list=[5, 8, 10, None], radius_init=0.06, max_iters=4000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Perturbation restarts
    dl = t0 + budget_s * 0.72
    scale = 0.025
    for attempt in range(400):
        if time.time() > dl:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl, time.time() + budget_s * 0.025)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k_list=[10, 5, None], radius_init=0.08, max_iters=400)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.12)

    # Phase 5: Pair polish
    if time.time() < t0 + budget_s * 0.82:
        dl = t0 + budget_s * 0.85
        hp, c5p = _pair_polish(best_h, dl, max_passes=10)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 6: Resolution sweep
    if time.time() < t0 + budget_s * 0.92:
        dl = t0 + budget_s * 0.94
        for dn in [-6, -5, -4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 8, 10]:
            if time.time() > dl:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 160 or n_try == best_n:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl, time.time() + budget_s * 0.025)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[10, None], radius_init=0.1, max_iters=300)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final ultra-fine LP polish
    if time.time() < t0 + budget_s * 0.97:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.995, top_k_list=[3, 5, 4], radius_init=0.015, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
