    cm = v.max()
    sh = np.clip(beta * (v - cm), -100, 100)
    w = np.exp(sh); w /= w.sum()
    g = np.zeros(n)
    for p in range(len(v)):
        if w[p] < 1e-14: continue
        s = p - (n-1)
        c = dx * w[p]
        if s >= 0:
            g[:n-s] += c * (1.0 - h[s:])
            g[s:] -= c * h[:n-s]
        else:
            t = -s
            g[t:] += c * (1.0 - h[:n-t])
            g[:n-t] -= c * h[t:]
    return g

def eqf(h): return np.sum(h) - len(h)/2.0
def eqj(h): return np.ones(len(h))

def run_slsqp(h0, betas, deadline, maxit=150):
    n = len(h0)
    bnd = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':eqf,'jac':eqj}]
    bh = h0.copy(); bs = official_score(h0)
    for beta in betas:
        if time.time() > deadline - 5: break
        try:
            res = minimize(sfunc, bh, args=(beta,), method='SLSQP',
                         jac=sgrad, bounds=bnd, constraints=cons,
                         options={'maxiter':maxit,'ftol':1e-15})
            if res.x is not None and np.all(np.isfinite(res.x)):
                hn = make_feasible(res.x)
                sc = official_score(hn)
                if sc < bs - 1e-14:
                    bh = hn; bs = sc
        except: pass
    return bh, bs

def subgrad_polish(h0, deadline, n_iter=500):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    beta = 500.0
    step = 0.05
    momentum = np.zeros(n)
    rho = 0.7
    for it in range(n_iter):
        if time.time() > deadline - 5: break
        g = sgrad(h, beta)
        momentum = rho * momentum + (1-rho) * g
        d = -momentum / (np.max(np.abs(momentum)) + 1e-15)
        # Line search
        best_alpha = 0; best_sc = best
        for alpha in [step, step*0.5, step*0.25, step*0.7, step*1.3]:
            ht = make_feasible(h + alpha * d)
            sc = official_score(ht)
            if sc < best_sc - 1e-15:
                best_alpha = alpha; best_sc = sc
