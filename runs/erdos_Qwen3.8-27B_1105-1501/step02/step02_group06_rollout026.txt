import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def grad_logsumexp(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    bc = beta * c
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w /= w.sum()
    g = np.zeros(n)
    wm = w * dx
    for p in range(2 * n - 1):
        k = p - (n - 1)
        wk = wm[p]
        if abs(wk) < 1e-15:
            continue
        if k >= 0:
            g[k:] += wk * (1.0 - h[:n - k])
            g[:n - k] -= wk * h[k:]
        else:
            j = -k
            g[:n - j] += wk * (1.0 - h[j:])
            g[j:] -= wk * h[:n - j]
    return g

def adam_optimize(h0, beta_start, beta_end, lr, n_iter, check_every=10):
    n = len(h0)
    h = h0.copy()
    best_score = official_score(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    for it in range(n_iter):
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, n_iter - 1))
        g = grad_logsumexp(h, beta)
        g -= np.mean(g)
        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step)
        if it % check_every == 0 or it == n_iter - 1:
            sc = official_score(h_new)
            if sc < best_score - 1e-15:
                h = h_new
                best_score = sc
            else:
                h = h_new
    return h, best_score

def slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    def obj(x):
        c = compute_c(x)
        bc = beta * c
        bc_max = bc.max()
        return bc_max + np.log(np.sum(np.exp(bc - bc_max))) / beta
    def obj_grad(x):
        return grad_logsumexp(x, beta)
    def sum_eq(x):
        return np.sum(x) - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return project_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    best_h = None
    best_score = float('inf')

    def update(h, sc):
        nonlocal best_h, best_score
        if h is not None and np.isfinite(sc) and sc < best_score - 1e-15:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = project_feasible(parent)
            ps = official_score(parent)
            update(parent, ps)
    except (NameError, AttributeError, UnboundLocalError):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x / 2.0))
        best_score = official_score(best_h)
        update(best_h, best_score)

    # Phase 1: Adam with beta annealing from parent
    if remaining() > 100:
        n = len(best_h)
        h_cur = best_h.copy()
        for attempt in range(3):
            if remaining() < 30:
                break
            b_s = 50.0 + 50.0 * attempt
            b_e = 5000.0 + 5000.0 * attempt
            lr_use = 0.05 / (1 + attempt * 0.5)
            n_it = min(3000, int(remaining() * 20))
            h_new, sc = adam_optimize(h_cur, b_s, b_e, lr_use, n_it, check_every=20)
            update(h_new, sc)
            h_cur = h_new

    # Phase 2: SLSQP with increasing beta
    if remaining() > 60:
        h_cur = best_h.copy()
        for beta in [200, 500, 2000, 10000]:
            if remaining() < 20:
                break
            h_new, sc = slsqp_smooth(h_cur, beta, maxiter=150)
            if h_new is not None:
                update(h_new, sc)
                if sc <= best_score + 1e-14:
                    h_cur = h_new

    # Phase 3: Resolution sweep
    if remaining() > 120:
        n_base = len(best_h)
        n_opts = sorted(set([max(24, n_base - 10), n_base - 5, n_base,
                             n_base + 5, n_base + 10, n_base + 20]))
        for n_opt in n_opts:
            if remaining() < 30:
                break
            if n_opt < 20 or n_opt > 100:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            if remaining() > 20:
                h_new, sc = adam_optimize(h_int, 100, 5000, 0.05, 1500, check_every=20)
                update(h_new, sc)
                if remaining() > 15:
                    h_new2, sc2 = slsqp_smooth(h_new, 3000, maxiter=100)
                    if h_new2 is not None:
                        update(h_new2, sc2)

    # Phase 4: Binary seeds with polishing
    if remaining() > 150:
        n = len(best_h)
        for attempt in range(8):
            if remaining() < 25:
                break
                h_bin = binary_seed(n, rng)
                sc_bin = official_score(h_bin)
                # Local search on binary
                h_bs, sc_bs = binary_local_search(h_bin, n_swaps=800, rng=rng)
                # Interpolate to continuous (already continuous since binary)
                # Polish with Adam
                h_new, sc = adam_optimize(h_bs, 50, 3000, 0.08, 1000, check_every=30)
                update(h_new, sc)
                # SLSQP polish
                if remaining() > 20:
                    h_new2, sc2 = slsqp_smooth(h_new, 1000, maxiter=100)
                    if h_new2 is not None:
                        update(h_new2, sc2)

    # Phase 5: Reversed complement + refine
    if remaining() > 40:
        h_sym = project_feasible(1.0 - best_h[::-1].copy())
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        if remaining() > 20:
            h_new, sc = adam_optimize(h_sym, 100, 8000, 0.03, 1000, check_every=20)
            update(h_new, sc)

    # Phase 6: Random perturbations + optimize
    if remaining() > 80:
        n = len(best_h)
        for i in range(12):
            if remaining() < 20:
                break
            sigma = 0.03 + 0.08 * rng.random()
            h_cand = project_feasible(best_h + rng.normal(0, sigma, size=n))
            h_new, sc = adam_optimize(h_cand, 80, 4000, 0.05, 800, check_every=25)
            update(h_new, sc)
            if remaining() > 15:
                h_new2, sc2 = slsqp_smooth(h_new, 2000, maxiter=80)
                if h_new2 is not None:
                    update(h_new2, sc2)

    # Phase 7: Final intensive polish
    if remaining() > 30:
        for lr_try in [0.02, 0.01, 0.005]:
            if remaining() < 10:
                break
            h_new, sc = adam_optimize(best_h, 1000, 20000, lr_try, 2000, check_every=30)
            update(h_new, sc)
        if remaining() > 15:
            h_new, sc = slsqp_smooth(best_h, 5000, maxiter=150)
            if h_new is not None:
                update(h_new, sc)

    # Final: try a few more n values around best
    if remaining() > 30:
        n_best = len(best_h)
        for dn in [-3, -2, -1, 1, 2, 3]:
            if remaining() < 10:
                break
            n_try = n_best + dn
            if n_try < 20 or n_try > 100:
                continue
            h_int = interpolate_to(best_h, n_try)
            h_new, sc = adam_optimize(h_int, 200, 10000, 0.03, 500, check_every=20)
            update(h_new, sc)

    final_h = project_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
