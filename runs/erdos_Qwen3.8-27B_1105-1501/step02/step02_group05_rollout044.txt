def two_point_search(h, n_tries, rng, is_dead):

        if is_dead():

        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        best_t = None
        best_local = best_sc
        for frac in [0.0, 0.25, 0.5, 0.75, 1.0, 0.382, 0.618]:
            t = t_lo + frac * span
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            tmp = best.copy()
            tmp[a] = val_a
            tmp[b] = val_b
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = best[a] + best_t
            best[b] = best[b] - best_t
            best[a] = np.clip(best[a], 0.0, 1.0)
            best[b] = np.clip(best[b], 0.0, 1.0)

    bits = (h0 >= thr).astype(float)
    if bits.sum() != target:
        bits = (h0 > 0.5).astype(float)
    if bits.sum() != target:
        idx_sorted = np.argsort(h0)[::-1]
        bits = np.zeros(n)
        bits[idx_sorted[:target]] = 1.0
    current = bits.copy()
    current_sc = score_exact(current)
    best_bits = bits.copy()
    best_sc = current_sc
    T = T_start
    rng_bin = np.random.default_rng(seed_val)
    ones = np.where(bits == 1)[0]
    zeros = np.where(bits == 0)[0]
    for it in range(n_steps):
        if is_dead and is_dead():
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng_bin.integers(0, len(ones))]
        j = zeros[rng_bin.integers(0, len(zeros))]
        bits[i] = 0.0
        bits[j] = 1.0
        ones = np.where(bits == 1)[0]
        zeros = np.where(bits == 0)[0]
        new_sc = score_exact(bits)
        delta = new_sc - current_sc
        if delta < -1e-12 or (T > 1e-8 and rng_bin.random() < np.exp(-delta / T)):
            current = bits.copy()
            current_sc = new_sc
            if new_sc < best_sc - 1e-12:
                best_sc = new_sc
                best_bits = bits.copy()
        else:
            bits[j] = 0.0
            bits[i] = 1.0
            ones = np.where(bits == 1)[0]
            zeros = np.where(bits == 0)[0]
        T *= cool
    return best_bits

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.07 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.35, 3, 0.12)
        best_score = score_exact(best_h)

    n = len(best_h)
    grad_ok = verify_gradient(best_h)
    use_fd = not grad_ok

    # Phase 1: Quick LP warmup on parent
    if not is_dead():
        h_lp, sc_lp = lp_refine(best_h, max_iters=200, is_dead=is_dead, use_fd=use_fd)
        if sc_lp < best_score - 1e-10:
            best_h = h_lp
            best_score = sc_lp

    # Phase 2: 2-point exchange from current
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 2))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 3: LP again on polished
    if not is_dead():
        h_lp, sc_lp = lp_refine(best_h, max_iters=300, is_dead=is_dead, use_fd=use_fd)
        if sc_lp < best_score - 1e-10:
            best_h = h_lp
            best_score = sc_lp

    # Phase 4: Anti-symmetric exploration
    if not is_dead():
        h_sym = anti_sym_map(best_h)
        sc_sym = score_exact(h_sym)
        if sc_sym < best_score - 1e-10:
            best_h = h_sym
            best_score = sc_sym
        if not is_dead():
            h_lp2, sc_lp2 = lp_refine(h_sym, max_iters=150, is_dead=is_dead, use_fd=use_fd)
            if sc_lp2 < best_score - 1e-10:
                best_h = h_lp2
                best_score = sc_lp2

    # Phase 5: Binary search for alternative basins
    if not is_dead():
        for sd in [0, 1, 2, 3]:
            if is_dead():
                break
            bb = binary_sa(best_h, n_steps=2000, seed_val=seed + sd, is_dead=is_dead)
            if bb is not None:
                sc_b = score_exact(bb)
                if sc_b < best_score - 1e-10:
                    best_h = bb.copy()
                    best_score = sc_b
                # LP polish from binary
                if not is_dead():
                    h_lp3, sc_lp3 = lp_refine(bb, max_iters=150, is_dead=is_dead, use_fd=use_fd)
                    if sc_lp3 < best_score - 1e-10:
                        best_h = h_lp3
                        best_score = sc_lp3

    # Phase 6: Resolution scanning
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set([n_base - 8, n_base - 4, n_base + 4, n_base + 8, 48, 56, 64, 72]))
        n_opts = [x for x in n_opts if 24 <= x <= 96]
        for n_opt in n_opts:
            if is_dead():
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-10:
                best_h = h_int
                best_score = sc_int
            if not is_dead():
                h_lp4, sc_lp4 = lp_refine(h_int, max_iters=200, is_dead=is_dead, use_fd=use_fd)
                if sc_lp4 < best_score - 1e-10:
                    best_h = h_lp4
                    best_score = sc_lp4
                if not is_dead():
                    h_2p, sc_2p = two_point_search(h_int, 1500, rng, is_dead)
                    if sc_2p < best_score - 1e-10:
                        best_h = h_2p
                        best_score = sc_2p

    # Phase 7: Intensive LP + 2-point alternating
    if not is_dead():
        for round_idx in range(4):
            if is_dead():
                break
            h_lp5, sc_lp5 = lp_refine(best_h, max_iters=250, is_dead=is_dead, use_fd=use_fd)
            if sc_lp5 < best_score - 1e-10:
                best_h = h_lp5
                best_score = sc_lp5
            if not is_dead():
                n_tries = max(1500, int((deadline - time.time()) * 1.5))
                best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 8: Random perturbation restarts
    if not is_dead():
        for sigma in [0.04, 0.08, 0.12]:
            if is_dead():
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=len(best_h)))
            h_lp6, sc_lp6 = lp_refine(h_pert, max_iters=100, is_dead=is_dead, use_fd=use_fd)
            if sc_lp6 < best_score - 1e-10:
                best_h = h_lp6
                best_score = sc_lp6

    # Phase 9: Final intensive polish
    if not is_dead():
        h_lp7, sc_lp7 = lp_refine(best_h, max_iters=400, is_dead=is_dead, use_fd=use_fd)
        if sc_lp7 < best_score - 1e-10:
            best_h = h_lp7
            best_score = sc_lp7
        if not is_dead():
            n_tries = max(3000, int((deadline - time.time()) * 3))
            best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)
        if not is_dead():
            h_lp8, sc_lp8 = lp_refine(best_h, max_iters=300, is_dead=is_dead, use_fd=use_fd)
            if sc_lp8 < best_score - 1e-10:
                best_h = h_lp8
                best_score = sc_lp8

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = make_feasible(final_h)
    final_score = score_exact(final_h)
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8

    return final_h, final_score, final_n
