def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

    """Compute gradients of all V[k] with respect to h. Returns (2n-1, n) array."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grads[k, s:n] += (1.0 - h[0:L])
            grads[k, 0:L] -= h[s:n]
        else:
            t = -s
            grads[k, 0:L] += (1.0 - h[t:n])
            grads[k, t:n] -= h[0:L]
    return grads

    """Solve LP with all constraints (not just active set)."""
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    grads = compute_all_grads(h)
    A_rows = []
    b_rows = []
    for k in range(2*n-1):
        row = np.zeros(n + 1)
        row[:n] = dx * grads[k]
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[k])
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc, d
    except Exception:
        pass
    return None, None, None

def lse_objective_and_grad(h, beta):
    """Compute log-sum-exp objective and gradient."""
    n = len(h)
    V = compute_values(h)
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= w_sum
    obj = V_max + (1.0/beta) * np.log(w_sum)
    # Gradient: sum_k w_k * grad_k
    grads = compute_all_grads(h)
    g = w @ grads  # (2n-1,) @ (2n-1, n) -> (n,)
    return obj, g

def slsqp_optimize(h_init, beta, max_iter=500, ftol=1e-12):
    """Optimize using SLSQP on log-sum-exp objective."""
    n = len(h_init)
    
    def obj_func(h):
        o, _ = lse_objective_and_grad(h, beta)
        return o
    
    def grad_func(h):
        _, g = lse_objective_and_grad(h, beta)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0,
                   'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_func, h_init, jac=grad_func,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.success and res.x is not None:
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            return h_new, sc
    except Exception:
        pass
    return h_init, official_score(h_init)
