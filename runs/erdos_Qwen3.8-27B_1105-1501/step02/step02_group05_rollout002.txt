def make_feasible(z):

        else:

import numpy as np
import time
from scipy.optimize import minimize_scalar
import cvxpy as cp

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def all_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def pair_score(h, a, b, t):
    """Score after h[a]+=t, h[b]-=t (assuming feasibility)."""
    h2 = h.copy()
    h2[a] += t
    h2[b] -= t
    return official_score(h2)

def two_point_ls(h, a, b, n_scan=16):
    """Line search for optimal t moving mass from b to a."""
    n = len(h)
    va, vb = h[a], h[b]
    t_lo = max(-va + 1e-12, vb - 1.0 - 1e-12)
    t_hi = min(1.0 - va - 1e-12, vb - 1e-12)
    if t_hi - t_lo < 1e-10:
        return 0.0, official_score(h)
    
    def f(t):
        h2 = h.copy()
        h2[a] = va + t
        h2[b] = vb - t
        return official_score(h2)
    
    # Coarse
    ts = np.linspace(t_lo, t_hi, n_scan)
    fs = np.array([f(t) for t in ts])
    bi = np.argmin(fs)
    
    # Fine
    lo = ts[max(0, bi-2)]
    hi = ts[min(n_scan-1, bi+2)]
    res = minimize_scalar(f, bounds=(lo, hi), method='bounded', options={'maxiter': 40, 'xatol': 1e-10})
    return res.x, res.fun

def sweep_pairs(h, is_dead, max_passes=3):
    """Multiple passes of systematic pair optimization with line search."""
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    
    for _pass in range(max_passes):
        if is_dead():
            break
        improved = False
        for a in range(n):
            if is_dead():
                break
            for b in range(n):
                if a == b:
                    continue
                t, sc = two_point_ls(best, a, b)
                if t is not None and abs(t) > 1e-8 and sc < best_sc - 1e-10:
                    best[a] += t
                    best[b] -= t
                    best = make_feasible(best)
                    best_sc = official_score(best)
                    improved = True
        if not improved:
            break
    return best, best_sc

def qp_step(h, mu=50.0, top_k=6, is_dead=None):
    """Trust-region QP step."""
    n = len(h)
    vals = all_vals(h)
    C = vals.max()
    npts = 2 * n - 1
    dx = 2.0 / n
    
    if top_k is not None and top_k < npts:
        sel = np.argsort(vals)[::-1][:top_k]
    else:
        sel = np.arange(npts)
    
    # Build gradient matrix
    Gmat = np.zeros((len(sel), n))
    for ii, p in enumerate(sel):
        s = int(p) - (n - 1)
        if s >= 0:
            idx = np.arange(n - s)
            Gmat[ii, idx + s] += (1.0 - h[idx]) * dx
            Gmat[ii, idx] -= h[idx + s] * dx
        else:
            t_neg = -s
            idx = np.arange(n - t_neg)
            Gmat[ii, idx + t_neg] += (1.0 - h[idx]) * dx
            Gmat[ii, idx] -= h[idx + t_neg] * dx
    
    rhs = vals[sel] - C
    
    d = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        Gmat @ d <= rhs + 1e-10,
        cp.sum(d) == 0,
        d >= -h + 1e-12,
        d <= 1.0 - h - 1e-12
    ]
    
    objective = cp.Minimize(t_var + mu * cp.sum_squares(d))
    prob = cp.Problem(objective, constraints)
    
    for solver in ['ECOS', 'SCS', 'CLARABEL', 'CVXOPT']:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate'] and d.value is not None:
                dv = np.asarray(d.value, dtype=float)
                if not np.all(np.isfinite(dv)):
                    continue
                # Line search
                best_sc = official_score(h)
                for alpha in [1.0, 0.6, 0.35, 0.2, 0.1, 0.05]:
                    trial = make_feasible(h + alpha * dv)
                    sc = official_score(trial)
                    if sc < best_sc - 1e-13:
                        return trial, sc
                break
        except Exception:
            continue
    return None, None

def make_wave(n, freqs, amps, phases=None):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for f, a in zip(freqs, amps):
        ph = 0.0
        if phases is not None:
            ph = phases[len([ff for ff in freqs if ff < f])]  # This won't work generically
        h += a * np.sin(2 * np.pi * f * x)
    return make_feasible(h)

    return make_feasible(h)
