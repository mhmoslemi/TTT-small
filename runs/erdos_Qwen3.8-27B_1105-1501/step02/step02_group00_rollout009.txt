import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad_matrix(h):
    """Build matrix G where G[k,j] = ∂s_k/∂h_j (without dx factor)
    k indexes the lag (position in correlate output), j indexes h coordinate."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)  # actual lag
        # For each j: (1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)
        # First term: j-k in [0,n-1] => j in [k, n-1+k]
        j1_lo = max(0, k)
        j1_hi = min(n, n + k)  # exclusive
        if j1_hi > j1_lo:
            G[r, j1_lo:j1_hi] += (1.0 - h[j1_lo - k : j1_hi - k])
        # Second term: j+k in [0,n-1] => j in [-k, n-1-k]
        j2_lo = max(0, -k)
        j2_hi = min(n, n - k)  # exclusive
        if j2_hi > j2_lo:
            G[r, j2_lo:j2_hi] += (-h[j2_lo + k : j2_hi + k])
    G *= dx
    return G

def _lp_minmax_direction(h, scores, G, radius):
    """Solve min t s.t. G*q + scores <= t, sum(q)=0, |q_j|<=radius
    Returns optimal q (direction to add to h) or None."""
    n = len(h)
    n_scores = len(scores)
    # Variables: q_0, ..., q_{n-1}, t  (n+1 variables)
    # Objective: minimize t => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality: G*q + scores <= t => G*q - t <= -scores
    A_ub = np.zeros((n_scores, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    # Equality: sum(q) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    # Bounds: -radius <= q_j <= radius, t free
    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            q = res.x[:n]
            t_pred = res.x[n]
            return q, t_pred
    except Exception:
        pass
    return None, None

def _line_search(h, direction, c5_cur, max_alpha=1.0):

    n = len(h)

    alpha_max = max(alpha_max, 1e-8)

        np.linspace(0, alpha_max, 20),
        np.logspace(-3, np.log10(max(alpha_max, 1e-3)), 10)
    ])
    alphas = alphas[alphas <= alpha_max + 1e-12]
    alphas = np.unique(alphas)
    
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        if a < 1e-10:
            continue
        h_t = h + a * direction

        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a

        return 0.0, c5_cur

    lo_a = max(0.0, best_a * 0.7)

def _smoothed_grad_direction(h, beta=20.0):
    """Compute smoothed gradient direction (softmax-weighted subgradient)"""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    weights = np.exp(shifted)
    total = float(np.sum(weights))
    if total < 1e-300:
        return None
    weights /= total
    
    # Compute weighted gradient
    G = _build_grad_matrix(h)
    d = G.T @ weights  # shape (n,)
    d -= np.mean(d)  # project to sum=0
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return -d / norm

def _random_direction(h, rng, scale=0.01):
    """Random feasible direction"""
    n = len(h)
    d = rng.normal(0, scale, n)
    # Only move coordinates that have room
    d[h >= 1.0 - 1e-8] = 0.0
    d[h <= 1e-8] = 0.0
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm * scale
