def _project_box_sum(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:

        else:

                val += (1.0 - h[j-k])
            if 0 <= j+k < n:
                val -= h[j+k]
            G[r, j] = val
    return scores, G

# Actually, the full G matrix is O(n^2 * n) = O(n^3) to compute and O(n^2) space.
# For n=64, that's 64^2 * 127 ≈ 524K operations. Fine.
# But for the subspace LP, I only need G[:, K] which is O(k*n).

def _subspace_lp_step(h, K, radius=0.1):
    """One subspace LP step on coordinates K."""
    n = len(h)
    k = len(K)
    if k == 0:
        return None
    
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute G[:, K]
    G_K = np.zeros((2*n-1, k))
    for idx, j in enumerate(K):
        for r in range(2*n-1):
            shift = r - (n-1)
            val = 0.0
            if 0 <= j - shift < n:
                val += (1.0 - h[j - shift])
            if 0 <= j + shift < n:
                val -= h[j + shift]
            G_K[r, idx] = val
    
    # LP: min t s.t. G_K @ delta - t <= -scores, sum(delta)=0, bounds
    c = np.zeros(k + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, k+1))
    A_ub[:, :k] = G_K
    A_ub[:, k] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, k+1))
    A_eq[0, :k] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for idx, j in enumerate(K):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success and res.x[-1] < np.max(scores) - 1e-10:
            return res.x[:k]
    except Exception:
        pass
    return None

def _pair_1d_optimize(h, j, k, radius=0.1):
    """Optimize over 2 coordinates (1D problem since d2=-d1)."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute G[:, j] - G[:, k]
    diff = np.zeros(2*n-1)
    for r in range(2*n-1):
        shift = r - (n-1)
        # G[r,j]
        gj = 0.0
        if 0 <= j - shift < n:
            gj += (1.0 - h[j - shift])
        if 0 <= j + shift < n:
            gj -= h[j + shift]
        # G[r,k]
        gk = 0.0
        if 0 <= k - shift < n:
            gk += (1.0 - h[k - shift])
        if 0 <= k + shift < n:
            gk -= h[k + shift]
        diff[r] = gj - gk
    
    # Find d that minimizes max_r (scores[r] + diff[r]*d)
    # subject to: -radius <= d <= radius, -h[j] <= d <= 1-h[j], -h[k] <= -d <= 1-h[k]
    d_lo = max(-radius, -h[j], h[k] - 1)  # d >= -h[j], -d >= -h[k] => d <= h[k]... wait
    # d2 = -d, so: -h[k] <= -d <= 1-h[k] => h[k]-1 <= d <= h[k]
    # Also: -h[j] <= d <= 1-h[j]
    # And: -radius <= d <= radius
    d_lo = max(-radius, -h[j], h[k] - 1.0)
    d_hi = min(radius, 1.0 - h[j], h[k])
    
    if d_lo >= d_hi:
        return 0.0
    
    # Bisection/ternary search for min of f(d) = max_r (scores[r] + diff[r]*d)
    # This is convex in d, so ternary search works.
    lo, hi = d_lo, d_hi
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(scores + diff * m1)
        f2 = np.max(scores + diff * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _lse_obj_and_grad(h, beta):
    """Compute LSE objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = np.max(scores)
    obj = max_s + np.log(np.sum(np.exp(beta * (scores - max_s)))) / beta
    
    weights = np.exp(beta * (scores - max_s))
    wsum = np.sum(weights)
    w = weights / wsum
    
    grad = np.zeros(n)
    for r in range(2*n-1):
        if w[r] < 1e-14:
            continue
        shift = r - (n-1)
        if shift >= 0:
            m = n - shift
            if m > 0:
                grad[shift:shift+m] += w[r] * dx * (1.0 - h[:m])
                grad[:m] += w[r] * dx * (-h[shift:shift+m])
        else:
            g = -shift
            m = n - g
            if m > 0:
                grad[:m] += w[r] * dx * (1.0 - h[g:g+m])
                grad[g:g+m] += w[r] * dx * (-h[:m])
    return obj, grad
