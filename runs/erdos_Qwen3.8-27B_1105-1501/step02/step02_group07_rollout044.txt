def full_lp(h, trust):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    G = all_gradients(h, n)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    dG = dx * G
    rhs = dG @ h - V
    
    A = np.hstack([dG, -np.ones((2 * n - 1, 1))])
    b = -rhs
    
    bounds = [(max(h[i] - trust, 0.0), min(h[i] + trust, 1.0)) for i in range(n)]
    bounds.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=[0.0],
                     bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = make_feasible(res.x[:n])
            return h_new
    except Exception:
        pass
    return None

def pg_step(h, beta, alpha_max=0.5, max_ls=14):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    
    g = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            g[s:n] += wi * (1.0 - h[0:L])
            g[0:L] -= wi * h[s:n]
        else:
            t = -s
            g[0:L] += wi * (1.0 - h[t:n])
            g[t:n] -= wi * h[0:L]
    
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-7:
            break
    return best_h, best_f

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.03 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)
    
    def time_left():
        return deadline - time.time()
    
    try:
        h = np.asarray(initial_h_values, dtype=float).ravel()
        h = make_feasible(h)
    except (NameError, AttributeError):
        n = 60
        x = np.arange(n) / n
        h = 0.5 + 0.2 * np.sin(np.pi * x) + 0.08 * np.sin(2 * np.pi * x)
        h = make_feasible(h)
    
    best_h = h.copy()
    best_score = official_score(h)
    
    # Phase 1: Iterative full LP (primary optimizer)
    h_cur = h.copy()
    trust = 0.3
    no_improve = 0
    for iteration in range(500):
        if time_left() < 350:
            break
        h_new = full_lp(h_cur, trust)
        if h_new is not None:
            sc = official_score(h_new)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
                h_cur = h_new
                trust = min(trust * 1.1, 0.5)
                no_improve = 0
            else:
                h_cur = h_new
                if sc > best_score + 1e-4:
                    h_cur = best_h.copy()
                trust = max(trust * 0.7, 0.02)
                no_improve += 1
            if no_improve > 60:
                break
        else:
            trust = max(trust * 0.7, 0.02)
            no_improve += 1
            if no_improve > 30:
                break
    
    # Phase 2: PG polish
    h_cur = best_h.copy()
    beta = 30.0
    stall = 0
    while time_left() > 280:
        h_new, sc = pg_step(h_cur, beta, alpha_max=0.5, max_ls=14)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
            stall = 0
            h_cur = h_new
        else:
            stall += 1
            h_cur = h_new
        beta = min(beta * 1.03, 300.0)
        if stall > 40:
            break
    
    # Phase 3: LP again with finer trust
    h_cur = best_h.copy()
    trust = 0.15
    no_improve = 0
    for iteration in range(300):
        if time_left() < 150:
            break
        h_new = full_lp(h_cur, trust)
        if h_new is not None:
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                h_cur = h_new
                trust = min(trust * 1.15, 0.3)
                no_improve = 0
            else:
                h_cur = h_new
                trust = max(trust * 0.75, 0.01)
                no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    # Phase 4: Try different n values
    if time_left() > 100:
        n_base = len(best_h)
        n_candidates = sorted(set([max(40, n_base - 12), max(40, n_base - 6),
                                     min(80, n_base + 6), min(100, n_base + 12)]))
        for n_try in n_candidates:
            if time_left() < 50:
                break
            if n_try == n_base:
                continue
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            h_cur = h_int.copy()
            
            trust = 0.25
            local_best = sc_int
            local_best_h = h_int.copy()
            no_improve = 0
            for it in range(200):
                if time_left() < 50:
                    break
                h_new = full_lp(h_cur, trust)
                if h_new is not None:
                    sc = official_score(h_new)
                    if sc < local_best - 1e-13:
                        local_best = sc
                        local_best_h = h_new.copy()
                        h_cur = h_new
                        trust = min(trust * 1.12, 0.45)
                        no_improve = 0
                    else:
                        h_cur = h_new
                        trust = max(trust * 0.7, 0.03)
                        no_improve += 1
                    if no_improve > 40:
                        break
                else:
                    no_improve += 1
                    if no_improve > 20:
                        break
            
            # PG polish at this n
            h_cur = local_best_h.copy()
            beta = 50.0
            for it in range(100):
                if time_left() < 50:
                    break
                h_new, sc = pg_step(h_cur, beta, alpha_max=0.4, max_ls=12)
                if sc < local_best - 1e-13:
                    local_best = sc
                    local_best_h = h_new.copy()
                h_cur = h_new
                beta = min(beta * 1.02, 200.0)
            
            if local_best < best_score - 1e-12:
                best_h = local_best_h.copy()
                best_score = local_best
    
    # Phase 5: Final multi-start + LP
    if time_left() > 60:
        for restart in range(5):
            if time_left() < 30:
                break
            noise_scale = 0.05 * (restart + 1) * 0.5
            h_try = make_feasible(best_h + rng.normal(0, noise_scale, len(best_h)))
            sc_try = official_score(h_try)
            if sc_try > best_score + 0.01:
                continue
            h_cur = h_try.copy()
            local_best = sc_try
            local_best_h = h_try.copy()
            trust = 0.2
            for it in range(100):
                if time_left() < 30:
                    break
                h_new = full_lp(h_cur, trust)
                if h_new is not None:
                    sc = official_score(h_new)
                    if sc < local_best - 1e-13:
                        local_best = sc
                        local_best_h = h_new.copy()
                        h_cur = h_new
                        trust = min(trust * 1.1, 0.4)
                    else:
                        h_cur = h_new
                        trust = max(trust * 0.7, 0.03)
                    if local_best < best_score - 1e-12:
                        best_h = local_best_h.copy()
                        best_score = local_best
    
    # Phase 6: Final fine LP + PG
    h_cur = best_h.copy()
    trust = 0.08
    for _ in range(100):
        if time_left() < 15:
            break
        h_new = full_lp(h_cur, trust)
        if h_new is not None:
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
    
    if time_left() > 10:
        h_cur = best_h.copy()
        beta = 200.0
        for _ in range(200):
            if time_left() < 10:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.2, max_ls=10)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.01, 500.0)
    
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
