from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f_raw(h):
    """Compute raw correlate values (without dx factor)."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_grads(h):
    """Compute gradients of all f_s components. Returns (2n-1, n) array."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for s_offset in range(2*n-1):
        s = s_offset - (n-1)
        g = np.zeros(n)
        if s == 0:
            g[:] = 1.0 - 2.0*h
        elif s > 0:
            for i in range(s, n):
                g[i] += (1.0 - h[i-s])
            for i in range(0, n-s):
                g[i] -= h[i+s]
        else:
            t = -s
            for i in range(0, n-t):
                g[i] += (1.0 - h[i+t])
            for i in range(t, n):
                g[i] -= h[i-t]
        grads[s_offset] = g
    return grads

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    # Bisection on shift to satisfy sum constraint
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)

def solve_lp(h, f_vals, grads, active_indices):
    """Solve the LP relaxation. Returns h_new or None."""
    n = len(h)
    K = len(active_indices)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.empty((K, n + 1))
    b_ub = np.empty(K)
    for i, idx in enumerate(active_indices):
        g = grads[idx]
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g.dot(h) - f_vals[idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:

            return res.x[:n]
    except Exception:
        pass
    return None
