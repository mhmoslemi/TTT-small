from scipy.optimize import linprog
import time

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Bisection to find shift that makes sum = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add > 1e-15:
                    h[idx] += add
                    residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub > 1e-15:
                    h[idx] -= sub
                    residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_gradients(h, active_k, n):
    """Compute gradients of correlation values at selected lags."""
    dx = 2.0 / n
    grads = np.zeros((len(active_k), n))
    for idx, k in enumerate(active_k):
        s = k - (n - 1)
        g = np.zeros(n)

            g[idx_arr] += (1.0 - h[idx_arr + t])
            g[idx_arr + t] -= h[idx_arr]
        grads[idx] = g * dx
    return grads

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()

    for _ in range(80):

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add > 1e-15:
                    h[idx] += add
                    residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub > 1e-15:
                    h[idx] -= sub
                    residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def binary_sa(h, n_steps, rng, is_dead):
    n = len(h)
    if n % 2 != 0:
        return h, official_score(h)
    target = n // 2
    bits = (h > 0.5).astype(float)
    excess = bits.sum() - target
    if excess > 0:
        idx = np.where(bits == 1)[0]
        order = np.argsort(h[idx])
        for i in range(int(excess)):
            bits[idx[order[i]]] = 0
    elif excess < 0:
        idx = np.where(bits == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(int(-excess)):
            bits[idx[order[i]]] = 1
    best_bits = bits.copy()
    best_sc = official_score(best_bits)
    cur_bits = bits.copy()
    cur_sc = best_sc
    T = 5e-4
    pos1 = np.where(cur_bits == 1)[0]
    pos0 = np.where(cur_bits == 0)[0]
    for step in range(n_steps):
        if is_dead():
            break
        i1 = int(rng.integers(0, len(pos1)))
        i0 = int(rng.integers(0, len(pos0)))
        a = pos1[i1]
        b = pos0[i0]
        cur_bits[a] = 0
        cur_bits[b] = 1
        cur_sc = official_score(cur_bits)
        delta = cur_sc - best_sc

            if cur_sc < best_sc:
                best_bits = cur_bits.copy()
                best_sc = cur_sc
            pos1 = np.where(cur_bits == 1)[0]

        else:
            cur_bits[a] = 1
            cur_bits[b] = 0
        T *= 0.9992

            h_new, sc_new = two_point_polish(h_sym, n_tries, rng, is_dead)
            if sc_new < best_score - 1e-10:
                best_h = h_new
                best_score = sc_new
            if not is_dead():
                remaining2 = deadline - time.time()
                if remaining2 > 30:
                    h_lp, sc_lp = lp_polish(h_new, time.time(), min(deadline, time.time() + remaining2 * 0.5),
                                            init_rho=0.05, max_iters=200)
                    if sc_lp < best_score - 1e-10:
                        best_h = h_lp
                        best_score = sc_lp

    # Phase 5: Binary SA for diverse starts
    if not is_dead():
        h_bin, sc_bin = binary_sa(best_h, 2000, rng, is_dead)
        if sc_bin < best_score - 1e-10:
            best_h = h_bin
            best_score = sc_bin
        # LP from binary
        if not is_dead():
            remaining = deadline - time.time()
            if remaining > 30:
                h_lp, sc_lp = lp_polish(h_bin, time.time(), min(deadline, time.time() + remaining * 0.5),
                                        init_rho=0.08, max_iters=200)
                if sc_lp < best_score - 1e-10:
                    best_h = h_lp
                    best_score = sc_lp

    # Phase 6: Multi-resolution
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set([n_base + d for d in [-8, -4, 4, 8, 12] if 24 <= n_base + d <= 100]))
        for nc in [40, 48, 56, 64, 72, 80]:
            if 24 <= nc <= 100 and nc not in n_opts:
                n_opts.append(nc)
        n_opts = sorted(set(n_opts))
        for n_opt in n_opts:
            if is_dead():
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-10:
                best_h = h_int
                best_score = sc_int
            if not is_dead():
                remaining = deadline - time.time()
                if remaining > 20:
                    t_p = min(deadline, time.time() + remaining * 0.4)
                    h_lp, sc_lp = lp_polish(h_int, time.time(), t_p, init_rho=0.06, max_iters=100)
                    if sc_lp < best_score - 1e-10:
                        best_h = h_lp
                        best_score = sc_lp

    # Phase 7: Final intensive LP + two-point on best
    if not is_dead():
        remaining = deadline - time.time()
        if remaining > 20:
            h_lp, sc_lp = lp_polish(best_h, time.time(), min(deadline, time.time() + remaining * 0.5),
                                    init_rho=0.04, max_iters=200)
            if sc_lp < best_score - 1e-10:
                best_h = h_lp
                best_score = sc_lp
    if not is_dead():
        remaining = deadline - time.time()
        n_tries = max(1000, int(remaining * 4))
        best_h, best_score = two_point_polish(best_h, n_tries, rng, is_dead)

    # Phase 8: Final LP polish
    if not is_dead():
        remaining = deadline - time.time()
        if remaining > 15:
            h_lp, sc_lp = lp_polish(best_h, time.time(), min(deadline, time.time() + remaining * 0.6),
                                    init_rho=0.03, max_iters=150, )
            if sc_lp < best_score - 1e-10:
                best_h = h_lp
                best_score = sc_lp

    # Phase 9: Random perturbations
    if not is_dead():
        for sigma in [0.03, 0.06, 0.1]:
            if is_dead():
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=len(best_h)))
            if not is_dead():
                remaining = deadline - time.time()
                if remaining > 10:
                    h_lp, sc_lp = lp_polish(h_pert, time.time(), min(deadline, time.time() + remaining * 0.4),
                                            init_rho=0.05, max_iters=100)
                    if sc_lp < best_score - 1e-10:
                        best_h = h_lp
                        best_score = sc_lp

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    if not np.isfinite(final_score) or final_score > 0.6:
        final_h = make_feasible(np.full(60, 0.5))
        final_score = official_score(final_h)
        final_n = 60

    assert np.all(np.isfinite(final_h))
    assert np.all(final_h >= -1e-10)
    assert np.all(final_h <= 1.0 + 1e-10)
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8
    assert np.isfinite(final_score)

    return final_h, final_score, final_n
