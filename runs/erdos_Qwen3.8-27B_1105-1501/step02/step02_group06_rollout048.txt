import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_raw(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    np.clip(h, 0.0, 1.0, out=h)
    return h

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def solve_1d_envelope(a, b):
    """Solve min_{x in [0,1]} max_j (a_j*x + b_j). Returns optimal x."""
    n_lines = len(a)
    if n_lines == 0:
        return 0.5
    if n_lines == 1:
        if a[0] > 0:
            return 0.0
        elif a[0] < 0:
            return 1.0
        else:
            return 0.5
    f0 = np.max(b)
    f1 = np.max(a + b)
    best_val = min(f0, f1)
    best_x = 0.0 if f0 <= f1 else 1.0
    pos_mask = a > 1e-12
    neg_mask = a < -1e-12
    zero_mask = (a >= -1e-12) & (a <= 1e-12)
    if np.any(zero_mask):
        zv = np.max(b[zero_mask])
        if zv < best_val:
            best_val = zv
            best_x = 0.5
    if np.any(pos_mask) and np.any(neg_mask):
        ap = a[pos_mask]
        bp = b[pos_mask]
        an = a[neg_mask]
        bn = b[neg_mask]
        denom = ap[:, None] - an[None, :]
        valid = np.abs(denom) > 1e-14
        if np.any(valid):
            x_star = np.where(valid, (bn[None, :] - bp[:, None]) / np.where(valid, denom, 1.0), 2.0)
            valid_x = valid & (x_star >= 0.0) & (x_star <= 1.0)
            if np.any(valid_x):
                xs = x_star[valid_x]
                vals = np.max(ap[:, None] * xs[None, :] + bp[:, None], axis=0)
                idx_best = np.argmin(vals)
                if vals[idx_best] < best_val - 1e-15:
                    best_val = vals[idx_best]
                    best_x = xs[idx_best]
    return float(np.clip(best_x, 0.0, 1.0))

def coordinate_sweep(h, lam, n_sweeps=5):
    """Sweep all coordinates minimizing max_j raw_j + lam*h[k] for each k."""
    h = h.copy()
    n = len(h)
    s_offsets = np.arange(2 * n - 1) - (n - 1)
    for _ in range(n_sweeps):
        raw = compute_raw(h)
        for k in range(n):
            coeff = np.zeros(2 * n - 1)
            ks = k + s_offsets
            mask1 = (ks >= 0) & (ks < n)
            coeff[mask1] += (1.0 - h[ks[mask1].astype(int)])
            km = k - s_offsets
            mask2 = (km >= 0) & (km < n)
            coeff[mask2] -= h[km[mask2].astype(int)]
            d = raw - coeff * h[k]
            x_opt = solve_1d_envelope(coeff + lam, d)
            h[k] = x_opt
    return h

def coordinate_sweep_bisect(h0, n_sweeps_inner=4, n_outer=15, tol=1e-10):
    """Bisect on lambda to satisfy sum(h)=n/2."""
    n = len(h0)
    target = n / 2.0
    lam = 0.0
    for _ in range(n_outer):
        h_new = coordinate_sweep(h0, lam, n_sweeps_inner)
        residual = np.sum(h_new) - target
        if abs(residual) < tol:
            break
        step = 0.1 * n
        if residual > 0:
            lam += step * residual / target
        else:
            lam += step * residual / target
        lam = np.clip(lam, -50.0, 50.0)
        h0 = h_new
    return h0, lam

def pair_coordinate_sweep(h, n_sweeps=3, rng=None):
    """Update pairs of coordinates maintaining sum constraint."""
    h = h.copy()
    n = len(h)
    s_offsets = np.arange(2 * n - 1) - (n - 1)
    for _ in range(n_sweeps):
        raw = compute_raw(h)
        for k in range(n):
            for m in range(k + 1, n):
                const = h[k] + h[m]
                u_lo = max(-const, const - 2.0)
                u_hi = min(2.0 - const, const)
                if u_hi - u_lo < 1e-12:
                    continue
                a_u = np.zeros(2 * n - 1)
                b_u = np.zeros(2 * n - 1)
                for j in range(2 * n - 1):
                    s = s_offsets[j]
                    ak = 0.0
                    bk = 0.0
                    am = 0.0
                    bm = 0.0
                    ks = k + s
                    if 0 <= ks < n and ks != m:
                        ak += (1.0 - h[ks]) * 0.5
                    km = k - s
                    if 0 <= km < n and km != m:
                        ak -= h[km] * 0.5
                    ms = m + s
                    if 0 <= ms < n and ms != k:
                        am += (1.0 - h[ms]) * (-0.5)
                    mm = m - s
                    if 0 <= mm < n and mm != k:
                        am -= h[mm] * (-0.5)
                    ak_part = (ak + am)
                    bk_part = raw[j] - (ak + am) * ((h[k] + h[m]) / 2.0)
                    a_u[j] = ak_part
                    b_u[j] = bk_part
                u_mid = (u_lo + u_hi) / 2.0
                u_range = u_hi - u_lo
                if u_range < 1e-12:
                    continue
                t_lo = (u_lo - u_mid) / u_range * 2.0
                t_hi = (u_hi - u_mid) / u_range * 2.0
                a_t = a_u * (u_range / 2.0)
                b_t = b_u + a_u * u_mid
                t_opt = solve_1d_envelope(a_t, b_t)
                u_opt = u_mid + t_opt * (u_range / 2.0)
                u_opt = np.clip(u_opt, u_lo, u_hi)
                h[k] = (const + u_opt) / 2.0
                h[m] = (const - u_opt) / 2.0
    return h

def gradient_p(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for k in range(n):
            if k >= s:
                g[k] += 1.0 - h[k - s]
            if k < n - s:
                g[k] -= h[k + s]
    else:
        t = -s
        for k in range(n):
            if k < n - t:
                g[k] += 1.0 - h[k + t]
            if k >= t:
                g[k] -= h[k - t]
    return g

def qp_step(h0, k_active, reg, budget=30):
    n = len(h0)
    raw = compute_raw(h0)
    dx = 2.0 / n
    vals = raw * dx
    top_idx = np.argsort(vals)[-k_active:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = dx * gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)
    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    status = None
    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                status = prob.status
                break
        except Exception:
            continue
    if status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)

def active_set_optimize(h0, max_iter=25, k_active=5, reg_start=3.0, reg_end=0.005,
                        deadline=None):
    h = h0.copy()
    best = official_score(h)
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline - 5:
            break
        reg = reg_start * (reg_end / reg_start) ** (it / max(1, max_iter - 1))
        h_new = qp_step(h, k_active=k_active, reg=reg)
        if h_new is None:
            break
        sc = official_score(h_new)
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.7, 0.4, 0.2, 0.1]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-14:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 6:
                    break
    return h, best

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))

def try_slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def flip_coordinates(h, rng, n_flips=20):
    """Try flipping coordinates to 0 or 1, maintaining sum constraint."""
    h = h.copy()
    n = len(h)
    target = n / 2.0
    best = official_score(h)
    for _ in range(n_flips):
        k = rng.integers(0, n)
        if h[k] < 0.05 or h[k] > 0.95:
            continue
        delta = 1.0 - h[k] if h[k] < 0.5 else -h[k]
        m = rng.integers(0, n)
        if m == k:
            continue
        if delta > 0:
            if h[m] < delta:
                continue
        else:
            if 1.0 - h[m] < -delta:
                continue
        h2 = h.copy()
        h2[k] = h[k] + delta
        h2[m] = h[m] - delta
        if h2[k] < -1e-10 or h2[k] > 1 + 1e-10 or h2[m] < -1e-10 or h2[m] > 1 + 1e-10:
            continue
        sc2 = official_score(h2)
        if sc2 < best - 1e-12:
            h = h2
            best = sc2
    return h, best

                                                reg_start=1.0, reg_end=0.01, deadline=deadline)
            update(h_opt, sc_opt)

    # Phase 9: Final intensive polish
    if remaining() > 30:
        for k_a in [3, 5, 8]:
            if remaining() < 10:
                break
            h_opt, sc_opt = active_set_optimize(best_h, max_iter=15, k_active=k_a,
                                                reg_start=0.5, reg_end=0.001, deadline=deadline)
            update(h_opt, sc_opt)
        h_sw = coordinate_sweep_bisect(best_h, n_sweeps_inner=5, n_outer=20, tol=1e-12)
        h_sw = make_feasible(h_sw)
        sc = official_score(h_sw)
        update(h_sw, sc)

    # Phase 10: Anti-symmetric transform
    if remaining() > 20:
        h_sym = make_feasible(1.0 - best_h[::-1].copy())
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        h_opt, sc_opt = active_set_optimize(h_sym, max_iter=10, k_active=5,
                                            reg_start=1.0, reg_end=0.01, deadline=deadline)
        update(h_opt, sc_opt)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = final_score + 1e-13
    return final_h, final_bound, len(final_h)
