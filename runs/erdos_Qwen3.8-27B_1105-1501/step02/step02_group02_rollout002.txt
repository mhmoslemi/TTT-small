    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for p in range(j0, j1):
            g[p] += wk * (1.0 - x[p - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for p in range(p0, p1):
            g[p] += wk * (-x[p + s])
    return g

def _slsqp_lse(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_active(h0, M=12, maxiter=120):
    n = len(h0)
    cv = _corr_vec(h0)
    active_idx = np.argsort(cv)[::-1][:M]
    t0 = cv[active_idx].max() + 0.002
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _afun(z):
        return z[-1]

    def _ajac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _aieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv[active_idx]

    def _aeq(z):
        return z[:-1].sum() - n / 2.0

    def _aeqjac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_afun, z0, jac=_ajac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': _aieq},
                                    {'type': 'eq', 'fun': _aeq, 'jac': _aeqjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _fista(h0, beta, n_iter, alpha_init=0.005, mu=0.9):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1.0
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        h_new = _project(y - alpha_init * grad + mu * (h_cur - h_prev))
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = _project(h_new + ((t - 1.0) / t_new) * (h_new - h_cur))
        t = t_new
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_cur - h_avg) / (it + 1)
        if it % 200 == 0 or it == n_iter - 1:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 500 == 0:
            sa = _score(_project(h_avg))
            if sa < best_s:
                best_s = sa
                best_h = _project(h_avg).copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s

def _ interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_anneal(n, rng, n_steps, T0=0.005):
    target = n // 2
    # Start with random balanced binary
    ones = np.zeros(n, dtype=int)
    pos = rng.choice(n, size=target, replace=False)
    ones[pos] = 1
    b = ones.astype(float)
    cur_score = _score(b)
    best_b = b.copy()
    best_score = cur_score
    T = T0
    cooling = 0.9995
    for _ in range(n_steps):
        # Pick one 1 and one 0 to swap
        one_idx = np.where(b == 1)[0]
        zero_idx = np.where(b == 0)[0]
        i = one_idx[rng.integers(len(one_idx))]
        j = zero_idx[rng.integers(len(zero_idx))]
        b[i] = 0
        b[j] = 1
        new_score = _score(b)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-abs(delta) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
        T *= cooling
    return best_b, best_score

def _make_binary(n, rng, kind='random'):
    target = n // 2
    b = np.zeros(n)
    if kind == 'random':
        pos = rng.choice(n, size=target, replace=False)
        b[pos] = 1
    elif kind == 'alternating':
        b[::2] = 1
        b[:target] = 1  # ensure exact count
        # fix: take first n/2 of the alternating
        b = np.zeros(n)
        b[np.arange(0, n, 2)[:target]] = 1
        if b.sum() != target:
            # fill remaining from odd positions
            odd = np.arange(1, n, 2)
            for j in odd:
                if b.sum() >= target:
                    break
                b[j] = 1
    elif kind == 'block':
        b[:target] = 1
    elif kind == 'golden':
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        pos = np.round(np.arange(target) * phi) % n
        b[np.unique(pos)] = 1
        if b.sum() != target:
            # adjust
            zeros = np.where(b == 0)[0]
            ones = np.where(b == 1)[0]
            if b.sum() < target:
                b[zeros[0]] = 1
            elif b.sum() > target:
                b[ones[-1]] = 0
    return b

def _active_cycle(h0, n_cycles=4, M=12, maxiter=100, deadline=None):
    cur = h0.copy()
    cur_s = _score(cur)
    for c in range(n_cycles):
        if deadline is not None and time.time() > deadline - 10:
            break
        new = _slsqp_active(cur, M=M, maxiter=maxiter)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns
        else:
            break
    return cur, cur_s

def _optimize_chain(h0, deadline):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)
    # LSE SLSQP with increasing beta
    for beta in [30, 100, 300, 800]:
        if time.time() > deadline - 15:
            break
        new = _slsqp_lse(cur, beta=beta, maxiter=120)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns
    # Active set cycling
    if time.time() < deadline - 15:
        cur, cur_s = _active_cycle(cur, n_cycles=3, M=14, maxiter=100, deadline=deadline)
    # FISTA polish
    if time.time() < deadline - 10:
        cur, cur_s = _fista(cur, beta=500, n_iter=4000, alpha_init=0.004, mu=0.88)
    # Final LSE
    if time.time() < deadline - 8:
        new = _slsqp_lse(cur, beta=1500, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns
    return cur, cur_s
