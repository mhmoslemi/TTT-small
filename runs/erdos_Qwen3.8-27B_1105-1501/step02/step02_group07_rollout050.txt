    target = n / 2.0

    lam = 0.0
    best_h = h.copy()

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]

        best_h = h_cd
        best_score = sc_cd

    # Phase 3: LP polishing
    h_cur = best_h.copy()
    if time_left() > 150:
        h_pol, sc_pol = full_lp_polish(h_cur, n_iters=50, trust_init=0.3)
        if sc_pol < best_score:
            best_h = h_pol
            best_score = sc_pol

    # Phase 4: More coordinate descent from LP result
    if time_left() > 120:
        h_cd2 = coord_descent(best_h, max_passes=60, tol=1e-12)
        sc_cd2 = official_score(h_cd2)
        if sc_cd2 < best_score:
            best_h = h_cd2
            best_score = sc_cd2

    # Phase 5: Another round of PG with high beta
    if time_left() > 100:
        h_cur = best_h.copy()
        beta = 100.0
        for _ in range(100):
            if time_left() < 60:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.4, max_ls=12)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.02, 500.0)

    # Phase 6: LP polish again
    if time_left() > 80:
        h_pol2, sc_pol2 = full_lp_polish(best_h, n_iters=40, trust_init=0.15)
        if sc_pol2 < best_score:
            best_h = h_pol2
            best_score = sc_pol2

    # Phase 7: Try perturbation and re-optimize
    if time_left() > 80:
        for scale in [0.05, 0.03, 0.02]:
            if time_left() < 40:
                break
            h_pert = make_feasible(best_h + rng.normal(0, scale, len(best_h)))
            sc_pert = official_score(h_pert)
            if sc_pert < best_score + 0.001:
                h_cur = h_pert.copy()
                beta = 50.0
                for _ in range(80):
                    if time_left() < 40:
                        break
                    h_new, sc = pg_step(h_cur, beta, alpha_max=0.5, max_ls=10)
                    if sc < best_score - 1e-12:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new
                    beta = min(beta * 1.02, 300.0)
                h_pol3, sc_pol3 = full_lp_polish(h_cur, n_iters=10, trust_init=0.1)
                if sc_pol3 < best_score:
                    best_h = h_pol3
                    best_score = sc_pol3
                h_cd3 = coord_descent(best_h, max_passes=30, tol=1e-12)
                sc_cd3 = official_score(h_cd3)
                if sc_cd3 < best_score:
                    best_h = h_cd3
                    best_score = sc_cd3

    # Phase 8: Try different n values
    if time_left() > 60:
        n_base = len(best_h)
        candidates = [n_base - 3, n_base - 2, n_base - 1, n_base + 1, n_base + 2, n_base + 3]
        candidates = [c for c in candidates if c >= 30 and c <= 80]
        h_try, sc_try = try_n_values(best_h, candidates, max_iters_each=60)
        if sc_try < best_score:
            best_h = h_try
            best_score = sc_try

    # Phase 9: Final fine PG sweep
    if time_left() > 30:
        h_cur = best_h.copy()
        beta = 200.0
        for _ in range(200):
            if time_left() < 15:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.2, max_ls=10)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.01, 800.0)

    # Phase 10: Final LP with very tight trust
    if time_left() > 15:
        h_final, sc_final = full_lp_polish(best_h, n_iters=20, trust_init=0.05)
        if sc_final < best_score:
            best_h = h_final
            best_score = sc_final

    # Phase 11: Final coordinate descent
    if time_left() > 10:
        h_cd_final = coord_descent(best_h, max_passes=30, tol=1e-13)
        sc_final = official_score(h_cd_final)
        if sc_final < best_score:
            best_h = h_cd_final
            best_score = sc_final

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
