def make_feasible(z):

    b_rows = []

        L = n - abs(s)

            gp[t:n] -= h[0:L]

        A_rows.append(row)

    bounds_list = []

        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')

def pg_step(h, beta, alpha_max=0.8, max_ls=18):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

def interpolate_to(h_src, n_tgt):

    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def antisymmetrize(h):
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n // 2):
        v = (h[i] + (1.0 - h[n - 1 - i])) / 2.0
        h_sym[i] = v
        h_sym[n - 1 - i] = 1.0 - v
    if n % 2 == 1:
        h_sym[n // 2] = 0.5
    return h_sym

def from_antisym_params(params):
    m = len(params)
    n = 2 * m
    h = np.zeros(n)
    for i in range(m):
        h[i] = params[i]
        h[n - 1 - i] = 1.0 - params[i]
    return h

def de_objective(params):

    return official_score(h)

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def pg_step(h, beta, alpha_max=0.8, max_ls=16):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    shifted = beta * (V - f)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    grad = fast_gradient(h, w)
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:] = 1.0 - h[0:L]
            gp[:L] -= h[s:]
        else:
            t = -s
            gp[:L] = 1.0 - h[t:]
            gp[t:] -= h[:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def pairwise_exchange_sa(h, rng, max_iter=200, T_init=0.01):
    n = len(h)
    best_h = h.copy()
    best_val = official_score(h)
    T = T_init
    cur_val = best_val
    for it in range(max_iter):
        V = compute_values(h)
        V_max = V.max()
        active = np.where(V >= V_max - 0.02)[0]
        coords = set()
        for idx in active:
            s = idx - (n - 1)
            if s >= 0:
                coords.update(range(s, n))
                coords.update(range(0, n - s))
            else:
                t = -s
                coords.update(range(0, n - t))
                coords.update(range(t, n))
        coords = sorted(coords)
        if len(coords) < 2:
            coords = list(range(n))
        i = coords[rng.integers(len(coords))]
        j = coords[rng.integers(len(coords))]
        if i == j:
            continue
        d = rng.uniform(0.01, 0.3)
        direction = 1 if rng.random() < 0.5 else -1
        h_new = h.copy()
        h_new[i] = h[i] + direction * d
        h_new[j] = h[j] - direction * d
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        f_new = official_score(h_new)
        if f_new < cur_val or rng.random() < np.exp(-(f_new - cur_val) / max(T, 1e-10)):
            h = h_new
            cur_val = f_new
            if f_new < best_val:
                best_val = f_new
                best_h = h.copy()
        T = T_init * np.exp(-it / (max_iter * 0.7))
    return best_h, best_val

def nelder_mead_polish(h, budget_s=40):
    n = len(h)
    t_start = time.time()
    V = compute_values(h)
    V_max = V.max()
    active_coords = set()
    for idx in range(len(V)):
        if V[idx] >= V_max - 0.01:
            s = idx - (n - 1)
            if s >= 0:
                active_coords.update(range(s, n))
                active_coords.update(range(0, n - s))
            else:
                t = -s
                active_coords.update(range(0, n - t))
                active_coords.update(range(t, n))
    coords = sorted(active_coords)
    if len(coords) > 16:
        step = max(1, len(coords) // 16)
        coords = coords[::step][:16]
    if len(coords) < 2:
        return h, official_score(h)

    def nm_obj(x):
        h_new = h.copy()
        for k in range(len(coords)):
            h_new[coords[k]] = x[k]
        h_feas = make_feasible(h_new)
        return official_score(h_feas)

    x0 = np.array([h[ci] for ci in coords])
    try:
        res = minimize(nm_obj, x0, method='Nelder-Mead',
                       options={'maxiter': 500, 'xatol': 1e-9, 'fatol': 1e-11,
                                'adaptive': True})
        if time.time() - t_start < budget_s:
            h_new = h.copy()
            for k in range(len(coords)):
                h_new[coords[k]] = res.x[k]
            h_feas = make_feasible(h_new)
            return h_feas, official_score(h_feas)
    except Exception:
        pass
    return h, official_score(h)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)
