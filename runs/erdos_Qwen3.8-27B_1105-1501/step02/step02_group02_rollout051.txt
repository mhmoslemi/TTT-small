def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    """Project z to [0,1]^n with sum = n/2."""
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_active(h, lags, weights):
    """Gradient of weighted sum of correlations at given lags."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k, w in zip(lags, weights):
        s = k - (n - 1)
        if s == 0:
            g += w * dx * (1.0 - 2.0 * h)
        else:
            t = abs(s)
            # (1-h[j-t]) for j >= t
            mask1 = np.zeros(n, dtype=bool)
            mask1[t:] = True
            g[mask1] += w * dx * (1.0 - h[:n-t])
            # -h[j+t] for j <= n-1-t
            mask2 = np.zeros(n, dtype=bool)
            mask2[:n-t] = True
            g[mask2] += w * dx * (-h[t:])
    return g

def _nesterov_opt(h0, deadline, m_start=6, m_end=18, beta_start=40, beta_end=400,
                  alpha_factor=1.0, mu=0.85, update_interval=20):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h_cur.copy()
    best_h = h_cur.copy()
    best_s = _score(h_cur)
    base_alpha = alpha_factor * n / 5.0
    it = 0
    active_set = None
    weights = None

    while time.time() < deadline:
        progress = min(1.0, it / 8000.0)
        m = int(m_start + (m_end - m_start) * progress)
        beta = beta_start + (beta_end - beta_start) * progress

        cv = _corr_vec(h_cur)
        if it % update_interval == 0 or active_set is None:
            top_m = np.argsort(cv)[::-1][:m]
            cv_top = cv[top_m]
            w = np.exp(beta * (cv_top - cv_top.max()))
            w = w / w.sum()
            active_set = top_m
            weights = w

        t_curr = it + 1
        mu_adj = mu * (1.0 - 0.3 * progress)
        y = h_cur + mu_adj * (h_cur - h_prev)
        y = _project(y)

        g = _grad_active(y, active_set, weights)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        s_cur = _score(h_cur)
        a = base_alpha
        best_trial = h_cur.copy()
        best_trial_s = s_cur
        for _ in range(12):
            h_trial = _project(y - a * g)
            s_trial = _score(h_trial)
            if s_trial < best_trial_s - 1e-15:
                best_trial = h_trial
                best_trial_s = s_trial
            if s_trial < s_cur:
                break
            a *= 0.7
            if a < 0.005:
                break

        h_new = best_trial
        s_new = best_trial_s
        h_prev = h_cur
        h_cur = h_new
        it += 1

        if s_new < best_s - 1e-14:
            best_s = s_new
            best_h = h_cur.copy()

        if it % 2000 == 1999:
            h_push = _project(h_cur ** 1.8)
            s_push = _score(h_push)
            if s_push < best_s:
                best_s = s_push
                best_h = h_push.copy()

        if time.time() < deadline and s_new >= best_s + 0.001 and it > 2000:
            h_recover = _project(0.5 * h_cur + 0.5 * best_h)
            s_rec = _score(h_recover)
            if s_rec < s_new:
                h_cur = h_recover
                h_prev = h_recover.copy()

    return best_h, best_s

def _slsqp_smooth(h0, beta=500, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta

    def _grad(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        g = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            if s == 0:
                g += wk * dx * (1.0 - 2.0 * x)
            else:
                t = abs(s)
                if t < n:
                    g[t:] += wk * dx * (1.0 - x[:n - t])
                    g[:n - t] += wk * dx * (-x[t:])
        return g

    def _con_fun(x):
        return x.sum() - n / 2.0

    def _con_jac(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _lbfgs_smooth(h0, beta=200, maxiter=300):

    def _obj(x):

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_push(h, p=2.0):
    h_pos = h - h.min()
    rng_ = h_pos.max() - h_pos.min()
    if rng_ < 1e-10:
        return h.copy()
    h_norm = h_pos / rng_
    return _project(h_norm ** p)

def _make_starts(h0, rng, n_pert=6):
    n = len(h0)
    starts = [h0.copy()]
    starts.append(_binary_push(h0, 1.5))
    starts.append(_binary_push(h0, 2.5))
    for sigma in [0.05, 0.1, 0.2]:
        starts.append(_project(h0 + sigma * rng.standard_normal(n)))
    for freq in [2, 3, 5]:
        phase = rng.uniform(0, 6.28)
        pert = 0.08 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        starts.append(_project(h0 + pert))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(12.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(57) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)

    # Phase 1: Dimension screening
    n_candidates = sorted(set(list(range(max(22, parent_n - 6), parent_n + 7, 2))))
    if parent_n not in n_candidates:
        n_candidates.append(parent_n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time.time() > deadline - 45:
            break
        h_try = _interp(parent, n)
        s0 = _score(h_try)
        if time.time() < deadline - 30 and s0 < best_score + 0.002:
            h_try = _slsqp_smooth(h_try, beta=80, maxiter=40)
            s_try = _score(h_try)
        else:
            s_try = s0
        if s_try < best_score - 1e-10:
            best_score = s_try
            best_h = h_try.copy()
            best_n = n

    # Phase 2: Main Nesterov optimization on best
    remain = deadline - time.time()
    if remain > 80:
        stage_deadline = time.time() + min(remain * 0.45, 400)
        best_h, best_score = _nesterov_opt(
            best_h, stage_deadline,
            m_start=5, m_end=14, beta_start=20, beta_end=200,
            alpha_factor=1.0, mu=0.82, update_interval=15)
        best_n = len(best_h)

    # Phase 3: L-BFGS polish
    if time.time() < deadline - 25:
        new = _lbfgs_smooth(best_h, beta=150, maxiter=200)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h, best_score = new, ns

    # Phase 4: Second Nesterov round with different params
    if time.time() < deadline - 30:
        stage_deadline = time.time() + min(deadline - time.time() - 35, 300)
        best_h, best_score = _nesterov_opt(
            best_h, stage_deadline,
            m_start=8, m_end=20, beta_start=80, beta_end=600,
            alpha_factor=0.8, mu=0.9, update_interval=30)
        best_n = len(best_h)

    # Phase 5: SLSQP polish
    if time.time() < deadline - 15:
        new = _slsqp_smooth(best_h, beta=1000, maxiter=80)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h, best_score = new, ns

    # Phase 6: Multi-start around best
    if time.time() < deadline - 25:
        starts = _make_starts(best_h, rng)
        for h0 in starts:
            if time.time() > deadline - 10:
                break
            stage_dl = min(time.time() + 40, deadline - 10)
            h_try, s_try = _nesterov_opt(h0, stage_dl,
                                          m_start=8, m_end=16,
                                          beta_start=100, beta_end=500,
                                          alpha_factor=0.9, mu=0.88,
                                          update_interval=20)
            if s_try < best_score - 1e-12:
                best_h, best_score = h_try, s_try
                best_n = len(h_try)

    # Phase 7: Final SLSQP
    if time.time() < deadline - 5:
        new = _slsqp_smooth(best_h, beta=2000, maxiter=50)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h, best_score = new, ns

    best_h = _project(best_h)
    best_score = _score(best_h)

    return (best_h, best_score, len(best_h))
