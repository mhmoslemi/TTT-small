    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

    Returns G of shape (2n-1, n) where G[p, i] = d(corr[p]*dx)/dh[i]
    """
    n = len(h)

    for p in range(2*n - 1):
        s = p - (n - 1)
        if s >= 0:

                G[p, i] += (1.0 - h[i + s])
            for i in range(s, n):
                G[p, i] += (-h[i - s])

                G[p, i] += (1.0 - h[i - t])
            for i in range(0, n - t):
                G[p, i] += (-h[i + t])

    return G

    #      -delta <= d_i <= delta
    #      -h_i <= d_i <= 1 - h_i

        slack = cmax - corr_vals[p]

    constraints.append(d_var <= upper)

    prob = cp.Problem(objective, constraints)

    d = d_var.value

def make_feasible(z):

                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.1)
        best_score = official_score(best_h)

    # Phase 1: LP minimax with large steps
    if time.time() < deadline:
        for delta in [0.15, 0.10, 0.08, 0.05, 0.03]:
            if time.time() >= deadline:
                break
            h_new, sc = lp_iterate(best_h, delta=delta, max_steps=150, deadline=deadline)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 2: Two-point search
    if time.time() < deadline:
        n_tries = max(2000, int((deadline - time.time()) * 2))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Phase 3: LP with smaller steps
    if time.time() < deadline:
        for delta in [0.03, 0.02, 0.01, 0.005, 0.002]:
            if time.time() >= deadline:
                break
            h_new, sc = lp_iterate(best_h, delta=delta, max_steps=200, deadline=deadline)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 4: Subgradient + two-point
    if time.time() < deadline:
        for _ in range(20):
            if time.time() >= deadline:
                break
            h_new, sc = subgradient_step(best_h, deadline)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
            if time.time() >= deadline:
                break
            best_h, best_score = two_point_search(best_h, 500, rng, deadline)

    # Phase 5: Try different resolutions
    if time.time() < deadline:
        n_base = len(best_h)
        n_opts = []
        for dn in [-12, -8, -4, 4, 8, 12, 16]:
            nt = n_base + dn
            if 24 <= nt <= 100:
                n_opts.append(nt)
        for nc in [36, 44, 50, 56, 64, 72, 80]:
            if nc not in n_opts and 24 <= nc <= 100:
                n_opts.append(nc)
        n_opts = sorted(set(n_opts))
        for n_opt in n_opts:
            if time.time() >= deadline:
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            h_work, sc_work = lp_iterate(h_int, delta=0.04, max_steps=80, deadline=deadline)
            sc_work = official_score(h_work)
            if time.time() < deadline:
                h_work, sc_work = two_point_search(h_work, 600, rng, deadline)
            if sc_work < best_score - 1e-14:
                best_h = h_work
                best_score = sc_work

    # Phase 6: Alternating LP + two-point (intensive)
    if time.time() < deadline:
        for i in range(8):
            if time.time() >= deadline:
                break
            delta = 0.02 / (1.0 + 0.5 * i)
            h_new, sc = lp_iterate(best_h, delta=delta, max_steps=100, deadline=deadline)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
            if time.time() >= deadline:
                break
            n_t = max(500, int((deadline - time.time()) * 1.5))
            best_h, best_score = two_point_search(best_h, n_t, rng, deadline)

    # Phase 7: Random restarts from perturbed best
    if time.time() < deadline:
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            if time.time() >= deadline:
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=len(best_h)))
            h_new, sc = lp_iterate(h_pert, delta=0.05, max_steps=50, deadline=deadline)
            if time.time() < deadline:
                h_new, sc = two_point_search(h_new, 500, rng, deadline)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 8: Final fine LP
    if time.time() < deadline:
        for delta in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() >= deadline:
                break
            h_new, sc = lp_iterate(best_h, delta=delta, max_steps=100, deadline=deadline)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 9: Final two-point polish
    if time.time() < deadline:
        n_tries = max(1000, int((deadline - time.time()) * 3))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Phase 10: Last LP touch
    if time.time() < deadline:
        h_new, sc = lp_iterate(best_h, delta=0.001, max_steps=50, deadline=deadline)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-10

    return final_h, final_score, final_n
