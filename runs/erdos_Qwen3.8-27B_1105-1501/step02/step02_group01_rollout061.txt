    corr = np.correlate(h, 1.0 - h, mode='full')

def max_corr_int(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(round(np.max(corr)))

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -4.0 - np.max(z), 4.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    if n % 2 == 0:
        n_ones = n // 2
        idx = np.argsort(h)[::-1][:n_ones]
        b = np.zeros(n)
        b[idx] = 1.0
    else:
        n_ones = n // 2
        idx = np.argsort(h)[::-1]
        b = np.zeros(n)
        b[idx[:n_ones]] = 1.0
        b[idx[n_ones]] = 0.5
    return b

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def anneal_binary(h0, rng, n_iters=100000, T_start=3.0, T_end=0.05, time_limit=None, t_ref=None):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_mc = max_corr_int(h)
    cur_mc = best_mc
    cool = (T_end / T_start) ** (1.0 / max(n_iters, 1))
    no_improve = 0
    for it in range(n_iters):
        if time_limit is not None and t_ref is not None:
            if time.time() - t_ref > time_limit:
                break
        T = T_start * cool ** it
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < 0.8:
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_mc = max_corr_int(h)
            delta = new_mc - cur_mc
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_mc = new_mc
                if new_mc < best_mc:
                    best_mc = new_mc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        else:
            # Block swap
            max_L = min(6, n // 4)
            L = int(rng.integers(1, max_L + 1))
            s1 = int(rng.integers(0, n - L))
            s2 = int(rng.integers(0, n - L))
            if abs(s1 - s2) < L:
                no_improve += 1
                continue
            tmp = h[s1:s1+L].copy()
            h[s1:s1+L] = h[s2:s2+L]
            h[s2:s2+L] = tmp
            new_mc = max_corr_int(h)
            delta = new_mc - cur_mc
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_mc = new_mc
                if new_mc < best_mc:
                    best_mc = new_mc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[s2:s2+L] = h[s1:s1+L]
                h[s1:s1+L] = tmp
                no_improve += 1
        if no_improve > 5000:
            break
    return best_h, best_mc

def local_search_binary(h0, max_rounds=200, time_limit=None, t_ref=None):
    h = h0.copy()
    n = len(h)
    cur_mc = max_corr_int(h)
    for rnd in range(max_rounds):
        if time_limit is not None and t_ref is not None:
            if time.time() - t_ref > time_limit:
                break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_mc = max_corr_int(h)
                if new_mc < cur_mc:
                    cur_mc = new_mc
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, cur_mc

def multi_start_binary(n, rng, parent_h=None, n_starts=8, time_limit=None, t_ref=None):
    best_h = None
    best_mc = 999
    starts = []
    if parent_h is not None:
        hi = interp_to_n(parent_h, n)
        starts.append(to_binary(hi))
    for _ in range(3):
        b = np.zeros(n)
        b[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(b)
    # Alternating
    alt = np.zeros(n)
    alt[::2] = 1.0
    if sum(alt) != n // 2:
        alt[n - 1] = 0.0
    starts.append(alt.copy())
    # Block patterns
    for p in [3, 4, 5]:
        b = np.zeros(n)
        chunk = p
        n_ones = n // 2
        pattern = [1.0] * (p // 2) + [0.0] * (p - p // 2)
        full = np.tile(pattern, n // len(pattern) + 1)[:n]
        # Adjust to get exactly n/2 ones
        s = np.sum(full)
        if s > n_ones:
            idx = np.where(full == 1.0)[0]
            full[idx[:int(s - n_ones)]] = 0.0
        elif s < n_ones:
            idx = np.where(full == 0.0)[0]
            full[idx[:int(n_ones - s)]] = 1.0
        starts.append(full)
    
    per_start_time = time_limit / max(len(starts), 1) if time_limit else None
    for b0 in starts:
        if time_limit is not None and t_ref is not None:
            if time.time() - t_ref > time_limit:
                break
        nt = 50000 if n <= 50 else 30000
        if per_start_time:
            sa_h, sa_mc = anneal_binary(b0, rng, n_iters=nt, T_start=3.0, T_end=0.05,
                                        time_limit=per_start_time * 0.6, t_ref=time.time())
        else:
            sa_h, sa_mc = anneal_binary(b0, rng, n_iters=nt, T_start=3.0, T_end=0.05)
        # Local search
        if time_limit is not None and t_ref is not None:
            if time.time() - t_ref > time_limit:
                break
        ls_h, ls_mc = local_search_binary(sa_h, max_rounds=100,
                                          time_limit=per_start_time * 0.3 if per_start_time else None,
                                          t_ref=time.time())
        if ls_mc < best_mc:
            best_mc = ls_mc
            best_h = ls_h.copy()
    return best_h, best_mc

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n - s2])
                grad[:n - s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n - m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n - m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def do_slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0, args=(beta, dx),
            jac=lambda x: smooth_lse_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(20.0, 0.06 * budget_s)
    deadline = t0 + budget_s - reserve
    
    rng = np.random.default_rng(seed)
    
    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)
    
    # Initial best
    best_h = parent.copy()
    best_score = official_score(best_h)
    
    # n values to try (even, focused on promising range)
    n_options = [44, 46, 48, 50, 52, 54, 56, 58, 60, 64]
    
    # Sort by potential: prefer larger n with achievable max_corr
    # n=48 target mc=9 -> 0.375; n=50 target mc=9 -> 0.36
    # Try in order of most promising
    n_options = [48, 50, 46, 52, 44, 54, 56, 58, 60, 64]
    
    for n in n_options:
        if time.time() > deadline - 60:
            break
        
        n_left_budget = deadline - time.time()
        if n_left_budget < 30:
            break
        
        # Try multiple starts for this n
        t_ref = time.time()
        t_lim = max(10, n_left_budget - 10)
        
        bh, bmc = multi_start_binary(n, rng, parent_h=parent, n_starts=6,
                                     time_limit=t_lim, t_ref=t_ref)
        
        if bh is not None:
            score = bmc * (2.0 / n)
            if score < best_score:
                best_score = score
                best_h = bh.copy()
                # Continue improving
                # Another SA round
                if time.time() < deadline - 20:
                    bh2, bmc2 = anneal_binary(bh, rng, n_iters=80000, T_start=1.0, T_end=0.02,
                                              time_limit=30, t_ref=time.time())
                    ls_h2, ls_mc2 = local_search_binary(bh2, max_rounds=100,
                                                       time_limit=15, t_ref=time.time())
                    s2 = ls_mc2 * (2.0 / n)
                    if s2 < best_score:
                        best_score = s2
                        best_h = ls_h2.copy()
                
                # Try continuous polish
                if time.time() < deadline - 15:
                    cont = do_slsqp_smooth(best_h, 200, maxiter=200)
                    cs = official_score(cont)
                    if cs < best_score:
                        best_score = cs
                        best_h = cont
                    # Try from binary too
                    cont2 = do_slsqp_smooth(best_h, 500, maxiter=150)
                    cs2 = official_score(cont2)
                    if cs2 < best_score:
                        best_score = cs2
                        best_h = cont2
    
    # Final refinement on best solution
    if time.time() < deadline - 5:
        n_best = len(best_h)
        # Try SA again on best
        b_bin = to_binary(best_h) if n_best % 2 == 0 else best_h
        if n_best % 2 == 0:
            bh3, bmc3 = anneal_binary(b_bin, rng, n_iters=50000, T_start=0.5, T_end=0.01,
                                      time_limit=20, t_ref=time.time())
            ls_h3, ls_mc3 = local_search_binary(bh3, max_rounds=50, time_limit=10, t_ref=time.time())
            s3 = ls_mc3 * (2.0 / n_best)
            if s3 < best_score:
                best_score = s3
                best_h = ls_h3.copy()
        
        # SLSQP polish
        if time.time() < deadline - 3:
            for beta in [300, 100]:
                if time.time() > deadline - 3:
                    break
                pol = do_slsqp_smooth(best_h, beta, maxiter=int(max(50, (deadline - time.time()) * 3)))
                ps = official_score(pol)
                if ps < best_score:
                    best_score = ps
                    best_h = pol
    
    # Ensure feasible
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
