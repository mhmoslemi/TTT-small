def _scores(h):
    """Return the raw correlation scores (before multiplying by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _soft_max(scores, beta):
    m = np.max(scores)
    if beta < 1e-6:
        return m
    exps = np.exp(beta * (scores - m))
    return (m + np.log(np.sum(exps)) / beta), exps / np.sum(exps)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _soft_descent(h, deadline, beta_start=30.0, beta_end=3000.0, max_iters=20000):

    target = n / 2.0

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    alphas = np.linspace(0.05, alpha_max, 12)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:

        h_t = _project(h_t, n / 2.0)

        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    for _ in range(20):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, n / 2.0))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_polish(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

            break

        G = _grad_matrix(h)

            radius *= 0.85

                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)

            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0

        else:

    return best_h, best_c5

def _coord_descent(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()

                    h_try = h.copy()
                    h_try[j] += d
                    # Compensate on another coordinate to maintain sum
                    # Find the coordinate with most room
                    comp = int(np.argmax(1.0 - h_try) if d > 0 else np.argmax(h_try))
                    if comp == j:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        if d > 0:
                            comp = int(np.argmax(np.where(mask, 1.0 - h_try, -1)))
                        else:

                        continue

                    if c5_try < best_c5 - 1e-12:

                        best_h = h_try.copy()

def _thue_morse_init(n, amplitude=0.3):
    tm = np.zeros(n, dtype=int)

        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amplitude * (2.0 * tm - 1.0)

    x = np.arange(n) / n
