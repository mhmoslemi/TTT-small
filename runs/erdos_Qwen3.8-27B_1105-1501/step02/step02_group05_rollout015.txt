import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(vals.max())

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    """Compute gradient of the lag value V_p (with shift s) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[s:n] += dx * (1.0 - h[0:m])
            grad[0:m] += dx * (-h[s:n])
    else:
        t = -s
        m = n - t
        if m > 0:
            grad[0:m] += dx * (1.0 - h[t:n])
            grad[t:n] += dx * (-h[0:m])
    return grad

def select_active(V, C, top_k=32, eta=0.006):
    """Select active lag indices: top-k and those within eta of C."""
    n = len(V)
    # Top k
    top_idx = np.argsort(V)[-top_k:][::-1]
    # Within eta of max
    thresh_idx = np.where(V >= C - eta)[0]
    # Union
    active = np.unique(np.concatenate([top_idx, thresh_idx]))
    return active

def lp_step(h, V, C, active, rho):
    """Solve the LP minimax subproblem. Returns direction d or None."""
    n = len(h)
    K = len(active)
    # Build constraint matrix
    # Variables: d[0..n-1], t
    # Objective: minimize t → c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: grad_p · d - t <= -V[p] for p in active
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, p in enumerate(active):
        s = int(p) - (n - 1)
        g = lag_gradient(h, s)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -V[p]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds for d: max(-h[i], -rho) <= d[i] <= min(1-h[i], rho)
    # t: -inf <= t <= C (upper bound to prevent increase)
    ub = np.minimum(1.0 - h, rho)
    lb = np.maximum(-h, -rho)
    bounds = list(zip(lb, ub))
    bounds.append((None, C))  # t upper bound
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.max(np.abs(d)) < 1e-15:
                return None
            return d
        elif res.status == 2:  # infeasible
            return None
        else:
            return None
    except Exception:
        return None

def binary_sa(h_float, n_steps=2000, seed=42, is_dead=None):
    """Simulated annealing on balanced binary sequences."""
    n = len(h_float)
    if n % 2 != 0:
        return None, None
    target = n // 2
    
    # Initialize from threshold
    bits = (h_float > 0.5).astype(int)
    ones = np.sum(bits)
    if ones > target:
        # Remove excess ones
        order = np.argsort(h_float)[::-1]  # highest h first (most likely to be 1)
        for idx in order:
            if ones <= target:
                break
            bits[idx] = 0
            ones -= 1
    elif ones < target:
        order = np.argsort(h_float)  # lowest h first
        for idx in order:
            if ones >= target:
                break
            bits[idx] = 1
            ones += 1
    
    # Ensure exact balance
    while np.sum(bits) > target:
        # Remove a 1 where h is lowest
        pos1 = np.where(bits == 1)[0]
        idx = pos1[np.argmin(h_float[pos1])]
        bits[idx] = 0
    while np.sum(bits) < target:
        pos0 = np.where(bits == 0)[0]
        idx = pos0[np.argmax(h_float[pos0])]
        bits[idx] = 1
    
    rng = np.random.default_rng(seed)
    curr_score = score_exact(bits.astype(float))
    best_bits = bits.copy()
    best_score = curr_score
    T = 2e-4
    
    pos1 = np.where(bits == 1)[0]
    
    for step in range(n_steps):
        if is_dead is not None and is_dead():
            break
        # Random swap
        i = rng.integers(0, len(pos1))
        # Find a random 0 position
        pos0 = np.where(bits == 0)[0]
        j = pos0[rng.integers(0, len(pos0))]
        pos1 = np.where(bits == 1)[0]
        i = pos1[rng.integers(0, len(pos1))]
        
        # Swap
        bits[i] = 0
        bits[j] = 1
        
        new_score = score_exact(bits.astype(float))
        delta = new_score - curr_score
        
        if delta <= 1e-15 or (T > 1e-8 and np.log(rng.random() + 1e-30) > delta / T):
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_bits = bits.copy()
        else:
            # Revert
            bits[j] = 0
            bits[i] = 1
        
        T *= 0.999
    
    return best_bits.astype(float), best_score

def interp_to(h_src, n_tgt):

    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2.0 * np.pi * freq1 * x + phase) + amp2 * np.cos(2.0 * np.pi * freq2 * x)
    return make_feasible(h)

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [t_lo + span * s for s in [0, 0.2, 0.38, 0.5, 0.62, 0.8, 1.0]]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None:
            best[a] += best_t
            best[b] -= best_t
            np.clip(best, 0, 1, out=best)
            best_sc = score_exact(best)
    return best, best_sc

import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[s:n] += dx * (1.0 - h[0:m])
            grad[0:m] += dx * (-h[s:n])
    else:
        t = -s
        m = n - t
        if m > 0:
            grad[0:m] += dx * (1.0 - h[t:n])
            grad[t:n] += dx * (-h[0:m])
    return grad

def select_active(V, C, top_k=32, eta=0.006):
    n = len(V)
    top_idx = np.argsort(V)[-top_k:][::-1]
    thresh_idx = np.where(V >= C - eta)[0]
    active = np.unique(np.concatenate([top_idx, thresh_idx]))
    return active

def lp_step(h, V, C, active, rho):
    n = len(h)
    K = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, p in enumerate(active):
        s = int(p) - (n - 1)
        g = lag_gradient(h, s)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -V[p]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    ub = np.minimum(1.0 - h, rho)
    lb = np.maximum(-h, -rho)
    bounds = list(zip(lb, ub))
    bounds.append((None, C))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.max(np.abs(d)) < 1e-14:
                return None
            return d
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='revised simplex')
        if res.status == 0:
            d = res.x[:n]
            if np.max(np.abs(d)) < 1e-14:
                return None
            return d
    except Exception:
        pass
    return None

def binary_sa(h_float, n_steps=2000, seed=42, is_dead=None):
    n = len(h_float)
    if n % 2 != 0:
        return None, None
    target = n // 2
    bits = (h_float > 0.5).astype(np.int8)
    ones = int(np.sum(bits))
    if ones > target:
        order = np.argsort(h_float)[::-1]
        for idx in order:
            if ones <= target:
                break
            bits[idx] = 0
            ones -= 1
    elif ones < target:
        order = np.argsort(h_float)
        for idx in order:
            if ones >= target:
                break
            bits[idx] = 1
            ones += 1
    while int(np.sum(bits)) > target:
        pos1 = np.where(bits == 1)[0]
        idx = pos1[np.argmin(h_float[pos1])]
        bits[idx] = 0
    while int(np.sum(bits)) < target:
        pos0 = np.where(bits == 0)[0]
        idx = pos0[np.argmax(h_float[pos0])]
        bits[idx] = 1

    rng = np.random.default_rng(seed)
    curr_score = score_exact(bits.astype(float))
    best_bits = bits.copy()
    best_score = curr_score
    T = 1.5e-4

    pos1_cache = np.where(bits == 1)[0]
    pos0_cache = np.where(bits == 0)[0]

    for step in range(n_steps):
        if is_dead is not None and is_dead():
            break
        if len(pos1_cache) == 0 or len(pos0_cache) == 0:
            break
        i = pos1_cache[rng.integers(0, len(pos1_cache))]
        j = pos0_cache[rng.integers(0, len(pos0_cache))]
        bits[i] = 0
        bits[j] = 1
        pos1_cache = np.where(bits == 1)[0]
        pos0_cache = np.where(bits == 0)[0]
        new_score = score_exact(bits.astype(float))
        delta = new_score - curr_score
        if delta <= 1e-15 or (T > 5e-9 and np.log(rng.random() + 1e-30) > delta / T):
            curr_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_bits = bits.copy()
        else:
            bits[j] = 0
            bits[i] = 1
            pos1_cache = np.where(bits == 1)[0]
            pos0_cache = np.where(bits == 0)[0]
        T *= 0.9985

    return best_bits.astype(float), best_score

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2.0 * np.pi * freq1 * x + phase) + amp2 * np.cos(2.0 * np.pi * freq2 * x)
    return make_feasible(h)

def two_point_search(h, n_tries, rng, is_dead):

    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break

        b = rng.integers(0, n)
        if a == b:
            continue

        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [t_lo + span * s for s in [0.0, 0.2, 0.38, 0.5, 0.62, 0.8, 1.0]]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None:
            best[a] += best_t
            best[b] -= best_t
            np.clip(best, 0.0, 1.0, out=best)
            best_sc = score_exact(best)
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    def is_dead():
        return time.time() >= deadline

    rng = np.random.default_rng(seed)
    best_h = None
    best_score = float('inf')
    fallback = None

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                fallback = parent.copy()
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.1)
        best_score = score_exact(best_h)
        fallback = best_h.copy()

    n = len(best_h)

    # Try simple transformations
    tfs = [best_h.copy(), best_h[::-1].copy(), (1.0 - best_h.copy()),
           (1.0 - best_h[::-1].copy()), anti_sym_map(best_h)]
    for tf in tfs:
        sc = score_exact(tf)
        if sc < best_score - 1e-12:
            best_h = tf
            best_score = sc
    fallback = best_h.copy()

    # Phase 1: Main LP refinement
    if not is_dead():
        h_new, sc = lp_refine(best_h, max_iter=400, is_dead=is_dead)
        if sc < best_score - 1e-13:
            best_h = h_new
            best_score = sc
    fallback = best_h.copy()

    # Phase 2: Binary SA with multiple starts, then LP on results
    if not is_dead():
        b_starts = [best_h.copy()]
        for freq in [1, 2, 3, 4]:
            b_starts.append(make_wave(n, freq, 0.35, freq + 1, 0.15))
        for bi, b_start in enumerate(b_starts):
            if is_dead():
                break
            b_arr, b_sc = binary_sa(b_start, n_steps=1200, seed=seed + bi * 997, is_dead=is_dead)
            if b_arr is not None:
                if b_sc < best_score - 1e-13:
                    best_h = make_feasible(b_arr)
                    best_score = score_exact(best_h)
                if not is_dead():
                    h_cont = make_feasible(b_arr)
                    h_new, sc = lp_refine(h_cont, max_iter=80, is_dead=is_dead)
                    if sc < best_score - 1e-13:
                        best_h = h_new
                        best_score = sc
    fallback = best_h.copy()

    # Phase 3: LP with varied parameters
    if not is_dead():
        temp = best_h.copy()
        for rho_try in [0.3, 0.4, 0.5]:
            if is_dead():
                break
            V = corr_values(temp)
            C = float(V.max())
            active = select_active(V, C, top_k=48, eta=0.012)
            d = lp_step(temp, V, C, active, rho_try)
            if d is not None:
                for scale in [0.5, 0.35, 0.25, 0.15, 0.08]:
                    trial = make_feasible(temp + scale * d)
                    sc = score_exact(trial)
                    if sc < best_score - 1e-13:
                        best_h = trial
                        best_score = sc
                        temp = trial
                        break
    fallback = best_h.copy()

    # Phase 4: Resolution scanning
    if not is_dead():
        n_base = len(best_h)
        n_opts = []
        for delta in [-8, -4, 4, 8, 12]:
            nt = n_base + delta
            if 24 <= nt <= 100:
                n_opts.append(nt)
        for nc in [36, 44, 50, 56, 64, 72, 80]:
            if nc not in n_opts:
                n_opts.append(nc)
        n_opts = sorted(set(n_opts))
        for n_opt in n_opts:
            if is_dead() or n_opt < 20 or n_opt > 100 or n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-13:
                best_h = h_int
                best_score = sc_int
            if not is_dead():
                h_new, sc = lp_refine(h_int, max_iter=120, is_dead=is_dead)
                if sc < best_score - 1e-13:
                    best_h = h_new
                    best_score = sc
    fallback = best_h.copy()

    # Phase 5: Intensive LP on current best
    if not is_dead():
        h_new, sc = lp_refine(best_h, max_iter=500, is_dead=is_dead)
        if sc < best_score - 1e-13:
            best_h = h_new
            best_score = sc
    fallback = best_h.copy()

    # Phase 6: Two-point polish
    if not is_dead():
        remaining = deadline - time.time()
        n_tries = max(2000, int(remaining * 3))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)
    fallback = best_h.copy()

    # Phase 7: Final LP
    if not is_dead():
        h_new, sc = lp_refine(best_h, max_iter=250, is_dead=is_dead)
        if sc < best_score - 1e-13:
            best_h = h_new
            best_score = sc
    fallback = best_h.copy()

    # Phase 8: Final two-point sweep
    if not is_dead():
        remaining = deadline - time.time()
        n_tries = max(1000, int(remaining * 2))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)) or final_n < 4:
        final_h = make_feasible(fallback)
        final_score = score_exact(final_h)
        final_n = len(final_h)

    final_h = np.clip(final_h, 0.0, 1.0)
    res = final_h.sum() - final_n / 2.0
    if abs(res) > 1e-10:
        final_h = make_feasible(final_h)

    final_score = score_exact(final_h)
    return final_h, final_score, len(final_h)
