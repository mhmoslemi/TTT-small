def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

        else:

        t = it + 2
        t_prev = it + 1
        t_new = 1.5 * t
        y_step = h_cur + (t_prev / t_new) * (h_cur - h_prev)
        y_step = _project(y_step)
        
        h_new = _project(y_step - alpha * grad)
        
        h_prev = h_cur

def _slsqp_smooth(h0, beta, maxiter=200):

    def _lse_grad(x):

        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()

        else:
            w = ev / es

                    g[i + j] += wk * (1.0 - x[i])
                    g[i] += wk * (-x[i + j])
        return g

        return np.ones(n)
    
    try:

                       method='SLSQP', bounds=bounds,

def _interp(h_src, n_dst):

    return _project(np.interp(x_dst, x_src, h_src))

def _block_init(n, n_blocks=2):
    """Create a block-structured initialization."""
    h = np.zeros(n)
    block_size = n // (2 * n_blocks)
    for i in range(n_blocks):
        start = i * 2 * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    return _project(h)

def _slsqp_lse(h0, beta, maxiter=150):
    """SLSQP minimizing soft-max (log-sum-exp) approximation of the max."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n

    def obj(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta

    def grad(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-18:
                continue
            s = k - (n - 1)
            if s == 0:
                g[:] += wk * (1.0 - x)
                g[:] += wk * (-x)
            elif s > 0:
                g[:n - s] += wk * (1.0 - x[s:])
                g[s:] += wk * (-x[:n - s])
            else:
                j = -s
                g[j:] += wk * (1.0 - x[:n - j])
                g[:n - j] += wk * (-x[j:])
        return g

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, args=(beta,), jac=grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

        return z[-1]

        g[-1] = 1.0
        return g

        t = z[-1]
        cv_full = _corr_vec(h)

            for a in range(len(active)):
                for b in range(a + 1, len(active)):
                    if abs(coeffs[a] - coeffs[b]) > 1e-15:
                        tc = (bases[b] - bases[a]) / (coeffs[a] - coeffs[b])
                        if -0.01 <= tc <= 1.01:
                            candidates.append(tc)
            best_t = old
            best_val = max(bases + coeffs * old)
            for t in candidates:
                t = float(np.clip(t, 0.0, 1.0))
                val = max(bases + coeffs * t)
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t
            if best_t != old:
                h[i] = best_t
                improved = True
        if not improved:
            break
        h = _project(h)
        s = _score(h)
        if s < best_s:
            best_s = s
    return h, best_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _perturb(h, sigma, rng):
    return _project(h + sigma * rng.standard_normal(len(h)))

def _optimize_chain(h0, deadline):
    """Multi-stage optimization pipeline for a single starting point."""
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)

    if time.time() < deadline - 20:
        h_new, s_new = _fista(cur, n_iter=4000, alpha=0.3, mu=0.87, m=8)
        if s_new < cur_s - 1e-12:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 15:
        h_new = _slsqp_lse(cur, beta=200, maxiter=120)
        s_new = _score(h_new)
        if s_new < cur_s - 1e-12:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 10:
        h_new, s_new = _fista_adaptive(cur, n_iter=5000, m=10)
        if s_new < cur_s - 1e-12:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 8:
        h_new = _slsqp_lse(cur, beta=800, maxiter=150)
        s_new = _score(h_new)
        if s_new < cur_s - 1e-12:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 6:
        h_new = _slsqp_direct(cur, m_active=10, maxiter=120)
        s_new = _score(h_new)
        if s_new < cur_s - 1e-12:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 4:
        h_new, s_new = _coord_descent(cur, n_sweeps=40, m=8)
        if s_new < cur_s - 1e-12:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 3:
        h_new, s_new = _fista(cur, n_iter=3000, alpha=0.15, mu=0.9, m=12, temp=5.0)
        if s_new < cur_s - 1e-12:
            cur, cur_s = h_new, s_new

    return cur, cur_s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(40) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)

    # Phase 1: Screen n values with quick optimization
    n_candidates = sorted(set(
        [parent_n] +
        list(range(max(24, parent_n - 12), min(72, parent_n + 14) + 1, 2))
    ))

    phase1_results = []
    for n in n_candidates:
        if time.time() > deadline - 100:
            break
        h_n = _interp(parent, n)
        # Quick FISTA to get a decent starting point
        h_opt, s_opt = _fista(h_n, n_iter=2000, alpha=0.3, m=8)
        phase1_results.append((s_opt, n, h_opt))
        if s_opt < best_score - 1e-12:
            best_h = h_opt.copy()
            best_score = s_opt
            best_n = n

    # Phase 2: Focused optimization on top candidates
    phase1_results.sort(key=lambda x: x[0])
    top_candidates = []
    seen_ns = set()
    for s, n, h in phase1_results:
        if n not in seen_ns:
            top_candidates.append((n, h))
            seen_ns.add(n)
        if len(top_candidates) >= 4:
            break

    for n, h_base in top_candidates:
        if time.time() > deadline - 80:
            break

        # Generate diverse starting points
        starts = [h_base.copy(), _mirror(h_base)]
        starts.append(_perturb(h_base, 0.1, rng))
        starts.append(_perturb(h_base, 0.2, rng))
        # Block-like perturbation
        block_h = np.zeros(n)
        bs = n // 6
        for bi in range(3):
            block_h[bi * 2 * bs:(bi * 2 + 1) * bs] = 1.0
        starts.append(_project(0.5 * h_base + 0.5 * _project(block_h)))

        for h0 in starts:
            if time.time() > deadline - 40:
                break
            cur, cur_s = _optimize_chain(h0, deadline)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)

    # Phase 3: Intensive refinement on the global best
    remaining = deadline - time.time()
    if remaining > 30:
        # Multi-pass FISTA with different parameters
        for m_val in [6, 10, 14]:
            for alpha_v in [0.2, 0.35, 0.5]:
                if time.time() > deadline - 10:
                    break
                n_iter = min(8000, max(1000, int(remaining * 0.12)))
                h_new, s_new = _fista(best_h, n_iter=n_iter, alpha=alpha_v,
                                      mu=0.88, m=m_val, temp=4.0)
                if s_new < best_score - 1e-12:
                    best_h = h_new
                    best_score = s_new
                    best_n = len(best_h)
                remaining = deadline - time.time()

    if time.time() < deadline - 25:
        # SLSQP with escalating beta
        for beta in [300, 1000, 2000]:
            if time.time() > deadline - 15:
                break
            h_new = _slsqp_lse(best_h, beta=beta, maxiter=150)
            s_new = _score(h_new)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new
                best_n = len(best_h)

    if time.time() < deadline - 15:
        # Direct min-max
        h_new = _slsqp_direct(best_h, m_active=12, maxiter=150)
        s_new = _score(h_new)
        if s_new < best_score - 1e-12:
            best_h = h_new
            best_score = s_new
            best_n = len(best_h)

    if time.time() < deadline - 10:
        # Coordinate descent
        h_new, s_new = _coord_descent(best_h, n_sweeps=60, m=10)
        if s_new < best_score - 1e-12:
            best_h = h_new
            best_score = s_new
            best_n = len(best_h)

    # Phase 4: Random restarts with remaining time
    remaining = deadline - time.time()
    n_restarts = min(int(remaining / 40), 10)
    for _ in range(n_restarts):
        if time.time() > deadline - 10:
            break
        for sigma in [0.02, 0.05, 0.1]:
            if time.time() > deadline - 10:
                break
            h_pert = _perturb(best_h, sigma, rng)
            cur, cur_s = _optimize_chain(h_pert, deadline)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)

    # Final polish
    if time.time() < deadline - 5:
        h_new, s_new = _fista(best_h, n_iter=4000, alpha=0.1,
                              mu=0.92, m=12, temp=6.0)
        if s_new < best_score - 1e-12:
            best_h = h_new
            best_score = s_new

    if time.time() < deadline - 3:
        h_new = _slsqp_lse(best_h, beta=3000, maxiter=100)
        s_new = _score(h_new)
        if s_new < best_score - 1e-12:
            best_h = h_new
            best_score = s_new

    # Final feasibility check
    best_h = _project(best_h)
    final_score = _score(best_h)

    return (best_h, final_score, len(best_h))
