import time

def score_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    n = z.size
    if n == 0:
        return z
    target = n / 2.0
    # Replace non-finite
    nan_mask = ~np.isfinite(z)
    if np.any(nan_mask):
        z[nan_mask] = 0.5
    z = np.clip(z, 0.0, 1.0)
    # Bisection to find shift that gives correct sum
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Greedy residual repair
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, p):
    """Compute gradient of the p-th correlation value (times dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)

        # dV/dh[i] = (1-h[i-s]) for i in s..n-1, minus h[i+s] for i in 0..n-1-s

        grad[idx1] += (1.0 - h[idx1 + t])

        grad[idx2] -= h[idx2 - t]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # A_ub: each row is [G[i], -1] <= [-c_active[i]]
    A_ub = np.hstack([G, -np.ones((len(active_idx), 1))])
    b_ub = -c_active
    
    # A_eq: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((-2.0, C))  # t bounds
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            u = res.x[:n]
            t_pred = res.x[-1]
            if t_pred < C - 1e-10:  # Predicted improvement
                return u
        elif res.status == 2:  # Infeasible
            return None
    except Exception:
        pass
    return None

def select_active(V, C, top_k=30, threshold=0.005):
    """Select active lag indices."""
    n = len(V)
    # Lags within threshold of max, plus top-k
    mask_thresh = V >= C - threshold
    top_idx = np.argsort(V)[-top_k:]
    active = np.unique(np.concatenate([np.where(mask_thresh)[0], top_idx]))
    # Sort by value descending
    active = active[np.argsort(V[active])[::-1]]
    return active

def lp_refine(h, n_iter, is_dead, top_k=30, threshold=0.005):

    best = h.copy()
    best_score = score_exact(best)
    n = len(best)
    
    rho_list = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01]
    alpha_list = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]
    
    stall = 0
    for it in range(n_iter):
        if is_dead():
            break
        V = np.correlate(best, 1.0 - best, mode='full') * (2.0 / n)
        C = V.max()
        
        active = select_active(V, C, top_k=top_k, threshold=threshold)
        if len(active) == 0:
            break
        
        improved = False
        for rho in rho_list:
            if is_dead() or improved:
                break
            u = lp_step(best, V, C, active, rho)
            if u is None:
                continue
            for alpha in alpha_list:
                h_trial = make_feasible(best + alpha * u)
                sc = score_exact(h_trial)
                if sc < best_score - 1e-12:
                    best = h_trial
                    best_score = sc
                    improved = True
                    stall = 0
                    break

            stall += 1

def binary_sa(h, n_steps, rng, is_dead):
    """Simulated annealing over balanced binary strings."""
    n = len(h)
    if n % 2 != 0:
        return None, None
    target_ones = n // 2
    
    # Initialize from threshold
    b = (h > 0.5).astype(float)
    n_ones = int(b.sum())
    if n_ones > target_ones:
        # Remove excess ones (those closest to 0.5 from below)
        one_idx = np.where(b == 1)[0]
        # Sort by h value ascending (closest to 0 first... actually closest to 0.5 from above)
        # We want to remove ones with lowest h values
        order = np.argsort(h[one_idx])
        for i in range(n_ones - target_ones):
            b[one_idx[order[i]]] = 0
    elif n_ones < target_ones:
        # Add ones (those closest to 0.5 from below)
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target_ones - n_ones):
            b[zero_idx[order[i]]] = 1
    
    assert int(b.sum()) == target_ones
    
    curr_score = score_exact(b)
    best_b = b.copy()
    best_score = curr_score
    
    T = 5e-4
    cool = 0.999
    
    pos1 = np.where(b == 1)[0]
    pos0 = np.where(b == 0)[0]
    
    for it in range(n_steps):
        if is_dead():
            break
        if len(pos1) == 0 or len(pos0) == 0:
            break
        
        i = pos1[rng.integers(len(pos1))]
        j = pos0[rng.integers(len(pos0))]
        
        # Swap
        b[i], b[j] = b[j], b[i]
        new_score = score_exact(b)
        delta = new_score - curr_score
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            curr_score = new_score
            # Update pos sets
            pos1[pos1 == i] = j
            pos0[pos0 == j] = i
            if new_score < best_score:
                best_score = new_score
                best_b = b.copy()
        else:
            # Revert
            b[i], b[j] = b[j], b[i]
        
        T *= cool
    
    return best_b, best_score

def two_point_search(h, n_tries, rng, is_dead):
    """Simple two-point exchange search (kept as fallback)."""
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [t_lo + 0.5 * span, t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + span / 2.618, t_lo + 2 * span / 2.618]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-12 or va > 1 + 1e-12 or vb < -1e-12 or vb > 1 + 1e-12:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = score_exact(best)
    return best, best_sc

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def interp_to(h_src, n_tgt):

    h_int = np.interp(tgt_c, src_c, h_src)

            best[b] = np.clip(best[b] - best_t, 0.0, 1.0)
            best_sc = score_exact(best)
    return best, best_sc


def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)


def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(12.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        n = 60
        x = (np.arange(n) + 0.5) * (2.0 / n)
        best_h = make_feasible(0.5 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 3 * x))
        best_score = score_exact(best_h)

    # Phase 1: Simple transforms
    if time.time() < deadline:
        h_t = anti_sym_map(best_h)
        sc_t = score_exact(h_t)
        if sc_t < best_score - 1e-12:
            best_h = h_t
            best_score = sc_t

    # Phase 2: Binary SA
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_steps = min(30000, int(remaining * 0.10 * 400))
        b_best, b_score = binary_sa(best_h, n_steps, rng, deadline)
        if b_best is not None:
            h_b = make_feasible(b_best)
            sc_b = score_exact(h_b)
            if sc_b < best_score - 1e-12:
                best_h = h_b
                best_score = sc_b
            h_ba = anti_sym_map(b_best)
            sc_ba = score_exact(h_ba)
            if sc_ba < best_score - 1e-12:
                best_h = h_ba
                best_score = sc_ba

    # Phase 3: Main LP refinement
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_iter = min(250, int(remaining * 2.5))
        best_h, best_score = lp_refine(best_h, n_iter, deadline, top_k=28, threshold=0.004)

    # Phase 4: Two-point polish
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_tries = min(50000, int(remaining * 0.12 * 400))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Phase 5: Resolution scanning
    if time.time() < deadline:
        n_base = len(best_h)
        n_opts = set()
        for dn in [-8, -4, 4, 8, 12]:
            nt = n_base + dn
            if 24 <= nt <= 100:
                n_opts.add(nt)
        for nc in [36, 44, 50, 56, 64, 72]:
            if 24 <= nc <= 100:
                n_opts.add(nc)
        n_opts = sorted(n_opts)
        for n_opt in n_opts:
            if time.time() >= deadline:
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int
                best_score = sc_int
            elif sc_int < best_score + 0.004:
                if time.time() < deadline:
                    h_new, sc_new = lp_refine(h_int, 40, deadline, top_k=24, threshold=0.004)
                    if sc_new < best_score - 1e-12:
                        best_h = h_new
                        best_score = sc_new

    # Phase 6: More LP from best
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_iter = min(300, int(remaining * 3))
        best_h, best_score = lp_refine(best_h, n_iter, deadline, top_k=32, threshold=0.003)

    # Phase 7: Random perturbations + LP
    if time.time() < deadline:
        n_best = len(best_h)
        for sigma in [0.03, 0.06, 0.10]:
            if time.time() >= deadline:
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n_best))
            h_new, sc = lp_refine(h_pert, 60, deadline, top_k=28, threshold=0.004)
            if sc < best_score - 1e-12:
                best_h = h_new
                best_score = sc

    # Phase 8: Final two-point polish
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_tries = min(40000, int(remaining * 0.25 * 400))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Phase 9: Final LP
    if time.time() < deadline:
        best_h, best_score = lp_refine(best_h, 40, deadline, top_k=36, threshold=0.002)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)) or final_h.min() < -1e-9 or final_h.max() > 1.0 + 1e-9:
        n_fb = 60
        final_h = np.full(n_fb, 0.5)
        final_h = make_feasible(final_h)
        final_score = score_exact(final_h)
        final_n = n_fb

    return final_h, final_score, final_n
