```python
    n = len(h0)
    h = _project(h0)

        cv = _corr(h)
        # Get top m active lags
        idx = np.argsort(cv)[::-1][:top_m]

        # s.t. t >= F_s for s in idx, sum(h) = n/2, 0<=h<=1, 0<=t<=1

            t_v = z[n]

            return t_v - cv_v[idx]
```

F_s(h) = dx * sum_i h[i] * (1 - h[i+s]) = dx * (sum_i h[i] - sum_i h[i]*h[i+s])

With the constraint sum(h) = n/2 (fixed), this becomes:
F_s(h) = dx * (n/2 - sum_i h[i]*h[i+s])

Minimizing max_s F_s is equivalent to minimizing max_s (n/2 - sum_i h[i]*h[i+s]) * dx, which is equivalent to maximizing min_s sum_i h[i]*h[i+s].

So the problem is:
maximize min_s sum_i h[i]*h[i+s]
subject to sum(h) = n/2, 0 <= h <= 1

The inner term sum_i h[i]*h[i+s] is a quadratic form. If h is restricted to {0,1}, this is the cross-correlation of a binary sequence, and maximizing the minimum cross-correlation is a classic combinatorial problem (related to Barker sequences, etc.).

For continuous h, the quadratic form h^T Q_s h is not convex or concave in general (Q_s is a circulant-like matrix with 1s on the s-th diagonal). So the problem remains nonconvex.

However, if we fix one set of variables and optimize over the other, we get a convex problem. This suggests an **alternating optimization** approach:

- Then fix h[1], h[3], h[5], ... and optimize h[0], h[2], h[4], ...
- Repeat

Each subproblem is:
maximize min_s (fixed_part_s + variable_part_s)
subject to sum(variable) = target_variable, 0 <= variable <= 1

This is still a minimax problem, but with fewer variables. It might be easier to solve.

```python
import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def _score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
            if abs(res) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, beta):
    cv = _corr(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    cv = _corr(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            g[:m] += dx * wk * (1.0 - x[s:s+m])
            g[s:s+m] -= dx * wk * x[:m]
        else:
            r = -s
            m = n - r
            g[r:r+m] += dx * wk * (1.0 - x[:m])
            g[:m] -= dx * wk * x[r:r+m]
    return g

def _eq_sum(z):
    return z.sum() - len(z) / 2.0

def _eq_sum_jac(z):
    return np.ones(len(z))

def _slsqp_lse(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_sum, 'jac': _eq_sum_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _act_obj(z):
    n = _G['n']
    return z[n] + 0.5 * _G['rho'] * np.sum((z[:n] - _G['center'])**2)

def _act_obj_jac(z):
    n = _G['n']
    g = np.zeros(n + 1)
    g[n] = 1.0
    g[:n] = _G['rho'] * (z[:n] - _G['center'])
    return g

def _act_ineq(z):
    n = _G['n']
    idx = _G['idx']
    cv = _corr(z[:n])
    return z[n] - cv[idx]

def _act_ineq_jac(z):
    n = _G['n']
    idx = _G['idx']
    h_v = z[:n]
    dx = 2.0 / n
    jac = np.zeros((len(idx), n + 1))
    for ii, k in enumerate(idx):
        s = k - (n - 1)
        jac[ii, n] = 1.0
        if s >= 0:
            m = n - s
            jac[ii, 0:m] -= dx * (1.0 - h_v[s:s+m])
            jac[ii, s:s+m] += dx * h_v[0:m]
        else:
            r = -s
            m = n - r
            jac[ii, r:r+m] -= dx * (1.0 - h_v[0:m])
            jac[ii, 0:m] += dx * h_v[r:r+m]
    return jac

def _act_eq(z):
    n = _G['n']
    return z[:n].sum() - n / 2.0

def _act_eq_jac(z):
    n = _G['n']
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _active_set_solve(h0, center, rho, top_m, max_rounds):
    n = len(h0)
    h = _project(h0)
    _G['n'] = n
    _G['center'] = center
    _G['rho'] = rho
    for _ in range(max_rounds):
        cv = _corr(h)
        idx = np.argsort(cv)[::-1][:top_m]
        _G['idx'] = idx
        z0 = np.concatenate([h, [cv.max() + 0.01]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        try:
            res = minimize(_act_obj, z0, jac=_act_obj_jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[
                               {'type': 'ineq', 'fun': _act_ineq, 'jac': _act_ineq_jac},
                               {'type': 'eq', 'fun': _act_eq, 'jac': _act_eq_jac}
                           ],
                           options={'maxiter': 250, 'ftol': 1e-14})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                h_new = _project(res['x'][:n])
                if _score(h_new) < _score(h) - 1e-14:
                    h = h_new
                else:
                    break
        except Exception:
            break
    return h

def _fista(h0, beta, n_iter, alpha_init, mu):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1.0
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        h_new = _project(y - alpha_init * grad + mu * mu * (h_cur - h_prev))
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = _project(h_new + (t_prev - 1.0) / t * (h_new - h_cur))
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_cur - h_avg) / (it + 1)
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
            if it > 0:
                sa = _score(_project(h_avg))
                if sa < best_s:
                    best_s = sa
                    best_h = _project(h_avg).copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    sa = _score(_project(h_avg))
    if sa < best_s:
        best_s = sa
        best_h = _project(h_avg).copy()
    return best_h, best_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _binary_push(h, p):
    return _project(h ** p)

def _gen_starts(h_base, n, rng):
    starts = [h_base.copy(), _mirror(h_base)]
    for sigma in [0.05, 0.1, 0.2]:
        starts.append(_project(h_base + sigma * rng.standard_normal(n)))
    for p in [1.5, 2.0, 3.0]:
        starts.append(_binary_push(h_base, p))
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 6.28)
        pert = 0.08 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
        starts.append(_project(h_base + pert))
    # Symmetrized
    starts.append(_project(0.5 * (h_base + h_base[::-1])))
    return starts

def _chain_optimize(h0, deadline):
    n = len(h0)
    cur = _project(h0)
    cur_s = _score(cur)

    # LSE stage 1: moderate beta
    if time.time() < deadline - 20:
        new = _slsqp_lse(cur, beta=60, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns

    # LSE stage 2: high beta
    if time.time() < deadline - 20:
        new = _slsqp_lse(cur, beta=300, maxiter=120)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns

    # Active-set stage
    if time.time() < deadline - 15:
        center = cur.copy()
        new = _active_set_solve(cur, center, rho=1e-4, top_m=5, max_rounds=4)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns

    # Active-set with larger active set
    if time.time() < deadline - 15:
        center = cur.copy()
        new = _active_set_solve(cur, center, rho=1e-5, top_m=8, max_rounds=3)
        ns = _score(new)

    # FISTA polishing

    # Final LSE polish
    if time.time() < deadline - 8:
        new = _slsqp_lse(cur, beta=1000, maxiter=100)
```
