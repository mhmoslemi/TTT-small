    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try random pairs
        for attempt in range(200):
            if time.time() > deadline:
                break
            i = int(np.random.randint(n))
            j = int(np.random.randint(n))
            if i == j:
                continue
            if cur_h[i] <= 1e-6 or cur_h[j] >= 1.0 - 1e-6:
                continue
            # Try transferring from j to i
            max_t = min(1.0 - cur_h[i], cur_h[j])
            if max_t < 1e-8:
                continue
            # Try several step sizes
            for frac in [0.2, 0.5, 0.8, 1.0]:
                t = frac * max_t
                cur_h[i] += t
                cur_h[j] -= t
                c5_t = _exact_c5(cur_h)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_h = cur_h.copy()
                    improved = True
                    break
                else:
                    cur_h[i] -= t
                    cur_h[j] += t
        
        if not improved:
            # Try more structured approach: target active lags
            scores = _scores(best_h)
            top_lag = int(np.argmax(scores))
            lag = top_lag - (n - 1)
            if lag >= 0:
                # score = sum h[j+lag]*(1-h[j]), want to reduce
                # Reduce h[j+lag] or increase h[j]
                # Find good pairs
                for j in range(n - lag):
                    if cur_h[j + lag] > 0.1 and cur_h[j] < 0.9:
                        t = 0.1 * min(cur_h[j + lag], 1.0 - cur_h[j])
                        cur_h[j + lag] -= t
                        cur_h[j] += t
                        c5_t = _exact_c5(cur_h)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = cur_h.copy()
                            improved = True
                            break
                        else:
                            cur_h[j + lag] += t
                            cur_h[j] -= t
                    if time.time() > deadline:
                        break
            if not improved:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _exact_c5(h):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

    hi = float(v.max()) + 2.0

            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))

        else:

    return y[-1]

    g[-1] = 1.0
    return g

class _EpiCon:
    def __init__(self, n, active_idx):
        self.n = n
        self.active_idx = active_idx
    def fun(self, y):

        return y[n] - _scores(y[:n])[self.active_idx]
    def jac(self, y):
        n = self.n
        h_y = y[:n]
        K = len(self.active_idx)
        J = np.zeros((K, n + 1))
        for k in range(K):
            p = int(self.active_idx[k])
            lag = p - (n - 1)
            if lag >= 0:
                m = n - lag
                J[k, :m] -= h_y[lag:lag + m]
                J[k, lag:] += 1.0 - h_y[:m]
            elif lag < 0:

                J[k, g:] -= h_y[:m]

    if time.time() < t0 + budget_s * 0.92:
        d5 = t0 + budget_s * 0.94
        for dn in [-4, -2, -1, 1, 2, 3, 4, 6, 8]:
            if time.time() > d5:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 200:
                continue
            h_r = _resample(best_h, n_try)
            h_r = _slsqp_one(h_r, min(d5, time.time() + 18), top_k=25, maxiter=120)
            c5_r = _exact_c5(h_r)
            if c5_r < best_c5 - 1e-13:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    # Phase 6: Final SLSQP polish
    if time.time() < final_dl:
        h_f = _slsqp_one(best_h, final_dl, top_k=50, maxiter=300, ftol=1e-12)
        c5_f = _exact_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
