import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_and_args(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find all indices where corr == max_val (within tolerance)
    idx = np.where(corr >= max_val - 1e-12)[0]
    return float(max_val * dx), idx

def _subgradient(h, active_indices):
    """Compute subgradient of C5 at h for the given active (tied max) indices."""
    n = len(h)
    grad = np.zeros(n)
    for r in active_indices:
        d = r - (n - 1)  # actual shift
        if d >= 0:
            # S_d = sum_{i=0}^{n-1-d} h[i] * (1 - h[i+d])
            m = n - d
            if m > 0:
                grad[:m] += (1.0 - h[d:])       # derivative of h[i] w.r.t. S_d
                grad[d:] += (-h[:m])             # derivative of (1-h[i+d]) w.r.t. h[i+d]
        else:
            g = -d
            m = n - g
            if m > 0:
                # S_{-g} = sum_{i=0}^{n-1-g} h[i+g] * (1-h[i])
                grad[g:] += (1.0 - h[:m])        # derivative of h[i+g] w.r.t. h[i+g]
                grad[:m] += (-h[g:])             # derivative of (1-h[i]) w.r.t. h[i]
    return grad

def _project_h(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    target = min(max(target, 0.0), float(n))
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binary_swap_search(h_init, n, target, deadline, rng, iters=50000):
    """Simulated annealing on binary configurations."""
    # Start from a binary projection of h_init
    h = (h_init >= 0.5).astype(float)
    # Fix to have exactly target ones
    num_ones = int(np.sum(h))
    if num_ones > target:
        ones = np.where(h == 1)[0]
        vals = h_init[ones]
        order = np.argsort(vals)  # remove the lowest-valued ones
        for idx in order[:num_ones - target]:
            h[idx] = 0
    elif num_ones < target:
        zeros = np.where(h == 0)[0]
        vals = h_init[zeros]
        order = np.argsort(-vals)  # add the highest-valued zeros
        for idx in order[:target - num_ones]:
            h[idx] = 1
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    temperature = 0.5
    cool_rate = 0.9995
    
    for it in range(iters):
        if time.time() > deadline:
            break
        temperature *= cool_rate
        if temperature < 1e-8:
            break
        
        # Random swap: pick a 1 and a 0, swap them
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_c5 = _exact_c5(h)
        
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / temperature):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def _subgradient_descent(h_init, deadline, lr_init=0.05, momentum=0.9, max_iters=20000):
    """Subgradient descent with Polyak-type step size and momentum."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Track best for Polyak step size
    best_c5_so_far = best_c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c5, active_idx = _c5_and_args(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        g = _subgradient(h, active_idx)
        
        # Normalize subgradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Polyak step size: t_k = (f_k - f*) / ||g_k||^2
        # We use best_c5 as f*
        gap = max(c5 - best_c5, 1e-10)
        t = lr_init * gap / (g_norm * g_norm + 1e-10)
        t = min(t, 0.1)  # cap step size
        
        # Momentum update
        v = momentum * v - t * g
        h = h + v
        h = _project_h(h, target)
        
        # Occasional re-centering
        if it % 500 == 0:
            c5_check = _exact_c5(h)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_h(h_new, n_new / 2.0)

def _lp_polish(h, deadline, max_iters=2000):
    """LP-based polish (similar to parent but simplified)."""
    from scipy.optimize import linprog
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_score = np.max(scores)
        
        # Build LP: minimize t subject to G @ p + max_score*p <= max_score + t*radius
        # Actually simplify: minimize max_d (current_score_d + G_d @ p)
        # where G_d is the gradient of score_d
        
        # Use the active set approach
        active = np.where(scores >= max_score - 1e-8)[0]
        
        # LP: min t s.t. for each active r: sum_i G[r,i] * p_i <= t
        #     subject to -radius <= p_i <= radius, sum p_i = 0
        #     0 <= h_i + p_i <= 1
        
        G = np.zeros((len(active), n))
        for idx, r in enumerate(active):
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    G[idx, :m] += (1.0 - h[d:])
                    G[idx, d:] += -h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    G[idx, g:] += (1.0 - h[:m])
                    G[idx, :m] += -h[g:]
        
        # LP formulation
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize t
        
        A_ub = np.zeros((len(active), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = np.zeros(len(active))
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                if np.max(np.abs(p)) > 1e-12:
                    # Line search
                    alpha_max = 0.5
                    for j in range(n):
                        if p[j] > 1e-14:
                            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                        elif p[j] < -1e-14:
                            alpha_max = min(alpha_max, h[j] / (-p[j]))
                    alpha_max = max(alpha_max, 1e-8)
                    
                    best_a = 0
                    best_c5_new = best_c5
                    for a in np.linspace(0, alpha_max, 20):
                        h_t = h + a * p
                        c5_t = _exact_c5(h_t)

                    if best_a > 0 and best_c5_new < best_c5 - 1e-12:

                        cur = _exact_c5(h)
                        if cur < best_c5:

                        stall = 0

            else:

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_with_active(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr))
    active = np.where(corr >= max_val - 1e-10)[0]
    return max_val * dx, active, corr

def _subgradient(h, active):
    n = len(h)
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += (1.0 - h[d:d + m])
                grad[d:] += -h[:m]
        else:
            m = -d
            mm = n - m
            if mm > 0:
                grad[m:] += (1.0 - h[:mm])
                grad[:mm] += -h[m:]
    return grad

def _project_h(v, target):

    target = max(0.0, min(float(n), target))
    if target <= 1e-15:
        return np.zeros(n)
    if target >= n - 1e-15:

    hi = float(np.max(v)) + 1.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_h(h_new, n_new / 2.0)

def _block_shuffle(h, rng, n_blocks):
    n = len(h)

    blocks = [h[i*bs:(i+1)*bs] for i in range(n_blocks)]

    if n_blocks * bs < n:
        h_new = np.concatenate(blocks + [h[n_blocks*bs:]])
    return _project_h(h_new, n / 2.0)
