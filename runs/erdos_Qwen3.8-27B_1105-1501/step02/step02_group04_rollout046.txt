def soft_objective_and_grad(h, n, T):
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')  # length 2n-1
    max_c = np.max(corr)
    # Softmax weights
    z = np.exp((corr - max_c) / T)
    w = z / np.sum(z)
    value = dx * (max_c + T * np.log(np.sum(z)))
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        s = m - (n - 1)
        # First term: (1-h[p-s]) for p in [max(0,s), min(n-1, s+n-1)]
        p_lo = max(0, s)
        p_hi = min(n - 1, s + n - 1)
        if p_lo <= p_hi:
            idx = np.arange(p_lo, p_hi + 1)
            grad[idx] += wm * (1.0 - h[idx - s])
        # Second term: -h[p+s] for p in [max(0,-s), min(n-1, n-1-s)]
        p_lo2 = max(0, -s)
        p_hi2 = min(n - 1, n - 1 - s)
        if p_lo2 <= p_hi2:
            idx = np.arange(p_lo2, p_hi2 + 1)
            grad[idx] -= wm * h[idx + s]
    
    grad *= dx
    return value, grad
