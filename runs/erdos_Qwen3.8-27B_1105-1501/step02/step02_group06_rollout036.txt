def softmax_descent_step(h, temperature, top_frac=1.0):
    """One step of softmax gradient descent with exact line search"""
    n = len(h)
    dx = 2.0/n
    q = 1.0 - h
    raw = np.correlate(h, q, mode='full')
    P = len(raw)
    f_max = raw.max()
    
    # Compute weights
    weights = np.exp(-(raw - f_max) / temperature)
    # Only keep top fraction
    threshold = np.percentile(raw, (1-top_frac)*100)
    weights[raw < threshold] = 0
    if weights.sum() < 1e-10:
        return h, official_score(h)
    weights /= weights.sum()
    
    # Compute weighted gradient
    # g for shift s: g[j] = (1-h[j+s])*I(j+s valid) - h[j-s]*I(j-s valid)
    # Sum over s of w_s * g_s
    d = np.zeros(n)
    for p in range(P):
        s = p - (n-1)  # shift
        w = weights[p]
        if w < 1e-12:
            continue
        # Gradient of f_s
        if s >= 0:
            # j+s valid for j in [0, n-1-s]
            d[0:n-s] += w * (1-h[s:n])
            # j-s valid for j in [s, n-1]
            d[s:] -= w * h[0:n-s]
        else:
            t = -s
            # j+t valid for j in [0, n-1-t]
            d[0:n-t] += w * (1-h[t:n])
            # j-t valid for j in [t, n-1]
            d[t:] -= w * h[0:n-t]
    
    # Project to sum=0
    d -= d.mean()
    
    # Scale direction
    d_scale = np.max(np.abs(d))

    d = d / d_scale * 0.5  # scale to reasonable step

    alpha_max = feasible_alpha_max(h, d)
    if alpha_max < 1e-12:
        return h, official_score(h)

        else:

    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:

            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:

            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def feasible_alpha_max(h, d):
    """Find max alpha such that h + alpha*d stays in [0,1]"""
    alpha_max = 1e10
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))
    return max(0.0, alpha_max)

def exact_line_search(h, d):
    """Find best alpha in [0, alpha_max] for direction d"""
    n = len(h)
    q = 1.0 - h
    raw = np.correlate(h, q, mode='full')
    P = len(raw)
    
    alpha_max = feasible_alpha_max(h, d)
    if alpha_max < 1e-12:
        return 0.0, h
    
    # Directional derivatives
    gd = np.correlate(d, q, mode='full')
    hd = np.correlate(h, d, mode='full')
    g = gd - hd
    
    # Check if direction is descent
    # At alpha=0, active set is ArgMax(raw)
    f_max = raw.max()
    active = raw >= f_max - 1e-10
    if np.min(g[active]) > 0:
        # All active constraints have positive derivative → F is increasing
        return 0.0, h
    
    # Find candidates: endpoints + breakpoints
    # For efficiency, only consider breakpoints in (0, alpha_max)
    best_alpha = 0.0
    best_val = float(np.max(raw))  # F(0)
    
    if best_val - (-np.inf) < 1e-15:  # always true, just for clarity
        pass
    
    # Evaluate at alpha_max
    val_end = np.max(raw + alpha_max * g)
    if val_end < best_val:
        best_val = val_end
        best_alpha = alpha_max
    
    # Breakpoints (only if P is small enough)
    if P <= 250:
        for i in range(P):
            for j in range(i+1, P):
                dg = g[i] - g[j]
                if abs(dg) > 1e-14:
                    a = (raw[j] - raw[i]) / dg
                    if 1e-10 < a < alpha_max - 1e-10:
                        val = np.max(raw + a * g)
                        if val < best_val - 1e-15:
                            best_val = val
                            best_alpha = a
    
    return best_alpha, h + best_alpha * d

def softmax_gradient(h, temperature):
    """Compute weighted negative gradient direction"""
    n = len(h)
    q = 1.0 - h
    raw = np.correlate(h, q, mode='full')
    P = len(raw)
    f_max = raw.max()
    
    # Softmax weights
    shifted = (raw - f_max) / temperature
    shifted = np.clip(shifted, -50, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-15:
        return None
    weights /= wsum
    
    # Compute weighted gradient: sum_s w_s * grad_f_s
    # grad_f_s[j] = (1-h[j+s])*I(j+s valid) - h[j-s]*I(j-s valid)
    # Using correlate:
    # For shift s (p = s + n - 1):
    #   grad = correlate(e_s, q, 'full') - correlate(q, e_s, 'full') ???
    # Actually, let's use the direct formula:
    # sum_s w_s * [(1-h[j+s])*I - h[j-s]*I]
    # = correlate(w_shifted, 1-h, 'full') - correlate(1-h, w_shifted, 'full') ???
    # Hmm, this is getting complicated. Let me just use the direct loop.
    
    # Actually, the gradient of f_s = sum_{valid} h[i]*(1-h[i+s]) w.r.t. h[j] is:
    #   (1-h[j+s]) if j+s is valid, minus h[j-s] if j-s is valid
    # 
    # Sum over s of w_s * grad_s[j] = sum_s w_s*(1-h[j+s])*I(j+s valid) - sum_s w_s*h[j-s]*I(j-s valid)
    # = sum_{t} w_{t-j}*(1-h[t])*I(t valid, t-j valid) - sum_{t} w_{j-t}*h[t]*I(t valid, j-t valid)
    # 
    # This is a cross-correlation of w with (1-h) and h!
    # Specifically: first term = (w * (1-h))[cross-corr at offset j]
    #            second term = ((1-h) * w)[cross-corr at offset j]... 
    # 
    # Let me think more carefully.
    # shift s corresponds to index p = s + (n-1) in the correlate output.
    # w_s is the weight for shift s.
    # First term at position j: sum_s w_s * (1-h[j+s]) * I(0≤j+s<n)
    #   = sum_{t=0}^{n-1} w_{t-j} * (1-h[t]) * I(0≤t-j<n)
    #   where t = j+s, so s = t-j
    #   = cross-correlation of w (indexed by s) with (1-h) (indexed by t) evaluated at j
    #   = np.correlate(1-h, w_padded, 'full')[j + offset]
    #
    # This is getting complex. For n≈60 and P≈119, a direct loop is fine.
    
    grad = np.zeros(n)
    for p in range(P):
        s = p - (n - 1)
        w = weights[p]
        if w < 1e-12:
            continue
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
        else:
            t = -s
            grad[:n-t] += w * (1.0 - h[t:])
            grad[t:] -= w * h[:n-t]
    
    # Negative gradient, projected to sum=0
    d = -grad
    d -= d.mean()
    
    # Scale
    dnrm = np.max(np.abs(d))
    if dnrm < 1e-15:
        return None
    d = d / dnrm
    
    return d

def qp_step_exact(h0, k_active, reg):
    """QP step with exact line search instead of accepting full step"""
    n = len(h0)
    vals = np.correlate(h0, 1.0 - h0, mode='full') * (2.0/n)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        # Linearize f_s at h0
        q = 1.0 - h0
        if s >= 0:
            g = np.zeros(n)
            g[:n-s] += (1.0 - h0[s:])
            g[s:] -= h0[:n-s]
        else:
            t = -s
            g = np.zeros(n)
            g[:n-t] += (1.0 - h0[t:])
            g[t:] -= h0[:n-t]
        # f_s(h) ≈ f_s(h0) + g^T (h - h0)
        # constraint: f_s(h0) + g^T(h-h0) <= t_var/dx
        constraints.append(vals[p]/dx + (g @ h_var) - (g @ h0) <= t_var)
    
    obj = cp.Minimize(t_var + reg * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None
    
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None, None
    
    h_new = make_feasible(h_new)
    sc = official_score(h_new)
    return h_new, sc

def qp_descent(h0, k_values, reg_schedule):
    """QP descent with exact scores"""
    h = h0.copy()
    best = official_score(h)
    
    for k in k_values:
        for reg in reg_schedule:
            h_new, sc = qp_step_exact(h, k, reg)
            if h_new is not None and sc < best - 1e-14:
                h = h_new
                best = sc
    
    return h, best

import numpy as np
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def feasible_alpha_max(h, d):
    alpha_max = 1e10
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    return max(0.0, alpha_max)

def exact_line_search(h, d):
    n = len(h)
    q = 1.0 - h
    raw = np.correlate(h, q, mode='full')
    P = len(raw)
    alpha_max = feasible_alpha_max(h, d)
    if alpha_max < 1e-12:
        return 0.0, h
    gd = np.correlate(d, q, mode='full')
    hd = np.correlate(h, d, mode='full')
    g = gd - hd
    f0 = float(raw.max())
    # Check if descent direction
    active = raw >= f0 - 1e-10
    if np.min(g[active]) > 1e-12:
        return 0.0, h
    # Vectorized breakpoint evaluation
    best_val = f0
    best_alpha = 0.0
    # Check endpoint
    val_end = float(np.max(raw + alpha_max * g))
    if val_end < best_val:
        best_val = val_end
        best_alpha = alpha_max
    # O(P^2) breakpoints - vectorized
    raw_diff = raw[None, :] - raw[:, None]
    g_diff = g[:, None] - g[None, :]
    valid = np.abs(g_diff) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        alphas_2d = raw_diff / g_diff
    alphas_2d[~valid] = -1.0
    mask = (alphas_2d > 1e-10) & (alphas_2d < alpha_max - 1e-10)
    candidate_alphas = alphas_2d[mask]
    if len(candidate_alphas) > 0:
        # Limit to avoid excessive memory for very large P
        if len(candidate_alphas) > 20000:
            candidate_alphas = candidate_alphas[:20000]
        vals = raw[None, :] + candidate_alphas[:, None] * g[None, :]
        max_vals = vals.max(axis=1)
        best_idx = int(np.argmin(max_vals))
        if max_vals[best_idx] < best_val:
            best_val = max_vals[best_idx]
            best_alpha = float(candidate_alphas[best_idx])
    if best_alpha < 1e-12:
        return 0.0, h
    return best_alpha, h + best_alpha * d

def softmax_gradient(h, temperature):
    n = len(h)
    q = 1.0 - h
    raw = np.correlate(h, q, mode='full')
    P = len(raw)
    f_max = raw.max()
    shifted = np.clip((raw - f_max) / temperature, -50.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-15:
        return None
    weights /= wsum
    grad = np.zeros(n)
    for p in range(P):
        s = p - (n - 1)
        w = weights[p]
        if w < 1e-12:
            continue
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            t = -s
            grad[:n - t] += w * (1.0 - h[t:])
            grad[t:] -= w * h[:n - t]
    d = -grad
    d -= d.mean()
    dnrm = float(np.max(np.abs(d)))
    if dnrm < 1e-15:
        return None
    return d / dnrm

def softmax_step(h, temperature, scale):
    d = softmax_gradient(h, temperature)
    if d is None:
        return h, official_score(h)
    d_scaled = d * scale
    alpha, h_prop = exact_line_search(h, d_scaled)
    if alpha < 1e-12:
        return h, official_score(h)
    h_new = make_feasible(h_prop)
    sc = official_score(h_new)
    return h_new, sc

def softmax_descent(h, temps, max_iter, scale, t_stop):
    h_cur = h.copy()
    best = official_score(h_cur)
    for temp in temps:
        if time.time() > t_stop:
            break
        for it in range(max_iter):
            if time.time() > t_stop:
                break
            h_new, sc = softmax_step(h_cur, temp, scale)
            if sc < best - 1e-14:
                h_cur = h_new
                best = sc
            else:
                break
    return h_cur, best

def subgradient_step(h, scale, rng):
    n = len(h)
    q = 1.0 - h
    raw = np.correlate(h, q, mode='full')
    P = len(raw)
    f_max = raw.max()
    active = np.where(raw >= f_max - 1e-10)[0]
    s_choice = active[rng.integers(len(active))]
    s = int(s_choice - (n - 1))
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        t = -s
        grad[:n - t] += (1.0 - h[t:])
        grad[t:] -= h[:n - t]
    d = -grad
    d -= d.mean()
    dnrm = float(np.max(np.abs(d)))
    if dnrm < 1e-15:
        return h, official_score(h)
    d = d / dnrm * scale
    alpha, h_prop = exact_line_search(h, d)
    if alpha < 1e-12:
        return h, official_score(h)
    h_new = make_feasible(h_prop)
    sc = official_score(h_new)
    return h_new, sc

def subgradient_descent(h, max_iter, scale, rng, t_stop):
    h_cur = h.copy()
    best = official_score(h_cur)
    for it in range(max_iter):
        if time.time() > t_stop:
            break
        h_new, sc = subgradient_step(h_cur, scale, rng)
        if sc < best - 1e-14:
            h_cur = h_new
            best = sc
        else:
            # Try smaller scale
            break
    return h_cur, best

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym(h):
    return make_feasible(1.0 - h[::-1].copy())

def qp_step(h0, k_active, reg):
    n = len(h0)
    q = 1.0 - h0
    raw = np.correlate(h0, q, mode='full')
    dx = 2.0 / n
    vals = raw * dx
    top_idx = np.argsort(vals)[-k_active:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[:n - s] += (1.0 - h0[s:])
            g[s:] -= h0[:n - s]
        else:
            t = -s
            g[:n - t] += (1.0 - h0[t:])
            g[t:] -= h0[:n - t]
        lhs = raw[p] + g @ (h_var - h0)
        constraints.append(lhs <= t_var / dx)
    obj = cp.Minimize(t_var + reg * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None, None
    h_new = make_feasible(h_new)
    sc = official_score(h_new)
    return h_new, sc

def qp_descent(h0, k_values, reg_schedule, t_stop):
    h = h0.copy()
    best = official_score(h)
    for k in k_values:
        if time.time() > t_stop:
            break
        for reg in reg_schedule:
            if time.time() > t_stop:
                break
            h_new, sc = qp_step(h, k, reg)
            if h_new is not None and sc < best - 1e-14:
                h = h_new
                best = sc
    return h, best
