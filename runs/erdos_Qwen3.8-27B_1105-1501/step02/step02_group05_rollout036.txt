```python
    h_int = np.interp(tgt_c, src_c, h_src)
```

```python
def solve_lp(direction_matrix, c_active, h, n_active):
    """Solve the minimax LP.
    direction_matrix: (n_active, n) - gradients for active lags (already * dx)
    c_active: (n_active,) - current lag values for active lags
    h: (n,) - current solution
    Returns: d (n,) or None if infeasible
    """
    n = len(h)
    
    # x = [t, d_1, ..., d_n], dimension n+1
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize t
    
    # A_ub: n_active rows
    # For each active p: G_p d - t ≤ -c_p
    # [-1, G_p[0], ..., G_p[n-1]] @ x ≤ -c_p
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = direction_matrix
    b_ub = -c_active
    
    # A_eq: 1 row (mass constraint)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0  # sum of d = 0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(None, None)]  # t is free
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[1:]
        if np.all(np.isfinite(d)):
            return d
    return None
```

```python
from scipy.optimize import linprog, minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def compute_lag_grads(h):
    n = len(h)
    npts = 2 * n - 1
    grads = np.zeros((npts, n))
    for p in range(npts):
        s = p - (n - 1)
        if s >= 0:
            for j in range(n - s):
                grads[p, j] += (1.0 - h[j + s])
                grads[p, j + s] -= h[j]
        else:
            t = -s
            for j in range(n - t):
                grads[p, j + t] += (1.0 - h[j])
                grads[p, j] -= h[j + t]
    return grads

def solve_minimax_lp(G, c_active, h):
    """Solve: min t s.t. G@d + c_active <= t, sum(d)=0, -h<=d<=1-h.
    G: (m, n) gradients
    c_active: (m,) current values
    h: (n,) current solution
    Returns d: (n,) or None
    """
    n = len(h)
    m = len(c_active)
    
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = -c_active
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-h[i], 1.0 - h[i]) for i in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        d = res.x[1:]
        if np.all(np.isfinite(d)):
            return d
    return None

def lp_active_set_optimize(h, max_steps=100, time_fn=None):

    n = len(h)

    best_h = h.copy()

def two_point_search(h, n_tries, rng, time_fn):
    """Simple two-point exchange search."""
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    
    for it in range(n_tries):
        if time_fn():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13:
                continue
            if val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            tmp[a] = val_a
            tmp[b] = val_b
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = max(0.0, min(1.0, best[a] + best_t))
            best[b] = max(0.0, min(1.0, best[b] - best_t))
            best_sc = official_score(best)
    
    return best, best_sc
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def compute_lag_grads(h):
    n = len(h)
    npts = 2 * n - 1
    grads = np.zeros((npts, n))
    for p in range(npts):
        s = p - (n - 1)
        if s >= 0:
            for j in range(n - s):
                grads[p, j] += (1.0 - h[j + s])
                grads[p, j + s] -= h[j]
        else:
            t = -s
            for j in range(n - t):
                grads[p, j + t] += (1.0 - h[j])
                grads[p, j] -= h[j + t]
    return grads

def solve_minimax_lp(G, c_active, h):
    n = len(h)
    m = len(c_active)
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = -c_active
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-h[i], 1.0 - h[i]) for i in range(n)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        d = res.x[1:]
        if np.all(np.isfinite(d)):
            return d, float(res.fun)
    return None, None

def lp_step_full(h):
    n = len(h)
    dx = 2.0 / n
    c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    all_grads = compute_lag_grads(h) * dx
    d, t_opt = solve_minimax_lp(all_grads, c_vals, h)
    return d, c_vals, t_opt

def lp_step_growing(h, max_grow=8):
    n = len(h)
    dx = 2.0 / n
    c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    all_grads = compute_lag_grads(h) * dx
    npts = 2 * n - 1
    order = np.argsort(c_vals)[::-1]
    active_set = order[:min(3, npts)].tolist()
    d = None
    for _ in range(max_grow):
        active_arr = np.array(active_set)
        G = all_grads[active_arr]
        c_act = c_vals[active_arr]
        d, t_opt = solve_minimax_lp(G, c_act, h)
        if d is None:
            break
        in_active = np.zeros(npts, dtype=bool)
        in_active[active_set] = True
        pred = c_vals + all_grads @ d
        pred_inactive = pred[~in_active]
        if pred_inactive.max() > t_opt + 1e-10:
            worst_local = np.argmax(pred_inactive)
            inactive_indices = np.where(~in_active)[0]
            active_set.append(int(inactive_indices[worst_local]))
        else:
            break
    return d, c_vals

def lp_optimize(h, max_steps, time_fn, use_growing=False):
    n = len(h)
    best_h = h.copy()
    best_sc = official_score(h)
    stall_count = 0
    for step in range(max_steps):
        if time_fn():
            break
        if use_growing:
            d, c_vals = lp_step_growing(best_h)
        else:
            d, c_vals, _ = lp_step_full(best_h)
        if d is None or np.max(np.abs(d)) < 1e-10:
            stall_count += 1
            if stall_count > 3:
                break
            continue
        stall_count = 0
        improved = False
        alphas = [1.0, 0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
        for a in alphas:
            h_trial = make_feasible(best_h + a * d)
            sc = official_score(h_trial)
            if sc < best_sc - 1e-15:
                best_h = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            for a in [0.005, 0.002, 0.001]:
                h_trial = make_feasible(best_h + a * d)
                sc = official_score(h_trial)
                if sc < best_sc - 1e-15:
                    best_h = h_trial
                    best_sc = sc
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count > 5:
                    break
    return best_h, best_sc

def two_point_search(h, n_tries, rng, time_fn):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if time_fn():
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13:
                continue
            if val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            tmp[a] = val_a
            tmp[b] = val_b
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = max(0.0, min(1.0, best[a] + best_t))
            best[b] = max(0.0, min(1.0, best[b] - best_t))
            best_sc = official_score(best)
    return best, best_sc

def subgradient_polish(h, n_iter, time_fn):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    dx = 2.0 / n
    alpha = 0.02
    for it in range(n_iter):
        if time_fn():
            break
        vals = np.correlate(best, 1.0 - best, mode='full') * dx
        cmax = vals.max()
        tol = 0.03
        active = np.where(vals >= cmax - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        w = np.exp(40.0 * (vals[active] - cmax))
        w = w / w.sum()
        all_grads = compute_lag_grads(best) * dx
        grad = (w[:, None] * all_grads[active]).sum(axis=0)
        d = -grad
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25]:
            h_trial = make_feasible(best + mult * d)
            sc = official_score(h_trial)
            if sc < best_sc - 1e-14:
                best = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:
            alpha = min(alpha * 1.1, 0.1)
    return best, best_sc
```
