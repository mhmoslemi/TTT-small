def _compute_partials(h):
    """Compute ∂F_q/∂h[i] for all i, q.
    Returns a matrix of shape (n, 2n-1).
    """
    n = len(h)
    dx = 2.0 / n
    P = np.zeros((n, 2 * n - 1))
    
    for q in range(2 * n - 1):
        s = q - (n - 1)
        if s >= 0:
            # F_q = dx * sum_{j=0}^{n-1-s} h[j+s] * (1-h[j])
            for j in range(n - s):
                # ∂/∂h[j+s]: (1-h[j]) * dx
                P[j + s, q] += (1.0 - h[j]) * dx
                # ∂/∂h[j]: -h[j+s] * dx
                P[j, q] += -h[j + s] * dx
        else:
            t = -s
            F_q = dx * sum_{j=t}^{n-1} h[j] * (1-h[j-t])
            for j in range(t, n):
                P[j, q] += (1.0 - h[j - t]) * dx
                P[j - t, q] += -h[j] * dx
    
    return P

    n = len(h0)
    h = h0.copy()
