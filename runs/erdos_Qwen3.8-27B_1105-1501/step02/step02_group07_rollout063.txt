def grad_V(h, p):
    """Gradient of V_p (correlate value at index p) with respect to h."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    g = np.zeros(n)
    L = n - abs(s)
    if L <= 0:
        return g * dx
    if s >= 0:
        # V_p = sum_{i=0}^{L-1} h[i+s] * (1-h[i]) * dx
        # dV/dh[i+s] = (1-h[i]) * dx for i in [0, L-1]
        # dV/dh[i] = -h[i+s] * dx for i in [0, L-1]
        g[s:s+L] += (1.0 - h[0:L]) * dx
        g[0:L] -= h[s:s+L] * dx
    else:
        t = -s
        # V_p = sum_{i=t}^{n-1} h[i] * (1-h[i-t]) * dx
        # dV/dh[i] = (1-h[i-t]) * dx for i in [t, n-1]
        # dV/dh[i-t] = -h[i] * dx for i in [t, n-1]
        g[t:n] += (1.0 - h[0:L]) * dx
        g[0:L] -= h[t:n] * dx
    return g

    """Gradients for multiple correlate indices."""
    n = len(h)
    g = np.zeros(n)
    for p in p_list:
        g += grad_V(h, p)
    return g

    #      sum(h) = n/2
    #      0 <= h <= 1
    # Variables: h[0..n-1], t (total n+1 variables)

    c[-1] = 1.0  # minimize t
    
    A_rows = []
    b_rows = []
    for p in idx:

        # Let me redo: V[p] + gp · (h_new - h) <= t

        # So: V(h) + gp · (h_new - h) <= t

def make_feasible(z):
    n = len(z)
    target = 0.5 * n
    z = np.clip(np.asarray(z, dtype=float), -3.0, 3.0)
    lo, hi = -3.0, 3.0

            lo = mid
        else:

    if abs(res) > 1e-12:

        else:

def grad_at(h, p):

    s = p - (n - 1)
