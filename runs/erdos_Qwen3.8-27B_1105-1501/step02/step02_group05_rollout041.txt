def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15: continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15: continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)

def lse_values(h, beta):
    """Compute log-sum-exp of beta*values and the weights."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    bv = beta * vals
    bv_max = bv.max()
    exp_vals = np.exp(bv - bv_max)
    log_sum = bv_max + np.log(exp_vals.sum())
    weights = exp_vals / exp_vals.sum()
    return log_sum / beta, vals, weights

def lse_grad(h, beta):
    """Gradient of log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    bv = beta * vals
    bv_max = bv.max()
    exp_vals = np.exp(bv - bv_max)
    W = exp_vals / exp_vals.sum()  # weights, length 2n-1

    """Find optimal t for pair (a,b) using the fact that C is convex in t."""
    n = len(h)
    dx = 2.0 / n
    
    t_lo = max(-h[a], h[b] - 1.0)
    t_hi = min(1.0 - h[a], h[b])
    if t_hi - t_lo < 1e-12:
        return h, exact_score(h)
    
    tmp = h.copy()
    
    def score_at_t(t):
        tmp[a] = h[a] + t
        tmp[b] = h[b] - t
        return exact_score(tmp)
    
    # Since C(t) is convex and piecewise linear, use ternary search
    # or just evaluate at a few strategic points
    # Actually, for convex PL function, the minimum is where the subgradient contains 0.
    # We can use bisection on the "slope": 
    # left slope = C(t+eps) - C(t) over eps
    # right slope = C(t) - C(t-eps) over eps
    # Find where left slope >= 0 and right slope <= 0
    
    # Simpler: just do a fine grid + local bisection
    # For n=60, a grid of 20 points is fine
    grid = np.linspace(t_lo, t_hi, 20)
    best_t = grid[0]
    best_sc = score_at_t(grid[0])
    for t in grid[1:]:
        sc = score_at_t(t)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    # Refine with bisection around best_t
    if 0 < best_t < t_hi:  # not at boundary
        # Find interval containing the minimum
        idx = int((best_t - t_lo) / (t_hi - t_lo) * 19)
        idx = min(max(idx, 0), 18)
        lo = grid[idx]
        hi = grid[min(idx+1, 19)]
        for _ in range(30):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            sc1 = score_at_t(mid1)
            sc2 = score_at_t(mid2)
            if sc1 < sc2:
                hi = mid2
            else:
                lo = mid1
        best_t = (lo + hi) / 2
        best_sc = score_at_t(best_t)
    
    if best_sc < exact_score(h) - 1e-14:
        h[a] = h[a] + best_t
        h[b] = h[b] - best_t
        # Clip for safety
        h[a] = np.clip(h[a], 0, 1)
        h[b] = np.clip(h[b], 0, 1)
    
    return h, best_sc

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15: continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15: continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)

def lse_fun_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    bv = beta * vals
    m = bv.max()
    ev = np.exp(bv - m)
    W = ev / ev.sum()
    fval = (m + np.log(ev.sum())) / beta
    
    # Gradient
    grad = np.zeros(n)
    h_inv = 1.0 - h
    # s=0
    grad += W[n-1] * dx * (1.0 - 2.0 * h)
    # s>0
    for s in range(1, n):
        w = W[n-1+s]
        if w < 1e-15: continue
        m2 = n - s
        grad[:m2] += w * dx * h_inv[s:]
        grad[s:] -= w * dx * h[:m2]
    # s<0
    for s in range(1, n):
        w = W[n-1-s]
        if w < 1e-15: continue
        m2 = n - s
        grad[s:] += w * dx * h_inv[:m2]
        grad[:m2] -= w * dx * h[s:]
    return fval, grad

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        f, _ = lse_fun_grad(x, beta)
        return f
    def jac(x):
        _, g = lse_fun_grad(x, beta)
        return g
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            return h_new, exact_score(h_new)
    except Exception:
        pass
    return None, None

def two_point_optimal(h, a, b, n_grid=15):
    """Find optimal t for pair (a,b)."""
    t_lo = max(-h[a], h[b] - 1.0)
    t_hi = min(1.0 - h[a], h[b])
    if t_hi - t_lo < 1e-12:
        return h[a], h[b], exact_score(h)
    
    tmp = h.copy()
    
    def sc(t):
        tmp[a] = h[a] + t
        tmp[b] = h[b] - t
        return exact_score(tmp)
    
    # Coarse grid
    grid = np.linspace(t_lo, t_hi, n_grid)
    best_t = grid[0]
    best_sc = sc(grid[0])
    for i in range(1, n_grid):
        s = sc(grid[i])
        if s < best_sc:
            best_sc = s
            best_t = grid[i]
    
    # Refine with ternary search
    if best_t > t_lo + 1e-10 and best_t < t_hi - 1e-10:
        idx = min(int((best_t - t_lo) / (t_hi - t_lo) * (n_grid-1)), n_grid-2)
        lo, hi = grid[idx], grid[idx+1]
        for _ in range(25):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            s1, s2 = sc(m1), sc(m2)
            if s1 < s2:
                hi = m2
            else:
                lo = m1
        best_t = (lo + hi) / 2.0
        best_sc = sc(best_t)
    
    return h[a] + best_t, h[b] - best_t, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def make_wave(n, freqs, amplitudes, phase=0.0):

    for f, a in zip(freqs, amplititudes):
        h += a * np.sin(2 * np.pi * f * x + phase)
    return make_feasible(h)

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lse_fun_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    bv = beta * vals
    m = bv.max()
    ev = np.exp(bv - m)
    W = ev / ev.sum()
    fval = (m + np.log(ev.sum())) / beta
    grad = np.zeros(n)
    h_inv = 1.0 - h
    grad += W[n - 1] * dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        w = W[n - 1 + s]
        if w < 1e-15:
            continue
        m2 = n - s
        grad[:m2] += w * dx * h_inv[s:]
        grad[s:] -= w * dx * h[:m2]
    for s in range(1, n):
        w = W[n - 1 - s]
        if w < 1e-15:
            continue
        m2 = n - s
        grad[s:] += w * dx * h_inv[:m2]
        grad[:m2] -= w * dx * h[s:]
    return fval, grad

def _slsqp_obj(x, beta):
    f, _ = lse_fun_grad(x, beta)
    return f

def _slsqp_jac(x, beta):
    _, g = lse_fun_grad(x, beta)
    return g

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, beta, maxiter=80):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_slsqp_obj, h0, args=(beta,), jac=_slsqp_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            return h_new, exact_score(h_new)
    except Exception:
        pass
    return None, None

def two_point_optimal_step(h, a, b, n_grid=12):
    t_lo = max(-h[a], h[b] - 1.0)
    t_hi = min(1.0 - h[a], h[b])
    if t_hi - t_lo < 1e-12:
        return h[a], h[b], None
    tmp = h.copy()

    def sc(t):
        tmp[a] = h[a] + t
        tmp[b] = h[b] - t
        return exact_score(tmp)

    grid = np.linspace(t_lo, t_hi, n_grid)
    best_t = grid[0]
    best_sc = sc(grid[0])
    for i in range(1, n_grid):
        s = sc(grid[i])
        if s < best_sc:
            best_sc = s
            best_t = grid[i]
    if best_t > t_lo + 1e-10 and best_t < t_hi - 1e-10:
        idx = min(int((best_t - t_lo) / (t_hi - t_lo) * (n_grid - 1)), n_grid - 2)
        lo, hi = grid[idx], grid[idx + 1]
        for _ in range(22):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            s1, s2 = sc(m1), sc(m2)
            if s1 < s2:
                hi = m2
            else:
                lo = m1
        best_t = (lo + hi) / 2.0
        best_sc = sc(best_t)
    return h[a] + best_t, h[b] - best_t, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()

    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

    # Phase 8: Resolution exploration
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set([max(20, n_base + d) for d in [-8, -4, 4, 8, 12, 16]
                            if 20 <= n_base + d <= 130] + [30, 36, 40, 48, 56, 64, 72, 80]))
        for n_opt in n_opts:
            if is_dead():
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = exact_score(h_int)
            if sc_int < best_score - 1e-14:
                best_h = h_int
                best_score = sc_int
            if not is_dead():
                h_new, sc = two_point_random(h_int, 1500, rng, is_dead)
                if sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc

    # Phase 9: Final intensive two-point
    if not is_dead():
        remaining = deadline - time.time()
        n_iter = max(3000, int(remaining * 6))
        best_h, best_score = two_point_random(best_h, n_iter, rng, is_dead)

    # Phase 10: Final active + SLSQP
    if not is_dead():
        best_h, best_score = active_correction(best_h, n_active=12, max_iter=100, is_dead=is_dead)
    if not is_dead():
        h_new, sc = slsqp_smooth(best_h, 1200, maxiter=40)
        if h_new is not None and sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 11: Last two-point polish
    if not is_dead():
        remaining = deadline - time.time()
        n_iter = max(2000, int(remaining * 5))
        best_h, best_score = two_point_random(best_h, n_iter, rng, is_dead)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = exact_score(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8

    return final_h, final_score, final_n
