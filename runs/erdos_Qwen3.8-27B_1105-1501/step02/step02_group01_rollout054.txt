def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection to find the right offset
    lo = -3.0 - np.max(z) if np.max(z) > 0 else -3.0
    hi = 3.0 - np.min(z) if np.min(z) < 1 else 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return make_feasible(np.interp(x_dst, x_src, h_src))

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

def compute_G(h):
    """Compute gradient matrix G where G[i, k] = d(corr[k])/d(h[i])."""
    n = len(h)
    u = 1.0 - h
    G = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for d in range(-(n-1), n):
            val = 0.0
            n1 = i - d
            if 0 <= n1 < n:
                val += u[n1]
            n2 = i + d
            if 0 <= n2 < n:
                val -= h[n2]
            G[i, d + n - 1] = val
    return G

def greedy_pair_descent(h0, max_rounds=5000, time_limit=30.0, t_start=None):
    """Greedy pair descent: at each step, find the best (i, j, delta) move."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_score = official_score(h)
    
    for rnd in range(max_rounds):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        G = compute_G(h)
        
        best_improve = 1e-12
        best_move = None
        
        # For efficiency, only consider pairs where h[i] > h[j] + threshold
        threshold = 0.02
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                if h[i] <= h[j] + threshold:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 0.01:
                    continue
                g_diff = G[j] - G[i]
                
                # Evaluate at a few delta values
                best_local = best_score
                best_delta = 0
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    delta = frac * delta_max
                    new_corr = corr + delta * g_diff
                    new_s = np.max(new_corr) * dx
                    if new_s < best_local:
                        best_local = new_s
                        best_delta = delta
                
                if best_local < best_score - best_improve:
                    best_improve = best_score - best_local
                    best_move = (i, j, best_delta)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            # Re-project for exact sum
            h = make_feasible(h)
            new_score = official_score(h)
            if new_score < best_score:
                best_score = new_score
            # Even if make_feasible changed things slightly, continue
        else:
            break
    
    return h, best_score
