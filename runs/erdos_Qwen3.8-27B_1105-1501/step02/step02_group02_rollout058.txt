import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_F_k(h, s):
    """Gradient of F_s with respect to h, where s is the shift (can be negative)."""
    n = len(h)
    d = abs(s)
    if d == 0:
        # F_0 = sum h[i](1-h[i])
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    # From (1-h[i-d]) term: contributes (1-h[i-d]) for i >= d
    for i in range(d, n):
        g[i] += (1.0 - h[i - d])
    # From -h[i+d] term: contributes -h[i+d] for i < n-d
    for i in range(0, n - d):
        g[i] -= h[i + d]
    return g

def _hess_F_k_vec(h, s):
    """Returns the sparse Hessian structure for F_s.
    H[i,p] = -delta_{p,i-d}*[i>=d] - delta_{p,i+d}*[i<n-d]
    We return it as a list of (row, col, value) tuples."""
    n = len(h)
    d = abs(s)
    entries = []
    if d == 0:
        for i in range(n):
            entries.append((i, i, -2.0))
        return entries
    for i in range(d, n):
        entries.append((i, i - d, -1.0))
    for i in range(0, n - d):
        entries.append((i, i + d, -1.0))
    return entries

def _newton_kkt(h0, active_lags, max_iter=50, tol=1e-10):
    """Solve the KKT system using Newton's method.
    
    active_lags: list of shift values s (can be negative)
    Returns (h, t, alpha) or None if failed.
    """
    n = len(h0)
    m = len(active_lags)
    # Variables: x = (h[0],...,h[n-1], alpha[0],...,alpha[m-1], lambda, t)
    # Size: n + m + 2
    
    h = h0.copy()
    alpha = np.ones(m) / m
    lam = 0.0
    t = _score(h0)
    
    shifts = np.array(active_lags)
    
    for it in range(max_iter):
        # Evaluate residuals
        # R1[i] = sum_j alpha_j * dF_kj/dh_i + lambda (for i=0..n-1)
        # R2[j] = F_kj(h) - t (for j=0..m-1)
        # R3 = sum(h) - n/2
        # R4 = sum(alpha) - 1
        
        grads = np.zeros((m, n))
        for j, s in enumerate(shifts):
            grads[j] = _grad_F_k(h, s)
        
        R1 = grads.T @ alpha + lam  # shape (n,)
        F_vals = np.array([_corr_k_at_shift(h, s) for s in shifts])
        R2 = F_vals - t
        R3 = h.sum() - n / 2.0
        R4 = alpha.sum() - 1.0
        
        residual = np.concatenate([R1, R2, [R3, R4]])
        if np.max(np.abs(residual)) < tol:
            break
        
        # Build Jacobian
        # Size: (n+m+2) x (n+m+2)
        # Variables ordered: h(0..n-1), alpha(0..m-1), lambda, t
        
        J = np.zeros((n + m + 2, n + m + 2))

        for j in range(m):
            s_j = shifts[j]
            d = abs(s_j)
            for i in range(d, n):
                J[i, i - d] += alpha[j] * (-1.0)
            for i in range(0, n - d):
                J[i, i + d] += alpha[j] * (-1.0)
            if d == 0:
                for i in range(n):
                    J[i, i] += alpha[j] * (-2.0)

        J[:n, n:n+m] = grads
        
        # dR1/dlambda: 1 (n x 1 block)
        J[:n, n+m] = 1.0
        # dR1/dt: 0
        
        # dR2/dh: row j of grads (m x n block)
        J[n:n+m, :n] = grads
        
        # dR2/dalpha: 0
        # dR2/dlambda: 0
        # dR2/dt: -I (m x m block)
        J[n:n+m, n+m+1] = -1.0  # Wait, this should be -I
        
        # Actually dR2_j/dt = -1 for all j, so it's a column vector of -1s
        # J[n:n+m, t_index] = -1 where t_index = n+m+1
        
        # dR3/dh: 1 (1 x n)
        J[n+m, :n] = 1.0
        # dR3/dalpha: 0, dR3/dlambda: 0, dR3/dt: 0
        
        # dR4/dalpha: 1 (1 x m)
        J[n+m+1, n:n+m] = 1.0
        
        # Solve J * delta = -residual
        try:
            delta = np.linalg.solve(J, -residual)
        except np.linalg.LinAlgError:
            delta = np.linalg.lstsq(J, -residual, rcond=None)[0]
        
        # Damped update
        step = 1.0
        for _ in range(30):
            h_new = h + step * delta[:n]
            alpha_new = alpha + step * delta[n:n+m]
            lam_new = lam + step * delta[n+m]
            t_new = t + step * delta[n+m+1]
            
            if np.all(h_new >= -1e-6) and np.all(h_new <= 1 + 1e-6):
                break
            step *= 0.5
        
        h = np.clip(h + step * delta[:n], 0.0, 1.0)
        alpha = np.maximum(alpha + step * delta[n:n+m], 0.0)
        lam = lam + step * delta[n+m]
        t = t + step * delta[n+m+1]
    
    # Check
    if np.any(h < -1e-8) or np.any(h > 1 + 1e-8):
        return None
    if np.any(alpha < -1e-6):
        return None
    
    h = _project(h)
    return h, t, alpha

- n ≤ q < n+m: 0
- q = n+m: 0
- q = n+m+1: -1  (derivative of F_j - t wrt t)

Row n+m (sum constraint), Col q:
- q < n: 1
- rest: 0

Row n+m+1 (alpha sum), Col q:
- q < n: 0
- n ≤ q < n+m: 1
- rest: 0

def _newton_kkt(h0, active_shifts, max_iter=100, tol=1e-12):
    """Newton method on KKT conditions.
    active_shifts: list of (s, k) where s is the shift (signed), k is lag index
    Returns (h, t, alphas) or None.
    """
    n = len(h0)
    m = len(active_shifts)
    N = n + m + 2
    
    h = h0.copy()
    alphas = np.ones(m) / m
    lam = 0.0
    t = _score(h)
    
    for it in range(max_iter):
        # Compute gradients and F values
        grads = np.zeros((m, n))
        F_vals = np.zeros(m)
        for j in range(m):
            s = active_shifts[j]
            grads[j] = _grad_at_lag(h, s)
            F_vals[j] = _corr_at_lag_signed(h, s)
        
        # Residuals
        R = np.zeros(N)
        R[:n] = grads.T @ alphas + lam
        R[n:n+m] = F_vals - t
        R[n+m] = h.sum() - n / 2.0
        R[n+m+1] = alphas.sum() - 1.0
        
        if np.max(np.abs(R)) < tol:
            break
        
        # Build Jacobian
        J = np.zeros((N, N))
        
        # Block (0:n, 0:n): sum_j alpha_j * H_j
        for j in range(m):
            d = abs(active_shifts[j])
            aj = alphas[j]
            if d == 0:
                J[:n, :n] += aj * (-2.0) * np.eye(n)
            else:
                for p in range(d, n):
                    J[p, p-d] += aj * (-1.0)
                for p in range(0, n-d):
                    J[p, p+d] += aj * (-1.0)
        
        # Block (0:n, n:n+m): gradients
        J[:n, n:n+m] = grads
        
        # Block (0:n, n+m): 1
        J[:n, n+m] = 1.0
        
        # Block (n:n+m, 0:n): gradients (transposed usage)
        J[n:n+m, :n] = grads
        
        # Block (n:n+m, n+m+1): -1
        J[n:n+m, n+m+1] = -1.0
        
        # Block (n+m, 0:n): 1
        J[n+m, :n] = 1.0
        
        # Block (n+m+1, n:n+m): 1
        J[n+m+1, n:n+m] = 1.0
        
        # Solve
        try:
            delta = np.linalg.solve(J, -R)
        except np.linalg.LinAlgError:
            try:
                delta = np.linalg.lstsq(J, -R, rcond=None)[0]
            except:
                break
        
        # Damped update
        step = 1.0
        for _ in range(40):
            h_trial = h + step * delta[:n]
            if np.all(h_trial >= 0) and np.all(h_trial <= 1):
                break
            step *= 0.5
        
        h = np.clip(h + step * delta[:n], 0.0, 1.0)
        alphas = np.maximum(alphas + step * delta[n:n+m], 0.0)
        alphas /= alphas.sum()  # renormalize
        lam = lam + step * delta[n+m]
        t = t + step * delta[n+m+1]
    
    # Final check
    h = _project(h)
    if np.any(h < -1e-10) or np.any(h > 1 + 1e-10):
        return None
    return h, t, alphas

One more consideration: the problem is symmetric in the sense that if h is a solution, then 1-h (reflected and shifted) might also be a solution. Specifically, if h(x) is a solution, then h(2-x) is also a solution (by the symmetry of the problem). Actually, let's check: corr at shift s (s>0) = sum h[i+s](1-h[i])
corr at shift -s = sum h[i](1-h[i+s])
These are NOT the same in general. So the problem is not symmetric under s → -s.

But the problem IS symmetric under h → h_reversed (i.e., h[i] → h[n-1-i]). Let me verify:
If h' = reversed(h), then:
corr_{h'}(s) = sum h'[i+s](1-h'[i]) = sum h[n-1-i-s](1-h[n-1-i])
Let j = n-1-i: = sum h[j-s](1-h[j]) for j = s to n-1
= sum_{j=s}^{n-1} h[j-s](1-h[j])
Let k = j-s: = sum_{k=0}^{n-1-s} h[k](1-h[k+s])
= corr_h(-s)

So corr_{reversed(h)}(s) = corr_h(-s). This means the set of all correlation values is the same! So the score is invariant under reversal. Good to know, but not directly useful for optimization.
