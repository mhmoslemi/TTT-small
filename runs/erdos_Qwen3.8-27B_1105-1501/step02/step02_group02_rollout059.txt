    n = len(h)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

        a = 0.0

            a -= h[p - s]
            b += h[p - s]

        B[i] = b

                    else:

                    else:

                else:

            elif r2 <= hi + 1e-13:

    B = np.zeros(2 * n - 2)
    s_arr = np.arange(1, n)
    # Positive lags
    ps = p + s_arr
    valid_ps = ps < n
    pm = p - s_arr
    valid_pm = pm >= 0
    if np.any(valid_ps):
        A[:n-1][valid_ps] += (1.0 - h[ps[valid_ps]])
    if np.any(valid_pm):
        A[:n-1][valid_pm] -= h[pm[valid_pm]]
        B[:n-1][valid_pm] += h[pm[valid_pm]]
    # Negative lags (shift = -s, s=1..n-1)
    pn = p - s_arr
    valid_pn = pn >= 0
    pp = p + s_arr
    valid_pp = pp < n
    if np.any(valid_pn):
        A[n-1:][valid_pn] += (1.0 - h[pn[valid_pn]])
    if np.any(valid_pp):
        A[n-1:][valid_pp] -= h[pp[valid_pp]]
        B[n-1:][valid_pp] += h[pp[valid_pp]]
    
    D0 = float(np.sum(h * (1.0 - h)) - h[p] * (1.0 - h[p]))
    
    # Bisection on v
    v_lo = 0.0
    v_hi = max(float(B.max()), float((A + B).max()), D0 + 0.25) + 1e-12
    
    pos_mask = A > 1e-13
    neg_mask = A < -1e-13
    zero_mask = ~(pos_mask | neg_mask)
    
    for _ in range(42):
        v = (v_lo + v_hi) * 0.5
        if zero_mask.any() and float(B[zero_mask].max()) > v:
            v_lo = v
            continue
        lo = 0.0
        hi = 1.0
        if pos_mask.any():
            ub = float(np.min((v - B[pos_mask]) / A[pos_mask]))
            if ub < hi:
                hi = ub
        if neg_mask.any():
            lb = float(np.max((v - B[neg_mask]) / A[neg_mask]))
            if lb > lo:
                lo = lb
        if lo > hi + 1e-13:
            v_lo = v
            continue
        disc = 1.0 - 4.0 * (v - D0)
        if disc > 0:
            r1 = (1.0 - np.sqrt(disc)) * 0.5
            r2 = (1.0 + np.sqrt(disc)) * 0.5
            if hi <= r1 + 1e-13:
                pass
            elif lo >= r2 - 1e-13:
                pass
            elif lo > r1 + 1e-13 and hi < r2 - 1e-13:
                v_lo = v
                continue
            else:
                if lo <= r1 + 1e-13:
                    hi = min(hi, r1)
                elif r2 <= hi + 1e-13:
                    lo = max(lo, r2)
                else:
                    v_lo = v
                    continue
        if lo <= hi + 1e-13:
            v_hi = v
        else:
            v_lo = v
    
    # Extract x
    v = v_hi + 1e-9
    lo = 0.0
    hi = 1.0
    if pos_mask.any():
        hi = min(hi, float(np.min((v - B[pos_mask]) / A[pos_mask])))
    if neg_mask.any():
        lo = max(lo, float(np.max((v - B[neg_mask]) / A[neg_mask])))
    disc = 1.0 - 4.0 * (v - D0)
    if disc > 0:
        r1 = (1.0 - np.sqrt(disc)) * 0.5
        r2 = (1.0 + np.sqrt(disc)) * 0.5
        if hi <= r1 + 1e-13:
            pass
        elif lo >= r2 - 1e-13:
            pass
        elif lo <= r1 + 1e-13:
            hi = min(hi, r1)
        elif r2 <= hi + 1e-13:
            lo = max(lo, r2)
    x = (lo + hi) * 0.5
    return float(np.clip(x, 0.0, 1.0))

def _cd_sweep(h, n, order):
    h_cur = h.copy()
    for p in order:
        h_cur[p] = _coord_opt(int(p), h_cur, n)
    return h_cur

def _beatty_sequence(n, k):
    """Create a Beatty-style binary-like sequence with k ones."""
    h = np.zeros(n)
    if k == 0:
        return h
    positions = np.unique((np.arange(k) * n // k).astype(int) % n)
    h[positions] = 1.0
    return h
