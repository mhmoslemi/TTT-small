def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -20.0, 20.0)
    lo, hi = -21.0, 21.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:

            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:

                residual += sub

    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

    m = n // 2
    u = np.clip(h[:m], 1e-10, 1.0 - 1e-10)
    return np.log(u / (1.0 - u))

def smooth_obj_grad_sym(z, n, beta):
    """Log-sum-exp smoothed objective and gradient for symmetric parameterization."""
    h = z_to_h(z, n)
    m = n // 2
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Numerical stability
    obj = cmax + np.log(np.sum(np.exp(shifted))) / beta
    
    # Weights
    exp_shifted = np.exp(shifted - shifted.max())
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of C_p with respect to h
    # Then chain rule through z
    n_full = n
    # Compute dC_p/dh for all p at once using the formula
    # For efficiency, compute the gradient contribution directly
    
    # dC/dz[j] = s_j*(1-s_j) * (dC/dh[j] - dC/dh[n-1-j])
    s = u = h[:m]  # sigmoid values
    sig_grad = s * (1.0 - s)  # s*(1-s)
    
    grad_h = np.zeros(n)
    for p in range(2 * n - 1):
        k = p - (n - 1)
        if abs(w[p]) < 1e-15:
            continue
        if k == 0:
            gh = dx * (1.0 - 2.0 * h)
        elif k > 0:
            L = n - k
            gh = np.zeros(n)
            gh[k:n] += dx * (1.0 - h[0:L])
            gh[0:L] -= dx * h[k:k+L]
        else:
            d = -k
            L = n - d
            gh = np.zeros(n)
            gh[0:L] += dx * (1.0 - h[d:d+L])
            gh[d:n] -= dx * h[0:L]
        grad_h += w[p] * gh
    
    grad_z = sig_grad * (grad_h[:m] - grad_h[m:])
    
    return obj, grad_z

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals))


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -20.0, 20.0)
    lo, hi = -21.0, 21.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)


def symmetrize(h):
    h_sym = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_sym)


def z_to_h(z, n):
    m = n // 2
    zc = np.clip(z, -40.0, 40.0)
    u = 1.0 / (1.0 + np.exp(-zc))
    h = np.empty(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h


def h_to_z(h, n):
    m = n // 2
    u = np.clip(h[:m], 1e-12, 1.0 - 1e-12)
    return np.log(u / (1.0 - u))


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def grad_c_ph(h, p, n):
    """Gradient of C_p with respect to h, where p is the output index 0..2n-2."""
    dx = 2.0 / n
    k = p - (n - 1)
    gh = np.zeros(n)
    if k == 0:
        gh[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        L = n - k
        gh[k:n] += dx * (1.0 - h[0:L])
        gh[0:L] -= dx * h[k:k + L]
    else:
        d = -k
        L = n - d
        gh[0:L] += dx * (1.0 - h[d:d + L])
        gh[d:n] -= dx * h[0:L]
    return gh


def smooth_obj_grad(z, n, beta):
    h = z_to_h(z, n)
    m = n // 2
    vals = compute_corr(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted_max = shifted.max()
    obj = cmax + (shifted_max + np.log(np.sum(np.exp(shifted - shifted_max)))) / beta
    w = np.exp(shifted - shifted_max)
    w /= w.sum()
    grad_h = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-15:
            continue
        grad_h += w[p] * grad_c_ph(h, p, n)
    sig = h[:m] * (1.0 - h[:m])
    if n % 2 == 0:
        grad_z = sig * (grad_h[:m] - grad_h[m:])
    else:
        grad_z = sig * (grad_h[:m] - grad_h[m + 1:])
    return obj, grad_z


def slsqp_constraint_z(z, p, n):
    h = z_to_h(z, n)
    vals = compute_corr(h)
    return vals[p]


def slsqp_constraint_grad_z(z, p, n):
    h = z_to_h(z, n)
    m = n // 2
    gh = grad_c_ph(h, p, n)
    sig = h[:m] * (1.0 - h[:m])
    if n % 2 == 0:
        return -(sig * (gh[:m] - gh[m:]))
    else:
        return -(sig * (gh[:m] - gh[m + 1:]))


def lbfags_sym_solve(z0, n, betas, time_limit, tol=1e-12):
    t0 = time.time()
    z = z0.copy()
    last_f = smooth_obj_grad(z, n, betas[0])[0]
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        res = minimize(
            fun=lambda x: smooth_obj_grad(x, n, beta)[0],
            x0=z,
            jac=lambda x: smooth_obj_grad(x, n, beta)[1],
            method='L-BFGS-B',
            options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            z = res.x
        f_new = smooth_obj_grad(z, n, beta)[0]
        if abs(f_new - last_f) < tol:
            break
        last_f = f_new
    return z


def active_set_solve(z0, n, max_active=30, max_rounds=8, maxiter_slsqp=300):

                else:

                J[i, -1] = 1.0
            return J

        bounds = [(-12.0, 12.0)] * z_dim + [(0.0, 1.0)]

def interp_sym(h_src, n_tgt):
    n_src = len(h_src)
    h_sym = symmetrize(h_src)
    m_src = n_src // 2
    m_tgt = n_tgt // 2
    if m_src == m_tgt:
        return h_to_z(h_sym, n_src)
    src_x = np.arange(m_src, dtype=float)
    tgt_x = np.arange(m_tgt, dtype=float)
    u_src = h_sym[:m_src]
    u_tgt = np.interp(tgt_x, src_x, u_src)
    z_tgt = h_to_z(z_to_h(u_tgt, 2 * m_tgt), 2 * m_tgt)
    return z_tgt


def try_full_pg(h, n_iters=2000, time_limit=30.0):
    """Projected gradient with backtracking on the full space."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    t0 = time.time()
    step = 0.05
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        vals = compute_corr(h)
        cmax = vals.max()
        thr = cmax - max(0.01 * cmax, 1e-5)
        active = np.where(vals >= thr)[0]
        gh = np.zeros(n)
        for p in active:
            gh += grad_c_ph(h, p, n)
        gh /= len(active)
        gn = np.linalg.norm(gh)
        if gn < 1e-12:
            break
        gh /= gn
        # Backtracking
        alpha = step
        f_cur = cmax
        for _ in range(30):
            h_trial = make_feasible(h - alpha * gh)
            f_trial = official_score(h_trial)
            if f_trial <= f_cur - 1e-4 * alpha:
                break
            alpha *= 0.5
        if alpha < 1e-10:
            break
        h = h_trial
        f_cur = f_trial
        if alpha == step:
            step = min(step * 1.3, 0.5)
    return h
