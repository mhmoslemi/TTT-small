def line_search_quadratic(h, d, active_lags, vals, b_coeffs, c_coeffs):
    """Find best t minimizing max of active quadratics, with box feasibility."""
    n = len(h)
    # Find max feasible step
    t_max = 1.0
    for i in range(n):
        if d[i] < 0 and h[i] > 0:
            t_max = min(t_max, h[i] / (-d[i]))
        elif d[i] > 0 and h[i] < 1:
            t_max = min(t_max, (1 - h[i]) / d[i])
    t_max = min(t_max, 1.0)  # Don't take full step initially
    
    # Collect intersection points
    t_candidates = [0.0, t_max]
    m = len(active_lags)
    for i in range(m):
        for j in range(i+1, m):
            # q_i(t) = q_j(t) => (c_i-c_j)t^2 + (b_i-b_j)t + (a_i-a_j) = 0
            a_diff = vals[active_lags[i]] - vals[active_lags[j]]
            b_diff = b_coeffs[active_lags[i]] - b_coeffs[active_lags[j]]

                    if 0 < t < t_max:
                        t_candidates.append(t)
            else:
                disc = b_diff**2 - 4*c_diff*a_diff
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    for sign in [1, -1]:
                        t = (-b_diff + sign*sqrt_disc) / (2*c_diff)
                        if 0 < t < t_max:

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float), -2.0, 3.0)
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            for idx in np.argsort(1.0 - h):
                amt = min(resid, 1.0 - h[idx])
                h[idx] += amt
                resid -= amt
                if abs(resid) < 1e-14: break
        else:
            for idx in np.argsort(h):
                amt = max(resid, -h[idx])
                h[idx] += amt
                resid -= amt
                if abs(resid) < 1e-14: break
    return h

def official_score(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def all_lags(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_off(h, off):
    n = len(h)
    g = np.zeros(n)
    lo = max(0, -off) if off > 0 else 0
    hi = min(n, n - off) if off > 0 else n
    for j in range(lo, hi):
        ip = j + off  # j + off is the "h" index
        if 0 <= ip < n:
            g[ip] += (1.0 - h[j])
            g[j] -= h[ip]
    return g

def bimetric_step(h, d, active_offs):
    """Perform 1D quadratic minimax line search along direction d."""
    n = len(h)
    # Max feasible step
    t_max = 5.0
    neg_d = d < -1e-12
    pos_d = d > 1e-12
    if neg_d.any():
        t_max = min(t_max, (h[neg_d] / (-d[neg_d])).min())
    if pos_d.any():
        t_max = min(t_max, ((1.0 - h[pos_d]) / d[pos_d]).min())
    t_max = max(t_max, 0.0)
    if t_max < 1e-12:
        return 0.0, 0.0
    
    # Compute Q_p coefficients for active lags
    m = len(active_offs)
    vals = np.zeros(m)
    Bs = np.zeros(m)
    Cs = np.zeros(m)
    for idx, off in enumerate(active_offs):
        # Find valid j range: 0 <= j < n and 0 <= j+off < n
        j_lo = max(0, -off)
        j_hi = min(n, n - off)
        js = np.arange(j_lo, j_hi)
        hs_j = h[js]
        hs_ip = h[js + off]
        vals[idx] = np.sum(hs_ip * (1.0 - hs_j))
        Bs[idx] = np.sum(d[js + off] * (1.0 - hs_j) - hs_ip * d[js])
        Cs[idx] = -np.sum(d[js + off] * d[js])
    
    base_max = vals.max()
    
    # Find candidate t values
    candidates = [0.0, t_max]
    # Add midpoints
    for frac in np.linspace(0.1, 0.9, 9):
        candidates.append(frac * t_max)
    # Add pairwise intersections
    for i in range(m):
        for j in range(i + 1, m):
            a_d = vals[i] - vals[j]
            b_d = Bs[i] - Bs[j]
            c_d = Cs[i] - Cs[j]
            if abs(c_d) > 1e-14:
                disc = b_d * b_d - 4.0 * c_d * a_d
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for sgn in (1.0, -1.0):
                        t = (-b_d + sgn * sq) / (2.0 * c_d)
                        if 1e-10 < t < t_max * (1 - 1e-10):
                            candidates.append(t)
            elif abs(b_d) > 1e-14:
                t = -a_d / b_d
                if 1e-10 < t < t_max * (1 - 1e-10):
                    candidates.append(t)
    
    best_t = 0.0
    best_score = base_max
    for t in candidates:
        q = vals + Bs * t + Cs * t * t
        mx = q.max()
        if mx < best_score:
            best_score = mx
            best_t = t
    return best_t, best_score

def project_to_direction(g, h):
    """Project gradient to a feasible descent direction."""
    n = len(h)
    d = g.copy()
    # Enforce sign constraints at boundaries
    tol = 1e-10
    lower = h <= tol
    upper = h >= 1.0 - tol
    d[lower] = np.maximum(d[lower], 0)
    d[upper] = np.minimum(d[upper], 0)
    # Enforce sum = 0 by subtracting mean
    d -= d.mean()
    # Re-clip for boundary violations introduced by mean subtraction
    d[lower] = np.maximum(d[lower], 0)
    d[upper] = np.minimum(d[upper], 0)
    # Re-normalize sum (small residual)
    d -= d.mean()
    return d

def solve_direction_lp(h, active_offs, max_time=2.0):
    """Solve a small QP for the best descent direction."""
    n = len(h)
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -1.0,
        d_var <= 1.0
    ]
    # Box constraints
    lower = h <= 1e-10
    upper = h >= 1.0 - 1e-10
    if lower.any():
        constraints.append(d_var[lower] >= 0)
    if upper.any():
        constraints.append(d_var[upper] <= 0)
    # Active constraints
    for off in active_offs:
        g = grad_for_off(h, off)
        constraints.append(cp.sum(g * d_var) <= t_var)
    obj = cp.Minimize(t_var + 0.01 * cp.sum_squares(d_var))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    d = d_var.value
    if d is None or not np.all(np.isfinite(d)):
        return None
    return d

        hi = min(t_max, best_t * 1.2)
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(30):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = (vals + Bs * x1 + Cs * x1 * x1).max()
            f2 = (vals + Bs * x2 + Cs * x2 * x2).max()
            if f1 < f2:
                hi = x2
            else:
                lo = x1
        best_t = (lo + hi) / 2
    
    return best_t

This is much better - it considers all lags in the line search, which prevents the "hidden constraint" problem.

Now, one concern: computing vals, Bs, Cs for all 2n-1 lags is O(n * n_lags) = O(n^2) per iteration. For n=60, that's 3600 operations, which is trivial.

The pairwise intersections for the top 50 lags: C(50,2) = 1225, also trivial.

The golden section refinement: 30 evaluations of max of 119 values = 3570 operations. Still trivial.

So each iteration is very fast. Good.

Now let me think about the overall strategy:
1. Start from initial_h_values
2. Main loop: compute direction, do full bimetric line search, update
3. Every few iterations, try:
   - Different active set sizes for the direction
   - LP-based direction
   - Random perturbation
   - Resolution change
4. Return best found

Let me also add a "smoothing" phase where I use a exponential approximation to the max:
f_β(h) = (1/β) * log(sum_p exp(β * F_p(h)/n_lags))

This gives a smooth objective whose gradient is a weighted average of all lag gradients, weighted by their exponentiated values. This can help escape from "stuck at isolated active sets" situations.

The gradient of f_β is:
∇f_β = (1/n_lags) * sum_p w_p * grad_p where w_p = exp(β * F_p / n_lags) / sum_q exp(β * F_q / n_lags)

For large β, this concentrates on the max; for small β, it's more spread out.

Let me include this as an additional phase.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            for idx in np.argsort(1.0 - h):
                amt = min(resid, 1.0 - h[idx])
                h[idx] += amt
                resid -= amt
                if abs(resid) < 1e-14:
                    break
        else:
            for idx in np.argsort(h):
                amt = max(resid, -h[idx])
                h[idx] += amt
                resid -= amt
                if abs(resid) < 1e-14:
                    break
    return h

def official_score(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def all_lags(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_off(h, off):
    n = len(h)
    g = np.zeros(n)
    j_lo = max(0, -off)
    j_hi = min(n, n - off)
    for j in range(j_lo, j_hi):
        ip = j + off
        g[ip] += (1.0 - h[j])
        g[j] -= h[ip]
    return g

def full_bimetric_step(h, d):
    n = len(h)
    t_max = 5.0
    neg_d = d < -1e-12
    pos_d = d > 1e-12
    if neg_d.any():
        t_candidate = (h[neg_d] / (-d[neg_d])).min()
        t_max = min(t_max, t_candidate)
    if pos_d.any():
        t_candidate = ((1.0 - h[pos_d]) / d[pos_d]).min()
        t_max = min(t_max, t_candidate)
    t_max = max(t_max, 0.0)
    if t_max < 1e-12:
        return 0.0
    n_lags = 2 * n - 1
    vals = np.zeros(n_lags)
    Bs = np.zeros(n_lags)
    Cs = np.zeros(n_lags)
    for p in range(n_lags):
        off = p - (n - 1)
        j_lo = max(0, -off)
        j_hi = min(n, n - off)
        js = np.arange(j_lo, j_hi)
        if len(js) == 0:
            continue
        hs_j = h[js]
        hs_ip = h[js + off]
        vals[p] = np.sum(hs_ip * (1.0 - hs_j))
        Bs[p] = np.sum(d[js + off] * (1.0 - hs_j) - hs_ip * d[js])
        Cs[p] = -np.sum(d[js + off] * d[js])
    base_max = vals.max()
    candidates = [0.0, t_max]
    for frac in np.linspace(0.05, 0.95, 12):
        candidates.append(frac * t_max)
    sorted_p = np.argsort(vals)[::-1]
    top_p = sorted_p[:min(60, n_lags)]
    for i in range(len(top_p)):
        for j in range(i + 1, len(top_p)):
            pi, pj = top_p[i], top_p[j]
            a_d = vals[pi] - vals[pj]
            b_d = Bs[pi] - Bs[pj]
            c_d = Cs[pi] - Cs[pj]
            if abs(c_d) > 1e-14:
                disc = b_d * b_d - 4.0 * c_d * a_d
                if disc >= 0:
                    sq = np.sqrt(disc)
                    t1 = (-b_d + sq) / (2.0 * c_d)
                    t2 = (-b_d - sq) / (2.0 * c_d)
                    if 1e-10 < t1 < t_max * 0.9999:
                        candidates.append(t1)
                    if 1e-10 < t2 < t_max * 0.9999:
                        candidates.append(t2)
            elif abs(b_d) > 1e-14:
                t = -a_d / b_d
                if 1e-10 < t < t_max * 0.9999:
                    candidates.append(t)
    best_t = 0.0
    best_score = base_max
    for t in candidates:
        q = vals + Bs * t + Cs * t * t
        mx = q.max()
        if mx < best_score:
            best_score = mx
            best_t = t
    if best_t > 1e-10:
        lo = max(0.0, best_t * 0.7)
        hi = min(t_max, best_t * 1.3)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(35):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = (vals + Bs * x1 + Cs * x1 * x1).max()
            f2 = (vals + Bs * x2 + Cs * x2 * x2).max()
            if f1 < f2:
                hi = x2
            else:
                lo = x1
        t_mid = (lo + hi) / 2.0
        f_mid = (vals + Bs * t_mid + Cs * t_mid * t_mid).max()
        if f_mid < best_score:
            best_t = t_mid
    return best_t

def project_direction(g, h):
    n = len(h)
    d = g.copy()
    tol = 1e-10
    lower = h <= tol
    upper = h >= 1.0 - tol
    d[lower] = np.maximum(d[lower], 0.0)
    d[upper] = np.minimum(d[upper], 0.0)
    d -= d.mean()
    d[lower] = np.maximum(d[lower], 0.0)
    d[upper] = np.minimum(d[upper], 0.0)
    d -= d.mean()
    return d

def softmax_direction(h, lags, beta):
    n = len(h)
    n_lags = len(lags)
    shifted = beta * (lags - lags.max())
    weights = np.exp(np.clip(shifted, -50, 50))
    weights /= weights.sum()
    grad_avg = np.zeros(n)
    for p in range(n_lags):
        if weights[p] > 1e-8:
            off = p - (n - 1)
            grad_avg += weights[p] * grad_for_off(h, off)
    return -grad_avg

def smooth_obj_grad(h, lags, beta):
    n = len(h)
    n_lags = len(lags)
    max_lag = lags.max()
    shifted = beta * (lags - max_lag)
    weights = np.exp(np.clip(shifted, -50, 50))
    weights /= weights.sum()
    val = max_lag + np.log(weights.sum()) / beta  # approximately
    grad = np.zeros(len(h))
    for p in range(n_lags):
        if weights[p] > 1e-8:
            off = p - (n - 1)
            grad += (weights[p] * beta / n_lags) * grad_for_off(h, off)
    return val, grad

def try_slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    def obj(h):
        lags = all_lags(h)
        mx = lags.max()
        shifted = beta * (lags - mx)
        return mx + np.log(np.sum(np.exp(np.clip(shifted, -50, 50)))) / beta
    def obj_fn(h):
        return obj(h)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x) / 2.0,
             'jac': lambda x: np.ones(len(x))}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def pair_descent(h, max_pairs=200):
    n = len(h)
    dx = 2.0 / n
    lags = all_lags(h)
    max_lag = lags.max()
    improved = False
    indices = list(range(n))
    np.random.shuffle(indices)
    for i in indices[:max(10, n // 3)]:
        for j in indices[:max(10, n // 3)]:
            if i == j:

            if i >= j:
                break
            h_save = h[i]
            delta = h[i] - h[j]

            lo = -min(h[i], 1.0 - h[j])

def interpolate_to(h_src, n_tgt):

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym(h):
    return make_feasible(1.0 - h[::-1].copy())
