    dx = 2.0 / n

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _smoothed_obj(h, alpha):
    """Log-sum-exp smoothing of the max of scores (unscaled)."""
    s = _scores(h)
    max_s = np.max(s)
    return (1.0/alpha) * np.log(np.sum(np.exp(alpha * (s - max_s)))) + max_s

    """Gradient of the smoothed max objective (unscaled scores)."""
    n = len(h)
    s = _scores(h)
    max_s = np.max(s)
    weights = np.exp(alpha * (s - max_s))
    weights /= np.sum(weights)
    
    # Compute gradient as weighted sum of individual score gradients
    g = np.zeros(n)
    for w_idx, w in enumerate(weights):
        if w < 1e-12:
            continue
        d = w_idx - (n - 1)  # shift

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):

        if d >= 0:

                G[r, :m] += -h[d:]

            m = n - g

    n = len(h)

        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _smooth_fun(x, alpha, target):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _lbfgs_step(h, alpha, maxiter=200):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_fun, h, args=(alpha, target),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _smooth_loop(h, deadline, alphas=None, start_alpha=5.0):
    if alphas is None:
        alphas = (5.0, 10.0, 20.0, 40.0, 80.0, 120.0)
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for alpha in alphas:
        if time.time() > deadline:
            break
        h_new, c5_new = _lbfgs_step(h, alpha, maxiter=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        # Second pass at same alpha with more iterations
        if time.time() < deadline:
            h_new2, c5_new2 = _lbfgs_step(h_new, alpha, maxiter=150)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
            h = h_new2

    return best_h, best_c5


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)

    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 14):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a

    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(14):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t

    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.15, max_iters=4000, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break

        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.4)
        else:
            stall += 1
            radius *= 0.72
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.08

    return best_h, best_c5


def _smooth_dir_step(h, alpha, radius=0.05):
    """A single gradient step on the smooth objective, projected."""
    n = len(h)
    target = n / 2.0
    s = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    G = _grad_matrix(h)
    d = -(G.T @ w)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return h.copy(), _exact_c5(h)
    d = d / norm * radius
    cur_c5 = _exact_c5(h)
    best_a, best_c5 = 0.0, cur_c5
    for t in np.linspace(0.0, 1.0, 8):
        a = t
        h_t = _project(h + a * d, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        return _project(h + best_a * d, target), best_c5
    return h.copy(), cur_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = [j for j in range(n) if 1e-6 < h[j] < 1.0 - 1e-6]

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in interior:
            if time.time() > deadline:
                break
            if 1e-6 <= h[j] <= 1.0 - 1e-6:
                for k in range(j + 1, n):
                    if 1e-6 <= h[k] <= 1.0 - 1e-6:
                        for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                            a_max = 0.0
                            if dj > 0:
                                a_max = min(a_max if a_max > 0 else 1e9, (1.0 - h[j]) / dj)
                            else:
                                a_max = min(a_max if a_max > 0 else 1e9, h[j] / (-dj))
                            if dk > 0:
                                a_max = min(a_max if a_max > 0 else 1e9, (1.0 - h[k]) / dk)
                            else:
                                a_max = min(a_max if a_max > 0 else 1e9, h[k] / (-dk))
                            if a_max < 1e-10:
                                continue
                            found = False
                            for frac in [0.5, 1.0, 0.25, 0.75, 0.125]:
                                a = frac * a_max
                                h_t = h.copy()
                                h_t[j] += dj * a
                                h_t[k] += dk * a
                                h_t = np.clip(h_t, 0.0, 1.0)
                                h_t = _project(h_t, target)
                                c5_t = _exact_c5(h_t)
                                if c5_t < best_c5 - 1e-12:
                                    best_c5 = c5_t
                                    best_h = h_t.copy()
                                    h = h_t
                                    improved = True
                                    a_max = a
                                    found = True
                                    break
                            if found:
                                break
                        if improved:
                            break
            if improved:
                break
        if not improved:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Multi-alpha smooth optimization (30% budget)
    d1 = t0 + budget_s * 0.30
    h, c5 = _smooth_loop(h, d1, alphas=(5.0, 10.0, 20.0, 40.0, 80.0, 150.0))
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: LP polishing (20% budget)
    d2 = t0 + budget_s * 0.50
    h, c5 = _lp_loop(h, d2, radius_init=0.15, max_iters=4000, stall_limit=50)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 3: Interleaved smooth + LP from multiple restarts (25% budget)
    d3 = t0 + budget_s * 0.75
    scale = 0.04
    for attempt in range(1000):
        if time.time() > d3:
            break
        if best_c5 < 0.3806:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        # Quick smooth step
        h_try, c5_try = _lbfgs_step(h_p, 30.0, maxiter=60)
        if c5_try < best_c5 - 1e-10:
            # Follow with LP
            sub_deadline = min(d3, time.time() + budget_s * 0.04)
            h_try, c5_try = _lp_loop(h_try, sub_deadline,
                                     radius_init=0.1, max_iters=600, stall_limit=40)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                scale = max(0.008, scale * 0.85)
        else:
            # Try a smooth-direction step instead
            h_try, c5_try = _smooth_dir_step(h_p, 50.0, radius=0.03)
            if c5_try < best_c5 - 1e-10:
                sub_deadline = min(d3, time.time() + budget_s * 0.02)
                h_try, c5_try = _lp_loop(h_try, sub_deadline,
                                         radius_init=0.08, max_iters=300, stall_limit=30)
                if c5_try < best_c5:
                    best_c5, best_h = c5_try, h_try.copy()
                    scale = max(0.008, scale * 0.9)
            else:
                scale = min(0.3, scale * 1.15)

    # Phase 4: Resolution search (12% budget)
    d4 = t0 + budget_s * 0.87
    for dn in [1, 2, 3, 4, 5, 6, 8, -1, -2, -3, -4]:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d4, time.time() + budget_s * 0.025)
        h_r, c5_r = _smooth_loop(h_r, sub_d, alphas=(15.0, 40.0))
        if time.time() < d4:
            sub_d2 = min(d4, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_loop(h_r, sub_d2, radius_init=0.1, max_iters=500, stall_limit=35)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Pair polish (5% budget)
    if time.time() < t0 + budget_s * 0.93:
        d5 = t0 + budget_s * 0.94
        hp, c5p = _pair_polish(best_h, d5, max_passes=60)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 6: Final LP polish (6% budget)
    if time.time() < t0 + budget_s * 0.97:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.99,
                             radius_init=0.04, max_iters=3000, stall_limit=80)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
