        for i in range(10):
            if remaining() < 15:
                break
            sigma = 0.02 + 0.06 * rng.random()
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=len(best_h)))
            h_opt, sc_opt = lp_optimize(h_pert, max_iter=80, k_active=5, max_step_size=0.3)
            update(h_opt, sc_opt)

    # Phase 7: Fine LP polish with small steps
    if remaining() > 30:
        for ms in [0.15, 0.08, 0.04]:
            if remaining() < 10:
                break
            h_opt, sc_opt = lp_optimize(best_h, max_iter=80, k_active=3, max_step_size=ms)
            update(h_opt, sc_opt)

    # Phase 8: Final coordinate descent
    if remaining() > 20:
        h_cd, sc_cd = coordinate_descent(best_h, rng, n_trials=400)
        update(h_cd, sc_cd)

    # Phase 9: Last LP from best
    if remaining() > 15:
        h_opt, sc_opt = lp_optimize(best_h, max_iter=50, k_active=4, max_step_size=0.1)
        update(h_opt, sc_opt)

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
