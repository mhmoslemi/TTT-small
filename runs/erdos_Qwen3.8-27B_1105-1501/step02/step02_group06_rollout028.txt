        best_score = official_score(best_h)

    n_best = len(best_h)

    # Phase 1: LP direction + line search with varying tolerances
    n_lp = len(best_h)
    for tol in [0.0, 0.0005, 0.002, 0.005, 0.01, 0.02]:
        if remaining() < 30:
            break
        for ka in [10, 20, 30]:
            if remaining() < 15:
                break
            h_cur = best_h.copy()
            sc_cur = best_score
            for it in range(50):
                if remaining() < 10:
                    break
                d = lp_direction(h_cur, k_active=ka, tol=tol, step_bound=1.0)
                if d is None:
                    break
                h_new, sc_new = step_direction(h_cur, d, n_trials=25)
                if sc_new < sc_cur - 1e-14:
                    h_cur = h_new
                    sc_cur = sc_new
                    update(h_cur, sc_cur)
                else:
                    break

    # Phase 2: QP refinement
    if remaining() > 60:
        for reg in [0.5, 0.1, 0.02, 0.005]:
            if remaining() < 20:
                break
            h_new, sc_new = qp_refine(best_h, k_active=8, reg=reg, max_iter=12)
            update(h_new, sc_new)

    # Phase 3: Coordinate pair descent
    if remaining() > 60:
        h_new, sc_new = coord_descent(best_h, n_rounds=4, n_delta_trials=8)
        update(h_new, sc_new)
        # Second pass with finer deltas
        h_new, sc_new = coord_descent(best_h, n_rounds=3, n_delta_trials=6)
        update(h_new, sc_new)

    # Phase 4: Symmetrize + refine
    if remaining() > 40:
        h_sym = symmetrize(best_h)
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        if remaining() > 30:
            h_new, sc_new = qp_refine(h_sym, k_active=8, reg=0.05, max_iter=10)
            update(h_new, sc_new)
            h_new, sc_new = coord_descent(h_sym, n_rounds=3, n_delta_trials=8)
            update(h_new, sc_new)

    # Phase 5: Resolution sweep
    if remaining() > 100:
        n_base = len(best_h)
        n_options = sorted(set([
            max(20, n_base - 10), n_base - 5, n_base,
            n_base + 5, n_base + 10, min(90, n_base + 20)
        ]))
        for n_opt in n_options:
            if remaining() < 30:
                break
            if n_opt < 16:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            # Refine at this resolution
            if remaining() > 25:
                for tol in [0.0, 0.003, 0.01]:
                    if remaining() < 10:
                        break
                    h_cur = h_int.copy()
                    for it in range(30):
                        if remaining() < 5:
                            break
                        d = lp_direction(h_cur, k_active=15, tol=tol, step_bound=1.0)
                        if d is None:
                            break
                        h_new2, sc_new2 = step_direction(h_cur, d, n_trials=20)
                        if sc_new2 < official_score(h_cur) - 1e-14:
                            h_cur = h_new2
                        else:
                            break
                    update(h_cur, official_score(h_cur))
                if remaining() > 15:
                    h_new, sc_new = coord_descent(h_int, n_rounds=2, n_delta_trials=6)
                    update(h_new, sc_new)

    # Phase 6: LP with random active-set subsets (diversity)
    if remaining() > 80:
        n = len(best_h)
        for trial in range(20):
            if remaining() < 15:
                break
            h_cur = best_h.copy()
            sc_cur = official_score(h_cur)
            for it in range(40):
                if remaining() < 5:
                    break
                lags = compute_all_lags(h_cur)
                # Randomly select 5-12 active lags from top 25
                top25 = np.argsort(lags)[-25:]
                n_sel = rng.integers(5, 13)
                sel = rng.choice(top25, size=n_sel, replace=False)
                # Solve LP with selected lags
                grads = np.array([gradient_lag(h_cur, p) for p in sel])
                k = len(sel)
                c = np.zeros(n + 1)
                c[n] = 1.0
                A_ub = np.zeros((k, n + 1))
                for i in range(k):
                    A_ub[i, :n] = grads[i]
                    A_ub[i, n] = -1.0
                b_ub = np.zeros(k)
                tol_box = 1e-6
                box_rows = []
                box_rhs = []
                for i in range(n):
                    if h_cur[i] <= tol_box:
                        row = np.zeros(n + 1)
                        row[i] = -1.0
                        box_rows.append(row)
                        box_rhs.append(0.0)
                    elif h_cur[i] >= 1.0 - tol_box:
                        row = np.zeros(n + 1)
                        row[i] = 1.0
                        box_rows.append(row)
                        box_rhs.append(0.0)
                if box_rows:
                    A_ub = np.vstack([A_ub, np.array(box_rows)])
                    b_ub = np.concatenate([b_ub, np.array(box_rhs)])
                A_eq = np.zeros((1, n + 1))
                A_eq[0, :n] = 1.0
                b_eq = np.zeros(1)
                bounds = [(-1.0, 1.0)] * n + [(-1.0, None)]
                try:
                    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                  bounds=bounds, method='highs')
                except Exception:
                    break
                if res.status != 0 or res.x is None:
                    break
                d = res.x[:n]
                norm_d = np.linalg.norm(d)
                if norm_d < 1e-10:
                    break
                d = d / norm_d
                h_new2, sc_new2 = step_direction(h_cur, d, n_trials=20)
                if sc_new2 < sc_cur - 1e-14:
                    h_cur = h_new2
                    sc_cur = sc_new2
                else:
                    break
            update(h_cur, official_score(h_cur))

    # Phase 7: Fourier multi-start
    if remaining() > 60:
        n = len(best_h)
        for trial in range(8):
            if remaining() < 15:
                break
            if trial == 0:
                h_f = symmetrize(best_h)
            else:
                max_harm = rng.integers(2, 5)
                coeffs = [(rng.normal(0, 0.15), rng.normal(0, 0.1)) for _ in range(max_harm)]
                h_f = fourier_start(n, coeffs, rng)
            sc_f = official_score(h_f)
            # Quick refine
            for it in range(20):
                if remaining() < 5:
                    break
                d = lp_direction(h_f, k_active=15, tol=0.005, step_bound=1.0)
                if d is None:
                    break
                h_new2, sc_new2 = step_direction(h_f, d, n_trials=20)
                if sc_new2 < official_score(h_f) - 1e-14:
                    h_f = h_new2
                else:
                    break
            h_f, sc_f2 = qp_refine(h_f, k_active=6, reg=0.1, max_iter=8)
            sc_f2 = official_score(h_f)
            update(h_f, sc_f2)

    # Phase 8: Final intensive polish
    if remaining() > 40:
        h_final = best_h.copy()
        for it in range(30):
            if remaining() < 5:
                break
            d = lp_direction(h_final, k_active=25, tol=0.003, step_bound=1.0)
            if d is None:
                break
            h_new2, sc_new2 = step_direction(h_final, d, n_trials=30)
            if sc_new2 < official_score(h_final) - 1e-14:
                h_final = h_new2
            else:
                break
        update(h_final, official_score(h_final))

    if remaining() > 30:
        h_final, sc = coord_descent(best_h, n_rounds=5, n_delta_trials=10)
        update(h_final, sc)

    if remaining() > 20:
        h_final, sc = qp_refine(best_h, k_active=12, reg=0.003, max_iter=10)
        update(h_final, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    # Ensure the reported bound is at least the actual score
    final_bound = final_score + 1e-12
    return final_h, final_bound, len(final_h)
