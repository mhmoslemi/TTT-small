def _pair_refine(h, deadline, max_passes=50):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()

            if time.time() > deadline:

        if time.time() > dl1:

        h_t = _slsqp_lse_solve(best_h, min(dl1, time.time() + budget_s*0.05), beta=beta, maxiter=120)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-10:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 2: Projected gradient with increasing beta
    dl2 = t0 + budget_s * 0.38
    for beta in [30.0, 80.0, 150.0, 300.0]:
        if time.time() > dl2:
            break
        h_pg, c5_pg = _proj_grad_lse(best_h, min(dl2, time.time() + budget_s*0.045), beta=beta, max_iters=350, lr_init=0.012)
        if c5_pg < best_c5 - 1e-10:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Diverse candidates
    dl3 = t0 + budget_s * 0.68
    cands = [best_h.copy()]
    # Symmetrized
    h_sym = _project((best_h + best_h[::-1]) / 2.0, best_n / 2.0)
    cands.append(h_sym)
    # Binary
    h_bin = (best_h >= 0.5).astype(float)
    if int(h_bin.sum()) != best_n // 2:
        idx = np.argsort(best_h)
        h_bin = np.zeros(best_n)
        h_bin[idx[best_n//2:]] = 1.0
    cands.append(h_bin)
    # Cosine variants
    for k in [1, 2, 3, 4, 5]:
        for ph in [0.0, np.pi*0.5, np.pi]:
            x = np.arange(best_n)
            hc = 0.5 + 0.38 * np.cos(2.0*np.pi*k*x/best_n + ph)
            hc = _project(hc, best_n / 2.0)
            cands.append(hc)
    # Perturbations
    for sc in [0.02, 0.05, 0.10, 0.18]:
        hp = best_h + rng.normal(0, sc, best_n)
        hp = _project(hp, best_n / 2.0)
        cands.append(hp)
    # Convex combos
    for w in [0.3, 0.5, 0.7]:
        hc = w * best_h + (1-w) * h_sym
        hc = _project(hc, best_n / 2.0)
        cands.append(hc)

    cands.sort(key=lambda x: _exact_c5(x))

    for cand in cands[:10]:
        if time.time() > dl3:
            break
        c_cand = _exact_c5(cand)
        if c_cand > best_c5 + 0.005:
            continue
        dl_sub = min(dl3, time.time() + budget_s * 0.03)
        h_s = _slsqp_lse_solve(cand, dl_sub, beta=100.0, maxiter=80)
        c5_s = _exact_c5(h_s)
        if c5_s < best_c5 - 1e-10:
            best_c5 = c5_s
            best_h = h_s.copy()
        dl_sub2 = min(dl3, time.time() + budget_s * 0.025)
        h_g, c5_g = _proj_grad_lse(h_s, dl_sub2, beta=200.0, max_iters=200, lr_init=0.01)
        if c5_g < best_c5 - 1e-10:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 4: Binary SA
    if time.time() < t0 + budget_s * 0.80:
        dl4 = t0 + budget_s * 0.82
        h_b, c5_b = _binary_sa(best_h, min(dl4, time.time() + 40), max_iters=80000, seed=seed)
        if h_b is not None and c5_b < best_c5 - 1e-10:
            best_c5 = c5_b
            best_h = h_b.copy()
        for ci in range(1, min(5, len(cands))):
            if time.time() > dl4:
                break
            h_b2, c5_b2 = _binary_sa(cands[ci], min(dl4, time.time() + 20), max_iters=40000, seed=seed+ci)
            if h_b2 is not None and c5_b2 < best_c5 - 1e-10:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 5: Pair refinement
    if time.time() < t0 + budget_s * 0.90:
        dl5 = t0 + budget_s * 0.92
        h_pr, c5_pr = _pair_refine(best_h, dl5, max_passes=80)
        if c5_pr < best_c5 - 1e-10:
            best_c5 = c5_pr
            best_h = h_pr.copy()

    # Phase 6: Epigraph polish
    if time.time() < t0 + budget_s * 0.95:
        dl6 = t0 + budget_s * 0.96
        h_ep = _epigraph_polish(best_h, dl6, maxiter=60)
        c5_ep = _exact_c5(h_ep)
        if c5_ep < best_c5 - 1e-10:
            best_c5 = c5_ep
            best_h = h_ep.copy()

    # Phase 7: Smaller resolutions
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.98
        for dn in [-4, -8, -2, -6, -12, -16, -10]:
            if time.time() > dl7:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            h_r = _proj_grad_lse(h_r, min(dl7, time.time() + budget_s*0.015), beta=100.0, max_iters=150, lr_init=0.01)[0]
            h_r = _pair_refine(h_r, min(dl7, time.time() + budget_s*0.01), max_passes=20)[0]
            c5_r = _exact_c5(h_r)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final
    if time.time() < t0 + budget_s * 0.993:
        h_f = _proj_grad_lse(best_h, t0 + budget_s * 0.995, beta=300.0, max_iters=100, lr_init=0.005)[0]
        c5_f = _exact_c5(h_f)
        if c5_f < best_c5 - 1e-10:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
