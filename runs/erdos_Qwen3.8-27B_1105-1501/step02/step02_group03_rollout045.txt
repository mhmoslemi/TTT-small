def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -11.0, 11.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in range(len(order)):
                add = min(residual, 1.0 - h[order[idx]])
                if add <= 0:
                    break
                h[order[idx]] += add
                residual -= add
                if abs(residual) < 1e-15:

def compute_grad_matrix(h):

    For even n=2M: h[i]=z[i], h[n-1-i]=1-z[i]

def antisym_grad(G, n):
    """Given full gradient matrix G (2n-1, n), compute gradient w.r.t. free anti-symmetric vars.
    Returns array of shape (2n-1, M) where M=n//2.
    grad[:, i] = G[:, i] - G[:, n-1-i]"""
    M = n // 2
    idx = np.arange(n - 1, n - 1 - M, -1)
    return G[:, :M].copy() - G[:, idx].copy()

def logsumexp(vals, beta):
    """Stable log-sum-exp of beta*(vals - max(vals)) + max(vals)*beta"""
    shifted = beta * (vals - vals.max())
    amax = shifted.max()
    if amax > 500:
        amax = 500
    return vals.max() + (amax + np.log(np.sum(np.exp(shifted - amax)))) / beta

def bfgs_sym(z_init, n, beta_schedule, time_limit, z_range=8.0):
    """Optimize anti-symmetric logistic parameters using BFGS with log-sum-exp."""
    M = n // 2
    z = z_init.copy()
    best_z = z.copy()
    best_score = official_score(antisym_h(np.clip(z, 0, 1), n))
    
    t0 = time.time()

            break

            amax = vshifted.max()

            # Gradient

            pass
