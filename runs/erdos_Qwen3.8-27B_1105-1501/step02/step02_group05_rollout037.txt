from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < s < n:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    elif -n < s < 0:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def solve_lp(h, active, delta):
    n = len(h)
    # Maximize t s.t. g_a . d <= -t for active a, sum(d)=0, bounds
    c = np.zeros(n + 1)
    c[n] = -1.0  # minimize -t = maximize t
    
    # A_ub * x <= b_ub
    # Bounds on d: d_lo <= d <= d_hi
    d_lo = np.maximum(h - 1.0, -delta)
    d_hi = np.minimum(1.0 - h, delta)
    
    # -d_i <= -d_lo_i  → rows of -I (for d part)
    #  d_i <= d_hi_i   → rows of  I (for d part)
    # g_a . d + t <= 0 → for each active
    
    n_active = len(active)
    n_rows = 2 * n + n_active
    A = np.zeros((n_rows, n + 1))
    b = np.zeros(n_rows)
    
    row = 0
    # -d <= -d_lo
    A[row:row+n, :n] = -np.eye(n)
    b[row:row+n] = -d_lo
    row += n
    # d <= d_hi
    A[row:row+n, :n] = np.eye(n)
    b[row:row+n] = d_hi
    row += n
    # g_a . d + t <= 0
    n_off = n - 1
    for idx in range(n_active):
        p = int(active[idx])
        s = p - n_off
        g = grad_lag(h, s)
        A[row, :n] = g
        A[row, n] = 1.0
        b[row] = 0.0
        row += 1
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] * (n + 1), method='highs')

        else:

    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
