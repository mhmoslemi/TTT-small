def qp_step(h, vals, active_indices, mu, trust_radius, solver='ECOS'):
    n = len(h)
    dx_corr = 2.0 / n
    C = vals.max()
    
    # Build gradient matrix for active lags
    # Each row corresponds to one active lag
    m = len(active_indices)
    G = np.zeros((m, n))
    
    for idx, p in enumerate(active_indices):
        k = p - (n - 1)
        if k >= 0:
            for j in range(n - k):
                G[idx, j + k] += (1.0 - h[j])
                G[idx, j] -= h[j + k]
        else:
            t = -k
            for j in range(n - t):
                G[idx, j + t] += (1.0 - h[j])
                G[idx, j] -= h[j + t]
    G *= dx_corr
    
    # QP: minimize t + mu * ||d||^2
    # subject to: G @ d <= t - vals[active_indices]  (linearized)
    #             0 <= h + d <= 1
    #             sum(d) = 0  (mass constraint)
    #             ||d|| <= trust_radius
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        h + d >= 0,
        h + d <= 1,
    ]
    
    for idx in range(m):
        constraints.append(G[idx] @ d <= t - vals[active_indices[idx]])
    
    if trust_radius > 0:
        constraints.append(cp.norm(d) <= trust_radius)
    
    objective = cp.Minimize(t + mu * cp.sum_squares(d))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status in ['optimal', 'optimal_inaccurate'] and d.value is not None:
            return np.asarray(d.value).ravel(), t.value
    except Exception:
        pass
    return None, None
