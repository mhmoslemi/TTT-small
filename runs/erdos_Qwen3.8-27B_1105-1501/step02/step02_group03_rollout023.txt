def official_score(h):

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -11.0, 11.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            cap = 1.0 - h
            order = np.argsort(-cap)
            for idx in order:
                add = min(residual, cap[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_all(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    center = n - 1
    for p in range(2 * n - 1):
        s = p - center
        if s == 0:
            G[p, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            L = n - s
            G[p, :L] -= dx * h[s:]
            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            L = n - t
            G[p, :L] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:L]
    return G

def smooth_obj_grad(h, beta):
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_all(h)
    grad = w @ G
    return obj, grad

def pg_solve(h0, beta, time_limit, max_iter):
    n = len(h0)
    h = h0.copy()
    f, g = smooth_obj_grad(h, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-30:
        return h
    alpha = 1.0 / (np.sqrt(gnorm2) + 1e-10)
    alpha = min(alpha, 5.0 / (float(np.max(g**2)) + 1e-12))
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-30:
            break
        step = alpha
        y = h.copy()
        f_trial = f
        for ls_iter in range(25):
            y = make_feasible(h - step * g)
            f_trial = smooth_obj_grad(y, beta)[0]
            if f_trial <= f - 1e-4 * step * gnorm2:
                break
            step *= 0.5
        if step < 1e-14 or not np.isfinite(f_trial):
            break
        f_new = f_trial
        h = y
        if f_new < f - 1e-8:
            alpha = min(alpha * 1.6, 10.0 / (gnorm2 + 1e-12))
        else:
            alpha *= 0.6
        f = f_new
    return h

def minimize_1d_max_linear(a, b):
    """Minimize max_i(a[i]*x + b[i]) over x in [0,1] via subgradient bisection."""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        vmax = vals.max()
        active = np.abs(vals - vmax) < 1e-13 + 1e-10 * abs(vmax)
        slopes_active = a[active]
        if slopes_active.min() > 1e-12:
            hi = mid
        elif slopes_active.max() < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def coord_descent_pass(h, n_sweeps=10):
    n = len(h)
    for sweep in range(n_sweeps):
        for j in range(n):
            vals = compute_vals(h)
            G = grad_all(h)
            a = G[:, j]
            b = vals - a * h[j]
            new_val = minimize_1d_max_linear(a, b)
            if abs(new_val - h[j]) > 1e-10:
                h[j] = new_val
                h = make_feasible(h)
    return h

class FullNLP:
    def __init__(self, n):
        self.n = n
        self.ncons = 2 * n - 1
    
    def objective(self, z):
        return z[-1]
    
    def objective_grad(self, z):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(self, z):
        h = z[:self.n]
        t = z[-1]
        vals = compute_vals(h)
        return t - vals
    
    def ineq_jac(self, z):
        h = z[:self.n]
        G = grad_all(h)
        return np.hstack([-G, np.ones((self.ncons, 1))])
    
    def eq_fun(self, z):
        return np.sum(z[:self.n]) - self.n / 2.0
    
    def eq_jac(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g
