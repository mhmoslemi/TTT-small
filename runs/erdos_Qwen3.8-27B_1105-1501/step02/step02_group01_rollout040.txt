    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    """Project z onto {h: 0<=h<=1, sum(h)=target}."""
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection on offset
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

    """Push h towards binary by raising to power p and reprojecting."""
    return make_feasible(h ** p)

def compute_grad(h):
    """Compute gradient matrix: grad[i, p] = ∂c[p]/∂h[i]
    where c = np.correlate(h, 1-h, 'full')"""
    n = len(h)
    # c[p] = sum_m h[m]*(1-h[m+q]) where q = n-1-p
    # ∂c[p]/∂h[i]: 
    #   term 1: m=i, gives (1-h[i+q]) if i+q in [0,n-1]
    #   term 2: m+q=i => m=i-q, gives -h[i-q] if i-q in [0,n-1]
    # So grad[i,p] = (1-h[i+q])*valid1 - h[i-q]*valid2 where q=n-1-p
    
    # Vectorized: for each i, grad[i,p] is nonzero only for specific p values.
    # p = n-1-q, so q = n-1-p
    # term1 valid: 0 <= i+q <= n-1 => 0 <= i+(n-1-p) <= n-1 => 1 <= p <= n+i
    #   value: 1-h[i+n-1-p] = 1-h[2n-1-p-i+... ]
    #   wait: i+q = i+n-1-p. For this to be in [0,n-1]:
    #   0 <= i+n-1-p <= n-1 => 1 <= p <= n+i (and p <= 2n-2)
    #   value: 1 - h[i+n-1-p]
    
    # term2 valid: 0 <= i-q <= n-1 => 0 <= i-(n-1-p) <= n-1 => n-1-i <= p <= 2n-1-i (and p >= 0)
    #   value: -h[i-n+1+p]
    
    # So grad[i] has non-zero entries in p range [max(1, n-1-i), min(2n-2, 2n-1-i)] 
    # Actually let me just compute it directly for clarity.
    
    grad = np.zeros((n, 2*n-1))
    u = 1.0 - h
    for i in range(n):
        # term1: p from max(0, 1) to min(2n-2, n+i), value = u[i+n-1-p]
        p_lo1 = max(0, 1)  # p >= 1 for i+q >= 0 means p <= n+i
        # i+n-1-p >= 0 => p <= i+n-1
        # i+n-1-p <= n-1 => p >= i
        p_lo1 = max(0, i)
        p_hi1 = min(2*n-2, i+n-1)
        if p_lo1 <= p_hi1:
            p_range = np.arange(p_lo1, p_hi1+1)
            idx = i + n - 1 - p_range  # index into u
            grad[i, p_range] = u[idx]
        
        # term2: i-n+1+p >= 0 => p >= n-1-i
        # i-n+1+p <= n-1 => p <= 2n-2-i+... 
        # i-(n-1-p) = i-n+1+p, need 0 <= i-n+1+p <= n-1
        # p >= n-1-i and p <= 2n-2-i
        p_lo2 = max(0, n-1-i)
        p_hi2 = min(2*n-2, 2*n-2-i+1)  # i-n+1+p <= n-1 => p <= 2n-2-i
        p_hi2 = min(2*n-2, 2*n-2-i)
        if p_lo2 <= p_hi2:
            p_range = np.arange(p_lo2, p_hi2+1)
            idx = i - n + 1 + p_range  # index into h
            grad[i, p_range] -= h[idx]
    
    return grad

                break

        d_max[np.diag_indices(n)] = 0

        new_scores = np.max(new_c, axis=2) * dx  # (n, n)

        current_score = np.max(c) * dx
        improvements = current_score - new_scores
        improvements[np.diag_indices(n)] = 0
        
        best_idx = np.unravel_index(np.argmax(improvements), improvements.shape)
        best_improve = improvements[best_idx]

        h[j] += d

            p = np.arange(p_lo, p_hi + 1)
            grad[i, p] = u[i + n - 1 - p]
        # Term 2: -h[i-r] where r = n-1-p, valid when 0<=i-r<=n-1
        # r in [i-n+1, i], so p in [n-1-i, 2n-2-i]
        p_lo = max(0, n - 1 - i)
        p_hi = min(2 * n - 2, 2 * n - 2 - i)
        if p_lo <= p_hi:
            p = np.arange(p_lo, p_hi + 1)
            grad[i, p] -= h[i - n + 1 + p]
    return grad

def vectorized_local_search(h0, max_rounds=2000, time_limit=None, t_start=None):
    """Best-improvement pairwise swap local search, vectorized."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        current_score = np.max(c) * dx
        
        # If current is worse than best, use best
        if current_score > best_score + 1e-10:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode='full')
            current_score = best_score
        
        grad = compute_grad(h)
        
        # d_max[i,j] = min(h[i], 1-h[j])
        d_max = np.minimum(h[:, None], 1.0 - h[None, :])
        d_max[np.diag_indices(n)] = 0.0
        
        # Only consider pairs where d_max > 0.01
        mask = d_max > 0.01
        
        if not np.any(mask):
            break
        
        # Compute new scores for all valid pairs
        # new_c[i,j,p] = c[p] + d_max[i,j] * (grad[j,p] - grad[i,p])
        diff = grad[None, :, :] - grad[:, None, :]  # (n, n, 2n-1)
        new_c = c[None, None, :] + d_max[:, :, None] * diff
        new_scores = np.max(new_c, axis=2) * dx  # (n, n)
        new_scores[~mask] = current_score + 1.0  # mask out invalid
        
        improvements = current_score - new_scores
        best_flat = np.argmax(improvements)
        best_i, best_j = np.unravel_index(best_flat, (n, n))
        best_improve = improvements[best_flat]
        
        if best_improve < 1e-12:
            no_improve += 1
            if no_improve > 3:
                break
            # Try a random perturbation to escape
            i = np.random.choice(n)
            j = np.random.choice(n)
            d = min(h[i], 1.0 - h[j])
            if d > 0.5:
                h[i] -= d
                h[j] += d
                h = np.clip(h, 0, 1)
            continue
        
        # Apply the best swap
        d = d_max[best_i, best_j]
        h[best_i] -= d
        h[best_j] += d
        h = np.clip(h, 0.0, 1.0)
        
        new_actual = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        if new_actual < best_score - 1e-14:
            best_score = new_actual
            best_h = h.copy()
            no_improve = 0
        else:
            # Rollback
            h[best_i] += d
            h[best_j] -= d
            no_improve += 1
            if no_improve > 3:
                break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_grad(h):
    n = len(h)
    u = 1.0 - h
    grad = np.zeros((n, 2 * n - 1))
    for i in range(n):
        p_lo = max(0, i)
        p_hi = min(2 * n - 2, n - 1 + i)
        if p_lo <= p_hi:
            p = np.arange(p_lo, p_hi + 1)
            grad[i, p] = u[i + n - 1 - p]
        p_lo2 = max(0, n - 1 - i)
        p_hi2 = min(2 * n - 2, 2 * n - 2 - i)
        if p_lo2 <= p_hi2:
            p2 = np.arange(p_lo2, p_hi2 + 1)
            grad[i, p2] -= h[i - n + 1 + p2]
    return grad

def local_search_v(h0, max_rounds=5000, time_limit=None, t_start=None, rng=None):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    stall = 0

    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = np.max(c) * dx
        if cur > best_score + 1e-10:
            h = best_h.copy()
            c = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
            cur = best_score
        grad = compute_grad(h)
        d_max = np.minimum(h[:, None], 1.0 - h[None, :])
        d_max[np.diag_indices(n)] = 0.0
        mask = d_max > 0.005
        if not np.any(mask):
            break
        diff = grad[None, :, :] - grad[:, None, :]
        new_c = c[None, None, :] + d_max[:, :, None] * diff
        new_sc = np.max(new_c, axis=2) * dx
        new_sc[~mask] = cur + 1.0
        imp = cur - new_sc
        bi, bj = np.unravel_index(np.argmax(imp), (n, n))
        if imp[bi, bj] < 1e-13:
            stall += 1
            if stall > 5:
                break
            if rng is not None:
                ii = rng.integers(n)
                jj = rng.integers(n)
                dd = min(h[ii], 1.0 - h[jj])
                if dd > 0.4:
                    h[ii] -= dd
                    h[jj] += dd
                    h = np.clip(h, 0, 1)
            continue
        d = d_max[bi, bj]
        h[bi] -= d
        h[bj] += d
        h = np.clip(h, 0.0, 1.0)
        na = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        if na < best_score - 1e-14:
            best_score = na
            best_h = h.copy()
            stall = 0
        else:
            h[bi] += d
            h[bj] -= d
            stall += 1
            if stall > 5:
                break
    return best_h, best_score

def local_search_sa(h0, rng, max_rounds=3000, T_start=0.01, T_end=0.0005,
                    time_limit=None, t_start=None):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    T = T_start
    cool = (T_end / T_start) ** (1.0 / max(max_rounds, 1))

    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = np.max(c) * dx
        grad = compute_grad(h)
        d_max = np.minimum(h[:, None], 1.0 - h[None, :])
        d_max[np.diag_indices(n)] = 0.0
        mask = d_max > 0.005
        if not np.any(mask):
            T *= cool
            continue
        diff = grad[None, :, :] - grad[:, None, :]
        new_c = c[None, None, :] + d_max[:, :, None] * diff
        new_sc = np.max(new_c, axis=2) * dx
        new_sc[~mask] = cur + 1.0
        flat_imp = (cur - new_sc).ravel()
        valid_flat = mask.ravel()
        if not np.any(valid_flat):
            break
        # Simulated annealing: accept with probability
        bad = flat_imp < 0
        good = flat_imp >= 0
        idx = np.where(valid_flat)[0]
        if np.any(good[idx]):
            candidates = idx[good[idx]]
            if len(candidates) > 20:
                candidates = rng.choice(candidates, 20, replace=False)
            pick = candidates[rng.integers(len(candidates))]
        else:
            candidates = idx[valid_flat]
            deltas = -flat_imp[candidates] / T
            probs = np.exp(np.clip(deltas, -500, 0))
            probs /= probs.sum()
            pick = candidates[rng.choice(len(candidates), p=probs)]
        bi, bj = np.unravel_index(pick, (n, n))
        d = d_max[bi, bj]
        h[bi] -= d
        h[bj] += d
        h = np.clip(h, 0.0, 1.0)
        na = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        if na < best_score - 1e-14:
            best_score = na
            best_h = h.copy()
        else:
            if flat_imp[pick] < 0 and rng.random() > np.exp(flat_imp[pick] / T):
                h[bi] += d
                h[bj] -= d
        T *= cool
    return best_h, best_score

def lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = e / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                g += (1.0 - x) - x
            else:
                g[s2:] += (1.0 - x[:n - s2])
                g[:n - s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                g += (1.0 - x) - x
            else:
                g[:n - m2] += (1.0 - x[m2:])
                g[m2:] += (-x[:n - m2])
    return g

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            lse_obj, h0, args=(beta, dx),
            jac=lambda x: lse_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def slsqp_direct(h0, maxiter=80):
    n = len(h0)
    dx = 2.0 / n
    t0 = np.max(np.correlate(h0, 1.0 - h0, mode='full')) * dx + 0.01
    t0 = min(max(t0, 0.01), 0.99)
    z0 = np.concatenate([h0, [t0]])

    def mm_obj(z):
        return z[-1]

    def mm_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def mm_ineq(z):
        h = z[:-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return z[-1] - corr

    def mm_ineq_jac(z):
        h = z[:-1]
        jjac = np.zeros((2 * n - 1, n + 1))
        jjac[:, -1] = 1.0
        for k in range(2 * n - 1):
            s2 = k - (n - 1)
            if s2 >= 0:
                if s2 == 0:
                    jjac[k, :n] = -dx * (2.0 * h - 1.0)
                else:
                    jjac[k, s2:n] -= dx * (1.0 - h[:n - s2])
                    jjac[k, :n - s2] -= dx * (-h[s2:])
            else:
                m2 = -s2
                if m2 == 0:
                    jjac[k, :n] = -dx * (2.0 * h - 1.0)
                else:
                    jjac[k, :n - m2] -= dx * (1.0 - h[m2:])
                    jjac[k, m2:n] -= dx * (-h[:n - m2])
        return jjac

    def mm_eq(z):
        return np.sum(z[:-1]) - n / 2.0

    def mm_eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(
            mm_obj, z0, jac=mm_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
                {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0

                best_h = fl_h
                best_score = fl_sc

    # Final feasibility check
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if final_score > best_score:
        final_score = best_score

    return (best_h, final_score, len(best_h))
