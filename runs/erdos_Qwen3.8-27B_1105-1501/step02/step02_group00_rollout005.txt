def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _shift_scores(h):
    """Compute all shift overlap scores (unscaled)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    """Compute the gradient matrix G where G[k, i] = ∂score_k/∂h[i]."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}."""

    return m + np.log(s) / beta

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _shift_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _logsumexp_val(scores, beta):
    m = float(np.max(scores))
    if beta < 1e-10:
        return m
    vals = np.exp(beta * (scores - m))
    s = float(np.sum(vals))
    if s < 1e-300:
        return m
    return m + float(np.log(s)) / beta

def _smoothed_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = _shift_scores(h)
    return _logsumexp_val(scores, beta) * dx

def _smoothed_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = _shift_scores(h)
    m = float(np.max(scores))
    w = np.exp(beta * (scores - m))
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    return (G.T @ w) * dx

def _lp_step(h, scores, G, reduction):
    n = len(h)
    target_level = float(np.max(scores)) - reduction
    A_ub = G.copy()
    b_ub = target_level - scores
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    c = np.zeros(n)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)

    best_a = 0.0
    best_c5 = c5_cur
    for a in np.linspace(0.01, alpha_max, 12):
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-11:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)

    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_optim(h, deadline, radius_init=0.1, max_iters=5000):

    target = n / 2.0
    h = h.copy()

    best_h = h.copy()

def _slsqp_optim(h, deadline, beta_start=40.0, beta_end=400.0, max_iters=25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (it / max(max_iters - 1, 1))

        try:
            res = minimize(lambda x: _smoothed_obj(x, beta), h,
                          jac=lambda x: _smoothed_grad(x, beta),
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 80, 'ftol': 1e-13})
            if res.x is not None:
                h_try = _project(res.x, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
        except Exception:
            pass

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                    best_a = 0.0
