import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            cands = np.where(result < 0.999)[0]
        else:
            cands = np.where(result > 0.001)[0]
        if len(cands) > 0:
            result[cands] += residual / len(cands)
            result = np.clip(result, 0.0, 1.0)
    return result


def smooth_value(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = np.max(corr)
    vals = (corr - M) / T
    logsum = np.log(np.sum(np.exp(vals)))
    return dx * (M + T * logsum)


def smooth_gradient(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = np.max(corr)
    vals = (corr - M) / T
    w = np.exp(vals)
    ws = np.sum(w)
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    v = 1.0 - x
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j_lo = max(0, s)
        j_hi = min(n - 1, s + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += wk * v[idx - s]
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1)
            grad[idx] -= wk * x[idx + s]
    return grad * dx


def _slsqp_fun(x, T, n):
    return smooth_value(x, T, n)


def _slsqp_jac(x, T, n):
    return smooth_gradient(x, T, n)


def _eq_fun(x, n):
    return np.sum(x) - n / 2.0


def _eq_jac(x, n):
    return np.ones(len(x))


def do_slsqp(h0, T, n, maxiter=80):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_fun, h0, jac=_slsqp_jac, args=(T, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if np.all(np.isfinite(res.x)) and res.fun < 1e6:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h0.copy()


def anneal(h0, n, T_schedule, maxiter_per_T=80, deadline=None):
    h = h0.copy()
    best_local = h.copy()
    best_c5 = compute_c5(h)
    for T in T_schedule:
        if deadline is not None and time.time() > deadline:
            break
