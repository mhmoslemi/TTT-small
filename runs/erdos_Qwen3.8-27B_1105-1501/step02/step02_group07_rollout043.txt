    h = h.copy()
    V = compute_values(h)
    best_h = h.copy()
    best_score = V.max()
    
    for p in range(n_passes):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for i in order:
            c = compute_sensitivities(h, i)
            x_opt = coord_minimize(c, V, h[i], lam)
            if abs(x_opt - h[i]) > 1e-10:
                # Update V incrementally
                delta = x_opt - h[i]
                V += c * delta
                h[i] = x_opt

        V = compute_values(h)  # Recompute exactly (to avoid accumulated error)
        score = V.max()
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score
