def pairwise_exchange(h, max_pairs=200, grid_points=15):
    n = len(h)
    best_score = official_score(h)
    improved = True
    while improved:
        improved = False
        # Sample pairs
        pairs = []
        for _ in range(max_pairs):
            i, j = np.random.randint(0, n, 2)
            if i != j:
                pairs.append((min(i,j), max(i,j)))
        # Also include some systematic pairs
        for i in range(0, n, 3):
            for j in range(i+1, n, 7):
                pairs.append((i, j))
        
        for (i, j) in pairs:
            S = h[i] + h[j]
            L = max(0.0, S - 1.0)
            U = min(1.0, S)
            if U - L < 1e-10:
                continue
            # Grid search
            xs = np.linspace(L, U, grid_points)
            best_x = xs[0]
            best_f = official_score(np.where(np.arange(n)==i, xs[0], np.where(np.arange(n)==j, S-xs[0], h)))
            for x in xs[1:]:
                h_try = h.copy()
                h_try[i] = x
                h_try[j] = S - x
                f = official_score(h_try)
                if f < best_f:
                    best_f = f
                    best_x = x
            if best_f < best_score - 1e-12:
                h[i] = best_x
                h[j] = S - best_x
                best_score = best_f
                improved = True
    return h, best_score

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -50.0, 50.0)
    ext = float(np.max(np.abs(z))) + 2.0
    lo, hi = -ext, ext
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            if residual <= 1e-14:
                break
            a = min(residual, 1.0 - h[idx])
            h[idx] += a
            residual -= a
    elif residual < -1e-12:
        order = np.argsort(h)
        for idx in order:
            if residual >= -1e-14:
                break
            a = min(-residual, h[idx])
            h[idx] -= a
            residual += a
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_basis(n, d):
    B = np.zeros((n, d))
    idx = np.arange(n) + 0.5
    for k in range(d):
        mode = k + 1
        B[:, k] = np.cos(np.pi * mode * idx / n)
        nm = np.linalg.norm(B[:, k])
        if nm > 1e-10:
            B[:, k] /= nm
    return B

def score_from_theta(B, theta):
    z = 0.5 + B @ theta
    h = make_feasible(z)
    return official_score(h), h

def dct_evolve(B, d, rng, n_iter, n_pop, deadline):
    scores = []
    thetas = []
    for _ in range(n_pop):
        th = rng.normal(0, 1.5, d)
        sc, _ = score_from_theta(B, th)
        scores.append(sc)
        thetas.append(th.copy())
    scores = np.array(scores)
    thetas = np.array(thetas)
    order = np.argsort(scores)
    scores = scores[order]
    thetas = thetas[order]
    best_sc = scores[0]
    best_th = thetas[0].copy()
    sigma = 0.7
    for gen in range(n_iter):
        if time.time() > deadline:
            break
        n_child = n_pop // 2
        child_sc = []
        child_th = []
        sel = max(1, n_pop // 4)
        for _ in range(n_child):
            ia = rng.integers(0, sel)
            ib = rng.integers(0, sel)
            ic = rng.integers(0, sel)
            ch = thetas[ia] + rng.uniform(0.5, 1.3) * (thetas[ib] - thetas[ic])
            ch = ch + rng.normal(0, sigma, d)
            ch = np.clip(ch, -5.0, 5.0)
            sc, _ = score_from_theta(B, ch)
            child_sc.append(sc)
            child_th.append(ch)

        child_th = np.array(child_th)

def pairwise_optimize(h, rng, max_passes, n_pairs, grid_n, deadline):
    n = len(h)
    best_score = official_score(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        nontriv = [i for i in range(n) if 0.03 < h[i] < 0.97]
        if len(nontriv) < 4:
            break
        pair_set = set()
        n_rand = min(n_pairs, len(nontriv) * 2)
        for _ in range(n_rand):
            i = nontriv[rng.integers(0, len(nontriv))]
            j = nontriv[rng.integers(0, len(nontriv))]
            if i != j:
                pair_set.add((min(i, j), max(i, j)))
        for i in range(0, n, 3):
            for dj in [1, 2, 4, 6, 10, 15]:
                j = (i + dj) % n
                if j != i:
                    pair_set.add((min(i, j), max(i, j)))
        for (i, j) in pair_set:
            if time.time() > deadline:
                break
            S = h[i] + h[j]
            L = max(0.0, S - 1.0)
            U = min(1.0, S)
            if U - L < 1e-8:
                continue
            xs = np.linspace(L, U, grid_n)
            best_f = best_score
            best_x = h[i]
            hi_save = h[i]
            hj_save = h[j]
            for x in xs:
                h[i] = x
                h[j] = S - x
                f = official_score(h)
                if f < best_f:
                    best_f = f
                    best_x = x
            h[i] = hi_save
            h[j] = hj_save
            if best_f < best_score - 1e-13:
                h[i] = best_x
                h[j] = S - best_x
                best_score = best_f
                improved = True
        if not improved:
            break
    return h, best_score

def active_lp(h, eps, trust):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None
    if len(active) > 50:
        top = np.argsort(-V[active])[:50]
        active = active[top]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] += (1.0 - h[0:L])
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] += (1.0 - h[t:n])
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None
    bounds_list = [(-trust, trust)] * n
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return (h_new, sc)
    except Exception:
        pass
    return None

def interpolate_to(h_src, n_tgt):

    h_int = np.interp(tgt_c, src_c, h_src)
