def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {x : 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny numerical error
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        # Find coordinates with room
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    return result

def smooth_objective_and_grad(x, n, T):
    """Compute smooth minimax objective and gradient."""
    dx = 2.0 / n
    h = x
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    m = np.max(corr)
    # Stable log-sum-exp
    shifted = (corr - m) / T
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    value = m + T * log_sum
    
    # Weights: softmax((corr - m)/T)
    weights = np.exp(shifted - max_shifted)
    weights /= np.sum(weights)
    
    # Gradient: for each i
    # grad[i] = dx * sum_k weights[k] * [(1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        # First term: (1-h[j]) where j = i-k, i.e., i = j+k
        # For fixed k, contributes to grad[i] for i = j+k where j in [0,n-1]
        # So i ranges from k to k+n-1, but i must be in [0, n-1]
        i_lo = max(0, k)
        i_hi = min(n - 1, k + n - 1)
        if i_lo <= i_hi:
            j_lo = i_lo - k
            j_hi = i_hi - k
            grad[i_lo:i_hi+1] += w * (1.0 - h[j_lo:j_hi+1])
        # Second term: -h[j] where j = i+k, i.e., i = j-k

        if i_lo2 <= i_hi2:
            j_lo2 = i_lo2 + k
            j_hi2 = i_hi2 + k
            grad[i_lo2:i_hi2+1] -= w * h[j_lo2:j_hi2+1]
    
    grad *= dx
    value *= dx
    return value, grad

def project_feasible(h, target_sum):
    n = len(h)
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-8)[0]
        else:
            idx = np.where(result > 1e-8)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    return result

def make_smooth_obj(n, T):
    dx = 2.0 / n
    def obj(x):
        h = x
        v = 1.0 - h
        corr = np.correlate(h, v, mode="full") * dx

        max_s = np.max(shifted)
