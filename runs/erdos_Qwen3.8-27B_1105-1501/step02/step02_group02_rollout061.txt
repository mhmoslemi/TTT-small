def _coord_min_single(h, i, n):
    """Find optimal h[i] given all other coordinates fixed. Returns (x_opt, g_at_x_opt)."""
    # Precompute for each s the coefficients
    # F_s(x) = (2/n) * (x * (1 - a_s - b_s) + b_s + C_s) for s > 0
    # F_0(x) = (2/n) * (-x^2 + x + C_0)
    
    # For s = 0:
    # F_0 = (2/n) * sum_j h[j](1-h[j])
    # = (2/n) * (h[i](1-h[i]) + sum_{j≠i} h[j](1-h[j]))
    C_0 = 0.0
    for j in range(n):
        if j != i:
            C_0 += h[j] * (1.0 - h[j])
    
    # For s > 0, s = 1, ..., n-1:
    # F_s = (2/n) * sum_{j=0}^{n-s-1} h[j+s](1-h[j])
    # h[i] appears as:
    #   h[j+s] when j = i-s (valid if s <= i <= n-1): contributes x*(1-h[i-s])
    #   (1-h[j]) when j = i (valid if 0 <= i <= n-s-1): contributes h[i+s]*(1-x)
    
    # Precompute rest of sum for each s
    coeffs = []  # list of (slope, intercept) for F_s, s > 0
    for s in range(1, n):
        # Compute C_s = sum of terms not involving h[i]
        C_s = 0.0
        a = 0.0  # h[i-s] if valid
        b = 0.0  # h[i+s] if valid
        for j in range(n - s):
            # term is h[j+s] * (1-h[j])
            if j + s == i:  # h[i] is the first factor
                # contributes x * (1 - h[j]) = x * (1 - h[i-s])
                a = h[i - s]  # this is h[j] where j = i-s
                # Don't add to C_s
            elif j == i:  # h[i] is in (1-h[j])
                # contributes h[j+s] * (1-x) = h[i+s] * (1-x)
                b = h[i + s]  # this is h[j+s] where j = i
                # Don't add to C_s
            else:
                C_s += h[j + s] * (1.0 - h[j])
        
        # F_s(x) = (2/n) * (x*(1-a) + b*(1-x) + C_s)
        # = (2/n) * (x - x*a + b - x*b + C_s)
        # = (2/n) * (x*(1-a-b) + b + C_s)
        slope = 1.0 - a - b
        intercept = b + C_s
        coeffs.append((slope, intercept, s))
    
    # Now find min of g(x) = max(F_0(x), F_1(x), ..., F_{n-1}(x)) on [0,1]
    # F_0(x) = (2/n) * (-x^2 + x + C_0)
    # F_s(x) = (2/n) * (slope_s * x + intercept_s) for s > 0
    
    factor = 2.0 / n
    
    def g0(x):
        return factor * (-x*x + x + C_0)
    
    def gs(slope, intercept, x):
        return factor * (slope * x + intercept)
    
    # Find intersection of F_0 and each F_s:
    # factor * (-x^2 + x + C_0) = factor * (slope * x + intercept)
    # -x^2 + x + C_0 = slope * x + intercept
    # -x^2 + (1-slope)*x + (C_0 - intercept) = 0
    # x^2 - (1-slope)*x - (C_0 - intercept) = 0
    
    candidates = [0.0, 1.0, 0.5]
    
    for slope, intercept, s in coeffs:
        # Solve: x^2 - (1-slope)*x - (C_0 - intercept) = 0
        A = 1.0
        B = -(1.0 - slope)
        C = -(C_0 - intercept)
        disc = B*B - 4*A*C
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            x1 = (-B + sqrt_disc) / (2*A)
            x2 = (-B - sqrt_disc) / (2*A)
            if 0.0 <= x1 <= 1.0:
                candidates.append(x1)
            if 0.0 <= x2 <= 1.0:
                candidates.append(x2)
    
    # Also find intersections between pairs of linear functions:
    # factor * (s1*x + i1) = factor * (s2*x + i2)
    # (s1-s2)*x = i2 - i1
    # This gives the vertices of the upper envelope of the linear functions
    # But we only need to check where the upper envelope changes
    
    # Actually, let's just be more systematic. The upper envelope of {F_0, F_1, ..., F_{n-1}}
    # changes at intersection points. Let's collect all pairwise intersections.
    
    # For efficiency, let's just add a denser grid as well
    for x in np.linspace(0, 1, 21):
        candidates.append(x)
    
    # For pairs of linear functions (s > 0, t > 0):
    for idx1 in range(len(coeffs)):
        for idx2 in range(idx1 + 1, len(coeffs)):
            s1, i1, ss1 = coeffs[idx1]
            s2, i2, ss2 = coeffs[idx2]
            if abs(s1 - s2) > 1e-15:
                x_int = (i2 - i1) / (s1 - s2)
                if 0.0 <= x_int <= 1.0:
                    candidates.append(x_int)
    
    # Evaluate g at all candidates
    best_x = 0.5
    best_g = float('inf')
    for x in candidates:
        x = max(0.0, min(1.0, x))
        val = g0(x)
        for slope, intercept, s in coeffs:
            v = gs(slope, intercept, x)
            if v > val:
                val = v
        if val < best_g:
            best_g = val
            best_x = x
    
    return best_x, best_g

    h = h0.astype(float).copy()
    n = len(h)
    factor = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * factor
    N = 2 * n - 1  # number of correlation values
    
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            c0 = corr[n - 1]

            # m ranges from -(n-1) to (n-1), k = m + (n-1) ranges from 0 to 2n-2
            m_vals = np.arange(N) - (n - 1)  # shape (N,)

            idx_plus = i + m_vals  # indices for h[i+m]
            a_arr = np.zeros(N)

            a_arr[valid_plus] = 1.0 - h[idx_plus[valid_plus]]

            idx_minus = i - m_vals  # indices for h[i-m]

            b_arr[valid_minus] = h[idx_minus[valid_minus]]

        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _coord_descent(h0, n_sweeps=200):
    """Exact coordinate descent with vectorized inner evaluation."""
    h = h0.astype(float).copy()
    n = len(h)
    factor = 2.0 / n
    N = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * factor
    m_vals = np.arange(N) - (n - 1)
    
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            c0 = corr[n - 1]
            idx_plus = i + m_vals
            idx_minus = i - m_vals
            a_arr = np.zeros(N)
            vp = (idx_plus >= 0) & (idx_plus < n)
            a_arr[vp] = 1.0 - h[idx_plus[vp]]
            b_arr = np.zeros(N)
            vm = (idx_minus >= 0) & (idx_minus < n)
            b_arr[vm] = h[idx_minus[vm]]
            a_coeff = a_arr - b_arr
            mask = np.ones(N, dtype=bool)
            mask[n - 1] = False
            ak = a_coeff[mask]
            ck = corr[mask]
            # Quadratic-linear intersections
            B_q = ak - 1.0
            C_q = old * old - old - ak * old + (ck - c0) / factor
            disc = B_q * B_q - 4.0 * C_q
            valid_d = disc >= 0
            cands = np.array([0.0, 1.0, old, 0.5])
            if np.any(valid_d):
                sq = np.sqrt(np.maximum(disc[valid_d], 0.0))
                Bv = B_q[valid_d]
                x1 = (-Bv + sq) / 2.0
                x2 = (-Bv - sq) / 2.0
                cands = np.concatenate([cands, x1, x2])
            cands = cands[(cands >= -1e-12) & (cands <= 1.0 + 1e-12)]
            cands = np.clip(cands, 0.0, 1.0)
            if len(cands) == 0:
                cands = np.array([old])
            # Vectorized evaluation
            qx = c0 + factor * (cands * (1.0 - cands) - old * (1.0 - old))
            lx = ck[None, :] + factor * (cands[:, None] - old) * ak[None, :]
            g_vals = np.maximum(qx, lx.max(axis=1))
            best_idx = np.argmin(g_vals)
            best_x = cands[best_idx]
            
            if abs(best_x - old) > 1e-12:
                h[i] = best_x
                max_change = max(max_change, abs(best_x - old))
                delta = factor * (best_x - old)
                corr[n - 1] += factor * (best_x * (1.0 - best_x) - old * (1.0 - old))
                corr[mask] += delta * a_coeff[mask]
        if max_change < 1e-11:
            break
    h = _project(h)
    return h

def _binary_local_search(h0, max_passes=80):
    """Greedy binary local search: single-swap moves."""
    n = len(h0)
    n_factor = 2.0 / n
    n_ones = n // 2
    idx = np.argsort(h0)[::-1][:n_ones]
    h = np.zeros(n)
    h[idx] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full') * n_factor
    cur_score = float(np.max(corr))
    ones_list = idx.tolist()
    zeros_list = [i for i in range(n) if i not in set(ones_list)]
    
    for _ in range(max_passes):
        best_improve = 0.0
        best_i = -1
        best_j = -1
        for i in zeros_list:
            for j in ones_list:
                h[i] = 1.0
                h[j] = 0.0
                new_corr = np.correlate(h, 1.0 - h, mode='full') * n_factor
                ns = float(np.max(new_corr))
                imp = cur_score - ns
                h[i] = 0.0
                h[j] = 1.0
                if imp > best_improve + 1e-15:
                    best_improve = imp
                    best_i = i
                    best_j = j
                    best_corr = new_corr
        if best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            ones_list.remove(best_j)
            ones_list.append(best_i)
            zeros_list.remove(best_i)
            zeros_list.append(best_j)
            corr = best_corr
            cur_score -= best_improve
        else:
            break
    return h

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _lse_obj(x, beta):
    n = len(x)
    cv = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    cv = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for p in range(j0, j1):
            g[p] += wk * (1.0 - x[p - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for p in range(p0, p1):
            g[p] += wk * (-x[p + s])
    return g

def _slsqp_smooth(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def _efun(x):
        return x.sum() - n / 2.0
    def _ejac(x):
        return np.ones(n)
    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _fista(h0, beta, n_iter, alpha_init=0.005, mu=0.88):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    vel = np.zeros(n)
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        h_new = y - alpha_init * (1.0 - mu) * vel + mu * mu * (h_cur - h_prev)
        h_new = _project(h_new)
        t += 1
        vel = h_cur - h_new
        h_prev = h_cur
        h_cur = h_new
        t_prev = t - 1
        t_new = 1.5 * t
        y = h_cur + (t_prev / t_new) * (h_cur - h_prev)
        y = _project(y)
        h_avg += (h_cur - h_avg) / t
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 2000 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.1 * budget_s)
    deadline = t_start + budget_s - reserve
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(4) * 0.5)
    
    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)
    rng = np.random.default_rng(seed)
    
    # Phase 1: Coordinate descent at parent's n
    if time.time() < deadline - 30:
        h_cd = _coord_descent(parent, n_sweeps=300)
        s = _score(h_cd)
        if s < best_score:
            best_h, best_score = h_cd, s
    
    # Phase 2: Try nearby n values with coordinate descent
    n_range = list(range(max(20, parent_n - 12), min(70, parent_n + 12) + 1, 2))
    for n_try in n_range:
        if time.time() > deadline - 20:
            break
        h_interp = _interp(parent, n_try)
        h_cd = _coord_descent(h_interp, n_sweeps=200)
        s = _score(h_cd)
        if s < best_score:
            best_h, best_score = h_cd, s
            best_n = n_try
    
    # Phase 3: Binary local search from best
    if time.time() < deadline - 20:
        h_bin = _binary_local_search(best_h, max_passes=60)
        s = _score(h_bin)
        if s < best_score:
            best_h, best_score = h_bin, s
            best_n = len(h_bin)
        # Also try binary search from rounded parent at different n
        for n_try in [best_n - 2, best_n, best_n + 2]:
            if time.time() > deadline - 15:
                break
            if n_try < 16:
                continue
            h_int = _interp(parent, n_try)
            h_bin2 = _binary_local_search(h_int, max_passes=40)
            s2 = _score(h_bin2)
            if s2 < best_score:
                best_h, best_score = h_bin2, s2
                best_n = n_try
    
    # Phase 4: Coordinate descent from binary solutions
    if time.time() < deadline - 20:
        h_cd2 = _coord_descent(best_h, n_sweeps=200)
        s = _score(h_cd2)
        if s < best_score:
            best_h, best_score = h_cd2, s
    
    # Phase 5: SLSQP polish
    if time.time() < deadline - 15:
        h_pol = _slsqp_smooth(best_h, beta=200, maxiter=150)
        s = _score(h_pol)
        if s < best_score:
            best_h, best_score = h_pol, s
        if time.time() < deadline - 15:
            h_pol2 = _slsqp_smooth(best_h, beta=800, maxiter=150)
            s2 = _score(h_pol2)
            if s2 < best_score:
                best_h, best_score = h_pol2, s2
    
    # Phase 6: Fista refinement
    if time.time() < deadline - 10:
        remaining = deadline - time.time()
        n_iters = min(20000, int(remaining * 3))
        best_h, best_score = _fista(best_h, beta=400, n_iter=n_iters,
                                     alpha_init=0.004, mu=0.88)
    
    # Phase 7: Final coordinate descent
    if time.time() < deadline - 5:
        h_final = _coord_descent(best_h, n_sweeps=300)
        s = _score(h_final)
        if s < best_score:
            best_h, best_score = h_final, s
    
    # Phase 8: More binary search at best n
    if time.time() < deadline - 10:
        h_bin3 = _binary_local_search(best_h, max_passes=100)
        s = _score(h_bin3)
        if s < best_score:
            best_h, best_score = h_bin3, s
            h_cd3 = _coord_descent(h_bin3, n_sweeps=200)
            s2 = _score(h_cd3)
            if s2 < best_score:
                best_h, best_score = h_cd3, s2
    
    # Phase 9: Diverse restarts
    if time.time() < deadline - 15:
        n_rt = len(best_h)
        for attempt in range(3):
            if time.time() > deadline - 5:
                break
            pert = 0.1 * rng.standard_normal(n_rt)
            h_pert = _project(best_h + pert)
            h_cd_r = _coord_descent(h_pert, n_sweeps=150)
            s = _score(h_cd_r)
            if s < best_score:
                best_h, best_score = h_cd_r, s
    
    # Phase 10: Mirror + CD
    if time.time() < deadline - 10:
        h_mir = _mirror(best_h)
        h_cd_m = _coord_descent(h_mir, n_sweeps=150)
        s = _score(h_cd_m)
        if s < best_score:
            best_h, best_score = h_cd_m, s
    
    # Final: one more CD pass
    if time.time() < deadline - 3:
        h_final2 = _coord_descent(best_h, n_sweeps=200)
        s = _score(h_final2)
        if s < best_score:
            best_h, best_score = h_final2, s
    
    best_h = _project(best_h)
    final_score = _score(best_h)
    
    return (best_h, final_score, len(best_h))
