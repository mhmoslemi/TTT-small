def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(h):
    """Clip and fix sum to n/2."""
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if residual > 1e-14:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0:
                break
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-15:
                break
    elif residual < -1e-14:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 0:
                break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

    n = len(h)

    """Convert logistic parameters z (length M=n//2) to full h (length n)."""
    M = n // 2
    u = expit(z)
    h = np.zeros(n)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    return h

def h_to_z(h, n):
    """Extract z from h assuming h[i] + h[N-1-i] = 1."""
    M = n // 2
    u = h[:M]
    u = np.clip(u, 1e-10, 1.0 - 1e-10)
    z = np.log(u / (1.0 - u))
    return z

def overlap_grad_rows(h):
    """Compute gradient of each overlap value w.r.t. h.
    Returns G of shape (2*n-1, n) where G[r, i] = d(C_r)/d(h_i)"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)  # shift
        if s >= 0:
            L = n - s
            # C = dx * sum_{i=0}^{L-1} h[i] * (1 - h[i+s])
            # dC/dh[i] (for i < L) = dx * (1 - h[i+s])
            # dC/dh[i+s] (for i < L) = -dx * h[i]
            G[p, :L] += dx * (1.0 - h[s:s+L])
            G[p, s:s+L] -= dx * h[:L]
        else:
            t = -s
            L = n - t
            # C = dx * sum_{i=0}^{L-1} h[i+t] * (1 - h[i])
            # dC/dh[i+t] = dx * (1 - h[i])
            # dC/dh[i] = -dx * h[i+t]
            G[p, t:t+L] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[t:t+L]
    return G

def soft_obj_and_grad_z(z, n, beta):
    """Log-sum-exp soft max of overlap values, with gradient w.r.t. z."""
    h = z_to_h(z, n)
    vals = compute_overlap_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Stable logsumexp
    max_shifted = shifted.max()
    logsum = cmax + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / beta
    
    # Weights
    w = np.exp(shifted - max_shifted)
    w /= w.sum()
    
    # Gradient w.r.t. h
    G = overlap_grad_rows(h)
    grad_h = w @ G  # shape (n,)
    
    # Chain rule: grad w.r.t. z
    M = n // 2
    u = expit(z)
    du = u * (1.0 - u)
    # h[:M] = u, h[M:] = 1-u[::-1]
    # dh[j]/dz[j] = du[j] for j < M
    # dh[M + (M-1-j)]/dz[j] = -du[j] i.e., h[n-1-j] has derivative -du[j]
    grad_z = np.zeros(M)
    for j in range(M):
        grad_z[j] = du[j] * (grad_h[j] - grad_h[n-1-j])
    
    return logsum, grad_z

    h = np.empty(n)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    vals = compute_overlap_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    w = exp_sh / exp_sh.sum()
    obj = cmax + (max_sh + np.log(exp_sh.sum())) / beta
    grad_h = grad_h_weighted(h, w)
    du = u * (1.0 - u)
    grad_z = du * (grad_h[:M] - grad_h[n - 1:n - 1 - M:-1])
    return obj, grad_z

def find_active_set(vals, frac=0.3, min_count=10):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-6)
    active = np.where(vals >= threshold)[0]
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active = np.unique(np.concatenate([active, top_idx]))
    return sorted(active.tolist())

def active_nlp_sym(z0, n, active_idx, max_iter=400, zbound=12.0):
    M = len(z0)
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(M + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(x):
        z = x[:M]
        t = x[-1]
        h = z_to_h(z, n)
        vals = compute_overlap_values(h)
        return t - vals[active_idx]
    
    def ineq_jac(x):
        z = x[:M]
        h = z_to_h(z, n)
        u = expit(z)
        du = u * (1.0 - u)
        G_act = np.array([grad_h_row(h, r) for r in active_idx])
        J = du * (G_act - G_act[:, ::-1])
        J = J[:, :M]
        return np.hstack([J, np.ones((len(active_idx), 1))])
    
    h0 = z_to_h(z0, n)
    vals0 = compute_overlap_values(h0)
    t0 = vals0[active_idx].max() + 1e-6
    x0 = np.concatenate([z0, [t0]])
    bounds = [(-zbound, zbound)] * M + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(z_to_h(res.x[:M], n))
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def active_nlp_full(h0, active_idx, max_iter=400):
    n = len(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return t - vals[active_idx]
    
    def ineq_jac(x):
        h = x[:n]
        G_act = np.array([grad_h_row(h, r) for r in active_idx])
        return np.hstack([-G_act, np.ones((len(active_idx), 1))])
    
    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    vals0 = compute_overlap_values(h0)
    t0 = vals0.max() + 1e-6
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])

    except Exception:
        pass
    return None, None

def soft_optimize_z(z0, n, beta_list, time_per_beta=25.0):
    z_cur = z0.copy()
    for beta in beta_list:
        t0 = time.time()
        try:
            res = minimize(soft_obj_grad_z, z_cur, args=(n, beta),
                           jac=True, method='L-BFGS-B',
                           bounds=[(-12.0, 12.0)] * len(z_cur),
                           options={'maxiter': 800, 'ftol': 1e-20,
                                    'max_fun': 3000, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_cur = res.x.copy()
        except Exception:
            pass
        if time.time() - t0 > time_per_beta:
            break
    return z_cur

def interpolate_h(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)
