import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_derivatives(h, i):
    """Compute dc[m]/dh[i] for all correlation indices m."""
    n = len(h)
    deriv = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        if s >= 0:
            if i >= s:
                deriv[m] += (1.0 - h[i - s])
            if i + s < n:
                deriv[m] -= h[i + s]
        else:
            t = -s
            if i + t < n:
                deriv[m] += (1.0 - h[i + t])
            if i >= t:
                deriv[m] -= h[i - t]
    return deriv

def solve_1d_convex(A, B):
    """Solve min_{x in [0,1]} max_m (A[m]*x + B[m]) using binary search on subgradient."""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        f_at = A * mid + B
        m_max = np.argmax(f_at)
        g = A[m_max]
        if g > 1e-14:
            hi = mid
        elif g < -1e-14:
            lo = mid
        else:
            break
    x_star = (lo + hi) / 2.0
    return x_star

def coord_optimize(h, i):
    """Find optimal value of coordinate i given other coordinates fixed."""
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    deriv = compute_derivatives(h, i)
    A = dx * deriv
    B = V - A * h[i]
    x_new = solve_1d_convex(A, B)
    return x_new

def coord_descent(h, max_sweeps=500, tol=1e-11):
    """Coordinate descent with make_feasible projection after each sweep."""
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    
    for sweep in range(max_sweeps):
        changed = False
        h_work = h.copy()
        for i in range(n):
            x_new = coord_optimize(h_work, i)
            if abs(x_new - h_work[i]) > 1e-12:
                h_work[i] = x_new
                changed = True
        h_work = make_feasible(h_work)
        sc = official_score(h_work)
        if sc < best_score - 1e-12:
            best_h = h_work.copy()
            best_score = sc
        if not changed and sweep > 3:
            break
        h = h_work
    
    return best_h, best_score

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        for i in range(n):
            if s >= 0:
                if i >= s:
                    gp[i] += (1.0 - h[i - s])
                if i + s < n:
                    gp[i] -= h[i + s]
            else:
                t = -s
                if i + t < n:
                    gp[i] += (1.0 - h[i + t])
                if i >= t:
                    gp[i] -= h[i - t]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def pg_step(h, beta, alpha_max=0.8, max_ls=14):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_shifted = beta * (V - f)
    V_shifted = np.clip(V_shifted, -500.0, 0.0)
    w = np.exp(V_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= w_sum
    
    g = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        deriv = compute_derivatives(h, 0)  # placeholder, compute properly
        # Actually compute gradient directly
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            for i in range(s, n):
                g[i] += wi * (1.0 - h[i - s])
            for i in range(s, n):
                g[i - s] -= wi * h[i]
        else:
            t = -s
            for i in range(t, n):
                g[i - t] += wi * (1.0 - h[i])
            for i in range(t, n):
                g[i] -= wi * h[i - t]
    
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_trig_initial(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n, endpoint=False)
    base = 0.5
    h = base.copy()
    for k in range(1, 6):
        amp = rng.uniform(0.02, 0.15) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * x + phase)
    return make_feasible(h)

def make_piecewise_initial(n, seed=0):
    rng = np.random.default_rng(seed)
    n_blocks = rng.integers(3, 8)
    block_size = n // n_blocks
    h = np.zeros(n)

        h[b * block_size:(b + 1) * block_size] = vals
    return make_feasible(h)
