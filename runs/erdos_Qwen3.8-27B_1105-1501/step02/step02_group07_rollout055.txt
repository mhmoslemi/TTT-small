import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def solve_linearized_lp(h, trust=None):

    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    
    # Build LP
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for k in range(n_constraints):
        s = k - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        
        # Compute coefficient c[m] for each h[m]
        # c[m] = (1-h[m+s]) if m < L (for s >= 0) or (1-h[m+|s|]) if m < L (for s < 0)
        #        + (-h[m-s]) if m >= s (for s >= 0) or (-h[m-|s|]) if m >= |s| (for s < 0)
        #        (with appropriate sign adjustments for negative s)

        # For shift s (which can be negative):
        # If s >= 0: pairs are (i, i+s) for i = 0..L-1
        #   coeff of h[m]: (1-h[m+s]) for m in [0, L-1], (-h[m-s]) for m in [s, n-1]
        # If s < 0: let t = -s, pairs are (i+t, i) for i = 0..L-1, i.e., (j, j-t) for j = t..n-1
        #   V = dx * sum_{i=0}^{L-1} h[i+t] * (1-h[i])
        #   coeff of h[m]: (1-h[m-t]) for m in [t, n-1] (from h[i+t] where m=i+t, i=m-t)
        #                   + (-h[m+t]) for m in [0, L-1] (from (1-h[i]) where m=i, i=m)
        
        if s >= 0:
            for m in range(L):
                A_ub[k, m] += dx * (1.0 - h[m + s])
            for m in range(s, n):
                A_ub[k, m] -= dx * h[m - s]
        else:
            t_ = -s
            for m in range(t_, n):
                A_ub[k, m] += dx * (1.0 - h[m - t_])
            for m in range(L):
                A_ub[k, m] -= dx * h[m + t_]
        
        A_ub[k, n] = -1.0
        
        # Constant: dx * sum h0[i]*h0[i+ss] over the overlap
        if s >= 0:
            B = np.dot(h[0:L], h[s:n])
        else:
            t_ = -s
            B = np.dot(h[t_:n], h[0:L])
        b_ub[k] = -dx * B
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    if trust is not None:
        bounds = []
        for i in range(n):
            lo = max(0.0, h[i] - trust)
            hi = min(1.0, h[i] + trust)
            bounds.append((lo, hi))
        bounds.append((0.0, None))
    else:
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.success and res.x is not None:
            return np.clip(res.x[:n], 0.0, 1.0)
    except Exception:
        pass
    return None

def lp_iteration(h, max_iter=10, trust_init=1.0, verbose=False):

    n = len(h)
    h_cur = h.copy()
    sc_cur = official_score(h_cur)
    trust = trust_init
    
    for it in range(max_iter):
        h_new = solve_linearized_lp(h_cur, trust=trust)
        if h_new is None:
            break
        
        sc_new = official_score(h_new)
        
        if sc_new < sc_cur - 1e-12:
            h_cur = h_new
            sc_cur = sc_new
            trust = min(trust * 1.2, 2.0)

            h_cur = h_new  # accept even if not better (for linearization update)
            trust = max(trust * 0.7, 0.05)
        else:

            h_new = solve_linearized_lp(h_cur, trust=trust)
            if h_new is None:
                break
            sc_new = official_score(h_new)
            if sc_new < sc_cur:
                h_cur = h_new
                sc_cur = sc_new
    
    return h_cur

def pnorm_gradient_step(h, p, alpha):
    """One gradient step on the p-norm smoothed objective."""
    n = len(h)
    V = compute_values(h)
    V = np.maximum(V, 1e-15)  # avoid 0^p issues
    
    # Weights: V[i]^(p-1) / sum(V^p)^(1/p) * p / sum(V^p)^(1/p)
    # Simplified: use V[i]^(p-1) as weights (normalized)
    w = V ** (p - 1)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w /= ws
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(V)):
        ww = w[k]
        if ww < 1e-15:
            continue
        s = k - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[0:L] += ww * (1.0 - h[s:n])
            grad[s:n] -= ww * h[0:L]
        else:
            t_ = -s
            grad[t_:n] += ww * (1.0 - h[0:L])
            grad[0:L] -= ww * h[t_:n]
    
    # Projected gradient step
    h_new = h - alpha * grad
    h_new = make_feasible(h_new)
    return h_new
