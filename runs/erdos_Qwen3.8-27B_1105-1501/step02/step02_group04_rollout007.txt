import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full")
    return float(np.max(c) * dx)


def project_feasible(x):
    n = len(x)
    target = n / 2.0
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            cands = np.where(result < 1.0 - 1e-10)[0]
        else:
            cands = np.where(result > 1e-10)[0]
        if len(cands) > 0:
            per = residual / len(cands)
            result[cands] = np.clip(result[cands] + per, 0.0, 1.0)
            residual2 = target - np.sum(result)
            if abs(residual2) > 1e-14:
                for idx in cands:
                    if residual2 > 0:
                        room = 1.0 - result[idx]
                        if room >= residual2:
                            result[idx] += residual2
                            break
                    else:
                        room = result[idx]
                        if room >= -residual2:
                            result[idx] += residual2
                            break
    return result


def smooth_value(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full")
    M = np.max(c)
    z = (c - M) / T
    zmax = np.max(z)
    if zmax > 500:
        z = z - zmax
    logsum = np.log(np.sum(np.exp(z))) + zmax
    return dx * (M + T * logsum)


def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full")
    M = np.max(c)
    z = (c - M) / T
    zmax = np.max(z)
    if zmax > 500:
        z = z - zmax
    w = np.exp(z)
    wsum = np.sum(w)
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        i_lo = max(0, s)
        i_hi = min(n - 1, s + n - 1)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += wk * (1.0 - h[i_lo - s:i_hi - s + 1])
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= wk * h[i_lo2 + s:i_hi2 + s + 1]
    return grad * dx


def slsqp_obj(x, T):
    return smooth_value(x, T)


def slsqp_grad(x, T):
    return smooth_grad(x, T)


def slsqp_cons(x, T):
    return np.sum(x) - len(x) / 2.0


def slsqp_cons_jac(x, T):
    return np.ones(len(x))


def slsqp_run(h0, T, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac, 'args': (T,)}]
    try:
        res = minimize(slsqp_obj, h0.copy(), args=(T,), jac=slsqp_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x)
    except Exception:
        pass
    return project_feasible(h0.copy())


def adaptive_temps(h, base_scale=None):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full")
    c_sorted = np.sort(c)[::-1]
    top = c_sorted[:min(20, len(c_sorted))]
    spread = top[0] - top[-1]
    if spread < 1e-10:
        spread = 1.0
    if base_scale is None:
        base_scale = max(spread * 0.5, dx * 3.0)
    ratios = [3.0, 1.5, 0.75, 0.4, 0.22, 0.12, 0.065, 0.035, 0.018, 0.010, 0.005]
    temps = [base_scale * r for r in ratios]
    min_t = dx * 0.008
    temps = [max(t, min_t) for t in temps]
    return temps


def make_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_sine(n, rng):
    x = np.arange(n) / n * 2.0 * np.pi
    h = 0.5 + 0.3 * np.sin(x + rng.rand() * 2 * np.pi) + 0.15 * np.sin(2 * x + rng.rand() * 2 * np.pi)
    return project_feasible(h)


def make_golden(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 - frac
    return project_feasible(h)


def make_balanced_wave(n, rng):
    h = 0.5 * np.ones(n)
    for freq in [1, 2, 3]:
        amp = 0.15 / freq
        phase = rng.rand() * 2 * np.pi
        h += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_feasible(h)


def interp_to_length(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_h = np.interp(new_x, old_x, h)
    return project_feasible(new_h)


def pair_sweep(h, n, rng, deadline, max_rounds=2):
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        perm = rng.permutation(n)
        improved = False
        for a in range(n):
            for b in range(a + 1, n):
                if time.time() > deadline:
                    return h
                i, j = perm[a], perm[b]
                if i > j:
                    i, j = j, i
                hi_v = h[i]
                hj_v = h[j]
                lo = max(-hi_v, hj_v - 1.0)
                hi = min(1.0 - hi_v, hj_v)
                if hi - lo < 1e-14:
                    continue
                a_t, b_t = lo, hi
                for _ in range(40):
                    m1 = a_t + (b_t - a_t) / 3.0
                    m2 = b_t - (b_t - a_t) / 3.0
                    h[i] = hi_v - m1
                    h[j] = hj_v + m1
                    f1 = compute_c5(h)
                    h[i] = hi_v - m2
                    h[j] = hj_v + m2
                    f2 = compute_c5(h)
                    h[i] = hi_v
                    h[j] = hj_v
                    if f1 < f2:
                        b_t = m2
                    else:
                        a_t = m1
                opt = (a_t + b_t) / 2.0
                h[i] = hi_v - opt
                h[j] = hj_v + opt
                new_c5 = compute_c5(h)
                if abs(opt) > 1e-12:
                    improved = True
        if not improved:
            break
    return h


def random_pair_polish(h, n, rng, n_pairs=1000, deadline=None):
    for _ in range(n_pairs):
        if deadline is not None and time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi_v = h[i]
        hj_v = h[j]
        lo = max(-hi_v, hj_v - 1.0)
        hi = min(1.0 - hi_v, hj_v)
        if hi - lo < 1e-14:
            continue
        a_t, b_t = lo, hi
        for _ in range(30):
            m1 = a_t + (b_t - a_t) / 3.0
            m2 = b_t - (b_t - a_t) / 3.0
            h[i] = hi_v - m1
            h[j] = hj_v + m1
            f1 = compute_c5(h)
            h[i] = hi_v - m2
            h[j] = hj_v + m2
            f2 = compute_c5(h)
            h[i] = hi_v
            h[j] = hj_v
            if f1 < f2:
                b_t = m2
            else:
                a_t = m1
        opt = (a_t + b_t) / 2.0
        h[i] = hi_v - opt
        h[j] = hj_v + opt
    return h


def slsqp_anneal(h_start, n, rng, deadline, verbose=False):
    cur_h = h_start.copy()
    cur_c5 = compute_c5(cur_h)
    temps = adaptive_temps(cur_h)
    for T in temps:
        if time.time() > deadline:
            break
        maxiter = 200
        prop = slsqp_run(cur_h, T, maxiter)
        prop = project_feasible(prop)
        prop_c5 = compute_c5(prop)
        if prop_c5 < cur_c5 - 1e-10:
            cur_h = prop
            cur_c5 = prop_c5
        elif prop_c5 < cur_c5 + 0.002:
            cur_h = prop
            cur_c5 = prop_c5
    return cur_h, cur_c5
