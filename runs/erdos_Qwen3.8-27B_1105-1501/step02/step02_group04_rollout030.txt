import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)

    hi = min(1.0 - hi_val, hj_val)
    if hi - lo < 1e-14:

        h[i] = hi_val + m2

    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_paley(n, rng):
    # Find a prime <= n
    def is_prime(k):
        if k < 2:
            return False
        if k == 2:
            return True
        if k % 2 == 0:
            return False
        for i in range(3, int(math.sqrt(k)) + 1, 2):
            if k % i == 0:
                return False
        return True
    
    p = n
    while p > 3 and not is_prime(p):
        p -= 1
    if p < 5:
        return make_random_binary(n, rng)
    
    # Build Paley sequence of length p
    sq = np.zeros(p, dtype=bool)
    for a in range(1, p):
        sq[(a * a) % p] = True
    paley = np.zeros(p)
    for k in range(1, p):
        if sq[k]:
            paley[k] = 1.0
    paley[0] = 0.5
    # Sum should be p/2
    # Interpolate to length n
    if p == n:
        return paley
    old_x = np.linspace(0, 2, p)
    new_x = np.linspace(0, 2, n)
    return np.interp(new_x, old_x, paley)

def make_golden(n, rng):

    best_c5 = compute_c5(h)

                current_h = perturbed
                current_c5 = compute_c5(current_h)
                stall = 0
    
    return best_h, best_c5

  Left: s ≥ n-1-i
  Right: s ≤ n-1-i+n-1 = 2n-2-i
  So for the "b=i" role, valid s range is [n-1-i, 2n-2-i].
  And the value at such s is -h[a] where a = i + s - n + 1.

So G_i[s] = (1-h[i-s+n-1]) * [i≤s≤i+n-1] + (-h[i+s-n+1]) * [n-1-i≤s≤2n-2-i]

This can be computed in O(n) for a given i.

Great, so the algorithm for each pair is:
1. Compute L = G_i - G_j in O(n)
2. Set Q = 0 vector, Q[i-j+n-1] += 1, Q[j-i+n-1] += 1
3. Ternary search: for each candidate t, compute new_result = result + t*L + t*t*Q, find max
   Cost per eval: O(n)
   Total ternary: O(n * n_steps)

With n_steps = 50 and n=70: 50 * 70 = 3500 operations per pair.
Number of pairs: 2415.
Total per sweep: 2415 * 3500 ≈ 8.5M operations. In numpy, this should be fast (~1-2 seconds).

This is much faster than calling np.correlate 100 times per pair!

```python
def compute_G_vector(h, i, n):
    """Compute G_i[s] for result index s = 0..2n-2."""
    m = 2 * n - 1
    G = np.zeros(m)
    # Role 1: i appears as 'a' (shifted), b = i - s + n - 1, valid for s in [i, i+n-1]
    s_lo1 = i
    s_hi1 = min(i + n - 1, m - 1)
    if s_lo1 <= s_hi1:
        b_vals = i - np.arange(s_lo1, s_hi1 + 1) + n - 1
        G[s_lo1:s_hi1+1] += (1.0 - h[b_vals])
    # Role 2: i appears as 'b' (unshifted), a = i + s - n + 1, valid for s in [n-1-i, 2n-2-i]
    s_lo2 = n - 1 - i
    s_hi2 = min(2 * n - 2 - i, m - 1)
    if s_lo2 <= s_hi2:
        a_vals = i + np.arange(s_lo2, s_hi2 + 1) - n + 1
        G[s_lo2:s_hi2+1] += -h[a_vals]
    return G
```
