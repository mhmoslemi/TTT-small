```python
def _optimal_coord(h, i, n):
    """Find x that minimizes C5 when h[i] = x, other h fixed."""
    dx_factor = 1.0  # We work with the unnormalized scores
    
    # Compute a_k and b_k for each lag k
    # score_k = sum_m h[m]*(1-h[m+k])
    # The terms involving h[i]:
    #   If 0 <= i <= n-1-k: h[i]*(1-h[i+k]) contributes
    #   If 0 <= i-k <= n-1-k (i.e., k <= i): h[i-k]*(1-h[i]) contributes
    
    a = np.zeros(2*n-1)  # coefficient for x = h[i]
    b = np.zeros(2*n-1)  # constant part
    
    # Current scores
    scores = np.correlate(h, 1.0-h, mode='full')
    
    for r in range(2*n-1):
        k = r - (n-1)
        # Current score at lag k
        b[r] = scores[r]
        a[r] = 0.0
        # If i is in range [0, n-1-|k|] as the "first" element:
        if k >= 0 and 0 <= i <= n-1-k:
            # h[i] appears as h[m] with m=i, term: h[i]*(1-h[i+k])
            b[r] -= h[i] * (1-h[i+k])  # remove current contribution
            a[r] += (1-h[i+k])  # coefficient of x
        if k >= 0 and 0 <= i-k <= n-1-k:  # k <= i
            # h[i] appears as h[m+k] with m=i-k, term: h[i-k]*(1-h[i])
            b[r] -= h[i-k] * (1-h[i])  # remove current contribution
            a[r] += -h[i-k]  # coefficient of x (since term is h[i-k]*(1-x))
        elif k < 0:
            k2 = -k  # symmetric case
            if 0 <= i <= n-1-k2:
                b[r] -= h[i] * (1-h[i+k2])  # h[i] as "first" element, h[i+k2] as "second"
                a[r] += (1-h[i+k2])
            if 0 <= i-k2 <= n-1-k2:
                b[r] -= h[i-k2] * (1-h[i])
                a[r] += -h[i-k2]
    
    # Now f(r) = a[r]*x + b[r], minimize max_r f(r) over x in [0,1]
    # Find breakpoints
    breakpoints = [0.0, 1.0]
    for r1 in range(2*n-1):
        for r2 in range(r1+1, 2*n-1):
            da = a[r1] - a[r2]
            if abs(da) > 1e-12:
                xp = (b[r2] - b[r1]) / da
                if 0 < xp < 1:
                    breakpoints.append(xp)
    
    best_x = 0.0
    best_val = np.inf
    for x in breakpoints:
        val = np.max(a * x + b)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x, best_val * (2.0/n)  # convert to C5 scale
```

```python
    # For positive lags k = 0 to n-1 (indices r = n-1 to 2n-2)
    ks_pos = np.arange(n)  # k = 0, 1, ..., n-1
    # "h" position: 0 <= i <= n-1-k → k <= n-1-i
    mask_h = ks_pos <= (n - 1 - i)
    # "(1-h)" position: k <= i
    mask_1h = ks_pos <= i
    
    # "h" position contribution
    r_h = (n - 1) + ks_pos[mask_h]
    p_h = i + ks_pos[mask_h]  # partner index
    b[r_h] -= h[i] * (1.0 - h[p_h])
    a[r_h] += (1.0 - h[p_h])
    
    # "(1-h)" position contribution
    r_1h = (n - 1) + ks_pos[mask_1h]
    p_1h = i - ks_pos[mask_1h]  # partner index
    b[r_1h] -= h[p_1h] * (1.0 - h[i])
    a[r_1h] += -h[p_1h]
    
    # For negative lags k = -1 to -(n-1) (indices r = n-2 down to 0)
    # Let k2 = -k > 0, so k2 = 1, ..., n-1
    ks_neg = np.arange(1, n)  # k2 = 1, ..., n-1
    r_neg = n - 1 - ks_neg  # indices 0 to n-2
    # "h" position: k2 <= i <= n-1 → i >= k2
    mask_h_neg = ks_neg <= i
    # "(1-h)" position: 0 <= i <= n-1-k2 → k2 <= n-1-i
    mask_1h_neg = ks_neg <= (n - 1 - i)
    
    r_h_neg = r_neg[mask_h_neg]
    p_h_neg = i - ks_neg[mask_h_neg]
    b[r_h_neg] -= h[i] * (1.0 - h[p_h_neg])
    a[r_h_neg] += (1.0 - h[p_h_neg])
    
    r_1h_neg = r_neg[mask_1h_neg]
    p_1h_neg = i + ks_neg[mask_1h_neg]
    b[r_1h_neg] -= h[p_1h_neg] * (1.0 - h[i])
    a[r_1h_neg] += -h[p_1h_neg]
    
    # Ternary search for min of max(a*x + b)
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if float(np.max(a * m1 + b)) < float(np.max(a * m2 + b)):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
```

This should be much faster since the main work is vectorized. The ternary search still does 80 iterations of O(n) numpy operations, but numpy operations on arrays of length 127 are very fast (~1-2μs each).

Total per coordinate: ~10μs (for the setup) + 80 * 2 * 1μs (ternary) ≈ 170μs. Per sweep: 64 * 170μs ≈ 11ms. Per second: ~90 sweeps. Very good.
