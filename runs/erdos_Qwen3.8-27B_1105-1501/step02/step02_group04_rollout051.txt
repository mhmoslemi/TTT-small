    if hi - lo < 1e-14:

        h[j] -= t

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h[~np.isfinite(h)] = 0.5
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(proj)
    if abs(residual) > 1e-13:
        if residual > 0:
            free = np.where(proj < 1.0 - 1e-12)[0]
        else:
            free = np.where(proj > 1e-12)[0]
        if len(free) > 0:
            per = residual / len(free)
            proj[free] = np.clip(proj[free] + per, 0.0, 1.0)
    return proj


def soft_value_grad(x, n, T):
    dx = 2.0 / n
    v = 1.0 - x
    corr = np.correlate(x, v, mode="full")
    M = np.max(corr)
    z = (corr - M) / T
    z = np.clip(z, -60.0, 60.0)
    expz = np.exp(z)
    S = np.sum(expz)
    w = expz / S
    val = (M + T * np.log(S)) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        t = n - 1 - k
        j_lo = max(0, -t)
        j_hi = min(n - 1, n - 1 - t)
        if j_lo <= j_hi:
            idx = np.arange(j_lo + t, j_hi + t + 1)
            grad[j_lo:j_hi + 1] += wk * v[idx]
        j_lo2 = max(0, t)
        j_hi2 = min(n - 1, n - 1 + t)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2 - t, j_hi2 - t + 1)
            grad[j_lo2:j_hi2 + 1] -= wk * x[idx2]
    grad *= dx
    return val, grad


def _slsqp_fun(x, n, T):
    val, _ = soft_value_grad(x, n, T)
    return val


def _slsqp_jac(x, n, T):
    _, g = soft_value_grad(x, n, T)
    return g


def _eq_fun(x, n, T):
    return np.sum(x) - n / 2.0


def _eq_jac(x, n, T):
    return np.ones(n)


def run_slsqp_smooth(h0, n, T, maxiter=120):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_slsqp_fun, h0, args=(n, T), jac=_slsqp_jac,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        x = res.x
        x = np.clip(x, 0.0, 1.0)
        x = project_feasible(x)
        return x
    except Exception:
        return h0.copy()


def pair_line_search(h, i, j):
    hi_val = min(1.0 - h[i], h[j])
    lo_val = max(-h[i], h[j] - 1.0)
    if hi_val - lo_val < 1e-14:
        return 0.0
    a, b = lo_val, hi_val
    best_t = 0.0
    best_f = compute_c5(h)
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] += m1
        h[j] -= m1
        f1 = compute_c5(h)
        if f1 < best_f:
            best_f = f1
            best_t = m1
        h[i] += m2 - m1
        h[j] -= m2 - m1
        f2 = compute_c5(h)
        if f2 < best_f:
            best_f = f2
            best_t = m2
        h[i] -= m2
        h[j] += m2
        if f1 < f2:
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    h[i] += t_opt
    h[j] -= t_opt
    f_opt = compute_c5(h)
    if f_opt < best_f:
        best_f = f_opt
        best_t = t_opt
    else:
        h[i] -= t_opt
        h[j] += t_opt
    h[i] += best_t - (h[i] - h[i])
    h[j] -= best_t - (h[j] - h[j])
    return best_t

def make_binary_random(n, rng):

def make_wave(n, freq, phase, rng):
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / n + phase) + rng.randn(n) * 0.05
    return project_feasible(h)


def transfer_resolution(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    positions_old = (np.arange(n) + 0.5) * (2.0 / n)
    positions_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(positions_new, positions_old, h, left=h[0], right=h[-1])
    return project_feasible(new_h)


def basin_hitch(h, rng, scale):
    return project_feasible(h + rng.randn(len(h)) * scale)

        h_fin = best_h.copy()
        gradient_guided_pair_pass(h_fin, rng, len(h_fin), deadline=hard_end)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-10:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)
    n_points = len(best_h)

    return best_h, final_c5, n_points
