import numpy as np

import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(res) < 1e-14:
                break
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad_vec(x, beta):

    n = len(x)

    cv = _corr_vec(x)
    mc = cv.max()

    if es < 1e-300:

        w = ev / es
    g = np.zeros(n)

            g[m:] += -wk * dx * x[:n - m]
        else:
            g += wk * dx * (1.0 - 2.0 * x)
    return g

def _fista(h0, beta, n_iter, alpha_init=0.01, mu=0.9):
    """FISTA with vectorized gradient and averaging."""
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1.0
    for it in range(n_iter):
        grad = _lse_grad_vec(y, beta)
        h_new = y - alpha_init * grad
        h_new = _project(h_new)
        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h_cur)
        y = _project(y)
        h_avg += (h_new - h_avg) / (t_new)
        t = t_new
        h_prev = h_cur
        h_cur = h_new
        if it % 200 == 0 or it == n_iter - 1:

            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    return best_h, best_s

def _slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def _efun(x):
        return x.sum() - n / 2.0
    def _ejac(x):
        return np.ones(n)
    try:

                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:

    return h0

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv
    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP', bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}

                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):

    except Exception:

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _make_binary(n, rng, pattern=0):
    """Create a random binary vector with exactly n/2 ones (n must be even)."""
    if n % 2 != 0:
        return _project(np.ones(n) * 0.5)
    h = np.zeros(n)
    if pattern == 0:
        # Random
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    elif pattern == 1:
        # Alternating
        h[::2] = 1.0
    elif pattern == 2:
        # Block
        h[:n // 2] = 1.0
    elif pattern == 3:
        # Sparse
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        # Sort and reverse for different pattern
        h = h[::-1]
    else:
        # Golden ratio
        positions = np.floor((np.arange(n) * (np.sqrt(5) - 1) / 2)) % n
        # Take first n/2 unique
        seen = set()
        count = 0
        for i in range(n * 2):
            p = int(np.floor((i * (np.sqrt(5) - 1) / 2))) % n
            if p not in seen:
                seen.add(p)
                h[p] = 1.0
                count += 1
                if count == n // 2:
                    break
    return h

def _anneal_binary(h0, n_iter, rng, t_start=0.3, t_end=0.01):
    """Simulated annealing on binary vectors preserving exact count."""
    n = len(h0)
    if n % 2 != 0:
        return h0, _score(h0)
    cur = h0.copy().astype(int)
    cur_s = _score(cur.astype(float))
    best = cur.copy()
    best_s = cur_s
    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        # Pick random swap
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_s = _score(cur.astype(float))
        delta = new_s - cur_s
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-8)):
            cur_s = new_s
            if new_s < best_s:
                best_s = new_s
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best.astype(float), best_s

def _smoothing(h, kernel_size=5):
    """Smooth a binary vector with a small kernel and project."""
    n = len(h)
    k = min(kernel_size, n)
    if k % 2 == 0:
        k += 1
    half = k // 2
    kernel = np.ones(k) / k
    # Circular convolution
    padded = np.concatenate([h[-half:], h, h[:half]])
    smoothed = np.convolve(padded, kernel, mode='valid')[:n]
    return _project(smoothed)
