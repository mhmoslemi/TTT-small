import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_raw(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_raw(h, s):
    """Gradient of raw(s) = sum_i h_i*(1-h_{i+s}) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        j = i + s
        if 0 <= j < n:
            g[i] += 1.0 - h[j]  # from d/dh_i of h_i*(1-h_{i+s})
            g[j] -= h[i]         # from d/dh_{i+s} of h_i*(1-h_{i+s})
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    # Find shift that makes sum(clip(z+shift, 0, 1)) = target
    lo, hi = -3.0, 3.0

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:

    return h

def subgradient_step(h, k_active, step_mult=1.0):
    """Take a projected subgradient descent step on the max of raw values"""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    # Find top k_active shifts

def qp_local(h0, k_active, reg, max_iter=30):
    """Active-set QP local optimization"""
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    
    for it in range(max_iter):
        raw = compute_raw(h)
        top_idx = np.argsort(raw)[-k_active:]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        for p in top_idx:
            s = p - (n - 1)
            g = grad_raw(h, s)
            constraints.append(raw[p] + g @ (h_var - h) <= t_var)
        
        obj = cp.Minimize(t_var + reg * cp.sum_squares(h_var - h))
        prob = cp.Problem(obj, constraints)
        
        h_new = None
        for solver in [cp.CLARABEL, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    if h_var.value is not None and np.all(np.isfinite(h_var.value)):

                    break
            except:
                continue

            break

            best = sc

                    h = h_try
                    best = sc_try
                    improved = True
