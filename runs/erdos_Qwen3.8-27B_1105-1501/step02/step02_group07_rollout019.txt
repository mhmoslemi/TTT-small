def softmax_weights(V, beta):
    V_max = V.max()
    shifted = np.clip(beta * (V - V_max), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws

def pg_step(h, beta, alpha_max=0.8, max_ls=15):
    n = len(h)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    f = V.max()
    w = softmax_weights(V, beta)
    grad = np.zeros(n)
    for idx in range(len(V)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f

            h_new, sc = active_set_lp(h_cur, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                    h_cur = h_new
                    trust = min(trust * 1.15, 0.35)
                else:
                    h_cur = h_new
                    trust = max(trust * 0.85, 0.03)

    # Phase 4: Coordinate descent after LP
    if time_left() > 100:
        h_cd = coord_sweep_max(best_h, n_sweeps=300, rng=rng)
        sc = official_score(h_cd)
        if sc < best_score - 1e-12:
            best_h = h_cd.copy()
            best_score = sc
        # Another LP round
        h_cur = best_h.copy()
        for _ in range(30):
            if time_left() < 60:
                break
            h_new, sc = active_set_lp(h_cur, eps=0.001, trust=0.08)
            if h_new is not None and sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
                h_cur = h_new

    # Phase 5: Resolution sweep
    if time_left() > 80:
        n_base = len(best_h)
        n_candidates = set()
        for offset in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 16, 20]:
            nt = n_base + offset
            if nt >= 20 and nt <= 200:
                n_candidates.add(nt)
        for nt in sorted(n_candidates):
            if time_left() < 40:
                break
            h_int = interpolate_to(best_h, nt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int.copy()
                best_score = sc_int
            # Optimize at this resolution
            h_opt = coord_sweep_max(h_int, n_sweeps=150, rng=rng)
            sc_opt = official_score(h_opt)
            if sc_opt < best_score - 1e-12:
                best_h = h_opt.copy()
                best_score = sc_opt
            # Quick LP
            h_cur2 = best_h.copy()
            for _ in range(15):
                if time_left() < 30:
                    break
                h_new, sc = active_set_lp(h_cur2, eps=0.002, trust=0.1)
                if h_new is not None and sc < best_score - 1e-13:
                    best_h = h_new.copy()
                    best_score = sc
                    h_cur2 = h_new

    # Phase 6: Final fine coordinate descent + LP
    if time_left() > 30:
        h_final = coord_sweep_max(best_h, n_sweeps=200, rng=rng)
        sc = official_score(h_final)
        if sc < best_score - 1e-13:
            best_h = h_final
            best_score = sc
        h_cur = best_h.copy()
        for _ in range(20):
            if time_left() < 10:
                break
            h_new, sc = active_set_lp(h_cur, eps=0.0005, trust=0.05)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
