def penalty_lse_obj(x, beta, dx, lam, target):
    corr = np.correlate(x, 1.0-x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    s = np.sum(np.exp(shifted))
    if s > 1e-300:
        obj = m + np.log(s) / beta
    else:
        obj = m
    # Penalty for sum constraint
    penalty = lam * (np.sum(x) - target)**2
    return obj + penalty

def penalty_lse_grad(x, beta, dx, lam, target):
    n = len(x)
    corr = np.correlate(x, 1.0-x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)

def fast_correlate(h, one_minus_h):
    """Use FFT for fast full correlation."""
    n = len(h)
    N = 2*n - 1
    # Pad to next power of 2 for FFT efficiency
    Npad = 1
    while Npad < 2*n:
        Npad *= 2
    H = np.fft.rfft(h, Npad)
    V = np.fft.rfft(one_minus_h, Npad)
    corr = np.fft.irfft(H * V, Npad)[:2*n-1]
    return corr

def lse_and_weights(corr, beta):
    """Compute LSE value and softmax weights."""
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        return m, w
    w = e / s
    return m + np.log(s) / beta, w

def compute_grad_at_lag(h, lag_idx):
    """Compute gradient of correlation value at a specific lag index."""
    n = len(h)
    s = lag_idx - (n-1)
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
    else:
        m = -s
        if m == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] += (-h[:n-m])
    return grad

def lse_obj_grad(x, beta, dx, lam=0.0, target=None):
    """Objective and gradient for LSE + penalty."""
    n = len(x)
    corr = np.correlate(x, 1.0-x, mode='full') * dx
    val, w = lse_and_weights(corr, beta)
    
    # Gradient: weighted sum of per-lag gradients
    grad = np.zeros(n)
    # Only compute for lags with significant weight
    significant = np.where(w > 1e-6)[0]
    for k in significant:
        w_k = w[k] * dx
        if w_k < 1e-10:
            continue
        grad += w_k * compute_grad_at_lag(x, k)
    
    # Penalty gradient
    if lam > 0 and target is not None:
        val += lam * (np.sum(x) - target)**2
        grad += 2.0 * lam * (np.sum(x) - target)
    
    return val, grad

    h = make_feasible(h0.copy())
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_score = official_score(h)
    
    bounds = [(0.0, 1.0)] * n
    
    for cont in range(n_continuation):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        frac = cont / max(n_continuation - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lam = lam_start + (lam_end - lam_start) * frac**2  # quadratic increase
        
        def obj(x):
            v, _ = lse_obj_grad(x, beta, dx, lam, target)
            return v
        
        def grad(x):
            _, g = lse_obj_grad(x, beta, dx, lam, target)
            return g
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16})
            h = make_feasible(res.x)
        except Exception:
            h = make_feasible(h)
        
        s = official_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    
    return best_h, best_score

def make_feasible(z, target=None):

        else:
            hi = mid

    n_evals = 0
    max_evals = max(10000, int(200 * n))

    for phase in range(6):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        for _ in range(n * 3):
            if n_evals > max_evals:
                break
            j = rng.integers(n)
            for d in [step, -step]:
                h2 = h.copy()
                h2[j] += d
                if h2[j] < 0 or h2[j] > 1:
                    continue
                h2 = make_feasible(h2)
                s = official_score(h2)
                n_evals += 1
                if s < best_score - 1e-12:
                    best_score = s
                    h = h2.copy()
                    improved = True
                    break
        if not improved:
            step *= 0.5
        if step < 0.001:
            break
    return make_feasible(h), best_score

def binary_round(h, rng):
    n = len(h)
    target = n / 2.0
    if n % 2 == 0:
        n_ones = n // 2
        h_bin = np.zeros(n)
        order = np.argsort(-h)
        h_bin[order[:n_ones]] = 1.0
        return h_bin
    else:
        n_ones = n // 2
        h_bin = np.zeros(n)
        order = np.argsort(-h)
        h_bin[order[:n_ones]] = 1.0
        h_bin[order[n_ones]] = 0.5
        return h_bin

def fourier_start(n, rng):
    x = np.arange(n)
    h = np.full(n, 0.5)
    n_harm = max(1, n // 4)
    for k in range(1, n_harm + 1):
        amp = 0.3 / (1.0 + 0.3 * k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / n + phase)
    return make_feasible(h)

def golden_start(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    if n % 2 == 0:
        n_ones = n // 2
        for i in range(n_ones):
            pos = int((i * phi) % n)
            h[pos] = 1.0
    else:
        n_ones = n // 2
        for i in range(n_ones):
            pos = int((i * phi) % n)
            h[pos] = 1.0
        h[int(n_ones * phi % n)] += 0.5
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = 25.0
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    n_list = [len(parent)]
    for nn in [48, 50, 52, 54, 56, 58, 60, 64, 68, 72, 80, 88, 96, 108, 120, 128, 144, 160]:
        if nn not in n_list:
            n_list.append(nn)
    n_list = sorted(set(nn for nn in n_list if 30 <= nn <= 180))

    def gen_starts(n, parent_arr):
        starts = []
        # Interpolated parent
        h_ip = make_feasible(interp_to_n(parent_arr, n))
        starts.append(h_ip)
        # Binary push
        for p in [2.0, 5.0]:
            starts.append(make_feasible(h_ip ** p))
        starts.append(binary_round(h_ip, rng))
        # Golden ratio
        starts.append(golden_start(n, rng))
        # Fourier
        starts.append(fourier_start(n, rng))
        # Random binary
        if n % 2 == 0:
            rb = np.zeros(n)
            rb[rng.choice(n, n // 2, replace=False)] = 1.0
            starts.append(rb)
        else:
            rb = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            rb[idx] = 1.0
            rb[rng.integers(n)] = 0.5
            starts.append(make_feasible(rb))
        # Random smooth
        starts.append(make_feasible(0.5 + 0.3 * np.sin(np.pi * np.arange(n) / n * rng.uniform(1, 4)) + 0.1 * rng.standard_normal(n)))
        return starts

    # Process resolutions in order of promise (closer to parent n first, then larger)
    n_sorted = sorted(n_list, key=lambda x: (abs(x - len(parent)), x))

    for n in n_sorted:
        if time.time() > deadline - 45:
            break
        remaining = deadline - time.time()
        n_starts = gen_starts(n, parent)
        # Limit starts based on remaining time
        max_starts = min(len(n_starts), max(2, int(remaining / 25)))

        for si in range(max_starts):
            if time.time() > deadline - 20:
                break
            h0 = n_starts[si].copy()
            # Time budget for this start
            ot = min(40.0, max(8.0, (deadline - time.time()) / max(1, max_starts - si)))

            h_opt, sc = optimize_cont_lse(h0, n_cont=20, maxiter=300,
                                          time_limit=ot, t_start=time.time())
            if official_score(h_opt) < best_score - 1e-12:
                best_score = official_score(h_opt)
                best_h = h_opt.copy()
                best_n = n

            # Local refinement
            if time.time() < deadline - 15:
                h_lr, sc_lr = local_refine(h_opt, rng, time_limit=min(8.0, deadline - time.time() - 10), t_start=time.time())
                if sc_lr < best_score - 1e-12:
                    best_score = sc_lr
                    best_h = h_lr.copy()
                    best_n = n

            # Binary round + re-optimize
            if time.time() < deadline - 20:
                h_bin = binary_round(h_opt, rng)
                h_bo, sc_bo = optimize_cont_lse(h_bin, n_cont=15, maxiter=200,
                                                time_limit=min(15.0, deadline - time.time() - 10), t_start=time.time())
                if official_score(h_bo) < best_score - 1e-12:
                    best_score = official_score(h_bo)
                    best_h = h_bo.copy()
                    best_n = n

    # Final polish on best solution
    remaining = deadline - time.time()
    if remaining > 30:
        h_p, sc_p = optimize_cont_lse(best_h, n_cont=25, maxiter=500,
                                      time_limit=min(remaining - 10, 60), t_start=time.time())
        if official_score(h_p) < best_score - 1e-12:
            best_score = official_score(h_p)
            best_h = h_p.copy()

    if time.time() < deadline - 10:
        h_lr, sc_lr = local_refine(best_h, rng, time_limit=min(remaining * 0.3, 20), t_start=time.time())
        if sc_lr < best_score - 1e-12:
            best_score = sc_lr
            best_h = h_lr.copy()

    # SLSQP direct minmax polish
    if time.time() < deadline - 8:
        n = len(best_h)
        dx = 2.0 / n
        t0_val = official_score(best_h) + 1e-6
        if t0_val > 0.99:
            t0_val = 0.99
        z0 = np.concatenate([best_h, [t0_val]])

        def mm_obj(z):
            return z[-1]

        def mm_obj_jac(z):
            g = np.zeros(len(z))
            g[-1] = 1.0
            return g

        def mm_ineq_fun(z):
            h = z[:-1]
            t = z[-1]
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            return t - corr

        def mm_ineq_jac(z):
            h = z[:-1]
            nn = len(h)
            jac = np.zeros((2*nn-1, nn+1))
            for k in range(2*nn-1):
                jac[k, -1] = 1.0
                g2 = grad_at_lag(h, k)
                jac[k, :nn] = -dx * g2
            return jac

        def mm_eq(z):
            return np.sum(z[:-1]) - len(z) / 2.0 + z[-1] * 0  # sum of h = n/2
            # Actually: sum(z[:-1]) = n/2

        def mm_eq(z):
            return np.sum(z[:-1]) - len(z[:-1]) / 2.0

        def mm_eq_jac(z):
            g = np.zeros(len(z))
            g[:-1] = 1.0
            return g

        bounds_mm = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        try:
            res = minimize(mm_obj, z0, jac=mm_obj_jac, method='SLSQP',
                           bounds=bounds_mm,
                           constraints=[
                               {'type': 'ineq', 'fun': mm_ineq_fun, 'jac': mm_ineq_jac},
                               {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac}
                           ],
                           options={'maxiter': 80, 'ftol': 1e-14})
            if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                h_mm = make_feasible(res.x[:-1])
                sc_mm = official_score(h_mm)
                if sc_mm < best_score - 1e-12:
                    best_score = sc_mm
                    best_h = h_mm.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
