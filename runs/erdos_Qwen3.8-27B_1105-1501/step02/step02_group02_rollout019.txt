    cv = _corr_vec(h)
    cv = np.maximum(cv - offset, 0.0)
    if p == np.inf:
        return cv.max()
    return np.sum(cv ** p) ** (1.0 / p)

    n = len(h)
    cv = _corr_vec(h)
    cv_pos = np.maximum(cv - offset, 0.0)
    if p == np.inf:
        # Just return gradient of max (subgradient)
        idx = np.argmax(cv)
        return _single_lag_grad(h, idx)
    cv_pos_p = cv_pos ** p
    total = cv_pos_p.sum()
    if total < 1e-300:
        return np.zeros(n)

def _block_cd(h, block_size=6, n_sweeps=50, deadline=None):
    n = len(h)
    cur = h.copy()
    cur_s = _score(cur)
    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            bsz = end - start
            # Optimize h[start:end] while fixing the rest
            # Objective: max_k F_k, which is a max of quadratic functions in the block variables
            # Use SLSQP on just these bsz variables
            block = cur[start:end].copy()
            
            def obj(x, start=start, end=end):
                h_temp = cur.copy()
                h_temp[start:end] = x
                return _pnorm_obj(h_temp, p=16)  # smooth surrogate
            
            def obj_grad(x, start=start, end=end):
                h_temp = cur.copy()
                h_temp[start:end] = x
                g_full = _pnorm_grad(h_temp, p=16)
                return g_full[start:end]
            
            # Constraint: sum of all h = n/2
            # sum of block + sum of rest = n/2
            rest_sum = cur.sum() - cur[start:end].sum()
            
            def eq_con(x):
                return x.sum() + rest_sum - n / 2.0
            
            def eq_jac(x):
                return np.ones(bsz)
            
            bounds = [(0.0, 1.0)] * bsz
            try:
                res = minimize(obj, block, jac=obj_grad, method='SLSQP',
                              bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
                              options={'maxiter': 30, 'ftol': 1e-10})
                if res.success and np.all(np.isfinite(res.x)):

                    if s_new < cur_s - 1e-12:
                        cur = cur_test
                        cur_s = s_new

            except Exception:
