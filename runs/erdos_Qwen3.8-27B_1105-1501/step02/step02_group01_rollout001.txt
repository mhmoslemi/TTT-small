def pair_local_search(h, score, rng, max_rounds=50, time_limit=None, t_start=None):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')  # raw correlation
    best_h = h.copy()
    best_score = score
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        
        # Compute gradient-like weights for active lags
        max_val = np.max(r)
        threshold = max_val - 0.05 * max_val
        active = np.where(r >= threshold)[0]
        
        # For each active lag, compute per-coordinate derivative
        # Using the formula: d(r_k)/d(h_m) for the active lags
        grad = np.zeros(n)
        for k in active:
            s = k - (n-1)
            if s >= 0:
                # r_k = sum_{m=0}^{n-1-s} h[m]*(1-h[m+s])
                # d/dh_m: (1-h[m+s]) + h[m]*(0) for m+s in range... 
                # Actually: d/dh[m] of h[m]*(1-h[m+s]) = (1-h[m+s])
                # d/dh[m+s] of h[m]*(1-h[m+s]) = -h[m]
                # So grad[m] += (1-h[m+s]) for m in [0, n-1-s]
                #    grad[m+s] += (-h[m]) for m in [0, n-1-s]
                # Combined: grad[0:n-s] += (1-h[s:n])
                #           grad[s:n] += (-h[0:n-s])
                grad[:n-s] += (1.0 - h[s:])
                grad[s:] += (-h[:n-s])
            else:
                m = -s
                # r_k = sum_{m=m}^{n-1} h[m]*(1-h[m-m])... 
                # r_k = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
                # d/dh[i] = (1-h[i-m]) + h[i-m]*(0)... wait
                # d/dh[i] of h[i]*(1-h[i-m]) = (1-h[i-m])
                # d/dh[i-m] of h[i]*(1-h[i-m]) = -h[i]
                # So for i in [m, n-1]: grad[i] += (1-h[i-m])
                #                           grad[i-m] += (-h[i])
                # Combined: grad[m:n] += (1-h[0:n-m])
                #           grad[0:n-m] += (-h[m:n])
                grad[m:] += (1.0 - h[:n-m])
                grad[:n-m] += (-h[m:])
        
        # Normalize gradient
        grad_norm = np.abs(grad)
        if np.max(grad_norm) > 0:
            grad /= np.max(grad_norm)
        
        # Select candidate pairs: (donor, receiver) where grad[donor] > 0 and grad[recv] < 0
        # (we want to decrease h at donor and increase at receiver)
        donors = np.where(grad > 0.1)[0]
        receivers = np.where(grad < -0.1)[0]
        
        if len(donors) == 0 or len(receivers) == 0:
            # Fallback: try random pairs
            donors = np.where(h > 0.3)[0]
            receivers = np.where(h < 0.7)[0]
        
        if len(donors) == 0 or len(receivers) == 0:
            no_improve += 1
            if no_improve > 10:
                break
            # Try a random perturbation
            h = make_feasible(h + 0.05 * rng.standard_normal(n))
            r = np.correlate(h, 1.0 - h, mode='full')
            score = official_score(h)
            if score < best_score:
                best_h = h.copy()
                best_score = score
                no_improve = 0
            continue
        
        # Try top pairs
        improved = False
        n_pairs_to_try = min(len(donors) * len(receivers), 30)
        
        # Score pairs by expected improvement
        pair_scores = []
        for d in donors[:10]:
            for rc in receivers[:10]:
                if d == rc:
                    continue
                # Expected improvement: -grad[d] - grad[rc] (we want to reduce the max)
                ps = -(grad[d] + grad[rc])
                pair_scores.append((ps, d, rc))
        
        pair_scores.sort(reverse=True)
        
        for ps, d, rc in pair_scores[:n_pairs_to_try]:
            # Try moving from d to rc (h[d] decreases, h[rc] increases)
            t_max = min(h[d], 1.0 - h[rc])
            if t_max < 1e-10:
                continue
            
            h_new, t_opt = optimize_pair_move(h, r, d, rc, dx)
            s_new = official_score(h_new)
            
            if s_new < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = s_new
                h = h_new.copy()
                r = np.correlate(h, 1.0 - h, mode='full')
                score = s_new
                improved = True
                # Recompute gradient
                break  # restart with new h
        
        if not improved:
            # Try some random pairs
            for _ in range(5):
                d = h > 0.3
                rc = h < 0.7
                if np.sum(d) == 0 or np.sum(rc) == 0:
                    break
                d_idx = rng.choice(np.where(d)[0])
                rc_idx = rng.choice(np.where(rc)[0])
                if d_idx == rc_idx:
                    continue
                t_max = min(h[d_idx], 1.0 - h[rc_idx])
                if t_max < 1e-10:

                h_new, t_opt = optimize_pair_move(h, r, d_idx, rc_idx, dx)

                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    
    return best_h, best_score
