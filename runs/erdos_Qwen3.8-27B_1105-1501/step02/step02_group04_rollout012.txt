    n = len(h)
    dx = 2.0 / n

    lo, hi = -2.0, 2.0

            hi = mid

    n = len(h)
    K = len(active_ells)

import time
from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_gradients(h, ells, offset):
    n = len(h)
    K = len(ells)
    G = np.zeros((K, n))
    p = np.arange(n)
    for r in range(K):
        lag = ells[r] - offset
        m1 = (p - lag >= 0) & (p - lag < n)
        G[r, m1] += (1.0 - h[p[m1] - lag])
        m2 = (p + lag >= 0) & (p + lag < n)
        G[r, m2] -= h[p[m2] + lag]
    return G

def solve_lp(h, active_ells, offset, delta_max):
    n = len(h)
    K = len(active_ells)
    G = build_gradients(h, active_ells, offset)
    
    # Variables: [d0, ..., d_{n-1}, theta], minimize -theta
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    
    # G @ d + theta <= 0 for each active lag
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K)
    
    lb = np.maximum(-h, -delta_max)
    ub = np.minimum(1.0 - h, delta_max)
    
    bounds = list(zip(lb, ub)) + [(0.0, 0.01)]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    try:
        res = linprog(c_vec, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, 0.0

def softmax_grad(h, temp):
    n = len(h)
    offset = n - 1
    corr = compute_corr(h)
    shifted = corr - np.max(corr)
    w = np.exp(shifted / temp)
    w /= w.sum()
    
    G = build_gradients(h, np.arange(2 * n - 1), offset)
    grad = w @ G
    return grad, corr
