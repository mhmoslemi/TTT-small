import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Residual repair
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def grad_lag_s(h, s):
    """Gradient of lag-s correlation value (without dx factor)."""
    n = len(h)
    grad = np.zeros(n)
    if s == 0:
        grad = 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        # df/dh[j] = (1-h[j-s]) for j in [s, n-1] minus h[j+s] for j in [0, n-s-1]
        grad[s:] += 1.0 - h[:m]  # from h[i+s]*(1-h[i]), derivative w.r.t. h[i+s]
        grad[:m] -= h[s:]         # from h[i+s]*(1-h[i]), derivative w.r.t. h[i]
    return grad

def active_values(h):
    """Compute all lag values (for lags 0 to n-1) with dx factor."""
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full')
    # Index n-1 is lag 0, indices n to 2n-2 are lags 1 to n-1
    # Due to symmetry, lags s and -s give same values
    # full[n-1+s] = lag s for s >= 0
    vals = full[n-1:n-1+n]  # lags 0 to n-1
    return vals * dx

def lp_direction(h, active_s, rho, C_val, V_active):
    """
    Solve LP to find direction u minimizing linearized max over active lags.
    
    h: current point
    active_s: list of lag indices (s >= 0)
    rho: trust radius
    C_val: current max value
    V_active: values at active lags (with dx)
    
    Returns direction u or None.
    """
    n = len(h)
    dx = 2.0 / n
    k = len(active_s)
    
    # Build constraint matrix
    # Variables: [u_0, ..., u_{n-1}, t]
    # Constraints: g_s^T u - t <= -V_s for each active s
    #              sum(u) = 0
    # Bounds: max(-h[i], -rho) <= u[i] <= min(1-h[i], rho), -inf <= t <= C_val
    
    A_ub = np.zeros((k, n + 1))
    b_ub = np.zeros(k)
    
    for idx, s in enumerate(active_s):
        g = grad_lag_s(h, s) * dx  # include dx
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -V_active[idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C_val))
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            u = res.x[:n]
            if np.max(np.abs(u)) > 1e-10:
                return u
    except Exception:
        pass
    return None

def lp_refine(h, n_iters, is_dead, eta_start=0.01, K=20, rho_start=0.15):
    """Iteratively refine h using LP directions."""
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    eta = eta_start
    rho = rho_start
    stall = 0
    
    for it in range(n_iters):
        if is_dead():
            break
        
        vals = active_values(best)
        C = vals.max()
        
        # Select active set
        threshold = C - eta
        active_mask = vals >= threshold
        active_idx = np.where(active_mask)[0]
        
        # If too many, take top K
        if len(active_idx) > K:
            top_vals = vals[active_idx]
            order = np.argsort(-top_vals)[:K]
            active_idx = active_idx[order]
        
        # Ensure at least the argmax is included
        argmax = np.argmax(vals)
        if argmax not in active_idx:
            active_idx = np.append(active_idx, argmax)
        
        V_active = vals[active_idx]
        active_s = active_idx.tolist()
        
        u = lp_direction(best, active_s, rho, C, V_active)
        
        accepted = False
        if u is not None:
            # Try various step sizes
            for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04]:
                trial = make_feasible(best + alpha * u)
                sc = score_exact(trial)
                if sc < best_sc - 1e-14:
                    best = trial
                    best_sc = sc
                    accepted = True
                    stall = 0
                    break
        
        if not accepted:
            stall += 1
            rho *= 0.7
            if stall % 5 == 0:
                eta = min(eta * 1.5, 0.1)
            if rho < 0.001 or eta > 0.1:
                # Try to recover
                if stall > 30:
                    break
        else:
            # Slightly increase rho if we made progress
            rho = min(rho * 1.1, 0.3)
    
    return best, best_sc

def binary_init_from_float(h, n):
    """Create a balanced binary vector from a float vector."""
    b = (h > 0.5).astype(float)
    target = n // 2
    cur = int(np.sum(b))
    if cur == target:
        return b
    elif cur > target:
        # Need to flip (cur - target) ones to zeros
        # Prefer flipping positions where h is closest to 0.5 (from above)
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])  # smallest h among ones = most easily flipped
        for i in range(cur - target):
            b[ones[order[i]]] = 0
    else:
        # Need to flip (target - cur) zeros to ones
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])  # largest h among zeros = most easily flipped to 1
        for i in range(target - cur):
            b[zeros[order[i]]] = 1
    return b

def binary_sa(h0, n_steps, rng, is_dead):
    """Simulated annealing on balanced binary strings."""
    n = len(h0)
    target = n // 2
    bits = binary_init_from_float(h0, n)
    cur_sc = score_exact(bits)
    best_bits = bits.copy()
    best_sc = cur_sc
    
    T = 0.001
    pos1 = np.where(bits == 1)[0].tolist()
    pos0 = np.where(bits == 0)[0].tolist()
    
    for step in range(n_steps):
        if is_dead():
            break
        
        # Random swap
        i = pos1[rng.integers(len(pos1))]
        j = pos0[rng.integers(len(pos0))]
        
        bits[i] = 0
        bits[j] = 1
        pos1.remove(i)
        pos1.append(j)
        pos0.remove(j)
        pos0.append(i)
        
        new_sc = score_exact(bits)
        delta = new_sc - cur_sc
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_bits = bits.copy()
                best_sc = new_sc
        else:
            # Revert
            bits[i] = 1
            bits[j] = 0
            pos1.remove(j)
            pos1.append(i)
            pos0.remove(i)
            pos0.append(j)
        
        T *= 0.999
    
    return best_bits, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [(t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = score_exact(best)
    
    return best, best_sc

import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def grad_lag_s(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        grad[s:] += 1.0 - h[:m]
        grad[:m] -= h[s:]
    return grad

def active_values(h):
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full')
    vals = full[n - 1:n - 1 + n]
    return vals * dx

def lp_direction(h, active_s, V_active, rho, C_val):
    n = len(h)
    dx = 2.0 / n
    k = len(active_s)
    if k == 0:
        return None
    A_ub = np.zeros((k, n + 1))
    b_ub = np.zeros(k)
    for idx, s in enumerate(active_s):
        g = grad_lag_s(h, s) * dx
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -V_active[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, C_val + 1e-12))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

def binary_init_from_float(h, n):

        order = np.argsort(-h[zeros])

def interp_to(h_src, n_tgt):

def two_point_search(h, n_tries, rng, is_dead):

    best = h.copy()

        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [(t_lo + t_hi) / 2.0,
                        t_lo + 0.3 * span, t_lo + 0.7 * span,
                        t_lo + 0.15 * span, t_lo + 0.85 * span,
                        t_lo + (1.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:

                best_t = t
        if best_t is not None:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
