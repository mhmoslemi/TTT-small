import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def build_D(h):
    """Build the derivative matrix D where D[l, i] = d/dh_i corr_full[l]"""
    n = len(h)
    D = np.zeros((2*n-1, n))
    one_minus_h = 1.0 - h
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        D[:, i] = np.correlate(e, one_minus_h, "full") - np.correlate(h, e, "full")
    return D

def project_feasible(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}"""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lp_direction(h, trust=0.05, active_threshold=0.01):
    """Find a direction d (sum=0, |d_i|<=trust) that decreases max correlation.
    Returns (d, theta) where theta is the predicted decrease."""
    n = len(h)
    dx = 2.0 / n
    corr0 = np.correlate(h, 1.0 - h, "full")
    max_corr = np.max(corr0)
    
    # Build D
    D = build_D(h)  # shape (2n-1, n)
    
    # LP: maximize theta
    # Variables: d (n) + theta (1) = n+1 vars
    # Constraints: D @ d + theta <= max_corr - corr0  (for all lags)
    #              sum(d) = 0
    #              |d_i| <= trust
    
    m = 2*n - 1
    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -theta = maximize theta
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, n] = 1.0
    b_ub = max_corr - corr0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = [(-trust, trust)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success and res.x[-1] > 1e-10:
        return res.x[:n], res.x[-1]
    return None, 0.0

def make_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

def make_golden_init(n):
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 - frac
    return project_feasible(h, n / 2.0)

def interpolate_h(h_from, n_to):
    """Interpolate h_from (length n_from) to length n_to."""
    n_from = len(h_from)
    x_from = np.linspace(0.5/n_from, 1.0 - 0.5/n_from, n_from)
    x_to = np.linspace(0.5/n_to, 1.0 - 0.5/n_to, n_to)
    h_to = np.interp(x_to, x_from, h_from)
    return project_feasible(h_to, n_to / 2.0)

def pair_polish(h, n_passes=3):
    """Do a few passes of pairwise optimization."""
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    
    for _ in range(n_passes):
        for i in range(n):
            for j in range(i+1, n):
                # Optimize h[i], h[j] while keeping sum fixed
                hi, hj = h[i], h[j]
                # Try moving mass from i to j or vice versa
                # h[i] - delta, h[j] + delta
                # Find optimal delta
                lo = max(-hj, hi - 1.0)
                hi_bound = min(hi, 1.0 - hj)
                if hi_bound - lo < 1e-14:
                    continue
                # Golden section search for optimal delta
                a, b = lo, hi_bound
                gr = (np.sqrt(5) - 1) / 2
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                h[i], h[j] = hi - c_pt, hj + c_pt
                fc = np.max(np.correlate(h, 1-h, "full"))
                h[i], h[j] = hi - d_pt, hj + d_pt
                fd = np.max(np.correlate(h, 1-h, "full"))
                for _ in range(20):
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        h[i], h[j] = hi - c_pt, hj + c_pt
                        c_pt = b - gr * (b - a)
                        fc = np.max(np.correlate(h, 1-h, "full"))
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        h[i], h[j] = hi - d_pt, hj + d_pt
                        d_pt = a + gr * (b - a)
                        fd = np.max(np.correlate(h, 1-h, "full"))
                # Use midpoint
                delta = (a + b) / 2
                h[i], h[j] = hi - delta, hj + delta
        
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h)
        if c5 >= best_c5:
            break
        best_c5 = c5
    
    return h, best_c5

import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_D(h):
    n = len(h)
    D = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        D[:, i] = np.correlate(e, omh, "full") - np.correlate(h, e, "full")
    return D


def lp_direction(h, trust=0.04):
    n = len(h)
    corr0 = np.correlate(h, 1.0 - h, "full")
    max_corr = float(np.max(corr0))
    D = build_D(h)
    m = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, n] = 1.0
    b_ub = max_corr - corr0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-trust, trust)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[n] > 1e-10:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, 0.0


def subgradient_direction(h, active_tol=0.005):
    n = len(h)
    corr0 = np.correlate(h, 1.0 - h, "full")
    max_corr = float(np.max(corr0))
    active = np.where(corr0 >= max_corr - active_tol * (2.0 / n))[0]
    D = build_D(h)
    d = -np.mean(D[active, :], axis=0)
    d -= np.mean(d)
    norm = np.max(np.abs(d))
    if norm < 1e-15:
        return None
    d /= norm
    return d

def _max_alpha(h, d):
    n = len(h)
    alpha_max = 1.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    return max(alpha_max, 0.0)


def pair_polish(h, n_passes=2, time_limit=None):
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()

    for p in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time_limit is not None and time.time() - t_start > time_limit:
                    return best_h, best_c5
                hi_v, hj_v = h[i], h[j]
                lo = max(-hj_v, hi_v - 1.0)
                hi_b = min(hi_v, 1.0 - hj_v)
                if hi_b - lo < 1e-13:
                    continue
                # Golden section
                a, b = lo, hi_b
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                h[i], h[j] = hi_v - c_pt, hj_v + c_pt
                fc = float(np.max(np.correlate(h, 1.0 - h, "full")))
                h[i], h[j] = hi_v - d_pt, hj_v + d_pt
                fd = float(np.max(np.correlate(h, 1.0 - h, "full")))
                for _ in range(16):
                    if fc < fd:
                        b = d_pt; d_pt = c_pt; fd = fc
                        h[i], h[j] = hi_v - c_pt, hj_v + c_pt
                        c_pt = b - gr * (b - a)
                        fc = float(np.max(np.correlate(h, 1.0 - h, "full")))
                    else:
                        a = c_pt; c_pt = d_pt; fc = fd
                        h[i], h[j] = hi_v - d_pt, hj_v + d_pt
                        d_pt = a + gr * (b - a)
                        fd = float(np.max(np.correlate(h, 1.0 - h, "full")))
                delta = (a + b) / 2.0
                h[i], h[j] = hi_v - delta, hj_v + delta
                if abs(delta) > 1e-12:
                    improved = True
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)

            n_p = len(parent_h)
        parent_h = project_feasible(parent_h, n_p / 2.0)
        c5_p = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = c5_p
        best_n = n_p

    if best_h is None:
        best_h = make_golden_init(70)
        best_c5 = compute_c5(best_h)
        best_n = 70

    # Determine sizes to explore
    base_n = best_n
    sizes = []
    for s in range(base_n, max(30, base_n - 4), -2):
        sizes.append(s)
    for s in range(base_n - 4, max(28, base_n - 30), -2):
        if s not in sizes:
            sizes.append(s)
    # Also try a couple larger
    for s in [base_n + 2, base_n + 4, base_n + 6]:
        if s <= 120:
            sizes.append(s)
    sizes = sorted(set(sizes), key=lambda x: abs(x - base_n))
    sizes = sizes[:14]  # limit to 14 sizes

    # Time allocation: more time for base_n and smaller sizes
    weights = np.array([1.0 / (1.0 + 0.3 * abs(s - base_n)) for s in sizes])
    weights /= weights.sum()

    # Phase 1: Optimize base_n with LP search (primary)
    phase1_time = time_limit * 0.45
    h_curr = best_h.copy()
    h_opt, c5_opt = local_search(h_curr, phase1_time, rng, trust_init=0.04)
    h_pol, c5_pol = pair_polish(h_opt, n_passes=2)
    h_pol = project_feasible(h_pol, best_n / 2.0)
    c5_pol = compute_c5(h_pol)
    if c5_pol < best_c5:
        best_h = h_pol.copy()
        best_c5 = c5_pol

    # Phase 2: Try other sizes
    phase2_start = time.time()
    phase2_end = t0 + time_limit * 0.85

    for n in sizes:
        if n == best_n:
            continue
        if time.time() > phase2_end:
            break

        # Allocate remaining time proportionally
        remaining = phase2_end - time.time()
        if remaining < 3:
            break
        alloc = max(3.0, remaining * weights[sizes.index(n)] * 1.5)
        alloc = min(alloc, remaining - 2.0)
        if alloc < 2.0:
            continue

        # Create initial h for this size
        h_init = interpolate_h(best_h, n)
        h_init = project_feasible(h_init, n / 2.0)

        # Run local search
        h_local, c5_local = local_search(h_init, alloc, rng, trust_init=0.04)

        # Polish
        h_pol2, c5_pol2 = pair_polish(h_local, n_passes=2)
        h_pol2 = project_feasible(h_pol2, n / 2.0)
        c5_pol2 = compute_c5(h_pol2)

        if c5_pol2 < best_c5:
            best_h = h_pol2.copy()
            best_c5 = c5_pol2
            best_n = n

    # Phase 3: Final refinement on best_n with remaining time
    remaining = t0 + time_limit - time.time()
    if remaining > 5:
        h_curr = best_h.copy()
        h_final, c5_final = local_search(h_curr, remaining * 0.6, rng, trust_init=0.03)
        h_pol3, c5_pol3 = pair_polish(h_final, n_passes=3, time_limit=remaining * 0.35)
        h_pol3 = project_feasible(h_pol3, best_n / 2.0)
        c5_pol3 = compute_c5(h_pol3)
        if c5_pol3 < best_c5:
            best_h = h_pol3.copy()
            best_c5 = c5_pol3

    # Phase 4: Try a few random restarts if time allows
    remaining = t0 + time_limit - time.time()
    if remaining > 15:
        for attempt in range(3):
            if time.time() > t0 + time_limit - 5:
                break
            n_try = best_n if rng.random() < 0.5 else max(32, best_n - 2 * (attempt + 1))
            if n_try % 2 != 0:
                n_try -= 1
            h_r = interpolate_h(best_h, n_try) + rng.randn(n_try) * 0.08
            h_r = project_feasible(h_r, n_try / 2.0)
            alloc_r = max(3.0, (t0 + time_limit - 5 - time.time()) * 0.4)
            h_l, c5_l = local_search(h_r, alloc_r, rng, trust_init=0.04)
            h_p, c5_p = pair_polish(h_l, n_passes=1)
            h_p = project_feasible(h_p, n_try / 2.0)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_h = h_p.copy()
                best_c5 = c5_p
                best_n = n_try

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    best_n = len(best_h)

    return best_h, final_c5, best_n
