def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.clip(z, -8.0, 8.0)
    lo, hi = -8.0, 8.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            if residual <= 1e-15:
                break
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
    elif residual < -1e-12:

        for idx in order:
            if -residual <= 1e-15:
                break
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
    return np.clip(h, 0.0, 1.0)


def grad_full(h):
    """Return matrix G of shape (2*n-1, n), G[p, i] = d/dh_i (vals[p]).
    vals[p] corresponds to correlate output index p, i.e. shift k = p-(n-1)."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    c = n - 1
    # p = c  -> k = 0
    G[c] = (1.0 - 2.0 * h) * dx
    for p in range(2 * n - 1):
        k = p - c
        if k == 0:
            continue
        if k > 0:
            L = n - k
            G[p, k:] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[k:]
        else:
            u = -k
            L = n - u
            G[p, :L] += dx * (1.0 - h[u:])
            G[p, u:] -= dx * h[:L]
    return G


def grad_selftest():
    """Finite-difference verification of grad_full on a tiny instance."""
    rng = np.random.default_rng(0)
    n = 8
    h = make_feasible(rng.uniform(0, 1, n))
    G = grad_full(h)
    vals = compute_vals(h)
    eps = 1e-6
    pidx = [0, n, 2 * n - 2, (2 * n - 1) // 2]
    for p in pidx:
        for i in range(n):
            hp = h.copy(); hp[i] = min(1.0, hp[i] + eps)
            hm = h.copy(); hm[i] = max(0.0, hm[i] - eps)
            fd = (compute_vals(hp)[p] - compute_vals(hm)[p]) / (2.0 * eps)
            if abs(fd - G[p, i]) > 1e-4:
                return False
    return True


# ---------------------------------------------------------------------------
# Smoothed (log-sum-exp) projected gradient
# ---------------------------------------------------------------------------

def smooth_obj_grad(h, beta):
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    mx = shifted.max()
    e = np.exp(shifted - mx)
    logsum = mx + np.log(e.sum())
    obj = cmax + logsum / beta
    w = e / e.sum()
    G = grad_full(h)
    grad = w @ G
    return obj, grad


def pg_solve(h0, beta, time_limit=20.0, max_iter=30000):
    h = np.asarray(h0, dtype=float).copy()
    f, g = smooth_obj_grad(h, beta)
    if not np.isfinite(f):
        return h
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-30:
        return h
    alpha = min(1.0 / (gnorm2 + 1e-12), 2.0 / (np.max(g * g) + 1e-12))
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        f, g = smooth_obj_grad(h, beta)
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-30:
            break
        step = alpha
        best_y, best_f = h, f
        for _ in range(40):
            y = make_feasible(h - step * g)
            fy = smooth_obj_grad(y, beta)[0]
            if np.isfinite(fy) and fy <= f - 1e-5 * step * gnorm2:
                best_y, best_f = y, fy
                break
            step *= 0.4
        if step < 1e-13:
            break
        if best_f >= f:
            break
        h, f = best_y, best_f
        alpha = min(alpha * 1.7, 4.0 / (gnorm2 + 1e-12))
    return h


# ---------------------------------------------------------------------------
# Subgradient exact step
# ---------------------------------------------------------------------------

def subgrad_step(h, step):
    vals = compute_vals(h)
    cmax = vals.max()
    thr = cmax - max(0.02 * cmax, 1e-5)
    act = np.where(vals >= thr)[0]
    G = grad_full(h)
    g = G[act].mean(axis=0)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h
    return make_feasible(h - step * g / gn)


# ---------------------------------------------------------------------------
# Symmetries / transformations
# ---------------------------------------------------------------------------

def symmetrize(h):
    h_sym = 0.5 * (h + (1.0 - h[::-1]))
    return make_feasible(h_sym)


def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)


# ---------------------------------------------------------------------------
# Structured initialisers
# ---------------------------------------------------------------------------

def sine_start(n, f1, f2, a1, a2, ph1, ph2):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + ph1) + a2 * np.cos(2 * np.pi * f2 * x + ph2)
    return make_feasible(h)


def block_start(n, m, rng):
    # m blocks, each with random level, then fitted to mean
    per = max(1, n // m)
    h = np.empty(n)
    j = 0
    while j < n:
        lvl = rng.uniform(0.0, 1.0)
        seg = int(rng.integers(max(1, per // 2), per + 1))
        h[j:j + seg] = lvl
        j += seg
    return make_feasible(h)


# ---------------------------------------------------------------------------
# Active-set SLSQP (exact constraints t >= vals[p])
# ---------------------------------------------------------------------------

class SlqProblem:
    def __init__(self, active, use_jac):
        self.active = np.asarray(active, dtype=int)
        self.n = len(self.active)  # placeholder; real n set later
        self.use_jac = use_jac

    def set_n(self, n):
        self.n = n
        self.target = n / 2.0

    def fun_obj(self, z):
        return z[-1]

    def jac_obj(self, z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def fun_in(self, z):
        h = z[:self.n]
        return z[-1] - compute_vals(h)[self.active]

    def jac_in(self, z):
        h = z[:self.n]
        G = grad_full(h)[self.active, :]
        r = G.shape[0]
        J = np.concatenate([G, np.ones((r, 1))], axis=1)
        return -J  # constraint is t - vals >= 0  ->  grad = [-dvals, +1]

    def fun_eq(self, z):
        return np.sum(z[:self.n]) - self.target

    def jac_eq(self, z):
        g = np.zeros(len(z))
        g[:self.n] = 1.0
        return g


def slsqp_solve(h0, active, max_iter=200, use_jac=True):
    n = len(h0)
    c = SlqProblem(list(active), use_jac)
    c.set_n(n)
    t0_init = max(0.0, official_score(h0) - 1e-4) + 5e-4
    z0 = np.concatenate([h0, [t0_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons_in = {'fun': c.fun_in}
    if use_jac:
        cons_in['jac'] = c.jac_in
    cons = [
        {'type': 'ineq', **cons_in},
        {'type': 'eq', 'fun': c.fun_eq, 'jac': c.jac_eq},
    ]
    try:
        res = minimize(c.fun_obj, z0, jac=c.jac_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            if np.isfinite(official_score(h_new)):
                return h_new
    except Exception:
        pass
    return None


def find_active(vals, frac=0.3, min_count=8):
    cmax = vals.max()
    thr = cmax - max(0.01 * cmax, 1e-6)
    act = set(np.where(vals >= thr)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    act.update(top_idx.tolist())
    return sorted(act)


def slsqp_refine(h0, use_jac, max_iter=250):
    h = h0.copy()
    for _ in range(3):
        vals = compute_vals(h)
        act = find_active(vals, frac=0.35, min_count=10)
        h_new = slsqp_solve(h, act, max_iter=max_iter, use_jac=use_jac)
        if h_new is None:
            break
        sc = official_score(h_new)
        if sc + 1e-12 < official_score(h):
            h = h_new
        else:
            break
    return h


# ---------------------------------------------------------------------------
# Refinement driver for a single candidate
# ---------------------------------------------------------------------------

def refine_candidate(h, use_jac, deadline):
    h = np.asarray(h, dtype=float).copy()
    sc = official_score(h)
    # PG with growing beta
    for beta in [30, 100, 400, 1500, 6000]:
        if time.time() >= deadline:
            break
        tl = min(8.0, max(2.0, (deadline - time.time()) * 0.15))
        h2 = pg_solve(h, beta, time_limit=tl)
        s2 = official_score(h2)
        if s2 + 1e-13 < sc:
            h, sc = h2, s2
    # SLSQP active-set
    if time.time() < deadline - 8:
        h2 = slsqp_refine(h, use_jac, max_iter=200)
        s2 = official_score(h2)
        if s2 + 1e-13 < sc:
            h, sc = h2, s2
    # subgradient polish
    if time.time() < deadline - 4:
        hs = h.copy()
        step = 0.15
        for _ in range(80):
            if time.time() >= deadline:
                break
            hs = subgrad_step(hs, step)
            ss = official_score(hs)
            if ss + 1e-13 < sc:
                h, sc = hs, ss
            step *= 0.96
    return h, sc


# ---------------------------------------------------------------------------
# Main
# ---------------------------------------------------------------------------

def accept(state, h, sc):
    best_h, best_score = state
    if not (np.all(np.isfinite(h)) and np.isfinite(sc)):
        return best_h, best_score
    if sc < best_score - 1e-13:
        return h.copy(), float(sc)
    if abs(sc - best_score) <= 1e-13 and best_h is not None and len(h) < len(best_h):
        return h.copy(), float(sc)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)
    use_jac = grad_selftest()

    best_h, best_score = None, float('inf')

    # --- parent ---
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h, best_score = parent.copy(), ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        best_h = make_feasible(np.concatenate([np.ones(33), np.zeros(33)]))
        best_score = official_score(best_h)

    # Quick refine of the parent first (highest value) so we always improve it.
    if time.time() < deadline - 20:
        rh, rs = refine_candidate(best_h, use_jac, deadline)
        best_h, best_score = accept((best_h, best_score), rh, rs)

    base_n = len(best_h)

    # --- Phase: anti-symmetry orbit ---
    if time.time() < deadline - 40:
        hs = symmetrize(best_h)
        hs, ss = refine_candidate(hs, use_jac, deadline)
        best_h, best_score = accept((best_h, best_score), hs, ss)

    # --- Phase: multi-resolution ---
    if time.time() < deadline - 60:
        n_opts = sorted(set([
            max(40, base_n - 6), max(42, base_n - 4), base_n - 2,
            base_n, base_n + 2, min(80, base_n + 4), min(84, base_n + 8),
        ]))
        for no in n_opts:
            if time.time() >= deadline - 30:
                break
            if no < 8 or no == base_n:
                continue
            h_int = interpolate_to(best_h, no)
            h_int = symmetrize(h_int)
            rh, rs = refine_candidate(h_int, use_jac, deadline)
            best_h, best_score = accept((best_h, best_score), rh, rs)

    # --- Phase: structured starts at a couple of resolutions ---
    if time.time() < deadline - 50:
        n_start = max(base_n, 60)
        starts = []
        # sine family
        for f1 in [1, 2]:
            for f2 in [2, 3, 4]:
                for a1 in [0.15, 0.3]:
                    for a2 in [0.05, 0.12]:
                        for ph in [0.0, 1.3]:
                            starts.append(sine_start(n_start, f1, f2, a1, a2, ph, 0.0))
        for s in starts[:6]:
            starts.append(s)
        # block family
        for m in [4, 6, 8, 10]:
            starts.append(block_start(n_start, m, rng))
        # binary-ish (thue-morse like) via block_start with tight levels
        rb = np.random.default_rng(seed + 7)
        binh = np.zeros(n_start)
        arr = 1.0
        for i in range(n_start):
            binh[i] = arr if (i & 1) == 0 else 0.0
            if i % (n_start // 2) == 0:
                arr = 1.0 - arr if arr == 1.0 else 1.0
        # simpler: use a periodic balanced 0/1 pattern
        pat = np.tile([0.0, 1.0, 1.0, 0.0, 1.0, 0.0, 0.0, 1.0], max(1, len(range(0, n_start, 8))))
        starts.append(make_feasible(pat[:n_start]))

        for s in starts[:10]:
            if time.time() >= deadline - 25:
                break
            if time.time() >= deadline:
                break
            rl, rs2 = refine_candidate(s, use_jac, deadline)
            best_h, best_score = accept((best_h, best_score), rl, rs2)

    # --- Phase: random restarts around current best ---
    if time.time() < deadline - 25:
        ncur = len(best_h)
        n_starts = max(3, min(12, int((deadline - time.time()) // 18)))
        sigmas = [0.02, 0.04, 0.06, 0.1, 0.15]
        for i in range(n_starts):
            if time.time() >= deadline - 15:
                break
            sig = sigmas[i % len(sigmas)]
            cand = make_feasible(best_h + rng.normal(0.0, sig, size=ncur))
            rh, rs = refine_candidate(cand, use_jac, deadline)
            best_h, best_score = accept((best_h, best_score), rh, rs)
            # also a symmetric version of the perturbation
            if time.time() < deadline - 12:
                cand2 = symmetrize(make_feasible(best_h + rng.normal(0.0, sig, size=ncur)))
                rh2, rs2 = refine_candidate(cand2, use_jac, deadline)
                best_h, best_score = accept((best_h, best_score), rh2, rs2)

    # --- Final intensive polish of the best point ---
    if time.time() < deadline - 2:
        rh, rs = refine_candidate(best_h, use_jac, deadline)
        best_h, best_score = accept((best_h, best_score), rh, rs)

    # --- Final validation ---
    final_h = make_feasible(best_h)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_n = len(final_h)
    # enforce exact sum
    diff = final_n / 2.0 - np.sum(final_h)
    if diff != 0.0:
        final_h = make_feasible(final_h)
    final_score = official_score(final_h)

    assert np.all(final_h >= -1e-15), "h negative"
    assert np.all(final_h <= 1.0 + 1e-15), "h > 1"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-9, "sum constraint violated"

    final_bound = float(final_score)
    return final_h, final_bound, final_n
