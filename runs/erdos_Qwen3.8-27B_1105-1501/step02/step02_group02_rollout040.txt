        y = _project(y)
        # Step 3: Gradient step with backtracking
        g = _lse_jac(y, beta)
        g = g - np.mean(g)
        x_new = _project(y - alpha * g)
        f_new = _score(x_new)
        f_y = _score(y)

        while f_new > f_y - 0.5 * alpha * np.dot(g, y - x_new) and bt < 15:
            alpha *= 0.5
            x_new = _project(y - alpha * g)
            f_new = _score(x_new)
            bt += 1
        
        x_old = x
        x = x_new
        
        if f_new < best_s:
            best_s = f_new
            best_h = x.copy()
        
        t = t_new
        # Periodically boost alpha
        if alpha < alpha_init * 0.01:
            alpha = alpha_init
    
    return best_h, best_s

import numpy as np
from scipy.optimize import minimize
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - float(np.max(z))
    hi = 2.0 - float(np.min(z))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.exp(sh).sum()
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta


def _lse_jac(x, beta):
    n = len(x)
    h = x
    cv = _corr_vec(h)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    dx = 2.0 / n
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        s = k_idx - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                g[s:] += wk * (1.0 - h[:ns])
                g[:ns] += -wk * h[s:]
        else:
            t = -s
            nt = n - t
            if nt > 0:
                g[:nt] += wk * (1.0 - h[t:])
                g[t:] += -wk * h[:nt]
    return g


def _slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_jac,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _slsqp_direct_active(h0, maxiter=200, tol=0.005, max_active=30):
    n = len(h0)
    cv = _corr_vec(h0)
    max_cv = cv.max()
    active_idx = np.argsort(cv)[::-1][:max_active]

    z0 = np.concatenate([h0, [max_cv + 0.002]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cvl = _corr_vec(h)
        return t - cvl[active_idx]

    def _ieq_jac(z):
        h = z[:-1]
        nl = len(h)
        jmat = np.zeros((len(active_idx), len(z)))
        dxl = 2.0 / nl
        for row, k_idx in enumerate(active_idx):
            s = k_idx - (nl - 1)
            if s >= 0:
                ns = nl - s
                if ns > 0:
                    jmat[row, s:s + ns] += dxl * (1.0 - h[:ns])
                    jmat[row, :ns] += -dxl * h[s:]
            else:
                t2 = -s
                nt = nl - t2
                if nt > 0:
                    jmat[row, :nt] += dxl * (1.0 - h[t2:])
                    jmat[row, t2:] += -dxl * h[:nt]
        return jmat

    def _eq(z):
        return z[:-1].sum() - n / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq, 'jac': _ieq_jac},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0


def _proj_grad_nesterov(h0, beta, n_iter=5000, alpha_init=0.05):
    n = len(h0)
    x = h0.copy()
    x_old = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    alpha = alpha_init
    t = 1.0
    stale = 0
    for it in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_old)
        y = _project(y)
        g = _lse_jac(y, beta)
        g = g - np.mean(g)
        x_new = _project(y - alpha * g)
        f_new = _score(x_new)
        f_y = _score(y)
        bt = 0
        while f_new > f_y - 0.5 * alpha * np.dot(g, y - x_new) and bt < 12:
            alpha *= 0.5
            x_new = _project(y - alpha * g)
            f_new = _score(x_new)
            bt += 1
        x_old = x.copy()
        x = x_new
        if f_new < best_s - 1e-14:
            best_s = f_new
            best_h = x.copy()
            stale = 0
        else:
            stale += 1
        if stale > 800:
            x_old = x.copy()
            stale = 0
        if alpha < alpha_init * 0.01:
            alpha = alpha_init
        t = t_new
    return best_h, best_s


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _full_chain(h0, deadline):
    h = h0.copy()
    s = _score(h)
    if time.time() < deadline - 20:

    if time.time() < deadline - 18:
