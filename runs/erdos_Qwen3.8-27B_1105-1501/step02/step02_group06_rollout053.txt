def lp_step(h0, k_active, radius):
    n = len(h0)
    vals_raw = np.correlate(h0, 1.0 - h0, mode='full')
    dx = 2.0 / n
    vals = vals_raw * dx
    top_idx = np.argsort(vals)[-k_active:]
    
    # Build LP: min t
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: [0, ..., 0, 1]
    # Constraints:
    #   For each active p: grad_p · h - t <= grad_p · h0 - raw[p]
    #   sum(h) = n/2
    #   -radius <= h_i - h0_i <= radius
    #   0 <= h_i <= 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    
    for p in top_idx:
        s = p - (n - 1)
        g = gradient_p(h0, s)
        # g · h - t <= g · h0 - raw[p]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(g @ h0 - vals_raw[p])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: max(0, h0_i - radius) <= h_i <= min(1, h0_i + radius)
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - radius)
        hi = min(1.0, h0[i] + radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status not in [0, 3]:  # 0=optimal, 3=inaccurate but feasible
        return None, None
    
    h_new = res.x[:n]
    t_new = res.x[-1]
    return h_new, t_new

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)

def lp_minimax(h0, radius, k_top=None):
    """Solve LP minimax subproblem. Returns (h_new, direction) or (None, None)."""
    n = len(h0)
    dx = 2.0 / n
    raw = np.correlate(h0, 1.0 - h0, mode='full')
    
    # Select constraints: top k by value, or all if k_top is None
    if k_top is not None and k_top < 2 * n - 1:
        top_idx = np.argsort(raw)[-k_top:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    m = len(top_idx)
    
    # Build LP
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for i, p in enumerate(top_idx):
        s = p - (n - 1)
        g = grad_for_shift(h0, s)
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = g @ h0 - raw[p]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - radius)
        hi = min(1.0, h0[i] + radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if res.status not in [0, 3]:
        return None, None
    
    h_new = res.x[:n]
    if not np.all(np.isfinite(h_new)):
        return None, None
    
    direction = h_new - h0
    return h_new, direction

def line_search(h0, direction, max_alpha=2.0, n_samples=8):
    """Find best alpha in [0, max_alpha] along direction."""
    n = len(h0)
    # Find max feasible alpha
    alpha_max = max_alpha
    for i in range(n):
        if direction[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h0[i]) / direction[i])
        elif direction[i] < -1e-14:
            alpha_max = min(alpha_max, h0[i] / (-direction[i]))
    
    if alpha_max < 1e-10:
        return h0, 0.0, official_score(h0)
    
    alpha_max = min(alpha_max, max_alpha)
    alphas = np.linspace(0.0, alpha_max, n_samples)
    
    best_h = h0.copy()
    best_sc = official_score(h0)
    best_alpha = 0.0
    
    for a in alphas[1:]:
        h_try = make_feasible(h0 + a * direction)
        sc = official_score(h_try)
        if sc < best_sc:
            best_sc = sc
            best_h = h_try
            best_alpha = a
    
    return best_h, best_alpha, best_sc

def lp_optimize(h0, max_iter=200, radius_init=0.1, k_top=None, time_limit=None, t_start=None):
    """Main LP-based minimax optimization loop."""
    h = h0.copy()
    best = official_score(h)
    radius = radius_init
    stall = 0
    
    for it in range(max_iter):
        if time_limit is not None:
            if t_start is not None and time.time() - t_start > time_limit:
                break
        
        h_new, direction = lp_minimax(h, radius, k_top=k_top)
        if h_new is None:
            radius *= 0.7
            stall += 1
            if stall > 10:
                break
            continue
        
        h_ls, alpha, sc_ls = line_search(h, direction, max_alpha=1.5, n_samples=6)
        
        if sc_ls < best - 1e-14:
            h = h_ls
            best = sc_ls
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            # Try smaller steps
            improved = False
            for a_frac in [0.5, 0.3, 0.1]:
                h_try = make_feasible(h + a_frac * alpha * direction)
                sc_try = official_score(h_try)
                if sc_try < best - 1e-14:
                    h = h_try
                    best = sc_try
                    improved = True
                    stall = 0
                    break
            if not improved:
                radius *= 0.6
                stall += 1
                if stall > 15:
                    break
    
    return h, best

                return h_new, sc
    except Exception:
        pass
    return None, None

def random_perturb(h0, sigma, rng):
    n = len(h0)
    return make_feasible(h0 + rng.normal(0, sigma, size=n))

def construct_initial(n, mode):
    x = (np.arange(n) + 0.5) * 2.0 / n
    if mode == 0:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    elif mode == 1:
        h = 0.5 + 0.35 * np.sin(4 * np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.3 * np.sign(np.sin(2 * np.pi * x))
    elif mode == 3:
        h = 0.5 + 0.25 * np.sin(3 * np.pi * x) + 0.1 * np.sin(5 * np.pi * x)
    elif mode == 4:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x) * np.sin(np.pi * x)
    else:
        h = 0.5 + 0.2 * np.sin(6 * np.pi * x)
    return make_feasible(h)
