    grad_z = np.zeros(M)
    for j in range(M):
        grad_z[j] = ds[j] * (grad_hz[j] - grad_hz[N - 1 - j])
    return obj, grad_z

def find_active_set(vals, frac=0.3, min_count=10, max_count=40):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 5e-5)
    active = np.where(vals >= threshold)[0]
    top_k = min(max_count, max(min_count, int(frac * len(vals))))
    top_idx = np.argsort(vals)[::-1][:top_k]
    active = np.union1d(active, top_idx)
    return np.sort(active)

def slsqp_sym(z0, N, active_idx, t0, bounds_z, max_iter=300, ftol=1e-12):
    M = N // 2
    n_vars = M + 1
    x0 = np.concatenate([z0, [t0]])
    bounds = list(bounds_z) + [(0.0, 2.0)]

    def obj_fun(xv):
        return xv[-1]

    def obj_jac(xv):
        g = np.zeros(n_vars)
        g[-1] = 1.0
        return g

    def con_fun(xv):
        zz = xv[:M]
        h = build_sym_h(zz, N)
        vals_ac = compute_overlaps(h)[active_idx]
        return xv[-1] - vals_ac

    def con_jac(xv):
        zz = xv[:M]
        h = build_sym_h(zz, N)
        G = compute_overlap_grads(h)
        s = 1.0 / (1.0 + np.exp(-np.clip(zz, -50, 50)))
        ds = s * (1.0 - s)
        n_act = len(active_idx)
        J_out = np.zeros((n_act, n_vars))
        for ai in range(n_act):
            gr = G[active_idx[ai]]
            for j in range(M):
                J_out[ai, j] = -ds[j] * (gr[j] - gr[N - 1 - j])
            J_out[ai, -1] = 1.0
        return J_out

    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fun, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        return res.x, res
    except Exception:
        return None, None

def slsqp_full(h0, N, active_idx, max_iter=200, ftol=1e-11):
    n_vars = N + 1
    x0 = np.concatenate([h0, [compute_overlaps(h0)[active_idx].max() + 1e-5]])
    bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]

    def obj_fun(xv):
        return xv[-1]

    def obj_jac(xv):
        g = np.zeros(n_vars)
        g[-1] = 1.0
        return g

    def eq_fun(xv):
        return np.sum(xv[:N]) - N / 2.0

    def eq_jac(xv):
        g = np.zeros(n_vars)
        g[:N] = 1.0
        return g

    def con_fun(xv):
        h = xv[:N]
        vals_ac = compute_overlaps(h)[active_idx]
        return xv[-1] - vals_ac

    def con_jac(xv):
        h = xv[:N]
        G = compute_overlap_grads(h)
        n_act = len(active_idx)
        J_out = np.zeros((n_act, n_vars))
        J_out[:, :N] = -G[active_idx]
        J_out[:, -1] = 1.0
        return J_out

    cons = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj_fun, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        return res.x, res
    except Exception:
        return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-15:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        N0 = 66
        h_init = np.full(N0, 0.5)
        h_init[:N0 // 2] += 0.2 * np.sin(np.linspace(0, np.pi, N0 // 2))
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    N = len(best_h)

    # Project to anti-symmetric and check improvement
    h_sym = 0.5 * (best_h + (1.0 - best_h[::-1]))
    h_sym = make_feasible(h_sym)
    sc_sym = official_score(h_sym)
    if sc_sym <= best_score + 1e-12:
        best_h = h_sym.copy()
        best_score = sc_sym

    N = len(best_h)
    M = N // 2
    z_init = h_to_z(best_h, N)

    # Phase 1: L-BFGS-B with increasing beta on anti-symmetric space
    z_cur = z_init.copy()
    beta_list = [15, 40, 100, 300, 800, 2000, 5000, 10000]
    for beta in beta_list:
        if time_left() < 10:
            break
        tl = min(time_left() * 0.2, 25.0)
        if tl < 3:
            break
        max_iter = max(50, int(tl * 30))
        try:
            res = minimize(
                fun=lambda zz: sym_soft_obj_grad(zz, N, beta)[0],
                x0=z_cur,
                jac=lambda zz: sym_soft_obj_grad(zz, N, beta)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'maxfun': max_iter * 10, 'gtol': 1e-10}
            )
            z_cur = np.clip(res.x, -25, 25)
            h_new = build_sym_h(z_cur, N)
            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if sc < best_score:
                update(h_new, sc)
                z_cur = h_to_z(best_h, len(best_h))
                N = len(best_h)
                M = N // 2
        except Exception:
            pass

    # Phase 2: Active set SLSQP in symmetric space (multiple rounds)
    for rnd in range(6):
        if time_left() < 20:
            break
        vals = compute_overlaps(best_h)
        fracs = [0.25, 0.35, 0.45, 0.55, 0.65, 0.75]
        frac = fracs[rnd % len(fracs)]
        active = find_active_set(vals, frac=frac, min_count=8, max_count=45)
        z0 = h_to_z(best_h, N)
        t0 = vals[active].max() + 1e-7
        bz = [(-20, 20)] * M
        max_iter = max(80, min(400, int(time_left() * 5)))
        x_res, res = slsqp_sym(z0, N, active, t0, bz, max_iter=max_iter, ftol=1e-13)
        if x_res is not None and np.all(np.isfinite(x_res)):
            h_new = build_sym_h(np.clip(x_res[:M], -30, 30), N)
            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if sc < best_score:
                update(h_new, sc)

    # Phase 3: Full-space non-symmetric refinement
    if time_left() > 30:
        vals = compute_overlaps(best_h)
        active = find_active_set(vals, frac=0.4, min_count=12, max_count=50)
        max_iter = max(50, min(300, int(time_left() * 3)))
        x_res, res = slsqp_full(best_h, N, active, max_iter=max_iter, ftol=1e-12)
        if x_res is not None and np.all(np.isfinite(x_res)):
            h_new = make_feasible(x_res[:N])
            sc = official_score(h_new)
            if sc < best_score:
                update(h_new, sc)
        # Second full-space attempt with different active set
        if time_left() > 15:
            vals2 = compute_overlaps(best_h)
            active2 = find_active_set(vals2, frac=0.3, min_count=8, max_color=existence)
            active2 = find_active_set(vals2, frac=0.3, min_count=8, max_count=30)
            if not np.array_equal(active2, active):
                x_res2, _ = slsqp_full(best_h, N, active2, max_iter=100, ftol=1e-12)
                if x_res2 is not None and np.all(np.isfinite(x_res2)):
                    h_new2 = make_feasible(x_res2[:N])
                    sc2 = official_score(h_new2)
                    if sc2 < best_score:
                        update(h_new2, sc2)

    # Phase 4: Multi-resolution search
    N_cur = len(best_h)
    n_opts = sorted(set([
        max(30, N_cur - 6), max(32, N_cur - 4),
        N_cur,
        min(80, N_cur + 2), min(80, N_cur + 4), min(80, N_cur + 6),
    ]))
    for N_try in n_opts:
        if time_left() < 25:
            break
        if N_try == N_cur:
            continue
        M_try = N_try // 2
        x_src = (np.arange(N_cur) + 0.5) * (2.0 / N_cur)
        x_tgt = (np.arange(N_try) + 0.5) * (2.0 / N_try)
        h_int = np.interp(x_tgt, x_src, best_h)
        h_int = make_feasible(h_int)
        # Symmetrize
        h_int_sym = 0.5 * (h_int + (1.0 - h_int[::-1]))
        h_int_sym = make_feasible(h_int_sym)
        sc_int = min(official_score(h_int), official_score(h_int_sym))
        h_cand = h_int_sym if official_score(h_int_sym) <= official_score(h_int) else h_int
        if sc_int < best_score:
            update(h_cand, sc_int)

        # Optimize at this resolution
        h_opt_src = h_cand
        M_opt = N_try // 2
        z_opt = h_to_z(h_opt_src, N_try)
        for beta in [300, 1500, 5000]:
            if time_left() < 8:
                break
            try:
                res = minimize(
                    fun=lambda zz, Nt=N_try, bt=beta: sym_soft_obj_grad(zz, Nt, bt)[0],
                    x0=z_opt,
                    jac=lambda zz, Nt=N_try, bt=beta: sym_soft_obj_grad(zz, Nt, bt)[1],
                    method='L-BFGS-B',
                    options={'maxiter': 200, 'maxfun': 2000, 'gtol': 1e-9}
                )
                h_new = build_sym_h(np.clip(res.x, -25, 25), N_try)
                h_new = make_feasible(h_new)
                sc = official_score(h_new)
                if sc < best_score:
                    update(h_new, sc)
                    z_opt = h_to_z(best_h, len(best_h))
                    N_try = len(best_h)
                    M_opt = N_try // 2
            except Exception:
                pass

        # Active set at this resolution
        if time_left() > 15:
            vals_t = compute_overlaps(h_cand if not (sc < best_score) else best_h)
            N_act = len(vals_t) // 2
            if N_act != M:
                pass
            h_use = best_h if sc < best_score else h_cand
            N_use = len(h_use)
            vals_use = compute_overlaps(h_use)
            active_use = find_active_set(vals_use, frac=0.35, min_count=8, max_count=35)
            M_use = N_use // 2
            z_use = h_to_z(h_use, N_use)
            t_use = vals_use[active_use].max() + 1e-7
            bz_use = [(-20, 20)] * M_use
            x_res, _ = slsqp_sym(z_use, N_use, active_use, t_use, bz_use,
                                  max_iter=150, ftol=1e-12)
            if x_res is not None and np.all(np.isfinite(x_res)):
                h_new = build_sym_h(np.clip(x_res[:M_use], -30, 30), N_use)
                h_new = make_feasible(h_new)
                sc = official_score(h_new)
                if sc < best_score:
                    update(h_new, sc)

    # Phase 5: Perturbation restarts
    if time_left() > 30:
        N_cur = len(best_h)
        M_cur = N_cur // 2
        z_best = h_to_z(best_h, N_cur)
        n_pert = max(3, min(12, int(time_left() / 15)))
        sigmas = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]
        for i in range(n_pert):
            if time_left() < 12:
                break
            sigma = sigmas[i % len(sigmas)]
            z_p = z_best + sigma * rng.standard_normal(M_cur)
            z_p = np.clip(z_p, -18, 18)
            h_p = build_sym_h(z_p, N_cur)
            h_p = make_feasible(h_p)
            sc_p = official_score(h_p)
            if sc_p < best_score:
                update(h_p, sc_p)
            # L-BFGS polish
            if time_left() > 8:
                try:
                    res = minimize(
                        fun=lambda zz: sym_soft_obj_grad(zz, N_cur, 3000)[0],
                        x0=z_p,
                        jac=lambda zz: sym_soft_obj_grad(zz, N_cur, 3000)[1],
                        method='L-BFGS-B',
                        options={'maxiter': 150, 'maxfun': 1500, 'gtol': 1e-8}
                    )
                    h_new = build_sym_h(np.clip(res.x, -25, 25), N_cur)
                    h_new = make_feasible(h
