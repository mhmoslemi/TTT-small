= dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

Hmm, that's V_{-k} = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

And V_k = dx * sum_{n=0}^{n-1-k} h[n+k]*(1-h[n]) = dx * sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])

These are NOT the same in general. So anti-symmetry does NOT imply V_k = V_{-k}.

Let me reconsider. Under anti-symmetry h[n-1-i] = 1-h[i]:
V_k = dx * sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
V_{-k} = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

Note: V_k + V_{-k} = dx * sum_{i=0}^{n-1-k} [h[i+k]*(1-h[i]) + h[i]*(1-h[i+k])]
= dx * sum_{i=0}^{n-1-k} [h[i+k] - h[i+k]*h[i] + h[i] - h[i]*h[i+k]]
= dx * sum_{i=0}^{n-1-k} [h[i+k] + h[i] - 2*h[i]*h[i+k]]

This is not particularly enlightening. The point is that anti-symmetry doesn't simplify the objective in an obvious way, but it does reduce the dimensionality and automatically satisfies the sum constraint.

Let me just focus on a clean implementation with:
1. Correct gradients (verified)
2. Log-sum-exp smoothing with projected gradient
3. SLSQP with all constraints (or a good active set)
4. Anti-symmetric initialization and projection
5. Multiple dimensions
6. Good time management

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    # Bisection for shift
    lo, hi = -15.0, 15.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if residual > 0:
        # Add to elements with most room
        room = 1.0 - h
        order = np.argsort(-room)  # most room first
        for idx in order:
            add = min(residual, room[idx])
            if add <= 0: break
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-15: break
    elif residual < 0:
        order = np.argsort(h)  # smallest first
        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 0: break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_all(h):
    """Compute gradient matrix G[k, i] = dV_k/dh_i for all k"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s == 0:
            G[p, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            L = n - s
            G[p, :L] -= dx * h[s:s+L]
            G[p, s:n] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:t+K])
            G[p, t:n] -= dx * h[:K]
    return G

def smooth_obj_grad(h, beta):
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_all(h)
    grad = w @ G
    return obj, grad

def pg_solve(h0, beta, time_limit=30.0, max_iter=50000):
    """Projected gradient with backtracking on smooth approximation"""
    n = len(h0)
    h = h0.copy()
    f, g = smooth_obj_grad(h, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-30:
        return h
    alpha = min(1.0 / (gnorm2 + 1e-12), 2.0 / (np.max(g**2) + 1e-12))
    t0 = time.time()
    best_f = f
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        f, g = smooth_obj_grad(h, beta)
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-30:
            break
        step = alpha
        accepted = False
        for _ in range(40):
            y = make_feasible(h - step * g)
            f_trial, _ = smooth_obj_grad(y, beta)
            if f_trial <= f - 1e-5 * step * gnorm2:
                accepted = True
                break
            step *= 0.5
        if not accepted or step < 1e-14:
            break
        h = y
        f = f_trial
        if f < best_f - 1e-12:
            best_f = f
            best_h = h.copy()
        alpha = min(alpha * 1.5, 5.0 / (gnorm2 + 1e-12))
    return best_h

def slsqp_solve(h0, active_idx, time_limit=60.0, max_iter=500):
    """SLSQP with constraints V_k <= t for k in active_idx"""
    n = len(h0)
    dx = 2.0 / n
    
    # We'll parameterize as [h (n vars), t (1 var)]
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def constr(z):
        h = z[:n]
        t = z[-1]
        # Inequality: t - V_k(h) >= 0
        vals = compute_vals(h)
        return t - vals[active_idx]
    
    def constr_jac(z):
        h = z[:n]
        G = grad_all(h)
        # For constraint j (corresponding to active_idx[j]):
        # d/dz of (t - V_{active_idx[j]}) = [-G[active_idx[j]], 1]
        rows = -G[active_idx, :]
        col = np.ones((len(active_idx), 1))
        return np.hstack([rows, col])
    
    def eq_fun(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    t0_init = official_score(h0) + 1e-4
    z0 = np.concatenate([h0, [t0_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': constr, 'jac': constr_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    start_time = time.time()
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def full_slsqp(h0, time_limit=120.0, max_iter=300):
    """SLSQP with ALL overlap constraints"""
    n = len(h0)
    all_idx = list(range(2 * n - 1))
    return slsqp_solve(h0, all_idx, time_limit=time_limit, max_iter=max_iter)

def find_active_set(vals, top_frac=0.3, min_count=10, threshold_frac=0.05):
    cmax = vals.max()
    threshold = cmax - max(threshold_frac * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0])
    top_k = max(min_count, int(top_frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def symmetrize(h):
    """Anti-symmetrize: h[i] = (h[i] + 1 - h[n-1-i]) / 2"""
    n = len(h)
    h_sym = np.zeros_like(h)
    for i in range(n):
        h_sym[i] = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
    return make_feasible(h_sym)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)

    tgt_pts = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_pts, src_pts, h_src)
    return make_feasible(h_int)
