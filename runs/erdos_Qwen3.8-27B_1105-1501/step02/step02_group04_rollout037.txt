from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h to have sum = target, maintaining [0,1] bounds."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_linear_matrix(h):
    """Build matrix A where L_k(d) = (A @ d)_k."""
    n = len(h)
    m = 2 * n - 1
    A = np.zeros((m, n))
    # L_k(d) = sum_j d[j] * [(1-h[j+k-(n-1)]) - h[j-(k-(n-1))]]
    # where h[out_of_range] = 0
    one_minus_h = 1.0 - h
    for k in range(m):
        offset = k - (n - 1)
        # term 1: sum_j d[j] * (1-h[j+offset])
        # For j from 0 to n-1, j+offset ranges from offset to n-1+offset
        # (1-h[j+offset]) is defined when 0 <= j+offset < n
        i1 = np.arange(n)
        j1 = i1 + offset
        mask1 = (j1 >= 0) & (j1 < n)
        col1 = np.zeros(n)
        col1[i1[mask1]] = one_minus_h[j1[mask1]]
        
        # term 2: sum_j d[j] * h[j-offset]
        j2 = i1 - offset
        mask2 = (j2 >= 0) & (j2 < n)
        col2 = np.zeros(n)
        col2[i1[mask2]] = h[j2[mask2]]
        
        A[k] = col1 - col2
    return A

def solve_lp_direction(h, corr, trust):
    """Solve LP to find descent direction d and expected decrease theta."""
    n = len(h)
    m = 2 * n - 1
    A = build_linear_matrix(h)
    M = np.max(corr)
    slack = M - corr  # non-negative, at least one zero
    
    # Variables: x = [d_0, ..., d_{n-1}, theta]
    # Minimize: -theta
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality: A @ d + theta <= slack

    A_ub = np.hstack([A, np.ones(m)])
    b_ub = slack.copy()
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-trust, trust)] * n + [(0, M)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, 0.0
    return res.x[:n], res.x[n]

def exact_line_eval(h, d, corr, L, Q, alphas):
    """Evaluate max_k [corr_k + α*L_k - α²*Q_k] for multiple alphas.
    Returns (best_alpha, best_value)."""
    best_val = np.max(corr)  # at alpha=0
    best_a = 0.0
    for a in alphas:
        vals = corr + a * L - a * a * Q
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_a = a
    return best_a, best_val

    c5_new = float(np.max(corr_new) * dx)

import numpy as np

from scipy.optimize import linprog


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_A(h):
    n = len(h)
    m = 2 * n - 1
    A = np.zeros((m, n))
    omh = 1.0 - h
    for k in range(m):
        off = k - (n - 1)
        # L_k(d) = sum_j d[j] * (omh[j+off] - h[j-off]) with zero extension
        j = np.arange(n)
        jp = j + off
        jm = j - off
        A[k, :] = np.where((jp >= 0) & (jp < n), omh[np.clip(jp, 0, n-1)], 0.0) - \
                  np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n-1)], 0.0)
    return A


def lp_direction(h, corr, trust):
    n = len(h)
    m = 2 * n - 1
    A = build_A(h)
    M = float(np.max(corr))
    slack = M - corr
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    A_ub = np.hstack([A, np.ones(m)])
    b_ub = slack
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-trust, trust)] * n + [(0.0, 2.0)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, 0.0
    return res.x[:n], float(res.x[n])


def line_search_best(h, d, corr, alpha_max, n_samples=50):
    n = len(h)
    m = 2 * n - 1
    if alpha_max <= 1e-12:
        return 0.0, float(np.max(corr))
    L = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
    Q = np.correlate(d, d, mode="full")
    alphas = np.linspace(0.0, alpha_max, n_samples)
    best_val = float(np.max(corr))
    best_a = 0.0
    for a in alphas:
        vals = corr + a * L - a * a * Q
        v = float(np.max(vals))
        if v < best_val:
            best_val = v
            best_a = a
    return best_a, best_val

    M = float(np.max(corr))
    d, theta = lp_direction(h, corr, trust)
    if d is None or theta < 1e-9:
        return h, M * dx
    # Find alpha_max
    alpha_max = 1.0
    for i in range(n):
        if d[i] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])

def pair_delta(h, i, j):
    n = len(h)
    hi_v, hj_v = h[i], h[j]

def pair_sweep(h, rng, max_pairs=None):
    n = len(h)
    if max_pairs is None:
        max_pairs = n * (n - 1) // 2
    perm = rng.permutation(n)
    count = 0
    for a in range(n):
        for b in range(a + 1, n):
            if count >= max_pairs:
                return
            i, j = perm[a], perm[b]
            d = pair_delta(h, i, j)
            if abs(d) > 1e-13:
                h[i] -= d
                h[j] += d
            count += 1

import numpy as np
import time
from scipy.optimize import linprog


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_A(h):
    n = len(h)
    m = 2 * n - 1
    A = np.zeros((m, n))
    omh = 1.0 - h
    for k in range(m):
        off = k - (n - 1)
        j = np.arange(n)
        jp = j + off
        jm = j - off
        t1 = np.where((jp >= 0) & (jp < n), omh[np.clip(jp, 0, n - 1)], 0.0)
        t2 = np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n - 1)], 0.0)
        A[k, :] = t1 - t2
    return A


def lp_direction(h, corr, trust):
    n = len(h)
    m = 2 * n - 1
    A = build_A(h)
    M = float(np.max(corr))
    slack = M - corr
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    A_ub = np.hstack([A, np.ones(m)])
    b_ub = slack
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-trust, trust)] * n + [(0.0, 5.0)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, 0.0
    return res.x[:n].copy(), float(res.x[n])


def line_search_best(h, d, corr, alpha_max):
    n = len(h)
    if alpha_max <= 1e-12:
        return 0.0, float(np.max(corr))
    L = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
    Q = np.correlate(d, d, mode="full")
    alphas = np.linspace(0.0, alpha_max, 40)
    best_val = float(np.max(corr))
    best_a = 0.0
    for a in alphas:
        v = float(np.max(corr + a * L - a * a * Q))
        if v < best_val:
            best_val = v
            best_a = a
    if best_a > 0.0:
        lo_a = max(0.0, best_a - alpha_max * 0.15)
        hi_a = min(alpha_max, best_a + alpha_max * 0.15)
        alphas_f = np.linspace(lo_a, hi_a, 25)
        for a in alphas_f:
            v = float(np.max(corr + a * L - a * a * Q))
            if v < best_val:
                best_val = v
                best_a = a
    return best_a, best_val


def do_lp_step(h, corr, trust):
    n = len(h)
    dx = 2.0 / n
    M = float(np.max(corr))
    d, theta = lp_direction(h, corr, trust)
    if d is None or theta < 1e-9:
        return h, M * dx, False
    alpha_max = 2.0
    for i in range(n):
        if d[i] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-12:
            alpha_max = min(alpha_max, -h[i] / d[i])
    alpha_max = max(alpha_max, 0.0)
    best_a, _ = line_search_best(h, d, corr, alpha_max)
    if best_a <= 1e-12:
        return h, M * dx, False
    h_new = h + best_a * d
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n / 2.0)
    corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
    c5_new = float(np.max(corr_new) * dx)
    if c5_new < M * dx - 1e-12:
        return h_new, c5_new, True
    return h, M * dx, False


def pair_delta(h, i, j):
    hi_v, hj_v = h[i], h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(18):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_sweep(h, rng, max_pairs=None):
    n = len(h)
    if max_pairs is None:
        max_pairs = n * (n - 1) // 2
    perm = rng.permutation(n)
    count = 0
    for a in range(n):
        for b in range(a + 1, n):
            if count >= max_pairs:
                return
            i, j = perm[a], perm[b]
            d = pair_delta(h, i, j)
            if abs(d) > 1e-13:
                h[i] -= d
                h[j] += d
            count += 1


def random_pairs(h, rng, n_pairs):
    n = len(h)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = pair_delta(h, i, j)
        if abs(d) > 1e-13:
            h[i] -= d
            h[j] += d


def transfer_h(h_from, n_to):
    n_from = len(h_from)
    xf = (np.arange(n_from) + 0.5) / n_from * 2.0
    xt = (np.arange(n_to) + 0.5) / n_to * 2.0
    h_to = np.interp(xt, xf, h_from, left=h_from[0], right=h_from[-1])
    h_to = np.clip(h_to, 0.0, 1.0)
    return project_sum(h_to, n_to / 2.0)


def make_binary_random(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_wave(n, rng):
    h = np.zeros(n)
    freq = 1.0 + rng.rand() * 3.0
    phase = rng.rand() * 2.0 * np.pi
    for i in range(n):
        h[i] = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * i / n + phase)
    h = project_sum(h, n / 2.0)
    return h


def local_optimize(h, rng, deadline, trust_init=0.05):
    n = len(h)
    c5_cur = compute_c5(h)
    trust = trust_init
    stall = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        h_new, c5_new, improved = do_lp_step(h, corr, trust)
        if improved:
            h = h_new
            c5_cur = c5_new
            stall = 0
            trust = min(trust * 1.04, 0.12)
        else:
            stall += 1
            if stall == 3:
                pair_sweep(h, rng, max_pairs=300)
                c5_cur = compute_c5(h)
                trust = max(trust * 0.75, 0.015)
                stall = 0
            elif stall >= 6:
                h = h + rng.randn(n) * (0.01 + 0.01 * (stall % 4))
                h = project_sum(h, n / 2.0)
                random_pairs(h, rng, 200)
                c5_cur = compute_c5(h)
                trust = max(trust * 0.7, 0.012)
                stall = 0
    return h, c5_cur
