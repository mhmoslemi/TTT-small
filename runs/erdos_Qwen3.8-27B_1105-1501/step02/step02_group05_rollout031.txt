                h[idx] -= sub

def lag_gradient(h, active_lags, weights):
    """Vectorized gradient of weighted sum of lag values"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)

        if s >= 0:
            m = n - s
            grad[:m] += wp * (1.0 - h[s:])
            grad[s:] -= wp * h[:m]
        else:
            t = -s
            m = n - t
            grad[t:] += wp * (1.0 - h[:m])
            grad[:m] -= wp * h[t:]
    return dx * grad

    n = len(h)

def official_score(h):

    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

        active = np.array([np.argmax(vals)])
    if len(active) > k:
        tv = vals[active]
        keep = np.argsort(tv)[-k:]
        active = active[keep]
    w = vals[active] - (cmax - tol) + 1e-8
    w = np.maximum(w, 0)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(len(active)) / len(active)
    else:
        w /= ws
    grad = np.zeros(n)
    for ii in range(len(active)):
        p = int(active[ii])
        wp = w[ii]
        s = p - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += wp * (1.0 - h[s:])
            grad[s:] -= wp * h[:m]
        else:
            t = -s
            m = n - t
            grad[t:] += wp * (1.0 - h[:m])
            grad[:m] -= wp * h[t:]
    return dx * grad, cmax

def ternary_full(h, a, b, n_iter=20):
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-12:
        return 0.0, official_score(h)
    ht = h.copy()
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        ht[a] = h[a] + m1
        ht[b] = h[b] - m1
        s1 = official_score(ht)
        ht[a] = h[a] + m2
        ht[b] = h[b] - m2
        s2 = official_score(ht)
        if s1 < s2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    ht[a] = h[a] + t
    ht[b] = h[b] - t
    ht[a] = np.clip(ht[a], 0, 1)
    ht[b] = np.clip(ht[b], 0, 1)
    return t, official_score(ht)

def build_ps(x):
    m = len(x)
    full = np.empty(2 * m)
    full[:m] = x
    full[m:] = 1.0 - x[::-1]
    return full

def score_ps(x):
    return official_score(build_ps(x))

def ps_grad(x, k=8, tol=0.015):
    full = build_ps(x)
    n = len(full)
    grad_full, cmax = full_grad(full, k, tol)
    m = len(x)
    grad_ps = grad_full[:m] - grad_full[m:][::-1]
    return grad_ps, cmax

def ternary_ps(x, i, j, n_iter=20):
    lo = max(-x[i], x[j] - 1.0)
    hi = min(1.0 - x[i], x[j])
    if hi - lo < 1e-12:
        return 0.0, score_ps(x)
    xt = x.copy()
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        xt[i] = x[i] + m1
        xt[j] = x[j] - m1
        s1 = score_ps(xt)
        xt[i] = x[i] + m2
        xt[j] = x[j] - m2
        s2 = score_ps(xt)
        if s1 < s2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    xt[i] = x[i] + t
    xt[j] = x[j] - t
    xt[i] = np.clip(xt[i], 0, 1)
    xt[j] = np.clip(xt[j], 0, 1)
    return t, score_ps(xt)

def full_space_search(h0, is_dead, max_iter=50000, seed=42):
    h = h0.copy()
    best_sc = official_score(h)
    n = len(h)
    rng = np.random.default_rng(seed)
    stagnant = 0
    tol_sched = [0.02, 0.015, 0.01, 0.005, 0.003]
    tol_idx = 0
    k_sched = [10, 8, 6, 5, 4]
    
    for it in range(max_iter):
        if is_dead():
            break
        if it % 5000 == 4999 and tol_idx < len(tol_sched) - 1:
            tol_idx += 1
        
        tol = tol_sched[tol_idx]
        k = k_sched[tol_idx]
        grad, cmax = full_grad(h, k, tol)
        
        source = int(np.argmax(grad))
        target = int(np.argmin(grad))
        improved = False
        
        if source != target:
            t, s_new = ternary_full(h, target, source, 22)
            if s_new < best_sc - 1e-15:
                h[target] = np.clip(h[target] + t, 0, 1)
                h[source] = np.clip(h[source] - t, 0, 1)
                h = make_feasible(h)
                best_sc = official_score(h)
                improved = True
                stagnant = 0
        
        # Try top few gradient pairs
        idx_sorted = np.argsort(grad)
        n_try = min(4, n)
        for trial in range(n_try):
            if is_dead():
                break
            ci = int(idx_sorted[trial])
            cj = int(idx_sorted[n - 1 - trial])
            if ci == cj:
                continue
            t, s_new = ternary_full(h, ci, cj, 16)
            if s_new < best_sc - 1e-15:
                h[ci] = np.clip(h[ci] + t, 0, 1)
                h[cj] = np.clip(h[cj] - t, 0, 1)
                h = make_feasible(h)
                best_sc = official_score(h)
                improved = True
                stagnant = 0
        
        # Random pairs when stagnant
        if stagnant > 20 or it % 50 == 0:
            nrand = 3 if stagnant > 100 else 2
            for _ in range(nrand):
                if is_dead():
                    break
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a == b:
                    continue
                t, s_new = ternary_full(h, a, b, 14)
                if s_new < best_sc - 1e-15:
                    h[a] = np.clip(h[a] + t, 0, 1)
                    h[b] = np.clip(h[b] - t, 0, 1)
                    h = make_feasible(h)
                    best_sc = official_score(h)
                    improved = True
                    stagnant = 0
        
        if not improved:
            stagnant += 1
        
        if stagnant > 500:
            # Perturbed restart
            sigma = min(0.15, 0.01 * stagnant / 100)
            h_pert = make_feasible(h + rng.normal(0, sigma, n))
            s_pert = official_score(h_pert)
            if s_pert < best_sc:
                h = h_pert
                best_sc = s_pert
            stagnant = 0
    
    return h, best_sc

def slsqp_smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def slsqp_smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w /= ws
    grad = np.zeros(n)
    for p in range(len(w)):
        if w[p] < 1e-10:
            continue
        s = p - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += w[p] * (1.0 - x[s:])
            grad[s:] -= w[p] * x[:m]
        else:
            t = -s
            m = n - t
            grad[t:] += w[p] * (1.0 - x[:m])
            grad[:m] -= w[p] * x[t:]
    return dx * grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def try_slsqp(h0, beta, maxiter=60):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(slsqp_smooth_obj, h0, args=(beta,), jac=slsqp_smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(12.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve
    
    def is_dead():
        return time.time() >= deadline
    
    best_h = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(initial_h_values, dtype=float).ravel()
        if p.size > 2 and np.all(np.isfinite(p)):
            best_h = make_feasible(p)
            best_score = official_score(best_h)
    except (NameError, Exception):
        pass
    
    if best_h is None:
        best_h = np.full(60, 0.5)
        best_score = official_score(best_h)
    
    n = len(best_h)
    
    # Phase 1: Quick SLSQP warmup
    if not is_dead():
        for beta in [50, 200, 600]:
            if is_dead():
                break
            h_new, sc = try_slsqp(best_h, beta, maxiter=50)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
    
    # Phase 2: Full-space gradient-guided search (main workhorse)
    if not is_dead():
        budget_phase2 = min(300, max(60, (deadline - time.time()) * 0.35))
        iters2 = int(budget_phase2 * 500)
        h2, s2 = full_space_search(best_h, is_dead, max_iter=iters2, seed=seed)
        if s2 < best_score:
            best_h, best_score = h2, s2
    
    # Phase 3: Anti-symmetric search
    if not is_dead():
        m = n // 2
        x0 = (best_h[:m] - best_h[n-1:n-m-1:-1] + 1.0) / 2.0
        x0 = np.clip(x0, 0, 1)
        budget_phase3 = min(150, max(30, (deadline - time.time()) * 0.2))
        iters3 = int(budget_phase3 * 500)
        x1, s1 = anti_sym_search(x0, is_dead, max_iter=iters3, seed=seed + 1)
        if s1 < best_score - 1e-14:
            best_h = build_ps(x1)
            best_score = s1
        # Also try SLSQP on anti-symmetric
        if not is_dead():
            for beta in [100, 400]:
                if is_dead():
                    break
                h_as = build_ps(x1)
                h_new, sc = try_slsqp(h_as, beta, maxiter=40)
                if h_new is not None and sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc
    
    # Phase 4: Multi-resolution exploration
    if not is_dead():
        n_opts = []
        for dn in [-6, -4, -2, 2, 4, 6, 8]:
            nt = n + dn
            if 36 <= nt <= 100:
                n_opts.append(nt)
        for nc in [44, 52, 64, 72, 80]:
            if nc not in n_opts and 36 <= nc <= 100:
                n_opts.append(nc)
        n_opts = sorted(set(n_opts))
        for n_opt in n_opts:
            if is_dead():
                break
            if n_opt == n:
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-14:
                best_score = sc_int
            iters4 = min(20000, int(max(30, (deadline - time.time()) * 0.1) * 500))
            if iters4 < 1000:
                continue
            h4, s4 = full_space_search(h_int, is_dead, max_iter=iters4, seed=seed + n_opt)
            if s4 < best_score - 1e-14:
                best_h = h4
                best_score = s4
    
    # Phase 5: Final intensive polish on best
    if not is_dead():
        iters5 = int(max(50, (deadline - time.time()) - 20) * 500)
        h5, s5 = full_space_search(best_h, is_dead, max_iter=iters5, seed=seed + 999)
        if s5 < best_score:
            best_h, best_score = h5, s2 if False else s5
    
    # Phase 6: One more anti-symmetric pass on final best
    if not is_dead():
        m = len(best_h) // 2
        x0 = (best_h[:m] - best_h[m*2-1:m-1:-1] + 1.0) / 2.0
        x0 = np.clip(x0, 0, 1)
        iters6 = int(max(20, (deadline - time.time()) - 10) * 500)
        if iters6 > 500:
            x6, s6 = anti_sym_search(x0, is_dead, max_iter=iters6, seed=seed + 7777)
            if s6 < best_score - 1e-14:
                best_h = build_ps(x6)
                best_score = s6
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    
    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8
    
    return final_h, final_score, final_n
