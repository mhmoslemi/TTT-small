import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -4.0 - np.max(z), 4.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p):
    return make_feasible(h ** p)

    n = len(h)
    h_work = h.copy()
    best_score = exact_score(h_work)
    
    for w in range(4, max_w + 1, 2):
        improved = True
        while improved:
            improved = False
            if time_limit is not None and t_start is not None:
                if time.time() - t_start > time_limit:
                    return h_work, best_score
            for s in range(0, n - w + 1):
                block = h_work[s:s+w]
                c_ones = int(round(np.sum(block)))
                if c_ones < 1 or c_ones > w - 1:
                    continue
                # Generate all positions for ones
                best_block = block.copy()
                best_local = best_score
                positions = np.arange(w)
                zero_positions = positions[block < 0.5]
                one_positions = positions[block >= 0.5]
                if len(one_positions) != c_ones:
                    continue
                # Enumerate all C(w, c_ones) assignments
                for new_ones in combinations(range(w), c_ones):
                    # Quick check: skip if same
                    new_set = set(new_ones)
                    old_set = set(one_positions.tolist())
                    if new_set == old_set:

                    h_work[s + np.array(new_ones)] = 1.0

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        s2 = k - (n-1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n-s2])
                grad[:n-s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n-m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n-m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0, args=(beta, dx),
            jac=lambda x: smooth_lse_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

                i = int(rng.integers(len(ones)))
                j = int(rng.integers(len(zeros)))
                pert[ones[i]] = 0.0
                pert[zeros[j]] = 1.0
            
            bh, bs = block_search(pert, max_w=10, time_limit=15, t_start=time.time())
            if bs < best_score - 1e-12:
                best_h = bh
                best_score = bs
        
        # Continuous polish on best
        if time.time() < deadline - 5:
            for beta in [100, 300]:
                if time.time() > deadline - 5:
                    break
                sl_h = slsqp_smooth(best_h, beta, maxiter=100)
                sl_score = exact_score(sl_h)
                if sl_score < best_score - 1e-12:
                    best_h = sl_h
                    best_score = sl_score
        
        # One final block search
        if time.time() < deadline - 3:
            bh, bs = block_search(to_binary(best_h), max_w=12, time_limit=10, t_start=time.time())
            if bs < best_score - 1e-12:
                best_h = bh
                best_score = bs
    
    best_h = make_feasible(best_h)
    final_score = exact_score(best_h)
    
    return (best_h, final_score, len(best_h))
