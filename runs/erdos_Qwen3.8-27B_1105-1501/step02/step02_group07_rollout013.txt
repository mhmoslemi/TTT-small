def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.0, posinf=5.0, neginf=-5.0)
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        smid = np.sum(np.clip(z + mid, 0.0, 1.0))
        if smid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w[0:n] * h[0:L]
    return grad

def pg_step(h, beta, alpha_max=0.8, max_ls=14):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_shifted = np.clip(beta * (V - V.max()), -500.0, 0.0)
    w = np.exp(V_shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    grad = compute_gradient(h, w)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-7:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.15):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return h, V_max
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return h, V_max
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),

                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc

def dct_search(h_init, n, d, pop_size, t_limit, rng):

        if sc < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc
        h_cur = h_new
        h_new2, sc2 = active_set_lp(h_cur, eps=0.001, trust=0.05)
        if sc2 < best_score - 1e-14:
            best_h = h_new2.copy()
            best_score = sc2
        h_cur = h_new2

    # Phase 7: Multi-resolution check
    if time_left() > 40:
        for n_try in [n - 4, n - 2, n + 2, n + 4, 50, 55, 65, 70]:
            if time_left() < 30:
                break
            if n_try < 20:
                continue
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int.copy()
                best_score = sc_int
            # Quick polish at this resolution
            h_cur2 = h_int.copy()
            for _ in range(50):
                if time_left() < 30:
                    break
                h_new3, sc3 = pairwise_exchange_v2(h_cur2, n_grid=5, n_passes=1)
                if sc3 < best_score - 1e-14:
                    best_h = h_new3.copy()
                    best_score = sc3
                h_cur2 = h_new3

    # Phase 8: Final polish
    if time_left() > 15:
        h_cur = best_h.copy()
        h_new, sc = pairwise_exchange_v2(h_cur, n_grid=10, n_passes=2)
        if sc < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc
        h_new2, sc2 = active_set_lp(h_cur, eps=0.0005, trust=0.03)
        if sc2 < best_score - 1e-14:
            best_h = h_new2.copy()
            best_score = sc2

    # Final validation
    final_h = make_feasible(best_h)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_score = official_score(final_h)
    final_n = len(final_h)

    return final_h, final_score, final_n
