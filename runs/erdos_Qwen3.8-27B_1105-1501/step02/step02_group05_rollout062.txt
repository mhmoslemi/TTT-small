def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Greedy repair
    residual = target - np.sum(h)
    if residual > 1e-13:
        order = np.argsort(1.0 - h)
        for idx in order:
            if residual < 1e-13:
                break
            add = min(residual, 1.0 - h[idx])
            if add > 1e-14:
                h[idx] += add
                residual -= add
    elif residual < -1e-13:
        order = np.argsort(h)
        for idx in order:
            if residual > -1e-13:
                break
            sub = min(-residual, h[idx])
            if sub > 1e-14:
                h[idx] -= sub
                residual += sub
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(h, s):
    """Compute gradient of lag s (shift in [-n+1, n-1]) w.r.t. h, including dx factor."""
    n = len(h)
    dx = 2.0 / n

    if s >= 0:
        m = n - s
        # For j in [0, m-1]: contribution from (1-h[j]) term: -dx * h[j+s]
        g[:m] += -dx * h[s:s+m]  # h[j+s] for j in 0..m-1
        # For j in [s, n-1]: contribution from h[j] term: dx * (1-h[j-s])

        # For j in [t, n-1]: contribution from (1-h[j]) term in h[j-t]*(1-h[j]): dx * h[j-t]

            continue
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04]:
            trial = make_feasible(best + alpha * u)
            sc = score_exact(trial)
            if sc < best_sc - 1e-12:
                best = trial
                best_sc = sc
                improved = True
                stall = 0
                G, V = all_lags_grad(best, use_fd=use_fd)
                break
        if not improved:
            stall += 1
            rho *= 0.6
            if rho < 0.003:
                break
    return best, best_sc

def binary_anneal(h_init, n_steps, rng, is_dead, T_start=5e-4):
    n = len(h_init)
    if n % 2 != 0 or n < 10:
        return None, None
    target_ones = n // 2
    b = (h_init >= 0.5).astype(float)
    n_ones = int(b.sum())
    if n_ones > target_ones:
        one_idx = np.where(b == 1)[0]
        remove = rng.choice(n_ones - target_ones, replace=False)
        b[one_idx[remove]] = 0
    elif n_ones < target_ones:
        zero_idx = np.where(b == 0)[0]
        need = target_ones - n_ones
        add = rng.choice(zero_idx.size, size=need, replace=False)
        b[zero_idx[add]] = 1
    cur_sc = score_exact(b)
    best_b = b.copy()
    best_sc = cur_sc
    T = T_start
    for step in range(n_steps):
        if is_dead():
            break
        one_idx = np.where(b == 1)[0]
        zero_idx = np.where(b == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        i = one_idx[rng.integers(len(one_idx))]
        j = zero_idx[rng.integers(len(zero_idx))]
        if i == j:
            continue
        b[i] = 0
        b[j] = 1
        new_sc = score_exact(b)
        delta = new_sc - cur_sc
        if delta < -1e-10 or (delta < 0.05 and rng.random() < np.exp(-delta / max(T, 1e-8))):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
        T *= 0.9995
    return best_b, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def make_wave(n, freq1, amp1, freq2, amp2):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def two_point_polish(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        cands = [t_lo + f * span for f in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]]
        tmp = best.copy()
        best_t = None
        best_local = best_sc
        for t in cands:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-12 or va > 1.0 + 1e-12 or vb < -1e-12 or vb > 1.0 + 1e-12:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local - 1e-13:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = best[a] + best_t
            best[b] = best[b] - best_t
            best[a] = np.clip(best[a], 0.0, 1.0)
            best[b] = np.clip(best[b], 0.0, 1.0)
            best_sc = score_exact(best)
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.07 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.1)
        best_score = score_exact(best_h)

    # Phase 1: LP refinement (main)
    if not is_dead():
        n_base = len(best_h)
        use_all = n_base <= 70
        for pass_i in range(6):
            if is_dead():
                break
            rho_val = 0.18 * (0.75 ** pass_i)
            k_val = 20 if not use_all else 2 * n_base - 1
            best_h, best_score = lp_refine(best_h, 250, is_dead, K=k_val,
                                           rho_start=rho_val, use_all_lags=use_all)

    # Phase 2: Binary annealing
    if not is_dead():
        n_cur = len(best_h)
        for n_bin in [n_cur] if n_cur % 2 == 0 else [n_cur + 1, n_cur - 1]:
            if is_dead() or n_bin < 10 or n_bin % 2 != 0:
                continue
            for T_try in [2e-4, 5e-4, 1e-3]:
                if is_dead():
                    break
                h_b, sc_b = binary_anneal(best_h, 4000, rng, is_dead, T_start=T_try)
                if h_b is not None:
                    if sc_b < best_score - 1e-10:
                        best_h = make_feasible(h_b)
                        best_score = score_exact(best_h)
                    if not is_dead():
                        h_c = make_feasible(h_b * 1.0)
                        h_new, sc_new = lp_refine(h_c, 120, is_dead, K=30,
                                                  rho_start=0.12, use_all_lags=(n_bin <= 70))
                        if sc_new < best_score - 1e-10:
                            best_h = h_new
                            best_score = sc_new
                    if not is_dead():
                        h_as = anti_sym_map(h_b * 1.0)
                        sc_as = score_exact(h_as)
                        if sc_as < best_score - 1e-10:
                            best_h = h_as
                            best_score = sc_as
                        h_new, sc_new = lp_refine(h_as, 100, is_dead, K=30,
                                                  rho_start=0.1, use_all_lags=(n_bin <= 70))
                        if sc_new < best_score - 1e-10:
                            best_h = h_new
                            best_score = sc_new

    # Phase 3: Resolution scanning
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set([max(24, n_base + d) for d in [-8, -4, 4, 8, 12]] + [40, 44, 48, 52, 56, 64, 72, 80]))
        n_opts = [x for x in n_opts if 24 <= x <= 100 and x != len(best_h)]
        for n_opt in n_opts:
            if is_dead():
                break
            h_int = interp_to(best_h, n_opt)
            h_new, sc_new = lp_refine(h_int, 180, is_dead, K=30,
                                      rho_start=0.14, use_all_lags=(n_opt <= 70))
            if sc_new < best_score - 1e-10:
                best_h = h_new
                best_score = sc_new

    # Phase 4: Fine polish
    if not is_dead():
        for rho in [0.06, 0.03, 0.015, 0.008]:
            if is_dead():
                break
            best_h, best_score = lp_refine(best_h, 200, is_dead, K=40,
                                           rho_start=rho, use_all_lags=(len(best_h) <= 70))

    # Phase 5: Two-point polish for final nudge
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 4))
        best_h, best_score = two_point_polish(best_h, n_tries, rng, is_dead)

    # Phase 6: Anti-symmetric + final LP
    if not is_dead():
        h_as = anti_sym_map(best_h)
        sc_as = score_exact(h_as)
        if sc_as < best_score - 1e-10:
            best_h = h_as
            best_score = sc_as
        best_h, best_score = lp_refine(best_h, 150, is_dead, K=40,
                                       rho_start=0.05, use_all_lags=(len(best_h) <= 70))
        if not is_dead():
            best_h, best_score = two_point_polish(best_h, 1500, rng, is_dead)

    # Final
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    if (not np.all(np.isfinite(final_h)) or final_h.min() < -1e-8 or
            final_h.max() > 1.0 + 1e-8 or not np.isfinite(final_score)):
        final_h = make_feasible(np.full(60, 0.5))
        final_score = score_exact(final_h)
        final_n = 60

    return final_h, final_score, final_n
