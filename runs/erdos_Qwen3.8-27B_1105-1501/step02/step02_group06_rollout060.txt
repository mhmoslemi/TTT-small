```python
def pair_optimize(h, i, j, iters=30):
    """Optimize the pair (h[i], h[j]) keeping their sum constant."""
    s = h[i] + h[j]  # constant
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if lo >= hi:
        return h
    
    def g(t):
        h[i] = t
        h[j] = s - t
        overlaps = np.correlate(h, 1.0-h, mode='full')
        return float(overlaps.max())
    
    # Golden section search on [lo, hi]
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = g(c)
    fd = g(d)
    for _ in range(iters):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a); fc = g(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a); fd = g(d)
    
    t_opt = (a + b) / 2
    h[i] = t_opt
    h[j] = s - t_opt
    return h
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=np.float64), -3.0, 3.0)
    lo, hi = -3.0, 3.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-12:
        if r > 0:
            for idx in np.argsort(1.0 - h):
                a = min(r, 1.0 - h[idx])
                h[idx] += a
                r -= a
                if abs(r) < 1e-15:

        else:

                s = min(-r, h[idx])
                h[idx] -= s
                r += s
                if abs(r) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)
```
