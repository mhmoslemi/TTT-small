Wait, I need to be more careful with the gradient. Let me re-derive.

The full correlation: np.correlate(h, 1-h, mode='full')
For arrays a, b of length n, the full correlation has length 2n-1.
The k-th element (0-indexed) of the full correlation is:
  c[k] = sum_i a[i] * b[i + k - (n-1)]
where the sum is over valid i such that both indices are in [0, n-1].

So c[k] = sum_i h[i] * (1 - h[i + k - (n-1)])

Let s = k - (n-1). Then:
c[k] = sum_i h[i] * (1 - h[i + s])

Valid i: 0 <= i < n and 0 <= i+s < n, i.e., max(0, -s) <= i < min(n, n-s).

For s >= 0: i ranges from 0 to n-s-1
For s < 0: i ranges from -s to n-1

Gradient of c[k] with respect to h[j]:
For s = 0: c[k] = sum_i h[i](1-h[i]), dc/dh[j] = 1 - 2h[j]
For s > 0: c[k] = sum_{i=0}^{n-s-1} h[i](1-h[i+s])
  dc/dh[j] = (1-h[j+s]) if j < n-s (j appears as first factor)
           + (-h[j-s]) if j >= s (j appears as second factor, i = j-s)
  So: dc/dh[j] = (1-h[j+s])*[j < n-s] + (-h[j-s])*[j >= s]

For s < 0, let t = -s > 0: c[k] = sum_{i=t}^{n-1} h[i](1-h[i-t])
  dc/dh[j] = (1-h[j-t])*[j >= t] + (-h[j+t])*[j < n-t]

dc/db = -a + (1-c) ✓ (j=1, j < 2 so (1-h[2])=1-c; j >= s=1 so -h[1-1]=-h[0]=-a)
dc/dc = -b ✓ (j=2, j < 2 is false; j >= 1 so -h[2-1]=-h[1]=-b)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_all_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_p(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        idx1 = np.arange(n - s)
        idx2 = np.arange(s, n)
        g[idx1] += (1.0 - h[idx1 + s])
        g[idx2] += (-h[idx2 - s])
    else:
        t = -s
        idx1 = np.arange(t, n)
        idx2 = np.arange(0, n - t)
        g[idx1] += (1.0 - h[idx1 - t])
        g[idx2] += (-h[idx2 + t])
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def solve_lp_subproblem(h0, selected_p, delta):
    """Solve trust-region LP for min-max linearization."""
    n = len(h0)
    f0 = compute_all_f(h0)
    m = len(selected_p)
    
    # Build LP: minimize t s.t. g_p · h - t <= g_p · h0 - f0[p]
    # Variables: [h (n), t (1)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, p in enumerate(selected_p):
        s = p - (n - 1)
        g = gradient_p(h0, s)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g @ h0 - f0[p]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lb = np.maximum(0.0, h0 - delta)
    ub = np.minimum(1.0, h0 + delta)
    bounds = list(zip(lb, ub)) + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_step(h0, k_active=10, delta=0.1):
    """One LP trust-region step."""
    n = len(h0)
    f0 = compute_all_f(h0)
    top_idx = np.argsort(f0)[-k_active:]
    
    h_lp = solve_lp_subproblem(h0, top_idx, delta)
    if h_lp is None:
        return None
    
    h_new = make_feasible(h_lp)
    return h_new

def official_score(h):
    n = len(h)

def compute_all_f(h):

def gradient_p(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        idx1 = np.arange(n - s)
        idx2 = np.arange(s, n)
        g[idx1] += (1.0 - h[idx1 + s])
        g[idx2] += (-h[idx2 - s])
    else:
        t = -s

        idx2 = np.arange(0, n - t)
        g[idx1] += (1.0 - h[idx1 - t])
        g[idx2] += (-h[idx2 + t])
    return g

                break
            sigma = 0.02 + 0.1 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = lp_refine(h_cand, k_active=12, delta_start=0.08,
                                      max_iter=12, time_limit=remaining() - 6)
            update(h_opt, sc_opt)

    # Phase 6: Final intensive polish with very small steps
    if remaining() > 30:
        for k_a in [6, 10, 15]:
            if remaining() < 10:
                break
            h_opt, sc_opt = lp_refine(best_h, k_active=k_a, delta_start=0.04,
                                      max_iter=30, time_limit=remaining() - 5)
            update(h_opt, sc_opt)

    # Phase 7: Combo - line search then LP
    if remaining() > 40:
        h_cur = best_h.copy()
        n = len(h_cur)
        x = (np.arange(n) + 0.5) * 2.0 / x[0] if n > 0 else None
        x = (np.arange(n) + 0.5) * 2.0 / n
        for freq in [2, 3, 5]:
            d = np.sin(2 * np.pi * freq * x / 2.0) * 0.1
            hs, ssc = line_search(h_cur, d, n_coarse=30, max_refine=3)
            if hs is not None:
                h_near = hs
            else:
                h_near = h_cur
            h_opt, sc_opt = lp_refine(h_near, k_active=15, delta_start=0.06,
                                      max_iter=15, time_limit=remaining() - 5)
            update(h_opt, sc_opt)

    # Final: try a few more resolutions from the best
    if remaining() > 20:
        for dn in [-4, -2, 2, 4]:
            if remaining() < 8:
                break
            nn = len(best_h) + dn
            if nn >= 30 and nn % 2 == 0:
                h_int = interpolate_to(best_h, nn)
                sc_int = official_score(h_int)
                update(h_int, sc_int)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = final_score + 1e-13
    return final_h, final_bound, len(final_h)
