import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    residual = target - np.sum(h)
    if abs(residual) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        order = np.argsort(1.0 - h) if residual > 0 else np.argsort(h)
        for idx in order:
            if abs(residual) < 1e-15:
                break
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
    return np.clip(h, 0.0, 1.0)

def build_sym_h(y, n):
    m = len(y)
    if n == 2 * m:
        return np.concatenate([y, 1.0 - y[::-1]])
    elif n == 2 * m + 1:
        return np.concatenate([y, [0.5], 1.0 - y[::-1]])
    else:
        raise ValueError("Incompatible sizes: m={}, n={}".format(m, n))

def grad_full(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] -= dx * h[s:]
            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K]
    return G

def sym_grad(y, n):
    m = len(y)
    h = build_sym_h(y, n)
    G = grad_full(h)
    if n == 2 * m:
        G_red = G[:, :m] - G[:, m:2*m][::-1, :]
    else:
        G_red = G[:, :m] - G[:, m+1:2*m+1][::-1, :]
    return G_red, h

def sym_vals(y, n):
    h = build_sym_h(y, n)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def parent_to_sym_y(h_parent, n):
    if len(h_parent) != n:
        n_src = len(h_parent)
        src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
        tgt_c = (np.arange(n) + 0.5) * (2.0 / n)
        h_int = np.interp(tgt_c, src_c, h_parent)
    else:
        h_int = h_parent.copy()
    h_sym = 0.5 * (h_int + (1.0 - h_int[::-1]))
    m = n // 2
    y = h_sym[:m].copy()
    return np.clip(y, 0.0, 1.0)

def solve_sym_nlp_full(y0, n, t_init, max_iter=300, time_limit=30.0):
    m = len(y0)

    def objective(x):
        return x[-1]

    def objective_grad(x):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g

    def constraint_fun(x):
        y = x[:m]
        t = x[-1]
        vals = sym_vals(np.clip(y, 0.0, 1.0), n)
        return t - vals

    def constraint_jac(x):
        y = np.clip(x[:m], 0.0, 1.0)
        G_red, _ = sym_grad(y, n)
        J = -G_red
        J_full = np.hstack([J, np.ones((2*n-1, 1))])
        return J_full

    x0 = np.concatenate([y0, [t_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}]

    t0 = time.time()
    try:
        res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            y_new = np.clip(res.x[:m], 0.0, 1.0)
            h_new = build_sym_h(y_new, n)
            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def smooth_pg_sym(y0, n, beta, max_iter=500, time_limit=20.0):
    y = y0.copy()
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        vals = sym_vals(y, n)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(vals)) / len(vals)
        else:
            w /= ws
        obj = cmax + np.log(ws) / beta
        G_red, _ = sym_grad(y, n)
        grad_y = w @ G_red
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-15:
            break
        alpha = 0.5 / (gnorm + 1e-10)
        for _ in range(40):
            y_trial = np.clip(y - alpha * grad_y, 0.0, 1.0)
            vals_trial = sym_vals(y_trial, n)
            cmax_t = vals_trial.max()
            shifted_t = beta * (vals_trial - cmax_t) - shifted_t.max() if beta > 0 else np.zeros_like(shifted)
            shifted_t = beta * (vals_trial - cmax_t)
            shifted_t -= shifted_t.max()
            w_t = np.exp(shifted_t)
            ws_t = w_t.sum()
            if ws_t < 1e-300:
                f_trial = cmax_t
            else:
                f_trial = cmax_t + np.log(ws_t) / beta
            if f_trial <= obj - 1e-7 * alpha * gnorm:
                break
            alpha *= 0.4
        if alpha < 1e-15:
            break
        y = np.clip(y - alpha * grad_y, 0.0, 1.0)
    return y
