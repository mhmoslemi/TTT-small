    #   -delta <= d[j] <= delta

        A_box[4*j+3, j] = -1.0

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    npts = 2 * n - 1
    grad = np.zeros((npts, n))
    for p in range(npts):
        k = p - (n - 1)
        if k >= 0:
            for j in range(n - k):
                grad[p, j + k] += (1.0 - h[j])
                grad[p, j] -= h[j + k]
        else:
            t = -k
            for j in range(t, n):
                grad[p, j - t] += (1.0 - h[j])
                grad[p, j] -= h[j - t]
    return grad * dx

def lp_minimax_step(h, delta):
    n = len(h)
    npts = 2 * n - 1
    vals = compute_lag_values(h)
    grad = compute_lag_gradients(h)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_lag = np.hstack([grad, -np.ones((npts, 1))])
    b_lag = -vals

    A_box = np.zeros((4 * n, n + 1))
    b_box = np.zeros(4 * n)
    for j in range(n):
        A_box[4 * j, j] = 1.0
        b_box[4 * j] = 1.0 - h[j]
        A_box[4 * j + 1, j] = -1.0
        b_box[4 * j + 1] = h[j]
        A_box[4 * j + 2, j] = 1.0
        b_box[4 * j + 2] = delta
        A_box[4 * j + 3, j] = -1.0
        b_box[4 * j + 3] = delta

    A_ub = np.vstack([A_lag, A_box])
    b_ub = np.concatenate([b_lag, b_box])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * (n + 1), method='highs')
    except Exception:
        return None, None

    if res.status != 0:
        return None, None

    d = res.x[:n]
    t_new = res.x[n]
    return d, t_new

def two_point_search(h, n_tries, rng):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            tmp[a] = val_a
            tmp[b] = val_b
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0.0, 1.0)
            best[b] = np.clip(best[b] - best_t, 0.0, 1.0)
            best_sc = official_score(best)
    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_optimize(h, is_dead, max_iters=500, delta_init=0.1):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    delta = delta_init
    no_improve = 0
    for _ in range(max_iters):
        if is_dead():
            break
        d, t_pred = lp_minimax_step(best, delta)
        if d is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        # Line search
        best_alpha = 0.0
        best_trial = best_sc
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            ht = make_feasible(best + alpha * d)
            sc = official_score(ht)
            if sc < best_trial - 1e-14:
                best_trial = sc
                best_alpha = alpha
        if best_alpha > 0:
            best = make_feasible(best + best_alpha * d)
            best_sc = best_trial
            no_improve = 0
            delta = max(0.01, delta * 0.92)
        else:
            no_improve += 1
            delta *= 0.6
            if delta < 0.005:
                break
    return best, best_sc
