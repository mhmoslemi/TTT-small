def corr_grad_mat(h):
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for p in range(K):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] += dx * (1.0 - h[s:s+L])
            G[p, s:] -= dx * h[:L]
        else:
            a = -s
            L = n - a
            G[p, a:] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[a:a+L]
    return G

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            best_h = parent.copy()
            best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        x = np.arange(n0)
        h_init = 0.5 + 0.3 * np.sin(np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)
    update(best_h, best_score)

    # Phase 1: Symmetric L-BFGS-B with progressive beta
    if time_left() > 40:
        h_s = symmetrize_h(best_h)
        sc_s = official_score(h_s)
        if sc_s < best_score - 1e-14:
            update(h_s, sc_s)
        n = len(h_s)
        if n % 2 == 0:
            m = n // 2
            x0 = np.clip(h_s[:m], 0.005, 0.995)
            for beta in [30, 100, 300, 1000, 3000, 10000, 30000, 100000]:
                if check():
                    break
                tl = min(time_left() * 0.12, 18.0)
                if tl < 3:
                    break
                try:
                    res = minimize(sym_obj_fun, x0, jac=sym_obj_grad,
                                  args=(beta,), method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * m,
                                  options={'maxiter': 800, 'ftol': 1e-19,
                                           'gtol': 1e-14, 'maxfun': 3000})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_new = h_from_half(np.clip(res.x, 0, 1))
                        sc = official_score(h_new)
                        if sc < best_score - 1e-14:
                            update(h_new, sc)
                            h_s = symmetrize_h(h_new)
                            m = len(h_s) // 2
                            x0 = np.clip(h_s[:m], 0.001, 0.999)
                except Exception:
                    pass
        # Re-symmetrize and continue
        h_s = symmetrize_h(best_h)
        sc_s = official_score(h_s)
        if sc_s < best_score - 1e-14:
            update(h_s, sc_s)

    # Phase 2: Active set minimax (symmetric)
    if time_left() > 30:
        h_s = symmetrize_h(best_h)
        n = len(h_s)
        if n % 2 == 0:
            m = n // 2
            x_s = np.clip(h_s[:m], 0.005, 0.995)
            vals = corr_vals(h_s)
            for rnd in range(5):
                if check():
                    break
                improved = False
                for rel_tol in [0.003, 0.005, 0.008, 0.012]:
                    if check():
                        break
                    active = find_active(vals, rel_tol=rel_tol, min_count=6 + rnd*2)
                    h_new, sc = active_minimax_sym(x_s, active, max_iter=250 + rnd*50)
                    if h_new is not None and sc < best_score - 1e-14:
                        update(h_new, sc)
                        h_s = symmetrize_h(h_new)
                        m = len(h_s) // 2
                        x_s = np.clip(h_s[:m], 0.005, 0.995)
                        vals = corr_vals(h_s)
                        improved = True
                        break
                if not improved:
                    break

    # Phase 3: Multi-resolution optimization
    if time_left() > 80:
        n_base = len(best_h)
        n_opts = sorted(set([n for n in [n_base-4, n_base-2, n_base, n_base+2, n_base+4, n_base+6]
                            if 40 <= n <= 96 and n % 2 == 0]))
        for n_opt in n_opts:
            if check():
                break
            h_int = interpolate_h(best_h, n_opt)
            sc_int = official_score(h_int)
            h_s = symmetrize_h(h_int)
            sc_s = official_score(h_s)
            x0 = np.clip(h_s[:n_opt//2], 0.005, 0.995)
            for beta in [500, 3000, 15000]:
                if check():
                    break
                tl = min(time_left() * 0.08, 12.0)
                if tl < 3:
                    break
                try:
                    res = minimize(sym_obj_fun, x0, jac=sym_obj_grad,
                                  args=(beta,), method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * (n_opt//2),
                                  options={'maxiter': 500, 'ftol': 1e-19, 'maxfun': 2000})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_new = h_from_half(np.clip(res.x, 0, 1))
                        sc = official_score(h_new)
                        if sc < best_score - 1e-14:
                            update(h_new, sc)
                            x0 = np.clip(h_new[:n_opt//2], 0.001, 0.999)
                except Exception:
                    pass
            # Active set at this resolution
            if check():
                break
            h_s = symmetrize_h(best_h)
            vals = corr_vals(h_s)
            if len(h_s) % 2 == 0:
                m2 = len(h_s) // 2
                x_s = np.clip(h_s[:m2], 0.005, 0.995)
                active = find_active(vals, rel_tol=0.005, min_count=8)
                h_new, sc = active_minimax_sym(x_s, active, max_iter=200)
                if h_new is not None and sc < best_score - 1e-14:
                    update(h_new, sc)

    # Phase 4: Full (non-symmetric) optimization via projected gradient
    if time_left() > 40:
        n = len(best_h)
        for beta in [500, 3000, 10000]:
            if check():
                break
            tl = min(time_left() * 0.1, 12.0)
            if tl < 3:
                break
            try:
                def full_obj(x):
                    h = make_feasible(x)
                    vals = corr_vals(h)
                    cmax = vals.max()
                    sh = beta * (vals - cmax)
                    return cmax + np.log(np.sum(np.exp(sh))) / beta
                def full_grad(x):
                    h = make_feasible(x)
                    vals = corr_vals(h)
                    cmax = vals.max()
                    sh = beta * (vals - cmax)
                    w = np.exp(sh)
                    ws = w.sum()
                    if ws < 1e-300:
                        w = np.ones(len(vals)) / len(vals)
                    else:
                        w /= ws
                    G = corr_grad_mat(h)
                    return w @ G

                def wrapped_obj(x):
                    return full_obj(x)
                def wrapped_grad(x):
                    return full_grad(x)

                res = minimize(wrapped_obj, best_h.copy(), jac=wrapped_grad,
                             method='L-BFGS-B', bounds=[(0, 1)] * n,
                             options={'maxiter': 500, 'ftol': 1e-19, 'maxfun': 2000})
                if res.x is not None:
                    h_new = make_feasible(res.x)
                    sc = official_score(h_new)
                    if sc < best_score - 1e-14:
                        update(h_new, sc)
            except Exception:
                pass

    # Phase 5: Random restarts with local optimization
    if time_left() > 50:
        n_cur = len(best_h)
        n_starts = max(4, min(12, int(time_left() / 12)))
        for i in range(n_starts):
            if check():
                break
            sigma = [0.02, 0.01, 0.04, 0.015, 0.03, 0.008, 0.05, 0.02,
                     0.025, 0.012, 0.035, 0.02][i % 12]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n_cur))
            h_cand = symmetrize_h(h_cand)
            if n_cur % 2 == 0:
                m = n_cur // 2
                x0 = np.clip(h_cand[:m], 0.005, 0.995)
                for beta in [1000, 8000]:
                    if check():
                        break
                    try:
                        res = minimize(sym_obj_fun, x0, jac=sym_obj_grad,
                                      args=(beta,), method='L-BFGS-B',
                                      bounds=[(0.0, 1.0)] * m,
                                      options={'maxiter': 300, 'ftol': 1e-19, 'maxfun': 1500})
                        if res.x is not None and np.all(np.isfinite(res.x)):
                            h_new = h_from_half(np.clip(res.x, 0, 1))
                            sc = official_score(h_new)
                            if sc < best_score - 1e-14:
                                update(h_new, sc)
                                x0 = np.clip(h_new[:m], 0.001, 0.999)
                    except Exception:
                        pass

    # Phase 6: Final intensive polish - very high beta + active set
    if time_left() > 20:
        h_s = symmetrize_h(best_h)
        n = len(h_s)
        if n % 2 == 0:
            m = n // 2
            x0 = np.clip(h_s[:m], 0.001, 0.999)
            for beta in [50000, 200000, 500000]:
                if check():
                    break
                try:
                    res = minimize(sym_obj_fun, x0, jac=sym_obj_grad,
                                  args=(beta,), method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * m,
                                  options={'maxiter': 1500, 'ftol': 1e-20,
                                           'gtol': 1e-15, 'maxfun': 5000})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_new = h_from_half(np.clip(res.x, 0, 1))
                        sc = official_score(h_new)
                        if sc < best_score - 1e-14:
                            update(h_new, sc)
                            x0 = np.clip(h_new[:m], 0.0001, 0.9999)
                except Exception:
                    pass
        # Final active set with multiple criteria
        if time_left() > 10:
            h_s = symmetrize_h(best_h)
            if len(h_s) % 2 == 0:
                m = len(h_s) // 2
                x_s = np.clip(h_s[:m], 0.001, 0.999)
                vals = corr_vals(h_s)
                for rel_tol in [0.002, 0.004, 0.006]:
                    if check():
                        break
                    active = find_active(vals, rel_tol=rel_tol, min_count=10)
                    h_new, sc = active_minimax_sym(x_s, active, max_iter=400)
                    if h_new is not None and sc < best_score - 1e-14:
                        update(h_new, sc)
                        h_s = symmetrize_h(h_new)
                        m = len(h_s) // 2
                        x_s = np.clip(h_s[:m], 0.001, 0.999)
                        vals = corr_vals(h_s)

    # Phase 7: Try slightly different even n values for the final best
    if time_left() > 30:
        n_best = len(best_h)
        for dn in [-2, 2, 4]:
            if check():
                break
            n_try = n_best + dn
            if n_try < 40 or n_try > 96 or n_try % 2 != 0:
                continue
            h_t = interpolate_h(best_h, n_try)
            h_t = symmetrize_h(h_t)
            m_t = n_try // 2
            x0 = np.clip(h_t[:m_t], 0.005, 0.995)
            for beta in [3000, 20000]:
                if check():
                    break
                try:
                    res = minimize(sym_obj_fun, x0, jac=sym_obj_grad,
                                  args=(beta,), method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * m_t,
                                  options={'maxiter': 600, 'ftol': 1e-19, 'maxfun': 2500})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_new = h_from_half(np.clip(res.x, 0, 1))
                        sc = official_score(h_new)
                        if sc < best_score - 1e-14:
                            update(h_new, sc)
                            x0 = np.clip(h_new[:m_t], 0.001, 0.999)
                except Exception:
                    pass
            # Active set polish
            if check():
                break
            h_t = symmetrize_h(best_h)
            vals_t = corr_vals(h_t)
            if len(h_t) % 2 == 0:
                m_t = len(h_t) // 2
                x_t = np.clip(h_t[:m_t], 0.005, 0.995)
                active_t = find_active(vals_t, rel_tol=0.004, min_count=8)
                h_new, sc = active_minimax_sym(x_t, active_t, max_iter=300)
                if h_new is not None and sc < best_score - 1e-14:
                    update(h_new, sc)

    # Final validation and return
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    if final_score > best_score + 1e-10:
        final_h = best_h.copy()
        final_score = best_score

    assert np.all(final_h >= -1e-12), "Negative values in h"
    assert np.all(final_h <= 1.0 + 1e-12), "Values > 1 in h"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-8, "Sum constraint violated"
    assert np.all(np.isfinite(final_h)), "Non-finite values"

    return final_h, final_score + 1e-13, final_n
