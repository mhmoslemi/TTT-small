def make_feasible(z, target=None):

    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:

            else:

            h[i] += add
            residual -= add

def interp_to_n(h_src, n_dst):

    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_at_lag(h, lag_idx):
    """Gradient of correlation at index lag_idx with respect to h."""
    n = len(h)
    d = lag_idx - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        for j in range(n - d):
            grad[j] += -(1.0 - h[j + d])  # d/dh[j] of h[j](1-h[j+d])
            grad[j + d] += -h[j]           # d/dh[j+d] of h[j](1-h[j+d])
    else:
        m = -d
        for j in range(m, n):
            grad[j] += -(1.0 - h[j - m])
            grad[j - m] += -h[j]
    return grad

    active = np.where(corr >= threshold)[0]

        active = np.array([np.argmax(corr)])
    
    for it in range(max_iter):

            if time.time() - t_start > time_limit:
                break

        def obj(z):
            return z[-1]
        
        def obj_jac(z):

            g[-1] = 1.0
            return g

        n_active = len(active)
        def ineq(z):
            h_var = z[:-1]
            t_var = z[-1]
            c = compute_corr(h_var)
            return t_var - c[active]
        
        def ineq_jac(z):

            jac = np.zeros((n_active, n + 1))
            jac[:, -1] = 1.0

                grad = compute_grad_at_lag(h_var, k)
                jac[i, :n] -= grad  # t - corr[k]: derivative w.r.t. h is -grad
            return jac

            return np.sum(z[:-1]) - n / 2.0
        
        def eq_jac(z):
            g = np.zeros(n + 1)
            g[:-1] = 1.0
            return g

Let me try a different tactic: **coordinate ascent on the dual**.

The problem is: min_h max_k f_k(h) where f_k(h) = corr at lag k.

The Lagrangian dual is: max_w min_h sum_k w_k f_k(h) subject to w >= 0, sum w = 1.

At the optimum, the inner min gives h*(w), and we update w in the primal direction: w_k = max(0, w_k + eta * (f_k(h*) - max_j f_j(h*))).

This is a saddle-point method. Let me implement it:

```python
def dual_ascent(h0, eta_schedule, n_iter):
    h = h0.copy()
    n = len(h)
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    
    for it in range(n_iter):
        eta = eta_schedule(it)
        # Inner: minimize sum_k w_k * f_k(h)
        # This is a quadratic (f_k is bilinear in h, so w^T f is quadratic)
        # Actually f_k(h) = sum_j h[j](1-h[j+d]) which is linear in h for fixed other entries
        # The weighted sum is sum_k w_k * sum_j h[j](1-h[j+d_k])
        # = sum_j h[j] * (1 - sum_k w_k * h[j+d_k])  [roughly]
        # This is not simply quadratic due to the overlapping sums
        
        # Use SLSQP to minimize the weighted sum
        def weighted_obj(h):
            corr = compute_corr(h)
            return np.dot(w, corr)
        
        def weighted_grad(h):
            corr = compute_corr(h)
            grad = np.zeros(n)
            for k in range(n_lags):
                if w[k] < 1e-10:
                    continue
                grad += w[k] * compute_grad_at_lag(h, k)
            return grad
        
        # Solve inner problem
        res = minimize(weighted_obj, h, jac=weighted_grad, method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':lambda x: np.sum(x)-n/2, 
                                     'jac':lambda x: np.ones(n)}],
                       options={'maxiter': 30, 'ftol': 1e-14})
        if res.success:
            h = make_feasible(res.x)
        
        # Outer: update weights
        corr = compute_corr(h)
        f_max = np.max(corr)
        w = np.maximum(0, w + eta * (corr - f_max))
        w_sum = np.sum(w)
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.ones(n_lags) / n_lags
    
    return h, official_score(h)
```

        # or projected gradient

            grad_w = w @ grads  # (n,)
            # Projected gradient step

            h_new = make_feasible(h_new)
            # Check if we're making progress
            obj_old = w @ compute_corr(h)

        score = official_score(h)

            best_h = h.copy()

        w = np.maximum(0.0, w + eta * (corr - f_max))
        w_sum = np.sum(w)
        if w_sum > 1e-15:
            w /= w_sum
        else:
            idx = np.argsort(corr)[::-1][:5]
            w = np.zeros(n_lags)
            w[idx] = 1.0 / 5.0
        
        # Track best
        score = official_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        eta *= eta_decay
    
    return best_h, best_score

def topm_subgrad(h0, n_iter=2000, m=4, time_limit=None, t_start=None):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    step = 0.03
    patience = 0
    max_pat = 300
    
    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:m]
        grads = compute_grad_all(h)
        grad_avg = np.mean(grads[top_idx], axis=0)
        
        h_new = h - step * grad_avg
        h_new = make_feasible(h_new)
        new_score = official_score(h_new)
        
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h_new.copy()
            h = h_new
            step = min(0.3, step * 1.03)
            patience = 0
        else:
            h = h_new
            step = max(0.001, step * 0.97)
            patience += 1
            if patience > max_pat:
                break
    
    return best_h, best_score
