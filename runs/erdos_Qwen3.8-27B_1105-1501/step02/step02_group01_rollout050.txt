    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p=2.0):
    return make_feasible(np.clip(h, 0, 1) ** p)

def compute_grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += (-h[s:])
    else:
        m = -s
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] += (-h[:n - m])
    return grad

def find_optimal_t(A, B, C, t_lo, t_hi, n_samples=200):
    """Find t in [t_lo, t_hi] minimizing max_k (A_k + B_k*t - C_k*t^2)."""
    if t_hi <= t_lo + 1e-14:
        return t_lo, np.max(A)
    t_grid = np.linspace(t_lo, t_hi, n_samples)
    vals = A[None, :] + B[None, :] * t_grid[:, None] - C[None, :] * t_grid[:, None] ** 2
    max_vals = np.max(vals, axis=1)
    idx = np.argmin(max_vals)
    best_t = t_grid[idx]
    best_val = max_vals[idx]
    # Refine
    if 0 < idx < n_samples - 1:
        t_fine = np.linspace(t_grid[idx - 1], t_grid[idx + 1], 200)
        vals_f = A[None, :] + B[None, :] * t_fine[:, None] - C[None, :] * t_fine[:, None] ** 2
        max_f = np.max(vals_f, axis=1)
        idx_f = np.argmin(max_f)
        if max_f[idx_f] < best_val:
            best_t = t_fine[idx_f]
            best_val = max_f[idx_f]
    return best_t, best_val

def pair_move_optimize(h0, max_rounds=200, time_limit=30, t_start=None, max_pairs=40):
    """Exact quadratic pair moves for local optimization."""
    h = h0.copy()
    n = len(h)
    A = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    best_score = float(np.max(A))
    best_h = h.copy()
    stall = 0
    max_stall = 40

    for rnd in range(max_rounds):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        active = np.where(A >= best_score * 0.97)[0]
        if len(active) == 0:
            active = np.array([np.argmax(A)])
        grad = np.zeros(n)
        for k in active:
            grad += compute_grad_k(h, k)
        grad /= len(active)

        donors = np.where(grad > 0.0005)[0]
        receivers = np.where(grad < -0.0005)[0]

        if len(donors) == 0 or len(receivers) == 0:
            # Try all pairs randomly
            perm = np.random.default_rng(rnd + 1).permutation(n)
            pairs = [(perm[i], perm[j]) for i in range(min(20, n)) for j in range(i + 1, min(40, n))]
        else:
            d_sorted = donors[np.argsort(-grad[donors])][:max_pairs]
            r_sorted = receivers[np.argsort(grad[receivers])][:max_pairs]
            pairs = [(d, r) for d in d_sorted for r in r_sorted]
            # Add some random pairs
            rng_local = np.random.default_rng(rnd * 7 + 13)
            for _ in range(10):
                i = rng_local.integers(n)
                j = rng_local.integers(n)
                if i != j:
                    pairs.append((i, j))

        best_improve = 0
        best_pair = None
        best_t = 0.0

        for i, j in pairs:
            if i == j:
                continue
            t_lo = max(h[i] - 1.0, -h[j])
            t_hi = min(h[i], 1.0 - h[j])
            if t_hi <= t_lo + 1e-12:
                continue
            v = np.zeros(n)
            v[i] = -1.0
            v[j] = 1.0
            B = (np.correlate(v, 1.0 - h, 'full') - np.correlate(h, v, 'full')) * (2.0 / n)
            C = np.correlate(v, v, 'full') * (2.0 / n)
            t_opt, score_opt = find_optimal_t(A, B, C, t_lo, t_hi)
            improve = best_score - score_opt
            if improve > best_improve:
                best_improve = improve
                best_pair = (i, j)
                best_t = t_opt

        if best_pair is not None and best_improve > 1e-10:
            i, j = best_pair
            h[i] -= best_t
            h[j] += best_t
            h = make_feasible(h)
            A = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            new_score = float(np.max(A))
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Random perturbation to escape
            if stall > 10:
                rng_esc = np.random.default_rng(rnd * 31 + 7)
                idx1 = rng_esc.integers(n)
                idx2 = rng_esc.integers(n)
                if idx1 != idx2:
                    amt = 0.1 * rng_esc.random()
                    if h[idx1] > 0.5 and h[idx2] < 0.5:
                        h[idx1] -= amt
                        h[idx2] += amt
                        h = make_feasible(h)
                        A = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                        if float(np.max(A)) > best_score + 0.01:
                            h = best_h.copy()
                            A = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                    else:
                        h = best_h.copy()
                        A = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

        if stall > max_stall:
            break

    return best_h, best_score

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -400, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -400, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        w *= dx
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x * 2.0 + (1.0 - x)
                # grad += 1 - 2x
            else:
                grad[s2:] += (1.0 - x[:n - s2])
                grad[:n - s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += 1.0 - 2.0 * x
            else:
                grad[:n - m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n - m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def do_slsqp_smooth(h0, beta, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0.copy(), args=(beta, dx),
            jac=lambda x: smooth_lse_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def do_direct_minimax(h0, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    t0 = official_score(h0) + 0.01
    z0 = np.concatenate([h0, [t0]])

    def mm_obj(z):
        return z[-1]

    def mm_obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def mm_ineq(z):
        h = z[:-1]
        t = z[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr

    def mm_ineq_jac(z):
        h = z[:-1]
        L = 2 * n - 1
        jac = np.zeros((L, n + 1))
        jac[:, -1] = 1.0
        for k in range(L):
            s2 = k - (n - 1)
            if s2 >= 0:
                if s2 == 0:
                    jac[k, :n] = dx * (2.0 * h - 1.0)
                else:
                    jac[k, s2:n] -= dx * (1.0 - h[:n - s2])
                    jac[k, :n - s2] -= dx * (-h[s2:])
            else:
                m2 = -s2
                if m2 == 0:
                    jac[k, :n] = dx * (2.0 * h - 1.0)
                else:
                    jac[k, :n - m2] -= dx * (1.0 - h[m2:])
                    jac[k, m2:n] -= dx * (-h[:n - m2])
        return jac

    def mm_eq(z):
        return np.sum(z[:-1]) - n / 2.0

    def mm_eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(
            mm_obj, z0, jac=mm_obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
                {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0

def simulated_annealing(h0, rng, n_swaps=20000, time_limit=40, t_start=None):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    T = 0.03
    T_end = 0.002
    cool_rate = (T_end / T) ** (1.0 / max(n_swaps, 1))

    for it in range(n_swaps):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if h[i] >= h[j]:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        delta = min(h[donor], 1.0 - h[receiver])
        if delta < 0.005:
            continue
        swap = delta * rng.uniform(0.3, 1.0)
        h[donor] -= swap
        h[receiver] += swap
        new_score = official_score(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
        else:
            if rng.random() > np.exp(-(new_score - best_score) / max(T, 1e-8)):
                h[donor] += swap
                h[receiver] -= swap
        T *= cool_rate
    return best_h, best_score

def build_starts(h_parent, n, rng):
    """Build diverse starting points for optimization at resolution n."""
    starts = []
    h_interp = make_feasible(interp_to_n(h_parent, n))
    starts.append(h_interp)

    for p in [1.5, 2.5, 5.0]:
        starts.append(push_binary(h_interp, p))

    if n % 2 == 0:
        rb = np.zeros(n)
        rb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(rb)
    else:
        rb = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        rb[idx] = 1.0
        rb[rng.integers(n)] = 0.5
        starts.append(make_feasible(rb))

    for sigma in [0.05, 0.12, 0.25]:
        starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))

    starts.append(make_feasible(1.0 - h_interp[::-1].copy()))

    # Periodic patterns
    for period in [3, 4, 5, 7]:
        base = np.tile(1.0 / period, n + period)[:n]
        pat = make_feasible(base + 0.02 * rng.standard_normal(n))
        starts.append(pat)

    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)

    # Try multiple resolutions
    parent_n = len(parent)
    n_options = sorted(set([
        max(24, parent_n - 4), parent_n, min(80, parent_n + 2),
        48, 50, 52, 54, 56
    ]))
    n_options = [n for n in n_options if 24 <= n <= 76]

    # Sort by proximity to parent for priority
    n_options.sort(key=lambda x: abs(x - parent_n))

    for n in n_options:
        if time.time() > deadline - 60:
            break

        starts = build_starts(parent, n, rng)
        # Deduplicate
        unique_starts = [starts[0]]
        for s in starts[1:]:
            if all(np.max(np.abs(s - us)) > 0.05 for us in unique_starts):
                unique_starts.append(s)

        for h0 in unique_starts:
            if time.time() > deadline - 30:
                break
            cur_h = make_feasible(h0.copy())
            cur_score = official_score(cur_h)

            # If already good, skip to fast polish
            if cur_score < best_score - 0.005:
                pass  # great start, go straight to optimization

            # Phase 1: Pair move optimization (primary)
            if time.time() < deadline - 25:
                pm_h, pm_score = pair_move_optimize(
                    cur_h, max_rounds=150, time_limit=50,
                    t_start=time.time(), max_pairs=35
                )
                if pm_score < cur_score - 1e-10:
                    cur_h, cur_score = pm_h, pm_score

            # Phase 2: SA for exploration
            if time.time() < deadline - 20 and cur_score > 0.3808:
                sa_h, sa_score = simulated_annealing(
                    cur_h, rng, n_swaps=15000,
                    time_limit=35, t_start=time.time()
                )
                if sa_score < cur_score - 1e-10:
                    cur_h, cur_score = sa_h, sa_score
                    # Refine with pair moves after SA
                    pm_h2, pm_score2 = pair_move_optimize(
                        cur_h, max_rounds=100, time_limit=25,
                        t_start=time.time(), max_pairs=35
                    )
                    if pm_score2 < cur_score - 1e-10:
                        cur_h, cur_score = pm_h2, pm_score2

            # Phase 3: SLSQP smooth polish
            for beta in [80, 300]:
                if time.time() > deadline - 15:
                    break
                mi = 120 if n <= 56 else 60
                new_h = do_slsqp_smooth(cur_h, beta, maxiter=mi)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-12:
                    cur_h, cur_score = new_h, new_score

            # Phase 4: More pair moves after smoothing
            if time.time() < deadline - 15:
                pm_h3, pm_score3 = pair_move_optimize(
                    cur_h, max_rounds=120, time_limit=30,
                    t_start=time.time(), max_pairs=40
                )
                if pm_score3 < cur_score - 1e-10:
                    cur_h, cur_score = pm_h3, pm_score3

            # Phase 5: Direct minimax
            if time.time() < deadline - 12:
                dc_h = do_direct_minimax(cur_h, maxiter=80)
                dc_score = official_score(dc_h)
                if dc_score < cur_score - 1e-12:
                    cur_h, cur_score = dc_h, dc_score

            # Phase 6: Final pair polish
            if time.time() < deadline - 10:
                pm_h4, pm_score4 = pair_move_optimize(
                    cur_h, max_rounds=80, time_limit=20,
                    t_start=time.time(), max_pairs=50
                )
                if pm_score4 < cur_score - 1e-10:
                    cur_h, cur_score = pm_h4, pm_score4

            # Update global best
            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
            elif abs(cur_score - best_score) <= 1e-11 and n < best_n:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = n

    # Final refinement on best solution
    remaining = deadline - time.time()
    if remaining > 15:
        # Multiple refinement trials
        n_trials = min(10, int(remaining / 12))
        for trial in range(n_trials):
            if time.time() > deadline - 8:
                break
            h_pert = make_feasible(best_h + 0.02 * rng.standard_normal(best_n))
            pm_h, pm_s = pair_move_optimize(
                h_pert, max_rounds=80, time_limit=10,
                t_start=time.time(), max_pairs=40
            )
            if pm_s < best_score - 1e-12:
                best_h, best_score = pm_h, pm_s

            # Quick SLSQP
            if time.time() < deadline - 10:
                sl_h = do_slsqp_smooth(best_h, 400, maxiter=60)
                sl_s = official_score(sl_h)
                if sl_s < best_score - 1e-12:
                    best_h, best_score = sl_h, sl_s
                    # Pair polish after
                    pm_h2, pm_s2 = pair_move_optimize(
                        best_h, max_rounds=50, time_limit=8,
                        t_start=time.time(), max_pairs=40
                    )
                    if pm_s2 < best_score - 1e-12:
                        best_h, best_score = pm_h2, pm_s2

    # Absolute final polish
    if time.time() < deadline - 5:
        final_h = make_feasible(best_h)
        final_score = official_score(final_h)
        if final_score > best_score:
            final_h = best_h.copy()
            final_score = best_score

        # One last pair move pass
        pm_final, pm_final_s = pair_move_optimize(
            final_h, max_rounds=50, time_limit=5,
            t_start=time.time(), max_pairs=50
        )
        if pm_final_s < final_score - 1e-12:
            final_h, final_score = pm_final, pm_final_s

    final_h = make_feasible(final_h)
    final_score = official_score(final_h)

    return (final_h, final_score, len(final_h))
