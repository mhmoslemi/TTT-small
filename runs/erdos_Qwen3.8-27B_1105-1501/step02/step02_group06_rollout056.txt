    """Iterative QP refinement with K active constraints."""
    n = len(h0)
    h = h0.copy()
    best_score = official_score(h)
    
    for it in range(n_iter):
        reg = reg_start * (reg_end / reg_start) ** (it / max(1, n_iter - 1))
        
        # Compute correlations
        raw = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        vals = raw * dx
        idx_max = np.argmax(vals)
        
        # Select top K constraints
        top_idx = np.argsort(vals)[-K:]

        h_var = cp.Variable(n)

            h_var >= 0,
            h_var <= 1

                    break

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        if resid > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(resid, 1.0 - h[idx])
                h[idx] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-resid, h[idx])
                h[idx] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)


def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for j in range(s, n):
            g[j] += (1.0 - h[j - s])
        for j in range(0, n - s):
            g[j] -= h[j + s]
    else:
        t = -s
        for j in range(0, n - t):
            g[j] += (1.0 - h[j + t])
        for j in range(t, n):
            g[j] -= h[j - t]
    return g


def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def qp_step(h, K, reg, use_trust, trust_r):
    n = len(h)
    vals = compute_vals(h)
    # Select top K shifts, ensure both positive and negative are represented
    sorted_idx = np.argsort(vals)[::-1]
    top_idx = sorted_idx[:K]
    # Ensure at least 2 from each side if possible
    positive = [i for i in top_idx if i > n - 1]
    negative = [i for i in top_idx if i < n - 1]
    zero = [i for i in top_idx if i == n - 1]
    if len(positive) < 2 and len(sorted_idx[sorted_idx > n - 1]) >= 2:
        extra = sorted_idx[sorted_idx > n - 1][:2]
        top_idx = np.concatenate([top_idx, extra])
        top_idx = np.unique(top_idx)
        top_idx = top_idx[:K + 2]
    if len(negative) < 2 and len(sorted_idx[sorted_idx < n - 1]) >= 2:
        extra = sorted_idx[sorted_idx < n - 1][:2]
        top_idx = np.concatenate([top_idx, extra])
        top_idx = np.unique(top_idx)
        top_idx = top_idx[:K + 2]

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = grad_shift(h, s)
        constraints.append(vals[p] + g @ (h_var - h) <= t_var)

    if use_trust:
        constraints.append(cp.norm(h_var - h, 2) <= trust_r)
        obj = cp.Minimize(t_var)
    else:
        obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h))

    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iter=10000, tol_gap_abs=1e-10,
                       tol_gap_rel=1e-10, tol_feas=1e-10)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                if h_new is not None and np.all(np.isfinite(h_new)):
                    return make_feasible(h_new)
        except Exception:
            continue
    return None


def line_search(h, h_target, n_pts=12):
    best_alpha = 0.0
    best_score = official_score(h)
    best_h = h
    for alpha in np.linspace(0.05, 1.0, n_pts):
        h_try = make_feasible((1.0 - alpha) * h + alpha * h_target)
        sc = official_score(h_try)
        if sc < best_score - 1e-15:
            best_score = sc
            best_alpha = alpha
            best_h = h_try
    return best_h, best_score, best_alpha


def qp_refine(h0, max_iter=30, K=12, reg_start=2.0, reg_end=0.005,
              trust_start=0.3, deadline=None):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    stall = 0

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline - 5:
            break
        frac = it / max(1, max_iter - 1)
        reg = reg_start * (reg_end / reg_start) ** frac
        trust_r = trust_start * (0.05 / trust_start) ** frac * 0.1 + 0.01

        h_new = qp_step(h, K=K, reg=reg, use_trust=True, trust_r=trust_r)
        if h_new is None:
            h_new = qp_step(h, K=K, reg=reg, use_trust=False, trust_r=0)

        if h_new is None:
            stall += 1
            if stall >= 3:
                K = max(4, K - 2)
                h_new = qp_step(h, K=K, reg=reg * 0.5, use_trust=False, trust_r=0)
                if h_new is None:
                    break
            continue

        h_best_ls, sc_best_ls, alpha = line_search(h, h_new)
        if sc_best_ls < best - 1e-14:
            h = h_best_ls
            best = sc_best_ls
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                break
    return h, best


def subgrad_refine(h0, max_iter=50, K=8, deadline=None):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    step_base = 0.4
    stall = 0

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline - 5:
            break
        vals = compute_vals(h)
        sorted_idx = np.argsort(vals)[::-1]
        active = sorted_idx[:K]

        # Weighted gradient: weight by how close to max
        vmax = vals[active[0]]
        vmin = vals[active[-1]]
        spread = max(vmax - vmin, 1e-10)
        weights = np.exp((vals[active] - vmax) / (spread * 0.3))
        weights /= weights.sum()

        d = np.zeros(n)
        for w, p in zip(weights, active):
            s = p - (n - 1)
            d += w * grad_shift(h, s)

        # Project to sum-zero space
        d -= d.mean()
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            break

        step = step_base * (0.95 ** stall)
        d = -d / d_norm * step

        h_new = make_feasible(h + d)
        sc = official_score(h_new)

        # Line search from h to h_new
        h_ls, sc_ls, _ = line_search(h, h_new, n_pts=8)
        if sc_ls < best - 1e-14:
            h = h_ls
            best = sc_ls
            stall = 0
        else:
            stall += 1
            if stall >= 10:
                break
    return h, best


def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)


def anti_sym(h):
    return make_feasible(1.0 - h[::-1].copy())


def smooth_obj(h, beta):
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta


def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0


def smooth_eq_jac(h):
    return np.ones(len(h))


def try_slsqp(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError, Exception):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.25 * np.sin(4 * np.pi * x))
        best_score = official_score(best_h)

    def try_update(h_cand, sc):
        nonlocal best_h, best_score
        if h_cand is not None and np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc

    # Phase 1: Aggressive QP at parent resolution
    if deadline - time.time() > 60:
        h_opt, sc = qp_refine(best_h, max_iter=25, K=12, reg_start=1.5,
                              reg_end=0.003, trust_start=0.25, deadline=deadline)
        try_update(h_opt, sc)

    # Phase 2: More QP iterations with different K
    if deadline - time.time() > 40:
        for K in [8, 15, 20]:
            if deadline - time.time() < 20:
                break
            h_opt, sc = qp_refine(best_h, max_iter=20, K=K, reg_start=1.0,
                                  reg_end=0.002, trust_start=0.2, deadline=deadline)
            try_update(h_opt, sc)

    # Phase 3: Subgradient method
    if deadline - time.time() > 40:
        h_opt, sc = subgrad_refine(best_h, max_iter=60, K=10, deadline=deadline)
        try_update(h_opt, sc)
        if deadline - time.time() > 20:
            h_opt, sc = subgrad_refine(best_h, max_iter=40, K=15, deadline=deadline)
            try_update(h_opt, sc)

    # Phase 4: Resolution sweep
    if deadline - time.time() > 80:
        n_base = len(best_h)
        n_opts = sorted(set([n_base - 4, n_base - 2, n_base, n_base + 2,
                             n_base + 4, n_base + 8]))
        n_opts = [x for x in n_opts if 24 <= x <= 120]
        for n_opt in n_opts:
            if deadline - time.time() < 25:
                break
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            try_update(h_int, sc_int)
            if n_opt != n_base and deadline - time.time() > 30:
                h_opt, sc = qp_refine(h_int, max_iter=15, K=10, reg_start=1.0,
                                      reg_end=0.003, trust_start=0.2, deadline=deadline)
                try_update(h_opt, sc)

    # Phase 5: Anti-symmetric transform + refine
    if deadline - time.time() > 40:
        h_sym = anti_sym(best_h)
        sc_sym = official_score(h_sym)
        try_update(h_sym, sc_sym)
        if deadline - time.time() > 30:
            h_opt, sc = qp_refine(h_sym, max_iter=20, K=12, reg_start=1.0,
                                  reg_end=0.002, trust_start=0.2, deadline=deadline)
            try_update(h_opt, sc)

    # Phase 6: SLSQP smooth approach
    if deadline - time.time() > 60:
        for beta in [500, 2000, 8000, 30000]:
            if deadline - time.time() < 20:
                break
            h_new, sc = try_slsqp(best_h, beta, maxiter=150)
            if h_new is not None:
                try_update(h_new, sc)
            if deadline - time.time() > 20:
                h_opt, sc = qp_refine(best_h, max_iter=10, K=10, reg_start=0.5,
                                      reg_end=0.001, trust_start=0.15, deadline=deadline)
                try_update(h_opt, sc)

    # Phase 7: Random perturbations + QP
    if deadline - time.time() > 60:
        for i in range(10):
            if deadline - time.time() < 20:
                break
            sigma = 0.01 + 0.06 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc = qp_refine(h_cand, max_iter=12, K=10, reg_start=1.0,
                                  reg_end=0.002, trust_start=0.2, deadline=deadline)
            try_update(h_opt, sc)

    # Phase 8: Final intensive polish
    if deadline - time.time() > 30:
        for K in [6, 10, 14, 18]:
            if deadline - time.time() < 10:
                break
            h_opt, sc = qp_refine(best_h, max_iter=15, K=K, reg_start=0.3,
                                  reg_end=0.0005, trust_start=0.12, deadline=deadline)
            try_update(h_opt, sc)

    # Phase 9: One more SLSQP at high beta
    if deadline - time.time() > 25:
        h_new, sc = try_slsqp(best_h, 50000, maxiter=100)
        if h_new is not None:
            try_update(h_new, sc)
        if deadline - time.time() > 15:
            h_opt, sc = qp_refine(best_h, max_iter=12, K=12, reg_start=0.2,
                                  reg_end=0.0003, trust_start=0.08, deadline=deadline)
            try_update(h_opt, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = final_score + 1e-12
    return final_h, final_bound, len(final_h)
