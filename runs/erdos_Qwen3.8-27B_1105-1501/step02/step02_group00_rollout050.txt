def _smoothed_grad(h, beta=20):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_s = np.max(scores)
    w = np.exp(beta * (scores - max_s))
    w /= np.sum(w)
    # Gradient = sum w_d * grad s(d)
    # grad s(d) w.r.t. h[j]:
    # s(d) = sum_i h[i]*(1-h[i+d])
    # d s(d)/d h[j] = (1-h[j+d]) - h[j-d] (with boundary handling)
    # More precisely:
    # s(d) = sum_{valid i} h[i]*(1-h[i+d])
    # d s(d)/d h[j] = sum_i [delta_{i,j}*(1-h[i+d]) + h[i]*(-delta_{i+d,j})]
    # = (1-h[j+d]) - h[j-d] (when both are in range)
    
    grad = np.zeros(n)
    for d_idx in range(2*n-1):
        if w[d_idx] < 1e-10:
            continue
        d = d_idx - (n-1)
        # Contribution to grad from lag d
        if d >= 0:
            # s(d) = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # d/dh[j]: 
            # if j <= n-1-d: (1-h[j+d])
            # if j >= d: -h[j-d]
            m = n - d
            for j in range(m):
                grad[j] += w[d_idx] * (1 - h[j+d])
            for j in range(d, n):
                grad[j] += w[d_idx] * (-h[j-d])
        else:
            g = -d
            m = n - g
            for j in range(m):
                grad[j] += w[d_idx] * (-h[j+g])
            for j in range(g, n):
                grad[j] += w[d_idx] * (1 - h[j-g])
    return grad

def _smooth_dir(h, beta=20.0):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w /= float(np.sum(w))
    G = _grad_matrix(h)
    d = G.T @ w
    d -= np.mean(d)  # Remove component along all-ones (to maintain sum constraint)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return -d / norm

def _line_search(h, p, c5_cur, n_evals=40):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = min(max(alpha_max, 1e-8), 1.0)
    
    # Coarse search
    alphas = np.linspace(0.0, alpha_max, n_evals)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        if a < 1e-10:
            continue
        h_t = h + a * p
        c5_t = _c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo_a = max(0.0, best_a - alpha_max * 0.1)
    hi_a = min(alpha_max, best_a + alpha_max * 0.1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _c5(h + c1 * p)
    f2 = _c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2; c2 = c1; f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _c5(h + c1 * p)
        else:
            lo_a = c1; c1 = c2; f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _c5(h + c2 * p)
    
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _opt_loop(h, deadline, method='lp', radius_init=0.15, max_iters=3000, beta=20.0):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        
        if method == 'lp':
            p = _lp_direction(h, scores, G, radius)
        else:
            p = _smooth_dir(h, beta=beta)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            if method == 'lp':
                radius *= 0.7
                if radius < 1e-5:
                    break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            if method == 'lp':
                radius = min(radius * 1.2, 0.5)
        else:
            stall += 1
            if method == 'lp':
                radius *= 0.7
                if stall == 10:
                    radius = min(0.3, radius * 3)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_local_search(h, deadline, max_iters=5000):
    """Local search on binary vectors by swapping 1↔0 pairs."""
    n = len(h)
    # Start with thresholded version
    h_bin = (h > 0.5).astype(float)
    ones = np.sum(h_bin)
    target = n / 2
    # Adjust to have exactly n/2 ones
    if ones < target:
        # Flip some 0s to 1s
        zero_idx = np.where(h_bin == 0)[0]
        for i in range(int(target - ones)):
            h_bin[zero_idx[i]] = 1.0
    elif ones > target:
        one_idx = np.where(h_bin == 1)[0]
        for i in range(int(ones - target)):
            h_bin[one_idx[i]] = 0.0
    
    cur_c5 = _c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        improved = False
        one_idx = np.where(h_bin == 1)[0]
        zero_idx = np.where(h_bin == 0)[0]
        # Try all swaps (expensive but n is small)
        # Be smart: only check promising swaps
        # For now, do random sampling of swaps
        rng = np.random.default_rng(it)
        n_trials = min(len(one_idx) * len(zero_idx), 200)
        for _ in range(n_trials):
            i = one_idx[rng.integers(len(one_idx))]
            j = zero_idx[rng.integers(len(zero_idx))]
            h_bin[i] = 0
            h_bin[j] = 1
            c5_t = _c5(h_bin)
            if c5_t < cur_c5 - 1e-12:
                cur_c5 = c5_t
                improved = True
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_bin.copy()
                one_idx = np.where(h_bin == 1)[0]
                zero_idx = np.where(h_bin == 0)[0]
                break
            else:
                h_bin[i] = 1
                h_bin[j] = 0
        if not improved:
            break
    
    return best_h, best_c5

def _cosine_init(n, k, phase):

    h = 0.5 + 0.5 * np.cos(2 * np.pi * k * x / 2.0 + phase)

            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _c5(h + c2 * p)

    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _opt_loop(h, deadline, radius_init=0.15, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    use_smooth = False

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break

        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)

        if use_smooth:
            p = _smooth_dir(h, G, scores, beta=15.0)
        else:
            p = _lp_direction(h, scores, G, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            if not use_smooth:
                radius *= 0.7
                if stall % 5 == 0:
                    use_smooth = True
                elif radius < 1e-5:
                    break
            else:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            use_smooth = False
            radius = min(radius * 1.25, 0.5)
        else:
            stall += 1
            if not use_smooth:
                radius *= 0.7
                if stall == 8:
                    use_smooth = True
                if stall == 20:
                    radius = 0.15
            else:
                use_smooth = False
                radius = 0.1

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _cosine_init(n, k, phase):
    x = (np.arange(n) + 0.5) / n * 2.0
    h = 0.5 + 0.5 * np.cos(np.pi * k * x + phase)
    return _project(h, n / 2.0)

def _binary_search(h_start, deadline, max_iter=8000):
    n = len(h_start)
    hb = (h_start > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(hb))
    if ones < target:
        zidx = np.where(hb == 0)[0]
        for i in range(target - ones):
            hb[zidx[i]] = 1.0
    elif ones > target:
        oidx = np.where(hb == 1)[0]
        for i in range(ones - target):
            hb[oidx[i]] = 0.0
    cur = _c5(hb)
    best_h = hb.copy()
    best = cur
    rng = np.random.default_rng(123)
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        improved = False
        oidx = np.where(hb == 1)[0]
        zidx = np.where(hb == 0)[0]
        n_tries = min(300, len(oidx) * len(zidx))
        for _ in range(n_tries):
            i = oidx[rng.integers(len(oidx))]
            j = zidx[rng.integers(len(zidx))]
            hb[i] = 0.0
            hb[j] = 1.0
            c5t = _c5(hb)
            if c5t < cur - 1e-12:
                cur = c5t
                improved = True
                if c5t < best:
                    best = c5t
                    best_h = hb.copy()
                oidx = np.where(hb == 1)[0]
                zidx = np.where(hb == 0)[0]
                break
            else:
                hb[i] = 1.0
                hb[j] = 0.0
        if not improved:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: Main LP optimization from initial (30% budget)
    deadline1 = t0 + budget_s * 0.30
    h1, c51 = _opt_loop(h, deadline1, radius_init=0.18, max_iters=8000)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Second pass with smaller radius for polish (8% budget)
    deadline2 = t0 + budget_s * 0.38
    h2, c52 = _opt_loop(best_h, deadline2, radius_init=0.06, max_iters=5000)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2.copy()

    # Phase 3: Resolution sweep (35% budget)
    deadline3 = t0 + budget_s * 0.73
    n_tries = [best_n + d for d in range(-14, 15, 2) if best_n + d >= 28 and best_n + d <= 120]
    n_tries.sort(key=lambda x: abs(x - best_n))
    time_per_res = budget_s * 0.028
    for n_try in n_tries:
        if time.time() > deadline3:
            break
        h_r = _resample(best_h, n_try)
        dl = min(deadline3, time.time() + time_per_res)
        h_r, c5_r = _opt_loop(h_r, dl, radius_init=0.12, max_iters=1500)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Diversified restarts (15% budget)
    deadline4 = t0 + budget_s * 0.88
    scale_seq = [0.06, 0.04, 0.08, 0.03, 0.10, 0.05, 0.07, 0.04, 0.12, 0.06]
    for attempt in range(200):
        if time.time() > deadline4:
            break
        r = rng.random()
        sc = scale_seq[attempt % len(scale_seq)]
        if r < 0.35:
            h_pert = best_h + rng.normal(0, sc, best_n)
        elif r < 0.60:
            k = int(rng.integers(1, 6))
            ph = float(rng.random() * 2 * np.pi)
            cos_h = _cosine_init(best_n, k, ph)
            h_pert = (1.0 - 0.3 * sc * 10) * best_h + 0.3 * sc * 10 * cos_h
        elif r < 0.80:
            h_pert = best_h.copy()
            idx = rng.choice(best_n, size=max(2, best_n // 8), replace=False)
            h_pert[idx] = np.clip(h_pert[idx] + rng.normal(0, 0.3, len(idx)), 0, 1)
        else:
            h_pert = _resample(best_h, max(28, best_n + int(rng.integers(-4, 5))))
            h_pert = np.resize(h_pert, best_n)
            h_pert = h_pert + rng.normal(0, sc * 0.5, best_n)

        h_pert = _project(h_pert, best_n / 2.0)
        dl = min(deadline4, time.time() + budget_s * 0.018)
        h_try, c5_try = _opt_loop(h_pert, dl, radius_init=0.1, max_iters=600)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Binary search + refinement (5% budget)
    deadline5 = t0 + budget_s * 0.93
    h_bin, c5_bin = _binary_search(best_h, min(deadline5, t0 + budget_s * 0.04))
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    if time.time() < deadline5:
        h_ref, c5_ref = _opt_loop(h_bin, deadline5, radius_init=0.12, max_iters=800)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Final aggressive polish (7% budget)
    deadline6 = t0 + budget_s * 0.995
    h_f, c5_f = _opt_loop(best_h, deadline6, radius_init=0.04, max_iters=10000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
