def make_feasible(z):

    z = np.clip(z, -10.0, 10.0)
    lo, hi = -11.0, 11.0

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break

    n = len(h)
    M = R.shape[1]

        elif D > 0:
            L = n - D
            grad[p] = dx * (R[:L, :].T @ (1.0 - h[D:]) - R[D:, :].T @ h[:L])
        else:
            a = -D
            L = n - a
            grad[p] = dx * (R[a:, :].T @ (1.0 - h[:L]) - R[:L, :].T @ h[a:])

    n = R.shape[0]
    M = R.shape[1]
    u = sigmoid(z)
    h = np.empty(n)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    
    vals = compute_vals(h, dx)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_s = shifted.max()  # should be 0
    obj = cmax + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / beta
    # Simplified: obj = cmax + logsumexp(shifted)/beta
