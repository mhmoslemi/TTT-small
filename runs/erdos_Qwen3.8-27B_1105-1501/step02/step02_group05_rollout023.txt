import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    """Compute gradient of lag value for shift s (where p = s + n - 1)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + s])
        grad[idx + s] -= h[idx]
    else:
        t = -s
        m = n - t
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + t])
        grad[idx + t] -= h[idx]
    return dx * grad

def build_lp_direction(h, V, C, active_idx, rho):
    """Build and solve the LP for active-lag minimax direction."""
    n = len(h)
    K = len(active_idx)
    
    # Build constraint matrix
    # Variables: u_0, ..., u_{n-1}, t
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    # Constraints: g_p^T u - t <= -V[p] for each active p
    # Equality: sum(u) = 0
    # Bounds: max(-h_i, -rho) <= u_i <= min(1-h_i, rho), t <= C
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for j, p in enumerate(active_idx):
        s = p - (n - 1)
        g = lag_gradient(h, s)
        A_ub[j, :n] = g
        A_ub[j, n] = -1.0
        b_ub[j] = -V[p]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C))  # t bounded above by C
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        return res.x[:n]
    return None

def binary_balance(h):
    """Create a balanced binary sequence from h."""
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    b = (h > 0.5).astype(int)
    ones = np.sum(b)
    if ones == target:
        return b
    if ones > target:
        # Remove ones where h is closest to 0.5 from above
        pos1 = np.where(b == 1)[0]
        pos0 = np.where(b == 0)[0]
        # Sort pos1 by h value ascending (closest to 0.5 first)
        order = np.argsort(h[pos1])
        to_remove = target - 0  # need to remove (ones - target)
        for idx in order[:ones - target]:
            b[pos1[idx]] = 0
    else:
        # Add ones where h is closest to 0.5 from below
        pos0 = np.where(b == 0)[0]
        order = np.argsort(-h[pos0])  # descending h
        to_add = target - ones
        for idx in order[:to_add]:
            b[pos0[idx]] = 1
    if np.sum(b) != target:
        # Fix
        ones = np.sum(b)
        if ones > target:
            pos1 = np.where(b == 1)[0]
            order = np.argsort(h[pos1])
            for idx in order[:ones - target]:

        else:
            pos0 = np.where(b == 0)[0]
            order = np.argsort(-h[pos0])
            for idx in order[:target - ones]:

    return b

def binary_sa(h_init, n_steps, rng, time_limit):
    """Simulated annealing on balanced binary sequences."""
    t_start = time.time()
    n = len(h_init)
    if n % 2 != 0:
        return None, None
    
    b = binary_balance(h_init)
    if b is None:
        return None, None
    
    cur_score = official_score(b.astype(float))
    best = b.copy()
    best_score = cur_score
    
    temp = 8e-4
    cool = 0.999
    
    one_set = set(np.where(b == 1)[0].tolist())
    zero_set = set(np.where(b == 0)[0].tolist())
    
    for it in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        
        # Random swap
        if len(one_set) == 0 or len(zero_set) == 0:
            break
        a = rng.choice(list(one_set))
        c = rng.choice(list(zero_set))
        
        b[a] = 0
        b[c] = 1
        one_set.discard(a)
        one_set.add(c)
        zero_set.discard(c)
        zero_set.add(a)
        
        new_score = official_score(b.astype(float))
        delta = new_score - cur_score
        
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            cur_score = new_score
            if new_score < best_score:
                best = b.copy()
                best_score = new_score
        else:
            # Revert
            b[a] = 1
            b[c] = 0
            one_set.add(a)
            one_set.discard(c)
            zero_set.discard(a)
            zero_set.add(c)
        
        temp *= cool
    
    return best, best_score

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + s])
        grad[idx + s] -= h[idx]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + t])
        grad[idx + t] -= h[idx]
    return dx * grad

def build_lp(h, V, C, active_idx, rho):
    n = len(h)
    K = len(active_idx)
    if K == 0:
        return None
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for j, p in enumerate(active_idx):
        s = p - (n - 1)

        A_ub[j, :n] = g

    A_eq = np.zeros((1, n + 1))

def binary_sa(h_init, n_steps, rng, time_limit):
    t0 = time.time()
    n = len(h_init)
    if n % 2 != 0:
        return None, float('inf')
    b = binary_balance(h_init)
    if b is None:
        return None, float('inf')
    cur_score = official_score(b)
    best = b.copy()
    best_score = cur_score
    temp = 6e-4
    cool = 0.9992
    one_pos = np.where(b == 1)[0]
    zero_pos = np.where(b == 0)[0]
    for it in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        a = int(rng.integers(0, len(one_pos)))
        c = int(rng.integers(0, len(zero_pos)))
        ia = one_pos[a]
        ic = zero_pos[c]
        b[ia] = 0.0
        b[ic] = 1.0
        one_pos = np.delete(one_pos, a)
        one_pos = np.append(one_pos, ic)
        zero_pos = np.delete(zero_pos, c)
        zero_pos = np.append(zero_pos, ia)
        new_score = official_score(b)
        delta = new_score - cur_score
        if delta <= 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / temp)):
            cur_score = new_score
            if new_score < best_score:
                best = b.copy()
                best_score = new_score
        else:
            b[ia] = 1.0
            b[ic] = 0.0
            one_pos = np.delete(one_pos, np.where(one_pos == ic)[0][0])
            one_pos = np.append(one_pos, ia)
            zero_pos = np.delete(zero_pos, np.where(zero_pos == ia)[0][0])
            zero_pos = np.append(zero_pos, ic)
        temp *= cool
    return best, best_score

    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def two_point_search(h, n_tries, rng, time_deadline):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if time.time() > time_deadline:
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [(t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + 0.1 * span, t_lo + 0.9 * span,
                        t_lo + (1.0 / 2.618) * span]
        tmp = best.copy()
        best_local = best_sc
        best_t = None
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = official_score(best)
    return best, best_sc

def subgradient_polish(h, n_iter, time_deadline):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    alpha = 0.015
    for it in range(n_iter):
        if time.time() > time_deadline:
            break
        dx = 2.0 / n
        V = np.correlate(best, 1.0 - best, mode='full') * dx
        C = V.max()
        tol = 0.04
        active = np.where(V >= C - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(V)])
        w = np.exp(80.0 * (V[active] - C))
        w /= w.sum()
        grad = np.zeros(n)
        for k, p in enumerate(active):
            s = int(p) - (n - 1)
            g = lag_gradient(best, s)
            grad += w[k] * g
        d = -grad
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d /= d_norm
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.75]:
            h_trial = make_feasible(best + mult * d)
            sc = official_score(h_trial)
            if sc < best_sc - 1e-14:
                best = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:
            alpha *= 1.08
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        n_fb = 60
        x = (np.arange(n_fb) + 0.5) * 2.0 / n_fb
        h_fb = 0.5 + 0.25 * np.sin(2 * np.pi * 2 * x) + 0.08 * np.cos(2 * np.pi * 3 * x)
        best_h = make_feasible(h_fb)
        best_score = official_score(best_h)

    # Phase 1: LP refinement (primary method)
    if not is_dead():
        remaining = deadline - time.time()
        t_alloc = min(remaining * 0.35, 350.0)
        if t_alloc > 20:
            h1, s1 = lp_refine(best_h, t_alloc, n_top=32, eta_start=0.008)
            if s1 < best_score - 1e-10:
                best_h, best_score = h1, s1
    if not is_dead():
        remaining = deadline - time.time()
        t_alloc = min(remaining * 0.2, 200.0)
        if t_alloc > 15:
            h1b, s1b = lp_refine(best_h, t_alloc, n_top=48, eta_start=0.012)
            if s1b < best_score - 1e-10:
                best_h, best_score = h1b, s1b

    # Phase 2: Subgradient polish
    if not is_dead():
        best_h, best_score = subgradient_polish(best_h, 150, deadline)

    # Phase 3: 2-point search
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 4))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Phase 4: LP again with tighter params
    if not is_dead():
        remaining = deadline - time.time()
        t_alloc = min(remaining * 0.25, 200.0)
        if t_alloc > 15:
            h2, s2 = lp_refine(best_h, t_alloc, n_top=24, eta_start=0.004)
            if s2 < best_score - 1e-10:
                best_h, best_score = h2, s2

    # Phase 5: Binary search
    if not is_dead():
        remaining = deadline - time.time()
        t_bin = min(remaining * 0.2, 100.0)
        if t_bin > 10:
            b_best, b_sc = binary_sa(best_h, 4000, rng, t_bin)
            if b_best is not None and b_sc < best_score - 1e-10:
                h_bin = make_feasible(b_best)
                sc_bin = official_score(h_bin)
                if sc_bin < best_score - 1e-10:
                    best_h, best_score = h_bin, sc_bin
                # Also try continuous refinement from binary
                if not is_dead():
                    h_bin_lp, sc_bin_lp = lp_refine(h_bin, 60, n_top=32, eta_start=0.01)
                    if sc_bin_lp < best_score - 1e-10:
                        best_h, best_score = h_bin_lp, sc_bin_lp

    # Phase 6: Resolution scanning
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set(
            [n_base + d for d in [-8, -4, -2, 2, 4, 8] if 30 <= n_base + d <= 90]
            + [40, 48, 52, 56, 60, 64, 68, 72, 76]
        ))
        for no in n_opts:
            if is_dead():
                break
            if no == n_base:
                continue
            h_int = interp_to(best_h, no)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-10:
                best_h = h_int
                best_score = sc_int
            if not is_dead():
                h_new, sc = two_point_search(h_int, 600, rng, deadline)
                if sc < best_score - 1e-10:
                    best_h = h_new
                    best_score = sc
                if not is_dead():
                    h_lp, sc_lp = lp_refine(h_new, 80, n_top=32, eta_start=0.006)
                    if sc_lp < best_score - 1e-10:
                        best_h, best_score = h_lp, sc_lp

    # Phase 7: Final LP polish
    if not is_dead():
        remaining = deadline - time.time()
        t_final = min(remaining * 0.4, 150.0)
        if t_final > 10:
            h_fin, s_fin = lp_refine(best_h, t_final, n_top=32, eta_start=0.003)
            if s_fin < best_score - 1e-10:
                best_h, best_score = h_fin, s_fin

    # Phase 8: Final 2-point + subgradient
    if not is_dead():
        best_h, best_score = subgradient_polish(best_h, 100, deadline)
    if not is_dead():
        n_tries = max(1500, int((deadline - time.time()) * 3))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)) or final_h.min() < -1e-8 or final_h.max() > 1.0 + 1e-8:
        final_h = make_feasible(np.full(60, 0.5))
        final_score = official_score(final_h)
        final_n = 60

    return final_h, final_score, final_n
