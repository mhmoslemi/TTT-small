def lp_minimax_step(h, tolerance=0.02, delta=0.1):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    
    # Active set: lags within tolerance of max
    active = np.where(vals >= cmax - tolerance)[0]
    
    # Compute gradients for active lags
    npts = 2 * n - 1
    grads = np.zeros((len(active), n))
    for idx, p in enumerate(active):
        s = p - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            for j in range(n - s):
                g[j] += dx * (1.0 - h[j + s])
                g[j + s] -= dx * h[j]
        else:
            t = -s
            for j in range(n - t):
                g[j + t] += dx * (1.0 - h[j])
                g[j] -= dx * h[j + t]
        grads[idx] = g
    
    # LP: maximize r
    # subject to: grads[i] @ d + r <= cmax - vals[active[i]] for all i
    #             sum(d) = 0
    #             -h[j] <= d[j] <= 1-h[j] for all j
    #             r >= 0
    
    # Using scipy.optimize.linprog
    # Variables: d[0..n-1], r
    # Total: n+1 variables
    
    from scipy.optimize import linprog
    
    # Maximize r => minimize -r
    c = np.zeros(n + 1)
    c[n] = -1.0  # coefficient of r in objective (minimize -r)
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag i: grads[i] @ d + r <= cmax - vals[active[i]]
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for i in range(len(active)):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = 1.0
        b_ub[i] = cmax - vals[active[i]]
    
    # Also add r >= 0 => -r <= 0
    A_ub = np.vstack([A_ub, np.array([np.zeros(n), -1.0])])
    b_ub = np.append(b_ub, 0.0)
    
    # Equality constraints: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -h[j] <= d[j] <= 1-h[j], r >= 0 (unbounded above)
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)] + [(0, None)]

def make_feasible(z):

    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        # Golden ratio search for optimal t
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        t1 = t_lo + gr * span
        t2 = t_lo + (1.0 - gr) * span
        tmp = best.copy()
        
        # Evaluate at t1 and t2
        tmp[a] = best[a] + t1
        tmp[b] = best[b] - t1
        sc1 = official_score(tmp)
        tmp[a] = best[a] + t2
        tmp[b] = best[b] - t2
        sc2 = official_score(tmp)
        
        # A few more points
        best_local = min(sc1, sc2, best_sc)
        best_t = 0.0
        for t in [t1, t2, (t_lo+t_hi)/2, t_lo+0.1*span, t_lo+0.9*span]:
            tmp[a] = best[a] + t
            tmp[b] = best[b] - t
            if tmp[a] < -1e-10 or tmp[a] > 1+1e-10 or tmp[b] < -1e-10 or tmp[b] > 1+1e-10:
                continue
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        
        if abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = official_score(best)
    
    return best, best_sc

def make_feasible(z):

    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
