```python
            # j from d to n-1
            direction[d:] += wd * h[:n-d]
        else:
            # j from 0 to n+d-1 (i.e., n+|d|-1)
            direction[:n+d] += wd * h[-d:]
    return direction
```

Now let me think about the step size. For the subgradient method on a minimax problem, a good approach is:

1. **Exponentiated gradient step**: Update weights w_d ∝ exp(η * score_d)
2. **Subgradient step with line search**: Take direction, do line search on actual objective

The line search is crucial. Since the objective is piecewise bilinear (max of bilinear forms), the line search function α ↦ F(h + α*dir) is the max of quadratic functions of α (each score_d becomes a quadratic in α along the direction). This is unimodal in the feasible range (since each quadratic is concave or convex... actually they're bilinear so along a line they're quadratic).

Wait, score_d(h+α*dir) = sum_i (h[i]+α*dir[i])*(1-h[i+d]-α*dir[i+d])
= score_d(h) + α*sum_i[dir[i]*(1-h[i+d]) - (h[i]+α*dir[i])*... 

Let me expand: = sum_i [h[i] + α*dir[i]] * [1 - h[i+d] - α*dir[i+d]]
= sum_i [h[i]*(1-h[i+d]) - α*h[i]*dir[i+d] + α*dir[i]*(1-h[i+d]) - α²*dir[i]*dir[i+d]]
= score_d(h) + α * sum_i [dir[i]*(1-h[i+d]) - h[i]*dir[i+d]] - α² * sum_i [dir[i]*dir[i+d]]

So each score_d is a quadratic in α: A_d - B_d*α - C_d*α² (approximately, with sign depending on convention).

Actually: score_d(α) = score_d(0) + α*G_d - α²*Q_d
where G_d = sum_i [dir[i]*(1-h[i+d]) - h[i]*dir[i+d]] and Q_d = sum_i dir[i]*dir[i+d]

Since Q_d ≥ 0 (it's a sum of products of dir[i] and dir[i+d], and for d=0 it's sum dir[i]² ≥ 0; for other d it could be negative... actually no, it's a sum of products, not necessarily non-negative).

Hmm, so the objective max_d score_d(α) is the max of quadratic functions, which may not be unimodal. But in practice, for small α, it should be well-behaved.

```python
def _subgrad_step(h, w, eta=1.0):
    """One subgradient step with weight update and line search."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Exponentiated weight update
    max_s = np.max(scores)
    new_w = np.exp(eta * (scores - max_s))
    new_w /= np.sum(new_w)
    
    # Compute descent direction
    direction = _neg_weighted_grad(h, new_w)
    
    # Center to maintain sum constraint
    direction -= np.mean(direction)
    
    # Scale direction
    max_abs = np.max(np.abs(direction))
    if max_abs < 1e-15:
        return h, new_w, _c5(h)
    direction /= max_abs
    
    # Line search
    c5_cur = _c5(h)
    # Find alpha range
    alpha_max = 0.5
    for j in range(n):
        if direction[j] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-12:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-6)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(h + c1 * direction)
    f2 = _c5(h + c2 * direction)
    for _ in range(30):
        if hi - lo < 1e-8:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(h + c1 * direction)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(h + c2 * direction)
    
    alpha = (lo + hi) / 2.0
    new_h = h + alpha * direction
    new_h = np.clip(new_h, 0, 1)
    new_h = _project(new_h, n / 2.0)
    
    return new_h, new_w, _c5(new_h)
```

```python
import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _neg_weighted_grad(h, w):
    n = len(h)
    omh = 1.0 - h
    direction = np.zeros(n)
    for d_idx in range(2 * n - 1):
        d = d_idx - (n - 1)
        wd = w[d_idx]
        if abs(wd) < 1e-15:
            continue
        if d >= 0:
            direction[:n - d] += -wd * omh[d:]
            direction[d:] += wd * h[:n - d]
        else:
            direction[-d:] += -wd * omh[:-d]
            direction[:n + d] += wd * h[-d:]
    return direction

def _line_search_c5(h, direction, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if direction[j] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-12:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-8)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(h + c1 * direction)
    f2 = _c5(h + c2 * direction)
    for _ in range(35):
        if hi - lo < 1e-9:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(h + c1 * direction)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(h + c2 * direction)
    alpha = (lo + hi) / 2.0
    best_f = min(f1, f2, c5_cur)
    f_half = _c5(h + 0.5 * (lo + hi) * direction)
    if f_half < best_f:
        best_f = f_half
        alpha = 0.5 * (lo + hi)
    return alpha, best_f

def _subgrad_optimize(h, deadline, eta_init=2.0, eta_rate=1.002, momentum=0.7, max_iters=500000):
    n = len(h)
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    w = np.ones(2 * n - 1) / (2 * n - 1)
    momentum_dir = np.zeros(n)
    eta = eta_init
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        scores = _scores(h)
        max_s = float(np.max(scores))
        new_w = np.exp(eta * (scores - max_s))
        ssum = float(np.sum(new_w))
        if ssum < 1e-300:
            new_w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            new_w /= ssum
        w = 0.5 * w + 0.5 * new_w
        w /= np.sum(w)
        direction = _neg_weighted_grad(h, w)
        direction -= np.mean(direction)
        momentum_dir = momentum * momentum_dir + (1 - momentum) * direction
        step_dir = momentum_dir
        max_ab = float(np.max(np.abs(step_dir)))
        if max_ab < 1e-14:
            stall += 1
            if stall > 200:
                break
            eta *= 1.5
            continue
        step_dir /= max_ab
        alpha, new_c5 = _line_search_c5(h, step_dir, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * step_dir
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, n / 2.0)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            eta = min(eta * 1.001, 50.0)
        else:
            stall += 1
            eta = max(eta * 0.95, 0.5)
            if stall > 500:
                break
        it += 1
    return best_h, best_c5

def _binary_anneal(h, deadline, max_iters=200000):
    n = len(h)
    target = n // 2
    if n % 2 != 0:
        target = (n + 1) // 2
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    cur_c5 = _c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    T = 0.1
    T_min = 0.001
    cooling = (T_min / T) ** (1.0 / max(max_iters, 1))
    rng = np.random.default_rng(123)
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        b[i] = 0.0
        b[j] = 1.0
        new_c5 = _c5(b)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            ones = np.where(b == 1.0)[0]
            zeros = np.where(b == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        T *= cooling
    return best_b, best_c5

def _pair_polish(h, deadline, max_passes=50):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    improved_any = False
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores_cache = _scores(h)
        for j in range(n):
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(n - 1, j, -1):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if dj > 0 and h[j] + dj > 1.0:
                        continue
                    if dj < 0 and h[j] + dj < 0.0:
                        continue
                    if dk > 0 and h[k] + dk > 1.0:
                        continue
                    if dk < 0 and h[k] + dk < 0.0:
                        continue
                    h_tmp = h.copy()
                    h_tmp[j] += dj
                    h_tmp[k] += dk
                    c5_tmp = _c5(h_tmp)
                    if c5_tmp < best_c5 - 1e-12:
                        best_c5 = c5_tmp
                        best_h = h_tmp.copy()
                        h = h_tmp.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            if not improved_any:
                break
            improved_any = False
        else:
            improved_any = True
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _try_different_eta(h, deadline, eta_list):
    n = len(h)
    best_h = h.copy()

        if c5_try < best_c5 - 1e-10:
```
