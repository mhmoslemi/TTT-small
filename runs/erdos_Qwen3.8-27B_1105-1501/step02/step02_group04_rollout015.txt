So the linear parts are fully captured, and the residual quadratic is +t². ✓

Similarly for k=j-i+offset, the term is h[j]*(1-h[i]), and after: (h[j]-t)(1-(h[i]+t)) = h[j](1-h[i]) - t(1-h[i]) - t*h[j] + t².
Linear: -t(1-h[i]) from -t*∂/∂h[j] (the first term of ∂/∂h[j] gives (1-h[j-k+offset]) = (1-h[i])), and -t*h[j] from t*∂/∂h[i] (the second term gives -h[i+k-offset] = -h[j]).
Residual: +t². ✓

So the exact formula is:
objective(t) / dx = max_k [corr[k] + t * L[k] + t² * Q[k]]

where L[k] = ∂corr[k]/∂h[i] - ∂corr[k]/∂h[j] and Q[k] is 1 at two positions, 0 elsewhere.

This is a max of quadratics with non-negative leading coefficients, hence convex in t. 

For the line search, I'll use ternary search on [t_lo, t_hi] with about 40-60 iterations (converging to within ~1e-15 of optimal).

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum} via bisection."""
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    """Compute the full correlation vector (without dx factor)."""
    return np.correlate(h, 1.0 - h, mode="full")

def pair_line_search(corr, h, i, j, n, iters=50):
    """
    For the transfer h[i] += t, h[j] -= t, find t that minimizes max(corr + t*L + t^2*Q).
    Uses ternary search on the convex function.
    Returns (best_t, best_value).
    """
    offset = n - 1
    m = 2 * n - 1  # length of corr
    
    # Compute L[k] = d(corr[k])/d(h[i]) - d(corr[k])/d(h[j])
    # d(corr[k])/d(h[a]) = (1-h[a-k+offset]) * 1_{0<=a-k+offset<n} - h[a+k-offset] * 1_{0<=a+k-offset<n}
    
    L = np.zeros(m)
    for k in range(m):
        # Contribution from h[i]
        idx1 = i - k + offset  # = i - (k - offset) = i - s
        idx2 = i + k - offset  # = i + s
        di = 0.0
        if 0 <= idx1 < n:
            di += 1.0 - h[idx1]
        if 0 <= idx2 < n:
            di -= h[idx2]
        
        # Contribution from h[j]
        idx3 = j - k + offset
        idx4 = j + k - offset
        dj = 0.0
        if 0 <= idx3 < n:
            dj += 1.0 - h[idx3]
        if 0 <= idx4 < n:
            dj -= h[idx4]
        
        L[k] = di - dj
    
    # Q positions
    q1 = i - j + offset  # = i - j + n - 1
    q2 = j - i + offset  # = j - i + n - 1
    
    # Feasible range for t
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    
    if t_hi - t_lo < 1e-13:
        return 0.0, float(np.max(corr))
    
    # Ternary search
    a, b = t_lo, t_hi
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        # Evaluate objective at m1 and m2
        # objective(t) = max(corr + t*L + t^2*Q)
        # Q is 1 at q1 and q2, 0 elsewhere
        v1 = corr + m1 * L
        v1[q1] += m1 * m1
        v1[q2] += m1 * m1
        f1 = float(np.max(v1))
        
        v2 = corr + m2 * L
        v2[q1] += m2 * m2
        v2[q2] += m2 * m2
        f2 = float(np.max(v2))
        
        if f1 < f2:
            b = m2
        else:
            a = m1
    
    t_opt = (a + b) / 2.0
    v = corr + t_opt * L
    v[q1] += t_opt * t_opt
    v[q2] += t_opt * t_opt
    best_val = float(np.max(v))
    
    return t_opt, best_val

def coordinate_descent(h, n, max_sweeps=100, deadline=None, rng=None, verbose=False):
    """
    Run coordinate descent with exact line search.
    Returns (best_h, best_c5).
    """
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = compute_corr(best_h)
    current_c5 = float(np.max(corr))  # without dx
    
    sweep = 0
    stall = 0
    
    while sweep < max_sweeps:
        if deadline is not None and time.time() >= deadline:
            break
        
        improved = False
        perm = rng.permutation(n)
        
        for a_idx in range(n):
            if deadline is not None and time.time() >= deadline:
                break
            for b_idx in range(a_idx + 1, n):
                i = perm[a_idx]
                j = perm[b_idx]
                
                t_opt, new_max = pair_line_search(corr, best_h, i, j, n, iters=45)
                
                current_max = float(np.max(corr))
                if new_max < current_max - 1e-10:
                    # Accept the move
                    best_h[i] += t_opt
                    best_h[j] -= t_opt
                    # Clip to handle numerical issues
                    best_h[i] = np.clip(best_h[i], 0.0, 1.0)
                    best_h[j] = np.clip(best_h[j], 0.0, 1.0)
                    # Recompute corr
                    corr = compute_corr(best_h)
                    improved = True
                    if np.max(corr) < best_c5:
                        best_c5 = float(np.max(corr))
        
        sweep += 1
        if not improved:
            stall += 1
            if stall >= 3:
                # Restart with perturbation
                break
        else:
            stall = 0
    
    # Final feasibility check
    best_h = project_onto_constraint(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def interpolate_to_size(h, new_n):
    """Interpolate h from its current size to new_n."""
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, new_n)
    new_h = np.interp(new_x, old_x, h)
    return project_onto_constraint(new_h, new_n / 2.0)

    h = np.zeros(n)

  Upper: i+k-n+1 < n → k < 2n-1-i → k <= 2n-2-i
  Combined with 0 <= k <= 2n-2:
  k ranges from max(0, n-1-i) to min(2n-2, 2n-2-i) = n-1-i to 2n-2-i (since i>=0, n-1-i <= n-1, and 2n-2-i >= n-1).
  
  Values: -h[i+k-offset] = -h[i+k-n+1]. Let j = i+k-n+1. When k=n-1-i: j = 0. When k=2n-2-i: j = n-1.
  So j ranges from 0 to n-1 as k goes from n-1-i to 2n-2-i.
  Values: -h[0], -h[1], ..., -h[n-1].
  
  So: For t = 0, 1, ..., n-1: grad_i[n-1-i+t] += -h[t]

Let me verify with the example. n=3, offset=2, i=0:

First part: k from 0 to 0+2=2. Values: (1-h[2]), (1-h[1]), (1-h[0]).

So grad_0[2] += -h[0], grad_0[3] += -h[1], grad_0[4] += -h[2].

Total grad_0 = [
  (1-h[2]),
