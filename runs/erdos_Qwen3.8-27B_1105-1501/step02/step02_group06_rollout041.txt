def gradient_of_f(h, s):
    """Gradient of f_s = Σ_i h[i]*(1-h[i+s]) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    # (1-h[k+s]) for valid k+s
    if s >= 0:
        for k in range(n - s):
            g[k] += 1.0 - h[k + s]
        for k in range(s, n):
            g[k] -= h[k - s]
    else:
        t = -s
        for k in range(t, n):
            g[k] += 1.0 - h[k - t]
        for k in range(n - t):
            g[k] -= h[k + t]
    return g

def make_feasible(z):

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_of_f(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        t = -s
        g[t:] += 1.0 - h[:n - t]
        g[:n - t] -= h[t:]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

            g += w[p] * dx * gradient_of_f(h, s)

def slsqp_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(h):
    return np.ones(len(h))

def try_slsqp_lse(h0, beta, maxiter=300):

    n = len(h0)
    h_cur = h0.copy()
    g_mom = np.zeros(n)
    best_h = h_cur.copy()
    best_sc = official_score(h_cur)
    step = step_start
    stall = 0
    tol = 1e-14
    
    for it in range(iters):
        vals = compute_all_f(h_cur)
        f_max = vals.max()
        sorted_idx = np.argsort(vals)[::-1]
        top_k = sorted_idx[:k_top]

            h_try2 = make_feasible(h_cur - 0.4 * step * g_mom)
            sc2 = official_score(h_try2)
            if sc2 < best_sc - tol:
                best_h = h_try2.copy()
                best_sc = sc2
                h_cur = h_try2
            stall += 1
        
        step *= step_decay
        if stall > 60:
            break
    
    return best_h, best_sc

def anti_sym_transform(h):
    return make_feasible(1.0 - h[::-1].copy())

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 10.0
    rng = np.random.default_rng(seed)
    
    def remaining():
        return deadline - time.time()
    
    best_h = None
    best_score = float('inf')
    
    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
    
    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass
    
    if best_h is None:
        n0 = 60
        best_h = np.full(n0, 0.5)
        best_score = official_score(best_h)
    update(best_h, best_score)
    
    # Phase 1: LSE-SLSQP continuation (main smooth optimizer)
    for beta in [50, 200, 1000, 5000, 20000, 50000, 100000]:
        if remaining() < 30:
            break
        h_new = try_slsqp_lse(best_h, beta, maxiter=400)
        if h_new is not None:
            sc = official_score(h_new)
            update(h_new, sc)
    
    # Phase 2: Cutting-plane QP with growing active set
    if remaining() > 60:
        h_cp, sc_cp = cutting_plane_h(best_h, max_rounds=40, tol=1e-9)
        update(h_cp, sc_cp)
    
    # Phase 3: Subgradient exploration with different parameters
    for k_t in [15, 30, 50]:
        if remaining() < 40:
            break
        for beta_sg in [30, 100, 300]:
            if remaining() < 20:
                break
            h_sg, sc_sg = subgradient_hedge(best_h, iters=200, k_top=k_t,
                                            beta=beta_sg, momentum=0.75,
                                            step_start=0.3, step_decay=0.997)
            update(h_sg, sc_sg)
            # Follow up with LSE from subgradient result
            if sc_sg < best_score + 0.003:
                h_new = try_slsqp_lse(h_sg, 5000, maxiter=200)
                if h_new is not None:
                    update(h_new, official_score(h_new))
    
    # Phase 4: Additional QP rounds
    if remaining() > 60:
        h_cp, sc_cp = cutting_plane_h(best_h, max_rounds=40, tol=1e-10)
        update(h_cp, sc_cp)
    
    # Phase 5: Anti-symmetric transform
    if remaining() > 40:
        h_sym = anti_sym_transform(best_h)
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        h_cp, sc_cp = cutting_plane_h(h_sym, max_rounds=20, tol=1e-9)
        update(h_cp, sc_cp)
        h_new = try_slsqp_lse(h_sym, 10000, maxiter=200)
        if h_new is not None:
            update(h_new, official_score(h_new))
    
    # Phase 6: Resolution sweep
    if remaining() > 100:
        n_base = len(best_h)
        n_opts = sorted(set([max(30, n_base - 10), max(30, n_base - 5),
                             n_base, n_base + 5, n_base + 10, n_base + 20, n_base + 40]))
        for n_opt in n_opts:
            if remaining() < 30:
                break
            if n_opt < 20 or n_opt > 150:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            # QP from interpolated
            h_cp, sc_cp = cutting_plane_h(h_int, max_rounds=20, tol=1e-8)
            update(h_cp, sc_cp)
            # LSE from interpolated
            h_new = try_slsqp_lse(h_int, 5000, maxiter=200)
            if h_new is not None:
                update(h_new, official_score(h_new))
    
    # Phase 7: Random perturbations + local optimization
    if remaining() > 80:
        for i in range(10):
            if remaining() < 20:
                break
            sigma = 0.01 + 0.05 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_cp, sc_cp = cutting_plane_h(h_cand, max_rounds=15, tol=1e-8)
            update(h_cp, sc_cp)
            if remaining() > 20:
                h_new = try_slsqp_lse(h_cp, 5000, maxiter=150)
                if h_new is not None:
                    update(h_new, official_score(h_new))
    
    # Phase 8: Final intensive polish
    if remaining() > 30:
        for beta_f in [20000, 50000, 100000]:
            if remaining() < 10:
                break
            h_new = try_slsqp_lse(best_h, beta_f, maxiter=300)
            if h_new is not None:
                update(h_new, official_score(h_new))
        h_cp, sc_cp = cutting_plane_h(best_h, max_rounds=30, tol=1e-10)
        update(h_cp, sc_cp)
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
