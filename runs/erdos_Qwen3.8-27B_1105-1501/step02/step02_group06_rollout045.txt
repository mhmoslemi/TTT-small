def grad_for_lag(h, p):

    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    d = p - (n - 1)  # offset
    if d >= 0:
        for i in range(n - d):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    else:
        for i in range(-d, n):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    return g

        h_var <= 1

Actually, I just realized something important. The `grad_for_lag` function needs to be correct. Let me verify:

`np.correlate(h, 1-h, 'full')` gives the cross-correlation. For the p-th element (0-indexed), with n = len(h):
- offset d = p - (n-1)
- The p-th element = sum over valid i of h[i] * (1-h)[i+d] = sum over valid i of h[i] * (1 - h[i+d])

where valid means 0 ≤ i < n and 0 ≤ i+d < n.

The derivative with respect to h[j]:
- If j is the "first" index in a valid pair (j, j+d): contribution is (1-h[j+d])
- If j is the "second" index in a valid pair (j-d, j): contribution is -h[j-d]
- (Note: if d < 0, these swap roles)

More precisely, for offset d:
- Valid pairs are (i, i+d) where 0≤i<n and 0≤i+d<n
- d/dh[i] of h[i]*(1-h[i+d]) = (1-h[i+d])

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_for_lag(h, p):
    n = len(h)
    dx = 2.0 / n
    d = p - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        for i in range(n - d):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    else:
        for i in range(-d, n):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    return g

def all_grads(h):
    n = len(h)
    return [grad_for_lag(h, p) for p in range(2 * n - 1)]

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:

                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def solve_lp_qp(h0, active_set, reg):
    n = len(h0)
    f0 = compute_f(h0)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in active_set:
        g = grad_for_lag(h0, p)
        constraints.append(f0[p] + g @ (h_var - h0) <= t_var)
    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return np.clip(h_new, 0.0, 1.0)

def line_search(h0, direction, n_samples=25):
    best_alpha = 0.0
    best_sc = official_score(h0)
    for alpha in np.linspace(0.0, 1.0, n_samples):
        h_try = make_feasible(h0 + alpha * direction)
        sc = official_score(h_try)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_alpha = alpha
    return best_alpha, best_sc

def cut_based_optimize(h0, max_rounds=40, initial_k=4, eps_expand=0.0008,
                       reg_start=2.0, reg_end=0.002, time_limit=60, t_start=None):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    f = compute_f(h)
    f_max = f.max()
    active = list(np.argsort(f)[-initial_k:])

    for rnd in range(max_rounds):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        reg = reg_start * (reg_end / reg_start) ** (rnd / max(1, max_rounds - 1))
        h_lp = solve_lp_qp(h, active, reg)
        if h_lp is None:
            break
        direction = h_lp - h
        if np.max(np.abs(direction)) < 1e-13:
            break
        alpha, sc_ls = line_search(h, direction)
        if alpha > 0.01:
            h = make_feasible(h + alpha * direction)
            best = min(best, official_score(h))
            f = compute_f(h)
            f_max = f.max()
            new_active = [p for p in range(2 * n - 1) if f[p] > f_max - eps_expand]
            if len(new_active) > 30:

            active = new_active
        else:

    return h, best

def coordinate_descent(h0, max_iter=50, delta=0.005, time_limit=40, t_start=None):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    for it in range(max_iter):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        improved = False
        order = np.random.default_rng(it).permutation(n)
        for i in order:
            for j in range(n):
                if i == j:
                    continue
                for sign in [1, -1]:
                    hi = h[i] + sign * delta
                    hj = h[j] - sign * delta
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i] = hi
                    h[j] = hj
                    sc = official_score(h)
                    if sc < best - 1e-14:
                        best = sc
                        improved = True
                    else:
                        h[i] -= sign * delta
                        h[j] += sign * delta
        if not improved:
            if delta > 0.0005:
                delta *= 0.6
            else:
                break
    return h, best

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))
