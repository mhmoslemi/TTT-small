```python
import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection for shift
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def to_binary(h):
    """Convert to balanced binary (for even n) or half-binary (for odd n)."""
    n = len(h)
    if n % 2 == 0:
        k = n // 2
        idx = np.argsort(h)[::-1]
        b = np.zeros(n)
        b[idx[:k]] = 1.0
        return b
    else:
        k = n // 2  # floor(n/2) = target - 0.5
        idx = np.argsort(h)[::-1]
        b = np.zeros(n)
        b[idx[:k]] = 1.0
        b[idx[k]] = 0.5
        return b

def block_search_binary(h, block_sizes, max_passes=5, rng=None, time_limit=None, t_start=None):
    """Exact block-level local search for binary vectors."""
    n = len(h)
    h = h.copy()
    best_score = exact_score(h)
    
    for block_size in block_sizes:
        for _pass in range(max_passes):
            if time_limit is not None and t_start is not None:
                if time.time() - t_start > time_limit:
                    break
            improved = False
            # Try all windows
            starts = rng.permutation(n - block_size + 1) if rng is not None else range(n - block_size + 1)
            for start in starts:
                if time_limit is not None and t_start is not None:
                    if time.time() - t_start > time_limit:
                        break
                end = start + block_size
                block = h[start:end]
                k = int(np.sum(block))
                if k <= 0 or k >= block_size:
                    continue
                # Enumerate all placements of k ones in block_size positions
                positions = list(range(block_size))
                best_local = best_score
                best_placement = None
                for combo in combinations(positions, k):
                    new_block = np.zeros(block_size)
                    new_block[list(combo)] = 1.0
                    new_h = h.copy()
                    new_h[start:end] = new_block
                    s = exact_score(new_h)
                    if s < best_local - 1e-14:
                        best_local = s
                        best_placement = new_block.copy()
                if best_placement is not None:
                    h[start:end] = best_placement
                    best_score = best_local
                    improved = True
            if not improved:
                break
    return h, best_score
```

```python
                order = rng.permutation(n_windows)
            else:
                order = np.arange(n_windows)
            for start in order:
                if deadline is not None and time.time() > deadline:
                    return h, best_score
                block = h[start:start + bs]
                k = int(round(np.sum(block)))
                if k <= 0 or k >= bs:
                    continue
                best_local = best_score
                best_block = None
                cur_tuple = tuple(block.astype(int))
                for combo in combinations(range(bs), k):
                    c = set(combo)
                    new_block = np.array([1.0 if i in c else 0.0 for i in range(bs)])
                    if tuple(new_block.astype(int)) == cur_tuple:
                        continue
                    new_h = h.copy()
                    new_h[start:start + bs] = new_block
                    s = exact_score(new_h)
                    if s < best_local - 1e-14:
                        best_local = s
                        best_block = new_block
                if best_block is not None:
                    h[start:start + bs] = best_block
                    best_score = best_local
                    improved = True
            if not improved:
                break
    return h, best_score

def block_search_parity(h, block_sizes, max_passes=4, rng=None, deadline=None):
    """Block search for odd-n half-binary vectors (one 0.5)."""
    n = len(h)
    h = h.copy()
    best_score = exact_score(h)
    half_idx = np.where(np.abs(h - 0.5) < 0.01)[0]
    for bs in block_sizes:
        if bs <= 0 or bs > n:
            continue
        for _pass in range(max_passes):
            if deadline is not None and time.time() > deadline:
                return h, best_score
            improved = False
            n_windows = n - bs + 1
            if rng is not None:
                order = rng.permutation(n_windows)
            else:
                order = np.arange(n_windows)
            for start in order:
                if deadline is not None and time.time() > deadline:
                    return h, best_score
                end = start + bs
                block = h[start:end]
                k_full = int(np.sum(block >= 0.99))
                has_half = any(np.abs(block - 0.5) < 0.01 for _ in [1])
                n_half = int(np.sum(np.abs(block - 0.5) < 0.01))
                if has_half and n_half == 1:
                    total_in_block = k_full + 0.5
                    k_pos = k_full
                    best_local = best_score
                    best_block = None
                    for half_pos in range(bs):
                        for combo in combinations([i for i in range(bs) if i != half_pos], k_pos):
                            new_block = np.zeros(bs)
                            new_block[half_pos] = 0.5
                            for c in combo:
                                new_block[c] = 1.0
                            new_h = h.copy()
                            new_h[start:end] = new_block
                            s = exact_score(new_h)
                            if s < best_local - 1e-14:
                                best_local = s
                                best_block = new_block
                else:
                    k = int(round(np.sum(block)))
                    if k <= 0 or k >= bs:
                        continue
                    best_local = best_score
                    best_block = None
                    for combo in combinations(range(bs), k):
                        new_block = np.zeros(bs)
                        new_block[list(combo)] = 1.0
                        new_h = h.copy()
                        new_h[start:end] = new_block
                        s = exact_score(new_h)
                        if s < best_local - 1e-14:
                            best_local = s
                            best_block = new_block
                if best_block is not None:
                    h[start:end] = best_block
                    best_score = best_local
                    improved = True
            if not improved:
                break
    return h, best_score

def slsqp_polish(h0, beta=200, maxiter=80, deadline=None):
    n = len(h0)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            return m
        return m + np.log(s) / beta

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights = e / s
        g = np.zeros(n)
        for l in range(2 * n - 1):
            w = weights[l] * dx
            if w < 1e-12:
                continue
            s2 = l - (n - 1)
            if s2 == 0:
                g += (1.0 - x) - 2.0 * x + x
                g[:]= (1.0 - x) - x
            elif s2 > 0:
                g[s2:] += (1.0 - x[:n - s2])
                g[:n - s2] += (-x[s2:])
            else:
                m2 = -s2
                g[:n - m2] += (1.0 - x[m2:])
                g[m2:] += (-x[:n - m2])
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def kick_binary(h, rng, n_kicks=4):
    """Randomly flip pairs of bits (1<->0) to create a perturbation."""
    n = len(h)
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) < n_kicks or len(zeros) < n_kicks:
        return h
    for _ in range(n_kicks):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        if i != j:
            h[i], h[j] = h[j], h[i]
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(25.0, 0.04 * budget_s)
    deadline = t0 + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        n_fb = 50
        rng_fb = np.random.default_rng(0)
        parent = np.zeros(n_fb)
        parent[rng_fb.choice(n_fb, n_fb // 2, replace=False)] = 1.0

    best_h = parent.copy()
    best_score = exact_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    n_options = sorted(set([48, 50, 52, 54, 56, 58, 60]))
    pn = len(parent)
    if pn not in n_options and 24 <= pn <= 76:
        n_options.append(pn)
    n_options = sorted(n_options)

    for n in n_options:
        if time.time() > deadline - 20:
            break
        time_left = deadline - time.time()
        if time_left < 30:
            break

        starts = []
        h_i = make_feasible(interp_to_n(parent, n))
        starts.append(h_i)

        if n % 2 == 0:
            starts.append(to_binary_even(h_i))
            rb = np.zeros(n)
            rb[rng.choice(n, n // 2, replace=False)] = 1.0
            starts.append(rb)
            for period in [6, 8, 12]:
                pat = np.zeros(n)
                for i in range(n):
                    pat[i] = 1.0 if (i % period) < (period // 2) else 0.0
                cur_ones = int(np.sum(pat))
                if cur_ones > n // 2:
                    oi = np.where(pat == 1.0)[0]
                    pat[oi[cur_ones - n // 2:]] = 0.0
                elif cur_ones < n // 2:
                    zi = np.where(pat == 0.0)[0]
                    pat[zi[:n // 2 - cur_ones]] = 1.0
                starts.append(pat)
            starts.append(to_binary_even(h_i ** 4.0))
        else:
            starts.append(to_binary_odd(h_i))
            starts.append(to_binary_odd(h_i ** 4.0))

        n_starts = min(len(starts), 4 if time_left > 120 else 2)

        for si in range(n_starts):
            if time.time() > deadline - 15:
                break
            h0 = starts[si]
            cur_h = h0.copy()
            cur_score = exact_score(cur_h)

            bs_list = [6, 8, 10]
            if time_left > 100:
                bs_list.append(12)
            if time_left > 200:
                bs_list.append(14)

            passes = 5 if n <= 52 else 4
            passes_large = 3 if n <= 52 else 2

            if n % 2 == 0:
                cur_h, cur_score = block_search(cur_h, bs_list[:3], max_passes=passes,
                                                rng=rng, deadline=deadline - 15)
                if len(bs_list) > 3 and time.time() < deadline - 20:
                    cur_h, cur_score = block_search(cur_h, bs_list[3:], max_passes=passes_large,
                                                    rng=rng, deadline=deadline - 15)
            else:
                cur_h, cur_score = block_search_parity(cur_h, bs_list[:3], max_passes=passes,
                                                       rng=rng, deadline=deadline - 15)
                if len(bs_list) > 3 and time.time() < deadline - 20:
                    cur_h, cur_score = block_search_parity(cur_h, bs_list[3:], max_passes=passes_large,
                                                           rng=rng, deadline=deadline - 15)

            if n % 2 == 0 and time.time() < deadline - 20:
                b_h = to_binary_even(cur_h)
                b_s = exact_score(b_h)
                if b_s < cur_score:
                    cur_h = b_h
                    cur_score = b_s
                cur_h, cur_score = block_search(cur_h, [8, 10], max_passes=3,
                                                rng=rng, deadline=deadline - 15)

            if time.time() < deadline - 25:
                for beta in [150, 500]:
                    if time.time() > deadline - 25:
                        break
                    new_h = slsqp_polish(cur_h, beta=beta, maxiter=60, deadline=deadline)
                    new_s = exact_score(new_h)
                    if new_s < cur_score - 1e-13:
                        cur_h = new_h
                        cur_score = new_s

            # Iterated local search: kick and re-search
            if n % 2 == 0 and time.time() < deadline - 30:
                n_kicks = max(1, int(time_left / 60))
                for _ in range(n_kicks):
                    if time.time() > deadline - 20:
                        break
                    kicked = kick_binary(cur_h, rng, n_kicks=min(6, n // 8))
                    kicked_score = exact_score(kicked)
                    if kicked_score < cur_score + 0.05:
                        kicked, kicked_score = block_search(kicked, [8, 10], max_passes=4,
                                                             rng=rng, deadline=deadline - 15)
                        if kicked_score < cur_score - 1e-13:
                            cur_h = kicked
                            cur_score = kicked_score

            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_h = cur_h.copy()
                best_n = len(cur_h)

    # Final polish on best
    if time.time() < deadline - 5:
        for beta in [300, 800]:
            if time.time() > deadline - 5:
                break
            new_h = slsqp_polish(best_h, beta=beta, maxiter=80, deadline=deadline)
            new_s = exact_score(new_h)
            if new_s < best_score - 1e-13:
                best_h = new_h
                best_score = new_s

    best_h = make_feasible(best_h)
    final_score = exact_score(best_h)

    return (best_h, final_score, len(best_h))
```
