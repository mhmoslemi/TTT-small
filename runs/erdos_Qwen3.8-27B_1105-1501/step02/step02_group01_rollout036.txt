    z = np.asarray(z, dtype=float).ravel().copy()

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)

        corr_max = np.max(corr)

        best_delta_corr = 0  # we want to minimize max corr
        best_move = None

        n_high = np.where(h > 0.01)[0]

                if delta < 0.001:
                    continue
                # New corr = corr + delta * (G[:, j] - G[:, i])

                new_max = np.max(new_corr)

                    best_move = (i, j, delta)

        h[i] -= delta
        h[j] += delta
        h = make_feasible(h)
        new_score = official_score(h)
        if new_score < best_score:
            best_score = new_score
        else:
            break

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]

            h[i] += add
            residual -= add

def build_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            for m in range(n):
                g = 0.0
                if m >= d:
                    g += (1.0 - h[m - d])
                if d > 0 and m + d < n:
                    g += (-h[m + d])
                elif d == 0:
                    g += (-h[m])
                G[k, m] = g
        else:
            r = -d
            for m in range(n):
                g = 0.0
                if m + r < n:
                    g += (-h[m + r])
                if m >= r:
                    g += (1.0 - h[m - r])
                G[k, m] = g
    return G
