def make_feasible(z):

        else:

                residual -= a

    for i in range(n_pop):
        # Select three distinct random indices
        a, b, c = rng.choice(n_pop, size=3, replace=False)
        while a == b or a == c or b == c:
            a, b, c = rng.choice(n_pop, size=3, replace=False)

                pop[i] = np.clip(pop[i], -4.0, 4.0)

    # Phase 3: Targeted pairwise + PG alternating
    h_cur = best_h.copy()
    rng_pw = np.random.default_rng(seed + 111)
    beta = 50.0
    stall2 = 0
    while time_left() > 180:
        h_new, sc = targeted_pairwise(h_cur, n_rounds=50, n_grid=10, rng=rng_pw)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
            stall2 = 0
        else:
            stall2 += 1
        h_cur = h_new

        h_new, sc = pg_step(h_cur, beta, alpha_max=0.5, max_ls=12)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
            stall2 = 0
        h_cur = h_new
        beta = min(beta * 1.02, 300.0)
        if stall2 > 60:
            h_cur = make_feasible(best_h + rng.normal(0, 0.015, len(best_h)))
            beta = 30.0
            stall2 = 0

    # Phase 4: Active set LP polishing
    h_cur = best_h.copy()
    trust = 0.25
    for eps in [0.008, 0.004, 0.002, 0.001, 0.0005]:
        if time_left() < 60:
            break
        for _ in range(30):
            if time_left() < 60:
                break
            h_new, sc = active_set_lp(h_cur, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                    h_cur = h_new
                    trust = min(trust * 1.1, 0.4)
                else:
                    h_cur = h_new
                    trust = max(trust * 0.8, 0.03)

    # Phase 5: Random pairwise + fine PG
    h_cur = best_h.copy()
    rng_pw2 = np.random.default_rng(seed + 222)
    beta = 100.0
    while time_left() > 80:
        h_new, sc = random_pairwise(h_cur, n_pairs=60, n_grid=12, rng=rng_pw2)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
        h_cur = h_new

        h_new, sc = pg_step(h_cur, beta, alpha_max=0.3, max_ls=10)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
        h_cur = h_new
        beta = min(beta * 1.015, 400.0)

    # Phase 6: Multi-resolution exploration
    if time_left() > 60:
        n_current = len(best_h)
        n_options = []
        for dn in [-10, -5, 5, 10, 15]:
            nt = n_current + dn
            if nt >= 30 and nt <= 100:
                n_options.append(nt)
        for n_try in n_options:
            if time_left() < 30:
                break
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int.copy()
                best_score = sc_int
            h_cur = h_int.copy()
            rng_mr = np.random.default_rng(seed + n_try)
            beta_mr = 40.0
            stall_mr = 0
            for _ in range(200):
                if time_left() < 30:
                    break
                h_new, sc = pg_step(h_cur, beta_mr, alpha_max=0.4, max_ls=10)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                    stall_mr = 0
                else:
                    stall_mr += 1
                h_cur = h_new
                beta_mr = min(beta_mr * 1.02, 200.0)
                if stall_mr > 30:
                    h_cur = make_feasible(best_h + rng_mr.normal(0, 0.02, len(best_h)))
                    beta_mr = 20.0
                    stall_mr = 0
            # Quick pairwise at this resolution
            if time_left() > 30:
                h_new, sc = targeted_pairwise(h_cur, n_rounds=40, n_grid=10, rng=rng_mr)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc

    # Phase 7: Final LP + PG polish
    if time_left() > 15:
        h_cur = best_h.copy()
        for eps in [0.001, 0.0005, 0.0002]:
            if time_left() < 10:
                break
            for _ in range(15):
                if time_left() < 10:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.05)
                if h_new is not None:
                    if sc < best_score - 1e-13:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    if time_left() > 8:
        h_cur = best_h.copy()
        beta = 200.0
        for _ in range(300):
            if time_left() < 5:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.15, max_ls=8)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(final_h >= -1e-10) and np.all(final_h <= 1.0 + 1e-10)
    assert abs(final_h.sum() - final_n / 2.0) < 1e-6

    return final_h, final_score, final_n
