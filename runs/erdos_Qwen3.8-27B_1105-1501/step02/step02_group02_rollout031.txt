Actually, let me think about this differently. The 1D problem min_x max_s (a_s*x + b_s) subject to 0 <= x <= 1 can be solved as follows:

The function f(x) = max_s (a_s*x + b_s) is convex and piecewise linear. Its minimum on [0,1] is at:
- x = 0 if f'(0+) >= 0 (all lines with the max at x=0 have non-negative slope)
- x = 1 if f'(1-) <= 0
- Otherwise, at an intersection point of two lines

The subgradient at x is the convex hull of {a_s : a_s*x + b_s = f(x)}.

A practical approach: binary search on the optimal value t*.
- For a given t, the feasible set for x is:
  - x >= 0, x <= 1
  - For each s with a_s > 0: x >= (t - b_s) / a_s
  - For each s with a_s < 0: x <= (t - b_s) / a_s
  - (For a_s = 0: need b_s <= t)
- Feasible iff max(0, max_{a_s>0} (t-b_s)/a_s) <= min(1, min_{a_s<0} (t-b_s)/a_s)

We can binary search on t in O(1) per iteration (just computing the max and min).

Total: O(n log(precision)) per coordinate.

        q = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        # h[j] as h[i] (i = j)
        if i_lo <= j < i_hi:
            if 0 <= j + s < n:
                a[q] += (2.0 / n) * (1.0 - h[j + s])
        # h[j] as h[i+s] (i = j-s)
        if 0 <= j - s < n:
            i2 = j - s
            if i_lo <= i2 < i_hi:
                a[q] += (2.0 / n) * (-h[i2])
                b[q] += (2.0 / n) * h[i2]
    return a, b

def _find_optimal_x(a, b, lam):
    """min_{x in [0,1]} max_q (a[q]*x + b[q]) + lam*x
    = min_{x in [0,1]} max_q ((a[q]+lam)*x + b[q])"""
    a_mod = a + lam
    # f(0) = max(b), f(1) = max(a_mod + b)
    f0 = np.max(b)
    f1 = np.max(a_mod + b)
    
    lo_t = min(f0, f1)
    hi_t = max(f0, f1) + 1e-10
    
    x_star = 0.0 if f0 <= f1 else 1.0
    
    for _ in range(60):
        if hi_t - lo_t < 1e-14:
            break
        t = (lo_t + hi_t) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feasible = True
        for q in range(len(a_mod)):
            if a_mod[q] > 1e-15:
                candidate = (t - b[q]) / a_mod[q]
                if candidate > x_lo:
                    x_lo = candidate
            elif a_mod[q] < -1e-15:
                candidate = (t - b[q]) / a_mod[q]
                if candidate < x_hi:
                    x_hi = candidate
            else:
                if b[q] > t + 1e-14:
                    feasible = False
                    break
        if feasible and x_lo <= x_hi + 1e-14:
            hi_t = t
            x_star = max(0.0, min(1.0, (x_lo + x_hi) / 2.0))
        else:
            lo_t = t
    
    # Determine final x_star
    x_lo = 0.0
    x_hi = 1.0
    for q in range(len(a_mod)):
        if a_mod[q] > 1e-15:
            candidate = (t - b[q]) / a_mod[q]
            if candidate > x_lo:
                x_lo = candidate
        elif a_mod[q] < -1e-15:
            candidate = (t - b[q]) / a_mod[q]
            if candidate < x_hi:
                x_hi = candidate
    x_star = max(0.0, min(1.0, x_lo))
    if x_lo > x_hi + 1e-14:
        x_star = 0.0 if f0 <= f1 else 1.0
    
    return x_star

def _cd_pass(h, lam, order=None):
    """One pass of coordinate descent with Lagrange multiplier lam."""
    n = len(h)
    h = h.copy()
    if order is None:
        order = range(n)
    for j in order:
        a, b = _compute_ab(h, j)
        x_new = _find_optimal_x(a, b, lam)
        h[j] = x_new
    return h

def _cd_solve(h0, max_outer=50, max_inner=100, tol=1e-10):
    """Solve min G(h) s.t. sum(h)=n/2, 0<=h<=1 using CD + Lagrangian."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = _score(h)
    
    # Binary search on lam for outer iterations
    lam_lo = -5.0
    lam_hi = 5.0
    
    for outer in range(max_outer):
        # Binary search on lam to get sum(h) ≈ target
        for _ in range(30):
            lam = (lam_lo + lam_hi) / 2.0
            # Do a few CD passes
            for _ in range(5):
                h = _cd_pass(h, lam)
            s = h.sum()
            if s > target:
                lam_lo = lam  # need to decrease h values, increase lam
            else:
                lam_hi = lam
        
        # Final pass with the best lam
        lam = (lam_lo + lam_hi) / 2.0
        for _ in range(10):
            h = _cd_pass(h, lam)
        
        # Project to exact constraint
        h = _project(h)
        score = _score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        if abs(h.sum() - target) < tol:
            break
    
    return best_h, best_score

import numpy as np

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _interp(h_src, n_dst):

def _cd_pass(h, lam, order=None):

    if order is None:
        order = list(range(n))

        lo_t = min(f0, f1)

def _slsqp_smooth(h0, beta, maxiter=200):
    """SLSQP with LSE smooth approximation."""
    n = len(h0)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    def _lse_obj(x):
        cv = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta

    def _lse_grad(x):
        cv = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            j0 = max(0, s)
            j1 = min(n, n + s)
            for p in range(j0, j1):
                g[p] += wk * (1.0 - x[p - s])
            p0 = max(0, -s)
            p1 = min(n, n - s)
            for p in range(p0, p1):
                g[p] += wk * (-x[p + s])
        return g

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

import numpy as np

from scipy.optimize import minimize

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _cd_pass(h, lam, order):
    n = len(h)
    h_w = h.copy()
    inv_n = 2.0 / n
    for j in order:
        a = np.zeros(2 * n - 1)
        b = np.zeros(2 * n - 1)
        for s in range(-(n - 1), n):
            q = s + n - 1
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            if i_lo <= j < i_hi and 0 <= j + s < n:
                a[q] += inv_n * (1.0 - h_w[j + s])
            i2 = j - s
            if 0 <= i2 < n and i_lo <= i2 < i_hi:
                a[q] += inv_n * (-h_w[i2])
                b[q] += inv_n * h_w[i2]
        a_mod = a + lam
        f0 = np.max(b)
        f1 = np.max(a_mod + b)
        lo_t = min(f0, f1)
        hi_t = max(f0, f1) + 1e-12
        x_star = 0.0 if f0 <= f1 else 1.0
        for _ in range(50):
            if hi_t - lo_t < 1e-13:
                break
            t = (lo_t + hi_t) / 2.0
            pos_m = a_mod > 1e-14
            neg_m = a_mod < -1e-14
            x_lo = 0.0
            x_hi = 1.0
            if np.any(pos_m):
                x_lo = max(x_lo, np.max((t - b[pos_m]) / a_mod[pos_m]))
            if np.any(neg_m):
                x_hi = min(x_hi, np.min((t - b[neg_m]) / a_mod[neg_m]))
            zero_m = ~(pos_m | neg_m)
            if np.any(zero_m) and np.max(b[zero_m]) > t + 1e-13:
                lo_t = t
                continue
            if x_lo <= x_hi + 1e-13:
                hi_t = t
                x_star = max(0.0, min(1.0, (x_lo + x_hi) / 2.0))
            else:
                lo_t = t
        h_w[j] = x_star
    return h_w

def _slsqp_lse(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def _efun(x):
        return x.sum() - n / 2.0
    def _ejac(x):
        return np.ones(n)
    def _obj(x):
        cv = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta
    def _grad(x):
        cv = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            j0 = max(0, s)
            j1 = min(n, n + s)
            for i in range(j0, j1):
                g[i] += wk * (1.0 - x[i - s])
            p0 = max(0, -s)
            p1 = min(n, n - s)
            for i in range(p0, p1):
                g[i] += wk * (-x[i + s])
        return g
    try:
        res = minimize(_obj, h0, args=(beta,), jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_direct(h0, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    cv = np.correlate(h0, 1.0 - h0, mode='full') * dx
    t0 = min(cv.max() + 0.003, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def _fun(z):
        return z[-1]
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv2 = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - cv2
    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': _ieq},
                                    {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):

    except Exception:
        pass
    return h0
