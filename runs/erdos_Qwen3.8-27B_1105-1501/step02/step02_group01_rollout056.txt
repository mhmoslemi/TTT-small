import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):

            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3:
                break

    return best_h, best_score

def smooth_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    shifted = np.clip(beta * (corr - np.max(corr)), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return np.max(corr)
    return np.max(corr) + np.log(s) / beta

def smooth_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n - s2])
                grad[:n - s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n - m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n - m2])
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, beta, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_obj, h0, args=(beta, dx),
            jac=lambda x: smooth_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def perturb_restart(h, rng, n_perturbs=5, sigma=0.05, time_limit=30, t_start=None):
    """Try random perturbations followed by pair moves."""
    h = h.copy()
    best_h = h.copy()
    best_score = exact_score(h)

    for p in range(n_perturbs):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        h_pert = make_feasible(h + sigma * rng.standard_normal(len(h)))
        h_new, s_new = greedy_pairs(h_pert, max_rounds=300, time_limit=10, t_start=time.time())
        if s_new < best_score - 1e-12:
            best_score = s_new
            best_h = h_new.copy()
            h = h_new.copy()

    return best_h, best_score

def build_starts(parent, n, rng):
    """Build starting points for a given n."""
    starts = []
    h_interp = make_feasible(interp_to_n(parent, n))
    starts.append(h_interp)
    for p in [1.5, 2.5, 5.0]:
        starts.append(push_binary(h_interp, p))
    if n % 2 == 0:
        n_ones = n // 2
        for trial in range(3):
            rb = np.zeros(n)
            rb[rng.choice(n, n_ones, replace=False)] = 1.0
            starts.append(rb)
    else:
        n_ones = n // 2
        for trial in range(3):
            rb = np.zeros(n)
            idx = rng.choice(n, n_ones, replace=False)
            rb[idx] = 1.0
            rb[rng.integers(n)] = 1.0
            rb = make_feasible(rb)
            starts.append(rb)
    for sigma in [0.03, 0.1]:
        starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))
    starts.append(make_feasible(1.0 - h_interp[::-1].copy()))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0_global = time.time()
    reserve = max(20.0, 0.06 * budget_s)
    deadline = t0_global + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)

    best_h = parent.copy()
    best_score = exact_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    n_base = len(parent)
    n_options = sorted(set([n for n in range(max(30, n_base - 6), min(72, n_base + 10)) if n >= 24]))

    for n in n_options:
        if time.time() > deadline - 60:
            break
        starts = build_starts(parent, n, rng)

        for h0 in starts:
            if time.time() > deadline - 40:
                break

            cur_h = h0.copy()
            cur_score = exact_score(cur_h)

            # Phase 1: Main greedy pair moves
            t_phase = time.time()
            cur_h, cur_score = greedy_pairs(cur_h, max_rounds=1500, time_limit=80, t_start=t_phase)

            # Phase 2: SLSQP smoothing
            if time.time() < deadline - 60:
                for beta in [100, 400]:
                    if time.time() > deadline - 60:
                        break
                    new_h = slsqp_smooth(cur_h, beta, maxiter=100)
                    new_s = exact_score(new_h)
                    if new_s < cur_score - 1e-13:
                        cur_h = new_h
                        cur_score = new_s

            # Phase 3: More pair moves after smoothing
            t_phase = time.time()
            cur_h, cur_score = greedy_pairs(cur_h, max_rounds=1000, time_limit=60, t_start=t_phase)

            # Phase 4: Perturbation + pair moves
            if time.time() < deadline - 50:
                t_phase = time.time()
                cur_h, cur_score = perturb_restart(
                    cur_h, rng, n_perturbs=4, sigma=0.04,
                    time_limit=40, t_start=t_phase
                )

            # Phase 5: Final SLSQP + pairs
            if time.time() < deadline - 40:
                new_h = slsqp_smooth(cur_h, 300, maxiter=80)
                new_s = exact_score(new_h)
                if new_s < cur_score - 1e-13:
                    cur_h = new_h
                    cur_score = new_s
                t_phase = time.time()
                cur_h, cur_score = greedy_pairs(cur_h, max_rounds=500, time_limit=30, t_start=t_phase)

            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
            elif abs(cur_score - best_score) <= 1e-12 and len(cur_h) < best_n:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)

    # Final refinement on best solution
    remaining = deadline - time.time()
    if remaining > 30:
        t_phase = time.time()
        best_h, best_score = greedy_pairs(best_h, max_rounds=2000, time_limit=remaining - 20, t_start=t_phase)

    if remaining > 60:
        for sigma in [0.02, 0.05, 0.08]:
            if time.time() > deadline - 15:
                break
            h_pert = make_feasible(best_h + sigma * rng.standard_normal(best_n))
            t_phase = time.time()
            h_new, s_new = greedy_pairs(h_pert, max_rounds=1000, time_limit=25, t_start=t_phase)
            if s_new < best_score - 1e-13:
                best_h = h_new
                best_score = s_new
            new_h = slsqp_smooth(h_pert, 500, maxiter=80)
            new_s = exact_score(new_h)
            if new_s < best_score - 1e-13:
                best_h = new_h
                best_score = new_s

    # Final pair polish
    if time.time() < deadline - 5:
        best_h, best_score = greedy_pairs(best_h, max_rounds=500, time_limit=10, t_start=time.time())

    best_h = make_feasible(best_h)
    final_score = exact_score(best_h)

    return (best_h, final_score, len(best_h))
