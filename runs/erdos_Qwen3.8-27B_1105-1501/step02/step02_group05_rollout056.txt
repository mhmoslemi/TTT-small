def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    z = np.clip(z, 0.0, 1.0)
    target = n / 2.0
    # Bisection to find shift that makes sum = target
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0, 1)
    # Greedy repair
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-12:
                break
    elif residual < -1e-12:
        order = np.argsort(h)[::-1]
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if -residual < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, s):
    """Compute gradient of dx * c_s with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)

        if m <= 0:

import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    z = np.clip(z, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0, 1)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-12:
                break
    elif residual < -1e-12:
        order = np.argsort(h)[::-1]
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if -residual < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return grad
        grad[s:] += dx * (1.0 - h[:m])
        grad[:m] -= dx * h[s:]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return grad
        grad[:m] += dx * (1.0 - h[t:])
        grad[t:] -= dx * h[:m]
    return grad

def lp_polish_step(h, eta=0.005, K_max=32, rho=0.15):
    """One LP-based polishing step. Returns improved h or None."""
    n = len(h)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    C = V.max()
    
    # Select active lags
    threshold = C - eta
    active_mask = V >= threshold
    active_idx = np.where(active_mask)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(V)])
    # Also include top K
    top_k = np.argsort(V)[-K_max:]
    active_idx = np.unique(np.concatenate([active_idx, top_k]))
    # Cap at max
    if len(active_idx) > K_max:
        # Keep the highest valued ones
        active_idx = active_idx[np.argsort(V[active_idx])[-K_max:]]
    
    # Compute gradients for active lags
    constraints_A = []
    constraints_b = []
    for p in active_idx:
        s = int(p - (n - 1))
        g = compute_lag_gradient(h, s)
        # g^T u - t <= -V[p]
        row = np.concatenate([g, [-1.0]])
        constraints_A.append(row)
        constraints_b.append(-V[p])
    
    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C + 0.01))  # t unbounded below, bounded above slightly above C
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success or (result.status in [0] and np.isfinite(result.fun)):
        u = result.x[:n]
        if not np.all(np.isfinite(u)):
            return None
        if np.max(np.abs(u)) < 1e-14:
            return None
        
        # Line search
        best_h = h.copy()
        best_score = C
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02]:
            trial = make_feasible(h + alpha * u)
            sc = score_exact(trial)
            if sc < best_score - 1e-14:
                best_h = trial
                best_score = sc
        if best_score < C - 1e-14:
            return best_h
    return None

def lp_polish(h, max_iters=200, eta=0.005, K_max=32, rho=0.15, time_limit=None, t_start=None):
    """Iterative LP polishing."""
    best = h.copy()
    best_score = score_exact(best)
    eta_cur = eta
    rho_cur = rho
    stall = 0
    
    for it in range(max_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        result = lp_polish_step(best, eta=eta_cur, K_max=K_max, rho=rho_cur)
        if result is not None:
            new_score = score_exact(result)
            if new_score < best_score - 1e-14:
                best = result
                best_score = new_score
                stall = 0
                # Slightly increase eta to capture more lags
                eta_cur = min(eta_cur * 1.3, 0.02)
                rho_cur = min(rho_cur * 1.1, 0.3)
            else:
                stall += 1
        else:
            stall += 1
        
        if stall >= 5:
            # Try different parameters
            if eta_cur > 0.002:
                eta_cur = max(eta_cur * 0.5, 0.001)
            elif rho_cur > 0.05:
                rho_cur = max(rho_cur * 0.6, 0.03)
            else:
                # Try decreasing K
                K_max = max(K_max - 4, 12)
            stall = 0
    
    return best, best_score

def binary_sa(h_init, n_steps=5000, t_start=0.0005, cool=0.998, rng=None, time_limit=None, t_start_time=None):
    """Binary simulated annealing on balanced binary strings."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(h_init)
    if n % 2 != 0:
        return None
    target_ones = n // 2
    
    # Initialize from thresholding
    b = (h_init >= 0.5).astype(int)
    # Balance
    n_ones = b.sum()
    if n_ones > target_ones:
        # Remove ones where h is smallest
        one_idx = np.where(b == 1)[0]
        order = np.argsort(h_init[one_idx])
        for i in range(n_ones - target_ones):
            b[one_idx[order[i]]] = 0
    elif n_ones < target_ones:
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h_init[zero_idx])
        for i in range(target_ones - n_ones):
            b[zero_idx[order[i]]] = 1
    
    cur_score = score_exact(b.astype(float))
    best_b = b.copy()
    best_score = cur_score
    T = t_start
    
    for step in range(n_steps):
        if time_limit is not None and t_start_time is not None:
            if time.time() - t_start_time > time_limit:
                break
        # Random swap
        pos1 = np.where(b == 1)[0]
        pos0 = np.where(b == 0)[0]
        i = pos1[rng.integers(len(pos1))]
        j = pos0[rng.integers(len(pos0))]
        
        b[i] = 0
        b[j] = 1
        new_score = score_exact(b.astype(float))
        
        delta = new_score - cur_score
        if delta <= 1e-14 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_b = b.copy()
        else:
            # Reject
            b[j] = 0
            b[i] = 1
        
        T *= cool
    
    return best_b.astype(float), best_score

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def two_point_search(h, n_tries, rng, is_dead):
    """Two-point exchange search (retained from previous as auxiliary)."""
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates = [t_lo + f * span for f in [0.0, 0.25, 0.5, 0.75, 1.0, 1/2.618, 2/2.618]]
        best_local = best_sc
        best_t = None
        tmp = best.copy()
        for t in candidates:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = score_exact(best)
    return best, best_sc

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    z = np.clip(z, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0, 1)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-12:
                break
    elif residual < -1e-12:
        order = np.argsort(h)[::-1]
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if -residual < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return grad
        grad[s:] += dx * (1.0 - h[:m])
        grad[:m] -= dx * h[s:]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return grad
        grad[:m] += dx * (1.0 - h[t:])
        grad[t:] -= dx * h[:m]
    return grad

def lp_step(h, eta, K_max, rho):
    n = len(h)
    V = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    C = V.max()
    
    # Active set
    active_idx = np.where(V >= C - eta)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(V)])
    top_k = np.argsort(V)[-K_max:]
    active_idx = np.unique(np.concatenate([active_idx, top_k]))
    if len(active_idx) > K_max:
        active_idx = active_idx[np.argsort(V[active_idx])[-K_max:]]
    
    # Build LP
    n_con = len(active_idx)
    A_ub = np.zeros((n_con, n + 1))
    b_ub = np.zeros(n_con)
    for k, p in enumerate(active_idx):
        s = int(p - (n - 1))
        g = compute_lag_gradient(h, s)
        A_ub[k, :n] = g
        A_ub[k, n] = -1.0
        b_ub[k] = -V[p]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(max(-h[i], -rho), min(1.0 - h[i], rho)) for i in range(n)]

    cur_sc = score_exact(b.astype(float))
    best_sc = cur_sc
    best_b = b.copy()
    T = T0
    
    for _ in range(n_steps):
        if time.time() - t0 > time_budget:
            break
        pos1 = np.where(b == 1)[0]
        pos0 = np.where(b == 0)[0]
        i = pos1[rng.integers(len(pos1))]
        j = pos0[rng.integers(len(pos0))]
        b[i] = 0
        b[j] = 1
        new_sc = score_exact(b.astype(float))
        delta = new_sc - cur_sc
        if delta <= 1e-14 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur_sc = new_sc
            if cur_sc < best_sc - 1e-14:
                best_sc = cur_sc
                best_b = b.copy()
        else:
            b[j] = 0
            b[i] = 1
        T *= cool
    
    return best_b.astype(float), best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for _ in range(n_tries):
        if is_dead():
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        fracs = [0.0, 0.3, 0.5, 0.7, 1.0, 0.382, 0.618, 0.2, 0.8]
        best_local = best_sc
        best_t = 0.0
        for f in fracs:
            t = t_lo + f * span
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp = best.copy()
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local - 1e-15:
                best_local = sc
                best_t = t
        if abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = score_exact(best)
    return best, best_sc
