V_0_new = V_0 + δ - (h[i]+δ)² + h[i]² - (h[j]-δ)² + h[j]²
= V_0 + δ - 2h[i]δ - δ² + 2h[j]δ - δ²
= V_0 + δ(1 - 2h[i] + 2h[j]) - 2δ²

For k≠0: V_k is linear in each h component, so:
V_k_new = V_k + δ*(∂V_k/∂h[i] - ∂V_k/∂h[j])

So the new max is max over all k of (V_k + linear_or_quadratic_in_δ).

To find the optimal δ, we need to minimize the max of (2n-1) affine/quadratic functions of δ, subject to 0 ≤ h[i]+δ ≤ 1 and 0 ≤ h[j]-δ ≤ 1.

This is a 1D minimax problem! We can solve it by:
1. Identifying the "upper envelope" of all the functions of δ
2. Finding where the envelope is minimized

For affine functions, the upper envelope is a piecewise-linear convex function, and its minimum is at a vertex (intersection of two lines) or at a boundary.

This is doable but complex to implement efficiently for n=60 (119 functions to consider per pair, and C(60,2) = 1770 pairs).

def make_feasible(z):
    n = len(z)

    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0

            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = np.clip(beta * (V - V_max), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.full(len(V), 1.0) / len(V)
    return w / ws

def gradient_max(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def pg_step(h, beta, alphas=(0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01)):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = gradient_max(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    for alpha in alphas:
        h_try = make_feasible(h - alpha * g)
        f_try = official_score(h_try)
        if f_try < best_f:
            best_f = f_try
            best_h = h_try
            break  # first successful step is enough (monotone)
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.15):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:

    bounds_list = [(max(-h[i], -trust), min(1.0 - h[i], trust)) for i in range(n)]
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None
