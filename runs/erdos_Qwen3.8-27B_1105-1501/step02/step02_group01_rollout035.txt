    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def thue_morse(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        tm[i] = cnt
    return tm

    n_ones = n // 2
    order = np.argsort(-scores)
    result = np.zeros(n)
    result[order[:n_ones]] = 1.0
    return result

def lp_unbinarize(h_bin, max_step=0.1):
    """Try to improve a binary solution by moving to non-binary."""
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    
    # Compute correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr))

    threshold = current_max - 0.05 * current_max
    active = np.where(corr >= threshold)[0]
    
    # Compute gradients for active lags
    # f_k(h) = dx * sum_i h[i] * (1-h[i+s_k]) where s_k = k - (n-1)
    grad_sum = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            # f = dx * sum_{p=0}^{n-1-s} h[p](1-h[p+s])
            # df/dh[p] = dx * [(1-h[p+s]) - h[p-s]] * (indicator that p is in range)
            for p in range(n):
                g = 0.0
                if p < n - s:
                    g += (1.0 - h[p + s])
                if p >= s:
                    g -= h[p - s]
                grad_sum[p] += g
        else:
            m = -s
            # f = dx * sum_{p=m}^{n-1} h[p](1-h[p-m])
            for p in range(n):
                g = 0.0
                if p >= m:
                    g += (1.0 - h[p - m])
                if p < n - m:
                    g -= h[p + m]
                grad_sum[p] += g
    
    grad_sum *= dx

    #   minimizes max over active k of grad_k · d
    #   subject to: sum(d) = 0, d_i >= 0 for zeros, d_i <= 0 for ones, |d_i| <= max_step

    # Equivalent: minimize t

    #              sum(d) = 0
