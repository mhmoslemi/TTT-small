def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def raw_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_all(h, s_list):
    """Compute gradients for a list of shifts. Returns array of shape (len(s_list), n)"""
    n = len(h)
    grads = np.zeros((len(s_list), n))
    for idx, s in enumerate(s_list):
        if s == 0:
            grads[idx] = 1.0 - 2.0 * h
        elif s > 0:
            g = np.zeros(n)
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
            grads[idx] = g
        else:
            t = -s
            g = np.zeros(n)
            g[t:] += (1.0 - h[:n-t])
            g[:n-t] -= h[t:]
            grads[idx] = g
    return grads

def make_feasible(z):

    target = n / 2.0

def qp_solve_linearized(h0, active_shifts, radius, reg):
    """Solve the linearized QP: minimize t + reg/2 * ||h-h0||^2 subject to linearized constraints"""
    n = len(h0)
    raw = raw_vals(h0)
    dx = 2.0 / n
    # active_shifts: list of shift values s (p - (n-1))
    # For each active shift, the linearized constraint is:
    #   raw[p] + dx * grad_s(h0) · (h - h0) <= t
    # i.e., raw[p] + dx * (grad_s · h - grad_s · h0) <= t
    # i.e., dx * grad_s · h - t <= raw[p] - dx * (grad_s · h0)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= np.maximum(0.0, h0 - radius),
        h_var <= np.minimum(1.0, h0 + radius),
    ]
    
    for s in active_shifts:
        p = s + (n - 1)
        g = grad_all(h0, [s])[0]
        # Constraint: raw[p] + dx * g · (h - h0) <= t
        # i.e., dx * g · h - t <= raw[p] - dx * g · h0
        rhs = raw[p] - dx * np.dot(g, h0)
        constraints.append(dx * g @ h_var - t_var <= rhs)
    
    obj = cp.Minimize(t_var + (reg / 2.0) * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)

            if prob.status in ['optimal', 'optimal_inaccurate']:

            continue

        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)
