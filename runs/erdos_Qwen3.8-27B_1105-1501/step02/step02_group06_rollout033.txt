    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_p(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i + s] += (1.0 - h[i])
            g[i] -= h[i + s]
    else:
        t = -s
        for i in range(t, n):
            g[i - t] += (1.0 - h[i])
            g[i] -= h[i - t]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def qp_step(h0, epsilon, reg, n_lags_cap=20):
    """QP step with soft epsilon-based active set."""
    n = len(h0)
    vals = compute_f(h0)
    t_max = vals.max()
    # Select lags within epsilon of max
    mask = vals >= t_max - epsilon
    if not np.any(mask):
        idx = np.argmax(vals)
        vals[idx] = t_max
        mask = vals >= t_max - 1e-10
    selected = np.where(mask)[0]
    # Cap the number of constraints
    if len(selected) > n_lags_cap:
        # Keep the ones closest to max
        order = np.argsort(vals[selected])[::-1]
        selected = selected[order[:n_lags_cap]]
    
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    for p in selected:
        s = p - (n - 1)
        g = dx * gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)
    
    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None
    
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None, None
    h_feas = make_feasible(h_new)
    sc = official_score(h_feas)
    return h_feas, sc

def active_set_loop(h0, max_iter=40, eps_start=0.02, eps_end=0.0005, reg_start=10.0, reg_end=0.001):
    """Iterative active-set QP with decreasing epsilon and regularization."""
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    stall = 0
    
    for it in range(max_iter):
        frac = it / max(1, max_iter - 1)
        eps = eps_start * (eps_end / eps_start) ** frac
        reg = reg_start * (reg_end / reg_start) ** frac
        h_new, sc = qp_step(h, eps, reg)
        if h_new is None:
            break
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.8, 0.5, 0.3, 0.15, 0.08]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-14:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 12:
                    break
    return h, best

def slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        vals = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        shifted = np.clip(shifted, -50, 50)
        obj = cmax + np.log(np.sum(np.exp(shifted))) / beta
        # Gradient
        probs = np.exp(shifted) / np.sum(np.exp(shifted))
        # d(vals[p])/d(x[i]) = dx * gradient of raw[p] at xi i
        grad = np.zeros(n)
        for p in range(2*n-1):
            if probs[p] < 1e-10:
                continue
            s = p - (n-1)
            g = dx * gradient_p(x, s)
            grad += probs[p] * g
        return obj, grad
    
    def eq_fun(x):
        return np.sum(x) - n/2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj_grad, h0.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            return h_new, sc
    except Exception:
        pass
    return None, None

def compute_f(h):

def gradient_p_vec(h, s):
    """Vectorized gradient of f_s with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] -= h[s:]
        g[s:] += 1.0 - h[:n-s]
    else:
        t = -s
        g[:n-t] += 1.0 - h[t:]
        g[t:] -= h[:n-t]
    return g

def qp_step(h0, epsilon, reg, n_lags_cap=15):
    n = len(h0)
    vals = compute_f(h0)
    t_max = vals.max()
    mask = vals >= t_max - epsilon
    if not np.any(mask):

        selected = selected[order[:n_lags_cap]]
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]

        g = dx * gradient_p_vec(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)
    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
