def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, target):
    # Project x onto {0 <= y <= 1, sum(y) = target}
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        if np.sum(y) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Log-sum-exp for numerical stability
    vals = (corr - m) / T
    S = np.sum(np.exp(vals))
    val = dx * (m + T * np.log(S))
    return val

def smooth_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    vals = (corr - m) / T
    exp_vals = np.exp(vals)
    w = exp_vals / np.sum(exp_vals)  # softmax weights
    
    # Gradient computation
    # w has length 2n-1, indexed 0 to 2n-2
    # w[n-1+k] corresponds to lag k
    grad = np.zeros(n)
    for j in range(n):

        if k_lo <= k_hi:
            k_indices = k_lo + np.arange(k_hi - k_lo + 1)
            t_indices = j + k_indices  # j+k

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        if np.sum(y) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    vals = (corr - m) / T
    S = np.sum(np.exp(vals))
    return dx * (m + T * np.log(S))

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    vals = (corr - m) / T

    M1 = w[M1_idx]
    
    first_sum = M1 @ (1.0 - h)
    second_sum = M1.T @ h
    return dx * (first_sum - second_sum)

def sum_eq(x):
    return np.sum(x) - x.size / 2.0

def sum_eq_jac(x):
    return np.ones(x.size)

def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(40):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def full_pair_sweep(h, rng):
    n = len(h)
    perm = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta

def random_pair_sweep(h, rng, n_pairs):

    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        if i > j:
            i, j = j, i
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:

            h[j] += delta

        return h_new
    return h.copy()
