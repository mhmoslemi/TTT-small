    dx = 2.0 / n

    """Gradient of correlation at offset 'off' with respect to h.
    
    off = p - (n-1) where p is the index in the full correlation.
    correlation[p] = sum_{i valid} h[i+off] * (1-h[i])
    valid i: max(0, -off) <= i <= min(n-1, n-1-off)
    """
    n = len(h)
    g = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n-1, n-1-off)
    for i in range(i_lo, i_hi+1):

        # d/dh[j]:

        # if j == i: contributes -h[i+off]
        j1 = i + off
        g[j1] += (1.0 - h[i])
        g[i] -= h[j1]
    return g

def lp_find_direction(h, k_active):
    """Find LP direction minimizing worst-case first-order change on active lags."""
    n = len(h)
    cors = compute_correlations(h)
    top_idx = np.argsort(cors)[-k_active:]
    
    grads = np.zeros((k_active, n))
    for i, p in enumerate(top_idx):
        off = p - (n - 1)
        grads[i] = grad_at_lag_off(h, off)
    
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(d) == 0, d >= -1, d <= 1]
    for i in range(k_active):
        constraints.append(grads[i] @ d <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.ECOS, cp.CVXOPT, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except:
            continue
    
    if prob.status in ['optimal', 'optimal_inaccurate'] and d.value is not None:
        if t.value < -1e-6:
            return d.value
    return None

def exact_line_search(h, d):
    """Golden section line search."""
    n = len(h)
    t_max = 1.0
    for i in range(n):
        if d[i] > 1e-14:
            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            t_max = min(t_max, -h[i] / d[i])
    t_max = max(0.0, min(1.0, t_max * 0.999))
    
    if t_max < 1e-10:
        return 0.0, official_score(h)
    
    best_t = 0.0
    best_sc = official_score(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, t_max
    for _ in range(40):
        c = b - gr * (b - a)
        m = a + gr * (b - a)
        fc = official_score(h + c * d)
        fm = official_score(h + m * d)
        if fc < fm:
            b = m
        else:
            a = c
        t_mid = (a + b) / 2.0
        sc_mid = official_score(h + t_mid * d)
        if sc_mid < best_sc:
            best_sc = sc_mid
            best_t = t_mid
    
    return best_t, best_sc

def lp_descent_loop(h0, max_iter=100, k_start=5, k_end=15, time_limit=None, t_start=None):
    """Iterative LP descent with adaptive active set size."""
    h = h0.copy()
    best_h = h.copy()
    best_sc = official_score(h)
    stall = 0
    
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        
        k_act = min(k_end, k_start + it // 10)
        d = lp_find_direction(h, k_act)
        if d is None:
            # Try with fewer active constraints
            d = lp_find_direction(h, max(2, k_act // 2))
        if d is None:
            stall += 2
        else:
            t, sc = exact_line_search(h, d)
            if t > 1e-8 and sc < best_sc - 1e-10:
                h = make_feasible(h + t * d)
                sc = official_score(h)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        
        if stall > 30:
            break
    
    return best_h, best_sc

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    
    def obj(x):
        h = np.clip(x, 0, 1)
        cors = compute_correlations(h)
        cmax = cors.max()
        shifted = beta * (cors - cmax)
        shifted = np.clip(shifted, -50, 50)
        return cmax + np.log(np.sum(np.exp(shifted))) / beta
    
    def obj_grad(x):
        h = np.clip(x, 0, 1)
        cors = compute_correlations(h)
        cmax = cors.max()
        shifted = beta * (cors - cmax)
        shifted = np.clip(shifted, -50, 50)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        # Gradient of cors with respect to h
        grad = np.zeros(n)
        for p in range(len(cors)):
            if weights[p] > 1e-10:
                off = p - (n - 1)
                grad += weights[p] * grad_at_lag_off(h, off)
        # Only for non-clipped
        mask = (x > 0) & (x < 1)
        return grad * mask
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            return h_new, sc
    except:
        pass
    return None, None

        h_opt, sc_opt = lp_descent_loop(h, max_iter=250, k_start=5, k_end=25,
                                         time_limit=remaining() - 80, t_ref=t_start)
        if sc_opt < best_score - 1e-12:
            best_h = h_opt.copy()
            best_score = sc_opt

    # Phase 2: SLSQP polish
    if remaining() > 40:
        for beta in [300, 1500, 8000]:
            if remaining() < 20:
                break
            h_new, sc = slsqp_smooth(best_h, beta, maxiter=100)
            if h_new is not None and sc < best_score - 1e-10:
                best_h = h_new
                best_score = sc

    # Phase 3: LP descent with wider active set
    if remaining() > 40:
        h_opt, sc_opt = lp_descent_loop(best_h, max_iter=200, k_start=8, k_end=35,
                                         time_limit=remaining() - 40, t_ref=t_start)
        if sc_opt < best_score - 1e-12:
            best_h = h_opt.copy()
            best_score = sc_opt

    # Phase 4: Subgradient with decreasing step
    if remaining() > 30:
        h_sub = best_h.copy()
        sc_sub = best_score
        for step_i in range(50):
            if remaining() < 10:
                break
            alpha = 0.05 / (1.0 + 0.3 * step_i)
            h_sub = subgradient_step(h_sub, 8, alpha)
            sc_sub = official_score(h_sub)
            if sc_sub < best_score - 1e-11:
                best_h = h_sub.copy()
                best_score = sc_sub

    # Phase 5: Multi-resolution
    if remaining() > 60:
        for n_low in [40, 50, 30]:
            if remaining() < 30:
                break
            h_low = interpolate_to(best_h, n_low)
            sc_low = official_score(h_low)
            h_opt_low, sc_opt_low = lp_descent_loop(h_low, max_iter=80, k_start=4,
                                                     k_end=12, time_limit=20, t_ref=time.time())
            # Interpolate back to original n
            h_back = interpolate_to(h_opt_low, n)
            sc_back = official_score(h_back)
            if sc_back < best_score - 1e-10:
                best_h = h_back
                best_score = sc_back
            # Also keep the low-res if it's better and we want fewer points
            if sc_opt_low < best_score - 1e-10 and n_low <= n:
                pass  # We prefer original n for now

        # Also try going up in resolution
        if remaining() > 40:
            n_high = min(n + 10, 80)
            h_high = interpolate_to(best_h, n_high)
            h_opt_high, sc_opt_high = lp_descent_loop(h_high, max_iter=60, k_start=5,
                                                       k_end=15, time_limit=25, t_ref=time.time())
            h_back_hi = interpolate_to(h_opt_high, n)
            sc_back_hi = official_score(h_back_hi)
            if sc_back_hi < best_score - 1e-10:
                best_h = h_back_hi
                best_score = sc_back_hi

    # Phase 6: Random perturbations + re-optimize
    if remaining() > 80:
        for attempt in range(20):
            if remaining() < 25:
                break
            sigma = 0.02 + 0.08 * rng.random()
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = lp_descent_loop(h_pert, max_iter=40, k_start=5, k_end=12,
                                             time_limit=remaining() - 25, t_ref=time.time())
            if sc_opt < best_score - 1e-10:
                best_h = h_opt.copy()
                best_score = sc_opt

    # Phase 7: Binary exploration
    if remaining() > 60:
        for n_b in [n, n - 10, n + 10]:
            if n_b < 20 or remaining() < 30:
                continue
            h_bin = binary_random_init(n_b, rng)
            sc_bin = official_score(h_bin)
            # Local search on binary
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            for _ in range(300):
                if remaining() < 20:
                    break
                oi = ones[rng.integers(len(ones))]
                oz = zeros[rng.integers(len(zeros))]
                h_bin[oi] = 0.0
                h_bin[oz] = 1.0
                if official_score(h_bin) >= sc_bin + 1e-12:
                    h_bin[oi] = 1.0
                    h_bin[oz] = 0.0
                else:
                    sc_bin = official_score(h_bin)
                    ones = np.where(h_bin == 1.0)[0]
                    zeros = np.where(h_bin == 0.0)[0]
            # Convert to continuous and optimize
            h_cont = binary_to_continuous(h_bin, n_b)
            sc_cont = official_score(h_cont)
            if n_b != n:
                h_back = interpolate_to(h_cont, n)
                sc_cont = official_score(h_back)
            else:
                h_back = h_cont
            h_opt, sc_opt = lp_descent_loop(h_back, max_iter=30, k_start=5, k_end=10,
                                             time_limit=min(15, remaining() - 20),
                                             t_ref=time.time())
            if sc_opt < best_score - 1e-10:
                best_h = h_opt.copy()
                best_score = sc_opt

    # Phase 8: Final intensive LP + SLSQP
    if remaining() > 20:
        h_opt, sc_opt = lp_descent_loop(best_h, max_iter=80, k_start=5, k_end=20,
                                         time_limit=min(20, remaining() - 10), t_ref=time.time())
        if sc_opt < best_score - 1e-12:
            best_h = h_opt.copy()
            best_score = sc_opt

    if remaining() > 15:
        h_new, sc = slsqp_smooth(best_h, 10000, maxiter=80)
        if h_new is not None and sc < best_score - 1e-10:
            best_h = h_new
            best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
