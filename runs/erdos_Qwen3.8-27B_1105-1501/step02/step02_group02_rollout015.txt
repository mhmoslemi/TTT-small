    """FISTA subgradient with averaging."""
    n = len(h0)
    dx = 2.0 / n
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    
    for t in range(1, n_iter + 1):
        cv = np.correlate(y, 1.0 - y, mode='full') * dx
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            if s >= 0:
                for i in range(n - s):
                    grad[i] += wk * dx * (1.0 - y[i + s])
                    grad[i + s] += wk * dx * (-y[i])
            else:
                ss = -s
                for i in range(n - ss):
                    grad[i + ss] += wk * dx * (1.0 - y[i])
                    grad[i] += wk * dx * (-y[i + ss])
        
        t_curr = float(t)
        y = h_cur + (t_curr - 1.0) / (t_curr + 1.5) * (h_cur - h_prev)
        y = _project(y)
        h_new = _project(y - alpha * grad)
        h_avg = h_avg + (h_cur - h_avg) / (t_curr + 1)
        h_prev = h_cur
        h_cur = h_new
        
        if t % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()

        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:

        best_h = ha.copy()

def _binary_sa(h_init, n_iter=3000, t0=0.5, t1=0.02, rng=None):
    """Simulated annealing on binary vectors."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(h_init)
    if n % 2 != 0:
        n -= 1
    target = n // 2
    
    h = np.zeros(n)
    # Initialize from h_init
    h_init_n = h_init[:n] if len(h_init) >= n else np.pad(h_init, (0, n - len(h_init)))
    h = (h_init_n > 0.5).astype(float)
    # Fix count
    ones_count = int(h.sum())
    if ones_count > target:
        ones_idx = np.where(h == 1)[0]
        h[rng.choice(ones_idx, ones_count - target, replace=False)] = 0
    elif ones_count < target:
        zeros_idx = np.where(h == 0)[0]
        h[rng.choice(zeros_idx, target - ones_count, replace=False)] = 1
    
    best_h = h.copy()
    best_s = _score(h)
    cur_s = best_s
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for it in range(n_iter):
        t = t0 * (t1 / t0) ** (it / max(1, n_iter - 1))
        if not ones_list or not zeros_list:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        
        h[i] = 0
        h[j] = 1
        s_new = _score(h)
        delta = s_new - cur_s
        
        if delta < 0 or rng.random() < np.exp(max(-50, -delta / max(t, 1e-10))):
            cur_s = s_new

            zeros_list[zeros_list.index(j)] = i

                best_s = cur_s
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_s

        h = np.zeros(n)
        h[:target] = 1.0
        h = h[rng.permutation(n)]
    return h

def _binary_sa(h_init, n_iter=3000, t0=0.4, t1=0.03, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(h_init)
    if n % 2 != 0:
        n -= 1
    target = n // 2
    h_in = h_init[:n] if len(h_init) >= n else np.pad(h_init, (0, n - len(h_init)))
    h = (h_in > 0.5).astype(float)
    cn = int(h.sum())
    if cn > target:
        oi = np.where(h == 1)[0]
        h[rng.choice(oi, cn - target, replace=False)] = 0
    elif cn < target:
        zi = np.where(h == 0)[0]
        h[rng.choice(zi, target - cn, replace=False)] = 1
    best_h = h.copy()
    best_s = _score(h)
    cur_s = best_s
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    t_start = time.time()
    for it in range(n_iter):
        if time.time() - t_start > 8.0:
            break
        t = t0 * (t1 / t0) ** (it / max(1, n_iter - 1))
        if not ones_list or not zeros_list:
            break
        ii = rng.integers(len(ones_list))
        jj = rng.integers(len(zeros_list))
        i = ones_list[ii]
        j = zeros_list[jj]
        h[i] = 0.0
        h[j] = 1.0
        s_new = _score(h)
        delta = s_new - cur_s
        if delta < 0 or rng.random() < np.exp(max(-50.0, -delta / max(t, 1e-10))):
            cur_s = s_new
            ones_list[ii] = j
            zeros_list[jj] = i
            if cur_s < best_s:
                best_s = cur_s
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_s

def _chain_optimize(h0, deadline):
    n = len(h0)
    h_cur = h0.copy()
    s_cur = _score(h_cur)
    if time.time() < deadline - 15:
        h_new = _active_slsqp(h_cur, maxiter=120, n_active=10, tol=0.008)
        s_new = _score(h_new)
        if s_new < s_cur - 1e-12:
            h_cur, s_cur = h_new, s_new
    if time.time() < deadline - 12:
        h_cur, s_cur = _fista(h_cur, beta=200, n_iter=6000, alpha=0.004, mu=0.82)
    if time.time() < deadline - 10:
        h_new = _active_slsqp(h_cur, maxiter=150, n_active=12, tol=0.004)
        s_new = _score(h_new)
        if s_new < s_cur - 1e-12:
            h_cur, s_cur = h_new, s_new
    if time.time() < deadline - 8:
        h_cur, s_cur = _fista(h_cur, beta=500, n_iter=6000, alpha=0.003, mu=0.88)
    if time.time() < deadline - 5:
        h_new = _active_slsqp(h_cur, maxiter=150, n_active=14, tol=0.002)
        s_new = _score(h_new)
        if s_new < s_cur - 1e-12:
            h_cur, s_cur = h_new, s_new
    return h_cur, s_cur

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(25.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(56) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(parent)
    parent_n = len(parent)
    rng = np.random.default_rng(seed)

    # Phase 1: Generate diverse candidates
    candidates = []
    # Interpolated parent at nearby lengths
    for dt in [-4, -2, 0, 2, 4, 6]:
        n_try = parent_n + dt
        if n_try >= 20:
            candidates.append(_interp(parent, n_try))
    # Mirror
    candidates.append(_project(parent[::-1]))
    # Binary pushes
    for p in [1.3, 2.0, 3.0]:
        candidates.append(_project(parent ** p))
    # Sinusoidal perturbations
    for freq in [1, 2, 3]:
        phase = rng.uniform(0, 6.28)
        pert = 0.12 * np.sin(2.0 * np.pi * freq * np.arange(parent_n) / parent_n + phase)
        candidates.append(_project(parent + pert))

    # Binary SA for even n values
    sa_deadline = min(deadline - 200, t_start + 180)
    for n_try in range(max(24, parent_n - 8), min(72, parent_n + 10), 2):
        if time.time() > sa_deadline:
            break
        if abs(n_try - parent_n) <= 2:
            continue
        h_bin = _generate_binary(n_try, rng, 'random')
        h_sa, s_sa = _binary_sa(h_bin, n_iter=2500, t0=0.35, t1=0.04, rng=rng)
        candidates.append(h_sa)
        # Also add smoothed version
        h_s = _project(h_sa + 0.3 * np.convolve(h_sa, [0.1, 0.25, 0.3, 0.25, 0.1], 'same'))
        candidates.append(h_s)

    # Phase 2: Quick screen
    scored = []
    for h_c in candidates:
        if time.time() > deadline - 80:
            break
        s = _score(h_c)
        scored.append((s, h_c))

    scored.sort(key=lambda x: x[0])
    for s, h_c in scored[:5]:
        if s < best_score - 1e-12:
            best_h = h_c.copy()
            best_score = s
            best_n = len(h_c)

    # Phase 3: Deep optimization on top candidates
    n_deep = min(10, len(scored))
    deep_deadline = deadline - 60
    for idx in range(n_deep):
        if time.time() > deep_deadline:
            break
        h_init = scored[idx][1].copy()
        if len(h_init) < 20:
            continue
        h_cur, s_cur = _chain_optimize(h_init, deep_deadline)
        if s_cur < best_score - 1e-12:
            best_h = h_cur.copy()
            best_score = s_cur
            best_n = len(h_cur)

    # Phase 4: Intensive refinement on best
    if time.time() < deadline - 20:
        for tol_val in [0.01, 0.005, 0.002, 0.001]:
            if time.time() > deadline - 10:
                break
            h_new = _active_slsqp(best_h, maxiter=200, n_active=15, tol=tol_val)
            s_new = _score(h_new)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new
        if time.time() < deadline - 10:
            best_h, best_score = _fista(best_h, beta=800, n_iter=10000, alpha=0.002, mu=0.9)
        if time.time() < deadline - 8:
            h_new = _active_slsqp(best_h, maxiter=200, n_active=16, tol=0.001)
            s_new = _score(h_new)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new

    # Phase 5: Random restarts around best
    remaining = deadline - time.time()
    n_restarts = min(12, int(remaining / 25))
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        h_pert = _project(best_h + 0.025 * rng.standard_normal(best_n))
        h_new = _active_slsqp(h_pert, maxiter=100, n_active=10, tol=0.005)
        s_new = _score(h_new)
        if s_new < best_score - 1e-12:
            best_h = h_new
            best_score = s_new
        if time.time() < deadline - 15:
            h_cur, s_cur = _fista(h_pert, beta=400, n_iter=3000, alpha=0.003, mu=0.85)
            if s_cur < best_score - 1e-12:
                best_h = h_cur.copy()
                best_score = s_cur

    # Final polish
    if time.time() < deadline - 5:
        h_new = _active_slsqp(best_h, maxiter=150, n_active=15, tol=0.001)
        s_new = _score(h_new)
        if s_new < best_score - 1e-12:
            best_h = h_new
            best_score = s_new

    best_h = _project(best_h)
    final_score = _score(best_h)
    return (best_h, final_score, len(best_h))
