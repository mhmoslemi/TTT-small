def official_score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Binary search for shift
    lo = -4.0 - np.max(z)
    hi = 4.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

_G_EQ_TARGET = None

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -5.0 - np.max(z)
    hi = 5.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p=2.0):
    return make_feasible(np.power(np.clip(h, 0, 1), p))

def _lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    s = np.sum(np.exp(shifted))
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def _lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        s2 = k - (n-1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n-s2])
                grad[:n-s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n-m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n-m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - _G_EQ_TARGET

def _eq_jac(x):
    return np.ones(len(x))

def do_slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    global _G_EQ_TARGET
    _G_EQ_TARGET = n / 2.0
    try:
        res = minimize(
            _lse_obj, h0, args=(beta, dx),
            jac=_lse_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def _do_slsqp_direct(h0, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    global _G_EQ_TARGET
    _G_EQ_TARGET = n / 2.0
    t0 = official_score(h0) + 0.01
    t0 = min(t0, 0.99)
    z0 = np.concatenate([h0, [t0]])

    def _mm_obj(z):
        return z[-1]

    def _mm_obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _mm_ineq(z):
        hh = z[:-1]
        t = z[-1]
        corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
        return t - corr

    def _mm_ineq_jac(z):
        hh = z[:-1]
        nn = len(hh)
        L = 2*nn - 1
        jac = np.zeros((L, nn + 1))
        jac[:, -1] = 1.0
        for k in range(L):
            s2 = k - (nn - 1)
            if s2 >= 0:
                if s2 == 0:
                    for jj in range(nn):
                        jac[k, jj] += dx * (2.0 * hh[jj] - 1.0)
                else:
                    for jj in range(s2, nn):
                        jac[k, jj] += dx * (1.0 - hh[jj - s2])
                    for jj in range(nn - s2):
                        jac[k, jj] += dx * (-hh[jj + s2])
            else:
                m2 = -s2
                if m2 == 0:
                    for jj in range(nn):
                        jac[k, jj] += dx * (2.0 * hh[jj] - 1.0)
                else:
                    for jj in range(nn - m2):
                        jac[k, jj] += dx * (1.0 - hh[jj + m2])
                    for jj in range(m2, nn):
                        jac[k, jj] += dx * (-hh[jj - m2])
        return jac

    try:
        res = minimize(
            _mm_obj, z0, jac=_mm_obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac},

            ],
            options={'maxiter': maxiter, 'ftol': 1e-13}

        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            hh = res['x'][:-1]
            if np.min(hh) > -0.01 and np.max(hh) < 1.01:
                return make_feasible(hh)
    except Exception:
        pass
    return h0

def pair_ratchet(h_init, rng, time_limit, t_start):
    """Exact two-coordinate polynomial line minimization."""
    h = h_init.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(best_h)

    while time.time() - t_start < time_limit:
        h_score = official_score(h)
        if h_score < best_score - 1e-13:
            best_score = h_score
            best_h = h.copy()

        # Compute active-set weights for gradient
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        corr_n = corr_raw * dx
        m = np.max(corr_n)
        beta = 300.0
        shifted = np.clip(beta * (corr_n - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s > 1e-300:
            weights = e / s
        else:
            weights = np.ones(2*n-1) / (2*n-1)

        # Compute gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            w = weights[k]
            if w < 1e-10:
                continue
            s2 = k - (n-1)
            if s2 >= 0:
                if s2 == 0:
                    grad += (1.0 - h) - h
                else:
                    grad[s2:] += (1.0 - h[:n-s2])
                    grad[:n-s2] += (-h[s2:])
            else:
                m2 = -s2
                if m2 == 0:
                    grad += (1.0 - h) - h
                else:
                    grad[:n-m2] += (1.0 - h[m2:])
                    grad[m2:] += (-h[:n-m2])
        grad *= dx

        # Build pair list: top gradient donors, bottom receivers
        n_sel = min(n, 35)
        top_idx = np.argsort(grad)[::-1][:n_sel]
        bot_idx = np.argsort(grad)[:n_sel]
        pairs = []
        for i in top_idx:
            for j in bot_idx:
                if i != j:
                    pairs.append((i, j))
        # Add random pairs for diversity
        for _ in range(min(30, n)):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((i, j))
        # Remove duplicates
        pairs = list(set(pairs))
        if len(pairs) > 1:
            perm = rng.permutation(len(pairs))
            pairs = [pairs[p] for p in perm]

        any_improve = False
        for (i, j) in pairs:
            if time.time() - t_start > time_limit:
                break
            lo_t = max(h[i] - 1.0, -h[j])
            hi_t = min(h[i], 1.0 - h[j])
            if hi_t - lo_t < 1e-10:
                continue

            v = np.zeros(n)
            v[i] = -1.0
            v[j] = 1.0

            A_raw = np.correlate(h, 1.0 - h, mode='full')
            B_raw = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
            C_raw = -np.correlate(v, v, mode='full')

            # Coarse grid
            n_grid = 64
            t_vals = np.linspace(lo_t, hi_t, n_grid)
            # Evaluate: f(t) = dx * max_k (A_k + B_k*t + C_k*t^2)
            f_vals = np.max(A_raw[None,:] + B_raw[None,:]*t_vals[:,None] + C_raw[None,:]*t_vals[:,None]**2, axis=1) * dx
            best_idx = int(np.argmin(f_vals))
            best_t = t_vals[best_idx]
            best_f = f_vals[best_idx]

            # Local refinement
            if 0 < best_idx < n_grid - 1:
                tl = t_vals[best_idx - 1]
                th = t_vals[best_idx + 1]
            else:
                tl, th = lo_t, hi_t
            tl = max(lt, tl)
            th = min(hi_t, th)
            t_fine = np.linspace(tl, th, 40)
            f_fine = np.max(A_raw[None,:] + B_raw[None,:]*t_fine[:,None] + C_raw[None,:]*t_fine[:,None]**2, axis=1) * dx
            fi = int(np.argmin(f_fine))
            if f_fine[fi] < best_f:
                best_t = t_fine[fi]
                best_f = f_fine[fi]

            # Check endpoints and 0
            for t_chk in [lo_t, hi_t, 0.0]:
                if lo_t - 1e-12 <= t_chk <= hi_t + 1e-12:
                    t_chk = max(lo_t, min(hi_t, t_chk))
                    fc = np.max(A_raw + B_raw * t_chk + C_raw * t_chk**2) * dx
                    if fc < best_f:
                        best_t = t_chk
                        best_f = fc

            # Accept if improvement
            cur_f = official_score(h)
            if best_f < cur_f - 1e-13:
                h[i] -= best_t
                h[j] += best_t
                h = np.clip(h, 0.0, 1.0)
                h = make_feasible(h)
                new_score = official_score(h)
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_h = h.copy()
                any_improve = True

        if not any_improve:
            break

    return best_h, best_score

def subgradient_optimize(h0, max_iter=2000, time_limit=None, t_start=None):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = official_score(h)
    alpha = 0.03
    patience = 0
    max_patience = 400

    for it in range(max_iter):
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = np.max(corr)
        threshold = max_val - 0.03 * max_val
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for k in active:
            s2 = k - (n-1)
            if s2 >= 0:
                if s2 == 0:
                    grad += (1.0 - h) - h
                else:
                    grad[s2:] += (1.0 - h[:n-s2])
                    grad[:n-s2] += (-h[s2:])
            else:
                m2 = -s2
                if m2 == 0:
                    grad += (1.0 - h) - h
                else:
                    grad[:n-m2] += (1.0 - h[m2:])
                    grad[m2:] += (-h[:n-m2])
        grad /= len(active)
        h = make_feasible(h - alpha * grad)
        new_score = official_score(h)
        if new_score < best_score - 1e-13:
            best_score = new_score
            best_h = h.copy()
            patience = 0
            alpha = min(alpha * 1.02, 0.5)
        else:
            patience += 1
            alpha = max(alpha * 0.99, 0.003)
        if patience > max_patience:
            break
    return best_h, best_score

def _make_random_binary(n, rng):
    if n % 2 == 0:
        h = np.zeros(n)
        h[rng.choice(n, n//2, replace=False)] = 1.0
    else:
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
        h[rng.integers(n)] = 0.5
        h = make_feasible(h)
    return h

def _make_periodic(n, phase, period):
    """Deterministic balanced binary with given period and phase."""
    h = np.zeros(n)
    duty = period // 2
    for i in range(n):
        if (i + phase) % period < duty:
            h[i] = 1.0
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)

    # n values to try (prefer smaller)
    n_options = [28, 30, 32, 34, 36, 38, 40, 42, 44, 46, 48, 50, 52, 54]
    pn = len(parent)
    if pn not in n_options:
        n_options.append(pn)
    n_options = sorted(set(n for n in n_options if 24 <= n <= 64))

    # Phase 1: Quick scan - interpolate parent to each n, do brief ratchet
    quick_results = []
    for n in n_options:
        if time.time() > deadline - 60:
            break
        h_try = make_feasible(interp_to_n(parent, n))
        h_try = pair_ratchet(h_try, rng, time_limit=8, t_start=time.time())
        h_try, s_try = h_try, official_score(h_try)
        h_try = make_feasible(h_try)
        s_try = official_score(h_try)
        quick_results.append((s_try, n, h_try))
        if s_try < best_score - 1e-12:
            best_h = h_try.copy()
            best_score = s_try
            best_n = n

    quick_results.sort(key=lambda x: x[0])

    # Phase 2: Focused optimization on best results + diverse starts
    processed = set()
    for tier in range(3):
        if time.time() > deadline - 40:
            break

        if tier == 0:
            # Best quick results
            candidates = quick_results[:min(4, len(quick_results))]
            t_limit = 80 if n <= 40 else 50
        elif tier == 1:
            # More diverse starts at best n values
            candidates = []
            best_ns = list(set([r[1] for r in quick_results[:4]]))
            for n in best_ns[:3]:
                # Random binary
                rb = _make_random_binary(n, rng)
                candidates.append((official_score(rb), n, rb))
                # Periodic
                for per in [3, 5, 7, 4, 6]:
                    if per < n // 2:
                        ph = _make_periodic(n, seed % per, per)
                        candidates.append((official_score(ph), n, ph))
            t_limit = 60 if n <= 40 else 45
        else:
            # Perturbed best
            candidates = []
            for sigma in [0.03, 0.08]:
                hp = make_feasible(best_h + sigma * rng.standard_normal(best_n))
                candidates.append((official_score(hp), best_n, hp))
            t_limit = 40

        for (init_score, n, h0) in candidates:
            if time.time() > deadline - 30:
                break
            key = (n, tuple(np.round(h0, 2)))
            if key in processed:
                continue
            processed.add(key)

            cur_h = h0.copy()
            cur_score = official_score(cur_h)

            # Pair ratchet (main workhorse)
            if time.time() < deadline - 20:
                h0, s0 = cur_h, cur_score
                remaining_for_ratchet = min(t_limit, (deadline - time.time()) - 20)
                cur_h, cur_score = pair_ratchet(h0, rng, time_limit=remaining_for_ratchet, t_start=time.time())

            # SLSQP smooth polish
            if time.time() < deadline - 15:
                for beta in [100, 400]:
                    if time.time() > deadline - 10:
                        break
                    mi = 100 if n <= 44 else 60
                    new_h = do_slsqp_smooth(cur_h, beta, maxiter=mi)
                    new_s = official_score(new_h)
                    if new_s < cur_score - 1e-13:
                        cur_h = new_h
                        cur_score = new_s

            # Subgradient
            if time.time() < deadline - 10:
                sg_h, sg_s = subgradient_optimize(cur_h, max_iter=1500, time_limit=15, t_start=time.time())
                if sg_s < cur_score - 1e-13:
                    cur_h = sg_h
                    cur_score = sg_s

            # More pair ratchet after smoothing
            if time.time() < deadline - 10:
                cur_h, cur_score = pair_ratchet(cur_h, rng, time_limit=20, t_start=time.time())

            # Direct minmax
            if time.time() < deadline - 8:
                dc_h = _do_slsqp_direct(cur_h, maxiter=80)
                dc_s = official_score(dc_h)
                if dc_s < cur_score - 1e-13:
                    cur_h = dc_h
                    cur_score = dc_s

            # Final SLSQP
            if time.time() < deadline - 5:
                fh = do_slsqp_smooth(cur_h, 500, maxiter=80)
                fs = official_score(fh)
                if fs < cur_score - 1e-13:
                    cur_h = fh
                    cur_score = fs

            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
            elif abs(cur_score - best_score) <= 1e-12 and n < best_n:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = n

    # Final polish on best
    if time.time() < deadline - 5:
        remaining = deadline - time.time()
        if remaining > 30:
            best_h, best_score = pair_ratchet(best_h, rng, time_limit=min(remaining-10, 60), t_start=time.time())
        if time.time() < deadline - 5:
            fh = do_slsqp_smooth(best_h, 800, maxiter=100)
            fs = official_score(fh)
            if fs < best_score - 1e-13:
                best_h = fh
                best_score = fs
        if time.time() < deadline - 3:
            best_h, _ = pair_ratchet(best_h, rng, time_limit=10, t_start=time.time())

    # Final feasibility check
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if final_score > best_score + 1e-10:
        # Repair made it worse; use pre-repair version clipped
        best_h = np.clip(best_h, 0, 1)
        best_h = make_feasible(best_h)
        final_score = official_score(best_h)

    return (best_h, final_score, len(best_h))
