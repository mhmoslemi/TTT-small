```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def grad_f(h, s):
    """Unnormalized gradient of F_s = sum h[n+s]*(1-h[n]) (the raw correlation at shift s)."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            # h[j] as h[n+s]: j-s = n, valid if 0 <= j-s < n-... 
            # Actually: n ranges 0 to n-1-s. j = n+s ranges s to n-1.
            for j in range(s, n):
                g[j] += (1.0 - h[j - s])
            # h[j] as (1-h[n]): n = j, valid if 0 <= j <= n-1-s
            for j in range(0, n - s):
                g[j] -= h[j + s]
    else:
        t = -s
        if t == 0:
            pass  # handled above
        else:
            # n ranges t to n-1. j = n-t ranges 0 to n-1-t.
            for j in range(0, n - t):
                g[j] += (1.0 - h[j + t])
            # h[j] as (1-h[n]): n = j, valid if t <= j <= n-1-t
            for j in range(t, n - t):
                g[j] -= h[j - t]
    return g


def make_feasible(z):
    """Project z to [0,1]^n with sum = n/2."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    
    # Bisection to find theta such that sum(clip(z + theta, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-15:
                break
    
    return np.clip(h, 0.0, 1.0)


def interpolate_to(h_src, n_tgt):
    """Interpolate h from source resolution to target resolution."""
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)


def line_search_grid(h0, direction, max_evals=30):
    """Grid line search: find best alpha in [0,1] along h0 + alpha*direction."""
    best_alpha = 0.0
    best_score = official_score(h0)
    
    # Coarse grid
    alphas = np.linspace(0.0, 1.0, 11)
    for a in alphas[1:]:  # skip a=0, we already have that score
        h_try = make_feasible(h0 + a * direction)
        sc = official_score(h_try)
        if sc < best_score:
            best_score = sc
            best_alpha = a
    
    # Fine grid around best
    if best_alpha > 0:
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        fine_alphas = np.linspace(lo, hi, 10)
        for a in fine_alphas:
            h_try = make_feasible(h0 + a * direction)
            sc = official_score(h_try)
            if sc < best_score:
                best_score = sc
                best_alpha = a
    
    if best_alpha == 0.0:
        return None, best_score
    return h0 + best_alpha * direction, best_score


def qp_step(h0, selected_shifts, lam):
    """Solve QP: min t + lam*||h-h0||^2 s.t. linearized constraints, box, sum."""
    n = len(h0)
    f_vals = compute_f(h0)  # normalized by dx
    # Work with raw values (multiply back by 1/dx... actually let's just use normalized)
    # Constraints: f_val[s] + (grad * dx) · (h - h0) <= t
    # where f_val is already dx * raw
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    for s in selected_shifts:
        g = grad_f(h0, s) * (2.0 / n)  # normalized gradient
        idx = s + (n - 1)  # index in the full correlation output
        constraints.append(f_vals[idx] + g @ (h_var - h0) <= t_var)
    
    obj = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    
    h_new = None
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                break
        except Exception:
            continue
    
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)


def select_constraints(f_vals, h, threshold_frac=0.03, max_count=40):
    """Select shifts whose f values are within threshold of the max."""
    f_max = f_vals.max()
    threshold = f_max - threshold_frac * f_max
    # Also cap: don't select more than max_count
    selected = np.where(f_vals >= threshold)[0]
    if len(selected) > max_count:
        # Keep only the top max_count
        top_vals = f_vals[selected]
        order = np.argsort(top_vals)[::-1][:max_count]
        selected = selected[order]
    return selected


def subgradient_step(h0, f_vals, step_size):
    """Take a subgradient step: negative weighted average of active constraint gradients."""
    n = len(h0)
    f_max = f_vals.max()
    # Weights: softmax over (f_vals - f_max)
    shifted = f_vals - f_max
    shifted = np.clip(shifted, -10, 0)
    w = np.exp(shifted / 0.01)
    w /= w.sum()
    
    # Subgradient directions (negative of weighted gradient)
    d = np.zeros(n)
    n_p = len(f_vals)
    for idx in range(n_p):
        s = idx - (n - 1)
        if w[idx] < 1e-6:
            continue
        g = grad_f(h0, s)
        d -= w[idx] * g
    
    # Zero mean to preserve sum constraint
    d -= d.mean()
    
    # Scale
    d_norm = np.abs(d).max()
    if d_norm < 1e-12:
        return None
    d = d / d_norm * step_size
    
    return make_feasible(h0 + d)


def optimize_qp(h0, max_iter=30, threshold_frac=0.03, lam_start=0.5, lam_end=0.01):
    """Run QP-based optimization loop."""
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    stall = 0
    
    for it in range(max_iter):
        f_vals = compute_f(h)
        selected = select_constraints(f_vals, h, threshold_frac, max_count=40)
        
        lam = lam_start * (lam_end / lam_start) ** (it / max(1, max_iter - 1))
        h_new = qp_step(h, selected, lam)
        
        if h_new is None:
            stall += 1
            if stall >= 5:
                break
            continue
        
        # Line search
        direction = h_new - h
        h_ls, sc_ls = line_search_grid(h, direction, max_evals=20)
        
        if h_ls is not None and sc_ls < best - 1e-13:
            h = make_feasible(h_ls)
            best = official_score(h)
            stall = 0
            # Tighten threshold slightly
            threshold_frac = max(0.01, threshold_frac * 0.95)
        else:
            # Try subgradient step
            d_sg = subgradient_step(h, f_vals, 0.05)
            if d_sg is not None:
                h_sg, sc_sg = line_search_grid(h, d_sg - h, max_evals=10)
                if h_sg is not None and sc_sg < best - 1e-13:
                    h = make_feasible(h_sg)
                    best = official_score(h)
                    stall = 0
                    continue
            stall += 1
            if stall >= 6:
                break
    
    return h, best
