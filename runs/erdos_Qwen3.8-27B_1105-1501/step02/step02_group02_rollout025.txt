    """Find optimal δ for pair swap (i,j). Returns best δ and new score."""
    d = G[:, i] - G[:, j]  # K-vector
    # Feasible range
    delta_lo = max(-h[i], -(1.0 - h[j]))
    delta_hi = min(1.0 - h[i], h[j])
    if delta_hi - delta_lo < 1e-14:
        return 0.0, cv.max()
    
    # Minimize max_k (cv[k] + δ * d[k]) over δ ∈ [delta_lo, delta_hi]
    # This is a convex piecewise linear function of δ.
    
    # Binary search for where subgradient contains 0
    lo, hi = delta_lo, delta_hi
    for _ in range(60):  # ~1e-18 precision
        mid = (lo + hi) / 2.0

        active = np.where(vals >= mx - 1e-14)[0]

            hi = mid  # function decreasing, go left
        elif d_active.max() < -1e-15:
            lo = mid  # function increasing, go right
        else:
            break  # optimal
    
    delta_opt = (lo + hi) / 2.0
    new_cv = cv + delta_opt * d
    new_score = new_cv.max()

                elif d_act.max() < -1e-14:
                    lo = mid
                else:
                    break
            
            delta_opt = (lo + hi) / 2.0
            if abs(delta_opt) < 1e-15:
                continue
            
            new_cv = cv_new + delta_opt * d
            new_score = new_cv.max()
            
            if new_score < best_round - tol:
                h_new[i] += delta_opt
                h_new[j] -= delta_opt
                cv_new = new_cv
                best_round = new_score
        
        # Check actual improvement
        actual_score = _score(h_new)
        if actual_score < best_score - 1e-15:
            h = h_new
            best_score = actual_score
        else:
            # No improvement, stop
            break
    
    return h, _score(h)

import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _compute_G(h):
    n = len(h)
    K = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((K, n))
    for k in range(K):
        s = k - (n - 1)
        if s >= 0:
            if n - s > 0:
                m1 = np.arange(s, n)
                G[k, m1] += (1.0 - h[m1 - s])
                m2 = np.arange(0, n - s)
                G[k, m2] += (-h[m2 + s])
        else:
            t = -s
            if n - t > 0:
                m1 = np.arange(0, n - t)
                G[k, m1] += (1.0 - h[m1 + t])
                m2 = np.arange(t, n)
                G[k, m2] += (-h[m2 - t])
    G *= dx
    return G

def _pair_swap_polish(h, n_rounds=300, rng=None, tol=1e-13, deadline=None):
    n = len(h)
    K = 2 * n - 1
    best_score = _score(h)

    if rng is None:
        rng = np.random.default_rng(12345)

    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    for rnd in range(n_rounds):
        if deadline is not None and time.time() > deadline - 5:
            break
        cv = _corr_vec(h).copy()
        G = _compute_G(h)

        h_new = h.copy()
        cv_new = cv.copy()
        best_round = best_score
        n_applied = 0

        order = all_pairs[:]
        rng.shuffle(order)

        for (i, j) in order:
            d = G[:, i] - G[:, j]
            if np.max(np.abs(d)) < 1e-15:
                continue

            delta_lo = max(-h_new[i], -(1.0 - h_new[j]))
            delta_hi = min(1.0 - h_new[i], h_new[j])
            if delta_hi - delta_lo < 1e-14:
                continue

            lo, hi = delta_lo, delta_hi
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = cv_new + mid * d
                mx = vals.max()
                active = vals >= mx - 1e-13
                d_act = d[active]
                if d_act.min() > 1e-14:
                    hi = mid
                elif d_act.max() < -1e-14:
                    lo = mid
                else:
                    break

            delta_opt = (lo + hi) / 2.0
            if abs(delta_opt) < 1e-15:
                continue

            new_cv = cv_new + delta_opt * d
            new_score = new_cv.max()

            if new_score < best_round - tol:
                h_new[i] += delta_opt
                h_new[j] -= delta_opt
                cv_new = new_cv
                best_round = new_score
                n_applied += 1

        actual_score = _score(h_new)
        if actual_score < best_score - 1e-15:
            h = h_new
            best_score = actual_score
            if n_applied == 0:
                break
        else:
            if rnd > 20:
                break
            # Try sequential order next
            h = h_new.copy()
            best_score = actual_score

    return h, _score(h)

def _pair_swap_sequential(h, n_rounds=100, tol=1e-13, deadline=None):
    n = len(h)
    best_score = _score(h)
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    for rnd in range(n_rounds):
        if deadline is not None and time.time() > deadline - 5:
            break
        cv = _corr_vec(h).copy()
        G = _compute_G(h)
        improved = False

        for (i, j) in all_pairs:
            d = G[:, i] - G[:, j]
            if np.max(np.abs(d)) < 1e-15:
                continue
            delta_lo = max(-h[i], -(1.0 - h[j]))
            delta_hi = min(1.0 - h[i], h[j])
            if delta_hi - delta_lo < 1e-14:
                continue
            lo, hi = delta_lo, delta_hi
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = cv + mid * d
                mx = vals.max()
                active = vals >= mx - 1e-13
                d_act = d[active]
                if d_act.min() > 1e-14:
                    hi = mid
                elif d_act.max() < -1e-14:
                    lo = mid
                else:
                    break
            delta_opt = (lo + hi) / 2.0
            if abs(delta_opt) < 1e-15:
                continue
            new_cv = cv + delta_opt * d
            new_score = new_cv.max()
            if new_score < _score(h) - tol:
                h[i] += delta_opt
                h[j] -= delta_opt
                cv = new_cv
                improved = True

        if not improved:
            break

    return h, _score(h)

def _slsqp_minmax(h0, maxiter=100):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = min(cv.max() + 0.003, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv

    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _slsqp_lse(h0, beta, maxiter=80):
    n = len(h0)
    dx = 2.0 / n

    def _obj(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta

    def _obj_grad(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        G = _compute_G(x)
        # _corr_vec = np.correlate(h, 1-h, mode='full') * dx
        # d(corr_vec[k])/dh[m] = G[k,m]
        # d(obj)/dh[m] = sum_k w[k] * G[k,m]
        return G.T @ w

    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _forget_polish(h, n_iter=500, eta1=0.01, eta2=0.5, deadline=None):
    """Forget algorithm: multiplicative weights on lags + QP-like step on h."""
    n = len(h)
    K = 2 * n - 1
    best_h = h.copy()
    best_s = _score(h)
    w = np.ones(K) / K
    h_cur = h.copy()

    for it in range(n_iter):
        if deadline is not None and time.time() > deadline - 5:
            break
        cv = _corr_vec(h_cur)
        G = _compute_G(h_cur)
        # Weighted gradient: sum_k w_k * G[k, :]
        grad = G.T @ w
        # Step: h_new = project(h_cur - eta1 * grad)
        h_new = _project(h_cur - eta1 * grad)
        # Check improvement
        s_new = _score(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h_cur = h_new
        # Update weights (multiplicative weights)
        cv_new = _corr_vec(h_cur)
        w = w * np.exp(-eta2 * cv_new)
        w = w / w.sum()
        # Decay eta1 slightly
        eta1 *= 0.999

    return best_h, best_s
