import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def pair_move(h, i, j, n_samples=80):
    """Exact two-coordinate move: h[i] decreases, h[j] increases, sum preserved.
    Returns best h and score, or (None, inf) if no improvement."""
    n = len(h)
    dx = 2.0 / n
    # Feasible range for t: h[i]-t >= 0, h[j]+t <= 1, h[i]-t >= 0, h[j]+t >= 0
    t_min = max(-h[j], 0.0)  # actually h[j]+t >= 0 is always true if h[j] >= 0 and t >= -h[j]
    t_max = min(h[i], 1.0 - h[j])
    if t_max - t_min < 1e-10:
        return None, float('inf')
    
    # Build v: v[i] = -1, v[j] = +1
    v = np.zeros(n)
    v[i] = -1.0
    v[j] = 1.0
    
    # A = corr(h, 1-h), B = corr(v, 1-h) - corr(h, v), C = -corr(v, v)
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    
    # Scale by dx
    A *= dx
    B *= dx
    C *= dx
    
    # Sample t values
    ts = np.linspace(t_min, t_max, n_samples)
    
    # Evaluate: for each t, compute max_k (A_k + B_k*t + C_k*t^2)
    # Shape: (n_samples, 2n-1)
    X = np.outer(ts, np.ones(2*n-1))  # broadcast t
    vals = A[np.newaxis, :] + B[np.newaxis, :] * X + C[np.newaxis, :] * X**2
    obj = vals.max(axis=1)
    
    best_idx = np.argmin(obj)
    best_t = ts[best_idx]
    best_val = obj[best_idx]
    
    if best_val >= exact_score(h) - 1e-12:
        return None, float('inf')
    
    # Refine around best_t
    t_lo = max(ts[max(0, best_idx-3)], t_min)
    t_hi = min(ts[min(n_samples-1, best_idx+3)], t_max)
    ts_fine = np.linspace(t_lo, t_hi, 50)
    X2 = np.outer(ts_fine, np.ones(2*n-1))
    vals2 = A[np.newaxis, :] + B[np.newaxis, :] * X2 + C[np.newaxis, :] * X2**2
    obj2 = vals2.max(axis=1)
    best_idx2 = np.argmin(obj2)
    best_t = ts_fine[best_idx2]
    best_val = obj2[best_idx2]
    
    if best_val >= exact_score(h) - 1e-12:
        return None, float('inf')
    
    # Apply the move
    h_new = h.copy()
    h_new[i] -= best_t
    h_new[j] += best_t
    h_new = make_feasible(h_new)
    return h_new, best_val

def pair_pass(h, rng, n_pairs=50, time_limit=None, t_start=None):
    """Do multiple pair moves, accepting improvements."""
    n = len(h)
    cur_score = exact_score(h)
    
    # Compute gradient to guide pair selection
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    
    # Gradient of the max: use active set
    threshold = max_val - 0.05 * max_val
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                grad += (1.0 - h) - h
            else:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:])
        else:
            m = -s
            if m == 0:
                grad += (1.0 - h) - h
            else:
                grad[:n-m] += (1.0 - h[m:])
                grad[m:] += (-h[:n-m])
    grad /= len(active)
    
    # Select pairs: high h (donor) where grad is high, low h (receiver) where grad is low
    # We want to move mass from high-gradient to low-gradient coordinates
    # Actually, we want to decrease coordinates where it helps and increase where it helps
    # grad[k] > 0 means increasing h[k] increases the objective (bad)
    # grad[k] < 0 means increasing h[k] decreases the objective (good)
    
    # Donors: coordinates with high h and positive grad (decreasing them helps)
    # Receivers: coordinates with low h and negative grad (increasing them helps)
    
    scores = h * grad  # positive means this coordinate contributes to objective
    
    # Generate candidate pairs
    pairs = []
    # Gradient-based: sort by score, pair high with low
    order = np.argsort(scores)
    high_half = order[n//2:]
    low_half = order[:n//2]
    
    # Add top gradient-based pairs
    for _ in range(min(30, n//2)):
        i = rng.integers(len(high_half))
        j = rng.integers(len(low_half))
        pi = high_half[i]
        pj = low_half[j]
        if pi != pj:
            pairs.append((pi, pj))
    
    # Add random pairs
    for _ in range(20):
        i = rng.integers(n)
        j = rng.integers(n)
        if i != j:
            pairs.append((i, j))
    
    # Remove duplicates
    seen = set()
    unique_pairs = []
    for p, q in pairs:
        key = (min(p,q), max(p,q))
        if key not in seen:
            seen.add(key)
            unique_pairs.append((p, q))
    
    best_h = h.copy()
    best_score = cur_score
    improvements = 0
    
    for p, q in unique_pairs[:n_pairs]:
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        # Try both directions
        h_new, val = pair_move(best_h, p, q, n_samples=60)
        if h_new is not None:
            score = exact_score(h_new)
            if score < best_score - 1e-12:
                best_h = h_new
                best_score = score
                improvements += 1
                continue
        h_new, val = pair_move(best_h, q, p, n_samples=60)
        if h_new is not None:
            score = exact_score(h_new)
            if score < best_score - 1e-12:
                best_h = h_new
                best_score = score
                improvements += 1
    
    return best_h, best_score

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -4.0 - np.max(z)
    hi = 4.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        order = np.argsort(1.0 - h) if residual > 0 else np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
            else:
                add = max(residual, -h[idx])
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_subgradient(h, beta_width=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    threshold = max_val - beta_width * max_val
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for k in active:
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                grad[s2:] += (1.0 - h[:n - s2])
                grad[:n - s2] += (-h[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                grad[:n - m2] += (1.0 - h[m2:])
                grad[m2:] += (-h[:n - m2])
    grad /= len(active)
    return grad

def pair_move_exact(h, i, j, n_samples=100):
    n = len(h)
    t_min = 0.0
    t_max = min(h[i], 1.0 - h[j])
    if t_max - t_min < 1e-10:
        return None, float('inf')
    v = np.zeros(n)
    v[i] = -1.0
    v[j] = 1.0
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    dx = 2.0 / n
    A *= dx
    B *= dx
    C *= dx
    ts = np.linspace(t_min, t_max, n_samples)
    L = 2 * n - 1
    vals = A[np.newaxis, :] + B[np.newaxis, :] * ts[:, np.newaxis] + C[np.newaxis, :] * (ts[:, np.newaxis] ** 2)
    obj = vals.max(axis=1)
    best_idx = np.argmin(obj)
    if obj[best_idx] >= exact_score(h) - 1e-13:
        return None, float('inf')
    lo_i = max(0, best_idx - 2)
    hi_i = min(n_samples, best_idx + 3)
    ts_fine = np.linspace(ts[lo_i], ts[hi_i - 1], 60)
    vals2 = A[np.newaxis, :] + B[np.newaxis, :] * ts_fine[:, np.newaxis] + C[np.newaxis, :] * (ts_fine[:, np.newaxis] ** 2)
    obj2 = vals2.max(axis=1)
    best_idx2 = np.argmin(obj2)
    best_t = ts_fine[best_idx2]
    best_val = obj2[best_idx2]
    if best_val >= exact_score(h) - 1e-13:
        return None, float('inf')
    h_new = h.copy()
    h_new[i] -= best_t
    h_new[j] += best_t
    h_new = make_feasible(h_new)
    return h_new, best_val

def pair_descent_pass(h, rng, max_pairs=60, time_limit=None, t_start=None):
    n = len(h)
    cur_score = exact_score(h)
    grad = compute_subgradient(h, beta_width=0.08)
    pair_list = []
    order = np.argsort(grad)
    low_half = order[:n // 2]
    high_half = order[n // 2:]
    for _ in range(min(40, n // 2)):
        i = int(rng.integers(len(high_half)))
        j = int(rng.integers(len(low_half)))
        pi, pj = high_half[i], low_half[j]
        if pi != pj and h[pi] > 0.01 and h[pj] < 0.99:
            pair_list.append((int(pi), int(pj)))
    for _ in range(20):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and h[i] > 0.01 and h[j] < 0.99:
            pair_list.append((i, j))
    seen = set()
    unique_pairs = []
    for p, q in pair_list:
        key = (min(p, q), max(p, q))
        if key not in seen:
            seen.add(key)
            unique_pairs.append((p, q))
    best_h = h
    best_score = cur_score
    for p, q in unique_pairs[:max_pairs]:
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
            break
        h_new, val = pair_move_exact(best_h, p, q, n_samples=80)
        if h_new is not None:
            s = exact_score(h_new)
            if s < best_score - 1e-13:
                best_h = h_new
                best_score = s
                continue
        h_new, val = pair_move_exact(best_h, q, p, n_samples=80)
        if h_new is not None:
            s = exact_score(h_new)
            if s < best_score - 1e-13:
                best_h = h_new
                best_score = s
    return best_h, best_score

def slsqp_smooth_v2(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    L = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        return m + np.log(s) / beta if s > 1e-300 else m
    def grad_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            weights = np.ones(L) / L
        else:
            weights = e / s
        grad = np.zeros(n)
        for k in range(L):
            w = weights[k]
            if w < 1e-14:
                continue
            s2 = k - (n - 1)
            if s2 >= 0:
                if s2 == 0:
                    grad[:] += w * (1.0 - 2.0 * x)
                else:
                    grad[s2:] += w * (1.0 - x[:n - s2])
                    grad[:n - s2] += w * (-x[s2:])
            else:
                m2 = -s2
                if m2 == 0:
                    grad[:] += w * (1.0 - 2.0 * x)
                else:
                    grad[:n - m2] += w * (1.0 - x[m2:])
                    grad[m2:] += w * (-x[:n - m2])
        return grad * dx
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0.copy(), jac=grad_func, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0
