from itertools import combinations

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -4.0 - float(np.max(z))
    hi = 4.0 - float(np.min(z))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
                if add <= 0:
                    break
                h[i] += add
                residual -= add
            else:
                sub = min(-residual, h[i])
                if sub <= 0:
                    break
                h[i] -= sub
                residual += sub
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    b = (h > 0.5).astype(float)
    count = int(np.sum(b))
    if count == target:
        return b
    if count > target:
        ones_idx = np.where(b == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(count - target):
            b[ones_idx[i]] = 0
    else:
        zeros_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(target - count):
            b[zeros_idx[i]] = 1
    return b

    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def nchoosek(n, k):

        return 0
    k = min(k, n - k)

    for i in range(1, k + 1):

def compute_grad_at_k(h, k):
    n = len(h)
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        if offset == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[:n - offset] += (1.0 - h[offset:])
            grad[offset:] += -h[:n - offset]
    else:
        m = -offset
        if m == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] += -h[m:]
    return grad

def projected_gradient(h, n_iters=5000, lr_init=0.08, active_frac=0.08,
                       time_limit=None, t_start=None):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best = h_cur.copy()
    best_score = exact_score(best)
    lr = lr_init
    patience = 0
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        max_val = np.max(corr)
        threshold = max_val - active_frac * max_val
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for k in active:
            grad += compute_grad_at_k(h_cur, k)
        grad /= len(active)
        h_new = make_feasible(h_cur - lr * grad)
        sc = exact_score(h_new)
        if sc < best_score - 1e-14:
            best = h_new.copy()
            best_score = sc
            h_cur = h_new
            patience = 0
            lr = min(lr * 1.01, 0.5)

            patience += 1
            lr = max(lr * 0.97, 0.002)
        if patience > 800:
            break
        lr = max(lr * 0.9998, 0.002)
    return best, best_score

def _slsqp_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def _slsqp_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            if offset == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n - offset] += w * dx * (1.0 - x[offset:])
                grad[offset:] += w * dx * (-x[:n - offset])
        else:
            m2 = -offset
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[m2:] += w * dx * (1.0 - x[:n - m2])
                grad[:n - m2] += w * dx * (-x[m2:])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_refine(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            _slsqp_obj, h0, args=(beta, dx),
            jac=_slsqp_grad, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

            candidates.append(h_r_opt.copy())

        # Periodic patterns
        for period in [4, 5, 6, 7, 8, 10, 12]:
            if time.time() > deadline - 30:
                break
            h_p = periodic_pattern(n, period, rng)
            widths_p = [6, 8]
            h_p_opt, _ = block_search(h_p, widths_p, max_comb=20000,
                                       time_limit=alloc * 0.1, t_start=time.time())
            try_update(h_p_opt)
            candidates.append(h_p_opt.copy())

    # Phase 2: Continuous refinement on top candidates
    if time.time() < deadline - 30:
        # Sort candidates by score
        scored = [(exact_score(c), c) for c in candidates if len(c) == best_n or True]
        scored.sort(key=lambda x: x[0])
        top_cands = [c for _, c in scored[:10]]
        # Also always include best_h
        top_cands.append(best_h.copy())

        for h_c in top_cands:
            if time.time() > deadline - 20:
                break
            n_c = len(h_c)
            t_sub = time.time()
            h_pg, sc_pg = projected_gradient(h_c, n_iters=3000, lr_init=0.06,
                                             active_frac=0.06,
                                             time_limit=60, t_start=t_sub)
            try_update(h_pg)

    # Phase 3: SLSQP polish on best
    if time.time() < deadline - 15:
        for beta in [80, 200, 500]:
            if time.time() > deadline - 10:
                break
            h_sl = slsqp_refine(best_h, beta, maxiter=150)
            try_update(h_sl)

    # Phase 4: Final alternating refinement
    if time.time() < deadline - 10:
        for round_i in range(5):
            if time.time() > deadline - 5:
                break
            # Binarize current best, block search, then re-fractionate
            n_best = len(best_h)
            if n_best % 2 == 0:
                h_b = to_binary(best_h, n_best)
            else:
                h_b = to_half_binary(best_h, n_best)
            h_b, _ = block_search(h_b, [8, 10, 12], max_comb=30000,
                                   time_limit=30, t_start=time.time())
            h_pg, _ = projected_gradient(h_b, n_iters=2000, lr_init=0.05,
                                         active_frac=0.05,
                                         time_limit=30, t_start=time.time())
            try_update(h_pg)
            # SLSQP
            if time.time() < deadline - 5:
                h_sl = slsqp_refine(best_h, 300, maxiter=100)
                try_update(h_sl)

    # Final: one more SLSQP with high beta
    if time.time() < deadline - 3:
        h_final = slsqp_refine(best_h, 800, maxiter=100)
        try_update(h_final)

    best_h = make_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
