        is_ = i + sq  # i + s_q for each q

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _min_max_1d(A, b):
    lo = 0.0
    hi = 1.0

            hi = mid

def _cd_round(h, sq, rng):
    n = len(h)
    m = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full')
    order = rng.permutation(n)
    total_change = 0.0
    for i in order:
        is_ = i + sq
        js_ = i - sq
        b = np.zeros(m)
        vp = (is_ >= 0) & (is_ < n)
        vm = (js_ >= 0) & (js_ < n)

            h[i] = x_star
