import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_single(h, idx):
    n = len(h)
    dx = 2.0 / n
    s = idx - (n - 1)
    L = n - abs(s)
    if L <= 0:
        return np.zeros(n)
    g = np.zeros(n)
    if s >= 0:
        g[s:n] += (1.0 - h[:L])
        g[:L] -= h[s:n]
    else:
        t = -s
        g[:L] += (1.0 - h[t:])
        g[t:] -= h[:L]
    return g * dx

def blended_gradient(h, beta):
    n = len(h)
    V = compute_values(h)
    V_max = V.max()
    shifted = np.clip(beta * (V - V_max), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0 / len(V))
    else:
        w /= ws
    g = np.zeros(n)
    for idx in range(len(V)):
        if w[idx] > 1e-12:
            g += w[idx] * grad_single(h, idx)
    return g

def nesterov_step(h, m_vec, beta, alpha_init=0.15):
    n = len(h)
    y = h + m_vec
    V = compute_values(y)
    f = V.max()
    g = blended_gradient(y, beta)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f, m_vec
    g /= gn
    best_h = h.copy()
    best_f = f
    best_m = m_vec
    alpha = alpha_init
    for _ in range(16):
        step = alpha * g
        h_cand = make_feasible(y - step)
        f_cand = official_score(h_cand)
        if f_cand < best_f:
            best_h = h_cand
            best_f = f_cand
            new_m = m_vec + alpha * g
            nm = np.linalg.norm(new_m)
            if nm > 2.0 * alpha:
                new_m = new_m * (2.0 * alpha / nm)
            best_m = new_m
        alpha *= 0.55
    return best_h, best_f, best_m

def lp_polish(h, trust, max_iters=50, eps=0.005):
    n = len(h)
    current = h.copy()
    current_sc = official_score(current)
    for it in range(max_iters):
        V = compute_values(current)
        V_max = V.max()
        active = np.where(V >= V_max - eps)[0]
        if len(active) == 0:
            break
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_rows = []
        b_rows = []
        for p in active:
            gp = grad_single(current, p)
            row = np.zeros(n + 1)
            row[:n] = gp
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(-V[p])
        if not A_rows:
            break
        bounds_list = []
        for i in range(n):
            lo = max(-current[i], -trust)
            hi = min(1.0 - current[i], trust)
            bounds_list.append((lo, hi))
        bounds_list.append((0.0, None))
        A_eq = np.ones((1, n + 1))
        try:
            res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                         A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
            if res.success and res.x is not None:
                d = res.x[:n]
                if np.max(np.abs(d)) > 1e-10:
                    h_new = make_feasible(current + d)
                    sc = official_score(h_new)
                    if sc < current_sc - 1e-13:
                        current = h_new
                        current_sc = sc
                    else:
                        break
                else:
                    break
            else:
                break
        except Exception:
            break
    return current, current_sc

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def coord_descent(h, max_sweeps=50):
    n = len(h)
    best_sc = official_score(h)
    for sweep in range(max_sweeps):
        improved = False
        order = np.arange(n)
        for k in order:
            for tv in [0.0, 1.0]:
                delta = tv - h[k]
                if abs(delta) < 1e-10:
                    continue
                h_test = h.copy()
                h_test[k] = tv
                if delta > 0:
                    rooms = h_test.copy()
                    rooms[k] = 0.0
                    tot = rooms.sum()
                    if tot > 1e-10:
                        h_test[k:] -= delta * rooms / tot
                else:
                    rooms = 1.0 - h_test
                    rooms[k] = 0.0
                    tot = rooms.sum()
                    if tot > 1e-10:
                        h_test -= delta * rooms / tot
                h_test = make_feasible(h_test)
                sc = official_score(h_test)
                if sc < best_sc - 1e-12:
                    h = h_test
                    best_sc = sc
                    improved = True
        if not improved:
            break
    return h, best_sc
