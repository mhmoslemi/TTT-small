        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-13:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - best_a * 0.3)
    hi = min(alpha_max, best_a + best_a * 0.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(20):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

def _subgrad_optimize(h, deadline, target, T_init=0.4, max_iters=20000):
    """Subgradient descent with momentum and adaptive step"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)
    alpha = 0.08
    beta = 0.6
    T = T_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        scores = _scores(h)
        grad = _softmax_grad(h, scores, T)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        grad /= gnorm
        
        v = beta * v - alpha * grad
        v -= np.mean(v)
        
        a, new_c5 = _line_search_dir(h, v, target, cur_c5, max_alpha=1.5)
        if a > 0 and new_c5 < cur_c5 - 1e-13:
            h = _project(h + a * v, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            T = max(0.05, T * 0.985)
            beta = min(0.92, beta + 0.001)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            alpha *= 0.85
            T = max(0.05, T * 0.92)
            if stall > 30:
                v *= 0.3
                alpha = 0.05
                T = 0.2
    
    return best_h, best_c5

def _pair_search_smart(h, deadline, target, max_passes=50):
    """Coordinate descent with gradient-guided pair selection"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        
        # Compute gradient to guide pair selection
        scores = _scores(h)
        grad = _softmax_grad(h, scores, 0.15)
        grad -= np.mean(grad)
        
        # Select top coordinates by |gradient|
        top_count = min(n, 24)
        top = np.argsort(-np.abs(grad))[:top_count]
        
        # Also include some random coordinates
        if n > top_count:
            extras = rng.choice(list(set(range(n)) - set(top.tolist())), size=min(4, n - top_count), replace=False)
            top = np.concatenate([top, extras])
        
        for ii in range(len(top)):
            if time.time() > deadline:
                return best_h, best_c5
            j = top[ii]
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for jj in range(len(top)):
                if jj == ii:
                    continue
                k = top[jj]
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                    a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                    a_max = min(a_max_j, a_max_k)
                    if a_max < 1e-10:
                        continue
                    for frac in [0.25, 0.5, 0.75, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-12:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            cur_c5 = c_t
                            improved = True
                            break
        if not improved:
            # Try random pairs as fallback
            for _ in range(200):
                if time.time() > deadline:
                    return best_h, best_c5
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k or h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                dj = 1.0 if rng.random() > 0.5 else -1.0
                dk = -dj
                a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                a_max = min(a_max_j, a_max_k)
                if a_max < 1e-10:
                    continue
                for frac in [0.3, 0.6, 1.0]:
                    a = a_max * frac
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-12:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
            if not improved:
                break
    
    return best_h, best_c5

def _multi_start(h0, deadline, target, rng, n_starts=30):
    """Try multiple perturbation starting points"""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    for s in range(n_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 3.0:
            break
        alloc = min(remaining * 0.9 / max(1, n_starts - s), 25.0)
        sub_dl = time.time() + alloc
        
        # Generate perturbation
        r = rng.random()
        if r < 0.3:
            # Gaussian
            h_try = h0 + rng.normal(0, 0.03, n)
        elif r < 0.5:
            # Sparse perturbation
            h_try = h0.copy()
            idx = rng.choice(n, size=n // 8, replace=False)
            h_try[idx] += rng.normal(0, 0.08, len(idx))
        elif r < 0.65:
            # Binary push
            h_try = (h0 + rng.normal(0, 0.05, n) > 0.5).astype(float)
        elif r < 0.8:
            # Shift-like
            shift = rng.integers(1, max(2, n // 8))
            h_try = np.roll(h0, shift)
        elif r < 0.9:
            # Scale around mean
            h_try = 0.5 + (h0 - 0.5) * (0.8 + 0.4 * rng.random())
        else:
            # Interpolate with random
            h_rand = (rng.random(n) > 0.5).astype(float)
            h_try = 0.5 * h0 + 0.5 * h_rand
        
        h_try = _project(h_try, target)
        
        # Optimize from this start
        h_opt, c5_opt = _subgrad_optimize(h_try, sub_dl, target, T_init=0.3, max_iters=3000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5

def _fine_polish(h, deadline, target):
    """Very fine local optimization with tiny steps"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(99)
    
    while time.time() < deadline:
        scores = _scores(h)
        max_s = float(np.max(scores))
        # Try directions from top lags
        top_lags = np.argsort(-scores)[:6]
        
        improved = False
        for r in top_lags:
            if time.time() > deadline:
                break
            d = int(r) - (n - 1)
            grad = _lag_gradient(h, d)
            grad -= np.mean(grad)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-10:
                continue
            grad /= gnorm
            
            # Try both directions
            for sign in [1.0, -1.0]:
                a, c_t = _line_search_dir(h, sign * grad, target, cur_c5, max_alpha=0.3)
                if a > 0 and c_t < best_c5 - 1e-13:
                    h = _project(h + a * sign * grad, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        
        # Also try random micro-perturbations
        if not improved:
            for _ in range(20):
                if time.time() > deadline:
                    break
                h_try = _project(h + rng.normal(0, 0.01, n), target)
                c_try = _exact_c5(h_try)
                if c_try < best_c5 - 1e-13:
                    a, c_t = _line_search_dir(h, h_try - h, target, cur_c5, max_alpha=0.5)
                    if c_t < best_c5 - 1e-13:
                        h = _project(h + a * (h_try - h), target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
        if not improved:
            # Try pairwise fine moves
            for _ in range(100):
                if time.time() > deadline:
                    break
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k:
                    continue
                if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6 or h[k] <= 1e-6 or h[k] >= 1.0 - 1e-6:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = min((1-h[j])/dj if dj>0 else h[j]/(-dj),
                                (1-h[k])/dk if dk>0 else h[k]/(-dk))
                    if a_max <= 0:
                        continue
                    for a in [a_max * 0.1, a_max * 0.3, a_max * 0.5, a_max * 0.7, a_max]:
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            cur_c5 = c_t
                            improved = True
                            break
        if not improved:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Direct subgradient from initial (25%)
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = _subgrad_optimize(h, dl1, target0, T_init=0.4, max_iters=50000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Multi-start with diverse perturbations (25%)
    dl2 = t0 + budget_s * 0.50
    h2, c5_2 = _multi_start(best_h, dl2, best_n / 2.0, rng, n_starts=25)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Smart pair search (20%)
    dl3 = t0 + budget_s * 0.70
    h3, c5_3 = _pair_search_smart(best_h, dl3, best_n / 2.0, max_passes=100)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Try different resolutions (15%)
    dl4 = t0 + budget_s * 0.85
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, 12, -12, 16, -16]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 160 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + min(20.0, (dl4 - time.time()) * 0.4))
        h_r, c5_r = _subgrad_optimize(h_r, sub_dl, n_try / 2.0, T_init=0.3, max_iters=5000)
        # Quick pair polish
        h_r, c5_r = _pair_search_smart(h_r, min(dl4, time.time() + 5.0), n_try / 2.0, max_passes=20)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 5: Final fine polish (15%)
    dl5 = t0 + budget_s * 0.985
    h5, c5_5 = _fine_polish(best_h, dl5, best_n / 2.0)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Final validation and return
    best_h = _project(best_h, best_n / 2.0)
    # Ensure exact sum
    s = float(np.sum(best_h))
    diff = best_n / 2.0 - s
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        # Fallback to initial if something went wrong
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
