        for idx in order:
            if abs(residual) < 1e-15:
                break
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
    return np.clip(h, 0.0, 1.0)

def compute_V(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def weighted_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wk = w[k_idx]
        if wk * dx < 1e-14:
            continue
        s = k_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        coef = wk * dx
        if s >= 0:
            grad[s:n] += coef * (1.0 - h[0:L])
            grad[0:L] -= coef * h[s:n]
        else:
            t = -s
            grad[0:L] += coef * (1.0 - h[t:n])
            grad[t:n] -= coef * h[0:L]
    return grad

def smoothed_obj_and_grad(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exps = np.exp(shifted)
    ws = np.sum(exps)
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w = exps / ws
    sm = V_max + np.log(ws) / beta
    grad = weighted_gradient(h, w, dx)
    cv = np.sum(h) - n / 2.0
    sm += gamma * cv * cv
    grad += 2.0 * gamma * cv
    return sm, grad

def lbgfs_optimize(h0, n, beta, gamma, maxiter=800):
    def obj(x):
        val, _ = smoothed_obj_and_grad(x, beta, gamma)
        return val
    def grad(x):
        _, g = smoothed_obj_and_grad(x, beta, gamma)
        return g
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40})
        if res.x is not None:
            return res.x
    except Exception:
        pass
    return h0

def active_set_lp(h, eps, trust):
    n = len(h)
    dx = 2.0 / n
    V = compute_V(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx] + 1e-15)
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_x = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_x = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_x, src_x, h_src)
    return make_feasible(h_int)

def pg_step_fine(h, beta, alpha_max, max_ls):
    n = len(h)
    dx = 2.0 / n
    V = compute_V(h)
    f = V.max()
    shifted = beta * (V - V.max())
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w /= ws
    g = weighted_gradient(h, w, dx)
    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(8.0, 0.03 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
    except (NameError, AttributeError):
        n_init = 60
        x = np.arange(n_init) / n_init
        h0 = 0.5 + 0.25 * np.sin(3 * np.pi * x) + 0.1 * np.sin(7 * np.pi * x)
        h0 = make_feasible(h0)

    n = len(h0)
    best_h = make_feasible(h0)
    best_score = official_score(best_h)

    # Phase 1: L-BFGS-B with increasing beta (main optimization)
    beta_schedule = [10, 20, 35, 60, 100, 180, 320, 560, 1000, 1800]
    for beta in beta_schedule:
        if time_left() < 180:
            break
        gamma = max(2.0, 80.0 / beta)
        h_cur = best_h.copy()
        h_new = lbgfs_optimize(h_cur, n, beta, gamma, maxiter=600)
        h_new = make_feasible(h_new)
        sc = official_score(h_new)
        if sc < best_score - 1e-10:
            best_h = h_new.copy()
            best_score = sc

    # Phase 2: Active set LP
    if time_left() > 100:
        h_cur = best_h.copy()
        for eps in [0.01, 0.005, 0.003, 0.0015, 0.001]:
            if time_left() < 70:
                break
            for _ in range(40):
                if time_left() < 70:
                    break
                h_new, sc = active_set_lp(h_cur, eps, 0.12)
                if h_new is not None:
                    if sc < best_score - 1e-12:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    # Phase 3: Very high beta L-BFGS-B
    if time_left() > 80:
        for beta in [3000, 6000, 12000]:
            if time_left() < 50:
                break
            gamma = 1.0
            h_cur = best_h.copy()
            h_new = lbgfs_optimize(h_cur, n, beta, gamma, maxiter=400)
            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if sc < best_score - 1e-10:
                best_h = h_new.copy()
                best_score = sc

    # Phase 4: Fine projected gradient
    if time_left() > 60:
        h_cur = best_h.copy()
        beta = 200.0
        for _ in range(300):
            if time_left() < 40:
                break
            h_new, sc = pg_step_fine(h_cur, beta, 0.3, 12)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.02, 800.0)

    # Phase 5: Try different n values
    if time_left() > 150:
        n_base = len(best_h)
        n_candidates = sorted(set([n_base - 15, n_base - 10, n_base - 5,
                                   n_base + 5, n_base + 10, n_base + 15]))
        for n_try in n_candidates:
            if time_left() < 80 or n_try < 20 or n_try > 120:
                continue
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            h_cur = h_int.copy()
            # Quick L-BFGS-B optimization at this n
            for beta in [40, 150, 600]:
                if time_left() < 50:
                    break
                gamma = max(1.0, 40.0 / beta)
                h_new = lbgfs_optimize(h_cur, n_try, beta, gamma, maxiter=400)
                h_new = make_feasible(h_new)
                sc = official_score(h_new)
                if sc < best_score - 1e-10:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
            # Quick LP at this n
            if time_left() > 50:
                for _ in range(20):
                    if time_left() < 50:
                        break
                    h_new, sc = active_set_lp(h_cur, 0.003, 0.08)
                    if h_new is not None:
                        if sc < best_score - 1e-12:
                            best_h = h_new.copy()
                            best_score = sc
                        h_cur = h_new

    # Phase 6: Random restarts on best structure
    if time_left() > 100:
        for attempt in range(3):
            if time_left() < 50:
                break
            noise = rng.normal(0, 0.05, len(best_h))
            h_restart = make_feasible(best_h + noise)
            h_cur = h_restart.copy()
            for beta in [30, 100, 400]:
                if time_left() < 30:
                    break
                gamma = max(1.0, 50.0 / beta)
                h_new = lbgfs_optimize(h_cur, n, beta, gamma, maxiter=300)
                h_new = make_feasible(h_new)
                sc = official_score(h_new)
                if sc < best_score - 1e-10:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new

    # Phase 7: Final LP polish
    if time_left() > 15:
        h_cur = best_h.copy()
        for eps in [0.002, 0.001, 0.0005, 0.0002]:
            if time_left() < 8:
                break
            for _ in range(25):
                if time_left() < 8:
                    break
                h_new, sc = active_set_lp(h_cur, eps, 0.04)
                if h_new is not None:
                    if sc < best_score - 1e-13:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    # Phase 8: Final fine gradient
    if time_left() > 10:
        h_cur = best_h.copy()
        for _ in range(200):
            if time_left() < 5:
                break
            h_new, sc = pg_step_fine(h_cur, 500.0, 0.15, 8)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    return final_h, final_score + 1e-12, final_n
