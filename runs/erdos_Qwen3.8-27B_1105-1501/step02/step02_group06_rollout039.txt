from scipy.optimize import linprog
import time

def official_score(h):

    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, k_lag):
    """Gradient of the k-th lag value (before dx scaling) w.r.t. h.
    k_lag is the index in the 'full' correlate output, ranging from 0 to 2n-2.
    The actual lag offset is off = k_lag - (n-1).
    """
    n = len(h)
    off = k_lag - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        # c = sum_{i=0}^{n-1-off} h[i+off] * (1-h[i])
        for i in range(n - off):
            g[i] += -h[i + off]
            g[i + off] += 1.0 - h[i]
    else:
        t = -off
        # c = sum_{i=0}^{n-1-t} h[i] * (1-h[i+t])
        for i in range(n - t):
            g[i] += 1.0 - h[i + t]
            g[i + t] += -h[i]
    return g

def project_feasible(z, n):

    target = n / 2.0
    # Bisection on shift

        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual

    if res > 1e-12:
        # Need to add more

    Returns (direction, step_bound) or (None, 0) if no descent direction found.
    """

    upper_bound = h >= 1.0 - tol_bound

    #   -1 <= d[i] <= 1 for all i

        else:
            bounds.append((-1.0, 1.0))

Actually, one more thought. Let me consider a completely different approach: **direct bilinear optimization using a penalty method**.

The objective is f(h) = max_k F_k(h) where F_k(h) = sum_i h_i (1-h_{i+k}) dx.

I can use the smooth approximation: f_smooth(h) = (1/beta) * log(sum_k exp(beta * F_k(h)))

And then minimize this with gradient descent / L-BFGS-B, subject to the constraints.

The gradient of f_smooth is:
∂f_smooth/∂h_i = (1/beta) * sum_k exp(beta*F_k) * (∂F_k/∂h_i) / sum_k exp(beta*F_k)
= sum_k (exp(beta*F_k) / sum_j exp(beta*F_j)) * (∂F_k/∂h_i)

This is a convex combination of the gradients of the individual F_k, weighted by their softmax. This is a well-known technique in robust optimization.

For large beta, this approaches the max. For smaller beta, it's smoother and easier to optimize.

def grad_lag(h, k_idx):
    n = len(h)

            g[i + off] += 1.0 - h[i]
