def lse_value_weight(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = vals.max()
    shifted = beta * (vals - m)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    val = m + np.log(max(w_sum, 1e-300)) / beta
    return val, w, vals

def lse_grad_fast(h, w):
    n = len(h)
    dx = 2.0 / n
    s1 = convolve(w, 1.0 - h, mode='full')
    s2 = convolve(w, h[::-1], mode='full')
    grad = dx * (s1[n - 1:2 * n - 1] - s2[n - 1:2 * n - 1][::-1])
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def slsqp_fast(h0, beta, maxiter=100):
    n = len(h0)
    def obj(x):
        val, w, _ = lse_value_weight(x, beta)
        return val
    def grad(x):
        _, w, _ = lse_value_weight(x, beta)
        return lse_grad_fast(x, w)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = exact_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def binary_local_search(h_bin, max_iters=500, is_dead=None):
    n = len(h_bin)
    best = h_bin.copy()
    best_sc = exact_score(best)
    ones = np.where(best == 1.0)[0]
    zeros = np.where(best == 0.0)[0]
    improved = True
    it = 0
    while improved and it < max_iters:
        if is_dead is not None and is_dead():
            break
        improved = False

        best_pair = None
        for a in ones:
            for b in zeros:
                best[a] = 0.0
                best[b] = 1.0
                sc = exact_score(best)
                best[a] = 1.0

            zeros = (zeros[zeros != b]).tolist()
            ones.append(b)
            zeros.append(a)
            best_sc = exact_score(best)
            improved = True
        it += 1
    return best, best_sc

def two_point_search(h, n_tries, rng, is_dead):

    best = h.copy()
    best_sc = exact_score(best)
    for it in range(n_tries):

    # Phase 6: More SLSQP with high beta
    if not is_dead():
        for beta in [500, 1000, 2000]:
            if is_dead():
                break
            h_new, sc = slsqp_fast(best_h, beta, maxiter=60)
            if h_new is not None and sc < best_score - 1e-12:
                best_h = h_new
                best_score = sc

    # Phase 7: Resolution exploration
    if not is_dead():
        n_base = len(best_h)
        n_opts = set()
        for dn in [-8, -4, 4, 8, 12, -12]:
            nt = n_base + dn
            if 30 <= nt <= 100:
                n_opts.add(nt)
        for nc in [40, 48, 56, 60, 64, 72, 80, 88]:
            if 30 <= nc <= 100:
                n_opts.add(nc)
        n_opts = sorted(n_opts)
        for n_opt in n_opts:
            if is_dead():
                break
            if n_opt == len(best_h):
                continue
            time_left = deadline - time.time()
            if time_left < 5:
                break
            h_int = interp_to(best_h, n_opt)
            sc_int = exact_score(h_int)
            # Quick SLSQP
            for beta in [80, 300]:
                if is_dead():
                    break
                h_new, sc = slsqp_fast(h_int, beta, maxiter=50)
                if h_new is not None:
                    h_int = h_new
                    sc_int = sc
            # Two-point
            if not is_dead():
                nt2 = max(500, int((deadline - time.time()) * 2))
                h_int, sc_int = two_point_search(h_int, nt2, rng, is_dead)
            if sc_int < best_score - 1e-12:
                best_h = h_int
                best_score = sc_int

    # Phase 8: Binary search at best resolution
    if not is_dead():
        n_cur = len(best_h)
        # Try binary at current and nearby resolutions
        for nb in [n_cur, n_cur - 2, n_cur + 2, n_cur - 4, n_cur + 4]:
            if is_dead() or nb < 20 or nb > 120:
                continue
            h_b = interp_to(best_h, nb)
            h_b = binarize(h_b)
            sc_b = exact_score(h_b)
            h_b, sc_b = binary_local_search(h_b, max_iters=300, is_dead=is_dead)
            if sc_b < best_score - 1e-12:
                best_h = h_b
                best_score = sc_b
            elif sc_b < best_score + 0.005 and not is_dead():
                # Polish from binary
                h_pol = make_feasible(0.5 * h_b + 0.5 * interp_to(best_h, nb))
                for beta in [100, 400]:
                    if is_dead():
                        break
                    h_new, sc = slsqp_fast(h_pol, beta, maxiter=50)
                    if h_new is not None and sc < best_score - 1e-12:
                        best_h = h_new
                        best_score = sc

    # Phase 9: Diverse starting points
    if not is_dead():
        # Wave-based starts
        for freqs, amps in [[(1, 2), (0.25, 0.2)], [(2, 3), (0.2, 0.15)],
                            [(1, 3, 5), (0.15, 0.1, 0.05)]]:
            if is_dead():
                break
            n_w = len(best_h)
            h_w = make_wave(n_w, list(freqs), list(amps))
            h_w, sc_w = two_point_search(h_w, 800, rng, is_dead)
            if not is_dead():
                h_new, sc = slsqp_fast(h_w, 200, maxiter=50)
                if h_new is not None and sc < sc_w:
                    h_w, sc_w = h_new, sc
            if sc_w < best_score - 1e-12:
                best_h = h_w
                best_score = sc_w

    # Phase 10: Final intensive polish
    if not is_dead():
        n_tries = max(3000, int((deadline - time.time()) * 5))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 11: Final subgradient
    if not is_dead():
        best_h, best_score = subgradient_polish(best_h, 200, is_dead)

    # Phase 12: One more SLSQP at very high beta
    if not is_dead():
        h_new, sc = slsqp_fast(best_h, 5000, maxiter=40)
        if h_new is not None and sc < best_score - 1e-12:
            best_h = h_new
            best_score = sc

    # Final two-point sweep
    if not is_dead():
        n_tries = max(1000, int((deadline - time.time()) * 3))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = exact_score(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-9

    return final_h, final_score, final_n
