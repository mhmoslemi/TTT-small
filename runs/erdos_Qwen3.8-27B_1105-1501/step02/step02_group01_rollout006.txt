from scipy.optimize import minimize
from itertools import combinations
import time

def exact_score(h):

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -4.0 - np.max(z), 4.0 - np.min(z)

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)

    if abs(residual) > 1e-10:
        if residual > 0:

        else:
            order = np.argsort(h)[::-1]

            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    assert n % 2 == 0
    k = n // 2
    idx = np.argsort(h)[::-1][:k]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def block_patterns(w, k):
    """Yield all binary patterns of length w with exactly k ones."""
    if k == 0:
        yield np.zeros(w)
        return
    if k == w:
        yield np.ones(w)
        return
    for ones in combinations(range(w), k):
        p = np.zeros(w)
        p[list(ones)] = 1.0
        yield p

def block_cd(h, max_w=10, max_rounds=50, time_limit=120.0, t_start=None):
    """Exact block coordinate descent on binary string."""
    n = len(h)
    h = h.copy()
    best_score = exact_score(h)
    
    for width in range(3, max_w + 1):
        for rnd in range(max_rounds):
            if t_start is not None and time.time() - t_start > time_limit:
                break
            improved_this_round = False
            for s in range(n - width + 1):
                if t_start is not None and time.time() - t_start > time_limit:
                    break
                k = int(np.round(np.sum(h[s:s+width])))
                if k < 0 or k > width:
                    continue
                cur_block = h[s:s+width]
                cur_score = best_score
                
                best_pattern = None
                best_pat_score = cur_score
                
                for pat in block_patterns(width, k):
                    if np.array_equal(pat, cur_block):
                        continue
                    h[s:s+width] = pat
                    sc = exact_score(h)
                    if sc < best_pat_score - 1e-13:
                        best_pat_score = sc
                        best_pattern = pat.copy()
                
                if best_pattern is not None:
                    h[s:s+width] = best_pattern
                    best_score = best_pat_score
                    improved_this_round = True
            
            if not improved_this_round:
                break
    
    return h, best_score

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        w = weights[k_idx] * dx
        if w < 1e-14:
            continue
        s2 = k_idx - (n-1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - 2.0 * x) * w
            else:
                grad[s2:] += (1.0 - x[:n-s2]) * w
                grad[:n-s2] += (-x[s2:]) * w
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - 2.0 * x) * w
            else:
                grad[:n-m2] += (1.0 - x[m2:]) * w
                grad[m2:] += (-x[:n-m2]) * w
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0, args=(beta, dx),
            jac=smooth_lse_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -4.0 - np.max(z), 4.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)


def to_binary(h):
    n = len(h)
    if n % 2 != 0:
        n = n - 1
        h = h[:n]
    k = n // 2
    idx = np.argsort(h)[::-1][:k]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)


def block_patterns_list(w, k):
    """Precompute all binary patterns of length w with exactly k ones as array of shape (C(w,k), w)."""
    if k == 0:
        return np.zeros((1, w))
    if k == w:
        return np.ones((1, w))
    patterns = []
    for ones in combinations(range(w), k):
        p = np.zeros(w)
        p[list(ones)] = 1.0
        patterns.append(p)
    return np.array(patterns)


def block_cd(h, max_w=10, max_rounds=40, time_limit=120.0, t_start=None, rng=None):
    """Exact block coordinate descent on binary string."""
    n = len(h)
    h = h.copy()
    best_score = exact_score(h)
    k_n = n // 2

    # Precompute pattern tables
    pattern_cache = {}
    for w in range(3, max_w + 1):
        for kk in range(0, w + 1):
            if kk <= w - kk:
                pattern_cache[(w, kk)] = block_patterns_list(w, kk)

    for width in range(3, max_w + 1):
        for rnd in range(max_rounds):
            if t_start is not None and time.time() - t_start > time_limit:
                break
            improved_this_round = False

            # Determine order: forward, backward, or random
            if rng is not None and rnd % 3 == 2:
                positions = list(range(n - width + 1))
                rng.shuffle(positions)
            elif rnd % 2 == 1:
                positions = list(range(n - width, -1, -1))
            else:
                positions = list(range(n - width + 1))

            for s in positions:
                if t_start is not None and time.time() - t_start > time_limit:
                    break
                k = int(np.round(np.sum(h[s:s + width])))
                if k < 0 or k > width:
                    continue
                cur_block = h[s:s + width].copy()
                pats = pattern_cache.get((width, k))
                if pats is None or len(pats) == 0:
                    continue
                if len(pats) == 1:
                    continue

                best_pat_idx = -1
                best_pat_score = best_score

                for pi in range(len(pats)):
                    if np.array_equal(pats[pi], cur_block):
                        continue
                    h[s:s + width] = pats[pi]
                    sc = exact_score(h)
                    if sc < best_pat_score - 1e-14:
                        best_pat_score = sc
                        best_pat_idx = pi

                if best_pat_idx >= 0:
                    h[s:s + width] = pats[best_pat_idx]
                    best_score = best_pat_score
                    improved_this_round = True
                else:
                    h[s:s + width] = cur_block

            if not improved_this_round:
                break

    return h, best_score


def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta


def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx] * dx
        if w < 1e-14:
            continue
        s2 = k_idx - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - 2.0 * x) * w
            else:
                grad[s2:] += (1.0 - x[:n - s2]) * w
                grad[:n - s2] += (-x[s2:]) * w
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - 2.0 * x) * w
            else:
                grad[:n - m2] += (1.0 - x[m2:]) * w
                grad[m2:] += (-x[:n - m2]) * w
    return grad


def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def _eq_jac(x):
    return np.ones(len(x))


def slsqp_smooth(h0, beta, maxiter=80):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0.copy(), args=(beta, dx),
            jac=smooth_lse_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0


def generate_seeds(parent, n, rng):
    """Generate diverse starting binary strings for dimension n."""
    seeds = []
    h_interp = make_feasible(interp_to_n(parent, n))
    seeds.append(to_binary(h_interp))

    # Push variants
    for p in [1.5, 3.0, 6.0]:
        pushed = make_feasible(h_interp ** p)
        seeds.append(to_binary(pushed))

    # Random binary
    k = n // 2
    rb = np.zeros(n)
    rb[rng.choice(n, k, replace=False)] = 1.0
    seeds.append(rb)

    # Periodic patterns with shifts
    for period in [4, 5, 6, 7, 8, 9, 10, 12]:
        n_ones_in_period = max(1, period // 2)
        base = np.zeros(period)
        base[:n_ones_in_period] = 1.0
        for shift in range(min(3, period)):
            shifted_base = np.roll(base, shift)
            tiled = np.tile(shifted_base, n // period + 1)[:n]
            n_ones = int(np.sum(tiled))
            if n_ones > k:
                ones_idx = np.where(tiled == 1)[0]
                to_remove = ones_idx[rng.choice(len(ones_idx), n_ones - k, replace=False)]
                tiled[to_remove] = 0
            elif n_ones < k:
                zeros_idx = np.where(tiled == 0)[0]
                to_add = zeros_idx[rng.choice(len(zeros_idx), k - n_ones, replace=False)]
                tiled[to_add] = 1
            seeds.append(tiled)

    # Inverse of interpolated
    seeds.append(to_binary(1.0 - h_interp))

    return seeds


def kick_restart(h, rng, n_kicks=3):
    """Randomly flip pairs of bits to escape local minima."""
    n = len(h)
    k = n // 2
    h = h.copy()
    score = exact_score(h)
    for _ in range(n_kicks):
        h2 = h.copy()
        for _ in range(3):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if h2[i] > 0.5 and h2[j] < 0.5:
                h2[i] = 0.0
                h2[j] = 1.0
            elif h2[i] < 0.5 and h2[j] > 0.5:
                h2[i] = 1.0
                h2[j] = 0.0
        s2 = exact_score(h2)
        if s2 < score:
            h = h2
            score = s2
    return h, score
