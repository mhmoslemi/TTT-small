        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_sweep(h, order, n_iter=20, temp=0.0, rng=None):
    cur_c5 = _exact_c5(h)
    for (j, k) in order:
        s = h[j] + h[k]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-12:
            continue
        hj0, hk0 = h[j], h[k]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h[j], h[k] = c1, s - c1
        f1 = _exact_c5(h)
        h[j], h[k] = c2, s - c2
        f2 = _exact_c5(h)
        for _ in range(n_iter):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi = c2; c2 = c1; f2 = f1
                c1 = hi - gr * (hi - lo)
                h[j], h[k] = c1, s - c1
                f1 = _exact_c5(h)
            else:
                lo = c1; c1 = c2; f1 = f2
                c2 = lo + gr * (hi - lo)
                h[j], h[k] = c2, s - c2
                f2 = _exact_c5(h)
        t_opt = (lo + hi) * 0.5
        h[j], h[k] = t_opt, s - t_opt
        c5_new = _exact_c5(h)
        if c5_new < cur_c5 - 1e-14:
            cur_c5 = c5_new
        elif temp > 0 and c5_new - cur_c5 < temp and rng is not None:
            if rng.random() < np.exp(-(c5_new - cur_c5) / temp):
                cur_c5 = c5_new
            else:
                h[j], h[k] = hj0, hk0
        else:
            h[j], h[k] = hj0, hk0
    return cur_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new, x_old, h), n_new / 2.0)
