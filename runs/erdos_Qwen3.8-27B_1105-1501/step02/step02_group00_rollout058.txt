def _grad_matrix(h):
    """Return G such that G[r, j] = d(corr[r])/d(h[j])."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(n):
        m = n - d
        if m <= 0: break
        r = n - 1 + d
        G[r, :m] += (1.0 - h[d:d+m])
        G[r, d:] += -h[:m]
        if d > 0:
            r2 = n - 1 - d
            G[r2, d:d+m] += (1.0 - h[:m])
            G[r2, :m] += -h[d:d+m]
    return G

def _lp_step(h, scores, G, radius):
    """Find direction p via LP: min t s.t. G*p <= -scores + t, sum(p)=0, |p|<=radius."""
    n = len(h)
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-10)
        hi = min(radius, 1.0 - h[j] - 1e-10)
        if lo >= hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x[-1] < -1e-10:
            return res.x[:n], res.x[-1]

        pass

The issue is that the LP finds the best single direction to reduce the max correlation, but it might take many small steps. Also, the LP might find a direction that reduces the current max but creates a new max elsewhere.

**Key insight**: The problem is a "min-max" problem. At the optimum, multiple lags will be tied for the maximum (similar to the minimax theorem in game theory). So the optimal solution should have several lags achieving the same (maximum) correlation value.

This suggests a different approach: **alternating between optimizing for different lags**. At each iteration:
1. Identify the top k lags (those closest to the maximum)
2. Optimize h to minimize the average (or max) of these k lags
3. Repeat

This is like a "mirror descent" on the max function, where we focus on the "active" constraints.

Another idea: **use the dual problem**. The primal is:
min_h max_k corr_k/h
= min_h t  s.t. corr_k ≤ t*h for all k, 0≤h≤1, sum(h)=n/2

This is bilinear (product of h and t, and corr is bilinear in h). The dual would be... complicated.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = v.astype(float).copy()
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n == n_new: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new, x_old, h), n_new / 2.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for d in range(n):
        m = n - d
        if m <= 0: break
        r = n - 1 + d
        G[r, :m] += omh[d:d+m]
        G[r, d:] += -h[:m]
        if d > 0:
            r2 = n - 1 - d
            G[r2, d:d+m] += omh[:m]
            G[r2, :m] += -h[d:d+m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-10)
        hi = min(radius, 1.0 - h[j] - 1e-10)
        if lo >= hi: lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x[-1] < -1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.4
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] * 0.99)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.99)
    alpha_max = max(alpha_max, 1e-10)
    best_a, best_c5 = 0.0, c5_cur
    for a in np.linspace(0.05, alpha_max, 15):
        h_t = h + a * p
        c = _exact_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo = max(0.0, best_a - alpha_max * 0.12)
    hi = min(alpha_max, best_a + alpha_max * 0.12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1, c2 = hi - gr*(hi-lo), lo + gr*(hi-lo)
    f1, f2 = _exact_c5(h+c1*p), _exact_c5(h+c2*p)
    for _ in range(22):
        if hi - lo < 1e-13: break
        if f1 < f2:
            hi, f2 = c2, f1; c2 = c1
            c1 = hi - gr*(hi-lo); f1 = _exact_c5(h+c1*p)
        else:
            lo, f1 = c1, f2; c1 = c2
            c2 = lo + gr*(hi-lo); f2 = _exact_c5(h+c2*p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5: return opt_a, opt_c5
    return best_a, best_c5

def _lp_optimize(h, deadline, radius_init=0.1, max_iters=3000):

    target = n / 2.0

        if stall > 60: break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None:
            stall += 2
            radius *= 0.85
            if radius < 1e-5: break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.35)
        else:

            radius *= 0.82
            if stall % 12 == 0 and radius < 0.012:
                radius = 0.04
    return best_h, best_c5

def _subgrad_optimize(h, deadline, lr_init=0.01, max_iters=2000):
    """Subgradient descent on the max of correlations."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    lr = lr_init
    for it in range(max_iters):
        if time.time() > deadline: break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Pick top 3 lags weighted
        top_k = 3
        idx = np.argsort(-corr)[:top_k]
        w = np.zeros(2*n-1)
        w[idx] = 1.0
        w /= top_k
        # Subgradient: weighted combination of gradients of those lags
        grad = np.zeros(n)
        omh = 1.0 - h
        for d in range(n):
            m = n - d
            if m <= 0: break
            wd = w[n-1+d] * dx
            if abs(wd) > 1e-14:
                grad[:m] += wd * omh[d:d+m]
                grad[d:] += -wd * h[:m]
            if d > 0:
                wd2 = w[n-1-d] * dx
                if abs(wd2) > 1e-14:
                    grad[d:d+m] += wd2 * omh[:m]
                    grad[:m] += -wd2 * h[d:d+m]
        h_new = h - lr * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.998
    return best_h, best_c5

def _penalty_lbbfgs(h0, deadline, lam_init=None):
    """Penalty method with L-BFGS-B."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h0 = h0.copy()
    best_c5 = _exact_c5(h0)
    best_h = h0.copy()
    if lam_init is None:
        lam_init = best_c5 * 1.05
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        exceed = np.maximum(0.0, corr - lam_init)
        pen = np.sum(exceed * exceed)
        csum = np.sum(x) - target
        return pen + 100.0 * csum * csum
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        active = np.maximum(0.0, corr - lam_init)
        g = np.zeros(n)
        omh = 1.0 - x
        for d in range(n):
            m = n - d
            if m <= 0: break
            a = 2.0 * active[n-1+d] * dx
            if abs(a) > 1e-14:
                g[:m] += a * omh[d:d+m]
                g[d:] += -a * x[:m]
            if d > 0:
                a2 = 2.0 * active[n-1-d] * dx
                if abs(a2) > 1e-14:
                    g[d:d+m] += a2 * omh[:m]
                    g[:m] += -a2 * x[d:d+m]
        g += 200.0 * (np.sum(x) - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10,
                               'maxfun': 5000})
    h_out = np.clip(result.x, 0, 1)
    h_out = _project(h_out, target)
    c5_out = _exact_c5(h_out)
    if c5_out < best_c5:
        best_c5 = c5_out
        best_h = h_out.copy()
    return best_h, best_c5

def _coord_polish(h, deadline, max_passes=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    deltas = [0.003, 0.007, -0.003, -0.007, 0.015, -0.015, 0.001, -0.001]
    for _pass in range(max_passes):
        if time.time() > deadline: break
        improved = False
        for j in range(n):
            if time.time() > deadline: break
            hv = h[j]
            if hv < 5e-5 or hv > 1.0 - 5e-5: continue
            for k in [(j+1) % n, (j-1) % n, (j+n//2) % n]:
                if k == j: continue
                hk = h[k]
                if hk < 5e-5 or hk > 1.0 - 5e-5: continue
                found = False
                for dt in deltas:
                    nj, nk = hv + dt, hk - dt
                    if nj < 0 or nj > 1 or nk < 0 or nk > 1: continue
                    h_t = h.copy()
                    h_t[j], h_t[k] = nj, nk
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        found = True
                        break
                if found: break
        if not improved: break
    return best_h, best_c5
