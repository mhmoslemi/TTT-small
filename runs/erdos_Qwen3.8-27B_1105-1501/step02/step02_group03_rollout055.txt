from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def symmetrize(h):
    n = len(h)
    h_sym = np.empty(n)
    for i in range(n):
        h_sym[i] = 0.5 * (h[i] + (1.0 - h[n-1-i]))
    # For odd n, center should be 0.5
    if n % 2 == 1:
        h_sym[n//2] = 0.5
    return make_feasible(h_sym)

    """Convert symmetric h to z variables (logistic inverse of left half)"""
    n = len(h)
    m = n // 2
    u = h[:m].copy()
    # Clip to avoid log(0)
    u = np.clip(u, 1e-10, 1.0 - 1e-10)
    z = np.log(u / (1.0 - u))
    return z

    m = len(z)
    n = 2 * m
    u = 1.0 / (1.0 + np.exp(-z))
    h = np.empty(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

    n = len(h)
    dx = 2.0 / n

            G[p, :L] -= dx * h[s:]

        else:

            G[p, :K] += dx * (1.0 - h[t:])

    n = 2 * len(z) + (1 if odd else 0)
    h = z_to_h_odd(z) if odd else z_to_h(z)
    m = len(z)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Numerically stable log-sum-exp
    max_shifted = shifted.max()
    logsum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    obj = cmax + logsum / beta
    
    w = np.exp(shifted - max_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:

    else:

    u = 1.0 / (1.0 + np.exp(-z))
    sigmoid_deriv = u * (1.0 - u)

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)


def symmetrize(h):
    n = len(h)
    h_sym = np.empty(n)
    for i in range(n):
        h_sym[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    if n % 2 == 1:
        h_sym[n // 2] = 0.5
    return make_feasible(h_sym)


def h_to_z(h):
    n = len(h)
    m = n // 2
    u = h[:m].copy()
    u = np.clip(u, 1e-10, 1.0 - 1e-10)
    z = np.log(u / (1.0 - u))
    return z


def z_to_h(z):
    m = len(z)
    n = 2 * m
    u = 1.0 / (1.0 + np.exp(-z))
    h = np.empty(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h


def z_to_h_odd(z):
    m = len(z)
    n = 2 * m + 1
    u = 1.0 / (1.0 + np.exp(-z))
    h = np.empty(n)
    h[:m] = u
    h[m] = 0.5
    h[m + 1:] = 1.0 - u[::-1]
    return h


def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def grad_vals(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] -= dx * h[s:]
            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K]
    return G


def smooth_obj_grad_z(z, beta, odd=False):
    h = z_to_h_odd(z) if odd else z_to_h(z)
    m = len(z)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_shifted = shifted.max()
    logsum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted - max_shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_vals(h)
    if odd:
        grad_u = G[:, :m] - G[:, m + 1:][:, ::-1]
    else:
        grad_u = G[:, :m] - G[:, m:][:, ::-1]
    u = 1.0 / (1.0 + np.exp(-z))
    sd = u * (1.0 - u)
    grad_z = (w @ grad_u) * sd
    return obj, grad_z


def find_active(vals, frac=0.2, min_count=8):
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)


def interpolate_sym(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return symmetrize(h_int)

def solve_sym_lbgfs(z0, beta_list, time_limit, odd=False, maxiter=300):

            if res.x is not None and np.all(np.isfinite(res.x)):
                z_cur = res.x.copy()
        except Exception:
            pass
    return z_cur

    def update(h_new, sc):
        nonlocal best_h, best_score
        if np.all(np.isfinite(h_new)) and np.isfinite(sc):
            if sc < best_score - 1e-15:
                best_h = h_new.copy()
                best_score = sc
            elif abs(sc - best_score) <= 1e-15 and best_h is not None and len(h_new) < len(best_h):
                best_h = h_new.copy()
                best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        x = np.arange(n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * x / n0) + 0.1 * np.cos(4 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    update(best_h, best_score)
    n_par = len(best_h)

    # Symmetrize and try
    h_sym_par = symmetrize(best_h)
    sc_sym = official_score(h_sym_par)
    if sc_sym < best_score - 1e-14:
        update(h_sym_par, sc_sym)

    def attempt_symmetric(n_target, z0, time_avail):
        """Run symmetric optimization at given resolution."""
        odd = (n_target % 2 == 1)
        m = n_target // 2
        # Validate z0 length
        if len(z0) != m:
            h_tmp = z_to_h(z0) if not odd else z_to_h_odd(z0)
            h_interp = interpolate_sym(best_h, n_target)
            h_interp = symmetrize(h_interp)
            z0 = h_to_z(h_interp)
            m = len(z0)

        t0 = time.time()
        # Phase 1: L-BFGS-B with increasing beta
        beta_list = [20, 50, 150, 500, 2000, 8000]
        remaining = time_avail - (time.time() - t0)
        if remaining > 3:
            z_new = solve_sym_lbgfs(z0, beta_list, time_limit=remaining * 0.5, odd=odd, maxiter=200)
            h_new = z_to_h_odd(z_new) if odd else z_to_h(z_new)
            h_new = make_feasible(h_new)
            sc_new = official_score(h_new)
            if sc_new < best_score - 1e-14:
                update(h_new, sc_new)
            # Re-extract z from improved h
            h_s = symmetrize(h_new)
            z_new = h_to_z(h_s)

        # Phase 2: Active set SLSQP
        remaining = time_avail - (time.time() - t0)
        if remaining > 3:
            h_cur = z_to_h_odd(z_new) if odd else z_to_h(z_new)
            h_cur = make_feasible(h_cur)
            for attempt in range(3):
                if check_deadline() or (time_avail - (time.time() - t0)) < 2:
                    break
                vals = compute_vals(h_cur)
                active = find_active(vals, frac=0.25, min_count=10)
                z_res = solve_sym_active(z_new, active, maxiter=400, odd=odd)
                h_res = z_to_h_odd(z_res) if odd else z_to_h(z_res)
                h_res = make_feasible(h_res)
                sc_res = official_score(h_res)
                if sc_res < best_score - 1e-14:
                    update(h_res, sc_res)
                h_cur = h_res
                z_new = h_to_z(symmetrize(h_res))

        # Phase 3: High beta L-BFGS-B polish
        remaining = time_avail - (time.time() - t0)
        if remaining > 3:
            for beta_hi in [15000, 50000]:
                if check_deadline() or (time_avail - (time.time() - t0)) < 2:
                    break
                prob = SmoothZProblem(beta_hi, odd)
                bounds = [(-9.0, 9.0)] * len(z_new)
                try:
                    res = minimize(prob.fun, z_new, jac=prob.jac, method='L-BFGS-B',
                                   bounds=bounds, options={'maxiter': 150, 'gtol': 1e-14})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_hi = z_to_h_odd(res.x) if odd else z_to_h(res.x)
                        h_hi = make_feasible(h_hi)
                        sc_hi = official_score(h_hi)
                        if sc_hi < best_score - 1e-14:
                            update(h_hi, sc_hi)
                        z_new = h_to_z(symmetrize(h_hi))
                except Exception:
                    pass

    # Main symmetric optimization at parent resolution
    if time_left() > 30:
        h_s = symmetrize(best_h)
        z_start = h_to_z(h_s)
        attempt_symmetric(n_par, z_start, time_left() * 0.4)

    # Try different resolutions
    if time_left() > 30:
        n_opts = sorted(set([
            max(40, n_par - 4),
            max(42, n_par - 2),
            n_par,
            min(76, n_par + 2),
            min(78, n_par + 4),
            min(80, n_par + 6),
        ]))
        for n_opt in n_opts:
            if check_deadline() or time_left() < 10:
                break
            if n_opt == n_par:
                continue
            hv = best_h.copy()
            h_int = interpolate_sym(hv, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-14:
                update(h_int, sc_int)
            z_opt = h_to_z(symmetrize(h_int))
            attempt_symmetric(n_opt, z_opt, min(time_left() * 0.25, 40.0))

    # Full (non-symmetric) SLSQP polish at best resolution
    if time_left() > 20:
        for _ in range(2):
            if check_deadline():
                break
            vals_f = compute_vals(best_h)
            active_f = find_active(vals_f, frac=0.3, min_count=12)
            h_f = solve_full_active(best_h, active_f, maxiter=400)
            h_f = make_feasible(h_f)
            sc_f = official_score(h_f)
            if sc_f < best_score - 1e-14:
                update(h_f, sc_f)
            # Symmetrize the result and continue
            h_sym_f = symmetrize(best_h)
            sc_sf = official_score(h_sym_f)
            if sc_sf < best_score - 1e-14:
                update(h_sym_f, sc_sf)

    # Random perturbation restarts in z-space
    if time_left() > 40:
        h_s_best = symmetrize(best_h)
        z_best = h_to_z(h_s_best)
        n_best = len(best_h)
        m_best = len(z_best)
        odd_best = (n_best % 2 == 1)
        sigmas = [0.3, 0.15, 0.08, 0.04, 0.02]
        n_restarts = max(3, min(10, int(time_left() / 12)))
        for i in range(n_restarts):
            if check_deadline() or time_left() < 8:
                break
            sigma = sigmas[i % len(sigmas)]
            z pert = np.clip(z_best + rng.normal(0, sigma, size=m_best), -9.0, 9.0)
            h_pert = z_to_h_odd(z_pert) if odd_best else z_to_h(z_pert)
            h_pert = make_feasible(h_pert)
            sc_pert = official_score(h_pert)
            # Quick L-BFGS-B polish
            if time_left() > 10:
                z_p = solve_sym_lbgfs(z_pert, [200, 2000], time_limit=min(8.0, time_left() * 0.3),
                                      odd=odd_best, maxiter=100)
                h_p2 = z_to_h_odd(z_p) if odd_best else z_to_h(z_p)
                h_p2 = make_feasible(h_p2)
                sc_p2 = official_score(h_p2)
                if sc_p2 < sc_pert:
                    h_pert, sc_pert, z_pert = h_p2, sc_p2, z_p
                if sc_pert < best_score - 1e-14:
                    update(h_pert, sc_pert)
                # Active set
                if time_left() > 8:
                    vals_p = compute_vals(h_pert)
                    act_p = find_active(vals_p, frac=0.25, min_count=10)
                    z_act = solve_sym_active(z_pert, act_p, maxiter=300, odd=odd_best)
                    h_act = z_to_h_odd(z_act) if odd_best else z_to_h(z_act)
                    h_act = make_feasible(h_act)
                    sc_act = official_score(h_act)
                    if sc_act < best_score - 1e-14:
                        update(h_act, sc_act)

    # Final intensive polish at best solution
    if time_left() > 15:
        h_final_s = symmetrize(best_h)
        sc_fs = official_score(h_final_s)
        if sc_fs < best_score - 1e-14:
            update(h_final_s, sc_fs)
        z_final = h_to_z(h_final_s)
        n_final = len(best_h)
        odd_final = (n_final % 2 == 1)
        m_final = len(z_final)

        # Very high beta
        for beta_vh in [20000, 80000]:
            if check_deadline():
                break
            prob = SmoothZProblem(beta_vh, odd_final)
            bounds = [(-10.0, 10.0)] * m_final
            try:
                res = minimize(prob.fun, z_final, jac=prob.jac, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 200, 'gtol': 1e-15})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_vh = z_to_h_odd(res.x) if odd_final else z_to_h(res.x)
                    h_vh = make_feasible(h_vh)
                    sc_vh = official_score(h_vh)
                    if sc_vh < best_score - 1e-14:
                        update(h_vh, sc_vh)
                    z_final = h_to_z(symmetrize(h_vh))
            except Exception:
                pass

        # Final active set
        if time_left() > 5:
            h_cur = z_to_h_odd(z_final) if odd_final else z_to_h(z_final)
            h_cur = make_feasible(h_cur)
            vals_fin = compute_vals(h_cur)
            for frac in [0.3, 0.5]:
                if check_deadline():
                    break
                act_fin = find_active(vals_fin, frac=frac, min_count=12)
                z_fin = solve_sym_active(z_final, act_fin, maxiter=600, odd=odd_final)
                h_fin = z_to_h_odd(z_fin) if odd_final else z_to_h(z_fin)
                h_fin = make_feasible(h_fin)
                sc_fin = official_score(h_fin)
                if sc_fin < best_score - 1e-14:
                    update(h_fin, sc_fin)
                    z_final = h_to_z(symmetrize(h_fin))
                    vals_fin = compute_vals(h_fin)

        # Full SLSQP final
        if time_left() > 5:
            vals_f2 = compute_vals(best_h)
            act_f2 = find_active(vals_f2, frac=0.4, min_count=15)
            h_f2 = solve_full_active(best_h, act_f2, maxiter=600)
            h_f2 = make_feasible(h_f2)
            sc_f2 = official_score(h_f2)
            if sc_f2 < best_score - 1e-14:
                update(h_f2, sc_f2)

    # Final validation
    final_h = make_feasible(best_h)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_n = len(final_h)
    final_score = official_score(final_h)

    assert np.all(final_h >= -1e-15), "Negative values"
    assert np.all(final_h <= 1.0 + 1e-15), "Values > 1"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-10, "Sum constraint violated"

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
