def project_feasible(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target} using bisection on threshold."""
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(proj)
    if abs(residual) < 1e-10:
        return proj
    # Distribute residual among non-boundary elements
    interior = (proj > 1e-10) & (proj < 1.0 - 1e-10)
    if np.sum(interior) > 0:
        per = residual / np.sum(interior)
        proj[interior] = np.clip(proj[interior] + per, 0.0, 1.0)
    return proj

def smooth_objective(h, T):
    """Compute smooth max of correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Stable log-sum-exp
    m = np.max(corr)
    vals = np.exp((corr - m) / T)
    s = np.sum(vals)
    return m + T * np.log(s)

def smooth_grad(h, T):
    """Gradient of smooth_max_T(correlation) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp((corr - m) / T)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    # For lag s = k - (n-1):
    # dr_k/dh_j = (1 - h[j+s]) - h[j-s]  (out of bounds = 0)

        if w < 1e-15:

        # (1-h[j+s]) for valid j+s
        for j in range(n):

            if j + s >= 0 and j + s < n:
                val += (1.0 - h[j+s])
            if j - s >= 0 and j - s < n:

def _smooth_value(x, T, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = (corr - m) / T
    scaled = np.clip(scaled, -60, 60)
    exps = np.exp(scaled)
    return dx * (m + T * np.log(np.sum(exps)))


def _smooth_grad(x, T, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

def pair_line_search(h, i, j):
    lo = -h[i]
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] -= m1
        h[j] += m1
        f1 = compute_c5(h)
        h[i] = h[i] + m1
        h[j] = h[j] - m1
        h[i] -= m2
        h[j] += m2
        f2 = compute_c5(h)
        h[i] = h[i] + m2
        h[j] = h[j] - m2
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_sweep(h, rng, n_pairs=None):
    n = len(h)
    if n_pairs is None:
        n_pairs = n * (n - 1) // 2
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    rng.shuffle(pairs)
    pairs = pairs[:n_pairs]
    improved = False
    for i, j in pairs:
        d = pair_line_search(h, i, j)
        if abs(d) > 1e-12:
            h[i] -= d
            h[j] += d
            improved = True
    return improved


def make_sine_init(n, freq, phase):

def make_random_binary(n, rng):

def resample_h(h_old, new_n):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / new_n
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    h_new = project_feasible(h_new, new_n / 2.0)
    return h_new

    h_new = h - step_size * grad
    return project_feasible(h_new, target)


def run_slsqp_anneal(h0, n, deadline, rng):
    dx = 2.0 / n
    h = h0.copy()
    target = n / 2.0
    best_local = h.copy()
    best_c5_local = compute_c5(h)
    temps = [4.0, 3.0, 2.2, 1.6, 1.2, 0.9, 0.65, 0.5, 0.38, 0.28, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]
    for T in temps:
        if time.time() > deadline:
            break
        maxiter = 40 if T > 1.5 else 30
        proposal = slsqp_optimize_with_grad(h, T, n, maxiter=maxiter)
        proposal = project_feasible(proposal, target)
        c5_prop = compute_c5(proposal)
        if c5_prop < best_c5_local - 1e-10:
            best_c5_local = c5_prop
            best_local = proposal.copy()
            h = proposal.copy()
        else:
            h = project_feasible(0.5 * h + 0.5 * proposal, target)
    return best_local, best_c5_local


def basin_hop_iter(h0, n, T_list, rng, deadline):
    h = h0.copy()
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(h)
    perturb_scale = 0.03
    for _ in range(20):
        if time.time() > deadline:
            break
        h_pert = h + rng.randn(n) * perturb_scale
        h_pert = project_feasible(h_pert, target)
        for T in T_list:
            if time.time() > deadline:
                break
            proposal = slsqp_optimize_with_grad(h_pert, T, n, maxiter=30)
            proposal = project_feasible(proposal, target)
            c5_p = compute_c5(proposal)
            if c5_p < best_c5 - 1e-10:
                best_c5 = c5_p
                best = proposal.copy()
                h = proposal.copy()
            elif c5_p < best_c5 + 0.001:
                h = proposal.copy()
            else:
                h = h_pert
                break
        perturb_scale = max(0.01, perturb_scale * 0.9)
    pair_sweep(best, rng, n_pairs=n * 15)
    best = project_feasible(best, target)
    best_c5 = compute_c5(best)
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = t0 + budget_s - 3.0

    # Get parent
    n_base = 70
    parent = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) == n_base and np.all(np.isfinite(h0)):
            parent = h0
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.full(n_base, 0.5)

    parent = project_feasible(parent, n_base / 2.0)
    parent_c5 = compute_c5(parent)

    best_h = parent.copy()
    best_c5 = parent_c5
    best_n = n_base

    def try_update(cand, cand_c5, cand_n):
        nonlocal best_h, best_c5, best_n
        if cand_c5 < best_c5 - 1e-12:
            best_c5 = cand_c5
            best_h = cand.copy()
            best_n = cand_n
        elif abs(cand_c5 - best_c5) < 1e-12 and cand_n < best_n:
            best_h = cand.copy()
            best_n = cand_n

    # Phase 1: SLSQP annealing on parent (35% time)
    phase1_end = t0 + budget_s * 0.35
    h_annealed, c5_annealed = run_slsqp_anneal(parent, n_base, phase1_end, rng)
    try_update(h_annealed, c5_annealed, n_base)

    # Quick pair polish
    hp = best_h.copy()
    pair_sweep(hp, rng, n_pairs=500)
    hp = project_feasible(hp, n_base / 2.0)
    c5p = compute_c5(hp)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Phase 2: Basin hopping (35% time)
    phase2_end = t0 + budget_s * 0.70
    T_hop = [1.5, 0.8, 0.4, 0.2, 0.1]
    for attempt in range(5):
        if time.time() > phase2_end:
            break
        scale = 0.04 + 0.02 * attempt
        h_start = best_h + rng.randn(n_base) * scale
        h_start = project_feasible(h_start, n_base / 2.0)
        h_bp, c5_bp = basin_hop_iter(h_start, n_base, T_hop, rng, phase2_end)
        try_update(h_bp, c5_bp, n_base)

        if time.time() > phase2_end:
            break
        # Also try from the annealed result
        h_start2 = h_annealed + rng.randn(n_base) * 0.03
        h_start2 = project_feasible(h_start2, n_base / 2.0)
        h_bp2, c5_bp2 = basin_hop_iter(h_start2, n_base, T_hop, rng, phase2_end)
        try_update(h_bp2, c5_bp2, n_base)

    # Phase 3: Multi-resolution (20% time)
    phase3_end = t0 + budget_s * 0.90
    test_ns = [46, 52, 58, 64, 72, 78, 84]
    for n_test in test_ns:
        if time.time() > phase3_end:
            break
        h_res = resample_h(best_h, n_test)
        # SLSQP at a few temperatures
        c5_res = compute_c5(h_res)
        if c5_res < best_c5 - 0.0005:
            # Worth optimizing further
            h_opt, c5_opt = run_slsqp_anneal(h_res, n_test, min(phase3_end - 1.0, t0 + budget_s * 0.04), rng)
            h_opt = project_feasible(h_opt, n_test / 2.0)
            c5_opt = compute_c5(h_opt)
            pair_sweep(h_opt, rng, n_pairs=n_test * 10)
            h_opt = project_feasible(h_opt, n_test / 2.0)
            c5_opt = compute_c5(h_opt)
            try_update(h_opt, c5_opt, n_test)
        else:
            # Quick local optimization
            h_q = slsqp_optimize_with_grad(h_res, 0.3, n_test, maxiter=30)
            h_q = project_feasible(h_q, n_test / 2.0)
            c5_q = compute_c5(h_q)
            if c5_q < best_c5 - 1e-10:
                pair_sweep(h_q, rng, n_pairs=n_test * 8)
                h_q = project_feasible(h_q, n_test / 2.0)
                c5_q = compute_c5(h_q)
                try_update(h_q, c5_q, n_test)

    # Phase 4: Final polish (remaining time)
    remaining = hard_deadline - time.time()
    if remaining > 5.0:
        n_final = best_n
        h_final = best_h.copy()
        target = n_final / 2.0
        # Multiple pair sweeps with random ordering
        n_rounds = max(1, int(remaining * 0.5))
        for _ in range(n_rounds):
            if time.time() > hard_deadline - 2.0:
                break
            pair_sweep(h_final, rng, n_pairs=n_final * 12)
            h_final = project_feasible(h_final, target)
            c5_f = compute_c5(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()
                best_n = n_final

        # Final SLSQP polish
        if time.time() < hard_deadline - 8.0:
            h_pol = slsqp_optimize_with_grad(best_h, 0.05, best_n, maxiter=50)
            h_pol = project_feasible(h_pol, best_n / 2.0)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_h = h_pol.copy()
                best_c5 = c5_pol
            if time.time() < hard_deadline - 5.0:
                h_pol2 = slsqp_optimize_with_grad(best_h, 0.02, best_n, maxiter=40)
                h_pol2 = project_feasible(h_pol2, best_n / 2.0)
                c5_pol2 = compute_c5(h_pol2)
                if c5_pol2 < best_c5:
                    best_h = h_pol2.copy()
                    best_c5 = c5_pol2

    # Final validation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
