def make_feasible(z):
    """Project z to feasible [0,1]^N with sum = N/2"""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0

            lo = mid
        else:

    residual = target - np.sum(h)

        for idx in order:

            if add <= 0: break
            h[idx] += add
            residual -= add

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -8.0, 8.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-14:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0:
                break
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    elif residual < -1e-14:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 0:
                break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def symmetrize(h):
    n = len(h)
    h_sym = np.zeros(n)

        j = n - 1 - i
        h_sym[i] = 0.5 * (h[i] + (1.0 - h[j]))
        h_sym[j] = 1.0 - h_sym[i]
    if n % 2 == 1:

    return h_sym

def compute_grad_matrix(h, dx):
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    for p in range(K):

            a = -s
            L = n - a
            G[p, a:a + L] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[a:a + L]

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -500.0, 500.0)))

def sigmoid_grad(z):
    s = sigmoid(z)
    return s * (1.0 - s)

def z_to_h(z, n):
    m = n // 2
    h = np.zeros(n)
    s = sigmoid(z)
    h[:m] = s
    h[m:] = 1.0 - s[::-1]
    return h

def h_to_z(h, n):
    m = n // 2
    u = h[:m].copy()
    u = np.clip(u, 1e-8, 1.0 - 1e-8)
    z = np.log(u / (1.0 - u))
    return np.clip(z, -12.0, 12.0)

def smooth_obj_grad_z(z, n, beta):
    m = n // 2
    h = z_to_h(z, n)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    logsum = max_shifted + np.log(exp_shifted.sum())
    obj = cmax + logsum / beta
    w = exp_shifted
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w = w / w_sum
    G = compute_grad_matrix(h, dx)
    sg = sigmoid_grad(z)
    diff = G[:, :m] - G[:, m:n][::-1]
    grad_z = sg * (w @ diff)
    return obj, grad_z

def find_active_set(h, dx, frac=0.3, min_count=10):
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    threshold = cmax - max(0.015 * cmax, 2e-5)

    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

    rng = np.random.default_rng(12345)

        i = rng.integers(0, n)
        eps = 1e-7
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        vp = np.correlate(h_plus, 1.0 - h_plus, mode='full')[p] * dx
        vm = np.correlate(h_minus, 1.0 - h_minus, mode='full')[p] * dx
        fd = (vp - vm) / (2 * eps)

    return True

                act_f2 = find_active_set(h_full, 2.0 / len(h_full), frac=0.5, min_count=18)
                h_new2, sc_new2 = run_full_nlp(h_full, act_f2, max_iter=200, time_limit=20.0)
                if h_new2 is not None and sc_new2 < best_score - 1e-13:
                    update(h_new2, sc_new2)
                    h_full = h_new2

    # Phase 4: Resolution search
    if time_left() > 80:
        n_base = len(best_h)
        n_opts = sorted(set([
            max(30, n_base - 6),
            max(34, n_base - 4),
            max(38, n_base - 2),
            n_base,
            n_base + 2,
            n_base + 4,
            n_base + 6,
        ]))
        for n_opt in n_opts:
            if check_deadline():
                break
            if n_opt == n_base or n_opt < 20:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-13:
                update(h_int, sc_int)
            # Optimize at new resolution
            if n_opt % 2 == 0 and time_left() > 20:
                m_opt = n_opt // 2
                h_sym_opt = symmetrize(h_int)
                z_opt = h_to_z(h_sym_opt, n_opt)
                for beta in [200, 1000, 5000]:
                    if check_deadline():
                        break
                    def bfgs_obj_opt(zv):
                        return smooth_obj_grad_z(zv, n_opt, beta)[0]
                    def bfgs_grad_opt(zv):
                        return smooth_obj_grad_z(zv, n_opt, beta)[1]
                    try:
                        res = minimize(bfgs_obj_opt, z_opt, jac=bfgs_grad_opt,
                                       method='L-BFGS-B',
                                       bounds=[(-12.0, 12.0)] * m_opt,
                                       options={'maxiter': 300, 'ftol': 1e-16,
                                                'gtol': 1e-12, 'maxfun': 1000})
                        if res.x is not None and np.all(np.isfinite(res.x)):
                            h_new = z_to_h(res.x, n_opt)
                            sc_new = official_score(h_new)
                            if np.isfinite(sc_new):
                                z_opt = res.x.copy()
                                if sc_new < best_score - 1e-13:
                                    update(h_new, sc_new)
                    except Exception:
                        pass
            # Active set at new resolution
            if time_left() > 15:
                dx_opt = 2.0 / n_opt
                act_opt = find_active_set(h_int, dx_opt, frac=0.3, min_count=8)
                h_new, sc_new = run_full_nlp(h_int, act_opt, max_iter=150, time_limit=15.0)
                if h_new is not None and sc_new < best_score - 1e-13:
                    update(h_new, sc_new)

    # Phase 5: Random restarts in z-space
    if time_left() > 80 and len(best_h) % 2 == 0:
        n_r = len(best_h)
        m_r = n_r // 2
        h_sym_r = symmetrize(best_h)
        z_base = h_to_z(h_sym_r, n_r)

        n_restarts = max(4, min(20, int(time_left() / 20)))
        sigmas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.15, 0.25, 0.4, 0.6]
        for i in range(n_restarts):
            if check_deadline():
                break
            sigma = sigmas[i % len(sigmas)]
            z_pert = np.clip(z_base + rng.normal(0, sigma, size=m_r), -12.0, 12.0)
            h_pert = z_to_h(z_pert, n_r)
            sc_pert = official_score(h_pert)
            if sc_pert < best_score - 1e-13:
                update(h_pert, sc_pert)
            # Local optimize
            if time_left() > 10:
                for beta in [500, 3000]:
                    if check_deadline():
                        break
                    def bfgs_obj_r(zv):
                        return smooth_obj_grad_z(zv, n_r, beta)[0]
                    def bfgs_grad_r(zv):
                        return smooth_obj_grad_z(zv, n_r, beta)[1]
                    try:
                        res = minimize(bfgs_obj_r, z_pert, jac=bfgs_grad_r,
                                       method='L-BFGS-B',
                                       bounds=[(-12.0, 12.0)] * m_r,
                                       options={'maxiter': 200, 'ftol': 1e-15,
                                                'gtol': 1e-11, 'maxfun': 800})
                        if res.x is not None and np.all(np.isfinite(res.x)):
                            h_new = z_to_h(res.x, n_r)
                            sc_new = official_score(h_new)
                            if np.isfinite(sc_new):
                                z_pert = res.x.copy()
                                if sc_new < best_score - 1e-13:
                                    update(h_new, sc_new)
                    except Exception:
                        pass

    # Phase 6: High-beta polish on best solution
    if time_left() > 40:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            m_fin = n_fin // 2
            h_sym_fin = symmetrize(best_h)
            z_fin = h_to_z(h_sym_fin, n_fin)
            for beta in [5000, 15000, 40000]:
                if check_deadline():
                    break
                tl = min(time_left() * 0.2, 20.0)
                if tl < 2:
                    break
                def bfgs_obj_fin(zv):
                    return smooth_obj_grad_z(zv, n_fin, beta)[0]
                def bfgs_grad_fin(zv):
                    return smooth_obj_grad_z(zv, n_fin, beta)[1]
                try:
                    res = minimize(bfgs_obj_fin, z_fin, jac=bfgs_grad_fin,
                                   method='L-BFGS-B',
                                   bounds=[(-12.0, 12.0)] * m_fin,
                                   options={'maxiter': 400, 'ftol': 1e-16,
                                            'gtol': 1e-13, 'maxfun': 1500})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_new = z_to_h(res.x, n_fin)
                        sc_new = official_score(h_new)
                        if np.isfinite(sc_new):
                            z_fin = res.x.copy()
                            if sc_new < best_score - 1e-13:
                                update(h_new, sc_new)
                except Exception:
                    pass
        # Final active-set refinement
        if time_left() > 15:
            dx_fin = 2.0 / len(best_h)
            act_fin = find_active_set(best_h, dx_fin, frac=0.4, min_count=15)
            h_new, sc_new = run_full_nlp(best_h, act_fin, max_iter=400,
                                         time_limit=min(time_left() * 0.3, 25.0))
            if h_new is not None and sc_new < best_score - 1e-13:
                update(h_new, sc_new)

    # Phase 7: One more symmetrization + optimize cycle
    if time_left() > 30:
        h_sym2 = symmetrize(best_h)
        sc_sym2 = official_score(h_sym2)
        if sc_sym2 < best_score - 1e-13:
            update(h_sym2, sc_sym2)
        if len(best_h) % 2 == 0 and time_left() > 15:
            n_s2 = len(best_h)
            m_s2 = n_s2 // 2
            h_s2 = symmetrize(best_h)
            z_s2 = h_to_z(h_s2, n_s2)
            for beta in [3000, 10000]:
                if check_deadline():
                    break
                def bfgs_obj_s2(zv):
                    return smooth_obj_grad_z(zv, n_s2, beta)[0]
                def bfgs_grad_s2(zv):
                    return smooth_obj_grad_z(zv, n_s2, beta)[1]
                try:
                    res = minimize(bfgs_obj_s2, z_s2, jac=bfgs_grad_s2,
                                   method='L-BFGS-B',
                                   bounds=[(-12.0, 12.0)] * m_s2,
                                   options={'maxiter': 300, 'ftol': 1e-16,
                                            'gtol': 1e-12, 'maxfun': 1000})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_new = z_to_h(res.x, n_s2)
                        sc_new = official_score(h_new)
                        if np.isfinite(sc_new) and sc_new < best_score - 1e-13:
                            update(h_new, sc_new)
                except Exception:
                    pass

    # Final validation and return
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Ensure feasibility
    final_h = np.clip(final_h, 0.0, 1.0)
    s = np.sum(final_h)
    target = final_n / 2.0
    if abs(s - target) > 1e-10:
        final_h = make_feasible(final_h)
        final_score = official_score(final_h)

    assert np.all(final_h >= -1e-15), "Negative values in solution"
    assert np.all(final_h <= 1.0 + 1e-15), "Values exceed 1"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-8, "Sum constraint violated"

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
