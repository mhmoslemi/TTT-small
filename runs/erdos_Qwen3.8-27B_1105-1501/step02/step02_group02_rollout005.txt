import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_smooth(h, p=4.0):
    """Fast gradient of p-norm smoothed objective using vectorized ops."""
    n = len(h)
    dx = 2.0 / n
    F = _corr_vec(h)  # length 2n-1
    F_safe = np.maximum(F, 1e-30)
    w = F_safe ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones_like(F) / len(F)
    else:
        w = w / wsum
    
    W_pos = w[n-1:2*n-1]  # weights for lags 0 to n-1, length n
    
    first_term = np.correlate(W_pos, 1.0 - h, mode='full')[n-1:2*n-1]
    second_term = np.convolve(h, W_pos, mode='full')[0:n]
    
    return dx * (first_term - second_term)

def _grad_lse(h, beta=100.0):
    """Fast gradient of LSE smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    F = _corr_vec(h)
    mc = F.max()
    sh = np.clip(beta * (F - mc), -500, 0)
    w = np.exp(sh)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones_like(F) / len(F)
    else:
        w = w / wsum
    
    W_pos = w[n-1:2*n-1]
    
    first_term = np.correlate(W_pos, 1.0 - h, mode='full')[n-1:2*n-1]
    second_term = np.convolve(h, W_pos, mode='full')[0:n]
    
    return dx * (first_term - second_term)
