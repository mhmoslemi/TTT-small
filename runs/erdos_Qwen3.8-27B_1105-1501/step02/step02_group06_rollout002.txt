    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_int = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_int)

def logsumexp_obj_and_grad(h, beta):
    """Returns (objective, gradient) for log-sum-exp approximation."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = cmax + np.log(sum_exp) / beta
    weights = exp_shifted / sum_exp
    g = np.zeros(n)
    for p in range(2 * n - 1):
        if weights[p] < 1e-10:
            continue
        s = p - (n - 1)
        g += weights[p] * grad_shift(h, s)
    g *= dx
    return obj, g

def slsqp_lse(h0, beta, maxiter=100):
    n = len(h0)
    def fun(h):
        v, _ = logsumexp_obj_and_grad(h, beta)
        return v
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            return h_new, official_score(h_new)
    except Exception:
        pass
    return None, None

def qp_solve(h0, k_active, reg):
    n = len(h0)
    vals = raw_vals(h0)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = dx * grad_shift(h0, s)
        cons.append(vals[p] + g @ (h_var - h0) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0)), cons)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)

def qp_optimize(h0, max_iter=30, k_active=5, reg_start=3.0, reg_end=0.002, deadline=None):
    h = h0.copy()
    best = official_score(h)
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        reg = reg_start * (reg_end / reg_start) ** (it / max(1, max_iter - 1))
        h_new = qp_solve(h, k_active, reg)
        if h_new is None:
            break
        sc = official_score(h_new)
        if sc < best - 1e-15:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.8, 0.5, 0.3, 0.15]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-15:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 8:
                    break
    return h, best

def subgrad_optimize(h0, steps=100, beta=50.0, deadline=None):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    for t in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        lr = 0.005 * (1.0 - t / max(1, steps)) + 0.0005
        _, g = logsumexp_obj_and_grad(h, beta)
        h_try = make_feasible(h - lr * g)
        sc = official_score(h_try)
        if sc < best:
            h = h_try
            best = sc
    return h, best
