    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project onto {x : 0 <= x <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        #   So term is h[i-d]*(1-h[i]) if 0 <= i-d < n and 0 <= i < n (the latter is always true)
        
        slope_i = 0.0
        intercept_i = 0.0
        # j = i contributes h[i] * (1-h[i+d]) if valid
        if 0 <= i + d < n:
            slope_i += (1.0 - h[i+d])
            intercept_i += 0  # the term is h[i]*(1-h[i+d]), slope is (1-h[i+d]), intercept is 0
        # j = i-d contributes h[i-d] * (1-h[i]) if valid
        if 0 <= i - d < n:
            # h[i-d] * (1-h[i]) = h[i-d] - h[i-d]*h[i]
            # slope w.r.t. h[i] is -h[i-d], intercept is h[i-d]
            slope_i += -h[i-d]
            intercept_i += h[i-d]
        
        # slope for this lag = slope_i
        slopes[r] = slope_i + lam  # add Lagrange term
        # For intercept, we need corr_r - h[i]*slope_i (without lambda)
        # Actually, the total contribution to the objective from h[i] at lag r is:
        # (2/n) * (slope_i * h[i] + (corr_r - slope_i * h[i]_current))
        # But this is getting complicated. Let me just compute the full corr_r and extract the linear part.
        
        # Simpler: compute corr_r at current h, then the linear part in h[i] is slope_i * (h[i] - h[i])
        # = 0. So corr_r = base_r + slope_i * h[i] where base_r = corr_r_current - slope_i * h[i]_current
        # But we don't have corr_r_current precomputed here.
        
        # Let me restructure: precompute corr, then for each r, the linear function is:
        # f_r(t) = corr_r_current + slope_i * (t - h[i])
        # = (corr_r_current - slope_i * h[i]) + slope_i * t
        # slope = slope_i, intercept = corr_r_current - slope_i * h[i]
        
        slopes[r] = slope_i  # slope of corr_r w.r.t. h[i]
        # We'll compute intercepts separately after we have corr
    
    # Compute current correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Now for each r: f_r(t) = corr[r] + slopes[r] * (t - h[i])
    # = (corr[r] - slopes[r] * h[i]) + slopes[r] * t
    # Adding lambda: f_r(t) + lam * t = (corr[r] - slopes[r]*h[i]) + (slopes[r] + lam) * t
    
    # We want to minimize max_r f_r(t) + lam*t over t in [0,1]
    # = max_r (a_r * t + b_r) where a_r = slopes[r] + lam, b_r = corr[r] - slopes[r]*h[i]
    
    a = slopes + lam
    b = corr - slopes * h[i]
    
    # Minimize max of lines a_r*t + b_r over t in [0,1]
    # The minimum is at a breakpoint or at 0 or 1
    # Breakpoints: t where a_i*t + b_i = a_j*t + b_j, i.e., t = (b_j - b_i) / (a_i - a_j)
    
    best_t = 0.0
    best_val = np.max(a * 0.0 + b)  # value at t=0
    
    for t_candidate in [0.0, 1.0]:
        val = np.max(a * t_candidate + b)
        if val < best_val:
            best_val = val
            best_t = t_candidate
    
    # Find breakpoints in (0, 1)
    # This is O(n²) in the number of lags, which is 2n-1
    # For n=64, that's about 127 lines, so ~8000 breakpoints to check
    # That's too slow per coordinate. Let me use a smarter approach.
    
    # Actually, the lower envelope of lines can be found in O(m log m) where m is the number of lines
    # But we want the upper envelope (max), and we want to minimize it.
    # The minimum of the upper envelope is at a point where two lines cross (or at a boundary).
    
    # For efficiency, let's just sample and use a few iterations:
    # Start with t=0.5, find the active line, find the crossing point, etc.
    
    # Simpler approach: ternary search or golden section on [0,1]
    # The function max_r (a_r*t + b_r) is convex in t, so we can use golden section.
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    t0, t1 = 0.0, 1.0
    c1 = t1 - gr * (t1 - t0)
    c2 = t0 + gr * (t1 - t0)
    f1 = np.max(a * c1 + b)
    f2 = np.max(a * c2 + b)
    for _ in range(50):
        if f1 < f2:
            t1 = c2
            c2 = c1
            f2 = f1
            c1 = t1 - gr * (t1 - t0)
            f1 = np.max(a * c1 + b)
        else:
            t0 = c1
            c1 = c2
            f1 = f2
            c2 = t0 + gr * (t1 - t0)
            f2 = np.max(a * c2 + b)
    
    t_opt = (t0 + t1) / 2.0
    f_opt = np.max(a * t_opt + b)
    if f_opt < best_val:
        best_val = f_opt
        best_t = t_opt
    
    return best_t

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _softmax_weights(scores, beta):
    max_s = float(np.max(scores))
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    s = float(np.sum(w))
    if s < 1e-30:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= s
    return w

def _grad_soft(h, beta):
    """Compute soft-max gradient of C5."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * (2.0 / n)
    w = _softmax_weights(scores, beta)
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        w_r = w[r]
        if w_r < 1e-8:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w_r * (1.0 - h[d:])
                grad[d:] += -w_r * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w_r * (1.0 - h[g:])
                grad[g:] += -w_r * h[:m]
    return grad * (2.0 / n)

def _nesterov_descent(h, deadline, max_iters=100000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    v = np.zeros(n)
    mu = 0.85  # momentum
    alpha = 0.05  # initial step size
    beta = 20.0  # initial soft-max parameter
    stall = 0
    
    for it in range(max_iters):
        if it > 0 and it % 200 == 0 and time.time() > deadline:
            break
        if time.time() > deadline:
            break
        
        grad = _grad_soft(h, beta)
        
        # Nesterov step
        v = mu * v - alpha * grad
        h_new = project(h + v, target)
        
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            # Occasionally increase step size
            if stall == 0 and it % 500 == 0:
                alpha = min(alpha * 1.3, 0.2)
        else:
            stall += 1
            # Reduce step size
            alpha *= 0.85
            # If stalled a lot, add noise
            if stall > 30:
                noise = np.random.normal(0, 0.01, n)
                h_new = _project(h + noise, target)
                c5_new = _exact_c5(h_new)
                v = np.zeros(n)
                stall = 0
                alpha = 0.02
        
        h = h_new
        
        # Anneal beta
        if it % 100 == 0:
            beta = min(beta * 1.02, 500.0)
        
        # Reset momentum periodically
        if it % 500 == 0:
            mu = max(0.5, mu * 0.98)
    
    return best_h, best_c5

def _binary_search(h, deadline, max_passes=5000):
    n = len(h)
    target = n / 2.0
    
    # Convert to binary
    h_bin = (h >= 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        # Remove excess ones (remove the smallest fractional values that became 1)
        idx = np.argsort(h)[::-1]  # from largest to smallest
        # Actually, we want to keep the n/2 largest values as 1
        thresh_idx = np.argsort(h)[::-1][:n//2]
        h_bin = np.zeros(n)
        h_bin[thresh_idx] = 1.0
    elif n_ones < n // 2:
        thresh_idx = np.argsort(h)[::-1][:n//2]
        h_bin = np.zeros(n)
        h_bin[thresh_idx] = 1.0
    
    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        
        improved = False
        best_swap_gain = 0
        best_i = -1
        best_j = -1
        
        # Find best single swap
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        
        for i in ones:
            for j in zeros:
                # Try swapping i and j
                h_try = h_bin.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                c5_try = _exact_c5(h_try)
                gain = best_c5 - c5_try
                if gain > best_swap_gain + 1e-12:
                    best_swap_gain = gain
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            best_c5 -= best_swap_gain
            best_h = h_bin.copy()
            improved = True
        
        if not improved:
            # Try double swaps
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            found = False
            for i in ones[:min(len(ones), 40)]:
                for j in zeros[:min(len(zeros), 40)]:
                    for i2 in ones:
                        if i2 == i:
                            continue
                        for j2 in zeros:
                            if j2 == j or j2 == i:
                                continue
                            h_try = h_bin.copy()
                            h_try[i] = 0.0
                            h_try[j] = 1.0
                            h_try[i2] = 0.0
                            h_try[j2] = 1.0
                            c5_try = _exact_c5(h_try)
                            if c5_try < best_c5 - 1e-12:
                                best_c5 = c5_try
                                h_bin = h_try
                                best_h = h_bin.copy()
                                found = True
                                improved = True
                                break
                        if found:
                            break
                    if found:
                        break
                if found:

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _softmax_weights(scores, beta):
    max_s = float(np.max(scores))
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    s = float(np.sum(w))
    if s < 1e-30:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= s
    return w

def _grad_soft(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * (2.0 / n)
    w = _softmax_weights(scores, beta)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        w_r = w[r]
        if w_r < 1e-8:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w_r * (1.0 - h[d:])
                grad[d:] += -w_r * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w_r * (1.0 - h[g:])
                grad[g:] += -w_r * h[:m]
    return grad * (2.0 / n)

def _nesterov_descent(h, deadline, max_iters=200000, alpha_init=0.03, beta_init=15.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    mu = 0.82
    alpha = alpha_init
    beta = beta_init
    stall = 0
    rng_local = np.random.default_rng(12345)

    for it in range(max_iters):
        if it > 0 and it % 500 == 0 and time.time() > deadline:
            break
        if time.time() > deadline:
            break

        grad = _grad_soft(h, beta)
        v = mu * v - alpha * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            if it % 1000 == 0:
                alpha = min(alpha * 1.2, 0.15)
        else:
            stall += 1
            alpha *= 0.88
            if stall > 50:
                noise = rng_local.normal(0, 0.02, n)
                h_new = _project(h + noise, target)
                v = np.zeros(n)
                stall = 0
                alpha = 0.015

        h = h_new

        if it % 200 == 0:
            beta = min(beta * 1.03, 800.0)
        if it % 1000 == 0:
            mu = max(0.5, mu * 0.97)

    return best_h, best_c5

def _binary_local_search(h, deadline, max_passes=3000):
    n = len(h)
    h_bin = np.zeros(n)
    idx_sorted = np.argsort(h)[::-1]
    h_bin[idx_sorted[:n // 2]] = 1.0
    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        best_gain = 0.0
        best_i = -1
        best_j = -1
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        for i in ones:
            for j in zeros:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c5_try = _exact_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                gain = best_c5 - c5_try
                if gain > best_gain + 1e-13:
                    best_gain = gain
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            best_c5 -= best_gain
            best_h = h_bin.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.05
    stall = 0

    for _it in range(3000):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue

        alpha = 1.0
        improved = False
        for _ in range(20):
            h_try = _project(h + alpha * p, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                improved = True
                break
            alpha *= 0.5
        if improved:
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _coordinate_refinement(h, deadline, max_sweeps=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        old_c5 = _exact_c5(h)
        for i in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            scores = corr * (2.0 / n)
            slopes = np.zeros(2 * n - 1)
            for d in range(-(n - 1), n):
                r = n - 1 + d
                sl = 0.0
                if 0 <= i + d < n:
                    sl += (1.0 - h[i + d])
                if 0 <= i - d < n:
                    sl += -h[i - d]
                slopes[r] = sl
            a = slopes
            b = scores - slopes * h[i]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t0, t1 = 0.0, 1.0
            c1 = t1 - gr * (t1 - t0)
            c2 = t0 + gr * (t1 - t0)
            f1 = float(np.max(a * c1 + b))
            f2 = float(np.max(a * c2 + b))
            for _ in range(40):
                if f1 < f2:
                    t1 = c2
                    c2 = c1
                    f2 = f1
                    c1 = t1 - gr * (t1 - t0)
                    f1 = float(np.max(a * c1 + b))
                else:
                    t0 = c1
                    c1 = c2
                    f1 = f2
                    c2 = t0 + gr * (t1 - t0)
                    f2 = float(np.max(a * c2 + b))
            t_opt = (t0 + t1) / 2.0
            f_opt = float(np.max(a * t_opt + b))
            f_0 = float(np.max(b))
            f_1 = float(np.max(a + b))
            best_t = t_opt
            best_f = f_opt
            if f_0 < best_f:
                best_f = f_0
                best_t = 0.0
            if f_1 < best_f:
                best_f = f_1
                best_t = 1.0
            h[i] = best_t
        h = _project(h, target)
        c5_new = _exact_c5(h)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h.copy()
        if abs(_exact_c5(h) - old_c5) < 1e-14:
            break
    return best_h, best_c5
