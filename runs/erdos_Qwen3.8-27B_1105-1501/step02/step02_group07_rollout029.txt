def slsqp_polish(h, n_constraints=15, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    
    # Select active constraints
    top_idx = np.argsort(V)[::-1][:n_constraints]
    
    # Define objective: minimize t
    def objective(vars_):
        h_v = vars_[:n]
        t = vars_[n]
        return t
    
    # Constraints
    def cons_j(j):
        def c(vars_):
            h_v = vars_[:n]
            t = vars_[n]
            # Compute V for lag j
            s = j - (n - 1)
            if s >= 0:
                v = np.sum(h_v[0:n-s] * (1 - h_v[s:n])) * dx
            else:
                t_ = -s
                v = np.sum(h_v[t_:n] * (1 - h_v[0:n-t_])) * dx
            return t - v  # ≥ 0 means constraint satisfied
        return c
    
    # Equality: sum(h) = n/2
    def eq_con(vars_):
        return np.sum(vars_[:n]) - n/2.0
    
    # Build constraint list
    constraints = []
    for j in top_idx:
        constraints.append({'type': 'ineq', 'fun': cons_j(j)})
    constraints.append({'type': 'eq', 'fun': eq_con})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    x0 = np.concatenate([h, [V.max() + 0.001]])
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if res.success:
        h_new = make_feasible(res.x[:n])
        return h_new, official_score(h_new)
    return None, None

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - np.sum(z)
    if abs(residual) < 1e-15:
        return z
    if residual > 0:
        order = np.argsort(1.0 - z)
    else:
        order = np.argsort(z)
    for idx in order:
        if residual > 0:
            a = min(residual, 1.0 - z[idx])
            z[idx] += a
            residual -= a
        else:
            a = min(-residual, z[idx])
            z[idx] -= a
            residual += a
        if abs(residual) < 1e-15:
            break
    return z

def compute_values_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    L = n - abs(s)
    if L <= 0:
        return grad
    if s >= 0:
        grad[0:L] += dx * (1.0 - h[s:s+L])
        grad[s:n] -= dx * h[0:L]
    else:
        t = -s
        L2 = n - t
        grad[t:n] += dx * (1.0 - h[0:L2])
        grad[0:L2] -= dx * h[t:n]
    return grad
