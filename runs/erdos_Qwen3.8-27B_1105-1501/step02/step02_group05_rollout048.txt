def solve_active_lp(h, V, active_p, rho):
    """
    Solve the active-lag LP.
    h: current feasible point
    V: current correlation values (length 2n-1)
    active_p: list of active correlation indices
    rho: trust region radius
    Returns: direction u (length n) or None if infeasible
    """
    n = len(h)
    dx = 2.0 / n
    K = len(active_p)
    
    # Build gradients for active lags
    G = np.zeros((K, n))
    for i, p in enumerate(active_p):
        s = p - (n-1)
        if s >= 0:
            if s == 0:
                G[i] = 1.0 - 2.0*h
            else:
                G[i, s:] += 1.0 - h[:n-s]
                G[i, :n-s] -= h[s:]
        else:
            t = -s
            G[i, :n-t] += 1.0 - h[t:]
            G[i, t:] -= h[:n-t]
    G *= dx
    
    c_vals = V[active_p]  # values of active lags
    
    # Variables: u_0, ..., u_{n-1}, t
    # Total: n+1 variables
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    # Inequality constraints: G[i] @ u - t <= -c_vals[i]
    # A_ub shape: (K, n+1)
    A_ub = np.zeros((K, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c_vals
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: u_i in [max(-h_i, -rho), min(1-h_i, rho)], t in [-inf, V.max()]
    t_max = float(V.max())
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, t_max))
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success and result.x is not None:
        u = result.x[:n]
        if np.all(np.isfinite(u)) and np.max(np.abs(u)) > 1e-12:
            return u
    return None

import numpy as np
from scipy.optimize import linprog
import time
import math

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def grad_row(h, p, n):
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    else:
        t = -s
        grad[:n - t] += 1.0 - h[t:]
        grad[t:] -= h[:n - t]
    return grad * dx

def solve_active_lp(h, V, active_p, rho):
    n = len(h)
    K = len(active_p)
    G = np.zeros((K, n))
    for i, p in enumerate(active_p):
        G[i] = grad_row(h, p, n)
    c_vals = V[active_p]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c_vals
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    t_max = float(V.max())
    bounds = []
    for i in range(n):
        lo_i = max(-h[i], -rho)
        hi_i = min(1.0 - h[i], rho)
        bounds.append((lo_i, hi_i))
    bounds.append((None, t_max))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            u = res.x[:n]
            if np.all(np.isfinite(u)) and np.max(np.abs(u)) > 1e-12:
                return u
    except Exception:
        pass
    return None

def lp_refine(h0, max_time, is_dead):
    h = make_feasible(h0)
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    cur_sc = best_sc
    stall = 0
    rho_idx = 0
    rho_list = [0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.018, 0.01]
    eta = 0.008
    K_top = 32
    t_start = time.time()
    while not is_dead() and (time.time() - t_start) < max_time:
        V_raw = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        V = V_raw * dx
        C = float(V.max())
        if C < cur_sc - 1e-14:
            cur_sc = C
        # Select active set
        sorted_idx = np.argsort(V)[::-1]
        active = sorted_idx[:K_top].tolist()
        # Add near-max
        thresh = C - eta
        for p in range(len(V)):
            if V[p] >= thresh and p not in active:
                active.append(p)
        if len(active) > 60:
            active = active[:60]
        u = None
        for ri in range(rho_idx, len(rho_list)):
            rho = rho_list[ri]
            u = solve_active_lp(h, V, active, rho)
            if u is not None:
                rho_idx = ri
                break
        if u is None:
            # Try with smaller active set
            active_small = sorted_idx[:min(12, K_top)].tolist()
            u = solve_active_lp(h, V, active_small, rho_list[min(rho_idx, 2)])
        if u is None:
            eta *= 0.8
            if eta < 1e-5:
                rho_idx = min(rho_idx + 1, len(rho_list) - 1)
            stall += 1
            if stall > 8:
                break
            continue
        # Line search
        u_norm = np.linalg.norm(u)
        if u_norm < 1e-12:
            break
        u = u / u_norm * min(1.0, rho_list[rho_idx])
        found = False
        for frac in [1.0, 0.7, 0.5, 0.35, 0.2, 0.12, 0.06, 0.03]:
            trial = make_feasible(h + frac * u)
            sc = official_score(trial)
            if sc < cur_sc - 1e-14:
                h = trial
                cur_sc = sc
                if cur_sc < best_sc - 1e-14:
                    best = h.copy()
                    best_sc = cur_sc
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            eta *= 0.85
            if stall > 10:
                break
    return best, best_sc

def binary_hill_climb(n, max_iters, rng, is_dead, t_start, max_time):
    target = n // 2
    # Start from a pattern
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h_wave = 0.5 + 0.4 * np.sin(2 * np.pi * 2.0 * x) + 0.15 * np.cos(2 * np.pi * 3.0 * x)
    h_wave = make_feasible(h_wave)
    b = (h_wave > 0.5).astype(float)
    # Balance
    n_ones = int(b.sum())
    if n_ones > target:
        excess = n_ones - target
        one_idx = np.where(b == 1)[0]
        # Remove ones with lowest h_wave values
        order = np.argsort(h_wave[one_idx])[:excess]
        b[one_idx[order]] = 0
    elif n_ones < target:
        deficit = target - n_ones
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h_wave[zero_idx])[:deficit]
        b[zero_idx[order]] = 1
    assert int(b.sum()) == target
    best_b = b.copy()
    best_sc = official_score(best_b)
    cur_sc = best_sc
    pos1 = np.where(b == 1)[0]
    pos0 = np.where(b == 0)[0]
    for it in range(max_iters):
        if is_dead() or (time.time() - t_start) > max_time:
            break
        i1 = pos1[rng.integers(len(pos1))]
        i0 = pos0[rng.integers(len(pos0))]
        if i1 == i0:
            continue
        b[i1] = 0
        b[i0] = 1
        sc = official_score(b)
        if sc < cur_sc - 1e-14:
            cur_sc = sc
            if sc < best_sc - 1e-14:
                best_b = b.copy()
                best_sc = sc
            pos1 = np.where(b == 1)[0]
            pos0 = np.where(b == 0)[0]
        else:
            b[i1] = 1
            b[i0] = 0
    return best_b, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0.0, 1.0)
            best[b] = np.clip(best[b] - best_t, 0.0, 1.0)
            best_sc = official_score(best)
    return best, best_sc
