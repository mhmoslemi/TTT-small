    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_raw(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_shift(h, s):
    """Gradient of raw correlation for shift s (unnormalized, no dx)."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]
    else:
        t = -s
        for i in range(n - t):
            g[i + t] += (1.0 - h[i])
            g[i] -= h[i + t]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -4.0, 4.0)
    lo, hi = -4.0, 4.0
    for _ in range(100):

            lo = mid
        else:

    h = np.clip(z + theta, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:

                h[idx] -= sub
                residual += sub

def line_search_1d(h, direction, alpha_lo=0.0, alpha_hi=1.0, tol=1e-12):

import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_raw(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]
    else:
        t = -s
        for i in range(n - t):
            g[i + t] += (1.0 - h[i])
            g[i] -= h[i + t]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -4.0, 4.0)
    lo, hi = -4.0, 4.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = np.clip(z + theta, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            caps = 1.0 - h
            order = np.argsort(-caps)
            for idx in order:
                add = min(residual, caps[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
    return h

    scores = np.array([official_score((1 - a) * h1 + a * h2) for a in alphas])
    bi = int(np.argmin(scores))

    if bi == len(alphas) - 1:
        return 1.0, scores[-1]
    lo_a, hi_a = alphas[bi - 1], alphas[bi + 1]

def subgradient_step(h, raw_vals, K, beta, trust_radius):
    n = len(h)
    P = len(raw_vals)
    order = np.argsort(raw_vals)[::-1]
    selected = order[:min(K, P)]
    sel_vals = raw_vals[selected]
    max_sel = sel_vals.max()
    weights = np.exp(beta * (sel_vals - max_sel))
    wsum = weights.sum()
    if wsum < 1e-30:
        return None
    weights /= wsum
    grad = np.zeros(n)
    for idx in range(len(selected)):
        s = int(selected[idx]) - (n - 1)
        grad += weights[idx] * grad_shift(h, s)
    grad -= np.mean(grad)
    dnorm = np.linalg.norm(grad)
    if dnorm < 1e-15:
        return None
    direction = -grad / dnorm * trust_radius
    h_new = make_feasible(h + direction)
    if np.max(np.abs(h_new - h)) < 1e-14:
        return None
    alpha, score = line_search_seg(h, h_new)
    if alpha < 1e-6:
        return None
    h_result = make_feasible((1 - alpha) * h + alpha * h_new)
    sc = official_score(h_result)
    return h_result, sc

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        for p in selected:
            s = int(p) - (n - 1)
            g = dx * grad_shift(h, s)
            constraints.append(raw[p] + g @ (h_var - h) <= t_var)
        obj = cp.Minimize(t_var + reg * cp.sum_squares(h_var - h))
        prob = cp.Problem(obj, constraints)
        h_new = None
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    h_new = h_var.value
                    break
            except Exception:
                continue
        if h_new is None or not np.all(np.isfinite(h_new)):
            stall += 1
            if stall >= 5:
                break
            continue
        h_new = make_feasible(h_new)
        sc = official_score(h_new)
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            alpha, sc_ls = line_search_seg(h, h_new, n_coarse=20)
            if sc_ls < best - 1e-14:
                h = make_feasible((1 - alpha) * h + alpha * h_new)
                best = sc_ls
                stall = 0
            else:
                stall += 1
                if stall >= 5:
                    break
    return h, best

def smooth_max_obj(h, beta_val):
    raw = compute_raw(h)
    cmax = raw.max()
    shifted = beta_val * (raw - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta_val

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))

def try_slsqp(h0, beta_val, maxiter=120):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_max_obj, h0, args=(beta_val,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    best_h = None
    best_score = float('inf')

    def try_update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-13:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x / 2.0))
        best_score = official_score(best_h)

    try_update(best_h, best_score)

    # Phase 1: Two-coordinate descent (exploits bilinear structure)
    if remaining() > 80:
        h_opt, sc_opt = two_coord_descent(best_h, max_iter=100, n_pairs=8)
        try_update(h_opt, sc_opt)

    # Phase 2: Subgradient with line search
    if remaining() > 80:
        for K in [10, 20, 30]:
            if remaining() < 30:
                break
            h_opt, sc_opt = subgradient_optimize(best_h, max_iter=60, K=K,
                                                  beta=40, trust_start=0.25, trust_end=0.01)
            try_update(h_opt, sc_opt)

    # Phase 3: QP local
    if remaining() > 80:
        for K in [10, 20, 35]:
            if remaining() < 25:
                break
            h_opt, sc_opt = qp_local(best_h, K=K, reg=0.5, max_iter=20)
            try_update(h_opt, sc_opt)

    # Phase 4: Two-coordinate + QP alternating
    if remaining() > 100:
        for rnd in range(4):
            if remaining() < 30:
                break
            h_tc, sc_tc = two_coord_descent(best_h, max_iter=60, n_pairs=10)
            try_update(h_tc, sc_tc)
            h_qp, sc_qp = qp_local(best_h, K=15, reg=0.3, max_iter=15)
            try_update(h_qp, sc_qp)

    # Phase 5: Resolution escalation
    n_base = len(best_h)
    n_candidates = sorted(set([
        max(30, n_base - 10), n_base,
        n_base + 15, n_base + 30, n_base + 50, n_base + 70,
        min(200, n_base + 100)
    ]))
    for n_tgt in n_candidates:
        if remaining() < 35:
            break
        if n_tgt < 24 or n_tgt > 180:
            continue
        h_int = interpolate_to(best_h, n_tgt)
        sc_int = official_score(h_int)
        try_update(h_int, sc_int)
        if remaining() > 25:
            h_opt, sc_opt = two_coord_descent(h_int, max_iter=50, n_pairs=8)
            try_update(h_opt, sc_opt)
        if remaining() > 20:
            h_opt, sc_opt = qp_local(best_h, K=12, reg=0.3, max_iter=15)
            try_update(h_opt, sc_opt)

    # Phase 6: SLSQP smooth max at current best
    if remaining() > 40:
        for bv in [300, 1000, 5000]:
            if remaining() < 12:
                break
            h_new, sc = try_slsqp(best_h, bv, maxiter=100)
            if h_new is not None:
                try_update(h_new, sc)

    # Phase 7: Random perturbations with 2-coord refine
    if remaining() > 60:
        for i in range(25):
            if remaining() < 15:
                break
            sigma = 0.01 + 0.06 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = two_coord_descent(h_cand, max_iter=40, n_pairs=8)
            try_update(h_opt, sc_opt)
            if remaining() > 10:
                h_opt2, sc_opt2 = qp_local(h_cand, K=12, reg=0.5, max_iter=10)
                try_update(h_opt2, sc_opt2)

    # Phase 8: Intensive final polish
    if remaining() > 50:
        h_opt, sc_opt = two_coord_descent(best_h, max_iter=80, n_pairs=12)
        try_update(h_opt, sc_opt)
        if remaining() > 20:
            for K in [8, 12, 18]:
                if remaining() < 8:
                    break
                h_opt, sc_opt = qp_local(best_h, K=K, reg=0.1, max_iter=12)
                try_update(h_opt, sc_opt)
        if remaining() > 15:
            h_opt, sc_opt = subgradient_optimize(best_h, max_iter=40, K=20,
                                                  beta=80, trust_start=0.1, trust_end=0.005)
            try_update(h_opt, sc_opt)

    # Phase 9: Final SLSQP
    if remaining() > 15:
        h_new, sc = try_slsqp(best_h, 10000, maxiter=80)
        if h_new is not None:
            try_update(h_new, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = final_score + 1e-13
    return final_h, final_bound, len(final_h)
