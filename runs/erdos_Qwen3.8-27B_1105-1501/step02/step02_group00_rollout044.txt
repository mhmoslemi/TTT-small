```python
    """
    Compute gradient of (1/beta)*log(sum exp(beta * s_r))
    where s_r = scores[r] * scores_scale
    Returns gradient w.r.t. h.
    """
    n = len(h)
    b = beta * scores_scale
    s = scores_scale * _scores(h)
    mx = s.max()
    w = np.exp(b * (s - mx))
    w_sum = w.sum()

        active = list(range(len(s)))
    else:

        w_top = w[top_idx]
        w_top /= w_top.sum()
        active = top_idx

    # ∂score[r]/∂h[j] = (1-h[j-d]) if 0<=j-d<n, else 0
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thr = (lo + hi) / 2.0
    x = np.clip(v - thr, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) & (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgrad_max(h, scores):
    """Compute subgradient of max(scores) w.r.t. h at point h.
    Uses the active (maximum) components.
    """
    n = len(h)
    mx = scores.max()
    active = np.where(scores >= mx - 1e-10)[0]
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            grad[d:] += (1.0 - h[:n-d])
            grad[:n-d] += -h[d:]
        else:
            g = -d
            grad[:n-g] += (1.0 - h[g:])
            grad[g:] += -h[:n-g]
    grad /= len(active)
    return grad
```

Let me try a different tactic. What if I use **Frank-Wolfe (conditional gradient)**?

Frank-Wolfe for our problem:
- At each step, find the linearization: L(h) = f(h_t) + <∇f(h_t), h - h_t>
- Solve: min_{h in P} <∇f(h_t), h> where P = {h: sum h = n/2, 0 ≤ h ≤ 1}
- The minimizer of a linear function over P is a vertex: exactly n/2 coordinates at 1, rest at 0
- Take a step: h_{t+1} = (1-γ) h_t + γ h_vw

The inner problem is: minimize sum_j g_j * h_j subject to sum h_j = n/2, 0 ≤ h_j ≤ 1
where g = ∇f(h_t) (or a subgradient).

The solution is: set h_j = 1 for the n/2 smallest g_j, h_j = 0 for the rest.

This is very fast! And Frank-Wolfe has good convergence properties.

```python
def _frank_wolfe(h, deadline, max_iters=5000, gamma_mode='standard'):
    """Frank-Wolfe method for the Erdős overlap problem."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        cur_c5 = _exact_c5(h)
        scores = _scores(h)
        mx = scores.max()
        active = np.where(scores >= mx - 1e-10)[0]
        
        # Subgradient (average of active gradients)
        grad = np.zeros(n)
        for r in active:
            d = r - (n - 1)
            if d >= 0:
                grad[d:] += (1.0 - h[:n-d])
                grad[:n-d] += -h[d:]
            else:
                g = -d
                grad[:n-g] += (1.0 - h[g:])
                grad[g:] += -h[:n-g]
        grad /= len(active)
        
        # Find FW point: min <grad, h'> s.t. sum h' = target, 0 <= h' <= 1
        # Solution: set h'[j] = 1 for the n/2 smallest grad[j]
        sorted_idx = np.argsort(grad)
        vw = np.zeros(n)
        vw[sorted_idx[:int(target)]] = 1.0
        
        # Step size
        if gamma_mode == 'standard':
            gamma = 2.0 / (it + 2)
        else:  'decrease'
            # Backtracking to find optimal gamma
            best_gamma = 0.0
            best_f = cur_c5
            for g in np.concatenate([np.linspace(0, 1, 20), np.array([0.1, 0.2, 0.3, 0.5, 0.7, 0.9])]):
                h_t = (1 - g) * h + g * vw
                f_t = _exact_c5(h_t)
                if f_t < best_f:
                    best_f = f_t
                    best_gamma = g
            gamma = best_gamma
        
        h_new = (1 - gamma) * h + gamma * vw
        h_new = _project_sum(h_new, target)
        
        f_new = _exact_c5(h_new)
        if f_new < best_c5:
            best_c5 = f_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Stopping criterion
        gap = grad @ (vw - h)
        if gap > -1e-8:  # primal gap is small (note: gap should be <= 0 for minimization)
            # Actually gap = <grad, vw - h> <= 0 since vw minimizes <grad, h'>
            # If gap is close to 0, we're at a (approximate) minimum
            break
    
    return best_h, best_c5
```
