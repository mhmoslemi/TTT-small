import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_corr_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    npts = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    grad = np.zeros((npts, n))
    for p in range(npts):
        s = p - (n - 1)
        if s >= 0:
            grad[p, s:] += (1.0 - h[:n-s])
            grad[p, :n-s] += -h[s:]
        else:
            t = -s
            grad[p, t:] += (1.0 - h[:n-t])
            grad[p, :n-t] += -h[t:]
    return corr, grad

def solve_lp_direction(h, delta):
    n = len(h)
    corr, grad = compute_corr_and_grad(h)
    npts = len(corr)
    
    d_lo = np.maximum(-delta, -h)
    d_hi = np.minimum(delta, 1.0 - h)
    if np.any(d_lo > d_hi + 1e-15):
        return None, None
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((npts, n + 1))
    A_ub[:, :n] = grad
    A_ub[:, n] = -1.0
    b_ub = -corr
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((d_lo[j], d_hi[j]))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None, None
        return res.x[:n], res.x[n]
    except Exception:
        return None, None

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = np.array([0.0, t_lo, t_hi, (t_lo+t_hi)/2,
                                  t_lo+0.25*span, t_lo+0.75*span,
                                  t_lo+span/2.618, t_lo+2*span/2.618])
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            if t < t_lo - 1e-13 or t > t_hi + 1e-13:
                continue
            tmp[a] = best[a] + t
            tmp[b] = best[b] - t
            if tmp[a] < -1e-13 or tmp[a] > 1+1e-13 or tmp[b] < -1e-13 or tmp[b] > 1+1e-13:
                continue
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = official_score(best)
    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)
