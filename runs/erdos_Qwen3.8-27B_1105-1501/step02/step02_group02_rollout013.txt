    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-18:
            continue
        s = k - (n - 1)
        # First term: (1-x[m+s]) for m where 0<=m+s<n, i.e., -s<=m<n-s
        m_start1 = max(0, -s)
        m_end1 = min(n, n - s)  # exclusive
        # x[m+s] for m in [m_start1, m_end1): indices [m_start1+s, m_end1+s)
        g[m_start1:m_end1] += wk * (1.0 - x[m_start1+s:m_end1+s])
        
        # Second term: -x[m-s] for m where 0<=m-s<n, i.e., s<=m<n+s
        m_start2 = max(0, s)
        m_end2 = min(n, n + s)  # exclusive
        # x[m-s] for m in [m_start2, m_end2): indices [m_start2-s, m_end2-s)
        g[m_start2:m_end2] -= wk * x[m_start2-s:m_end2-s]
    return g * dx

import numpy as np
from scipy.optimize import minimize
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _weighted_corr_grad(x, w):
    n = len(x)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-18:
            continue
        s = k - (n - 1)
        m_start1 = max(0, -s)
        m_end1 = min(n, n - s)
        g[m_start1:m_end1] += wk * (1.0 - x[m_start1 + s:m_end1 + s])
        m_start2 = max(0, s)
        m_end2 = min(n, n + s)
        g[m_start2:m_end2] -= wk * x[m_start2 - s:m_end2 - s]
    return g * dx


def _pnorm_obj(x, p):
    cv = _corr_vec(x)
    F = np.maximum(cv, 1e-30)
    Sp = np.sum(F ** p)
    return Sp ** (1.0 / p)


def _pnorm_grad(x, p):
    cv = _corr_vec(x)
    F = np.maximum(cv, 1e-30)
    Sp = np.sum(F ** p)
    coeff = Sp ** (1.0 / p - 1.0)
    w = F ** (p - 1) * coeff
    return _weighted_corr_grad(x, w)


def _pnorm_slsqp(h0, p, maxiter=150, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def _efun(x):
        return x.sum() - target

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_pnorm_obj, h0, args=(p,), jac=_pnorm_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _direct_mm_obj(z, n):
    return z[-1]


def _direct_mm_obj_jac(z, n):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def _direct_mm_ieq(z, n):
    h = z[:-1]
    t = z[-1]
    cv = _corr_vec(h)
    return t - cv


def _direct_mm_eq(z, n):
    return z[:-1].sum() - n / 2.0


def _direct_mm_eq_jac(z, n):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g


def _direct_minmax(h0, maxiter=150):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = min(cv.max() + 0.003, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(_direct_mm_obj, z0, args=(n,),
                       jac=lambda z: _direct_mm_obj_jac(z, n),
                       method='SLSQP', bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': lambda z: _direct_mm_ieq(z, n)},
                           {'type': 'eq', 'fun': lambda z: _direct_mm_eq(z, n),
                            'jac': lambda z: _direct_mm_eq_jac(z, n)}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _binary_push(h, p):
    return _project(h ** p)


def _mirror(h):
    return _project(h[::-1].copy())


def _de_global(h0, n, pop_size, n_gen, rng, deadline, reserve=10.0):
    """Differential evolution for global search."""
    best = h0.copy()
    best_s = _score(best)
    pop = [best.copy()]
    # Initialize population
    transforms = [best.copy(), _mirror(best), _binary_push(best, 2.0)]
    for sigma in [0.1, 0.2, 0.4]:
        transforms.append(_project(best + sigma * rng.standard_normal(n)))
    # Binary-ish
    thresh = rng.uniform(0.3, 0.7)
    transforms.append(_project((best > thresh).astype(float)))
    for t in transforms:
        if len(pop) >= pop_size:
            break
        pop.append(t.copy())
    while len(pop) < pop_size:
        base = pop[rng.integers(0, len(pop))]
        pop.append(_project(base + 0.15 * rng.standard_normal(n)))

    pop_scores = [_score(p) for p in pop]
    best_idx = int(np.argmin(pop_scores))
    best = pop[best_idx].copy()
    best_s = pop_scores[best_idx]

    F = 0.7
    CR = 0.8
    for gen in range(n_gen):
        if time.time() > deadline - reserve:
            break
        best_idx = int(np.argmin(pop_scores))
        for i in range(pop_size):
            if time.time() > deadline - reserve:
                break
            r1, r2 = rng.integers(0, pop_size), rng.integers(0, pop_size)
            while r1 == r2 or r1 == i or r2 == i:
                r1, r2 = rng.integers(0, pop_size), rng.integers(0, pop_size)
            v = pop[best_idx] + F * (pop[r1] - pop[r2])
            mask = rng.random(n) < CR
            u = np.where(mask, v, pop[i])
            u = _project(u)
            su = _score(u)
            if su <= pop_scores[i]:
                pop[i] = u
                pop_scores[i] = su
                if su < best_s:
                    best_s = su
                    best = u.copy()
        # Elitism check
        if gen % 5 == 4:
            new_best_idx = int(np.argmin(pop_scores))
            if pop_scores[new_best_idx] < best_s - 1e-12:
                best_s = pop_scores[new_best_idx]
                best = pop[new_best_idx].copy()
    return best, best_s


def _simulated_anneal(h0, n, n_steps, T_init, T_final, rng, deadline, reserve=10.0):
    """Simulated annealing with sum-preserving moves."""
    cur = h0.copy()
    cur_s = _score(cur)
    best = cur.copy()
    best_s = cur_s

    T = T_init
    cool = (T_final / T_init) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.time() > deadline - reserve:
            break
        # Choose two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            j = (j + 1) % n
        # Choose delta that preserves sum and stays in [0,1]
        if cur[i] >= cur[j]:
            max_d = min(cur[i], 1.0 - cur[j])
        else:
            max_d = min(cur[j], 1.0 - cur[i])
        if max_d < 1e-12:
            continue
        d = rng.random() * max_d
        # Move: transfer d from higher to lower (or random direction)
        if rng.random() < 0.5:
            cur[i] -= d
            cur[j] += d
        else:
            cur[i] += d
            cur[j] -= d
        # Check bounds
        if cur[i] < -1e-12 or cur[i] > 1.0 + 1e-12 or cur[j] < -1e-12 or cur[j] > 1.0 + 1e-12:
            cur[i] = np.clip(cur[i], 0, 1)
            cur[j] = np.clip(curlj(cur[i], cur[j], d, i, j, cur), 0, 1)
            # Just revert if out of bounds
            cur[i] -= (cur[i] - np.clip(cur[i], 0, 1))
            continue
        s_new = _score(cur)
        delta = s_new - cur_s
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_s = s_new
            if s_new < best_s:
                best_s = s_new
                best = cur.copy()
        else:
            cur[i] += d if (cur[i] < cur[j] or rng.random() < 0.5) else -d
            cur[j] -= d if (cur[i] < cur[j] or rng.random() < 0.5) else -d
            # Revert
            cur = best.copy() if False else cur  # keep cur
            # Actually just undo
            pass
        T *= cool
    # Project final
    best = _project(best)
    best_s = _score(best)
    return best, best_s

def _sa_clean(h0, n, n_steps, T_init, T_final, rng, deadline, reserve=10.0):
    """Clean simulated annealing with sum-preserving moves."""
    cur = h0.copy()
    cur_s = _score(cur)
    best = cur.copy()
    best_s = cur_s

    T = T_init
    cool = (T_final / T_init) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.time() > deadline - reserve:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n
        max_d = min(min(cur[i], cur[j]), max(0, 1.0 - max(cur[i], cur[j])))
        # Actually: we want to transfer d from one to other
        # If we move d from i to j: need cur[i]>=d and cur[j]+d<=1
        # If we move d from j to i: need cur[j]>=d and cur[i]+d<=1
        d_max = min(cur[i], 1.0 - cur[j])  # move from i to j
        if d_max < 1e-10:
            d_max = min(cur[j], 1.0 - cur[i])  # move from j to i
        if d_max < 1e-10:
            continue
        d = float(rng.random()) * d_max
        # Random direction
        if rng.random() < 0.5:
            old_i, old_j = cur[i], cur[j]
            cur[i] -= d
            cur[j] += d
        else:
            old_i, old_j = cur[i], cur[j]
            cur[i] += d
            cur[j] -= d
        # Verify feasibility
        if cur[i] < -1e-14 or cur[i] > 1.0 + 1e-14 or cur[j] < -1e-14 or cur[j] > 1.0 + 1e-14:
            cur[i], cur[j] = old_i, old_j
            continue
        s_new = _score(cur)
        delta = s_new - cur_s
        if delta < 0 or float(rng.random()) < np.exp(-delta / max(T, 1e-12)):
            cur_s = s_new
            if s_new < best_s:
                best_s = s_new
                best = cur.copy()
        else:
            cur[i], cur[j] = old_i, old_j
        T *= cool

    best = _project(best)
    best_s = _score(best)
    return best, best_s


def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta


def _lse_grad(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * len(x) - 1, 1.0 / (2 * len(x) - 1))
    else:
        w = ev / es
    return _weighted_corr_grad(x, w)


def _lse_slsqp(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def _efun(x):
        return x.sum() - target

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _fista(h, beta, n_iter, alpha_init=0.006, mu=0.9):
    """FISTA accelerated subgradient with averaging."""
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    y = h.copy()
    h_avg = h.copy()
    best_h = h.copy()
    best_s = _score(h)
    t = 1
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        t_new = 1.5 * t
        y = h_cur + ((t - 1) / t_new) * (h_cur - h_prev)
        y = _project(y)
        h_new = y - alpha_init * grad
        h_new = _project(h_new)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_cur - h_avg) / (t)
        t = t_new
        if it % 200 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 1000 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s


def _polish(h0, deadline, reserve=8.0):
    """Multi-stage polishing of a candidate."""
    cur = h0.copy()
    cur_s = _score(cur)

    # P-norm SLSQP with increasing p
    for p in [4, 8, 16, 32]:
        if time.time() > deadline - reserve:
            break
        new = _pnorm_slsqp(cur, p, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    # LSE SLSQP
    if time.time() < deadline - reserve:
        for beta in [100, 300, 800]:
            if time.time() > deadline - reserve:
                break
            new = _lse_slsqp(cur, beta, maxiter=120)
            ns = _score(new)
            if ns < cur_s - 1e-12:
                cur, cur_s = new, ns

    # FISTA
    if time.time() < deadline - reserve:
        cur, cur_s = _fista(cur, beta=300, n_iter=6000, alpha_init=0.005, mu=0.88)

    # P-norm final
    if time.time() < deadline - reserve:
        new = _pnorm_slsqp(cur, 32, maxiter=120)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    # Direct min-max
    if time.time() < deadline - reserve:
        new = _direct_minmax(cur, maxiter=120)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    # Final FISTA
    if time.time() < deadline - reserve:
        cur, cur_s = _fista(cur, beta=600, n_iter=4000, alpha_init=0.003, mu=0.92)

    return cur, cur_s

            h2 = _pnorm_slsqp(h, 8, maxiter=60)
            s2 = _score(h2)
            if s2 < s:
                h, s = h2, s2
        results.append((s, n, h))
    results.sort(key=lambda x: x[0])
    return results


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(48) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(parent)
    parent_n = len(parent)

    # Phase 1: Try different n values
    n_candidates = []
    for n in range(max(20, parent_n - 10), min(80, parent_n + 20) + 1, 2):
        n_candidates.append(n)
    if parent_n not in n_candidates:
        n_candidates.append(parent_n)

    screen_results = _try_all_ns(parent, n_candidates, deadline, reserve=60.0)

    for s, n, h in screen_results[:3]:
        if s < best_score - 1e-12:
            best_h = h.copy()
            best_score = s
            best_n = n

    # Phase 2: Global search using DE and SA on best dimensions
    top_ns = list(set([r[1] for r in screen_results[:4]]))
    if best_n not in top_ns:
        top_ns.append(best_n)

    de_deadline = deadline - 120
    for n in top_ns[:3]:
        if time.time() > de_deadline:
            break
        # Get starting point for this n
        h_start = best_h
        if len(h_start) != n:
            h_start = _interp(best_h, n)
        h_start = _project(h_start)

        # DE
        h_de, s_de = _de_global(h_start, n, pop_size=24, n_gen=30,
                                rng=rng, deadline=de_deadline, reserve=20.0)
        if s_de < best_score - 1e-12:
            best_h = h_de.copy()
            best_score = s_de
            best_n = n

        # SA
        h_sa, s_sa = _sa_clean(h_de, n, n_steps=3000,
                               T_init=max(0.01, best_score * 0.05),
                               T_final=0.001, rng=rng,
                               deadline=de_deadline, reserve=10.0)
        if s_sa < best_score - 1e-12:
            best_h = h_sa.copy()
            best_score = s_sa
            best_n = n

    # Phase 3: Polish best solution
    if time.time() < deadline - 30:
        best_h, best_score = _polish(best_h, deadline, reserve=15.0)

    # Phase 4: More diverse starts and polish
    if time.time() < deadline - 60:
        n = best_n
        starts = [best_h.copy()]
        starts.append(_mirror(best_h))
        starts.append(_binary_push(best_h, 1.5))
        starts.append(_binary_push(best_h, 3.0))
        for sigma in [0.05, 0.1, 0.2]:
            starts.append(_project(best_h + sigma * rng.standard_normal(n)))
        # Structured perturbation
        for freq in [1, 2, 3, 5]:
            phase = rng.uniform(0, 6.28)
            pert = 0.08 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            starts.append(_project(best_h + pert))

        for h0 in starts[:8]:
            if time.time() > deadline - 20:
                break
            h0 = _project(h0)
            cur, cur_s = _polish(h0, deadline, reserve=10.0)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)

    # Phase 5: Focused refinement with multiple p values
    if time.time() < deadline - 20:
        for p in [6, 12, 24, 48, 96]:
            if time.time() > deadline - 8:
                break
            new = _pnorm_slsqp(best_h, p, maxiter=80)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h = new
                best_score = ns

    # Phase 6: Direct min-max final
    if time.time() < deadline - 10:
        new = _direct_minmax(best_h, maxiter=200)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h = new
            best_score = ns

    # Phase 7: Final FISTA
    if time.time() < deadline - 5:
        best_h, best_score = _fista(best_h, beta=1000, n_iter=5000,
                                    alpha_init=0.002, mu=0.93)

    # Final check
    best_h = _project(best_h)
    final_score = _score(best_h)
    if final_score < best_score:
        best_score = final_score

    return (best_h, final_score, len(best_h))
