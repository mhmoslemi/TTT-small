    n = len(h)

        h = np.clip(h, 0, 1)

def _pair_optimize(h, i, j):

    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)

        v = np.max(_corr(h))
        return v

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

2. Ensure sum(p) = 0 (for feasibility)
3. Find α* that minimizes max_d (corr[d] + b_d*α - c_d*α²) subject to 0 ≤ h+α*p ≤ 1
4. Update h ← h + α**p
5. Repeat

- Binary search on α in the feasible range
- Or: find all breakpoints (where one quadratic crosses another) and check each

def _direction_eval(h, p, n):
    """Compute the correlation values as functions of alpha: a_d + b_d*alpha - c_d*alpha^2"""
    corr = _corr(h)  # a_d
    # b_d = sum p[k]*(1-h[k+d]) - sum h[k]*p[k+d]
    # c_d = sum p[k]*p[k+d]
    b = np.zeros(2*n-1)
    c = np.zeros(2*n-1)
    for d in range(-(n-1), n):
        r = n-1+d
        if d >= 0:
            m = n-d
            b[r] = np.sum(p[:m]*(1-h[d:])) - np.sum(h[:m]*p[d:])
            c[r] = np.sum(p[:m]*p[d:])
        else:
            g = -d
            m = n-g
            b[r] = np.sum(p[:m]*(1-h[g:])) - np.sum(h[g:]*p[:m])
            c[r] = np.sum(p[:m]*p[g:])
    return corr, b, c

def _minimize_direction(h, p, n):
    """Find best alpha to minimize max_d (a_d + b_d*alpha - c_d*alpha^2)"""
    corr, b, c = _direction_eval(h, p, n)
    
    # Feasible range for alpha
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1-h[j])/p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j]/(-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    def f(alpha):
        vals = corr + b*alpha - c*alpha*alpha
        return np.max(vals)
    
    # Golden section search on [0, alpha_max]
    gr = (np.sqrt(5)-1)/2
    lo, hi = 0.0, alpha_max
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1, f2 = f(c1), f(c2)
    
    for _ in range(30):
        if hi-lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr*(hi-lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr*(hi-lo)
            f2 = f(c2)
    
    best_alpha = (lo+hi)/2
    best_f = f(best_alpha)
    return best_alpha, best_f

    n = len(h)

        # Smoothed subgradient using softmax weights
        weights = np.exp(smooth_beta * (corr - np.max(corr)))
        weights /= np.sum(weights)
    else:
        # Uniform over top-k
        top_idx = np.argsort(corr)[-top_k:]
        weights = np.zeros(2*n-1)
        weights[top_idx] = 1.0/top_k

    for d in range(-(n-1), n):

        if d >= 0:
            m = n - d
            grad[:m] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:m]
        else:
            g = -d
            m = n - g
            grad[:m] += w * (1.0 - h[g:])
            grad[g:] -= w * h[:m]

The Erdős problem is related to the following: find a set E ⊂ [0,1] of measure 1/2 that minimizes max_t m(E ∩ (E+t)).

Wait, that's the complement of what we have. Let me re-read the problem.

We have h: [0,2] → [0,1] with ∫h = 1. We want to minimize:
max_k ∫ h(x)(1-h(x+k)) dx

This is the measure of points where h=1 and h shifted by k = 0. Equivalently, it's:
max_k (m({h=1}) - m({h=1} ∩ {h(·+k)=1}))
= max_k (1 - m({h=1} ∩ {h(·+k)=1}))

So we want to maximize min_k m({h=1} ∩ {h(·+k)=1}), i.e., make the autocorrelation as large as possible at all shifts simultaneously.

This is equivalent to finding a function h with "flat" autocorrelation (as uniform as possible).

For the step function discretization, this means we want the autocorrelation vector to be as flat as possible (subject to the boundary effects).

**Key insight for a new approach**: Use the structure of the autocorrelation.

The autocorrelation R(d) = sum h[k]h[k+d] is a positive-definite sequence (it's the autocorrelation of a finite signal). So it has a non-negative Fourier transform (periodogram).

If we want R(d) to be "flat" (constant for all d), we'd need a signal with a delta-function periodogram, which is an impulse - not possible for a bounded signal.

But we can get close by using a signal whose periodogram is as concentrated as possible at frequency 0 (corresponding to a constant signal) while satisfying the constraint.

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _project(h, target):
    n = len(h)
    h = h.astype(float).copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt < 1e-10:
            h = np.clip(h + diff/n, 0, 1)
            break
        h[mask] += diff / cnt
    return np.clip(h, 0, 1)

def _line_search_quad(h, p):

def _subgrad(h, top_k=3):

    corr = _corr(h)
