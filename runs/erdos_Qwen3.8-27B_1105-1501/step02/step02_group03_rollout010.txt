import numpy as np
from scipy.optimize import minimize
import time

# Top-level helper functions

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(h):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    residual = target - np.sum(h)
    if abs(residual) < 1e-14:
        return h
    # Distribute residual proportionally among non-saturated entries
    if residual > 0:
        headroom = 1.0 - h
        total_headroom = headroom.sum()
        if total_headroom > 0:
            h += headroom * (residual / total_headroom)
        else:
            # All at 1, can't add
            pass
    else:
        space = h
        total_space = space.sum()
        if total_space > 0:
            h += space * (residual / total_space)
    # Final clip and repair
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Sequential fix
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return h

def antisym_h(left):
    """Build full h from left half (anti-symmetric)."""
    m = len(left)
    n = 2 * m
    h = np.empty(n)
    h[:m] = left
    h[m:] = 1.0 - left[::-1]
    return h

def symmetrize(h):
    """Project h onto anti-symmetric subspace."""
    n = len(h)
    m = n // 2
    left = 0.5 * (h[:m] + 1.0 - h[n-1:n-m-1:-1])
    return antisym_h(left)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_full(h):
    """Full gradient matrix (2n-1) x n."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        D = p - (n-1)
        if D >= 0:
            L = n - D
            G[p, :L] += dx * (1.0 - h[D:D+L])
            G[p, D:] -= dx * h[:L]
        else:
            a = -D
            L = n - a
            G[p, a:] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[a:a+L]
    return G

def smooth_obj_grad_sym(z, beta):
    """Smooth objective and gradient in anti-symmetric logistic space."""
    m = len(z)
    n = 2 * m
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))
    h = antisym_h(u)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -700, 700)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_full(h)
    sigma = u * (1.0 - u)
    G_z = G[:, :m] * sigma[np.newaxis, :] - G[:, n-1:n-m-1:-1] * sigma[np.newaxis, :]
    grad = w @ G_z
    return obj, grad

# Class for SLSQP in symmetric space
class SymMinimaxProb:
    def __init__(self, active_idx, n):
        self.active = np.array(active_idx, dtype=int)
        self.n = n
        self.m = n // 2

            J[i, :self.n] = -G[p, :]
            J[i, -1] = 1.0
        return J
    
    def constr_eq(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0
    
    def constr_eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def select_active(vals, frac=0.3, min_count=10):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def interp_to(h_src, n_tgt):
    """Interpolate from source to target resolution, preserve anti-symmetry."""
    m_src = len(h_src) // 2
    m_tgt = n_tgt // 2
    left_src = h_src[:m_src]
    x_src = (np.arange(m_src) + 0.5) * (2.0 / len(h_src))
    x_tgt = (np.arange(m_tgt) + 0.5) * (2.0 / n_tgt)
    left_tgt = np.interp(x_tgt, x_src, left_src)
    left_tgt = np.clip(left_tgt, 0.0, 1.0)
    return antisym_h(left_tgt)

def h_to_z(h):
    """Convert anti-symmetric h to z variables (left half logit)."""
    m = len(h) // 2
    left = h[:m]
    left = np.clip(left, 1e-10, 1.0 - 1e-10)
    z = np.log(left / (1.0 - left))
    return np.clip(z, -20, 20)

def z_to_h(z):
    m = len(z)
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))
    return antisym_h(u)
