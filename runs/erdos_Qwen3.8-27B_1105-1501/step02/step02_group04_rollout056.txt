    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(result)
    if abs(diff) > 1e-15:
        mask = (result > 1e-14) & (result < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result[mask] = np.clip(result[mask], 0.0, 1.0)
        else:

            if diff > 0:

            else:
