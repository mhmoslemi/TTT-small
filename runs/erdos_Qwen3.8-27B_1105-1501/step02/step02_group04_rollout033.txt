import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pair_line_search(corr, h, i, j, n, n_iter=60):
    """
    Find optimal t for h[i] += t, h[j] -= t
    using the exact formula corr_new[s] = corr[s] + t*A[s] + B[s]*t^2
    """
    t_low = -min(h[i], 1.0 - h[j])
    t_high = min(1.0 - h[i], h[j])
    if t_high - t_low < 1e-13:
        return 0.0
    
    # Compute A vector
    A = np.zeros(2 * n - 1)
    offset = n - 1

        if 0 <= i - s < n:
            A[idx] += (-h[i - s])

    for s in range(-(n-1), n):

            A[idx] += (-(1.0 - h[j + s]))
        if 0 <= j - s < n:
            A[idx] += h[j - s]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pair_line_search(corr, h, i, j, n, grid_size=500, n_refine=25):
    """Fast pairwise line search using exact quadratic formula."""
    t_low = -min(h[i], 1.0 - h[j])
    t_high = min(1.0 - h[i], h[j])
    if t_high - t_low < 1e-13:
        return 0.0
    
    # Compute A vector (linear coefficient for each lag)
    A = np.zeros(2 * n - 1)
    # h[i] += t: first factor
    A[n-1-i : 2*n-1-i] += (1.0 - h)
    # h[i] += t: second factor  
    A[i : i+n] += -h[::-1]
    # h[j] -= t: first factor
    A[n-1-j : 2*n-1-j] += (h - 1.0)
    # h[j] -= t: second factor
    A[j : j+n] += h[::-1]
    
    d = j - i
    B_pos1 = d + n - 1
    B_pos2 = -d + n - 1
    
    # Grid search for approximate minimum
    t_grid = np.linspace(t_low, t_high, grid_size)
    val = corr[None, :] + t_grid[:, None] * A[None, :]
    val[:, B_pos1] += t_grid**2
    val[:, B_pos2] += t_grid**2

    best_idx = int(np.argmin(max_vals))

        lo = t_low
        hi = t_grid[1]

def full_sweep(h, corr, rng, grid_size=500, n_refine=25):
    """One full sweep of pairwise optimization."""
    n = len(h)
    improved = False
    perm = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            t = pair_line_search(corr, h, i, j, n, grid_size, n_refine)
            if abs(t) > 1e-10:
                h[i] += t
                h[j] -= t
                improved = True
                # Recompute corr
                corr = np.correlate(h, 1.0 - h, mode="full")
    return h, corr, improved

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def pair_line_search(corr, h, i, j, n, grid_size=500, n_refine=25):
    t_low = -min(h[i], 1.0 - h[j])
    t_high = min(1.0 - h[i], h[j])
    if t_high - t_low < 1e-13:
        return 0.0
    
    A = np.zeros(2 * n - 1)
    A[n-1-i : 2*n-1-i] += (1.0 - h)
    A[i : i+n] += -h[::-1]
    A[n-1-j : 2*n-1-j] += (h - 1.0)
    A[j : j+n] += h[::-1]
    
    d = j - i
    B_pos1 = d + n - 1
    B_pos2 = -d + n - 1
    
    # Quick check: if t=0 is already optimal (gradient check)
    # f'(0+) from the active constraint
    # For now, just do the search
    
    t_grid = np.linspace(t_low, t_high, grid_size)
    val = corr[None, :] + t_grid[:, None] * A[None, :]
    val[:, B_pos1] += t_grid**2
    val[:, B_pos2] += t_grid**2
    max_vals = np.max(val, axis=1)
    best_idx = int(np.argmin(max_vals))
    
    if best_idx == 0:
        lo = t_low
        hi = t_grid[min(1, grid_size-1)]
    elif best_idx == grid_size - 1:
        lo = t_grid[max(0, grid_size-2)]
        hi = t_high
    else:
        lo = t_grid[best_idx - 1]
        hi = t_grid[best_idx + 1]
    
    for _ in range(n_refine):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = corr + m1 * A
        v1[B_pos1] += m1 * m1
        v1[B_pos2] += m1 * m1
        f1 = np.max(v1)
        
        v2 = corr + m2 * A
        v2[B_pos1] += m2 * m2
        v2[B_pos2] += m2 * m2
        f2 = np.max(v2)
        
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    return (lo + hi) / 2.0
