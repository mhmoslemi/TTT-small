def project_onto_constraint(h, target_sum):

    lo, hi = -2.0, 2.0

import numpy as np
import time
from scipy.optimize import linprog


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_jacobian(h):
    n = len(h)
    omh = 1.0 - h
    D = np.zeros((2 * n - 1, n))
    for j in range(n):
        ej = np.zeros(n)
        ej[j] = 1.0
        D[:, j] = np.correlate(ej, omh, "full") - np.correlate(h, ej, "full")
    return D


def solve_lp_direction(h, D, trust):
    n = len(h)
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    # Find d,theta: minimize theta s.t. D@d - theta <= max_corr - corr, sum(d)=0, |d|<=trust
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-trust, trust)] * n + [(-1.0, 1.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n].copy()
            d -= np.mean(d)
            return d, res.x[n]
    except Exception:
        pass
    return None, 0.0


def line_search(h, d, D, corr0, t_max=1.0, n_samples=150):
    n = len(h)
    dd_corr = np.correlate(d, d, "full")
    linear = D @ d
    t_lim = t_max
    for i in range(n):
        if d[i] > 1e-12:
            t_lim = min(t_lim, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-12:
            t_lim = min(t_lim, h[i] / (-d[i]))
    t_lim = min(t_lim, 1.0)
    if t_lim <= 0:
        return 0.0, np.max(corr0)
    best_t = 0.0
    best_val = np.max(corr0)
    ts = np.linspace(0.0, t_lim, n_samples)
    for t in ts:
        if t <= 0:
            continue
        corr_t = corr0 + t * linear - t * t * dd_corr
        val = np.max(corr_t)
        if val < best_val - 1e-12:
            best_val = val
            best_t = t
    if best_t > 0:
        span = max(t_lim / n_samples, 1e-4)
        ts2 = np.linspace(max(0, best_t - span), min(t_lim, best_t + span), 40)
        for t in ts2:
            if t <= 0:

            corr_t = corr0 + t * linear - t * t * dd_corr
            val = np.max(corr_t)
            if val < best_val - 1e-12:

def pair_optimal_delta(h, i, j):

def pair_polish(h, n_passes=3, rng=None, time_budget=None, t_start=None):
    n = len(h)
    target = n / 2.0
    for p in range(n_passes):
        if time_budget is not None and time.time() - t_start > time_budget:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for a in range(n):
            if time_budget is not None and time.time() - t_start > time_budget:
                break
            for b in range(a + 1, n):
                i, j = order[a], order[b]
                if i > j:
                    i, j = j, i
                delta = pair_optimal_delta(h, i, j)
                if abs(delta) > 1e-12:
                    h[i] -= delta
                    h[j] += delta
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_constraint(h, target)
    return h


def make_random_binary(n, rng):

def make_sinusoidal(n, freq, phase):

def make_golden(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = max(0.0, 1.0 - 2.0 * abs(frac - 0.5))
    return project_onto_constraint(h, n / 2.0)


def make_burst(n, rng):
    h = np.zeros(n)
    center = rng.randint(0, n)
    width = rng.randint(n // 4, n // 2)
    for i in range(n):

        if dist < width:
            h[i] = 1.0 - 0.5 * (dist / width)**2
    h = project_onto_constraint(h, n / 2.0)
    return h

            best_h = h_lp.copy()
            best_n = len(h_lp)

    # Phase 2: Pair polish
    if remaining() > 10:
        h_p = best_h.copy()
        h_p = pair_polish(h_p, n_passes=4, rng=rng, time_budget=remaining() * 0.15, t_start=t0)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 - 1e-10:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = len(h_p)

    # Phase 3: Multi-size, multi-start exploration
    sizes = [best_n]
    for s in [65, 60, 55, 50, 45, 40, 70, 75]:
        if s not in sizes and 30 <= s <= 80:
            sizes.append(s)
    sizes = sorted(set(sizes))

    for n_try in sizes:
        if remaining() < 10:
            break
        if n_try == best_n and best_c5 < 0.381:
            continue

        starts = []
        if best_h is not None:
            if best_n == n_try:
                starts.append(best_h.copy())
            else:
                starts.append(transfer_size(best_h, n_try))

        for _ in range(2):
            starts.append(make_random_binary(n_try, rng))
        starts.append(make_sinusoidal(n_try, 1.0, rng.uniform(0, 2 * np.pi)))
        starts.append(make_sinusoidal(n_try, 2.0, rng.uniform(0, 2 * np.pi)))
        starts.append(make_sinusoidal(n_try, 3.0, rng.uniform(0, 2 * np.pi)))
        starts.append(make_golden(n_try))
        starts.append(make_burst(n_try, rng))

        best_start = None
        best_start_c5 = float('inf')
        for s in starts:
            s = np.clip(s, 0.0, 1.0)
            s = project_onto_constraint(s, n_try / 2.0)
            c5_s = compute_c5(s)
            if c5_s < best_start_c5:
                best_start_c5 = c5_s
                best_start = s.copy()

        if best_start is None:
            continue

        alloc = min(remaining() * 0.18, 80.0)
        h_lp, c5_lp = lp_local_search(best_start, alloc, rng)

        if remaining() > 5:
            h_lp = pair_polish(h_lp, n_passes=2, rng=rng, time_budget=remaining() * 0.08, t_start=t0)
            c5_lp = compute_c5(h_lp)

        if c5_lp < best_c5 - 1e-10:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            best_n = n_try

    # Phase 4: Final refinement on best solution with more LP iterations
    if remaining() > 15:
        alloc = min(remaining() * 0.5, 120.0)
        h_f, c5_f = lp_local_search(best_h, alloc, rng)
        if c5_f < best_c5 - 1e-10:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(h_f)

    # Phase 5: Final pair polish
    if remaining() > 5:
        h_pp = best_h.copy()
        h_pp = pair_polish(h_pp, n_passes=3, rng=rng, time_budget=remaining() * 0.6, t_start=t0)
        c5_pp = compute_c5(h_pp)
        if c5_pp < best_c5 - 1e-10:
            best_c5 = c5_pp
            best_h = h_pp.copy()

    # Ensure validity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
