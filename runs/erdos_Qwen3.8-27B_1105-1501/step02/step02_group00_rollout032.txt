    """Compute constraint values and Jacobian for SLSQP."""
    h = x[:N]
    z = x[N]
    scores = np.correlate(h, 1.0 - h, mode='full')
    # Constraint values: z - scores[d] >= 0
    cvals = z - scores
    
    # Jacobian: row d is [1, -G[d, :]]
    # where G is the gradient matrix of scores w.r.t. h
    G = np.zeros((2*N-1, N))
    # d = 0 row
    G[N-1, :] = 1.0 - 2.0 * h  # score_0 = sum h*(1-h), grad = 1-2h
    
    # For d >= 1 (row index N-1+d in scores array)
    for d in range(1, N):
        row = N - 1 + d
        # Term 1: (1-h[d:]) in columns d..N-1
        G[row, d:] += (1.0 - h[:N-d])
        # Term 2: -h[:N-d] in columns 0..N-1-d  
        G[row, :N-d] += -h[d:]
    
    # For d < 0, g = -d (row index N-1-d = N-1+g... wait)
    # scores array index: scores[k] corresponds to d = k - (N-1)
    # So d = -g corresponds to index k = N-1-g

    scores = np.correlate(h, 1.0 - h, mode='full')

            best_h = h_new.copy()
        
        h = h_new
        alpha *= 0.999  # slow decay

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = np.minimum(1.0, np.maximum(0.0, x + res / cnt))
    return x

    h = x[:n]
    z = x[n]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return z - scores[n - 1:]

    h = x[:n]
    jac = np.zeros((n, n + 1))
    jac[:, n] = 1.0

        g[k:] = (1.0 - h[:n - k])
        g[:n - k] += -h[k:]
        jac[k, :n] = -g
    return jac

    constraints = [
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac},
