        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(res) < 1e-15:
                break
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_active(x, active_idx, weights, n):
    g = np.zeros(n)
    dx = 2.0 / n
    for ii, k in enumerate(active_idx):
        w = weights[ii]
        if w < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                g[i] += w * dx * (-x[i + s])
                g[i + s] += w * dx * (1.0 - x[i])
        else:
            t = -s
            for i in range(t, n):
                g[i] += w * dx * (-x[i - t])
                g[i - t] += w * dx * (1.0 - x[i])
    return g

def _active_weights(cv, beta, top_m=10):
    idx = np.argsort(cv)[::-1][:top_m]
    vals = cv[idx]
    shifted = beta * (vals - vals[0])
    w = np.exp(np.clip(shifted, -100, 0))
    s = w.sum()
    if s < 1e-300:
        w = np.ones(len(idx)) / len(idx)
    else:
        w = w / s
    return idx, w

def _grad_descent_active(h0, deadline, beta_start=50, beta_end=500,
                         n_iter=20000, step_init=0.01, momentum=0.9,
                         top_m=8):
    n = len(h0)
    h_cur = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    vel = np.zeros(n)
    alpha = step_init
    stall_count = 0

    for it in range(n_iter):
        if time.time() > deadline:
            break
        cv = _corr_vec(h_cur)
        idx, w = _active_weights(cv, beta_start if it < n_iter // 3 else
                                  beta_start + (beta_end - beta_start) * min(1.0, (it - n_iter // 3) / (n_iter * 2.0 / 3)),
                                  top_m=top_m)
        g = _grad_active(h_cur, idx, w, n)
        vel = momentum * vel - alpha * g
        h_new = _project(h_cur + vel)
        s_new = _score(h_new)
        if s_new < best_s - 1e-14:
            best_s = s_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        if s_new < _score(h_cur) - 1e-15:
            alpha = min(alpha * 1.04, 0.04)
        else:
            alpha = max(alpha * 0.75, 0.001)
        if stall_count > 500:
            alpha = max(alpha, 0.02)
            stall_count = 0
        h_cur = h_new

    return best_h, best_s

def _slsqp_active(h0, beta=200, maxiter=100, top_m=8):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        cv = _corr_vec(x)
        idx, w = _active_weights(cv, beta, top_m=top_m)
        return float(np.dot(w, cv[idx]))

    def _obj_grad(x):
        cv = _corr_vec(x)
        idx, w = _active_weights(cv, beta, top_m=top_m)
        return _grad_active(x, idx, w, n)

    def _eq_fun(x):
        return x.sum() - n / 2.0

    def _eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _direct_minmax(h0, maxiter=100, top_m=8):
    n = len(h0)
    cv = _corr_vec(h0)
    idx_active, _ = _active_weights(cv, beta=1, top_m=top_m)
    t0 = cv.max() + 0.003
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv[idx_active]

    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _fixed_binary_optimize(h0, threshold=0.04, maxiter=150, beta=400):
    n = len(h0)
    binary_mask = (h0 <= threshold) | (h0 >= 1.0 - threshold)
    free_mask = ~binary_mask
    n_free = int(np.sum(free_mask))
    if n_free < 2 or n_free > n - 2:
        return h0.copy(), False

    free_idx = np.where(free_mask)[0]
    h0_copy = h0.copy()
    h0_copy[binary_mask] = np.where(h0_copy[binary_mask] > 0.5, 1.0, 0.0)
    fixed_sum = h0_copy[binary_mask].sum()
    target = n / 2.0 - fixed_sum
    x0 = h0_copy[free_mask].copy()
    bounds = [(0.0, 1.0)] * n_free

    def _obj(x):
        h = h0_copy.copy()
        h[free_idx] = x
        cv = _corr_vec(h)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        return mc + np.log(es) / beta

    def _obj_grad(x):
        h = h0_copy.copy()
        h[free_idx] = x
        cv = _corr_vec(h)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        full_g = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            if s >= 0:
                for i in range(n - s):
                    full_g[i] += wk * dx * (-h[i + s])
                    full_g[i + s] += wk * dx * (1.0 - h[i])
            else:
                t = -s
                for i in range(t, n):
                    full_g[i] += wk * dx * (-h[i - t])
                    full_g[i - t] += wk * dx * (1.0 - h[i])
        return full_g[free_idx]

    def _eq_fun(x):
        return x.sum() - target

    def _eq_jac(x):
        return np.ones(n_free)

    try:
        res = minimize(_obj, x0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            h_new = h0_copy.copy()
            h_new[free_idx] = res['x']
            h_new = _project(h_new)
            if np.sum(binary_mask) > 0:
                h_new[binary_mask] = np.where(h0_copy[binary_mask] > 0.5, 1.0, 0.0)
                h_new = _project(h_new)
            return h_new, True
    except Exception:
        pass
    return h0.copy(), False

def _binary_promote(h, threshold=0.05):
    n = len(h)
    best_h = h.copy()
    best_s = _score(h)

    can_zero = h < threshold
    can_one = h > 1.0 - threshold

    trials = []
    if np.any(can_zero):
        trials.append(can_zero.copy())
    if np.any(can_one):
        trials.append(can_one.copy())
    if np.any(can_zero) and np.any(can_one):
        t = can_zero.copy()
        t |= can_one
        trials.append(t)

    for mask in trials:
        h_try = h.copy()
        h_try[mask & (h < 0.5)] = 0.0
        h_try[mask & (h >= 0.5)] = 1.0
        h_try = _project(h_try)
        s_try = _score(h_try)
        if s_try < best_s - 1e-12:
            best_h, best_s = h_try, s_try

    return best_h, best_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _local_search(h0, deadline, rng, n_trials=300):
    n = len(h0)
    best_h = h0.copy()
    best_s = _score(h0)

    for trial in range(n_trials):
        if time.time() > deadline:
            break
        h_try = best_h.copy()
        r = rng.random()
        if r < 0.25:
            i, j = rng.choice(n, size=2, replace=False)
            amt = min(0.04, best_h[i], 1.0 - best_h[j])
            if amt > 1e-6:
                h_try[i] -= amt
                h_try[j] += amt
        elif r < 0.45:
            h_try += 0.015 * rng.standard_normal(n)
        elif r < 0.65:
            j = rng.integers(n)
            h_try[j] = 1.0 if best_h[j] > 0.5 else 0.0
        elif r < 0.8:
            j = rng.integers(1, n - 1)
            h_try[j] = 0.5 * (best_h[j - 1] + best_h[j + 1])
        else:
            free = (best_h > 0.02) & (best_h < 0.98)
            if np.sum(free) > 2:
                h_try[free] += 0.02 * rng.standard_normal(np.sum(free))

        h_try = _project(h_try)
        s_try = _score(h_try)
        if s_try < best_s - 1e-12:
            best_h, best_s = h_try, s_try

    return best_h, best_s

def _chain_optimize(h0, deadline, rng):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)

    if time.time() < deadline - 40:
        cur, cur_s = _grad_descent_active(cur, max(deadline - 60, time.time() + 5),
                                          beta_start=30, beta_end=300,
                                          n_iter=4000, step_init=0.01, momentum=0.88, top_m=8)

    if time.time() < deadline - 30:
        cur, cur_s = _binary_promote(cur, threshold=0.04)

    if time.time() < deadline - 30:
        new, promoted = _fixed_binary_optimize(cur, threshold=0.05, maxiter=120, beta=300)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    if time.time() < deadline - 25:
        new = _slsqp_active(cur, beta=80, maxiter=80, top_m=8)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    if time.time() < deadline - 20:
        cur, cur_s = _grad_descent_active(cur, max(deadline - 25, time.time() + 5),
                                          beta_start=100, beta_end=600,
                                          n_iter=4000, step_init=0.005, momentum=0.9, top_m=6)

    if time.time() < deadline - 15:
        new, promoted = _fixed_binary_optimize(cur, threshold=0.03, maxiter=100, beta=500)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    if time.time() < deadline - 15:
        cur, cur_s = _local_search(cur, max(deadline - 20, time.time() + 5), rng, n_trials=100)

    if time.time() < deadline - 10:
        new = _direct_minmax(cur, maxiter=80, top_m=8)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    if time.time() < deadline - 8:
        new = _slsqp_active(cur, beta=400, maxiter=60, top_m=6)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    return cur, cur_s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.1 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(57) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)
    rng = np.random.default_rng(seed)

    # Phase 1: Screen nearby n values
    n_candidates = sorted(set(range(max(42, parent_n - 10), min(72, parent_n + 10) + 1, 2)))
    phase1_results = [(best_score, best_n, best_h.copy())]

    for n in n_candidates:
        if time.time() > deadline - 150:
            break
        if n == parent_n:
            continue
        h_interp = _interp(parent, n)
        s0 = _score(h_interp)
        if time.time() < deadline - 80:
            h_opt = _slsqp_active(h_interp, beta=60, maxiter=40, top_m=8)
            s1 = _score(h_opt)
            if s1 < s0:
                h_interp, s0 = h_opt, s1
        phase1_results.append((s0, n, h_interp))
        if s0 < best_score - 1e-12:
            best_h = h_interp.copy()
            best_score = s0
            best_n = n

    # Phase 2: Focused optimization on top candidates
    phase1_results.sort(key=lambda x: x[0])
    top_candidates = phase1_results[:6]

    for s_init, n_cand, h_base in top_candidates:
        if time.time() > deadline - 80:
            break
        starts = [h_base.copy(), _mirror(h_base)]
        for sigma in [0.06, 0.12]:
            starts.append(_project(h_base + sigma * rng.standard_normal(n_cand)))

        for h0 in starts:
            if time.time() > deadline - 40:
                break
            cur, cur_s = _chain_optimize(h0, max(deadline - 30, time.time() + 10), rng)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)

    # Phase 3: Intensive refinement
    remaining = deadline - time.time()
    if remaining > 50:
        for bs, be in [(50, 200), (200, 800), (800, 2000)]:
            if time.time() > deadline - 20:
                break
            best_h, best_score = _grad_descent_active(
                best_h, max(deadline - 25, time.time() + 5),
                beta_start=bs, beta_end=be,
                n_iter=min(8000, int(remaining / 6)),
                step_init=0.004, momentum=0.9, top_m=6)
            remaining = deadline - time.time()

    if time.time() < deadline - 30:
        best_h, best_score = _binary_promote(best_h, threshold=0.03)
        if time.time() < deadline - 20:
            new, _ = _fixed_binary_optimize(best_h, threshold=0.04, maxiter=100, beta=500)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h, best_score = new, ns

    if time.time() < deadline - 20:
        new = _slsqp_active(best_h, beta=500, maxiter=100, top_m=6)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h, best_score = new, ns

    if time.time() < deadline - 15:
        new = _direct_minmax(best_h, maxiter=100, top_m=8)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h, best_score = new, ns

    # Phase 4: Random restarts
    remaining = deadline - time.time()
    n_restarts = min(8, int(remaining / 35))
    for _ in range(n_restarts):
        if time.time() > deadline - 20:
            break
        h_pert = _project(best_h + 0.015 * rng.standard_normal(best_n))
        cur, cur_s = _chain_optimize(h_pert, max(deadline - 20, time.time() + 5), rng)
        if cur_s < best_score - 1e-12:
            best_h = cur.copy()
            best_score = cur_s
            best_n = len(cur)

    # Final polish
    if time.time() < deadline - 10:
        best_h, best_score = _local_search(best_h, max(deadline - 5, time.time() + 3), rng, n_trials=200)

    if time.time() < deadline - 5:
        best_h, best_score = _grad_descent_active(
            best_h, deadline, beta_start=1000, beta_end=3000,
            n_iter=3000, step_init=0.002, momentum=0.92, top_m=5)

    best_h = _project(best_h)
    final_score = _score(best_h)
    return (best_h, final_score, len(best_h))
