def make_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    n = z.size
    if n == 0:
        return z
    z[np.isnan(z) | np.isinf(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    target = n / 2.0
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-14:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-14:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

This is related to the autocorrelation of the binary sequence.

For the continuous case, we can interpolate between 0 and 1, which might give better results than pure binary.

The key insight: the optimal h is likely to be "mostly" 0 or 1, with some intermediate values at the boundaries of runs. This suggests that binary search + continuous polishing is a good strategy.

import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(vals.max())

def make_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    n = z.size
    if n == 0:
        return z
    z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-14:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-14:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, p):
    """Compute gradient of V[p] = dx * corr[p] with respect to h."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        # V[p] = dx * sum_{j=0}^{n-1-s} h[j+s] * (1-h[j])
        # dV/dh[j] = dx * [(1-h[j-s]) for j in [s, n-1] - h[j+s] for j in [0, n-1-s]]
        j_range_1 = np.arange(s, n)  # where h appears as h[j+s] → h[j] = h[j] with shift
        # Actually: h[j+s] appears, so dV/dh[j+s] += dx*(1-h[j]) for j in [0, n-1-s]
        # and (1-h[j]) appears, so dV/dh[j] += dx*(-h[j+s]) for j in [0, n-1-s]
        m = n - s
        j = np.arange(m)
        grad[j + s] += dx * (1.0 - h[j])       # from h[j+s] term
        grad[j] -= dx * h[j + s]                 # from (1-h[j]) term
    else:
        t = -s
        # V[p] = dx * sum_{j=0}^{n-1-t} h[j+t] * (1-h[j])
        m = n - t
        j = np.arange(m)
        grad[j + t] += dx * (1.0 - h[j])
        grad[j] -= dx * h[j + t]
    return grad

def lp_step(h, active_indices, V, C, rho, mu=0.0):
    """Solve the LP minimax step. Returns direction u or None."""
    n = len(h)
    k = len(active_indices)
    
    # Build gradient matrix G (k x n)
    G = np.zeros((k, n))
    for i, p in enumerate(active_indices):
        G[i] = compute_lag_gradient(h, p)
    
    # LP: minimize t + mu * sum(v)
    # subject to:
    #   G @ u - t <= -V[active]  (linearized active constraints)
    #   sum(u) = 0
    #   -min(h, rho) <= u <= min(1-h, rho)
    #   v >= u, v >= -u (if mu > 0)
    #   t <= C
    
    if mu > 0:
        # Variables: u (n), t (1), v (n) → total 2n+1
        nv = 2 * n + 1
        # Objective: [0, 0, ..., 0, 1, mu, mu, ..., mu]
        c_obj = np.zeros(nv)
        c_obj[n] = 1.0  # t coefficient
        c_obj[n+1:] = mu  # v coefficients
        
        # Inequality constraints: A_ub @ x <= b_ub
        # G @ u - t <= -V[active]: G blocks on u, -1 on t
        A_ub = np.zeros((k + 2*n, nv))
        b_ub = np.zeros(k + 2*n)
        for i in range(k):
            A_ub[i, :n] = G[i]
            A_ub[i, n] = -1.0
            b_ub[i] = -V[active_indices[i]]
        # v >= u → -u + v >= 0 → u - v <= 0
        for i in range(n):
            A_ub[k+i, i] = 1.0
            A_ub[k+i, n+1+i] = -1.0
        # v >= -u → u + v >= 0 → -u - v <= 0
        for i in range(n):
            A_ub[k+n+i, i] = -1.0
            A_ub[k+n+i, n+1+i] = -1.0
        
        # Equality: sum(u) = 0
        A_eq = np.zeros((1, nv))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds
        bounds = []
        for i in range(n):
            lo = max(-h[i], -rho)
            hi = min(1.0 - h[i], rho)
            bounds.append((lo, hi))
        bounds.append((0.0, C))  # t
        for i in range(n):
            bounds.append((0.0, rho))  # v
        
    else:
        # Variables: u (n), t (1) → total n+1
        nv = n + 1
        c_obj = np.zeros(nv)
        c_obj[n] = 1.0  # minimize t
        
        A_ub = np.zeros((k, nv))
        b_ub = np.zeros(k)
        for i in range(k):
            A_ub[i, :n] = G[i]
            A_ub[i, n] = -1.0
            b_ub[i] = -V[active_indices[i]]
        
        A_eq = np.zeros((1, nv))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for i in range(n):
            lo = max(-h[i], -rho)
            hi = min(1.0 - h[i], rho)
            bounds.append((lo, hi))
        bounds.append((0.0, C))  # t
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            u = res.x[:n]
            t_opt = res.x[n]
            if t_opt < C - 1e-10:
                return u, t_opt
            # Even if t_opt ≈ C, the direction might still be useful
            if np.abs(u).max() > 1e-8:
                return u, t_opt
    except Exception:
        pass
    return None, None

def lp_refine(h, is_dead, max_iters=200):
    """Main LP refinement loop."""
    n = len(h)
    best_h = h.copy()
    best_score = score_exact(best_h)
    
    eta_start = 0.005
    eta = eta_start
    rho_start = 0.15
    rho = rho_start
    stall = 0
    mu = 0.0
    
    line_scales = [0.95, 0.75, 0.55, 0.4, 0.28, 0.2, 0.12, 0.07, 0.035, 0.017]
    
    for it in range(max_iters):
        if is_dead():
            break
        
        h = best_h
        V = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        C = V.max()
        
        # Select active set
        sorted_idx = np.argsort(V)[::-1]
        active = sorted_idx[:min(30, len(sorted_idx))]
        # Also include all within eta of max
        near_max = np.where(V >= C - eta)[0]
        active = np.unique(np.concatenate([active, near_max]))
        active = active[:50]  # Cap
        
        # Try LP
        u, t_opt = lp_step(h, active, V, C, rho, mu)
        
        accepted = False
        if u is not None:
            for alpha in line_scales:
                if is_dead():
                    break
                trial = make_feasible(h + alpha * u)
                sc = score_exact(trial)
                if sc < best_score - 1e-12:
                    best_h = trial
                    best_score = sc
                    stall = 0
                    accepted = True
                    break
        
        if not accepted:
            stall += 1
            # Adapt parameters
            if stall % 5 == 1:
                rho *= 0.7
                if rho < 0.01:
                    rho = rho_start
            elif stall % 5 == 2:
                eta *= 0.5
                if eta < 0.0005:
                    eta = eta_start
            elif stall % 5 == 3:
                mu = 0.05
            elif stall % 5 == 4:
                mu = 0.0
                eta = eta_start
                rho = rho_start
            if stall > 25:
                # Reset and try from slightly perturbed point
                rng_local = np.random.default_rng(it)
                h_pert = make_feasible(h + rng_local.normal(0, 0.02, n))
                sc_pert = score_exact(h_pert)
                if sc_pert < best_score:
                    best_h = h_pert
                    best_score = sc_pert
                stall = 0
                rho = rho_start
                eta = eta_start
    
    return best_h, best_score

def two_point_search(h, n_tries, rng, is_dead):
    """Random two-point exchange search."""
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, (t_lo + t_hi) / 2.0, t_lo + 0.25 * span, t_lo + 0.75 * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] += best_t
            best[b] -= best_t
            np.clip(best, 0, 1, out=best)
            best_sc = score_exact(best)
    
    return best, best_sc

def binary_sa(h_init, n_steps, rng, is_dead):
    """Simulated annealing on binary sequences."""
    n = len(h_init)
    if n % 2 != 0:
        return None, None
    
    target = n // 2
    # Initial binary from threshold
    h = h_init.copy()
    idx = np.argsort(-h)
    b = np.zeros(n, dtype=int)
    b[idx[:target]] = 1
    
    def binary_score(bits):
        return score_exact(bits.astype(float))
    
    current_score = binary_score(b)
    best_bits = b.copy()
    best_score = current_score
    
    T = 5e-4
    cool = 0.999
    
    pos1 = np.where(b == 1)[0]
    pos0 = np.where(b == 0)[0]
    
    for it in range(n_steps):
        if is_dead():
            break
        i = pos1[rng.integers(0, len(pos1))]
        j = pos0[rng.integers(0, len(pos0))]
        
        b[i] = 0
        b[j] = 1
        new_score = binary_score(b)
        
        delta = new_score - current_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_score = new_score
            pos1 = pos0.copy()
            pos0 = pos1.copy()
            # Actually need to swap the positions
            pos1 = np.where(b == 1)[0]
            pos0 = np.where(b == 0)[0]
            if new_score < best_score:
                best_score = new_score
                best_bits = b.copy()
        else:
            b[i] = 1
            b[j] = 0
        
        T *= cool
        if T < 1e-6:
            break
    
    return best_bits.astype(float), best_score

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

So: ∂V/∂h[j] = dx * [(1-h[j+t]) * 1_{0≤j≤n-1-t} - h[j-t] * 1_{t≤j≤n-1}]

For t=2, n=3:
j=0: [(1-h[2]) * 1_{0≤0≤0} - h[-2] * 1_{2≤0≤2}] = (1-h[2]) - 0 = 0.2 ✓
j=1: [(1-h[3]) * 1_{0≤1≤0} - h[-1] * 1_{2≤1≤2}] = 0 - 0 = 0 ✓
j=2: [(1-h[4]) * 1_{0≤2≤0} - h[0] * 1_{2≤2≤2}] = 0 - h[0] = -0.2 ✓
