import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def push_to_binary(h):
    """Push h towards binary (0 or 1) while maintaining feasibility."""
    # Sort by distance from 0.5, push extremes first
    d = np.abs(h - 0.5)
    order = np.argsort(d)[::-1]
    h_new = h.copy()
    for i in order:
        if h[i] > 0.5:
            h_new[i] = 1.0
        else:
            h_new[i] = 0.0
    return make_feasible(h_new)

def binary_all_swaps_score(h, ones, zeros):
    """Try all pair swaps and return the best (score, i, j)."""
    n = len(h)
    best_score = official_score(h)
    best_improve = 0
    best_swap = None
    for ii in range(len(ones)):
        i = ones[ii]
        for jj in range(len(zeros)):
            j = zeros[jj]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            s = official_score(h)
            h[i] = 1.0
            h[j] = 0.0
            improve = best_score - s
            if improve > best_improve:
                best_improve = improve
                best_swap = (i, j)
    return best_swap, best_improve

def binary_greedy(h, max_rounds=10000, time_limit=None, t_start=None):
    """Greedy local search: try all pair swaps, accept best improvement."""
    h = h.copy()
    n = len(h)
    best_score = official_score(h)
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:

        ones = np.where(h > 0.5)[0]

        if len(ones) == 0 or len(zeros) == 0:

        best_swap, best_improve = binary_all_swaps_score(h, ones, zeros)

            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        best_score = official_score(h)

def compute_corr_jacobian(h):
    """Compute the full Jacobian of the correlation w.r.t. h."""
    n = len(h)
    J = np.zeros((n, 2*n-1))
    for d in range(-(n-1), n):  # d is the shift
        k = n - 1 + d
        for m in range(n):
            val = 0.0
            if d > 0:
                if d <= m <= n-1:
                    val += (1.0 - h[m-d])
                if 0 <= m <= n-1-d:
                    val -= h[m+d]
            elif d < 0:
                dp = -d
                if 0 <= m <= n-1-dp:
                    val += (1.0 - h[m+dp])
                if dp <= m <= n-1:
                    val -= h[m-dp]
            else:  # d = 0
                val = 1.0 - 2.0*h[m]
            J[m, k] = val
    return J

def lse_objective_and_grad(h, beta, dx):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = np.max(corr)
    shifted = np.clip(beta * (corr - max_corr), -500, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        active = [n-1]  # just the max
        weights_active = [1.0]
    else:
        active = np.where(weights > 1e-10 * wsum)[0]
        weights_active = weights[active] / wsum
    
    obj = max_corr + np.log(wsum) / beta if wsum > 1e-300 else max_corr
    
    grad = np.zeros(n)
    for k, w in zip(active, weights_active):
        d = k - (n-1)
        if d == 0:
            g = 1.0 - 2.0*h
        elif d > 0:
            g = np.zeros(n)
            g[d:] += (1.0 - h[:n-d])
            g[:n-d] += (-h[d:])
        else:
            dp = -d
            g = np.zeros(n)
            g[:n-dp] += (1.0 - h[dp:])
            g[dp:] += (-h[:n-dp])
        grad += w * g * dx
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_to_binary(h):
    n = len(h)
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for k in range(n_ones - target):
            h_bin[ones_idx[order[k]]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(h[zeros_idx])[::-1]
        for k in range(target - n_ones):
            h_bin[zeros_idx[order[k]]] = 1
    return h_bin

def binary_greedy(h0, max_rounds=50000, time_limit=None, t_start=None):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_score = official_score(h)
    no_improve = 0
    max_no_improve = 50
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = 0.0
        best_i = -1
        best_j = -1
        for ii in range(len(ones)):
            i = ones[ii]
            for jj in range(len(zeros)):
                j = zeros[jj]
                h[i] = 0.0
                h[j] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                s = np.max(corr) * dx
                h[i] = 1.0
                h[j] = 0.0
                imp = best_score - s
                if imp > best_improve:
                    best_improve = imp
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        best_score -= best_improve
        no_improve = 0
    return h, best_score

def binary_sa(h0, rng, n_iters=30000, time_limit=None, t_start=None):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    T = 0.05
    T_end = 0.0001
    cool = (T_end / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        s = official_score(h)
        delta = s - best_score
        if s < best_score or rng.random() < np.exp(-delta / T):
            if s < best_score:
                best_score = s
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_score

def _lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def _lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        active = np.array([n - 1])
        w_active = np.array([1.0])
    else:
        threshold = 1e-12 * s
        active = np.where(e > threshold)[0]
        w_active = e[active] / s
    grad = np.zeros(n)
    for k, w in zip(active, w_active):
        d = k - (n - 1)
        if d == 0:
            g = 1.0 - 2.0 * x
        elif d > 0:
            g = np.zeros(n)
            g[d:] += (1.0 - x[:n - d])
            g[:n - d] += (-x[d:])
        else:
            dp = -d
            g = np.zeros(n)
            g[:n - dp] += (1.0 - x[dp:])
            g[dp:] += (-x[:n - dp])
        grad += w * g * dx
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_lse(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    h0 = make_feasible(h0)
    try:
        res = minimize(
            _lse_obj, h0, args=(beta, dx),
            jac=_lse_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def slsqp_continuation(h0, betas=(20, 50, 100, 300, 800), maxiter=150, time_limit=None, t_start=None):
    h = make_feasible(h0)
    for beta in betas:
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        h_new = slsqp_lse(h, beta, maxiter=maxiter)
        s_new = official_score(h_new)
        s_old = official_score(h)
        if s_new < s_old + 1e-10:
            h = h_new
    return h

def structured_binary(n, rng):
    h = np.zeros(n)
    target = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.zeros(n)
    for i in range(n):
        positions[i] = (i * golden) % 1.0
    order = np.argsort(positions)
    h[order[:target]] = 1.0
    return h

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h
