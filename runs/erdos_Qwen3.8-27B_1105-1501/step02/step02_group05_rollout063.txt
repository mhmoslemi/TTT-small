    alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.12, 0.06, 0.03]
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        u = None
        try:
            u = lp_step(best, rho, top_k, eta)
        except Exception:
            u = None
        if u is None:
            stall += 1
            cycle = (stall // 4) % 4
            if cycle == 0:
                eta = max(0.001, eta * 0.6)
                top_k = max(10, top_k - 4)
            elif cycle == 1:
                rho = max(0.008, rho * 0.6)
            elif cycle == 2:
                eta = min(0.02, eta * 1.8)
                top_k = min(2 * n - 1, top_k + 8)
            else:
                rho = 0.04
                eta = 0.003
                top_k = 24
            if stall > 60 and rho < 0.01:
                break
            continue
        stall = 0
        improved = False
        for alpha in alphas:
            trial = make_feasible(best + alpha * u)
            sc = score_exact(trial)
            if sc < best_sc - 1e-13:
                best = trial
                best_sc = sc
                improved = True
                break
        if improved:
            rho = min(0.12, rho * 1.05)
            eta = max(0.001, eta * 0.9)
        else:
            rho *= 0.7
            if rho < 0.006:
                rho = 0.04
                eta = 0.003
                top_k = 24
    return best, best_sc


def binary_score_fast(b):
    n = len(b)
    dx = 2.0 / n
    h = b.astype(np.float64)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def binary_sa(h_init, n_steps, T_start, cool, rng, deadline):
    n = len(h_init)
    half = n // 2
    sorted_idx = np.argsort(h_init)
    b = np.zeros(n, dtype=np.int8)
    b[sorted_idx[half:]] = 1
    cur_sc = binary_score_fast(b)
    best_b = b.copy()
    best_sc = cur_sc
    T = T_start
    pos1 = list(np.where(b == 1)[0])
    pos0 = list(np.where(b == 0)[0])
    for _ in range(n_steps):
        if time.time() >= deadline:
            break
        ii = rng.integers(len(pos1))
        jj = rng.integers(len(pos0))
        i = pos1[ii]
        j = pos0[jj]
        b[i] = 0
        b[j] = 1
        pos1[ii] = j
        pos0[jj] = i
        new_sc = binary_score_fast(b)
        delta = new_sc - cur_sc
        if delta < -1e-15 or (delta >= 1e-15 and rng.random() < np.exp(-delta / max(T, 1e-12))):
            cur_sc = new_sc
            if new_sc < best_sc - 1e-15:
                best_sc = new_sc
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
            pos1[ii] = i
            pos0[jj] = j
        T *= cool
    return best_b, best_sc


def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_x = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_x = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_x, src_x, h_src)
    return make_feasible(h_int)


def make_wave(n, f1, a1, f2, a2):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.sin(2.0 * np.pi * f1 * x) + a2 * np.cos(2.0 * np.pi * f2 * x)
    return make_feasible(h)


def two_point_local(h, n_tries, rng, deadline):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for _ in range(n_tries):
        if time.time() >= deadline:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        best_t = None
        best_local = best_sc
        for frac in [0.0, 0.25, 0.5, 0.75, 1.0, 0.382, 0.618]:
            t = t_lo + frac * span
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-12 or va > 1.0 + 1e-12 or vb < -1e-12 or vb > 1.0 + 1e-12:
                continue
            best[a] = va
            best[b] = vb
            sc = score_exact(best)
            if sc < best_local - 1e-13:
                best_local = sc
                best_t = t
        if best_t is None:
            best[a] -= 0  # no change
        else:
            best_sc = best_local
            best[a] = np.clip(best[a] + best_t - best_t, 0.0, 1.0)
    return best, best_sc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(12.0, 0.06 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_sc = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 4 and np.all(np.isfinite(parent)):
            best_h = make_feasible(parent)
            best_sc = score_exact(best_h)
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.28, 3, 0.12)
        best_sc = score_exact(best_h)

    n_base = len(best_h)

    # Phase 1: Quick LP warmup
    if time.time() < deadline:
        d1 = min(deadline, t0 + 45)
        h1, sc1 = lp_local(best_h, d1, max_iters=5000)
        if sc1 < best_sc:
            best_h, best_sc = h1, sc1

    # Phase 2: Binary annealing with multiple restarts
    if time.time() < deadline - 90:
        for restart in range(12):
            if time.time() >= deadline - 90:
                break
            if restart == 0:
                h_init = best_h
            elif restart == 1:
                h_init = best_h[::-1].copy()
            elif restart == 2:
                h_init = 1.0 - best_h
            elif restart == 3:
                h_init = make_wave(n_base, 1, 0.35, 4, 0.15)
            elif restart == 4:
                h_init = make_wave(n_base, 3, 0.2, 1, 0.3)
            else:
                h_init = make_feasible(best_h + rng.normal(0, 0.12, size=n_base))
            T_start = 0.4e-3 + 0.1e-3 * restart
            n_steps = 15000
            b, b_sc = binary_sa(h_init, n_steps, T_start, 0.998, rng, deadline - 60)
            h_bin = make_feasible(b.astype(float))
            sc_bin = score_exact(h_bin)
            if sc_bin < best_sc:
                best_h, best_sc = h_bin, sc_bin
            if time.time() < deadline - 60:
                h_pol, sc_pol = lp_local(h_bin, deadline - 60, max_iters=3000)
                if sc_pol < best_sc:
                    best_h, best_sc = h_pol, sc_pol

    # Phase 3: Main LP optimization (main workhorse)
    if time.time() < deadline - 40:
        h2, sc2 = lp_local(best_h, deadline - 40, max_iters=50000)
        if sc2 < best_sc:
            best_h, best_sc = h2, sc2

    # Phase 4: Two-point exchange refinement
    if time.time() < deadline - 20:
        h4, sc4 = two_point_local(best_h, 5000, rng, deadline - 10)
        if sc4 < best_sc:
            best_h, best_sc = h4, sc4
        # LP after two-point
        if time.time() < deadline:
            h4b, sc4b = lp_local(best_h, deadline, max_iters=10000)
            if sc4b < best_sc:
                best_h, best_sc = h4b, sc4b

    # Phase 5: Resolution scanning
    if time.time() < deadline - 15:
        offsets = [-8, -6, -4, -2, 2, 4, 6, 8]
        tried = set()
        for d in offsets:
            if time.time() >= deadline - 15:
                break
            nt = n_base + d
            if nt < 20 or nt > 120 or nt in tried:
                continue
            tried.add(nt)
            h_int = interp_to(best_h, nt)
            sc_int = score_exact(h_int)
            if sc_int < best_sc:
                best_h, best_sc = h_int, sc_int
            if time.time() < deadline - 15:
                h_pol, sc_pol = lp_local(h_int, deadline - 15, max_iters=3000)
                if sc_pol < best_sc:
                    best_h, best_sc = h_pol, sc_pol

    # Phase 6: Final LP polish
    if time.time() < deadline:
        h6, sc6 = lp_local(best_h, deadline, max_iters=50000)
        if sc6 < best_sc:
            best_h, best_sc = h6, sc6

    # Final validation and return
    final_h = make_feasible(best_h)
    final_sc = score_exact(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)):
        final_h = np.full(60, 0.5)
        final_sc = score_exact(final_h)
        final_n = 60

    return final_h, final_sc, final_n
