def make_feasible(z):

    n = len(z)

            if add <= 0: break

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    vals = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        k = p - (n - 1)
        a = abs(k)
        L = n - a
        if L <= 0:
            continue
        if k >= 0:
            vals[p] = dx * np.dot(h[k:k+L], 1.0 - h[0:L])
        else:
            vals[p] = dx * np.dot(h[0:L], 1.0 - h[a:a+L])
    return vals

def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    nrows = 2 * n - 1
    G = np.zeros((nrows, n))
    for p in range(nrows):
        k = p - (n - 1)
        a = abs(k)
        L = n - a
        if L <= 0:
            continue
        if k >= 0:
            G[p, k:k+L] += dx * (1.0 - h[0:L])
            G[p, 0:L] -= dx * h[k:k+L]
        else:
            G[p, 0:L] += dx * (1.0 - h[a:a+L])
            G[p, a:a+L] -= dx * h[0:L]
    return G

def sym_h_from_u(u):
    m = len(u)
    n = 2 * m
    h = np.empty(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

def sym_grad_v(G, m):
    n = 2 * m
    idx = np.arange(n - 1, m - 1, -1)
    return G[:, :m] - G[:, idx]

def sym_obj_func(x, beta):
    m = len(x)
    h = sym_h_from_u(x)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    mx = shifted.max()
    logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    return cmax + (logsum - 0.0) / beta

def sym_jac_func(x, beta):
    m = len(x)
    h = sym_h_from_u(x)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    mx = shifted.max()
    logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    w = np.exp(shifted - (logsum - 0.0))
    G = compute_grad_matrix(h)
    Gz = sym_grad_v(G, m)
    return w @ Gz

def sym_lbfgs_solve(u0, beta, max_iter=500):

    bounds = [(0.0, 1.0)] * m
    res = minimize(sym_obj_func, u0, method='L-BFGS-B',
                   jac=lambda x: sym_jac_func(x, beta),
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    if res.success or res.fun < sym_obj_func(u0, beta):

    return u0
