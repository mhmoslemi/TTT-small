                h[idx] += a

            else:
                a = min(-residual, h[idx] + 1e-15)
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-14:

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_gradient(h, idx, n):
    """Gradient of V_idx w.r.t. h"""
    s = idx - (n - 1)
    L = n - abs(s)
    if L <= 0:
        return np.zeros(n)
    g = np.zeros(n)
    if s >= 0:
        g[s:n] += (1.0 - h[0:L])
        g[0:L] -= h[s:n]
    else:
        t = -s
        g[0:L] += (1.0 - h[t:n])
        g[t:n] -= h[0:L]
    return g

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx] + 1e-15)
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx] + 1e-15)
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, idx, n):
    s = idx - (n - 1)
    L = n - abs(s)
    if L <= 0:
        return np.zeros(n)
    g = np.zeros(n)
    if s >= 0:
        g[s:n] += (1.0 - h[0:L])
        g[0:L] -= h[s:n]
    else:
        t = -s
        g[0:L] += (1.0 - h[t:n])
        g[t:n] -= h[0:L]
    return g

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def fast_grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def pg_step(h, beta, alpha_max=0.8, max_ls=16):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = fast_grad_weighted(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h.copy(), f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue

        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo_b = max(-h[i], -trust)
        hi_b = min(1.0 - h[i], trust)
        bounds_list.append((lo_b, hi_b))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def fourier_basis(n, n_modes):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    cols = []
    for j in range(1, n_modes + 1):
        cols.append(np.cos(np.pi * j * x))
        cols.append(np.sin(np.pi * j * x))
    B = np.array(cols).T
    B = B - B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-10] = 1.0
    B = B / norms
    return B

def pair_coord_descent(h, n_iter=200, rng=None):
    n = len(h)
    best_h = h.copy()
    best_f = official_score(h)
    h_cur = best_h.copy()
    dx = 2.0 / n
    for it in range(n_iter):
        if rng is not None:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            i = (2 * it) % n
            j = (2 * it + 1) % n
        s = h_cur[i] + h_cur[j]
        lo_v = max(0.0, s - 1.0)
        hi_v = min(1.0, s)
        if hi_v - lo_v < 1e-12:
            continue
        grid = np.linspace(lo_v, hi_v, 16)
        best_v = h_cur[i]
        best_local_f = best_f
        h_save = h_cur.copy()
        for v in grid:
            h_cur[i] = v
            h_cur[j] = s - v
            f_local = official_score(h_cur)
            if f_local < best_local_f:
                best_local_f = f_local
                best_v = v
        h_cur = h_save.copy()
        h_cur[i] = best_v
        h_cur[j] = s - best_v
        if best_local_f < best_f - 1e-14:
            best_h = h_cur.copy()
            best_f = best_local_f
    return best_h, best_f
