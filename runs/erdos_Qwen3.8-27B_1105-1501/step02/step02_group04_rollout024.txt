def transfer_h(h_src, n_dst):
    """Transfer h from source length to destination length via interpolation."""
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src, left=h_src[0], right=h_src[-1])
    h_dst = project_onto_constraint(h_dst, n_dst / 2.0)
    return h_dst

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def build_D(h):
    n = len(h)
    m = 2 * n - 1
    D = np.zeros((m, n))
    one_minus_h = 1.0 - h
    for p in range(n):
        e = np.zeros(n)
        e[p] = 1.0
        D[:, p] = np.correlate(e, one_minus_h, mode="full") - np.correlate(h, e, mode="full")
    return D

def bilinear_line_search(h, d, corr0, dx):
    lin = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
    quad = np.correlate(d, d, mode="full")
    
    # Find alpha_max
    alpha_max = 1.0
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    if np.any(pos_mask):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        alpha_max = min(alpha_max, float(np.min(-h[neg_mask] / d[neg_mask])))
    alpha_max = min(alpha_max, 1.0)
    if alpha_max < 1e-14:
        return 0.0, compute_c5(h)
    
    # Grid search
    best_alpha = 0.0
    best_val = float(np.max(corr0) * dx)
    
    # Coarse grid
    for alpha in np.linspace(0.01, alpha_max, 40):
        val = float(np.max(corr0 + alpha * lin - alpha * alpha * quad) * dx)
        if val < best_val:
            best_val = val
            best_alpha = float(alpha)
    
    # Fine grid
    if best_alpha > 0:
        span = alpha_max / 40.0
        lo = max(0.0, best_alpha - 3 * span)
        hi = min(alpha_max, best_alpha + 3 * span)
        for alpha in np.linspace(lo, hi, 30):
            if alpha < 0 or alpha > alpha_max:
                continue
            val = float(np.max(corr0 + alpha * lin - alpha * alpha * quad) * dx)
            if val < best_val:
                best_val = val
                best_alpha = float(alpha)
    
    # Final: also check alpha=0
    if best_alpha < 1e-10:
        return 0.0, best_val
    
    # Verify with exact computation
    h_new = np.clip(h + best_alpha * d, 0.0, 1.0)
    exact_val = compute_c5(h_new)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = [(-trust, trust)] * n + [(0.0, trust)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success and res.x[n] > 1e-7:
            d = res.x[:n].copy()
            d -= np.mean(d)
            nrm = np.max(np.abs(d))
            if nrm > 1e-12:
                d *= trust / nrm
                d -= np.mean(d)
            return d
    except Exception:
        pass
    return None

def pair_optimal(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(hi_v - 1.0, -hj_v)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(35):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) * 0.5

def local_search_lp(h, rng, time_limit, trust_init=0.05):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    trust = trust_init
    stall = 0
    best_c5 = compute_c5(h)
    
    while time.time() - t0 < time_limit:
        # LP direction
        D = build_D(h)
        corr0 = np.correlate(h, 1.0 - h, mode="full")
        d = lp_find_direction(h, corr0, D, trust)
        improved = False
        
        if d is not None and np.max(np.abs(d)) > 1e-10:
            alpha, new_c5 = bilinear_line_search(h, d, corr0, dx)
            if alpha > 1e-10 and new_c5 < best_c5 - 1e-12:
                h_new = np.clip(h + alpha * d, 0.0, 1.0)
                h_new = project_onto_constraint(h_new, n * 0.5)
                ec5 = compute_c5(h_new)
                if ec5 < best_c5 - 1e-12:
                    h[:] = h_new
                    best_c5 = ec5
                    stall = 0
                    trust = min(trust * 1.15, 0.15)
                    improved = True
        
        if not improved:
            # Try random directions
            found = False
            for _ in range(3):
                rd = rng.randn(n)
                rd -= np.mean(rd)
                nrm = np.linalg.norm(rd)
                if nrm < 1e-10:
                    continue
                rd *= 0.05 / nrm
                alpha, new_c5 = bilinear_line_search(h, rd, corr0, dx)
                if alpha > 1e-10 and new_c5 < best_c5 - 1e-12:
                    h_new = np.clip(h + alpha * rd, 0.0, 1.0)
                    h_new = project_onto_constraint(h_new, n * 0.5)
                    ec5 = compute_c5(h_new)
                    if ec5 < best_c5 - 1e-12:
                        h[:] = h_new
                        best_c5 = ec5
                        found = True
                        break
            if found:
                stall = 0
                continue
            
            # Pair moves
            for _ in range(100):
                if time.time() - t0 > time_limit:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                dlt = pair_optimal(h, i, j)
                if abs(dlt) > 1e-12:
                    h[i] -= dlt
                    h[j] += dlt
                    c5v = compute_c5(h)
                    if c5v < best_c5 - 1e-12:
                        best_c5 = c5v
                        stall = 0
                    else:
                        stall += 1
            
            if stall > 80:
                trust *= 0.6
                if trust < 0.01:
                    # Restart
                    h_pert = h + rng.randn(n) * 0.03
                    h_pert = project_onto_constraint(h_pert, n * 0.5)
                    for _ in range(300):
                        i = rng.randint(n)
                        j = rng.randint(n)
                        if i == j:
                            continue
                        dlt = pair_optimal(h_pert, i, j)
                        if abs(dlt) > 1e-12:
                            h_pert[i] -= dlt
                            h_pert[j] += dlt
                    cp = compute_c5(h_pert)
                    if cp < best_c5:
                        h[:] = h_pert
                        best_c5 = cp
                        trust = trust_init
                    stall = 0
    
    return h, best_c5
