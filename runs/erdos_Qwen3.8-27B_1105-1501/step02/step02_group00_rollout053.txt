import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _precompute_d(n):
    return np.arange(2 * n - 1) - (n - 1)

def _shift_sum(h, pos, d, n):
    idx1 = pos - d
    idx2 = pos + d
    result = np.zeros(len(d))
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    result[m1] += h[idx1[m1]]
    result[m2] += h[idx2[m2]]
    return result

def _pair_cd_sweep(h, n, d, max_sweeps=200, deadline=None, return_history=False):
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    no_improve = 0
    prev_best = best_c5
    improvements = []

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        if no_improve > 8:
            break

        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            for j in range(i + 1, n):
                hi_val = h[i]
                hj_val = h[j]
                if hi_val <= 1e-14 or hi_val >= 1.0 - 1e-14:
                    if hj_val <= 1e-14 or hj_val >= 1.0 - 1e-14:
                        continue
                corr0 = _corr(h)
                c5_0 = float(np.max(corr0))

                si = _shift_sum(h, i, d, n)
                sj = _shift_sum(h, j, d, n)
                s = sj - si

                t_lo = max(-hi_val + 1e-15, hj_val - 1.0 + 1e-15)
                t_hi = min(1.0 - hi_val - 1e-15, hj_val - 1e-15)

                if t_hi - t_lo < 1e-14:
                    continue

                grid = np.linspace(t_lo, t_hi, 16)
                f_vals = np.max(corr0[:, None] + s[:, None] * grid[None, :], axis=0)
                best_idx = int(np.argmin(f_vals))
                f_best = float(f_vals[best_idx])

                if f_best >= c5_0 - 1e-13:
                    continue

                idx = best_idx
                lo = grid[max(0, idx - 1)]
                hi = grid[min(len(grid) - 1, idx + 1)]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1p = hi - gr * (hi - lo)
                c2p = lo + gr * (hi - lo)
                f1 = float(np.max(corr0 + s * c1p))
                f2 = float(np.max(corr0 + s * c2p))

                for _ in range(20):
                    if hi - lo < 1e-14:
                        break
                    if f1 < f2:
                        hi = c2p
                        c2p = c1p
                        f2 = f1
                        c1p = hi - gr * (hi - lo)
                        f1 = float(np.max(corr0 + s * c1p))
                    else:
                        lo = c1p
                        c1p = c2p
                        f1 = f2
                        c2p = lo + gr * (hi - lo)
                        f2 = float(np.max(corr0 + s * c2p))

                t_opt = (lo + hi) / 2.0
                f_opt = float(np.max(corr0 + s * t_opt))
                if f_opt < f_best:
                    t_best, f_best = t_opt, f_opt
                else:
                    t_best = grid[best_idx]

                h[i] += t_best
                h[j] -= t_best

        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
            if return_history:
                improvements.append(cur_c5)
        else:
            no_improve += 1
            if return_history:
                improvements.append(cur_c5)

    return best_h, best_c5

def _lp_step(h, n, d, radius, deadline=None):
    corr0 = _corr(h)
    scores = corr0.copy()
    max_s = float(np.max(scores))
    active_thresh = max_s - 0.01

    active = np.where(scores >= active_thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > 40:
        top_idx = np.argsort(scores)[-40:]
        active = top_idx

    A_rows = []
    b_rows = []
    for r in active:
        dr = r - (n - 1)
        row = np.zeros(n)
        for i in range(n):
            val = 1.0
            i_m = i - dr
            i_p = i + dr
            if 0 <= i_m < n:
                val -= h[i_m]
            if 0 <= i_p < n:
                val -= h[i_p]
            row[i] = val
        A_rows.append(row)
        b_rows.append(-scores[r])

    if not A_rows:
        return None

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_full = np.zeros((len(A_rows), n + 1))
    A_full[:, :n] = A_ub
    A_full[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))

    try:
        res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_dir(h, p, c5_cur, n):
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)

    alphas = np.linspace(0.0, alpha_max, 20)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        if a < 1e-12:
            continue
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    lo_a = max(0.0, best_a - alpha_max * 0.1)
    hi_a = min(alpha_max, best_a + alpha_max * 0.1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)

    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _simulated_annealing(h, n, d, t_start, t_end, n_steps, deadline):
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5

    n_pairs = n * (n - 1) // 2
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        if T < 1e-10:
            T = 1e-10

        i = int(np.random.randint(0, n))
        j = int(np.random.randint(0, n))
        if i == j:
            continue

        corr0 = _corr(h)
        c5_0 = float(np.max(corr0))

        si = _shift_sum(h, i, d, n)
        sj = _shift_sum(h, j, d, n)
        s = sj - si

        t_lo = max(-h[i] + 1e-15, h[j] - 1.0 + 1e-15)
        t_hi = min(1.0 - h[i] - 1e-15, h[j] - 1e-15)
        if t_hi - t_lo < 1e-14:
            continue

        t_try = np.random.uniform(t_lo, t_hi)
        f_try = float(np.max(corr0 + s * t_try))
        delta = f_try - c5_0

        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h[i] += t_try
            h[j] -= t_try
            new_c5 = f_try
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            cur_c5 = new_c5

    return best_h, best_c5

def _binary_random(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
