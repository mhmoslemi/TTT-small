def make_feasible(z):

    target = float(n) / 2.0
    z = np.asarray(z, dtype=float).ravel().copy()
    z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(residual) < 1e-14:
                break
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add > 1e-15:
                    h[idx] += add
                    residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub > 1e-15:
                    h[idx] -= sub
                    residual += sub
    return np.clip(h, 0.0, 1.0)

    bounds.append((0.0, C))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            u = res.x[:n]
            if np.all(np.isfinite(u)) and np.max(np.abs(u)) > 1e-12:
                return u
    except Exception:
        pass
    return None

def lp_polish(h, deadline, eta0=0.02, Kmax=40, rho_seq=None):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    if rho_seq is None:
        rho_seq = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01]
    alpha_scales = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]
    stall = 0
    rho_idx = 0
    eta = eta0
    while time.time() < deadline:
        V = all_lag_values(best)
        C = V.max()
        if C <= best_sc - 1e-12:
            best_sc = C
        active = select_active(V, C, Kmax, eta)
        if len(active) < 2:
            active = np.array([np.argmax(V)])
        improved = False
        u = solve_lp_direction(best, V, C, active, rho_seq[rho_idx])
        if u is not None:
            for alpha in alpha_scales:
                trial = make_feasible(best + alpha * u)
                sc = score_exact(trial)
                if sc < best_sc - 1e-13:
                    best = trial
                    best_sc = sc
                    improved = True
                    break
        if improved:
            stall = 0
            if rho_idx < len(rho_seq) - 1 and stall < 3:
                rho_idx += 1
            eta = max(0.003, eta * 0.95)
        else:
            stall += 1
            rho_idx = max(0, rho_idx - 1)
            eta = max(0.003, eta * 1.3)
            if rho_idx == 0 and eta > 0.02:
                eta = min(0.04, eta)
            if stall > 15:
                u_rand = np.random.randn(n) * 0.05
                u_rand -= u_rand.mean()
                trial = make_feasible(best + u_rand)
                sc = score_exact(trial)
                if sc < best_sc:
                    best = trial
                    best_sc = sc
                    stall = 5
                else:
                    break
    return best, best_sc

def binary_threshold(h):
    n = len(h)
    target = n // 2
    b = (h > 0.5).astype(float)
    cur = int(b.sum())
    if cur > target:
        order = np.argsort(h)[- (cur - target):]
        for i in order:
            b[i] = 0.0
    elif cur < target:
        order = np.argsort(h)[:target - cur]
        for i in order:
            b[i] = 1.0
    return b

def binary_sa(h0, deadline, rng, max_steps=8000):
    n = len(h0)
    if n % 2 != 0:
        return None
    b = binary_threshold(h0)
    best_b = b.copy()
    best_sc = score_exact(b)
    cur_sc = best_sc
    T = 0.002
    cool = 0.999
    pos1 = np.where(b == 1.0)[0]
    pos0 = np.where(b == 0.0)[0]
    steps = 0
    while time.time() < deadline and steps < max_steps:
        i = pos1[rng.integers(len(pos1))]
        j = pos0[rng.integers(len(pos0))]
        b[i] = 0.0
        b[j] = 1.0
        pos1 = np.where(b == 1.0)[0]
        pos0 = np.where(b == 0.0)[0]
        new_sc = score_exact(b)
        delta = new_sc - cur_sc
        if delta <= 1e-14 or rng.random() < np.exp(-delta / T):
            cur_sc = new_sc
            if new_sc < best_sc - 1e-14:
                best_b = b.copy()
                best_sc = new_sc
        else:
            b[i] = 1.0
            b[j] = 0.0
            pos1 = np.where(b == 1.0)[0]
            pos0 = np.where(b == 0.0)[0]
        T *= cool
        steps += 1
    if best_sc < score_exact(h0):
        return best_b, best_sc
    return None

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def two_point_search(h, n_tries, rng, deadline):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for _ in range(n_tries):
        if time.time() >= deadline:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        best_t = None
        best_local = best_sc
        for frac in [0.0, 0.25, 0.5, 0.75, 1.0, 0.382, 0.618]:
            t = t_lo + frac * span
            tmp = best.copy()
            tmp[a] = best[a] + t
            tmp[b] = best[b] - t
            if tmp[a] < -1e-12 or tmp[a] > 1.0 + 1e-12 or tmp[b] < -1e-12 or tmp[b] > 1.0 + 1e-12:
                continue
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] += best_t
            best[b] -= best_t
            best[a] = np.clip(best[a], 0, 1)
            best[b] = np.clip(best[b], 0, 1)
            best_sc = best_local
    return best, best_sc

def make_wave(n, freq1=2, amp1=0.3, freq2=3, amp2=0.1, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 8 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60)
        best_score = score_exact(best_h)

    n = len(best_h)
    is_dead = lambda: time.time() >= deadline

    # Phase 1: Quick transformations
    transforms = [best_h.copy(), best_h[::-1].copy(), 1.0 - best_h, anti_sym_map(best_h)]
    for t in transforms:
        t = make_feasible(t)
        sc = score_exact(t)
        if sc < best_score - 1e-12:
            best_h = t
            best_score = sc

    n = len(best_h)

    # Phase 2: Main LP polishing (primary new method)
    if not is_dead():
        t_phase = min(time.time() + 0.45 * (deadline - t_start), deadline - 60)
        best_h, best_score = lp_polish(best_h, t_phase, eta0=0.025, Kmax=40)
        n = len(best_h)

    # Phase 3: More LP with different params
    if not is_dead():
        t_phase = min(time.time() + 0.25 * (deadline - t_start), deadline - 30)
        best_h, best_score = lp_polish(best_h, t_phase, eta0=0.012, Kmax=50,
                                       rho_seq=[0.06, 0.03, 0.015, 0.008])
        n = len(best_h)

    # Phase 4: Binary search for alternative basins
    if not is_dead():
        t_bin_end = min(time.time() + 120, deadline - 60)
        res = binary_sa(best_h, t_bin_end, rng, max_steps=6000)
        if res is not None:
            b_arr, b_sc = res
            # Polish the binary result continuously
            b_cont = make_feasible(b_arr)
            if not is_dead():
                b_cont, b_sc2 = lp_polish(b_cont, min(deadline - 30, time.time() + 60),
                                          eta0=0.03, Kmax=30)
            if b_sc2 < best_score - 1e-12:
                best_h = b_cont
                best_score = b_sc2

    # Phase 5: Two-point search (classic, but useful for fine tuning)
    if not is_dead():
        remaining = deadline - time.time()
        n_tries = max(2000, int(remaining * 4))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)
        n = len(best_h)

    # Phase 6: Resolution changes
    if not is_dead():
        n_base = len(best_h)
        n_opts = set()
        for dn in [-8, -4, 4, 8]:
            nt = n_base + dn
            if 24 <= nt <= 100 and nt % 2 == 0:
                n_opts.add(nt)
        for nc in [36, 44, 50, 56, 64, 72, 80]:
            if 24 <= nc <= 100:
                n_opts.add(nc)
        for n_opt in sorted(n_opts):
            if is_dead():
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int
                best_score = sc_int
            if not is_dead() and best_score < 0.381:
                h_new, sc_new = lp_polish(h_int, min(deadline - 20, time.time() + 30),
                                          eta0=0.02, Kmax=30)
                if sc_new < best_score - 1e-12:
                    best_h = h_new
                    best_score = sc_new

    # Phase 7: Final LP polish on best
    if not is_dead():
        best_h, best_score = lp_polish(best_h, deadline, eta0=0.008, Kmax=50,
                                       rho_seq=[0.03, 0.015, 0.008, 0.004])

    # Phase 8: Final two-point sweep
    if not is_dead():
        remaining = max(0, deadline - time.time())
        n_tries = max(1000, int(remaining * 5))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)) or final_h.min() < -1e-9 or final_h.max() > 1.0 + 1e-9:
        final_h = make_feasible(np.full(n, 0.5))
        final_score = score_exact(final_h)
        final_n = n

    return final_h, final_score, final_n
