```python
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        return np.sum(np.clip(z + t, 0.0, 1.0))
    lo, hi = -1.0, 1.0
    s_lo, s_hi = sum_clipped(lo), sum_clipped(hi)
    if s_lo > target:
        # all zeros at lo, need to go up... but that can't happen if z is clipped
        pass
    for _ in range(60):

            lo = mid
        else:
            hi = mid

            h[idx] += add
            residual -= add

    elif residual < -1e-12:

            if sub <= 0:
                break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def coord_descent_step(h, k, n_evals=200):
    """For coordinate k, find best value that minimizes max overlap."""
    n = len(h)
    dx = 2.0 / n
    best_val = h[k]
    best_score = official_score(h)
    
    # Evaluate at grid points
    alphas = np.linspace(0, 1, n_evals)
    scores = np.empty(n_evals)
    for i, x in enumerate(alphas):
        h[k] = x
        scores[i] = np.correlate(h, 1.0 - h, mode='full').max()
    h[k] = best_val  # restore
    
    # Find best on grid
    idx_best = np.argmin(scores)
    best_grid_val = alphas[idx_best]
    
    # Refine around best

        hi = alphas[min(n_evals - 1, idx_best + 1)]
    else:

    # Fine grid refinement
    fine_alphas = np.linspace(lo, hi, 100)
    fine_scores = np.empty(len(fine_alphas))

        h[k] = x
        fine_scores[i] = np.correlate(h, 1.0 - h, mode='full').max()
    h[k] = best_val
    
    idx_fine = np.argmin(fine_scores)
    refined_val = fine_alphas[idx_fine]
    
    if fine_scores[idx_fine] < scores[idx_best]:
        best_val = refined_val
    
    h[k] = best_val
    return h, official_score(h)

def coord_descent_full(h, max_sweeps=10, n_evals=100, order='sequential'):
    """Full coordinate descent."""
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    
    for sweep in range(max_sweeps):

        if order == 'sequential':

        elif order == 'reverse':

            if sc_new < best_score - 1e-10:
                h = h_new
                best_score = sc_new
                improved = True
        
        if not improved:
            break
```

Check: A + B*x + C*x^2 = f_{n-1} - h[k] + h[k]^2 + x - x^2
At x = h[k]: f_{n-1} - h[k] + h[k]^2 + h[k] - h[k]^2 = f_{n-1}. ✓

For s > 0 (p = n-1+s):
f_p = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
Terms with h[k]:
- If k ≤ n-1-s: h[k]*(1-h[k+s]) contributes to A (constant part is 0 since h[k] is the variable) and B = (1-h[k+s])
- If k ≥ s: h[k-s]*(1-h[k]) = h[k-s] - h[k-s]*h[k], so A += h[k-s], B += -h[k-s]

So: B_p_k = (1-h[k+s])*1_{k≤n-1-s} + (-h[k-s])*1_{k≥s}
A_p_k = f_p - h[k]*B_p_k (since f_p = A + B*h[k] + 0*h[k]^2, so A = f_p - B*h[k])
C = 0

For s < 0, t = -s (p = n-1-t):
f_p = sum_{i=0}^{n-1-t} h[i+t]*(1-h[i])
Terms with h[k]:
- If k ≥ t (i.e., k-t ≥ 0) and k-t ≤ n-1-t (i.e., k ≤ n-1): h[k]*(1-h[k-t]), B += (1-h[k-t])
- If k ≤ n-1-t (i.e., k+t ≤ n-1): h[k+t]*(1-h[k]) = h[k+t] - h[k+t]*h[k], A += h[k+t], B += -h[k+t]

So: B_p_k = (1-h[k-t])*1_{t≤k≤n-1} + (-h[k+t])*1_{k≤n-1-t}
A_p_k = f_p - h[k]*B_p_k
C = 0

Great! So for each k, I can precompute A_p and B_p for all p=0,...,2n-2, and then the objective as a function of x=h[k] is:
g(x) = max_p (A_p + B_p*x + C_p*x^2) where C_p = -1 if p=n-1 else 0.

To minimize g(x) over x ∈ [0,1]:
- g(x) is the max of (2n-2) linear functions and one concave quadratic
- I can evaluate g(x) at many points and find the minimum

But the key optimization is: for a given k, computing A and B for all p takes O(n) work (just reading h values), and then evaluating g at M points takes O(M * 2n) work. Total: O(n + M*2n) = O(n*M) per coordinate. For n=60, M=500: about 60,000 operations per coordinate, 3.6 million per full sweep. That's very fast!

```python
def precompute_coefficients(h, k):
    """Precompute A_p, B_p, C_p for all p when varying h[k]."""
    n = len(h)
    f_all = np.correlate(h, 1.0 - h, mode='full')
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    C = np.zeros(2*n-1)
    
    for p in range(2*n-1):
        s = p - (n-1)
        if s == 0:
            C[p] = -1.0
            B[p] = 1.0
        elif s > 0:
            b = 0.0
            if k <= n-1-s:
                b += (1.0 - h[k+s])
            if k >= s:
                b += (-h[k-s])
            B[p] = b
        else:
            t = -s
            b = 0.0
            if t <= k <= n-1:
                b += (1.0 - h[k-t])
            if k <= n-1-t:
                b += (-h[k+t])
            B[p] = b
        A[p] = f_all[p] - h[k] * B[p]
    
    return A, B, C
```

```python
def lp_step(h, k_active=20, radius=0.1):
    """LP-based step: minimize t subject to linearized constraints and trust region."""
    n = len(h)
    dx = 2.0 / n
    f_all = np.correlate(h, 1.0 - h, mode='full')
    f_max = f_all.max()
    
    # Select active constraints (top k by value)
    top_idx = np.argsort(f_all)[-k_active:]
    
    # Build constraint matrix
    # For each active p: f_p(h_new) ≈ f_p(h) + grad_p @ (h_new - h) ≤ t
    # => grad_p @ h_new - t ≤ f_p(h) - grad_p @ h
    # Note: f_p(h) - grad_p @ h = A_p (the constant term from precomputation)
    
    m = len(top_idx)
    # A_ub: m x (n+1) matrix, last column is for t
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for i, p in enumerate(top_idx):
        s = p - (n - 1)
        # Compute gradient of f_p at h
        grad = np.zeros(n)
        if s == 0:
            grad = 1.0 - 2.0 * h
        elif s > 0:
            for k in range(n):
                if k <= n - 1 - s:
                    grad[k] += (1.0 - h[k + s])
                if k >= s:
                    grad[k] += (-h[k - s])
        else:
            t = -s
            for k in range(n):
                if t <= k <= n - 1:
                    grad[k] += (1.0 - h[k - t])
                if k <= n - 1 - t:
                    grad[k] += (-h[k + t])
        
        A_ub[i, :n] = grad
        A_ub[i, n] = -1.0  # coefficient of t
        b_ub[i] = f_all[p] - grad @ h
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: max(0, h - radius) ≤ h_i ≤ min(1, h + radius), t ≥ 0
    bounds = []
    for k in range(n):
        lo = max(0.0, h[k] - radius)
        hi = min(1.0, h[k] + radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t ≥ 0
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    h_new = res.x[:n]
    return make_feasible(h_new)
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

    n = len(h)

    indices = range(n-1, -1, -1) if reverse else range(n)
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0:

            residual -= add
            if residual < 1e-15:

        h_lp, sc_lp = lp_optimize(best_h, max_iter=60, r_init=0.35)
        try_update(h_lp)

    # Phase 3: Coordinate descent + LP alternation
    if remaining() > 60:
        for it in range(20):
            if remaining() < 10:
                break
            h_cd = coord_sweep(best_h, n_evals=500)
            try_update(h_cd)
            h_lp, _ = lp_optimize(best_h, max_iter=30, r_init=0.25)
            try_update(h_lp)

    # Phase 4: Antisymmetric-restricted optimization
    if remaining() > 40:
        x_anti = antisym_project(best_h)
        h_anti = antisym_construct(x_anti)
        try_update(h_anti)
        # Coordinate descent on antisymmetric
        for sweep in range(30):
            if remaining() < 5:
                break
            h_new = coord_sweep(h_anti, n_evals=300)
            # Project back to antisymmetric
            x_new = antisym_project(h_new)
            h_anti = antisym_construct(x_new)
            try_update(h_anti)

    # Phase 5: LP on antisymmetric with extra constraints
    if remaining() > 40:
        h_anti_cur = antisym_construct(antisym_project(best_h))
        h_lp2, sc_lp2 = lp_optimize(h_anti_cur, max_iter=40, r_init=0.3)
        h_anti2 = antisym_construct(antisym_project(h_lp2))
        try_update(h_anti2)
        for sweep in range(15):
            if remaining() < 5:
                break
            h_new = coord_sweep(h_anti2, n_evals=300)
            h_anti2 = antisym_construct(antisym_project(h_new))
            try_update(h_anti2)

    # Phase 6: Random perturbations + coordinate descent
    if remaining() > 60:
        for i in range(20):
            if remaining() < 10:
                break
            sigma = 0.03 + 0.08 * rng.random()
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_cd = coord_sweep(h_pert, n_evals=300)
            try_update(h_cd)
            if remaining() > 15:
                h_lp3, _ = lp_optimize(h_cd, max_iter=

20, r_init=0.2)
                try_update(h_lp3)

    # Phase 7: Resolution changes
    if remaining() > 60:
        for n_try in [n - 4, n - 2, n + 2, n + 4, n - 8, n + 8]:
            if remaining() < 15 or n_try < 20:
                continue
            h_int = interp_resample(best_h, n_try)
            try_update(h_int)
            h_cd = coord_sweep(h_int, n_evals=300)
            try_update(h_cd)
            if remaining() > 10:
                h_lp4, _ = lp_optimize(h_cd, max_iter=25, r_init=0.25)
                try_update(h_lp4)

    # Phase 8: Final intensive polish
    if remaining() > 30:
        for sweep in range(30):
            if remaining() < 5:
                break
            best_h = coord_sweep(best_h, n_evals=600)
            try_update(best_h)
        if remaining() > 20:
            h_lp5, _ = lp_optimize(best_h, max_iter=50, r_init=0.2)
            try_update(h_lp5)
        if remaining() > 15:
            for sweep in range(20):
                if remaining() < 3:
                    break
                best_h = coord_sweep(best_h, n_evals=800, reverse=(sweep % 2 == 1))
                try_update(best_h)

    # Final: ensure feasibility and compute exact score
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
```
