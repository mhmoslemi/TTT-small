import numpy as np
import time
from scipy.optimize import linprog

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return grad
        j = np.arange(s, n)
        grad[j] += (1.0 - h[j - s])
        j = np.arange(0, m)
        grad[j] += -h[j + s]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return grad
        j = np.arange(0, m)
        grad[j] += (1.0 - h[j + t])
        j = np.arange(t, n)
        grad[j] += -h[j - t]
    return dx * grad

def select_active(V, C, K_max, eta):
    n_lags = len(V)
    sorted_idx = np.argsort(V)[::-1]
    top_k = sorted_idx[:min(K_max, n_lags)]
    near_max = np.where(V >= C - eta)[0]
    active = np.unique(np.concatenate([top_k, near_max]))
    return active

def lp_step(h, V, C, active, rho):
    n = len(h)
    n_active = len(active)
    if n_active == 0:
        return None
    G = np.zeros((n_active, n))
    c_vals = np.zeros(n_active)
    for idx, p in enumerate(active):
        s = int(p) - (n - 1)
        G[idx] = lag_gradient(h, s)
        c_vals[idx] = V[p]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c_vals
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_b = max(-h[i], -rho)
        hi_b = min(1.0 - h[i], rho)
        bounds.append((lo_b, hi_b))
    bounds.append((None, C))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            u = res.x[:n]
            if np.all(np.isfinite(u)) and np.linalg.norm(u) > 1e-10:
                return u
    except Exception:
        pass
    return None

def lp_refine(h, deadline, max_iters=300):
    n = len(h)
    best_h = h.copy()
    best_score = score_exact(best_h)
    rho = 0.12
    eta = 0.006
    K_max = 28
    stall = 0
    step_scales = [1.0, 0.75, 0.55, 0.4, 0.28, 0.18, 0.12, 0.07, 0.04, 0.02]
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        V = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / n)
        C = float(V.max())
        active = select_active(V, C, K_max, eta)
        direction = lp_step(best_h, V, C, active, rho)
        if direction is None:
            stall += 1
            if stall % 5 == 0:
                rho *= 0.75
                if rho < 0.01:
                    rho = 0.12
                    K_max = min(50, K_max + 4)
                    eta *= 1.4
            continue
        improved = False
        for scale in step_scales:
            trial = make_feasible(best_h + scale * direction)
            sc = score_exact(trial)
            if sc < best_score - 1e-13:
                best_h = trial
                best_score = sc
                improved = True
                break
        if improved:
            stall = 0
            rho = min(0.22, rho * 1.15)
        else:
            stall += 1
            rho *= 0.85
            if stall > 12:
                eta *= 0.7
                K_max = max(16, K_max - 3)
                rho = 0.1
                stall = 0
                if eta < 0.001:
                    break
    return best_h, best_score

def two_point_refine(h, n_tries, rng, deadline):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if time.time() >= deadline:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates = [0.0, (t_lo + t_hi) / 2.0, t_lo + 0.25 * span,
                      t_lo + 0.75 * span, t_lo + 0.382 * span, t_lo + 0.618 * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = best[a] + best_t
            best[b] = best[b] - best_t
            best[a] = np.clip(best[a], 0.0, 1.0)
            best[b] = np.clip(best[b], 0.0, 1.0)
            best_sc = score_exact(best)
    return best, best_sc

def balance_bits(h):
    n = len(h)
    target = n // 2
    b = (h > 0.5).astype(int)
    cur = int(np.sum(b))
    if cur == target:
        return b
    if cur > target:
        excess = cur - target
        ones = np.where(b == 1)[0]
        vals = h[ones]
        order = np.argsort(vals)
        for i in range(excess):
            b[ones[order[i]]] = 0
    else:
        deficit = target - cur
        zeros = np.where(b == 0)[0]
        vals = h[zeros]
        order = np.argsort(vals)[::-1]
        for i in range(deficit):
            b[zeros[order[i]]] = 1
    return b

def binary_anneal(h, rng, deadline, n_iter=5000):
    n = len(h)
    if n % 2 != 0:
        return None
    b = balance_bits(h)
    cur_score = score_exact(b.astype(float))
    best_b = b.copy()
    best_score = cur_score
    pos1 = list(np.where(b == 1)[0])
    pos0 = list(np.where(b == 0)[0])
    T = 0.002
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        i = rng.integers(0, len(pos1))
        j = rng.integers(0, len(pos0))
        a, c = pos1[i], pos0[j]
        b[a] = 0
        b[c] = 1
        pos1[i] = c
        pos0[j] = a
        sc = score_exact(b.astype(float))
        delta = sc - cur_score
        if delta <= 1e-14 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cur_score = sc
            if sc < best_score - 1e-14:
                best_b = b.copy()
                best_score = sc
        else:
            b[a] = 1
            b[c] = 0
            pos1[i] = a
            pos0[j] = c
        T *= 0.9995
    if best_b.size == 0:
        return None
    return best_b.astype(float), best_score

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def make_wave_init(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    f1 = rng.integers(2, 5)
    f2 = rng.integers(5, 9)
    a1 = rng.uniform(0.2, 0.4)
    a2 = rng.uniform(0.05, 0.2)
    ph = rng.uniform(0, 2 * np.pi)
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + ph) + a2 * np.cos(2 * np.pi * f2 * x)
    return make_feasible(h)

def subgradient_polish(h, deadline, n_iter=200):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    alpha = 0.02
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        dx = 2.0 / n
        vals = np.correlate(best, 1.0 - best, mode='full') * dx
        cmax = vals.max()
        tol = 0.04
        active = np.where(vals >= cmax - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        w = np.exp(30.0 * (vals[active] - cmax))
        w /= w.sum()
        grad = np.zeros(n)
        for k, p in enumerate(active):
            s = int(p) - (n - 1)
            grad += w[k] * lag_gradient(best, s)
        d = -grad
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25]:
            h_trial = make_feasible(best + mult * d)
            sc = score_exact(h_trial)
            if sc < best_sc - 1e-14:
                best = h_trial
                best_sc = sc
                improved = True
                break
        if improved:
            alpha = min(0.05, alpha * 1.1)
        else:
            alpha *= 0.7
            if alpha < 1e-7:
                break
    return best, best_sc
