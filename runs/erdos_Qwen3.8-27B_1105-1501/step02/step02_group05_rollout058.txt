    bounds.append((C - 2.0 * rho, C))
    try:

def two_point_exchange(h, n_tries, rng, deadline):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for _ in range(n_tries):
        if time.time() >= deadline:
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        cands = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                 t_lo + 0.25 * span, t_lo + 0.75 * span,
                 t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in cands:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:

def balance_binary(h_float, rng):
    n = len(h_float)
    target = n // 2
    b = (h_float > 0.5).astype(float)
    cur = int(np.sum(b))
    if cur > target:
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        to_remove = rng.choice(ones, size=cur - target, replace=False)
        b[to_remove] = 0.0
    elif cur < target:
        zeros = np.where(b == 0)[0]
        to_add = rng.choice(zeros, size=target - cur, replace=False)
        b[to_add] = 1.0
    return b

            b[pj] = 0.0

    if time.time() < deadline:
        d2 = min(deadline, time.time() + budget_s * 0.15)
        h2, s2 = two_point_exchange(best_h, 5000, rng, d2)
        if s2 < best_score:
            best_h, best_score = h2, s2

    # Phase 3: Second LP round from improved point
    if time.time() < deadline:
        d3 = min(deadline, time.time() + budget_s * 0.20)
        h3, s3 = lp_local_search(best_h, d3, eta0=0.006, kmax=50, rho0=0.15)
        if s3 < best_score:
            best_h, best_score = h3, s3

    # Phase 4: Binary SA for basin jumping
    if time.time() < deadline:
        d4 = min(deadline, time.time() + budget_s * 0.10)
        for trial in range(3):
            if time.time() >= d4:
                break
            start = best_h + rng.normal(0, 0.1, size=n)
            start = make_feasible(start)
            hb, sb = binary_sa(start, 3000, rng, d4)
            if sb < best_score:
                best_h, best_score = hb, sb
            # Also try LP polishing of binary result
            if sb < best_score + 0.01 and time.time() < d4:
                d4b = min(deadline, time.time() + 30)
                hlp, slp = lp_local_search(hb, d4b, eta0=0.01, kmax=35, rho0=0.2)
                if slp < best_score:
                    best_h, best_score = hlp, slp

    # Phase 5: Anti-symmetric transform + polish
    if time.time() < deadline:
        h_sym = anti_sym(best_h)
        sc_sym = official_score(h_sym)
        if sc_sym < best_score:
            best_h, best_score = h_sym, sc_sym
        if time.time() < deadline:
            d5 = min(deadline, time.time() + budget_s * 0.08)
            h5, s5 = lp_local_search(h_sym, d5, eta0=0.008, kmax=35, rho0=0.2)
            if s5 < best_score:
                best_h, best_score = h5, s5

    # Phase 6: Resolution changes
    if time.time() < deadline:
        d6 = min(deadline, time.time() + budget_s * 0.08)
        n_base = len(best_h)
        n_opts = sorted(set([n_base - 8, n_base - 4, n_base + 4, n_base + 8,
                             n_base - 12, n_base + 12, 40, 48, 56, 64, 72]))
        n_opts = [x for x in n_opts if 20 <= x <= 100 and x != n_base]
        for nt in n_opts:
            if time.time() >= d6:
                break
            h_int = interp_to(best_h, nt)
            sc_int = official_score(h_int)
            if sc_int < best_score:
                best_h, best_score = h_int, sc_int
            if time.time() < d6:
                d6b = min(d6, time.time() + 20)
                h_new, sc_new = lp_local_search(h_int, d6b, eta0=0.01, kmax=30, rho0=0.25)
                if sc_new < best_score:
                    best_h, best_score = h_new, sc_new

    # Phase 7: Final intensive polish
    if time.time() < deadline:
        d7 = deadline
        h7, s7 = lp_local_search(best_h, d7, eta0=0.004, kmax=45, rho0=0.1)
        if s7 < best_score:
            best_h, best_score = h7, s7

    # Final: two-point sweep on best
    if time.time() < deadline - 5:
        d_final = deadline
        h_f, s_f = two_point_exchange(best_h, 3000, rng, d_final)
        if s_f < best_score:
            best_h, best_score = h_f, s_f

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    if not np.all(np.isfinite(final_h)):
        final_h = np.full(60, 0.5)
        final_score = official_score(final_h)
        final_n = 60
    final_h = np.clip(final_h, 0.0, 1.0)
    final_score = official_score(final_h)
    return final_h, final_score, final_n
