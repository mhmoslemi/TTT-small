def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target_sum):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target_sum - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _lse_grad(h, beta):

    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    exp_s = np.exp(beta * (scores - max_s))
    p = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        w = p[r]
        if w < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            grad[:m] += w * (1.0 - h[d:d + m])
            grad[d:] += -w * h[:m]
        else:
            g = -d
            m = n - g
            if m <= 0:

            grad[g:] += w * (1.0 - h[:m])
            grad[:m] += -w * h[g:]
    return grad

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    lo_a = max(0.0, best_a - alpha_max * 0.1)

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _gradient_flow(h, deadline, beta=50, lr=0.002, max_iters=200000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = float(_exact_c5(h))
    momentum = np.zeros(n)
    mu = 0.9
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = _lse_grad(h, beta)
        momentum = mu * momentum - lr * g
        h_new = h + momentum
        h_new = _project(h_new, target)
        c5 = float(_exact_c5(h_new))
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = float(_exact_c5(h))
    best_n = n0

    # Phase 1: Multi-beta Adam (30% budget)
    deadline1 = t0 + budget_s * 0.30
    h1, c5_1 = _multi_beta(h, deadline1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient flow with medium beta (10% budget)
    deadline2 = t0 + budget_s * 0.40
    if time.time() < deadline2:
        h2, c5_2 = _gradient_flow(best_h, deadline2, beta=80, lr=0.001, max_iters=200000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: LP refinement (20% budget)
    deadline3 = t0 + budget_s * 0.60
    h3, c5_3 = _lp_loop(best_h, deadline3, radius_init=0.12, max_iters=5000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Restarts with perturbations + LP (15% budget)
    deadline4 = t0 + budget_s * 0.75
    scale = 0.03
    for attempt in range(200):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        h_try, c5_try = _lp_loop(h_pert, min(deadline4, time.time() + budget_s * 0.02),
                                  radius_init=0.08, max_iters=600)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.25, scale * 1.2)

    # Phase 5: Multi-resolution sweep (12% budget)
    deadline5 = t0 + budget_s * 0.87
    n_best = len(best_h)
    try_ns = [n_best + dn for dn in [1, -1, 2, -2, 3, -3, 4, -4, 6, -6, 8, -8]]
    for n_try in try_ns:
        if time.time() > deadline5:
            break
        if n_try < 24 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _multi_beta(h_r, min(deadline5, time.time() + budget_s * 0.025))
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 6: Coordinate descent polish (8% budget)
    deadline6 = t0 + budget_s * 0.95
    h6, c5_6 = _coord_descent(best_h, deadline6, max_passes=1000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.995,
                              radius_init=0.04, max_iters=2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = float(_exact_c5(best_h))
    return best_h, final_c5, len(best_h)
