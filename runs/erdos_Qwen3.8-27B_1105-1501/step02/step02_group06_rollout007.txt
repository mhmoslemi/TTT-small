import numpy as np
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(vals.max())

def compute_f_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return compute_f_raw(h) * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -2.0, 3.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-13:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
            if residual <= 1e-14:
                break
    elif residual < -1e-13:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if -residual <= 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def gradient_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] += (-h[i])
    else:
        t = -s
        for i in range(t, n):
            g[i] += (1.0 - h[i - t])
            g[i - t] += (-h[i])
    return g

def subgradient_at(h, weights):
    n = len(h)
    g = np.zeros(n)
    for p, w in enumerate(weights):
        s = p - (n - 1)
        g += w * gradient_shift(h, s)
    return g

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym(h):
    return make_feasible(1.0 - h[::-1].copy())

def qp_active_step(h0, k_active, reg, radius):
    n = len(h0)
    vals = compute_f(h0)
    n_val = len(vals)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        h_var >= h0 - radius,
        h_var <= h0 + radius
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = dx * gradient_shift(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)

    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)

    h_new = None
    for solver in [cp.CLARABEL, cp.ECOS, cp.OSQP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                break
        except Exception:
            continue

    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    h_new = make_feasible(h_new)
    return h_new

def qp_optimize(h0, max_iter, k_active, reg_start, reg_end, radius_start, radius_end):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    stall = 0

    for it in range(max_iter):
        if it == max_iter - 1:
            t = 1.0
        else:
            t = it / max_iter
        reg = reg_start * (reg_end / reg_start) ** t
        radius = radius_start * (radius_end / radius_start) ** t

        h_new = qp_active_step(h, k_active, reg, radius)
        if h_new is None:
            stall += 1
            if stall >= 6:
                break
            continue

        sc = official_score(h_new)
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.8, 0.6, 0.4, 0.2]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-14:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 8:
                    break
    return h, best

def steepest_descent_refine(h0, n_iters, k_active, step_decay):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    dx = 2.0 / n

    for it in range(n_iters):
        vals = compute_f(h)
        v_max = vals.max()
        raw = compute_f_raw(h)
        n_val = len(raw)

        weights = np.zeros(n_val)
        for p in range(n_val):
            s = p - (n - 1)
            diff = v_max - vals[p]
            if diff < 0.02:
                weights[p] = np.exp(-diff / max(0.005, 0.02 * dx))
        w_sum = weights.sum()
        if w_sum < 1e-10:
            break
        weights /= w_sum

        g = subgradient_at(h, weights)
        d = -g + np.mean(g)

        alpha_max = np.inf
        for i in range(n):
            if d[i] > 1e-10:
                alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
            elif d[i] < -1e-10:
                alpha_max = min(alpha_max, h[i] / (-d[i]))
        alpha_max = max(0.0, min(alpha_max, 0.3))

        if alpha_max < 1e-12:
            break

        step = 0.1 / (it * step_decay + 1.0)
        alpha_try = min(step, alpha_max)

        best_alpha = 0.0
        best_sc = best
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15]:
            a = alpha_try * frac
            h_try = make_feasible(h + a * d)
            sc_try = official_score(h_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_alpha = a
        if best_alpha > 0:
            h = make_feasible(h + best_alpha * d)
            best = best_sc

    return h, best

def paired_cd(h0, n_iters, rng, n_candidates=5):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    raw = compute_f_raw(h)
    n_val = len(raw)

    for it in range(n_iters):
        vals = raw * (2.0 / n)
        v_max = vals.max()

        weights = np.zeros(n_val)
        for p in range(n_val):
            diff = v_max - vals[p]
            if diff < 0.03:
                weights[p] = np.exp(-diff / max(0.003, 0.015 * (2.0 / n)))
        w_sum = weights.sum()
        if w_sum < 1e-10:
            break
        weights /= w_sum

        g = subgradient_at(h, weights)
        d = -g + np.mean(g)

        pos_idx = np.argsort(d)[-n_candidates:]
        neg_idx = np.argsort(d)[:n_candidates]

        best_pair_score = best
        best_pair_move = None

        for ii in pos_idx:
            for jj in neg_idx:
                if ii == jj:
                    continue
                t_lo = max(-h[ii], h[jj] - 1.0)
                t_hi = min(1.0 - h[ii], h[jj])
                if t_hi - t_lo < 1e-12:
                    continue

                coeffs = np.zeros(n_val)
                for p in range(n_val):
                    s = p - (n - 1)
                    gs = gradient_shift(h, s)
                    coeffs[p] = gs[ii] - gs[jj]

                a, b = t_lo, t_hi
                for _ in range(40):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    if np.max(raw + m1 * coeffs) < np.max(raw + m2 * coeffs):
                        b = m2
                    else:
                        a = m1
                t_opt = (a + b) / 2.0

                sc = float(np.max(raw + t_opt * coeffs) * (2.0 / n))
                if sc < best_pair_score - 1e-14:
                    best_pair_score = sc
                    best_pair_move = (ii, jj, t_opt)

        if best_pair_move is not None:
            ii, jj, t_opt = best_pair_move
            h[ii] += t_opt
            h[jj] -= t_opt
            h = np.clip(h, 0.0, 1.0)
            if abs(np.sum(h) - n / 2.0) > 1e-10:
                h = make_feasible(h)
            best = best_pair_score
            raw = compute_f_raw(h)

    return h, best

def random_perturb_optimize(h0, rng, n_trials, sigma_min, sigma_max):
    n = len(h0)
    best_h = h0.copy()
    best_sc = official_score(h0)

    for i in range(n_trials):
        sigma = sigma_min + (sigma_max - sigma_min) * rng.random()
        h_try = make_feasible(h0 + rng.normal(0, sigma, size=n))
        sc = official_score(h_try)
        if sc < best_sc - 1e-14:
            best_h = h_try.copy()
            best_sc = sc
    return best_h, best_sc
