def make_feasible(h):
    """Project h onto [0,1]^n with sum = n/2"""

        s = np.sum(h)
        residual = target - s

            break
        if residual > 0:
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            h[mask] += residual / np.sum(mask)
        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            h[mask] += residual / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)

    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:

    """Compute gradient of all correlation values w.r.t. y for anti-symmetric h."""
    m = n // 2
    h = build_symmetric_h(y, n)
    dx = 2.0 / n
    K = 2 * n - 1

        s = p - (n - 1)
        if s >= 0:
            L = n - s
            Gh[p, :L] += dx * (1.0 - h[s:])
            Gh[p, s:] -= dx * h[:L]
        else:
            t = -s
            K2 = n - t
            Gh[p, t:] += dx * (1.0 - h[:K2])
            Gh[p, :K2] -= dx * h[t:]
    
    # Chain rule: h[i] = y[i] for i < m, h[i] = 1 - y[n-1-i] for i >= m
    Gy = np.zeros((K, m))
    Gy[:, :m] = Gh[:, :m]  # dh[i]/dy[i] = 1 for i < m
    # For i >= m: h[i] = 1 - y[n-1-i], so dh[i]/dy[j] = -1 if j = n-1-i
