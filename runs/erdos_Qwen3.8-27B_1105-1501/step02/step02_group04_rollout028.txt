import numpy as np

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    # Bisection for the threshold that makes sum(clip(h - t, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    z = np.exp((corr - cmax) / T)
    val = cmax + T * np.log(np.sum(z))
    return dx * val

def smooth_gradient(h, n, T):

    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    z = np.exp((corr - cmax) / T)
    w = z / np.sum(z)

            jp = i + s
            if 0 <= jp < n:
                val -= h[jp]
            grad[i] += w_s * val

        i_hi2 = min(n-1, n-1-s)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2+1)
            grad[idx] -= w_s * h[idx + s]
    
    grad *= dx
    return grad

            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
    return h


def pair_sweep_ordered(h, rng, n_max_pairs):
    n = len(h)
    total = n * (n - 1) // 2
    order = rng.permutation(total)
    count = 0
    for idx in order:
        if count >= n_max_pairs:
            break
        i = idx // (n - 1)
        j = (idx % (n - 1)) + i + 1
        delta = _pair_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
        count += 1
    return h


def interpolate_to_length(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_onto_constraint(h_dst, n_dst / 2.0)


def make_random_vector(n, rng):
    h = rng.rand(n)
    return project_onto_constraint(h, n / 2.0)


def make_sinusoidal(n, freq, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.25 * np.sin(np.pi * freq * x + phase) + 0.1 * np.sin(np.pi * 2 * freq * x + rng.uniform(0, 6.28))
    return project_onto_constraint(h, n / 2.0)


def make_binary_balanced(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return project_onto_constraint(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3.0

    # Initialize from parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        best_h = np.full(70, 0.5)

    n = len(best_h)
    best_h = project_onto_constraint(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Multi-start candidates
    candidates = [best_h.copy()]
    for _ in range(4):
        candidates.append(make_sinusoidal(n, 2 + int(rng.rand() * 4), rng))
    for _ in range(3):
        candidates.append(make_binary_balanced(n, rng))
    candidates.append(make_random_vector(n, rng))

    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c5:
            best_c5 = c
            best_h = cand.copy()

    # T schedules in raw correlation units
    T_coarse = [10.0, 5.0, 2.5, 1.2]
    T_mid = [0.7, 0.4, 0.2]
    T_fine = [0.1, 0.05, 0.02, 0.01]

    # Phase 1: Annealed SLSQP on best candidate (~30% budget)
    phase1_end = t0 + budget_s * 0.30
    if time.time() < phase1_end:
        h1 = best_h.copy()
        h1, c1 = anneal_slsqp(h1, n, T_coarse + T_mid, maxiter_per_T=50)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1
    if time.time() < phase1_end:
        h1 = best_h.copy()
        h1, c1 = anneal_slsqp(h1, n, T_mid + T_fine, maxiter_per_T=60)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    # Phase 2: Basin hopping with SLSQP polishing (~40% budget)
    phase2_end = t0 + budget_s * 0.72
    cur_h = best_h.copy()
    cur_c5 = best_c5
    hop_iter = 0
    while time.time() < phase2_end:
        hop_iter += 1
        # Adaptive perturbation scale
        if hop_iter % 7 == 0:
            sigma = 0.12 + 0.12 * rng.rand()
        elif hop_iter % 3 == 0:
            sigma = 0.04 + 0.04 * rng.rand()
        else:
            sigma = 0.015 + 0.025 * rng.rand()

        h_pert = cur_h + rng.randn(len(cur_h)) * sigma
        h_pert = project_onto_constraint(h_pert, n / 2.0)

        # Short annealing
        T_hop = [0.4, 0.15, 0.06, 0.02, 0.008]
        h_new, c5_new = anneal_slsqp(h_pert, n, T_hop, maxiter_per_T=25)

        # Acceptance: strict improvement on best, or simulated annealing on current
        if c5_new < cur_c5:
            cur_h = h_new
            cur_c5 = c5_new
        elif c5_new < cur_c5 + 0.003 and rng.rand() < np.exp(-(c5_new - cur_c5) / 0.001):
            cur_h = h_new
            cur_c5 = c5_new

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            cur_h = h_new.copy()
            cur_c5 = c5_new

        # Periodic pair polish on best
        if hop_iter % 20 == 0 and time.time() < phase2_end:
            hp = best_h.copy()
            hp = pair_sweep_random(hp, rng, 150)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp

    # Phase 3: Try neighboring lengths (~15% budget)
    phase3_end = t0 + budget_s * 0.88
    test_lengths = []
    for dn in [-14, -7, 7, 14, -20, 20]:
        tn = n + dn
        if tn >= 30 and tn <= 160 and tn % 2 == 0:
            test_lengths.append(tn)
    # Remove duplicates, keep unique
    seen = set()
    unique_lengths = []
    for tl in test_lengths:
        if tl not in seen:
            seen.add(tl)
            unique_lengths.append(tl)

    for tl in unique_lengths:
        if time.time() > phase3_end:
            break
        h_test = interpolate_to_length(best_h, tl)
        h_test, c_test = anneal_slsqp(h_test, tl, T_mid + T_fine[:3], maxiter_per_T=35)
        if c_test < best_c5:
            best_c5 = c_test
            best_h = h_test
            n = tl

    # Phase 4: Final intensive pair polish + micro-SLSQP
    phase4_end = deadline
    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    n = len(best_h)
    while time.time() < phase4_end:
        # Random pair sweep
        hp = best_h.copy()
        hp = pair_sweep_random(hp, rng, 300)
        cp = compute_c5(hp)
        if cp < best_c5 - 1e-10:
            best_c5 = cp
            best_h = hp

        # Micro SLSQP at very fine T
        if time.time() < phase4_end - 5:
            hq = best_h.copy()
            hq, cq = anneal_slsqp(hq, n, [0.03, 0.01, 0.005], maxiter_per_T=40)
            if cq < best_c5 - 1e-10:
                best_c5 = cq
                best_h = hq

        # Check if we're stalling
        if time.time() > phase4_end - 10:
            break

    # Final projection and evaluation
    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
