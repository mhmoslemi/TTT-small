    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, -2.0, 3.0)
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-13:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]

            else:
                add = max(res, h[i] - 1.0)

                           constraints=[
                               {'type': 'ineq', 'fun': _as_ieq, 'jac': _as_ieq_jac},
                               {'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}
                           ],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                new_h = _project(res['x'][:-1])
                new_s = _score(new_h)
                if new_s <= cur_s + 1e-10:
                    old_count = len(active_idx)
                    cur = new_h
                    cur_s = new_s
                    cv_new = _corr_vec(cur)
                    max_new = cv_new.max()
                    active_new = np.where(cv_new >= max_new - 0.003)[0]
                    if max_new < max_cv - 1e-6 and len(active_new) <= max_active:
                        if set(active_new.tolist()) == set(active_idx.tolist()):
                            break
                else:
                    break
        except Exception:
            break
    return cur, cur_s

def _fista(h0, beta, n_iter, alpha=None):
    n = len(h0)
    if alpha is None:
        alpha = 0.8 / (beta * 2.0 / n * 4.0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t_val = 1.0
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        if it > 0:
            y = h_cur + ((t_val - 1.0) / t_new) * (h_cur - h_prev)
            y = _project(y)
        t_val = t_new
        h_new = _project(y - alpha * grad)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_cur - h_avg) / (it + 1)
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s

def _chain(h0, deadline):
    cur = h0.copy()
    cur_s = _score(cur)
    if time.time() < deadline - 12:
        cur = _slsqp_lse(cur, beta=40, maxiter=150)
        ns = _score(cur)
        if ns < cur_s:
            cur_s = ns
    if time.time() < deadline - 12:
        cur = _slsqp_lse(cur, beta=200, maxiter=200)
        ns = _score(cur)
        if ns < cur_s:
            cur_s = ns
    if time.time() < deadline - 10:
        cur, cur_s = _fista(cur, beta=300, n_iter=8000, alpha=0.004)
    if time.time() < deadline - 10:
        cur = _slsqp_lse(cur, beta=1000, maxiter=200)
        ns = _score(cur)
        if ns < cur_s:
            cur_s = ns
    if time.time() < deadline - 8:
        new, ns = _as_solve(cur, max_active=10, max_outer=6, maxiter=200, time_limit=deadline - 5)
        if ns < cur_s:
            cur, cur_s = new, ns
    if time.time() < deadline - 5:
        cur, cur_s = _fista(cur, beta=500, n_iter=5000, alpha=0.003)
    return cur, cur_s

def _make_binary_pattern(n, mode, rng):
    h = np.zeros(n)
    if mode == 0:
        h[::2] = 1.0
    elif mode == 1:
        h[:n // 2] = 1.0
    elif mode == 2:
        idx = np.sort(rng.choice(n, n // 2, replace=False))
        h[idx] = 1.0
    elif mode == 3:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        positions = (np.floor(np.arange(n // 2) * phi) % n).astype(int)
        h[positions] = 1.0
    elif mode == 4:
        for i in range(n):
            if bin(i).count('1') % 2 == 0:
                h[i] = 1.0
        ones = int(np.sum(h))
        if ones > n // 2:
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            remove = oi[:ones - n // 2]
            h[remove] = 0
        elif ones < n // 2:
            zi = np.where(h == 0)[0]
            add = zi[:n // 2 - ones]
            h[add] = 1
    return h

def _binary_sa(h0, n_iter, t_start, t_end, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    cur_s = best_s
    for it in range(n_iter):
        t = t_start * (t_end / max(t_start, 1e-10)) ** (it / max(1, n_iter - 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        new_s = _score(h)
        if new_s < cur_s or (t > 1e-10 and rng.random() < np.exp(-(new_s - cur_s) / t)):
            cur_s = new_s
            if new_s < best_s:
                best_s = new_s
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = 30.0
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(56) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(parent)

    # Phase 1: Screen n values with quick LSE
    n_range = []
    for n in range(max(48, best_n - 12), min(72, best_n + 14) + 1):
        n_range.append(n)

    phase1_best = []
    for n in n_range:
        if time.time() > deadline - 120:
            break
        h_interp = _interp(parent, n)
        s0 = _score(h_interp)
        if time.time() < deadline - 80:
            h_opt = _slsqp_lse(h_interp, beta=80, maxiter=100)
            s1 = _score(h_opt)
            if s1 < s0:
                s0 = s1
                h_interp = h_opt
        phase1_best.append((s0, n, h_interp.copy()))
        if s0 < best_score - 1e-10:
            best_h = h_interp.copy()
            best_score = s0
            best_n = n

    # Phase 2: Try binary patterns for even n (gives diverse starting points)
    even_ns = [n for n in n_range if n % 2 == 0]
    for n in even_ns[:6]:
        if time.time() > deadline - 80:
            break
        for mode in [0, 3, 4]:
            h_bin = _make_binary_pattern(n, mode, rng)
            h_smooth = _project(h_bin + 0.1 * rng.standard_normal(n))
            s = _score(h_smooth)
            if s < best_score - 1e-4:
                h_opt = _slsqp_lse(h_smooth, beta=100, maxiter=100)
                s_opt = _score(h_opt)
                if s_opt < best_score - 1e-10:
                    best_h = h_opt
                    best_score = s_opt
                    best_n = n

    # Phase 3: Focused multi-start optimization at best n and neighbors
    top_starts = [best_h.copy(), _mirror(best_h)]
    for sigma in [0.03, 0.06, 0.12]:
        top_starts.append(_project(best_h + sigma * rng.standard_normal(best_n)))
    # Also try at nearby n values
    for n_delta in [-2, 2]:
        n_try = best_n + n_delta
        if 48 <= n_try <= 72:
            h_near = _interp(best_h, n_try)
            top_starts.append(h_near)

    for h0 in top_starts:
        if time.time() > deadline - 40:
            break
        cur, cur_s = _chain(h0, deadline)
        if cur_s < best_score - 1e-10:
            best_h = cur.copy()
            best_score = cur_s
            best_n = len(cur)

    # Phase 4: Intensive active-set refinement
    remaining = deadline - time.time()
    if remaining > 40:
        for ma in [8, 12, 16]:
            if time.time() > deadline - 15:
                break
            new, ns = _as_solve(best_h, max_active=ma, max_outer=8, maxiter=250,
                                time_limit=deadline - 10)
            if ns < best_score - 1e-10:
                best_h = new
                best_score = ns

    # Phase 5
