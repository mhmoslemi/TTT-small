import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(h, p):
    """Compute gradient of V[p] = dx * correlate(h, 1-h)[p] w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        # V = sum_{j=0}^{m-1} h[j+s]*(1-h[j])
        # d/dh[i]: (1-h[i-s]) if s<=i<=n-1, plus -h[i+s] if 0<=i<=n-1-s
        # First part: i from s to n-1, j = i-s, contribution (1-h[i-s])
        idx1 = np.arange(s, s + m)  # i = s, s+1, ..., s+m-1 = n-1
        grad[idx1] += (1.0 - h[np.arange(m)])
        # Second part: i from 0 to m-1, contribution -h[i+s]
        idx2 = np.arange(m)
        grad[idx2] -= h[idx2 + s]
    else:
        t = -s
        m = n - t
        # V = sum_{j=0}^{m-1} h[j]*(1-h[j+t])
        # d/dh[i]: (1-h[i+t]) if 0<=i<=n-1-t, plus -h[i-t] if t<=i<=n-1
        idx1 = np.arange(m)  # i = 0, ..., m-1
        grad[idx1] += (1.0 - h[idx1 + t])
        idx2 = np.arange(t, t + m)  # i = t, ..., t+m-1 = n-1
        grad[idx2] -= h[idx2 - t]
    return dx * grad

def lp_step(h, C, active_p, rho, with_penalty=0.0):
    """Solve LP to find direction u minimizing linearized max over active lags."""
    n = len(h)
    V = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    
    # Build gradient matrix
    G = np.zeros((len(active_p), n))
    c = np.zeros(len(active_p))
    for i, p in enumerate(active_p):
        if 0 <= p <= 2*n-2:
            G[i] = grad_for_lag(h, p)
            c[i] = V[p]
    
    if with_penalty > 0:
        # Add L1 penalty: min t + lam * sum(v) s.t. -v <= u <= v
        nv = n + 1  # u (n vars) + t (1 var)
        nl1 = n  # v variables
        nvars = nv + nl1
        # Objective: [0,...,0, 1, lam,...,lam]
        c_obj = np.zeros(nvars)
        c_obj[n] = 1.0  # t
        c_obj[n+1:] = with_penalty  # v_i
        # Active constraints: G u - t <= -c  =>  [G, -1, 0] x <= -c
        A_ub = np.zeros((len(active_p)+(2+n), nvars))
        b_ub = np.zeros(len(active_p)+(2+n))
        A_ub[:len(active_p), :n] = G
        A_ub[:len(active_p), n] = -1.0
        b_ub[:len(active_p)] = -c
        # L1 constraints: u_i - v_i <= 0
        A_ub[len(active_p):len(active_p)+n, 0:n] = np.eye(n)
        A_ub[len(active_p):len(active_p)+n, n+1:] = -np.eye(n)
        b_ub[len(active_p):len(active_p)+n] = 0.0
        # -u_i - v_i <= 0
        A_ub[len(active_p)+n:len(active_p)+2*n, 0:n] = -np.eye(n)
        A_ub[len(active_p)+n:len(active_p)+2*n, n+1:] = -np.eye(n)
        b_ub[len(active_p)+n:len(active_p)+2*n] = 0.0
        # Equality: sum(u) = 0
        A_eq = np.zeros((1, nvars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        # Bounds
        bounds = []
        for i in range(n):
            lo_i = max(-h[i], -rho)
            hi_i = min(1.0 - h[i], rho)
            bounds.append((lo_i, hi_i))
        bounds.append((0.0, C))  # t
        bounds.extend([(0.0, rho)] * n)  # v_i
    else:
        nvars = n + 1
        c_obj = np.zeros(nvars)
        c_obj[n] = 1.0  # minimize t
        A_ub = np.zeros((len(active_p), nvars))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -c
        A_eq = np.zeros((1, nvars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = []
        for i in range(n):
            lo_i = max(-h[i], -rho)
            hi_i = min(1.0 - h[i], rho)
            bounds.append((lo_i, hi_i))
        bounds.append((0.0, C + 0.01))  # t bounded above
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            u = res.x[:n]
            if np.all(np.isfinite(u)) and np.max(np.abs(u)) > 1e-12:
                return u
    except Exception:
        pass
    return None

def select_active(V, top_k=24, eta=0.005):

def binary_sa(h0, n_steps, t_deadline, rng, top_k=10):
    """Binary simulated annealing starting from threshold of h0."""
    n = len(h0)
    if n % 2 != 0:
        return None, None
    
    # Initialize from thresholding
    h_sorted_idx = np.argsort(h0)[::-1]
    bits = np.zeros(n, dtype=int)
    bits[h_sorted_idx[:n//2]] = 1
    
    cur_score = official_score(bits.astype(float))
    best_bits = bits.copy()
    best_score = cur_score
    
    T = 0.3 * max(0.01, 1.0 - cur_score)
    pos1 = np.where(bits == 1)[0]
    pos0 = np.where(bits == 0)[0]
    
    for it in range(n_steps):
        if time.time() >= t_deadline:
            break
        # Random swap
        i = pos1[rng.integers(0, len(pos1))]
        j = pos0[rng.integers(0, len(pos0))]
        bits[i], bits[j] = bits[j], bits[i]
        pos1 = np.where(bits == 1)[0]
        pos0 = np.where(bits == 0)[0]
        
        new_score = official_score(bits.astype(float))
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_bits = bits.copy()
        else:
            bits[i], bits[j] = bits[j], bits[i]
            pos1 = np.where(bits == 1)[0]
            pos0 = np.where(bits == 0)[0]
        
        T *= 0.999
    
    return best_bits.astype(float), best_score

def periodic_bits(n, freq, phase=0.0):
    """Generate periodic binary pattern with exactly n/2 ones."""
    if n % 2 != 0:
        return None
    x = np.arange(n)
    raw = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    # Threshold to get balanced
    sorted_idx = np.argsort(raw)[::-1]
    bits = np.zeros(n, dtype=int)
    bits[sorted_idx[:n//2]] = 1
    return bits

def two_point_search(h, n_tries, rng, t_deadline):
    """Random two-point exchange search."""
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for _ in range(n_tries):
        if time.time() >= t_deadline:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates = [(t_lo + t_hi)/2, t_lo + 0.25*span, t_lo + 0.75*span]
        for t in candidates:
            tmp = best.copy()
            tmp[a] = best[a] + t
            tmp[b] = best[b] - t
            if tmp[a] < -1e-13 or tmp[a] > 1.0+1e-13 or tmp[b] < -1e-13 or tmp[b] > 1.0+1e-13:
                continue
            sc = official_score(tmp)
            if sc < best_sc - 1e-12:
                best = tmp
                best_sc = sc
    return best, best_sc

def make_feasible(z):

            grad[idx1] += (1.0 - h[np.arange(m)])

            grad[idx2] -= h[idx2 + s]
    else:

            grad[idx1] += (1.0 - h[idx1 + t])

def binary_sa(h0, n_steps, t_deadline, rng):
    n = len(h0)
    if n % 2 != 0:
        return None, None
    h_sorted_idx = np.argsort(h0)[::-1]
    bits = np.zeros(n, dtype=int)
    bits[h_sorted_idx[:n//2]] = 1
    cur_score = official_score(bits.astype(float))
    best_bits = bits.copy()
    best_score = cur_score
    T = 0.3 * max(0.01, 1.0 - cur_score)
    decay = 0.999
    for _ in range(n_steps):
        if time.time() >= t_deadline:
            break
        i = bits.copy()
        pos1 = np.where(bits == 1)[0]
        pos0 = np.where(bits == 0)[0]
        a = pos1[rng.integers(0, len(pos1))]
        b = pos0[rng.integers(0, len(pos0))]
        bits[a], bits[b] = bits[b], bits[a]
        new_score = official_score(bits.astype(float))
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_bits = bits.copy()
        else:
            bits[a], bits[b] = bits[b], bits[a]
        T *= decay
    return best_bits.astype(float), best_score

def periodic_bits(n, freq, phase=0.0):

        return None

    raw = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    sorted_idx = np.argsort(raw)[::-1]
    bits = np.zeros(n, dtype=int)
    bits[sorted_idx[:n//2]] = 1
    return bits

def two_point_search(h, n_tries, rng, t_deadline):

    best = h.copy()
    best_sc = official_score(h)
    for _ in range(n_tries):
        if time.time() >= t_deadline:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        for frac in [0.5, 0.25, 0.75, 0.35, 0.65]:
            t = t_lo + frac * span
            tmp_a = best[a] + t
            tmp_b = best[b] - t
            if tmp_a < -1e-13 or tmp_a > 1.0 + 1e-13:
                continue
            if tmp_b < -1e-13 or tmp_b > 1.0 + 1e-13:
                continue
            tmp = best.copy()
            tmp[a] = tmp_a
            tmp[b] = tmp_b
            sc = official_score(tmp)
            if sc < best_sc - 1e-13:
                best = tmp
                best_sc = sc
    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def subgradient_polish(h, n_iter, t_deadline):
    n = len(h)
    best = h.copy()
    best_sc = official_score(h)
    alpha = 0.04
    for _ in range(n_iter):
        if time.time() >= t_deadline:
            break
        V = np.correlate(best, 1.0 - best, mode='full') * (2.0 / n)
        cmax = V.max()
        active = np.where(V >= cmax - 0.04)[0]
        if len(active) == 0:
            active = np.array([np.argmax(V)])
        w = np.exp(60.0 * (V[active] - cmax))
        w /= w.sum()
        grad = np.zeros(n)
        for idx, p in enumerate(active):
            grad += w[idx] * grad_for_lag(best, int(p))
        d = -grad
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 0.25, alpha * 2.0]:
            trial = make_feasible(best + mult * d)
            sc = official_score(trial)
            if sc < best_sc - 1e-13:
                best = trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.6
            if alpha < 1e-6:
                break
        else:
            alpha = min(alpha * 1.1, 0.12)
    return best, best_sc

def make_wave(n, f1, a1, f2, a2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.sin(2*np.pi*f1*x + phase) + a2 * np.cos(2*np.pi*f2*x)
    return make_feasible(h)
