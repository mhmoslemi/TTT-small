def make_feasible(z):

    target = float(n) / 2.0
    z = np.asarray(z, dtype=float).copy()

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)

                    continue

def solve_lp_direction(h, V, active_idx, rho, C):

def binary_anneal(n, max_steps, is_dead, rng):

    if n % 2 != 0:
        return None

    bits = np.zeros(n, dtype=int)

    best_h = h_cur.copy()
    best_score = cur_score
    
    T = 5e-4
    pos1 = np.where(bits == 1)[0]
    pos0 = np.where(bits == 0)[0]
    
    for step in range(max_steps):
        if is_dead():
            break
        # Propose a swap
        i = pos1[rng.integers(0, len(pos1))]
        j = pos0[rng.integers(0, len(pos0))]
        
        # Perform swap
        bits[i], bits[j] = bits[j], bits[i]
        h_new = bits.astype(float)
        new_score = official_score(h_new)
        
        delta = new_score - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # Accept
            cur_score = new_score
            # Update pos arrays
            pos1 = np.where(bits == 1)[0]
            pos0 = np.where(bits == 0)[0]
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            # Reject
            bits[i], bits[j] = bits[j], bits[i]
        
        T *= 0.999
        
        if T < 1e-6:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = float(n) / 2.0
    z = np.asarray(z, dtype=float).copy()
    if not np.all(np.isfinite(z)):
        z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return grad
        i = np.arange(m)
        grad[i] += (1.0 - h[i + s])
        grad[i + s] -= h[i]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return grad
        j = np.arange(m)
        grad[j + t] += (1.0 - h[j])
        grad[j] -= h[j + t]
    return dx * grad

def solve_lp_direction(h, V, active_idx, rho, C):
    n = len(h)
    K = len(active_idx)
    if K == 0:
        return None
    c_active = V[active_idx]
    G = np.zeros((K, n))
    for i, p in enumerate(active_idx):
        s = int(p) - (n - 1)
        G[i] = lag_gradient(h, s)
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c_active
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            u = res.x[:n]
            if np.all(np.isfinite(u)) and np.max(np.abs(u)) < rho + 1e-10:
                return u
    except Exception:

    return None

    return make_feasible(0.5 * (h + 1.0 - h[::-1].copy()))

    n_src = len(h_src)

    h_int = np.interp(tgt_c, src_c, h_src)

def two_point_refine(h, n_tries, deadline, rng):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):

        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:

            tmp[b] = best[b] - t
            sc = official_score(tmp)
            if sc < best_sc - 1e-14:
                best = tmp
                best_sc = sc
                break
    return best, best_sc
