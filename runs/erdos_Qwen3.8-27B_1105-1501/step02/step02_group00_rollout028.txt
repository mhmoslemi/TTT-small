import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_scores(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    idx = np.arange(n)
    for k in range(2*n-1):
        d = k - (n - 1)
        if d == 0:
            G[k] = 1.0 - 2.0 * h
        elif d > 0:
            m1 = idx <= (n - 1 - d)
            if np.any(m1):
                G[k, m1] += (1.0 - h[idx[m1] + d])
            m2 = idx >= d
            if np.any(m2):
                G[k, m2] += -h[idx[m2] - d]
        else:
            g = -d
            m1 = idx >= g
            if np.any(m1):
                G[k, m1] += (1.0 - h[idx[m1] - g])
            m2 = idx <= (n - 1 - g)
            if np.any(m2):
                G[k, m2] += -h[idx[m2] + g]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _logsumexp_obj(h, beta):
    s = _scores(h)
    m = float(np.max(s))
    return m + beta * float(np.log(np.sum(np.exp((s - m) / beta))))

def _logsumexp_grad(h, beta):
    n = len(h)
    s = _scores(h)
    m = float(np.max(s))
    w = np.exp((s - m) / beta)
    w_sum = float(np.sum(w))
    w /= w_sum
    G = _grad_scores(h)
    return G.T @ w
