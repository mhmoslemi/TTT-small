def _score(h):
    n = len(h)
    dx = 2.0 / n

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

    return mc + np.log(es) / beta

def _fista(h0, beta, n_iter, alpha_init=None, deadline=None):
    """Nesterov accelerated gradient with projection."""

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

    ev = np.exp(sh)

```python
def _active_set_solve(h0, deadline, n_active=8, maxiter=100):
    """Solve min t s.t. t >= C_k(h) for active k, sum(h) = n/2, 0<=h<=1"""
    n = len(h0)
    cv = _corr_vec(h0)
    # Get active lags
    active = np.argsort(cv)[::-1][:n_active]
    
    # Variables: z = (h, t)
    z0 = np.concatenate([h0, [cv.max() + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def fun(z):
        return z[-1]
    
    def fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def constr_fun(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv[active]
    
    def constr_jac(z):
        h = z[:-1]
        n = len(h)
        dx = 2.0 / n
        m = len(active)
        J = np.zeros((m, n + 1))
        for ii, k in enumerate(active):
            J[ii, -1] = 1.0
            s = k - (n - 1)
            if s >= 0:
                J[ii, :n-s] -= dx * (1.0 - h[s:])
                J[ii, s:] += dx * h[:n-s]
            else:
                r = -s
                J[ii, r:] -= dx * (1.0 - h[:n-r])
                J[ii, :n-r] += dx * h[r:]
        return J
    
    def eq_fun(z):
        return z[:-1].sum() - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    try:
        res = minimize(fun, z0, jac=fun_jac, method='SLSQP',
                      bounds=bounds,
                      constraints=[
                          {'type': 'ineq', 'fun': constr_fun, 'jac': constr_jac},
                          {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
                      ],
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0
```
