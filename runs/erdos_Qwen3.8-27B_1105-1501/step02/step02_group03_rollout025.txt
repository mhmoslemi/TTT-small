def build_full(half):
    m = len(half)
    h = np.empty(2 * m)
    h[:m] = half
    h[m:] = 1.0 - half[::-1]
    return h

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_V_full(h, s):
    n = len(h)
    if s == 0:
        return 1.0 - 2.0 * h
    elif s > 0:
        L = n - s
        g = np.zeros(n)
        g[:L] += (1.0 - h[s:])
        g[s:] += (-h[:L])
        return g
    else:
        t = -s
        L = n - t
        g = np.zeros(n)
        g[t:] += (1.0 - h[:L])
        g[:L] += (-h[t:])
        return g

def all_grads_half(h, m):
    n = len(h)
    G = np.zeros((2 * n - 1, m))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        gf = grad_V_full(h, s)
        G[p] = gf[:m] - gf[m:2*m][::-1]
    return G

def smooth_obj_grad(x, m, beta):
    h = build_full(x)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    logsum = mx + np.log(exps.sum())
    obj = (cmax - cmax) + logsum / beta  # = logsum/beta... wait
    # Actually: F_beta = (1/beta) * log(sum exp(beta*V_s))
    # = (1/beta) * log(sum exp(beta*V_s - beta*cmax) * exp(beta*cmax))
    # = cmax + (1/beta) * log(sum exp(beta*(V_s - cmax)))
    # = cmax + (1/beta) * log(sum exp(shifted))
    # For numerical stability:
    # log(sum exp(shifted)) = mx + log(sum exp(shifted - mx))
    obj = cmax + (mx + np.log(exps.sum())) / beta
    w = exps / exps.sum()
    G = all_grads_half(h, m)
    grad = w @ G
    return obj, grad

def pg_optimize(x, m, beta, max_iter=10000, time_limit=30.0):
    x = x.copy()
    t0 = time.time()
    f, g = smooth_obj_grad(x, m, beta)
    f_prev = f
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-28:
        return x
    step = min(1.0 / (gnorm2 + 1e-12), 2.0 / (np.max(g**2) + 1e-12))
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        f, g = smooth_obj_grad(x, m, beta)
        gnorm = float(np.sqrt(g @ g))
        if gnorm < 1e-13:
            break
        
        alpha = step
        x_trial = x  # in case line search fails
        success = False
        for _ in range(40):
            x_trial = np.clip(x - alpha * g, 0.0, 1.0)
            f_trial = smooth_obj_grad(x_trial, m, beta)[0]
            if f_trial <= f - 1e-4 * alpha * gnorm:
                success = True
                break
            alpha *= 0.4
        if alpha < 1e-15:
            break
        if not success or not np.isfinite(f_trial):
            break
        improved = f_trial < f - 1e-12
        x = x_trial
        f_prev, f = f, f_trial
        if improved:
            step = min(step * 1.7, 5.0 / (gnorm2 + 1e-12))
    
    return x

def slsqp_solve(x0, m, active_shifts, max_iter=200):
    """Solve NLP: min t s.t. 0<=x<=1, t >= V(s) for s in active_shifts."""
    n = 2 * m
    
    h0 = build_full(x0)
    vals0 = compute_vals(h0)
    t_init = float(vals0.max()) + 1e-5
    z0 = np.concatenate([x0, [t_init]])
    
    ns = len(active_shifts)
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(z):
        x = z[:m]
        t = z[-1]
        h = build_full(x)
        v = compute_vals(h)
        # Map shift index to correlate index
        # Shift s corresponds to correlate index p = s + (n-1)
        v_active = np.array([v[s + (n-1)] for s in active_shifts])
        return t - v_active
    
    def ineq_jac(z):
        x = z[:m]
        h = build_full(x)
        G = all_grads_half(h, m)
        # G has rows indexed by p = 0,...,2n-2
        # Shift s corresponds to row p = s + (n-1)
        J = np.zeros((ns, m + 1))
        for i, s in enumerate(active_shifts):
            p = s + (n - 1)
            J[i, :m] = -G[p]
            J[i, -1] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            x_new = np.clip(res.x[:m], 0.0, 1.0)
            h_new = build_full(x_new)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return x_new, sc
    except Exception:
        pass
    return None, None

def active_set_optimize(x0, m, time_limit=60.0):
    """Iterative active-set SLSQP optimization."""
    n = 2 * m
    x = x0.copy()
    best_score = official_score(build_full(x))
    t0 = time.time()
    
    for iteration in range(20):
        if time.time() - t0 > time_limit:
            break
        
        h = build_full(x)
        vals = compute_vals(h)
        cmax = float(vals.max())
        
        # Find active shifts
        threshold = cmax - max(0.01 * cmax, 1e-5)
        active = [p - (n-1) for p in range(2*n-1) if vals[p] >= threshold]
        # Also include top K
        top_k = max(10, (2*n-1) // 5)
        top_idx = np.argsort(vals)[::-1][:top_k]
        for p in top_idx:
            s = p - (n-1)
            if s not in active:
                active.append(s)
        active = sorted(set(active))
        
        # Solve
        tl = max(5.0, time_limit - (time.time() - t0) - 5.0)
        x_new, sc = slsqp_solve(x, m, active, max_iter=200)
        if x_new is not None and sc < best_score - 1e-14:
            x = x_new
            best_score = sc
        
        # Check for violated constraints
        h = build_full(x)
        vals = compute_vals(h)
        cmax_new = float(vals.max())
        if cmax_new > best_score + 1e-6:
            # Some constraint was violated; add it
            violated = [p - (n-1) for p in range(2*n-1) if vals[p] > best_score + 1e-8]
            for s in violated:
                if s not in active:
                    active.append(s)
            active = sorted(set(active))
        else:
            # Converged
            break
    
    return x, best_score

def antisymmetrize(h):
    n = len(h)
    if n % 2 == 0:
        h_sym = np.empty(n)
        h_sym[:n//2] = 0.5 * (h[:n//2] + 1.0 - h[n//2:][::-1])
        h_sym[n//2:] = 1.0 - h_sym[:n//2][::-1]
