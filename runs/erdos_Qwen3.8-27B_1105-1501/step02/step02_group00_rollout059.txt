    idx = int(np.argmax(corr))

def _binary_c5(h_bin):
    n = len(h_bin)
    # For binary, we can compute more efficiently
    # S(e) = sum_{i=0}^{n-1-e} h[i] * (1-h[i+e]) for e = 0..n-1
    # T(e) = sum_{i=0}^{n-1-e} h[i+e] * (1-h[i]) for e = 0..n-1

    for e in range(1, n):
        s = np.sum(h_bin[:n-e] * (1 - h_bin[e:]))
        t = np.sum(h_bin[e:] * (1 - h_bin[:n-e]))
        max_s = max(max_s, s, t)
    return max_s * 2.0 / n

    # Take top n/2 values
    idx = np.argsort(-h)[:n//2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

    """Simulated annealing on binary sequences"""
    best = h_init.copy()
    best_c5 = _binary_c5(best)
    h = h_init.copy()
    cur_c5 = best_c5
    cur_ones = np.where(h > 0.5)[0]
    cur_zeros = np.where(h <= 0.5)[0]
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max(n_iters, 1))

**Alternative: Use the fact that the objective is piecewise bilinear**

The objective F(h) = max_k O_k(h) where each O_k is bilinear. At any given h, only one (or a few) O_k are active. For the active one, we can do a proper descent on that bilinear function.

For a bilinear function O_k(h) = sum_i h[i](1-h[i+d]), the gradient is:
∇O_k = ((1-h[d:]) - h[:n-d], ...) as computed above.

But O_k is not convex (it's bilinear = neither convex nor concave in general). So gradient descent on O_k alone won't work globally.

However, if we fix the constraint that h stays near its current value (a trust region), we can linearize O_k and use the LP approach (which is what the parent did). The difference is we should be smarter about:
1. Which constraints to activate
2. The step size
3. When to restart

Let me try a different angle: **Nesterov's accelerated gradient for the smoothed max**

Consider the smoothed objective:
F_μ(h) = μ * log(sum_k exp(O_k/μ))

This is a smooth approximation of max_k O_k, and it's convex in h... wait, no. O_k is bilinear, so F_μ is not convex either.

But we can still do gradient descent on it. The gradient is:
∇F_μ(h) = (1/sum_k exp(O_k/μ)) * sum_k exp(O_k/μ) * ∇O_k

This is a weighted average of the gradients of all O_k, weighted by their relative values.

For large μ, this is smooth and we can use a fixed step size. As μ → 0, it approaches the true max.

def _smoothed_c5(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr values are the raw correlations (before multiplying by dx)
    # We want mu * log(sum exp(corr/mu)) * dx
    max_corr = np.max(corr)
    logsum = max_corr + mu * np.log(np.sum(np.exp((corr - max_corr) / mu)))
    return float(logsum) * dx

def _smoothed_grad(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp((corr - max_corr) / mu)
    weights /= np.sum(weights)
    
    # grad = sum_k weights[k] * grad_O_k
    # grad_O_k has the form computed above
    # But we need to compute it for all k simultaneously
    
    # For shift d (where d ranges from -(n-1) to n-1):
    # grad_O_d[j] = (1-h[j+d])*1[j+d<n] - h[j-d]*1[j-d>=0] for d >= 0
    #              = (1-h[j-d])*1[j-d>=0] - h[j+d]*1[j+d<n] for d < 0
    
    # Let's compute this efficiently
    # For d >= 0: grad = (1-h[d:])[:n-d] padded + (-h[:n-d]) shifted by d
    # We can use the correlation structure
    
    # Actually, let's just loop over the top few active constraints
    top_k = np.argsort(-weights)[:10]  # top 10
    grad = np.zeros(n)
    for idx in top_k:
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += weights[idx] * (1.0 - h[d:])
            grad[d:] += -weights[idx] * h[:m]
        else:
            e = -d
            m = n - e
            grad[e:] += weights[idx] * (1.0 - h[:m])
            grad[:m] += -weights[idx] * h[e:]
    
    return grad * dx
```

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smoothed_obj_grad(h, mu):
    """Compute smoothed C5 and its gradient using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    # Numerical stability
    shifted = (corr - max_c) / mu
    exp_s = np.exp(shifted)
    logsum = max_c + mu * np.log(np.sum(exp_s))
    obj = float(logsum) * dx
    
    # Weights for each shift
    w = exp_s / np.sum(exp_s)
    
    # Gradient: weighted sum of individual gradients
    # For shift index idx, d = idx - (n-1)
    grad = np.zeros(n)
    # Only use top shifts for efficiency
    top_idxs = np.argsort(-w)
    n_top = min(20, len(top_idxs))
    for idx in top_idxs[:n_top]:
        wi = float(w[idx])
        if wi < 1e-10:
            break
        d = int(idx) - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += wi * (1.0 - h[d:])
            grad[d:] += -wi * h[:m]
        else:
            e = -d
            m = n - e
            grad[e:] += wi * (1.0 - h[:m])
            grad[:m] += -wi * h[e:]
    
    grad *= dx
    return obj, grad

def _smooth_descent(h_init, deadline, mu_start=0.3, mu_end=0.01, n_phases=5):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        mu_s = mu_start * (mu_end / mu_start) ** (phase / n_phases)
        mu_e = mu_start * (mu_end / mu_start) ** ((phase + 1) / n_phases)
        
        # Number of iterations for this phase
        phase_deadline = min(deadline, time.time() + (deadline - time.time()) / (n_phases - phase))
        
        it = 0
        while time.time() < phase_deadline:
            it += 1
            t = it / 10000.0
            mu = mu_s * (mu_e / mu_s) ** min(t, 1.0)
            
            obj, grad = _smoothed_obj_grad(h, mu)
            grad -= np.mean(grad)
            
            # Step size: for smooth function with Lipschitz L, optimal step is 1/L
            # L ≈ 1/mu for the log-sum-exp smoothing of a bilinear function
            step = min(0.5, 2.0 * mu)
            
            h_new = _project(h - step * grad, target)
            new_c5 = _c5(h_new)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            h = h_new
    
    return best_h, best_c5

def _binary_sa(h_frac, n, deadline, rng, n_iters=50000):
    """Simulated annealing on binary sequences of length n"""
    # Convert to binary
    if len(h_frac) == n:
        h_bin = np.zeros(n)
        idx = np.argsort(-h_frac)[:n//2]
        h_bin[idx] = 1.0
    else:
        # Resample
        x_old = (np.arange(len(h_frac)) + 0.5) / len(h_frac)
        x_new = (np.arange(n) + 0.5) / n
        h_resampled = np.interp(x_new, x_old, h_frac)
        h_bin = np.zeros(n)
        idx = np.argsort(-h_resampled)[:n//2]
        h_bin[idx] = 1.0
    
    def bin_c5(h_b):
        dx = 2.0 / n
        corr = np.correlate(h_b, 1.0 - h_b, mode='full')
        return float(np.max(corr)) * dx
    
    cur_c5 = bin_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    T = 0.1
    T_end = 0.001
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = 0.1 * (0.001 / 0.1) ** (it / n_iters)
        
        # Random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = bin_c5(h_bin)

            cur_c5 = new_c5

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0

Wait, I think there's an issue with my smoothed gradient. The log-sum-exp gives a smooth approximation of the max, but the gradient I'm computing might not be exactly right because the bilinear structure makes things non-standard.

Let me verify: if F(h) = mu * log(sum_k exp(O_k(h)/mu)), then:
∇F(h) = (1/sum_k exp(O_k/mu)) * sum_k exp(O_k/mu) * ∇O_k(h)
        = sum_k w_k * ∇O_k(h)

where w_k = exp(O_k/mu) / sum_j exp(O_j/mu).

And ∇O_k(h) is the gradient of the bilinear form O_k(h) = sum_i h[i](1-h[i+d_k]).

For d ≥ 0:

def _overlap_array(h_bin, n):
    """Compute S(d) and T(d) for all d"""
    S = np.zeros(n)  # S[d] = #{i: h[i]=1, h[i+d]=0}
    T = np.zeros(n)  # T[d] = #{i: h[i]=0, h[i+d]=1}
    for d in range(1, n):
        S[d] = np.sum(h_bin[:n-d] * (1 - h_bin[d:]))
        T[d] = np.sum((1 - h_bin[:n-d]) * h_bin[d:])
    return S, T
```
