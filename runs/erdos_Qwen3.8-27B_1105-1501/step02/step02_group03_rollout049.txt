import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -12.0, 12.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_all(h):
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for p in range(K):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] -= dx * h[s:]
            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K2 = n - t
            G[p, :K2] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K2]
    return G

def build_h_from_z(z):
    M = len(z)
    N = 2 * M
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
    h = np.empty(N)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    return h

def z_obj_grad(z, beta, need_grad=True):
    h = build_h_from_z(z)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    mx = shifted.max()
    logsum = cmax + (mx + np.log(np.sum(np.exp(shifted - mx)))) / beta
    if not need_grad:
        return logsum, vals
    w = np.exp(shifted - mx)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_all(h)
    M = len(z)
    u = h[:M]
    du = u * (1.0 - u)
    Gz = (G[:, :M] - G[:, M:][::-1]) * du[np.newaxis, :]
    grad = w @ Gz
    return logsum, vals, grad

def bfgs_z(z0, beta, max_iter=400):
    def _f(z):
        return z_obj_grad(z, beta, need_grad=False)[0]

    def _g(z):
        return z_obj_grad(z, beta, need_grad=True)[2]

    try:
        res = minimize(_f, z0, jac=_g, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 30})
        if res.success or res.status == 0:
            return res.x
        return res.x
    except Exception:
        return z0

def find_active(vals, frac=0.3, min_count=8):
    cmax = vals.max()
    threshold = cmax - max(0.02 * cmax, 1e-5)
    active = np.where(vals >= threshold)[0]
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active = np.union1d(active, top_idx)
    return sorted(active.tolist())

def h_to_z(h):
    N = len(h)
    M = N // 2
    u = h[:M].copy()
    u = np.clip(u, 1e-8, 1.0 - 1e-8)
    z = np.log(u / (1.0 - u))
    return np.clip(z, -15.0, 15.0)

    N = len(h)
    h_sym = 0.5 * (h + 1.0 - h[::-1].copy())

            active_f = find_active(vals_f, frac=0.4, min_count=12)
            h_new, sc = solve_full_slsqp(best_h, active_f, max_iter=300)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_n = len(final_h)
    final_score = official_score(final_h)
    assert np.all(final_h >= -1e-10)
    assert np.all(final_h <= 1.0 + 1e-10)
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-9

    final_h = np.clip(final_h, 0.0, 1.0)
    # Re-check after clip
    if abs(np.sum(final_h) - final_n / 2.0) > 1e-9:
        final_h = make_feasible(final_h)
    final_score = official_score(final_h)

    return final_h, final_score, final_n
