And we want to minimize max_s f_s(δ) over δ in a feasible range. This is a 1D convex optimization (max of linear functions is convex), solvable exactly.

This should work well! Let me implement it.

Algorithm:
1. Start from the parent solution.
2. Repeatedly do sweeps over all pairs (j,k), for each pair find the optimal δ and update.
3. Continue until no improvement.
4. Try random restarts / perturbations to escape local optima.

Actually, for n=60, O(n^2) = 3600 pairs per sweep, and each pair takes O(n) time, so each sweep is O(n^3) = 216K operations. That's very fast. We can do thousands of sweeps in a second.

But the question is whether pairwise descent will actually improve the solution. The issue is that the optimal direction might require coordinated changes to many coordinates simultaneously.

    b_ub = np.zeros(k_active)

        h_try = h + alpha * d
        h_try = make_feasible(h_try)
        f_try = official_score(h_try)
        if f_try < best_f - 1e-14:
            best_h = h_try
            best_f = f_try
            break
    return best_h, best_f

def lp_optimize(h, max_iter=80, k_active=5):
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    stall = 0
    for it in range(max_iter):
        d = lp_direction(h, k_active)
        if d is None:
            break
        h_new, sc_new = line_search(h, d)
        if sc_new < best_score - 1e-14:
            h = h_new
            best_score = sc_new
            stall = 0
        else:
            stall += 1
            if stall >= 10:
                break
    return h, best_score

def pair_polish(h, max_sweeps=10):
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    f = compute_f(h)
    G = compute_gradients(h)
    P = 2 * n - 1
    n_grid = 40
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            for k in range(n):
                if j == k:
                    continue
                c = G[:, k] - G[:, j]
                delta_lo = max(-h[k], h[j] - 1.0)
                delta_hi = min(h[j], 1.0 - h[k])
                if delta_hi - delta_lo < 1e-13:
                    continue
                grid = np.linspace(delta_lo, delta_hi, n_grid)
                vals = np.max(f[None, :] + grid[:, None] * c[None, :], axis=1)
                best_idx = np.argmin(vals)
                if vals[best_idx] < best_score - 1e-13 and abs(grid[best_idx]) > 1e-14:
                    h[j] -= grid[best_idx]
                    h[k] += grid[best_idx]
                    h = np.clip(h, 0.0, 1.0)
                    f = f + grid[best_idx] * c
                    improved = True
        if improved:
            f = compute_f(h)
            G = compute_gradients(h)
            sc = official_score(h)
            if sc < best_score:
                best_score = sc
            if not improved:
                break
    h = make_feasible(h)
    sc = official_score(h)
    if sc < best_score:
        best_score = sc
    return h, best_score

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def smooth_obj_fun(h, beta):
    n = len(h)
    f = compute_f(h)
    cmax = f.max()
    shifted = np.clip(beta * (f - cmax), -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))

def slsqp_smooth(h0, beta, maxiter=100):
    from scipy.optimize import minimize
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj_fun, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    best_h = None
    best_score = float('inf')
    best_n = 0

    def update(h, sc):
        nonlocal best_h, best_score, best_n
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                best_n = len(parent)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x / 2.0))
        best_score = official_score(best_h)
        best_n = len(best_h)

    update(best_h, best_score)

    # Phase 1: LP optimization with varying k_active
    for k_a in [3, 5, 7, 10]:
        if remaining() < 30:
            break
        h_opt, sc = lp_optimize(best_h, max_iter=60, k_active=k_a)
        update(h_opt, sc)

    # Phase 2: Pairwise polish
    if remaining() > 30:
        h_opt, sc = pair_polish(best_h, max_sweeps=8)
        update(h_opt, sc)

    # Phase 3: SLSQP smooth max
    if remaining() > 60:
        cur = best_h.copy()
        for beta in [500, 2000, 10000]:
            if remaining() < 20:
                break
            h_new, sc = slsqp_smooth(cur, beta, maxiter=80)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
                cur = h_new
            elif h_new is not None:
                cur = h_new

    # Phase 4: LP again from best
    if remaining() > 40:
        for k_a in [5, 8, 12]:
            if remaining() < 15:
                break
            h_opt, sc = lp_optimize(best_h, max_iter=40, k_active=k_a)
            update(h_opt, sc)

    # Phase 5: Resolution sweep
    if remaining() > 80:
        n_base = len(best_h)
        n_opts = sorted(set([max(24, n_base - 10), n_base - 4, n_base,
                             min(80, n_base + 4)]))
        for n_opt in n_opts:
            if remaining() < 20:
                break
            if n_opt < 16:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            h_opt, sc = lp_optimize(h_int, max_iter=40, k_active=5)
            update(h_opt, sc)
            if remaining() > 15:
                h_opt2, sc2 = pair_polish(h_opt, max_sweeps=5)
                update(h_opt2, sc2)

    # Phase 6: Random perturbations + LP
    if remaining() > 60:
        for i in range(20):
            if remaining() < 15:
                break
            sigma = 0.02 + 0.08 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc = lp_optimize(h_cand, max_iter=30, k_active=5)
            update(h_opt, sc)
            if remaining() > 20:
                h_new, sc2 = slsqp_smooth(h_opt, 2000, maxiter=50)
                if h_new is not None:
                    update(h_new, sc2)

    # Phase 7: Intensive final polish
    if remaining() > 40:
        for k_a in [3, 5, 7]:
            if remaining() < 10:
                break
            h_opt, sc = lp_optimize(best_h, max_iter=30, k_active=k_a)
            update(h_opt, sc)
        if remaining() > 20:
            h_opt, sc = pair_polish(best_h, max_sweeps=10)
            update(h_opt, sc)
        if remaining() > 15:
            h_new, sc = slsqp_smooth(best_h, 20000, maxiter=80)
            if h_new is not None:
                update(h_new, sc)

    # Phase 8: Final LP burst
    if remaining() > 20:
        for k_a in [5, 10, 15, 20]:
            if remaining() < 8:
                break
            h_opt, sc = lp_optimize(best_h, max_iter=20, k_active=k_a)
            update(h_opt, sc)

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
