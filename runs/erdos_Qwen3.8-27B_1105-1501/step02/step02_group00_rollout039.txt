    """Find optimal alpha for h[j]+=alpha, h[k]-=alpha."""
    # Compute c_r = b_r(j) - b_r(k) for each lag r
    c = np.zeros(2*n - 1)
    for r in range(2*n - 1):
        d = r - (n - 1)
        # b_r(j)
        if d >= 0:
            bj = 0.0
            if j <= n - 1 - d:
                bj += (1.0 - h[j + d])
            if j >= d:
                bj -= h[j - d]
        else:
            g = -d
            bj = 0.0
            if j >= g:
                bj += (1.0 - h[j - g])
            if j + g <= n - 1:
                bj -= h[j + g]
        # b_r(k)
        if d >= 0:
            bk = 0.0
            if k <= n - 1 - d:
                bk += (1.0 - h[k + d])
            if k >= d:
                bk -= h[k - d]
        else:
            g = -d
            bk = 0.0
            if k >= g:
                bk += (1.0 - h[k - g])
            if k + g <= n - 1:
                bk -= h[k + g]
        c[r] = bj - bk

    alpha_hi = min(1.0 - h[j], h[k])

    lo, hi = alpha_lo, alpha_hi

def precompute_B(h, n):
    """Compute B where B[r, j] = coefficient of h[j] in corr[r]."""
    B = np.zeros((2*n - 1, n))
    for r in range(2*n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                # corr[r] = sum_{i=0}^{m-1} h[i]*(1-h[i+d])
                B[r, :m] += (1.0 - h[d:d+m])  # h[i] coefficient: (1-h[i+d])
                B[r, d:d+m] -= h[:m]  # h[i+d] coefficient: -h[i]
        else:
            g = -d
            m = n - g
            if m > 0:
                # corr[r] = sum_{i=g}^{n-1} h[i]*(1-h[i-g])
                B[r, g:] += (1.0 - h[:m])  # h[i] coefficient: (1-h[i-g])
                B[r, :m] -= h[g:]  # h[i-g] coefficient: -h[i]

Now, for the pair update:
c_r = B[r, j] - B[r, k]

And we want to minimize max_r (corr[r] + α * c_r) over α in [α_lo, α_hi].

This is a 1D convex PL problem. The optimal α can be found by:
1. Computing all "breakpoints" where a different lag becomes active
2. Or by binary search on the subgradient

The binary search approach: at α, the active set is {r : corr[r] + α*c_r = max}. The subgradient is {c_r : r active}. We want 0 to be in the subgradient, i.e., min(c_r for active r) ≤ 0 ≤ max(c_r for active r).

If max(c_r for active r) < 0, the function is decreasing at α, so we should increase α → lo = mid.
If min(c_r for active r) > 0, the function is increasing at α, so we should decrease α → hi = mid.
Otherwise, we've found the optimum.

This converges in O(log(1/ε)) iterations.

    d = r - (n - 1)
    if d >= 0:

        if j <= n - 1 - d:
            c += (1.0 - h[j + d])
        if j >= d:
            c -= h[j - d]
        return c
    else:

        c = 0.0
        if j >= g:
            c += (1.0 - h[j - g])
        if j + g <= n - 1:

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_B(h):
    n = len(h)
    B = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        r = n - 1 + d
        m = n - d
        B[r, :m] += (1.0 - h[d:d+m])
        B[r, d:n] -= h[:m]
    for g in range(1, n):
        r = n - 1 - g
        m = n - g
        B[r, g:n] += (1.0 - h[:m])
        B[r, :m] -= h[g:n]
    return B

def _pair_descent(h, deadline, max_sweeps=500, refresh_every=15, anneal_temp=0.0):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    B = _compute_B(h)
    cur_c5 = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_c5
    updates_count = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_improvement = False
        sweep_improvements = 0
        
        for j in range(n):
            if time.time() > deadline:
                break
            hj = h[j]
            if hj <= 1e-6 or hj >= 1.0 - 1e-6:
                continue
            for k in range(j + 1, n):
                hk = h[k]
                if hk <= 1e-6 or hk >= 1.0 - 1e-6:
                    continue
                
                a_lo = max(-hj, hk - 1.0)
                a_hi = min(1.0 - hj, hk)
                if a_hi - a_lo < 1e-12:
                    continue
                
                c = B[:, j] - B[:, k]
                max_abs_c = float(np.max(np.abs(c)))
                if max_abs_c < 1e-12:
                    continue
                
                # Binary search for optimal alpha that minimizes max(corr + alpha*c)
                lo, hi = a_lo, a_hi
                for _ in range(35):
                    mid = (lo + hi) / 2.0
                    vals = corr + mid * c
                    mx = float(np.max(vals))
                    active = vals >= mx - 1e-9
                    mc = c[active]
                    if float(np.max(mc)) < -1e-12:
                        lo = mid
                    elif float(np.min(mc)) > 1e-12:
                        hi = mid
                    else:
                        break
                alpha = (lo + hi) / 2.0
                
                if abs(alpha) < 1e-10:
                    continue
                
                new_corr = corr + alpha * c
                new_c5 = float(np.max(new_corr))
                delta = new_c5 - cur_c5
                
                # Accept if improvement or (rarely) worsening for annealing
                accept = delta < -1e-11
                if not accept and anneal_temp > 0:
                    accept = np.random.random() < np.exp(-delta / anneal_temp)
                
                if accept:
                    h[j] += alpha
                    h[k] -= alpha
                    np.clip(h, 0.0, 1.0, out=h)
                    corr = new_corr
                    cur_c5 = new_c5
                    any_improvement = True
                    sweep_improvements += 1
                    updates_count += 1
                    
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    
                    if updates_count % refresh_every == 0:
                        B = _compute_B(h)
        
        if not any_improvement:
            break
        B = _compute_B(h)
    
    return best_h, best_c5

def _subgradient_step(h, corr, step):
    n = len(h)
    target = n / 2.0
    max_idx = int(np.argmax(corr))
    d = max_idx - (n - 1)
    if d >= 0:
        m = n - d
        grad = np.zeros(n)
        grad[:m] += (1.0 - h[d:d + m])
        grad[d:n] -= h[:m]
    else:
        g = -d
        m = n - g
        grad = np.zeros(n)
        grad[g:n] += (1.0 - h[:m])
        grad[:m] -= h[g:]
    h_new = h - step * grad
    return _project(h_new, target)

    # Phase 5: Try nearby resolutions
    deadline5 = t0 + budget_s * 0.94
    for dn in [8, -8, 16, -16, 32, 4, -4]:
        if time.time() > deadline5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200:
            continue
        h_r = _resample_h(best_h, n_try)
        t_alloc = min(deadline5 - time.time(), budget_s * 0.035)
        if t_alloc < 2.0:
            break
        h_t, c5_t = _pair_descent(h_r, time.time() + t_alloc, max_sweeps=200)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_try

    # Phase 6: Final intensive polish
    if time.time() < t0 + budget_s * 0.98:
        h_f, c5_f = _pair_descent(best_h, t0 + budget_s * 0.995, max_sweeps=3000, refresh_every=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
