            alpha = step

            f, g = f_new, g_new
            step = min(step * 1.5, 2.0 / (gnorm + 1e-10))

def antisym_from_h(h):
    """Extract the anti-symmetric part: u = first half, checking h[i]+h[n-1-i]≈1."""
    n = len(h)
    m = n // 2
    # Force anti-symmetry: u[k] = (h[k] + (1-h[n-1-k])) / 2
    u = 0.5 * (h[:m] + 1.0 - h[n-1-np.arange(m)])
    return np.clip(u, 0.0, 1.0)

def antisym_symmetrize(h):
    """Enforce anti-symmetry on h and return the anti-symmetric u."""
    n = len(h)
    m = n // 2
    u = 0.5 * (h[:m] + 1.0 - h[n-1-np.arange(m)])
    return np.clip(u, 0.0, 1.0)

def interpolate_u(u_src, n_tgt):
    """Interpolate anti-symmetric representation to different size."""
    m_src = len(u_src)
    m_tgt = n_tgt // 2
    x_src = np.arange(m_src) / m_src
    x_tgt = np.arange(m_tgt) / m_tgt
    u_tgt = np.interp(x_tgt, x_src, u_src)
    return np.clip(u_tgt, 0.0, 1.0)

def slsqp_solve(u0, active_idx, max_iter=500, tol=1e-14):
    """SLSQP minimax in anti-symmetric space."""
    m = len(u0)
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def cons_fun(z):
        u = z[:m]
        t = z[-1]
        h = build_sym(u)
        vals = compute_vals(h)
        return t - vals[active_idx]
    
    def cons_jac(z):
        u = z[:m]
        h = build_sym(u)
        Gr = grad_u_fast(h)
        J = -Gr[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])
    
    t0_init = compute_vals(build_sym(u0)).max() + 1e-6
    z0 = np.concatenate([u0, [t0_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 3.0)]
    cons = [
        {'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac},
    ]
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            u_new = np.clip(res.x[:m], 0.0, 1.0)
            return u_new
    except Exception:
        pass
    return None

def find_active(vals, frac=0.3, min_count=6):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0])
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_sym(u):
    m = len(u)
    n = 2 * m
    h = np.empty(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

def grad_u_fast(h):
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    Gr = np.zeros((2 * n - 1, m))
    ar = np.arange(m)
    mirror = n - 1 - ar
    for p in range(2 * n - 1):
        s = p - (n - 1)
        row = np.zeros(n)
        if s >= 0:
            L = n - s
            row[s:] += dx * (1.0 - h[:L])
            row[:L] -= dx * h[s:]
        else:
            t = -s
            K = n - t
            row[:K] += dx * (1.0 - h[t:])
            row[t:] -= dx * h[:K]
        Gr[p] = row[ar] - row[mirror]
    return Gr

def smooth_obj_grad_u(u, beta):
    h = build_sym(u)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    Gr = grad_u_fast(h)
    grad = w @ Gr
    return obj, grad

def smoothed_pg_solve(u0, betas, time_limit=60.0, max_iter=30000):
    u = u0.copy()
    t0 = time.time()

            break
        f, g = smooth_obj_grad_u(u, beta)
        gnorm = float(g @ g) ** 0.5
        if gnorm < 1e-12:
            continue
        step = 1.0 / (gnorm + 1e-10)
        step = min(step, 0.5 / (np.max(np.abs(g)) + 1e-10))
        for it in range(max_iter):
            if time.time() - t0 > time_limit:
                break
            if gnorm < 1e-11:
                break
            alpha = step
            accepted = False
            for _ in range(20):
                u_trial = np.clip(u - alpha * g, 0.0, 1.0)

                    u = u_trial
                    f = f_trial
                    accepted = True
                    break
                alpha *= 0.4

                break
            f, g = smooth_obj_grad_u(u, beta)
            gnorm = float(g @ g) ** 0.5
            step = min(step * 1.5, 2.0 / (gnorm + 1e-10))
    return u

def subgrad_polish(u, steps=200, start_step=0.1):
    u_cur = u.copy()
    step = start_step
    for _ in range(steps):
        h = build_sym(u_cur)
        vals = compute_vals(h)
        cmax = vals.max()
        threshold = cmax - max(0.02 * cmax, 1e-5)
        active = np.where(vals >= threshold)[0]
        Gr = grad_u_fast(h)

            break
        u_cur = np.clip(u_cur - step * g / gnorm, 0.0, 1.0)
        step *= 0.97
    return u_cur

def antisym_symmetrize(h):
    n = len(h)
    m = n // 2
    u = 0.5 * (h[:m] + 1.0 - h[n - 1 - np.arange(m)])
    return np.clip(u, 0.0, 1.0)

def interpolate_u(u_src, n_tgt):
    m_src = len(u_src)
    m_tgt = n_tgt // 2
    x_src = (np.arange(m_src) + 0.5) / m_src
    x_tgt = (np.arange(m_tgt) + 0.5) / m_tgt
    u_tgt = np.interp(x_tgt, x_src, u_src)
    return np.clip(u_tgt, 0.0, 1.0)

def slsqp_solve(u0, active_idx, max_iter=400):
    m = len(u0)
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def cons_fun(z):
        u = z[:m]
        t = z[-1]
        h = build_sym(u)
        vals = compute_vals(h)
        return t - vals[active_idx]
    
    def cons_jac(z):
        u = z[:m]
        h = build_sym(u)
        Gr = grad_u_fast(h)
        J = -Gr[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])
    
    t0_init = compute_vals(build_sym(u0)).max() + 1e-5
    z0 = np.concatenate([u0, [t0_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 3.0)]
    cons = [{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        pass
    return None

    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)
