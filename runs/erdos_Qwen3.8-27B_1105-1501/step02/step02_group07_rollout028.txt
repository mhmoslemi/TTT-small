```python
    h_idx2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    J = valid1 * (1.0 - h_idx1) - valid2 * h_idx2
    return J
```

This is fully vectorized and should be fast. For n=60, the arrays are 119x60 = 7140 elements, which is tiny.

Now the pairwise line search:
```python
def pairwise_line_search(h, i, j, V, J):
    n = len(h)
    D = J[:, i] - J[:, j]  # (2n-1,)
    
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    
    if delta_hi - delta_lo < 1e-15:
        return 0.0
    
    # Minimize f(δ) = max(V + D*δ) over δ in [delta_lo, delta_hi]
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = delta_lo, delta_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = np.max(V + D * c_pt)
    fd = np.max(V + D * d_pt)
    
    for _ in range(50):
        if b - a < 1e-14:
            break
        if fc < fd:
            b = d_pt; d_pt = c_pt; fd = fc
            c_pt = b - gr * (b - a)
            fc = np.max(V + D * c_pt)
        else:
            a = c_pt; c_pt = d_pt; fc = fd
            d_pt = a + gr * (b - a)
            fd = np.max(V + D * d_pt)
    
    return (a + b) / 2.0
```

```python
def pair_sweep(h, V, J, max_improvement_check=False):
```

import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return h

def compute_V(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_J(h):
    n = len(h)
    num_sh = 2 * n - 1
    s = np.arange(num_sh) - (n - 1)
    p = np.arange(n)
    s_col = s[:, np.newaxis]
    p_row = p[np.newaxis, :]
    idx1 = p_row - s_col
    idx2 = p_row + s_col
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h_idx1 = np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
    h_idx2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return valid1 * (1.0 - h_idx1) - valid2 * h_idx2

def line_search_1d(V, D, delta_lo, delta_hi, max_iter=50):
    if delta_hi - delta_lo < 1e-15:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = delta_lo, delta_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = np.max(V + D * c_pt)
    fd = np.max(V + D * d_pt)
    for _ in range(max_iter):
        if b - a < 1e-15:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = np.max(V + D * c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = np.max(V + D * d_pt)
    return (a + b) / 2.0

def pair_sweep(h, V, J):
    n = len(h)
    old_score = V.max()
    for i in range(n):
        for j in range(i + 1, n):
            D = J[:, i] - J[:, j]
            delta_lo = max(-h[i], h[j] - 1.0)
            delta_hi = min(1.0 - h[i], h[j])
            if delta_hi - delta_lo < 1e-15:
                continue
            delta = line_search_1d(V, D, delta_lo, delta_hi)
            if abs(delta) > 1e-14:
                h[i] += delta
                h[j] -= delta
                V = V + delta * D
                J = compute_J(h)
