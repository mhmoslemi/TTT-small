Wait, I think there's an issue. Let me re-derive.

The LSE objective is: f(x) = max_k corr_k + (1/beta) * log(sum_k exp(beta*(corr_k - max_j corr_j)))

where corr_k = sum_j h[j]*(1-h[j+s_k]) * dx and h = [x, 1-x].

The gradient:
df/dx[i] = sum_k w_k * dcorr_k/dx[i]

where w_k = exp(beta*(corr_k - cmax)) / sum_j exp(beta*(corr_j - cmax))

And dcorr_k/dx[i] = dcorr_k/dh[i] * 1 + dcorr_k/dh[i+m] * (-1)
= dcorr_k/dh[i] - dcorr_k/dh[i+m]

Now, dcorr_k/dh[i]:
corr_k = sum_j h[j]*(1-h[j+s_k])*dx

dcorr_k/dh[i] = dx * [term where j=i in first factor + term where j+s_k=i in second factor]
= dx * [(1-h[i+s_k]) * 1_{0<=i+s_k<n} + (-h[i-s_k]) * 1_{0<=i-s_k<n}]

def anti_slsqp(h0, maxiter=200):
    """SLSQP on anti-symmetric subspace."""
    x0 = h0[:len(h0)//2].copy()
    m = len(x0)
    dx = 2.0 / len(h0)
    n = len(h0)

        shifted = np.clip(beta*(corr-cmax), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            return cmax
        return cmax + np.log(s)/beta

        return lse_and_grad(x, 2000)[1]

def coord_descent_anti(x, n_rounds=50, time_limit=30, t_start=None):
    """Coordinate descent on anti-symmetric parameterization."""
    m = len(x)
    best_h = build_anti(x)
    best_score = official_score(best_h)
    
    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(m):
            if time.time() - t_start > time_limit:
                break
            # Optimize x[i] while keeping others fixed
            x_save = x[i]
            # The objective as a function of x[i] is the max of several quadratic functions
            # Each corr_k is quadratic in x[i] (at most quadratic since h[i] and h[i+m] both depend on x[i])
            # Try a few candidate values
            candidates = [0.0, 1.0, x_save, 
                         x_save + 0.5, x_save - 0.5,
                         0.1, 0.9, 0.2, 0.8, 0.3, 0.7, 0.4, 0.6]
            candidates = [c for c in candidates if 0.0 <= c <= 1.0]
            # Also try values near current
            for delta in [0.01, 0.05, 0.1, 0.2]:
                candidates.append(x_save + delta)
                candidates.append(x_save - delta)
            candidates = [c for c in candidates if 0.0 <= c <= 1.0]
            
            best_c = x_save
            best_s = best_score
            for c in candidates:
                x[i] = c
                h = build_anti(x)
                s = official_score(h)
                if s < best_s - 1e-14:
                    best_s = s
                    best_c = c
            x[i] = best_c
            if best_c != x_save:
                h = build_anti(x)
                s = official_score(h)
                if s < best_score - 1e-14:
                    best_score = s
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    
    x = best_h[:m]
    return x, best_score

def binary_local_search(h, rng, max_rounds=300, time_limit=30, t_start=None):
    """Local search on full h: try swaps."""
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        # Find positions to swap
        high = np.where(h > 0.6)[0]
        low = np.where(h < 0.4)[0]
        if len(high) == 0 or len(low) == 0:
            break
        
        best_improve = 0
        best_swap = None
        # Sample pairs
        n_samples = min(100, len(high)*len(low))
        for _ in range(n_samples):
            i = high[rng.integers(len(high))]
            j = low[rng.integers(len(low))]
            if i == j:
                continue
            delta = min(h[i], 1.0-h[j])
            if delta < 0.1:
                continue
            h[i] -= delta
            h[j] += delta
            s = official_score(h)
            imp = best_score - s
            if imp > best_improve:
                best_improve = imp
                best_swap = (i, j, delta)
                best_h = h.copy()
            h[i] += delta
            h[j] -= delta
        
        if best_swap is not None:
            h = best_h
            best_score = best_score - best_improve
        else:
            break
    
    return h, best_score

def subgradient_anti(x0, max_iter=2000, time_limit=30, t_start=None):
    """Subgradient method for the max in anti-symmetric parameterization."""
    x = x0.copy()
    m = len(x)
    best_x = x.copy()
    best_score = official_score(build_anti(x))
    alpha = 0.05
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        h = build_anti(x)
        corr = np.correlate(h, 1.0-h, mode='full') * (2.0/len(h))
        cmax = np.max(corr)
        threshold = cmax - 0.05
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Average gradient over active lags
        grad = np.zeros(m)
        for k in active:
            # Compute gradient of corr[k] w.r.t. x
            n = len(h)
            s_k = k - (n-1)
            df = np.zeros(n)
            if s_k >= 0:
                if s_k == 0:
                    df = 1.0 - 2.0*h
                else:
                    df[s_k:] += (1.0 - h[:n-s_k])
                    df[:n-s_k] += -h[s_k:]
            else:
                t = -s_k
                df[:n-t] += (1.0 - h[t:])
                df[t:] += -h[:n-t]
            grad += df[:m] - df[m:]
        grad /= len(active)
        
        x_new = np.clip(x - alpha * grad, 0.0, 1.0)
        score = official_score(build_anti(x_new))
        if score < best_score - 1e-14:
            best_score = score
            best_x = x_new.copy()
            x = x_new
            alpha = min(alpha * 1.02, 0.3)
        else:
            x = x_new
            alpha = max(alpha * 0.98, 0.01)
    
    return best_x, best_score

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -n + 2.0, n + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-14:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)


def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def lse_full_obj(h):

def lse_full_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    beta = 2000.0
    shifted = np.clip(beta * (corr - cmax), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = e / s
    grad = np.zeros(n)

        if w < 1e-14:
            continue
        s_k = k - (n - 1)

            if s_k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:

                grad += w * (1.0 - 2.0 * h)
            else:
                grad[:n - t] += w * (1.0 - h[t:])
                grad[t:] += -w * h[:n - t]
    return grad

def subgradient_anti(x0, max_iter=3000, time_limit=30, t_start=None):
    x = x0.copy()
    m = len(x)
    best_x = x.copy()
    best_score = official_score(build_anti(x))
    alpha = 0.03
    patience = 0
    max_patience = 400

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        h = build_anti(x)
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        threshold = cmax - max(0.01, 0.03 * cmax)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(m)
        for k in active:
            s_k = k - (n - 1)
            df = np.zeros(n)
            if s_k >= 0:
                if s_k == 0:
                    df = 1.0 - 2.0 * h
                else:
                    df[s_k:] += (1.0 - h[:n - s_k])
                    df[:n - s_k] += -h[s_k:]
            else:
                t = -s_k
                if t == 0:
                    df = 1.0 - 2.0 * h
                else:
                    df[:n - t] += (1.0 - h[t:])
                    df[t:] += -h[:n - t]
            grad += df[:m] - df[m:]
        grad /= len(active)

        x_new = np.clip(x - alpha * grad, 0.0, 1.0)
        score = official_score(build_anti(x_new))
        if score < best_score - 1e-14:
            best_score = score
            best_x = x_new.copy()
            alpha = min(alpha * 1.01, 0.2)
            patience = 0
        else:
            patience += 1
            alpha = max(alpha * 0.99, 0.005)
        x = x_new
        if patience > max_patience:
            break

    return best_x, best_score


def coord_descent_anti(x, n_rounds=80, time_limit=25, t_start=None):
    m = len(x)
    x = x.copy()
    best_score = official_score(build_anti(x))
    best_x = x.copy()

    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(m):
            if time.time() - t_start > time_limit:
                break
            x_save = x[i]
            candidates = [0.0, 1.0, x_save]
            for d in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                candidates.append(x_save + d)
                candidates.append(x_save - d)
            for c in [0.05, 0.1, 0.15, 0.2, 0.3, 0.35, 0.5, 0.65, 0.7, 0.8, 0.85, 0.9, 0.95]:
                if abs(c - x_save) > 0.01:
                    candidates.append(c)
            candidates = [c for c in candidates if 0.0 <= c <= 1.0]

            best_c = x_save
            best_s = best_score
            for c in candidates:
                x[i] = c
                s = official_score(build_anti(x))
                if s < best_s - 1e-14:
                    best_s = s
                    best_c = c
            x[i] = best_c
            if best_c != x_save:
                s = official_score(build_anti(x))
                if s < best_score - 1e-14:
                    best_score = s
                    best_x = x.copy()
                    improved = True
        if not improved:
            break

    return best_x, best_score


def binary_local_search(h, rng, max_rounds=300, time_limit=20, t_start=None):
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        high = np.where(h > 0.5)[0]
        low = np.where(h < 0.5)[0]
        if len(high) == 0 or len(low) == 0:
            break

        best_improve = 0
        best_swap = None
        best_h_try = None
        n_samples = min(150, len(high) * len(low))
        for _ in range(n_samples):
            i = high[rng.integers(len(high))]
            j = low[rng.integers(len(low))]
            if i == j:
                continue
            delta = min(h[i], 1.0 - h[j])
            if delta < 0.05:
                continue
            h[i] -= delta
            h[j] += delta
            s = official_score(h)
            imp = best_score - s
            if imp > best_improve:
                best_improve = imp
                best_swap = (i, j, delta)
                best_h_try = h.copy()
            h[i] += delta
            h[j] -= delta

        if best_swap is not None:
            h = best_h_try
            best_score = official_score(h)
            best_h = h.copy()
        else:
            break

    return best_h, best_score


def sa_anti(x0, rng, n_iters=10000, time_limit=25, t_start=None):
    x = x0.copy()
    m = len(x)
    best_x = x.copy()
    best_score = official_score(build_anti(x))
    T = 0.03
    T_end = 0.002
    cool = (T_end / T) ** (1.0 / max(n_iters, 1))

    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        i = rng.integers(m)
        delta = rng.uniform(0.05, 0.5)
        direction = 1.0 if rng.random() < 0.5 else -1.0
        old = x[i]
        x[i] = np.clip(old + direction * delta, 0.0, 1.0)
        if x[i] == old:
            T *= cool
            continue
        new_score = official_score(build_anti(x))
        if new_score < best_score:
            best_score = new_score
            best_x = x.copy()
        else:
            delta_obj = new_score - best_score
            if rng.random() > np.exp(-delta_obj / max(T, 1e-10)):
                x[i] = old
        T *= cool

    return best_x, best_score


def full_slsqp_polish(h0, maxiter=120):
    n = len(h0)
    h0 = make_feasible(h0)
    try:
        res = minimize(
            lse_full_obj, h0,
            jac=lse_full_grad,

        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

                if fs < cur_score - 1e-13:
                    cur_h = h_full
                    cur_score = fs

            # Phase 7: Binary local search
            if time.time() < deadline - 5:
                h_bs, bs_score = binary_local_search(
                    cur_h, rng, max_rounds=250, time_limit=15, t_start=time.time())
                if bs_score < cur_score - 1e-13:
                    cur_h = h_bs
                    cur_score = bs_score

            # Phase 8: Final SLSQP after binary search
            if time.time() < deadline - 5 and cur_score < best_score:
                h_fin = full_slsqp_polish(cur_h, maxiter=80)
                ff = official_score(h_fin)
                if ff < cur_score - 1e-13:
                    cur_h = h_fin
                    cur_score = ff

            if cur_score < best_score - 1e-13:
                best_h = cur_h.copy()
                best_score = cur_score

    # Final refinement loop
    remaining = deadline - time.time()
    if remaining > 15:
        n_trials = min(8, int(remaining / 18))
        for trial in range(n_trials):
            if time.time() > deadline - 8:
                break
            n_b = len(best_h)
            pert = make_feasible(best_h + 0.03 * rng.standard_normal(n_b))
            h_pol = full_slsqp_polish(pert, maxiter=100)
            s_pol = official_score(h_pol)
            if s_pol < best_score - 1e-13:
                best_h = h_pol
                best_score = s_pol

            # Try anti-symmetric version of the perturbation
            if n_b % 2 == 0:
                x_pert = np.clip(pert[:n_b // 2], 0, 1)
                x_cd, cd_s = coord_descent_anti(
                    x_pert, n_rounds=60, time_limit=10, t_start=time.time())
                if cd_s < best_score - 1e-13:
                    h_new = full_slsqp_polish(build_anti(x_cd), maxiter=80)
                    s_new = official_score(h_new)
                    if s_new < best_score - 1e-13:
                        best_h = h_new
                        best_score = s_new

    # Final polish
    if time.time() < deadline - 3:
        best_h = full_slsqp_polish(best_h, maxiter=150)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
