def coordinate_descent_step(h, i):
    """Find optimal h[i] given all other coordinates fixed."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation contributions
    # c[k] for k = 0, ..., 2n-2
    # c[k] = sum over valid (a,b) of h[a]*(1-h[b]) where a-b = k-(n-1)
    
    # For each lag, compute:
    # A_k = sum of (1-h[b]) for pairs where a=i
    # B_k = sum of h[a] for pairs where b=i
    # C_k = contribution from pairs not involving i
    
    n_lags = 2*n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    C = np.zeros(n_lags)
    
    for k in range(n_lags):
        d = k - (n-1)  # lag
        if d >= 0:
            # c[k] = sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])
            for j in range(n-d):
                a = j + d
                b = j
                if a == i:
                    A[k] += (1 - h[b])
                elif b == i:
                    B[k] += h[a]
                else:
                    C[k] += h[a]*(1-h[b])
        else:
            d2 = -d
            # c[k] = sum_{j=d2}^{n-1} h[j]*(1-h[j-d2])
            for j in range(d2, n):
                a = j
                b = j - d2
                if a == i:
                    A[k] += (1 - h[b])
                elif b == i:
                    B[k] += h[a]
                else:
                    C[k] += h[a]*(1-h[b])
    
    # f(h[i]) = max_k (C[k] + B[k] + h[i]*(A[k] - B[k])) * dx
    # = max_k (C[k] + B[k] + h[i]*D[k]) * dx where D[k] = A[k] - B[k]
    D = A - B
    base = C + B
    
    # f(h[i])/dx = max_k (base[k] + h[i]*D[k])
    # This is convex piecewise linear in h[i]
    # Minimum is at a breakpoint where two lines cross, or at boundary 0 or 1
    
    # Find all breakpoints in [0,1]
    breakpoints = [0.0, 1.0]
    for k1 in range(n_lags):
        for k2 in range(k1+1, n_lags):
            if D[k1] != D[k2]:
                x = (base[k2] - base[k1]) / (D[k1] - D[k2])
                if 0 <= x <= 1:
                    breakpoints.append(x)
    
    breakpoints = sorted(set(breakpoints))
    
    best_x = 0
    best_val = np.inf
    for x in breakpoints:
        val = np.max(base + x * D) * dx
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x, best_val

        return b
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0

                    return h, best_score
            h[start:start+w] = current_block
            if best_block is not None:

                best_score = best_local_score
                improved = True

def exact_score(h):
