    z = np.asarray(z, dtype=float).ravel().copy()

- flip_off[i, d_idx] = change in corr at lag d if h[i] goes from 1 to 0
  = -(1-h[i+d]) if 0<=i+d<n, else 0, for the "a=i" term
  + h[i-d] if 0<=i-d<n, else 0, for the "a=i-d" term

- flip_on[j, d_idx] = change in corr at lag d if h[j] goes from 0 to 1
  = +(1-h[j+d]) if 0<=j+d<n, else 0
  + (-h[j-d]) if 0<=j-d<n, else 0

Then delta(i,j) = flip_off[i] + flip_on[j]

This is a great optimization! Precompute two (n, 2n-1) matrices, then each pair evaluation is just an array addition.

For n=50: matrix size is 50 × 99 = 4950 elements. Two matrices = ~10k elements. Very manageable.

And the pair evaluation is: new_corr = corr + flip_off[i] + flip_on[j], then check max. That's O(n) but with very small constant (just array addition and max).

Even better: I can precompute the max more cleverly. If I know the current max value M and the set of indices where corr=d_max, I can check:
- For those indices, does the delta decrease the value?
- For all other indices, does corr + delta stay below M?

from scipy.optimize import minimize
import time

def score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.clip(z, -5.0, 5.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def precompute_flip_off(h, n):
    """For each position i, compute the change in correlation if h[i] goes 1->0.
    Returns array of shape (n, 2n-1)."""
    d = np.arange(-(n-1), n)  # shape (2n-1,)
    flip_off = np.zeros((n, 2*n-1))
    for i in range(n):
        if h[i] < 0.5:
            continue  # only meaningful if h[i]=1
        ip = i + d
        valid_ip = (ip >= 0) & (ip < n)
        flip_off[i, valid_ip] = -(1.0 - h[ip[valid_ip]])
        im = i - d
        valid_im = (im >= 0) & (im < n)
        flip_off[i, valid_im] += h[im[valid_im]]
    return flip_off

def precompute_flip_on(h, n):
    """For each position j, compute the change in correlation if h[j] goes 0->1.
    Returns array of shape (n, 2n-1)."""
    d = np.arange(-(n-1), n)
    flip_on = np.zeros((n, 2*n-1))
    for j in range(n):
        if h[j] > 0.5:
            continue
        jp = j + d
        valid_jp = (jp >= 0) & (jp < n)
        flip_on[j, valid_jp] = (1.0 - h[jp[valid_jp]])
        jm = j - d
        valid_jm = (jm >= 0) & (jm < n)
        flip_on[j, valid_jm] += -h[jm[valid_jm]]
    return flip_on

def fast_binary_local_search(h0, rng, max_rounds=2000, time_limit=60, t_start=None):
    """Fast greedy local search for binary sequences using precomputed flip deltas."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Fix sum constraint: exactly n/2 ones (if n even)
    n_ones = int(round(sum(h)))
    target_ones = n // 2 if n % 2 == 0 else n // 2
    # For odd n, we handle separately
    if n % 2 == 0:
        if n_ones > target_ones:
            # Remove excess ones
            excess = n_ones - target_ones
            for i in np.where(h > 0.5)[0][-excess:]:
                h[i] = 0.0
        elif n_ones < target_ones:
            deficit = target_ones - n_ones
            for i in np.where(h < 0.5)[0][-deficit:]:
                h[i] = 1.0
    
    corr = compute_corr(h)
    current_max = float(np.max(corr))
    
    for rnd in range(max_rounds):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute flip deltas
        flip_off = precompute_flip_off(h, n)
        flip_on = precompute_flip_on(h, n)
        
        best_improve = 0.0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                delta = flip_off[i] + flip_on[j]
                new_max = float(np.max(corr + delta))
                improve = current_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            corr = compute_corr(h)
            current_max = float(np.max(corr))
        else:
            break
    
    return h, current_max * dx

```python
def fix_binary_count(h, n):
    """Ensure exactly n//2 ones in a binary array."""
    target = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones > target:
        excess = n_ones - target
        idx = np.where(h > 0.5)[0]
        h[idx[-excess:]] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        idx = np.where(h < 0.5)[0]
        h[idx[:deficit]] = 1.0
    return h

def make_periodic(period, n):
    """Create a periodic binary pattern with the given period."""
    # Pattern: first half of period is 1, second half is 0
    half = period // 2
    pattern = np.zeros(period)
    pattern[:half] = 1.0
    h = np.tile(pattern, n // period + 1)[:n]
    return h
```
