def compute_smooth_gradient(h, temp):
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    M = np.max(corr)
    exp_vals = np.exp((corr - M) / temp)
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        tau = k - (n - 1)
        # First term: w[k] * (1 - h[j+tau]) for valid j
        # j+tau valid: 0 <= j+tau < n => -tau <= j < n-tau
        j_lo = max(0, -tau)
        j_hi = min(n, n - tau)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w[k] * (1.0 - h[j_lo + tau : j_hi + tau])

        # j-tau valid: 0 <= j-tau < n => tau <= j < n+tau

        j_hi2 = min(n, n + tau)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w[k] * h[j_lo2 - tau : j_hi2 - tau]
    
    return grad

        j_lo = max(0, -tau)
        j_hi = min(n, n - tau)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wk * (1.0 - h[j_lo + tau:j_hi + tau])
        # Second term: -wk * h[j-tau] for 0 <= j-tau < n
        j_lo2 = max(0, tau)
        j_hi2 = min(n, n + tau)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wk * h[j_lo2 - tau:j_hi2 - tau]
    return grad

def slsqp_smooth_step(h, temp, maxiter=50, ftol=1e-10):
    n = len(h)
    
    def obj(x):
        return smooth_value(x, temp)
    
    def jac(x):
        return smooth_gradient(x, temp)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
    if res.success or res.x is not None:
        return res.x
    return h

def pair_line_search(h, i, j, n_iter=40):
    """Ternary search for optimal delta to transfer from h[i] to h[j]."""
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)  # h[i]-delta >= 0, h[j]+delta <= 1
    hi = min(hi_v, 1.0 - hj_v)   # h[i]-delta <= 1 (always true if hi_v<=1), h[j]+delta >= 0
    # Actually: h[i]-delta >= 0 => delta <= hi_v; h[j]+delta <= 1 => delta <= 1-hj_v
    # h[i]-delta <= 1 => delta >= hi_v-1 (always true); h[j]+delta >= 0 => delta >= -hj_v
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-15:
        return 0.0
    
    a, b = lo, hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    delta = (a + b) / 2.0
    return delta

def full_pair_sweep(h, rng, n_iter=32):
    n = len(h)
    perm = rng.permutation(n)
    improved = False
    for a in range(n):
        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            delta = pair_line_search(h, i, j, n_iter)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
                improved = True
    return improved
