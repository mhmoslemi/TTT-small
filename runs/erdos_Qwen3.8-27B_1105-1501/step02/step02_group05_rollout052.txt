import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, p):
    """Compute gradient of lag p of correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        j = np.arange(m)
        grad[j] += (1.0 - h[j + s])
        grad[j + s] -= h[j]
    else:
        t = -s
        m = n - t
        j = np.arange(m)
        grad[j + t] += (1.0 - h[j])
        grad[j] -= h[j + t]
    return dx * grad

def lp_active_step(h, rho=0.15, top_k=32, eta=0.006):
    """Solve LP for active lag minimax direction"""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    C = vals.max()
    
    # Select active lags
    sorted_idx = np.argsort(vals)[::-1]
    active = []
    for p in sorted_idx:
        if len(active) >= top_k:
            break
        if vals[p] >= C - eta:
            active.append(p)
    if not active:
        active = [int(np.argmax(vals))]
    active = np.array(active)
    
    # Build gradients
    G = np.array([lag_gradient(h, p) for p in active])  # shape (|A|, n)
    c = vals[active]  # shape (|A|,)
    
    # LP: minimize t
    # vars: u_0,...,u_{n-1}, t
    # constraints: G @ u - t <= -c  (for each active lag)
    #              sum(u) = 0
    # bounds: max(-h_i, -rho) <= u_i <= min(1-h_i, rho)
    #         t unbounded (or <= C for safety)
    
    n_vars = n + 1
    # Objective: minimize t -> c_obj has 1 in last position
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    # Inequality constraints: G @ u - t <= -c
    # A_ub shape: (len(active), n+1)
    A_ub = np.zeros((len(active), n_vars))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_i = max(-h[i], -rho)
        hi_i = min(1.0 - h[i], rho)
        bounds.append((lo_i, hi_i))
    bounds.append((None, C))  # t <= C (for safety; t will naturally be <= C)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            u = res.x[:n]
            if np.max(np.abs(u)) > 1e-12:
                return u, C
    except Exception:
        pass
    return None, C

def lp_refine(h, rng, max_iters=200, is_dead=None):
    """Iteratively apply LP active step with line search"""
    if is_dead is None:
        is_dead = lambda: False
    n = len(h)
    best = h.copy()
    best_score = score_exact(best)
    rho = 0.15
    eta = 0.006
    stall = 0
    
    for it in range(max_iters):
        if is_dead():
            break
        u, C = lp_active_step(best, rho=rho, top_k=32, eta=eta)
        if u is None:
            # Try smaller eta
            u, C = lp_active_step(best, rho=rho, top_k=32, eta=eta*0.5)
        if u is None:
            rho *= 0.7
            if rho < 0.005:
                break
            continue
        
        # Line search
        improved = False
        for alpha in [1.0, 0.75, 0.55, 0.4, 0.28, 0.19, 0.12, 0.07, 0.035]:
            trial = make_feasible(best + alpha * u)
            sc = score_exact(trial)
            if sc < best_score - 1e-12:
                best = trial
                best_score = sc
                improved = True
                break
        
        if improved:
            stall = 0
            rho = min(0.25, rho * 1.1)
            eta = max(0.001, eta * 0.95)
        else:
            stall += 1
            rho *= 0.8
            eta *= 1.2
            if stall > 30:
                break
    
    return best, best_score

import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            j = np.arange(m)
            grad[j + s] += (1.0 - h[j])
            grad[j] -= h[j + s]
    else:
        t = -s
        m = n - t
        if m > 0:
            j = np.arange(m)
            grad[j] += (1.0 - h[j + t])
            grad[j + t] -= h[j]
    return dx * grad

def lp_active_step(h, rho=0.15, top_k=32, eta=0.006):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    C = vals.max()
    
    # Select active lags
    sorted_idx = np.argsort(vals)[::-1]
    active = []
    for p in sorted_idx:
        if len(active) >= top_k:
            break
        if vals[p] >= C - eta:
            active.append(int(p))
    if not active:
        active = [int(np.argmax(vals))]
    active_arr = np.array(active)
    
    # Build gradients
    G = np.vstack([lag_grad(h, p) for p in active])
    c = vals[active_arr]
    
    # LP setup
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((len(active), n_vars))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo_i = max(-h[i], -rho)
        hi_i = min(1.0 - h[i], rho)
        bounds.append((lo_i, hi_i))
    bounds.append((None, C + 1.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            u = res.x[:n]
            if np.max(np.abs(u)) > 1e-10:
                return u, C
    except Exception:
        pass
    return None, C

def lp_refine(h0, max_iters=500, is_dead=None):
    if is_dead is None:
        is_dead = lambda: False
    n = len(h0)
    best = np.asarray(h0, dtype=float).copy()
    best_score = score_exact(best)
    rho = 0.15
    eta = 0.006
    top_k = 32
    stall = 0
    best_rho = rho
    best_eta = eta
    
    for it in range(max_iters):
        if is_dead():
            break
        u, C = lp_active_step(best, rho=rho, top_k=top_k, eta=eta)
        if u is None:
            # Try smaller eta
            u, C = lp_active_step(best, rho=rho, top_k=top_k, eta=max(0.001, eta * 0.5))
        if u is None:
            # Try larger eta
            u, C = lp_active_step(best, rho=rho*1.5, top_k=min(2*n-1, top_k+8), eta=eta*2)
        if u is None:
            stall += 1
            rho *= 0.8
            if rho < 0.003:
                rho = 0.15
                eta *= 1.3
                if eta > 0.05:
                    break
            continue
        
        # Line search
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.07, 0.04]:
            trial = make_feasible(best + alpha * u)
            sc = score_exact(trial)
            if sc < best_score - 1e-12:
                best = trial
                best_score = sc
                improved = True
                break
        
        if improved:
            stall = 0
            rho = min(0.25, rho * 1.05)
            eta = max(0.001, eta * 0.98)
        else:
            stall += 1
            rho *= 0.85
            if stall > 20:
                # Reset with broader search
                rho = 0.2
                eta = min(0.05, eta * 1.5)
                stall = 0
                if eta > 0.05:
                    break
    
    return best, best_score

def binary_sa(h0, max_iters=5000, is_dead=None):
    if is_dead is None:
        is_dead = lambda: False
    n = len(h0)
    if n % 2 != 0:
        return make_feasible(h0), score_exact(make_feasible(h0))
    target = n // 2
    # Initialize from thresholding
    h = np.asarray(h0, dtype=float)
    bits = (h > 0.5).astype(int)
    # Balance
    cur_sum = int(bits.sum())
    if cur_sum > target:
        # Remove excess 1s: choose positions closest to 0.5
        one_pos = np.where(bits == 1)[0]
        dist = np.abs(h[one_pos] - 0.5)
        order = np.argsort(dist)
        for k in range(cur_sum - target):
            bits[one_pos[order[k]]] = 0
    elif cur_sum < target:
        zero_pos = np.where(bits == 0)[0]
        dist = np.abs(1.0 - h[zero_pos] - 0.5)  # distance of (1-h) from 0.5 = h from 0.5
        dist = np.abs(h[zero_pos] - 0.5)
        order = np.argsort(dist)
        for k in range(target - cur_sum):
            bits[zero_pos[order[k]]] = 1
    
    current = bits.astype(float).copy()
    current_score = score_exact(current)
    best = current.copy()
    best_score = current_score
    
    T = 5e-4
    cool = 0.999
    pos1 = np.where(bits == 1)[0]
    pos0 = np.where(bits == 0)[0]
    
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if is_dead():
            break
        if len(pos1) == 0 or len(pos0) == 0:
            break
        i = pos1[rng.integers(len(pos1))]
        j = pos0[rng.integers(len(pos0))]
        
        # Swap
        bits[i] = 0
        bits[j] = 1
        trial = bits.astype(float)
        sc = score_exact(trial)
        
        delta = sc - current_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_score = sc
            # Update position arrays
            pos1 = np.where(bits == 1)[0]
            pos0 = np.where(bits == 0)[0]
            if sc < best_score:
                best = trial.copy()
                best_score = sc
        else:
            # Reject
            bits[i] = 1
            bits[j] = 0
        
        T *= cool
    
    return best, best_score

def two_point_search(h0, n_tries, rng, is_dead):

    best = h0.copy()

    for it in range(n_tries):

            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            tmp[a] = val_a
            tmp[b] = val_b
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0.0, 1.0)
            best[b] = np.clip(best[b] - best_t, 0.0, 1.0)
            best_sc = score_exact(best)
    return best, best_sc
