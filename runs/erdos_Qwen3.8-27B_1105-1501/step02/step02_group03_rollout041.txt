    z = np.asarray(z, dtype=float).copy()

    if abs(s - target) < 1e-12:
        return z
    # Bisection on additive shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)

    residual = target - np.sum(h)
    if residual > 0:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0: break

    """Compute soft max of overlap values and its gradient w.r.t. u"""
    vals = compute_vals(h_full)

    n = len(z)

    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-14:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0:
                break
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-15:
                break
    elif residual < -1e-14:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 0:
                break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def antisym_project(h):
    n = len(h)
    m = n // 2
    h_s = np.empty(n)
    for i in range(m):
        val = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
        val = np.clip(val, 0.0, 1.0)
        h_s[i] = val
        h_s[n - 1 - i] = 1.0 - val
    if n % 2 == 1:
        h_s[m] = 0.5
    return h_s

def build_h_from_u(u, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

def build_u_from_h(h):
    n = len(h)
    m = n // 2
    return h[:m].copy()

def compute_vals_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_full_matrix(h):
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for p in range(K):
        lag = p - (n - 1)
        if lag >= 0:
            L = n - lag
            G[p, lag:] += dx * (1.0 - h[:L])
            G[p, :L] += -dx * h[lag:]
        else:
            L = -lag
            K2 = n - L
            G[p, :K2] += dx * (1.0 - h[L:L + K2])
            G[p, L:] += -dx * h[:K2]
    return G

def grad_reduced_matrix(G_full):
    n = G_full.shape[1]
    m = n // 2
    idx_r = np.arange(n - 1, n - 1 - m, -1)
    return G_full[:, :m] - G_full[:, idx_r]

def soft_obj(u, n, beta):
    h = build_h_from_u(u, n)
    vals = compute_vals_arr(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def soft_grad(u, n, beta):
    h = build_h_from_u(u, n)
    vals = compute_vals_arr(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    w /= w.sum()
    G = grad_full_matrix(h)
    G_red = grad_reduced_matrix(G)
    return w @ G_red

def find_active(vals, frac=0.2, min_count=5):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 5e-5)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def minimax_obj(x, n, active):
    return x[-1]

def minimax_grad(x, n, active):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def minimax_ineq(x, n, active):
    u = x[:-1]
    t = x[-1]
    h = build_h_from_u(u, n)
    vals = compute_vals_arr(h)
    return t - vals[active]

def minimax_ineq_jac(x, n, active):
    u = x[:-1]
    m = len(u)
    h = build_h_from_u(u, n)
    G = grad_full_matrix(h)
    G_red = grad_reduced_matrix(G)
    J = -G_red[active, :]
    J_full = np.hstack([J, np.ones((len(active), 1))])
    return J_full

def active_set_solve(u0, n, active, max_iter=300):
    m = len(u0)
    h0 = build_h_from_u(u0, n)
    t0 = float(compute_vals_arr(h0).max()) + 1e-8
    x0 = np.concatenate([u0, [t0]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 3.0)]
    cons = [
        {'type': 'ineq', 'fun': minimax_ineq, 'jac': minimax_ineq_jac, 'args': (n, active)}
    ]
    try:
        res = minimize(minimax_obj, x0, jac=minimax_grad, args=(n, active),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            u_new = np.clip(res.x[:m], 0.0, 1.0)
            h_new = build_h_from_u(u_new, n)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return u_new, sc
    except Exception:
        pass
    return None, None

def soft_solve(u0, n, beta, time_limit=20.0):
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    t0 = time.time()
    try:
        res = minimize(soft_obj, u0, jac=soft_grad, args=(n, beta),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 500, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': 2000})
        if time.time() - t0 > time_limit * 2:
            return u0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return u0

def interpolate_u(u_src, n_src, n_tgt):
    m_src = n_src // 2
    m_tgt = n_tgt // 2
    if m_src == m_tgt:
        return u_src.copy()
    src_x = np.linspace(0.0, 1.0, m_src)
    tgt_x = np.linspace(0.0, 1.0, m_tgt)
    u_tgt = np.interp(tgt_x, src_x, u_src)
    return np.clip(u_tgt, 0.0, 1.0)

def full_dim_soft_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    vals = compute_vals_arr(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def subgradient_step(h, step_size):
    vals = compute_vals_arr(h)
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-5)
    active = np.where(vals >= threshold)[0]
    G = grad_full_matrix(h)
    g = G[active].mean(axis=0)
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-15:
        return h
    return make_feasible(h - step_size * g / gnorm)

def subgradient_reduced_step(u, n, step_size):
    h = build_h_from_u(u, n)
    vals = compute_vals_arr(h)
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-5)
    active = np.where(vals >= threshold)[0]
    G = grad_full_matrix(h)
    G_red = grad_reduced_matrix(G)
    g = G_red[active].mean(axis=0)
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-15:
        return u
    u_new = u - step_size * g / gnorm
    return np.clip(u_new, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def elapsed():
        return time.time() - t_start

    def time_left():
        return deadline - time.time()

    def check_deadline():
        return time.time() >= deadline

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
        elif abs(sc - best_score) <= 1e-14 and best_h is not None and len(h) < len(best_h):
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            best_h = parent.copy()
            best_score = official_score(parent)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        h_init = np.full(n0, 0.5)
        for i in range(n0):
            h_init[i] = 0.5 + 0.35 * np.sin(2 * np.pi * i / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    update(best_h, best_score)

    # Project to anti-symmetric
    n_parent = len(best_h)
    if n_parent % 2 == 1:
        n_parent = n_parent + 1
    # Interpolate parent to even length if needed
    if len(best_h) != n_parent:
        src_x = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
        tgt_x = (np.arange(n_parent) + 0.5) * (2.0 / n_parent)
        h_int = np.interp(tgt_x, src_x, best_h)
        h_int = make_feasible(h_int)
        sc = official_score(h_int)
        update(h_int, sc)

    # Phase 1: Anti-symmetric soft optimization with annealing
    n = len(best_h)
    h_as = antisym_project(best_h)
    sc_as = official_score(h_as)
    if sc_as < best_score - 1e-14:
        update(h_as, sc_as)
    u = build_u_from_h(h_as)

    betas = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000]
    for beta in betas:
        if check_deadline() or time_left() < 5:
            break
        tl = min(time_left() * 0.25, 25.0)
        if tl < 2:
            break
        u = soft_solve(u, n, beta, time_limit=tl)
        h_new = build_h_from_u(u, n)
        sc = official_score(h_new)
        if sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 2: Active-set SLSQP refinement
    if time_left() > 20:
        h_cur = best_h.copy()
        if len(h_cur) % 2 == 0:
            h_as2 = antisym_project(h_cur)
            sc_as2 = official_score(h_as2)
            if sc_as2 < best_score - 1e-14:
                update(h_as2, sc_as2)
            h_cur = h_as2
            n_cur = len(h_cur)
            u_cur = build_u_from_h(h_cur)
        else:
            n_cur = len(h_cur)
            u_cur = None

        for round_num in range(5):
            if check_deadline() or time_left() < 10:
                break
            h_eval = build_h_from_u(u_cur, n_cur) if u_cur is not None else h_cur
            vals = compute_vals_arr(h_eval)
            active = find_active(vals, frac=0.15, min_count=4)
            if u_cur is not None:
                u_new, sc = active_set_solve(u_cur, n_cur, active, max_iter=400)
                if u_new is not None and sc < best_score - 1e-14:
                    update(build_h_from_u(u_new, n_cur), sc)
                    u_cur = u_new
                # Try with broader active set
                if time_left() > 15:
                    vals2 = compute_vals_arr(build_h_from_u(u_cur, n_cur))
                    active2 = find_active(vals2, frac=0.3, min_count=6)
                    if active2 != active:
                        u_new2, sc2 = active_set_solve(u_cur, n_cur, active2, max_iter=300)
                        if u_new2 is not None and sc2 < best_score - 1e-14:
                            update(build_h_from_u(u_new2, n_cur), sc2)
                            u_cur = u_new2

    # Phase 3: More aggressive soft optimization from current best
    if time_left() > 60:
        h_as3 = antisym_project(best_h)
        u3 = build_u_from_h(h_as3)
        n3 = len(h_as3)
        for beta in [2000, 5000, 10000, 20000]:
            if check_deadline() or time_left() < 5:
                break
            tl = min(time_left() * 0.2, 20.0)
            if tl < 2:
                break
            u3 = soft_solve(u3, n3, beta, time_limit=tl)
            h_new = build_h_from_u(u3, n3)
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                update(h_new, sc)

    # Phase 4: Active-set rounds from improved point
    if time_left() > 30:
        h_as4 = antisym_project(best_h)
        u4 = build_u_from_h(h_as4)
        n4 = len(h_as4)
        for round_num in range(6):
            if check_deadline() or time_left() < 10:
                break
            vals = compute_vals_arr(build_h_from_u(u4, n4))
            active = find_active(vals, frac=0.25, min_count=5)
            u_new, sc = active_set_solve(u4, n4, active, max_iter=500)
            if u_new is not None and sc < best_score - 1e-14:
                update(build_h_from_u(u_new, n4), sc)
                u4 = u_new
            else:
                # Try different active set selection
                vals2 = compute_vals_arr(build_h_from_u(u4, n4))
                cmax = vals2.max()
                thr = cmax - 2e-4
                active_alt = sorted(np.where(vals2 >= thr)[0].tolist())
                if len(active_alt) >= 3 and active_alt != active:
                    u_new2, sc2 = active_set_solve(u4, n4, active_alt, max_iter=300)
                    if u_new2 is not None and sc2 < best_score - 1e-14:
                        update(build_h_from_u(u_new2, n4), sc2)
                        u4 = u_new2

    # Phase 5: Try nearby resolutions
    if time_left() > 60:
        n_base = len(best_h)
        n_opts = []
        for dn in [-4, -2, 2, 4, 6]:
            nn = n_base + dn
            if nn >= 40 and nn <= 90 and nn % 2 == 0:
                n_opts.append(nn)
        n_opts = sorted(set(n_opts))

        for n_opt in n_opts:
            if check_deadline() or time_left() < 15:
                break
            u_src = build_u_from_h(antisym_project(best_h))
            u_opt = interpolate_u(u_src, len(best_h), n_opt)
            h_opt = build_h_from_u(u_opt, n_opt)
            sc_opt = official_score(h_opt)
            # Soft optimize
            for beta in [500, 2000, 5000]:
                if check_deadline() or time_left() < 5:
                    break
                tl = min(time_left() * 0.2, 15.0)
                if tl < 2:
                    break
                u_opt = soft_solve(u_opt, n_opt, beta, time_limit=tl)
                h_new = build_h_from_u(u_opt, n_opt)
                sc = official_score(h_new)
                if sc < best_score - 1e-14:
                    update(h_new, sc)
            # Active refine
            if time_left() > 10:
                vals_opt = compute_vals_arr(build_h_from_u(u_opt, n_opt))
                active_opt = find_active(vals_opt, frac=0.2, min_count=4)
                u_new, sc = active_set_solve(u_opt, n_opt, active_opt, max_iter=300)
                if u_new is not None and sc < best_score - 1e-14:
                    update(build_h_from_u(u_new, n_opt), sc)

    # Phase 6: Random restarts in reduced space
    if time_left() > 40:
        n_r = len(best_h)
        h_as_r = antisym_project(best_h)
        u_r = build_u_from_h(h_as_r)
        sigmas = [0.05, 0.1, 0.2, 0.3, 0.15]
        n_starts = max(2, min(8, int(time_left() / 20)))
        for i in range(n_starts):
            if check_deadline() or time_left() < 10:
                break
            sigma = sigmas[i % len(sigmas)]
            u_pert = np.clip(u_r + rng.normal(0, sigma, size=len(u_r)), 0.0, 1.0)
            h_pert = build_h_from_u(u_pert, n_r)
            sc_pert = official_score(h_pert)
            if sc_pert < best_score - 1e-14:
                update(h_pert, sc_pert)
            # Quick soft polish
            if time_left() > 8:
                u_pert = soft_solve(u_pert, n_r, 3000, time_limit=min(time_left() * 0.15, 10.0))
                h_new = build_h_from_u(u_pert, n_r)
                sc = official_score(h_new)
                if sc < best_score - 1e-14:
                    update(h_new, sc)

    # Phase 7: Subgradient polishing on full h
    if time_left() > 30:
        h_sub = best_h.copy()
        step = 0.05
        for it in range(300):
            if check_deadline():
                break
            h_sub = subgradient_step(h_sub, step)
            sc_sub = official_score(h_sub)
            if sc_sub < best_score - 1e-14:
                update(h_sub, sc_sub)
            step *= 0.97

    # Phase 8: Final intensive polish
    if time_left() > 25:
        h_final_as = antisym_project(best_h)
        u_fin = build_u_from_h(h_final_as)
        n_fin = len(h_final_as)
        # Very high beta soft
        for beta in [5000, 15000, 30000]:
            if check_deadline() or time_left() < 5:
                break
            tl = min(time_left() * 0.3, 15.0)
            if tl < 2:
                break
            u_fin = soft_solve(u_fin, n_fin, beta, time_limit=tl)
            h_new = build_h_from_u(u_fin, n_fin)
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                update(h_new, sc)
        # Final active set
        if time_left() > 10:
            vals_fin = compute_vals_arr(build_h_from_u(u_fin, n_fin))
            active_fin = find_active(vals_fin, frac=0.3, min_count=5)
            u_new, sc = active_set_solve(u_fin, n_fin, active_fin, max_iter=600)
            if u_new is not None and sc < best_score - 1e-14:
                update(build_h_from_u(u_new, n_fin), sc)
            # One more with tighter active set
            if time_left() > 10:
                vals_fin2 = compute_vals_arr(build_h_from_u(u_fin, n_fin))
                cmax2 = vals_fin2.max()
                active_tight = sorted(np.where(vals_fin2 >= cmax2 - 1e-4)[0].tolist())
                if len(active_tight) >= 3:
                    u_new2, sc2 = active_set_solve(u_fin, n_fin, active_tight, max_iter=400)
                    if u_new2 is not None and sc2 < best_score - 1e-14:
                        update(build_h_from_u(u_new2, n_fin), sc2)

    # Phase 9: Try full-space SLSQP (non-anti-symmetric)
    if time_left() > 30:
        n_f = len(best_h)
        vals_f = compute_vals_arr(best_h)
        active_f = find_active(vals_f, frac=0.15, min_count=5)

        class FullMinimaxSolver:
            def __init__(self, n, active):
                self.n = n
                self.active = active
            def obj(self, x):
                return x[-1]
            def grad(self, x):
                g = np.zeros(len(x))
                g[-1] = 1.0
                return g
            def ineq(self, x):
                h = np.clip(x[:-1], 0.0, 1.0)
                t = x[-1]
                vals = compute_vals_arr(h)
                return t - vals[self.active]
            def ineq_jac(self, x):
                h = np.clip(x[:-1], 0.0, 1.0)
                G = grad_full_matrix(h)
                J = -G[self.active, :]
                return np.hstack([J, np.ones((len(self.active), 1))])
            def eq(self, x):
                return np.sum(x[:-1]) - self.n / 2.0
            def eq_jac(self, x):
                g = np.zeros(len(x))
                g[:-1] = 1.0
                return g

        solver = FullMinimaxSolver(n_f, active_f)
        t0_f = float(vals_f.max()) + 1e-8
        x0_f = np.concatenate([best_h, [t0_f]])
        bounds_f = [(0.0, 1.0)] * n_f + [(0.0, 3.0)]
        cons_f = [
            {'type': 'ineq', 'fun': solver.ineq, 'jac': solver.ineq_jac},
            {'type': 'eq', 'fun': solver.eq, 'jac': solver.eq_jac}
        ]
        try:
            res_f = minimize(solver.obj, x0_f, jac=solver.grad,
                             method='SLSQP', bounds=bounds_f, constraints=cons_f,
                             options={'maxiter': 300, 'ftol': 1e-16})
            if res_f.x is not None and np.all(np.isfinite(res_f.x)):
                h_f = make_feasible(res_f.x[:n_f])
                sc_f = official_score(h_f)
                if sc_f < best_score - 1e-14:
                    update(h_f, sc_f)
        except Exception:
            pass

    # Final validation and repair
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Ensure validity
    final_h = np.clip(final_h, 0.0, 1.0)
    sum_check = np.sum(final_h)
    if abs(sum_check - final_n / 2.0) > 1e-10:
        final_h = make_feasible(final_h)
        final_score = official_score(final_h)

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
