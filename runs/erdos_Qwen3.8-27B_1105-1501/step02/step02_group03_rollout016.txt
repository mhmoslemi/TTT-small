def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def make_feasible(z):

    z = np.asarray(z, dtype=float).copy()

    lo, hi = -11.0, 11.0
    for _ in range(60):

            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14: break

def compute_overlap_vals(h):
    """Compute all overlap values: V_p = dx * sum h[i]*(1-h[i+p]) for valid shifts"""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return vals

            #   j-s >= 0 means j >= s, and j < L+S = n (automatic since j < n)
            #   But also need j-s < L, i.e., j < L+s = n (automatic)
            #   And j-s >= 0, i.e., j >= s

def antisym_embed(z, n):
    """Embed z (length m=n//2) into antisymmetric h of length n"""
    m = n // 2
    h = np.zeros(n)
    for i in range(m):
        h[i] = z[i]
        h[n - 1 - i] = 1.0 - z[i]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def antisym_proj(h, n):
    """Project h onto antisymmetric subspace, return z vector"""
    m = n // 2
    z = np.zeros(m)
    for i in range(m):
        z[i] = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
    return z

def solve_antisym_nlp(z0, n, t0, maxiter=300, time_limit=30.0):
    """Solve min t s.t. V_p(antisym_embed(z)) <= t for all p, 0<=z<=1"""
    m = n // 2
    t_start = time.time()
    
    def obj(z_t):
        return z_t[-1]
    
    def obj_grad(z_t):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def con_fun(z_t):
        z = z_t[:m]
        t = z_t[-1]
        h = antisym_embed(z, n)
        # Clip to be safe
        h = np.clip(h, 0.0, 1.0)
        vals = compute_overlap_vals(h)
        return t - vals  # >= 0 means feasible
    
    def con_jac(z_t):
        z = z_t[:m]
        t = z_t[-1]
        h = antisym_embed(z, n)
        h = np.clip(h, 0.0, 1.0)
        G = grad_overlaps(h)  # shape (K, n)
        # dV_p/dz_j = dV_p/dh_j + dV_p/dh_{n-1-j} * (-1)
        # (since h[n-1-j] = 1 - z[j], dh[n-1-j]/dz[j] = -1)
        K = 2 * n - 1
        J = np.zeros((K, m))
        # For j in 0..m-1:
        #   contribution to h[j] is +1, to h[n-1-j] is -1
        #   So dV_p/dz_j = G[p, j] - G[p, n-1-j]
        for j in range(m):
            J[:, j] = G[:, j] - G[:, n - 1 - j]
        # Full Jacobian: [J, -I] for constraint t - V_p >= 0
        # Actually constraint is t - V_p >= 0, so gradient is [-dV_p/dz, 1]
        # SLSQP expects constraint fun >= 0
        Jac = np.zeros((K, m + 1))
        Jac[:, :m] = -J  # -dV_p/dz_j
        Jac[:, m] = 1.0   # dt/dt = 1
        return Jac
    
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    
    zt0 = np.concatenate([z0, [t0]])
    
    res = minimize(obj, zt0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = antisym_embed(res.x[:m], n)
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        target = n / 2.0
        residual = target - np.sum(h_new)
        if abs(residual) > 1e-12:
            h_new += residual / n
            h_new = np.clip(h_new, 0.0, 1.0)
        sc = official_score(h_new)
        if np.isfinite(sc):
            return h_new, sc, res
    return None, None, None

def pg_solve_antisym(z0, n, beta, time_limit=15.0):
    """Projected gradient on the antisymmetric subspace using log-sum-exp smoothing"""
    m = n // 2
    z = z0.copy()
    z = np.clip(z, 1e-6, 1.0 - 1e-6)
    t0 = time.time()
    
    def smooth_obj(z):
        h = antisym_embed(z, n)
        vals = compute_overlap_vals(h)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        logsum = np.log(np.sum(np.exp(shifted)))
        return cmax + logsum / beta
    
    def smooth_grad(z):
        h = antisym_embed(z, n)
        vals = compute_overlap_vals(h)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(vals)) / len(vals)
        else:
            w /= ws
        G = grad_overlaps(h)
        # Gradient w.r.t. z: dJ/dz_j = sum_p w_p * (G[p,j] - G[p,n-1-j])
        grad_h = w @ G  # shape (n,)
        grad_z = np.zeros(m)
        for j in range(m):
            grad_z[j] = grad_h[j] - grad_h[n - 1 - j]
        return grad_z
    
    f = smooth_obj(z)
    g = smooth_grad(z)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-28:
        return z
    
    step = 1.0 / (gnorm2 + 1e-12)
    step = min(step, 2.0 / (np.max(g**2) + 1e-12))
    
    for it in range(5000):
        if time.time() - t0 > time_limit:
            break
        g = smooth_grad(z)
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-28:
            break
        s = step
        for _ in range(30):
            z_trial = np.clip(z - s * g, 1e-6, 1.0 - 1e-6)
            f_trial = smooth_obj(z_trial)
            if f_trial <= f - 1e-4 * s * gnorm2:
                break
            s *= 0.4
        if s < 1e-14:
            break
        z = np.clip(z - s * g, 1e-6, 1.0 - 1e-6)
        f_new = smooth_obj(z)
        if not np.isfinite(f_new):
            break
        if f_new < f:
            step = min(step * 1.5, 5.0 / (gnorm2 + 1e-12))
        f = f_new
    
    return z

            w_t = np.exp(sh_t - sh_t.max())

            if f_trial <= f_now - 1e-4 * s * gnorm2:
                break
            s *= 0.4
        if s < 1e-14:
            break
        z = np.clip(z - s * gz, 1e-8, 1.0 - 1e-8)
    
    return z

def coord_descent_antisym(z0, n, time_limit=10.0):
    """Coordinate descent on the antisymmetric subspace"""
    m = n // 2
    z = z0.copy()
    t0 = time.time()
    order = np.arange(m)
    
    for outer in range(20):
        if time.time() - t0 > time_limit:
            break
        np.random.default_rng(outer).shuffle(order)
        improved = False
        for j in order:
            if time.time() - t0 > time_limit:
                break
            
            def f_1d(t_val):
                z[j] = t_val
                h = antisym_embed(z, n)
                vals = compute_vals(h)
                return vals.max()
            
            # Save original
            z_orig = z[j]
            res = minimize_scalar(f_1d, bounds=(1e-8, 1.0-1e-8),
                                  method='bounded', options={'xatol': 1e-10})
            z[j] = res.x
            if res.fun < f_1d(z_orig) - 1e-14:
                improved = True
            else:
                z[j] = z_orig  # No improvement, revert
        if not improved:
            break
    
    return z
