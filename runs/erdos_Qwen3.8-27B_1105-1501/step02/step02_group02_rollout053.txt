def _grad_lag(x, s):
    """Gradient of F_{n-1+s}(x) = sum of h[i+s]*(1-h[i]) * (2/n)."""
    n = len(x)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i + s] += dx * (1.0 - x[i])
            g[i] += dx * (-x[i + s])
    else:
        for i in range(-s, n):
            g[i + s] += dx * (1.0 - x[i])
            g[i] += dx * (-x[i + s])
    return g

def _grad_active(x, active_lags):
    """Gradient of the max approximated by weighted average of active lags."""
    n = len(x)
    cv = _corr_vec(x)
    vals = [cv[k] for k in active_lags]
    vmax = max(vals)
    # Weight by proximity to max
    eps = 1e-6
    weights = [((v - min(vals)) / (vmax - min(vals) + eps)) ** 2 for v in vals]
    wsum = sum(weights)
    if wsum < 1e-10:
        weights = [1.0/len(active_lags)] * len(active_lags)
        wsum = 1.0
    else:
        weights = [w/wsum for w in weights]
    
    g = np.zeros(n)
    for ii, k in enumerate(active_lags):
        s = k - (n - 1)
        g += weights[ii] * _grad_lag(x, s)
    return g

def _binary_push(h, t=0.5):

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

            s = k - (n-1)
            g += _grad_lag(y, s)

        t_prev = it

        h_cur = h_new
        y = h_cur + alpha * (h_cur - h_prev)
        y = _project(y)
        
        if it % 200 == 0:
            s = _score(h_cur)
            if s < best_s:

    def fun(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta

    def jac(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        return _grad_all_lags(x, w)

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _slsqp_active(h0, active_lags, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    k = len(active_lags)

    def fun(x):
        cv = _corr_vec(x)
        return np.mean([cv[j] for j in active_lags])

    def jac(x):
        cv = _corr_vec(x)
        vals = np.array([cv[j] for j in active_lags])
        vrange = vals.max() - vals.min()
        if vrange < 1e-12:
            w = np.full(k, 1.0 / k)
        else:
            w = (vals - vals.min()) / vrange
            w = w ** 2
            w = w / w.sum()
        g = np.zeros(n)
        for ii, j in enumerate(active_lags):
            s = j - (n - 1)
            g += w[ii] * _grad_lag_vec(x, s)
        return g

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _adaptive_active(h0, max_rounds=5, topk=4, maxiter=80):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)

    for rnd in range(max_rounds):
        cv = _corr_vec(cur)
        idx = np.argsort(cv)[::-1]
        active = list(idx[:topk])

        new = _slsqp_active(cur, active, maxiter=maxiter)
        new_s = _score(new)

        if new_s < cur_s - 1e-14:
            cur = new
            cur_s = new_s

        cv2 = _corr_vec(cur)
        idx2 = np.argsort(cv2)[::-1]
        if list(idx2[:topk]) == active:
            active6 = list(idx2[:min(6, 2 * n - 1)])
            new = _slsqp_active(cur, active6, maxiter=maxiter // 2)
            new_s = _score(new)
            if new_s < cur_s - 1e-14:
                cur = new
                cur_s = new_s
            break

    return cur, cur_s


def _coord_descent(h0, max_sweeps=10):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)

    for sweep in range(max_sweeps):
        improved = False
        for p in range(n):
            old_hp = cur[p]
            best_hp = old_hp
            best_val = cur_s
            for hp in np.linspace(0.0, 1.0, 11):
                if abs(hp - old_hp) < 1e-4:
                    continue
                cur[p] = hp
                v = _score(cur)
                if v < best_val - 1e-14:
                    best_val = v
                    best_hp = hp
            cur[p] = best_hp
            if best_hp != old_hp:
                improved = True
                cur_s = best_val
        if not improved:
            break
        cur = _project(cur)
        cur_s = _score(cur)

    return cur, cur_s


def _fista_subgrad(h0, n_iter=5000, alpha0=0.004):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)

    for it in range(n_iter):
        cv = _corr_vec(y)
        max_val = cv.max()
        tol = 1e-8
        active = np.where(cv >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(cv)])
        g = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            g += _grad_lag_vec(y, s)
        g /= len(active)

        tt = it + 1
        t_prev = it
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tt * tt))
        alpha = t_prev / t_new

        step = alpha0 / (1.0 + 0.01 * it)
        y_new = y + alpha * (h_cur - h_prev)
        h_new = _project(y_new - step * g)

        h_prev = h_cur
        h_cur = h_new
        y = h_new

        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()

    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    return best_h, best_s


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _binary_round(h0):
    n = len(h0)
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def _slsqp_direct_minimax(h0, maxiter=150):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = min(cv.max() + 0.003, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def fun(z):
        return z[-1]

    def fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv

    def eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0

    def eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(fun, z0, jac=fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': ieq},
                                    {'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(57) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)
    rng = np.random.default_rng(seed)

    # Phase 1: Screen n values with quick optimization
    n_list = sorted(set(list(range(max(48, parent_n - 5), min(72, parent_n + 7) + 1, 2))))
    n_scores = []

    for n in n_list:
        if time.time() > deadline - 80:
            break
        if time.time() - t_start > (budget_s - 80) * 0.35 and n < parent_n:
            pass

        h_interp = _interp(parent, n)
        s0 = _score(h_interp)

        # Quick LSE pass
        if time.time() < deadline - 50:
            h1 = _slsqp_lse(h_interp, beta=60, maxiter=80)
            s1 = _score(h1)
            if s1 < s0:
                h_interp = h1
                s0 = s1

        # Quick active set
        if time.time() < deadline - 30:
            h2, s2 = _adaptive_active(h_interp, max_rounds=3, topk=4, maxiter=50)
            if s2 < s0:
                h_interp = h2
                s0 = s2

        n_scores.append((s0, n, h_interp.copy()))

        if s0 < best_score - 1e-14:
            best_h = h_interp.copy()
            best_score = s0
            best_n = n

    # Sort by score, focus on top candidates
    n_scores.sort(key=lambda x: x[0])

    # Phase 2: Focused optimization on top 3 n values
    for s0, n, h_base in n_scores[:3]:
        if time.time() > deadline - 40:
            break

        # Generate diverse starts
        starts = [h_base.copy()]
        starts.append(_project(h_base[::-1].copy()))
        for sigma in [0.05, 0.12, 0.2]:
            starts.append(_project(h_base + sigma * rng.standard_normal(n)))
        # Binary round then smooth
        h_bin = _binary_round(h_base)
        h_smooth = _project(h_bin + 0.1 * rng.standard_normal(n))
        starts.append(h_smooth)

        for h0 in starts:
            if time.time() > deadline - 25:
                break

            # Pipeline: LSE(med) → active → LSE(high) → coord
            h1 = _slsqp_lse(h0, beta=80, maxiter=100)
            s1 = _score(h1)

            if time.time() < deadline - 20:
                h2, s2 = _adaptive_active(h1, max_rounds=4, topk=4, maxiter=70)
                if s2 < s1:
                    h1, s1 = h2, s2

            if time.time() < deadline - 15:
                h3 = _slsqp_lse(h1, beta=400, maxiter=100)
                s3 = _score(h3)
                if s3 < s1:
                    h1, s1 = h3, s3

            if time.time() < deadline - 10:
                h4, s4 = _coord_descent(h1, max_sweeps=6)
                if s4 < s1:
                    h1, s1 = h4, s4

            if s1 < best_score - 1e-14:
                best_h = h1.copy()
                best_score = s1
                best_n = n

    # Phase 3: Intensive refinement on overall best
    if time.time() < deadline - 20:
        # Multiple adaptive active rounds
        for _ in range(4):
            if time.time() > deadline - 15:
                break
            h_new, s_new = _adaptive_active(best_h, max_rounds=6, topk=5, maxiter=90)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new

    if time.time() < deadline - 15:
        # FISTA with exact subgradient
        remaining = deadline - time.time()
        n_iter = min(15000, max(2000, int(remaining * 800)))
        h_new, s_new = _fista_subgrad(best_h, n_iter=n_iter, alpha0=0.003)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new

    if time.time() < deadline - 10:
        # Direct min-max formulation
        h_new = _slsqp_direct_minimax(best_h, maxiter=120)
        s_new = _score(h_new)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new

    if time.time() < deadline - 8:
        # Coordinate descent final polish
        h_new, s_new = _coord_descent(best_h, max_sweeps=8)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new

    # Phase 4: Small perturbations + re-optimize (escape local min)
    remaining = deadline - time.time()
    if remaining > 30:
        n_pert = min(8, int(remaining / 35))
        for _ in range(n_pert):
            if time.time() > deadline - 10:
                break
            h_p = _project(best_h + 0.02 * rng.standard_normal(best_n))
            h1 = _slsqp_lse(h_p, beta=200, maxiter=80)
            s1 = _score(h1)
            if time.time() < deadline - 8:
                h2, s2 = _adaptive_active(h1, max_rounds=3, topk=4, maxiter=50)
                if s2 < s1:
                    h1, s1 = h2, s2
            if s1 < best_score - 1e-14:
                best_h = h1.copy()
                best_score = s1
                best_n = len(h1)

    # Final polish
    if time.time() < deadline - 5:
        h_new = _slsqp_lse(best_h, beta=1000, maxiter=150)
        s_new = _score(h_new)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new

    if time.time() < deadline - 3:
        h_new, s_new = _coord_descent(best_h, max_sweeps=5)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new

    best_h = _project(best_h)
    final_score = _score(best_h)
    return (best_h, final_score, len(best_h))
