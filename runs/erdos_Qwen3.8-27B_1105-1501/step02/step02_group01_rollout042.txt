def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]

            else:

            if abs(add) < 1e-15:

            h[i] += add
            residual -= add

    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

    """Compute full correlation h with (1-h), scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_corr_h(h):
    """Compute gradient of all correlations w.r.t. h. Returns array of shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        k = s + (n-1)  # index in full correlation
        lo = max(0, -s)
        hi = min(n, n - s)
        if s == 0:

        else:

            for i in range(lo, hi):
                grad[k, i] += (1.0 - h[i+s]) * dx
                grad[k, i+s] += (-h[i]) * dx
    return grad

def sigmoid_lbfgs(z0, beta, maxiter=200):
    n = len(z0)
    
    def objective(z):
        h = h_from_z(z, n)
        corr = compute_corr(h)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            return m
        return m + np.log(s) / beta
    
    def gradient(z):
        h = h_from_z(z, n)
        corr = compute_corr(h)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            weights = np.ones(2*n-1) / (2*n-1)
        else:
            weights = e / s
        
        # Compute g = weighted gradient in h-space
        g = np.zeros(n)
        dx = 2.0 / n
        for k in range(2*n-1):
            w = weights[k] * dx
            if w < 1e-14:
                continue
            s_shift = k - (n-1)
            lo = max(0, -s_shift)
            hi = min(n, n - s_shift)
            for i in range(lo, hi):
                g[i] += w * (1.0 - h[i + s_shift])
                g[i + s_shift] += w * (-h[i])
        
        # Convert to z-gradient
        v = h * (1.0 - h)
        V = np.sum(v)
        if V < 1e-15:
            return g * v
        B = np.sum(g * v) / V
        return v * (g - B)
    
    result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    
    if result.success or hasattr(result, 'x'):
        return make_feasible(h_from_z(result.x, n))
    return make_feasible(h_from_z(z0, n))

                h[i] = save_i - amt
                h[j] = save_j + amt
                s = official_score(h)
                if s < best_score:
                    best_score = s
                    improved = True
                else:
                    h[i], h[j] = save_i, save_j
            if not improved:
                break
    return h, best_score

def targeted_repair(h, rng, max_iters=3000, time_limit=30, t_start=None):
    """Target the max correlation lag specifically."""
    h = h.copy()
    n = len(h)
    best_score = official_score(h)
    
    for it in range(max_iters):
        if time_start_exceeded(t_start, time_limit):
            break
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        s = max_idx - (n - 1)
        
        # Find positions where h[i]=1 and h[i+s]=0 (or vice versa) to reduce this lag
        if s >= 0:
            conflict_10 = [i for i in range(n - s) if h[i] > 0.3 and h[i + s] < 0.7]
            conflict_01 = [i for i in range(n - s) if h[i] < 0.7 and h[i + s] > 0.3]
        else:
            m = -s
            conflict_10 = [i + m for i in range(n - m) if h[i + m] > 0.3 and h[i] < 0.7]
            conflict_01 = [i for i in range(n - m) if h[i] < 0.7 and h[i + m] > 0.3]
        
        # Try to flip one of the conflict positions
        candidates = conflict_10[:20] + conflict_01[:20]
        if not candidates:
            # Random perturbation
            idx = rng.integers(n)
            h[idx] += 0.05 * (1 if h[idx] < 0.5 else -1)
            h = make_feasible(h)
            continue
        
        best_gain = 0
        best_move = None
        for pos in candidates:
            # Try moving this position toward the "other" value
            if h[pos] > 0.5:
                amt = h[pos]
            else:
                amt = 1.0 - h[pos]
            save = h[pos]
            h[pos] = save - amt
            # Compensate: move a random opposite position
            comp_candidates = np.where((h > 0.5) != (save > 0.5))[0]
            if len(comp_candidates) == 0:
                h[pos] = save
                continue
            comp = comp_candidates[rng.integers(len(comp_candidates))]
            h[comp] += amt
            h = make_feasible(h)
            new_score = official_score(h)
            gain = best_score - new_score
            if gain > best_gain:
                best_gain = gain
                best_move = (pos, comp, amt, save)
            # Undo
            h[pos] = save
            h[comp] -= amt
            h = make_feasible(h)
        
        if best_move is not None:
            pos, comp, amt, save = best_move
            h[pos] = save - amt
            h[comp] += amt
            h = make_feasible(h)
            new_score = official_score(h)
            if new_score < best_score:
                best_score = new_score
            else:
                h = make_feasible(h)
        else:
            break
    return h, best_score

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)
    
    best_h = parent.copy()
    best_score = official_score(parent)
    best_n = len(best_h)
    
    rng = np.random.default_rng(seed)
    
    def time_ok(margin=20):
        return time.time() < deadline - margin
    
    n_list = [48, 50, 51, 52, 54, 56, 60, 64]
    
    for n in n_list:
        if not time_ok(80):
            break
        
        starts = []
        h_interp = make_feasible(interp_to_n(parent, n))
        starts.append(('interp', h_interp))
        
        # Thue-Morse
        tm = thue_morse(n)
        tm = make_feasible(tm)
        starts.append(('tm', tm))
        
        # Random binary
        for _ in range(3):
            rb = np.zeros(n)
            n_ones = n // 2
            idx = rng.choice(n, n_ones, replace=False)
            rb[idx] = 1.0
            if n % 2 == 1:
                rb[rng.integers(n)] = 0.5
                rb = make_feasible(rb)
            starts.append(('rand', rb.copy()))
        
        # Binary push of interpolated
        for p in [1.5, 3.0]:
            starts.append(('push', push_binary(h_interp, p)))
        
        # Thue-Morse shifted variants
        for shift in [1, 3]:
            tm2 = thue_morse(n)
            tm2 = np.roll(tm2, shift)
            tm2 = make_feasible(tm2)
            starts.append(('tm_shift', tm2))
        
        for label, h0 in starts:
            if not time_ok(30):
                break
            cur_h = h0.copy()
            cur_score = official_score(cur_h)
            
            # Phase 1: Sigmoid L-BFGS with continuation
            z = logit(cur_h)
            for beta in [20, 50, 150, 400]:
                if not time_ok(20):
                    break
                new_h = sigmoid_lbfgs(z, beta, maxiter=200)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-12:
                    cur_score = new_score
                    cur_h = new_h
                    z = logit(cur_h)
            
            # Phase 2: Adam refinement
            if time_ok(30):
                z = logit(cur_h)
                adam_h, adam_score = adam_optimize(z, 300, lr=0.05, max_iter=3000,
                                                   time_limit=30, t_start=time.time())
                if adam_score < cur_score - 1e-12:
                    cur_score = adam_score
                    cur_h = adam_h
            
            # Phase 3: Binary local search
            if time_ok(25):
                h_bin = to_binary(cur_h, n)
                bls_h, bls_score = binary_local_search(h_bin, rng, max_rounds=1500,
                                                       time_limit=35, t_start=time.time())
                if bls_score < cur_score - 1e-12:
                    cur_score = bls_score
                    cur_h = bls_h
            
            # Phase 4: L-BFGS from binary solution
            if time_ok(20):
                z = logit(make_feasible(cur_h + 1e-3 * rng.standard_normal(n)))
                for beta in [100, 300]:
                    if not time_ok(15):
                        break
                    new_h = sigmoid_lbfgs(z, beta, maxiter=200)
                    new_score = official_score(new_h)
                    if new_score < cur_score - 1e-12:
                        cur_score = new_score
                        cur_h = new_h
            
            # Phase 5: Targeted repair
            if time_ok(15):
                tr_h, tr_score = targeted_repair(cur_h, rng, max_iters=2000,
                                                 time_limit=20, t_start=time.time())
                if tr_score < cur_score - 1e-12:
                    cur_score = tr_score
                    cur_h = tr_h
            
            # Phase 6: Another L-BFGS polish
            if time_ok(15):
                z = logit(cur_h)
                new_h = sigmoid_lbfgs(z, 500, maxiter=150)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-12:
                    cur_score = new_score
                    cur_h = new_h
            
            # Update best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_h = cur_h.copy()
                best_n = n
    
    # Final refinement on the best solution
    if time_ok(30):
        n_best = len(best_h)
        z = logit(best_h)
        # L-BFGS with very high beta
        for beta in [1000, 2000]:
            if not time_ok(10):
                break
            new_h = sigmoid_lbfgs(z, beta, maxiter=300)
            new_score = official_score(new_h)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_h = new_h
                z = logit(best_h)
    
    # Final binary search
    if time_ok(20):
        n_best = len(best_h)
        h_bin = to_binary(best_h, n_best)
        bls_h, bls_score = binary_local_search(h_bin, rng, max_rounds=3000,
                                               time_limit=25, t_start=time.time())
        if bls_score < best_score - 1e-12:
            best_score = bls_score
            best_h = bls_h
    
    # Final L-BFGS
    if time_ok(10):
        z = logit(best_h)
        new_h = sigmoid_lbfgs(z, 1000, maxiter=200)
        new_score = official_score(new_h)
        if new_score < best_score:
            best_score = new_score
            best_h = new_h
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
