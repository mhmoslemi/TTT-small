import numpy as np
from scipy.optimize import minimize
import time

def _score_tau(h, d):
    """Compute score for lag d: sum h[n+d]*(1-h[n]) for d>=0, or h[n]*(1-h[n+d]) for d<0"""
    n = len(h)
    if d >= 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _all_scores(h):
    """Compute all scores matching np.correlate(h, 1-h, 'full')"""
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    """Gradient of score_tau w.r.t. h for lag d"""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m <= 0:
            return grad
        # h[j+d] * (1-h[j]) for j=0..m-1
        # d/dh[j] of this: -h[j+d] for j=0..m-1
        grad[:m] += -h[d:d+m]
        # h[j] * (1-h[j-d]) for j=d..n-1
        # d/dh[j] of this: (1-h[j-d]) for j=d..n-1
        grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m <= 0:
            return grad
        # h[j] * (1-h[j+g]) for j=0..m-1
        # d/dh[j] of this: (1-h[j+g]) for j=0..m-1
        grad[:m] += (1.0 - h[g:g+m])
        # h[j-g] * (1-h[j]) for j=g..n-1
        # d/dh[j] of this: -h[j-g] for j=g..n-1
        grad[g:] += -h[:m]
    return grad

    n = len(h_init)
    target = n / 2.0
    scores_init = _all_scores(h_init)
    z0 = float(np.max(scores_init)) + 1e-6
    
    # Variable: x = [h_0, ..., h_{n-1}, z]
    x0 = np.zeros(n + 1)
    x0[:n] = h_init
    x0[n] = z0

    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 1.0)]
    
    # Equality constraint: sum(h) = target
    cons_eq = ({'type': 'eq',
                'fun': lambda x: np.sum(x[:n]) - target,
                'jac': lambda x: np.concatenate([np.ones(n), [0.0]])})

    # We'll use all lags or top_k
    all_lags = list(range(-(n-1), n))

        # Sort by score descending, keep top_k
        sorted_lags = sorted(all_lags, key=lambda d: _score_tau(h_init, d), reverse=True)
        active_lags = sorted_lags[:top_k]

    constraints = [cons_eq]

        def _make_cons(d):

            }
        constraints.append(_make_cons(d))
    
    # Objective: minimize z
    def obj(x):
        return x[n]

        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def callback(xk):

    dx = 2.0 / n

    if d == 0:

    elif d > 0:
        m = n - d
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))

        m = n - g
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        grad[:m] += -h[d:d+m]
        grad[d:] += (1.0 - h[:m])

        g = -d
        m = n - g
        grad[:m] += (1.0 - h[g:g+m])

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
    f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
    for _ in range(20):
        if hi_a - lo_a < 1e-10 or time.time() > deadline:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))

    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(np.clip(h + opt_a * p, 0.0, 1.0))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break

        scores = _all_scores(h)
        p = _lp_direction(h, scores, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.85
            if radius < 1e-6:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5, deadline)
        if new_c5 < cur_c5 - 1e-12:
            h = np.clip(h + alpha * p, 0.0, 1.0)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.04

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_gradient_step(h, scores, beta=80.0):
    """Compute smooth max gradient direction"""
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        G[r] = _score_grad(h, d)
    d = G.T @ w
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return -d / norm

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP with broad active set
    deadline1 = t0 + budget_s * 0.20
    result = _slsqp_solve(h, deadline1, top_k=50, maxiter=120, ftol=1e-11)
    if result is not None:
        h_try, c5_try = result
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 2: LP loop for quick improvement
    deadline2 = t0 + budget_s * 0.35
    h_lp, c5_lp = _lp_loop(best_h, deadline2, radius_init=0.12, max_iters=3000)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: Multi-restart with SLSQP
    deadline3 = t0 + budget_s * 0.78
    scale = 0.06
    attempts = 0
    while time.time() < deadline3:
        attempts += 1
        if best_c5 < 0.3807:
            break
        # Adaptive scale
        scale = 0.06 * (0.92 ** min(attempts, 30))
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)

        per_budget = max(3.0, min(25.0, (deadline3 - time.time()) * 0.3))
        per_deadline = time.time() + per_budget

        result = _slsqp_solve(h_pert, per_deadline, top_k=30, maxiter=80, ftol=1e-10)
        if result is not None:
            h_try, c5_try = result
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.015, scale * 0.85)
            else:
                scale = min(0.15, scale * 1.1)
        else:
            # Fallback: try LP on perturbed
            h_lp2, c5_lp2 = _lp_loop(h_pert, time.time() + min(10.0, (deadline3 - time.time()) * 0.2),
                                       radius_init=0.08, max_iters=500)
            if c5_lp2 < best_c5 - 1e-12:
                best_c5 = c5_lp2
                best_h = h_lp2.copy()

    # Phase 4: SLSQP refinement with tight active set
    if time.time() < t0 + budget_s * 0.92:
        dl4 = t0 + budget_s * 0.94
        h_ref, c5_ref = _slsqp_refine(best_h, dl4, top_k=15, maxiter=250, ftol=1e-12)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 5: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.95:
        dl5 = t0 + budget_s * 0.97
        for dn in [-4, 4, -8, 8, -2, 2]:
            if time.time() > dl5:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 160 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _slsqp_solve(h_r, min(dl5, time.time() + 20.0),
                                      top_k=30, maxiter=60, ftol=1e-10)
            if h_r is not None:
                if c5_r < best_c5 - 1e-12:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = n_try

    # Phase 6: Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.995,
                              radius_init=0.03, max_iters=2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        # Projection degraded; try without projection if already feasible
        if abs(float(np.sum(best_h)) - best_n / 2.0) < 1e-6:
            pass
        else:
            # Use the pre-projection version if it was better
            pass

    return best_h, final_c5, best_n
