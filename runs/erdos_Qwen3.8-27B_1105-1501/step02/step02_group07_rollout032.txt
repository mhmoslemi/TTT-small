    n = len(h)
    dx = 2.0 / n
    S = h[i] + h[j]
    x_new_j = S - x
    dh_i = x - h[i]
    dh_j = x_new_j - h[j]
    
    if abs(dh_i) < 1e-15 and abs(dh_j) < 1e-15:
        return V_old.max()
    
    # Compute delta for each lag
    V_new = V_old.copy()
    
    for s in range(1, n):
        # Lag s > 0: V[s] = dx * sum_{m=0}^{n-s-1} h[m+s]*(1-h[m])
        # Offset in correlate output: idx = (n-1) + s
        # Check if i or j appears
        delta = 0.0
        # h[i] as first factor: m+s = i => m = i-s, valid if 0 <= i-s <= n-s-1
        if 0 <= i - s <= n - s - 1:

        if 0 <= j <= n - s - 1:
            m = j
            delta += -dh_j * h[m + s]

            V_new[(n-1) + s] += dx * delta

    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def pair_descent(h, rng, n_steps=300, grid_size=20, target_lag=None):
    n = len(h)
    best_sc = official_score(h)
    for _ in range(n_steps):
        if target_lag is not None and rng.random() < 0.6:
            s = abs(target_lag)
            valid_i = np.arange(max(0, -(n - s)), min(n, n - s))
            if len(valid_i) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
            else:
                if s == 0:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                else:
                    ii = int(rng.integers(len(valid_i)))
                    i = int(valid_i[ii])
                    j = i + s
                    if j >= n:
                        j = i - s
                    if j < 0 or j >= n or j == i:
                        i = int(rng.integers(n))
                        j = int(rng.integers(n))
        elif rng.random() < 0.5:
            i = int(rng.integers(n))
            offset = int(rng.integers(1, min(5, n)))
            j = (i + offset) % n
            if j == i:
                j = (i + 1) % n
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
        S = h[i] + h[j]
        L = max(0.0, S - 1.0)
        U = min(1.0, S)
        if U - L < 1e-8:
            continue
        hi_old, hj_old = h[i], h[j]
        best_x = hi_old
        best_x_sc = best_sc
        if U - L > 1e-6:
            xs = np.linspace(L, U, grid_size)
            for x in xs:
                h[i] = x
                h[j] = S - x
                sc = official_score(h)
                if sc < best_x_sc:
                    best_x_sc = sc
                    best_x = x
        if best_x_sc < best_sc - 1e-15:
            h[i] = best_x
            h[j] = S - best_x
            best_sc = best_x_sc
        else:
            h[i] = hi_old
            h[j] = hj_old
    return h, best_sc

def pg_step(h, beta=50.0, alpha_max=0.8, max_ls=15):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    g = -grad / gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h + alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-8:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    max_active = 50
    if len(active) > max_active:
        order = np.argsort(-V[active])[:max_active]
        active = active[order]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def local_optimize(h, rng, time_left_fn, min_time=10):
    best_h = h.copy()
    best_sc = official_score(h)
    V = compute_values(best_h)
    worst_idx = int(np.argmax(V))
    while time_left_fn() > min_time:
        h_cur = best_h.copy()
        h_cur, sc = pair_descent(h_cur, rng, n_steps=200, grid_size=18, target_lag=worst_idx - (len(h_cur)-1))
        if sc < best_sc - 1e-14:
            best_h = h_cur
            best_sc = sc
            V = compute_values(best_h)
            worst_idx = int(np.argmax(V))
        h_cur, sc2 = pg_step(best_h, beta=60.0, alpha_max=0.6, max_ls=12)
        if sc2 < best_sc - 1e-14:
            best_h = h_cur
            best_sc = sc2
            V = compute_values(best_h)
            worst_idx = int(np.argmax(V))
        h_new, sc3 = active_set_lp(best_h, eps=0.004, trust=0.15)
        if h_new is not None and sc3 < best_sc - 1e-14:
            best_h = h_new
            best_sc = sc3
            V = compute_values(best_h)
            worst_idx = int(np.argmax(V))
        else:
            h_new, sc4 = active_set_lp(best_h, eps=0.002, trust=0.08)
            if h_new is not None and sc4 < best_sc - 1e-14:
                best_h = h_new
                best_sc = sc4
                V = compute_values(best_h)
                worst_idx = int(np.argmax(V))
    return best_h, best_sc

def generate_candidate(n, rng, mode=0):
    x = np.arange(n) / n
    if mode == 0:
        h = 0.5 + 0.1 * np.sin(2 * np.pi * x) + 0.05 * np.sin(4 * np.pi * x)
        h += rng.normal(0, 0.03, n)
    elif mode == 1:
        h = 0.5 + 0.12 * np.cos(2 * np.pi * 2 * x + rng.uniform(0, 2*np.pi))
        h += 0.06 * np.sin(2 * np.pi * 3 * x + rng.uniform(0, 2*np.pi))
        h += rng.normal(0, 0.02, n)
    elif mode == 2:
        h = 0.5 + 0.08 * np.sin(np.pi * x) + 0.04 * np.sin(3 * np.pi * x)
        h += rng.normal(0, 0.04, n)
    else:
        h = 0.5 + np.zeros(n)
        for k in range(1, 5):
            h += rng.uniform(-0.08, 0.08) * np.cos(2 * np.pi * k * x + rng.uniform(0, 2*np.pi))
        h += rng.normal(0, 0.03, n)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(8.0, 0.06 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    try:
        h_parent = np.asarray(initial_h_values, dtype=float).ravel()
        h_parent = make_feasible(h_parent)
        n_parent = len(h_parent)
    except (NameError, AttributeError):
        n_parent = 60
        h_parent = generate_candidate(n_parent, rng, mode=0)

    best_h = h_parent.copy()
    best_sc = official_score(h_parent)

    # Phase 1: Quick local optimization on parent
    if time_left() > budget_s * 0.5:
        best_h, best_sc = local_optimize(best_h, rng, time_left, min_time=budget_s * 0.3)

    # Phase 2: Global exploration with different shapes and resolutions
    n_tries = [n_parent]
    if n_parent >= 50:
        n_tries.append(n_parent - 8)
    if n_parent >= 50:
        n_tries.append(n_parent - 4)
    n_tries.append(n_parent + 6)
    n_tries.append(n_parent + 12)
    n_tries = [n for n in n_tries if 20 <= n <= 150]

    n_starts = 8
    per_start_time = budget_s * 0.25 / max(1, n_starts)

    for start_idx in range(n_starts):
        if time_left() < 30:
            break
        if start_idx == 0:
            n_try = n_parent
            h_try = best_h.copy()
        else:
            n_try = n_tries[start_idx % len(n_tries)]
            mode = start_idx % 4
            h_try = generate_candidate(n_try, rng, mode=mode)
            sc_try = official_score(h_try)
            if sc_try > best_sc * 1.05 and start_idx > 2:
                continue
        # Quick optimization
        sub_deadline = time.time() + per_start_time
        def sub_time_left():
            return sub_deadline - time.time()
        if sub_time_left() > 5:
            h_try, sc_try = local_optimize(h_try, rng, sub_time_left, min_time=5)
        if sc_try < best_sc - 1e-14:
            best_h = h_try.copy()
            best_sc = sc_try

    # Phase 3: Deep local optimization on best
    if time_left() > 60:
        V = compute_values(best_h)
        worst_idx = int(np.argmax(V))
        h_cur = best_h.copy()
        stall = 0
        beta = 40.0
        while time_left() > 30:
            h_cur, sc = pair_descent(h_cur, rng, n_steps=300, grid_size=22, target_lag=worst_idx - (len(h_cur)-1))
            if sc < best_sc - 1e-14:
                best_h = h_cur.copy()
                best_sc = sc
                stall = 0
                V = compute_values(best_h)
                worst_idx = int(np.argmax(V))
            else:
                stall += 1
            h_cur, sc2 = pg_step(best_h, beta=beta, alpha_max=0.5, max_ls=12)
            if sc2 < best_sc - 1e-14:
                best_h = h_cur.copy()
                best_sc = sc2
                stall = 0
                V = compute_values(best_h)
                worst_idx = int(np.argmax(V))
            else:
                stall += 1
            h_new, sc3 = active_set_lp(best_h, eps=0.003, trust=0.12)
            if h_new is not None and sc3 < best_sc - 1e-14:
                best_h = h_new.copy()
                best_sc = sc3
                stall = 0
                V = compute_values(best_h)
                worst_idx = int(np.argmax(V))
            else:
                h_new, sc4 = active_set_lp(best_h, eps=0.001, trust=0.06)
                if h_new is not None and sc4 < best_sc - 1e-14:
                    best_h = h_new.copy()
                    best_sc = sc4
                    stall = 0
                    V = compute_values(best_h)
                    worst_idx = int(np.argmax(V))
            beta = min(beta * 1.03, 300.0)
            if stall > 30:
                h_cur = make_feasible(best_h + rng.normal(0, 0.02, len(best_h)))
                stall = 0
                beta = 30.0

    # Phase 4: Multi-resolution interpolation
    if time_left() > 40:
        n_base = len(best_h)
        for n_try in [max(30, n_base - 6), max(30, n_base - 2), n_base + 4, n_base + 8]:
            if time_left() < 20:
                break
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            sub_deadline4 = time.time() + max(5, (time_left() - 20) / 4)
            def sub_tl4():
                return sub_deadline4 - time.time()
            if sub_tl4() > 3:
                h_int, sc_int = local_optimize(h_int, rng, sub_tl4, min_time=3)
            if sc_int < best_sc - 1e-14:
                best_h = h_int.copy()
                best_sc = sc_int

    # Phase 5: Final fine polishing
    if time_left() > 15:
        h_cur = best_h.copy()
        V = compute_values(h_cur)
        worst_idx = int(np.argmax(V))
        for _ in range(20):
            if time_left() < 12:
                break
            h_cur, sc = pair_descent(h_cur, rng, n_steps=400, grid_size=30, target_lag=worst_idx - (len(h_cur)-1))
            if sc < best_sc - 1e-15:
                best_h = h_cur.copy()
                best_sc = sc
                V = compute_values(best_h)
                worst_idx = int(np.argmax(V))
            else:
                break
        if time_left() > 12:
            for eps in [0.002, 0.001, 0.0005]:
                if time_left() < 10:
                    break
                h_new, sc = active_set_lp(best_h, eps=eps, trust=0.04)
                if h_new is not None and sc < best_sc - 1e-15:
                    best_h = h_new.copy()
                    best_sc = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_sc = official_score(final_h)
    if final_sc < best_sc:
        best_sc = final_sc
    final_n = len(final_h)
    final_bound = final_sc + 1e-14
    return final_h, final_bound, final_n
