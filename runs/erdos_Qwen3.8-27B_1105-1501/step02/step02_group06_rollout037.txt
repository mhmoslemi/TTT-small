        if np.any(free):
            adj = residual / np.sum(free)

def target_step(h, k_active=5, tol=0.001):
    """Solve QP to target making top k active lags equal to their current max."""
    n = len(h)
    f = compute_f(h)  # array of length 2n-1
    C = f.max()
    
    # Select active lags
    threshold = C - tol
    active_indices = np.where(f > threshold)[0]
    if len(active_indices) > k_active:
        # Keep only top k
        sorted_idx = np.argsort(f[active_indices])[-k_active:]
        active_indices = active_indices[sorted_idx]
    
    if len(active_indices) == 0:
        active_indices = np.array([np.argmax(f)])
    
    # Compute gradients for active lags
    grads = []
    for p in active_indices:
        lag = p - (n - 1)
        grads.append(gradient_at_lag(h, lag))
    G = np.array(grads)  # shape (k, n)
    
    # Set up QP:
    # minimize ||h_new - h||^2
    # subject to: G @ h_new - C_new = G @ h - f[active_indices]
    #             sum(h_new) = n/2
    #             0 <= h_new <= 1
    
    a = f[active_indices]  # current values of active lags
    
    h_var = cp.Variable(n)
    c_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        G @ h_var - c_var == G @ h - a
    ]
    
    obj = cp.Minimize(cp.sum_squares(h_var - h))
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, C
    
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None, C
    
    h_new = make_feasible(h_new)
    return h_new, c_var.value

import numpy as np
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
            else:
                add = max(residual, h[idx] - 1.0)
            if abs(add) < 1e-15:
                break
            h[idx] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def gradient_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        s = lag
        if s == 0:
            return 1.0 - 2.0 * h
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        t = -lag
        g[:n - t] += 1.0 - h[t:]
        g[t:] -= h[:n - t]
    return g

def target_qp_step(h, k_active=5, tol=0.0):
    n = len(h)
    f = compute_f(h)
    C = f.max()
    threshold = C - tol
    active_mask = f >= threshold
    active_indices = np.where(active_mask)[0]
    if len(active_indices) > k_active:
        sorted_idx = np.argsort(f[active_indices])[-k_active:]
        active_indices = active_indices[sorted_idx]
    if len(active_indices) < 1:
        active_indices = np.array([np.argmax(f)])
    grads = np.array([gradient_at_lag(h, p - (n - 1)) for p in active_indices])
    a = f[active_indices]
    h_var = cp.Variable(n)
    c_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        grads @ h_var - c_var == grads @ h - a
    ]
    obj = cp.Minimize(cp.sum_squares(h_var - h))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, C
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None, C
    h_new = make_feasible(h_new)
    return h_new, float(c_var.value) if c_var.value is not None else C

def lp_direction(h, k_active=5):
    n = len(h)
    f = compute_f(h)
    C = f.max()
    top_k = np.argsort(f)[-k_active:]
    worst = top_k[0]
    worst_grad = gradient_at_lag(h, worst - (n - 1))
    other_grads = [gradient_at_lag(h, p - (n - 1)) for p in top_k[1:]]
    d = cp.Variable(n)
    constraints = [
        cp.sum(d) == 0,
        d >= -1.0,
        d <= 1.0
    ]
    for og in other_grads:
        constraints.append(og @ d <= 1e-6)
    obj = cp.Minimize(worst_grad @ d)
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate'] or d.value is None:
        return None
    d_val = np.asarray(d.value, dtype=float)
    if not np.all(np.isfinite(d_val)) or np.linalg.norm(d_val) < 1e-12:
        return None
    d_val = d_val / np.linalg.norm(d_val)
    return d_val

def line_search(h, direction, best_score, n_steps=12):
    n = len(h)
    for i in range(n_steps):
        alpha = 1.0 / (2.0 ** i)
        h_try = make_feasible(h + alpha * direction)
        sc = official_score(h_try)
        if sc < best_score - 1e-14:
            return h_try, sc, alpha
    return None, best_score, 0.0

def subgradient_step(h, k_active=5, step_size=0.05):
    n = len(h)
    f = compute_f(h)
    top_k = np.argsort(f)[-k_active:]
    avg_grad = np.mean([gradient_at_lag(h, p - (n - 1)) for p in top_k], axis=0)
    avg_grad = avg_grad - np.mean(avg_grad)
    d = -step_size * avg_grad / (np.linalg.norm(avg_grad) + 1e-15)
    h_try = make_feasible(h + d)
    return h_try

def smoothing_move(h, i, j, magnitude):
    n = len(h)
    h_try = h.copy()
    delta = (h[i] - h[j]) * magnitude
    if h[i] - delta >= 0 and h[j] + delta <= 1:
        h_try[i] -= delta
        h_try[j] += delta
    else:
        max_d = min(h[i], 1.0 - h[j]) * magnitude
        if max_d > 0:
            h_try[i] -= max_d
            h_try[j] += max_d
    return h_try

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)

    h_int = np.interp(tgt_c, src_c, h_src)
