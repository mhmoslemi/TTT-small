```python
def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    # Project z to [0,1]^n with sum = n/2
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    # Binary search for shift
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_of_all_lags(h):
    """Compute gradients of all lag functions."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for s in range(n):

            if k >= t:

            grads[n-1+t, k] = g
```

The problem is: minimize max_k ∫₀^{2-|k|} h(x)(1-h(x+k)) dx

```python
import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_all_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_value(h, idx):
    n = len(h)
    dx = 2.0 / n
    s = idx - (n - 1)
    if s >= 0:
        L = n - s
        return float(np.sum(h[s:n] * (1.0 - h[0:L])) * dx)
    else:
        t = -s
        L = n - t
        return float(np.sum(h[0:L] * (1.0 - h[t:n])) * dx)

def lag_grad(h, idx):
    n = len(h)
    dx = 2.0 / n
    s = idx - (n - 1)
    g = np.zeros(n)
    if s == 0:
        g = dx * (1.0 - 2.0 * h)
    elif s > 0:
        L = n - s
        g[s:] += dx * (1.0 - h[0:L])
        g[:L] -= dx * h[s:]
    else:
        t = -s
        L = n - t
        g[:L] += dx * (1.0 - h[t:])
        g[t:] -= dx * h[:L]
    return g

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = np.clip(beta * (V - V_max), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w /= ws
    return w

def full_gradient(h):
    n = len(h)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        g += lag_grad(h, idx)
    return g

def projected_descent_step(h, direction, alpha_max=0.5):
    best_h = h.copy()
    best_f = official_score(h)
    alpha = alpha_max
    for _ in range(14):
        h_new = make_feasible(h - alpha * direction)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.55
        if alpha < 1e-9:
            break
    return best_h, best_f

def gradient_phase(h, duration_s=30.0):
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_f = official_score(h)
    t0 = time.time()
    beta = 25.0
    stall = 0
    step = 0
    while time.time() - t0 < duration_s:
        V = compute_all_values(h_cur)
        w = softmax_weights(V, beta)
        g = np.zeros(n)
        for idx in range(len(V)):
            if w[idx] > 1e-5:
                g += w[idx] * lag_grad(h_cur, idx)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new, f_new = projected_descent_step(h_cur, g, alpha_max=0.4)
        if f_new < best_f - 1e-13:
            best_h = h_new
            best_f = f_new
        elif f_new < official_score(h_cur) - 1e-13:
            stall = 0
        else:
            stall += 1
        h_cur = h_new
        beta = min(beta * 1.02, 400.0)
        if stall > 80:
            h_cur = make_feasible(best_h + 0.02 * np.random.default_rng(step).standard_normal(n))
            beta = 20.0
            stall = 0
        step += 1
    return best_h, best_f

def _slsqp_ineq(v, n_pts, lag_idx):
    h = v[:-1]
    t = v[-1]
    return t - lag_value(h, lag_idx)

def _slsqp_eq(v, n_pts):
    return float(np.sum(v[:-1]) - n_pts / 2.0)

def _slsqp_obj(v):
    return v[-1]

def slsqp_minimax(h0, n_active=20, maxiter=150, t_margin=0.005):
    n = len(h0)
    V = compute_all_values(h0)
    active = np.argsort(V)[-n_active:]
    x0 = np.concatenate([h0, [V.max() + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = []
    for idx in active:
        cons.append({'type': 'ineq', 'fun': partial(_slsqp_ineq, n_pts=n, lag_idx=int(idx))})
    cons.append({'type': 'eq', 'fun': partial(_slsqp_eq, n_pts=n)})
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None:
            h_new = make_feasible(res.x[:-1])
            sc = official_score(h_new)
            if sc < V.max() + 0.05:
                return h_new, sc
    except Exception:
        pass
    return None, None

def pair_descent(h, n_pairs=300, seed=99):
    n = len(h)
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_f = official_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_pairs):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-10:
            continue
        a, b = d_lo, d_hi
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        hc = h.copy(); hc[i] += c; hc[j] -= c
        hd = h.copy(); hd[i] += d; hd[j] -= d
        fc = official_score(hc)
        fd = official_score(hd)
        for _ in range(25):
            if fc < fd:
                b = d; d = c; fd = fc
                c = b - gr * (b - a)
                hc[i] = h[i] + c; hc[j] = h[j] - c
                fc = official_score(hc)
            else:
                a = c; c = d; fc = fd
                d = a + gr * (b - a)
                hd[i] = h[i] + d; hd[j] = h[j] - d
                fd = official_score(hd)
            if b - a < 1e-8:
                break
        d_opt = (a + b) / 2.0
        h_new = h.copy()
        h_new[i] += d_opt
        h_new[j] -= d_opt
        f_new = official_score(h_new)
        if f_new < best_f - 1e-12:
            best_h = h_new
            best_f = f_new
            h = h_new.copy()
    return best_h, best_f

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)

            # Follow up with gradient
            h_g, sg = gradient_phase(best_h, duration_s=5)
            if sg < best_f:

    return best_h, best_f

            h_pert = make_feasible(best_h + 0.05 * rng.standard_normal(n_cur))
            s_pert = official_score(h_pert)
            if s_pert < best_score + 0.02:
                h_opt, s_opt = gradient_phase(h_pert, duration_s=10)
                if s_opt < s_pert:
                    h_pert, s_pert = h_opt, s_opt
                if time_ok(40):
                    h_s, s_s = slsqp_minimax(h_pert, n_active=15, maxiter=80)
                    if h_s is not None and s_s < s_pert:
                        h_pert, s_pert = h_s, s_s
                if s_pert < best_score:
                    best_h, best_score = h_pert, s_pert

    # Phase 8: Final polishing - SLSQP + gradient
    if time_ok(30):
        h_f, s_f = slsqp_rounds(best_h, n_rounds=2, duration_s=20)
        if s_f < best_score:
            best_h, best_score = h_f, s_f
    if time_ok(15):
        h_f2, s_f2 = gradient_phase(best_h, duration_s=10)
        if s_f2 < best_score:
            best_h, best_score = h_f2, s_f2

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
```
