import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    """Project x onto {0 <= y <= 1, sum(y) = target_sum} using bisection."""
    if not np.all(np.isfinite(x)):
        x = np.zeros(len(x))
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(proj)
    if abs(residual) > 1e-12:
        # Distribute residual among non-saturated coordinates
        if residual > 0:
            candidates = np.where(proj < 1.0 - 1e-10)[0]
        else:
            candidates = np.where(proj > 1e-10)[0]
        if len(candidates) > 0:
            per = residual / len(candidates)
            proj[candidates] = np.clip(proj[candidates] + per, 0.0, 1.0)
    return proj

def smooth_obj_and_grad(x, n, T):
    """Compute smooth log-sum-exp objective and gradient in raw correlation units."""
    v = 1.0 - x
    corr = np.correlate(x, v, mode="full")
    corr_max = np.max(corr)
    shifted = (corr - corr_max) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum_exp = np.log(np.sum(exp_vals))
    value = corr_max + T * log_sum_exp
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient
    m = len(corr)  # 2n-1
    n_half = n - 1
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for s in range(-(n-1), n):
            m_idx = n_half + s
            w = weights[m_idx]
            if w < 1e-15:
                continue
            # Contribution from h[j] as first factor: (1-h)[j-s] if j-s in [0,n-1]

    return value, grad

    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)

        h[j] = hj_v

def pair_sweep(h, rng, max_pairs=None):
    n = len(h)
    if max_pairs is None:
        max_pairs = n * (n - 1)
    perm = rng.permutation(n * n)
    count = 0
    improved = True
    while improved and count < max_pairs:
        improved = False
        for idx in perm[:max_pairs]:
            i = idx % n
            j = (idx // n) % n
            if i == j:
                continue
            if i > j:
                i, j = j, i
            delta = pair_line_search(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
                improved = True
        count += max_pairs
