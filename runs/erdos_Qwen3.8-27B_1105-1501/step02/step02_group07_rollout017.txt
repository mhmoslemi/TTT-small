    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.0, posinf=5.0, neginf=-5.0)

            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break

    """LP to reduce active overlaps."""
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    # Limit active set size

        top_idx = np.argsort(V[active])[::-1][:50]
        active = active[top_idx]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo_b = max(-h[i], -trust)
        hi_b = min(1.0 - h[i], trust)
        bounds_list.append((lo_b, hi_b))
    bounds_list.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def make_feasible(z):
    n = len(z)
    target = n / 2.0

        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:

                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:

def active_set_lp(h, eps=0.005, trust=0.1):

    b_rows = []

            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]

            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)

def pg_step(h, beta, alpha_max=0.7, max_ls=14):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -400.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)

            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-7:
            break
    return best_h, best_f

                    i = rng.integers(n_t)
                    j = rng.integers(n_t)
                    if i == j:
                        i = (i + 1) % n_t
                    h_new, sc = pair_optimize(h_cur, i, j, n_grid=15)
                else:
                    h_new, sc = pg_step(h_cur, 60.0, alpha_max=0.4, max_ls=10)
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc
                if sc < official_score(h_cur) - 1e-14:
                    h_cur = h_new
            # Quick LP at this resolution
            for _ in range(15):
                if time_left() < 50:
                    break
                h_new, sc = active_set_lp(h_cur, eps=0.003, trust=0.06)
                if h_new is not None and sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc
                    h_cur = h_new

    # Phase 5: Final intensive polish on best
    h_cur = best_h.copy()
    n_cur = len(h_cur)
    for _ in range(2000):
        if time_left() < 20:
            break
        r = rng.random()
        if r < 0.6:
            i = rng.integers(n_cur)
            j = rng.integers(n_cur)
            if i == j:
                i = (i + 1) % n_cur
            h_new, sc = pair_optimize(h_cur, i, j, n_grid=21)
        elif r < 0.85:
            h_new, sc = active_set_lp(h_cur, eps=0.001, trust=0.04)
            if h_new is None:
                h_new, sc = pair_optimize(h_cur, int(rng.integers(n_cur)), int(rng.integers(n_cur)), n_grid=15)
        else:
            h_new, sc = pg_step(h_cur, 200.0, alpha_max=0.15, max_ls=8)
        if sc < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc
        if sc < official_score(h_cur) - 1e-14:
            h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
