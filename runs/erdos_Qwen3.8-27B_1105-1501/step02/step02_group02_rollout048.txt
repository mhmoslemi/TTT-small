def _score(h):

def _corr_arr(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if res > 0:

            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
            if abs(res) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, beta):
    cv = _corr_arr(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    cv = _corr_arr(x)
    mc = cv.max()

                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_direct(h0, maxiter=100):
    n = len(h0)
    cv = _corr_arr(h0)
    t0 = min(cv.max() + 0.005, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_arr(h)
        return t - cv

    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _fista(h0, beta, n_iter, alpha_init=0.005, mu=0.88):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        t_cur = it + 1
        t_new = 1.5 * t_cur
        h_new = _project(y - alpha_init * grad + (h_cur - h_prev) * (t_cur-1) / t_new)
        # FISTA extrapolation
        y = h_new
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_cur - h_avg) / (t_new)
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 2000 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
