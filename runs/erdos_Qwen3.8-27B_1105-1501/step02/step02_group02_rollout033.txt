    n = len(h)
    h = h.copy()
    cv = _corr_vec(h)  # array of 2n-1 lag values
    grid = np.linspace(0, 1, n_grid)
    
    for pass_i in range(n_passes):
        improved = False
        
        for p in range(n):
            old_val = h[p]
            # Compute the coefficient of h[p] in each lag
            # For lag k (shift s = k - (n-1)):
            # h[p] contributes (1-h[p-s]) if p-s is valid in sum
            # h[p] contributes -h[p+s] if p is valid in sum (as the (1-h[i]) factor)
            
            coeffs = np.zeros(2*n-1)
            for k in range(2*n-1):
                s = k - (n-1)
                i_min = max(0, s)
                i_max = min(n, n+s)
                
                # h[p] as h[i+s]: i = p-s, valid if i_min <= p-s < i_max
                coeff = 0.0
                if i_min <= p-s < i_max and 0 <= p-s < n:
                    coeff += (1.0 - h[p-s])

                if i_min <= p < i_max:
                    coeff -= h[p+s] if 0 <= p+s < n else 0.0

            best_g = cv.max()
            best_t = old_val

                    new_cv = cv + coeffs * (t - old_val)
                    g = new_cv.max()
                    if g < best_g - 1e-12:
                        best_g = g
                        best_t = t

                    val = t - old_val
                    new_max = (cv + coeffs * val).max()
                    if new_max < best_g - 1e-12:
                        best_g = new_max
                        best_t = t

                cv = cv + coeffs * (best_t - old_val)

import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
            if abs(res) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def _coord_cd(h, n_passes=500, grid_size=50, use_shuffle=False, rng=None):
    n = len(h)
    h = h.copy()
    cv = _corr_vec(h)
    grid = np.linspace(0.0, 1.0, grid_size)
    s0_idx = n - 1
    shifts = np.arange(2 * n - 1) - (n - 1)
    v_i_min = np.zeros(2 * n - 1, dtype=int)
    v_i_max = np.zeros(2 * n - 1, dtype=int)
    for k in range(2 * n - 1):
        sk = shifts[k]
        v_i_min[k] = max(0, sk)
        v_i_max[k] = min(n, n + sk)
    coords = np.arange(n)
    for pass_i in range(n_passes):
        if use_shuffle and rng is not None:
            idx = rng.permutation(n)
        else:
            idx = coords
        improved = False
        for p in idx:
            old_val = h[p]
            ps = p - shifts
            pss = p + shifts
            mask1 = (ps >= v_i_min) & (ps < v_i_max) & (ps >= 0) & (ps < n) & (shifts != 0)
            mask2 = (p >= v_i_min) & (p < v_i_max) & (pss >= 0) & (pss < n) & (shifts != 0)
            coeffs = np.zeros(2 * n - 1)
            if mask1.any():
                coeffs[mask1] += 1.0 - h[ps[mask1]]
            if mask2.any():
                coeffs[mask2] -= h[pss[mask2]]
            deltas = grid - old_val
            close = np.abs(deltas) < 1e-10
            if not np.any(~close):
                continue
            g_deltas = deltas[~close]
            g_grid = grid[~close]
            new_cv = cv[None, :] + coeffs[None, :] * g_deltas[:, None]
            new_cv[:, s0_idx] = cv[s0_idx] + g_deltas * (1.0 - g_grid - old_val)
            g_vals = new_cv.max(axis=1)
            best_local = np.argmin(g_vals)
            cur_max = cv.max()
            if g_vals[best_local] < cur_max - 1e-13:
                best_t = g_grid[best_local]
                delta = best_t - old_val
                cv = cv + coeffs * delta
                cv[s0_idx] += delta * (1.0 - best_t - old_val)
                h[p] = best_t
                improved = True
        if not improved:

    h = _project(h)
    return h, _score(h)

def _slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    def _lse_obj(x, b):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(b * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / b

    def _lse_grad(x, b):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(b * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            j0 = max(0, s)
            j1 = min(n, n + s)
            for p in range(j0, j1):
                g[p] += wk * (1.0 - x[p - s])
            p0 = max(0, -s)
            p1 = min(n, n - s)
            for p in range(p0, p1):
                g[p] += wk * (-x[p + s])
        return g

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

    """Create a block-structured initialization."""
    block_vals = rng.uniform(0.3, 0.7, m)
    h = np.repeat(block_vals, n // m + 1)[:n]
