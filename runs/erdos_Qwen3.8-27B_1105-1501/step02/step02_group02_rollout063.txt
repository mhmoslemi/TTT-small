```python
def _corr_vec(h):

def _project(z):
    # Project z onto {h: 0<=h<=1, sum(h)=n/2}
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)
```

```python
    n = len(h)
```

```python
def _coord_descent(h, top_m=10, n_passes=50):
    n = len(h)
    h = h.copy()
    for _ in range(n_passes):
        for p in range(n):
            # Compute active correlations and their derivatives w.r.t. h[p]
            cv = _corr_vec(h)
            idx = np.argsort(cv)[::-1][:top_m]
            # For each active lag k, compute F_k and dF_k/dh[p]
            vals = []
            derivs = []
            for k in idx:
                s = k - (n - 1)
                # F_k = dx * sum_{valid j} h[j+s] * (1-h[j])
                # dF_k/dh[p]:
                # If p is in h[j+s] part: j = p-s, contribution (1-h[p-s])
                # If p is in (1-h[j]) part: j = p, contribution -h[p+s]
                d = 0.0
                # p in h[j+s] part: need 0<=p-s<n and 0<=p<n => s<=p and p-s>=0
                if 0 <= p - s < n and 0 <= p < n:
                    d += (1.0 - h[p - s])
                # p in (1-h[j]) part: need 0<=p<n and 0<=p+s<n
                if 0 <= p < n and 0 <= p + s < n:
                    d += -h[p + s]
                vals.append(cv[k])
                derivs.append(d * (2.0 / n))
            
            # Now find h[p] in [0,1] that minimizes max_k (vals[k] + derivs[k] * (h[p] - h[p]))
            # = max_k (vals[k] + derivs[k] * delta) where delta = h[p]_new - h[p]
            # Let f(δ) = max_k (a_k + b_k * δ) where a_k = vals[k], b_k = derivs[k]
```

```python
def _score(h):

def _corr_vec(h):

def _project(z):

    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        else:

    res = target - h.sum()
    if abs(res) > 1e-12:

        else:
            idx = np.argsort(h)[::-1]
```

```python
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _lse_value(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta


def _lse_grad_full(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        for p in range(max(0, s), min(n, n + s)):
            g[p] += wk * dx * (1.0 - x[p - s])
        for p in range(max(0, -s), min(n, n - s)):
            g[p] += wk * dx * (-x[p + s])
    return g


def _weighted_grad(h, lams, k_indices):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for ii, k in enumerate(k_indices):
        wk = lams[ii] * dx
        if abs(wk) < 1e-15:
            continue
        s = k - (n - 1)
        for p in range(max(0, s), min(n, n + s)):
            g[p] += wk * (1.0 - h[p - s])
        for p in range(max(0, -s), min(n, n - s)):
            g[p] += wk * (-h[p + s])
    return g


def _slsqp_solve(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return _lse_value(x, beta)

    def _grad(x):
        return _lse_grad_full(x, beta)

    def _eq(x):
        return x.sum() - n / 2.0

    def _eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return _project(h0)


def _lbfgsb_solve(h0, beta, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        v = _lse_value(x, beta)
        pen = 50.0 * (x.sum() - n / 2.0) ** 2
        return v + pen

    def _grad(x):
        g = _lse_grad_full(x, beta)
        g += 100.0 * (x.sum() - n / 2.0) * np.ones(n)
        return g

    try:
        res = minimize(_obj, h0, jac=_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return _project(h0)


def _fista_smooth(h0, beta, n_iter, alpha_init=0.005, mu=0.9):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1
    for it in range(n_iter):
        grad = _lse_grad_full(y, beta)
        t_prev = t
        t = 1.5 * t
        t_new = t
        y = h_cur + (t_prev / t_new) * (h_cur - h_prev)
        y = _project(y)
        grad = _lse_grad_full(y, beta)
        h_new = _project(y - alpha_init * (1.0 - mu) * (y - h_prev) - alpha_init * mu * (y - h_cur) + alpha_init * grad * 0 + 0)
        # Simpler FISTA:
        h_new = _project(y - alpha_init * grad)
        h_avg += (h_new - h_avg) / (t)
        h_prev = h_cur
        h_cur = h_new
        if it % 200 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 500 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s


def _mwu_optimize(h0, n_outer=40, n_inner=60, eta=0.04, top_m=8, alpha_inner=0.004):
    n = len(h0)
    h = h0.copy()
    for t in range(n_outer):
        cv = _corr_vec(h)
        idx = np.argsort(cv)[::-1][:top_m]
        active_cv = cv[idx]
        lams = np.exp(eta * active_cv)
        lams /= lams.sum()
        # Inner: projected gradient on weighted sum
        for _ in range(n_inner):
            g = _weighted_grad(h, lams, idx)
            h = _project(h - alpha_inner * g)
    return h


def _coord_descent(h, top_m=8, n_passes=80):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    for _ in range(n_passes):
        improved_any = False
        for p in range(n):
            cv = _corr_vec(h)
            idx = np.argsort(cv)[::-1][:top_m]
            vals = cv[idx].copy()
            derivs = np.zeros(len(idx))
            for ii, k in enumerate(idx):
                s = k - (n - 1)
                d = 0.0
                # h[n+m] part: p = n+m, contribution (1-h[p-s])
                if max(0, s) <= p < min(n, n + s):
                    d += (1.0 - h[p - s])
                # (1-h[n]) part: p = n, contribution -h[p+s]
                if max(0, -s) <= p < min(n, n - s):
                    d += (-h[p + s])
                derivs[ii] = d * dx
            # Bisection for optimal delta
            lo_d = -h[p]
            hi_d = 1.0 - h[p]
            for _ in range(50):
                mid_d = (lo_d + hi_d) / 2.0
                new_vals = vals + derivs * mid_d
                mx = new_vals.max()
                active_mask = new_vals >= mx - 1e-12
                sg = derivs[active_mask].sum()
                if sg > 1e-14:
                    hi_d = mid_d
                elif sg < -1e-14:
                    lo_d = mid_d
                else:
                    break
            new_val = np.clip(h[p] + (lo_d + hi_d) / 2.0, 0.0, 1.0)
            if abs(new_val - h[p]) > 1e-12:
                h[p] = new_val
                improved_any = True
        h = _project(h)
        if not improved_any:
            break
    return h


def _direct_minmax(h0, maxiter=150):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = min(cv.max() + 0.005, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv

    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return _project(h0)


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _chain_refine(h0, deadline, tag=""):
    """Multi-stage refinement chain."""
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)

    if time.time() < deadline - 30:
        new = _slsqp_solve(cur, beta=60, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns

    if time.time() < deadline - 25:
        new = _lbfgsb_solve(cur, beta=100, maxiter=200)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns

    if time.time() < deadline - 20:
        cur, cur_s = _fista_smooth(cur, beta=150, n_iter=4000, alpha_init=0.004, mu=0.88)

    if time.time() < deadline - 15:
        new = _slsqp_solve(cur, beta=400, maxiter=120)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns

    if time.time() < deadline - 10:
        cur = _coord_descent(cur, top_m=8, n_passes=40)
        cs = _score(cur)
        if cs < cur_s - 1e-13:
            cur_s = cs

    if time.time() < deadline - 8:
        new = _direct_minmax(cur, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns

    if time.time() < deadline - 5:
        cur, cur_s = _fista_smooth(cur, beta=500, n_iter=3000, alpha_init=0.003, mu=0.92)

    return cur, cur_s


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(40) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(parent)
    parent_n = len(parent)
    rng = np.random.default_rng(seed)

    # Phase 1: Screen n values with quick optimization
    n_lo = max(20, parent_n - 20)
    n_hi = min(80, parent_n + 20)
    n_candidates = sorted(set(list(range(n_lo, n_hi + 1, 2)) + [parent_n]))

    phase1_results = []
    for n in n_candidates:
        if time.time() > deadline - 350:
            break
        h_i = _interp(parent, n)
        s_i = _score(h_i)
        if time.time() < deadline - 330:
            h_opt = _slsqp_solve(h_i, beta=50, maxiter=50)
            s_opt = _score(h_opt)
            if s_opt < s_i:
                s_i, h_i = s_opt, h_opt
        phase1_results.append((s_i, n, h_i.copy()))
        if s_i < best_score - 1e-12:
            best_score = s_i
            best_h = h_i.copy()
            best_n = n

    # Also try 1-h of parent (same C5 by symmetry)
    h_complement = _project(1.0 - parent)
    s_comp = _score(h_complement)
    if s_comp < best_score - 1e-12:
        best_score = s_comp
        best_h = h_complement.copy()

    # Phase 2: MWU on top candidates
    phase1_results.sort(key=lambda x: x[0])
    top_candidates = phase1_results[:4]

    for s_val, n_val, h_val in top_candidates:
        if time.time() > deadline - 200:
            break
        h_mwu = _mwu_optimize(h_val, n_outer=25, n_inner=50, eta=0.03, top_m=8)
        s_mwu = _score(h_mwu)
        if s_mwu < best_score - 1e-12:
            best_score = s_mwu
            best_h = h_mwu.copy()
            best_n = n_val

    # Phase 3: Full chain refinement on best
    if time.time() < deadline - 100:
        best_h, best_score = _chain_refine(best_h, deadline)

    # Phase 4: Try different starting points around best
    n_starts = min(6, int((deadline - time.time()) / 30))
    for i in range(n_starts):
        if time.time() > deadline - 40:
            break
        if i == 0:
            h_start = _project(1.0 - best_h)
        elif i == 1:
            h_start = _project(best_h[::-1].copy())
        elif i == 2:
            h_start = _project(best_h + 0.05 * rng.standard_normal(best_n))
        elif i == 3:
            h_start = _project(np.clip(best_h ** 1.8, 0.0, 1.0))
        elif i == 4:
            freq = rng.integers(2, 5)
            phase = rng.uniform(0, 6.28)
            pert = 0.08 * np.sin(2 * np.pi * freq * np.arange(best_n) / best_n + phase)
            h_start = _project(best_h + pert)
        else:
            h_start = _project(best_h + 0.03 * rng.standard_normal(best_n))

        h_new = _mwu_optimize(h_start, n_outer=15, n_inner=40, eta=0.03, top_m=6)
        s_new = _score(h_new)
        if s_new < best_score - 1e-12:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 5: Final intensive polish
    if time.time() < deadline - 50:
        best_h, best_score = _chain_refine(best_h, deadline)

    # Phase 6: Final coordinate descent + SLSQP
    if time.time() < deadline - 20:
        best_h = _coord_descent(best_h, top_m=6, n_passes=50)
        cs = _score(best_h)
        if cs < best_score:
            best_score = cs

    if time.time() < deadline - 10:
        new = _slsqp_solve(best_h, beta=800, maxiter=150)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_score = ns
            best_h = new

    if time.time() < deadline - 8:
        best_h = _coord_descent(best_h, top_m=6, n_passes=30)

    # Final projection and score
    best_h = _project(best_h)
    final_score = _score(best_h)

    return (best_h, final_score, len(best_h))
```
