**Level method for minimax:**

**Key insight for step size**: In the projected gradient method, the optimal step size for the smoothed problem is:
alpha = (f(h_k) - τ) / ||g_k||^2
where τ is a target value (like the current best from a lower beta). This is the **Polyak step size** for subgradient methods.

```python
def pg_with_polyak(h0, beta, tau, time_limit=20.0, max_iter=5000):
    n = len(h0)
    h = h0.copy()
    f, g = smooth_obj_and_grad(h, beta)
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-20:
            break
        # Polyak step size
        alpha = (f - tau) / gnorm2
        alpha = max(alpha, 1e-8)
        alpha = min(alpha, 2.0 / (npm.max(g**2) + 1e-12))
        
        # Try the step
        h_new = make_feasible(h - alpha * g)
        f_new, g_new = smooth_obj_and_grad(h_new, beta)
        
        if f_new > f:
            # Backtracking
            alpha *= 0.5
            h_new = make_feasible(h - alpha * g)
            f_new, g_new = smooth_obj_and_grad(h_new, beta)
            if f_new > f:
                # Try even smaller
                alpha *= 0.5
                h_new = make_feasible(h - alpha * g)
                f_new, g_new = smooth_obj_and_grad(h_new, beta)
        
        if not np.isfinite(f_new):
            break
        h, f, g = h_new, f_new, g_new
    
    return h
```

Actually, I realize the main issue with the previous code might be that it's spending too much time on each iteration (computing the full gradient matrix is O(n^2) per evaluation) and not doing enough iterations.

Let me optimize the gradient computation. Instead of computing the full matrix G (size (2n-1, n)), I can compute the weighted gradient directly:

```python
def smooth_grad_fast(h, w):
    """Compute weighted sum of row gradients without building the full matrix."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n-1)
        if s >= 0:
            L = n - s
            grad[:L] -= wk * dx * h[s:]
            grad[s:] += wk * dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            grad[:K] += wk * dx * (1.0 - h[t:])
            grad[t:] -= wk * dx * h[:K]
    return grad
```

This is O(n^2) per call (loop over 2n-1 shifts, each touching O(n) elements). For n=66, that's about 131 * 66 ≈ 8600 operations, which is fast.

Actually, we can vectorize this! The key observation:

For s ≥ 0: grad[:n-s] -= w_s * dx * h[s:], and grad[s:] += w_s * dx * (1-h[:n-s])
For s < 0 (t = -s > 0): grad[:n-t] += w_{-t} * dx * (1-h[t:]), and grad[t:] -= w_{-t} * dx * h[:n-t]

This can be computed using cumulative sums or FFT, but for n=66, the loop is fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14: break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj_grad(h, beta):
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w = w / ws
    grad = smooth_grad_from_weights(h, w)
    return obj, grad, vals

def smooth_grad_from_weights(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            L = n - s
            grad[:L] -= wk * dx * h[s:]
            grad[s:] += wk * dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            grad[:K] += wk * dx * (1.0 - h[t:])
            grad[t:] -= wk * dx * h[:K]
    return grad

def pg_optimize(h0, beta, time_limit=15.0, max_iter=10000):
    n = len(h0)
    h = h0.copy()
    f, g, _ = smooth_obj_grad(h, beta)
    t0 = time.time()
    step = 1.0 / (float(g @ g) + 1e-12)
    step = min(step, 2.0 / (np.max(g**2) + 1e-12))
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-25:
            break
        
        # Try the step with backtracking
        alpha = step
        accepted = False
        for bt in range(20):
            h_trial = make_feasible(h - alpha * g)
            f_trial, _, _ = smooth_obj_grad(h_trial, beta)
            if f_trial <= f - 1e-4 * alpha * gnorm2 or alpha < 1e-14:
                h = h_trial
                f = f_trial
                _, g, _ = smooth_obj_grad(h, beta)
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            break
        
        # Adaptive step size
        gnorm2_new = float(g @ g)
        if f < f - alpha * 0.1 * gnorm2:  # good descent
            step = min(step * 1.5, 5.0 / (gnorm2_new + 1e-12))
        else:
            step = max(step * 0.7, 1e-12)
    
    return h

def slsqp_solve(h0, active_idx, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    
    def obj_func(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(z):
        h = z[:n]
        t = z[-1]
        vals = compute_vals(h)
        return t - vals[active_idx]
    
    def ineq_jac(z):
        h = z[:n]
        n_a = len(active_idx)
        J = np.zeros((n_a, n + 1))
        for i, k in enumerate(active_idx):
            s = k - (n - 1)
            if s >= 0:
                L = n - s
                J[i, :L] = dx * h[s:]
                J[i, s:] -= dx * (1.0 - h[:L])
            else:
                t = -s
                K = n - t
                J[i, :K] = -dx * (1.0 - h[t:])
                J[i, t:] = dx * h[:K]
            J[i, n] = 1.0
        return J
    
    def eq_fun(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    t0 = official_score(h0) + 1e-6
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj_func, z0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def find_active(vals, margin=0.005, min_count=10):
    cmax = vals.max()
    threshold = cmax - margin
    active = set(np.where(vals >= threshold)[0])

    top_idx = np.argsort(vals)[::-1][:top_k]
    active.update(top_idx.tolist())
    return sorted(active)
