def _min_max_lines(d, e, t_lo=0.0, t_hi=1.0):
    """Find t in [t_lo, t_hi] minimizing max_i(d[i] + e[i]*t)."""
    n = len(d)
    if n == 0:
        return t_lo
    
    # Create lines: y = e[i]*t + d[i]
    # Sort by slope
    order = np.argsort(e)
    e_sorted = e[order]
    d_sorted = d[order]
    
    # Build upper hull
    # A line (e, d) is on the upper hull if it's not dominated
    # Use the "convex hull of lines" algorithm
    hull = []  # list of indices into sorted arrays
    for i in range(len(e_sorted)):
        ei, di = e_sorted[i], d_sorted[i]
        # Check if this line is useful
        # A line is useful if at its intersection with the last hull line,
        # it's above the second-to-last hull line
        while len(hull) >= 2:
            last = hull[-1]
            second_last = hull[-2]
            # Intersection of last and new: t_cross = (d_new - d_last) / (e_last - e_new)
            if abs(ei - e_sorted[last]) < 1e-15:
                # Same slope: keep the one with larger d
                if di <= d_sorted[last]:
                    break  # new line is dominated
                else:
                    hull.pop()
                    continue

            # t_cross is where new line crosses last line

            if len(hull) >= 2:

                    continue

    """Minimize max(d + e*t) over t in [0,1]."""
    n = len(d)
    # All pairwise intersections
    # t_ij = (d_i - d_j) / (e_j - e_i) for e_i != e_j
    de = e[:, None] - e[None, :]  # n x n
    dd = d[None, :] - d[:, None]  # n x n
    # t[i,j] = (d_j - d_i) / (e_i - e_j) = dd[j,i] / de[i,j]

    target = n / 2.0

    lo = -2.0 - np.max(z)
