import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    """Return all correlation values (undecayed)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # These are the actual C5 contributions for each lag

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_gradient(h, tau):
    """Compute gradient of soft-max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx  # length 2n-1
    
    # Softmax weights
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    # Compute gradient
    # For lag d (where d = k - (n-1)), C_d = sum_n h[n+d]*(1-h[n])
    # dC_d/dh[j] = (1-h[j-d]) * 1[j-d valid] - h[j+d] * 1[j+d valid]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        # (1-h[j-d]) for j where j-d is in [0, n-1]
        # j-d in [0,n-1] means j in [d, d+n-1], intersect with [0, n-1]
        j_lo = max(0, d)
        j_hi = min(n - 1, d + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            src = idx - d  # indices into h for the (1-h) term
            grad[idx] += wk * (1.0 - h[src])
        
        # -h[j+d] for j where j+d is in [0, n-1]
        # j+d in [0,n-1] means j in [-d, n-1-d], intersect with [0, n-1]
        j_lo2 = max(0, -d)
        j_hi2 = min(n - 1, n - 1 - d)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            src2 = idx2 + d  # indices into h for the h term
            grad[idx2] += wk * (-h[src2])
    
    return grad

def _exact_subgradient(h):
    """Compute subgradient of the exact max C5 (using argmax lag)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    d = k_max - (n - 1)
    
    grad = np.zeros(n)
    # (1-h[j-d]) for valid j
    j_lo = max(0, d)
    j_hi = min(n - 1, d + n - 1)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        src = idx - d
        grad[idx] = (1.0 - h[src])
    
    # -h[j+d] for valid j
    j_lo2 = max(0, -d)
    j_hi2 = min(n - 1, n - 1 - d)
    if j_lo2 <= j_hi2:
        idx2 = np.arange(j_lo2, j_hi2 + 1)
        src2 = idx2 + d
        grad[idx2] += (-h[src2])
    
    return grad * dx  # Scale by dx since C5 = max(corr) * dx

def _gd_step(h, grad, learning_rate, momentum_buf, beta):
    """One step of gradient descent with momentum and projection."""
    momentum_buf = beta * momentum_buf + (1 - beta) * grad
    h_new = h - learning_rate * momentum_buf
    return h_new, momentum_buf

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _anneal_search(h, deadline, tau_start=50.0, tau_end=200.0,
                   lr_start=0.02, lr_end=0.001, n_iters=5000,
                   beta=0.9, restart_every=200, restart_scale=0.05):
    """Animated gradient descent with simulated annealing."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    
    n_iters_actual = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        n_iters_actual += 1
        progress = it / max(1, n_iters - 1)
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute gradient
        if progress < 0.7:
            grad = _soft_gradient(h, tau)
        else:
            # Mix of soft and exact
            g_soft = _soft_gradient(h, tau)
            g_exact = _exact_subgradient(h)
            grad = 0.5 * g_soft + 0.5 * g_exact
        
        h_new, momentum = _gd_step(h, grad, lr, momentum, beta)
        h_new = _project(h_new, target)
        
        c5_new = _exact_c5(h_new)
        
        # Acceptance criterion (simulated annealing)
        if c5_new < cur_c5:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Accept with probability exp(-delta / T)
            delta = c5_new - cur_c5
            T = max(0.001, 0.01 * (1 - progress))
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                cur_c5 = c5_new
        
        # Periodic restart
        if it > 0 and it % restart_every == 0:
            h_pert = best_h + np.random.normal(0, restart_scale * (1 - progress), n)
            h_pert = _project(h_pert, target)
            c5_pert = _exact_c5(h_pert)
            if c5_pert < cur_c5:
                h = h_pert
                cur_c5 = c5_pert
            momentum = np.zeros(n)
    
    return best_h, best_c5

def _local_refine(h, deadline, max_iters=2000):
    """Fine-tune using exact subgradient with small steps."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = _exact_subgradient(h)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        
        # Backtracking line search
        lr = 0.01
        for _trial in range(10):
            h_new = h - lr * grad
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            else:
                lr *= 0.5
        
        if lr < 1e-8:
            break
    
    return best_h, best_c5

            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        j_lo = max(0, d)
        j_hi = min(n - 1, d + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += wk * (1.0 - h[idx - d])
        j_lo2 = max(0, -d)
        j_hi2 = min(n - 1, n - 1 - d)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad[idx2] += wk * (-h[idx2 + d])
    return grad

def _grad_search(h, deadline, tau=100.0, lr=0.01, momentum_beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        grad = _soft_grad(h, tau)
        mom = momentum_beta * mom + (1.0 - momentum_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            lr *= 0.9
            if stall > 50:
                break
            if stall % 20 == 0:
                h = best_h + np.random.normal(0, 0.02, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if dj > 0:
                        am = (1.0 - h[j]) / dj
                    else:
                        am = h[j] / (-dj)
                    if dk > 0:
                        am = min(am, (1.0 - h[k]) / dk)
                    else:
                        am = min(am, h[k] / (-dk))
                    if am < 1e-10:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

def _evolutionary(h_init, deadline, pop_size=15, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.06, n)
        pop.append(_project(h_i, target))
    fits = [_exact_c5(h) for h in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.04
    while time.time() < deadline:
        idxs = rng.integers(0, pop_size, size=3)
        w1 = pop[idxs[int(np.argmin([fits[i] for i in idxs]))]].copy()
        idxs2 = rng.integers(0, pop_size, size=3)
        w2 = pop[idxs2[int(np.argmin([fits[i] for i in idxs2]))]].copy()
        child = 0.5 * (w1 + w2)
        child += rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.01, mut_scale * 0.95)
            else:
                mut_scale = min(0.12, mut_scale * 1.05)
        else:
            mut_scale = min(0.12, mut_scale * 1.03)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick gradient warmup
    d1 = t0 + budget_s * 0.05
    h, c5 = _grad_search(h, d1, tau=80.0, lr=0.015, momentum_beta=0.85)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Intensive LP local search
    d2 = t0 + budget_s * 0.35
    h, c5 = _lp_loop(best_h, d2, radius_init=0.12)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Multi-resolution search
    d3 = t0 + budget_s * 0.65
    offsets = [0, -1, 1, -2, 2, -3, 3, -4, 4, -6, 6, -8, 8, -12, 12, -16, 16, 20, -20, 24, -24]
    tried_n = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 160:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1)
        if time.time() < d3:
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_r, c5_r = _grad_search(h_r, sub_dl2, tau=80.0, lr=0.01, momentum_beta=0.85)
            h_r, c5_r = _lp_loop(h_r, min(d3, time.time() + budget_s * 0.008), radius_init=0.05)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Perturbation + re-optimization
    d4 = t0 + budget_s * 0.82
    scale = 0.035
    for attempt in range(80):
        if time.time() > d4:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_try, c5_try = _lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.88)
        else:
            scale = min(0.2, scale * 1.18)

    # Phase 5: Evolutionary search at current best resolution
    if time.time() < t0 + budget_s * 0.90:
        d5 = t0 + budget_s * 0.90
        h_evo, c5_evo = _evolutionary(best_h, d5, pop_size=12, seed=seed + 7)
        if c5_evo < best_c5 - 1e-11:
            best_c5 = c5_evo
            best_h = h_evo.copy()

    # Phase 6: Final LP polish + pair refinement
    if time.time() < t0 + budget_s * 0.95:
        d6 = t0 + budget_s * 0.95
        h_f, c5_f = _lp_loop(best_h, min(d6, time.time() + budget_s * 0.03), radius_init=0.04)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < t0 + budget_s * 0.98:
        d7 = t0 + budget_s * 0.98
        h_p, c5_p = _pair_refine(best_h, d7, max_passes=40)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final
    if time.time() < t0 + budget_s * 0.995:
        h_f2, c5_f2 = _lp_loop(best_h, t0 + budget_s * 0.998, radius_init=0.02)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
