import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Greedy residual fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, idx):
    """Compute gradient of lag index idx (in full correlation output)."""
    n = len(h)
    s = idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[s:] += (1.0 - h[:m])
            grad[:m] -= h[s:]
    else:
        t = -s
        m = n - t
        if m > 0:
            grad[:m] += (1.0 - h[t:])
            grad[t:] -= h[:m]
    return grad

def all_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lp_step(h, V, active_idx, rho, n):
    """Solve LP to find direction that reduces active lags."""
    dx = 2.0 / n
    n_active = len(active_idx)
    if n_active == 0:
        return None
    
    # Build constraint matrix
    # Variables: u[0..n-1], t (total n+1)
    # Minimize: t (i.e., [0, ..., 0, 1])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: (dx * g_p)^T u - t <= -V[p]
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for i, p in enumerate(active_idx):
        g = lag_gradient(h, p) * dx
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -V[p]
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_i = max(-h[i], -rho)
        hi_i = min(1.0 - h[i], rho)
        bounds.append((lo_i, hi_i))
    # t is unbounded (or bounded by [0, V.max()])
    bounds.append((0.0, V.max()))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        u = res.x[:n]
        if np.max(np.abs(u)) > 1e-14:
            return u
    return None

def select_active(V, top_k=30, eta=0.01):
    """Select active lag indices: top-K plus those within eta of max."""
    cmax = V.max()
    mask = V >= cmax - eta
    active = np.where(mask)[0]
    if len(active) > top_k:
        # Keep top-K by value
        order = np.argsort(V)[::-1]
        active = order[:top_k]
    return active

def lp_refine(h0, max_iter=500, is_dead=None, verbose=False):
    """Main LP refinement loop."""
    h = h0.copy()
    n = len(h)
    best = h.copy()
    best_score = score_exact(h)
    
    rho = 0.2
    eta = 0.01
    top_k = 30
    stall = 0
    max_stall = 50
    
    for it in range(max_iter):
        if is_dead and is_dead():
            break
        
        V = all_lag_values(h)
        C = V.max()
        
        if C < best_score - 1e-14:
            best = h.copy()
            best_score = C
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            break
        
        # Select active lags
        active = select_active(V, top_k=top_k, eta=eta)
        
        # Solve LP
        u = lp_step(h, V, active, rho, n)
        if u is None:
            # Shrink or adjust
            rho *= 0.7
            if rho < 0.01:
                # Try expanding active set
                top_k = min(top_k + 10, 2 * n - 1)
                rho = 0.1
            continue
        
        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            trial = make_feasible(h + alpha * u)
            sc = score_exact(trial)
            if sc < best_score - 1e-12:
                h = trial
                best = h.copy()
                best_score = sc
                improved = True
                break
        
        if not improved:
            rho *= 0.8
            eta *= 0.9
            if rho < 0.005 or eta < 0.001:
                # Reset
                rho = 0.15
                eta = 0.005
                # Small random kick
                h = make_feasible(h + 0.02 * np.random.randn(n))
                stall = 0
    
    return best, best_score

def binary_from_continuous(h):
    """Create a balanced binary vector from continuous h."""
    n = len(h)
    target = n // 2
    # Sort by value, pick top target
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def binary_anneal(h0, max_steps=5000, is_dead=None):
    """Simulated annealing on balanced binary vectors."""
    n = len(h0)
    if n % 2 != 0:
        return h0, score_exact(h0)
    
    b = binary_from_continuous(h0)
    cur_score = score_exact(b)
    best_b = b.copy()
    best_score = cur_score
    
    pos1 = np.where(b == 1.0)[0]
    pos0 = np.where(b == 0.0)[0]
    
    T = 2e-4
    cool = 0.999
    
    for step in range(max_steps):
        if is_dead and is_dead():
            break
        
        # Random swap
        i = pos1[np.random.randint(len(pos1))]
        j = pos0[np.random.randint(len(pos0))]
        
        b[i] = 0.0
        b[j] = 1.0
        new_score = score_exact(b)
        
        delta = new_score - cur_score
        if delta < -1e-14 or np.random.random() < np.exp(-delta / T):
            cur_score = new_score
            # Update pos arrays
            pos1 = np.where(b == 1.0)[0]
            pos0 = np.where(b == 0.0)[0]
            if new_score < best_score:
                best_b = b.copy()
                best_score = new_score
        else:
            b[i] = 1.0
            b[j] = 0.0
        
        T *= cool
    
    return best_b, best_score

def interp_to(h_src, n_tgt):
    n_src = len(h_src)

def score_exact(h):
    n = len(h)
    if n < 2:
        return float('inf')

    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, idx):
    n = len(h)
    s = idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[s:] += (1.0 - h[:m])
            grad[:m] -= h[s:]
    else:
        t = -s
        m = n - t
        if m > 0:
            grad[:m] += (1.0 - h[t:])
            grad[t:] -= h[:m]
    return grad

        order = np.argsort(V[active])[::-1]

    return active

def lp_solve(h, V, active_idx, rho, n):

    n_active = len(active_idx)
    if n_active == 0:
        return None

    c_obj[n] = 1.0

    for i, p in enumerate(active_idx):
        g = lag_gradient(h, p) * dx
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -V[p]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_i = max(-h[i], -rho)
        hi_i = min(1.0 - h[i], rho)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, V.max() + 0.1))

        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:

        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Phase 2: LP with wider active set and larger trust region
    if not is_dead():
        h2 = best_h.copy()
        n2 = len(h2)
        V2 = all_lag_values(h2)
        active2 = select_active(V2, top_k=50, eta=0.025)
        result2 = lp_solve(h2, V2, active2, 0.35, n2)
        if result2 is not None:
            u2, _ = result2
            for alpha in [0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
                trial = make_feasible(h2 + alpha * u2)
                sc = score_exact(trial)
                if sc < best_score - 1e-13:
                    best_h = trial
                    best_score = sc
                    break

    # Phase 3: Multiple LP rounds with perturbations
    for round_i in range(6):
        if is_dead():
            break
        sigma = 0.02 + 0.01 * round_i
        h_pert = make_feasible(best_h + sigma * rng.normal(0, 1, size=len(best_h)))
        h_new, sc = lp_refine(h_pert, max_iter=250, is_dead=is_dead,
                              initial_rho=0.2, initial_eta=0.01)
        if sc < best_score - 1e-13:
            best_h = h_new
            best_score = sc

    # Phase 4: Anti-symmetric exploration
    if not is_dead():
        h_sym = anti_sym_map(best_h)
        sc_sym = score_exact(h_sym)
        if sc_sym < best_score - 1e-13:
            best_h = h_sym
            best_score = sc_sym
        if not is_dead():
            h_new, sc = lp_refine(h_sym, max_iter=200, is_dead=is_dead)
            if sc < best_score - 1e-13:
                best_h = h_new
                best_score = sc

    # Phase 5: Binary annealing for escape
    if not is_dead():
        b, b_score = binary_anneal(best_h, max_steps=5000, is_dead=is_dead)
        if not is_dead():
            h_from_bin = b.copy()
            h_new, sc = lp_refine(h_from_bin, max_iter=250, is_dead=is_dead,
                                  initial_rho=0.3, initial_eta=0.015)
            if sc < best_score - 1e-13:
                best_h = h_new
                best_score = sc

    # Phase 6: Resolution scanning
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set([n_base + dn for dn in [-10, -6, -4, 4, 6, 10]
                             if 32 <= n_base + dn <= 100]))
        for nc in [44, 52, 56, 64, 72, 80]:
            if nc not in n_opts and nc != n_base:
                n_opts.append(nc)
        n_opts = sorted(set(n_opts))
        for n_opt in n_opts:
            if is_dead():
                break
            h_int = interp_to(best_h, n_opt)
            h_new, sc = lp_refine(h_int, max_iter=200, is_dead=is_dead,
                                  initial_rho=0.2, initial_eta=0.01)
            if sc < best_score - 1e-13:
                best_h = h_new
                best_score = sc

    # Phase 7: Final intensive LP
    if not is_dead():
        h_new, sc = lp_refine(best_h, max_iter=500, is_dead=is_dead,
                              initial_rho=0.15, initial_eta=0.008)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Phase 8: Two-point polish
    if not is_dead():
        remaining = deadline - time.time()
        if remaining > 20:
            n_tries = max(2000, int(remaining * 3))
            best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 9: Final LP after 2-point
    if not is_dead():
        h_new, sc = lp_refine(best_h, max_iter=300, is_dead=is_dead,
                              initial_rho=0.12, initial_eta=0.006)
        if sc < best_score:
            best_h = h_new
            best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8

    return final_h, final_score, final_n
