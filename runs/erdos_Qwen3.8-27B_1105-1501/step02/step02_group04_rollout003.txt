```python
def lp_direction(h, corr, trust):
    n = len(h)
    m = len(corr)
    max_corr = np.max(corr)
    slack = max_corr - corr  # shape (m,)
    
    D = build_direction_matrix(h)  # shape (m, n)
    
    # Variables: [d_0, ..., d_{n-1}, theta]
    # Minimize -theta => c = [0,...,0, -1]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality: D[l, :] @ d - theta <= slack[l] for all l
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, -1] = -1.0
    b_ub = slack
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: d_i in [-trust, trust], theta free
    bounds = [(-trust, trust)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, 0.0
    
    d = res.x[:n]
    theta = res.x[-1]
    
    if theta < 1e-10:
        return None, 0.0
    
    return d, theta
```

```python
import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pair_optimal_delta(h, i, j):
    if i == j:
        return 0.0
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(25):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def full_pair_sweep(h, deadline=None, max_passes=3):
    n = len(h)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            return
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                return
            for j in range(i + 1, n):
                delta = pair_optimal_delta(h, i, j)
                if abs(delta) > 1e-12:
                    h[i] -= delta
                    h[j] += delta

def build_direction_matrix(h):
    n = len(h)
    m = 2 * n - 1
    D = np.zeros((m, n))
    omh = 1.0 - h
    for p in range(n):
        e = np.zeros(n)
        e[p] = 1.0
        D[:, p] = np.correlate(e, omh, mode="full") - np.correlate(h, e, mode="full")
    return D

def lp_direction(h, corr, trust):
    n = len(h)
    m = len(corr)
    max_corr = np.max(corr)
    slack = max_corr - corr
    
    D = build_direction_matrix(h)
    
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, -1] = -1.0
    b_ub = slack.copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-trust, trust)] * n + [(None, None)]
    
    try:

    except Exception:

        return None, 0.0
    
    d = res.x[:n].copy()
    theta = res.x[-1]
    
    if theta < 1e-8:
        return None, 0.0
    
    # Ensure sum(d) = 0 (numerical fix)
    d -= np.mean(d)
    
    return d, theta

def max_feasible_alpha(h, d):

    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
```
