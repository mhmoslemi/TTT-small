    h = h0.copy()

    dx = 2.0 / n
    best_score = official_score(h)
    
    for it in range(max_iter):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break

        max_val = np.max(corr)
        # Active set: lags within 1% of max
        threshold = max_val * 0.99
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Average gradient over active lags
        grad = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            for i in range(n):
                g = 0.0
                if 0 <= i + d < n:
                    g += (1.0 - h[i + d])
                if 0 <= i - d < n:
                    g += -h[i - d]
                grad[i] += g
        grad /= len(active)
        grad *= dx
        
        # Try step sizes
        best_step = None
        best_s = best_score
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.5]:
            h_new = make_feasible(h - alpha * grad)
            s = official_score(h_new)
            if s < best_s:
                best_s = s
                best_step = h_new
        
        if best_step is not None:
            h = best_step
            best_score = best_s
        else:
            break  # No improvement possible with this gradient
    
    return h, best_score
