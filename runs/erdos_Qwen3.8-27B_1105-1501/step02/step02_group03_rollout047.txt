import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    # Bisection to make sum = n/2, clip to [0,1]
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0: break
            h[idx] += add
            residual -= add
            if residual < 1e-14: break
    elif residual < -1e-12:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 0: break
            h[idx] -= sub
            residual += sub
            if residual > -1e-14: break
    return h

def antisym_project(h):
    # Project h to anti-symmetric family: h[i] + h[n-1-i] = 1
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = 0.5 * (h[i] + 1.0 - h[n-1-i])
    # For even n, this automatically satisfies sum = n/2
    # For odd n, center element is 0.5
    return make_feasible(h_sym)

def antisym_to_left(h):
    # Extract left half from anti-symmetric h
    n = len(h)
    m = n // 2
    return h[:m].copy()

def left_to_antisym(left, n):
    # Reconstruct full h from left half
    m = n // 2
    h = np.zeros(n)
    h[:m] = left
    h[m:] = 1.0 - left[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def compute_overlap_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_overlap(h):
    # Returns gradient of each overlap value with respect to h
    # Shape: (2n-1, n)
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        s = p - (n-1)
        if s >= 0:
            L = n - s
            # val = dx * sum_{i=0}^{L-1} h[i+s]*(1-h[i])
            # d/dh[i+s] = dx*(1-h[i]), d/dh[i] = -dx*h[i+s]
            G[p, s:s+L] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[s:s+L]
        else:
            t = -s
            L = n - t
            # val = dx * sum_{i=0}^{L-1} h[i]*(1-h[i+t])
            # d/dh[i] = dx*(1-h[i+t]), d/dh[i+t] = -dx*h[i]
            G[p, :L] += dx * (1.0 - h[t:t+L])
            G[p, t:t+L] -= dx * h[:L]
    return G

def grad_overlap_reduced(h, left):
    # Gradient of each overlap value with respect to left variables
    # h is full, left is the left half (m values)
    n = len(h)
    m = n // 2
    G_full = grad_overlap(h)
    G_red = np.zeros((2*n-1, m))
    for j in range(m):
        G_red[:, j] = G_full[:, j] - G_full[:, n-1-j]
    return G_red

def logsumexp_obj_grad(h, left, beta):
    vals = compute_overlap_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    
    G_red = grad_overlap_reduced(h, left)
    grad = w @ G_red
    return obj, grad

def optimize_reduced(left, beta, time_limit, max_iter=5000):

    n_total = 2 * len(left)  # assume even

                       options={'maxiter': max_iter, 'gtol': 1e-12, 'disp': False})

            if res.success or res.status == 1:

def active_set_solve(h, active_idx, max_iter=300, time_limit=30.0):
    # SLSQP minimax with active constraints in full space
    n = len(h)
    t0 = time.time()
    
    def make_func(z):
        h_z = z[:n]
        t = z[n]
        return t
    
    def make_jac(z):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def make_ineq(z):
        h_z = z[:n]
        t = z[n]
        vals = compute_overlap_values(h_z)
        return t - vals[active_idx]
    
    def make_ineq_jac(z):
        h_z = z[:n]
        G = grad_overlap(h_z)
        J = -G[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])
    
    def make_eq(z):
        return np.sum(z[:n]) - n/2.0
    
    def make_eq_jac(z):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    cmax = compute_overlap_values(h).max()
    z0 = np.concatenate([h, [cmax + 1e-6]])
    
    cons = [
        {'type': 'ineq', 'fun': make_ineq, 'jac': make_ineq_jac},
        {'type': 'eq', 'fun': make_eq, 'jac': make_eq_jac}
    ]
    
    try:
        res = minimize(make_func, z0, jac=make_jac, method='SLSQP',
                       bounds=[(0.0, 1.0)]*n + [(0.0, 3.0)],

        if res.x is not None and np.all(np.isfinite(res.x)):

def find_active(vals, top_frac=0.3, min_count=6):

    threshold = cmax - max(0.005 * cmax, 1e-6)
