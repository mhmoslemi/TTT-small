        grad[:n-m] += (-h[m:])

def projected_gradient_step(h, grad, step_size):

    h_new = h - step_size * grad

    """
    Projected gradient descent focusing on active lags.
    """
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = official_score(h)
    step = 0.1
    min_step = 0.001
    
    for it in range(n_iter):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        
        corr = corr_full(h)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]

        threshold = max_val - 0.05 * abs(max_val) - 0.1
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([max_idx])
        
        # Weighted gradient
        # Use exponential weighting
        weights = np.exp((corr[active] - max_val) * 10)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, active):
            s = k - (n - 1)
            grad += w * lag_grad_vec(h, s)
        
        # Backtracking line search
        cur_score = official_score(h)
        for trial in range(20):
            h_new = projected_gradient_step(h, grad, step)
            new_score = official_score(h_new)
            if new_score < cur_score:
                h = h_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
                step = min(step * 1.1, 0.5)
                break
            else:
                step *= 0.7
                if step < min_step:
                    break
        else:
            # No improvement, try smaller step
            step *= 0.5
            if step < min_step:
                break
    
    return best_h, best_score

def binary_2flip_local_search(h, max_rounds=100, time_limit=None, t_start=None):
    """
    Local search on binary (or near-binary) h using 2-flip moves.
    Tries all pairs (i,j) with h[i] > h[j] and accepts the best improvement.
    """
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_score = official_score(h)
    
    for rnd in range(max_rounds):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        
        corr = corr_full(h)  # length 2n-1
        current_max = np.max(corr)
        
        # Precompute profiles
        # profile[p](s) = (1-h[p+s]) - h[p-s] for valid indices
        # s ranges from -(n-1) to n-1, indexed as 0 to 2n-2 in corr
        profiles = np.zeros((n, 2*n-1))
        for p in range(n):
            for s_idx in range(2*n-1):
                s = s_idx - (n-1)
                val = 0.0
                ps = p + s
                pm = p - s
                if 0 <= ps < n:
                    val += (1.0 - h[ps])
                if 0 <= pm < n:
                    val -= h[pm]
                profiles[p, s_idx] = val
        
        # Find positions to swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 0
        best_pair = None
        best_new_corr = None
        
        # For near-binary, handle the 0.5 case
        # For now, just use threshold
        for i in ones:
            for j in zeros:
                # ΔR(s) = profiles[i](s) - profiles[j](s)
                new_corr = corr + profiles[i] - profiles[j]
                new_max = np.max(new_corr)
                improve = current_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
                    best_new_corr = new_corr
        
        if best_pair is None or best_improve <= 1e-14:
            break
        
        i, j = best_pair
        # Swap: h[i] -> 1-h[i], h[j] -> 1-h[j] (for binary)
        # For near-binary: h[i] += (1-2*h[i]) * 0.5... 
        # Actually for binary: just flip
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        h = make_feasible(h)
        
        new_score = official_score(h)
        if new_score < best_score:
            best_score = new_score
    
    return h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def two_flip_search(h, rng, max_rounds=500, time_limit=None, t_start=None, temperature=0.0):
    """
    2-flip local search with optional annealing.
    For binary h, this is exact.
    """
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = official_score(h)
    
    for rnd in range(max_rounds):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        
        corr = corr_full(h)
        current_max = np.max(corr)
        profiles = compute_profiles(h)
        
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            # Try 1-flip or accept
            break
        
        best_improve = -np.inf
        best_pair = None
        
        # Vectorized: for each pair (i,j), new_max = max(corr + profiles[i] - profiles[j])
        # This is O(|ones| * |zeros| * n)
        
        for i in ones:
            delta_i = profiles[i]  # shape (2n-1,)
            for j in zeros:
                delta = delta_i - profiles[j]
                new_corr = corr + delta
                new_max = np.max(new_corr)
                improve = current_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        
        # Accept with probability based on temperature
        if best_improve >= 0 or temperature > 0:
            if best_improve >= 0 or rng.random() < np.exp(best_improve / (temperature + 1e-10)):
                h[i] = 1.0 - h[i]
                h[j] = 1.0 - h[j]
                new_score = official_score(h)
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
        else:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_profiles(h):
    n = len(h)
    one_minus_h = 1.0 - h
    h_rev = h[::-1].copy()
    nm1 = 2 * n - 1
    profiles = np.zeros((n, nm1))
    for p in range(n):
        profiles[p, n - 1 - p: 2 * n - 1 - p] = one_minus_h
        profiles[p, p: p + n] -= h_rev
    return profiles

def round_to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    if n % 2 == 0:
        n_ones = n // 2
        h_bin[sorted_idx[:n_ones]] = 1.0
    else:
        n_ones = (n - 1) // 2
        h_bin[sorted_idx[:n_ones]] = 1.0
        h_bin[sorted_idx[n_ones]] = 0.5
    return h_bin

def two_flip_search(h, rng, max_rounds=500, time_limit=None, t_start=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = official_score(h)
    stall_count = 0
    
    for rnd in range(max_rounds):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        corr = corr_full(h)
        cur_max = corr.max()
        profiles = compute_profiles(h)
        
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = -np.inf
        best_i, best_j = -1, -1
        best_new_max = cur_max

def lag_grad_vec(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    else:
        m = -s
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] += (-h[m:])
    return grad

def projected_gradient_refine(h, max_iter=300, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = official_score(h)
    step = 0.08
    min_step = 0.002
    
    for it in range(max_iter):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        corr = corr_full(h)
        max_val = corr.max()
        
        # Active set: lags within 10% of max
        threshold = max_val - 0.1 * abs(max_val)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Exponential weighting
        w = np.exp((corr[active] - max_val) * 8.0)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, k in zip(w, active):
            s = k - (n - 1)
            grad += wi * lag_grad_vec(h, s)
        
        # Backtracking
        improved = False
        for trial in range(12):
            h_new = make_feasible(h - step * grad)
            s_new = official_score(h_new)
            s_cur = official_score(h)
            if s_new < s_cur - 1e-14:
                h = h_new
                if s_new < best_score:
                    best_score = s_new
                    best_h = h.copy()
                step = min(step * 1.1, 0.4)
                improved = True
                break
            step *= 0.65
            if step < min_step:
                break
        if not improved:
            step = max(step * 0.5, min_step)
            if step < min_step * 1.1:
                break
    
    return best_h, best_score

def simulated_annealing_hill_climb(h, rng, max_iter=5000, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = official_score(h)
    T = 0.015
    T_min = 0.0005
    cool = (T_min / T) ** (1.0 / max(max_iter, 1))
    
    for it in range(max_iter):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Swap amounts to keep sum
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.3, 1.0)
        if delta < 0.005:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = official_score(h)
        diff = new_score - best_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        T *= cool
    
    return best_h, best_score
