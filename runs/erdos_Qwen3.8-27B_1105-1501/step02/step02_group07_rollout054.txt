def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue

            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:

    c = np.zeros(n + 1)

    b_rows = []

            gp[0:L] -= h[s:n]

  - This sets grad[k] -= w*h[k+s] for k in [0, L-1] = [0, n-s-1]. ✓ matches second term.

For s < 0 (t = -s):
- `grad[0:L] += w * (1.0 - h[t:n])` where L = n-t
  - This sets grad[k] += w*(1-h[k+t]) for k in [0, L-1] = [0, n-t-1]. ✓ matches first term.
- `grad[t:n] -= w * h[0:L]` where L = n-t
  - This sets grad[k] -= w*h[k-t] for k in [t, n-1]. ✓ matches second term.

Great, the gradient computation is correct.

Now, one issue: the gradient is with respect to the unweighted sum. The weights w are already incorporated (they're the softmax weights). So `fast_gradient(h, w)` gives ∑_k w_k * dV_k/dh, which is the gradient of ∑_k w_k * V_k = the gradient of the log-sum-exp (up to the 1/beta factor and the V_max offset).

Wait, actually the gradient of logsumexp(V, beta) = (∑_k exp(beta*(V_k - Vmax)) * dV_k) / (∑_k exp(beta*(V_k - Vmax))) = ∑_k w_k * dV_k where w_k are the softmax weights. So yes, `fast_gradient(h, w)` is exactly the gradient of the log-sum-exp.

But wait, there's a subtlety: the log-sum-exp is (1/beta) * log(∑exp(beta*V_k)) + V_max. The gradient is (∑exp(beta*V_k)*dV_k) / (∑exp(beta*V_k)) = ∑w_k*dV_k. So the gradient of the log-sum-exp is just the weighted average of the individual gradients, regardless of beta. The beta only affects the weights w_k.

So the gradient magnitude doesn't change with beta, only the direction (which V_k's are emphasized). This means Adam's adaptive learning rate should work well: for coordinates where the gradient is consistently large, Adam will take smaller steps (dividing by sqrt(v)), and for coordinates where the gradient varies, it will take larger steps.

import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def adam_optimize(h, deadline, beta0=5.0, beta1=300.0, lr=0.02, max_iters=100000, cooldown=200):
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_score = official_score(h)
    no_improve = 0
    
    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        if no_improve > cooldown and t % 100 == 0:
            h = make_feasible(best_h + np.random.default_rng(t).normal(0, 0.01, n))
            m *= 0.5
            v *= 0.5
            no_improve = 0
            continue
        V = compute_values(h)
        fraction = min(t / max_iters, 1.0)
        beta_t = beta0 * (beta1 / beta0) ** fraction
        w = softmax_weights(V, beta_t)
        grad = fast_gradient(h, w)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        h_new = make_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps))
        score = official_score(h_new)
        
        if score < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
    
    return best_h, best_score

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_polish(h, deadline, max_rounds=200):
    h_cur = h.copy()
    best_h = h.copy()
    best_score = official_score(h)
    for eps in [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]:
        for _ in range(max_rounds):

                break
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
    return best_h, best_score
