        valid2 = (j + s_vals >= 0) & (j + s_vals < n) & (s_vals != 0)

    return sens

def _gradient_lse(h, beta):
    n = len(h)
    corr = _corr_full(h)
    sh = beta * (corr - corr.max())
    ev = np.exp(np.clip(sh, -500, 0))
    w = ev / ev.sum()
    grad = np.zeros(n)
    for j in range(n):
        sens = _sensitivity(h, j)
        grad[j] = np.dot(w, sens)
    return grad

def _coord_descent(h, n_sweeps=100, n_grid=200):
    n = len(h)
    h = h.copy()
    s_vals = np.arange(2 * n - 1) - (n - 1)
    s0_idx = n - 1
    xs_grid = np.linspace(0.0, 1.0, n_grid)
    for sweep in range(n_sweeps):
        for j in range(n):
            old = h[j]
            corr = _corr_full(h)
            sens = _sensitivity(h, j)
            # Evaluate on grid
            dx_arr = xs_grid - old
            vals = corr[np.newaxis, :] + dx_arr[:, np.newaxis] * sens[np.newaxis, :]
            vals[:, s0_idx] = corr[s0_idx] + dx_arr * (1.0 - xs_grid - old)
            objs = np.max(vals, axis=1)
            best_idx = int(np.argmin(objs))
            best_x = xs_grid[best_idx]
            # Refine if interior
            if 0 < best_idx < n_grid - 1:
                lo = xs_grid[best_idx - 1]
                hi = xs_grid[best_idx + 1]
                for _ in range(20):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    d1 = m1 - old
                    d2 = m2 - old
                    v1 = corr + d1 * sens
                    v1[s0_idx] = corr[s0_idx] + d1 * (1.0 - m1 - old)
                    v2 = corr + d2 * sens
                    v2[s0_idx] = corr[s0_idx] + d2 * (1.0 - m2 - old)
                    if np.max(v1) < np.max(v2):
                        hi = m2
                    else:
                        lo = m1
                best_x = (lo + hi) / 2.0
            h[j] = best_x
        h = _project(h)
    return h

def _fista_topk(h, beta, n_iter, top_k=5, alpha=0.01, mu=0.9):
    """FISTA with top-k gradient."""
    n = len(h)
    x = h.copy()
    x_prev = h.copy()
    y = h.copy()
    x_avg = h.copy()
    best = h.copy()
    best_s = _score(h)
    t = 1.0
    s_vals = np.arange(2 * n - 1) - (n - 1)
    for it in range(n_iter):
        corr = _corr_full(y)
        # Top-k weights
        idx = np.argsort(corr)[::-1][:top_k]
        cv = corr[idx]
        cv_shifted = beta * (cv - cv.max())
        ev = np.exp(np.clip(cv_shifted, -500, 0))
        w_top = ev / ev.sum()
        # Gradient
        grad = np.zeros(n)
        for j in range(n):
            sens = _sensitivity(y, j)
            # Only use weights for top-k lags
            w_full = np.zeros(2 * n - 1)
            w_full[idx] = w_top
            grad[j] = np.dot(w_full, sens)
        # FISTA update
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x + (t_prev / t) * (x - x_prev)
        y_new = _project(y_new)
        x_prev = x
        x = _project(y_new - alpha * grad)
        x_avg += (x - x_avg) / (it + 1)
        y = y_new
        if it % 50 == 0:
            s = _score(x)
            if s < best_s:
                best_s = s
                best = x.copy()
    # Check averaged
    x_avg = _project(x_avg)
    s_avg = _score(x_avg)
    if s_avg < best_s:
        best = x_avg.copy()
        best_s = s_avg
    s_final = _score(x)
    if s_final < best_s:
        best = x.copy()
        best_s = s_final
    return best, best_s

def _slsqp_topk(h0, top_k=5, maxiter=100):
    """SLSQP with top-k max objective."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        corr = _corr_full(x)
        idx = np.argsort(corr)[::-1][:top_k]
        return corr[idx].max()
    
    def _obj_grad(x):
        corr = _corr_full(x)
        idx = np.argsort(corr)[::-1][:top_k]
        # Gradient of max of top-k (approximate: use the argmax)
        k_max = np.argmax(corr[idx])
        k_actual = idx[k_max]
        s = k_actual - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            sens = _sensitivity(x, j)
            grad[j] = sens[k_actual]
        return grad
    
    def _eq(x):
        return x.sum() - n / 2.0
    
    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sens(h, j):
    n = len(h)
    s_vals = np.arange(2 * n - 1) - (n - 1)
    sv = np.zeros(2 * n - 1)
    v1 = (j - s_vals >= 0) & (j - s_vals < n) & (s_vals != 0)
    v2 = (j + s_vals >= 0) & (j + s_vals < n) & (s_vals != 0)
    sv[v1] = 1.0 - h[j - s_vals[v1]]
    sv[v2] += -h[j + s_vals[v2]]
    sv[n - 1] = 1.0 - 2.0 * h[j]
    return sv

def _grad_lse(h, beta, top_k=None):
    n = len(h)
    c = _corr(h)
    if top_k is not None:
        idx = np.argsort(c)[::-1][:top_k]
        cv = c[idx]
        sh = beta * (cv - cv.max())
        ev = np.exp(np.clip(sh, -500, 0))
        w = np.zeros(2 * n - 1)
        w[idx] = ev / ev.sum()
    else:
        sh = beta * (c - c.max())
        ev = np.exp(np.clip(sh, -500, 0))
        w = ev / ev.sum()
    g = np.zeros(n)
    for j in range(n):
        g[j] = np.dot(w, _sens(h, j))
    return g

def _cd(h, n_sweeps, n_grid=200):
    """Coordinate descent."""
    n = len(h)
    h = h.copy()
    s0 = n - 1
    xs = np.linspace(0.0, 1.0, n_grid)
    for _ in range(n_sweeps):
        for j in range(n):
            old = h[j]
            c = _corr(h)
            sv = _sens(h, j)
            d = xs - old
            v = c[np.newaxis, :] + d[:, np.newaxis] * sv[np.newaxis, :]
            v[:, s0] = c[s0] + d * (1.0 - xs - old)
            obj = np.max(v, axis=1)
            bi = int(np.argmin(obj))
            bx = xs[bi]
            if 0 < bi < n_grid - 1:
                lo = xs[bi - 1]
                hi = xs[bi + 1]
                for _ in range(20):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    d1 = m1 - old
                    d2 = m2 - old
                    v1 = c + d1 * sv
                    v1[s0] = c[s0] + d1 * (1.0 - m1 - old)
                    v2 = c + d2 * sv
                    v2[s0] = c[s0] + d2 * (1.0 - m2 - old)
                    if np.max(v1) < np.max(v2):
                        hi = m2
                    else:
                        lo = m1
                bx = (lo + hi) / 2.0
            h[j] = bx
        h = _project(h)
    return h

def _fista(h0, beta, n_iter, top_k=5, alpha=0.005):
    n = len(h0)
    x = h0.copy()
    xp = h0.copy()
    y = h0.copy()
    xavg = h0.copy()
    best = h0.copy()
    best_s = _score(h0)
    t = 1.0
    for it in range(n_iter):
        g = _grad_lse(y, beta, top_k)
        tp = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = _project(x + (tp / t) * (x - xp))
        xp = x
        x = _project(y - alpha * g)
        xavg += (x - xavg) / (it + 1)
        if it % 100 == 0:
            s = _score(x)
            if s < best_s:
                best_s = s
                best = x.copy()
    xavg = _project(xavg)
    sa = _score(xavg)
    if sa < best_s:
        best, best_s = xavg.copy(), sa
    sf = _score(x)
    if sf < best_s:
        best, best_s = x.copy(), sf
    return best, best_s

def _slsqp_direct(h0, maxiter=100):
    n = len(h0)
    c0 = _corr(h0)
    t0 = min(c0.max() * (2.0 / n) + 0.002, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    
    def _f(z):
        return z[-1]
    
    def _fjac(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def _ieq(z):
        h = z[:-1]
        t = z[-1] * (2.0 / n)
        return t - _corr(h)
    
    def _eq(z):
        return z[:-1].sum() - n / 2.0
    
    def _eqjac(z):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    try:
        res = minimize(_f, z0, jac=_fjac, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'ineq', 'fun': _ieq},

                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0
