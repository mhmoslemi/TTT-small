        idx = np.arange(s, n)
        g[idx] = (1.0 - h[idx - s])
        idx2 = np.arange(0, n - s)

    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)

                residual += sub

def interp_to(h_src, n_tgt):
    ns = len(h_src)
    sc = (np.arange(ns) + 0.5) * 2.0 / ns
    tc = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return make_feasible(np.interp(tc, sc, h_src))

def qp_step(h0, k_active, reg):
    n = len(h0)
    vals = compute_f(h0)
    top = np.argsort(vals)[-k_active:]
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    for p in top:
        s = p - (n - 1)
        g = dx * gradient_vec(h0, s)
        cons.append(vals[p] + g @ (h_var - h0) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0)), cons)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    hx = h_var.value
    if hx is None or not np.all(np.isfinite(hx)):
        return None
    return make_feasible(hx)

def qp_loop(h0, max_iter, k_active, reg_start, reg_end):
    h = h0.copy()
    best = official_score(h)
    stall = 0
    for it in range(max_iter):
        reg = reg_start * (reg_end / reg_start) ** (it / max(1, max_iter - 1))
        hnew = qp_step(h, k_active, reg)
        if hnew is None:
            break
        sc = official_score(hnew)
        if sc < best - 1e-14:
            h, best, stall = hnew, sc, 0
        else:
            acc = False
            for a in [0.5, 0.25, 0.1, 0.05]:
                ht = make_feasible(h + a * (hnew - h))
                st = official_score(ht)
                if st < best - 1e-14:
                    h, best, stall = ht, st, 0
                    acc = True
                    break
            if not acc:
                stall += 1
                if stall >= 5:
                    break
    return h, best

def smooth_obj(h, beta):
    vals = compute_f(h)
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def slsqp_eq_fun(x):
    return x.sum() - len(x) / 2.0

def slsqp_eq_jac(x):
    return np.ones(len(x))

def try_slsqp(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hnew = make_feasible(res.x)
            sc = official_score(hnew)
            if np.isfinite(sc):
                return hnew, sc
    except Exception:
        pass
    return None, None

def subgradient_opt(h0, n_iter=200, rate_start=0.02, rate_end=0.001):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best = official_score(h)
    best_h = h.copy()
    for it in range(n_iter):
        vals = compute_f(h)
        vmax = vals.max()
        active = np.where(vals > vmax - max(1e-4, 0.005 * vmax))[0]
        if len(active) == 0:
            active = np.argmax(vals)
            active = np.array([active])
        grad = np.zeros(n)
        for p in active:
            s = p - (n - 1)
            grad += dx * gradient_vec(h, s)
        grad /= len(active)
        rate = rate_start * (rate_end / rate_start) ** (it / max(1, n_iter - 1))
        h_try = make_feasible(h - rate * grad)
        sc = official_score(h_try)
        if sc < best - 1e-15:
            h, best, best_h = h_try, sc, h_try.copy()
    return best_h, best

def anti_sym(h):
    return make_feasible(1.0 - h[::-1].copy())

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15.0
    rng = np.random.default_rng(seed)

    def rem():
        return deadline - time.time()

    best_h = None
    best_sc = float('inf')

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_sc = ps
    except (NameError, AttributeError, Exception):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x / 2.0))
        best_sc = official_score(best_h)

    n_base = len(best_h)

    # Quick transforms
    for tf_name in ['flip', 'anti', 'complement']:
        if rem() < 5:
            break
        if tf_name == 'flip':
            cand = make_feasible(best_h[::-1].copy())
        elif tf_name == 'anti':
            cand = anti_sym(best_h)
        else:
            cand = make_feasible(1.0 - best_h)
        sc = official_score(cand)
        if sc < best_sc - 1e-14:
            best_h, best_sc = cand, sc

    # Phase 1: Active-set QP with various K
    for k_a in [3, 5, 2, 8, 10]:
        if rem() < 15:
            break
        h_opt, sc_opt = qp_loop(best_h, max_iter=30, k_active=k_a,
                                reg_start=2.0, reg_end=0.003)
        if sc_opt < best_sc - 1e-14:
            best_h, best_sc = h_opt, sc_opt

    # Phase 2: SLSQP polish
    if rem() > 30:
        cur = best_h.copy()
        for beta in [2000, 10000, 50000, 200000]:
            if rem() < 10:
                break
            h_new, sc = try_slsqp(cur, beta, maxiter=150)
            if h_new is not None and sc < best_sc - 1e-14:
                best_h, best_sc = h_new, sc
                cur = h_new
            elif h_new is not None:
                sc_cur = official_score(cur)
                if sc < sc_cur - 1e-14:
                    cur = h_new

    # Phase 3: Subgradient with different rates
    if rem() > 30:
        for rs in [0.03, 0.01, 0.005]:
            if rem() < 10:
                break
            h_opt, sc_opt = subgradient_opt(best_h, n_iter=150,
                                             rate_start=rs, rate_end=0.0005)
            if sc_opt < best_sc - 1e-14:
                best_h, best_sc = h_opt, sc_opt
            # Follow with QP
            if rem() > 10:
                h_opt2, sc_opt2 = qp_loop(h_opt, max_iter=20, k_active=5,
                                           reg_start=1.0, reg_end=0.003)
                if sc_opt2 < best_sc - 1e-14:
                    best_h, best_sc = h_opt2, sc_opt2

    # Phase 4: Random perturbations + re-optimize
    if rem() > 60:
        for i in range(12):
            if rem() < 15:
                break
            sigma = 0.05 + 0.15 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = qp_loop(h_cand, max_iter=20, k_active=5,
                                     reg_start=1.0, reg_end=0.005)
            if sc_opt < best_sc - 1e-14:
                best_h, best_sc = h_opt, sc_opt
            if rem() > 15:
                h_new, sc = try_slsqp(h_cand, 5000, maxiter=100)
                if h_new is not None and sc < best_sc - 1e-14:
                    best_h, best_sc = h_new, sc

    # Phase 5: Resolution sweep
    if rem() > 80:
        n_base = len(best_h)
        n_opts = sorted(set([n_base - 4, n_base - 2, n_base,
                             n_base + 2, n_base + 4, n_base + 8,
                             max(20, n_base - 8), min(100, n_base + 12),
                             max(16, n_base - 12)]))
        for n_opt in n_opts:
            if rem() < 15:
                break
            if n_opt < 16 or n_opt % 2 != 0:
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_sc - 1e-14:
                best_h, best_sc = h_int, sc_int
            # Optimize at this resolution
            h_opt, sc_opt = qp_loop(h_int, max_iter=25, k_active=5,
                                     reg_start=2.0, reg_end=0.003)
            if sc_opt < best_sc - 1e-14:
                best_h, best_sc = h_opt, sc_opt
            if rem() > 20:
                h_new, sc = try_slsqp(h_opt, 20000, maxiter=120)
                if h_new is not None and sc < best_sc - 1e-14:
                    best_h, best_sc = h_new, sc
                h_opt2, sc_opt2 = qp_loop(h_new if h_new is not None else h_opt,
                                           max_iter=15, k_active=4,
                                           reg_start=0.5, reg_end=0.002)
                if sc_opt2 < best_sc - 1e-14:
                    best_h, best_sc = h_opt2, sc_opt2

    # Phase 6: Anti-symmetric refinement
    if rem() > 30:
        h_sym = anti_sym(best_h)
        sc_sym = official_score(h_sym)
        if sc_sym < best_sc - 1e-14:
            best_h = h_sym
            best_sc = sc_sym
        h_opt, sc_opt = qp_loop(h_sym, max_iter=25, k_active=5,
                                 reg_start=1.5, reg_end=0.003)
        if sc_opt < best_sc - 1e-14:
            best_h, best_sc = h_opt, sc_opt
        if rem() > 10:
            h_new, sc = try_slsqp(h_opt, 50000, maxiter=120)
            if h_new is not None and sc < best_sc - 1e-14:
                best_h, best_sc = h_new, sc

    # Phase 7: Intensive final polish
    if rem() > 40:
        for k_a in [2, 3, 5, 7]:
            if rem() < 10:
                break
            h_opt, sc_opt = qp_loop(best_h, max_iter=25, k_active=k_a,
                                     reg_start=0.5, reg_end=0.001)
            if sc_opt < best_sc - 1e-14:
                best_h, best_sc = h_opt, sc_opt
        if rem() > 20:
            for beta in [100000, 500000]:
                h_new, sc = try_slsqp(best_h, beta, maxiter=150)
                if h_new is not None and sc < best_sc - 1e-14:
                    best_h, best_sc = h_new, sc
        if rem() > 15:
            h_opt, sc_opt = subgradient_opt(best_h, n_iter=100,
                                             rate_start=0.005, rate_end=0.0003)
            if sc_opt < best_sc - 1e-14:
                best_h, best_sc = h_opt, sc_opt
            h_opt2, sc_opt2 = qp_loop(best_h, max_iter=20, k_active=5,
                                       reg_start=0.3, reg_end=0.001)
            if sc_opt2 < best_sc - 1e-14:
                best_h, best_sc = h_opt2, sc_opt2

    # Final validation
    final_h = make_feasible(best_h)
    final_sc = official_score(final_h)
    final_bound = final_sc + 5e-15
    return final_h, final_bound, len(final_h)
