from scipy.optimize import linprog

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_D(h):
    """Build the D matrix where D[:, k] = correlate(e_k, 1-h) - correlate(h, e_k)."""
    n = len(h)
    one_minus_h = 1.0 - h
    D = np.zeros((2*n - 1, n))
    for k in range(n):
        e_k = np.zeros(n)
        e_k[k] = 1.0
        D[:, k] = np.correlate(e_k, one_minus_h, mode="full") - np.correlate(h, e_k, mode="full")
    return D

def solve_lp_direction(D, corr, trust):
    """Solve the LP to find direction d and predicted max correlation t."""
    n = D.shape[1]
    m = D.shape[0]
    # Variables: [d_0, ..., d_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, -1] = -1.0
    b_ub = -corr
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-trust, trust)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def line_search(h, d, corr, trust):
    """Find best alpha in [0, alpha_max] minimizing max_l f_l(alpha)."""
    n = len(h)
    one_minus_h = 1.0 - h
    a = np.correlate(d, one_minus_h, mode="full") - np.correlate(h, d, mode="full")
    b = np.correlate(d, d, mode="full")
    
    # Feasibility: alpha_max

        return 0.0
    alpha_max = np.inf

    if np.any(neg):
        alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))

    alpha_max = min(alpha_max, 1.0)  # cap at 1 for safety

        vals = corr + alpha * a - alpha * alpha * b
        return np.max(vals)
    
    # Golden section search

            d_val = g(d_pt)
    
    best_alpha = (lo + hi) / 2.0

    best_alpha = max([0.0, best_alpha, alpha_max], key=lambda x: -g(x))

import numpy as np
import time
from scipy.optimize import linprog


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_D(h):
    n = len(h)
    v = 1.0 - h
    D = np.zeros((2 * n - 1, n))
    for k in range(n):
        e_k = np.zeros(n)
        e_k[k] = 1.0
        D[:, k] = np.correlate(e_k, v, mode="full") - np.correlate(h, e_k, mode="full")
    return D


def solve_lp_direction(D, corr, trust):
    n = D.shape[1]
    m = D.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, -1] = -1.0
    b_ub = -corr
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-trust, trust)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        d = res.x[:n]
        t = res.x[-1]
        if np.max(np.abs(d)) > 1e-10:
            return d, t
    return None, None


def line_search_exact(h, d, corr):
    n = len(h)
    v = 1.0 - h
    a = np.correlate(d, v, mode="full") - np.correlate(h, d, mode="full")
    b = np.correlate(d, d, mode="full")

        return 0.0
    
    alpha_max = np.inf
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))
    if not np.isfinite(alpha_max) or alpha_max <= 0:
        return 0.0
    alpha_max = min(alpha_max, 1.0)
    
    def g(alpha):
        return float(np.max(corr + alpha * a - alpha * alpha * b))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_a, hi_a = 0.0, alpha_max
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = g(c1)
    f2 = g(c2)
    for _ in range(50):
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = g(c1)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = g(c2)
    
    best_alpha = (lo_a + hi_a) / 2.0
    # Check endpoints
    f0 = g(0.0)
    ff = g(best_alpha)
    fm = g(alpha_max)
    if f0 <= ff and f0 <= fm:
        return 0.0
    if fm <= ff and fm <= f0:
        return alpha_max
    return best_alpha


def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi

        m2 = b - (b - a) / 3.0

        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def full_pair_sweep(h, rng, time_limit=None, t0=None):
    n = len(h)
    perm = rng.permutation(n)
    count = 0
    for a_idx in range(n):

            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
            count += 1
            if time_limit is not None and t0 is not None:
                if time.time() - t0 > time_limit:
                    return

            h = project_onto_constraint(h, target)
