```python
def make_feasible(z):
    N = len(z)
    target = N / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    """Compute gradient of lag value c_s / dx w.r.t. h, then multiply by dx."""
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    if s >= 0:
        # c_s = sum_{n=0}^{N-1-s} h[n+s] * (1-h[n])

            if i >= s:
                g += (1.0 - h[i - s])
            if i + s < N:
                g += (-h[i + s])
            grad[i] = dx * g

            if i >= t:
                g += (-h[i - t])
            grad[i] = dx * g
    return grad
```

```python
def lp_minimax_step(h, V, active_p, rho, C):
    N = len(h)
    K = len(active_p)
    # Variables: u[0..N-1], t
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    c = np.zeros(N + 1)
    c[N] = 1.0
    
    # Inequality constraints: g_p^T u - t <= -V[p] for each active p
    # A_ub shape: K x (N+1)
    A_ub = np.zeros((K, N + 1))
    b_ub = np.zeros(K)
    for i, p in enumerate(active_p):
        s = p - (N - 1)
        g = lag_gradient(h, s)
        A_ub[i, :N] = g
        A_ub[i, N] = -1.0
        b_ub[i] = -V[p]
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(N):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C))  # t <= C
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:N]
    return None
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    N = len(z)
    target = N / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s):
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    if s >= 0:
        m = N - s
        grad[s:] += (1.0 - h[:m])
        grad[:m] += -h[s:]
    else:
        t = -s
        m = N - t
        grad[:m] += (1.0 - h[t:])
        grad[t:] += -h[:m]
    grad *= dx
    return grad

def all_lag_values(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def select_active(V, K_top=24, tol=0.004):
    C = V.max()
    mask = V >= C - tol
    if np.sum(mask) < 4:
        mask = np.zeros(len(V), dtype=bool)
        idx = np.argsort(V)[-max(4, K_top):]
        mask[idx] = True
    top_idx = np.argsort(V)[-K_top:]
    mask[top_idx] = True
    return np.where(mask)[0]

def lp_minimax_direction(h, V, active_p, rho, C):
    N = len(h)
    K = len(active_p)
    if K == 0:
        return None
    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0
    A_ub = np.zeros((K, N + 1))
    b_ub = np.zeros(K)
    for i, p in enumerate(active_p):
        s = p - (N - 1)
        g = lag_gradient(h, s)
        A_ub[i, :N] = g
        A_ub[i, N] = -1.0
        b_ub[i] = -V[p]
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(N):
        lo_b = max(-h[i], -rho)
        hi_b = min(1.0 - h[i], rho)
        if lo_b > hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, C + 1e-12))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            direction = res.x[:N]
            t_pred = res.x[N]
            if t_pred < C - 1e-10 and np.max(np.abs(direction)) > 1e-12:
                return direction
            elif np.max(np.abs(direction)) > 1e-10:
                return direction
    except Exception:
        pass
    return None

def two_point_search(h, n_tries, rng, is_dead):
    N = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = int(rng.integers(0, N))
        b = int(rng.integers(0, N))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates = np.array([0.0, t_lo, t_hi, 0.5 * span,
                               0.25 * span, 0.75 * span,
                               span / 2.618, 2.0 * span / 2.618]) + t_lo
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-12 or va > 1.0 + 1e-12 or vb < -1e-12 or vb > 1.0 + 1e-12:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] += best_t
            best[b] -= best_t
            np.clip(best, 0.0, 1.0, out=best)
            best_sc = best_local
    return best, best_sc

def subgradient_polish(h, n_iter, is_dead):
    N = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    V = all_lag_values(best)
    alpha = 0.015
    beta = 60.0
    for it in range(n_iter):
        if is_dead():
            break
        C = V.max()
        shifted = beta * (V - C)
        w = np.exp(shifted)
        w /= w.sum()
        active_idx = np.where(w > 1.0 / (N * 2))[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(V)])
        w_a = w[active_idx]
        w_a /= w_a.sum()
        grad = np.zeros(N)
        for xp in active_idx:
            s = xp - (N - 1)
            g = lag_gradient(best, s)
            grad += w_a[list(active_idx).index(xp)] * g
        d = -grad
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d /= d_norm
        improved = False

                best_sc = sc

                improved = True
                alpha *= 1.1
                break
        if not improved:
            alpha *= 0.65
            if alpha < 1e-7:

    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_tgt == n_src:
        return make_feasible(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.09 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.1)
        best_score = score_exact(best_h)

    n = len(best_h)

    # Phase 1: Quick LP warm-up from parent
    if not is_dead():
        h_new, sc = lp_refine(best_h, 80, is_dead, K_top=16, tol=0.005, rho_start=0.12)
        if sc < best_score - 1e-12:
            best_h = h_new
            best_score = sc

    # Phase 2: Generate diverse starting points
    starts = [best_h.copy()]
    h_anti = anti_sym_map(best_h)
    sc_anti = score_exact(h_anti)
    if sc_anti < best_score - 1e-12:
        best_h = h_anti
        best_score = sc_anti
    starts.append(h_anti)
    for freq1 in [1, 3, 5]:
        for amp in [0.2, 0.35]:
            h_w = make_wave(n, freq1, amp, freq1 + 2, 0.08)
            sc_w = score_exact(h_w)
            if sc_w < best_score + 0.05:
                starts.append(h_w)

    # Phase 3: Binary exploration
    if not is_dead():
        h_bin, sc_bin = binary_sa(best_h, 8000, 8e-4, rng, is_dead)
        if h_bin is not None and sc_bin < best_score + 0.1:
            h_bin_cont = make_feasible(h_bin)
            sc_bin_cont = score_exact(h_bin_cont)
            starts.append(h_bin_cont)
            if sc_bin_cont < best_score - 1e-12:
                best_h = h_bin_cont
                best_score = sc_bin_cont
            h_bin_anti = anti_sym_map(h_bin_cont)
            sc_ba = score_exact(h_bin_anti)
            starts.append(h_bin_anti)
            if sc_ba < best_score - 1e-12:
                best_h = h_bin_anti
                best_score = sc_ba

    # Phase 4: LP refine best start
    if not is_dead():
        h_new, sc = lp_refine(best_h, 150, is_dead, K_top=20, tol=0.004, rho_start=0.15)
        if sc < best_score - 1e-12:
            best_h = h_new
            best_score = sc

    # Phase 5: Structured 2-point on best
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 2.5))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 6: Subgradient polish
    if not is_dead():
        best_h, best_score = subgradient_polish(best_h, 150, is_dead)

    # Phase 7: LP refine again (tighter)
    if not is_dead():
        h_new, sc = lp_refine(best_h, 200, is_dead, K_top=24, tol=0.002, rho_start=0.08)
        if sc < best_score - 1e-12:
            best_h = h_new
            best_score = sc

    # Phase 8: More 2-point
    if not is_dead():
        n_tries = max(3000, int((deadline - time.time()) * 3))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 9: Try other diverse starts with LP
    if not is_dead():
        for st in starts[1:4]:
            if is_dead():
                break
            if len(st) != len(best_h):
                st = interp_to(st, len(best_h))
            remaining = deadline - time.time()
            if remaining < 10:
                break
            h_new, sc = lp_refine(st, 60, is_dead, K_top=20, tol=0.004, rho_start=0.12)
            if sc < best_score - 1e-12:
                best_h = h_new
                best_score = sc
                h_new2, sc2 = two_point_search(h_new, 1000, rng, is_dead)
                if sc2 < best_score - 1e-12:
                    best_h = h_new2
                    best_score = sc2

    # Phase 10: Resolution scanning
    if not is_dead():
        n_cur = len(best_h)
        test_ns = sorted(set([n_cur - 8, n_cur - 4, n_cur + 4, n_cur + 8,
                              44, 52, 56, 64, 72]))
        for nt in test_ns:
            if is_dead():
                break
            if nt < 20 or nt > 100 or nt == n_cur:
                continue
            h_int = interp_to(best_h, nt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int
                best_score = sc_int
                h_new, sc_new = lp_refine(h_int, 40, is_dead, K_top=16, tol=0.004, rho_start=0.1)
                if sc_new < best_score - 1e-12:
                    best_h = h_new
                    best_score = sc_new

    # Phase 11: Final intensive LP
    if not is_dead():
        h_new, sc = lp_refine(best_h, 300, is_dead, K_top=28, tol=0.0015, rho_start=0.06)
        if sc < best_score - 1e-12:
            best_h = h_new
            best_score = sc

    # Phase 12: Final 2-point polish
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 3))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 13: One more subgradient
    if not is_dead():
        best_h, best_score = subgradient_polish(best_h, 100, is_dead)

    # Phase 14: Final LP with very tight params
    if not is_dead():
        h_new, sc = lp_refine(best_h, 200, is_dead, K_top=32, tol=0.001, rho_start=0.04)
        if sc < best_score - 1e-12:
            best_h = h_new
            best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)) or final_h.min() < -1e-9 or final_h.max() > 1.0 + 1e-9:
        final_h = make_feasible(np.full(60, 0.5))
        final_score = score_exact(final_h)
        final_n = 60

    return final_h, final_score, final_n
```
