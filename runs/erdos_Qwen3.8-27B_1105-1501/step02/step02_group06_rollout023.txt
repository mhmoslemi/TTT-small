def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:

                if add <= 0: break

            top_k = np.argsort(raw_vals)[-k_active:]
            active_mask[top_k] = True
            active_idx = np.where(active_mask)[0]
        reg = reg_init * (0.7 ** it)
        h_new = solve_qp(h, raw_vals, active_idx, radius, reg)
        if h_new is None:
            radius *= 0.6
            if radius < 0.005:
                stall += 2
            else:
                stall += 1
            if stall > 8:
                break
            continue
        h_new = make_feasible(h_new)
        best_alpha = 0
        best_local = best_score
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            h_try = make_feasible(h + alpha * (h_new - h))
            sc_try = official_score(h_try)
            if sc_try < best_local - 1e-15:
                best_local = sc_try
                best_alpha = alpha
        if best_alpha > 0:
            h = make_feasible(h + best_alpha * (h_new - h))
            if best_local < best_score - 1e-15:
                best_score = best_local
                stall = 0
                radius = min(radius * 1.15, 0.8)
            else:
                stall += 1
        else:
            stall += 1
            radius *= 0.6
        if stall > 12:
            break
    return h, best_score

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = raw.max()
    shifted = np.clip(beta * (raw - max_val), -50, 50)
    return max_val + np.log(np.sum(np.exp(shifted))) / beta

def eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def eq_jac(h):
    return np.ones(len(h))

def try_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_max_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, float('inf')

def subgrad_dir(h):
    n = len(h)
    raw_vals = compute_all_raw(h)
    raw_max = raw_vals.max()
    beta = 300.0
    w = np.exp(beta * (raw_vals - raw_max))
    w /= w.sum()
    d = np.zeros(n)
    for p in range(len(raw_vals)):
        if w[p] > 1e-8:
            d += w[p] * gradient_raw(h, p - (n-1))
    d -= np.mean(d)
    d = -d
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    return d / norm

def line_search_dir(h, d, radius=0.3):
    n = len(h)
    best_sc = official_score(h)
    best_alpha = 0
    alpha_max = radius
    for i in range(n):
        if d[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i] * 0.99)
        elif d[i] < -1e-14:
            alpha_max = min(alpha_max, -h[i] / d[i] * 0.99)
    if alpha_max <= 0:
        return None, best_sc
    for alpha in np.linspace(0.05 * alpha_max, alpha_max, 15):
        h_try = make_feasible(h + alpha * d)
        sc = official_score(h_try)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_alpha = alpha
    if best_alpha == 0:
        return None, best_sc
    return make_feasible(h + best_alpha * d), best_sc
