Actually, let me think more carefully. The correlation `np.correlate(h, 1-h, mode='full')` gives:
- For lag k = 0: sum_i h[i]*(1-h[i])
- For positive lag s (correlation index n-1+s): sum_{i=s}^{n-1} h[i]*(1-h[i-s])
- For negative lag -s (correlation index n-1-s): sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])

When we change h_i → h_i - t and h_j → h_j + t:
- Each term involving position i changes by -t*(1-h[...]) or -t*h[...] or similar
- Each term involving position j changes by +t*(1-h[...]) or +t*h[...]

So the change in each correlation value is linear in t individually, but since h appears in both factors of the product h[i]*(1-h[j]), the total change is actually quadratic in t.

More precisely, for a correlation term h[a]*(1-h[b]):
- If a=i and b=j: (h_i-t)*(1-h_j-t) = h_i*(1-h_j) - t*(1-h_j) - t*h_i + t²
- If a=i, b≠j: (h_i-t)*(1-h_b) = h_i*(1-h_b) - t*(1-h_b)
- If a=j, b=i: (h_j+t)*(1-h_i+t) = h_j*(1-h_i) + t*(1-h_i) + t*h_j - t²
- If a=j, b≠i: (h_j+t)*(1-h_b) = h_j*(1-h_b) + t*(1-h_b)
- If a≠i,j, b=i: h_a*(1-h_i+t) = h_a*(1-h_i) + t*h_a
- If a≠i,j, b=j: h_a*(1-h_j-t) = h_a*(1-h_j) - t*h_a
- If a≠i,j, b≠i,j: unchanged

So each correlation value c_k changes as: c_k + (A_k)*t + (B_k)*t²

where A_k and B_k depend on the specific positions i, j and the lag k.

For B_k: it's non-zero only when both i and j appear in the same product term of c_k. This happens when |i-j| = k (or |i-j| = 0 and i=j which is impossible). So B_k is non-zero for at most 2 lags (k = j-i and k = i-j).

This is great! For most lags, the change is linear in t. For at most 2 lags, it's quadratic.

So the approach is:
1. For a given pair (i,j), compute the current correlation vector c
2. Compute the linear and quadratic coefficients A_k and B_k for each lag k
3. The new max is max_k (c_k + A_k*t + B_k*t²)
4. Find t that minimizes this, subject to 0 ≤ h_i - t and 0 ≤ h_j + t ≤ 1

This is a min-max of piecewise quadratic functions, which can be solved by:
- Finding the "critical" t values where the active lag changes
- Checking the minimum of each active quadratic over its valid t range

Actually, for a simpler but still effective approach: sample t densely on the valid range and pick the best. With 100+ samples, this should be very accurate.

                    continue

                    continue

                best_local_max = np.inf

                    h[j] += t
                    c = np.max(np.correlate(h, 1.0 - h, mode='full'))

                    if c < best_local_max:

            best_score = official_score(h)
            improved = True
        
        if not improved:
            break
    
    return h, best_score

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n - s2])
                grad[:n - s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n - m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n - m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            lse_obj, h0, args=(beta, dx),
            jac=lambda x: lse_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def pair_move_optimize(h, max_passes=200, n_t=80, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    dx = 2.0 / n

    for passage in range(max_passes):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break

        corr0 = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        best_impr = 0.0
        best_i = best_j = -1
        best_t = 0.0
        best_new_max = np.inf

        # Compute gradient for pair selection
        max_c = np.max(corr0)
        thresh = max_c * 0.97
        active = np.where(corr0 >= thresh)[0]
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                if s == 0:
                    grad += (1.0 - h) - h
                else:
                    grad[s:] += (1.0 - h[:n - s])
                    grad[:n - s] += (-h[s:])
            else:
                m = -s
                if m == 0:
                    grad += (1.0 - h) - h
                else:
                    grad[m:] += (1.0 - h[:n - m])
                    grad[:n - m] += (-h[m:])
        grad /= len(active)

        # Select candidate pairs
        med = np.median(grad)
        donors = np.where((h > 0.02) & (grad >= med))[0]
        receivers = np.where((h < 0.98) & (grad < med))[0]
        if len(donors) == 0:
            donors = np.argsort(h)[-12:]
        if len(receivers) == 0:
            receivers = np.argsort(h)[:12]
        donors = donors[:min(12, len(donors))]
        receivers = receivers[:min(12, len(receivers))]

        # Also add some random pairs for diversity
        for _ in range(6):
            ri = np.random.randint(n)
            rj = np.random.randint(n)
            if ri != rj and h[ri] > 0.02 and h[rj] < 0.98:
                if ri not in donors:
                    donors = np.append(donors, ri)
                if rj not in receivers:
                    receivers = np.append(receivers, rj)

        for i in donors:
            if i < 0 or i >= n:
                continue
            for j in receivers:
                if j < 0 or j >= n or i == j:
                    continue
                t_hi = min(h[i], 1.0 - h[j])
                if t_hi < 0.005:
                    continue

                # Compute sparse correlations with d = e_j - e_i
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                Ad = np.correlate(h, d, mode='full')
                Bd = np.correlate(d, 1.0 - h, mode='full')
                Cd = np.correlate(d, d, mode='full')

                # new_corr(t) = corr0 - t*Ad - t*Bd + t^2*Cd
                t_vals = np.linspace(0.0, t_hi, n_t)
                t_col = t_vals[:, np.newaxis]
                nc = corr0[np.newaxis, :] - t_col * Ad[np.newaxis, :] - t_col * Bd[np.newaxis, :] + t_col**2 * Cd[np.newaxis, :]
                maxes = np.max(nc, axis=1)
                bi = np.argmin(maxes)
                if maxes[bi] < best_new_max:
                    best_new_max = maxes[bi]
                    best_i, best_j, best_t = int(i), int(j), float(t_vals[bi])

        if best_i >= 0 and best_new_max * dx < best_score - 1e-12:
            h[best_i] -= best_t
            h[best_j] += best_t
            h = make_feasible(h)
            new_s = official_score(h)
            if new_s < best_score:
                best_score = new_s
                best_impr = best_score - new_s
                improved = True

        if not improved:
            break

    return h, best_score

def sa_optimize(h0, rng, n_iters=20000, time_limit=None, t_start=None):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    T = 0.015
    T_end = 0.0005
    cool = (T_end / T) ** (1.0 / max(n_iters, 1))

    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if h[i] >= h[j]:
            d, r = i, j
        else:
            d, r = j, i
        delta = min(h[d], 1.0 - h[r])
        if delta < 0.01:
            continue
        swap = delta * rng.uniform(0.3, 1.0)
        h[d] -= swap
        h[r] += swap
        s = official_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
        else:
            if rng.random() <= np.exp(-(s - best_score) / T):
                pass
            else:
                h[d] += swap
                h[r] -= swap
        T *= cool

    return best_h, best_score

def periodic_pattern(n, phase, k=3):
    x = np.arange(n)
    h = np.zeros(n)
    for m in range(1, k + 1):
        h += np.sin(2.0 * np.pi * m * (x + phase) / n)
    h = h - np.min(h)
    hmax = np.max(h)
    if hmax > 0:
        h /= hmax
    return make_feasible(h)

def build_starts(parent, n, rng):
    starts = []
    hi = make_feasible(interp_to_n(parent, n))
    starts.append(hi)
    starts.append(make_feasible(np.clip(hi, 0, 1) ** 1.5))
    starts.append(make_feasible(np.clip(hi, 0, 1) ** 3.0))
    starts.append(make_feasible(np.clip(hi, 0, 1) ** 5.0))

    rb = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    rb[idx] = 1.0
    starts.append(make_feasible(rb))

    for ph in [0, 1, 2, 3]:
        starts.append(periodic_pattern(n, ph))

    for sig in [0.1, 0.2]:
        starts.append(make_feasible(hi + sig * rng.standard_normal(n)))

    starts.append(make_feasible(1.0 - hi[::-1].copy()))

    # Deduplicate by score (keep best per approximate score)
    unique = []
    seen_scores = []
    for s in starts:
        sc = official_score(s)
        dup = False
        for ss in seen_scores:
            if abs(sc - ss) < 1e-6:
                dup = True
                break
        if not dup:
            unique.append((sc, s))
            seen_scores.append(sc)
    unique.sort(key=lambda x: x[0])
    return [s for _, s in unique[:8]]

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    # Determine n values to try
    pn = len(parent)
    n_opts = sorted(set([36, 38, 40, 42, 44, 46, 48, 50, 52, 54, 56, 58, 60]))
    if pn in n_opts:
        n_opts.remove(pn)
    n_opts.insert(0, pn)
    n_opts = [n for n in n_opts if 28 <= n <= 70]

    # Process n values
    for n in n_opts:
        if time.time() > deadline - 40:
            break

        starts = build_starts(parent, n, rng)
        starts.sort(key=lambda x: official_score(x))

        for h0 in starts[:4]:
            if time.time() > deadline - 20:
                break

            cur = h0.copy()
            cur_score = official_score(cur)

            # Phase 1: SA
            if time.time() < deadline - 30:
                nh, ns = sa_optimize(cur, rng, n_iters=15000, time_limit=40, t_start=time.time())
                if ns < cur_score - 1e-10:
                    cur, cur_score = nh, ns

            # Phase 2: Pair moves (main optimization)
            if time.time() < deadline - 20:
                tl = max(10.0, min(50.0, (deadline - time.time()) * 0.3))
                cur, cur_score = pair_move_optimize(cur, max_passes=200, n_t=80, time_limit=tl, t_start=time.time())

            # Phase 3: SLSQP polish
            if time.time() < deadline - 15:
                for beta in [80, 300, 800]:
                    if time.time() > deadline - 10:
                        break
                    nh = slsqp_smooth(cur, beta, maxiter=80)
                    ns = official_score(nh)
                    if ns < cur_score - 1e-12:
                        cur, cur_score = nh, ns

            # Phase 4: More pair moves
            if time.time() < deadline - 15:
                tl = max(5.0, min(30.0, (deadline - time.time()) * 0.3))
                cur, cur_score = pair_move_optimize(cur, max_passes=150, n_t=80, time_limit=tl, t_start=time.time())

            # Phase 5: Final SLSQP
            if time.time() < deadline - 10:
                nh = slsqp_smooth(cur, 500, maxiter=80)
                ns = official_score(nh)
                if ns < cur_score - 1e-12:
                    cur, cur_score = nh, ns

            # Phase 6: One more SA + pair
            if time.time() < deadline - 15:
                nh, ns = sa_optimize(cur, rng, n_iters=8000, time_limit=20, t_start=time.time())
                if ns < cur_score - 1e-10:
                    cur, cur_score = nh, ns
                    tl = max(5.0, min(20.0, (deadline - time.time()) * 0.3))
                    cur, cur_score = pair_move_optimize(cur, max_passes=100, n_t=80, time_limit=tl, t_start=time.time())
                    if time.time() < deadline - 5:
                        nh2 = slsqp_smooth(cur, 400, maxiter=60)
                        ns2 = official_score(nh2)
                        if ns2 < cur_score - 1e-12:
                            cur, cur_score = nh2, ns2

            # Update global best
            if cur_score < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_score
                best_n = len(cur)
            elif abs(cur_score - best_score) < 1e-12 and len(cur) < best_n:
                best_h = cur.copy()
                best_score = cur_score
                best_n = len(cur)

    # Final refinement on best solution
    remaining = deadline - time.time()
    if remaining > 15:
        # Try perturbations of best found
        for trial in range(min(8, int(remaining / 10))):
            if time.time() > deadline - 5:
                break
            pert = make_feasible(best_h + 0.05 * rng.standard_normal(best_n))
            pert_score = official_score(pert)
            if pert_score > best_score + 0.01:
                continue
            tl = max(3.0, min(15.0, (deadline - time.time()) * 0.4))
            nh, ns = pair_move_optimize(pert, max_passes=80, n_t=64, time_limit=tl, t_start=time.time())
            if time.time() < deadline - 5:
                nh2 = slsqp_smooth(nh, 300, maxiter=60)
                ns2 = official_score(nh2)
                if ns2 < ns:
                    nh, ns = nh2, ns2
            if ns < best_score - 1e-12:
                best_h = nh.copy()
                best_score = ns

    # Final polish
    if time.time() < deadline - 5:
        final_h = slsqp_smooth(best_h, 600, maxiter=100)
        final_s = official_score(final_h)
        if final_s < best_score:
            best_h = final_h
            best_score = final_s
        tl = max(3.0, min(10.0, (deadline - time.time()) * 0.5))
        final_h2, final_s2 = pair_move_optimize(best_h, max_passes=50, n_t=64, time_limit=tl, t_start=time.time())
        if final_s2 < best_score:
            best_h = final_h2
            best_score = final_s2

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
