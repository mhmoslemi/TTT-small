import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    n = len(h)

    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

    n = len(h)

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2*n-1):
        w_val = weights[k] * dx
        if w_val < 1e-14:
            continue
        s2 = k - (n-1)
        if s2 >= 0:
            if s2 == 0:
                grad[:] += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n-s2])
                grad[:n-s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad[:] += (1.0 - x) - x
            else:
                grad[:n-m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n-m2])
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

            jac=lambda x: smooth_lse_grad(x, beta, dx),

            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],

import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:k]] = 1.0
    return h_bin

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def block_steepest(h, w, time_limit, t_start):
    n = len(h)
    if n % 2 != 0:
        return h, exact_score(h)
    h = h.copy()
    best_score = exact_score(h)
    for _ in range(300):
        if time.time() - t_start > time_limit:
            break
        best_improve = 0
        best_new_h = None
        for start in range(0, n - w + 1):
            k_block = int(np.sum(h[start:start+w]))
            if k_block == 0 or k_block == w:
                continue
            current_combo = tuple(np.where(h[start:start+w] == 1)[0])
            for combo in combinations(range(w), k_block):
                if combo == current_combo:
                    continue
                new_block = np.zeros(w)
                for pos in combo:
                    new_block[pos] = 1.0
                h_new = h.copy()
                h_new[start:start+w] = new_block
                score = exact_score(h_new)
                improve = best_score - score
                if improve > best_improve:
                    best_improve = improve
                    best_new_h = h_new
        if best_improve < 1e-14:
            break
        h = best_new_h
        best_score = exact_score(h)
    return h, best_score

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2*n-1):
        w_val = weights[k] * dx
        if w_val < 1e-14:
            continue
        s2 = k - (n-1)
        if s2 >= 0:
            if s2 == 0:
                grad[:] += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n-s2])
                grad[:n-s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad[:] += (1.0 - x) - x
            else:
                grad[:n-m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n-m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0.copy(), args=(beta, dx),
            jac=lambda x: smooth_lse_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0
