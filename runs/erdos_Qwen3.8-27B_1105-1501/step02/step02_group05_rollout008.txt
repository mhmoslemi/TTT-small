import time

from scipy.optimize import minimize

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return vals

def compute_lag_gradient(h, lag_idx):
    """Compute gradient of lag value at index lag_idx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            grad[i] += (1.0 - h[i + s])
            grad[i + s] -= h[i]
    else:
        t = -s
        for i in range(n - t):
            grad[i + t] += (1.0 - h[i])
            grad[i] -= h[i + t]
    return dx * grad

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    npts = 2 * n - 1
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(npts) / npts
    else:
        w /= ws
    grad = np.zeros(n)
    for p in range(npts):
        s = p - (n - 1)
        if s >= 0:
            for j in range(n - s):
                grad[j] += w[p] * (1.0 - x[j + s])
                grad[j + s] -= w[p] * x[j]
        else:
            t = -s
            for j in range(n - t):
                grad[j + t] += w[p] * (1.0 - x[j])
                grad[j] -= w[p] * x[j + t]
    return dx * grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def try_slsqp(h0, beta, maxiter=80):

    bounds = [(0.0, 1.0)] * n

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    n = len(h0)
    vals = compute_lag_values(h0)
    cmax = vals.max()

    n_lags = len(vals)

        G[i] = compute_lag_gradient(h0, p)

    # subject to: sum(d) = 0
    #             -h <= d <= 1-h
    #             G @ d <= t - vals[active]
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    obj = cp.Minimize(t + mu * cp.sum_squares(d))
    rhs = cmax - vals[active]  # vals[active] <= cmax, so rhs >= 0

        cp.sum(d) == 0,
        d >= -h0,
        d <= 1.0 - h0,
        G @ d <= t + rhs  # G@d <= t - vals[active] + cmax = t + (cmax - vals[active])
    ]
    
    prob = cp.Problem(obj, constraints)
    
    solvers = ['ECOS', 'CLARABEL', 'SCS', 'OSQP']

        try:
            prob.solve(solver=solver, verbose=False)
            break
        except Exception:
            continue
    else:
        return None, None
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None

def gradient_informed_pairs(h, n_pairs=200):
    """Find promising pairs based on gradient structure."""
    n = len(h)
    vals = compute_lag_values(h)
    cmax = vals.max()
    tol = 0.03
    active = np.where(vals >= cmax - tol)[0]
    
    # Compute combined gradient over active lags
    weights = np.exp(30.0 * (vals[active] - cmax))
    weights /= weights.sum()
    
    grad = np.zeros(n)

        g = compute_lag_gradient(h, p)
        grad += weights[i] * g
    
    # Pairs where one has positive grad and other negative
    pos = np.where(grad > 0.01)[0]
    neg = np.where(grad < -0.01)[0]

            a = pos[np.random.randint(len(pos))]
            b = neg[np.random.randint(len(neg))]

def anti_sym_map(h):

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):

    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, lag_idx):
    n = len(h)
    dx = 2.0 / n
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            grad[i] += (1.0 - h[i + s])
            grad[i + s] -= h[i]
    else:
        t = -s
        for i in range(n - t):
            grad[i + t] += (1.0 - h[i])
            grad[i] -= h[i + t]
    return dx * grad

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    npts = 2 * n - 1
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(npts) / npts

                grad[j + s] -= w[p] * x[j]

            t = -s
            for j in range(n - t):
                grad[j + t] += w[p] * (1.0 - x[j])
                grad[j] -= w[p] * x[j + t]
    return dx * grad
