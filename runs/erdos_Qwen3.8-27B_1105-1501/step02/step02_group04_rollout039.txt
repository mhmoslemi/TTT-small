    offset = n - 1
    max_corr = np.max(corr)
    K = len(active_lags)
    A = build_active_gradient(h, n, active_lags)
    
    # slack[k] = max_corr - corr[active_lags[k]]
    slack = max_corr - corr[active_lags]
    slack = np.maximum(slack, 0.0)
    
    # LP: maximize theta
    # s.t. A @ d + theta <= slack  (for each active lag)
    #      sum(d) = 0
    #      -trust_radius <= d_i <= min(trust_radius, 1-h_i) for positive moves
    #      0-h_i <= d_i (so h_i + d_i >= 0)
    #      theta >= 0
    
    # Variables: d_0, ..., d_{n-1}, theta (total n+1 variables)
    # Objective: minimize -theta (to maximize theta)
    c = np.zeros(n + 1)
    c[n] = -1.0  # minimize -theta = maximize theta
    
    # Inequality constraints: A @ d + theta <= slack
    # In linprog form: A_ub @ x <= b_ub
    A_ub = np.zeros((K, n + 1))

    A_ub[:, n] = 1.0  # coefficient of theta
    b_ub = slack

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # For theta: [0, inf)
    lbs = np.maximum(-h, -trust_radius)
    ubs = np.minimum(1.0 - h, trust_radius)
    bounds = list(zip(lbs, ubs)) + [(0.0, None)]
    
    try:

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.success and res.fun < 0:

    return None, 0.0

        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def build_active_gradient(h, n, active_lags):
    K = len(active_lags)
    A = np.zeros((K, n))
    offset = n - 1
    for r, ell in enumerate(active_lags):
        s = ell - offset
        # First term: (1-h[i-s]) where 0 <= i-s < n
        i_lo1 = max(0, s)
        i_hi1 = min(n - 1, n - 1 + s)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            A[r, idx] += (1.0 - h[idx - s])
        # Second term: -h[i+s] where 0 <= i+s < n
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            A[r, idx] -= h[idx + s]
    return A

def solve_lp_direction(h, n, active_lags, corr, trust_radius):
    max_corr = np.max(corr)
    K = len(active_lags)
    A = build_active_gradient(h, n, active_lags)
    slack = np.maximum(max_corr - corr[active_lags], 0.0)
    
    # Variables: d_0,...,d_{n-1}, theta
    c_obj = np.zeros(n + 1)
    c_obj[n] = -1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = 1.0
    b_ub = slack
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    lbs = np.maximum(-h, -trust_radius)
    ubs = np.minimum(1.0 - h, trust_radius)
    bounds = list(zip(lbs, ubs)) + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and -res.fun > 1e-8:
            return res.x[:n], -res.fun
    except Exception:
        pass
    return None, 0.0

def do_line_search(h, n, direction, current_max):
    best_val = current_max
    best_alpha = 0.0
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
        trial = h + alpha * direction
        if np.any(trial < -1e-9) or np.any(trial > 1.0 + 1e-9):
            continue
        trial = np.clip(trial, 0.0, 1.0)
        # Sum is preserved since sum(direction)=0
        # But numerical error might creep in, so re-project
        if abs(np.sum(trial) - n/2.0) > 1e-6:
            trial = project_onto_constraint(trial, n / 2.0)
        corr = np.correlate(trial, 1.0 - trial, mode="full")
        val = np.max(corr)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_val, best_alpha

def limited_pair_sweep(h, rng, n, max_pairs, time_limit):
    perm = rng.permutation(n)
    count = 0
    for a in range(n):
        for b in range(a + 1, n):
            if count >= max_pairs or time.time() >= time_limit:
                return h
            i = perm[a]
            j = perm[b]
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
                count += 1
    return h

def random_kick(h, rng, n, kick_size, n_pairs, time_limit):
    h_try = h + rng.randn(n) * kick_size
    h_try = project_onto_constraint(h_try, n / 2.0)

            h_try[i] -= d
            h_try[j] += d
    return h_try

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pair_optimal_delta(h, i, j):
    hi_v, hj_v = h[i], h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0

import numpy as np
import time
from scipy.optimize import linprog


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def pair_optimal_delta(h, i, j):
    hi_v, hj_v = h[i], h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(24):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i], h[j] = hi_v - m1, hj_v + m1
        f1 = compute_c5(h)
        h[i], h[j] = hi_v - m2, hj_v + m2
        f2 = compute_c5(h)
        h[i], h[j] = hi_v, hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def build_full_gradient(h, n):
    m = 2 * n - 1
    A = np.zeros((m, n))
    offset = n - 1
    for ell in range(m):
        s = ell - offset
        i_lo1 = max(0, s)
        i_hi1 = min(n - 1, n - 1 + s)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            A[ell, idx] += (1.0 - h[idx - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            A[ell, idx] -= h[idx + s]
    return A


def solve_lp_direction(h, n, A, corr, trust_radius):
    max_corr = np.max(corr)
    slack = np.maximum(max_corr - corr, 0.0)
    m = len(corr)
    c_obj = np.zeros(n + 1)
    c_obj[n] = -1.0
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = 1.0
    b_ub = slack
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    lbs = np.maximum(-h, -trust_radius)
    ubs = np.minimum(1.0 - h, trust_radius)
    bounds = list(zip(lbs, ubs)) + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res is not None and res.success and -res.fun > 1e-9:
            return res.x[:n].copy(), -res.fun
    except Exception:
        pass
    return None, 0.0


def do_line_search(h, n, direction, current_max):
    best_val = current_max
    best_alpha = 0.0
    for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
        trial = h + alpha * direction
        if np.any(trial < -1e-8) or np.any(trial > 1.0 + 1e-8):
            continue
        val = np.max(np.correlate(trial, 1.0 - trial, mode="full"))
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_val, best_alpha


def random_pair_polish(h, rng, n, n_pairs, time_limit):
    for _ in range(n_pairs):
        if time.time() >= time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = pair_optimal_delta(h, i, j)
        if abs(d) > 1e-12:
            h[i] -= d
            h[j] += d
    return h


def full_ordered_pair_sweep(h, rng, n, time_limit, max_pairs=None):
    perm = rng.permutation(n)
    count = 0
    for a in range(n):
        for b in range(a + 1, n):
            if max_pairs is not None and count >= max_pairs:
                return h
            if time.time() >= time_limit:
                return h
            i, j = perm[a], perm[b]
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
                count += 1
    return h


def softmax_subgradient_direction(h, n, A, corr, temp=0.08):
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / temp)
    w /= np.sum(w)
    grad = A.T @ w
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return None
    return -grad / norm
