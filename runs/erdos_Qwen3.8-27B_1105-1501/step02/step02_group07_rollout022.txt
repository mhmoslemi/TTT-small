    deadline = time.time() + time_limit
    V = compute_V(h)
    obj_cur = lse_val(V, beta)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        w = lse_weights(V, beta)
        g = grad_V(h, w)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn  # normalize
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(25):
            h_new = make_feasible(h - alpha * g)
            V_new = compute_V(h_new)
            obj_new = lse_val(V_new, beta)
            if obj_new < obj_cur - 1e-14:
                h = h_new
                V = V_new
                obj_cur = obj_new
                improved = True
                sc = official_score(h)
                if sc < best_score - 1e-13:
                    best_h = h.copy()
                    best_score = sc
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        
        if not improved:
            lr *= 0.5
        else:
            lr = min(lr * 1.1, 2.0)
    
    return best_h, best_score

def active_set_lp(h, eps=0.003, trust=0.15):
    n = len(h)
    dx = 2.0 / n
    V = compute_V(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[:L]
            gp[:L] -= h[s:n]
        else:
            t = -s
            gp[t:n] = 1.0 - h[:L]
            gp[:L] -= h[t:n]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                for frac in [1.0, 0.75, 0.5, 0.25]:
                    h_new = make_feasible(h + frac * d)
                    sc = official_score(h_new)
                    if sc < V_max - 1e-12:
                        return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def random_binary(n, rng):
    """Create a random binary vector with exactly n/2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h
