```python
    dx = 2.0 / n

    lo, hi = -2.0, 2.0

        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((n, m))
    offset = n - 1
    v = 1.0 - h
    for i in range(n):
        # Part 1: (1-h[i+s]) where s = p-offset, valid when 0<=i+s<=n-1
        # p in [offset-i, offset-i+n-1]
        p_lo = max(0, offset - i)
        p_hi = min(m - 1, offset - i + n - 1)
        if p_lo <= p_hi:
            idx = np.arange(p_lo, p_hi + 1)
            j_idx = i + idx - offset
            G[i, p_lo:p_hi+1] += v[j_idx]
        # Part 2: -h[i-s] where s = p-offset, i-s = i+offset-p, valid when 0<=i-s<=n-1
        # p in [i+offset-(n-1), i+offset]
        p_lo2 = max(0, i + offset - n + 1)
        p_hi2 = min(m - 1, i + offset)
        if p_lo2 <= p_hi2:
            idx2 = np.arange(p_lo2, p_hi2 + 1)
            j_idx2 = i + offset - idx2
            G[i, p_lo2:p_hi2+1] -= h[j_idx2]
    return G

def exact_line_search(C, L, k1, k2, t_lo, t_hi, steps=35):
    """Minimize max(C + t*L + t^2 at positions k1,k2) over t in [t_lo, t_hi]"""
    a, b = t_lo, t_hi
    for _ in range(steps):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        v1 = C + m1 * L
        if k1 >= 0:
            v1[k1] += m1 * m1
        if k2 >= 0 and k2 != k1:
            v1[k2] += m1 * m1
        f1 = np.max(v1)
        
        v2 = C + m2 * L
        if k1 >= 0:
            v2[k1] += m2 * m2
        if k2 >= 0 and k2 != k1:
            v2[k2] += m2 * m2
        f2 = np.max(v2)
        
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def coordinate_descent(h, n, rng, max_sweeps=20, time_limit=None, t0=None):
    """Run coordinate descent with exact line search"""
    offset = n - 1
    m = 2 * n - 1
    h = h.copy()
    best_local = float('inf')
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > t0 + time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        current_score = float(np.max(corr))
        
        G = build_grad_matrix(h)
        
        perm = rng.permutation(n)
        improved = False
        
        for a in range(n):
            for b in range(a + 1, n):
                i, j = perm[a], perm[b]
                
                # Feasible range for t (h[i] += t, h[j] -= t)
                t_lo = max(-h[i], h[j] - 1.0)
                t_hi = min(1.0 - h[i], h[j])
                
                if t_hi - t_lo < 1e-12:
                    continue
                
                L = G[i] - G[j]
                
                # Quadratic positions
                d = j - i
                k1 = offset + d
                k2 = offset - d
                # k1 and k2 are valid if 0 <= k < m
                if k1 < 0 or k1 >= m:
                    k1 = -1
                if k2 < 0 or k2 >= m:
                    k2 = -1
                
                t_opt = exact_line_search(corr, L, k1, k2, t_lo, t_hi)
                
                # Evaluate
                new_score = current_score
                v = corr + t_opt * L
                if k1 >= 0:
                    v[k1] += t_opt * t_opt
                if k2 >= 0 and k2 != k1:
                    v[k2] += t_opt * t_opt
                new_score = float(np.max(v))
                
                if new_score < current_score - 1e-12:
                    h[i] += t_opt
                    h[j] -= t_opt
                    corr = v
                    # Rebuild G since h changed
                    G = build_grad_matrix(h)
                    current_score = new_score
                    improved = True
        
        score = float(np.max(corr))
        if score < best_local - 1e-10:
            best_local = score
            no_improve = 0
        else:
            no_improve += 1
        
        if not improved:
            # Try perturbation
            if no_improve >= 3:
                h = h + rng.randn(n) * 0.05
                h = project_onto_constraint(h, n / 2.0)
                no_improve = 0
    
    return h, best_local
```

```python
def compute_L(h, i, j, n):
    """Compute L_p = G[i,p] - G[j,p] for all p"""
    offset = n - 1
    m = 2 * n - 1
    s = np.arange(m) - offset  # s[p] = p - offset
    
    # L[p] = (1-h[i+s]) - h[i-s] - (1-h[j+s]) + h[j-s] (with validity)
    L = np.zeros(m)
    
    # (1-h[i+s]) where 0<=i+s<=n-1
    valid1 = (i + s >= 0) & (i + s < n)
    L[valid1] += 1.0 - h[i + s[valid1]]
    
    # -h[i-s] where 0<=i-s<=n-1
    valid2 = (i - s >= 0) & (i - s < n)
    L[valid2] -= h[i - s[valid2]]
    
    # -(1-h[j+s]) where 0<=j+s<=n-1
    valid3 = (j + s >= 0) & (j + s < n)
    L[valid3] -= 1.0 - h[j + s[valid3]]
    
    # +h[j-s] where 0<=j-s<=n-1
    valid4 = (j - s >= 0) & (j - s < n)
    L[valid4] += h[j - s[valid4]]
    
    return L
```
