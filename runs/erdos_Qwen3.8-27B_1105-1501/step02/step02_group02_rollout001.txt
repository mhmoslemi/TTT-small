def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
            if abs(res) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def _grad_corr(h):
    """Gradient of each correlation component wrt h."""
    n = len(h)
    dx = 2.0 / n
    cv = np.correlate(h, 1.0 - h, mode='full')
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for p in range(j0, j1):
            grads[k, p] += (1.0 - h[p - s]) * dx
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for p in range(p0, p1):
            grads[k, p] += (-h[p + s]) * dx
    return cv, grads

def _pnorm_obj(x, p):
    n = len(x)
    cv = _corr_vec(x)
    cmax = np.max(cv)
    shifted = cv - cmax + 1e-30
    val = np.sum(shifted ** p) ** (1.0 / p)
    return cmax + val

def _pnorm_grad(x, p):
    n = len(x)
    cv = _corr_vec(x)
    cmax = np.max(cv)
    shifted = cv - cmax + 1e-30
    norm = np.sum(shifted ** p) ** (1.0 / p)
    w = (p / norm ** p) * shifted ** (p - 1)
    g = np.zeros(n)
    for k in range(len(cv)):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for pp in range(j0, j1):
            g[pp] += wk * (1.0 - x[pp - s]) * (2.0 / n)
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for pp in range(p0, p1):
            g[pp] += wk * (-x[pp + s]) * (2.0 / n)
    return g

def _lse_obj(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(cv)):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for pp in range(j0, j1):
            g[pp] += wk * (1.0 - x[pp - s]) * dx
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for pp in range(p0, p1):
            g[pp] += wk * (-x[pp + s]) * dx
    return g

def _slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def _efun(x):
        return x.sum() - n / 2.0
    def _ejac(x):
        return np.ones(n)
    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_direct(h0, maxiter=100):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = min(cv.max() + 0.003, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def _fun(z):
        return z[-1]
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv
    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _interp(h_src, n_dst):
    n_src = len(h_src)

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _sa_search(h_best, n, n_iter, rng, deadline):
    """Sum-preserving simulated annealing."""
    cur = h_best.copy()
    cur_s = _score(cur)
    best = cur.copy()
    best_s = cur_s
    T = 0.005
    T_min = 1e-6
    cooling = (T_min / T) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        if time.time() > deadline - 20:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        delta = rng.uniform(-0.15, 0.15)
        hi = cur[i] + delta
        hj = cur[j] - delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        cur[i] = hi
        cur[j] = hj
        new_s = _score(cur)
        ds = new_s - cur_s
        if ds < 0 or rng.random() < np.exp(-ds / T):
            cur_s = new_s
            if new_s < best_s:
                best_s = new_s
                best = cur.copy()
        else:
            cur[i] -= delta
            cur[j] += delta
        T *= cooling
    return best, best_s

def _gen_structured_start(n, rng):
    """Generate structured starting points."""
    starts = []
    x = np.linspace(0, 2, n)
    # Symmetric triangular
    tri = np.clip(1.0 - np.abs(x - 1.0), 0, 1)
    starts.append(_project(tri))
    # Two bumps
    bump = np.exp(-4.0 * (x - 0.5) ** 2) + np.exp(-4.0 * (x - 1.5) ** 2)
    starts.append(_project(bump))
    # Periodic
    for freq in [2, 3, 4, 5]:
        per = 0.5 + 0.3 * np.sin(np.pi * freq * x)
        starts.append(_project(per))
    # Ramp
    ramp = np.clip(x, 0, 1)
    starts.append(_project(ramp))
    # Random binary-like
    for _ in range(3):
        r = (rng.random(n) < 0.5).astype(float)
        starts.append(_project(r))
    return starts

            best_h, best_score = _fista(
                best_h, beta=beta, n_iter=min(12000, int(remaining / 8)),
                alpha_init=0.003, mu=0.9)
            remaining = deadline - time.time()

    if time.time() < deadline - 25:
        for p in [6.0, 16.0, 32.0]:
            if time.time() > deadline - 12:
                break
            new = _slsqp_pnorm(best_h, p=p, maxiter=100)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h = new
                best_score = ns

    if time.time() < deadline - 20:
        for beta in [500, 1500]:
            if time.time() > deadline - 10:
                break
            new = _slsqp_smooth(best_h, beta=beta, maxiter=120)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h = new
                best_score = ns

    if time.time() < deadline - 15:
        new = _slsqp_direct(best_h, maxiter=120)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h = new
            best_score = ns

    # Phase 5: More SA rounds + restarts
    remaining = deadline - time.time()
    n_restarts = min(10, int(remaining / 30))
    for _ in range(n_restarts):
        if time.time() > deadline - 15:
            break
        # SA from current best
        if time.time() < deadline - 10:
            n_sa = min(5000, int((deadline - time.time() - 10) * 150))
            h_sa, s_sa = _sa_search(best_h, best_n, max(n_sa, 2000), rng, deadline)
            if s_sa < best_score - 1e-12:
                best_h = h_sa.copy()
                best_score = s_sa
        # Small perturbation + quick chain
        if time.time() < deadline - 8:
            h_pert = _project(best_h + 0.02 * rng.standard_normal(best_n))
            new = _slsqp_smooth(h_pert, beta=400, maxiter=80)
            ns = _score(new)
            if ns < best_score - 1e-12:
                best_h = new
                best_score = ns

    # Final polish
    if time.time() < deadline - 5:
        best_h, best_score = _fista(best_h, beta=1000, n_iter=3000,
                                    alpha_init=0.002, mu=0.93)

    if time.time() < deadline - 3:
        new = _slsqp_smooth(best_h, beta=2000, maxiter=80)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h = new
            best_score = ns

    # Final feasibility check
    best_h = _project(best_h)
    final_score = _score(best_h)

    return (best_h, final_score, len(best_h))
