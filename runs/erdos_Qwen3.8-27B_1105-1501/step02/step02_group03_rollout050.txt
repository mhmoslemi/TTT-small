from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def repair_feasible(z, n):
    target = n / 2.0
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(z + mid, 0.0, 1.0)
        if h.sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
            else:
                add = max(residual, -h[idx])
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)


def vals_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def grad_row(h, p):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    s = p - (n - 1)
    if s >= 0:
        L = n - s
        g[:L] += dx * (1.0 - h[s:s + L])
        g[s:s + L] -= dx * h[:L]
    else:
        t = -s
        L = n - t
        g[t:t + L] += dx * (1.0 - h[:L])
        g[:L] -= dx * h[t:t + L]
    return g


def symmetrize_h(h):
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n // 2):
        val = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
        h_sym[i] = val
        h_sym[n - 1 - i] = 1.0 - val
    if n % 2 == 1:
        h_sym[n // 2] = 0.5
    return repair_feasible(h_sym, n)


def find_active(vals, tol_rel, min_count):
    cmax = vals.max()
    threshold = cmax * (1.0 - tol_rel)
    active = np.where(vals >= threshold)[0]
    if len(active) < min_count:
        k = min(min_count, len(vals))
        active = np.argsort(vals)[::-1][:k]
    return sorted(active)


class MinimaxSolver:
    def __init__(self, n, active_idx):
        self.n = n
        self.active_idx = np.array(active_idx)
        self.K = len(self.active_idx)

    def obj(self, x):
        return x[-1]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g

    def constr_fun(self, x):
        h = x[:self.n]
        t = x[-1]
        v = vals_of(h)
        return t - v[self.active_idx]

    def constr_jac(self, x):
        h = x[:self.n]
        J = np.zeros((self.K, self.n + 1))
        for i, k in enumerate(self.active_idx):
            J[i, :self.n] = -grad_row(h, k)
            J[i, self.n] = 1.0
        return J

    def eq_fun(self, x):
        return x[:self.n].sum() - self.n / 2.0

    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

    def solve(self, h0, max_iter=800, ftol=1e-15):
        v0 = vals_of(h0)
        t0 = max(v0[self.active_idx]) + 1e-6
        x0 = np.concatenate([h0, [t0]])
        bounds = [(0.0, 1.0)] * self.n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': self.constr_fun, 'jac': self.constr_jac},
            {'type': 'eq', 'fun': self.eq_fun, 'jac': self.eq_jac}
        ]
        try:
            res = minimize(self.obj, x0, jac=self.obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': ftol})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = repair_feasible(res.x[:self.n], self.n)
                sc = official_score(h_new)
                if np.isfinite(sc):
                    return h_new, sc
        except Exception:
            pass
        return None, None


def smooth_obj_grad(h, beta):
    v = vals_of(h)
    cmax = v.max()
    shifted = beta * (v - cmax)
    shifted_max = shifted.max()
    logsum = np.log(np.sum(np.exp(shifted - shifted_max))) + shifted_max
    obj = cmax + logsum / beta
    w = np.exp(shifted - shifted_max)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(v)) / len(v)
    else:
        w = w / ws
    n = len(h)
    g = np.zeros(n)
    for i in range(len(v)):
        if w[i] > 1e-15:
            g += w[i] * grad_row(h, i)
    return obj, g
