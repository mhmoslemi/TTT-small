            lo = mid
        else:
            hi = mid

    n = len(h)
    K = len(active_indices)
    G = np.zeros((K, n))
    for r, ell in enumerate(active_indices):
        s = ell - offset  # lag
        i_lo = max(0, s)
        i_hi = min(n - 1, n - 1 + s)
        if i_lo > i_hi:
            continue
        iarr = np.arange(i_lo, i_hi + 1)
        jarr = iarr - s

        np.add.at(G[r], iarr, 1.0 - h[jarr])
        # d(corr)/d(h[j]) for j in jarr: += -h[i]
        np.add.at(G[r], jarr, -h[iarr])
    return G

    n = len(h)
    K = G.shape[0]
    # Variables: d[0..n-1], theta
    # Minimize -theta

    ub = np.minimum(1.0 - h, trust)
    bounds = list(zip(lb, ub)) + [(0, None)]  # theta >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        d = res.x[:n]
        theta = res.x[n]
        return d, theta
    return None, 0.0

    best_corr_max = np.max(compute_corr(h))

        h_new = h + alpha * d
        if np.any(h_new < -1e-12) or np.any(h_new > 1.0 + 1e-12):
            continue
        h_new = np.clip(h_new, 0.0, 1.0)
        corr = compute_corr(h_new)
        cm = np.max(corr)
        if cm < best_corr_max - 1e-12:
            best_corr_max = cm
            best_alpha = alpha
    return best_alpha, best_corr_max

def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)

def make_alternating(n, rng):
    h = np.zeros(n)
    skip = rng.randint(2, 5)
    for i in range(0, n, skip):
        h[i] = 1.0
    return project_onto_constraint(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    n_base = 70
    candidates = []

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) == n_base:
            candidates.append(h0)
    except (NameError, Exception):
        pass

    if not candidates:
        candidates.append(np.full(n_base, 0.5))

    for _ in range(10):
        candidates.append(make_random_binary(n_base, rng))
    candidates.append(make_fibonacci_structure(n_base))
    for _ in range(4):
        candidates.append(make_alternating(n_base, rng))

    best_h = None
    best_c5 = float('inf')
    for h_c in candidates:
        h_c = project_onto_constraint(h_c.copy(), n_base / 2.0)
        c5 = compute_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    n = n_base
    offset = n - 1

    # Phase 1: LP-based active-lag descent
    phase1_end = t0 + budget_s * 0.65
    base_alphas = np.array([0.03, 0.05, 0.08, 0.12, 0.2, 0.35, 0.6, 1.0])
    K_active = 8
    margin = 0.015
    trust = 0.025
    stall = 0
    iters = 0
    trust_schedule = [0.02, 0.025, 0.03, 0.04, 0.05, 0.06, 0.08]
    trust_idx = 1

    while time.time() < phase1_end:
        h = best_h.copy()
        corr = compute_corr(h)
        max_corr = np.max(corr)

        # Select active lags: those within margin of max
        active_mask = corr >= max_corr - margin
        n_active = np.sum(active_mask)
        if n_active < 4:
            sorted_idx = np.argsort(corr)[::-1]
            active_indices = sorted_idx[:K_active]
        elif n_active > 25:
            # Pick the top 25 from the active set
            active_sorted = np.where(active_mask)[0]
            order = np.argsort(corr[active_sorted])[::-1]
            active_indices = active_sorted[order[:25]]
        else:
            active_indices = np.where(active_mask)[0]

        # Build gradient and solve LP
        G = build_active_gradients(h, active_indices, offset)
        slacks = max_corr - corr[active_indices]
        # Filter out zero-slack rows (exact ties) - they'll just constrain
        # but keep them for correctness
        d, theta = solve_lp_direction(h, G, slacks, trust)

        accepted = False
        if d is not None and theta > 1e-9:
            alpha, new_raw = line_search(h, d, base_alphas)
            if alpha > 0:
                h_new = h + alpha * d
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_onto_constraint(h_new, n / 2.0)
                new_c5 = compute_c5(h_new)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    accepted = True
                    stall = 0
                    iters += 1
                    # Occasionally increase trust
                    if iters % 80 == 0 and trust_idx < len(trust_schedule) - 1:
                        trust_idx += 1
                        trust = trust_schedule[trust_idx]
                    continue

        # Fallback: pairwise polish
        h2 = best_h.copy()
        random_pair_polish(h2, rng, 150)
        c5_pol = compute_c5(h2)
        if c5_pol < best_c5 - 1e-15:
            best_c5 = c5_pol
            best_h = h2.copy()
            accepted = True
            stall = 0
            iters += 1
            continue

        # Stagnation handling
        stall += 1
        if stall >= 4:
            # Random restart with noise
            scale = 0.015 + 0.01 * (stall % 4)
            h3 = best_h + rng.randn(n) * scale
            h3 = project_onto_constraint(h3, n / 2.0)
            random_pair_polish(h3, rng, 400)
            c5_r = compute_c5(h3)
            if c5_r < best_c5 - 1e-15:
                best_c5 = c5_r
                best_h = h3.copy()
                stall = 0
                iters += 1
            else:
                # Try a different subset of active lags
                corr3 = compute_corr(best_h)
                mc3 = np.max(corr3)
                # Try reducing only the single maximum
                max_idx = np.argmax(corr3)
                G_single = build_active_gradients(best_h, np.array([max_idx]), offset)
                sl_single = np.array([mc3 - corr3[max_idx] + 1e-6])
                d_s, th_s = solve_lp_direction(best_h, G_single, sl_single, trust * 0.5)
                if d_s is not None and th_s > 1e-9:
                    a_s, _ = line_search(best_h, d_s, base_alphas)
                    if a_s > 0:
                        h4 = np.clip(best_h + a_s * d_s, 0.0, 1.0)
                        h4 = project_onto_constraint(h4, n / 2.0)
                        c5_s = compute_c5(h4)
                        if c5_s < best_c5 - 1e-15:
                            best_c5 = c5_s
                            best_h = h4.copy()
                            stall = 0
                            iters += 1
                stall = 0

        # Vary parameters periodically
        if iters > 0 and iters % 120 == 0:
            margin = 0.01 + 0.01 * (iters // 120 % 3)
            K_active = 6 + 2 * (iters // 120 % 4)

    # Phase 2: Full pair sweeps with restarts
    phase2_end = t0 + budget_s * 0.88
    stall = 0
    while time.time() < phase2_end:
        h = best_h.copy()
        full_pair_sweep(h, rng)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                h = best_h + rng.randn(n) * 0.02
                h = project_onto_constraint(h, n / 2.0)
                full_pair_sweep(h, rng)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0

    # Phase 3: Final fine polishing
    h_final = best_h.copy()
    while time.time() < t0 + budget_s - 2:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = pair_optimal_delta(h_final, i, j)
        if abs(d) > 1e-12:
            h_final[i] -= d
            h_final[j] += d

    h_final = project_onto_constraint(h_final, n / 2.0)
    final_c5 = compute_c5(h_final)
    # Use the best of what we found
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5
    return best_h, best_c5, n
