def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

            best_score = s3

    # Phase 4: Try different resolutions
    n_base = len(best_h)
    res_list = sorted(set([n_base - 8, n_base - 4, n_base, n_base + 4, n_base + 8,
                            n_base + 12, 48, 56, 64, 72, 80, 96]))
    res_list = [r for r in res_list if 32 <= r <= 100]

    for n_try in res_list:
        if is_dead():
            break
        if n_try == n_base:
            continue
        h_int = interp_to(best_h, n_try)
        sc_int = official_score(h_int)
        if sc_int < best_score - 1e-12:
            best_h = h_int
            best_score = sc_int
            n_base = n_try
        # Quick QP on this resolution
        if not is_dead() and (sc_int < best_score + 0.005):
            h_q, s_q = qp_optimize(h_int, 50, is_dead)
            if s_q < best_score - 1e-12:
                best_h = h_q
                best_score = s_q
                n_base = n_try

    # Phase 5: QP at best resolution
    if not is_dead():
        h5, s5 = qp_optimize(best_h, 200, is_dead)
        if s5 < best_score - 1e-12:
            best_h = h5
            best_score = s5

    # Phase 6: Two-point polish
    if not is_dead():
        n_tries = max(2000, int((deadline - time.time()) * 3))
        h6, s6 = two_point_search(best_h, n_tries, rng, is_dead)
        if s6 < best_score - 1e-12:
            best_h = h6
            best_score = s6

    # Phase 7: QP again after two-point
    if not is_dead():
        h7, s7 = qp_optimize(best_h, 100, is_dead)
        if s7 < best_score - 1e-12:
            best_h = h7
            best_score = s7

    # Phase 8: Random perturbations + QP
    if not is_dead():
        n_best = len(best_h)
        for sigma in [0.03, 0.08, 0.12]:
            if is_dead():
                break
            for rep in range(2):
                if is_dead():
                    break
                h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n_best))
                sc_pert = official_score(h_pert)
                if sc_pert < best_score + 0.01:
                    h_p, s_p = qp_optimize(h_pert, 30, is_dead)
                    if s_p < best_score - 1e-12:
                        best_h = h_p
                        best_score = s_p

    # Phase 9: Final QP + subgradient
    if not is_dead():
        h9, s9 = qp_optimize(best_h, 80, is_dead)
        if s9 < best_score - 1e-12:
            best_h = h9
            best_score = s9
    if not is_dead():
        h10, s10 = subgradient_step(best_h, is_dead)
        if s10 < best_score - 1e-12:
            best_h = h10
            best_score = s10

    # Phase 10: Final two-point
    if not is_dead():
        n_tries = max(1000, int((deadline - time.time()) * 4))
        h11, s11 = two_point_search(best_h, n_tries, rng, is_dead)
        if s11 < best_score - 1e-12:
            best_h = h11
            best_score = s11

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8

    return final_h, final_score, final_n
