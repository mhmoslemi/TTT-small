import numpy as np
import time
from scipy.optimize import linprog


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_D_matrix(h):
    n = len(h)
    m = 2 * n - 1
    D = np.zeros((m, n))
    for j in range(n):
        ej = np.zeros(n)
        ej[j] = 1.0
        D[:, j] = np.correlate(ej, 1.0 - h, "full") - np.correlate(h, ej, "full")
    return D


def lp_full(h, D, trust):
    n = len(h)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = D
    A_ub[:, n] = 1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-trust, trust)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(hi_v - 1.0, -hj_v)
    hi = min(1.0 - hj_v, hi_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(24):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_sweep(h, rng, deadline, max_iter=None):
    n = len(h)
    count = 0
    perm = rng.permutation(n)
    for a in range(n):
        if time.time() > deadline:
            break
        if max_iter and count >= max_iter:
            break
        for b in range(a + 1, n):
            if time.time() > deadline:
                break
            if max_iter and count >= max_iter:
                break
            i, j = perm[a], perm[b]
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
            count += 1
    return h


def random_pairs(h, rng, n_pairs, deadline):
    n = len(h)
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if i > j:
            i, j = j, i
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta
    return h


def gradient_direction(h, D, corr, temperature=0.05):
    n = len(h)
    m = len(corr)
    w = np.exp((corr - np.max(corr)) / temperature)
    w /= np.sum(w)
    grad = D.T @ w
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return -grad / norm

            mode = rng.randint(3)
            if mode == 0:
                h_new, c5_new = lp_local_search(h_cur.copy(), rng, time.time() + sub_time,
                                                trust_init=0.03)
            elif mode == 1:
                h_new, c5_new = pair_local_search(h_cur.copy(), rng, time.time() + sub_time)
            else:
                h_new = h_cur + rng.randn(n_cur) * 0.02
                h_new = project_onto_constraint(h_new, n_cur / 2.0)
                random_pairs(h_new, rng, 500, time.time() + sub_time)
                c5_new = compute_c5(h_new)

            h_new, c5_new = safe_evaluate(h_new)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new
                best_c5 = c5_cur
                best_h = h_new
                stall = 0
            else:
                stall += 1
                if stall > 15:
                    break

    # Final validation
    final_h = np.clip(best_h.copy(), 0.0, 1.0)
    final_h = project_onto_constraint(final_h, len(final_h) / 2.0)
    final_c5 = compute_c5(final_h)

    if not np.isfinite(final_c5) or final_c5 > best_c5 + 1e-6:
        final_h = project_onto_constraint(best_h.copy(), len(best_h) / 2.0)
        final_c5 = compute_c5(final_h)

    return final_h, final_c5, len(final_h)
