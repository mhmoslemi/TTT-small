import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_of_g(h, s):
    """Compute gradient of g_s with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    L = n - abs(s)
    if L <= 0:
        return grad
    if s >= 0:
        grad[s:n] += (1.0 - h[0:L])
        grad[0:L] -= h[s:n]
    else:
        t = -s
        grad[0:L] += (1.0 - h[t:n])
        grad[t:n] -= h[0:L]
    return dx * grad

def weighted_gradient(h, weights):
    """Compute weighted sum of gradients."""
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def selections_search(h, top_k, beta):
    """Compute selection weights using top-k + softmax blend."""
    n = len(h)
    V = compute_values(h)
    # Top-k uniform + softmax blend

    w_top = np.zeros(len(V))

def lp_polish(h, active_shifts, trust=0.15):
    """Solve LP to minimize t subject to linearized constraints."""
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    
    # Build LP: variables are d[0..n-1] and t
    # minimize t
    # subject to: for each active s: g_s(h) + ∇g_s(h)·d ≤ t
    #             -trust ≤ d_i ≤ trust
    #             sum(d) = 0
    #             t ≥ 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub_rows = []
    b_ub_rows = []
    
    for s_idx in active_shifts:
        s = s_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = (1.0 - h[0:L])
            gp[0:L] -= h[s:n]
        else:
            t_neg = -s
            gp[0:L] = (1.0 - h[t_neg:n])
            gp[t_neg:n] -= h[0:L]
        gp *= dx
        
        # g_s(h) + gp·d ≤ t  =>  gp·d - t ≤ -g_s(h)
        row = np.zeros(n + 1)
        row[:n] = gp
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(-V[s_idx])
    
    if not A_ub_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))  # t ≥ 0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c, A_ub=np.array(A_ub_rows), b_ub=np.array(b_ub_rows),
                      A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            h_new = make_feasible(h + d)
            sc = official_score(h_new)
            return h_new, sc
    except Exception:
        pass
    return None, None

Let me also think about whether the correlate computation is exactly what we want. The problem says:
∫ h(x)(1-h(x+k)) dx

In the discrete case with h defined at points x_i = (i+0.5)*dx:
∫ h(x)(1-h(x+k)) dx ≈ ∑_i h_i * (1 - h_{i+k}) * dx (when both indices are valid)

This is exactly what np.correlate gives (times dx). Good.

Now, one subtle point: the "full" mode correlate gives 2n-1 values. The shift s = j - (n-1) ranges from -(n-1) to n-1. For each shift s, the number of terms is n - |s|. So:
- s=0: n terms
- s=±1: n-1 terms
- ...
- s=±(n-1): 1 term

The value g_s = (∑ h_i (1-h_{i+s})) * dx, where the sum has n-|s| terms.

For the max to be small, we need all g_s to be small. The ones with the most terms (small |s|) are the hardest to keep small.
