import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_G(h):
    """Build gradient matrix G[i,m] = d corr[m]/d h[i]"""
    n = len(h)
    m_total = 2 * n - 1
    G = np.zeros((n, m_total))
    for i in range(n):
        # First term: (1-h[i-m+n-1]) for i <= m <= i+n-1
        m_lo = max(0, i)
        m_hi = min(m_total - 1, i + n - 1)
        if m_lo <= m_hi:
            indices = i - np.arange(m_lo, m_hi + 1) + n - 1
            G[i, m_lo:m_hi+1] += (1.0 - h[indices])
        # Second term: -h[i+m-n+1] for n-1-i <= m <= 2n-2-i
        m_lo2 = max(0, n - 1 - i)
        m_hi2 = min(m_total - 1, 2 * n - 2 - i)
        if m_lo2 <= m_hi2:
            indices2 = i + np.arange(m_lo2, m_hi2 + 1) - n + 1
            G[i, m_lo2:m_hi2+1] -= h[indices2]
    return G

def line_search_pair(corr, G, h, i, j, n):

        improved = False
        for a_idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            i = perm[a_idx]
            for b_idx in range(a_idx + 1, n):
                j = perm[b_idx]
                t = line_search_pair(corr, G, h, i, j, n)
                if abs(t) > 1e-10:
                    h[i] += t
                    h[j] -= t
                    # Rebuild G and corr
                    G = build_G(h)
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    improved = True
        if not improved:
            break
        # Check improvement
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return best_c5

def make_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_structured(n, freq, shift=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / 2.0 + shift)
    return np.clip(h, 0, 1)

def make_golden(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * (phi - 1.0)) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    # Adjust to get exactly n/2 ones
    n_ones = np.sum(h)
    if n_ones < n / 2:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:int(n/2 - n_ones)]:
            h[idx] = 1.0
    elif n_ones > n / 2:
        ones = np.where(h == 1)[0]
        for idx in ones[:int(n_ones - n/2)]:
            h[idx] = 0.0
    return h
