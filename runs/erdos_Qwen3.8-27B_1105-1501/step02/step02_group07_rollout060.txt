    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def nesterov_optimize(h0, beta_start, n_iters, alpha_max=0.7, beta_rate=1.025, beta_cap=300.0):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    beta = beta_start
    best_h = h.copy()
    best_f = official_score(h)

    for it in range(n_iters):
        V = compute_values(y)
        w = softmax_weights(V, beta)
        g = fast_gradient(y, w)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn

        best_h_it = h
        best_f_it = official_score(h)
        alpha = alpha_max
        for _ in range(8):
            h_new = make_feasible(y - alpha * g)
            f_new = official_score(h_new)
            if f_new < best_f_it - 1e-14:
                best_h_it = h_new
                best_f_it = f_new
            alpha *= 0.55
            if alpha < 1e-7:
                break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = best_h_it + ((t - 1.0) / t_new) * (best_h_it - h)
        y = np.clip(y, 0.0, 1.0)
        h = best_h_it
        t = t_new

        if best_f_it < best_f - 1e-14:
            best_f = best_f_it
            best_h = h.copy()

        beta = min(beta * beta_rate, beta_cap)

    return best_h, best_f

def cooperative_shift_improve(h, n_iter=200):
    """Equalize pairs at the worst shift cooperatively."""
    n = len(h)
    best_h = h.copy()
    best_f = official_score(h)
    stuck = 0

    for _ in range(n_iter):
        if stuck > 30:
            break
        V = compute_values(h)
        k_max = int(np.argmax(V))
        s = k_max - (n - 1)
        if s == 0:
            stuck += 1
            continue

        improved = False
        for frac in [0.5, 0.3, 0.15, 0.08]:
            h_test = h.copy()
            if s > 0:
                L = n - s
                imbalances = h[s:s+L] - h[0:L]
                for i in range(L):
                    delta = imbalances[i] * frac / 2.0
                    h_test[i] += delta
                    h_test[i + s] -= delta
            else:
                t = -s
                L = n - t
                imbalances = h[0:L] - h[t:t+L]
                for i in range(L):
                    delta = imbalances[i] * frac / 2.0
                    h_test[i + t] += delta
                    h_test[i] -= delta

            h_test = make_feasible(h_test)
            f_test = official_score(h_test)
            if f_test < best_f - 1e-14:
                best_f = f_test
                best_h = h_test.copy()
                h = h_test.copy()
                improved = True
                break

        if improved:
            stuck = 0
        else:
            stuck += 1

    return best_h, best_f

def pair_swap_improve(h, n_iter=400):
    n = len(h)
    best_h = h.copy()
    best_f = official_score(h)
    stuck = 0

    for _ in range(n_iter):
        if stuck > 40:
            break
        V = compute_values(h)
        k_max = int(np.argmax(V))
        s = k_max - (n - 1)
        if s == 0:
            stuck += 1
            continue

        if s > 0:
            L = n - s
            contribs = h[0:L] * (1.0 - h[s:n])
        else:
            t = -s
            L = n - t
            contribs = h[t:n] * (1.0 - h[0:L])

        if contribs.max() < 1e-10:
            break

        idx = int(np.argmax(contribs))
        if s > 0:
            i, j = idx, idx + s
        else:
            i, j = idx + (-s), idx

        if h[i] <= h[j]:
            i, j = j, i

        improved = False
        for frac in [0.8, 0.5, 0.3, 0.15]:
            delta = (h[i] - h[j]) * frac
            h_test = h.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = make_feasible(h_test)
            f_test = official_score(h_test)
            if f_test < best_f - 1e-14:
                best_f = f_test
                best_h = h_test.copy()
                h = h_test.copy()
                improved = True
                break

        if improved:
            stuck = 0
        else:
            stuck += 1

    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.15):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def binary_project(h):
    """Project to binary preserving sum constraint exactly."""
    n = len(h)
    target = int(n // 2)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(10.0, 0.03 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    # Get initial values
    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
        h0 = make_feasible(h0)
    except (NameError, AttributeError):
        n0 = 60
        x = np.arange(n0) / n0
        h0 = 0.5 + 0.2 * np.sin(np.pi * x) + 0.08 * np.sin(2 * np.pi * x)
        h0 = make_feasible(h0)

    best_h = h0.copy()
    best_score = official_score(h0)

    # Phase 1: Nesterov optimization on initial n
    h_cur = h0.copy()
    h_new, sc = nesterov_optimize(h_cur, beta_start=20.0, n_iters=400,
                                   alpha_max=0.7, beta_rate=1.025, beta_cap=250.0)
    if sc < best_score:
        best_h = h_new.copy()
        best_score = sc
    h_cur = best_h.copy()

    # Phase 2: Cooperative shift improvement
    if time_left() > 400:
        h_new, sc = cooperative_shift_improve(h_cur, n_iter=300)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc
        h_cur = best_h.copy()

    # Phase 3: Pair swap
    if time_left() > 350:
        h_new, sc = pair_swap_improve(h_cur, n_iter=400)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc
        h_cur = best_h.copy()

    # Phase 4: LP polishing
    if time_left() > 250:
        trust = 0.15
        for eps in [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]:
            if time_left() < 150:
                break
            for _ in range(25):
                if time_left() < 150:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=trust)
                if h_new is not None:
                    if sc < best_score - 1e-12:
                        best_h = h_new.copy()
                        best_score = sc
                        h_cur = h_new
                        trust = min(trust * 1.1, 0.25)
                    else:
                        h_cur = h_new
                        trust = max(trust * 0.8, 0.02)

    # Phase 5: High-beta Nesterov
    if time_left() > 200:
        h_new, sc = nesterov_optimize(h_cur, beta_start=80.0, n_iters=200,
                                       alpha_max=0.4, beta_rate=1.02, beta_cap=400.0)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc

    # Phase 6: Try different n values
    n_base = len(best_h)
    n_candidates = []
    for dn in [-15, -10, -5, 5, 10, 15, 20, -25, 25]:
        nt = n_base + dn
        if 24 <= nt <= 120 and nt not in n_candidates:
            n_candidates.append(nt)

    for n_try in n_candidates:
        if time_left() < 100:
            break
        h_int = interpolate_to(best_h, n_try)
        sc_int = official_score(h_int)
        if sc_int < best_score - 1e-12:
            best_h = h_int.copy()
            best_score = sc_int

        h_cur = h_int.copy()
        h_new, sc = nesterov_optimize(h_cur, beta_start=30.0, n_iters=150,
                                       alpha_max=0.6, beta_rate=1.02, beta_cap=200.0)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc

        if time_left() < 80:
            break

        h_new, sc = pair_swap_improve(best_h, n_iter=150)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc

        if time_left() < 60:
            break

        h_new, sc = cooperative_shift_improve(best_h, n_iter=100)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc

    # Phase 7: Binary projection attempts
    n_cur = len(best_h)
    if time_left() > 60:
        h_bin = binary_project(best_h)
        sc_bin = official_score(h_bin)
        if sc_bin < best_score:
            best_h = h_bin.copy()
            best_score = sc_bin
        else:
            # Try binary then optimize from there
            h_bin_opt, sc_bin_opt = nesterov_optimize(
                h_bin.astype(float), beta_start=50.0, n_iters=100,
                alpha_max=0.5, beta_rate=1.03, beta_cap=200.0)
            if sc_bin_opt < best_score:
                best_h = h_bin_opt.copy()
                best_score = sc_bin_opt

    # Phase 8: Random restarts
    for restart in range(4):
        if time_left() < 60:
            break
        n_r = len(best_h)
        h_pert = best_h + rng.normal(0, 0.04, n_r)
        h_pert = make_feasible(h_pert)
        h_new, sc = nesterov_optimize(h_pert, beta_start=40.0, n_iters=120,
                                       alpha_max=0.6, beta_rate=1.03, beta_cap=250.0)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc

    # Phase 9: Final LP + Nesterov polish
    if time_left() > 20:
        h_cur = best_h.copy()
        for eps in [0.001, 0.0005]:
            if time_left() < 15:
                break
            for _ in range(12):
                if time_left() < 15:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.04)
                if h_new is not None:
                    if sc < best_score - 1e-13:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    if time_left() > 10:
        h_new, sc = nesterov_optimize(best_h, beta_start=200.0, n_iters=50,
                                       alpha_max=0.2, beta_rate=1.01, beta_cap=500.0)
        if sc < best_score:
            best_h = h_new.copy()
            best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
