import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_G(h, n):
    m = 2 * n - 1
    offset = n - 1
    G = np.zeros((n, m))
    for p in range(n):
        s_min = max(p - (n - 1), -(n - 1))
        s_max = min(p, n - 1)
        if s_min <= s_max:
            s_range = np.arange(s_min, s_max + 1)
            G[p, s_range + offset] += (1.0 - h[p - s_range])
        s_max2 = n - 1 - p
        if s_max2 >= 0:
            s_range2 = np.arange(0, s_max2 + 1)
            G[p, s_range2 + offset] += -h[p + s_range2]
    return G

def optimize_pairs(h, n, rng, deadline, steps=20):
    """Iterative pairwise coordinate descent."""
    best_c5 = compute_c5(h)
    h = h.copy()
    
    while time.time() < deadline:
        # Rebuild G and corr
        corr = np.correlate(h, 1.0 - h, mode="full")
        G = build_G(h, n)
        
        improved = False
        perm = rng.permutation(n)
        
        for ai in range(n):
            for aj in range(ai + 1, n):
                i, j = int(perm[ai]), int(perm[aj])
                if time.time() > deadline:
                    return h, best_c5
                
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-13:
                    continue
                
                offset = n - 1
                direction = G[i] - G[j]
                k1 = i - j + offset
                k2 = j - i + offset
                
                # Ternary search
                a, b = lo, hi
                for _ in range(steps):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    v1 = corr + m1 * direction
                    v1[k1] += m1 * m1
                    v1[k2] += m1 * m1
                    f1 = np.max(v1)
                    v2 = corr + m2 * direction
                    v2[k1] += m2 * m2
                    v2[k2] += m2 * m2
                    f2 = np.max(v2)
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                
                t_opt = (a + b) / 2.0
                if abs(t_opt) < 1e-10:
                    continue
                
                # Apply
                h[i] += t_opt
                h[j] -= t_opt
                # Clip for safety
                h[i] = np.clip(h[i], 0.0, 1.0)
                h[j] = np.clip(h[j], 0.0, 1.0)
                
                # Update corr and G
                corr = np.correlate(h, 1.0 - h, mode="full")
                G = build_G(h, n)
                improved = True
        
        if not improved:
            break
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5
