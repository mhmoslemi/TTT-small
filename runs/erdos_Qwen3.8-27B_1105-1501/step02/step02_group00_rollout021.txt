def _smooth_step(h, sigma):
    """Apply a small smoothing and re-project"""
    n = len(h)
    # Manual Gaussian smoothing (circular to avoid boundary effects)
    # Use a simple 3-point or 5-point kernel
    kernel_size = max(3, int(2 * sigma * 3) | 1)  # odd
    kernel = np.zeros(kernel_size)
    center = kernel_size // 2
    for i in range(kernel_size):
        kernel[i] = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    kernel /= np.sum(kernel)
    
    # Circular convolution
    h_smooth = np.convolve(np.tile(h, 3), kernel, mode='same')[n:2*n]
    return _project(h_smooth, n / 2.0)

        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]

    hi_a = min(alpha_max, best_a + span)
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi_a - lo_a < 1e-11:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_a = opt_a
    return best_a, best_c5

def _lp_optimize(h, deadline, radius_init=0.12, max_iters=4000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            p = _subgrad_direction(h, G, beta=4.0)
            if p is None:
                stall += 1
                radius *= 0.8
                if radius < 1e-5:
                    break
                continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall % 12 == 0 and radius < 0.03:
                radius = 0.04
                kick = rng.standard_normal(n)
                kick -= np.mean(kick)
                kick /= (np.max(np.abs(kick)) + 1e-15)
                h_try = _project(h + 0.008 * kick, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5:
                    h = h_try
                    cur_c5 = c5_try
    return best_h, best_c5

def _smooth_step(h, sigma):
    n = len(h)
    ksize = max(3, int(4 * sigma) | 1)
    center = ksize // 2
    kernel = np.array([np.exp(-0.5 * ((i - center) / max(sigma, 0.3)) ** 2) for i in range(ksize)])
    kernel /= np.sum(kernel)
    h_ext = np.concatenate([h[-center:], h, h[:center]])
    h_smooth = np.convolve(h_ext, kernel, mode='valid')[:n]
    return _project(h_smooth, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 0.4
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for a in [a_max * t for t in [0.25, 0.5, 0.75, 1.0]]:
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            h = h_t
                            improved = True
                            break
        if not improved and _pass > 0:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Main LP optimization (32%)
    deadline1 = t0 + budget_s * 0.32
    h, c5 = _lp_optimize(h, deadline1, radius_init=0.15, max_iters=6000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Resolution sweep (18%)
    deadline2 = t0 + budget_s * 0.50
    n_candidates = list(range(max(32, best_n - 10), min(128, best_n + 12)))
    n_candidates.sort(key=lambda x: abs(x - best_n))
    for n_try in n_candidates:
        if time.time() > deadline2:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        sub_deadline = min(deadline2, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_deadline, radius_init=0.1, max_iters=400)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 3: Diverse perturbations + LP (25%)
    deadline3 = t0 + budget_s * 0.75
    solutions = [(best_h.copy(), best_c5)]
    scale = 0.04
    for attempt in range(500):
        if time.time() > deadline3:
            break
        if best_c5 < 0.3806:
            break
        pt = attempt % 5
        h_pert = best_h.copy()
        if pt == 0:
            h_pert += rng.normal(0, scale, best_n)
        elif pt == 1:
            freq = rng.integers(1, best_n // 2 + 1)
            amp = scale * 2.5
            phase = rng.uniform(0, 2 * np.pi)
            h_pert += amp * np.sin(2 * np.pi * freq * np.arange(best_n) / best_n + phase)
        elif pt == 2:
            h_pert = _smooth_step(best_h, rng.uniform(0.5, 2.5))
        elif pt == 3:
            bstart = rng.integers(0, max(1, best_n - 12))
            blen = rng.integers(4, max(5, best_n // 3))
            h_pert[bstart:bstart + blen] += scale * 3 * rng.uniform(-1, 1)
        else:
            h_pert += scale * rng.choice([-1, 1], size=best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(deadline3, time.time() + budget_s * 0.018)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.08, max_iters=350)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            solutions.append((best_h.copy(), best_c5))
            scale = max(0.01, scale * 0.88)
        else:
            scale = min(0.25, scale * 1.12)

    # Phase 4: Convex combinations (8%)
    deadline4 = t0 + budget_s * 0.83
    if len(solutions) >= 2:
        solutions.sort(key=lambda x: x[1])
        thetas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]
        for i in range(min(len(solutions), 6)):
            if time.time() > deadline4:
                break
            for j in range(i + 1, min(len(solutions), 10)):
                if time.time() > deadline4:
                    break
                if len(solutions[i][0]) != len(solutions[j][0]):
                    continue
                n_mix = len(solutions[i][0])
                for theta in thetas:
                    h_mix = theta * solutions[i][0] + (1 - theta) * solutions[j][0]
                    h_mix = _project(h_mix, n_mix / 2.0)
                    c5_mix = _exact_c5(h_mix)
                    if c5_mix < best_c5 - 1e-10:
                        best_c5 = c5_mix
                        best_h = h_mix.copy()
                        best_n = n_mix
                        solutions.append((best_h.copy(), best_c5))

    # Phase 5: Pair polish (8%)
    deadline5 = t0 + budget_s * 0.91
    h, c5 = _pair_polish(best_h, deadline5, max_passes=40)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Final LP polish (9%)
    if time.time() < t0 + budget_s * 0.98:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.993,
                                  radius_init=0.04, max_iters=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
