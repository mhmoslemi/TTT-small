        g[t:n] = -h[0:L]
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, V.max()
    g /= gn
    best_h = h.copy()
    best_f = V.max()
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def lp_polish_step(h, eps, trust, max_active=120):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    if len(active) > max_active:
        sorted_idx = np.argsort(V[active])
        active = active[sorted_idx[-max_active:]]
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, p_idx in enumerate(active):
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            A_ub[i, s:n] = 1.0 - h[0:L]
            A_ub[i, 0:L] = -h[s:n]
        else:
            t_shift = -s
            A_ub[i, 0:L] = 1.0 - h[t_shift:n]
            A_ub[i, t_shift:n] = -h[0:L]
        A_ub[i, :n] *= dx
        A_ub[i, n] = -1.0
        b_ub[i] = -V[p_idx]
    bounds = [(max(-h[i], -trust), min(1.0 - h[i], trust)) for i in range(n)]
    bounds.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                     bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                for frac in [1.0, 0.75, 0.5, 0.3, 0.15, 0.08]:
                    h_try = make_feasible(h + frac * d)
                    sc = official_score(h_try)
                    if sc < V_max - 1e-13:
                        return h_try, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(8.0, 0.03 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        h = np.asarray(initial_h_values, dtype=float).ravel()
        h = make_feasible(h)
    except (NameError, AttributeError):
        n = 60
        x = np.arange(n) / n
        h = 0.5 + 0.2 * np.sin(np.pi * x) + 0.08 * np.sin(2 * np.pi * x)
        h = make_feasible(h)

    def time_left():
        return deadline - time.time()

    best_h = h.copy()
    best_score = official_score(h)

    # Phase 1: Alternating gradient methods
    h_cur = h.copy()
    beta = 12.0
    stall = 0
    use_steepest = False
    while time_left() > 350:
        if use_steepest:
            h_new, sc = steepest_step(h_cur, alpha_max=0.7, max_ls=10)
            use_steepest = False
        else:
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.8, max_ls=12)
            use_steepest = True
        if sc < best_score - 1e-12:
            best_h = h_new.copy()
            best_score = sc
            stall = 0
        elif sc < official_score(h_cur) - 1e-12:
            stall = 0
        else:
            stall += 1
        h_cur = h_new
        beta = min(beta * 1.03, 300.0)
        if stall > 50:
            h_cur = make_feasible(best_h + rng.normal(0, 0.015, len(best_h)))
            beta = 8.0
            stall = 0

    # Phase 2: Aggressive LP with cycling eps and trust management
    h_cur = best_h.copy()
    trust = 0.10
    consecutive_fail = 0
    total_lp = 0
    eps_cycle = [0.005, 0.0025, 0.001, 0.0005, 0.0002, 0.0001]
    eps_idx = 0
    while time_left() > 150:
        eps = eps_cycle[eps_idx % len(eps_cycle)]
        eps_idx += 1
        h_new, sc = lp_polish_step(h_cur, eps=eps, trust=trust)
        total_lp += 1
        if h_new is not None:
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
                h_cur = h_new
                trust = min(trust * 1.04, 0.35)
                consecutive_fail = 0
            else:
                h_cur = h_new
                consecutive_fail += 1
        else:
            consecutive_fail += 1
        if consecutive_fail > 20:
            h_cur = make_feasible(best_h + rng.normal(0, 0.002, len(best_h)))
            trust = 0.06
            consecutive_fail = 0

    # Phase 3: Fine LP with shrinking trust
    h_cur = best_h.copy()
    trust = 0.015
    fine_fail = 0
    for _ in range(2000):
        if time_left() < 40:
            break
        eps = 0.0003 + 0.0002 * (trust / 0.015)
        h_new, sc = lp_polish_step(h_cur, eps=eps, trust=trust)
        if h_new is not None:
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                trust = min(trust * 1.02, 0.03)
            else:
                h_cur = h_new
            fine_fail = 0
        else:
            fine_fail += 1
        trust = max(trust * 0.995, 0.001)
        if fine_fail > 50:
            h_cur = make_feasible(best_h + rng.normal(0, 0.001, len(best_h)))
            trust = 0.01
            fine_fail = 0

    # Phase 4: Multi-resolution exploration
    if time_left() > 100:
        n_base = len(best_h)
        n_options = [n_base - 4, n_base - 2, n_base + 2, n_base + 4, n_base + 8, n_base + 12]
        for n_try in n_options:
            if time_left() < 60 or n_try < 24:
                continue
            h_int = interpolate_to(best_h, n_try)
            sc_cur = official_score(h_int)
            if sc_cur >= best_score + 0.001:
                continue
            h_cur = h_int.copy()
            beta = 20.0
            for _ in range(60):
                if time_left() < 60:
                    break
                h_new, sc = pg_step(h_cur, beta, alpha_max=0.5, max_ls=8)
                if sc < sc_cur - 1e-13:
                    sc_cur = sc
                h_cur = h_new
                beta = min(beta * 1.03, 150.0)
            trust = 0.08
            for _ in range(60):
                if time_left() < 60:
                    break
                h_new, sc = lp_polish_step(h_cur, eps=0.002, trust=trust)
                if h_new is not None:
                    if sc < sc_cur - 1e-14:
                        sc_cur = sc
                    h_cur = h_new
                    trust = max(trust * 0.92, 0.005)
                else:
                    trust = max(trust * 0.95, 0.003)
            if sc_cur < best_score - 1e-12:
                best_h = h_cur.copy()
                best_score = sc_cur

    # Phase 5: Final ultra-fine LP on best
    if time_left() > 15:
        h_cur = best_h.copy()
        trust = 0.005
        for _ in range(800):
            if time_left() < 8:
                break
            h_new, sc = lp_polish_step(h_cur, eps=0.0001, trust=trust, max_active=80)
            if h_new is not None:
                if sc < best_score - 1e-15:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
            trust = max(trust * 0.99, 0.0005)

    # Phase 6: Last resort - steepest on worst then LP
    if time_left() > 8:
        h_cur = best_h.copy()
        for _ in range(30):
            if time_left() < 5:
                break
            h_new, sc = steepest_step(h_cur, alpha_max=0.3, max_ls=8)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            h_new2, sc2 = lp_polish_step(h_cur, eps=0.0005, trust=0.01)
            if h_new2 is not None and sc2 < best_score - 1e-14:
                best_h = h_new2.copy()
                best_score = sc2
                h_cur = h_new2

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
