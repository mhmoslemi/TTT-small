```python
    dx = 2.0 / n
    
    # Get active lags
    corr = np.correlate(h0, 1.0 - h0, mode='full') * dx

    def obj(z):
        return z[-1]

        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    cons = []

                    if i + d != i:

        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
```

```python
def coord_descent_greedy(h0, max_moves=500, time_limit=None, t_start=None, n_try=100):
    h = h0.copy()
    n = len(h)
    best_score = score(h)
    
    for move in range(max_moves):
        if time_limit and t_start and time.time() - t_start > time_limit + t_start:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        
        best_improve = 0
        best_pair = None
        best_delta = 0
        
        for _ in range(n_try):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            g = pair_grad_diff(h, i, j)
            delta = find_best_delta(corr, g, h[i], h[j])
            if abs(delta) > 1e-11:
                new_max = np.max(corr + delta * g)
                improve = max_corr - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
                    best_delta = delta
        
        if best_pair is None or best_improve < 1e-12:
            break
        
        i, j = best_pair
        h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
        h[j] = np.clip(h[j] - best_delta, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - n/2.0) > 1e-10:
            # Redistribute the error
            err = n/2.0 - s
            # Add to the coordinate with more room
            if err > 0:
                room = 1.0 - h
            else:
                room = h.copy()
            idx = np.argmax(room)
            h[idx] += err
            h = np.clip(h, 0.0, 1.0)
        
        new_score = score(h)
        if new_score < best_score:
            best_score = new_score
    
    return h, best_score
```
