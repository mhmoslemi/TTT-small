import numpy as np

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def build_gradient_matrix(h):
    n = len(h)
    offset = n - 1
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        e = k - offset
        # p appears as i: j = p - e, valid if 0 <= p-e < n
        # p appears as j: i = p + e, valid if 0 <= p+e < n
        for p in range(n):

            if 0 <= j < n:
                g += (1.0 - h[j])
            i = p + e
            if 0 <= i < n:
                g += (-h[i])
            G[k, p] = g
    return G

        cp.sum(d) == 0,
        d >= -h,
        d <= 1.0 - h,
        cp.abs(d) <= trust_radius,
        theta >= 0
    ]
    obj = cp.Maximize(theta)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iter=10000)
    except:
        try:
            prob.solve(solver=cp.CLARABEL)
        except:
            return None
    if d.value is None:

import numpy as np
import time
import cvxpy as cp


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def build_gradient_matrix(h):
    n = len(h)
    offset = n - 1
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        e = k - offset
        for p in range(n):
            g = 0.0
            j = p - e
            if 0 <= j < n:
                g += (1.0 - h[j])
            i = p + e
            if 0 <= i < n:
                g -= h[i]
            G[k, p] = g
    return G


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def solve_lp_direction(h, corr, trust_radius):
    n = len(h)
    corr_max = np.max(corr)
    G = build_gradient_matrix(h)
    d = cp.Variable(n)
    theta = cp.Variable()
    constraints = [
        G @ d + theta <= (corr_max - corr),
        cp.sum(d) == 0,
        d >= -h + 1e-10,
        d <= 1.0 - h - 1e-10,
        cp.abs(d) <= trust_radius,
        theta >= 0
    ]
    prob = cp.Problem(cp.Maximize(theta), constraints)
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, max_iter=10000)
            if d.value is not None:
                d_val = np.array(d.value).flatten()
                if np.all(np.isfinite(d_val)):
                    d_val = np.clip(d_val, -0.999, 0.999)
                    d_val -= np.sum(d_val) / n
                    return d_val, float(theta.value) if theta.value is not None else 0.0
        except Exception:
            continue
    return None, 0.0

    dx = 2.0 / n
    best_alpha = 0.0
    best_val = best_c5

        if np.any(h_new < -1e-10) or np.any(h_new > 1.0 + 1e-10):
            continue

        h_new = project_onto_constraint(h_new, n / 2.0)
        c5 = compute_c5(h_new)
        if c5 < best_val:
            best_val = c5
            best_alpha = alpha
    return best_alpha, best_val


def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_sweep(h, rng, num_pairs):
    n = len(h)
    for _ in range(num_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta

import numpy as np
import time
import cvxpy as cp


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def build_gradient_matrix(h):
    n = len(h)
    offset = n - 1
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        e = k - offset
        for p in range(n):
            g = 0.0
            j = p - e
            if 0 <= j < n:
                g += (1.0 - h[j])
            i = p + e
            if 0 <= i < n:
                g -= h[i]
            G[k, p] = g
    return G


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def solve_lp_direction(h, corr, trust_radius):
    n = len(h)
    corr_max = np.max(corr)
    G = build_gradient_matrix(h)
    d = cp.Variable(n)
    theta = cp.Variable()
    constraints = [
        G @ d + theta <= (corr_max - corr),
        cp.sum(d) == 0,
        d >= -h + 1e-12,
        d <= 1.0 - h - 1e-12,
        d <= trust_radius,
        d >= -trust_radius,
        theta >= 0
    ]
    prob = cp.Problem(cp.Maximize(theta), constraints)
    solvers = [cp.SCS, cp.CLARABEL, cp.ECOS, cp.PDLP, cp.CBC]
    for solver in solvers:
        try:
            prob.solve(solver=solver, max_iter=20000)
            if d.value is not None:
                d_val = np.array(d.value).flatten()
                if np.all(np.isfinite(d_val)) and np.linalg.norm(d_val) > 1e-12:
                    d_val = np.clip(d_val, -0.999, 0.999)
                    d_val = np.clip(d_val, -trust_radius, trust_radius)
                    d_val -= np.sum(d_val) / n
                    tv = float(theta.value) if theta.value is not None else 0.0
                    return d_val, tv
        except Exception:
            continue
    return None, 0.0


def line_search(h, d, best_c5):
    n = len(h)
    best_alpha = 0.0
    best_val = best_c5
    alphas = [0.05, 0.1, 0.15, 0.25, 0.35, 0.5, 0.7, 1.0]
    for alpha in alphas:
        h_new = h + alpha * d
        if np.any(h_new < -1e-9) or np.any(h_new > 1.0 + 1e-9):
            continue
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_constraint(h_new, n / 2.0)
        c5 = compute_c5(h_new)
        if c5 < best_val:
            best_val = c5
            best_alpha = alpha
    return best_alpha, best_val


def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(25):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_sweep(h, rng, num_pairs):
    n = len(h)
    for _ in range(num_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta


def full_pair_sweep(h, rng):
    n = len(h)
    perm = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = int(perm[a]), int(perm[b])
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
