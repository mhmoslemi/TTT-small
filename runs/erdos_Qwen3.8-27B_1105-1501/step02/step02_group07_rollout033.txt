def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def fourier_parameterize(coeffs, n):
    """Convert Fourier coefficients to h vector.
    coeffs = [a1, b1, a2, b2, ..., aM, bM]
    h[i] = 0.5 + sum_m (a_m * cos(2*pi*m*(i+0.5)/n) + b_m * sin(2*pi*m*(i+0.5)/n))
    """
    M = len(coeffs) // 2
    i = (np.arange(n) + 0.5) / n
    h = np.full(n, 0.5)
    for m in range(M):
        a = coeffs[2*m]
        b = coeffs[2*m + 1]
        h += a * np.cos(2 * np.pi * m * i) + b * np.sin(2 * np.pi * m * i)
    return make_feasible(h)

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def pg_step(h, beta, alpha_max=0.8, max_ls=20):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = fast_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def pg_step_momentum(h, mom_state, beta, alpha=0.3):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = fast_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f, mom_state
    g /= gn
    m = 0.9 * mom_state + 0.1 * g
    mn = np.linalg.norm(m)
    if mn < 1e-15:
        return h, f, g
    m = m / mn
    h_new = make_feasible(h - alpha * m)
    f_new = official_score(h_new)
    if f_new < f - 1e-14:
        return h_new, f_new, m
    return h, f, m

def pairwise_exchange(h, max_pairs=300, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_f = official_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_pairs):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d_lo = max(-best_h[i], best_h[j] - 1.0)
        d_hi = min(1.0 - best_h[i], best_h[j])
        if d_hi - d_lo < 1e-10:
            continue
        a = d_lo
        b = d_hi
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        h_c = best_h.copy()
        h_c[i] += c
        h_c[j] -= c
        fc = official_score(h_c)
        h_d = best_h.copy()
        h_d[i] += d
        h_d[j] -= d
        fd = official_score(h_d)
        for _ in range(35):
            if b - a < 1e-10:
                break
            if fc < fd:
                b = d
                d = c
                h_d, fd = h_c, fc
                c = b - gr * (b - a)
                h_c = best_h.copy()
                h_c[i] += c
                h_c[j] -= c
                fc = official_score(h_c)
            else:
                a = c
                h_c, fc = h_d, fd
                c = b - gr * (b - a)
                h_c = best_h.copy()
                h_c[i] += c
                h_c[j] -= c
                fc = official_score(h_c)
        d_best = (a + b) / 2.0
        d_best = max(d_lo, min(d_hi, d_best))
        h_try = best_h.copy()
        h_try[i] += d_best
        h_try[j] -= d_best
        f_try = official_score(h_try)
        if f_try < best_f - 1e-14:
            best_h = h_try
            best_f = f_try
    return best_h, best_f

def pairwise_directed(h, max_pairs=200):
    """Use gradient to choose pairs: high gradient with low gradient."""
    n = len(h)
    V = compute_values(h)
    w = softmax_weights(V, 50.0)
    g = fast_gradient(h, w)
    g_centered = g - g.mean()
    best_h = h.copy()
    best_f = official_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    idx_hi = np.argsort(g_centered)[-5:]
    idx_lo = np.argsort(g_centered)[:5]
    for _ in range(max_pairs):
        i = int(idx_hi[_ % len(idx_hi)])
        j = int(idx_lo[_ % len(idx_lo)])
        if i == j:
            continue
        # We want to increase h[i] (negative gradient means decrease obj) and decrease h[j]
        # Actually: g[i] = dV/dh[i]. To decrease V, we want to move h[i] in -g[i] direction.
        # If g[i] < 0, increasing h[i] helps. If g[j] > 0, decreasing h[j] helps.
        # So d > 0 means h[i] += d, h[j] -= d
        # This helps if g[i] < 0 (increasing h[i] decreases V) and g[j] > 0 (decreasing h[j] decreases V)
        d_lo = max(-best_h[i], best_h[j] - 1.0)
        d_hi = min(1.0 - best_h[i], best_h[j])
        if d_hi - d_lo < 1e-10:
            continue
        a = d_lo
        b = d_hi
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        h_c = best_h.copy()
        h_c[i] += c
        h_c[j] -= c
        fc = official_score(h_c)
        h_d = best_h.copy()
        h_d[i] += d
        h_d[j] -= d
        fd = official_score(h_d)
        for _ in range(30):
            if b - a < 1e-10:
                break
            if fc < fd:
                b = d
                d = c
                h_d, fd = h_c, fc
                c = b - gr * (b - a)
                h_c = best_h.copy()
                h_c[i] += c
                h_c[j] -= c
                fc = official_score(h_c)
            else:
                a = c
                h_c, fc = h_d, fd
                c = b - gr * (b - a)
                h_c = best_h.copy()
                h_c[i] += c
                h_c[j] -= c
                fc = official_score(h_c)
        d_best = (a + b) / 2.0
        d_best = max(d_lo, min(d_hi, d_best))
        h_try = best_h.copy()
        h_try[i] += d_best
        h_try[j] -= d_best
        f_try = official_score(h_try)
        if f_try < best_f - 1e-14:
            best_h = h_try
            best_f = f_try
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def fourier_perturb(h, n_modes, scale, rng):
    """Perturb h in Fourier space."""
    n = len(h)
    t = (np.arange(n) + 0.5) / n
    ac = h - 0.5
    coeffs = np.zeros(2 * n_modes)
    for m in range(n_modes):
        coeffs[2*m] = 2.0 * np.mean(ac * np.cos(2*np.pi*m*t))
        coeffs[2*m+1] = 2.0 * np.mean(ac * np.sin(2*np.pi*m*t))
    coeffs += rng.normal(0, scale, 2 * n_modes)
    h_new = np.full(n, 0.5)
    for m in range(n_modes):
        h_new += coeffs[2*m] * np.cos(2*np.pi*m*t)
        h_new += coeffs[2*m+1] * np.sin(2*np.pi*m*t)
    # Blend with original to keep it in range
    h_new = 0.5 * h + 0.5 * h_new
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.04 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        h_init = make_feasible(h_init)
        n_base = len(h_init)
    except (NameError, AttributeError):
        n_base = 60
        h_init = make_feasible(np.full(n_base, 0.5))

    best_h = h_init.copy()
    best_score = official_score(h_init)

    # Phase 1: Projected gradient from initial (use ~35% budget)
    h_cur = h_init.copy()
    beta = 20.0
    stall = 0
    while time_left() > budget_s * 0.35:
        h_new, sc = pg_step(h_cur, beta, alpha_max=0.8, max_ls=20)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
            stall = 0
        elif sc < official_score(h_cur) - 1e-13:
            stall = 0
        else:
            stall += 1
        h_cur = h_new
        beta = min(beta * 1.04, 400.0)
        if stall > 80:
            h_cur = make_feasible(best_h + rng.normal(0, 0.03, len(best_h)))
            beta = 15.0
            stall = 0

    # Phase 2: LP polish
    h_cur = best_h.copy()
    for eps_val in [0.01, 0.005, 0.002, 0.001]:
        if time_left() < budget_s * 0.25:
            break
        for _ in range(40):
            if time_left() < budget_s * 0.25:
                break
            h_new, sc = active_set_lp(h_cur, eps=eps_val, trust=0.15)
            if h_new is not None:
                if sc < best_score - 1e-13:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new

    # Phase 3: Directed pairwise exchange
    if time_left() > budget_s * 0.2:
        for _ in range(3):
            if time_left() < budget_s * 0.2:
                break
            h_new, sc = pairwise_directed(best_h, max_pairs=300)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc

    # Phase 4: Random pairwise exchange
    if time_left() > budget_s * 0.15:
        pair_rng = np.random.default_rng(seed + 1)
        h_cur = best_h.copy()
        for _ in range(3):
            if time_left() < budget_s * 0.15:
                break
            h_new, sc = pairwise_exchange(h_cur, max_pairs=400, rng=pair_rng)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new

    # Phase 5: Fourier perturbation + restart
    if time_left() > budget_s * 0.15:
        for attempt in range(8):
            if time_left() < budget_s * 0.15:
                break
            scale = 0.08 / (attempt + 1)
            h_try = fourier_perturb(best_h, 5, scale, rng)
            h_cur = h_try
            beta = 30.0
            for _ in range(300):
                if time_left() < budget_s * 0.15:
                    break
                h_new, sc = pg_step(h_cur, beta, alpha_max=0.5, max_ls=15)
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
                beta = min(beta * 1.02, 300.0)
            # Quick LP polish
            for eps_val in [0.005, 0.002]:
                if time_left() < budget_s * 0.15:
                    break
                for _ in range(15):
                    if time_left() < budget_s * 0.15:
                        break
                    h_new, sc = active_set_lp(h_cur, eps=eps_val, trust=0.1)
                    if h_new is not None:
                        if sc < best_score - 1e-14:
                            best_h = h_new.copy()
                            best_score = sc
                        h_cur = h_new

    # Phase 6: Multi-resolution
    if time_left() > budget_s * 0.1:
        for n_try in [n_base - 5, n_base + 5, n_base + 10, n_base - 10, n_base + 15]:
            if n_try < 20 or n_try > 130:
                continue
            if time_left() < budget_s * 0.04:
                break
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            h_cur = h_int
            beta = 30.0
            for _ in range(400):
                if time_left() < budget_s * 0.04:
                    break
                h_new, sc = pg_step(h_cur, beta, alpha_max=0.6, max_ls=15)
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
                if _ % 15 == 14:
                    beta *= 1.06
            # LP at this resolution
            for eps_val in [0.003, 0.001]:
                if time_left() < budget_s * 0.04:
                    break
                for _ in range(20):
                    if time_left() < budget_s * 0.04:
                        break
                    h_new, sc = active_set_lp(h_cur, eps=eps_val, trust=0.08)
                    if h_new is not None:
                        if sc < best_score - 1e-14:
                            best_h = h_new.copy()
                            best_score = sc
                        h_cur = h_new
            # Pairwise at this resolution
            if time_left() > budget_s * 0.04:
                pair_rng2 = np.random.default_rng(seed + n_try)
                h_new, sc = pairwise_exchange(h_cur, max_pairs=200, rng=pair_rng2)
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc

    # Phase 7: Final intensive polish on best
    h_cur = best_h.copy()
    # Alternating methods
    for it in range(800):
        if time_left() < 10:
            break
        if it % 15 == 0:
            h_new, sc = active_set_lp(h_cur, eps=0.001, trust=0.05)
            if h_new is not None:
                h_cur = h_new
        elif it % 15 == 7:
            h_new, sc = pairwise_directed(h_cur, max_pairs=50)
            h_cur = h_new
        else:
            beta = 200 + it * 0.5
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.25, max_ls=12)
            h_cur = h_new
        sc = official_score(h_cur)
        if sc < best_score - 1e-14:
            best_h = h_cur.copy()
            best_score = sc

    # Phase 8: Final verification
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    if final_score >= best_score + 1e-10:
        final_h = best_h
        final_score = best_score

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
