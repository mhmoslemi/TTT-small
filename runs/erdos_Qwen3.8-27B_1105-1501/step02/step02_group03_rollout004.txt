import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_all(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, s:] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[s:]
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K]
    return G

def symmetrize_h(h):
    """Project h onto symmetric subspace h[i] + h[n-1-i] = 1"""
    n = len(h)
    if n % 2 != 0:
        # For odd n, the middle element must be 0.5
        m = n // 2
        h_sym = h.copy()
        for i in range(m):
            avg = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
            h_sym[i] = avg
            h_sym[n - 1 - i] = 1.0 - avg
        h_sym[m] = 0.5
        return make_feasible(h_sym)
    m = n // 2
    h_sym = np.empty(n)
    for i in range(m):
        avg = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
        h_sym[i] = np.clip(avg, 0.0, 1.0)
        h_sym[n - 1 - i] = 1.0 - h_sym[i]
    return h_sym

def embed_sym(x):
    """Embed symmetric free variables into full h"""
    m = len(x)
    n = 2 * m
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def vals_from_x(x):
    """Compute correlation values from symmetric free variables"""
    h = embed_sym(x)
    return compute_vals(h)

def grad_from_x(x):
    """Compute gradient of all correlation values w.r.t. symmetric free variables"""
    h = embed_sym(x)
    G = grad_all(h)  # (2n-1, n)
    m = len(x)
    # dV/dx_j = G[:,j] - G[:,n-1-j]
    G_sym = G[:, :m] - G[:, n-1-::-1]
    return G_sym

def smooth_obj_grad_sym(x, beta):
    """Smooth log-sum-exp objective and gradient in symmetric subspace"""
    vals = vals_from_x(x)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = grad_from_x(x)
    grad = w @ G
    return obj, grad

def pg_solve_sym(x0, beta, time_limit=30.0, max_iter=5000):
    """Projected gradient in symmetric subspace"""
    x = np.clip(x0.copy(), 0.0, 1.0)
    f, g = smooth_obj_grad_sym(x, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-28:
        return x
    alpha = 1.0 / (gnorm2 + 1e-12)
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-28:
            break
        step = alpha
        for _ in range(30):
            x_trial = np.clip(x - step * g, 0.0, 1.0)
            f_trial = smooth_obj_grad_sym(x_trial, beta)[0]
            if f_trial <= f - 1e-4 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-14:
            break
        x = np.clip(x - step * g, 0.0, 1.0)
        f_new, g = smooth_obj_grad_sym(x, beta)
        if not np.isfinite(f_new):
            break
        f = f_new
        if f < f_new:  # this shouldn't happen, but safety
            alpha = min(alpha * 1.8, 5.0 / (gnorm2 + 1e-12))
    return x

def slsqp_sym_full(x0, max_iter=500, ftol=1e-15):
    """SLSQP in symmetric subspace with ALL constraints"""
    m = len(x0)
    n = 2 * m
    t0_init = vals_from_x(x0).max() + 1e-6
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def con_fun(z):
        x = z[:m]
        t = z[-1]
        vals = vals_from_x(x)
        return t - vals
    
    def con_jac(z):
        x = z[:m]
        G = grad_from_x(x)  # (2n-1, m)
        J = np.zeros((2 * m - 1, m + 1))
        J[:, :m] = -G
        J[:, m] = 1.0
        return J
    
    z0 = np.concatenate([np.clip(x0, 0.0, 1.0), [t0_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            x_new = np.clip(res.x[:m], 0.0, 1.0)
            sc = vals_from_x(x_new).max()
            if np.isfinite(sc):
                return x_new, sc
    except Exception:
        pass
    return None, None

def find_active(vals, frac=0.25, min_count=8):
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0])
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

    h_int = np.interp(tgt_c, src_c, h_src)
