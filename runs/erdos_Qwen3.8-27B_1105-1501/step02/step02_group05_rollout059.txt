        order = np.argsort(1.0 - h)  # smallest slack first? No, largest slack first
        # Actually: sort by (1-h) descending to add to those with most room
        order = np.argsort(-(1.0 - h))
    else:
        order = np.argsort(h)  # smallest h first to subtract from
    for idx in order:
        if residual > 0:
            add = min(residual, 1.0 - h[idx])
            if add <= 1e-15: continue
            h[idx] += add
            residual -= add
        else:
            sub = min(-residual, h[idx])
            if sub <= 1e-15: continue
            h[idx] -= sub
            residual += sub
        if abs(residual) < 1e-14:
            break
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(h, p):
    """Compute gradient of the p-th correlation value (times dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += -h[s:]
        grad[s:] += 1.0 - h[:m]
    else:
        t = -s
        m = n - t
        grad[:m] += 1.0 - h[t:]
        grad[t:] += -h[:m]
    return dx * grad

def all_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def solve_lp_step(h, V, C, active_p, rho):
    """Solve LP to find direction that reduces all active lags."""
    n = len(h)
    K = len(active_p)
    if K == 0:
        return None, C
    
    # Build gradient matrix
    G = np.array([grad_for_lag(h, p) for p in active_p])  # shape (K, n)
    
    # LP variables: [u_0, ..., u_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: G @ u - t <= -V[active_p]
    A_ub = np.hstack([G, -np.ones((K, 1))])
    b_ub = -V[active_p]
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((0.0, C))  # t bounds
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.fun < C - 1e-10:
            u = res.x[:n]
            t_pred = res.x[n]
            return u, t_pred
    except Exception:
        pass
    return None, C

def binary_sa(h_init, n_max_iter, T_start, T_end, rng, is_dead):

def lp_refine(h, max_iter, eta_start, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    eta = eta_start
    rho = 0.25
    stall = 0
    for it in range(max_iter):
        if is_dead():
            break
        V = all_lag_values(best)
        C = float(V.max())
        active_mask = V >= C - eta
        active_p = np.where(active_mask)[0]
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(V)[-top_k:]
        active_p = np.unique(np.concatenate([active_p, top_idx]))
        if len(active_p) > 50:
            active_p = np.sort(active_p)[-50:]
        u, t_pred = solve_lp_step(best, V, C, active_p, rho)
        if u is not None:
            h_new, sc_new = line_search(best, u, C, is_dead)
            if sc_new < best_sc - 1e-13:
                best = h_new
                best_sc = sc_new
                stall = 0
                rho = min(rho * 1.1, 0.4)
            else:
                stall += 1
                if stall % 8 == 0:
                    rho *= 0.7
                    if rho < 0.02:
                        rho = 0.3
                        eta = min(eta * 1.8, 0.04)
        else:
            stall += 1
            if stall % 10 == 0:
                rho = 0.3
                eta = min(eta * 1.5, 0.04)
        if stall > 60:
            break
    return best, best_sc
