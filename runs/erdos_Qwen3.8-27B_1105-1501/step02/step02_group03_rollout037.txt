def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    # Bisection to fix sum
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))

    if abs(residual) > 1e-14:

            for idx in order:

def solve_antisym(h0_full, n, t_init, active_set=None, maxiter=300):
    """
    Solve the anti-symmetric minimax problem using SLSQP.
    h0_full: initial full h vector (length n)
    n: number of samples
    t_init: initial upper bound on max value
    active_set: optional list of constraint indices to use
    """
    m = n // 2
    h0 = antisym_project(h0_full, n)[:m]
    
    def build_h(free_vars):
        """Reconstruct full h from free variables"""
        h = np.zeros(n)
        h[:m] = free_vars
        for i in range(m):
            h[n-1-i] = 1.0 - free_vars[i]
        return h
    
    def compute_vals_from_free(free_vars):
        h = build_h(free_vars)
        return np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    
    def compute_grad_from_free(free_vars):
        h = build_h(free_vars)
        n_h = len(h)
        dx = 2.0 / n_h
        G = np.zeros((n_h, 2*n_h - 1))
        for s in range(-(n_h-1), n_h):
            p = s + (n_h - 1)
            if s >= 0:
                L = n_h - s
                G[:L, p] += dx * (1.0 - h[s:])
                G[s:, p] -= dx * h[:L]
            else:
                a = -s
                L = n_h - a
                G[a:, p] += dx * (1.0 - h[:L])
                G[:L, p] -= dx * h[a:]
        # Project to anti-symmetric
        G_sym = G[:m, :] - G[m:, :][::-1, :]
        return G_sym
    
    # Objective and constraints
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    if active_set is None:
        active_set = list(range(2*n - 1))
    
    def constraint_fun(x):
        free = x[:m]
        t = x[-1]
        vals = compute_vals_from_free(free)
        return t - vals[active_set]
    
    def constraint_jac(x):
        free = x[:m]
        G_sym = compute_grad_from_free(free)
        J = -G_sym[:, active_set].T
        full_J = np.hstack([J, np.ones((len(active_set), 1))])
        return full_J
    
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    x0 = np.concatenate([h0, [t_init]])
    
    try:
        res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_full = build_h(res.x[:m])
            h_full = make_feasible(h_full)
            score = official_score(h_full)
            if np.isfinite(score):
                return h_full, score
    except Exception:
        pass
    return None, None

import numpy as np
from scipy.optimize import minimize
import time

# Global problem state for SLSQP callables (no closures allowed)
_pa = {}  # antisymmetric problem
_pf = {}  # full problem


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def official_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)


def antisym_project(h):
    return 0.5 * (h + 1.0 - h[::-1].copy())


def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    return np.interp(tgt_c, src_c, h_src)


def build_h_antisym(free, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = free
    h[m:] = 1.0 - free[::-1]
    return h


def compute_grad_full(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((n, 2 * n - 1))
    for s in range(-(n - 1), n):
        p = s + (n - 1)
        if s >= 0:
            L = n - s
            G[:L, p] += dx * (1.0 - h[s:])
            G[s:, p] -= dx * h[:L]
        else:
            a = -s
            L = n - a
            G[a:, p] += dx * (1.0 - h[:L])
            G[:L, p] -= dx * h[a:]
    return G


def find_active(h, frac=0.3, min_count=10):
    vals = official_vals(h)
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 5e-5)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)


# === Antisymmetric SLSQP ===
def _a_obj(x):
    return x[-1]


def _a_obj_grad(x):
    m = _pa['m']
    g = np.zeros(m + 1)
    g[-1] = 1.0
    return g


def _a_con(x):
    n = _pa['n']
    m = _pa['m']
    active = _pa['active']
    h = build_h_antisym(x[:m], n)
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return x[-1] - vals[active]


def _a_con_jac(x):
    n = _pa['n']
    m = _pa['m']
    active = _pa['active']
    h = build_h_antisym(x[:m], n)
    dx = 2.0 / n
    Gf = np.zeros((n, 2 * n - 1))
    for s in range(-(n - 1), n):
        p = s + (n - 1)
        if s >= 0:
            L = n - s
            Gf[:L, p] += dx * (1.0 - h[s:])
            Gf[s:, p] -= dx * h[:L]
        else:
            a = -s
            L = n - a
            Gf[a:, p] += dx * (1.0 - h[:L])
            Gf[:L, p] -= dx * h[a:]
    Gs = Gf[:m, :] - Gf[m:, :][::-1, :]
    J = -Gs[:, active].T
    return np.hstack([J, np.ones((len(active), 1))])


def solve_antisym(h0, n, t_init, active=None, maxiter=500, ftol=1e-10):
    m = n // 2
    if active is None:
        active = list(range(2 * n - 1))
    _pa['n'] = n
    _pa['m'] = m
    _pa['active'] = active

    h_sym = antisym_project(h0)
    h0_free = h_sym[:m]
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': _a_con, 'jac': _a_con_jac}]
    x0 = np.concatenate([h0_free, [t_init]])
    try:
        res = minimize(_a_obj, x0, jac=_a_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_full = build_h_antisym(res.x[:m], n)
            h_full = make_feasible(h_full)
            sc = official_score(h_full)
            if np.isfinite(sc) and sc < t_init + 1e-9:
                return h_full, sc
    except Exception:
        pass
    return None, None


# === Full (non-antisymmetric) SLSQP ===
def _f_obj(x):
    return x[-1]


def _f_obj_grad(x):
    n = _pf['n']
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g


def _f_con(x):
    n = _pf['n']
    active = _pf['active']
    h = x[:n]
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return x[-1] - vals[active]


def _f_con_jac(x):
    n = _pf['n']
    active = _pf['active']
    h = x[:n]
    Gf = compute_grad_full(h)
    J = -Gf[:, active].T
    return np.hstack([J, np.ones((len(active), 1))])


def _f_eq(x):
    n = _pf['n']
    return np.sum(x[:n]) - n / 2.0


def _f_eq_jac(x):
    n = _pf['n']
    g = np.zeros(n + 1)
    g[:n] = np.ones(n)
    return g


def solve_full(h0, n, t_init, active=None, maxiter=300, ftol=1e-10):
    if active is None:
        active = list(range(2 * n - 1))
    _pf['n'] = n
    _pf['active'] = active
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': _f_con, 'jac': _f_con_jac},
        {'type': 'eq', 'fun': _f_eq, 'jac': _f_eq_jac}
    ]
    x0 = np.concatenate([h0, [t_init]])
    try:
        res = minimize(_f_obj, x0, jac=_f_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_full = make_feasible(res.x[:n])
            sc = official_score(h_full)
            if np.isfinite(sc) and sc < t_init + 1e-9:
                return h_full, sc
    except Exception:
        pass
    return None, None
