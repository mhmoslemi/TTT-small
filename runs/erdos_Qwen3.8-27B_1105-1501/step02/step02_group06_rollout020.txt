def official_score(h):
    n = len(h)
    dx = 2.0 / n

def compute_corr(h):

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - z.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - z)
            for idx in order:
                add = min(residual, 1.0 - z[idx])
                if add <= 0: break
                z[idx] += add
                residual -= add
        else:
            order = np.argsort(z)
            for idx in order:
                sub = min(-residual, z[idx])
                if sub <= 0: break
                z[idx] -= sub
                residual += sub
    return np.clip(z, 0.0, 1.0)

def lp_direction(h, top_k):
    """Find a direction d that reduces all top-k correlation values."""
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    dx = 2.0 / n
    
    # For each active lag s = idx - (n-1), the gradient of corr[s] w.r.t. h is:
    # d/dh_i corr[s] = dx * (1 - h[i+s]) if i+s valid, minus dx * h[i-s] if i-s valid
    # (gradient of h[i]*(1-h[i+s]) w.r.t. h[i] is (1-h[i+s]), w.r.t. h[i+s] is -h[i])
    
    # We want: for each active lag p, sum_i grad_p[i] * d[i] <= 0
    # Subject to: sum(d) = 0, ||d|| <= 1
    
    # This is an LP: minimize 0 (or some small objective) subject to constraints
    # Or: minimize -sum of slacks (maximize the minimum rate of decrease)
    
    # Simpler: use the average gradient of top-k lags as negative direction
    avg_grad = np.zeros(n)
    for p in top_idx:
        s = p - (n - 1)
        g = np.zeros(n)
        for i in range(n):

            j2 = i - s
            if 0 <= j2 < n:
                g[i] -= h[j2]
        avg_grad += g
    avg_grad /= len(top_idx)
    
    # Project to sum-zero subspace
    d = -avg_grad - np.mean(-avg_grad)
    return d

def qp_refine(h, max_iter=20, k_active=5, reg=0.01):

        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1
        ]

            s = p - (n - 1)
            g = np.zeros(n)

                    g[i] += (1.0 - cur[j])

                    g[i] -= cur[j2]

        prob = cp.Problem(obj, constraints)

            break

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_for_lag(h, s):
    """Gradient of corr at lag s (where corr index = s + n - 1) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        j = i + s
        if 0 <= j < n:
            g[i] += (1.0 - h[j])
        j2 = i - s
        if 0 <= j2 < n:
            g[i] -= h[j2]
    return g

def softmax_subgradient(h, beta):
    """Smoothed subgradient of max using softmax over correlations."""
    n = len(h)
    corr = compute_corr(h)
    corr = corr - corr.min()  # shift for numerical stability
    weights = np.exp(beta * (corr - corr.max()))
    weights /= weights.sum()
    g = np.zeros(n)
    for p in range(len(corr)):
        if weights[p] > 1e-10:
            s = p - (n - 1)
            g += weights[p] * grad_for_lag(h, s)
    return g

            update(h_opt, sc_opt)
            if sc_opt < best_score + 0.01 and remaining() > 15:
                h_new, sc = slsqp_smooth(h_opt, 3000, maxiter=60)
                update(h_new, sc)

    # Phase 6: Intensive final polish
    if remaining() > 40:
        for k_a in [3, 5, 7, 10, 15]:
            if remaining() < 10:
                break
            h_opt, sc_opt = qp_refine(best_h, max_iter=15, k_active=k_a, reg=0.003)
            update(h_opt, sc_opt)

        if remaining() > 20:
            h_new, sc = slsqp_smooth(best_h, 30000, maxiter=150)
            update(h_new, sc)

        if remaining() > 20:
            h_gd, sc_gd = subgradient_descent(best_h, max_iter=200, beta=80, eta0=0.02)
            update(h_gd, sc_gd)
            h_pol, sc_pol = qp_refine(h_gd, max_iter=15, k_active=5, reg=0.002)
            update(h_pol, sc_pol)

    # Phase 7: Fine n-adjustment around best
    best_n = len(best_h)
    if remaining() > 30:
        for dn in [-3, -2, -1, 1, 2, 3, -5, 5]:
            if remaining() < 10:
                break
            n_try = best_n + dn
            if n_try < 14 or n_try > 120:
                continue
            h_try = interpolate_to(best_h, n_try)
            h_opt, sc_opt = qp_refine(h_try, max_iter=12, k_active=5, reg=0.01)
            update(h_opt, sc_opt)
            if remaining() > 12:
                h_new, sc = slsqp_smooth(h_opt, 5000, maxiter=80)
                update(h_new, sc)

    # Phase 8: Very final micro-polish
    if remaining() > 15:
        for reg_v in [0.001, 0.0005, 0.0001]:
            if remaining() < 8:
                break
            h_opt, sc_opt = qp_refine(best_h, max_iter=20, k_active=6, reg=reg_v)
            update(h_opt, sc_opt)

    if remaining() > 10:
        h_new, sc = slsqp_smooth(best_h, 50000, maxiter=100)
        update(h_new, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    # Verify feasibility
    assert abs(np.sum(final_h) - len(final_h) / 2.0) < 1e-10
    assert np.all(final_h >= -1e-12) and np.all(final_h <= 1.0 + 1e-12)
    final_h = np.clip(final_h, 0.0, 1.0)
    # Re-check score after clip
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
