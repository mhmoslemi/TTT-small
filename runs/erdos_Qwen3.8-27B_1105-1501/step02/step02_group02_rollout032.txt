```python
def _coord_descent(h, n_sweeps=100):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    cv = _corr_vec(h)  # already multiplied by dx
    
    for sweep in range(n_sweeps):
        max_improve = 0.0
        for i in range(n):
            old_val = h[i]
            
            # Compute coefficient array
            # coef[k] = (1-h[i+k-n+1]) * [n-1-i <= k <= 2n-2-i] - h[i-k+n-1] * [i <= k <= i+n-1]
            coef = np.zeros(2*n-1)
            
            # Case 1: k in [n-1-i, 2n-2-i]
            k1_lo = n - 1 - i
            k1_hi = 2*n - 2 - i
            if k1_lo >= 0 and k1_hi < 2*n-1:
                # second index = i + k - n + 1, for k in [k1_lo, k1_hi]
                # = i + (n-1-i) - n + 1 = 0, up to i + (2n-2-i) - n + 1 = n-1
                idx2 = np.arange(k1_lo, k1_hi+1) - (n - 1 - i)  # 0, 1, 2, ..., n-1-i
                coef[k1_lo:k1_hi+1] = 1.0 - h[idx2]
            
            # Case 2: k in [i, i+n-1]
            k2_lo = i
            k2_hi = i + n - 1
            if k2_lo >= 0 and k2_hi < 2*n-1:
                # p = i - k + n - 1, for k in [k2_lo, k2_hi]
                # = i - i + n - 1 = n-1, down to i - (i+n-1) + n - 1 = 0
                idx1 = np.arange(k2_hi, k2_lo-1, -1) - (k2_hi - k2_lo)  # n-1, n-2, ..., 0
                # Actually: for k=k2_lo=i, p = i-i+n-1 = n-1
                # for k=k2_hi=i+n-1, p = i-(i+n-1)+n-1 = 0
                # So p = n-1-(k-k2_lo) = n-1-k+i
                # Let's just compute directly
                ks = np.arange(k2_lo, k2_hi+1)
                ps = i - ks + n - 1  # p values
                coef[k2_lo:k2_hi+1] -= h[ps]
            
            coef = dx * coef
            
            # Minimize max_k (cv[k] + (v - old_val)*coef[k]) over v in [0,1]
            # = max_k ((cv[k] - old_val*coef[k]) + v*coef[k])
            a = cv - old_val * coef
            b = coef
            
            # Ternary search
            lo, hi_v = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi_v - lo) * 0.333
                m2 = hi_v - (hi_v - lo) * 0.333
                f1 = np.max(a + m1 * b)
                f2 = np.max(a + m2 * b)
                if f1 <= f2:
                    hi_v = m2
                else:
                    lo = m1
            
            v_opt = (lo + hi_v) * 0.5
            v_opt = max(0.0, min(1.0, v_opt))
            
            delta = v_opt - old_val
            if abs(delta) > 1e-12:
                new_score = np.max(cv + delta * coef)
                old_score = np.max(cv)
                if new_score < old_score - 1e-14:
                    h[i] = v_opt
                    cv = cv + delta * coef
                    max_improve = max(max_improve, old_score - new_score)
        
        if max_improve < 1e-13:
            break
    
    return h, _score(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(res) < 1e-14:
                break
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _coord_descent(h, n_sweeps=200, alpha=1.0, project_every=10):
    """Fast coordinate descent with incremental correlation updates."""
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    cv = _corr_vec(h)
    
    for sweep in range(n_sweeps):
        max_improve = 0.0
        for i in range(n):
            old_val = h[i]
            
            # Compute coefficient array for h[i] in each correlation
            coef = np.zeros(2 * n - 1)
            
            # Case 1: h[i] appears as first factor h[p] where p=i
            # k in [n-1-i, 2n-2-i], second index = i + k - n + 1
            k1_lo = n - 1 - i
            k1_hi = 2 * n - 2 - i
            if k1_lo >= 0 and k1_hi < 2 * n - 1:
                idx2 = np.arange(k1_lo, k1_hi + 1) - (n - 1 - i)
                coef[k1_lo:k1_hi + 1] = 1.0 - h[idx2]
            
            # Case 2: h[i] appears as second factor (1-h[p+s]) where p+s=i
            # k in [i, i+n-1], first index p = i - k + n - 1
            k2_lo = i
            k2_hi = i + n - 1
            if k2_lo >= 0 and k2_hi < 2 * n - 1:
                ks = np.arange(k2_lo, k2_hi + 1)
                ps = i - ks + n - 1
                coef[k2_lo:k2_hi + 1] -= h[ps]
            
            coef *= dx
            
            # Minimize max_k (cv[k] + (v - old_val)*coef[k]) over v in [0,1]
            a = cv - old_val * coef
            b = coef
            
            # Check if current value is already optimal
            f_lo = np.max(a + 0.0 * b)  # = max(a) at v=0
            f_hi = np.max(a + 1.0 * b)  # = max(a+b) at v=1
            f_cur = np.max(cv)
            
            if f_lo >= f_cur - 1e-14 and f_hi >= f_cur - 1e-14:
                # Current is at boundary or interior minimum
                # Try ternary search anyway for potential improvement
                pass
            
            lo, hi_v = 0.0, 1.0
            for _ in range(35):
                m1 = lo + (hi_v - lo) / 3.0
                m2 = hi_v - (hi_v - lo) / 3.0
                f1 = np.max(a + m1 * b)
                f2 = np.max(a + m2 * b)
                if f1 <= f2:
                    hi_v = m2
                else:
                    lo = m1
            
            v_opt = (lo + hi_v) * 0.5
            v_opt = max(0.0, min(1.0, v_opt))
            
            new_val = old_val + alpha * (v_opt - old_val)
            new_val = max(0.0, min(1.0, new_val))
            
            delta = new_val - old_val
            if abs(delta) > 1e-12:
                new_max = np.max(cv + delta * coef)
                old_max = np.max(cv)
                if new_max < old_max - 1e-14:
                    h[i] = new_val
                    cv = cv + delta * coef
                    max_improve = max(max_improve, old_max - new_max)
        
        # Periodic projection to maintain sum constraint
        if (sweep + 1) % project_every == 0:
            h_new = _project(h)
            s_new = _score(h_new)
            s_old = np.max(cv)
            if s_new <= s_old + 1e-10:
                h = h_new
                cv = _corr_vec(h)
        
        if max_improve < 1e-14:
            break
    
    h = _project(h)
    return h, _score(h)
```

```python
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(res) < 1e-14:
                break
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _coord_descent(h, n_sweeps=200, project_every=8, tol=1e-14):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    cv = _corr_vec(h)
    for sweep in range(n_sweeps):
        max_imp = 0.0
        for i in range(n):
            old_val = h[i]
            coef = np.zeros(2 * n - 1)
            k1_lo = n - 1 - i
            k1_hi = 2 * n - 2 - i
            if k1_lo >= 0 and k1_hi < 2 * n - 1:
                idx2 = np.arange(k1_lo, k1_hi + 1) - (n - 1 - i)
                coef[k1_lo:k1_hi + 1] = 1.0 - h[idx2]
            k2_lo = i
            k2_hi = i + n - 1
            if k2_lo >= 0 and k2_hi < 2 * n - 1:
                ks = np.arange(k2_lo, k2_hi + 1)
                ps = i - ks + n - 1
                coef[k2_lo:k2_hi + 1] -= h[ps]
            coef *= dx
            a = cv - old_val * coef
            b = coef
            lo_v, hi_v = 0.0, 1.0
            for _ in range(40):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                f1 = np.max(a + m1 * b)
                f2 = np.max(a + m2 * b)
                if f1 <= f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) * 0.5
            v_opt = max(0.0, min(1.0, v_opt))
            delta = v_opt - old_val
            if abs(delta) > 1e-13:
                new_max = np.max(cv + delta * coef)
                old_max = np.max(cv)
                if new_max < old_max - tol:
                    h[i] = v_opt
                    cv = cv + delta * coef
                    max_imp = max(max_imp, old_max - new_max)
        if (sweep + 1) % project_every == 0:
            h_pr = _project(h)
            s_pr = np.max(_corr_vec(h_pr))
            s_cur = np.max(cv)
            if s_pr <= s_cur + 1e-10:
                h = h_pr
                cv = _corr_vec(h)
        if max_imp < tol:
            break
    h = _project(h)
    return h, _score(h)


def _coord_descent_momentum(h, n_sweeps=300, alpha=0.7, project_every=10):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    cv = _corr_vec(h)
    vel = np.zeros(n)
    for sweep in range(n_sweeps):
        max_imp = 0.0
        for i in range(n):
            old_val = h[i]
            coef = np.zeros(2 * n - 1)
            k1_lo = n - 1 - i
            k1_hi = 2 * n - 2 - i
            if k1_lo >= 0 and k1_hi < 2 * n - 1:
                idx2 = np.arange(k1_lo, k1_hi + 1) - (n - 1 - i)
                coef[k1_lo:k1_hi + 1] = 1.0 - h[idx2]
            k2_lo = i
            k2_hi = i + n - 1
            if k2_lo >= 0 and k2_hi < 2 * n - 1:
                ks = np.arange(k2_lo, k2_hi + 1)
                ps = i - ks + n - 1
                coef[k2_lo:k2_hi + 1] -= h[ps]
            coef *= dx
            a = cv - old_val * coef
            b = coef
            lo_v, hi_v = 0.0, 1.0
            for _ in range(35):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                f1 = np.max(a + m1 * b)
                f2 = np.max(a + m2 * b)
                if f1 <= f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) * 0.5
            v_opt = max(0.0, min(1.0, v_opt))
            target = old_val + alpha * (v_opt - old_val)
            vel[i] = alpha * vel[i] + (1.0 - alpha) * (target - old_val)
            new_val = max(0.0, min(1.0, old_val + vel[i]))
            delta = new_val - old_val
            if abs(delta) > 1e-13:
                new_max = np.max(cv + delta * coef)
                old_max = np.max(cv)
                if new_max < old_max - 1e-14:
                    h[i] = new_val
                    cv = cv + delta * coef
                    max_imp = max(max_imp, old_max - new_max)
                else:
                    vel[i] *= 0.5
        if (sweep + 1) % project_every == 0:
            h_pr = _project(h)
            s_pr = np.max(_corr_vec(h_pr))
            s_cur = np.max(cv)
            if s_pr <= s_cur + 1e-10:
                h = h_pr
                cv = _corr_vec(h)
        if max_imp < 1e-14:
            break
    h = _project(h)
    return h, _score(h)


def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta


def _lse_grad(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for p in range(j0, j1):
            g[p] += wk * (1.0 - x[p - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for p in range(p0, p1):
            g[p] += wk * (-x[p + s])
    return g


def _slsqp_smooth(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _pnorm_obj(x, p):
    cv = _corr_vec(x)
    cv_pos = np.maximum(cv, 0.0)
    return float(np.sum(cv_pos ** p) ** (1.0 / p))


def _pnorm_grad(x, p):
    cv = _corr_vec(x)
    cv_pos = np.maximum(cv, 0.0)
    base = np.sum(cv_pos ** p)
    if base < 1e-300:
        return np.zeros(len(x))
    norm = base ** (1.0 / p)
    w = p * cv_pos ** (p - 1) / base
    g = np.zeros(len(x))
    n = len(x)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for ii in range(j0, j1):
            g[ii] += wk * (1.0 - x[ii - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for ii in range(p0, p1):
            g[ii] += wk * (-x[ii + s])
    return g * norm


def _slsqp_pnorm(h0, p, maxiter=120):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_pnorm_obj, h0, args=(p,), jac=_pnorm_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _direct_minmax(h0, maxiter=120):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = cv.max() + 0.003
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        return t - _corr_vec(h)

    def _eq(z):
        return z[:-1].sum() - (len(z) - 1) / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _mirror(h):
    return _project(h[::-1].copy())


def _binary_push(h, p=2.0):
    m = h.max()
    if m < 1e-10:
        return h.copy()
    return _project((h / m) ** p)


def _annealing_step(h, cv, i, rng, temperature):
    n = len(h)
    dx = 2.0 / n
    old_val = h[i]
    new_val = max(0.0, min(1.0, old_val + rng.uniform(-0.05, 0.05)))
    delta = new_val - old_val
    if abs(delta) < 1e-12:
        return h, cv, False
    coef = np.zeros(2 * n - 1)
    k1_lo = n - 1 - i
    k1_hi = 2 * n - 2 - i
    if k1_lo >= 0 and k1_hi < 2 * n - 1:
        idx2 = np.arange(k1_lo, k1_hi + 1) - (n - 1 - i)
        coef[k1_lo:k1_hi + 1] = 1.0 - h[idx2]
    k2_lo = i
    k2_hi = i + n - 1
    if k2_lo >= 0 and k2_hi < 2 * n - 1:
        ks = np.arange(k2_lo, k2_hi + 1)
        ps = i - ks + n - 1
        coef[k2_lo:k2_hi + 1] -= h[ps]
    coef *= dx
    old_max = np.max(cv)
    new_cv = cv + delta * coef
    new_max = np.max(new_cv)
    delta_obj = new_max - old_max
    if delta_obj < 0 or rng.uniform() < np.exp(-delta_obj / max(temperature, 1e-10)):
        h[i] = new_val
        cv = new_cv
        return h, cv, True
    return h, cv, False


def _sa_optimize(h0, n_iter=5000, t_start=0.005, t_end=0.0001, seed=0):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    cv = _corr_vec(h)
    best_h = h.copy()
    best_s = np.max(cv)
    t_cur = t_start
    cool = (t_end / t_start) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        i = rng.integers(0, n)
        h, cv, accepted = _annealing_step(h, cv, i, rng, t_cur)
        if accepted:
            cur_max = np.max(cv)
            if cur_max < best_s - 1e-12:
                best_s = cur_max
                best_h = h.copy()
        t_cur *= cool
        if (it + 1) % 500 == 0:
            h_pr = _project(h)
            s_pr = _score(h_pr)
            if s_pr < best_s:
                best_s = s_pr
                best_h = h_pr.copy()
                cv = _corr_vec(h)
    best_h = _project(best_h)
    return best_h, _score(best_h)


def _make_slsqp_helpers(n):
    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    return _efun, _ejac


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(30.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(50) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)

    def _elapsed():
        return time.time() - t_start

    def _update_best(h, s):
        nonlocal best_h, best_score, best_n
        if s < best_score - 1e-13:
            best_h = h.copy()
            best_score = s
            best_n = len(h)

    # Phase 1: Dimension scan with fast coordinate descent
    n_candidates = []
    lo_n = max(30, parent_n - 15)
    hi_n = min(80, parent_n + 20)
    for n in range(lo_n, hi_n + 1):
        n_candidates.append(n)

    dim_results = []
    for n in n_candidates:
        if _elapsed() > budget_s * 0.22:
            break
        h_interp = _interp(parent, n)
        h_cd, s_cd = _coord_descent(h_interp, n_sweeps=80, project_every=10)
        dim_results.append((s_cd, n, h_cd))
        _update_best(h_cd, s_cd)

    dim_results.sort(key=lambda x: x[0])
    top_dims = []
    for s, n, h in dim_results[:6]:
        if n not in top_dims:
            top_dims.append(n)

    # Phase 2: Intensive optimization on top dimensions
    for n in top_dims:
        if _elapsed() > budget_s * 0.55:
            break
        h_base = None
        for s, nn, hh in dim_results:
            if nn == n:
                h_base = hh.copy()
                break
        if h_base is None:
            h_base = _interp(parent, n)

        cur = h_base.copy()
        cur_s = _score(cur)

        # Long coordinate descent
        if _elapsed() < deadline - 30:
            cur, cur_s = _coord_descent(cur, n_sweeps=500, project_every=8)
            _update_best(cur, cur_s)

        # SLSQP with LSE
        if _elapsed() < deadline - 30:
            cur2 = _slsqp_smooth(cur, beta=100, maxiter=150)
            s2 = _score(cur2)
            if s2 < cur_s:
                cur, cur_s = cur2, s2
            _update_best(cur, cur_s)

        # p-norm SLSQP
        if _elapsed() < deadline - 30:
            for p in [8, 16]:
                cur3 = _slsqp_pnorm(cur, p=p, maxiter=100)
                s3 = _score(cur3)
                if s3 < cur_s:
                    cur, cur_s = cur3, s3
            _update_best(cur, cur_s)

        # More coordinate descent
        if _elapsed() < deadline - 25:
            cur, cur_s = _coord_descent(cur, n_sweeps=400, project_every=6)
            _update_best(cur, cur_s)

        # Simulated annealing
        if _elapsed() < deadline - 25:
            cur_sa, cur_sa_s = _sa_optimize(cur, n_iter=3000,
                                            t_start=0.004, t_end=0.0002,
                                            seed=seed + n)
            if cur_sa_s < cur_s:
                cur, cur_s = cur_sa, cur_sa_s
            _update_best(cur, cur_s)

        # Direct min-max
        if _elapsed() < deadline - 20:
            cur_dm = _direct_minmax(cur, maxiter=100)
            s_dm = _score(cur_dm)
            if s_dm < cur_s:
                cur, cur_s = cur_dm, s_dm
            _update_best(cur, cur_s)

        # Final coord descent
        if _elapsed() < deadline - 15:
            cur, cur_s = _coord_descent(cur, n_sweeps=300, project_every=5)
            _update_best(cur, cur_s)

    # Phase 3: Deep optimization on the global best
    cur = best_h.copy()
    cur_s = best_score

    if _elapsed() < deadline - 40:
        cur, cur_s = _coord_descent(cur, n_sweeps=1000, project_every=5)
        _update_best(cur, cur_s)

    if _elapsed() < deadline - 35:
        for beta in [200, 500, 1000]:
            if _elapsed() > deadline - 20:
                break
            cur2 = _slsqp_smooth(cur, beta=beta, maxiter=120)
            s2 = _score(cur2)
            if s2 < cur_s:
                cur, cur_s = cur2, s2
            _update_best(cur, cur_s)
            cur, cur_s = _coord_descent(cur, n_sweeps=200, project_every=5)
            _update_best(cur, cur_s)

    if _elapsed() < deadline - 30:
        for p in [16, 32, 64]:
            if _elapsed() > deadline - 15:
                break
            cur3 = _slsqp_pnorm(cur, p=p, maxiter=100)
            s3 = _score(cur3)
            if s3 < cur_s:
                cur, cur_s = cur3, s3
            _update_best(cur, cur_s)
            cur, cur_s = _coord_descent(cur, n_sweeps=150, project_every=5)
            _update_best(cur, cur_s)

    if _elapsed() < deadline - 25:
        cur_dm = _direct_minmax(cur, maxiter=150)
        s_dm = _score(cur_dm)
        if s_dm < cur_s:
            cur, cur_s = cur_dm, s_dm
        _update_best(cur, cur_s)
        cur, cur_s = _coord_descent(cur, n_sweeps=300, project_every=5)
        _update_best(cur, cur_s)

    # Phase 4: Random restarts around best
    if _elapsed() < deadline - 20:
        n_restarts = min(10, int((deadline - time.time()) / 40))
        for ri in range(n_restarts):
            if _elapsed() > deadline - 10:
                break
            sigma = 0.02 + 0.01 * ri
            h_pert = _project(cur + sigma * rng.standard_normal(len(cur)))
            h_r, s_r = _coord_descent(h_pert, n_sweeps=200, project_every=8)
            if s_r < cur_s:
                cur, cur_s = h_r, s_r
            _update_best(cur, cur_s)
            if _elapsed() < deadline - 15:
                h_sa, s_sa = _sa_optimize(cur, n_iter=2000,
                                          t_start=0.003, t_end=0.0003,
                                          seed=seed + 100 + ri)
                if s_sa < cur_s:
                    cur, cur_s = h_sa, s_sa
                _update_best(cur, cur_s)

    # Phase 5: Final intensive polish
    if _elapsed() < deadline - 10:
        cur, cur_s = _coord_descent_momentum(cur, n_sweeps=500, alpha=0.8,
                                             project_every=5)
        _update_best(cur, cur_s)

    if _elapsed() < deadline - 8:
        cur, cur_s = _coord_descent(cur, n_sweeps=500, project_every=4)
        _update_best(cur, cur_s)

    if _elapsed() < deadline - 5:
        cur2 = _slsqp_smooth(best_h, beta=2000, maxiter=100)
        s2 = _score(cur2)
        if s2 < best_score:
            best_h, best_score = _project(cur2), s2
        cur, cur_s = _coord_descent(best_h, n_sweeps=200, project_every=4)
        _update_best(cur, cur_s)

    # Final
    best_h = _project(best_h)
    final_score = _score(best_h)
    return (best_h, final_score, len(best_h))
```
