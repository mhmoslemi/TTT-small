def gradient_for_shift(h, s):
    """
    Gradient of f_s(h) = sum_{valid i} h[i]*(1-h[i+s]) w.r.t. h.
    s is the shift in index space (can be negative).
    """
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # f = sum_{i=0}^{n-s-1} h[i]*(1-h[i+s])
        for j in range(n):
            if j < n - s:
                g[j] += (1.0 - h[j + s])
            if j >= s:
                g[j] -= h[j - s]
    else:
        t = -s
        for j in range(n):
            if j >= t:
                g[j] += (1.0 - h[j - t])
            if j < n - t:
                g[j] -= h[j + t]
    return g

def make_feasible(z):
    """Project z onto the feasible set: 0<=h<=1, sum(h)=n/2."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    # Bisection to find offset
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def qp_step(h0, k_active, reg, solver_list=None):
    """
    One active-set QP step.
    h0: current feasible point
    k_active: number of top constraints to include
    reg: regularization parameter
    """
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = gradient_for_shift(h0, s)
        lhs = vals[p] + g @ (h_var - h0)
        constraints.append(lhs <= t_var)
    
    obj = cp.Minimize(t_var + (reg / 2.0) * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)

def active_set_optimize(h0, max_iter, k_active, reg_start, reg_end, deadline):
    """Run active-set QP with decreasing regularization."""
    h = h0.copy()
    best = official_score(h)
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        reg = reg_start * (reg_end / reg_start) ** (it / max(1, max_iter - 1))
        h_new = qp_step(h, k_active=k_active, reg=reg)
        if h_new is None:
            break
        sc = official_score(h_new)
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.8, 0.5, 0.3, 0.15]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-14:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 6:
                    break
                # Try with larger K
                h_new2 = qp_step(h, k_active=min(k_active + 3, len(h) * 2 - 1), reg=reg * 0.5)
                if h_new2 is not None:
                    sc2 = official_score(h_new2)
                    if sc2 < best - 1e-14:
                        h = h_new2
                        best = sc2
                        stall = 0
    return h, best

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals_raw = np.correlate(h, 1.0 - h, mode='full')
    vals = vals_raw * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient of smooth max
    g = np.zeros(n)
    for idx, w in enumerate(weights):
        if w > 1e-10:
            s = idx - (n - 1)
            g += w * gradient_for_shift(h, s) * dx
    return g

def try_slsqp(h0, beta, maxiter, deadline):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
                       'jac': lambda h: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h0.copy(), args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def anti_sym(h):
    return make_feasible(1.0 - h[::-1].copy())
