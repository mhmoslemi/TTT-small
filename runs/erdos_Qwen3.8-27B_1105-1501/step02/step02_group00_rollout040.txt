def _subgrad_step(h, step_size, n_top=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top lags
    top_idx = np.argsort(corr)[-n_top:]
    # Average gradient of top lags
    grad = np.zeros(n)
    for idx in top_idx:
        d = idx - (n - 1)  # actual lag
        if d >= 0:
            m = n - d
            grad[d:] += (1.0 - h[:m])
            grad[:m] += -h[d:]
        else:
            g = -d
            m = n - g
            grad[:m] += (1.0 - h[g:])
            grad[g:] += -h[:m]
    grad /= n_top
    # Negative subgradient direction
    direction = -grad
    direction -= np.mean(direction)  # maintain sum constraint approximately
    norm = np.linalg.norm(direction)
    if norm < 1e-15:
        return h, _exact_c5(h)
    direction /= norm
    h_new = h + step_size * direction
    h_new = _project(h_new, n/2.0)
    return h_new, _exact_c5(h_new)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_at_lag(h, lag_idx):
    n = len(h)
    d = lag_idx - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        gmag = -d
        m = n - gmag
        if m > 0:
            g[:m] += (1.0 - h[gmag:])
            g[gmag:] += -h[:m]
    return g

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, p, c5_cur):

            best_a = a

        gr = (np.sqrt(5.0) - 1.0) / 2.0

        c2 = lo_a + gr * (hi_a - lo_a)

            if f1 < f2:

            else:

        if opt_c5 < best_c5:

            best_c5 = opt_c5

    n = len(h_frac)
    target = int(n // 2)
    
    # Initialize from fractional
    h = (h_frac > 0.5).astype(float)
    # Fix cardinality
    ones = int(np.sum(h))
    if ones > target:
        order = np.argsort(h_frac)
        removed = 0
        for i in order:
            if h[i] == 1 and removed < ones - target:
                h[i] = 0
                removed += 1
    elif ones < target:
        order = np.argsort(-h_frac)
        added = 0
        for i in order:
            if h[i] == 0 and added < target - ones:
                h[i] = 1
                added += 1

    best_h = h.copy()
    best_c5 = cur_c5
    
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]

    decay = (T_min / T_init) ** (1.0 / max(max_iters, 1))
    accepted = 0
    total = 0

        if time.time() > deadline:
            break
        
        i = one_idx[np.random.randint(len(one_idx))]
        j = zero_idx[np.random.randint(len(zero_idx))]
        if i == j:

        total += 1

            accepted += 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            # Update indices
            one_idx = np.where(h == 1)[0]
            zero_idx = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]  # Revert
        
        T *= decay
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_at_lag(h, lag_idx):
    n = len(h)
    d = lag_idx - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        gmag = -d
        m = n - gmag
        if m > 0:
            g[:m] += (1.0 - h[gmag:])
            g[gmag:] += -h[:m]
    return g

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.05, 1.0, 12) * alpha_max
    for a in alphas:
        if a > alpha_max + 1e-12:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    if best_a > 0:
        lo_a = max(0.0, best_a * 0.7)
        hi_a = min(alpha_max, best_a * 1.3)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(12):
            if hi_a - lo_a < 1e-11:
                break
            if f1 < f2:
                hi_a = c2
                c2 = c1
                f2 = f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo_a = c1
                c1 = c2
                f1 = f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(h + c2 * p)
        opt_a = (lo_a + hi_a) / 2.0
        opt_c5 = _exact_c5(h + opt_a * p)
        if opt_c5 < best_c5:
            best_a = opt_a
            best_c5 = opt_c5
    return best_a, best_c5

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=5000, n_top=1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    last_alpha = 0.1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        corr = _corr_values(h)
        top_idx = np.argsort(corr)[-max(n_top, 1):]
        scores = corr[top_idx]
        G = np.zeros((len(top_idx), n))
        for i, idx in enumerate(top_idx):
            G[i] = _grad_at_lag(h, idx)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((len(top_idx), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = np.zeros(len(top_idx))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-radius, radius)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            p = res.x[:n] if res.success else None
        except Exception:
            p = None
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.9
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-10:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            last_alpha = alpha
            radius = min(max(radius * 1.08, alpha * 1.5), 0.5)
        else:
            stall += 1
            radius *= 0.88
            if stall == 15:
                radius = min(radius * 2.5, 0.4)
            if radius < 1e-5:
                break
    return best_h, best_c5

def _binary_sa(h_frac, deadline, max_iters=150000, T_init=0.02, T_min=0.001):
    n = len(h_frac)
    target = int(n // 2)
    h = (h_frac > 0.5).astype(float)
    ones = int(np.sum(h))
    if ones > target:
        order = np.argsort(h_frac)
        removed = 0
        for i in order:
            if h[i] == 1 and removed < ones - target:
                h[i] = 0
                removed += 1
    elif ones < target:
        order = np.argsort(-h_frac)
        added = 0
        for i in order:
            if h[i] == 0 and added < target - ones:
                h[i] = 1
                added += 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]
    T = T_init
    decay = (T_min / T_init) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = one_idx[np.random.randint(len(one_idx))]
        j = zero_idx[np.random.randint(len(zero_idx))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < -1e-15 or (T > T_min and np.random.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
            one_idx = np.where(h == 1)[0]
            zero_idx = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
        T *= decay
    return best_h, best_c5

def _binary_greedy(h_bin, deadline, max_passes=20):
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

        if time.time() > deadline:

        improved = False
        one_idx = np.where(h == 1)[0]

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, seed_val):
    rng_local = np.random.default_rng(seed_val)
    h = np.zeros(n)
    idx = rng_local.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def _subgrad_momentum(h, deadline, max_iters=20000, n_top=3, mom=0.9):

    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.05
    v = np.zeros(n)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        corr = _corr_values(h)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for idx in top_idx:
            grad += _grad_at_lag(h, idx)
        grad /= n_top
        grad -= np.mean(grad)
        v = mom * v - step * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 0.3:
            v *= 0.3 / v_norm
        h_new = h + v
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step = min(step * 1.02, 0.2)
        else:
            step *= 0.92
            stall += 1
    return best_h, best_c5

        h_r, c5_r = _lp_optimize(h_r, deadline_sub, radius_init=0.12, max_iters=3000, n_top=1)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Also try with top-3
            deadline_sub2 = min(d5, time.time() + budget_s * 0.03)
            h_r2, c5_r2 = _lp_optimize(h_r, deadline_sub2, radius_init=0.08, max_iters=2000, n_top=3)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()

    # Phase 6: Subgradient with momentum at best resolution (7%)
    d6 = t0 + budget_s * 0.92
    if best_n <= 160:
        hs, c5_s = _subgrad_momentum(best_h, d6, max_iters=15000, n_top=3)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = hs.copy()

    # Phase 7: Final LP polish (8%)
    d7 = t0 + budget_s * 0.995
    h_f, c5_f = _lp_optimize(best_h, d7, radius_init=0.04, max_iters=3000, n_top=2)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
