                best_s = sa
                best_h = ha.copy()
    ha = _project(avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s

def _active_slsqp(h0, n_active=12, maxiter=120, extra_margin=0.008):
    n = len(h0)
    cv = _corr_full(h0)
    current_max = cv.max()
    
    # Find active lags
    order = np.argsort(cv)[::-1]
    active_idx = set(order[:n_active].tolist())
    for i in range(len(cv)):
        if cv[i] > current_max - extra_margin:
            active_idx.add(int(i))
            if len(active_idx) >= n_active + 6:
                break
    active_lags = sorted([int(i - (n - 1)) for i in active_idx])
    
    t0 = min(current_max + 0.002, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def _fun(z):
        return z[-1]
    
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def _ineq(z):
        h = z[:-1]
        t = z[-1]
        out = np.empty(len(active_lags))
        for ii, s in enumerate(active_lags):
            out[ii] = t - _corr_lag_val(s, h)
        return out
    
    def _ineq_jac(z):
        h = z[:-1]
        m = len(active_lags)
        J = np.zeros((m, len(z)))
        J[:, -1] = 1.0
        for ii, s in enumerate(active_lags):
            g = _corr_grad_single(s, h)
            J[ii, :n] = -g
        return J
    
    def _eq(z):
        return z[:-1].sum() - n / 2.0
    
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:n] = 1.0
        return g
    
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP', bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _corr_lag_val(s, h):
    n = len(h)
    if s == 0:
        return float(np.sum(h * (1.0 - h)))
    elif s > 0:
        if s >= n:
            return 0.0
        return float(np.sum(h[s:] * (1.0 - h[:-s])))
    else:
        sp = -s
        if sp >= n:
            return 0.0
        return float(np.sum(h[:-sp] * (1.0 - h[sp:])))

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_balanced(n, rng, method='random'):
    if n % 2 != 0:
        return _project(np.ones(n) * 0.5)
    target = n // 2
    if method == 'random':
        pos = rng.choice(n, size=target, replace=False)
        h = np.zeros(n)
        h[pos] = 1.0
    elif method == 'golden':
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        pos = np.zeros(target, dtype=int)
        for i in range(target):
            pos[i] = int((i * gr) % n)
        pos = np.unique(pos)
        if len(pos) < target:
            remaining = np.setdiff1d(np.arange(n), pos)
            extra = rng.choice(remaining, size=target - len(pos), replace=False)
            pos = np.concatenate([pos, extra])
        pos = pos[:target]
        h = np.zeros(n)
        h[pos] = 1.0
    elif method == 'alt':
        h = np.zeros(n)
        h[::2] = 1.0
        if h.sum() != target:
            h[target] = 1.0
            h[n//2] = 0.0
    elif method == 'block':
        h = np.zeros(n)
        block = max(2, n // 10)
        for i in range(0, n, block * 2):
            h[i:i+block] = 1.0
        # Fix count
        cur = h.sum()
        if cur < target:
            h = h.copy()
            for i in range(int(cur), target):
                if h[i] < 1:
                    h[i] = 1.0
        elif cur > target:
            h = h.copy()
            for i in range(int(cur) - 1, target - 1, -1):
                h[i] = 0.0
        h = np.round(h)
        # Ensure exact count
        deficit = target - int(h.sum())
        if deficit > 0:
            zeros = np.where(h == 0)[0]
            for i in range(min(deficit, len(zeros))):
                h[zeros[i]] = 1.0
        elif deficit < 0:
            ones = np.where(h == 1)[0]
            for i in range(min(-deficit, len(ones))):
                h[ones[i]] = 0.0
    else:
        h = np.zeros(n)
        h[:target] = 1.0
    return _project(h)

def _sinusoidal_start(n, rng, freq1, freq2, amp1, amp2, phase1, phase2):
    x = np.arange(n) / n
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase1) + amp2 * np.sin(2 * np.pi * freq2 * x + phase2)
    return _project(h)

def _smooth_binary(h_bin, kernel_width=3):
    if kernel_width <= 1:
        return h_bin.copy()
    k = np.array([1, 4, 6, 4, 1], dtype=float)
    k = k / k.sum()
    h_s = np.convolve(h_bin, k, mode='same')
    return _project(h_s)

def _chain_optimize(h0, deadline, rng, n_active=12):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)
    
    # Stage 1: LSE with moderate beta
    if time.time() < deadline - 20:
        new, ns = _fista(cur, beta=50, n_iter=3000, alpha=0.005, mu=0.85, deadline=deadline)
        if ns < cur_s:
            cur, cur_s = new, ns
    
    # Stage 2: Active-set SLSQP
    if time.time() < deadline - 15:
        new = _active_slsqp(cur, n_active=n_active, maxiter=100)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns
    
    # Stage 3: LSE with higher beta
    if time.time() < deadline - 15:
        new, ns = _fista(cur, beta=300, n_iter=5000, alpha=0.003, mu=0.88, deadline=deadline)
        if ns < cur_s:
            cur, cur_s = new, ns
    
    # Stage 4: Active-set SLSQP again
    if time.time() < deadline - 12:
        new = _active_slsqp(cur, n_active=n_active + 4, maxiter=120, extra_margin=0.005)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns
    
    # Stage 5: Final LSE
    if time.time() < deadline - 10:
        new, ns = _fista(cur, beta=1000, n_iter=4000, alpha=0.002, mu=0.9, deadline=deadline)
        if ns < cur_s:
            cur, cur_s = new, ns
    
    # Stage 6: Final SLSQP
    if time.time() < deadline - 8:
        new = _active_slsqp(cur, n_active=n_active + 6, maxiter=150, extra_margin=0.003)
        ns = _score(new)
        if ns < cur_s - 1e-13:
            cur, cur_s = new, ns
    
    return cur, cur_s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(30.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(56) * 0.5)
    
    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    pn = len(parent)
    
    # Phase 1: Screen n values
    n_range = list(range(max(20, pn - 15), min(76, pn + 15) + 1, 2))
    if pn not in n_range:
        n_range.append(pn)
    n_range.append(pn - 1 if pn - 1 >= 20 else pn + 1)
    n_range = sorted(set(n_range))
    
    screen_results = []
    for n in n_range:
        if time.time() > deadline - 120:
            break
        h_i = _interp(parent, n)
        s_i = _score(h_i)
        # Quick LSE
        if time.time() < deadline - 80:
            h_q, s_q = _fista(h_i, beta=80, n_iter=2000, alpha=0.005, mu=0.85, deadline=deadline)
            if s_q < s_i:
                screen_results.append((s_q, n, h_q))
            else:
                screen_results.append((s_i, n, h_i))
        else:
            screen_results.append((s_i, n, h_i))
        if s_i < best_score:
            best_h = h_i.copy()
            best_score = s_i
            best_n = n
    
    screen_results.sort(key=lambda x: x[0])
    
    # Phase 2: Generate diverse starting points for top n values
    top_ns = []
    for s, n, h in screen_results[:5]:
        if n not in top_ns:
            top_ns.append(n)
    
    all_starts = []
    for n in top_ns:
        if time.time() > deadline - 60:
            break
        # Get best for this n
        h_base = None
        for s, nn, hh in screen_results:
            if nn == n:
                h_base = hh.copy()
                break
        if h_base is None:
            h_base = _interp(parent, n)
        
        # Continuous perturbations
        for sigma in [0.05, 0.1, 0.2]:
            h_p = _project(h_base + sigma * rng.standard_normal(n))
            all_starts.append((n, h_p, _score(h_p)))
        
        # Mirror
        h_m = _project(h_base[::-1].copy())
        all_starts.append((n, h_m, _score(h_m)))
        
        # Binary starts (only for even n)
        if n % 2 == 0:
            for method in ['random', 'golden', 'block']:
                h_b = _binary_balanced(n, rng, method)
                # Smooth version
                h_bs = _smooth_binary(h_b)
                all_starts.append((n, h_b, _score(h_b)))
                all_starts.append((n, h_bs, _score(h_bs)))
        
        # Sinusoidal starts
        for freq in [1, 2, 3]:
            for amp in [0.3, 0.4, 0.5]:
                ph = rng.uniform(0, 2 * np.pi)
                h_s = _sinusoidal_start(n, rng, freq, 2*freq, amp, amp*0.5, ph, rng.uniform(0, 2*np.pi))
                all_starts.append((n, h_s, _score(h_s)))
    
    all_starts.sort(key=lambda x: x[2])
    
    # Phase 3: Optimize top diverse starts
    n_optimized = 0
    for n, h_start, s_start in all_starts:
        if time.time() > deadline - 40:
            break
        n_optimized += 1
        if n_optimized > 12:
            break
        if s_start >= best_score + 0.01:
            continue
        cur, cur_s = _chain_optimize(h_start, deadline, rng, n_active=12)
        if cur_s < best_score - 1e-12:
            best_h = cur.copy()
            best_score = cur_s
            best_n = len(cur)
    
    # Phase 4: Intensive refinement on best
    remaining = deadline - time.time()
    if remaining > 30:
        # Multi-beta LSE on best
        for beta in [80, 300, 800, 2000]:
            if time.time() > deadline - 10:
                break
            iters = min(8000, int(remaining / 5))
            best_h, best_score = _fista(best_h, beta=beta, n_iter=iters,
                                         alpha=0.003, mu=0.88, deadline=deadline)
            remaining = deadline - time.time()
    
    if time.time() < deadline - 20:
        # Active SLSQP with increasing active set
        for na in [8, 14, 20]:
            if time.time() > deadline - 10:
                break
            new = _active_slsqp(best_h, n_active=na, maxiter=150, extra_margin=0.004)
            ns = _score(new)
            if ns < best_score - 1e-13:
                best_h = new
                best_score = ns
    
    if time.time() < deadline - 15:
        # Another LSE pass
        best_h, best_score = _fista(best_h, beta=1500, n_iter=5000,
                                     alpha=0.0015, mu=0.92, deadline=deadline)
    
    # Phase 5: Random restarts
    remaining = deadline - time.time()
    n_restarts = min(8, int(remaining / 20))
    for _ in range(n_restarts):
        if time.time() > deadline - 10:
            break
        h_p = _project(best_h + 0.04 * rng.standard_normal(best_n))
        cur, cur_s = _chain_optimize(h_p, deadline, rng, n_active=14)
        if cur_s < best_score - 1e-12:
            best_h = cur.copy()
            best_score = cur_s
            best_n = len(cur)
    
    # Phase 6: Try slightly different n near best
    if time.time() < deadline - 25:
        for dn in [-1, 1, -2, 2]:
            if time.time() > deadline - 10:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 80:
                continue
            h_try = _interp(best_h, n_try)
            cur, cur_s = _chain_optimize(h_try, deadline, rng, n_active=12)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)
    
    # Final polish
    if time.time() < deadline - 5:
        new = _active_slsqp(best_h, n_active=16, maxiter=200, extra_margin=0.002)
        ns = _score(new)
        if ns < best_score - 1e-13:
            best_h = new
            best_score = ns
    
    # Final LSE micro-refinement
    if time.time() < deadline - 3:
        best_h, best_score = _fista(best_h, beta=3000, n_iter=2000,
                                     alpha=0.001, mu=0.95, deadline=deadline)
    
    # Final feasibility check and evaluation
    best_h = _project(best_h)
    final_score = _score(best_h)
    
    return (best_h, final_score, len(best_h))
