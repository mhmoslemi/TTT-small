def grad_val_single(h, s):
    """Compute dv_p/dh for shift s (where p = n-1+s)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        s2 = -s
        g[s2:] += (1.0 - h[:n-s2])
        g[:n-s2] -= h[s2:]
    return dx * g

def smooth_grad_antisym(u, n, beta):
    """Compute gradient of smoothed max wrt u in anti-symmetric space."""
    m = n // 2
    h = build_antisym(u, n)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    Z = w.sum()
    if Z < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= Z
    
    # Compute gradient: sum_p w[p] * ∂v_p/∂u_j
    n = len(h)
    dx = 2.0 / n
    grad_h = np.zeros(n)  # weighted sum of ∂v_p/∂h_j
    for p in range(2*n-1):
        s = p - (n-1)
        g = grad_val_single(h, s)  # this is ∂v_p/∂h (already includes dx)
        grad_h += w[p] * g
    
    # Reduce to u-space: ∂v/∂u_j = ∂v/∂h_j - ∂v/∂h_{n-1-j}
    grad_u = grad_h[:m] - grad_h[n-1-:m-1:-1]  # grad_h[n-1], grad_h[n-2], ..., grad_h[n-m]
    # Wait, n-1-j for j=0,...,m-1 gives n-1, n-2, ..., n-m
    # That's grad_h[m:] reversed for even n
    if n % 2 == 0:

    else:
        # n = 2m+1, center at m
        grad_u = grad_h[:m] - grad_h[m+1:][::-1]
        # Note: grad_h[m] doesn't contribute to any u_j since h[m]=0.5 is fixed
    
    return grad_u

def official_score(h):
    n = len(h)

    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_vals(h):
    n = len(h)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def build_antisym(u, n):
    h = np.zeros(n)
    m = n // 2
    uc = np.clip(u, 0.0, 1.0)[:m]
    h[:m] = uc
    if n % 2 == 0:
        h[m:] = 1.0 - uc[::-1]
    else:
        h[m] = 0.5
        h[m+1:] = 1.0 - uc[::-1]

def grad_val_single_dx(h, s):
    """Compute dv_p/dh for shift s, including dx factor."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        s2 = -s
        g[s2:] += (1.0 - h[:n-s2])
        g[:n-s2] -= h[s2:]
    return dx * g

def smooth_obj_grad_antisym(u, n, beta):
    """Compute smoothed objective and gradient in anti-symmetric space."""
    m = n // 2
    h = build_antisym(u, n)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50.0, 50.0)
    w = np.exp(shifted)
    Z = w.sum()
    if Z < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= Z
    obj = cmax + np.log(Z) / beta  # log-sum-exp approximation
    
    # Gradient
    grad_h = np.zeros(n)
    for p in range(2 * n - 1):
        s = p - (n - 1)
        g_p = grad_val_single_dx(h, s)
        if w[p] > 1e-15:
            grad_h += w[p] * g_p
    
    # Reduce to u-space
    if n % 2 == 0:
        grad_u = grad_h[:m] - grad_h[m:][::-1]
    else:
        grad_u = grad_h[:m] - grad_h[m+1:][::-1]
    
    return obj, grad_u

def pg_antisym(u0, n, beta, time_limit=20.0, max_iter=5000):
    """Projected gradient in anti-symmetric space."""
    m = n // 2
    u = np.clip(u0[:m].copy(), 0.0, 1.0)
    f, g = smooth_obj_grad_antisym(u, n, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-30:
        return u
    alpha = 1.0 / (gnorm2 + 1e-12)
    alpha = min(alpha, 1.0 / (np.max(g**2) + 1e-12))
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-28:
            break
        step = alpha
        for _ in range(20):
            u_trial = np.clip(u - step * g, 0.0, 1.0)
            f_trial, _ = smooth_obj_grad_antisym(u_trial, n, beta)
            if f_trial <= f - 1e-4 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-14:
            break
        u = np.clip(u - step * g, 0.0, 1.0)
        f, g = smooth_obj_grad_antisym(u, n, beta)
        if f < f:  # This won't trigger, but keep for structure
            alpha = min(alpha * 1.5, 2.0 / (gnorm2 + 1e-12))
    return u
