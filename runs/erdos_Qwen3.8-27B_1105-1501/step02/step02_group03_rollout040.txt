def make_feasible(h):

    h = np.clip(h.copy(), 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) < 1e-12:
        return h
    # Iterative redistribution
    for _ in range(200):
        if abs(residual) < 1e-14:
            break
        if residual > 0:

            room = 1.0 - h
            order = np.argsort(-room)

                    residual -= add

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    residual = target - np.sum(h)
    for _ in range(300):
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                if residual <= 0:
                    break
                add = min(residual, room[idx])
                if add > 0:
                    h[idx] += add
                    residual -= add
        else:
            order = np.argsort(h)
            for idx in order:
                if residual >= 0:
                    break
                sub = min(-residual, h[idx])
                if sub > 0:
                    h[idx] -= sub
                    residual += sub
    return np.clip(h, 0.0, 1.0)

def project_antisym(h):
    n = len(h)
    return 0.5 * (h + 1.0 - h[::-1].copy())

def build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        L = n - s
        G[n - 1 + s, s:] += dx * (1.0 - h[:L])
        G[n - 1 + s, :L] -= dx * h[s:s + L]
        G[n - 1 - s, :L] += dx * (1.0 - h[s:s + L])
        G[n - 1 - s, s:] -= dx * h[:L]
    return G

def sigmoid(x):
    x = np.clip(x, -500, 500)
    return 1.0 / (1.0 + np.exp(-x))

def z_to_y(z, alpha):
    return sigmoid(alpha * np.asarray(z, dtype=float))

def y_to_antisym_h(y, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = y
    for i in range(m):
        h[n - 1 - i] = 1.0 - y[i]
    return h

def h_to_z_antisym(h, alpha):
    n = len(h)
    m = n // 2
    y = h[:m]
    y = np.clip(y, 1e-10, 1.0 - 1e-10)
    return np.log(y / (1.0 - y)) / alpha

def vals_grad_z(z, alpha, n):
    m = n // 2
    y = z_to_y(z, alpha)
    h = y_to_antisym_h(y, n)
    vals = compute_vals(h)
    G = build_G(h)
    sy = alpha * y * (1.0 - y)
    Gz = np.empty((2 * n - 1, m))
    for j in range(m):
        Gz[:, j] = sy[j] * (G[:, j] - G[:, n - 1 - j])
    return vals, Gz

def smooth_obj_z(z, alpha, n, beta):
    vals, Gz = vals_grad_z(z, alpha, n)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    grad = w @ Gz
    return obj, grad

def lse_opt_z(z0, alpha, n, beta, time_limit=30.0, max_iter=8000):
    z = z0.copy()
    f, g = smooth_obj_z(z, alpha, n, beta)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-30:
        return z
    step = min(1.0 / (gnorm2 + 1e-12), 2.0 / (np.max(g**2) + 1e-12))
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        f, g = smooth_obj_z(z, alpha, n, beta)
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-30:
            break
        s = step
        z_new = z
        for _ in range(40):
            z_new = z - s * g
            f_new = smooth_obj_z(z_new, alpha, n, beta)[0]
            if f_new <= f - 1e-5 * s * gnorm2:
                break
            s *= 0.4
        if s < 1e-15:
            break
        z = z_new
        f = f_new
        if f < f + 1e-20:
            step = min(step * 1.6, 5.0 / (gnorm2 + 1e-12))
    return z

def slsqp_antisym(z0, alpha, n, t_init, maxiter=400, active_idx=None):
    m = n // 2
    def _obj(zf):
        return zf[-1]
    def _obj_grad(zf):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    def _con_fun(zf):
        z = zf[:m]
        t = zf[-1]
        vals = compute_vals_z(z, alpha, n)
        if active_idx is not None:
            return t - vals[active_idx]
        return t - vals
    def _con_jac(zf):
        z = zf[:m]
        _, Gz = vals_grad_z(z, alpha, n)
        if active_idx is not None:
            J = np.zeros((len(active_idx), m + 1))
            J[:, :m] = -Gz[active_idx, :]
            J[:, m] = 1.0
        else:
            J = np.zeros((2 * n - 1, m + 1))
            J[:, :m] = -Gz
            J[:, m] = 1.0
        return J
    z_init = np.concatenate([z0, [t_init]])
    bounds = [(-20, 20)] * m + [(0.01, 2.0)]
    try:
        res = minimize(_obj, z_init, jac=_obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints={'type': 'ineq', 'fun': _con_fun, 'jac': _con_jac},
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            z_new = res.x[:m]
            y = z_to_y(z_new, alpha)
            h = y_to_antisym_h(y, n)
            h = np.clip(h, 0, 1)
            sc = official_score(h)
            if np.isfinite(sc):
                return h, sc
    except Exception:
        pass
    return None, None

def slsqp_full(h0, t_init, maxiter=400, active_idx=None):
    n = len(h0)
    def _obj(zf):
        return zf[-1]
    def _obj_grad(zf):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def _eq_fun(zf):
        return np.sum(zf[:n]) - n / 2.0
    def _eq_jac(zf):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    def _ineq_fun(zf):
        h = zf[:n]
        t = zf[-1]
        vals = compute_vals(h)
        if active_idx is not None:
            return t - vals[active_idx]
        return t - vals
    def _ineq_jac(zf):
        h = zf[:n]
        G = build_G(h)
        if active_idx is not None:
            J = np.zeros((len(active_idx), n + 1))
            J[:, :n] = -G[active_idx, :]
            J[:, n] = 1.0
        else:
            J = np.zeros((2 * n - 1, n + 1))
            J[:, :n] = -G
            J[:, n] = 1.0
        return J
    z_init = np.concatenate([h0, [t_init]])
    bounds = [(0, 1)] * n + [(0.01, 2.0)]
    try:
        res = minimize(_obj, z_init, jac=_obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
                           {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def find_active_idx(vals, frac=0.3, min_count=10):
    cmax = vals.max()
    threshold = cmax - max(0.02 * cmax, 5e-6)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top.tolist())
    return sorted(active)
