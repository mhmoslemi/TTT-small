def _project_box_mean(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    # Binary search for threshold
    lo, hi = -2.0, 2.0
    for _ in range(100):

            for j in range(0, n-k):
                grad[j] += (-h[j+k])
        elif k < 0:
            g = -k
            # c[-g] = sum_{n=g}^{n-1} h[n-g] * (1-h[n])
            # ∂/∂h[j] = (1-h[j+g]) for j in [0, n-1-g] + (-h[j-g]) for j in [g, n-1]
            for j in range(0, n-g):
                grad[j] += (1.0 - h[j+g])
            for j in range(g, n):
                grad[j] += (-h[j-g])
        # k=0: c[0] = sum h[j]*(1-h[j]), grad = 1-2*h[j] (but this is 0 for binary)
        else:
            grad += (1.0 - 2.0 * h)
    return dx * grad

    """Find lags where f_k is within tol of the max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    active = []
    for idx in range(2*n-1):
        lag = idx - (n-1)
        if corr[idx] * dx >= c5 - tol:
            active.append(lag)
    return active

    return -d / norm, corr

    """Log-sum-exp gradient descent with adaptive beta."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = beta_init
    alpha = 0.5
    patience = 0
    n_iter = 0
    
    while time.time() < deadline and n_iter < 20000:
        if patience > 500:
            break
        n_iter += 1
        
        direction, corr = _lse_direction(h, beta)
        if direction is None:
            break
        
        # Try several step sizes
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05]:
            h_try = _project_box_mean(h + a * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
            elif c5_try < _exact_c5(h) - 1e-13:
                h = h_try
                improved = True
                break
        
        if improved:
            patience = 0
            alpha = min(alpha * 1.1, 2.0)
            beta = min(beta * 1.01, beta_max)
        else:
            patience += 1
            alpha *= 0.8
            if alpha < 0.02:
                beta = max(beta * 0.9, beta_init)
                alpha = 0.5
    
    return best_h, best_c5
