def project_feasible(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    # Bisection on threshold
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual among non-saturated coordinates
        if residual > 0:
            candidates = np.where(result < 1.0 - 1e-10)[0]
        else:
            candidates = np.where(result > 1e-10)[0]
        if len(candidates) > 0:
            per = residual / len(candidates)
            result[candidates] += per
            result = np.clip(result, 0.0, 1.0)
    return result

        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            if i > j:
                i, j = j, i
            pairs.append((i, j))
            if max_pairs and len(pairs) >= max_pairs:
                break
        if max_pairs and len(pairs) >= max_pairs:
            break
    for i, j in pairs:
        d = pair_line_search(h, i, j, n_iter=20)
        if abs(d) > 1e-12:
            h[i] -= d
            h[j] += d
    return h


def random_perturb_sweep(h, rng, scale, n_iter):
    n = len(h)
    h = h + rng.randn(n) * scale
    h = project_feasible(h, n / 2.0)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = pair_line_search(h, i, j, n_iter=15)
        if abs(d) > 1e-12:
            h[i] -= d
            h[j] += d
    return h


def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_golden(n, rng):
    h = np.zeros(n)
    phi = 0.6180339887
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 - frac
    h = project_feasible(h, n / 2.0)
    return h


def make_sine(n, rng):
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.randint(3, 12)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.3 * np.sin(freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n / 2.0)
    return h


def interpolate_solution(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    h_new = project_feasible(h_new, n_new / 2.0)
    return h_new


def anneal_slsqp(h_start, n, T_list, deadline):
    h_cur = h_start.copy()
    c_cur = compute_c5(h_cur)
    for T in T_list:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_smooth(h_cur, n, T, maxiter=60)
        h_new = project_feasible(h_new, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < c_cur - 1e-10:
            h_cur = h_new
            c_cur = c_new
    return h_cur, c_cur


def basin_hopping_step(h_best, c_best, n, rng, deadline, T_list_fine):
    n_trial = 3
    for _ in range(n_trial):
        if time.time() > deadline - 10:
            break
        scale = rng.uniform(0.02, 0.08)
        h_trial = h_best + rng.randn(n) * scale
        h_trial = project_feasible(h_trial, n / 2.0)
        c_trial = compute_c5(h_trial)
        if c_trial < c_best + 0.002:
            h_trial, c_trial = anneal_slsqp(h_trial, n, T_list_fine, deadline)
            if c_trial < c_best:
                h_best = h_trial
                c_best = c_trial
    return h_best, c_best


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 4

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h_init = np.full(70, 0.5)

    n_init = len(h_init)
    if n_init < 10:
        h_init = np.full(70, 0.5)
        n_init = 70
    h_init = project_feasible(h_init, n_init / 2.0)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Verify gradient once
    grad_ok = verify_gradient(best_h, 0.5, n_init)
    if not grad_ok:
        pass  # SLSQP will still work, just slower

    # Phase 1: SLSQP annealing on parent length
    T_list_coarse = [4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.008, 0.003]
    phase1_end = t0 + budget_s * 0.25
    if time.time() < phase1_end:
        h_cur = best_h.copy()
        c_cur = compute_c5(h_cur)
        for T in T_list_coarse:
            if time.time() > phase1_end - 3:
                break
            h_new = slsqp_smooth(h_cur, n_init, T, maxiter=80)
            h_new = project_feasible(h_new, n_init / 2.0)
            c_new = compute_c5(h_new)
            if c_new < c_cur - 1e-10:
                h_cur = h_new
                c_cur = c_new
        if c_cur < best_c5:
            best_c5 = c_cur
            best_h = h_cur.copy()

    # Phase 2: Basin hopping + SLSQP on parent length
    T_list_fine = [1.0, 0.5, 0.2, 0.1, 0.04, 0.015]
    phase2_end = t0 + budget_s * 0.50
    hop_count = 0
    while time.time() < phase2_end - 10:
        best_h, best_c5 = basin_hopping_step(best_h, best_c5, n_init, rng, phase2_end, T_list_fine)
        hop_count += 1
        if hop_count >= 25:
            break

    # Phase 3: Try different lengths
    lengths_to_try = []
    for n_try in [40, 50, 60, 80, 90, 100, 120]:
        if n_try != n_init:
            lengths_to_try.append(n_try)
    lengths_to_try.sort(key=lambda x: abs(x - n_init))

    phase3_end = t0 + budget_s * 0.75
    for n_try in lengths_to_try:
        if time.time() > phase3_end - 15:
            break
        # Interpolate best solution to new length
        h_try = interpolate_solution(best_h, n_try)
        c_try = compute_c5(h_try)

        # Also try fresh candidates
        candidates = [h_try]
        candidates.append(make_golden(n_try, rng))
        candidates.append(make_sine(n_try, rng))
        candidates.append(make_random_binary(n_try, rng))

        best_local_h = None
        best_local_c5 = float('inf')
        for h_c in candidates:
            h_c = project_feasible(h_c, n_try / 2.0)
            c_c = compute_c5(h_c)
            if c_c < best_local_c5:
                best_local_c5 = c_c
                best_local_h = h_c.copy()

        if best_local_h is None:
            continue

        # SLSQP annealing on this length
        T_list_short = [2.0, 1.0, 0.5, 0.2, 0.08, 0.03, 0.01]
        h_local = best_local_h.copy()
        c_local = best_local_c5
        for T in T_list_short:
            if time.time() > phase3_end - 10:
                break
            h_new = slsqp_smooth(h_local, n_try, T, maxiter=50)
            h_new = project_feasible(h_new, n_try / 2.0)
            c_new = compute_c5(h_new)
            if c_new < c_local - 1e-10:
                h_local = h_new
                c_local = c_new

        # Quick pair polish
        if time.time() < phase3_end - 10:
            h_local = full_pair_sweep(h_local, rng, max_pairs=n_try * n_try // 4)
            c_local = compute_c5(h_local)

        if c_local < best_c5 - 1e-8:
            best_c5 = c_local
            best_h = h_local.copy()

    # Phase 4: Final pair polishing on best solution
    n_best = len(best_h)
    h_final = best_h.copy()
    c_final = best_c5
    final_end = deadline

    # Multiple full pair sweeps
    while time.time() < final_end - 20:
        h_sw = h_final.copy()
        full_pair_sweep(h_sw, rng, max_pairs=None)
        c_sw = compute_c5(h_sw)
        if c_sw < c_final - 1e-10:
            c_final = c_sw
            h_final = h_sw.copy()
        else:
            # Random perturbation + short pair sweep
            h_p = h_final + rng.randn(n_best) * 0.015
            h_p = project_feasible(h_p, n_best / 2.0)
            for _ in range(300):
                i = rng.randint(n_best)
                j = rng.randint(n_best)
                if i == j:
                    continue
                d = pair_line_search(h_p, i, j, n_iter=15)
                if abs(d) > 1e-12:
                    h_p[i] -= d
                    h_p[j] += d
            c_p = compute_c5(h_p)
            if c_p < c_final:
                c_final = c_p
                h_final = h_p.copy()

    # Final fine random pairs
    while time.time() < final_end - 3:
        i = rng.randint(n_best)
        j = rng.randint(n_best)
        if i == j:
            continue
        d = pair_line_search(h_final, i, j, n_iter=20)
        if abs(d) > 1e-12:
            h_final[i] -= d
            h_final[j] += d
            c_now = compute_c5(h_final)
            if c_now < c_final - 1e-12:
                c_final = c_now

    # Final projection and evaluation
    h_final = project_feasible(h_final, n_best / 2.0)
    c_final = compute_c5(h_final)

    # Update best if final is better
    if c_final < best_c5:
        best_c5 = c_final
        best_h = h_final.copy()
        n_best = len(best_h)

    return best_h, best_c5, len(best_h)
