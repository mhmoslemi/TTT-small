import numpy as np
from scipy.optimize import minimize
import time

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _score(h):
    return float(np.max(_corr_vec(h)))

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_k(h, k):
    """Gradient of correlation[k] w.r.t. h."""
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    # h[i+s] * (1-h[i]) terms: d/dh[p] = (1-h[p-s]) for p = i+s
    i_start = max(0, s)
    i_end = min(n, n + s)
    for i in range(i_start, i_end):
        g[i] += (1.0 - h[i - s])
    # (1-h[i]) part: d/dh[p] = -h[p+s] for p = i
    i_start2 = max(0, -s)
    i_end2 = min(n, n - s)
    for i in range(i_start2, i_end2):
        g[i] += (-h[i + s])
    return g

def _minimax_dir(h, active):
    """Min-norm subgradient direction (negated) for the max of active correlations."""
    n = len(h)
    m = len(active)
    if m == 0:
        return np.zeros(n)
    grads = np.array([_grad_k(h, k) for k in active])
    # Project to sum-zero subspace
    means = grads.mean(axis=1)
    grads = grads - means[:, None]
    if m == 1:
        d = -grads[0]
        nr = np.linalg.norm(d)
        return d / nr if nr > 1e-15 else np.zeros(n)
    # QP: min w^T (G^T G) w s.t. 1^T w = 1, w >= 0
    # where G = grads (m x n)
    M = grads @ grads.T  # m x m
    def _qp_obj(w):
        return float(w @ M @ w)
    def _qp_grad(w):
        return 2.0 * (M @ w)
    w0 = np.ones(m) / m
    cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0, 'jac': lambda w: np.ones(m)}]
    try:
        res = minimize(_qp_obj, w0, jac=_qp_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-18})
        w = np.clip(res.x, 0, 1)
    except Exception:
        w = np.ones(m) / m
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(m) / m
    else:
        w = w / ws
    d = -(grads.T @ w)
    nr = np.linalg.norm(d)
    return d / nr if nr > 1e-15 else np.zeros(n)

def _max_step(h, d):
    """Max step maintaining 0 <= h + alpha*d <= 1."""
    alphas = []
    for i in range(len(h)):
        if d[i] > 1e-14:
            alphas.append((1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            alphas.append(h[i] / (-d[i]))
    return max(0.0, min(alphas)) if alphas else 0.0

def _minimax_optimize(h0, max_iter=5000, active_tol=1e-4, top_m=15):
    """Minimax direction optimization with full steps."""
    h = h0.copy()
    s = _score(h)
    stall = 0
    for it in range(max_iter):
        cv = _corr_vec(h)
        cmax = cv.max()
        # Get top lags
        idx = np.argsort(cv)[::-1][:top_m]
        active = idx[idx >= 0]  # all are >= 0
        # Filter to those within active_tol of max
        active = idx[cv[idx] >= cmax - active_tol]
        if len(active) == 0:
            active = idx[:1]
        
        d = _minimax_dir(h, active)
        nr = np.linalg.norm(d)
        if nr < 1e-14:
            break
        
        alpha = _max_step(h, d)
        if alpha < 1e-10:
            stall += 1
            if stall > 10:
                break
            # Try a small random perturbation to unstuck
            h = _project(h + 0.01 * np.random.randn(len(h)))
            continue
        
        h_new = _project(h + alpha * d)
        s_new = _score(h_new)
        
        if s_new < s - 1e-14:
            h = h_new
            s = s_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    
    return h, s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_local_search(h0, max_flips=5000):

    n = len(h)

    if excess > 0:
        # Need to flip 'excess' ones to zeros
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with smallest impact (or random)
        for i in range(int(excess)):
            h_bin[ones[i]] = 0
    elif excess < 0:
        zeros = np.where(h_bin == 0)[0]
        for i in range(int(-excess)):
            h_bin[zeros[i]] = 1
    
    h = _project(h_bin)
    s = _score(h)
    
    # Local search: flip pairs (one 1->0, one 0->1)
    for it in range(max_flips):
        cv = _corr_vec(h)
        ones = np.where(h > 0.9)[0]
        zeros = np.where(h < 0.1)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improvement = 1e-14
        best_pair = None
        # Try a subset of pairs for speed
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):

            h_test = h.copy()
            h_test[i] = 0.0

            h_test = _project(h_test)
            s_test = _score(h_test)
            if s_test < s - best_improvement:
                best_improvement = s - s_test
                best_pair = (i, j, s_test)
        
        if best_pair is not None:
            i, j, s_new = best_pair
            h[i] = 0.0
            h[j] = 1.0
            h = _project(h)
            s = _score(h)
        else:
            break
    
    return h, s

def _slsqp_polish(h0, maxiter=200):
    """SLSQP polish with direct min-max formulation."""
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = cv.max() + 0.001
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def _fun(z):
        return z[-1]
    
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        return t - _corr_vec(h)
    
    def _eq(z):
        return z[:-1].sum() - n / 2.0
    
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': _ieq},
                                    {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

import numpy as np
from scipy.optimize import minimize
import time

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _score(h):
    return float(np.max(_corr_vec(h)))

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    i_start = max(0, s)
    i_end = min(n, n + s)
    for i in range(i_start, i_end):
        g[i] += (1.0 - h[i - s])
    i_start2 = max(0, -s)
    i_end2 = min(n, n - s)
    for i in range(i_start2, i_end2):
        g[i] += (-h[i + s])
    return g

def _minimax_dir(h, active):
    n = len(h)
    m = len(active)
    if m == 0:
        return np.zeros(n)
    grads = np.array([_grad_k(h, k) for k in active])
    means = grads.mean(axis=1)
    grads = grads - means[:, None]
    if m == 1:
        d = -grads[0]
        nr = np.linalg.norm(d)
        return d / nr if nr > 1e-15 else np.zeros(n)
    M = grads @ grads.T
    def _qp_obj(w):
        return float(w @ M @ w)
    def _qp_grad(w):
        return 2.0 * (M @ w)
    w0 = np.ones(m) / m
    cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0, 'jac': lambda w: np.ones(m)}]
    try:
        res = minimize(_qp_obj, w0, jac=_qp_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-18})
        w = np.clip(res.x, 0, 1)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(m) / m
        else:
            w = w / ws
    except Exception:
        w = np.ones(m) / m
    d = -(grads.T @ w)
    nr = np.linalg.norm(d)
    return d / nr if nr > 1e-15 else np.zeros(n)

def _max_step(h, d):
    alphas = []
    for i in range(len(h)):
        if d[i] > 1e-14:
            alphas.append((1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            alphas.append(h[i] / (-d[i]))
    if not alphas:
        return 0.0
    a = min(alphas)
    return max(0.0, min(a, 5.0))

def _minimax_optimize(h0, max_iter=5000, active_tol=1e-4, top_m=15, deadline=None):
    h = h0.copy()
    s = _score(h)
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        cv = _corr_vec(h)
        cmax = cv.max()
        idx = np.argsort(cv)[::-1][:top_m]
        active = idx[cv[idx] >= cmax - active_tol]
        if len(active) == 0:
            active = idx[:1]
        d = _minimax_dir(h, active)
        nr = np.linalg.norm(d)
        if nr < 1e-14:
            break
        alpha = _max_step(h, d)
        if alpha < 1e-10:
            stall += 1
            if stall > 15:
                break
            h = _project(h + 0.005 * np.random.randn(len(h)))
            s = _score(h)
            continue
        h_new = _project(h + alpha * d)
        s_new = _score(h_new)
        if s_new < s - 1e-15:
            h = h_new
            s = s_new
            stall = 0
        else:
            stall += 1
            if stall > 25:

                active2 = idx[cv[idx] >= cmax - active_tol * 0.2]

                        s_test = _score(h_test)
                        if s_test < s - 1e-15:
                            h = h_test
                            s = s_test
                            stall = 0
                            continue
                break
    return h, s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_local_search(h0, rng, max_iters=3000, deadline=None):
    h = h0.copy()
    n = len(h)
    s = _score(h)
    h_b = (h > 0.5).astype(float)
    excess = h_b.sum() - n / 2.0
    if excess > 0:
        ones = np.where(h_b == 1)[0]
        rng.shuffle(ones)
        for i in range(int(np.ceil(excess))):
            h_b[ones[i]] = 0.0
    elif excess < 0:
        zeros = np.where(h_b == 0)[0]
        rng.shuffle(zeros)
        for i in range(int(np.ceil(-excess))):
            h_b[zeros[i]] = 1.0
    h = _project(h_b)
    s = _score(h)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h > 0.8)[0]
        zeros = np.where(h < 0.2)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_try = min(100, len(ones) * len(zeros))
        best_s = s
        best_i, best_j = -1, -1
        for _ in range(n_try):
            i = rng.integers(len(ones))
            j = rng.integers(len(zeros))
            ii = ones[i]
            jj = zeros[j]
            h_test = h.copy()
            h_test[ii] = 0.0
            h_test[jj] = 1.0
            h_test = _project(h_test)
            s_test = _score(h_test)
            if s_test < best_s - 1e-15:
                best_s = s_test
                best_i, best_j = ii, jj
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            h = _project(h)
            s = best_s
        else:
            break
    return h, s

def _slsqp_polish(h0, maxiter=200):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = cv.max() + 0.002
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        return z[-1] - _corr_vec(h)

    def _eq(z):
        return z[:-1].sum() - n / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': _ieq},
                                    {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _nesterov_subgrad(h0, beta, n_iter, alpha_init, mu, deadline=None):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        cv = _corr_vec(y)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -400, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.ones(len(cv)) / len(cv)
        else:
            w = ev / es
        g = np.zeros(n)
        for k in range(len(cv)):
            wk = w[k]
            if wk < 1e-12:
                continue
            gk = _grad_k(y, k)
            g += wk * gk
        h_new = _project(y - alpha_init * g)
        t_prev = t
        t = 1.5 * t
        h_prev = h_cur
        h_cur = h_new
        y = _project(h_cur + (t_prev / t) * (h_cur - h_prev))
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    return best_h, best_s
