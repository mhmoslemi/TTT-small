def gradient_p(h, s):
    """Gradient of the raw correlation at shift s with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for k in range(n - s):
            g[k] += 1.0 - h[k + s]
            g[k + s] -= h[k]
    else:
        t = -s
        for k in range(t, n):
            g[k] -= h[k - t]
            g[k - t] += 1.0 - h[k]
    return g

    """One active-set QP step."""
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        cp.norm(h_var - h0, 2) <= trust_radius
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = dx * gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)

    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:

            if prob.status in ['optimal', 'optimal_inaccurate']:
                break

            continue

        return None
    h_new = h_var.value

        return None
    return make_feasible(h_new)

def subgradient_step(h, top_k, step_size):
    """Take a subgradient step on the max function."""
    n = len(h)
    vals = compute_f(h)
    top_idx = np.argsort(vals)[-top_k:]
    # Subgradient of max is the gradient of an active constraint
    # Use weighted average of top-k gradients
    g_sum = np.zeros(n)
    for p in top_idx:
        s = p - (n - 1)
        g_sum += gradient_p(h, s)
    g_sum /= len(top_idx)
    # Normalize and take step
    g_norm = np.linalg.norm(g_sum)
    if g_norm < 1e-15:
        return h
    d = -g_sum / g_norm
    # Project to feasible set
    h_new = make_feasible(h + step_size * d)
    return h_new

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_p(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        idx = np.arange(n - s)
        g[idx] += 1.0 - h[idx + s]
        g[idx + s] -= h[idx]
    else:
        t = -s
        idx = np.arange(t, n)
        g[idx] -= h[idx - t]
        g[idx - t] += 1.0 - h[idx]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_step(h0, k_active, box_radius):
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n
    lb = np.maximum(0.0, h0 - box_radius)
    ub = np.minimum(1.0, h0 + box_radius)
    if np.any(lb > ub):
        return None
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= lb,
        h_var <= ub
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = dx * gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    for solver in [cp.GLOP, cp.PDLP, cp.CBC, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)

def qp_step(h0, k_active, reg):
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = dx * gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)
    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break

            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value

        return None
    return make_feasible(h_new)

def active_set_optimize(h0, max_iter=30, k_active=5, reg_start=3.0, reg_end=0.001,
                        trust_start=0.5, trust_end=0.03, use_lp=True):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    stall = 0
    for it in range(max_iter):
        t_frac = it / max(1, max_iter - 1)
        if use_lp:
            trust = trust_start * (trust_end / trust_start) ** t_frac
            h_new = lp_step(h, k_active=k_active, box_radius=trust)
        else:
            reg = reg_start * (reg_end / reg_start) ** t_frac
            h_new = qp_step(h, k_active=k_active, reg=reg)
        if h_new is None:
            if not use_lp and it < 3:
                continue
            break
        sc = official_score(h_new)
        if sc < best - 1e-15:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.8, 0.5, 0.3, 0.15, 0.05]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-15:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 12:
                    break
    return h, best

def smooth_obj(h, beta, dx):
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_obj_grad(h, beta, dx, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted)
    weights /= weights.sum()
    g = np.zeros(n)
    for i in range(2 * n - 1):
        if weights[i] < 1e-12:
            continue
        s = i - (n - 1)
        g += weights[i] * gradient_p(h, s)
    return g * dx

def slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    def obj(h):
        return smooth_obj(h, beta, dx)
    def obj_grad(h):
        return smooth_obj_grad(h, beta, dx, n)
    def eq_fun(h):
        return np.sum(h) - n / 2.0
    def eq_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            return h_new, official_score(h_new)
    except Exception:
        pass
    return None, None

def subgradient_step(h, top_k, step_size):
    n = len(h)
    vals = compute_f(h)
    top_idx = np.argsort(vals)[-top_k:]
    g_sum = np.zeros(n)
    for p in top_idx:
        s = p - (n - 1)
        g_sum += gradient_p(h, s)
    g_sum /= len(top_idx)
    g_norm = np.linalg.norm(g_sum)
    if g_norm < 1e-15:
        return h
    d = -g_sum / g_norm
    return make_feasible(h + step_size * d)

def pair_swap_optimize(h, max_sweeps=3, n_pairs=50):
    """Try pairwise mass exchanges to reduce the max correlation."""
    n = len(h)
    h = h.copy()
    best_sc = official_score(h)
    for sweep in range(max_sweeps):
        improved = False
        # Select interesting pairs: prefer coordinates where h is fractional
        frac_idx = np.where((h > 0.01) & (h < 0.99))[0]
        if len(frac_idx) < 2:
            frac_idx = np.arange(n)
        # Sample pairs
        rng_local = np.random.default_rng(42 + sweep)
        tried = 0
        for _ in range(200):
            if tried >= n_pairs:
                break
            if len(frac_idx) >= 2:
                i, j = frac_idx[rng_local.integers(0, len(frac_idx), size=2)]
            else:
                i, j = rng_local.integers(0, n, size=2)
            if i == j:
                continue
            tried += 1
            for delta in [0.01, -0.01, 0.02, -0.02, 0.03, -0.03]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                if h[j] - delta < 0 or h[j] - delta > 1:
                    continue
                h[i] += delta
                h[j] -= delta
                sc_try = official_score(h)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    improved = True
                    break
                else:
                    h[i] -= delta
                    h[j] += delta
        if not improved:
            break
    return h, best_sc
