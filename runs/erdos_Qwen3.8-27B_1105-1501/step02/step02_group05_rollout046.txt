import numpy as np

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def lag_grad(h, p, n):
    """Gradient of lag p (raw, before dx) of correlate(h, 1-h, mode='full')."""
    k = p - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        for i in range(n):
            if i < m:
                g[i] -= h[i + k]
            if i >= k:
                g[i] += (1.0 - h[i - k])
    else:
        t = -k
        m = n - t
        for i in range(n):
            if i < m:
                g[i] += (1.0 - h[i + t])
            if i >= t:
                g[i] -= h[i - t]
    return g

def lag_grad_vec(h, p_list, n):
    """Vectorized gradient computation for multiple lags."""
    dims = len(p_list)
    G = np.zeros((dims, n))
    for idx, p in enumerate(p_list):
        k = p - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[idx, :m] -= h[k:k+m]
                if k + m <= n:
                    G[idx, k:k+m] += (1.0 - h[:m])
        else:
            t = -k
            m = n - t
            if m > 0:
                G[idx, :m] += (1.0 - h[t:t+m])
                if t + m <= n:
                    G[idx, t:t+m] -= h[:m]
    return G

    else:

        if is_dead():
            break
        V = np.correlate(h, 1.0 - h, mode='full') * dx_val(n)
        cmax = V.max()
        active_idx = select_active(V, kmax, eta)
        
        rho = rho_start * (0.85 ** (stall // 3))
        rho = max(rho, 0.01)
        lam = lam_cycle[lam_idx % len(lam_cycle)]
        lam_idx += 1
        
        u = solve_lp_direction(h, active_idx, rho, lam)
        if u is None:
            stall += 1
            if stall % 5 == 0:
                eta *= 1.5
                kmax = min(kmax + 4, 2 * n - 1)
            continue
        
        # Line search
        found = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03]:
            trial = make_feasible(h + alpha * u)
            sc = score_exact(trial)
            if sc < best_sc - 1e-14:
                h = trial
                best_h = trial.copy()
                best_sc = sc
                found = True
                stall = 0
                eta = max(0.002, eta * 0.9)
                kmax = max(16, int(kmax * 0.95))
                rho_start = min(0.35, rho_start * 1.1)
                break
        if not found:
            stall += 1
            if stall >= 15:
                # Perturb to escape
                rng_local = np.random.default_rng(it * 7919 + 13)
                h = make_feasible(h + rng_local.normal(0, 0.02, size=n))
                sc = score_exact(h)
                if sc < best_sc - 1e-14:
                    best_h = h.copy()
                    best_sc = sc
                stall = 0
                eta = 0.01
                kmax = 32
                rho_start = 0.15
    
    return best_h, best_sc

def dx_val(n):
    return 2.0 / n

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        best_t = None
        best_local = best_sc
        for frac in [0.0, 0.25, 0.5, 0.75, 1.0, 0.382, 0.618]:
            t = t_lo + frac * span
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            best[a] = val_a
            best[b] = val_b
            sc = score_exact(best)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None:
            best[a] = best[a] + best_t
            best[b] = best[b] - best_t
            best_sc = best_local
        else:
            best[a] = best[a]
            best[b] = best[b]
    return best, best_sc

def binary_init(h):
    n = len(h)
    target = n // 2
    b = (h > 0.5).astype(float)
    ones = np.sum(b)
    if ones > target:
        idx = np.where(b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(int(ones - target)):
            b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(int(target - ones)):
            b[idx[order[i]]] = 1
    return b

def binary_sa(b0, n_steps, T0, cool, rng, is_dead):
    n = len(b0)
    target = n // 2
    b = b0.copy()
    best_b = b.copy()
    best_sc = score_exact(b)
    pos1 = np.where(b == 1)[0]
    pos0 = np.where(b == 0)[0]
    T = T0
    for it in range(n_steps):
        if is_dead():
            break
        i = pos1[rng.integers(0, len(pos1))]
        j = pos0[rng.integers(0, len(pos0))]
        if i == j:
            T *= cool
            continue
        b[i] = 0
        b[j] = 1
        pos1[pos1 == i] = j
        pos1[pos1 == j] = i
        pos0[pos0 == j] = i
        pos0[pos0 == i] = j
        sc = score_exact(b)
        delta = sc - best_sc
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            if sc < best_sc:
                best_sc = sc
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
            pos1[pos1 == j] = i
            pos1[pos1 == i] = j
            pos0[pos0 == i] = j
            pos0[pos0 == j] = i
        T *= cool
    return best_b, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)
    
    def is_dead():
        return time.time() >= deadline
    
    best_h = None
    best_score = float('inf')
    
    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass
    
    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.1)
        best_score = score_exact(best_h)
    
    n = len(best_h)
    
    # Build initial candidate pool
    pool = [best_h.copy()]
    # Anti-symmetric
    h_sym = anti_sym_map(best_h)
    sc_sym = score_exact(h_sym)
    if sc_sym < best_score - 1e-14:
        best_h = h_sym.copy()
        best_score = sc_sym
    if sc_sym < best_score + 0.01:
        pool.append(h_sym)
    
    # Phase 1: LP refinement on best (primary method)
    if not is_dead():
        hp, scp = lp_refine(best_h, is_dead, max_iters=300)
        if scp < best_score - 1e-14:
            best_h = hp
            best_score = scp
    
    # Phase 2: Binary SA exploration
    if not is_dead():
        b_init = binary_init(best_h)
        T0 = max(2e-4, 0.3 * (best_score - 0.38))
        T0 = max(T0, 1e-4)
        n_sa = int(min(20000, max(5000, (deadline - time.time()) * 0.15)))
        b_rand = rng.integers(0, 2, size=n).astype(float)
        # Balance binary
        while np.sum(b_rand) > n // 2 and np.sum(b_rand) > 0:
            idx1 = np.where(b_rand == 1)[0]
            if len(idx1) > 0:
                b_rand[idx1[rng.integers(0, len(idx1))]] = 0
        while np.sum(b_rand) < n // 2:
            idx0 = np.where(b_rand == 0)[0]
            if len(idx0) > 0:
                b_rand[idx0[rng.integers(0, len(idx0))]] = 1
        
        b_starts = [b_init, b_rand]
        for bs in b_starts:
            if is_dead():
                break
            n_steps = max(3000, int((deadline - time.time()) * 0.08))
            b_best, b_sc = binary_sa(bs, n_steps, T0, 0.9992, rng, is_dead)
            if b_sc < best_score - 1e-14:
                # Convert to continuous and polish
                h_cont = make_feasible(b_best)
                sc_cont = score_exact(h_cont)
                if sc_cont < best_score - 1e-14:
                    best_h = h_cont
                    best_score = sc_cont
                # Also try anti-symmetric of binary
                h_anti = anti_sym_map(b_best)
                sc_anti = score_exact(h_anti)
                if sc_anti < best_score - 1e-14:
                    best_h = h_anti
                    best_score = sc_anti
    
    # Phase 3: More LP refinement after binary exploration
    if not is_dead():
        hp, scp = lp_refine(best_h, is_dead, max_iters=300)
        if scp < best_score - 1e-14:
            best_h = hp
            best_score = scp
    
    # Phase 4: Two-point search (proven effective)
    if not is_dead():
        n_tries = int(min(15000, max(5000, (deadline - time.time()) * 0.12)))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)
    
    # Phase 5: LP refactor
    if not is_dead():
        hp, scp = lp_refine(best_h, is_dead, max_iters=200)
        if scp < best_score - 1e-14:
            best_h = hp
            best_score = scp
    
    # Phase 6: Resolution scanning
    if not is_dead():
        n_cur = len(best_h)
        n_opts = sorted(set([n_cur + d for d in [-8, -4, 4, 8] if 20 <= n_cur + d <= 100]))
        for nc in [40, 44, 48, 52, 56, 64, 72, 80]:
            if nc not in n_opts and 20 <= nc <= 100:
                n_opts.append(nc)
        n_opts = sorted(n_opts)
        for n_opt in n_opts:
            if is_dead():
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-14:
                best_h = h_int
                best_score = sc_int
                # Quick LP on new resolution
                if not is_dead():
                    hp2, sc2 = lp_refine(best_h, is_dead, max_iters=80)
                    if sc2 < best_score - 1e-14:
                        best_h = hp2
                        best_score = sc2
    
    # Phase 7: Intensive two-point polishing
    if not is_dead():
        n_tries = int(min(20000, max(5000, (deadline - time.time()) * 0.15)))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)
    
    # Phase 8: Final LP pass
    if not is_dead():
        hp, scp = lp_refine(best_h, is_dead, max_iters=200)
        if scp < best_score - 1e-14:
            best_h = hp
            best_score = scp
    
    # Phase 9: Random perturbations + LP
    if not is_dead():
        for sigma in [0.02, 0.04, 0.06]:
            if is_dead():
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=len(best_h)))
            hp, scp = lp_refine(h_pert, is_dead, max_iters=100)
            if scp < best_score - 1e-14:
                best_h = hp
                best_score = scp
    
    # Phase 10: Final two-point sweep
    if not is_dead():
        n_tries = int(min(10000, max(3000, (deadline - time.time()) * 0.2)))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)
    
    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8
    
    return final_h, final_score, final_n
