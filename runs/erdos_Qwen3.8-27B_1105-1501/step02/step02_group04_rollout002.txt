Hmm, this has an issue with the indexing when s is negative or positive. Let me be more careful.

For a given lag s (shift), and variable index j:
- Term 1: h[j] appears as h[i+s] where i = j-s. Valid if 0 <= i < n, i.e., 0 <= j-s < n.
  Contribution: (1 - h[i]) = (1 - h[j-s])
- Term 2: h[j] appears as part of (1-h[i]) where i = j. Valid if 0 <= i+s < n, i.e., 0 <= j+s < n.
  Contribution: -h[i+s] = -h[j+s]

So:
grad[j, m] = (1-h[j-s]) * [0 <= j-s < n] - h[j+s] * [0 <= j+s < n]

For s > 0: j-s >= 0 means j >= s; j+s < n means j < n-s
For s < 0: j-s >= 0 means j >= s (always true since j>=0 and s<0, so j-s > 0); j+s < n means j < n-s (always true since s<0)
Wait no: for s < 0, j-s = j+|s|, which is in [0, n-1] when j < n-|s|. And j+s = j-|s|, in [0,n-1] when j >= |s|.

```python
import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_grad_matrix(h):
    n = len(h)
    num_lags = 2 * n - 1
    grad = np.zeros((n, num_lags))
    js = np.arange(n)
    for m in range(num_lags):
        s = m - (n - 1)
        idx1 = js - s
        mask1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.zeros(n)
        if np.any(mask1):
            term1[mask1] = 1.0 - h[idx1[mask1]]
        idx2 = js + s
        mask2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.zeros(n)
        if np.any(mask2):
            term2[mask2] = h[idx2[mask2]]
        grad[:, m] = term1 - term2
    return grad

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lp_direction(h, grad, corr, trust_radius):
    """Solve LP to find descent direction.
    Returns (d, t) where d is the direction and t is the predicted max change.
    """
    n = len(h)
    num_lags = 2 * n - 1
    c_max = np.max(corr)
    
    # Variables: d[0..n-1], t  (n+1 total)
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: G[m,:].d - t <= c_max - corr[m] for each m
    # A_ineq shape: (num_lags, n+1)
    A_ineq = np.zeros((num_lags, n + 1))
    # Transpose grad: grad is (n, num_lags), we need (num_lags, n)
    G = grad.T  # (num_lags, n)
    A_ineq[:, :n] = G
    A_ineq[:, n] = -1.0
    b_ineq = c_max - corr
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    bounds.append((-c_max, c_max))  # bound on t
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, 0.0
    
    d = res.x[:n]
    t = res.x[n]
    return d, t

def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(32):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def full_pair_sweep(h, rng, max_iterations=None):
    n = len(h)
    perm = rng.permutation(n)
    count = 0
    for a in range(n):
        for b in range(a + 1, n):
            if max_iterations and count >= max_iterations:
                return
            count += 1
            i, j = perm[a], perm[b]
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    n = 70
    n_half = n // 2
    
    # Get initial solution
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        assert len(h) == n

    except (NameError, Exception):
        h = np.full(n, 0.5)
    
    h = project_onto_constraint(h, n_half)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Phase 1: LP-based descent (main algorithm)
    # Phase 2: Pair polishing
    # Phase 3: Random restarts + polish
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: LP descent
    phase1_end = t0 + budget_s * 0.55
    while elapsed() < phase1_end:
        corr = compute_correlations(h)
        c_max = np.max(corr)
        grad = compute_grad_matrix(h)
        
        d, t_pred = lp_direction(h, grad, corr, 0.1)
        if d is None:
            break
        
        # Check if t_pred < 0 (descent)
        if t_pred < -1e-10:
            # Line search: try alpha in [0, 1]
            best_alpha = 0.0
            best_f = compute_c5(h)
            for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_trial = h + alpha * d
                h_trial = project_onto_constraint(h_trial, n_half)
                f = compute_c5(h_trial)
                if f < best_f - 1e-12:
                    best_f = f
                    best_alpha = alpha
            
            if best_alpha > 0:
                h = h + best_alpha * d
                h = project_onto_constraint(h, n_half)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        else:
            # No LP descent found, break to polishing
            break
    
    # Phase 2: Pair polishing
    phase2_end = t0 + budget_s * 0.80
    stall = 0
    while elapsed() < phase2_end:

        h = project_onto_constraint(h, n_half)

            best_h = h.copy()
            stall = 0
        else:

        h = best_h + rng.randn(n) * scale
        h = project_onto_constraint(h, n_half)

            if elapsed() > phase3_end:
