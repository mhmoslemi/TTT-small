Now for the anti-symmetric LP:
If we parameterize by the first half `x = h[:m]` where m = n/2, and h = as_from_half(x),
then a perturbation δx in the first half gives δh = [δx, -δx[::-1]].

The change in f_s is: G[p] @ δh = G[p, :m] @ δx + G[p, m:] @ (-δx[::-1])

Let me define the "effective gradient" for the anti-symmetric parameterization:
g_eff[p, j] = G[p, j] - G[p, n-1-j] for j in [0, m)

Because δh[j] = δx[j] and δh[n-1-j] = -δx[j], so:
G[p] @ δh = ∑_j G[p,j] δx[j] + ∑_j G[p, n-1-j] (-δx[j])
          = ∑_j (G[p,j] - G[p,n-1-j]) δx[j]
          = g_eff[p] @ δx

Great! So the LP becomes:
Variables: δx ∈ ℝ^m
minimize t
subject to: g_eff[active] @ δx + vals[active] ≤ t
            -h[j] ≤ δx[j] ≤ 1-h[j] for j in [0, m)
            |δx[j]| ≤ δ for j in [0, m)

Note: the mass constraint ∑δh = 0 is automatically satisfied since ∑δx - ∑δx[::-1] = 0.

This is a clean LP with m variables and ~|active| + 2m constraints.

    n = len(h)

    x: first half (m values)
    h: full anti-symmetric h (2m values)

    m = len(x)
    dx = 2.0 / n
    
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    active = np.where(vals >= cmax - tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(vals)])

    t = cp.Variable()

        g_eff @ d + vals[active] <= t,
        d >= lb,
        d <= ub,

    max_a = max(max_a, 0.0)

            best_x = x_trial.copy()

def two_search(x, n_tries, rng, is_dead):

    m = len(x)
    best = x.copy()
    h_best = as_from_half(best)
    best_sc = official_score(h_best)
    
    for it in range(n_tries):
        if is_dead():
            break
        a = int(rng.integers(0, m))
        b = int(rng.integers(0, m))
        if a == b:
            continue
        t_lo = max(-best[a], -(1.0 - best[b]))
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        # Golden ratio candidates + endpoints + midpoint
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        candidates = [t_lo + 0.5 * span, t_lo + (1.0/gr) * span, t_lo + (1.0 - 1.0/gr) * span,
                      t_lo, t_hi, t_lo + 0.25 * span, t_lo + 0.75 * span]
        best_t = None
        best_local = best_sc
        for t in candidates:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13:
                continue
            if vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp = best.copy()
            tmp[a] = va
            tmp[b] = vb
            tmp = np.clip(tmp, 0.0, 1.0)
            sc = official_score(as_from_half(tmp))
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = best[a] + best_t
            best[b] = best[b] - best_t
            best = np.clip(best, 0.0, 1.0)
            best_sc = best_local
    
    return best, best_sc

    n_src = len(h_full)
    m_tgt = n_tgt // 2

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_full)
    return extract_half(anti_symmetrize(h_int))

def subgradient_step(x, h, alpha, is_dead):
    """One subgradient step in anti-sym param."""
    n = len(h)
    m = len(x)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    # Use all lags within tolerance
    tol = max(0.01, 0.1 * cmax)
    active = np.where(vals >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(vals))])
    # Weighted average of gradients
    w = np.exp(30.0 * (vals[active] - cmax))
    w = w / w.sum()
    g = np.zeros(m)
    for k, p in enumerate(active):
        g += w[k] * compute_eff_grad_row(h, int(p))
    # Move opposite to gradient
    d = -g
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-14:
        return x.copy(), official_score(h)
    d = d / d_norm
    # Projected step
    x_trial = np.clip(x + alpha * d, 0.0, 1.0)
    h_trial = as_from_half(x_trial)
    sc = official_score(h_trial)
    return x_trial, sc

import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        order = np.argsort(1.0 - h) if residual > 0 else np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def as_from_half(x):
    m = len(x)
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def extract_half(h):
    return h[:len(h) // 2].copy()

def anti_symmetrize(h):
    n = len(h)
    m = n // 2
    half = 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])
    return as_from_half(np.clip(half, 0.0, 1.0))

def eff_grad_row(h, p):
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    s = p - (n - 1)
    row = np.zeros(n)
    if s >= 0:
        if s == 0:
            row[:] = 1.0 - 2.0 * h
        else:
            row[:n - s] += (1.0 - h[s:])
            row[s:] -= h[:n - s]
    else:
        t = -s
        row[t:] += (1.0 - h[:n - t])
        row[:n - t] -= h[t:]
    return (row[:m] - row[n - 1:m - 1:-1]) * dx

def lp_step(x, h, delta, tol):
    n = len(h)
    m = len(x)
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cmax = vals.max()
    active = np.where(vals >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(vals))])
    G = np.array([eff_grad_row(h, int(p)) for p in active])
    d = cp.Variable(m)
    t = cp.Variable()
    lb = np.maximum(-x, -delta)
    ub = np.minimum(1.0 - x, delta)
    constraints = [G @ d + vals[active] <= t, d >= lb, d <= ub]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if d.value is not None and np.all(np.isfinite(d.value)):
                return d.value.copy()
        except Exception:
            continue
    return None

        best_x, best_score = two_search(best_x, n_fp, rng, is_dead)

    h_final = as_from_half(best_x)
    h_final = np.clip(h_final, 0.0, 1.0)
    final_n = len(h_final)
    final_score = official_score(h_final)

    assert np.all(np.isfinite(h_final))
    assert h_final.min() >= -1e-10
    assert h_final.max() <= 1.0 + 1e-10
    assert abs(h_final.sum() - final_n / 2.0) < 1e-8

    return h_final, final_score, final_n
