def make_feasible(z, target=None):

                    J[i, m2:] = self.dx * h[:self.n-m2]

    def eq(self, z):
        return np.sum(z[:-1]) - self.n / 2.0
    
    def eq_jac(self, z):

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -np.max(z) - 1.0, 1.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    return h


def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)


def push_binary(h, p=2.0):
    return make_feasible(h ** p)


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] += -h[s:]
    else:
        m2 = -s
        if m2 == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - m2] += 1.0 - h[m2:]
            g[m2:] += -h[:n - m2]
    return g


def nesterov_subgrad(h0, max_iter=5000, t_start=None, time_limit=None):
    h = h0.copy()
    n = len(h)
    y = h0.copy()
    best_h = h0.copy()
    best_score = official_score(h0)
    c = 0.4
    beta = 1.0
    patience = 0
    max_patience = 500

    for k in range(1, max_iter + 1):
        if t_start is not None and time_limit is not None:
            if time.time() - t_start > time_limit:
                break

        corr = compute_corr(y)
        m = np.max(corr)
        threshold = m - 0.03 * m
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for ki in active:
            grad += _grad_k(y, ki)
        grad /= len(active)

        alpha = c / np.sqrt(k)
        h_new = make_feasible(y - alpha * grad)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = (t_k - 1.0) / (t_k + 1.0)
        y_new = make_feasible(h_new + momentum * (h_new - h))

        h = h_new
        y = y_new
        beta += 1.0

        score = official_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

    return best_h, best_score


class LSEProblem:
    def __init__(self, n, dx, beta):
        self.n = n
        self.dx = dx
        self.beta = beta

    def obj(self, x):
        corr = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = np.max(corr)
        shifted = np.clip(self.beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            return m
        return m + np.log(s) / self.beta

    def grad(self, x):
        n = self.n
        corr = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = np.max(corr)
        shifted = np.clip(self.beta * (corr - m), -500, 0)
        e = np.exp(shifted)
        s = np.sum(e)
        if s < 1e-300:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights = e / s
        g = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k] * self.dx
            if w < 1e-15:
                continue
            g += w * _grad_k(x, k)
        return g

    def eq_fun(self, x):
        return np.sum(x) - self.n / 2.0

    def eq_jac(self, x):
        return np.ones(self.n)


class MinMaxProblem:
    def __init__(self, n, dx, active_lags):
        self.n = n
        self.dx = dx
        self.active = active_lags

    def obj(self, z):
        return z[-1]

    def obj_jac(self, z):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g

    def ineq(self, z):
        h = z[:-1]
        t = z[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return t - c[self.active]

    def ineq_jac(self, z):
        h = z[:-1]
        J = np.zeros((len(self.active), self.n + 1))
        for i, ki in enumerate(self.active):
            J[i, -1] = 1.0
            J[i, :self.n] = -self.dx * _grad_k(h, ki)
        return J

    def eq(self, z):
        return np.sum(z[:-1]) - self.n / 2.0

    def eq_jac(self, z):
        g = np.zeros(self.n + 1)
        g[:-1] = 1.0
        return g


def lse_slsqp(h0, beta, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    prob = LSEProblem(n, dx, beta)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            prob.obj, h0, jac=prob.grad, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0


def direct_minmax(h0, n_active=5, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    corr = compute_corr(h0)
    m = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:min(n_active, 2 * n - 1)]
    z0 = np.concatenate([h0, [m + 0.005]])
    prob = MinMaxProblem(n, dx, active)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(
            prob.obj, z0, jac=prob.obj_jac, method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': prob.ineq, 'jac': prob.ineq_jac},
                {'type': 'eq', 'fun': prob.eq, 'jac': prob.eq_jac}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0


def binary_search(h0, rng, max_rounds=300, time_limit=None, t_start=None):
    h = h0.copy()
    n = len(h)
    best_score = official_score(h)

    for rnd in range(max_rounds):
        if t_start is not None and time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        ones = np.where(h > 0.6)[0]
        zeros = np.where(h < 0.4)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improve = 1e-10
        best_move = None
        n_try = min(80, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            for frac in [0.3, 0.6, 1.0]:
                delta = frac * h[i]
                if delta > 1.0 - h[j]:
                    delta = 1.0 - h[j]
                if delta < 0.02:
                    continue
                si, sj = h[i], h[j]
                h[i] = si - delta
                h[j] = sj + delta
                s = official_score(h)
                if best_score - s > best_improve:
                    best_improve = best_score - s
                    best_move = (i, j, delta)
                h[i], h[j] = si, sj

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            best_score = official_score(h)
        else:
            break

    return h, best_score
