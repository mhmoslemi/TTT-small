def official_score(h):
    n = len(h)
    dx = 2.0 / n

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def lbfgsb_objective_grad(h, beta, gamma):
    """Objective and gradient for L-BFGS-B."""
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    beta_V = beta * V
    V_max = beta_V.max()
    exp_shifted = np.exp(beta_V - V_max)
    exp_sum = exp_shifted.sum()
    logsumexp = V_max + np.log(exp_sum)
    value = logsumexp / beta

    value += gamma * (s - n/2.0) ** 2
    # Gradient
    w = exp_shifted / exp_sum
    grad = dx * fast_gradient(h, w)
    grad += 2.0 * gamma * (s - n/2.0) * 1.0

def lbfgsb_optimize(h0, beta, gamma, maxiter=500, gtol=1e-8):
    """Run L-BFGS-B optimization."""
    n = len(h0)
    
    def fun(x):
        val, _ = lbfgsb_objective_grad(x, beta, gamma)
        return val
    
    def grad(x):
        _, g = lbfgsb_objective_grad(x, beta, gamma)
        return g
    
    bounds = [(0.0, 1.0)] * n

def pg_step(h, beta, alpha_max=0.8, max_ls=18):

    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = fast_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)

def active_set_lp(h, eps=0.005, trust=0.2):

    dx = 2.0 / n

    V_max = V.max()

        return None, None

    b_rows = []
    for p_idx in active:

        L = n - abs(s)
        if L <= 0:
            continue

        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]

            gp[t:n] -= h[0:L]

        b_rows.append(-V[p_idx])

        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))

        pass
    return None, None

def make_feasible(z):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

    V = compute_values(x)

    V_max = beta_V.max()

    exp_sum = exp_shifted.sum()
    value = (V_max + np.log(exp_sum)) / beta
    s = x.sum()
    value += gamma * (s - n / 2.0) ** 2
    return value

def lse_grad(x, beta, gamma):

    dx = 2.0 / n
    V = compute_values(x)
    beta_V = beta * V
    V_max = beta_V.max()

    w = exp_shifted / exp_sum

    s = x.sum()
    grad += 2.0 * gamma * (s - n / 2.0)
    return grad

    f = V.max()

        n_base = len(best_h)
        n_tries = sorted(set([n_base - 4, n_base - 2, n_base + 2, n_base + 4,
                              n_base - 6, n_base + 6]))
        for n_try in n_tries:
            if time_left() < 50 or n_try < 20 or n_try > 120:
                continue
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            if sc_int < best_score:
                best_h = h_int.copy()
                best_score = sc_int
            h_cur = h_int.copy()
            for beta in [30.0, 80.0, 200.0]:
                if time_left() < 35:
                    break
                for gamma in [1000.0, 100.0, 10.0]:
                    if time_left() < 25:
                        break
                    h_new, sc = run_lbfgsb(h_cur, beta, gamma, maxiter=120)
                    if sc < best_score:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    # Phase 6: Multi-start from structured inits
    if time_left() > 120:
        n_use = len(best_h)
        starts = []
        # Sinusoidal
        x = np.arange(n_use) / n_use
        s1 = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        starts.append(make_feasible(s1))
        # Binary-like
        s2 = np.zeros(n_use)
        s2[::2] = 1.0
        starts.append(make_feasible(s2))
        # Block structure
        block = n_use // 6
        s3 = np.zeros(n_use)
        s3[:block] = 1.0
        s3[2*block:3*block] = 1.0
        s3[4*block:5*block] = 1.0
        starts.append(make_feasible(s3))
        # Random
        s4 = rng.uniform(0.2, 0.8, n_use)
        starts.append(make_feasible(s4))

        for s in starts:
            if time_left() < 60:
                break
            h_cur = s.copy()
            for beta in [20.0, 80.0]:
                if time_left() < 40:
                    break
                for gamma in [1000.0, 100.0]:
                    if time_left() < 30:
                        break
                    h_new, sc = run_lbfgsb(h_cur, beta, gamma, maxiter=150)
                    if sc < best_score:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    # Phase 7: Final fine PG
    if time_left() > 30:
        h_cur = best_h.copy()
        beta = 300.0
        for _ in range(800):
            if time_left() < 15:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.15, max_ls=10)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new

    # Phase 8: Final LP with very tight trust
    if time_left() > 15:
        h_cur = best_h.copy()
        for eps in [0.0005, 0.0002, 0.0001]:
            if time_left() < 10:
                break
            for _ in range(40):
                if time_left() < 10:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.03)
                if h_new is not None:
                    if sc < best_score - 1e-13:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
