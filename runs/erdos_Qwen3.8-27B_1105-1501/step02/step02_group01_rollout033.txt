```python
    """Return new score after swapping h[i](=1) and h[j](=0)"""
    n = len(h)
    s = np.arange(-(n-1), n)

    delta[mask] -= h[idx[mask]]
    return np.max(C + delta) * dx
```

```python
def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):

    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p=2.0):

    return make_feasible(h_pushed)

    h = make_feasible(h0)
    n = len(h)
    best_score = official_score(h)
    
    for rnd in range(max_rounds):

            if time.time() - t_start > time_limit:
                break
        # Find positions to swap
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        C = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n

        best_improve = 1e-15

                delta[mask] += h[idx[mask]]
                idx = i + s
                mask = (idx >= 0) & (idx < n)

                new_max = np.max(C + delta)
                if new_max < current_max - 1e-14:
                    improve = current_max - new_max
```

```python
h_i_minus_s = h_pad[(I + pad - S).astype(int)]  # (n_ones, 1, 2n-1)
h_j_minus_s = h_pad[(J + pad - S).astype(int)]  # (1, n_zeros, 2n-1)
h_j_plus_s = h_pad[(J + pad + S).astype(int)]  # (1, n_zeros, 2n-1)
h_i_plus_s = h_pad[(I + pad + S).astype(int)]  # (n_ones, 1, 2n-1)

# delta[i,j,s] = h[i-s] - h[j-s] + h[j+s] - h[i+s]
# = h_i_minus_s[i,0,k] - h_j_minus_s[0,j,k] + h_j_plus_s[0,j,k] - h_i_plus_s[i,0,k]
delta = h_i_minus_s - h_j_minus_s + h_j_plus_s - h_i_plus_s  # (n_ones, n_zeros, 2n-1)

# new_max[i,j] = max_s (C[k] + delta[i,j,k]) where k = s + (n-1)
# C is indexed by k = 0..2n-2, s = k-(n-1) = -(n-1)..(n-1)
# So C corresponds to s in order
new_max = np.max(C[None, None, :] + delta, axis=2)  # (n_ones, n_zeros)

# Find best pair
best_idx = np.unravel_index(np.argmin(new_max), new_max.shape)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
            else:
                add = max(residual, -h[idx])
            if abs(add) < 1e-15:
                break
            h[idx] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

def to_binary(h, n_ones=None):
    """Convert to exact binary (n/2 ones for even n)."""
    n = len(h)
    if n_ones is None:
        n_ones = n // 2
    # Take the n_ones largest values as 1s
    thresh_idx = np.partition(h, n - n_ones)
    thresh = thresh_idx[n - n_ones]
    b = (h >= thresh).astype(float)
    # Ensure exactly n_ones are 1
    n_actual = int(np.sum(b))
    if n_actual > n_ones:
        # Too many 1s, set some to 0
        ones_pos = np.where(b == 1)[0]
        # Set the smallest ones to 0
        order = np.argsort(h[ones_pos])
        for idx in order[:n_actual - n_ones]:
            b[ones_pos[idx]] = 0.0
    elif n_actual < n_ones:
        zeros_pos = np.where(b == 0)[0]
        order = np.argsort(-h[zeros_pos])
        for idx in order[:n_ones - n_actual]:
            b[zeros_pos[idx]] = 1.0
    return b

def vectorized_local_search(h0, max_rounds=20000, time_limit=None, t_start=None):
    """Fast vectorized binary local search."""
    h = to_binary(h0).copy()
    n = len(h)
    dx = 2.0 / n
    best_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    
    s_arr = np.arange(-(n-1), n)  # (2n-1,)
    pad = n - 1
    C = np.correlate(h, 1.0 - h, mode='full')
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Build padded h
        h_pad = np.zeros(2 * pad + 1, dtype=float)
        h_pad[pad:pad+n] = h
        
        # Vectorized delta computation
        # I shape: (n_ones, 1, 1), J shape: (1, n_zeros, 1), S shape: (1, 1, 2n-1)
        I = ones.astype(int)[:, None, None] + pad
        J = zeros.astype(int)[None, :, None] + pad
        S = s_arr[None, None, :]
        
        # h[i-s] → h_pad[I - S], h[j-s] → h_pad[J - S], etc.
        idx_iminus_s = (I - S).astype(int)
        idx_jminus_s = (J - S).astype(int)
        idx_jplus_s = (J + S).astype(int)
        idx_iplus_s = (I + S).astype(int)
        
        # Clip to valid range [0, len(h_pad)-1]
        max_idx = 2 * pad  # last valid index
        idx_iminus_s = np.clip(idx_iminus_s, 0, max_idx)
        idx_jminus_s = np.clip(idx_jminus_s, 0, max_idx)
        idx_jplus_s = np.clip(idx_jplus_s, 0, max_idx)
        idx_iplus_s = np.clip(idx_iplus_s, 0, max_idx)
        
        h_iminus = h_pad[idx_iminus_s]  # (n_ones, 1, 2n-1) broadcast to (n_ones, n_zeros, 2n-1)
        h_jminus = h_pad[idx_jminus_s]  # (1, n_zeros, 2n-1)
        h_jplus = h_pad[idx_jplus_s]    # (1, n_zeros, 2n-1)
        h_iplus = h_pad[idx_iplus_s]    # (n_ones, 1, 2n-1)
        
        # Wait, broadcasting won't give (n_ones, n_zeros, 2n-1) directly
        # h_iminus is (n_ones, 1, 2n-1), h_jminus is (1, n_zeros, 2n-1)
        # Their difference will broadcast to (n_ones, n_zeros, 2n-1) ✓
        
        delta = h_iminus - h_jminus + h_jplus - h_iplus  # (n_ones, n_zeros, 2n-1)
        
        # new_max for each pair
        new_max = np.max(C[None, None, :] + delta, axis=2)  # (n_ones, n_zeros)
        
        current_max = C.max()
        min_new = new_max.min()
        
        if min_new >= current_max - 1e-14:
            break  # No improving swap
        
        # Find best pair
        best_pos = np.unravel_index(new_max.argmin(), new_max.shape)
        i = ones[best_pos[0]]
        j = zeros[best_pos[1]]
        
        h[i] = 0.0
        h[j] = 1.0
        
        C = np.correlate(h, 1.0 - h, mode='full')
        best_score = C.max() * dx
    
    return h, best_score

def smooth_refine(h0, max_iter=200, time_limit=None, t_start=None):
    """Simple projected gradient (subgradient) method for smooth refinement."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = official_score(h)
    lr = 0.02
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = corr.max()
        
        # Active lags: within 5% of max
        thresh = max_val - 0.05 * max_val
        active = np.where(corr >= thresh)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Compute average gradient over active lags
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:])
            else:
                m = -s
                grad[:n-m] += (1.0 - h[m:])
                grad[m:] += (-h[:n-m])
        grad /= len(active)
        
        h_new = make_feasible(h - lr * grad)
        score = official_score(h_new)
        
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr = max(lr * 0.8, 0.005)
    
    return best_h, best_score

def generate_starts(n, parent, rng):
    """Generate diverse starting points of length n."""
    starts = []
    
    # Interpolated parent
    if parent is not None:
        pi = interp_to_n(parent, n)
        starts.append(pi)
        starts.append(push_binary(pi, 2.0))
        starts.append(push_binary(pi, 4.0))
        starts.append(to_binary(pi) * 1.0)

        rb[rng.choice(n, n//2, replace=False)] = 1.0
        starts.append(rb)

    for block_size in [2, 3, 4, 5, 7]:
        pattern = np.zeros(n)
        for i in range(n):
            cycle_pos = (i // block_size) % 2
            if cycle_pos == 0:
                pattern[i] = 1.0
        # Adjust to have exactly n/2 ones
        starts.append(to_binary(pattern) * 1.0)
    
    # Golden ratio
    phi = (1 + np.sqrt(5)) / 2
    gr = np.zeros(n)
    for i in range(n):
        gr[i] = 1.0 if ((i + 1) * phi) % 1.0 < 0.5 else 0.0
    starts.append(to_binary(gr) * 1.0)
    
    # Alternating small blocks
    for small in [1, 2]:
        alt = np.zeros(n)
        for i in range(n):
            if (i // (small * 2)) % 2 == 0:
                alt[i] = 1.0
            elif i % (small * 2) < small:
                alt[i] = 1.0
        starts.append(to_binary(alt) * 1.0)
    
    return starts
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
            else:
                add = max(residual, -h[idx])
            if abs(add) < 1e-15:
                break
            h[idx] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

def to_binary(h):
    n = len(h)
    n_ones = n // 2
    b = np.zeros(n)
    order = np.argsort(-h)
    b[order[:n_ones]] = 1.0
    return b

def vectorized_local_search(h0, max_rounds=50000, time_limit=None, t_start=None):
    h = to_binary(h0).copy()
    n = len(h)
    dx = 2.0 / n
    pad = n - 1
    C = np.correlate(h, 1.0 - h, mode='full')
    best_score = C.max() * dx
    s_arr = np.arange(-(n - 1), n)
    max_idx = 2 * pad
    h_pad = np.zeros(2 * pad + 1, dtype=float)
    h_pad[pad:pad + n] = h
    stagnation = 0
    max_stagnation = 50

    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break

        I = ones.astype(np.int32)[:, None, None].flatten()
        I = np.tile(I[:, :, None], (1, 1, 2 * n - 1))
        I = I + pad
        J = zeros.astype(np.int32)[None, :, None].flatten()
        J = np.tile(J[None, :, :], (n_ones, 1, 1))
        J = J + pad
        S = s_arr[None, None, :].astype(np.int32)
        S = np.tile(S, (n_ones, n_zeros, 1))

        idx_iminus = np.clip(I - S, 0, max_idx)
        idx_jminus = np.clip(J - S, 0, max_idx)
        idx_jplus = np.clip(J + S, 0, max_idx)
        idx_iplus = np.clip(I + S, 0, max_idx)

        h_iminus = h_pad[idx_iminus]
        h_jminus = h_pad[idx_jminus]
        h_jplus = h_pad[idx_jplus]
        h_iplus = h_pad[idx_iplus]

        delta = h_iminus - h_jminus + h_jplus - h_iplus
        new_max = np.max(C[None, None, :] + delta, axis=2)

        current_max = C.max()
        min_new = new_max.min()

        if min_new >= current_max - 1e-14:
            stagnation += 1
            if stagnation > max_stagnation:
                break
            # Random perturbation to escape
            if n_ones > 2 and n_zeros > 2:
                i_rand = ones[rng_local.integers(n_ones)] if 'rng_local' in dir() else ones[np.random.randint(n_ones)]
                j_rand = zeros[np.random.randint(n_zeros)]
                h[i_rand] = 0.0
                h[j_rand] = 1.0
                h_pad = np.zeros(2 * pad + 1, dtype=float)
                h_pad[pad:pad + n] = h
                C = np.correlate(h, 1.0 - h, mode='full')
                stagnation = 0
            continue

        best_pos = np.unravel_index(new_max.argmin(), new_max.shape)
        i = ones[best_pos[0]]
        j = zeros[best_pos[1]]

        C += delta[best_pos[0], best_pos[1], :]
        h[i] = 0.0
        h[j] = 1.0
        h_pad[pad + i] = 0.0
        h_pad[pad + j] = 1.0

        new_score = C.max() * dx
        if new_score < best_score:
            best_score = new_score
        stagnation = 0

    return h, best_score

def smooth_refine(h0, max_iter=300, time_limit=None, t_start=None):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = official_score(h)
    lr = 0.03
    patience = 0
    max_patience = 80

    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = corr.max()
        thresh = max_val - 0.08 * max_val
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] += (-h[s:])
            else:
                m = -s
                grad[:n - m] += (1.0 - h[m:])
                grad[m:] += (-h[:n - m])
        grad /= len(active)

        h_new = make_feasible(h - lr * grad)
        score = official_score(h_new)

        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            patience = 0
            lr = min(lr * 1.05, 0.4)
        else:
            h = h_new
            patience += 1
            lr = max(lr * 0.92, 0.003)

        if patience > max_patience:
            break

    return best_h, best_score

def greedy_construct(n):
    h = np.zeros(n)
    ones_needed = n // 2
    for _ in range(ones_needed):
        best_pos = 0
        best_max = np.inf
        for pos in range(n):
            if h[pos] == 1.0:
                continue
            h[pos] = 1.0
            new_C = np.correlate(h, 1.0 - h, mode='full')
            new_max = new_C.max()
            h[pos] = 0.0
            if new_max < best_max:
                best_max = new_max
                best_pos = pos
        h[best_pos] = 1.0
    return h

def generate_starts(n, parent, rng):
    starts = []
    if parent is not None:
        pi = interp_to_n(parent, n)
        starts.append(to_binary(pi))
        starts.append(to_binary(push_binary(pi, 2.0)))
        starts.append(to_binary(push_binary(pi, 4.0)))

    for _ in range(4):
        rb = np.zeros(n)
        rb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(rb)

    for block_size in [2, 3, 4, 5, 6, 8]:
        pattern = np.zeros(n)
        for i in range(n):
            if (i // block_size) % 2 == 0:
                pattern[i] = 1.0
        starts.append(to_binary(pattern))

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    gr = np.zeros(n)
    for i in range(n):
        gr[i] = 1.0 if ((i + 1) * phi) % 1.0 < 0.5 else 0.0
    starts.append(to_binary(gr))

    starts.append(greedy_construct(n))

    # Varying offset periodic
    for offset in [0, 1, 2]:
        pattern = np.zeros(n)
        period = 7
        for i in range(n):
            if ((i + offset) % period) < period // 2:
                pattern[i] = 1.0
        starts.append(to_binary(pattern))

    return starts
```
