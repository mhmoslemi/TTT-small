                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def build_gradient(h, w, n):
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.full(len(V), 1.0) / len(V)
    return w / ws

def pg_step(h, beta, alpha_max=0.7, max_ls=14):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = build_gradient(h, w, n)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def lp_direction(h, k_active=5, max_step=1.0):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    total = 2 * n - 1
    k = min(k_active, total)
    top_idx = np.argsort(-V)[:k]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_rows = []
    b_rows = []
    for p_idx in top_idx:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, V
    
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -max_step)
        hi = min(1.0 - h[i], max_step)
        bounds_list.append((lo, hi))
    bounds_list.append((None, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                return d, V
    except Exception:
        pass
    return None, V

def line_search(h, d, max_alphas=20):
    n = len(h)
    best_f = official_score(h)
    best_h = h.copy()
    
    max_alpha = 10.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

    if np.any(neg):
        max_alpha = min(max_alpha, (-h[neg] / d[neg]).min())
    max_alpha = max(max_alpha, 1e-10)
    
    for i in range(max_alphas):
        frac = 1.0 - 0.12 * i
        if frac < 0.01:
            break
        alpha = max_alpha * frac
        h_new = make_feasible(h + alpha * d)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-15:
            best_f = f_new
            best_h = h_new
    
    return best_h, best_f
