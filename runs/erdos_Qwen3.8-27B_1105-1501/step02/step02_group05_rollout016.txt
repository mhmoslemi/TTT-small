```python
def lp_minimimax_step(h, tol=0.05, delta=0.1):
    """Find a step direction using LP-based minimax.
    
    At current point h, linearize active lags and find direction d
    that maximizes the minimum decrease in active lag values.
    """
    n = len(h)
    vals, grads = compute_lag_gradients_v(h)
    c_max = vals.max()
    
    # Active set: lags within tol of max
    active = np.where(vals >= c_max - tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(vals)])
    
    # Build LP:
    # Variables: d[0..n-1], t
    # Maximize: t
    # Subject to:
    #   grads[active, j] · d >= t for each active lag (i.e., sum_j grads[k,j]*d[j] >= t)
    #   sum(d) = 0
    #   -h[j] <= d[j] <= 1-h[j]
    #   -delta <= d[j] <= delta (trust region)
    
    # cvxpy formulation:
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = []
    for k in active:
        constraints.append(grads[k] @ d_var >= t_var)
    constraints.append(cp.sum(d_var) == 0)
    constraints.append(d_var >= -h)
    constraints.append(d_var <= 1.0 - h)
    constraints.append(d_var >= -delta)
    constraints.append(d_var <= delta)
    
    objective = cp.Maximize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None, None
    
    if d_var.value is None or not np.all(np.isfinite(d_var.value)):
        return None, None
    
    d_opt = np.asarray(d_var.value).flatten()
    t_opt = t_var.value if t_var.value is not None else 0.0
    
    if t_opt <= 1e-10:
        return None, None
    
    return d_opt, t_opt
```

```python
        d = d / d_norm
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25, alpha * 4.0]:
            h_trial = make_feasible(best + mult * d)
            sc = official_score(h_trial)
            if sc < best_sc - 1e-14:
                best = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:
            alpha = min(0.05, alpha * 1.05)
    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_wave(n, freq1, amp1, freq2, amp2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except Exception:
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.3, 3, 0.1)
        best_score = official_score(best_h)

    # Phase 1: LP-based optimization (primary)
    if not is_dead():
        h_new, sc = lp_optimize(best_h, n_steps=400, is_dead=is_dead)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 2: 2-point fine-tuning
    if not is_dead():
        n_tries = max(3000, int((deadline - time.time()) * 2))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 3: LP again with tighter focus
    if not is_dead():
        h_new, sc = lp_optimize(best_h, n_steps=300, is_dead=is_dead,
                                initial_delta=0.15, initial_tol=0.03)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 4: Subgradient polish
    if not is_dead():
        h_new, sc = subgradient_polish(best_h, 200, is_dead)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 5: More 2-point
    if not is_dead():
        n_tries = max(3000, int((deadline - time.time()) * 2))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 6: Try different resolutions
    if not is_dead():
        n_base = len(best_h)
        tried = {n_base}
        candidates = sorted(set(list(range(max(32, n_base - 16), min(88, n_base + 18), 4)) + [36, 44, 50, 56, 64, 72, 80]))
        for nc in candidates:
            if is_dead():
                break
            if nc in tried or nc < 28 or nc > 96:
                continue
            tried.add(nc)
            h_int = interp_to(best_h, nc)
            sc_int = official_score(h_int)
            if not is_dead():
                h_np, sc_np = lp_optimize(h_int, n_steps=60, is_dead=is_dead,
                                          initial_delta=0.2, initial_tol=0.05)
                if sc_np < sc_int:
                    sc_int = sc_np
                    h_int = h_np
            if sc_int < best_score - 1e-14:
                best_h = h_int
                best_score = sc_int

    # Phase 7: Final LP polish on best
    if not is_dead():
        h_new, sc = lp_optimize(best_h, n_steps=200, is_dead=is_dead,
                                initial_delta=0.1, initial_tol=0.02)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 8: Anti-symmetry exploration
    if not is_dead():
        h_sym = anti_sym_map(best_h)
        sc_sym = official_score(h_sym)
        if sc_sym < best_score - 1e-14:
            best_h = h_sym
            best_score = sc_sym
            if not is_dead():
                h_new, sc = lp_optimize(best_h, n_steps=100, is_dead=is_dead)
                if sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc

    # Phase 9: Random perturbation + re-optimize
    if not is_dead():
        n_best = len(best_h)
        for sigma in [0.04, 0.10, 0.18]:
            if is_dead():
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n_best))
            h_new, sc = lp_optimize(h_pert, n_steps=80, is_dead=is_dead)
            if not is_dead():
                h_new2, sc2 = two_point_search(h_new, 800, rng, is_dead)
                if sc2 < sc:
                    h_new, sc = h_new2, sc2
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 10: Final intensive 2-point
    if not is_dead():
        n_tries = max(4000, int((deadline - time.time()) * 4))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 11: Final subgradient
    if not is_dead():
        h_new, sc = subgradient_polish(best_h, 150, is_dead)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-9

    return final_h, final_score, final_n
```
