def lp_step(h_current, n, active_lags, c_current):
    """One LP step: minimize t subject to linearized constraints."""
    half = n // 2
    dx = 2.0 / n
    
    # Variables: params[0..half-1], t
    # Objective: min t
    c_obj = np.zeros(half + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Constraints: for each active lag k
    # corr_k_linearized = corr_k_current + sum_i (d_corr_k/d_params_i) * (params_i - params_i_current)
    # We want: corr_k_linearized <= t
    # => sum_i (d_corr_k/d_params_i) * params_i - t <= corr_k_current - sum_i (d_corr_k/d_params_i) * params_i_current
    
    # Actually: corr_k_current + grad · (params - params_current) <= t
    # => grad · params - t <= corr_k_current - grad · params_current
    
    A_ub = []
    b_ub = []
    
    for k in active_lags:
        # Compute gradient of corr_k with respect to params
        s = k - (n - 1)
        h = h_current  # current full h
        
        grad_params = np.zeros(half)
        for i in range(half):

            # d corr_k / d h[j] = dx * [(1-h[j+s])*1_{j+s valid} + (-h[j-s])*1_{j-s valid}]

            term1 = 0.0
            if 0 <= i + s < n:
                term1 += (1.0 - h[i + s])
            if 0 <= i - s < n:
                term1 += (-h[i - s])
            
            # For h[n-1-i]:
            j = n - 1 - i
            term2 = 0.0
            if 0 <= j + s < n:
                term2 += (1.0 - h[j + s])
            if 0 <= j - s < n:
                term2 += (-h[j - s])

        row = np.zeros(half + 1)
        row[:half] = grad_params
        row[-1] = -1.0  # -t
        
        rhs = c_current[k] - np.dot(grad_params, h_current[:half])

            trial = cur + alpha * (new_params - cur)
            trial = np.clip(trial, 0.0, 1.0)
            s = official_score(antisym_h(trial, n))
            if s < best_s - 1e-12:
                best_s = s
                best_p = trial
        if not np.any(best_p != cur):
            break
        if cur_score - best_s < tol:
            cur = best_p
            cur_score = best_s
            break
        cur = best_p
        cur_score = best_s
    return cur, cur_score

def binary_tabu_search(n, rng, max_iters=20000, tabu_size=50, time_limit=30, t_start=None):
    """Tabu search for binary anti-symmetric solutions (even n only)."""
    if n % 2 != 0:
        return None, None
    half = n // 2
    params = rng.choice([0.0, 1.0], half)
    best_params = params.copy()
    best_score = official_score(antisym_h(best_params, n))
    tabu = {}
    T = 0.03
    for it in range(max_iters):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        i = rng.integers(half)
        flip = 1.0 - params[i]
        key = (i, params[i])
        if key in tabu and tabu[key] > it - tabu_size:
            continue
        save = params[i]
        params[i] = flip
        tabu[key] = it
        s = official_score(antisym_h(params, n))
        if s < best_score:
            best_score = s
            best_params = params.copy()
        elif rng.random() > np.exp(-(s - best_score) / T):
            pass
        else:
            params[i] = save
        T = max(T * 0.9995, 0.001)
    return best_params, best_score

def coord_descent(params, n, max_rounds=100, time_limit=20, t_start=None):
    """Coordinate descent on anti-symmetric params."""
    cur = params.copy()
    cur_score = official_score(antisym_h(cur, n))
    half = len(cur)
    for rnd in range(max_rounds):
        if t_start is not None and time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(half):
            if t_start is not None and time.time() - t_start > time_limit:
                break
            save = cur[i]
            for delta in [0.005, 0.01, 0.02, 0.05, 0.1, -0.005, -0.01, -0.02, -0.05, -0.1]:
                nv = save + delta
                if nv < 0 or nv > 1:
                    continue
                cur[i] = nv
                s = official_score(antisym_h(cur, n))
                if s < cur_score - 1e-12:
                    cur_score = s
                    improved = True
                    break
            cur[i] = save
            # Re-find best for this coordinate
            best_v = save
            best_s = official_score(antisym_h(cur, n))
            cur[i] = save
            lo, hi = max(0.0, save - 0.5), min(1.0, save + 0.5)
            for _ in range(30):
                mid1 = lo + (hi - lo) / 3
                mid2 = hi - (hi - lo) / 3
                cur[i] = mid1
                s1 = official_score(antisym_h(cur, n))
                cur[i] = mid2
                s2 = official_score(antisym_h(cur, n))
                if s1 < s2:
                    hi = mid2
                else:
                    lo = mid1
                if s1 < best_s:
                    best_s = s1
                    best_v = mid1
                if s2 < best_s:
                    best_s = s2
                    best_v = mid2
            cur[i] = best_v
            if best_s < cur_score:
                cur_score = best_s
                improved = True
        if not improved:
            break
    return cur, cur_score

def slsq_antisym(params, n, beta=100, maxiter=80):
    try:
        res = minimize(
            lse_obj, params, args=(n, beta),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * len(params),
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return params.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(48, 0.5)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    n_list = [50, 52, 54, 56]
    pn = len(parent)
    if pn not in n_list and 40 <= pn <= 68:
        n_list.append(pn)
    if pn % 2 == 0 and pn - 2 >= 40:
        n_list.append(pn - 2)
    n_list = sorted(set(n_list))
    n_list = [n for n in n_list if 40 <= n <= 68]

    for n in n_list:
        if time.time() > deadline - 40:
            break
        starts = []
        h_interp = make_feasible(interp_to_n(parent, n))
        h_as = project_antisym(h_interp, n)
        starts.append(get_params(h_as, n))
        hb = make_feasible(h_interp ** 2.0)
        starts.append(get_params(project_antisym(hb, n), n))
        if n % 2 == 0:
            half = n // 2
            starts.append(rng.choice([0.0, 1.0], half))
        starts.append(rng.uniform(0.0, 1.0, n // 2))
        h_flip = project_antisym(1.0 - h_interp[::-1], n)
        starts.append(get_params(h_flip, n))
        for p_val in [1.3, 2.5, 5.0]:
            hb2 = make_feasible(h_as ** p_val)
            starts.append(get_params(project_antisym(hb2, n), n))

        for p0 in starts:
            if time.time() > deadline - 25:
                break
            cur = np.clip(p0.copy(), 0.0, 1.0)
            cur_s = official_score(antisym_h(cur, n))

            # Phase 1: Binary tabu (if even n and solution is near-binary)
            if n % 2 == 0 and time.time() < deadline - 20:
                tp, ts = binary_tabu_search(n, rng, max_iters=15000, tabu_size=40,
                                            time_limit=25, t_start=time.time())
                if tp is not None and ts < cur_s - 1e-10:
                    cur = tp
                    cur_s = ts

            # Phase 2: LP active-set iteration
            if time.time() < deadline - 15:
                lp_p, lp_s = lp_iteration(cur, n, max_iters=300, tol=1e-11)
                if lp_s < cur_s - 1e-12:
                    cur = lp_p
                    cur_s = lp_s

            # Phase 3: SLSQP with LSE (multiple betas)
            for beta in [30, 150, 600]:
                if time.time() > deadline - 10:
                    break
                sp = slsq_antisym(cur, n, beta=beta, maxiter=60)
                ss = official_score(antisym_h(sp, n))
                if ss < cur_s - 1e-12:
                    cur = sp
                    cur_s = ss
                    # Follow up with LP
                    if time.time() < deadline - 10:
                        lp_p2, lp_s2 = lp_iteration(cur, n, max_iters=100, tol=1e-11)
                        if lp_s2 < cur_s - 1e-12:
                            cur = lp_p2
                            cur_s = lp_s2

            # Phase 4: Coordinate descent
            if time.time() < deadline - 8:
                cd_p, cd_s = coord_descent(cur, n, max_rounds=50,
                                           time_limit=15, t_start=time.time())
                if cd_s < cur_s - 1e-12:
                    cur = cd_p
                    cur_s = cd_s

            # Phase 5: Another LP pass
            if time.time() < deadline - 6:
                lp_p3, lp_s3 = lp_iteration(cur, n, max_iters=200, tol=1e-11)
                if lp_s3 < cur_s - 1e-12:
                    cur = lp_p3
                    cur_s = lp_s3

            # Evaluate
            cur_h = antisym_h(cur, n)
            final_s = official_score(cur_h)
            if final_s < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = final_s
                best_n = n

    # Additional refinement on best solution
    remaining = deadline - time.time()
    if remaining > 15:
        # Try small perturbations and re-optimize
        n_b = best_n
        p_b = get_params(project_antisym(best_h, n_b), n_b)
        n_trials = min(10, int(remaining / 12))
        for trial in range(n_trials):
            if time.time() > deadline - 8:
                break
            sigma = 0.02 + 0.01 * trial
            p_pert = np.clip(p_b + sigma * rng.standard_normal(len(p_b)), 0.0, 1.0)
            lp_p, lp_s = lp_iteration(p_pert, n_b, max_iters=150, tol=1e-11)
            if lp_s < best_score - 1e-12:
                best_h = antisym_h(lp_p, n_b)
                best_score = lp_s
                p_b = lp_p
        # Final LP polish
        p_final = get_params(project_antisym(best_h, best_n), best_n)
        lp_pf, lp_sf = lp_iteration(p_final, best_n, max_iters=300, tol=1e-12)
        if lp_sf < best_score:
            best_h = antisym_h(lp_pf, best_n)
            best_score = lp_sf

    # If best_n is odd, try non-antisymmetric subgradient
    if best_n % 2 == 1 and time.time() < deadline - 10:
        h = best_h.copy()
        n = best_n
        alpha = 0.02
        for it in range(2000):
            if time.time() > deadline - 5:
                break
            corr = corr_all(h, n)
            m = np.max(corr)
            thresh = m - 0.02 * m
            active = np.where(corr >= thresh)[0]
            dx = 2.0 / n
            grad = np.zeros(n)
            for k in active:
                s = k - (n - 1)
                g = np.zeros(n)
                for j in range(n):
                    t1 = 0.0
                    if 0 <= j + s < n:
                        t1 += (1.0 - h[j + s])
                    if 0 <= j - s < n:
                        t1 += (-h[j - s])
                    g[j] = t1
                grad += dx * g
            grad /= len(active)
            h_new = make_feasible(h - alpha * grad)
            if official_score(h_new) < best_score:
                best_score = official_score(h_new)
                best_h = h_new
                h = h_new
            else:
                h = h_new
                alpha *= 0.995

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
