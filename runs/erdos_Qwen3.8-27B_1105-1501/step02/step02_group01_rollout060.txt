def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)

        target = n / 2.0

    lo = -3.0 - np.max(z)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:

        else:

            else:

            if abs(add) < 1e-15:

            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

                C = np.correlate(v, comp, mode='full')
                D = np.correlate(v, v, mode='full')
                diff = C - B
                t_vals = np.linspace(0, t_max, n_grid)
                scores_mat = A[None,:] + t_vals[:,None]*diff[None,:] - (t_vals[:,None]**2)*D[None,:]
                max_s = np.max(scores_mat, axis=1)
                idx = np.argmin(max_s)
                if max_s[idx] < best_new_score:
                    best_new_score = max_s[idx]
                    best_new_h = h.copy()
                    best_new_h[j] += t_vals[idx]
                    best_new_h[i] -= t_vals[idx]
        
        if best_new_h is not None and best_new_score < current_max - 1e-12:
            h = np.clip(best_new_h, 0.0, 1.0)
            # Fix sum if needed
            s = np.sum(h)
            if abs(s - n/2.0) > 1e-10:
                diff_sum = n/2.0 - s
                # Add to a random interior point
                interior = np.where((h > 1e-6) & (h < 1-1e-6))[0]
                if len(interior) > 0:
                    h[interior[0]] += diff_sum
                    h = np.clip(h, 0.0, 1.0)
            new_score = exact_score(h)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -4.0 - np.max(z)
    hi = 4.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
            else:
                add = max(residual, -h[idx])
            if abs(add) < 1e-15:
                break
            h[idx] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def push_binary(h, p=2.0):
    return make_feasible(np.clip(h, 0, 1) ** p)

def pair_move_search(h, n_grid=97, max_iter=800, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy()
    best_score = exact_score(h)
    best_h = h.copy()
    no_improve_count = 0

    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break

        comp = 1.0 - h
        A = np.correlate(h, comp, mode='full')
        current_max = np.max(A)

        donors = np.where(h > 1e-10)[0]
        receivers = np.where(h < 1.0 - 1e-10)[0]
        if len(donors) == 0 or len(receivers) == 0:
            break

        best_new_score = current_max
        best_move = None

        for i in donors:
            hi = h[i]
            for j in receivers:
                if i == j:
                    continue
                t_max = min(hi, 1.0 - h[j])
                if t_max <= 1e-10:
                    continue
                v = np.zeros(n)
                v[j] = 1.0
                v[i] = -1.0
                B = np.correlate(h, v, mode='full')
                C = np.correlate(v, comp, mode='full')
                D = np.correlate(v, v, mode='full')
                diff = C - B
                t_vals = np.linspace(0.0, t_max, n_grid)
                scores_mat = A[None, :] + t_vals[:, None] * diff[None, :] - (t_vals[:, None] ** 2) * D[None, :]
                max_s = np.max(scores_mat, axis=1)
                idx = np.argmin(max_s)
                if max_s[idx] < best_new_score:
                    best_new_score = max_s[idx]
                    best_move = (i, j, t_vals[idx])

        if best_move is not None and best_new_score < current_max - 1e-13:

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    L = 2 * n - 1
    if s < 1e-300:
        weights = np.ones(L) / L
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(L):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s2 = k - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n - s2])
                grad[:n - s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n - m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n - m2])
    return grad

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0, args=(beta, dx),
            jac=smooth_lse_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0
