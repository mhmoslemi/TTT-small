import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def pair_eval(h, i, j, A, B, C):
    """Evaluate objective at t for pair (i,j) using precomputed A,B,C."""
    n = len(h)
    dx = 2.0 / n
    lower = max(h[i] - 1.0, -h[j])
    upper = min(h[i], 1.0 - h[j])
    if upper - lower < 1e-12:
        return 0.0, np.max(A) * dx
    # Ternary search
    a, b = lower, upper
    for _ in range(60):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        v1 = A + m1 * B + m1 * m1 * C
        v2 = A + m2 * B + m2 * m2 * C
        if np.max(v1) < np.max(v2):
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    vals = A + t_opt * B + t_opt * t_opt * C
    return t_opt, np.max(vals) * dx

def compute_pair_B(h, i, j):
    """Compute B vector for pair (i,j)."""
    n = len(h)
    v = np.zeros(n)
    v[i] = -1.0
    v[j] = 1.0
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    return B, C

def pair_local_search(h, max_rounds=300, n_try=12, time_limit=None, t_start=None):
    """Exact pair-move local search."""
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    best_score = official_score(h)
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        A = corr
        k = int(np.argmax(corr))
        
        # Compute g(m) for directional derivative at active lag
        g = np.zeros(n)
        for m in range(n):
            idx1 = m + k - (n - 1)
            idx2 = m + (n - 1) - k
            val1 = (1.0 - h[idx1]) if 0 <= idx1 < n else 0.0
            val2 = h[idx2] if 0 <= idx2 < n else 0.0
            g[m] = val1 - val2
        
        # Find candidate pairs: want g(j) - g(i) to be most negative
        # i should have large g(i), j should have small g(j), h[i] > h[j]
        order_desc = np.argsort(g)[::-1]
        order_asc = np.argsort(g)
        
        best_improve = 0
        best_move = None
        
        # Try top candidates
        tried = set()
        for ii in range(min(n_try, n)):
            i = order_desc[ii]
            for jj in range(min(n_try, n)):
                j = order_asc[jj]
                if i == j or h[i] <= h[j] + 1e-8:
                    continue
                key = (i, j)
                if key in tried:
                    continue
                tried.add(key)
                B, C = compute_pair_B(h, i, j)
                t_opt, new_score = pair_eval(h, i, j, A, B, C)
                if new_score < best_score - best_improve - 1e-13:
                    best_improve = best_score - new_score
                    best_move = (i, j, t_opt)
        
        # If no improvement from top candidates, try broader search
        if best_move is None:
            # Try all pairs with negative derivative
            best_improve = 0
            best_move = None
            for i in range(n):
                for j in range(n):
                    if i == j or h[i] <= h[j] + 1e-8:
                        continue
                    deriv = g[j] - g[i]
                    if deriv > 1e-10:
                        continue
                    key = (i, j)
                    if key in tried:
                        continue
                    tried.add(key)
                    B, C = compute_pair_B(h, i, j)
                    t_opt, new_score = pair_eval(h, i, j, A, B, C)
                    if new_score < best_score - best_improve - 1e-13:
                        best_improve = best_score - new_score
                        best_move = (i, j, t_opt)
        
        if best_move is not None:
            i, j, t = best_move
            h[i] -= t
            h[j] += t
            h = make_feasible(h)
            new_actual = official_score(h)
            if new_actual < best_score - 1e-13:
                best_score = new_actual
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
    
    return h, best_score

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    L = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(L) / L
    else:
        weights = e / s
    grad = np.zeros(n)
    for kk in range(L):
        w = weights[kk] * dx
        if w < 1e-14:
            continue
        s2 = kk - (n - 1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n - s2])
                grad[:n - s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n - m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n - m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def do_slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0, args=(beta, dx),
            jac=smooth_lse_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def do_slsqp_direct(h0, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    t0 = official_score(h0) + 1e-4
    if t0 > 0.99:
        t0 = 0.99
    z0 = np.concatenate([h0, [t0]])
    L = 2 * n - 1
    
    def mm_obj(z):
        return z[-1]
    
    def mm_obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def mm_ineq(z):
        hh = z[:-1]
        t = z[-1]
        corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
        return t - corr
    
    def mm_ineq_jac(z):
        hh = z[:-1]
        jac = np.zeros((L, n + 1))
        jac[:, -1] = 1.0
        for kk in range(L):
            s2 = kk - (n - 1)
            if s2 >= 0:
                if s2 == 0:
                    for jj in range(n):
                        jac[kk, jj] = dx * (2.0 * hh[jj] - 1.0)
                else:
                    for jj in range(n - s2):
                        jac[kk, jj + s2] = dx * (1.0 - 2.0 * hh[jj + s2])
                        jac[kk, jj] = dx * hh[jj]
            else:
                m2 = -s2
                if m2 == 0:
                    for jj in range(n):
                        jac[kk, jj] = dx * (2.0 * hh[jj] - 1.0)
                else:
                    for jj in range(n - m2):
                        jac[kk, jj] = dx * (1.0 - 2.0 * hh[jj + m2])
                        jac[kk, jj + m2] = dx * hh[jj]
        return jac
    
    def mm_eq(z):
        return np.sum(z[:-1]) - n / 2.0
    
    def mm_eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(
            mm_obj, z0, jac=mm_obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
                {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'][:-1])
    except Exception:
        pass
    return h0

def simulated_annealing(h0, rng, n_swaps=15000, t_start=None, time_limit=None):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    T = 0.03
    T_end = 0.002
    cool_rate = (T_end / T) ** (1.0 / max(n_swaps, 1))
    
    for it in range(n_swaps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if h[i] >= h[j]:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        delta = min(h[donor], 1.0 - h[receiver])
        if delta < 0.005:
            continue
        swap = delta * rng.uniform(0.3, 1.0)
        h[donor] -= swap
        h[receiver] += swap
        new_score = official_score(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
        else:
            if rng.random() > np.exp(-(new_score - best_score) / T):
                h[donor] += swap
                h[receiver] -= swap
        T *= cool_rate
    return best_h, best_score

def push_binary(h, p=2.0):
    return make_feasible(h ** p)

                )
            
            # Phase 2: SLSQP smooth
            for beta in [80, 300]:
                if time.time() > deadline - 20:
                    break
                new_h = do_slsqp_smooth(cur_h, beta, maxiter=120)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-12:
                    cur_h = new_h
                    cur_score = new_score
            
            # Phase 3: More pair search after smoothing
            if time.time() < deadline - 20:
                cur_h, cur_score = pair_local_search(
                    cur_h, max_rounds=300, n_try=12,
                    time_limit=min(40, deadline - 20 - time.time()),
                    t_start=time.time()
                )
            
            # Phase 4: Simulated annealing for diversity
            if time.time() < deadline - 25:
                sa_h, sa_score = simulated_annealing(
                    cur_h, rng, n_swaps=12000,
                    time_limit=min(30, deadline - 25 - time.time()),
                    t_start=time.time()
                )
                if sa_score < cur_score - 1e-10:
                    cur_h = sa_h
                    cur_score = sa_score
                    # Polish with pair search
                    if time.time() < deadline - 15:
                        cur_h, cur_score = pair_local_search(
                            cur_h, max_rounds=200, n_try=10,
                            time_limit=min(25, deadline - 15 - time.time()),
                            t_start=time.time()
                        )
            
            # Phase 5: Direct min-max SLSQP
            if time.time() < deadline - 15:
                dc_h = do_slsqp_direct(cur_h, maxiter=80)
                dc_score = official_score(dc_h)
                if dc_score < cur_score - 1e-12:
                    cur_h = dc_h
                    cur_score = dc_score
                # Pair polish after direct
                if time.time() < deadline - 10:
                    cur_h, cur_score = pair_local_search(
                        cur_h, max_rounds=150, n_try=8,
                        time_limit=min(20, deadline - 10 - time.time()),
                        t_start=time.time()
                    )
            
            # Update global best
            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
            elif abs(cur_score - best_score) <= 1e-11 and len(cur_h) < best_n:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
    
    # Final refinement on the best solution
    remaining = deadline - time.time()
    if remaining > 20:
        # Try perturbing the best and re-optimizing
        for trial in range(min(5, int(remaining / 20))):
            if time.time() > deadline - 10:
                break
            h_pert = make_feasible(best_h + 0.02 * rng.standard_normal(best_n))
            h_pert, s_pert = pair_local_search(
                h_pert, max_rounds=200, n_try=10,
                time_limit=min(15, deadline - 10 - time.time()),
                t_start=time.time()
            )
            if s_pert < best_score - 1e-12:
                best_h = h_pert
                best_score = s_pert
    
    # Final polish
    if time.time() < deadline - 5:
        best_h, best_score = pair_local_search(
            best_h, max_rounds=200, n_try=15,
            time_limit=min(20, deadline - 5 - time.time()),
            t_start=time.time()
        )
    
    # One more SLSQP pass
    if time.time() < deadline - 3:
        fh = do_slsqp_smooth(best_h, 500, maxiter=80)
        fs = official_score(fh)
        if fs < best_score - 1e-12:
            best_h = fh
            best_score = fs
        
        # Final pair polish
        if time.time() < deadline - 2:
            best_h, best_score = pair_local_search(
                best_h, max_rounds=100, n_try=10,
                time_limit=max(1.0, deadline - 2 - time.time()),
                t_start=time.time()
            )
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
