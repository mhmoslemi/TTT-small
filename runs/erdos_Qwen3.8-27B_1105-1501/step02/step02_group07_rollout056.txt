def active_set_lp_v2(h, eps, trust):
    n = len(h)
    dx = 2.0 / n
    V = compute_V(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    
    if len(active) == 0:
        return None, None
    
    # Minimize t subject to:
    # dx * g_k · d <= t - V_k for k in active
    # sum(d) = 0
    # |d[i]| <= trust
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_rows = []
    b_rows = []
    
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] += 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t_shift = -s
            gp[0:L] += 1.0 - h[t_shift:n]
            gp[t_shift:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    # Bounds: d[i] in [-min(h[i], trust), min(1-h[i], trust)]
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:

            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_V(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _1d_min_from_V(h, j, V):
    n = len(h)
    dx = 2.0 / n
    h_old = h[j]
    K = 2 * n - 1
    a = np.zeros(K)
    b = np.zeros(K)
    for idx in range(K):
        k = idx - (n - 1)
        a[idx] = 0.0
        if 0 <= j + k < n:
            a[idx] += dx * (1.0 - h[j + k])
        if 0 <= j - k < n:
            a[idx] += dx * (-h[j - k])
        b[idx] = V[idx] - a[idx] * h_old
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = (a * m1 + b).max()
        v2 = (a * m2 + b).max()
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    x_opt = (lo + hi) / 2.0
    v_opt = (a * x_opt + b).max()
    return x_opt, v_opt

def coord_optimize(h, max_sweeps=300, tol=1e-11):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for sweep in range(max_sweeps):
        V = compute_V(h)
        max_change = 0.0
        for j in range(n):
            new_j, _ = _1d_min_from_V(h, j, V)
            delta = new_j - h[j]
            if abs(delta) > max_change:
                max_change = abs(delta)
            h[j] = new_j
            r = target - np.sum(h)
            if abs(r) > 1e-14:
                if r > 0:
                    room = 1.0 - h
                else:
                    room = h
                room[j] = 0.0
                total = room.sum()
                if total > 1e-15:
                    h += r * (room / total)
        if max_change < tol:
            break
    return make_feasible(h)

def active_set_lp(h, eps, trust):
    n = len(h)
    dx = 2.0 / n
    V = compute_V(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] += 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t_sh = -s
            gp[0:L] += 1.0 - h[t_sh:n]
            gp[t_sh:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    bounds_list = []
    for i in range(n):
        lo_b = max(-h[i], -trust)
        hi_b = min(1.0 - h[i], trust)
        bounds_list.append((lo_b, hi_b))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) < 1e-13:
                return None, None
            h_new = make_feasible(h + d)
            sc = official_score(h_new)
            return h_new, sc
    except Exception:
        pass
    return None, None

def lp_polish(h, budget_fn, max_iter=5000):
    n = len(h)
    trust = 0.25
    best_h = h.copy()
    best_score = official_score(h)
    h_cur = h.copy()
    no_improve = 0
    for it in range(max_iter):
        if budget_fn() < 15:
            break
        eps = 0.002 if trust < 0.03 else 0.005
        h_new, sc = active_set_lp(h_cur, eps, trust)
        if h_new is not None:
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                no_improve = 0
                trust = min(trust * 1.08, 0.4)
            else:
                no_improve += 1
                trust *= 0.6
            h_cur = h_new
        else:
            if trust > 0.002:
                trust *= 0.5
            else:
                break
            no_improve += 1
        if no_improve > 80:
            break
    return best_h, best_score

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def pg_step(h, beta, alpha_max=0.8, max_ls=14):
    n = len(h)
    V = compute_V(h)
    f = V.max()
    V_shifted = beta * (V - f)
    V_shifted = np.clip(V_shifted, -500.0, 0.0)
    w = np.exp(V_shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-7:
            break
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(10.0, 0.04 * budget_s)
    deadline = t0 + budget_s - reserve

    def time_left():
        return deadline - time.time()

    rng = np.random.default_rng(seed)

    try:
        h = np.asarray(initial_h_values, dtype=float).ravel()
        h = make_feasible(h)
    except (NameError, AttributeError):
        n = 60
        x = np.linspace(0, 1, n, endpoint=False)
        h = 0.5 + 0.15 * np.sin(2 * np.pi * x) + 0.05 * np.sin(4 * np.pi * x)
        h = make_feasible(h)

    best_h = h.copy()
    best_score = official_score(h)

    # Phase 1: Coordinate descent for fast initial convergence
    if time_left() > 200:
        h_cd = coord_optimize(h, max_sweeps=200)
        sc_cd = official_score(h_cd)
        if sc_cd < best_score:
            best_h = h_cd
            best_score = sc_cd

    # Phase 2: PG with adaptive beta to refine
    if time_left() > 300:
        h_cur = best_h.copy()
        beta = 30.0
        stall = 0
        while time_left() > 250:
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.7, max_ls=14)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
                stall = 0
            elif sc < official_score(h_cur) - 1e-13:
                stall = 0
            else:
                stall += 1
            h_cur = h_new
            beta = min(beta * 1.05, 300.0)
            if stall > 60:
                h_cur = make_feasible(best_h + rng.normal(0, 0.02, len(best_h)))
                beta = 20.0
                stall = 0

    # Phase 3: Aggressive LP polishing (main workhorse)
    h_cur = best_h.copy()
    best_h, best_score = lp_polish(h_cur, time_left, max_iter=5000)

    # Phase 4: Multi-resolution search
    n_base = len(best_h)
    n_candidates = sorted(set([
        max(25, n_base - 15), max(30, n_base - 10),
        n_base, min(90, n_base + 10), min(100, n_base + 20)
    ]))

    for n_try in n_candidates:
        if time_left() < 80:
            break
        if n_try == n_base:
            continue
        h_int = interpolate_to(best_h, n_try)
        sc_int = official_score(h_int)
        # Refine at this resolution
        if time_left() > 100:
            h_cd2 = coord_optimize(h_int, max_sweeps=100)
            sc_cd2 = official_score(h_cd2)
            if sc_cd2 < sc_int:
                h_int = h_cd2
                sc_int = sc_cd2
        if time_left() > 60:
            h_pol, sc_pol = lp_polish(h_int, time_left, max_iter=1500)
            if sc_pol < sc_int:
                h_int = h_pol
                sc_int = sc_pol
        if sc_int < best_score:
            best_h = h_int.copy()
            best_score = sc_int

    # Phase 5: Final LP polish on best found
    if time_left() > 30:
        h_cur = best_h.copy()
        best_h, best_score = lp_polish(h_cur, time_left, max_iter=3000)

    # Phase 6: Very fine PG polish
    if time_left() > 20:
        h_cur = best_h.copy()
        beta = 200.0
        for _ in range(300):
            if time_left() < 15:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.3, max_ls=10)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.01, 800.0)

    # Phase 7: Final LP micro-polish
    if time_left() > 10:
        h_cur = best_h.copy()
        lv_ = 0.02
        for _ in range(200):
            if time_left() < 8:
                break
            h_new, sc = active_set_lp(h_cur, 0.001, lv_)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
                h_cur = h_new
                lv_ = min(lv_ * 1.1, 0.1)
            else:
                lv_ = max(lv_ * 0.5, 0.001)

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
