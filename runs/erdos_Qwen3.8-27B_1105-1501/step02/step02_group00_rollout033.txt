import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_score_matrix(var, fixed, n, var_is_h):
    """Build matrix A where score[r] = sum_j A[r,j] * var[j].
    var_is_h: True if var is h and fixed is g=1-h; False if var is g and fixed is h.
    """
    A = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            if var_is_h:
                # score = sum_{j=d}^{n-1} var[j] * fixed[j-d]
                A[r, d:] = fixed[:m]
            else:
                # score = sum_{j=0}^{m-1} fixed[j+d] * var[j]
                A[r, :m] = fixed[d:]
        else:
            k = -d
            m = n - k
            if m <= 0:
                continue
            if var_is_h:
                # score = sum_{j=0}^{m-1} var[j] * fixed[j+k]
                A[r, :m] = fixed[k:]
            else:
                # score = sum_{j=k}^{n-1} fixed[j-k] * var[j]
                A[r, k:] = fixed[:m]
    return A

def _solve_block_lp(var, fixed, n, target, var_is_h):

    A = _build_score_matrix(var, fixed, n, var_is_h)

    c[-1] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0

            break
        
        g = 1.0 - h
        c5_before = _exact_c5(h)
        
        # Step A: optimize h (fix g)
        h_new = _solve_block_lp(h, g, n, target, True)
        if h_new is not None:
            h_candidate = _project(h_new, target)
            c5_a = _exact_c5(h_candidate)
            if c5_a < c5_before:
                h = h_candidate
                c5_before = c5_a
        
        # Step B: optimize g (fix h)
        g = 1.0 - h
        g_new = _solve_block_lp(g, h, n, target, False)
        if g_new is not None:
            g_candidate = _project(g_new, target)
            h_candidate = 1.0 - g_candidate
            h_candidate = _project(h_candidate, target)
            c5_b = _exact_c5(h_candidate)
            if c5_b < c5_before:
                h = h_candidate
                c5_before = c5_b
        
        # Update best
        if c5_before < best_c5:
            best_c5 = c5_before
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def _lse_gradient_descent(h, deadline, T_init=20.0, T_final=200.0, max_iters=2000):
    """Log-sum-exp smoothed gradient descent."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    T = T_init
    lr = 0.01
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 100:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = np.max(scores)
        
        # Soft-max weights
        shifted = scores - max_s
        w = np.exp(T * shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(scores)] = 1.0
        else:
            w /= w_sum
        
        # Gradient: weighted average of individual score gradients
        # score[r] = sum_j A[r,j] * h[j] (approximately, using current 1-h)
        # ds/dh[j] for each r is A[r,j] where A depends on current h
        # Actually, let's compute it properly
        # score[r] is the r-th element of correlate(h, 1-h)
        # d(score[r])/dh[j] = contribution of h[j] to score[r]
        
        # For shift d = r - (n-1):
        # If d >= 0: score = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        #   d/dh[j]: if j >= d and j-d < n-d (i.e., j-d < n-1-d+1... always true for valid j)
        #   d/dh[j] = (1-h[j-d]) if j >= d (from h[j] term with i=j-d)
        #            - h[j+d] if j+d < n (from -h[j] in the (1-h) term with i+d=j, i=j-d... wait)
        
        # Let me redo this more carefully.
        # score[r] = correlate(h, 1-h)[r]
        # For d >= 0: score = sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
        #   = sum_{i=0}^{n-1-d} h[i+d] - sum_{i=0}^{n-1-d} h[i+d]*h[i]
        #   d/dh[j]:
        #     From first term sum: +1 if j in [d, n-1] (i.e., j = i+d for some valid i)
        #     From second term: -h[j-d] if j >= d and j-d < n-d (i.e., j in [d, n-1])... 
        #       Actually h[i+d]*h[i]: if j = i+d, contributes -h[i] = -h[j-d]
        #       if j = i, contributes -h[i+d] = -h[j+d] (if j+d < n)
        
        # This is getting complicated. Let me just use the G matrix approach.
        
        # Build gradient matrix G where G[r, j] = d(score[r])/dh[j]
        # For d >= 0 (r = n-1+d):
        #   score = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        #   G[r, j] = (1-h[j-d]) - h[j+d] if j >= d and j < n  ... no wait
        #   Let me be very explicit:
        #   score = sum_{i=0}^{m-1} h[i+d] * (1-h[i]) where m = n-d
        #   = sum_{i=0}^{m-1} h[i+d] - sum_{i=0}^{m-1} h[i+d]*h[i]
        #   d/dh[j] (for j in 0..n-1):
        #     First term: d/dh[j] sum_{i=0}^{m-1} h[i+d] = 1 if j-d in [0,m-1] i.e. j in [d, d+m-1] = [d, n-1]
        #     Second term: d/dh[j] sum_{i=0}^{m-1} h[i+d]*h[i]
        #       = sum over i where i+d=j: h[i] = h[j-d] (if j-d in [0,m-1])
        #         + sum over i where i=j: h[i+d] = h[j+d] (if j in [0,m-1] and j+d in [d, d+m-1]=[d,n-1])
        #       = h[j-d] (if j >= d) + h[j+d] (if j < m = n-d, i.e., j+d < n)
        #     So G[r,j] = [1 if j >= d] - h[j-d] (if j >= d) - h[j+d] (if j < n-d)
        #     = (1 - h[j-d]) if j >= d, minus h[j+d] if j < n-d
        
        # Hmm, this is getting complex. Let me just build it numerically.
        
        # Actually, for the gradient of the LSE objective, I can use:
        # grad[j] = sum_r w[r] * G[r, j]
        # where G is the Jacobian of scores w.r.t. h.
        
        # Let me build G efficiently.
        G = np.zeros((2*n-1, n))
        for r in range(2*n-1):
            d = r - (n-1)
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                # score = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
                # G[r, j] for j in [d, n-1]:
                #   from h[i+d] term: +1 (when i = j-d)
                #   from -h[i+d]*h[i] term: -h[j-d] (when i = j-d, j >= d)
                #   from -h[i+d]*h[i] term: -h[j+d] (when i = j, need j < m)
                # So G[r, j] = (1 - h[j-d]) - h[j+d]*[j < n-d] for j >= d
                # Wait, I need to be more careful.
                # score = sum_{i=0}^{m-1} h[i+d] - sum_{i=0}^{m-1} h[i+d]*h[i]
                # Term 1: sum_{i=0}^{m-1} h[i+d], derivative w.r.t. h[j] is 1 if j in [d, n-1]
                # Term 2: sum_{i=0}^{m-1} h[i+d]*h[i]
                #   d/dh[j]: 
                #     h[i+d] depends on h[j] when i+d = j, i.e., i = j-d, valid if 0 <= j-d <= m-1, i.e., d <= j <= d+m-1 = n-1
                #       contribution: h[i] = h[j-d]
                #     h[i] depends on h[j] when i = j, valid if 0 <= j <= m-1
                #       contribution: h[i+d] = h[j+d]
                #   So d/dh[j] of term 2 = h[j-d]*[d<=j<=n-1] + h[j+d]*[0<=j<=m-1]
                # G[r,j] = [d<=j<=n-1] - h[j-d]*[d<=j<=n-1] - h[j+d]*[0<=j<=m-1]
                #        = (1-h[j-d])*[j>=d] - h[j+d]*[j < n-d]
                
                js = np.arange(d, n)
                G[r, js] = 1.0 - h[js - d]
                js2 = np.arange(0, m)
                G[r, js2] -= h[js2 + d]
            else:
                k = -d
                m = n - k
                if m <= 0:
                    continue
                # score = sum_{i=0}^{m-1} h[i]*(1-h[i+k])
                # = sum_{i=0}^{m-1} h[i] - sum_{i=0}^{m-1} h[i]*h[i+k]
                # Term 1: d/dh[j] = 1 if j in [0, m-1]
                # Term 2: d/dh[j]:
                #   h[i] when i=j, valid if j < m: contribution h[j+k]
                #   h[i+k] when i+k=j, i=j-k, valid if 0<=j-k<m i.e. k<=j<=k+m-1=n-1: contribution h[j-k]
                # G[r,j] = [j<m] - h[j+k]*[j<m] - h[j-k]*[j>=k]
                #        = (1-h[j+k])*[j<m] - h[j-k]*[j>=k]
                
                js = np.arange(0, m)
                G[r, js] = 1.0 - h[js + k]
                js2 = np.arange(k, n)
                G[r, js2] -= h[js2 - k]
        
        # Gradient of LSE objective
        grad = (w @ G) * dx  # shape (n,)
        
        # Line search
        lr_try = lr
        best_f = _exact_c5(h)
        best_h_step = h.copy()
        for a in [lr_try, lr_try*0.5, lr_try*0.25, lr_try*0.1, lr_try*0.01]:
            h_trial = _project(h - a * grad, target)
            f_trial = _exact_c5(h_trial)
            if f_trial < best_f:
                best_f = f_trial
                best_h_step = h_trial
        
        if best_f < best_c5 - 1e-12:
            h = best_h_step
            if best_f < best_c5:
                best_c5 = best_f
                best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.1, 0.1)
        else:
            no_improve += 1
            lr *= 0.7
            if lr < 1e-6:
                break
        
        # Gradually increase T
        if it % 20 == 0:
            T = min(T * 1.05, T_final)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_linearize(h, deadline, radius_init=0.1, max_iters=2000):
    """Standard LP linearization approach (similar to previous program)."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 30:
            break
        
        g = 1.0 - h
        scores = np.correlate(h, g, mode='full')
        
        # Build gradient matrix
        G = np.zeros((2*n-1, n))
        for r in range(2*n-1):
            d = r - (n-1)
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                js = np.arange(d, n)
                G[r, js] = 1.0 - g[js - d]  # 1-h[j-d] = g[j-d]
                js2 = np.arange(0, m)
                G[r, js2] -= h[js2 + d]
            else:
                k = -d
                m = n - k
                if m <= 0:
                    continue
                js = np.arange(0, m)
                G[r, js] = 1.0 - g[js + k]
                js2 = np.arange(k, n)
                G[r, js2] -= h[js2 - k]
        
        # LP: find direction p that reduces the worst score
        # min t s.t. G @ (h + p) <= scores + t*... 
        # Actually: minimize max_r (scores[r] + G[r,:] @ p)
        # = minimize t s.t. G[r,:] @ p <= t - scores[r] for all r
        # with ||p||_inf <= radius, sum(p) = 0
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores  # G@p - t <= -scores, i.e., scores + G@p - t <= 0
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                if np.max(np.abs(p)) > 1e-12:
                    # Line search
                    alpha_max = 0.5
                    for j in range(n):
                        if p[j] > 1e-14:
                            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                        elif p[j] < -1e-14:
                            alpha_max = min(alpha_max, h[j] / (-p[j]))
                    alpha_max = max(alpha_max, 1e-8)
                    
                    best_a = 0.0
                    best_c5_try = _exact_c5(h)
                    for a in np.linspace(0, alpha_max, 20):
                        if a < 1e-10:
                            continue
                        h_t = _project(h + a * p, target)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5_try:
                            best_c5_try = c5_t
                            best_a = a
                    
                    if best_a > 0 and best_c5_try < best_c5 - 1e-12:
                        h = _project(h + best_a * p, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                        no_improve = 0
                        radius = min(radius * 1.1, 0.3)
                        continue
        except Exception:
            pass
        
        no_improve += 1
        radius *= 0.8
        if radius < 1e-4:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad_matrix(h, g):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            js = np.arange(d, n)
            G[r, js] = g[js - d]
            js2 = np.arange(0, m)
            G[r, js2] -= h[js2 + d]
        else:
            k = -d
            m = n - k
            if m <= 0:
                continue
            js = np.arange(0, m)
            G[r, js] = g[js + k]
            js2 = np.arange(k, n)
            G[r, js2] -= h[js2 - k]
    return G

def _build_block_matrix(fixed, n, var_is_h):
    A = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            if var_is_h:
                A[r, d:] = fixed[:m]
            else:
                A[r, :m] = fixed[d:]
        else:
            k = -d
            m = n - k
            if m <= 0:
                continue
            if var_is_h:
                A[r, :m] = fixed[k:]
            else:
                A[r, k:] = fixed[:m]
    return A

def _solve_block_lp(fixed, n, target, var_is_h):
    A = _build_block_matrix(fixed, n, var_is_h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _alternating_lp(h, deadline, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 25:
            break

        g = 1.0 - h
        c5_cur = _exact_c5(h)

        h_new = _solve_block_lp(g, n, target, True)
        if h_new is not None:
            h_cand = _project(h_new, target)
            c5_a = _exact_c5(h_cand)
            if c5_a < c5_cur - 1e-12:
                h = h_cand
                c5_cur = c5_a

        g = 1.0 - h
        g_new = _solve_block_lp(h, n, target, False)
        if g_new is not None:
            h_cand = 1.0 - g_new
            h_cand = _project(h_cand, target)
            c5_b = _exact_c5(h_cand)
            if c5_b < c5_cur - 1e-12:
                h = h_cand
                c5_cur = c5_b

        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

def _lse_gd(h, deadline, T_init=20.0, T_max=500.0, max_iters=3000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    T = T_init
    lr = 0.008
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 150:
            break

        g = 1.0 - h
        scores = np.correlate(h, g, mode='full') * dx
        max_s = np.max(scores)
        shifted = scores - max_s
        w = np.exp(T * shifted)
        ws = float(np.sum(w))
        if ws < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= ws

        G = _build_grad_matrix(h, g)
        grad = (w @ G) * dx

        best_f = best_c5
        best_step = None
        for a in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_t = _project(h - a * grad, target)
            f_t = _exact_c5(h_t)
            if f_t < best_f - 1e-13:
                best_f = f_t
                best_step = h_t

        if best_step is not None:
            h = best_step
            if best_f < best_c5 - 1e-12:
                best_c5 = best_f
                best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.05)
        else:
            no_improve += 1
            lr *= 0.75
            if lr < 1e-7:
                break

        if it % 30 == 0:
            T = min(T * 1.06, T_max)

    return best_h, best_c5

def _lp_direction_search(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 40:
            break

        g = 1.0 - h
        G = _build_grad_matrix(h, g)
        scores = np.correlate(h, g, mode='full')

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))

        p = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and np.max(np.abs(res.x[:n])) > 1e-12:
                p = res.x[:n]
        except Exception:
            pass

        if p is None:
            no_improve += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue

        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)

        best_a = 0.0
        best_c = _exact_c5(h)
        alphas = np.concatenate([np.linspace(0.01, alpha_max, 10),
                                 np.logspace(-2, np.log10(max(alpha_max, 1e-3)), 10)])
        alphas = alphas[alphas <= alpha_max + 1e-12]
        for a in alphas:
            if a < 1e-10:
                continue
            h_t = _project(h + a * p, target)
            c_t = _exact_c5(h_t)
            if c_t < best_c:
                best_c = c_t
                best_a = a

        if best_a > 0 and best_c < _exact_c5(h) - 1e-12:
            h = _project(h + best_a * p, target)
            cur = _exact_c5(h)
            if cur < best_c5 - 1e-12:
                best_c5 = cur
                best_h = h.copy()
            no_improve = 0
            radius = min(radius * 1.1, 0.4)
        else:
            no_improve += 1
            radius *= 0.75
            if stall := no_improve % 10 == 0 and radius < 0.03:
                radius = 0.06

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    h_t = h.copy()
                    h_t[j] += dj * a_max
                    h_t[k] += dk * a_max
                    h_t = _project(h_t, target)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
        if not improved:
            break

    return best_h, best_c5
