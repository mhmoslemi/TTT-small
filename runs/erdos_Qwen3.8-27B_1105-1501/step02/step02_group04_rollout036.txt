def smooth_objective(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    # log-sum-exp for stability
    exp_vals = np.exp((corr - max_c) / temp)
    val = max_c + temp * np.log(np.sum(exp_vals))
    return dx * val

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_obj(x):
    n = _MODULE_N
    dx = _MODULE_DX
    T = _MODULE_TEMP
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)

    shifted = (corr - max_c) / T
    # Clip to avoid overflow (shouldn't be needed if T > 0)
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return dx * (max_c + T * log_sum)

def _smooth_jac(x):
    n = _MODULE_N
    dx = _MODULE_DX
    T = _MODULE_TEMP
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = (corr - max_c) / T

    weights = exp_vals / np.sum(exp_vals)
    # Build gradient

    W = weights
    # first_term[m] = W[n-1-m : 2n-1-m] @ (1-x)
    # second_term[m] = W[m : m+n][::-1] @ x
    one_minus_x = 1.0 - x
    first = np.empty(n)

        first[m] = np.dot(W[n-1-m : 2*n-1-m], one_minus_x)
        second[m] = np.dot(W[m : m+n][::-1], x)
    return dx * (first - second)

        else:

def pair_line_search(h, i, j, current_c5):
    """Optimize transfer between h[i] and h[j]."""
    hi_val = h[i]
    hj_val = h[j]
    # t: h[i] -> h[i] + t, h[j] -> h[j] - t
    # Constraints: 0 <= h[i]+t <= 1, 0 <= h[j]-t <= 1
    t_lo = max(-hi_val, hj_val - 1.0)

    a, b = t_lo, t_hi
