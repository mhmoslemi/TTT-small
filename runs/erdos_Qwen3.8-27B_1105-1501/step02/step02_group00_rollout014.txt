        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    target = float(target)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _slsqp_optimize(h, beta, maxiter=300):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        s = _scores(x)
        return float(logsumexp(beta * s)) / beta
    
    def obj_grad(x):
        s = _scores(x)
        m = logsumexp(beta * s)
        w = np.exp(beta * s - m)
        G = _grad_matrix(x)
        return G.T @ w
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.x is not None:
            x = _project(res.x, target)
            return x
    except Exception:
        pass
    return h

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a = 0.0
    best_c5 = c5_cur
    for a in np.linspace(0.0, alpha_max, 20):
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    
    if best_a > 0:
        lo_a = max(0.0, best_a * 0.8)
        hi_a = min(alpha_max, best_a * 1.2)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(20):
            if hi_a - lo_a < 1e-10:
                break
            if f1 < f2:
                hi_a = c2; c2 = c1; f2 = f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo_a = c1; c1 = c2; f1 = f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(h + c2 * p)
        opt_a = (lo_a + hi_a) / 2.0
        opt_c5 = _exact_c5(h + opt_a * p)
        if opt_c5 < best_c5:
            return opt_a, opt_c5
    
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.8
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.03
    
    return best_h, best_c5

def _subgrad_momentum(h, deadline, lr_init=0.01, max_iters=2000):
    """Subgradient method with momentum and projection."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    lr = lr_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        scores = _scores(h)
        max_s = float(np.max(scores))
        active = scores >= max_s - 1e-8
        G = _grad_matrix(h)
        grad = G[active].mean(axis=0)
        grad -= np.mean(grad)  # make it tangent to sum constraint
        
        # Momentum
        momentum = 0.9 * momentum + 0.1 * grad
        h_new = h - lr * momentum
        h_new = _project(h_new, target)
        
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            lr *= 0.9
    
    return best_h, best_c5

def _fourier_perturb(h, rng, n_modes=3, amp=0.05):
    """Add random low-frequency Fourier perturbations."""
    n = len(h)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        amp_i = amp * rng.uniform(0.5, 1.5)
        pert += amp_i * np.sin(2 * np.pi * freq * x + phase)
    return h + pert

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                # Try moving mass from j to k and k to j
                for direction, a_vals in [((1.0, -1.0), [0.3, 0.5, 0.7]),
                                          ((-1.0, 1.0), [0.3, 0.5, 0.7])]:
                    a_max = min(h[j] if direction[0] < 0 else (1-h[j]),
                                h[k] if direction[1] < 0 else (1-h[k]))
                    for a in a_vals:
                        if a > a_max:
                            continue
                        h_t = h.copy()
                        h_t[j] += direction[0] * a
                        h_t[k] += direction[1] * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    
    return best_h, best_c5

    if target >= n:
        return np.ones(n)
