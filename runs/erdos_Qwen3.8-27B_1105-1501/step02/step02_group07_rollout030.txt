    n = len(h_init)
    dx = 2.0 / n
    V = compute_overlaps_raw(h_init)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h_init[0:L]
            gp[0:L] -= h_init[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h_init[t:n]
            gp[t:n] -= h_init[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo = max(-h_init[i], -trust)
        hi = min(1.0 - h_init[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                h_new = make_feasible(h_init + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)
    
    def time_left():
        return deadline - time.time()
    
    # Get initial values
    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        h_init = make_feasible(h_init)
    except (NameError, AttributeError):
        n = 60
        x = np.linspace(0, 1, n, endpoint=False)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
        h_init = make_feasible(h_init)
    
    n0 = len(h_init)
    best_h = h_init.copy()
    best_score = official_score(h_init)
    
    # Phase 1: Binary local search from binarized initial
    h_bin = (h_init > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target_ones = n0 // 2
    if n_ones > target_ones:
        order = np.argsort(h_bin.sum(axis=0) if h_bin.ndim > 1 else h_init)
        excess = n_ones - target_ones
        for i in range(excess):
            h_bin[order[i]] = 0.0
    elif n_ones < target_ones:
        order = np.argsort(-h_init)
        deficit = target_ones - n_ones
        for i in range(deficit):
            h_bin[order[i]] = 1.0
    
    t1_end = t0 + min(120, budget_s * 0.15)
    h_bin, bin_max = binary_local_search(h_bin, max_passes=200000, time_fn=lambda: t1_end - time.time())
    bin_score = bin_max * 2.0 / n0
    if bin_score < best_score:
        best_h = h_bin.copy()
        best_score = bin_score
    
    # Phase 2: Try a few random binary patterns
    t2_end = t0 + min(250, budget_s * 0.25)
    for trial in range(8):
        if time_left() < 30:
            break
        h_rand = np.zeros(n0)
        indices = rng.permutation(n0)
        h_rand[indices[:target_ones]] = 1.0
        h_b, b_max = binary_local_search(h_rand, max_passes=50000, time_fn=lambda: time_left())
        b_score = b_max * 2.0 / n0
        if b_score < best_score:
            best_h = h_b.copy()
            best_score = b_score
    
    # Phase 3: Continuous refinement from best binary
    t3_end = t0 + min(450, budget_s * 0.45)
    if best_h.max() > 0.99:
        h_cont = 0.05 + 0.9 * best_h
        h_cont = make_feasible(h_cont)
    else:
        h_cont = best_h.copy()
    
    h_ref, ref_score = continuous_refine(h_cont, lambda: time_left(), max_iters=3000)
    if ref_score < best_score:
        best_h = h_ref.copy()
        best_score = ref_score
    
    # Phase 4: Active set LP polishing from initial_h_values
    h_lp = h_init.copy()
    t4_end = t0 + min(700, budget_s * 0.7)
    trust = 0.2
    for eps in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        while time_left() > (t4_end - t0) and time_left() > 80:
            h_new, sc = active_set_lp(h_lp, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-13:
                    best_h = h_new.copy()
                    best_score = sc
                    h_lp = h_new
                    trust = min(trust * 1.05, 0.35)
                else:
                    h_lp = h_new
                    if sc < official_score(h_lp) + 1e-10:
                        trust = max(trust * 0.9, 0.02)
                    else:
                        break
            else:
                break
    
    # Phase 5: LP polishing from the best solution found
    h_lp2 = best_h.copy()
    t5_end = t0 + min(900, budget_s * 0.85)
    trust = 0.15
    for eps in [0.005, 0.002, 0.001, 0.0005]:
        while time_left() > (t5_end - t0) and time_left() > 60:
            h_new, sc = active_set_lp(h_lp2, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-13:
                    best_h = h_new.copy()
                    best_score = sc
                    h_lp2 = h_new
                    trust = min(trust * 1.05, 0.3)
                else:
                    h_lp2 = h_new
                    trust = max(trust * 0.85, 0.02)
            else:
                break
    
    # Phase 6: Try different resolutions
    if time_left() > 80:
        for n_try in [n0 - 5, n0 - 3, n0 + 3, n0 + 5, n0 + 10]:
            if time_left() < 50 or n_try < 30:
                break
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int.copy()
                best_score = sc_int
            # Quick continuous refinement
            sub_deadline = min(time_left() - 20, 40)
            if sub_deadline > 10:
                t_sub = time.time() + sub_deadline
                h_r, s_r = continuous_refine(h_int, lambda: t_sub - time.time(), max_iters=1000)
                if s_r < best_score - 1e-12:
                    best_h = h_r.copy()
                    best_score = s_r
    
    # Phase 7: Final fine polishing
    h_final = best_h.copy()
    trust = 0.05
    for eps in [0.002, 0.001, 0.0005]:
        while time_left() > 10:
            h_new, sc = active_set_lp(h_final, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc
                    h_final = h_new
                else:
                    h_final = h_new
                    trust = max(trust * 0.7, 0.005)
                    if trust <= 0.005:
                        break
            else:
                break
    
    # Phase 8: Very fine gradient steps
    if time_left() > 15:
        h_cur = best_h.copy()
        beta = 200.0
        for _ in range(2000):
            if time_left() < 10:
                break
            V = compute_overlaps_raw(h_cur)
            V_max = V.max()
            shifted = beta * (V - V_max)
            shifted = np.clip(shifted, -500.0, 0.0)
            exp_v = np.exp(shifted)
            wsv = exp_v.sum()
            if wsv < 1e-300:
                break
            w = exp_v / wsv
            grad = gradient_overlaps(h_cur, w)
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad /= gn
            alpha = 0.1
            improved = False
            for _ in range(10):
                h_new = make_feasible(h_cur - alpha * grad)
                f_new = official_score(h_new)
                if f_new < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = f_new
                    h_cur = h_new
                    improved = True
                    break
                alpha *= 0.5
                if alpha < 1e-10:
                    break
            if not improved:
                beta = min(beta * 1.5, 2000.0)
                if beta >= 2000:
                    break
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12
    
    return final_h, final_bound, final_n
