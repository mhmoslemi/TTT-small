def exact_score(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:

                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, s, dx):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[:m] += (1.0 - h[s:s+m])
            g[s:] += (-h[:m])
    else:
        t = -s
        m = n - t
        if m > 0:
            g[t:] += (1.0 - h[:m])
            g[:m] += (-h[t:t+m])
    return g * dx

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def lse_objective(h, beta):
    vals = compute_lags(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return cmax + log_sum / beta

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = compute_lags(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w /= wsum
    grad = np.zeros(n)
    for p in range(len(w)):
        if w[p] < 1e-15:
            continue
        s = p - (n - 1)
        grad += w[p] * lag_gradient(h, s, dx)
    return grad

def slsqp_run(h0, beta, maxiter=80):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(lse_objective, h0, args=(beta,), jac=lse_gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = exact_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def proj_gradient(h, is_dead, active_tol=0.015, max_iter=500, n_grid=80):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = exact_score(h)
    step_scale = 1.0
    stagnant = 0

    for it in range(max_iter):
        if is_dead():
            break
        vals = compute_lags(best_h)
        cmax = vals.max()

        active_idx = np.where(vals >= cmax - active_tol)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(vals)])

        g = np.zeros(n)
        for idx in active_idx:
            s = idx - (n - 1)
            g += lag_gradient(best_h, s, dx)
        g /= len(active_idx)

        d = -g
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        pos = d > 1e-14
        neg = d < -1e-14
        t_max = 5.0
        if np.any(pos):
            t_max = min(t_max, np.min((1.0 - best_h[pos]) / d[pos]))
        if np.any(neg):
            t_max = min(t_max, np.min(best_h[neg] / (-d[neg])))

        if t_max < 1e-10:
            break

        t_max = min(t_max, step_scale)
        ts = np.linspace(0, t_max, n_grid)
        best_t = 0.0
        best_sc = cmax
        for t in ts:
            if t < 1e-15:
                continue
            h_tr = best_h + t * d
            sc = exact_score(h_tr)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_t = t

        if best_t < 1e-10:
            step_scale *= 0.5
            stagnant += 1
            if stagnant > 15:
                break
            if step_scale < 1e-6:
                break
            continue

        h_new = make_feasible(best_h + best_t * d)
        sc_new = exact_score(h_new)
        if sc_new < best_score - 1e-14:
            best_h = h_new
            best_score = sc_new
            stagnant = 0
            step_scale = max(step_scale * 1.05, 0.1)
            step_scale = min(step_scale, 5.0)
        else:
            step_scale *= 0.7
            stagnant += 1
            if stagnant > 20:
                break

    return best_h, best_score

def two_point_polish(h, n_iter, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = exact_score(best)
    pairs = list(range(n * (n - 1) // 2))
    for it in range(n_iter):
        if is_dead():
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = exact_score(tmp)
            if sc < best_local - 1e-15:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] += best_t
            best[b] -= best_t
            best[a] = np.clip(best[a], 0.0, 1.0)
            best[b] = np.clip(best[b], 0.0, 1.0)
            best_sc = exact_score(best)
    return best, best_sc

def interp_res(h_src, n_tgt):
    n_src = len(h_src)
    if n_tgt == n_src:
        return make_feasible(h_src.copy())
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def make_wave(n, freqs, amps, phases):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for f, a, ph in zip(freqs, amps, phases):
        h += a * np.sin(2 * np.pi * f * x + ph)
    return make_feasible(h)

def make_block(n, block_vals):
    h = np.array([block_vals[i % len(block_vals)] for i in range(n)], dtype=float)
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(12.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    def try_update(h_candidate):
        nonlocal best_h, best_score
        if h_candidate is None:
            return
        h_c = make_feasible(h_candidate)
        sc = exact_score(h_c)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h_c
            best_score = sc
        elif best_h is None and np.isfinite(sc):
            best_h = h_c
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = exact_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, [2, 3, 5], [0.3, 0.15, 0.08], [0.0, 1.0, 2.0])
        best_score = exact_score(best_h)

    n = len(best_h)

    # Phase 1: SLSQP smoothing with multiple betas
    if not is_dead():
        for beta in [20, 50, 120, 300, 800]:
            if is_dead():
                break
            h_new, sc = slsqp_run(best_h, beta, maxiter=100)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 2: Projected gradient with active set
    if not is_dead():
        for tol_seq in [0.03, 0.015, 0.005, 0.002]:
            if is_dead():
                break
            h_new, sc = proj_gradient(best_h, is_dead, active_tol=tol_seq, max_iter=300, n_grid=100)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 3: 2-point polishing
    if not is_dead():
        n_tries = max(5000, int((deadline - time.time()) * 4))
        best_h, best_score = two_point_polish(best_h, n_tries, rng, is_dead)

    # Phase 4: Another projected gradient pass
    if not is_dead():
        h_new, sc = proj_gradient(best_h, is_dead, active_tol=0.003, max_iter=400, n_grid=120)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    # Phase 5: 2-point again
    if not is_dead():
        n_tries = max(5000, int((deadline - time.time()) * 4))
        best_h, best_score = two_point_polish(best_h, n_tries, rng, is_dead)

    # Phase 6: Anti-symmetric transform
    if not is_dead():
        h_sym = anti_sym_map(best_h)
        sc_sym = exact_score(h_sym)
        if sc_sym < best_score - 1e-14:
            best_h = h_sym
            best_score = sc_sym
            # Optimize from symmetric point
            if not is_dead():
                h_new, sc = proj_gradient(best_h, is_dead, active_tol=0.01, max_iter=200, n_grid=80)
                if sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc
            if not is_dead():
                n_tries = max(3000, int((deadline - time.time()) * 3))
                best_h, best_score = two_point_polish(best_h, n_tries, rng, is_dead)

    # Phase 7: Resolution exploration
    if not is_dead():
        n_base = len(best_h)
        n_opts = sorted(set([n_base + dn for dn in [-12, -8, -4, -2, 2, 4, 8, 12]
                             if 24 <= n_base + dn <= 100]))
        for nc in [36, 40, 44, 48, 52, 56, 64, 72, 80]:
            if nc not in n_opts and 24 <= nc <= 100:
                n_opts.append(nc)
        n_opts = sorted(set(n_opts))

        for n_opt in n_opts:
            if is_dead():
                break
            if n_opt == n_base:
                continue
            h_int = interp_res(best_h, n_opt)
            sc_int = exact_score(h_int)
            h_work = h_int.copy()
            sc_work = sc_int
            if sc_int < best_score - 1e-14:
                best_h = h_int
                best_score = sc_int

            if not is_dead():
                h_new, sc = proj_gradient(h_work, is_dead, active_tol=0.01, max_iter=100, n_grid=60)
                if sc < sc_work - 1e-14:
                    h_work = h_new
                    sc_work = sc
            if not is_dead():
                h_work, sc_work = two_point_polish(h_work, 2000, rng, is_dead)
            try_update(h_work)

    # Phase 8: Try wave starting points at different frequencies
    if not is_dead():
        for n_wave in [48, 56, 60, 64, 72]:
            if is_dead():
                break
            for seed_try in range(3):
                if is_dead():
                    break
                freqs = [rng.uniform(1.5, 5.0) for _ in range(3)]
                amps = [rng.uniform(0.2, 0.4), rng.uniform(0.05, 0.2), rng.uniform(0.02, 0.1)]
                phases = [rng.uniform(0, 2*np.pi) for _ in range(3)]
                h_w = make_wave(n_wave, freqs, amps, phases)
                h_w, sc_w = two_point_polish(h_w, 1500, rng, is_dead)
                if not is_dead():
                    h_w, sc_w = proj_gradient(h_w, is_dead, active_tol=0.01, max_iter=80, n_grid=60)
                if sc_w < sc_w:
                    pass
                try_update(h_w)

    # Phase 9: Intensive final polishing
    if not is_dead():
        h_new, sc = proj_gradient(best_h, is_dead, active_tol=0.002, max_iter=300, n_grid=150)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    if not is_dead():
        n_tries = max(8000, int((deadline - time.time()) * 5))
        best_h, best_score = two_point_polish(best_h, n_tries, rng, is_dead)

    # Phase 10: Final SLSQP polish at high beta
    if not is_dead() and best_score > 0.3805:
        h_new, sc = slsqp_run(best_h, 2000, maxiter=60)
        if h_new is not None and sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
        if not is_dead():
            n_tries = max(3000, int((deadline - time.time()) * 3))
            best_h, best_score = two_point_polish(best_h, n_tries, rng, is_dead)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = exact_score(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)):
        final_h = np.full(60, 0.5)
        final_score = exact_score(final_h)
        final_n = 60

    final_h = np.clip(final_h, 0.0, 1.0)
    if abs(final_h.sum() - final_n / 2.0) > 1e-10:
        final_h = make_feasible(final_h)
        final_score = exact_score(final_h)

    return final_h, final_score, final_n
