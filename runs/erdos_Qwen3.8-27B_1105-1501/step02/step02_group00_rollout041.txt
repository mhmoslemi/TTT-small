- score for lag 0: sum h[i]*(1-h[i]) = 0 (since h is binary)
- score for lag d: number of i where h[i]=1 and h[i+d]=0 (for valid i)

Ah, right. With the equality constraint sum(h) = target, we can't independently optimize each coordinate. We need to optimize in pairs (or use a Lagrange multiplier approach).

For a pair of coordinates (j, k), we optimize:
min_{h[j], h[k]} max_l f_l(h)
subject to h[j] + h[k] = target - sum(h_others)
and 0 ≤ h[j] ≤ 1, 0 ≤ h[k] ≤ 1.

The constraint becomes: 0 ≤ x ≤ 1 and 0 ≤ S - x ≤ 1, so max(0, S-1) ≤ x ≤ min(1, S).

Each f_l is at most quadratic in x (due to the h[j]*(1-h[j]) term for lag 0 if j is the first index, and h[k]*(1-h[k]) for lag 0 if k is the first index).

So f_l(x) is at most quadratic (degree ≤ 2) for all l.
The max of quadratics is not necessarily convex, but in 1D it can be found by evaluating at critical points.

The critical points of max_l f_l(x) occur where:
- Two or more f_l's are equal and their derivatives differ (nonsmooth point)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = max_s + log_sum / beta
    weights = np.exp(shifted - log_sum)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        wr = weights[r] * dx
        if wr < 1e-16:
            continue
        if d >= 0:
            m = n - d
            grad[:m] += wr * (1.0 - h[d:])
            grad[d:] += wr * (-h[:m])
        else:
            g = -d
            m = n - g
            grad[g:] += wr * (1.0 - h[:m])
            grad[:m] += wr * (-h[g:])
    return obj, grad

def _eq_fun(h, target):
    return float(np.sum(h)) - target

def _eq_jac(h, target):
    return np.ones(len(h))

def _slsqp_opt(h0, beta, maxiter=300, tol=1e-12):
    n = len(h0)
    target = n / 2.0
    res = minimize(_lse_obj_grad_wrap, h0, args=(beta,), jac=_lse_grad_wrap,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (target,)}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': tol})
    if res.success and np.all(res.x >= -1e-8) and np.all(res.x <= 1 + 1e-8):
        return _project(res.x, target)
    return h0

def _lse_obj_grad_wrap(h, beta):
    return _lse_obj_grad(h, beta)[0]

def _lse_grad_wrap(h, beta):
    return _lse_obj_grad(h, beta)[1]

def _proj_grad(h0, beta, maxiter=8000, deadline=None):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    obj_prev, g = _lse_obj_grad(h, beta)
    step = 1.0 / (float(np.max(np.abs(g))) + 1e-10)
    prev_h = h.copy()
    prev_g = g.copy()
    for it in range(maxiter):
        if deadline is not None and time.time() > deadline:
            break
        h_new = h - step * g
        h_new = _project(h_new, target)
        obj_new, g_new = _lse_obj_grad(h_new, beta)
        if it > 0:
            s = h_new - prev_h
            y = g_new - prev_g
            ss = float(np.dot(s, s))
            ys = float(np.dot(y, s))
            yy = float(np.dot(y, y))
            if ss > 1e-15 and yy > 1e-15:
                if it % 2 == 0:
                    step = ss / max(ys, 1e-12)
                else:
                    step = ys / max(yy, 1e-12)
                step = float(np.clip(step, 1e-7, 5.0))
        prev_h = h.copy()
        prev_g = g.copy()
        h = h_new
        g = g_new
        obj_prev = obj_new
        if it % 20 == 0 or it == maxiter - 1:
            c5 = _exact_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
    c5 = _exact_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _lp_loop(h0, deadline, radius_init=0.1, max_iters=3000):

        corr_raw = np.correlate(h, 1.0 - h, mode='full')

        c_vec = np.zeros(n + 1)

        A_ub[:, :n] = G

        A_eq = np.ones((1, n))

            for _ in range(12):

                    hi_a, c2, f2 = c2, c1, f1
                    c1 = hi_a - gr * (hi_a - lo_a)
                    f1 = _exact_c5(h + c1 * p)
                else:
                    lo_a, c1, f1 = c1, c2, f2

                    f2 = _exact_c5(h + c2 * p)
            opt_a = (lo_a + hi_a) / 2.0
            opt_c5 = _exact_c5(h + opt_a * p)
            if opt_c5 < best_c5_try:
                best_c5_try = opt_c5
                best_a = opt_a
            h = h + best_a * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:

                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7

def _pair_sweep(h0, deadline, max_passes=50):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            for k in rng.permutation(n):
                if k == j:
                    continue
                S = h[j] + h[k]
                lo = max(0.0, S - 1.0)
                hi = min(1.0, S)
                if hi - lo < 1e-10:
                    continue
                best_x = h[j]
                best_val = best_c5
                a = lo
                b = hi
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                for _ in range(18):
                    if b - a < 1e-10:
                        break
                    h[j] = c1
                    h[k] = S - c1
                    v1 = _exact_c5(h)
                    h[j] = c2
                    h[k] = S - c2
                    v2 = _exact_c5(h)
                    if v1 < v2:
                        b, c2 = c2, c1
                    else:
                        a, c1 = c1, c2
                    c1 = b - gr * (b - a)
                    c2 = a + gr * (b - a)
                opt_x = (a + b) / 2.0
                h[j] = opt_x
                h[k] = S - opt_x
                c5_val = _exact_c5(h)
                if c5_val < best_val - 1e-12:
                    improved = True
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h.copy()
                else:
                    h[j] = best_x
                    h[k] = S - best_x
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_anneal(h0, deadline, max_iters=50000):
    n = len(h0)
    target = int(n / 2)
    h = np.where(h0 > 0.5, 1.0, 0.0)
    ones = np.where(h == 1.0)[0]
    if len(ones) != target:
        idx = np.argsort(h0)
        h = np.zeros(n)
        h[idx[-target:]] = 1.0
        ones = np.where(h == 1.0)[0]
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    temp = 0.005
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it > 0 and it % 5000 == 0:
            temp *= 0.8
        a = ones[rng_binary(it)]
        zeros = np.where(h == 0.0)[0]
        b = zeros[rng_binary(it + 100000)]
        h[a] = 0.0
        h[b] = 1.0
        ones[ones == a] = b
        c5_new = _exact_c5(h)
        if c5_new < _exact_c5(best_h) or rng_accept(it, c5_new, best_c5, temp):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return _project(best_h, float(target))

def rng_binary(x):
    return int((np.sin(x * 12.9898 + 78.233) * 43758.5453) % 1000000) % 200000

def rng_accept(it, new_c5, old_c5, temp):
    if new_c5 <= old_c5:
        return True
    diff = new_c5 - old_c5
    return float(np.random.default_rng(it).random()) < np.exp(-diff / max(temp, 1e-8))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP with increasing beta
    deadline1 = t0 + budget_s * 0.20
    for beta in [5.0, 10.0, 20.0, 40.0, 80.0, 150.0]:
        if time.time() > deadline1:
            break
        h_s = _slsqp_opt(best_h, beta, maxiter=200)
        c5_s = _exact_c5(h_s)
        if c5_s < best_c5 - 1e-10:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Projected gradient refinement
    deadline2 = t0 + budget_s * 0.35
    for beta in [30.0, 60.0, 120.0, 250.0]:
        if time.time() > deadline2:
            break
        h_pg, c5_pg = _proj_grad(best_h, beta, maxiter=5000, deadline=deadline2)
        if c5_pg < best_c5 - 1e-10:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: LP polishing
    deadline3 = t0 + budget_s * 0.50
    h_lp, c5_lp = _lp_loop(best_h, deadline3, radius_init=0.08, max_iters=2000)
    if c5_lp < best_c5 - 1e-10:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 4: Pair coordinate descent
    deadline4 = t0 + budget_s * 0.60
    h_pd, c5_pd = _pair_sweep(best_h, deadline4, max_passes=80)
    if c5_pd < best_c5 - 1e-10:
        best_c5 = c5_pd
        best_h = h_pd.copy()

    # Phase 5: Multi-start exploration
    deadline5 = t0 + budget_s * 0.82
    base = best_h.copy()
    bn = best_n
    for attempt in range(500):
        if time.time() > deadline5:
            break
        if best_c5 < 0.3805:
            break
        scale = 0.02 + 0.03 * (1.0 + np.sin(attempt * 0.7)) * 0.5
        h_start = base + rng.normal(0, scale, bn)
        h_start = _project(h_start, bn / 2.0)
        # Quick SLSQP from this start
        sub_deadline = min(deadline5, time.time() + budget_s * 0.04)
        h_try = _slsqp_opt(h_start, 50.0, maxiter=80)
        h_try = _lp_loop(h_try, sub_deadline, radius_init=0.05, max_iters=300)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            base = best_h.copy()

    # Phase 6: Resolution sweep
    deadline6 = t0 + budget_s * 0.95
    for dn in [-3, -2, -1, 1, 2, 3, -4, 4, -5, 5, -6, 6]:
        if time.time() > deadline6:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline6, time.time() + budget_s * 0.05)
        h_r = _slsqp_opt(h_r, 40.0, maxiter=100)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.06, max_iters=500)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.995, radius_init=0.03, max_iters=1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        h_f2, c5_f2 = _pair_sweep(best_h, t0 + budget_s * 0.998, max_passes=30)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
