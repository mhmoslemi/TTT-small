def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0:

            h[idx] += add
            residual -= add
            if residual < 1e-14:

        for idx in order:

            if sub <= 0:

            residual += sub
            if -residual < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 0:
                break
            h[idx] += add
            residual -= add
            if residual < 1e-14:
                break
    elif residual < -1e-12:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 0:
                break
            h[idx] -= sub
            residual += sub
            if -residual < 1e-14:
                break
    return h

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        D = p - (n - 1)
        if D >= 0:
            L = n - D
            if L > 0:
                G[p, D:D + L] += dx * (1.0 - h[:L])
                G[p, :L] -= dx * h[D:D + L]
        else:
            a = -D
            L = n - a
            if L > 0:
                G[p, :L] += dx * (1.0 - h[a:a + L])
                G[p, a:a + L] -= dx * h[:L]
    return G

    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
    return make_feasible(h_sym)

def smooth_obj_grad_z(z, n, beta):

    h = build_h_from_z(z, n)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    logsum = shifted.max() + np.log(np.sum(exp_shifted))
    obj = cmax + logsum / beta
    w = exp_shifted / exp_shifted.sum()
    G = build_grad_matrix(h)
    v = w @ G

def find_active_set(vals, frac=0.3, min_count=10, max_count=50):

    threshold = cmax - max(0.005 * cmax, 5e-5)
    active = set(np.where(vals >= threshold)[0].tolist())

        return smooth_obj_grad_z(z, n, beta)[1]

    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'gtol': 1e-10,
                                'maxfun': 500})
        if np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return z0


def interpolate_h(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    def past_deadline():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 64
        x = np.arange(n0)
        h_init = 0.5 + 0.2 * np.sin(2 * np.pi * x / n0) + 0.05 * np.cos(4 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    update(best_h, best_score)

    # Symmetrize and check
    h_sym = symmetrize_h(best_h)
    sc_sym = official_score(h_sym)
    if sc_sym < best_score:
        update(h_sym, sc_sym)

    # Phase 1: BFGS in anti-symmetric space with increasing beta
    n = len(best_h)
    if n % 2 != 0:
        n = n - 1  # use even
        best_h = interpolate_h(best_h, n)
        best_score = official_score(best_h)
        h_sym = symmetrize_h(best_h)
        sc_sym = official_score(h_sym)
        if sc_sym < best_score:
            update(h_sym, sc_sym)

    M = n // 2
    h_work = symmetrize_h(best_h)
    sc_work = official_score(h_work)
    if sc_work < best_score:
        update(h_work, sc_work)

    z_work = extract_z(h_work, n)

    for beta in [50, 200, 1000, 5000, 20000]:
        if past_deadline():
            break
        tl = min(time_left() * 0.2, 25.0)
        if tl < 5:
            break
        z_new = bfgs_smooth_solve(z_work, n, beta, time_limit=tl, maxiter=200)
        h_new = build_h_from_z(z_new, n)
        sc_new = official_score(h_new)
        if sc_new < best_score:
            update(h_new, sc_new)
            z_work = z_new.copy()
            h_work = h_new.copy()

    # Phase 2: SLSQP active-set in anti-symmetric space
    if not past_deadline() and time_left() > 30:
        vals = compute_vals(h_work)
        for frac in [0.3, 0.5, 0.7]:
            if past_deadline():
                break
            active = find_active_set(vals, frac=frac, min_count=8, max_count=40)
            h_new, sc_new, z_new = solve_sym_minimax(n, active, z_work, maxiter=300)
            if h_new is not None and sc_new < best_score:
                update(h_new, sc_new)
                z_work = z_new.copy()
                h_work = h_new.copy()
                vals = compute_vals(h_work)

    # Phase 3: Full (non-symmetric) SLSQP from best point
    if not past_deadline() and time_left() > 30:
        vals = compute_vals(best_h)
        for frac in [0.3, 0.5]:
            if past_deadline():
                break
            active = find_active_set(vals, frac=frac, min_count=10, max_count=45)
            h_new, sc_new = solve_full_minimax(n, active, best_h, maxiter=300)
            if h_new is not None and sc_new < best_score:
                update(h_new, sc_new)
                vals = compute_vals(best_h)

    # Phase 4: Iterate refinement
    for round_idx in range(4):
        if past_deadline() or time_left() < 30:
            break
        # Symmetric BFGS polish
        h_sym_cur = symmetrize_h(best_h)
        sc_sym_cur = official_score(h_sym_cur)
        if sc_sym_cur < best_score:
            update(h_sym_cur, sc_sym_cur)
        z_cur = extract_z(h_sym_cur, n)
        z_cur = bfgs_smooth_solve(z_cur, n, 20000, time_limit=min(time_left() * 0.15, 15.0), maxiter=150)
        h_cur = build_h_from_z(z_cur, n)
        sc_cur = official_score(h_cur)
        if sc_cur < best_score:
            update(h_cur, sc_cur)
            h_sym_cur = h_cur
            z_cur = z_cur

        # Active set SLSQP symmetric
        vals_cur = compute_vals(h_sym_cur)
        active_cur = find_active_set(vals_cur, frac=0.4, min_count=10, max_count=45)
        h_new, sc_new, z_new = solve_sym_minimax(n, active_cur, z_cur, maxiter=400)
        if h_new is not None and sc_new < best_score:
            update(h_new, sc_new)
            z_cur = z_new
            h_sym_cur = h_new

        # Non-symmetric SLSQP
        if not past_deadline():
            vals_ns = compute_vals(best_h)
            active_ns = find_active_set(vals_ns, frac=0.5, min_count=12, max_count=50)
            h_new2, sc_new2 = solve_full_minimax(n, active_ns, best_h, maxiter=250)
            if h_new2 is not None and sc_new2 < best_score:
                update(h_new2, sc_new2)

    # Phase 5: Resolution changes
    if not past_deadline() and time_left() > 60:
        n_base = len(best_h)
        n_opts = sorted(set([max(40, n_base - 4), max(42, n_base - 2), n_base,
                             min(74, n_base + 2), min(76, n_base + 4)]))
        for n_opt in n_opts:
            if past_deadline() or time_left() < 20:
                break
            if n_opt == n_base:
                continue
            if n_opt % 2 != 0:
                n_opt += 1
            h_int = interpolate_h(best_h, n_opt)
            sc_int = official_score(h_int)
            # Symmetric BFGS on new resolution
            h_sym_int = symmetrize_h(h_int)
            sc_si = official_score(h_sym_int)
            if sc_si < sc_int:
                h_int = h_sym_int
                sc_int = sc_si
            M_int = n_opt // 2
            z_int = extract_z(h_sym_int, n_opt)
            for b in [1000, 10000]:
                if past_deadline():
                    break
                z_int = bfgs_smooth_solve(z_int, n_opt, b, time_limit=min(time_left() * 0.15, 15.0), maxiter=150)
                h_new_r = build_h_from_z(z_int, n_opt)
                sc_r = official_score(h_new_r)
                if sc_r < best_score:
                    update(h_new_r, sc_r)
            # Active set
            if not past_deadline():
                vals_r = compute_vals(best_h)
                active_r = find_active_set(vals_r, frac=0.4, min_count=10, max_count=40)
                h_new_a, sc_new_a, _ = solve_sym_minimax(n_opt, active_r, z_int, maxiter=200)
                if h_new_a is not None and sc_new_a < best_score:
                    update(h_new_a, sc_new_a)

    # Phase 6: Random perturbations
    if not past_deadline() and time_left() > 40:
        n_cur = len(best_h)
        n_starts = min(8, max(3, int(time_left() / 15)))
        for i in range(n_starts):
            if past_deadline() or time_left() < 15:
                break
            sigma = [0.02, 0.04, 0.06, 0.10][i % 4]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n_cur))
            sc_cand = official_score(h_cand)
            if sc_cand < best_score:
                update(h_cand, sc_cand)
            # Quick BFGS from perturbed point
            h_sym_c = symmetrize_h(h_cand)
            sc_sc = official_score(h_sym_c)
            if sc_sc < best_score:
                update(h_sym_c, sc_cand)
            z_c = extract_z(h_sym_c, n_cur)
            z_c = bfgs_smooth_solve(z_c, n_cur, 5000, time_limit=min(time_left() * 0.1, 10.0), maxiter=100)
            h_c2 = build_h_from_z(z_c, n_cur)
            sc_c2 = official_score(h_c2)
            if sc_c2 < best_score:
                update(h_c2, sc_c2)

    # Phase 7: Final intensive polish
    if not past_deadline() and time_left() > 20:
        # Very high beta BFGS
        h_fin_sym = symmetrize_h(best_h)
        sc_fin_sym = official_score(h_fin_sym)
        if sc_fin_sym < best_score:
            update(h_fin_sym, sc_fin_sym)
        z_fin = extract_z(h_fin_sym, len(best_h))
        z_fin = bfgs_smooth_solve(z_fin, len(best_h), 50000, time_limit=min(time_left() * 0.3, 20.0), maxiter=200)
        h_fin = build_h_from_z(z_fin, len(best_h))
        sc_fin = official_score(h_fin)
        if sc_fin < best_score:
            update(h_fin, sc_fin)
            h_fin_sym = h_fin
            z_fin = z_fin

        # Final SLSQP with large active set
        if not past_deadline():
            vals_fin = compute_vals(best_h)
            active_fin = find_active_set(vals_fin, frac=0.6, min_count=15, max_count=60)
            h_new, sc_new, z_new = solve_sym_minimax(len(best_h), active_fin, z_fin, maxiter=500)
            if h_new is not None and sc_new < best_score:
                update(h_new, sc_new)

        # Final non-symmetric SLSQP
        if not past_deadline():
            vals_fin2 = compute_vals(best_h)
            active_fin2 = find_active_set(vals_fin2, frac=0.5, min_count=12, max_count=50)
            h_new2, sc_new2 = solve_full_minimax(len(best_h), active_fin2, best_h, maxiter=400)
            if h_new2 is not None and sc_new2 < best_score:
                update(h_new2, sc_new2)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(final_h >= -1e-14), "Negative values"
    assert np.all(final_h <= 1.0 + 1e-14), "Values > 1"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-8, "Sum constraint violated"

    final_h = np.clip(final_h, 0.0, 1.0)
    final_score = official_score(final_h)

    return final_h, final_score, final_n
