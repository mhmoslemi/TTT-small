def build_grad_rows(h, indices, n):
    """Build gradient rows for specified correlation indices."""
    m = len(indices)
    G = np.zeros((m, n))
    offset = n - 1
    for r, idx in enumerate(indices):
        lag = idx - offset
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        for i in range(i_lo, i_hi + 1):
            p1 = i + lag
            p2 = i
            G[r, p1] += (1.0 - h[p2])
            G[r, p2] -= h[p1]
    return G

    M = np.max(corr)

        sorted_idx = np.argsort(-corr)
        active = sorted_idx[:max(4, k_top)]

    # LP: minimize t
    # s.t. G[r,:] @ d - t <= -corr[active[r]] for r = 0..m-1
    #      sum(d) = 0
    #      -trust <= d_i <= trust
    #      t in [-M, M]

    b_ub = np.zeros(m)

    A_eq = np.zeros((1, N))

    t = res.x[n]
