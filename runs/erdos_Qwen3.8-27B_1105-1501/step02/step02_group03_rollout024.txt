def grad_anti_sym(h, n):
    """Compute gradient of all values w.r.t. u (left half) for anti-symmetric h."""
    m = n // 2
    dx = 2.0 / n
    vals = compute_vals(h)
    # Full gradient
    G_full = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        # v_p = dx * sum over valid (i,j) pairs where j-i = p-(n-1)
        # shift s = p - (n-1)
        s = p - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                if room[idx] <= 0:
                    break
                add = min(residual, room[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                if h[idx] <= 0:
                    break
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def anti_sym_build(u):
    m = len(u)
    n = 2 * m
    h = np.empty(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

def anti_sym_project(h):
    n = len(h)
    m = n // 2
    u = 0.5 * (h[:m] + 1.0 - h[m:][::-1].copy())
    return anti_sym_build(np.clip(u, 0.0, 1.0))

def interpolate_to(h_src, n_tgt):

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def find_active(vals, top_frac=0.3, min_count=8, tol_frac=0.01):

    threshold = cmax - max(tol_frac * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(top_frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:top_k]
    active.update(top_idx.tolist())
    return sorted(active)

    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))

def logit_smooth_solve(h_init, beta, max_iter=5000, time_limit=20.0):
    """Optimize using log-sum-exp smooth objective with projected gradient."""
    n = len(h_init)
    h = h_init.copy()
    t0 = time.time()
    alpha = 0.1
    best_f = float('inf')
    best_h = h.copy()

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        vals = compute_vals(h)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        shifted -= shifted.max()
        logsum = np.log(np.sum(np.exp(shifted)))
        f = cmax + logsum / beta
        if f < best_f:
            best_f = f
            best_h = h.copy()

        w = np.exp(shifted)
        w /= w.sum()
        G = full_grad(h)
        g = w @ G
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-28:
            break

        step = min(alpha, 2.0 / (gnorm2 + 1e-12))
        for _ in range(20):
            y = make_feasible(h - step * g)
            vals_y = compute_vals(y)
            cmax_y = vals_y.max()
            shifted_y = beta * (vals_y - cmax_y)
            shifted_y -= shifted_y.max()
            f_y = cmax_y + np.log(np.sum(np.exp(shifted_y))) / beta
            if f_y <= f - 1e-5 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-14:
            break
        h = y
        if f_y < f:
            alpha = min(alpha * 1.5, 5.0 / (gnorm2 + 1e-12))

    return best_h

        h_init = 0.5 + 0.25 * np.sin(2 * np.pi * x / n0) + 0.05 * np.cos(4 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    n_parent = len(best_h)
    parent_score = best_score

    # Phase 1: Quick anti-symmetric SLSQP at parent resolution
    if n_parent % 2 == 0 and time_ok():
        h_as = anti_sym_project(best_h)
        sc_as = official_score(h_as)
        cur = h_as if sc_as <= parent_score else best_h

        for (tf, mc, tol) in [(0.15, 6, 0.008), (0.25, 8, 0.01), (0.35, 10, 0.012)]:
            if not time_ok():
                break
            vals = compute_vals(cur)
            active = find_active(vals, top_frac=tf, min_count=mc, tol_frac=tol)
            if len(cur) % 2 == 0:
                h_new, sc = solve_anti_sym_slsqp(cur, active, max_iter=250)
            else:
                h_new, sc = solve_full_slsqp(cur, active, max_iter=250)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
                cur = h_new

    # Phase 2: Multi-resolution with anti-symmetric optimization
    n_list = []
    for d in range(-8, 9, 2):
        n = n_parent + d
        if n % 2 == 0 and 36 <= n <= 84:
            n_list.append(n)
    n_list = sorted(set(n_list))

    for n in n_list:
        if not time_ok():
            break
        h_n = interpolate_to(best_h, n)
        h_as = anti_sym_project(h_n)

        # Try several active set configurations
        for (tf, mc, tol, mi) in [(0.15, 6, 0.008, 200), (0.25, 8, 0.01, 200), (0.35, 10, 0.012, 250)]:
            if not time_ok():
                break
            vals = compute_vals(h_as)
            active = find_active(vals, top_frac=tf, min_count=mc, tol_frac=tol)
            h_new, sc = solve_anti_sym_slsqp(h_as, active, max_iter=mi)
            if h_new is not None:
                if sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc
                h_as = h_new

        # Also try full-space SLSQP (without anti-sym constraint)
        if time_ok():
            vals_f = compute_vals(h_n)
            active_f = find_active(vals_f, top_frac=0.25, min_count=8, tol_frac=0.01)
            h_new, sc = solve_full_slsqp(h_n, active_f, max_iter=200)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 3: Log-sum-exp smooth refinement
    if time_ok():
        for beta in [500, 2000, 8000]:
            if not time_ok():
                break
            tl = min(30.0, max(5.0, (deadline - time.time()) * 0.15))
            h_smooth = logit_smooth_solve(best_h, beta, max_iter=3000, time_limit=tl)
            sc = official_score(h_smooth)
            if sc < best_score - 1e-14:
                best_h = h_smooth
                best_score = sc

            # Follow up with SLSQP
            if time_ok():
                vals = compute_vals(best_h)
                active = find_active(vals, top_frac=0.25, min_count=8, tol_frac=0.01)
                if len(best_h) % 2 == 0:
                    h_as2 = anti_sym_project(best_h)
                    h_new, sc2 = solve_anti_sym_slsqp(h_as2, active, max_iter=300)
                else:
                    h_new, sc2 = solve_full_slsqp(best_h, active, max_iter=300)
                if h_new is not None and sc2 < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc2

    # Phase 4: Intensive final polish with tight active sets
    if time_ok():
        n_best = len(best_h)
        for (tf, mc, tol, mi) in [(0.1, 5, 0.005, 600), (0.2, 8, 0.008, 500), (0.4, 12, 0.015, 400)]:
            if not time_ok():
                break
            vals = compute_vals(best_h)
            active = find_active(vals, top_frac=tf, min_count=mc, tol_frac=tol)
            if n_best % 2 == 0:
                h_as = anti_sym_project(best_h)
                h_new, sc = solve_anti_sym_slsqp(h_as, active, max_iter=mi)
            else:
                h_new, sc = solve_full_slsqp(best_h, active, max_iter=mi)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc

    # Phase 5: Try random perturbations + local refine
    if time_ok():
        n_cur = len(best_h)
        n_starts = min(6, max(2, int(max(0, deadline - time.time()) / 20)))
        for i in range(n_starts):
            if not time_ok():
                break
            sigma = 0.02 * (i + 1)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n_cur))
            sc_cand = official_score(h_cand)
            if sc_cand < best_score - 1e-14:
                best_h = h_cand
                best_score = sc_cand

            if time_ok():
                if n_cur % 2 == 0:
                    h_as = anti_sym_project(h_cand)
                else:
                    h_as = h_cand
                vals_c = compute_vals(h_as)
                active_c = find_active(vals_c, top_frac=0.25, min_count=8, tol_frac=0.01)
                if n_cur % 2 == 0:
                    h_new, sc = solve_anti_sym_slsqp(h_as, active_c, max_iter=200)
                else:
                    h_new, sc = solve_full_slsqp(h_as, active_c, max_iter=200)
                if h_new is not None and sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc

    # Phase 6: Final micro-polish - iterate SLSQP with re-selected active sets
    if time_ok():
        for _ in range(4):
            if not time_ok():
                break
            vals = compute_vals(best_h)
            cmax = vals.max()
            # Use very tight active set: only values within 0.1% of max
            threshold = cmax - max(0.001 * cmax, 1e-5)
            active_tight = sorted(np.where(vals >= threshold)[0].tolist())
            if len(active_tight) < 3:
                active_tight = find_active(vals, top_frac=0.1, min_count=5, tol_frac=0.005)

            n_b = len(best_h)
            if n_b % 2 == 0:
                h_as = anti_sym_project(best_h)
                h_new, sc = solve_anti_sym_slsqp(h_as, active_tight, max_iter=400)
            else:
                h_new, sc = solve_full_slsqp(best_h, active_tight, max_iter=400)
            if h_new is not None and sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
            else:
                break

    # Final validation and return
    final_h = make_feasible(best_h)
    final_n = len(final_h)
    final_score = official_score(final_h)

    # Ensure feasibility
    final_h = np.clip(final_h, 0.0, 1.0)
    sum_target = final_n / 2.0
    if abs(np.sum(final_h) - sum_target) > 1e-10:
        final_h = make_feasible(final_h)
        final_score = official_score(final_h)

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
