def pg_step(h, beta, alpha_max=0.8, max_ls=16):

    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = compute_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h.copy(), f
    g = g / gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

def fwl_step(h, beta, gamma=0.5):
    """Frank-Wolfe step."""
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = compute_gradient(h, w)
    # FWL vertex
    z = fwl_vertex(g)
    # Convex combination
    h_new = (1.0 - gamma) * h + gamma * z
    h_new = make_feasible(h_new)
    f_new = official_score(h_new)
    return h_new, f_new, f

def active_set_lp(h, eps=0.005, trust=0.2, max_active=30):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    # Get top active indices
    active = np.where(V >= V_max - eps)[0]
    if len(active) > max_active:
        # Keep only the top ones
        active = active[np.argsort(V[active])[-max_active:]]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):

The minimum of a convex piecewise-linear function occurs at:
- An endpoint (L or U), or
- A point where two lines cross and the derivative changes from negative to positive.

To find this: we can binary search on x' in [L, U]. At a given x', the subgradient of the max function is the set of slopes c_s for all s that achieve the maximum. 1. Evaluate at L and U.
2. Find the upper envelope of the lines.
3. The minimum is at the lowest point on the envelope.

            #   as h[k-t] when k-t=i, i.e., k=i+t (if t<=i+t<=n-1, i.e., if 2t<=... 
            #   Actually k ranges from t to n-1, so k=i+t is valid if t <= i+t <= n-1, i.e., i >= 0 and i+t <= n-1
            #   The term is h[i+t]*(1-h[i]), change = -h[i+t]*Δx
            a_i = 0.0
            if t <= i <= n - 1:
                a_i += (1.0 - h[i-t])
            if t <= i + t <= n - 1:
                a_i += (-h[i+t])
            
            a_j = 0.0
            if t <= j <= n - 1:
                a_j += (1.0 - h[j-t])
            if t <= j + t <= n - 1:
                a_j += (-h[j+t])
            
            c_r = a_i - a_j
        
        slopes[r] = dx * c_r
        intercepts[r] = V_current[r] - dx * c

r * x
    
    # Now find min over x' in [L, U] of max_r (slopes[r]*x' + intercepts[r])
    # This is a convex piecewise-linear function.
    # Evaluate at L, U, and find the minimum.
    # The minimum of a convex piecewise-linear function on [L,U] is at:
    # - L, or U, or a point where the subgradient changes sign.
    
    # Method: binary search for the point where the derivative (subgradient) contains 0.
    # At x', the subgradient is the set of slopes[r] for all r achieving the max.
    # If min(subgradient) > 0: move left
    # If max(subgradient) < 0: move right
    # Else: x' is optimal
    
    # But since the function is piecewise linear, the exact minimum is at a breakpoint.
    # Let's find all breakpoints and evaluate.
    
    # Simpler: evaluate at many points and also find exact breakpoints.
    # For 119 lines, there are at most 119*118/2 ≈ 7000 breakpoints.
    # But we can be smarter: the minimum of the upper envelope is at a vertex.
    
    # Let's just do a fine grid (200 points) plus the endpoints.
    n_eval = 200
    xs = np.linspace(L, U, n_eval)
    best_f = np.inf
    best_xp = x
    
    for xp in xs:
        vals = slopes * xp + intercepts
        f_val = vals.max()
        if f_val < best_f:
            best_f = f_val
            best_xp = xp
    
    # Also check exact breakpoints near the best
    # Refine with a finer grid around best_xp
    if L < best_xp < U:
        margin = (U - L) / n_eval
        xs_fine = np.linspace(max(L, best_xp - 3*margin), min(U, best_xp + 3*margin), 200)
        for xp in xs_fine:
            vals = slopes * xp + intercepts
            f_val = vals.max()
            if f_val < best_f:
                best_f = f_val
                best_xp = xp
    
    # Apply
    h_new = h.copy()
    h_new[i] = best_xp
    h_new[j] = S - best_xp
    h_new = make_feasible(h_new)
    f_new = official_score(h_new)
    
    if f_new < official_score(h) - 1e-14:
        return h_new, f_new
    return h, official_score(h)
