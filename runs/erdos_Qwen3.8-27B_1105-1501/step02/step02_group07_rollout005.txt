from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.0, posinf=5.0, neginf=-5.0)
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def pg_step(h, beta, alpha_max=0.8, max_ls=16):

def dct_basis(N, D):
    cols = np.zeros((N, D))
    i = np.arange(N) + 0.5
    for k in range(1, D + 1):
        cols[:, k-1] = np.cos(np.pi * k * i / N)
        norm = np.linalg.norm(cols[:, k-1])
        if norm > 1e-12:
            cols[:, k-1] /= norm
    return cols

def eval_theta(B, theta):

def de_populate(pop, scores, F, CR, rng, B, max_evals):

def pair_line_search(h, i, j, n_pts=20):
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)

            best_h = h_new

                h_new = make_feasible(h + d)
                sc = official_score(h_new)
