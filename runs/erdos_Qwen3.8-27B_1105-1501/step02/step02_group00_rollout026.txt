```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_objective(h, G, beta):
    scores = np.correlate(h, 1.0 - h, mode='full')
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    f = (s_max + np.log(np.sum(exp_s)) / beta)
    # gradient
    w = exp_s / np.sum(exp_s)
    grad = G.T @ w
    return f, grad

def _slsqp_optimize(h, target, beta, max_iter=200):
    n = len(h)
    G = _grad_matrix(h)
    
    def obj(x):
        scores = np.correlate(x, 1.0 - x, mode='full')
        s_max = float(np.max(scores))
        return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta
    
    def obj_grad(x):

        scores = np.correlate(x, 1.0 - x, mode='full')
        s_max = float(np.max(scores))
        exp_s = np.exp(beta * (scores - s_max))
        w = exp_s / np.sum(exp_s)
        return G_x.T @ w
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    if res.success:
        return _project(res.x, target)
    return h
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_and_grad(h):
    """Return scores and gradient matrix G where G[k,:] = ds_k/dh"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return scores, G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, beta):
    """Compute gradient of log-sum-exp smoothed max of scores"""
    scores, G = _scores_and_grad(h)
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    return G.T @ w, scores

def _slsqp_phase(h, target, deadline, betas):
    """Run SLSQP with increasing beta values"""
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        n = len(h)
        
        def obj(x):
            s = np.correlate(x, 1.0 - x, mode='full')
            s_max = float(np.max(s))
            return s_max + np.log(np.sum(np.exp(beta * (s - s_max)))) / beta
        
        def obj_grad(x):
            g, _ = _smooth_grad(x, beta)
            return g
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                       'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-10})
            if res.x is not None:
                h_try = _project(res.x, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
        except Exception:
            pass
    return best_h, best_c5

def _momentum_descent(h, target, deadline, lr_init=0.01, momentum=0.9, beta=20.0):
    """Momentum-based gradient descent on smooth objective"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    
    while time.time() < deadline:
        grad, scores = _smooth_grad(h, beta)
        # Project gradient onto feasible direction (sum = 0)
        grad = grad - np.mean(grad)
        
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = _project(h_new, target)
        
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            h = _project(h, target)  # ensure feasibility
            lr *= 0.7
            v *= 0.5
            if lr < 1e-6:
                break
    
    return best_h, best_c5

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.0, alpha_max, 20)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.1)
    hi = min(alpha_max, best_a + alpha_max * 0.1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break

        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _scores_and_grad(h)[1]
        p = _lp_direction(h, scores, G, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-10:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.04

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_search(h, deadline, max_passes=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for t in [0.3, 0.6, 0.9, 1.0]:
                        a = a_max * t
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break

    return best_h, best_c5
```

```python
from scipy.optimize import linprog, minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores_and_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return scores, G


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    T = 0.005

    while time.time() < deadline:
        if nesterov:
            y = h + momentum * v
            grad = _smooth_jac(y, beta)
        else:
            grad = _smooth_jac(h, beta)
        grad = grad - np.mean(grad)

        v = momentum * v + lr * grad
        h_new = _project(h - v, target)
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h)

        delta = c5_new - c5_cur
        accept = delta < 0
        if not accept and T > 1e-6:
            if np.random.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.7
            v *= 0.3

        T *= 0.998
        if lr < 1e-7:
            break
    return best_h, best_c5


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)

    best_a = 0.0
    best_c5 = c5_cur
    n_eval = 25
    alphas = np.linspace(0.0, alpha_max, n_eval)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    lo = max(0.0, best_a - alpha_max * 0.12)
    hi = min(alpha_max, best_a + alpha_max * 0.12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)

    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.1, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break

        scores, G = _scores_and_grad(h)
        p = _lp_direction(h, scores, G, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.72
            if stall % 12 == 0 and radius < 0.015:
                radius = 0.04

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_search(h, deadline, max_passes=25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-12:
                        continue
                    for t in [0.25, 0.5, 0.75, 1.0]:
                        a = a_max * t
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return h, best_c5


def _steepest_ascent_jump(h, target, rng, jump_size=0.03):
    n = len(h)
    scores, G = _scores_and_grad(h)
    s_max = float(np.max(scores))
    active = np.where(np.abs(scores - s_max) < 0.05 * s_max)[0]
    if len(active) == 0:
        active = np.where(scores == s_max)[0]
    G_active = G[active, :]
    grad = G_active.mean(axis=0)
    grad = grad - np.mean(grad)
    norm = float(np.sqrt(np.sum(grad ** 2)))
    if norm < 1e-12:
        direction = rng.standard_normal(n)
        direction -= np.mean(direction)
        norm = float(np.sqrt(np.sum(direction ** 2)))
        if norm < 1e-12:
            return h.copy()
        direction = direction / norm
    else:
        direction = grad / norm
    h_new = h + jump_size * direction
    h_new = _project(h_new, target)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP + trust-constr annealing
    deadline1 = t0 + budget_s * 0.25
    betas = [5.0, 12.0, 25.0, 50.0, 100.0, 200.0]
    h_s, c5_s = _slsqp_anneal(h, target0, deadline1, betas)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()
        h = h_s.copy()

    # Phase 2: Momentum / Nesterov descent with annealing
    deadline2 = t0 + budget_s * 0.45
    for beta in [8.0, 15.0, 30.0, 60.0]:
        if time.time() > deadline2:
            break
        h_m, c5_m = _momentum_descent(best_h, best_n / 2.0,
                                      min(deadline2, time.time() + budget_s * 0.05),
                                      beta=beta, lr_init=0.012, momentum=0.82,
                                      nesterov=True)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    # Phase 3: LP-based local search
    deadline3 = t0 + budget_s * 0.62
    h_lp, c5_lp = _lp_loop(best_h, deadline3, radius_init=0.12, max_iters=5000)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 4: Steepest ascent jumps + re-optimization
    deadline4 = t0 + budget_s * 0.80
    for attempt in range(200):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3807:
            break
        # Jump in steepest ascent direction
        h_jump = _steepest_ascent_jump(best_h, best_n / 2.0, rng,
                                       jump_size=rng.uniform(0.01, 0.05))
        # Quick SLSQP
        h_try = _slsqp_optimize(h_jump, best_n / 2.0, beta=30.0, max_iter=80)
        # Quick LP
        h_try, c5_try = _lp_loop(h_try, min(deadline4, time.time() + budget_s * 0.015),
                                  radius_init=0.06, max_iters=400)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Pair search polish
    if time.time() < t0 + budget_s * 0.88:
        deadline5 = t0 + budget_s * 0.90
        hp, c5p = _pair_search(best_h, deadline5, max_passes=30)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 6: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.95:
        deadline6 = t0 + budget_s * 0.96
        for dn in [1, -1, 2, -2, 3, -3]:
            if time.time() > deadline6:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 160:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _lp_loop(h_r, min(deadline6, time.time() + budget_s * 0.02),
                                  radius_init=0.08, max_iters=500)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.995,
                              radius_init=0.02, max_iters=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
