So the gradient is simply: for each j, grad[j] = h[j+s] (if j+s < n) + h[j-s] (if j-s ≥ 0)

This is just the cross-correlation of h with itself!

Actually, np.correlate(h, h, mode='full') gives the auto-correlation. For lag s, the auto-correlation is Σ_{i} h[i]*h[i+s] (with appropriate bounds), which is exactly S_s.

∂c_s/∂h = dx * (∂(Σ h[i])/∂h - ∂S_s/∂h) = dx * (1_vector - auto_corr_s)

where auto_corr_s is the s-th lag auto-correlation vector.

Then we can find the best value of h[i] (given all other coordinates fixed) by minimizing the resulting max. This is a 1D problem that can be solved exactly (the c_k are linear in h[i] for fixed other coordinates... wait, no, they're not linear because h[i] appears in 1-h[i] and h[i] both).

Actually, for fixed other coordinates, c_k as a function of h[i] is:
c_k = dx * [terms not involving h[i] + h[i]*(1-h[i]) * (contribution if i is at the "self" position) + h[i] * (terms from cross terms)]

More precisely, if h[i] appears in the product h[i]*h[i+s] or h[i+s]*h[i], then:

- The term (1-h[i])*h[i+s_a] is linear in h[i]
- The term h[i]*(1-h[i]) at k=0 is quadratic in h[i]

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -np.max(z) - 1.0, 1.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def lse_value_and_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m, np.ones(len(corr)) / len(corr)
    return m + np.log(s) / beta, e / s

def compute_lse_gradient(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    _, weights = lse_value_and_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                grad += (1.0 - 2.0 * h) * w * dx
            else:
                grad[s:] += (1.0 - h[:n - s]) * w * dx
                grad[:n - s] += (-h[s:]) * w * dx
        else:
            m = -s
            if m == 0:
                grad += (1.0 - 2.0 * h) * w * dx
            else:
                grad[:n - m] += (1.0 - h[m:]) * w * dx
                grad[m:] += (-h[:n - m]) * w * dx
    return grad

def _eq_fun(x, n):
    return np.sum(x) - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def slsqp_lse(h0, beta, maxiter=100, t_start=None, time_limit=None):
    n = len(h0)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        v, _ = lse_value_and_weights(corr, beta)
        return v

    def grad(x):
        return compute_lse_gradient(x, beta, dx)

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: _eq_fun(x, n),
                                     'jac': lambda x: _eq_jac(x, n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0.copy()

def active_gradient_step(h, n_active=5, step=0.02):
    """Take a step in the negative gradient direction of the top-n_active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # Active: lags within 5% of max
    active = np.where(corr >= m * 0.95)[0]
    if len(active) > n_active:
        idx = np.argsort(corr[active])[::-1][:n_active]
        active = active[idx]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                grad += (1.0 - 2.0 * h) * dx
            else:
                grad[s:] += (1.0 - h[:n - s]) * dx
                grad[:n - s] += (-h[s:]) * dx
        else:
            m2 = -s
            if m2 == 0:
                grad += (1.0 - 2.0 * h) * dx
            else:
                grad[:n - m2] += (1.0 - h[m2:]) * dx
                grad[m2:] += (-h[:n - m2]) * dx
    return make_feasible(h - step * grad)

def gradient_descent_loop(h0, max_iter=3000, time_limit=30, t_start=None):
    h = h0.copy()
    best_h = h.copy()
    best_score = official_score(h)
    n = len(h)
    beta = 80.0
    step = 0.05
    stall = 0

    for it in range(max_iter):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        # LSE-based gradient
        grad = compute_lse_gradient(h, beta, 2.0 / n)
        h_new = make_feasible(h - step * grad)
        new_score = official_score(h_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h_new.copy()
            stall = 0
            step = min(step * 1.03, 0.3)
        else:
            stall += 1
            step = max(step * 0.97, 0.003)
        h = h_new
        beta = min(beta * 1.005, 500.0)
        if stall > 300:
            break
    return best_h, best_score

def coord_descent(h0, max_rounds=50, time_limit=20, t_start=None):
    """Coordinate descent: for each i, try adjusting h[i] to reduce the max."""
    h = h0.copy()
    n = len(h)
    best_score = official_score(h)
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            # Try moving h[i] up or down in small steps
            orig = h[i]
            best_local = orig
            best_local_score = best_score
            for delta in [0.05, 0.1, 0.15, 0.2, -0.05, -0.1, -0.15, -0.2]:
                val = orig + delta
                if val < 0.0 or val > 1.0:
                    continue
                h[i] = val
                h = make_feasible(h)
                s = official_score(h)
                if s < best_local_score - 1e-10:
                    best_local_score = s
                    best_local = val
            h = np.clip(h, 0, 1)
            if best_local != orig:
                # Redo with best_local
                h[i] = best_local
                h = make_feasible(h)
                best_score = official_score(h)
                improved = True
            else:
                h[i] = orig
                h = make_feasible(h)
        if not improved:
            break
    return h, best_score

def swap_local_search(h0, rng, max_rounds=200, time_limit=25, t_start=None):
    """Pairwise swap local search guided by active lags."""
    h = h0.copy()
    n = len(h)
    best_score = official_score(h)
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        active = np.where(corr >= m * 0.9)[0]

        # Compute gradient from active lags
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                if s == 0:
                    grad += (1.0 - 2.0 * h) * dx
                else:
                    grad[s:] += (1.0 - h[:n - s]) * dx
                    grad[:n - s] += (-h[s:]) * dx
            else:
                m2 = -s
                if m2 == 0:
                    grad += (1.0 - 2.0 * h) * dx
                else:
                    grad[:n - m2] += (1.0 - h[m2:]) * dx
                    grad[m2:] += (-h[:n - m2]) * dx

        # Find donor (high h, positive grad) and receiver (low h, negative grad)
        donor_score = h * np.maximum(grad, 0)
        recv_score = (1.0 - h) * np.maximum(-grad, 0)

        sum_d = donor_score.sum()
        sum_r = recv_score.sum()
        if sum_d < 1e-15 or sum_r < 1e-15:
            break

        best_improve = 0
        best_move = None
        # Try several candidate moves
        for _ in range(30):
            i = rng.choice(n, p=donor_score / sum_d)
            j = rng.choice(n, p=recv_score / sum_r)
            if i == j:
                continue
            delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.4, 0.9)
            if delta < 0.02:
                continue
            save = h.copy()
            h[i] -= delta
            h[j] += delta
            h = make_feasible(h)
            s = official_score(h)
            h[:] = save
            if best_score - s > best_improve:
                best_improve = best_score - s
                best_move = (i, j, delta)

        if best_move:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = make_feasible(h)
            best_score = official_score(h)
        else:
            break
    return h, best_score
