def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Binary search for the shift that makes sum(clip(z+shift, 0, 1)) = target
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
            else:
                add = max(residual, -h[idx])
                if add >= 0:
                    break
                h[idx] += add
                residual -= add
    return np.clip(h, 0.0, 1.0)

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

        # This is correlate(h, e_p, 'full') for each p

        L = 2 * n - 1
        S = np.zeros((n, L))
        for p in range(n):
            start = n - 1 + p
            S[p, start:start+n] = h
        
        # S1mh for (1-h)
        S1mh = np.zeros((n, L))
        one_m_h = 1.0 - h
        for p in range(n):
            start = n - 1 + p
            S1mh[p, start:start+n] = one_m_h
        
        best_improve = current_max_raw + 1e-12  # We want to find t that gives max < current
        best_pair = None
        best_t = None

                t_hi = min(h[i], 1.0 - h[j])

                # B = correlate(d, 1-h) where d = e_j - e_i

                # So correlate(e_j, 1-h) = S1mh[j][::-1]
                B = S1mh[j][::-1] - S1mh[i][::-1]
                # C = correlate(d, d): has nonzeros at n-1 (val=2), n-1+i-j (val=-1), n-1+j-i (val=-1)
