def make_feasible(z):
    z = np.asarray(z, dtype=float).copy()

    target = n / 2.0
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_lag_value(h, s, dx):
    """Compute the raw correlation value for lag s (not multiplied by dx for gradient)."""
    n = len(h)
    if s >= 0:
        m = n - s
        if m <= 0:
            return 0.0
        return np.sum(h[s:s+m] * (1.0 - h[:m]))
    else:
        t = -s
        m = n - t
        if m <= 0:
            return 0.0
        return np.sum(h[:m] * (1.0 - h[t:t+m]))

def compute_lag_grad(h, s, dx):
    """Compute gradient of dx * lag_value(h, s) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return grad
        # grad[i] gets contribution from (1-h[i-s]) if i >= s
        # and from -h[i+s] if i <= n-1-s
        if m > 0:
            # For i in [s, n-1]: += (1 - h[i-s])
            grad[s:] += (1.0 - h[:m])
            # For i in [0, n-1-s]: += -h[i+s]
            grad[:m] -= h[s:]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return grad
        # For i in [0, n-1-t]: += (1 - h[i+t])
        grad[:m] += (1.0 - h[t:t+m])
        # For i in [t, n-1]: += -h[i-t]
        grad[t:] -= h[:m]
    return grad * dx

def solve_lp_direction(h, V, active_p, rho, n):
    """Solve the LP subproblem to find a direction u that reduces the max."""
    dx = 2.0 / n
    C = V.max()
    K = len(active_p)
    
    # Variables: [u_0, ..., u_{n-1}, t]
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: g_p^T u - t <= -V[p] for each active p
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, p in enumerate(active_p):
        s = int(p) - (n - 1)
        g = compute_lag_grad(h, s, dx)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -V[p]
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C))  # t bounded above by current C
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            u = res.x[:n]
            t_pred = res.x[n]
            if np.all(np.isfinite(u)):
                return u, t_pred
    except Exception:
        pass
    return None, None

def binary_anneal(h0, n_steps, is_dead, rng):
    """Simulated annealing on binary states."""
    n = len(h0)
    if n % 2 != 0:
        return None
    
    # Initialize from threshold
    b = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(b))
    target_ones = n // 2
    
    # Fix to exactly target_ones ones
    if n_ones > target_ones:
        # Remove excess ones (those with lowest h0 values among ones)
        one_idx = np.where(b == 1.0)[0]
        zero_idx = np.where(b == 0.0)[0]
        # Sort ones by h0 ascending (least "ones-like")
        order = np.argsort(h0[one_idx])
        for i in range(n_ones - target_ones):
            b[one_idx[order[i]]] = 0.0
    elif n_ones < target_ones:
        zero_idx = np.where(b == 0.0)[0]
        order = np.argsort(-h0[zero_idx])
        for i in range(target_ones - n_ones):
            b[zero_idx[order[i]]] = 1.0
    
    curr = b.copy()
    curr_score = score_exact(curr)
    best = curr.copy()
    best_score = curr_score
    
    T = max(1e-4, 0.5 * (0.5 - curr_score))  # start temperature
    # Actually let's use a fixed reasonable temperature
    T = 3e-4
    
    for step in range(n_steps):
        if is_dead():
            break
        # Pick a random 1 and a random 0
        ones = np.where(curr == 1.0)[0]
        zeros = np.where(curr == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        # Swap
        curr[i] = 0.0
        curr[j] = 1.0
        new_score = score_exact(curr)
        delta = new_score - curr_score
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            curr_score = new_score
            if new_score < best_score:
                best = curr.copy()
                best_score = new_score
        else:
            # Revert
            curr[i] = 1.0
            curr[j] = 0.0
        
        T *= 0.999
    
    return best

def two_point_search(h, n_tries, rng, is_dead):
    """Quick two-point exchange search (kept from previous, but simplified)."""
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b_idx = rng.integers(0, n)
        if a == b_idx:
            continue
        t_lo = max(-best[a], best[b_idx] - 1.0)
        t_hi = min(1.0 - best[a], best[b_idx])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span,
                        t_lo + (1.0 / 2.618) * span, t_lo + (2.0 / 2.618) * span]
        best_t = None
        best_local = best_sc
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b_idx] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            best[a] = val_a
            best[b_idx] = val_b
            sc = score_exact(best)
            if sc < best_local:
                best_local = sc
                best_t = t
            best[a] = val_a - t  # restore
            best[b_idx] = val_b + t  # restore
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] += best_t
            best[b_idx] -= best_t
            best_sc = best_local
    return best, best_sc
