```python
    n = len(z)
    target = n / 2.0

            else:

            h[i] += add
            res -= add
            if abs(res) < 1e-14:

    n = len(h)
    dx = 2.0 / n
    cv = _corr_vec(h)

        threshold = np.sort(cv)[::-1][top_k-1]
        mask = cv >= threshold
    else:

    mc = cv[mask].max()
    sh = np.where(mask, beta * (cv - mc), -500.0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.zeros(len(cv))
        w[mask] = 1.0 / mask.sum()
    else:
        w = ev / es
```

Let me also think about the coordinate descent approach. For a fixed set of active lags, if I fix all coordinates except one, the objective becomes a function of a single variable. Specifically, if I fix h[j] and vary it, the active lag values F_s become quadratic in h[j] (since each F_s is bilinear). So I can minimize the max of a few quadratics in one variable, which is a 1D minimization.

Actually, F_s as a function of h[j] (with all other h fixed) is:
F_s = (terms not involving h[j]) + h[j] * (1-h[j-s]) * 1_{valid} + (1-h[j]) * h[j+s] * 1_{valid}
= const + h[j]*(1-h[j-s]) - h[j]*h[j+s] + h[j+s]
Wait, let me be more careful.

F_s = sum_n h[n+s](1-h[n])

If I vary h[j]:
- h[j] appears as h[n+s] when n = j-s: contributes h[j]*(1-h[j-s]) (if valid)
- h[j] appears as h[n] when n = j: contributes h[j+s]*(1-h[j]) (if valid)

So F_s = F_s^rest + [h[j]*(1-h[j-s]) + h[j+s]*(1-h[j])] * 1_{valid terms}
where F_s^rest is the part not involving h[j].

Expanding: h[j]*(1-h[j-s]) - h[j]*(h[j+s]) + h[j+s] (for the valid terms)
= h[j] - h[j]*h[j-s] - h[j]*h[j+s] + h[j+s]
= h[j] * (1 - h[j-s] - h[j+s]) + h[j+s]

This is linear in h[j]! (Not quadratic, because h[j] doesn't appear in both a h[n+s] and h[n] term for the same s... wait, it can.)

Actually, can h[j] appear in both roles for the same s? h[j] = h[n+s] means n = j-s. h[j] = h[n] means n = j. These are the same only if j-s = j, i.e., s=0.

For s ≠ 0: F_s is linear in h[j].
For s = 0: F_0 = sum_n h[n](1-h[n]) = sum h[n] - sum h[n]². The part involving h[j] is h[j] - h[j]², which is quadratic.

So for s ≠ 0, varying h[j] gives a linear function of h[j]. The max of linear functions is convex, so we can find the minimum of max(linear functions) in 1D. This is a small convex optimization problem.

For s = 0, it's the max of a concave quadratic (h-j - h[j]²) and linear functions, which is nonconvex in 1D.

This coordinate descent idea is interesting! Let me implement it.

For a given j and current h, to find the optimal h[j]:
- For each active lag s, compute the coefficient of h[j] in F_s
- The constraint is 0 ≤ h[j] ≤ 1 and sum constraint (which fixes the sum of all h)
- Actually, if I change h[j], I need to maintain the sum constraint. So I should vary h[j] and compensate by changing another coordinate, or just ignore the sum constraint temporarily and re-project.

Hmm, this gets complicated. Let me just do:
1. Fix all h except h[j]

Or better:
1. Fix all h except h[j]
2. Find h[j] that minimizes the LSE (or max) of active F_s
3. Re-project and check if score improved

Actually, for the coordinate descent, I think the simplest approach is:
- For each j, try h[j] + δ for δ in a small grid
- Re-project
- Keep the change if it improves the score

```python
import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
            if abs(res) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def _grad_lse_fast(h, beta, top_k=0):
    n = len(h)
    dx = 2.0 / n
    cv = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k > 0 and top_k < 2 * n - 1:
        sorted_cv = np.sort(cv)[::-1]
        threshold = sorted_cv[top_k - 1]
        mask = cv >= threshold
    else:
        mask = np.ones(2 * n - 1, dtype=bool)
    mc = cv[mask].max()
    sh = np.where(mask, beta * (cv - mc), -600.0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    o1h = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                g[s:] += wk * o1h[:n - s]
                g[:n - s] -= wk * h[s:]
        else:
            p = -s
            if p < n:
                g[:n - p] += wk * o1h[p:]
                g[p:] -= wk * h[:n - p]
    return g * dx

def _lse_objective(x, beta, top_k=0):
    n = len(x)
    cv = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    return mc + np.log(ev.sum()) / beta

def _slsqp_lse(h0, beta, top_k=0, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return _lse_objective(x, beta, top_k)
    
    def grad(x):
        return _grad_lse_fast(x, beta, top_k)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _lbfgs_with_proj(h0, beta, top_k=0, n_iter=500):
    """L-BFGS with projection fallback."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return _lse_objective(x, beta, top_k)
    
    def grad(x):
        return _grad_lse_fast(x, beta, top_k)
    
    # Use penalty for sum constraint
    penalty = 1000.0
    
    def fun_pen(x):
        return _lse_objective(x, beta, top_k) + penalty * (x.sum() - n / 2.0) ** 2
    
    def grad_pen(x):
        return _grad_lse_fast(x, beta, top_k) + 2 * penalty * (x.sum() - n / 2.0)
    
    try:
        res = minimize(fun_pen, h0, jac=grad_pen, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': n_iter, 'ftol': 1e-15, 'gtol': 1e-10})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _fista(h0, beta, n_iter, alpha_init=0.005, mu=0.85, top_k=0):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    t = 1.0
    for it in range(n_iter):
        grad = _grad_lse_fast(y, beta, top_k)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x_new = _project(y - alpha_init * grad)
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _project(y)
        x = x_new
        t = t_new
        if it % 100 == 0:
            s = _score(x)
            if s < best_s:
                best_s = s
                best_h = x.copy()
    s = _score(x)
    if s < best_s:
        best_s = s
        best_h = x.copy()
    return best_h, best_s

def _coord_descend(h, n_sweeps=10):
    """Coordinate descent: for each coordinate, try to improve it."""
    n = len(h)
    best_s = _score(h)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            # Try small perturbations
            base = h.copy()
            for delta in [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]:
                h[j] = base[j] + delta
                if h[j] < 0 or h[j] > 1:
                    continue
                h = _project(h)
                s = _score(h)
                if s < best_s - 1e-12:
                    best_s = s
                    improved = True
                else:
                    h = base.copy()
        if not improved:
            break
    return h, best_s

def _slsqp_direct(h0, top_lags=4, maxiter=150):
    """Direct min-max: minimize t s.t. t >= corr_k for top_lags k."""
    n = len(h0)
    cv = _corr_vec(h0)
    # Get top lag indices
    top_idx = np.argsort(cv)[::-1][:top_lags]
    t0 = cv.max() + 0.001
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def fun(z):
        return z[-1]
    
    def fun_grad(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def ineq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv[top_idx]
    
    def eq_fun(z):
        return z[:-1].sum() - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    try:
        res = minimize(fun, z0, jac=fun_grad, method='SLSQP', bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': ineq},
                           {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0
```

def _project(z):

        else:
            hi = mid

    if abs(res) > 1e-10:
