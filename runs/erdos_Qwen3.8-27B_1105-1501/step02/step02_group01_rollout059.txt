from scipy.optimize import linprog, minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
    else:
        m = -s
        if m == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] += (-h[:n-m])
    return grad

    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(80):

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)

        else:
            order = np.argsort(h)
        for i in order:

            else:

            if abs(add) < 1e-15:

    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)

            grad[:] = 1.0 - 2.0 * h

For each coordinate i, the function f_k(h) is linear in h_i (for fixed other coordinates). So for each coordinate, the max of linear functions is a piecewise-linear convex function of h_i. We can find the optimal h_i in O(n_active) time.

This gives us a coordinate descent algorithm that's guaranteed to make progress (or prove optimality) at each coordinate.

**Algorithm: Min-max coordinate descent**
For each coordinate i (cycling through):
  1. Fix all h_j for j≠i
  2. For each active k, compute the slope of f_k w.r.t. h_i
  3. The function to minimize is max_k (a_k * h_i + b_k) where a_k is the slope and b_k is the intercept
  4. Find h_i* that minimizes this piecewise-linear function (subject to 0 ≤ h_i ≤ 1 and sum constraint)
  5. Update h_i = h_i*

- Move mass from j to i

def greedy_mass_transfer(h, max_iter=10000, time_limit=None, t_start=None, tol=1e-10):
    """Greedy mass transfer with active set tracking."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_t = np.max(np.correlate(h, 1-h, mode='full'))
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        f_vals = np.correlate(h, 1.0 - h, mode='full')
        t_cur = np.max(f_vals)
        
        # Active set: top lags
        n_active = min(10, 2*n-1)
        active = np.argsort(f_vals)[::-1][:n_active]
        
        # Compute gradient matrix: G[i,k] = g_k[i]
        G = np.zeros((n, n_active))
        for i, k in enumerate(active):
            G[:, i] = compute_grad_k(h, k)
        
        # For each coordinate, compute max active gradient
        coord_max = G.max(axis=1)  # shape (n,)
        coord_min = G.min(axis=1)
        
        # Best donor: highest coord_max (most dangerous to increase)
        j = np.argmax(coord_max)
        # Best receiver: lowest coord_min (safest to increase)
        i = np.argmin(coord_min)
        
        if i == j:
            # Try second best
            if j == np.argmax(coord_max):
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                i = np.argmin(coord_min * mask + 100*(1-mask))
        
        if i == j:
            break
        
        # Step size: try a few values
        best_delta = 0
        best_new_t = t_cur
        for delta in [0.05, 0.1, 0.2, 0.02, 0.01]:
            if h[j] < delta or h[i] > 1 - delta:
                delta = min(h[j], 1-h[i])
                if delta < 1e-6:
                    continue
            h[j] -= delta
            h[i] += delta
            new_t = np.max(np.correlate(h, 1.0-h, mode='full'))
            h[j] += delta
            h[i] -= delta
            if new_t < best_new_t:
                best_new_t = new_t
                best_delta = delta
        
        if best_delta == 0:
            break
        
        h[j] -= best_delta
        h[i] += best_delta
        h = make_feasible(h)  # ensure exact feasibility
        
        new_t = np.max(np.correlate(h, 1.0-h, mode='full'))
        if new_t < best_t:
            best_t = new_t
            best_h = h.copy()
        
        if t_cur - new_t < 1e-8:
            break
    
    return best_h, best_t

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
    else:
        m = -s
        if m == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] += (-h[:n-m])
    return grad

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        if s < target:
