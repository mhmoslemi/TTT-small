import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target=None):
    n = len(x)
    if target is None:
        target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    # Bisection on the threshold
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add: find entries with room to increase
            order = np.argsort(1.0 - result)
            for idx in order:
                room = 1.0 - result[idx]
                add = min(room, residual)
                result[idx] += add
                residual -= add
                if residual <= 1e-14:
                    break
        else:
            # Need to subtract
            order = np.argsort(result)
            for idx in order:
                sub = min(result[idx], -residual)
                result[idx] -= sub
                residual += sub
                if residual >= -1e-14:
                    break
    return result

def soft_objective_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    M = np.max(corr)
    # Softmax weights
    z = np.exp((corr - M) / T)
    S = np.sum(z)
    w = z / S
    value = dx * (M + T * np.log(S))
    # Gradient: grad[j] = dx * sum_l w[j+l] * (1-2*h[l])
    g = 1.0 - 2.0 * h  # length n
    grad = np.zeros(n)
    # For each j: grad[j] = dx * sum_{l=0}^{min(2n-2-j, n-1)} w[j+l] * g[l]
    # Build the Hankel matrix
    # W[j,l] = w[j+l]
    idx_l = np.arange(n)
    idx_j = np.arange(n)
    W = w[idx_j[:, None] + idx_l[None, :]]  # n x n matrix
    grad = dx * (W @ g)
    return value, grad

def slsqp_step(h, T, maxiter=50, ftol=1e-10):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        val, _ = soft_objective_and_grad(x, T)
        return val
    
    def jac(x):
        _, g = soft_objective_and_grad(x, T)
        return g
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    result = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
    if result.success or result.fun < obj(h):
        return result.x
    return h

def pair_line_search(h, i, j, iters=30):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hi_v, hj_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def full_pair_sweep(h, rng, deadline=None):
    n = len(h)
    improved = True
    passes = 0
    while improved and passes < 5:
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        perm = rng.permutation(n)
        for a in range(n):
            for b in range(a + 1, n):
                if deadline is not None and time.time() > deadline:
                    return
                i, j = int(perm[a]), int(perm[b])
                if i > j:
                    i, j = j, i
                delta = pair_line_search(h, i, j)
                if abs(delta) > 1e-12:
                    h[i] -= delta
                    h[j] += delta
                    improved = True
        passes += 1

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target=None):
    n = len(x)
    if target is None:
        target = n / 2.0
    x = np.asarray(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0

        proj = np.clip(x - mid, 0.0, 1.0)

        else:

    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(-(1.0 - result))
            for idx in order:
                room = 1.0 - result[idx]
                if room < 1e-15:
                    continue
                add = min(room, residual)
                result[idx] += add
                residual -= add
                if residual <= 1e-13:

            order = np.argsort(result)

                    continue
                sub = min(result[idx], -residual)
                result[idx] -= sub
                residual += sub
                if residual >= -1e-13:
                    break

    corr = np.correlate(h, v, mode="full")

    g = 1.0 - 2.0 * h
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    grad = dx * (W @ g)
    return value, grad

    target = n / 2.0
    
    def obj(x):

def pair_line_search(h, i, j, iters=25):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hi_v, hj_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def pair_sweep(h, rng, deadline=None, max_passes=3):
    n = len(h)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        perm = rng.permutation(n)
        for a in range(n):
            for b in range(a + 1, n):
                if deadline is not None and time.time() > deadline:
                    return
                i, j = int(perm[a]), int(perm[b])
                if i > j:
                    i, j = j, i
                d = pair_line_search(h, i, j)
                if abs(d) > 1e-12:
                    h[i] -= d
                    h[j] += d

def random_binary(n, rng):
    h = np.zeros(n)

def make_wave(n, freq, phase, rng):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / n + phase)
    h += rng.randn(n) * 0.05
    return project_feasible(h, n / 2.0)

def transfer_to_length(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_tgt = np.interp(x_tgt, x_src, h_src, left=0.0, right=0.0)
    return project_feasible(h_tgt, n_target / 2.0)

def compute_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target=None):
    n = len(x)
    if target is None:
        target = n / 2.0
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(-(1.0 - result))
            for idx in order:
                room = 1.0 - result[idx]
                if room < 1e-15:
                    continue
                add = min(room, residual)
                result[idx] += add
                residual -= add
                if residual <= 1e-13:
                    break
        else:
            order = np.argsort(result)
            for idx in order:
                if result[idx] < 1e-15:
                    continue
                sub = min(result[idx], -residual)
                result[idx] -= sub
                residual += sub
                if residual >= -1e-13:
                    break
    return result

def soft_value_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    M = np.max(corr)
    shifted = (corr - M) / T
    # Clamp for numerical stability
    shifted = np.clip(shifted, -500.0, 500.0)
    z = np.exp(shifted)
    S = np.sum(z)
    if S < 1e-300:
        S = 1e-300
    w = z / S
    value = dx * (M + T * np.log(S))
    g = 1.0 - 2.0 * h
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    grad = dx * (W @ g)
    return value, grad

def pair_line_search(h, i, j, iters=25):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hi_v, hj_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def pair_sweep(h, rng, deadline=None, max_passes=2):
    n = len(h)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            return
        perm = rng.permutation(n)
        for a in range(n):
            if deadline is not None and time.time() > deadline:
                return
            for b in range(a + 1, n):
                i, j = int(perm[a]), int(perm[b])
                if i > j:
                    i, j = j, i
                d = pair_line_search(h, i, j)
                if abs(d) > 1e-12:
                    h[i] -= d
                    h[j] += d
