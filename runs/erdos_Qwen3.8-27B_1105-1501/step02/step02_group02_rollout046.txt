Let me be careful about the gradient computation. For the LSE objective with top_m lags:

obj = mc + (1/beta) * log(sum_{k in A} exp(beta * (F_k - mc)))

grad = (1/sum_{k in A} exp(beta*(F_k-mc))) * sum_{k in A} exp(beta*(F_k-mc)) * dF_k/dh

= sum_{k in A} w_k * dF_k/dh

where w_k = exp(beta*(F_k-mc)) / sum_{j in A} exp(beta*(F_j-mc))

And dF_k/dh[p] for lag s = k-(n-1):
If s >= 0:
  dF_k/dh[p] = (1-h[p-s]) for p in [s, n-1]  (from h[p] * (1-h[p-s]))
              + (-h[p+s]) for p in [0, n-s-1]  (from -h[p] * h[p+s])

If s < 0, let t = -s:
  dF_k/dh[p] = (1-h[p+t]) for p in [0, n-t-1]
              + (-h[p-t]) for p in [t, n-1]

This is O(n) per lag, so O(n * top_m) total.

Let me code this efficiently using numpy vectorization:

```python
def _lse_obj_grad(x, beta, top_m=None):
    n = len(x)
    cv = _corr_full(x)  # O(n^2) but fast with numpy
    if top_m is not None and top_m < 2*n-1:
        idx = np.argsort(cv)[::-1][:top_m]
    else:
        idx = np.arange(2*n-1)
    
    cv_a = cv[idx]
    mc = cv_a.max()
    sh = np.clip(beta * (cv_a - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        obj = mc
        w = np.ones(len(idx)) / len(idx)
    else:
        obj = mc + np.log(es) / beta
        w = ev / es
    
    # Gradient: O(n * top_m)
    g = np.zeros(n)
    ones = 1.0 - x
    for ii, k in enumerate(idx):
        wk = w[ii]
        if wk < 1e-15: continue
        s = k - (n - 1)
        if s >= 0:
            # (1-x[p-s]) for p in [s, n-1]
            g[s:] += wk * ones[:n-s]
            # -x[p+s] for p in [0, n-s-1]
            g[:n-s] -= wk * x[s:]
        else:
            t = -s
            # (1-x[p+t]) for p in [0, n-t-1]
            g[:n-t] += wk * ones[t:]
            # -x[p-t] for p in [t, n-1]
            g[t:] -= wk * x[:n-t]
    
    return obj, g
```

```python
def _lse_obj_grad(x, beta, top_m=None):

    cv = _corr_full(x)
    if top_m is not None and top_m < 2 * n - 1:
        idx = np.argsort(cv)[::-1][:top_m]

        w = np.ones(len(idx)) / len(idx)

        else:
            t = -s
            g[:n - t] += wk * ones[t:]
            g[t:] -= wk * x[:n - t]
    return obj, g


def _slsqp_lse(h0, beta, top_m, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return _lse_obj_grad(x, beta, top_m)[0]

    def jac(x):
        return _lse_obj_grad(x, beta, top_m)[1]

    def efun(x):
        return x.sum() - n / 2.0

    def ejac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': efun, 'jac': ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res.x), _score(_project(res.x))
    except Exception:
        pass
    return h0, _score(h0)


def _fista_lse(h0, beta, n_iter, top_m, lr=0.005):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    x_old = h0.copy()
    t_param = 1.0
    best_x = h0.copy()
    best_s = _score(h0)
    for it in range(n_iter):
        _, g = _lse_obj_grad(y, beta, top_m)
        x_new = _project(y - lr * g)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = _project(x_new + ((t_param - 1.0) / t_new) * (x_new - x_old))
        x_old = x.copy()
        x = x_new
        t_param = t_new
        if it % 500 == 0:

            if s < best_s:

                best_x = x.copy()
    s = _score(x)
    if s < best_s:
        best_s = s
        best_x = x.copy()
    return best_x, best_s


def _active_polish(h0, n_outer=80):
    cur = h0.copy()
    cur_s = _score(cur)
    n = len(cur)
    for _ in range(n_outer):

                    cur_s = ts

def _direct_minmax(h0, maxiter=200):
    n = len(h0)
    cv = _corr_full(h0)
    t0 = min(cv.max() + 0.003, 0.98)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def fun(z):
        return z[-1]

    def fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def ieq(z):
        h = z[:-1]
        t = z[-1]
        return t - _corr_full(h)

    def eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0

    def eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(fun, z0, jac=fun_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': ieq},
                                    {'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res.x[:-1]), _score(_project(res.x[:-1]))
    except Exception:
        pass
    return h0, _score(h0)


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))
```
