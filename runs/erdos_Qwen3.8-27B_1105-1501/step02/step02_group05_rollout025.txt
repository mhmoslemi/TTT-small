```python
                    cj = (1.0 - h[j-s] if s <= j else 0.0) - (h[j+s] if j+s <= n-1 else 0.0)
                else:
                    t = -s
                    ci = (1.0 - h[i+t] if i+t <= n-1 else 0.0) - (h[i-t] if t <= i else 0.0)
                    cj = (1.0 - h[j+t] if j+t <= n-1 else 0.0) - (h[j-t] if t <= j else 0.0)
                d[p] = ci - cj
            
            # Golden section search for optimal delta
            a, b = d_lo, d_hi
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(50):
                m1 = a + gr * (b - a)
                m2 = b - gr * (b - a)
                if np.max(V + d * m1) <= np.max(V + d * m2):
                    b = m2
                else:
                    a = m1
            delta = 0.5 * (a + b)
            
            new_score = float(np.max(V + d * delta))
            if new_score < current_score - 1e-14:
                h[i] = max(0.0, min(1.0, hi + delta))
                h[j] = max(0.0, min(1.0, hj - delta))
                V = V + d * delta
                current_score = new_score
                sweep_improve += 1
        
        # Check convergence
        if sweep_improve == 0:
            break
    
    return h, V
```

This looks good. But the inner loop computing `d` for each p is O(n) in Python, which is slow. Let me vectorize it.

For a given pair (i, j), I need to compute d[p] for all p. Let me think about how to vectorize this.

For s ≥ 0 (p from n-1 to 2n-2):
c_s(i) = (1-h[i-s]) * [s ≤ i] - h[i+s] * [i+s ≤ n-1]

Let me define arrays:
- For index i: 
  - Term 1: (1-h[i-s]) for s = 0, 1, ..., min(i, n-1) → this is (1-h[i]), (1-h[i-1]), ..., (1-h[max(0,i-(n-1))])
    As an array indexed by s: (1-h[i-s]) for s=0..min(i,n-1)
  - Term 2: -h[i+s] for s = 0, 1, ..., n-1-i
    As an array indexed by s: -h[i], -h[i+1], ..., -h[n-1]

Hmm, it's a bit tricky to vectorize cleanly because of the different ranges. Let me just use numpy indexing.

For the "positive lag" part (s = 0 to n-1, corresponding to p = n-1 to 2n-2):
- For coordinate i:
  - a1[i, s] = 1-h[i-s] if s ≤ i else 0  → this is (1-h[max(0, i-s)]) * (s ≤ i)
  - a2[i, s] = -h[i+s] if i+s ≤ n-1 else 0

Actually, let me just precompute the "influence matrix" M where M[p, k] = c_p(k) (the rate of change of V[p] when h[k] increases by 1).

M has shape (2n-1, n). I can compute it once for a given h, and then for each pair (i,j), d = M[:, i] - M[:, j].

But M changes every time h changes. So I'd need to recompute M after each update. That's O(n²) per update, and there are O(n²) updates per sweep, giving O(n⁴) per sweep. For n=60, that's 12.96M per sweep. Still feasible.

Actually wait, I don't need to recompute all of M. After updating h[i] and h[j], only the rows of M that involve h[i] or h[j] change. But since the correlation at lag p involves h[i] and h[j] if they're in the right position, it's hard to tell which rows change.

Let me just recompute M from scratch after each update. No wait, that defeats the purpose of the incremental V update.

Alternative: don't maintain V incrementally. Instead, after each pair update, recompute V from scratch (O(n²) using np.correlate). Then the total cost per sweep is O(n² * n²) = O(n⁴) = 13M for n=60. In numpy, np.correlate for n=60 is very fast (maybe 10μs). So 1770 * 10μs ≈ 18ms per sweep. We could do ~50000 sweeps in the budget!

This is O(n) in Python. For the golden section search, I need c_i and c_j, and then d = c_i - c_j. Then I do ~50 evaluations of max(V + d*delta), each O(n).

Total per pair: O(n) [compute c_i, c_j] + O(50*n) [golden section] = O(51n).
Total per sweep: O(n² * 51n) = O(51n³) ≈ 51*216000 ≈ 11M for n=60.

```python
def compute_c_vec(h, k):
    """Compute c_s(k) for all lags, vectorized."""
    n = len(h)
    c = np.zeros(2*n-1)
    
    # For s >= 0 (p = n-1 to 2n-2):
    s_vals = np.arange(0, n)  # s = 0, 1, ..., n-1
    # Term 1: (1-h[k-s]) for s <= k
    valid1 = s_vals <= k
    idx1 = k - s_vals[valid1]  # indices into h
    c[n-1 + s_vals[valid1]] += (1.0 - h[idx1])
    # Term 2: -h[k+s] for k+s <= n-1
    valid2 = (k + s_vals) <= (n-1)
    idx2 = k + s_vals[valid2]
    c[n-1 + s_vals[valid2]] -= h[idx2]
    
    # For s < 0, i.e., t = -s > 0 (p = 0 to n-2):
    t_vals = np.arange(1, n)  # t = 1, 2, ..., n-1
    # c_{-t}(k) = (1-h[k+t]) * [k+t <= n-1] - h[k-t] * [t <= k]
    valid3 = (k + t_vals) <= (n-1)
    idx3 = k + t_vals[valid3]
    c[n-1 - t_vals[valid3]] += (1.0 - h[idx3])
    valid4 = t_vals <= k
    idx4 = k - t_vals[valid4]
    c[n-1 - t_vals[valid4]] -= h[idx4]
    
    return c
```

This should be fast in numpy.

```python
def eval_max(V, d, delta):
```

This is O(n) in numpy. With 50 iterations of golden section, it's 50 O(n) operations = very fast.

```python
import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    """Compute all correlation values (already scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c(h, k):
    """Compute the rate of change of each correlation value when h[k] increases by 1."""
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2*n-1)
    
    # For s >= 0: correlation index p = n-1+s
    # c_p = dx * [(1-h[k-s]) * (s <= k) - h[k+s] * (k+s <= n-1)]
    for s in range(n):
        p = n - 1 + s
        val = 0.0
        if s <= k:
            val += (1.0 - h[k-s])
        if k + s <= n-1:
            val -= h[k+s]
        c[p] = dx * val
    
    # For s < 0, t = -s: correlation index p = n-1-t
    for t in range(1, n):
        p = n - 1 - t
        val = 0.0
        if k + t <= n-1:
            val += (1.0 - h[k+t])
        if t <= k:
            val -= h[k-t]
        c[p] = dx * val
    
    return c
```
