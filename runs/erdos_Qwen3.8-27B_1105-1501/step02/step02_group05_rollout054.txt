```python
def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float)
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Greedy residual repair
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)  # most slack first
        for idx in order:
            if residual < 1e-12:
                break
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
    elif residual < -1e-12:

        order = np.argsort(-h)

                break

            h[idx] -= sub
            residual += sub
    return np.clip(h, 0.0, 1.0)

        c_vals[i] = V[p]

    A_ub = np.zeros((k, n_vars))

    A_ub[:, n] = -1.0
```

A good binary sequence would have the property that for every shift s, roughly half of the ones in S are "covered" by non-ones, and vice versa. This is related to "difference sets" or "Bose-Chowla" type constructions.

- Start from threshold of the continuous h
- Perform random swaps (swap a 1 and a 0)
- Accept based on simulated annealing criterion
- Track best binary state
- Convert best binary state to continuous (it's already continuous, just 0/1 values)
- Also try anti-symmetric transform of the binary state

```python
import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            if residual < 1e-13:
                break
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
    elif residual < -1e-12:
        order = np.argsort(-h)
        for idx in order:
            if -residual < 1e-13:
                break
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
    return np.clip(h, 0.0, 1.0)

def compute_lag_grad(h, p, n, dx):
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        idx_a = np.arange(m)
        grad[idx_a] += dx * (-h[idx_a + s])
        idx_b = np.arange(s, n)
        grad[idx_b] += dx * (1.0 - h[idx_b - s])
    else:
        t = -s
        m = n - t
        idx_a = np.arange(t, n)
        grad[idx_a] += dx * (-h[idx_a - t])
        idx_b = np.arange(0, m)
        grad[idx_b] += dx * (1.0 - h[idx_b + t])
    return grad

def lp_direction(h, V, active_p, rho, C):
    n = len(h)
    dx = 2.0 / n
    k = len(active_p)
    G = np.zeros((k, n))
    c_vals = np.zeros(k)
    for i, p in enumerate(active_p):
        G[i] = compute_lag_grad(h, int(p), n, dx)
        c_vals[i] = V[int(p)]
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((k, n_vars))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c_vals
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_b = max(-h[i], -rho)
        hi_b = min(1.0 - h[i], rho)
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, C))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            u = res.x[:n]
            t_pred = float(res.x[n])
            if np.max(np.abs(u)) < 1e-14:
                return None, None
            return u, t_pred
    except Exception:
        pass
    return None, None

def lp_refine(h, is_dead, time_budget, eta_init=0.008, rho_init=0.15):
    n = len(h)
    dx = 2.0 / n
    V = np.asarray(np.correlate(h, 1.0 - h, mode='full'), dtype=float) * dx
    C = float(V.max())
    best_h = h.copy()
    best_C = C
    eta = eta_init
    rho = rho_init
    stall = 0
    t_start = time.time()
    while True:
        if is_dead() or (time.time() - t_start) > time_budget:
            break
        active_mask = V >= C - eta
        active_p = np.where(active_mask)[0]
        pmax = int(np.argmax(V))
        if pmax not in active_p:
            active_p = np.append(active_p, pmax)
        if len(active_p) > 44:
            srt = active_p[np.argsort(-V[active_p])][:44]
            active_p = srt
        u, t_pred = lp_direction(h, V, active_p, rho, C)
        if u is None:
            rho *= 0.5
            if rho < 0.005:
                eta = max(1e-5, eta * 0.6)
                rho = rho_init
                stall += 1
                if stall > 30:
                    break
            continue
        improved = False
        for scale in [1.0, 0.75, 0.55, 0.4, 0.28, 0.18, 0.1, 0.05, 0.02]:
            h_trial = make_feasible(h + scale * u)
            V_trial = np.asarray(np.correlate(h_trial, 1.0 - h_trial, mode='full'), dtype=float) * dx
            C_trial = float(V_trial.max())
            if C_trial < C - 1e-12:
                h = h_trial
                V = V_trial
                C = C_trial
                improved = True
                if C < best_C:
                    best_C = C
                    best_h = h.copy()
                stall = 0
                break
        if not improved:
            rho *= 0.6
            if rho < 0.008:
                eta = max(1e-5, eta * 0.7)
                rho = rho_init
            stall += 1
            if stall > 50:
                break
        if improved and stall < 5:
            eta = max(1e-5, eta * 0.85)
    return best_h, best_C

def binary_from_continuous(h, rng):
    n = len(h)
    target = n // 2
    b = (h > 0.5).astype(float)
    cur_ones = int(np.sum(b))
    if cur_ones > target:
        idx_ones = np.where(b == 1)[0]
        idx_zeros = np.where(b == 0)[0]
        to_remove = idx_ones[np.argsort(h[idx_ones])[:cur_ones - target]]
        b[to_remove] = 0
    elif cur_ones < target:
        idx_zeros = np.where(b == 0)[0]
        to_add = idx_zeros[np.argsort(-h[idx_zeros])[:target - cur_ones]]
        b[to_add] = 1
    if int(np.sum(b)) != target:
        idx_ones = np.where(b == 1)[0]
        idx_zeros = np.where(b == 0)[0]
        while int(np.sum(b)) > target and len(idx_zeros) > 0:
            rm = idx_ones[rng.integers(0, len(idx_ones))]
            b[rm] = 0
            idx_ones = np.where(b == 1)[0]
        while int(np.sum(b)) < target and len(idx_ones) > 0:
            add = idx_zeros[rng.integers(0, len(idx_zeros))]
            b[add] = 1
            idx_zeros = np.where(b == 0)[0]
    return b

def binary_anneal(h0, rng, is_dead, time_budget):
    n = len(h0)
    target = n // 2
    if n % 2 != 0:
        target = n // 2
    b = binary_from_continuous(h0, rng)
    if int(np.sum(b)) != target:
        b = binary_from_continuous(h0, rng)
    cur_score = score_exact(b)
    best_b = b.copy()
    best_score = cur_score
    T = 3e-4
    t_start = time.time()
    it = 0
    while True:
        if is_dead() or (time.time() - t_start) > time_budget:
            break
        pos1 = np.where(b == 1)[0]
        pos0 = np.where(b == 0)[0]
        if len(pos1) == 0 or len(pos0) == 0:
            break
        i = pos1[rng.integers(0, len(pos1))]
        j = pos0[rng.integers(0, len(pos0))]
        b[i] = 0
        b[j] = 1
        new_score = score_exact(b)
        delta = new_score - cur_score
        if delta < -1e-13:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_b = b.copy()
        else:
            if rng.random() < np.exp(-max(delta, 0) / max(T, 1e-8)):
                cur_score = new_score
            else:
                b[i] = 1
                b[j] = 0
        T *= 0.9992
        it += 1
        if it % 5000 == 0 and T < 5e-6:
            break
    return best_b, best_score

def two_point_exchange(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for _ in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        best_local = best_sc
        best_t = None
        for frac in [0.0, 0.5, 0.25, 0.75, 1.0 / 2.618, 2.0 / 2.618]:
            t = t_lo + frac * span
            tmp = best.copy()
            tmp[a] = best[a] + t
            tmp[b] = best[b] - t
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] += best_t
            best[b] -= best_t
            best[a] = np.clip(best[a], 0, 1)
            best[b] = np.clip(best[b], 0, 1)
            best_sc = score_exact(best)
    return best, best_sc

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def subgrad_step(h, is_dead, n_iter=200):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_sc = score_exact(best)
    alpha = 0.03
    for _ in range(n_iter):
        if is_dead():
            break
        V = np.asarray(np.correlate(best, 1.0 - best, mode='full'), dtype=float) * dx
        C = float(V.max())
        tol = 0.03
        active = np.where(V >= C - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(V)])
        w = np.exp(40.0 * (V[active] - C))
        w = w / w.sum()
        grad = np.zeros(n)
        for idx, p in enumerate(active):
            g = compute_lag_grad(best, int(p), n, dx)
            grad += w[idx] * g
        d = -grad
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 1.5, alpha * 0.25]:
            h_trial = make_feasible(best + mult * d)
            sc = score_exact(h_trial)
            if sc < best_sc - 1e-13:
                best = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-6:
                break
        else:
            alpha = min(alpha * 1.1, 0.15)
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):

            if np.isfinite(ps):

                best_score = ps
    except (NameError, Exception):

    if best_h is None:
        n = 60
        x = (np.arange(n) + 0.5) * (2.0 / n)

    # Phase 2: More LP with different parameters

        h2, s2 = lp_refine(best_h, is_dead, time_budget=alloc, eta_init=0.015, rho_init=0.10)

            best_h = h2
            best_score = s2

    # Phase 3: Subgradient polish

        h3, s3 = subgrad_step(best_h, is_dead, n_iter=300)
        if s3 < best_score - 1e-13:
            best_h = h3
            best_score = s3

    # Phase 4: LP again from polished state
    if not is_dead():
        remaining = deadline - time.time()
        alloc = min(0.3 * remaining, 150.0)
        h4, s4 = lp_refine(best_h, is_dead, time_budget=alloc, eta_init=0.005, rho_init=0.12)

    # Phase 5: Binary annealing
    if not is_dead():
        n_b = len(best_h)
        if n_b % 2 == 0:

            alloc = min(0.15 * remaining, 120.0)
            bb, bs = binary_anneal(best_h, rng, is_dead, alloc)
            if bs < best_score - 1e-13:
                best_h = bb.copy()
                best_score = bs
                # LP refine from binary
                if not is_dead():
                    alloc2 = min(0.2 * (deadline - time.time()), 100.0)
                    hb, sb = lp_refine(bb, is_dead, time_budget=alloc2, eta_init=0.01, rho_init=0.15)
                    if sb < best_score - 1e-13:
                        best_h = hb
                        best_score = sb

    # Phase 6: Anti-symmetric + refine
    if not is_dead():
        h_sym = anti_sym_map(best_h)
        sc_sym = score_exact(h_sym)
        if sc_sym < best_score - 1e-13:
            best_h = h_sym
            best_score = sc_sym
        else:

                remaining = deadline - time.time()
                alloc = min(0.1 * remaining, 60.0)
                hs, ss = lp_refine(h_sym, is_dead, time_budget=alloc, eta_init=0.008, rho_init=0.12)
                if ss < best_score - 1e-13:
                    best_h = hs
                    best_score = ss

    # Phase 7: Resolution changes

        n_cur = len(best_h)
        candidates = []
        for dn in [-8, -6, -4, -2, 2, 4, 6, 8]:
            nt = n_cur + dn

                candidates.append(nt)
        for nc in [40, 48, 52, 56, 64, 72, 80]:

                candidates.append(nc)

        if s_tp < best_score - 1e-13:
            best_h = h_tp
            best_score = s_tp

    # Phase 9: Final LP polish
    if not is_dead() and (deadline - time.time()) > 20:
        alloc = min(deadline - time.time() - 10, 80.0)
        h_final, s_final = lp_refine(best_h, is_dead, time_budget=alloc, eta_init=0.004, rho_init=0.08)
        if s_final < best_score - 1e-13:

    # Phase 10: Random perturbations + local search
    if not is_dead() and (deadline - time.time()) > 40:
        n_best = len(best_h)
        for sigma in [0.04, 0.08, 0.12]:
            if is_dead():
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n_best))
            sc_pert = score_exact(h_pert)
            if sc_pert < best_score - 1e-13:
                best_h = h_pert
                best_score = sc_pert
            alloc = min(30.0, max(10.0, (deadline - time.time()) * 0.1))
            h_pp, s_pp = lp_refine(h_pert, is_dead, time_budget=alloc, eta_init=0.008, rho_init=0.12)
            if s_pp < best_score - 1e-13:
                best_h = h_pp
                best_score = s_pp

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-8

    return final_h, final_score, final_n
```
