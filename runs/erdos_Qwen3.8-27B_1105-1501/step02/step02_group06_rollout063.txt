import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, s):
    """Gradient of raw correlation at lag s (s = p - (n-1))."""
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    t = abs(s)
    # grad[j] = (1-h[j-t]) * [j >= t] - h[j+t] * [j < n-t]
    g[t:] += (1.0 - h[:n-t])
    g[:n-t] -= h[t:]
    return g

def make_feasible(z):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(h)
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        rem = abs(res)
        sign = 1 if res > 0 else -1
        for i in idx:
            if rem < 1e-15:
                break
            if sign > 0:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
            else:
                sub = min(rem, h[i])
                h[i] -= sub
                rem -= sub
    return np.clip(h, 0.0, 1.0)

def lp_direction(h0, k):
    """Find LP descent direction for top-k active lags."""
    n = len(h0)
    raw = compute_raw(h0) * (2.0 / n)
    top_idx = np.argsort(raw)[-k:]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = []
    b_ub = []
    for p in top_idx:
        s = p - (n - 1)
        g = grad_lag(h0, s)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(raw[p] - g @ h0)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    h_new = res.x[:n]
    return h_new - h0  # return direction

def exact_line(h0, d):
    """Find best alpha in [0, alpha_max] minimizing max raw at h0 + alpha*d."""
    n = len(h0)
    # alpha_max from box constraints
    amax = 10.0
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        amax = min(amax, np.min((1.0 - h0[pos]) / d[pos]))
    if np.any(neg):
        amax = min(amax, np.min(-h0[neg] / d[neg]))
    amax = min(amax, 1.0)  # cap at 1 for safety
    
    if amax < 1e-10:
        return 0.0
    
    def f(a):
        h = np.clip(h0 + a * d, 0.0, 1.0)
        return official_score(h)
    
    # Grid search
    alphas = np.linspace(0, amax, 40)
    vals = np.array([f(a) for a in alphas])
    best_i = int(np.argmin(vals))
    
    # Refine
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(len(alphas) - 1, best_i + 1)]
    
    res = minimize_scalar(f, bounds=(lo, hi), method='bounded', options={'xatol': 1e-8})
    if res.success:
        return float(res.x)
    return float(alphas[best_i])

def binary_local_search(h, max_iters=100):
    """Local search on binary patterns."""
    n = len(h)
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        # Flip some 1s to 0s (the ones with lowest "importance")
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by h value (lower h means more likely should be 0)
        order = np.argsort(h[ones_idx])
        for i in order[:n_ones - target_ones]:
            h_bin[ones_idx[i]] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in order[:target_ones - n_ones]:
            h_bin[zeros_idx[i]] = 1
    
    best_score = official_score(h_bin)
    best_h = h_bin.copy()
    
    for it in range(max_iters):
        # Try all pairs (flip one 1 to 0 and one 0 to 1)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        
        best_pair = None
        best_s = best_score
        
        for a in ones:
            for b in zeros:
                if a == b:
                    continue
                trial = best_h.copy()
                trial[a] = 0
                trial[b] = 1
                s = official_score(trial)
                if s < best_s - 1e-14:
                    best_s = s
                    best_pair = (a, b)
        
        if best_pair is None:
            break
        
        a, b = best_pair
        best_h[a] = 0
        best_h[b] = 1
        best_score = best_s
    
    return best_h, best_score

    raw = compute_raw(h0) * (2.0 / n)
    top_idx = np.argsort(raw)[-k:]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = []
    b_ub = []
    for p in top_idx:
        s = p - (n - 1)
        g = grad_lag(h0, s)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(raw[p] - g @ h0)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status != 0 or res.x is None:
        return None
    return res.x[:n] - h0

def exact_line(h0, d):
    n = len(h0)
    amax = 2.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        amax = min(amax, np.min((1.0 - h0[pos]) / d[pos]))
    if np.any(neg):
        amax = min(amax, np.min(-h0[neg] / d[neg]))
    if amax < 1e-12:
        return 0.0
    def f(a):
        h = np.clip(h0 + a * d, 0.0, 1.0)
        return official_score(h)
    alphas = np.linspace(0.0, amax, 30)
    vals = np.array([f(a) for a in alphas])
    bi = int(np.argmin(vals))
    lo = alphas[max(0, bi - 1)]
    hi = alphas[min(len(alphas) - 1, bi + 1)]
    if hi - lo < 1e-10:
        return float(alphas[bi])
    try:
        r = minimize_scalar(f, bounds=(lo, hi), method='bounded', options={'xatol': 1e-9})
        return float(r.x) if r.success else float(alphas[bi])
    except Exception:
        return float(alphas[bi])

def binary_local_search(h0, max_iters=50):
    n = len(h0)
    h_b = (h0 > 0.5).astype(float)
    n1 = int(h_b.sum())
    tgt = n // 2
    if n1 > tgt:
        oi = np.where(h_b == 1)[0]
        order = np.argsort(h0[oi])
        for i in order[:n1 - tgt]:
            h_b[oi[i]] = 0.0
    elif n1 < tgt:
        zi = np.where(h_b == 0)[0]
        order = np.argsort(-h0[zi])
        for i in order[:tgt - n1]:
            h_b[zi[i]] = 1.0
    best_s = official_score(h_b)
    best_cpp = h_b.copy()
    for _ in range(max_iters):
        ones = np.where(best_cpp == 1)[0]
        zeros = np.where(best_cpp == 0)[0]
        improved = False
        for a in ones:
            for b in zeros:
                trial = best_cpp.copy()
                trial[a] = 0.0
                trial[b] = 1.0
                s = official_score(trial)
                if s < best_s - 1e-13:
                    best_s = s
                    best_cpp = trial
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_cpp, best_s

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    return make_feasible(np.interp(tgt_c, src_c, h_src))

def smooth_obj(h_val, beta):
    raw = compute_raw(h_val) * (2.0 / len(h_val))
    cmax = raw.max()
    shifted = beta * (raw - cmax)
    shifted = np.clip(shifted, -50.0, 50.0)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def eq_fun(h_val):
    return np.sum(h_val) - len(h_val) / 2.0

def eq_jac(h_val):
    return np.ones(len(h_val))

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 12.0
    rng = np.random.default_rng(seed)

    def time_ok(margin=5.0):
        return (deadline - margin) - time.time() > 0

    try:
        h = make_feasible(np.asarray(initial_h_values, dtype=float).ravel())
    except (NameError, AttributeError, ValueError):
        h = make_feasible(np.full(60, 0.5))

    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_sc = official_score(best_h)

    # Phase 1: LP active-set with multiple k values
    for k_try in [3, 5, 7, 4, 6, 8, 5, 3]:
        if not time_ok(20):
            break
        h_cur = best_h.copy()
        stall = 0
        for it in range(200):
            if not time_ok(5):
                break
            d = lp_direction(h_cur, k_try)
            if d is None or np.max(np.abs(d)) < 1e-10:
                stall += 1
                if stall >= 5:
                    break
                continue
            alpha = exact_line(h_cur, d)
            if alpha < 1e-10:
                stall += 1
                if stall >= 5:
                    break
                continue
            h_new = make_feasible(h_cur + alpha * d)
            sc = official_score(h_new)
            if sc < best_sc - 1e-14:
                best_h = h_new
                best_sc = sc
                h_cur = h_new
                stall = 0
            elif sc < official_score(h_cur) - 1e-13:
                h_cur = h_new
                stall = 0
            else:
                improved = False
                for frac in [0.5, 0.25, 0.1]:
                    h_t = make_feasible(h_cur + alpha * frac * d)
                    sc_t = official_score(h_t)
                    if sc_t < best_sc - 1e-14:
                        best_h = h_t
                        best_sc = sc_t
                        h_cur = h_t
                        improved = True
                        stall = 0
                        break
                    if sc_t < official_score(h_cur) - 1e-13:
                        h_cur = h_t
                        improved = True
                        stall = 0
                        break
                if not improved:
                    stall += 1
                    if stall >= 8:
                        break

    # Phase 2: Binary local search + polish
    if time_ok(50):
        for attempt in range(4):
            if not time_ok(15):
                break
            h_bin, sc_bin = binary_local_search(best_h, max_iters=40)
            if sc_bin < best_sc - 1e-14:
                best_h = h_bin
                best_sc = sc_bin
            h_cur = best_h.copy()
            for k_try in [4, 6]:
                if not time_ok(5):
                    break
                for it in range(50):
                    if not time_ok(3):
                        break
                    d = lp_direction(h_cur, k_try)
                    if d is None or np.max(np.abs(d)) < 1e-10:
                        break
                    alpha = exact_line(h_cur, d)
                    if alpha < 1e-10:
                        break
                    h_new = make_feasible(h_cur + alpha * d)
                    sc = official_score(h_new)
                    if sc < best_sc - 1e-14:
                        best_h = h_new
                        best_sc = sc
                    h_cur = h_new if sc < official_score(h_cur) else h_cur

    # Phase 3: Subgradient with momentum
    if time_ok(50):
        h_cur = best_h.copy()
        mom = np.zeros(n)
        mu = 0.85
        lr = 0.03
        for it in range(300):
            if not time_ok(3):
                break
            raw = compute_raw(h_cur) * dx
            top5 = np.argsort(raw)[-5:]
            g = np.zeros(n)
            for p in top5:
                g += grad_lag(h_cur, p - (n - 1))
            g /= len(top5)
            norm_g = np.linalg.norm(g)
            if norm_g < 1e-10:
                break
            mom = mu * mom - lr * g
            h_new = make_feasible(h_cur + mom)
            sc = official_score(h_new)
            if sc < best_sc - 1e-14:
                best_h = h_new
                best_sc = sc
            if sc < official_score(h_cur) - 1e-14:
                h_cur = h_new
            else:
                lr *= 0.9
                if lr < 1e-7:
                    break

    # Phase 4: Random perturbations + LP
    if time_ok(60):
        for i in range(25):
            if not time_ok(8):
                break
            sigma = 0.03 + 0.08 * rng.random()
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_cur = h_cand
            for k_try in [5, 3]:
                if not time_ok(3):
                    break
                for it in range(20):
                    if not time_ok(2):
                        break
                    d = lp_direction(h_cur, k_try)
                    if d is None or np.max(np.abs(d)) < 1e-10:
                        break
                    alpha = exact_line(h_cur, d)
                    if alpha < 1e-10:
                        break
                    h_new = make_feasible(h_cur + alpha * d)
                    sc = official_score(h_new)
                    if sc < best_sc - 1e-14:
                        best_h = h_new
                        best_sc = sc
                    h_cur = h_new if sc < official_score(h_cur) else h_cur

    # Phase 5: Resolution sweep
    if time_ok(80):
        n_base = len(best_h)
        for n2 in sorted(set([max(16, n_base - 6), n_base - 3, n_base + 3, min(90, n_base + 6)])):
            if not time_ok(15):
                break
            h_int = interpolate_to(best_h, n2)
            sc_int = official_score(h_int)
            if sc_int < best_sc - 1e-14:
                best_h = h_int
                best_sc = sc_int
            h_cur = h_int
            for it in range(30):
                if not time_ok(5):
                    break
                d = lp_direction(h_cur, 5)
                if d is None or np.max(np.abs(d)) < 1e-10:
                    break
                alpha = exact_line(h_cur, d)
                if alpha < 1e-10:
                    break
                h_new = make_feasible(h_cur + alpha * d)
                sc = official_score(h_new)
                if sc < best_sc - 1e-14:
                    best_h = h_new
                    best_sc = sc
                h_cur = h_new if sc < official_score(h_cur) else h_cur

    # Phase 6: Alternative starting points
    if time_ok(60):
        n_alt = len(best_h)
        starts = []
        # Periodic pattern
        pat = np.tile([1, 1, 1, 0, 1, 0, 0, 0], n_alt // 8 + 1)[:n_alt]
        starts.append(make_feasible(pat.astype(float)))
        # Alternating
        alt = np.zeros(n_alt)
        alt[::2] = 1.0
        starts.append(make_feasible(alt * n_alt / 2.0))
        # Random binary
        rb = np.zeros(n_alt)
        rb[rng.choice(n_alt, n_alt // 2, replace=False)] = 1.0
        starts.append(rb.copy())
        for h_s in starts:
            if not time_ok(12):
                break
            h_cur = h_s
            for k_try in [5, 7]:
                if not time_ok(5):
                    break
                for it in range(25):
                    if not time_ok(2):
                        break
                    d = lp_direction(h_cur, k_try)
                    if d is None or np.max(np.abs(d)) < 1e-10:
                        break
                    alpha = exact_line(h_cur, d)
                    if alpha < 1e-10:
                        break
                    h_new = make_feasible(h_cur + alpha * d)
                    sc = official_score(h_new)
                    if sc < best_sc - 1e-14:
                        best_h = h_new
                        best_sc = sc
                    h_cur = h_new if sc < official_score(h_cur) else h_cur

    # Phase 7: Final intensive LP polish
    if time_ok(30):
        h_cur = best_h.copy()
        for k_try in [3, 4, 5, 6, 8]:
            if not time_ok(5):
                break
            for it in range(100):
                if not time_ok(2):
                    break
                d = lp_direction(h_cur, k_try)
                if d is None or np.max(np.abs(d)) < 1e-11:
                    break
                alpha = exact_line(h_cur, d)
                if alpha < 1e-11:
                    break
                h_new = make_feasible(h_cur + alpha * d)
                sc = official_score(h_new)
                if sc < best_sc - 1e-14:
                    best_h = h_new
                    best_sc = sc
                h_cur = h_new if sc < official_score(h_cur) else h_cur

    # Phase 8: SLSQP smooth max polish
    if time_ok(25):
        n_fin = len(best_h)
        for beta in [3000, 10000, 30000]:
            if not time_ok(8):
                break
            try:
                res = minimize(smooth_obj, best_h, args=(beta,), method='SLSQP',
                               bounds=[(0.0, 1.0)] * n_fin,
                               constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                               options={'maxiter': 80, 'ftol': 1e-14})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = make_feasible(res.x)
                    sc = official_score(h_new)
                    if sc < best_sc - 1e-14:
                        best_h = h_new
                        best_sc = sc
            except Exception:
                pass

    # Final validation
    best_h = make_feasible(best_h)
    final_sc = official_score(best_h)
    return best_h, final_sc + 1e-13, len(best_h)
