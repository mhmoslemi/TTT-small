Let me think about the mathematical structure:

The objective is C(h) = max_p dx * ∑_r h[r+k_p](1-h[r])

For a fixed lag k ≥ 0:
- V(k) = dx * ∑_{r=0}^{n-1-k} h[r+k](1-h[r])
- ∂V(k)/∂h[r+k] = dx * (1-h[r]) for r = 0,...,n-1-k
- ∂V(k)/∂h[r] = dx * (-h[r+k]) for r = 0,...,n-1-k

For k < 0, let q = -k:
- V(k) = dx * ∑_{r=q}^{n-1} h[r-q](1-h[r])
- ∂V(k)/∂h[r-q] = dx * (1-h[r]) for r = q,...,n-1
- ∂V(k)/∂h[r] = dx * (-h[r-q]) for r = q,...,n-1

The QP formulation:
- minimize t + μ * sum(d²)
- subject to:
  - For each active lag k_i: g_i · d ≤ vals[k_i] - C (where C = max(vals))
  - 0 ≤ h + d ≤ 1 (i.e., -h ≤ d ≤ 1-h)
  - sum(d) = 0

This is a convex QP with linear constraints. The key is that if we can find a direction d that decreases the predicted maximum, the actual maximum will (approximately) decrease.

Let me think about what makes this problem hard:
- It's a max of many linear-ish functions (the correlations)
- We want to minimize the maximum
- This is naturally a Chebyshev/semi-infinite type problem

The QP approach is essentially a trust-region method for minimizing the maximum of functions.

import numpy as np
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_for_lag(h, lag_idx):
    """Compute gradient of V(p) where p is index into the full correlation array."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        for r in range(n - k):
            grad[r + k] += dx * (1.0 - h[r])
            grad[r] -= dx * h[r + k]
    else:
        q = -k
        for r in range(q, n):
            grad[r - q] += dx * (1.0 - h[r])
            grad[r] -= dx * h[r - q]
    return grad

def gradient_matrix(h, active_indices):
    """Compute gradient rows for a set of active lag indices."""
    n = len(h)
    K = len(active_indices)
    G = np.zeros((K, n))
    for i, p in enumerate(active_indices):
        G[i] = gradient_for_lag(h, p)
    return G

    n = len(h)
    d = cp.Variable(n)
    t = cp.Variable()
    
    G = gradient_matrix(h, active_indices)

For lag k ≥ 0 (lag_idx = n-1+k):
V(k) = dx * Σ_{r=0}^{n-1-k} h[r+k] * (1-h[r])

∂V(k)/∂h[j] for j = r+k (where r = j-k): dx * (1-h[j-k]) = dx * (1-h[r])
∂V(k)/∂h[j] for j = r: dx * (-h[r+k]) = dx * (-h[j+k])

So for a fixed j:
- If j ≥ k: ∂V/∂h[j] gets contribution dx*(1-h[j-k]) from the h[r+k] term (where r=j-k)

import numpy as np
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_for_lag(h, lag_idx):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        for r in range(n - k):
            grad[r + k] += dx * (1.0 - h[r])
            grad[r] -= dx * h[r + k]
    else:
        q = -k
        for r in range(q, n):
            grad[r - q] += dx * (1.0 - h[r])
            grad[r] -= dx * h[r - q]
    return grad

def solve_qp(h, vals, active, mu, solver='ECOS'):
    n = len(h)
    d = cp.Variable(n)
    t = cp.Variable()
    G = np.array([gradient_for_lag(h, int(p)) for p in active])
    rhs = -vals[active]
    constraints = [
        G @ d - t <= rhs,
        d >= -h,
        d <= 1.0 - h,
        cp.sum(d) == 0
    ]
    objective = cp.Minimize(t + mu * cp.sum_squares(d))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
        if d.value is None:
            prob.solve(solver='SCS')
    except Exception:
        try:
            prob.solve(solver='SCS')
        except Exception:
            return None
    if d.value is None or not np.all(np.isfinite(d.value)):
        return None
    return np.asarray(d.value, dtype=float)

def qp_direction(h, mu, topK, solver='ECOS'):
    vals = compute_vals(h)
    C = vals.max()
    n_vals = len(vals)
    K = min(topK, n_vals)
    active = np.argsort(vals)[-K:]
    d = solve_qp(h, vals, active, mu, solver)
    return d, vals, C

def line_search(h, d, current_score):
    best_h = h.copy()
    best_sc = current_score
    for alpha in [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
        h_trial = make_feasible(h + alpha * d)
        sc = official_score(h_trial)
        if sc < best_sc:
            best_sc = sc
            best_h = h_trial
    return best_h, best_sc

def two_point_optimize(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for _ in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        # Try several t values
        span = t_hi - t_lo
        candidates = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                      t_lo + 0.2 * span, t_lo + 0.8 * span,
                      t_lo + 0.382 * span, t_lo + 0.618 * span]
        best_t = None
        best_local = best_sc
        for t in candidates:
            if t < t_lo - 1e-12 or t > t_hi + 1e-12:
                continue
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-12 or val_a > 1.0 + 1e-12 or val_b < -1e-12 or val_b > 1.0 + 1e-12:
                continue
            test = best.copy()
            test[a] = val_a
            test[b] = val_b
            sc = official_score(test)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = min(1.0, max(0.0, best[a] + best_t))
            best[b] = min(1.0, max(0.0, best[b] - best_t))
            best_sc = official_score(best)
    return best, best_sc

def subgradient_step(h, current_score, topK=5):
    vals = compute_vals(h)
    C = vals.max()
    n_vals = len(vals)
    K = min(topK, n_vals)
    active = np.argsort(vals)[-K:]
    # Average gradient of active lags
    G = np.array([gradient_for_lag(h, int(p)) for p in active])
    grad_avg = G.mean(axis=0)
    # Zero-mean direction
    d = -grad_avg
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return h, current_score
    d = d / norm
    return line_search(h, d, current_score)

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_for_lag(h, lag_idx):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        for r in range(n - k):
            grad[r + k] += dx * (1.0 - h[r])
            grad[r] -= dx * h[r + k]
    else:
        q = -k
        for r in range(q, n):
            grad[r - q] += dx * (1.0 - h[r])
            grad[r] -= dx * h[r - q]
    return grad

def build_qp(h, active, mu, solver):
    n = len(h)
    d = cp.Variable(n)
    t = cp.Variable()
    G = np.zeros((len(active), n))
    for i, p in enumerate(active):
        G[i] = gradient_for_lag(h, int(p))
    vals = compute_vals(h)
    rhs = -vals[active]
    constraints = [
        G @ d - t <= rhs,
        d >= -h,
        d <= 1.0 - h,
        cp.sum(d) == 0
    ]
    objective = cp.Minimize(t + mu * cp.sum_squares(d))
    prob = cp.Problem(objective, constraints)
    solvers = [solver, 'ECOS', 'SCS', 'CLARABEL', 'CVXOPT']
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and d.value is not None:
                dv = np.asarray(d.value, dtype=float)
                if np.all(np.isfinite(dv)):
                    return dv
        except Exception:
            continue
    return None

def qp_step(h, current_score, mu, topK, solver='ECOS'):
    vals = compute_vals(h)
    n_vals = len(vals)
    K = min(topK, n_vals)
    active = np.argsort(vals)[-K:]
    d = build_qp(h, active, mu, solver)
    if d is None:
        return None, current_score
    h_new = h.copy()
    best_sc = current_score
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08, 0.04]:
        h_trial = make_feasible(h + alpha * d)
        sc = official_score(h_trial)
        if sc < best_sc:
            best_sc = sc
            h_new = h_trial

def subgradient_step(h, current_score, topK=5):

def two_point_optimize(h, n_tries, rng, deadline_fn):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for _ in range(n_tries):
        if deadline_fn():
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        fracs = [0.0, 1.0, 0.5, 0.25, 0.75, 0.125, 0.875, 0.382, 0.618]
        best_t = None
        best_local = best_sc
        for fr in fracs:
            t = t_lo + fr * span
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-12 or val_a > 1.0 + 1e-12 or val_b < -1e-12 or val_b > 1.0 + 1e-12:
                continue
            test = best.copy()
            test[a] = val_a
            test[b] = val_b
            sc = official_score(test)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = min(1.0, max(0.0, best[a] + best_t))
            best[b] = min(1.0, max(0.0, best[b] - best_t))
            best_sc = official_score(best)
    return best, best_sc
