    dx = 2.0 / n

    for k in range(m):
        J[k, k] = 1.0
        J[n - 1 - k, k] = -1.0
    return J

def compute_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] += dx * (1.0 - h[s:])
            G[p, s:] -= dx * h[:L]
        else:
            t = -s
            K = n - t
            G[p, t:] += dx * (1.0 - h[:K])
            G[p, :K] -= dx * h[t:]
    return G

def logsumexp_obj_and_grad(u, n, beta):
    m = n // 2
    h = build_sym_h(u, n)
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    
    G = compute_G(h, n)
    J = build_J(n, m)
    grad = (w @ G) @ J
    return obj, grad

    """Projected gradient with log-sum-exp smoothing."""
    m = n // 2
    u = u0.copy()
    t0 = time.time()
    
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        obj, grad = logsumexp_obj_and_grad(u, n, beta)
        
        # Projected gradient descent with backtracking
        alpha = 1.0 / (np.linalg.norm(grad)**2 + 1e-12)
        for it in range(5000):
            if time.time() - t0 > time_limit:
                break
            grad = logsumexp_obj_and_grad(u, n, beta)[1]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:

            step = alpha

                u_new = np.clip(u - step * grad, 0.0, 1.0)
                obj_new = logsumexp_obj_and_grad(u_new, n, beta)[0]
                if obj_new <= obj - 1e-4 * step * gnorm**2:

                step *= 0.4
            if step < 1e-14:
                break
            
            u = u_new
            obj = obj_new
            alpha = min(alpha * 1.5, 2.0 / (gnorm**2 + 1e-12))

from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def build_sym_h(u, n):
    m = n // 2
    h = np.empty(n)
    if n % 2 == 0:
        h[:m] = u
        h[m:] = 1.0 - u[::-1]
    else:
        h[:m] = u
        h[m] = 0.5
        h[m+1:] = 1.0 - u[::-1]
    return h

def antisym_project(h):
    n = len(h)
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return np.clip(h_sym, 0.0, 1.0)

def extract_u(h, n):
    m = n // 2
    return h[:m].copy()

def interp_to_n(h_src, n_tgt):
    n_src = len(h_src)
    src_x = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_x = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_x, src_x, h_src)
    h_sym = antisym_project(h_int)
    return extract_u(h_sym, n_tgt)

def compute_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, :L] += dx * (1.0 - h[s:])
            G[p, s:] -= dx * h[:L]
        else:
            t = -s
            K = n - t
            G[p, t:] += dx * (1.0 - h[:K])
            G[p, :K] -= dx * h[t:]
    return G

def build_J(n, m):
    J = np.zeros((n, m))
    for k in range(m):
        J[k, k] = 1.0
        J[n - 1 - k, k] = -1.0
    return J

def lse_obj_grad(u, n, beta):
    m = n // 2
    h = build_sym_h(u, n)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    logsum = shifted.max() + np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    G = compute_G(h, n)
    J = build_J(n, m)
    grad = (w @ G) @ J
    return obj, grad

def pg_solve_u(u0, n, betas, time_limit):
    m = n // 2
    u = u0.copy()
    t0 = time.time()
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        obj, grad = lse_obj_grad(u, n, beta)
        gnorm2 = float(grad @ grad)
        if gnorm2 < 1e-28:
            continue
        alpha = min(1.0 / (gnorm2 + 1e-12), 2.0 / (np.max(grad**2) + 1e-12))
        stall = 0
        for it in range(8000):
            if time.time() - t0 > time_limit:
                break
            obj, grad = lse_obj_grad(u, n, beta)
            gnorm2 = float(grad @ grad)
            if gnorm2 < 1e-26:
                break
            step = alpha
            for _ in range(40):
                u_new = np.clip(u - step * grad, 0.0, 1.0)
                obj_new, _ = lse_obj_grad(u_new, n, beta)
                if obj_new <= obj - 1e-5 * step * gnorm2:
                    break
                step *= 0.4

            obj = obj_new

def slsqp_lse(u0, n, beta, max_iter=300):
    m = n // 2
    def f_obj(u):
        return lse_obj_grad(u, n, beta)[0]
    def f_grad(u):
        return lse_obj_grad(u, n, beta)[1]
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(f_obj, u0, jac=f_grad, method='SLSQP',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return u0

def lbfgs_lse(u0, n, beta, max_iter=500):
    m = n // 2
    def f_obj(u):
        return lse_obj_grad(u, n, beta)[0]
    def f_grad(u):
        return lse_obj_grad(u, n, beta)[1]

        res = minimize(f_obj, u0, jac=f_grad, method='L-BFGS-B',

    except Exception:
        pass
    return u0

def active_set_solve(u0, n, max_iter=400):
    """Direct minimax with active set constraints using SLSQP."""
    m = n // 2
    h0 = build_sym_h(u0, n)
    dx = 2.0 / n
    vals = np.correlate(h0, 1.0 - h0, mode='full') * dx
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-7)
    active = np.where(vals >= threshold)[0]
    top_k = max(10, int(0.2 * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active = np.unique(np.concatenate([active, top_idx]))
    
    def obj_func(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(z):
        h = build_sym_h(z[:m], n)
        cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return z[-1] - cv[active]
    
    def ineq_jac(z):
        h = build_sym_h(z[:m], n)
        G = compute_G(h, n)
        J = build_J(n, m)
        Gj = G[active, :] @ J
        Jout = np.zeros((len(active), m + 1))
        Jout[:, :m] = -Gj
        Jout[:, m] = 1.0
        return Jout
    
    t0_start = cmax + 2e-7
    z0 = np.concatenate([u0, [t0_start]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    try:
        res = minimize(obj_func, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            u_new = np.clip(res.x[:m], 0.0, 1.0)
            return u_new
    except Exception:
        pass
    return u0

def subgradient_step_u(u, n, step_size):
    m = n // 2
    h = build_sym_h(u, n)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-6)
    active = np.where(vals >= threshold)[0]
    G = compute_G(h, n)
    J = build_J(n, m)
    g = G[active, :].mean(axis=0) @ J
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-15:
        return u
    return np.clip(u - step_size * g / gnorm, 0.0, 1.0)
