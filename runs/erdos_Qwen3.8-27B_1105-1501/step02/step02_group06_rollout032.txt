```python
def compute_Bprime(h, i, j):
    n = len(h)
    p = np.arange(2*n - 1)
    offs = (n - 1) - p
    
    # For coordinate i
    i_plus = i + offs
    i_minus = i - offs
    B = np.zeros(2*n - 1)
    C = np.zeros(2*n - 1)
    mask_bp = (i_plus >= 0) & (i_plus < n)
    mask_cp = (i_minus >= 0) & (i_minus < n)
    B[mask_bp] = 1.0 - h[i_plus[mask_bp]]
    C[mask_cp] = h[i_minus[mask_cp]]
    
    # For coordinate j
    j_plus = j + offs
    j_minus = j - offs
    D = np.zeros(2*n - 1)
    E = np.zeros(2*n - 1)
    mask_dp = (j_plus >= 0) & (j_plus < n)
    mask_ep = (j_minus >= 0) & (j_minus < n)
    D[mask_dp] = 1.0 - h[j_plus[mask_dp]]
    E[mask_ep] = h[j_minus[mask_ep]]
    
    return (B - C) - (D - E)
```

```python
    n = len(h)
```

```python
def compute_f(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    # Bisection on shift to make sum(clip(z+shift, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix small residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
```

The minimum of max_p (F_p + δ * Bprime_p) occurs at:

- An intersection of two lines f_p(δ) = f_q(δ) where p has Bprime > 0 and q has Bprime < 0 (or vice versa)

2. Find q* at δ_max: q* = argmax_p (F_p + δ_max * Bprime_p)
3. If p* == q*, the minimum is at the boundary where f_{p*} is smallest
4. Otherwise, compute δ_cross = (F_{q*} - F_{p*}) / (Bprime_{p*} - Bprime_{q*})
5. If δ_cross ∈ [δ_min, δ_max], check if it's valid (no other f_r exceeds at δ_cross)
6. If valid, use δ_cross; otherwise, use the boundary

This is O(n) per step instead of O(n * 60). Much faster!

```python
    # Find active at boundaries
    f_at_min = F + delta_min * Bprime
    f_at_max = F + delta_max * Bprime
    p_star = np.argmax(f_at_min)
    q_star = np.argmax(f_at_max)

        if Bprime[p_star] >= 0:
            return delta_min
        else:
            return delta_max

    denom = Bprime[p_star] - Bprime[q_star]

        return delta_min if F[p_star] < F[q_star] else delta_max

    max_at_cross = np.max(f_at_cross)
    target_val = F[p_star] + delta_cross * Bprime[p_star]
    
    # Allow small tolerance
    if max_at_cross <= target_val + 1e-12:
        return delta_cross
    
    # Intersection not valid, fall back to golden section
    gr = (np.sqrt(5.0) + 3.0) / 2.0
    inv_gr = 2.0 - gr
    a, b = delta_min, delta_max
    c = b - inv_gr * (b - a)
    d = a + inv_gr * (b - a)
    fc = np.max(F + c * Bprime)
    fd = np.max(F + d * Bprime)
    
    for _ in range(50):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - inv_gr * (b - a)
            fc = np.max(F + c * Bprime)
        else:
            a = c; c = d; fc = fd
            d = a + inv_gr * (b - a)
            fd = np.max(F + d * Bprime)
    
    return (a + b) / 2.0
```

```python
def simultaneous_pair_sweep(h, dx, F, rng, iters_per_sweep=60):
    n = len(h)
    h = h.copy()
    p = np.arange(2*n - 1)
    offs = (n-1) - p
    
    for it in range(iters_per_sweep):
        # Create random pairing
        perm = rng.permutation(n)
        pairs = [(perm[2*k], perm[2*k+1]) for k in range(n//2)]
        
        # For each pair, compute Bprime and solve 1D problem
        deltas = np.zeros(len(pairs))
        for k, (i, j) in enumerate(pairs):
            # Compute Bprime
            i_plus = i + offs
            i_minus = i - offs
            j_plus = j + offs
            j_minus = j - offs
            
            B = np.where((i_plus >= 0) & (i_plus < n), 1.0 - h[np.clip(i_plus, 0, n-1)], 0.0)
            C = np.where((i_minus >= 0) & (i_minus < n), h[np.clip(i_minus, 0, n-1)], 0.0)
            D = np.where((j_plus >= 0) & (j_plus < n), 1.0 - h[np.clip(j_plus, 0, n-1)], 0.0)
            E = np.where((j_minus >= 0) & (j_minus < n), h[np.clip(j_minus, 0, n-1)], 0.0)
            
            # Zero out invalid positions
            B[~((i_plus >= 0) & (i_plus < n))] = 0.0
            C[~((i_minus >= 0) & (i_minus < n))] = 0.0
            D[~((j_plus >= 0) & (j_plus < n))] = 0.0
            E[~((j_minus >= 0) & (j_minus < n))] = 0.0
            
            Bprime = (B - C) - (D - E)
            delta_min = max(-h[i], -(1.0 - h[j]))
            delta_max = min(1.0 - h[i], h[j])
            
            if delta_max - delta_min < 1e-14:
                continue
            
            deltas[k] = solve_1d_minimax(F, Bprime, delta_min, delta_max)
        
        # Apply all updates
        for k, (i, j) in enumerate(pairs):
            h[i] += deltas[k]
            h[j] -= deltas[k]
        
        # Update F: for each pair, F += delta * Bprime
        # But this is approximate since Bprime was computed before the update
        # For accuracy, recompute F
        F = compute_f(h)
    
    return h, F
```

```python
def simultaneous_sweep(h, F, rng, n_pairs=None):
    n = len(h)
    dx = 2.0 / n
    p_idx = np.arange(2*n - 1)
    offs = (n-1) - p_idx
    
    if n_pairs is None:
        n_pairs = n // 2
    
    perm = rng.permutation(n)
    
    # Compute all Bprimes using current h
    Bprimes = []
    for k in range(n_pairs):
        i, j = perm[2*k], perm[2*k+1]
        i_plus = i + offs
        i_minus = i - offs
        j_plus = j + offs
        j_minus = j - offs
        
        mask_ip = (i_plus >= 0) & (i_plus < n)
        mask_im = (i_minus >= 0) & (i_minus < n)
        mask_jp = (j_plus >= 0) & (j_plus < n)
        mask_jm = (j_minus >= 0) & (j_minus < n)
        
        B = np.zeros(2*n-1); C = np.zeros(2*n-1)
        D = np.zeros(2*n-1); E = np.zeros(2*n-1)
        B[mask_ip] = 1.0 - h[i_plus[mask_ip]]
        C[mask_im] = h[i_minus[mask_im]]
        D[mask_jp] = 1.0 - h[j_plus[mask_jp]]
        E[mask_jm] = h[j_minus[mask_jm]]
        
        Bprimes.append((B - C) - (D - E))
    
    # Solve all 1D problems
    F_total_change = np.zeros(2*n-1)
    for k in range(n_pairs):
        i, j = perm[2*k], perm[2*k+1]
        Bprime = Bprimes[k]
        delta_min = max(-h[i], -(1.0 - h[j]))
        delta_max = min(1.0 - h[i], h[j])
        if delta_max - delta_min < 1e-14:
            continue
        delta = solve_1d_minimax(F, Bprime, delta_min, delta_max)
        h[i] += delta
        h[j] -= delta
        F_total_change += delta * Bprime
    
    F = F + F_total_change * dx
    return h, F
```
