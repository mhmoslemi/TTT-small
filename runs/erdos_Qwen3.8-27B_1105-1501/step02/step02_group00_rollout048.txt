    A_eq = np.zeros((1, n + 1))

    grad = G.T @ w
    return grad, corr

def _sumexp_descent(h, deadline, n_iter=500, beta_init=20.0, beta_max=200.0):
    """Sum-exp smoothed projected gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = beta_init
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        grad, corr = _sumexp_gradient(h, beta)
        c5_cur = float(np.max(corr) * (2.0/n))
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Step size: start with a reasonable value, use backtracking
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-10:
            break
        
        # Try several step sizes
        base_alpha = 0.01 / grad_norm
        improved = False
        for trial in range(10):
            alpha = base_alpha * (0.5 ** trial)
            h_new = h - alpha * grad
            # Project onto box [0,1] and sum constraint
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                improved = True
                break
        
        if not improved:
            # Try applying directly without the initial clip
            for trial in range(5):
                alpha = base_alpha * (0.5 ** trial)
                h_new = _project(h - alpha * grad, target)
                c5_new = _exact_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    h = h_new
                    improved = True
                    break
        
        if not improved:
            # Increase beta to sharpen
            beta = min(beta * 1.3, beta_max)
            if beta > beta_max * 0.9:
                break
        else:
            # Slightly increase beta
            beta = min(beta * 1.05, beta_max)
    
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-10:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 10 == 0 and radius < 0.03:
                radius = 0.08
    
    return best_h, best_c5

def _perturb_lp(h, deadline, rng, max_attempts=100, scale_init=0.05):
    """Try random perturbations followed by short LP optimization."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = scale_init
    
    for attempt in range(max_attempts):
        if time.time() > deadline:
            break
        if best_c5 < 0.3800:
            break
        
        # Structured perturbation
        pert_type = attempt % 3
        if pert_type == 0:
            # Gaussian
            h_p = h + rng.normal(0, scale, n)
        elif pert_type == 1:
            # Low-frequency sinusoidal
            freq = rng.integers(1, 4)
            phase = rng.uniform(0, 2*np.pi)
            h_p = h + scale * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
        else:
            # Block perturbation
            block_size = rng.integers(4, n//4 + 1)
            start = rng.integers(0, n - block_size)
            h_p = h.copy()
            h_p[start:start+block_size] += scale * rng.uniform(-1, 1, block_size)
        
        h_p = _project(h_p, target)
        
        # Short LP from perturbed point
        sub_deadline = min(deadline, time.time() + max(0.5, (deadline - time.time()) / max(1, max_attempts - attempt)))
        h_try, c5_try = _lp_loop(h_p, sub_deadline, radius_init=0.08, max_iters=300)
        
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.2)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_search(h, deadline, max_passes=20):
    """Pair coordinate descent for final polish."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                # Try increasing j, decreasing k (and vice versa)
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    # Try a few step sizes
                    for a in [a_max * 0.5, a_max * 0.25, a_max * 0.75]:
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    
    return best_h, best_c5

from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        m = n - d
        if m <= 0:
            continue
        r = n - 1 + d
        G[r, :m] += (1.0 - h[d:d + m])
        G[r, d:d + m] += (-h[0:m])
    for g in range(1, n):
        m = n - g
        if m <= 0:
            continue
        r = n - 1 - g
        G[r, 0:m] += (-h[g:g + m])
        G[r, g:g + m] += (1.0 - h[0:m])
    return G


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

                alpha_max = am
        elif p[j] < -1e-14:
            am = h[j] / (-p[j])
            if am < alpha_max:
                alpha_max = am
    alpha_max = max(alpha_max, 1e-8)

    alphas = np.linspace(0.005, alpha_max, 25)

            best_h = h_try.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.25, scale * 1.2)

    return best_h, best_c5


def _pair_search(h, deadline, max_passes=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for a in [a_max * 0.5, a_max * 0.25, a_max * 0.75]:
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _triple_search(h, deadline, max_passes=15):
    """3-coordinate descent for final polish."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                for k in range(j + 1, n):
                    if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                        continue
                    # Try: inc i, dec j, keep k balanced (net zero sum)
                    # Or: inc i, dec j, dec k (would violate sum) - skip
                    # Only do (inc i, dec j) and (dec i, inc j) as in pair
                    # For triple: (inc i, dec j, dec k) and (dec i, inc j, inc k)
                    for di, dj, dk in [(1.0, -0.5, -0.5), (-1.0, 0.5, 0.5)]:
                        a_max = 1e9
                        if di > 0:
                            a_max = min(a_max, (1.0 - h[i]) / di)
                        else:
                            a_max = min(a_max, h[i] / (-di))
                        if dj > 0:
                            a_max = min(a_max, (1.0 - h[j]) / dj)
                        else:
                            a_max = min(a_max, h[j] / (-dj))
                        if dk > 0:
                            a_max = min(a_max, (1.0 - h[k]) / dk)
                        else:
                            a_max = min(a_max, h[k] / (-dk))
                        if a_max < 1e-10:
                            continue
                        for a in [a_max * 0.4, a_max * 0.7]:
                            h_t = h.copy()
                            h_t[i] += di * a
                            h_t[j] += dj * a
                            h_t[k] += dk * a
                            c5_t = _exact_c5(h_t)
                            if c5_t < best_c5 - 1e-12:
                                best_c5 = c5_t
                                best_h = h_t.copy()
                                h = h_t.copy()
                                improved = True
                                break
        if not improved:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Aggressive LP optimization (40% of budget)
    deadline1 = t0 + budget_s * 0.35
    h, c5 = _lp_loop(h, deadline1, radius_init=0.25, max_iters=15000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Sum-exp gradient descent to escape LP local optima
    deadline2 = t0 + budget_s * 0.48
    h_se, c5_se = _sumexp_descent(best_h, deadline2, n_iter=400, beta_init=25.0, beta_max=400.0)
    if c5_se < best_c5:
        best_c5 = c5_se
        best_h = h_se.copy()

    # Phase 3: LP from sumexp result
    deadline3 = t0 + budget_s * 0.58
    h, c5 = _lp_loop(best_h, deadline3, radius_init=0.15, max_iters=8000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Structured perturbations + LP
    deadline4 = t0 + budget_s * 0.78
    h_p, c5_p = _perturb_lp(best_h, deadline4, rng, max_attempts=200, scale_init=0.035)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 5: Another sumexp pass from best
    deadline5 = t0 + budget_s * 0.85
    h_se2, c5_se2 = _sumexp_descent(best_h, deadline5, n_iter=200, beta_init=50.0, beta_max=600.0)
    if c5_se2 < best_c5:
        best_c5 = c5_se2
        best_h = h_se2.copy()

    # Phase 6: Pair search polish
    deadline6 = t0 + budget_s * 0.92
    h_pair, c5_pair = _pair_search(best_h, deadline6, max_passes=50)
    if c5_pair < best_c5:
        best_c5 = c5_pair
        best_h = h_pair.copy()

    # Phase 7: Resolution exploration
    deadline7 = t0 + budget_s * 0.97
    for dn in [2, -2, 4, -4, 8, -8, 16, -16, 12, -12]:
        if time.time() > deadline7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline7, time.time() + budget_s * 0.025)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, max_iters=1000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final LP + sumexp polish
    deadline_final = t0 + budget_s * 0.995
    h_f, c5_f = _lp_loop(best_h, min(deadline_final, t0 + budget_s * 0.99),
                         radius_init=0.04, max_iters=5000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(best_h)

    if time.time() < deadline_final:
        h_fe, c5_fe = _sumexp_descent(best_h, deadline_final, n_iter=100,
                                      beta_init=80.0, beta_max=800.0)
        if c5_fe < best_c5:
            best_c5 = c5_fe
            best_h = h_fe.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
