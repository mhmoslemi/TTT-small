def find_direction_lp(h, f_vals, delta, max_active):
    n = len(h)
    f_max = f_vals.max()
    
    # Select active set
    active = np.where(f_vals >= f_max - delta)[0]
    if len(active) > max_active:
        # Keep only the top max_active
        top_order = np.argsort(f_vals[active])[::-1]
        active = active[top_order[:max_active]]
    if len(active) < 2:
        # Include at least the top 2
        top2 = np.argsort(f_vals)[::-1][:2]
        active = np.unique(np.concatenate([active, top2]))
    
    # Build LP
    # Variables: d[0..n-1], t
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: for each active s, grad_s · d - t <= -f_s
    A_ub_rows = []
    b_ub = []
    for p in active:
        s = p - (n - 1)  # shift from np.correlate index
        g = overlap_grad(h, s)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub.append(-f_vals[p])
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-h[k], 1.0 - h[k]) for k in range(n)]
    bounds.append((0.0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    t_new = res.x[-1]
    
    # Check if we actually have descent
    if t_new >= f_max - 1e-10:
        return None  # No improvement in linearized model
    
    return d

- a_s = dx * Σ_{valid i} [d[i]*(1-h[i+s]) - h[i]*d[i+s]] = g_s · d (directional derivative) ✓
- b_s = -dx * Σ_{valid i} d[i]*d[i+s] (quadratic coefficient)

And the model is: f_s(t) ≈ f_s(0) + t*a_s + t²*b_s

Score model: S(t) = max_s (f_s(0) + t*a_s + t²*b_s)

This is what I want to minimize over t ∈ [0, t_max].

Now let me implement this cleanly. I'll compute a and b vectors for all shifts:

```python
def compute_line_model(h, d, f_vals):
    """Compute a_s and b_s for all shifts.
    a_s = directional derivative of f_s along d
    b_s = quadratic coefficient
    """
    n = len(h)
    dx = 2.0 / n
    P = 2 * n - 1
    a = np.zeros(P)
    b = np.zeros(P)
    
    for p in range(P):
        s = p - (n - 1)
        if s >= 0:
            nv = n - s
            idx1 = np.arange(nv)
            idx2 = idx1 + s
            a[p] = dx * np.sum(d[idx1] * (1.0 - h[idx2]) - h[idx1] * d[idx2])
            b[p] = -dx * np.sum(d[idx1] * d[idx2])
        else:
            t = -s
            nv = n - t
            # h[i]*(1-h[i-t]) for i=t..n-1
            idx2 = np.arange(nv)  # i-t = 0..nv-1
            idx1 = idx2 + t       # i = t..n-1
            a[p] = dx * np.sum(d[idx1] * (1.0 - h[idx2]) - h[idx1] * d[idx2])
            b[p] = -dx * np.sum(d[idx1] * d[idx2])
    
    return a, b
```

This is O(n²) total (P shifts, each O(n) work). For n=60, that's about 7000 operations. Very fast.

Actually, let me reconsider the LP approach. There's a potential issue: the LP finds a direction that decreases the linearized max, but the actual improvement might be much smaller (due to the quadratic terms). If the quadratic terms are strongly positive (b_s > 0 for active s), the function curves upward and the linearized decrease might not materialize.

To handle this, the line search will naturally find a small step. If the step is too small (less than 1e-6), we should consider that we're at a true local minimum and try a different strategy.

Also, I should consider: what if the best solution has a specific structure? For example, what if the optimal h has some periodicity or symmetry? The anti-symmetric transform h → 1-h[::-1] is a symmetry, so at the optimum, h and its anti-symmetric transform should give the same score. This means the average (h + 1-h[::-1])/2 = (h - h[::-1] + 1)/2 is also optimal (by convexity of the max of linear functions... wait, the max of bilinear functions is not convex).

Actually, the max of bilinear functions is NOT convex in general, so the average of two optimal solutions is NOT necessarily optimal. But it's a useful heuristic.

Let me also think about: what structure does the optimal solution have?
- It should be "as flat as possible" in the sense that all overlaps are equal at the optimum (by the minimax principle)
- But this is only possible if the problem is "fully degenerate" (all shifts tied)
- More likely, a subset of shifts are tied at the maximum, and the others are below

At the optimum, the KKT conditions say:
- For active shifts (at the max): the weighted sum of their gradients is zero (in the feasible direction space)
- The weights are the "Lagrange multipliers" which are non-negative

This is exactly what the LP is finding: a direction that decreases all active constraints simultaneously.

```python
import numpy as np
from scipy.optimize import linprog
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def overlap_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def overlap_grad_unnorm(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]
    else:
        t = -s
        for i in range(t, n):
            g[i] += (1.0 - h[i - t])
            g[i - t] -= h[i]
    return g


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -2.0, 2.0)
    # Bisection for theta: find theta such that sum(clip(z+theta, 0, 1)) = target
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)


def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)


def anti_sym(h):
    return make_feasible(1.0 - h[::-1].copy())


def compute_line_model(h, d, f_vals):
    n = len(h)
    dx = 2.0 / n
    P = 2 * n - 1
    a = np.zeros(P)
    b = np.zeros(P)
    for p in range(P):
        s = p - (n - 1)
        if s >= 0:
            nv = n - s
            a[p] = dx * np.sum(d[:nv] * (1.0 - h[s:]) - h[:nv] * d[s:])
            b[p] = -dx * np.sum(d[:nv] * d[s:])
        else:
            t = -s
            nv = n - t
            a[p] = dx * np.sum(d[t:] * (1.0 - h[:nv]) - h[t:] * d[:nv])
            b[p] = -dx * np.sum(d[t:] * d[:nv])
    return a, b


def line_search_model(f_vals, a, b, h, d):
    n = len(h)
    t_max = np.inf
    for k in range(n):
        if d[k] < -1e-15:
            t_max = min(t_max, h[k] / (-d[k]))
        elif d[k] > 1e-15:
            t_max = min(t_max, (1.0 - h[k]) / d[k])
    t_max = max(t_max, 1e-8)
    t_max = min(t_max, 1.0)
    
    n_grid = 60
    t_grid = np.linspace(0.0, t_max, n_grid)
    scores = np.empty(n_grid)
    for i in range(n_grid):
        t = t_grid[i]
        scores[i] = np.max(f_vals + t * a + t * t * b)
    
    best_idx = np.argmin(scores)
    best_t = t_grid[best_idx]
    best_score = scores[best_idx]
    
    # Golden section refinement
    if 0 < best_idx < n_grid - 1:
        lo_t = t_grid[max(0, best_idx - 2)]
        hi_t = t_grid[min(n_grid - 1, best_idx + 2)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = hi_t - gr * (hi_t - lo_t)
        d_pt = lo_t + gr * (hi_t - lo_t)
        f_c = np.max(f_vals + c_pt * a + c_pt * c_pt * b)
        f_d = np.max(f_vals + d_pt * a + d_pt * d_pt * b)
        for _ in range(40):
            if f_c < f_d:
                hi_t = d_pt
                d_pt = c_pt
                f_d = f_c
                c_pt = hi_t - gr * (hi_t - lo_t)
                f_c = np.max(f_vals + c_pt * a + c_pt * c_pt * b)
            else:
                lo_t = c_pt
                c_pt = d_pt
                f_c = f_d
                d_pt = lo_t + gr * (hi_t - lo_t)
                f_d = np.max(f_vals + d_pt * a + d_pt * d_pt * b)
        best_t = (lo_t + hi_t) / 2.0
        best_score = np.max(f_vals + best_t * a + best_t * best_t * b)
    
    return float(best_t), float(best_score)


def find_direction_lp(h, f_vals, delta, max_active):
    n = len(h)
    dx = 2.0 / n
    f_max = f_vals.max()
    
    active = np.where(f_vals >= f_max - delta)[0]
    if len(active) > max_active:
        order = np.argsort(f_vals[active])[::-1]
        active = active[order[:max_active]]
    elif len(active) < 2:
        top2 = np.argsort(f_vals)[::-1][:2]
        active = np.unique(np.concatenate([active, top2]))
    
    # Build LP: minimize t s.t. F[p] + dx*grad_p . d <= t for p in active, sum(d)=0, box
    n_vars = n + 1  # d[0..n-1] and t
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    A_ub_list = []
    b_ub_list = []
    for p in active:
        s = p - (n - 1)
        g = dx * overlap_grad_unnorm(h, s)
        row = np.zeros(n_vars)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-f_vals[p])
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds_list = [(-h[k], 1.0 - h[k]) for k in range(n)]
    bounds_list.append((0.0, None))
    bounds = bounds_list
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res is None or res.status != 0:
        return None
    
    d = res.x[:n].copy()
    t_new = res.x[-1]
    
    if t_new >= f_max - 1e-10:
        return None
    
    if np.abs(d).max() < 1e-10:
        return None
    
    return d


def smooth_direction(h, f_vals, temperature):
    """Compute weighted subgradient direction."""
    n = len(h)
    dx = 2.0 / n
    P = 2 * n - 1
    
    # Softmax weights
    shifted = f_vals - f_vals.max()
    weights = np.exp(shifted / temperature)
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for p in range(P):
        if weights[p] < 1e-10:
            continue
        s = p - (n - 1)
        g += weights[p] * dx * overlap_grad_unnorm(h, s)
    
    # Project to sum=0
    g -= g.mean()
    
    # Negate for descent
    d = -g
    if np.abs(d).max() < 1e-12:
        return None
    return d
