import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = dx * w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        # s >= 0: F_s = dx * sum_{j=0}^{n-s-1} h[j](1-h[j+s])
        # dF_s/dh[j] for j in [0, n-s-1]: dx*(1-h[j+s])
        # dF_s/dh[j] for j in [s, n-1]: dx*(-h[j-s])
        j0 = max(0, s)
        j1 = min(n, n + s)
        for p in range(j0, j1):
            g[p] += wk * (1.0 - x[p - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for p in range(p0, p1):
            g[p] += wk * (-x[p + s])
    return g

def _slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def _efun(x):
        return x.sum() - n / 2.0
    
    def _ejac(x):
        return np.ones(n)
    
    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _active_set_solve(h0, max_active=12, max_iter=5):
    """Active-set minimax SLSQP."""
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)
    
    for it in range(max_iter):
        cv = _corr_vec(cur)
        mc = cv.max()
        # Select top-m lags
        m = min(max_active, len(cv))
        active_idx = np.argsort(cv)[::-1][:m]
        
        # Build the active-set SLSQP problem
        z0 = np.concatenate([cur, [mc + 0.001]])
        bounds = [(0.0, 1.0)] * n + [(0.01, 1.0)]
        
        def _aobj(z):
            return z[-1]
        
        def _aobj_jac(z):
            g = np.zeros(len(z))
            g[-1] = 1.0
            return g
        
        def _acons(z):
            h = z[:-1]
            t = z[-1]
            cv = _corr_vec(h)
            return t - cv[active_idx]
        
        def _acons_jac(z):
            h = z[:-1]
            dx = 2.0 / n
            n_con = len(active_idx)
            J = np.zeros((n_con, len(z)))
            for ci, k in enumerate(active_idx):
                s = k - (n - 1)
                J[ci, -1] = 1.0  # d(t - F_s)/dt = 1
                # dF_s/dh
                j0 = max(0, s)
                j1 = min(n, n + s)

                for p in range(p0, p1):
                    J[ci, p] += -dx * (-h[p + s])
            return J
        
        def _eeq(z):
            return z[:-1].sum() - n / 2.0
        
        def _eeq_jac(z):
            g = np.zeros(len(z))
            g[:-1] = 1.0
            return g
        
        try:
            res = minimize(_aobj, z0, jac=_aobj_jac, method='SLSQP',

                               {'type': 'ineq', 'fun': _acons, 'jac': _acons_jac},
                               {'type': 'eq', 'fun': _eeq, 'jac': _eeq_jac}
                           ],
                           options={'maxiter': 300, 'ftol': 1e-12})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                new_h = _project(res['x'][:-1])
                new_s = _score(new_h)
                if new_s < cur_s - 1e-14:
                    cur = new_h
                    cur_s = new_s
                else:
                    break
            else:
                break
        except Exception:
            break
    
    return cur, cur_s

def _coord_descent(h0, n_sweeps=20, project_every=3):
    """Coordinate descent with PWL minimization."""
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)
    
    for sweep in range(n_sweps):
        for i in range(n):
            # Compute a_s and b_s for coordinate i
            h_save = cur[i]
            
            # Set h[i] = 0
            cur[i] = 0.0
            cv0 = _corr_vec(cur)
            
            # Set h[i] = 1
            cur[i] = 1.0
            cv1 = _corr_vec(cur)
            
            a = cv1 - cv0  # coefficient of h[i] in each F_s
            b = cv0  # constant part (F_s when h[i]=0)
            
            # Find min of max_s(a_s * x + b_s) over x in [0,1]
            # Compute breakpoints
            breakpoints = [0.0, 1.0]
            m = len(a)
            for p in range(m):
                for q in range(p + 1, m):
                    da = a[p] - a[q]
                    if abs(da) > 1e-15:
                        x_bp = (b[q] - b[p]) / da
                        if 0.0 < x_bp < 1.0:
                            breakpoints.append(x_bp)
            
            breakpoints = sorted(set(breakpoints))
            
            # Evaluate at all breakpoints
            best_x = 0.0
            best_v = np.inf
            for x in breakpoints:
                vals = a * x + b
                v = vals.max()
                if v < best_v:
                    best_v = v
                    best_x = x
            
            cur[i] = best_x
        
        # Project every few sweeps
        if (sweep + 1) % project_every == 0:
            cur = _project(cur)
        
        s = _score(cur)
        if s < cur_s - 1e-14:
            cur_s = s
    
    cur = _project(cur)
    cur_s = _score(cur)
    return cur, cur_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

                           bounds=bounds,
                           constraints=[
                               {'type': 'ineq', 'fun': _acons, 'jac': _acons_jac},
                               {'type': 'eq', 'fun': _eeq, 'jac': _eeq_jac}
                           ],
                           options={'maxiter': 400, 'ftol': 1e-13})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                new_h = _project(res['x'][:-1])
                new_s = _score(new_h)
                if new_s < cur_s - 1e-14:
                    cur = new_h
                    cur_s = new_s
                else:
                    break
            else:
                break
        except Exception:
            break
    return cur, cur_s

def _coord_descent(h0, n_sweeps=20, project_every=4):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)
    for sweep in range(n_sweps):
        for i in range(n):
            save = cur[i]
            cur[i] = 0.0
            cv0 = _corr_vec(cur)
            cur[i] = 1.0
            cv1 = _corr_vec(cur)
            a = cv1 - cv0
            b = cv0
            breakpoints = [0.0, 1.0]
            m = len(a)
            for p in range(m):
                for q in range(p + 1, m):
                    da = a[p] - a[q]
                    if abs(da) > 1e-14:
                        xb = (b[q] - b[p]) / da
                        if 1e-9 < xb < 1.0 - 1e-9:
                            breakpoints.append(xb)
            breakpoints = sorted(set(breakpoints))
            best_x = 0.0
            best_v = np.inf
            for x in breakpoints:
                vals = a * x + b
                v = vals.max()
                if v < best_v:
                    best_v = v
                    best_x = x
            cur[i] = best_x
        if (sweep + 1) % project_every == 0:
            cur = _project(cur)
        s = _score(cur)
        if s < cur_s:
            cur_s = s
    cur = _project(cur)
    cur_s = _score(cur)
    return cur, cur_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _binary_push(h, p):
    return _project(h ** p)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.10 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(56) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)
    rng = np.random.default_rng(seed)

    # Phase 1: Resolution exploration
    n_list = sorted(set(parent_n + d for d in range(-8, 9, 2)) | {parent_n})
    n_list = [n for n in n_list if 24 <= n <= 80]

    candidates = []
    for n in n_list:
        if time.time() > deadline - 180:
            break
        h_i = _interp(parent, n)
        h_opt = _slsqp_smooth(h_i, beta=120, maxiter=80)
        s0 = _score(h_i)
        s1 = _score(h_opt)
        h_b = h_opt if s1 <= s0 else h_i
        s_b = min(s0, s1)
        candidates.append((s_b, n, h_b.copy()))
        if s_b < best_score - 1e-12:
            best_h = h_b.copy()
            best_score = s_b
            best_n = n

    # Phase 2: Active-set + coordinate descent on top candidates
    candidates.sort(key=lambda x: x[0])
    top_k = min(6, len(candidates))
    for idx in range(top_k):
        if time.time() > deadline - 120:
            break
        s_c, n_c, h_c = candidates[idx]
        # Active-set
        h_as, s_as = _active_set_solve(h_c, max_active=14, max_rounds=6)
        if s_as < best_score - 1e-12:
            best_h = h_as.copy()
            best_score = s_as
            best_n = n_c
        # Coordinate descent
        h_cd, s_cd = _coord_descent(h_as, n_sweeps=20, project_every=4)
        if s_cd < best_score - 1e-12:
            best_h = h_cd.copy()
            best_score = s_cd
            best_n = n_c

    # Phase 3: Diversified starts
    if time.time() < deadline - 80:
        starts = [best_h.copy(), _mirror(best_h)]
        for sigma in [0.06, 0.12, 0.2]:
            starts.append(_project(best_h + sigma * rng.standard_normal(best_n)))
        for p in [1.5, 2.5, 0.7]:
            starts.append(_binary_push(best_h, p))
        # Fourier perturbation
        for freq in [1, 2, 3, 5]:
            phase = rng.uniform(0, 6.283)
            pert = 0.08 * np.sin(2.0 * np.pi * freq * np.arange(best_n) / best_n + phase)
            starts.append(_project(best_h + pert))

        for h0 in starts:
            if time.time() > deadline - 60:
                break
            h_lse = _slsqp_smooth(h0, beta=200, maxiter=100)
            h_as, s_as = _active_set_solve(h_lse, max_active=12, max_rounds=4)
            h_cd, s_cd = _coord_descent(h_as, n_sweeps=15, project_every=4)
            if s_cd < best_score - 1e-12:
                best_h = h_cd.copy()
                best_score = s_cd
                best_n = len(best_h)

    # Phase 4: Intensive refinement on best
    # Alternating LSE + active-set + coord descent
    for round_i in range(4):
        if time.time() > deadline - 40:
            break
        beta = 400 + round_i * 600
        h_pol = _slsqp_smooth(best_h, beta=beta, maxiter=120)
        s_pol = _score(h_pol)
        if s_pol < best_score - 1e-12:
            best_h = h_pol
            best_score = s_pol
        h_as, s_as = _active_set_solve(best_h, max_active=16, max_rounds=6)
        if s_as < best_score - 1e-12:
            best_h = h_as
            best_score = s_as
        h_cd, s_cd = _coord_descent(best_h, n_sweeps=25, project_every=5)
        if s_cd < best_score - 1e-12:
            best_h = h_cd
            best_score = s_cd

    # Phase 5: Random restarts
    remaining = deadline - time.time()
    n_restarts = min(8, max(2, int(remaining / 25)))
    for r in range(n_restarts):
        if time.time() > deadline - 15:
            break
        h_pert = _project(best_h + 0.04 * rng.standard_normal(best_n))
        h_lse = _slsqp_smooth(h_pert, beta=500, maxiter=80)
        h_as, s_as = _active_set_solve(h_lse, max_active=12, max_rounds=4)
        h_cd, s_cd = _coord_descent(h_as, n_sweeps=15, project_every=4)
        if s_cd < best_score - 1e-12:
            best_h = h_cd.copy()
            best_score = s_cd
            best_n = len(best_h)

    # Final polish
    if time.time() < deadline - 8:
        h_cd, s_cd = _coord_descent(best_h, n_sweeps=30, project_every=5)
        if s_cd < best_score - 1e-12:
            best_h = h_cd
            best_score = s_cd

    if time.time() < deadline - 4:
        h_pol = _slsqp_smooth(best_h, beta=5000, maxiter=100)
        s_pol = _score(h_pol)
        if s_pol < best_score - 1e-12:
            best_h = h_pol
            best_score = s_pol

    best_h = _project(best_h)
    final_score = _score(best_h)
    return (best_h, final_score, len(best_h))
