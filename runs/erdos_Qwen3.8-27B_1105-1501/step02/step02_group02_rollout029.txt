def _lse_grad(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()

    ev = np.exp(sh)

    # This is: corr_k(h) + (contribution change due to h_i changing)

    # - If i is in the "second position" (j = i): old term was h[i+s]*(1-h[i]), new is h[i+s]*(1-x)

    #   change = x*(1-h[i-s]) - h[i]*(1-h[i-s]) = (x-h[i])*(1-h[i-s])
