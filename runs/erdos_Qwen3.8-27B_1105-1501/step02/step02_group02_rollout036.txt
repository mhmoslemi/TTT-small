Now, for the **linearized LP** approach:

At current point h*, compute:
- cv = F(h*) (all 2n-1 values, including dx)
- For each active lag k (top few), compute ∂F_k/∂h_j at h*

The LP is:
minimize t
subject to:
  t ≥ F_k(h*) + ∇F_k(h*)^T (h - h*) for k in active set
  sum(h) = n/2
  0 ≤ h ≤ 1

In standard LP form (for scipy.optimize.linprog):
minimize c^T x where x = [h_0, ..., h_{n-1}, t]
c = [0, ..., 0, 1]

Inequalities (A_ub @ x ≤ b_ub):
For each active k: -∇F_k(h*)^T h + t ≤ F_k(h*) - ∇F_k(h*)^T h*
  → -g_k^T h + t ≤ F_k(h*) - g_k^T h*
  where g_k = ∂F_k/∂h at h* (this already includes dx)

Wait, F_k(h) ≈ F_k(h*) + g_k^T (h - h*) = F_k(h*) - g_k^T h* + g_k^T h
So t ≥ F_k(h*) - g_k^T h* + g_k^T h
→ -g_k^T h + t ≤ F_k(h*) - g_k^T h*

Equality: sum(h) = n/2 → A_eq @ x = b_eq where A_eq = [1,...,1, 0], b_eq = n/2

Bounds: 0 ≤ h_j ≤ 1 for all j, t ≥ 0 (or t ≥ some lower bound)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad_fast(x, beta):
    n = len(x)
    dx = 2.0 / n
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:

        w = ev / es

            i2 = j + s
            if 0 <= i2 < n:

        g[j] = dx * val
    return g

def _grad_single_lag(x, k):
    """Gradient of F_k (the k-th correlation value) at x."""
    n = len(x)
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = j - s
        if 0 <= i1 < n:
            val += (1.0 - x[i1])
        i2 = j + s
        if 0 <= i2 < n:
            val -= x[i2]
        g[j] = dx * val
    return g

def _solve_linearized_lp(h_cur, active_lags, top_k=5):
    """Solve the linearized min-max LP around h_cur."""
    n = len(h_cur)
    cv = _corr_vec(h_cur)
    
    # Get gradients for active lags
    A_ub_rows = []
    b_ub = []
    for k in active_lags:
        g_k = _grad_single_lag(h_cur, k)
        # t - g_k^T h ≤ cv[k] - g_k^T h_cur
        # -g_k^T h + t ≤ cv[k] - g_k^T h_cur
        row = np.zeros(n + 1)
        row[:n] = -g_k
        row[n] = 1.0
        A_ub_rows.append(row)
        b_ub.append(cv[k] - g_k @ h_cur)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: 0 ≤ h_j ≤ 1, t ≥ 0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return _project(res.x[:n])
    except Exception:
        pass
    return h_cur.copy()

def _binary_local_search(h0, n_iter=5000, rng=None):
    """Local search on binary points near h0."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(h0)
    # Initialize from h0
    h_bin = (h0 >= 0.5).astype(float)
    # Fix sum

        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones)] = 0
    while h_bin.sum() < n / 2:

        h_bin[rng.choice(zeros)] = 1

    T = 0.005
    for it in range(n_iter):

        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_score = _score(h_bin)

            cur_score = new_score

                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    
    return best_h, best_score

def _slsqp_lse(h0, beta, maxiter=200):
    """SLSQP with LSE objective."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad_fast,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return _project(res.x)
    except Exception:
        pass
    return h0

def _fista(h0, beta, n_iter, alpha=0.005):
    """FISTA with LSE objective."""
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t_val = 1.0
    best_h = h0.copy()
    best_s = _score(h0)
    
    for it in range(n_iter):
        g = _lse_grad_fast(y, beta)
        x_new = _project(y - alpha * g)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - x)
        y = _project(y)
        x = x_new
        t_val = t_new
        
        if it % 200 == 0:
            s = _score(x)
            if s < best_s:
                best_s = s
                best_h = x.copy()
    
    s = _score(x)
    if s < best_s:
        best_s = s
        best_h = x.copy()
    return best_h, best_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _cutting_plane_iterate(h_cur, n_lags=8, n_lp_iters=10):

    h = h_cur.copy()
    for _ in range(n_lp_iters):

        # Top active lags
        idx = np.argsort(cv)[::-1][:n_lags]

            if s_new <= s_old + 1e-10:
                h = new_h
            else:
                # LP didn't help, break
                break
        else:
            break
    return h

    betas = cv0[active_k_list]

    # subject to sum constraint: h[j] = t, and sum(h) = n/2
    # (other coords are fixed, so sum is fixed except for h[j])
    # t must be in [max(0, n/2 - sum(others)), min(1, n/2 - sum(others))]
    
    sum_others = h.sum() - h[j]
    t_lo = max(0.0, n/2.0 - sum_others)
    t_hi = min(1.0, n/2.0 - sum_others)
    
    if t_lo > t_hi:
        return h[j]  # infeasible, shouldn't happen
    
    # Find t in [t_lo, t_hi] that minimizes max_k (betas[k] + alphas[k]*t)
    # This is a 1D convex minimization (max of affine functions is convex)
    # Use golden section or ternary search
    
    best_t = t_lo
    best_val = np.max(betas + alphas * t_lo)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(betas + alphas * c)
    fd = np.max(betas + alphas * d)

            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)

            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta


def _lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            i1 = j - s
            if 0 <= i1 < n:
                val += wk * (1.0 - x[i1])
            i2 = j + s
            if 0 <= i2 < n:
                val -= wk * x[i2]
        g[j] = dx * val
    return g


def _grad_single(h, k):
    n = len(h)
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = j - s
        if 0 <= i1 < n:
            val += (1.0 - h[i1])
        i2 = j + s
        if 0 <= i2 < n:
            val -= h[i2]
        g[j] = dx * val
    return g


def _coord_descent(h, n_cycles=50, active_frac=0.05, max_active=25):
    """Cyclic coordinate descent with exact 1D minimization on active lags."""
    n = len(h)
    h = h.copy()
    cv = _corr_vec(h)
    cv_max = cv.max()
    threshold = cv_max - active_frac
    active_k = np.where(cv >= threshold)[0]
    if len(active_k) > max_active:
        active_k = np.argsort(cv)[::-1][:max_active]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sum_target = n / 2.0
    
    for cyc in range(n_cycles):
        improved = False
        for j in range(n):
            h0v = h.copy()
            h0v[j] = 0.0
            h1v = h.copy()
            h1v[j] = 1.0
            cv0 = np.correlate(h0v, 1.0 - h0v, mode='full') * (2.0 / n)
            cv1 = np.correlate(h1v, 1.0 - h1v, mode='full') * (2.0 / n)
            
            alphas = cv1[active_k] - cv0[active_k]
            betas = cv0[active_k]
            
            sum_others = h.sum() - h[j]
            t_lo = max(0.0, sum_target - sum_others)
            t_hi = min(1.0, sum_target - sum_others)
            
            if t_lo >= t_hi - 1e-15:
                continue
            
            a = t_lo
            b = t_hi
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = np.max(betas + alphas * c)
            fd = np.max(betas + alphas * d)
            
            for _ in range(40):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = np.max(betas + alphas * c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = np.max(betas + alphas * d)
            
            t_opt = (a + b) / 2.0
            val_opt = np.max(betas + alphas * t_opt)
            val_lo = np.max(betas + alphas * t_lo)
            if val_opt < val_lo - 1e-14:
                h[j] = t_opt
                improved = True

def _cutting_plane(h, n_lags=10, n_iters=8):
    """Linearized cutting-plane iterations."""
    n = len(h)
    h = h.copy()
    for _ in range(n_iters):
        cv = _corr_vec(h)
        idx = np.argsort(cv)[::-1][:n_lags]
        
        A_ub_rows = []
        b_ub = []
        for k in idx:
            g_k = _grad_single(h, k)
            row = np.zeros(n + 1)
            row[:n] = -g_k
            row[n] = 1.0
            A_ub_rows.append(row)
            b_ub.append(cv[k] - g_k @ h)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        c = np.zeros(n + 1)
        c[n] = 1.0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                new_h = _project(res.x[:n])
                s_old = _score(h)
                s_new = _score(new_h)
                if s_new <= s_old + 1e-12:
                    h = new_h

            else:
                break
        except Exception:
            break
    return h


def _fista(h0, beta, n_iter, alpha=0.004):
    """FISTA with LSE objective."""
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t_val = 1.0
    best_h = h0.copy()
    best_s = _score(h0)
    
    for it in range(n_iter):

        x_new = _project(y - alpha * g)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - x)
        y = _project(y)
        x = x_new
        t_val = t_new
        
        if it % 300 == 0:

                best_h = x.copy()
    
    s = _score(x)
    if s < best_s:
        best_s = s
        best_h = x.copy()
    return best_h, best_s


def _slsqp_lse(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,

                       method='SLSQP', bounds=bounds, constraints=cons,

        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return _project(res.x)
    except Exception:
        pass
    return h0

            phase1_results.append((s0, n, h_interp))
        sc = phase1_results[-1][0]
        if sc < best_score:
            best_score = sc
            best_h = phase1_results[-1][2].copy()
    
    # Phase 2: Focused optimization on top dimensions
    phase1_results.sort(key=lambda x: x[0])
    top_dims = []
    for s, n, h in phase1_results[:4]:
        if n not in top_dims:
            top_dims.append(n)
    
    for n in top_dims:
        if time.time() > deadline - 150:
            break
        h_base = None
        for s, nn, hh in phase1_results:
            if nn == n:
                h_base = hh.copy()
                break
        if h_base is None:
            h_base = _interp(parent, n)
        
        starts = [h_base.copy(), h_base[::-1].copy()]
        for sigma in [0.05, 0.12, 0.2]:
            starts.append(_project(h_base + sigma * rng.standard_normal(n)))
        # Binary push variants
        for p in [2.0, 3.0, 5.0]:
            starts.append(_project(h_base ** p))
        # Sinusoidal perturbation
        for freq in [1, 2, 3]:
            phase = rng.uniform(0, 6.28)
            pert = 0.08 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            starts.append(_project(h_base + pert))
        
        for h0 in starts:
            if time.time() > deadline - 120:
                break
            h_opt = _full_optimize(h0, deadline)
            s_opt = _score(h_opt)
            if s_opt < best_score:
                best_score = s_opt
                best_h = h_opt.copy()
    
    # Phase 3: Intensive refinement on best
    n_best = len(best_h)
    
    for round_i in range(6):
        if time.time() > deadline - 40:
            break
        # Coordinate descent with shrinking active set
        frac = max(0.005, 0.03 * (0.7 ** round_i))
        max_act = max(8, 25 - round_i * 2)
        best_h = _coord_descent(best_h, n_cycles=100, active_frac=frac, max_active=max_act)
        
        # Cutting plane
        if time.time() < deadline - 30:
            best_h = _cutting_plane(best_h, n_lags=15, n_iters=8)
        
        # FISTA with increasing beta
        if time.time() < deadline - 25:
            beta = 200 + round_i * 300
            best_h, best_score_local = _fista(best_h, beta=beta, n_iter=3000, alpha=0.003)
            if best_score_local < best_score:
                best_score = best_score_local
        
        # SLSQP polish
        if time.time() < deadline - 20:
            h_new = _slsqp_lse(best_h, beta=800, maxiter=120)
            s_new = _score(h_new)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new
        
        best_score = _score(best_h)
    
    # Phase 4: Random restarts
    remaining = deadline - time.time()
    n_restarts = min(10, int(remaining / 40))
    for _ in range(n_restarts):
        if time.time() > deadline - 20:
            break
        h_pert = _project(best_h + 0.03 * rng.standard_normal(n_best))
        h_opt = _full_optimize(h_pert, deadline)
        s_opt = _score(h_opt)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()
    
    # Final intensive coordinate descent
    if time.time() < deadline - 8:
        best_h = _coord_descent(best_h, n_cycles=200, active_frac=0.005, max_active=8)
    
    if time.time() < deadline - 5:
        best_h = _cutting_plane(best_h, n_lags=15, n_iters=10)
    
    if time.time() < deadline - 3:
        best_h = _coord_descent(best_h, n_cycles=100, active_frac=0.003, max_active=6)
    
    if time.time() < deadline - 2:
        best_h, _ = _fista(best_h, beta=3000, n_iter=2000, alpha=0.001)
        best_h = _coord_descent(best_h, n_cycles=50, active_frac=0.003, max_active=5)
    
    best_h = _project(best_h)
    final_score = _score(best_h)
    
    return (best_h, final_score, len(best_h))
