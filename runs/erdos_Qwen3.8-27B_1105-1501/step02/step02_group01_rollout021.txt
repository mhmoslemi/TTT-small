def official_score(h):

def make_feasible(z, target=None):
    # Project onto {0<=x<=1, sum(x)=target}
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)

        target = n / 2.0
    # Bisection on shift

    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:

        else:

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    m = np.max(corr)

                break

        j = rng.integers(n)
        if i == j:
            continue
        if h[i] <= h[j]:
            i, j = j, i
        # Move from i (high) to j (low)

        if delta < 0.01:
            continue
        # Try full swap and partial swaps
        best_delta = 0
        best_s = best_score

            h[j] += d

            h[j] += best_delta

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)

        target = n / 2.0

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:

        else:

    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:

    return m + np.log(s) / beta

            if rng.random() > accept_prob:

                h[j] -= d

            h[j] += best_d

    return best_h, best_score

                    grad[m2:] += (-h[:n - m2])
        grad /= len(active)
        v = momentum * v - alpha * grad
        h_new = make_feasible(h + v)
        new_score = official_score(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            h = h_new
            patience = 0
            alpha = min(alpha * 1.01, 0.5)
        else:
            h = h_new
            patience += 1
            alpha = max(alpha * 0.98, 0.01)
        if patience > 400:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(25.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    n_options = [50, 51, 52]
    if len(parent) not in n_options and 24 <= len(parent) <= 80:
        n_options.append(len(parent))

    for n in n_options:
        if time.time() > deadline - 80:
            break

        h_parent = make_feasible(interp_to_n(parent, n))

        # Build diverse starting points
        starts = []
        starts.append(h_parent)
        for p in [2.0, 4.0, 6.0, 10.0]:
            starts.append(push_binary(h_parent, p))
        for trial in range(6):
            rb = np.zeros(n)
            if n % 2 == 0:
                rb[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                idx = rng.choice(n, (n - 1) // 2, replace=False)
                rb[idx] = 1.0
                rb[rng.integers(n)] = 1.0
                rb = make_feasible(rb)
            starts.append(rb)
        for sig in [0.08, 0.2, 0.4]:
            starts.append(make_feasible(h_parent + sig * rng.standard_normal(n)))
        # Alternating
        alt = np.zeros(n)
        alt[::2] = 1.0
        starts.append(make_feasible(alt + 0.1 * rng.standard_normal(n)))
        # Block patterns
        for bs in [3, 5, 7]:
            bp = np.zeros(n)
            for pos in range(0, n, 2 * bs):
                bp[pos:pos + bs] = 1.0
            starts.append(make_feasible(bp + 0.05 * rng.standard_normal(n)))
        # Anti-symmetric
        starts.append(make_feasible(1.0 - h_parent[::-1]))
        # Coarse optimization then interpolate
        if n >= 28:
            nc = n // 2 if n % 2 == 0 else (n - 1) // 2
            h_coarse = make_feasible(interp_to_n(h_parent, nc))
            h_coarse = push_binary(h_coarse, 4.0)
            h_coarse_opt, _ = do_cd(h_coarse, rng, 2000, 10, time.time())
            starts.append(make_feasible(interp_to_n(h_coarse_opt, n)))

        # Deduplicate by similar score
        seen = set()
        unique_starts = []
        for s in starts:
            sc = round(official_score(s), 5)
            if sc not in seen:
                seen.add(sc)
                unique_starts.append(s)
        starts = unique_starts

        for h0 in starts:
            if time.time() > deadline - 40:
                break
            cur_h = h0.copy()
            cur_score = official_score(cur_h)

            # Stage 1: SA
            if time.time() < deadline - 35:
                n_sa = 25000 if n <= 52 else 15000
                sa_h, sa_score = do_sa(cur_h, rng, n_sa, 45, time.time())
                if sa_score < cur_score - 1e-10:
                    cur_h = sa_h
                    cur_score = sa_score

            # Stage 2: SLSQP with increasing beta
            for beta in [30, 100, 400, 1500]:
                if time.time() > deadline - 20:
                    break
                mi = 250 if n <= 52 else 120
                new_h = do_slsqp(cur_h, beta, maxiter=mi)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-13:
                    cur_h = new_h
                    cur_score = new_score

            # Stage 3: CD
            if time.time() < deadline - 25:
                cd_h, cd_score = do_cd(cur_h, rng, 6000, 35, time.time())
                if cd_score < cur_score - 1e-10:
                    cur_h = cd_h
                    cur_score = cd_score

            # Stage 4: Subgradient
            if time.time() < deadline - 25:
                sg_h, sg_score = do_subgrad(cur_h, max_iter=2500, time_limit=25, t_start=time.time())
                if sg_score < cur_score - 1e-10:
                    cur_h = sg_h
                    cur_score = sg_score

            # Stage 5: SA again on refined
            if time.time() < deadline - 20:
                sa_h, sa_score = do_sa(cur_h, rng, 15000, 25, time.time())
                if sa_score < cur_score - 1e-10:
                    cur_h = sa_h
                    cur_score = sa_score

            # Stage 6: Final SLSQP
            for beta in [500, 2000, 5000]:
                if time.time() > deadline - 10:
                    break
                new_h = do_slsqp(cur_h, beta, maxiter=200)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-13:
                    cur_h = new_h
                    cur_score = new_score

            # Stage 7: Final CD
            if time.time() < deadline - 10:
                cd_h, cd_score = do_cd(cur_h, rng, 4000, 15, time.time())
                if cd_score < cur_score - 1e-10:
                    cur_h = cd_h
                    cur_score = cd_score

            if cur_score < best_score - 1e-13:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
                # Record improvement

    # Final polish loop
    remaining = deadline - time.time()
    if remaining > 20:
        n_trials = min(15, int(remaining / 12))
        for trial in range(n_trials):
            if time.time() > deadline - 8:
                break
            pert = 0.02 + 0.02 * trial
            h_pert = make_feasible(best_h + pert * rng.standard_normal(best_n))
            sa_h, sa_s = do_sa(h_pert, rng, 8000, 8, time.time())
            sl_h = do_slsqp(sa_h, 500, maxiter=150)
            sl_s = official_score(sl_h)
            if sl_s < best_score - 1e-13:
                best_h = sl_h
                best_score = sl_s

    # Very final polish
    if time.time() < deadline - 3:
        final_h = do_slsqp(best_h, 3000, maxiter=200)
        final_s = official_score(final_h)
        if final_s < best_score:
            best_h = final_h
            best_score = final_s
        cd_h, cd_s = do_cd(best_h, rng, 3000, 5, time.time())
        if cd_s < best_score:
            best_h = cd_h
            best_score = cd_s

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
