```python
def find_optimal_t(R, c, t_lo, t_hi, n_iter=40):
    """Find t that minimizes dx * max(R + c*t) on [t_lo, t_hi]."""
    for _ in range(n_iter):
        m1 = t_lo + (t_hi - t_lo) / 3.0
        m2 = t_hi - (t_hi - t_lo) / 3.0
        f1 = np.max(R + c * m1)
        f2 = np.max(R + c * m2)
        if f1 < f2:
            t_hi = m2
        else:
            t_lo = m1
    return (t_lo + t_hi) / 2.0
```

```python
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))

def smooth_grad(x, beta):
```

```python
def compute_pair_coeff(h, i, j, n):
    """Compute c array for pair (i,j). c[p] = dR[p]/dh[i] - dR[p]/dh[j]."""
    c = np.zeros(2*n-1)
    for s in range(-(n-1), n):
        p = n - 1 + s
        if s >= 0:
            # dR/dh[i]
            di = 0.0
            if 0 <= i <= n-1-s:
                di += (1.0 - h[i+s])
            if s <= i <= n-1:
                di += (-h[i-s])
            # dR/dh[j]
            dj = 0.0
            if 0 <= j <= n-1-s:
                dj += (1.0 - h[j+s])
            if s <= j <= n-1:
                dj += (-h[j-s])
        else:
            t2 = -s
            # dR/dh[i]
            di = 0.0
            if t2 <= i <= n-1:
                di += (1.0 - h[i-t2])
            if 0 <= i <= n-1-t2:
                di += (-h[i+t2])
            # dR/dh[j]
            dj = 0.0
            if t2 <= j <= n-1:
                dj += (1.0 - h[j-t2])
            if 0 <= j <= n-1-t2:
                dj += (-h[j+t2])
        c[p] = di - dj
    return c
```

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:

                residual -= add
            else:

            if abs(residual) < 1e-14:

    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta
