        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def binary_hill_climb(h_bin, seed=42, max_rounds=200):
    """Hill climbing on binary patterns."""
    n = len(h_bin)
    rng = np.random.default_rng(seed)
    h = h_bin.copy()
    best_score = official_score(h)
    
    for rnd in range(max_rounds):
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        if not ones or not zeros:
            break
        improved = False
        n_pairs = len(ones) * len(zeros)
        n_tries = min(n_pairs, 3000)
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            sc = official_score(h)
            if sc < best_score - 1e-14:
                best_score = sc
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            # Try all pairs (slower but more thorough)
            improved = False
            for i in ones:
                for j in zeros:
                    h[i], h[j] = 0.0, 1.0
                    sc = official_score(h)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        improved = True
                        ones = list(np.where(h > 0.5)[0])
                        zeros = list(np.where(h < 0.5)[0])
                        break
                    else:
                        h[i], h[j] = 1.0, 0.0
                if improved:
                    break
            if not improved:
                break
    return h, best_score

def interpolate_to(h_src, n_tgt):

def pg_loop(h_cur, beta_start, beta_max, n_steps, alpha_max, time_check):
    """Run PG for n_steps with adaptive beta."""
    h = h_cur
    beta = beta_start
    best_h = h.copy()
    best_f = official_score(h)
    for _ in range(n_steps):
        if time_check() < 0:
            break
        h_new, sc = pg_step(h, beta, alpha_max=alpha_max, max_ls=16)
        if sc < best_f - 1e-13:
            best_h = h_new.copy()
            best_f = sc
        h = h_new
        beta = min(beta * 1.03, beta_max)
    return best_h, best_f

def lp_polish(h_cur, eps_list, trust_list, max_iters_each, time_check):

                else:
                    break
    return h, best_score

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0

            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a

            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def pg_optimize(h_start, beta_start, beta_rate, beta_max, max_steps, alpha_max, time_fn):
    h_cur = h_start.copy()
    beta = beta_start
    best_h = h_cur.copy()
    best_f = official_score(h_cur)
    for _ in range(max_steps):
        if time_fn() < 0:
            break
        h_new, sc = pg_step(h_cur, beta, alpha_max=alpha_max, max_ls=18)
        if sc < best_f - 1e-13:
            best_h = h_new.copy()
            best_f = sc
        elif sc < official_score(h_cur) - 1e-13:
            pass
        h_cur = h_new
        beta = min(beta * beta_rate, beta_max)
    return best_h, best_f

def lp_polish(h_start, time_fn, min_time=5.0):
    h = h_start.copy()
    best_score = official_score(h)
    eps_seq = [0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0007, 0.0005]
    trust_seq = [0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008]
    for ep, tr in zip(eps_seq, trust_seq):
        if time_fn() < min_time:
            break
        for _ in range(80):
            if time_fn() < min_time:
                break
            h_new, sc = active_set_lp(h, eps=ep, trust=tr)
            if h_new is not None:
                if sc < best_score - 1e-14:
                    best_score = sc
                h = h_new
            else:
                break
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.04 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        h_init = make_feasible(h_init)
    except (NameError, AttributeError):
        n = 60
        h_init = make_feasible(np.full(n, 0.5) + rng.normal(0, 0.15, n))

    best_h = h_init.copy()
    best_score = official_score(h_init)

    # Phase 1: PG optimization from initial with restarts
    if time_left() > 200:
        h_cur = h_init.copy()
        beta = 25.0
        stall = 0
        while time_left() > 180:
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.8, max_ls=18)
            if sc < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = sc
                stall = 0
            elif sc < official_score(h_cur) - 1e-12:
                stall = 0
            else:
                stall += 1
            h_cur = h_new
            beta = min(beta * 1.03, 350.0)
            if stall > 40:
                noise = 0.015 + 0.01 * (stall - 40) / 60.0
                h_cur = make_feasible(best_h + rng.normal(0, noise, len(best_h)))
                beta = 15.0
                stall = 0

    # Phase 2: Binary pattern search (different approach)
    if time_left() > 200:
        n = len(best_h)
        n_ones = n // 2
        # Get binary skeleton from current best
        idx_sorted = np.argsort(best_h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx_sorted[:n_ones]] = 1.0

        for bs in range(8):
            if time_left() < 120:
                break
            h_b, sc_b = binary_hill_climb(h_bin, seed=seed + 200 + bs * 37,
                                          max_rounds=80, rng=rng)
            # Create continuous version: blend binary with best_h
            blend = 0.7
            h_cont = make_feasible(h_b * blend + best_h * (1.0 - blend))
            sc_cont = official_score(h_cont)
            if sc_cont < best_score - 1e-12:
                best_h = h_cont.copy()
                best_score = sc_cont

            # PG refine from this starting point
            if time_left() > 120:
                h_cur2 = h_cont.copy()
                res_h, res_sc = pg_optimize(h_cur2, 40.0, 1.02, 300.0,
                                           300, 0.5, time_left)
                if res_sc < best_score - 1e-12:
                    best_h = res_h.copy()
                    best_score = res_sc

            # Update binary skeleton
            idx_sorted = np.argsort(best_h)[::-1]
            h_bin = np.zeros(n)
            h_bin[idx_sorted[:n_ones]] = 1.0

    # Phase 3: Aggressive LP polishing (many small steps)
    if time_left() > 100:
        h_cur = best_h.copy()
        h_pol, sc_pol = lp_polish(h_cur, time_left, min_time=50.0)
        if sc_pol < best_score - 1e-13:
            best_h = h_pol.copy()
            best_score = sc_pol
        else:
            h_cur = best_h.copy()

        # Second round with even smaller trust
        if time_left() > 60:
            h_pol2, sc_pol2 = lp_polish(h_cur, time_left, min_time=40.0)
            if sc_pol2 < best_score - 1e-13:
                best_h = h_pol2.copy()
                best_score = sc_pol2

    # Phase 4: Fine PG with high beta (converge to local opt)
    if time_left() > 50:
        h_cur = best_h.copy()
        beta = 150.0
        for _ in range(800):
            if time_left() < 40:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.3, max_ls=14)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.008, 600.0)

    # Phase 5: Interpolation to nearby n and re-optimize
    if time_left() > 80:
        n_base = len(best_h)
        for n_try in [n_base - 2, n_base + 1, n_base + 2, n_base - 1]:
            if time_left() < 40:
                break
            if n_try < 20:
                continue
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int.copy()
                best_score = sc_int
            h_cur = h_int.copy()
            # Quick PG + LP
            if time_left() > 40:
                res_h, res_sc = pg_optimize(h_cur, 60.0, 1.015, 250.0,
                                           200, 0.4, lambda: time_left() - 5)
                if res_sc < best_score - 1e-12:
                    best_h = res_h.copy()
                    best_score = res_sc
                if time_left() > 40:
                    h_pol, sc_pol = lp_polish(best_h, time_left, min_time=35.0)
                    if sc_pol < best_score - 1e-13:
                        best_h = h_pol.copy()
                        best_score = sc_pol

    # Phase 6: Final LP with tiny trust regions
    if time_left() > 20:
        h_cur = best_h.copy()
        for ep in [0.001, 0.0005, 0.0003]:
            if time_left() < 15:
                break
            for tr in [0.005, 0.003, 0.002]:
                if time_left() < 15:
                    break
                for _ in range(50):
                    if time_left() < 15:
                        break
                    h_new, sc = active_set_lp(h_cur, eps=ep, trust=tr)
                    if h_new is not None:
                        if sc < best_score - 1e-14:
                            best_h = h_new.copy()
                            best_score = sc
                        h_cur = h_new
                    else:
                        break

    # Phase 7: Ultimate fine PG
    if time_left() > 10:
        h_cur = best_h.copy()
        beta = 300.0
        for _ in range(1000):
            if time_left() < 8:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.15, max_ls=10)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
