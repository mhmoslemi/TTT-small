def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    """Project x onto {0 <= y <= 1, sum(y) = target_sum}"""
    x = np.asarray(x, dtype=float)
    n = len(x)
    # Bisection on the threshold
    lo, hi = -1.0 - max(abs(x).max(), 1.0), 1.0 + max(abs(x).max(), 1.0)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        # Find coordinates with slack and adjust
        if residual > 0:
            order = np.argsort(-result)  # Most room first
            for idx in order:
                room = 1.0 - result[idx]
                adj = min(room, residual)
                result[idx] += adj
                residual -= adj
                if abs(residual) < 1e-12:
                    break
        else:
            order = np.argsort(result)  # Least value first
            for idx in order:
                room = result[idx]
                adj = min(room, -residual)
                result[idx] -= adj
                residual += adj
                if abs(residual) < 1e-12:
                    break
    return result

def soft_objective_and_grad(h, temp):
    """Compute log-sum-exp smoothed objective and its gradient.
    
    Returns (objective_value, gradient_vector)
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = len(corr)  # 2n - 1
    
    # Stable log-sum-exp
    cm = np.max(corr)
    z = np.exp((corr - cm) / temp)
    S = np.sum(z)
    weights = z / S
    value = cm + temp * np.log(S)
    
    # Gradient: grad[j] = dx * sum_k weights[k] * dC_k/dh[j]
    # where C_k = corr[k] = C_s with s = k - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for s in range(-(n - 1), n):
            k = s + n - 1
            w = weights[k]
            if w < 1e-15:
                continue
            # First term: (1-h[j+s]) if 0 <= j+s < n
            if 0 <= j + s < n:
                g += w * (1.0 - h[j + s])
            # Second term: -h[j-s] if 0 <= j-s < n
            if 0 <= j - s < n:
                g -= w * h[j - s]
        grad[j] = g * dx
    
    return value, grad

    """Run SLSQP to minimize the smooth objective starting from h_init."""
    
    def obj(x):
        val, _ = soft_objective_and_grad(x, temp)
        return val
    
    def jac(x):
        _, g = soft_objective_and_grad(x, temp)
        return g
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})

            x = project_feasible(res.x, target_sum)
            if np.sum((x >= -1e-10) & (x <= 1 + 1e-10)):
                return x
    except Exception:
        pass
    return h_init

def pair_line_search(h, i, j):
    """Find optimal delta to transfer from h[i] to h[j]."""
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)

        else:

def full_pair_sweep(h, rng, time_limit=None, t0=None):
