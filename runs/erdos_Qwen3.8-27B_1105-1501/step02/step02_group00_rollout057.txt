    target = n / 2.0

        # Randomize order

        for i in order:
            if time.time() > deadline:
                break
            for j in order:
                if i == j:
                    continue
                # Try moving mass from j to i (t > 0) and from i to j (t < 0)
                t_opt = _optimal_pair_step(h, i, j)

                    h_new[j] -= t_opt
                    c5_new = _exact_c5(h_new)

                        h = h_new
                        best_h = h.copy()
                        best_c5 = c5_new
                        improved = True
        if not improved:
            break
    return best_h, best_c5

            x = np.clip(x, 0.0, 1.0)

        vp = (i + s >= 0) & (i + s < n)
        vm = (i - s >= 0) & (i - s < n)
        G[vp, i] = 1.0 - h[i + s[vp]]
        G[vm, i] -= h[i - s[vm]]
