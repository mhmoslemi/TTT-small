import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    """Project z onto {h: 0<=h<=1, sum(h)=N/2}"""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    # Bisection on shift
    lo, hi = -11.0, 11.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_full(h):
    """Compute gradient matrix G of shape (K, N) where G[r,i] = dC_r/dh_i"""
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for p in range(K):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            # dC/dh_i for i in [s, n-1]: (1-h[i-s])
            G[p, s:] += dx * (1.0 - h[:L])
            # dC/dh_i for i in [0, L-1]: -h[i+s]
            G[p, :L] -= dx * h[s:]
        else:
            a = -s
            L = n - a
            # dC/dh_i for i in [0, L-1]: (1-h[i+a])
            G[p, :L] += dx * (1.0 - h[a:])
            # dC/dh_i for i in [a, n-1]: -h[i-a]
            G[p, a:] -= dx * h[:L]
    return G

def symmetrize(h):
    n = len(h)
    h_sym = 0.5 * (h + (1.0 - h[::-1].copy()))
    # For even n, this automatically gives sum = n/2
    # For odd n, center is 0.5
    return make_feasible(h_sym)

def build_h_from_z(z, n):
    """Build anti-symmetric h from left-half z values. n must be even."""
    m = n // 2
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
    h = np.zeros(n)
    h[:m] = u
    h[m:] = 1.0 - u[::-1]
    return h

def z_from_h(h, n):
    """Extract z from anti-symmetric h (left half via logit)."""
    m = n // 2
    u = h[:m].copy()
    eps = 1e-12
    u = np.clip(u, eps, 1.0 - eps)
    z = np.log(u / (1.0 - u))
    return z

def sym_grad_matrix(h, n):
    """Compute the anti-symmetric gradient matrix of shape (K, M).
    Sz[r, j] = sigmoid'(z[j]) * (G[r, j] - G[r, n-1-j])
    But we need z to compute sigmoid'. Instead, we compute the h-gradient
    and transform on the fly.
    
    Actually, let's just return G (full gradient) and the transformation is done
    in the objective function.
    """
    return grad_full(h)

def smooth_obj_sym(z, n, beta):
    """Compute the log-sum-exp smooth objective and gradient in z-space."""
    m = n // 2
    h = build_h_from_z(z, n)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Numerically stable log-sum-exp
    max_shifted = shifted.max()
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    obj = cmax + log_sum / beta
    # Weights
    w = np.exp(shifted - max_shifted)
    w_sum = w.sum()
    w = w / w_sum
    
    # Full gradient
    G = grad_full(h)
    # Transform to z-space
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
    dsig = u * (1.0 - u)
    # Sz[r, j] = dsig[j] * (G[r, j] - G[r, n-1-j])
    # Shape: G is (K, n), we need G[:, :m] and G[:, n-1:n-m-1:-1]
    # G[r, n-1-j] for j=0..m-1 → G[r, n-1], G[r, n-2], ..., G[r, m]
    G_left = G[:, :m]  # (K, m)
    G_right = G[:, n-1:n-m-1:-1]  # (K, m) - this is G[:, [n-1, n-2, ..., m]]
    Sz = dsig * (G_left - G_right)  # (K, m)
    grad_z = w @ Sz  # (m,)
    return obj, grad_z

        g = np.zeros(m + 1)
        g[-1] = 1.0

        zz = x[:m]
        t = x[-1]
        hh = build_h_from_z(zz, n)
        vv = compute_vals(hh)
        return t - vv[active_idx]
    
    def ineq_jac(x):
        zz = x[:m]
        hh = build_h_from_z(zz, n)
        G = grad_full(hh)
        u = 1.0 / (1.0 + np.exp(-np.clip(zz, -30, 30)))
        dsig = u * (1.0 - u)
        G_left = G[:, :m]
        G_right = G[:, n-1:n-m-1:-1]
        Sz = dsig * (G_left - G_right)
        # Jacobian of (t - C_r) for r in active_idx:
        # d/dz: -Sz[r, :]
        # d/dt: 1
        J = -Sz[active_idx, :]  # (len(active), m)
        Jt = np.ones((len(active_idx), 1))
        return np.hstack([J, Jt])
    
    bounds_z = [(-12.0, 12.0)] * m
    bounds_t = [(0.0, 2.0)]
    bounds = bounds_z + bounds_t
    
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    
    x0 = np.concatenate([z, [t0]])
    t_start = time.time()
    
    try:
        res = minimize(obj_fun, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if time.time() - t_start > time_limit:
            return None, None  # timeout (though SLSQP doesn't have internal timeout)
        if res.x is not None and np.all(np.isfinite(res.x)):
            z_new = res.x[:m]
            h_new = build_h_from_z(z_new, n)
            # Verify
            sc = official_score(h_new)
            if np.isfinite(sc):
                return z_new, sc
    except Exception:
        pass
    return None, None

def find_active(vals, frac=0.3, min_count=10):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0])
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lgbeta_solve_sym(z, n, beta, time_limit=30.0, maxiter=500):
    """Run L-BFGS-B on the smooth symmetric objective."""
    m = n // 2
    
    def fun(x):
        obj, _ = smooth_obj_sym(x, n, beta)
        return obj
    
    def jac(x):
        _, g = smooth_obj_sym(x, n, beta)
        return g
    
    bounds = [(-12.0, 12.0)] * m
    t_start = time.time()
    
    try:
        res = minimize(fun, z, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return z

def full_space_lgbeta(h, beta, time_limit=30.0, maxiter=300):
    """L-BFGS-B in full h-space with penalty for constraint violation."""
    n = len(h)
    
    def fun(x):
        hh = np.clip(x, 0.0, 1.0)
        vals = compute_vals(hh)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        max_s = shifted.max()
        log_sum = max_s + np.log(np.sum(np.exp(shifted - max_s)))
        obj = cmax + log_sum / beta
        # Penalty for constraint violation
        penalty = 100.0 * (np.sum(hh) - n / 2.0) ** 2
        # Penalty for bound violations
        bound_pen = 100.0 * (np.sum(np.maximum(0, -x))**2 + np.sum(np.maximum(0, x-1))**2)
        return obj + penalty + bound_pen
    
    def jac(x):
        hh = np.clip(x, 0.0, 1.0)
        vals = compute_vals(hh)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w_sum = w.sum()
        w = w / w_sum
        G = grad_full(hh)
        grad_obj = w @ G
        # Penalty gradient
        grad_pen = 200.0 * (np.sum(hh) - n / 2.0) * np.ones(n)
        grad_bp = 200.0 * (np.maximum(0, -x) * (-1) + np.maximum(0, x-1))
        # Note: gradient of clip is 1 where in bounds
        grad = np.where((x >= 0) & (x <= 1), grad_obj + grad_pen + grad_bp, 0)
        return grad
    
    t_start = time.time()
    try:
        res = minimize(fun, h.copy(), jac=jac, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            return h_new
    except Exception:
        pass
    return h

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

                if sub <= 0:

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_full(h):
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for p in range(K):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, s:] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[s:]
        else:
            a = -s
            L = n - a
            G[p, :L] += dx * (1.0 - h[a:])
            G[p, a:] -= dx * h[:L]
    return G

def symmetrize(h):

def z_from_h(h, n):

def smooth_obj_sym(z, n, beta):
    m = n // 2
    h = build_h_from_z(z, n)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_s = shifted.max()
    log_sum = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    obj = cmax + log_sum / beta
    w = np.exp(shifted - max_s)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= w_sum
    G = grad_full(h)
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -30.0, 30.0)))

    G_left = G[:, :m]
    G_right = G[:, n - 1:n - m - 1:-1]
    Sz = dsig * (G_left - G_right)
    grad_z = w @ Sz
    return obj, grad_z

        self.beta = beta

def find_active(vals, frac=0.3, min_count=10):
    cmax = vals.max()
    threshold = cmax - max(0.004 * cmax, 1e-7)
    active = set(np.where(vals >= threshold)[0])
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)
