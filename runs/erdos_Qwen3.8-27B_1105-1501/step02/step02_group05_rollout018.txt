import numpy as np
from scipy.optimize import linprog
import time

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Residual repair
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def lag_gradient(h, s):
    """Gradient of lag s of correlate(h, 1-h, mode='full') * dx.
    
    s is the lag: s = p - (n-1) where p is the correlation index.
    Returns array of length n.
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        idx = np.arange(m)
        # c = sum_{i=0}^{m-1} h[i+s] * (1-h[i])
        # dc/dh[i+s] = (1-h[i]) -> contributes to grad[i+s]
        # dc/dh[i] = -h[i+s] -> contributes to grad[i]
        grad[idx + s] += (1.0 - h[idx])
        grad[idx] -= h[idx + s]
    else:
        t = -s
        m = n - t
        idx = np.arange(m)
        # c = sum_{i=0}^{m-1} h[i] * (1-h[i+t])
        # dc/dh[i] = (1-h[i+t]) -> contributes to grad[i]
        # dc/dh[i+t] = -h[i] -> contributes to grad[i+t]
        grad[idx] += (1.0 - h[idx + t])
        grad[idx + t] -= h[idx]
    return dx * grad

def verify_gradient(h, n_check=3):
    """Verify gradient formula against finite differences."""
    n = len(h)
    dx = 2.0 / n
    eps = 1e-7
    # Test a few lags
    for s in range(-n_check, n_check + 1):
        p = s + (n - 1)
        grad = lag_gradient(h, s)
        # Finite difference
        fd = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            vp = np.correlate(h_plus, 1.0 - h_plus, mode='full')[p] * dx
            vm = np.correlate(h_minus, 1.0 - h_minus, mode='full')[p] * dx
            fd[i] = (vp - vm) / (2 * eps)
        if not np.allclose(grad, fd, atol=1e-5):
            return False
    return True

def finite_diff_lag_grad(h, p):
    """Finite difference gradient for correlation index p."""
    n = len(h)
    dx = 2.0 / n
    eps = 1e-7
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        vp = np.correlate(h_plus, 1.0 - h_plus, mode='full')[p] * dx
        vm = np.correlate(h_minus, 1.0 - h_minus, mode='full')[p] * dx
        grad[i] = (vp - vm) / (2 * eps)
    return grad

def lp_step(h, active_p, V, rho, use_fd=False):
    """Solve LP for direction u that reduces active lags.
    
    Returns direction u (array of length n) or None if failed.
    """
    n = len(h)
    C = V.max()
    
    # Build gradient matrix
    G = np.zeros((len(active_p), n))
    for row, p in enumerate(active_p):
        if use_fd:
            G[row] = finite_diff_lag_grad(h, p)
        else:
            s = p - (n - 1)
            G[row] = lag_gradient(h, s)
    
    # LP: minimize t
    # subject to: G@u - t <= -V[active_p]
    #             sum(u) = 0
    #             max(-h_i, -rho) <= u_i <= min(1-h_i, rho)
    #             t <= C
    #             t >= C - 2*rho (loose lower bound to help solver)
    
    # Variables: [u_0, ..., u_{n-1}, t]
    # Objective: c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # G@u - t <= -V[active]
    A_ub = np.zeros((len(active_p), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -V[active_p]
    
    # t <= C (to prevent t from being too large)
    A_ub = np.vstack([A_ub, np.zeros((1, n + 1))])
    A_ub[-1, n] = 1.0
    b_ub = np.append(b_ub, C)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_ub = max(-h[i], -rho)
        hi_ub = min(1.0 - h[i], rho)
        bounds.append((lo_ub, hi_ub))
    bounds.append((C - 2.0 * rho, C + 0.01))
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        u = res.x[:n]
        if np.all(np.isfinite(u)):
            return u
    return None

def lp_refine(h, time_limit, rng, eta_start=0.006, k_max=28, rho_list=None):
    """Iterative LP-based refinement."""
    if rho_list is None:
        rho_list = [0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01, 0.005]
    
    n = len(h)
    h = make_feasible(h.copy())
    best = h.copy()
    best_sc = score_exact(best)
    t_start = time.time()
    
    eta = eta_start
    rho_idx = 0
    stall = 0
    use_fd = not verify_gradient(h)
    
    while time.time() - t_start < time_limit:
        V = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        C = V.max()
        
        # Select active set
        sorted_idx = np.argsort(V)[::-1]
        active = sorted_idx[:k_max].copy()
        # Also include lags within eta of max
        near_max = np.where(V >= C - eta)[0]
        active = np.unique(np.concatenate([active, near_max]))
        active = active[:min(len(active), k_max + 8)]  # Cap total
        
        # Try different rho values

                break

                    improved = True
                    break
            if improved:

def binary_init(h, rng):
    """Create a balanced binary array from continuous h."""
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    # Threshold at 0.5, then adjust count
    b = (h >= 0.5).astype(float)
    cnt = int(b.sum())
    if cnt > target:
        # Remove (cnt - target) ones with smallest h values
        one_idx = np.where(b == 1)[0]
        # Sort by h value ascending
        order = np.argsort(h[one_idx])
        for i in range(cnt - target):
            b[one_idx[order[i]]] = 0
    elif cnt < target:
        # Add (target - cnt) zeros with largest h values
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - cnt):
            b[zero_idx[order[i]]] = 1
    return b

def binary_sa(h_start, n_iter, time_limit, rng, t_start_temp=8e-4, cool=0.997):
    """Simulated annealing on balanced binary strings."""
    n = len(h_start)
    if n % 2 != 0:
        return None, None
    
    b = binary_init(h_start, rng)
    if b is None:
        return None, None
    
    current_sc = score_exact(b)
    best_b = b.copy()
    best_sc = current_sc
    
    t_start = time.time()
    temp = t_start_temp
    
    # Precompute positions
    pos1 = np.where(b == 1)[0]
    pos0 = np.where(b == 0)[0]
    
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        temp *= cool
        if temp < 1e-7:
            temp = 1e-7
        
        # Random swap
        i1 = rng.integers(0, len(pos1))
        i0 = rng.integers(0, len(pos0))
        p1 = pos1[i1]
        p0 = pos0[i0]
        if p1 == p0:
            continue
        
        b[p1] = 0
        b[p0] = 1
        
        new_sc = score_exact(b)
        delta = new_sc - current_sc
        
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            current_sc = new_sc
            # Update position arrays
            pos1 = np.where(b == 1)[0]
            pos0 = np.where(b == 0)[0]
            if new_sc < best_sc - 1e-12:
                best_sc = new_sc
                best_b = b.copy()
        else:
            b[p1] = 1
            b[p0] = 0
    
    return best_b, best_sc

def anti_sym_map(h):
    return make_feasible(0.5 * (h + 1.0 - h[::-1].copy()))

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def two_point_search(h, n_tries, rng, time_limit):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    t_start = time.time()
    
    for it in range(n_tries):
        if time.time() - t_start > time_limit:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            tmp[a] = val_a
            tmp[b] = val_b
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = best[a] + best_t
            best[b] = best[b] - best_t
            best[a] = np.clip(best[a], 0, 1)
            best[b] = np.clip(best[b], 0, 1)
            best_sc = score_exact(best)
    
    return best, best_sc

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:

                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def lag_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx + s] += (1.0 - h[idx])
        grad[idx] -= h[idx + s]
    else:
        t = -s
        m = n - t
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + t])
        grad[idx + t] -= h[idx]
    return dx * grad

def verify_gradient_form(h):
    n = len(h)
    dx = 2.0 / n
    eps = 1e-6
    for s in [0, 1, -1, 2, -2, 5, -5, n - 1, -(n - 1)]:
        s = max(-(n - 1), min(n - 1, s))
        p = s + (n - 1)
        grad = lag_gradient(h, s)
        fd = np.zeros(n)
        for i in range(min(n, 12)):
            hp = h.copy()
            hm = h.copy()
            hp[i] = np.clip(hp[i] + eps, 0, 1)
            hm[i] = np.clip(hm[i] - eps, 0, 1)
            vp = np.correlate(hp, 1.0 - hp, mode='full')[p] * dx
            vm = np.correlate(hm, 1.0 - hm, mode='full')[p] * dx
            fd[i] = (vp - vm) / (hp[i] - hm[i] if hp[i] != hm[i] else 2 * eps)
        if not np.allclose(grad[:12], fd[:12], atol=1e-4):

def finite_diff_lag_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    eps = 1e-6
    grad = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hm = h.copy()

        vm = np.correlate(hm, 1.0 - hm, mode='full')[p] * dx
        denom = hp[i] - hm[i]

def lp_step(h, active_p, V, rho, use_fd=False):
    n = len(h)
    C = V.max()
    G = np.zeros((len(active_p), n))
    for row, p in enumerate(active_p):
        if use_fd:
            G[row] = finite_diff_lag_grad(h, p)
        else:
            s = int(p) - (n - 1)
            G[row] = lag_gradient(h, s)

    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active_p) + 1, n + 1))
    A_ub[:len(active_p), :n] = G
    A_ub[:len(active_p), n] = -1.0
    b_ub = -V[active_p].copy()
    A_ub[len(active_p), n] = 1.0
    b_ub = np.append(b_ub, C + 0.01)
    bounds = []
    for i in range(n):

        hi_ub = min(1.0 - h[i] - 1e-10, rho)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

def binary_init_from_h(h):

def anti_sym_map(h):

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)

def two_point_search(h, time_limit, rng, deadline=1e18):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    t_start = time.time()
    scales = [0.5, 0.25, 0.75, 0.125, 0.875, 0.0]
    while time.time() < deadline and time.time() - t_start < time_limit:
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates = [t_lo + f * span for f in scales]
        best_local = best_sc
        best_t = None
        tmp = best.copy()
        for t in candidates:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-12 or va > 1.0 + 1e-12 or vb < -1e-12 or vb > 1.0 + 1e-12:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = score_exact(best)
    return best, best_sc

def perturb_and_optimize(h, sigma, time_limit, rng, deadline=1e18):
    n = len(h)
    noise = rng.normal(0, sigma, size=n)
    noise -= np.mean(noise)
    h_pert = make_feasible(h + noise)
    sc_pert = score_exact(h_pert)
    if sc_pert > score_exact(h) + 0.05:
        return h, score_exact(h)
    return two_point_search(h_pert, time_limit, rng, deadline=deadline)

def run(seed=42, budget_s=1100, **kwargs):
    t_global_start = time.time()
    reserve = max(15.0, 0.06 * budget_s)
    deadline = t_global_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = score_exact(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except Exception:
        pass

    if best_h is None:
        n_fallback = 60
        x = (np.arange(n_fallback) + 0.5) * (2.0 / n_fallback)
        h_wave = 0.5 + 0.25 * np.sin(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 3 * x)
        best_h = make_feasible(h_wave)
        best_score = score_exact(best_h)

    n = len(best_h)

    # Phase 1: Quick 2-point warmup
    if time.time() < deadline:
        tl = min(20.0, (deadline - time.time()) * 0.05)
        if tl > 3:
            h_new, sc = two_point_search(best_h, tl, rng, deadline=deadline)
            if sc < best_score - 1e-12:
                best_h = h_new
                best_score = sc

    # Phase 2: Binary annealing (multiple starts)
    if time.time() < deadline and n % 2 == 0:
        tl = min(80.0, (deadline - time.time()) * 0.1)
        if tl > 15:
            chunk = min(30.0, tl / 3.0)
            for attempt in range(3):
                if time.time() >= deadline:
                    break
                if attempt == 0:
                    start = best_h
                elif attempt == 1:
                    start = anti_sym_map(best_h)
                else:
                    noise = rng.normal(0, 0.1, size=n)
                    noise -= np.mean(noise)
                    start = make_feasible(best_h + noise)
                b_best, b_sc = binary_sa(start, chunk, rng, deadline=deadline)
                if b_best is not None:
                    h_from_bin = make_feasible(b_best.astype(float))
                    sc_bw = score_exact(h_from_bin)
                    if sc_bw < best_score - 1e-12:
                        best_h = h_from_bin
                        best_score = sc_bw
                    h_sym_bin = anti_sym_map(b_best.astype(float))
                    sc_sb = score_exact(h_sym_bin)
                    if sc_sb < best_score - 1e-12:
                        best_h = h_sym_bin
                        best_score = sc_sb

    # Phase 3: Main LP refinement
    if time.time() < deadline:
        tl = min(500.0, (deadline - time.time()) * 0.5)
        if tl > 30:
            h_lp, sc_lp = lp_refine(best_h, tl, rng, eta_start=0.006, k_max=28,
                                    deadline=deadline)
            if sc_lp < best_score - 1e-12:
                best_h = h_lp
                best_score = sc_lp

    # Phase 4: 2-point polish
    if time.time() < deadline:
        tl = min(40.0, (deadline - time.time()) * 0.06)
        if tl > 5:
            h_new, sc = two_point_search(best_h, tl, rng, deadline=deadline)
            if sc < best_score - 1e-12:
                best_h = h_new
                best_score = sc

    # Phase 5: LP with different params on anti-symmetric
    if time.time() < deadline:
        h_sym = anti_sym_map(best_h)
        sc_sym = score_exact(h_sym)
        if sc_sym < best_score - 1e-12:
            best_h = h_sym
            best_score = sc_sym
        tl = min(200.0, (deadline - time.time()) * 0.2)
        if tl > 20:
            h_lp2, sc_lp2 = lp_refine(h_sym, tl, rng, eta_start=0.004, k_max=22,
                                      rho_list=[0.2, 0.12, 0.07, 0.04, 0.02, 0.01],
                                      deadline=deadline)
            if sc_lp2 < best_score - 1e-12:
                best_h = h_lp2
                best_score = sc_lp2

    # Phase 6: Resolution changes
    if time.time() < deadline:
        n_base = len(best_h)
        n_opts = []
        for dn in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16]:
            nt = n_base + dn
            if 32 <= nt <= 100:
                n_opts.append(nt)
        for nc in [36, 44, 50, 56, 64, 72, 80]:
            if nc not in n_opts and 32 <= nc <= 100:
                n_opts.append(nc)
        n_opts = sorted(set(n_opts))
        for n_opt in n_opts:
            if time.time() >= deadline:
                break
            if n_opt == len(best_h):
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = score_exact(h_int)
            if sc_int < best_score - 1e-12:
                best_h = h_int
                best_score = sc_int
            tl = min(12.0, (deadline - time.time()) * 0.02)
            if tl > 3:
                h_new, sc = two_point_search(h_int, tl, rng, deadline=deadline)
                if sc < best_score - 1e-12:
                    best_h = h_new
                    best_score = sc
                if sc_int < best_score - 1e-12:
                    best_h = interp_to(h_new, n_opt)
                    best_score = score_exact(best_h)

    # Phase 7: Final LP with tighter params
    if time.time() < deadline:
        tl = min(150.0, (deadline - time.time()) * 0.2)
        if tl > 15:
            h_lp3, sc_lp3 = lp_refine(best_h, tl, rng, eta_start=0.003, k_max=18,
                                      rho_list=[0.15, 0.08, 0.05, 0.03, 0.015, 0.008],
                                      deadline=deadline)
            if sc_lp3 < best_score - 1e-12:
                best_h = h_lp3
                best_score = sc_lp3

    # Phase 8: Perturbation + optimization
    if time.time() < deadline:
        for sigma in [0.03, 0.08, 0.12]:
            if time.time() >= deadline:
                break
            tl = min(20.0, (deadline - time.time()) * 0.03)
            if tl > 3:
                h_pert, sc_p = perturb_and_optimize(best_h, sigma, tl, rng, deadline=deadline)
                if sc_p < best_score - 1e-12:
                    best_h = h_pert
                    best_score = sc_p

    # Phase 9: Final polish
    if time.time() < deadline:
        tl = min(30.0, (deadline - time.time()) * 0.08)
        if tl > 3:
            h_new, sc = two_point_search(best_h, tl, rng, deadline=deadline)
            if sc < best_score - 1e-12:
                best_h = h_new
                best_score = sc
        if time.time() < deadline:
            tl2 = min(40.0, (deadline - time.time()) * 0.1)
            if tl2 > 10:
                h_lp4, sc_lp4 = lp_refine(best_h, tl2, rng, eta_start=0.002, k_max=16,
                                          rho_list=[0.1, 0.05, 0.03, 0.015, 0.008],
                                          deadline=deadline)
                if sc_lp4 < best_score - 1e-12:
                    best_h = h_lp4
                    best_score = sc_lp4

    # Final validation
    final_h = make_feasible(best_h)
    final_score = score_exact(final_h)
    final_n = len(final_h)

    if not np.all(np.isfinite(final_h)) or final_h.min() < -1e-8 or final_h.max() > 1.0 + 1e-8:
        final_h = make_feasible(0.5 * np.ones(final_n))
        final_score = score_exact(final_h)

    if abs(final_h.sum() - final_n / 2.0) > 1e-8:
        final_h = make_feasible(final_h)
        final_score = score_exact(final_h)

    return final_h, final_score, final_n
