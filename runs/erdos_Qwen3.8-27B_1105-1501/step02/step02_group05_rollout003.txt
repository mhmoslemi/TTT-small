The gradient of each lag value vals[p] with respect to h is well-defined. For a given lag k (where p = n-1+k):
- If k >= 0: vals[p] = dx * sum_{r=0}^{n-1-k} h[r+k]*(1-h[r])

  - d/dh[r+k] = dx * (1-h[r]) (for r in 0..n-1-k)
- If k < 0 (let q = -k): vals[p] = dx * sum_{r=q}^{n-1} h[r-q]*(1-h[r])

  - d/dh[r] = -dx * h[r-q] (for r in q..n-1)

    dx = 2.0 / n

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:

                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:

def grad_single_lag(h, p):
    """Gradient of vals[p] with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    k = p - (n - 1)
    if k >= 0:

            grad[r] += -dx * h[r + k]
            grad[r + k] += dx * (1.0 - h[r])
    else:
        q = -k
        for r in range(q, n):
            grad[r - q] += dx * (1.0 - h[r])
            grad[r] += -dx * h[r - q]

    n = len(h)

        G[i] = grad_single_lag(h, p)
    return G

def qp_solve(h, mu, k_active, trust_radius, solver_timeout=30):
    """Solve trust-region QP to find a descent direction."""
    n = len(h)
    vals = vals_of(h)
    C = vals.max()
    
    # Select active lags
    if k_active >= 2 * n - 1:
        sel = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(vals)[-k_active:]
        sel = top_idx
    
    G = build_gradient_matrix(h, sel)
    v_sel = vals[sel]
    
    # QP: minimize t + mu * ||d||^2
    # s.t. G @ d <= t - C + v_sel (i.e., G@d - t <= v_sel - C, all <= 0)
    #      sum(d) = 0
    #      -h <= d <= 1-h
    #      ||d|| <= trust_radius (optional, use L2 norm as constraint)
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    obj = cp.Minimize(t + mu * cp.sum_squares(d))
    constraints = [
        G @ d - t <= (v_sel - C),
        cp.sum(d) == 0,
        d >= -h,
        d <= 1.0 - h
    ]
    
    # Add trust region
    if trust_radius > 0:
        constraints.append(cp.norm(d, 2) <= trust_radius)
    
    prob = cp.Problem(obj, constraints)
    
    # Try solvers in order
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.OSQP, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                if d.value is not None and np.all(np.isfinite(d.value)):
                    return d.value.copy()
        except Exception:

    return None

- vals[p] = a_p * h[i] + b_p (where a_p and b_p depend on the other variables)
- We want to minimize max_p (a_p * h[i] + b_p) subject to 0 <= h[i] <= 1 and the sum constraint.

The sum constraint: sum_j h[j] = n/2, so if we change h[i], we need to adjust another variable. This makes it a 2D problem (which is what the two-point search does).

import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def vals_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_single_lag(h, p):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    k = p - (n - 1)
    if k >= 0:
        for r in range(n - k):
            grad[r] -= dx * h[r + k]
            grad[r + k] += dx * (1.0 - h[r])
    else:
        q = -k
        for r in range(q, n):
            grad[r - q] += dx * (1.0 - h[r])
            grad[r] -= dx * h[r - q]
    return grad

def build_gradient_matrix(h, indices):
    n = len(h)
    G = np.zeros((len(indices), n))
    for i, p in enumerate(indices):
        G[i] = grad_single_lag(h, p)
    return G

def qp_solve(h, mu, k_active, trust_radius):
    n = len(h)
    vals = vals_of(h)
    C = vals.max()
    total_lags = 2 * n - 1
    if k_active >= total_lags:
        sel = np.arange(total_lags)
    else:
        sel = np.argsort(vals)[-k_active:]
    G = build_gradient_matrix(h, sel)
    v_sel = vals[sel]
    d = cp.Variable(n)
    t = cp.Variable()
    obj = cp.Minimize(t + mu * cp.sum_squares(d))
    constraints = [
        G @ d - t <= (v_sel - C),
        cp.sum(d) == 0,
        d >= -h - 1e-9,
        d <= 1.0 - h + 1e-9
    ]
    if trust_radius > 0:
        constraints.append(cp.norm(d, 2) <= trust_radius)
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.OSQP]:
        try:
            prob.solve(solver=solver)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                if d.value is not None and np.all(np.isfinite(d.value)):
                    return np.asarray(d.value, dtype=float).copy()
        except Exception:
            continue
    return None

def two_point_search(h, n_tries, rng, is_dead):
