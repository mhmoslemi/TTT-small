        lo = max(0.0, (idx - 1) / 50.0)
        hi = min(1.0, (idx + 1) / 50.0)
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = np.max(orig + m1 * L - m1 * m1 * q) * dx
            v2 = np.max(orig + m2 * L - m2 * m2 * q) * dx
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        best_alpha = (lo + hi) / 2.0
        best_c5 = np.max(orig + best_alpha * L - best_alpha * best_alpha * q) * dx
    
    return best_alpha, best_c5

def pair_optimal_delta_fast(h, i, j, orig, g):
    N = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-13:
        return 0.0
    
    L = g[:, j] - g[:, i]
    # q for unit pair direction
    # q[m] is correlate of d with itself where d[i]=-1, d[j]=1
    # q = sum_{n:valid} d[n+k]*d[n]
    # Nonzero only when {n, n+k} intersects {i, j}
    # For k=0: q = d[i]^2 + d[j]^2 = 2
    # For k=j-i: q = d[j]*d[i] = -1 (if valid)
    # For k=i-j: q = d[i]*d[j] = -1 (if valid)
    # All others: 0
    q = np.zeros(2 * N - 1)
    q[N - 1] = 2.0  # k=0
    if abs(i - j) <= N - 1:
        q[N - 1 + (j - i)] = -1.0  # k = j-i
        q[N - 1 + (i - j)] = -1.0  # k = i-j
    
    # Line search
    deltas = np.linspace(lo, hi, 20)
    best_delta = 0.0
    best_val = np.max(orig)
    for delta in deltas:
        val = np.max(orig + delta * L - delta * delta * q)
        if val < best_val:
            best_val = val
            best_delta = delta
    
    if -1e-12 < best_delta < 1e-12:
        return 0.0
    
    # Refine
    idx = int((best_delta - lo) / (hi - lo) * 20) if hi > lo else 0
    lo2 = max(lo, deltas[max(0, idx - 1)])
    hi2 = min(hi, deltas[min(19, idx + 1)])
    
    for _ in range(35):
        m1 = lo2 + (hi2 - lo2) / 3.0
        m2 = hi2 - (hi2 - lo2) / 3.0
        v1 = np.max(orig + m1 * L - m1 * m1 * q)
        v2 = np.max(orig + m2 * L - m2 * m2 * q)
        if v1 < v2:
            hi2 = m2
        else:
            lo2 = m1
    
    return (lo2 + hi2) / 2.0

    except (NameError, Exception):
        h = np.full(70, 0.5)

    h = project_onto_constraint(h, n / 2.0)
    h = np.clip(h, 0.0, 1.0)
    
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    def remaining_time():
        return budget_s - (time.time() - t0)
