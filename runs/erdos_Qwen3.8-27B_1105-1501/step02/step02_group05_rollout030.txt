    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] += (-h[k:])
    else:
        q = -k
        if q < n:
            grad[:n-q] += (1.0 - h[q:])
            grad[q:] += (-h[:n-q])
    return dx * grad

def solve_qp(h, vals, C, active_indices, mu, solver='ECOS'):
    n = len(h)
    d = cp.Variable(n)
    t = cp.Variable()
    
    # Build gradient rows for active lags
    constraints = [cp.sum(d) == 0, d >= -h, d <= 1.0 - h]
    for idx in active_indices:
        k = idx - (n - 1)
        g = lag_gradient(h, k)
        constraints.append(vals[idx] + g @ d <= t)
    
    objective = cp.Minimize(t + mu * cp.sum_squares(d))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if d.value is not None and np.all(np.isfinite(d.value)):
            return d.value
    except Exception:
        pass
    
    # Try other solvers
    for alt_solver in ['OSQP', 'SCS', 'CLARABEL', 'CVXOPT']:
        try:
            prob.solve(solver=alt_solver)
            if d.value is not None and np.all(np.isfinite(d.value)):
                return d.value
        except Exception:
            continue
    return None

def two_point_search(h, n_tries, rng, is_dead):

        if is_dead():
            break

        b = rng.integers(0, n)

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def objective_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] += (-h[k:])
    else:
        q = -k
        if q < n:
            grad[:n - q] += (1.0 - h[q:])
            grad[q:] += (-h[:n - q])
    return dx * grad

def solve_qp(h, vals, active_indices, mu):
    n = len(h)
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(d) == 0, d >= -h, d <= 1.0 - h]
    for idx in active_indices:
        k = idx - (n - 1)
        g = lag_gradient(h, k)
        constraints.append(vals[idx] + g @ d <= t)
    obj = cp.Minimize(t + mu * cp.sum_squares(d))
    prob = cp.Problem(obj, constraints)
    for solver in ['ECOS', 'OSQP', 'SCS', 'CLARABEL', 'CVXOPT']:
        try:
            prob.solve(solver=solver)
            if d.value is not None and np.all(np.isfinite(d.value)):
                return np.asarray(d.value, dtype=float)
        except Exception:
            continue
    return None

def solve_qp_sym(h, vals, active_indices, mu):
    """QP with anti-symmetry constraint d[i] + d[n-1-i] = 0."""
    n = len(h)
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(d) == 0, d >= -h, d <= 1.0 - h]
    for i in range(n // 2):
        constraints.append(d[i] + d[n - 1 - i] == 0)
    for idx in active_indices:

def qp_one_step(h, mu, K, use_sym=False, is_dead=None):
    n = len(h)
    vals = objective_values(h)
    C = vals.max()
    if K >= len(vals):

    else:
        active = np.argsort(vals)[::-1][:K]
    if use_sym:
        d = solve_qp_sym(h, vals, active, mu)
    else:
        d = solve_qp(h, vals, active, mu)

        return None, None
    if np.max(np.abs(d)) < 1e-12:
        return None, None
    best_trial = None

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        cands = [0.5 * span, 0.25 * span, 0.75 * span,
                 span / 2.618, span * 2 / 2.618]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in cands:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = official_score(best)
    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + phase) + a2 * np.cos(2 * np.pi * f2 * x)
    return make_feasible(h)
