import numpy as np
import time
import cvxpy as cp


def compute_c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_gradient_rows(h, lags):
    """Build gradient rows for given lags. Returns matrix of shape (len(lags), n)."""
    n = len(h)
    off = n - 1
    G = np.zeros((len(lags), n))
    for r, ell in enumerate(lags):
        k = ell - off  # the lag index
        # dc_k/dh[p] = (1 - h[p-k]) - h[p+k] (with OOB = 0)
        for p in range(n):
            g = 0.0
            pk = p - k  # index in (1-h)
            if 0 <= pk < n:
                g += 1.0 - h[pk]
            pk2 = p + k  # index in h
            if 0 <= pk2 < n:
                g -= h[pk2]
            G[r, p] = g
    return G


def solve_lp_direction(h, active_lags, corr_values, trust_radius=0.05):
    """Solve LP to find a direction that decreases active lags."""
    n = len(h)
    off = n - 1
    raw_max = np.max(corr_values)
    
    G = build_gradient_rows(h, active_lags)
    # G has shape (K, n)
    
    d = cp.Variable(n)
    theta = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        d >= -h,
        d <= 1.0 - h,
        d >= -trust_radius,
        d <= trust_radius,
        theta >= 0,
    ]
    
    # For each active lag: G[k] @ d <= -theta
    # i.e., G @ d + theta <= 0
    constraints.append(G @ d + theta <= np.zeros(len(active_lags)))
    
    obj = cp.Maximize(theta)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None, 0.0
    
    if d.value is None or not np.all(np.isfinite(d.value)):
        return None, 0.0
    
    d_val = d.value
    # Project to exactly satisfy sum constraint
    d_val = d_val - np.mean(d_val)
    # Clip to bounds
    d_val = np.clip(d_val, -h + 1e-12, 1.0 - h - 1e-12)
    d_val = d_val - np.mean(d_val)
    
    theta_val = theta.value if theta.value is not None else 0.0
    return d_val, theta_val


def pair_optimal_delta(h, i, j, n_iter=40):
    """Find optimal delta for transferring from i to j."""
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5_raw(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5_raw(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pairwise_polish(h, n_pairs, rng):
    """Do random pairwise polish steps."""
    n = len(h)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = pair_optimal_delta(h, i, j, n_iter=25)
        if abs(delta) > 1e-10:
            h[i] -= delta
            h[j] += delta


def lp_descent_step(h, rng, trust_radius=0.05):
    """One LP-based descent step."""
    n = len(h)
    corr = compute_corr(h)
    raw_max = np.max(corr)
    off = n - 1
    
    # Find active lags: those within a small margin of the max
    margin = 0.02
    active_mask = corr >= raw_max - margin
    active_lags = np.where(active_mask)[0]
    
    # Limit to top K
    if len(active_lags) > 15:
        top_k = np.argsort(corr[active_lags])[-15:]
        active_lags = active_lags[top_k]
    
    if len(active_lags) == 0:
        return h, compute_c5_raw(h), False
    
    d, theta = solve_lp_direction(h, active_lags, corr, trust_radius)
    
    if d is None or np.linalg.norm(d) < 1e-12:
        return h, compute_c5_raw(h), False
    
    # Line search
    best_h = h.copy()
    best_c5 = compute_c5_raw(h)
    
    for alpha in [1.0, 0.5, 0.25, 0.1, 0.5 * 0.5, 0.25 * 0.5]:
        h_trial = h + alpha * d
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = project_onto_constraint(h_trial, n / 2.0)
        c5_trial = compute_c5_raw(h_trial)
        if c5_trial < best_c5 - 1e-12:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    
    if c5_trial < best_c5 or best_c5 < compute_c5_raw(h) - 1e-12:
        return best_h, best_c5, True
    
    return h, best_c5, False


def full_pair_sweep(h, rng, n_rounds=1):
    """Full sweep over all pairs."""
    n = len(h)
    for _ in range(n_rounds):
        perm = rng.permutation(n)
        improved = False
        for a_idx in range(n):
            for b_idx in range(a_idx + 1, n):
                i, j = perm[a_idx], perm[b_idx]
                delta = pair_optimal_delta(h, i, j, n_iter=20)
                if abs(delta) > 1e-10:
                    h[i] -= delta
                    h[j] += delta
                    improved = True
        if not improved:
            break


def stochastic_kick(h, rng, scale=0.03):
    """Add random perturbation and re-project."""
    n = len(h)
    h_kick = h + rng.randn(n) * scale
    h_kick = project_onto_constraint(h_kick, n / 2.0)
    return h_kick

import numpy as np
import time
import scipy.optimize as sco


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_gradient_rows(h, ell_indices):
    n = len(h)
    off = n - 1
    K = len(ell_indices)
    G = np.zeros((K, n))
    for r in range(K):
        k = ell_indices[r] - off
        for p in range(n):
            g = 0.0
            idx1 = p - k
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]
            idx2 = p + k
            if 0 <= idx2 < n:
                g -= h[idx2]
            G[r, p] = g
    return G


def solve_lp(h, active_ells, trust_radius):
    n = len(h)
    off = n - 1
    G = build_gradient_rows(h, active_ells)
    K = len(active_ells)
    
    # Variables: x = [d_0, ..., d_{n-1}, theta], size n+1
    # Minimize: -theta => c = [0, ..., 0, -1]
    c = np.zeros(n + 1)
    c[n] = -1.0
    
    # Inequalities: A_ub @ x <= b_ub
    # 1) G[k] @ d + theta <= 0 for each active lag
    A_parts = []
    b_parts = []
    
    A1 = np.zeros((K, n + 1))
    A1[:, :n] = G
    A1[:, n] = 1.0
    A_parts.append(A1)
    b_parts.append(np.zeros(K))
    
    # 2) d_i <= 1 - h[i]: e_i @ d <= 1 - h[i]
    A2 = np.zeros((2 * n, n + 1))
    A2[:n, :n] = np.eye(n)
    A2[n:, :n] = -np.eye(n)
    b2 = np.concatenate([1.0 - h, h])
    A_parts.append(A2)
    b_parts.append(b2)
    
    # 3) Trust region: |d_i| <= trust_radius
    A3 = np.zeros((2 * n, n + 1))
    A3[:n, :n] = np.eye(n)
    A3[n:, :n] = -np.eye(n)
    b3 = np.full(2 * n, trust_radius)
    A_parts.append(A3)
    b_parts.append(b3)
    
    # 4) theta >= 0: -theta <= 0
    A4 = np.zeros((1, n + 1))
    A4[0, n] = -1.0
    A_parts.append(A4)
    b_parts.append(np.zeros(1))
    
    A_ub = np.vstack(A_parts)
    b_ub = np.concatenate(b_parts)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = sco.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          method='highs', options={'maxiter': 5000})
    except:
        try:
            res = sco.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              method='interior-point')
        except:
            return None, 0.0
    
    if not res.success:
        return None, 0.0
    
    d = res.x[:n].copy()
    theta = res.x[n]
    
    # Clean up: ensure sum(d) = 0 exactly
    d = d - np.mean(d)
    # Clip to bounds
    lb = np.maximum(-h, -trust_radius) + 1e-10
    ub = np.minimum(1.0 - h, trust_radius) - 1e-10
    d = np.clip(d, lb, ub)
    d = d - np.mean(d)
    
    if np.linalg.norm(d) < 1e-12:
        return None, 0.0
    
    return d, max(0.0, theta)


def pair_optimal_delta(h, i, j, n_iter=35):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5_raw(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5_raw(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    best_m = (a + b) / 2.0
    h[i] = hi_v - best_m
    h[j] = hj_v + best_m
    return best_m


def pairwise_polish(h, n_pairs, rng, n_iter=25):
    n = len(h)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        pair_optimal_delta(h, i, j, n_iter=n_iter)


def full_pair_sweep(h, rng):
    n = len(h)
    perm = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            dz = max(-h[j], h[i] - 1.0)
            dzhi = min(h[i], 1.0 - h[j])
            if dzhi - dz < 1e-14:
                continue
            pair_optimal_delta(h, i, j, n_iter=20)


def lp_descent(h, rng, trust_radius):
    n = len(h)
    corr = compute_corr(h)
    raw_max = np.max(corr)
    
    # Select active lags
    margin = max(0.01, raw_max * 0.005)
    active_mask = corr >= raw_max - margin
    active_ells = np.where(active_mask)[0]
    
    if len(active_ells) > 20:
        idx_sorted = np.argsort(corr[active_ells])
        active_ells = active_ells[idx_sorted[-20:]]
    
    if len(active_ells) == 0:
        return h, compute_c5_raw(h), False
    
    d, theta = solve_lp(h, active_ells, trust_radius)
    
    if d is None:
        return h, compute_c5_raw(h), False
    
    # Exact line search
    best_h = h.copy()
    best_c5 = compute_c5_raw(h)
    
    alphas = [1.0, 0.7, 0.5, 0.3, 0.1]
    for alpha in alphas:
        h_trial = h + alpha * d
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = project_onto_constraint(h_trial, n / 2.0)
        c5t = compute_c5_raw(h_trial)
        if c5t < best_c5 - 1e-13:
            best_c5 = c5t
            best_h = h_trial.copy()
    
    improved = best_c5 < compute_c5_raw(h) - 1e-13
    return best_h, best_c5, improved


def gradient_descent_step(h, temp=0.05, step_size=0.01):
    n = len(h)
    corr = compute_corr(h)
    raw_max = np.max(corr)
    
    # Softmax weights
    exp_vals = np.exp((corr - raw_max) / temp)
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    off = n - 1
    grad = np.zeros(n)
    for r in range(len(corr)):
        w = weights[r]
        if w < 1e-10:
            continue
        k = r - off
        for p in range(n):
            g = 0.0
            idx1 = p - k
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]
            idx2 = p + k
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[p] += w * g
    
    # Project gradient step
    h_trial = h - step_size * grad
    h_trial = np.clip(h_trial, 0.0, 1.0)
    h_trial = project_onto_constraint(h_trial, n / 2.0)
    return h_trial


def make_binary_start(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_golden_start(n):
    h = np.zeros(n)
    inv_phi = (np.sqrt(5) - 1) / 2.0
    for i in range(n):
        h[i] = 1.0 - ((i * inv_phi) % 1.0)
    return project_onto_constraint(h, n / 2.0)


def make_cosine_start(n):

    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    return project_onto_constraint(h, n / 2.0)

import numpy as np
import time
import scipy.optimize as sco


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_gradient_rows(h, ell_indices):
    n = len(h)
    off = n - 1
    K = len(ell_indices)
    G = np.zeros((K, n))
    for r in range(K):
        k = int(ell_indices[r]) - off
        for p in range(n):
            g = 0.0
            idx1 = p - k
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]
            idx2 = p + k
            if 0 <= idx2 < n:
                g -= h[idx2]
            G[r, p] = g
    return G


def solve_lp(h, active_ells, trust_radius):
    n = len(h)
    G = build_gradient_rows(h, active_ells)
    K = len(active_ells)
    c = np.zeros(n + 1)
    c[n] = -1.0
    A_parts = []
    b_parts = []
    A1 = np.zeros((K, n + 1))
    A1[:, :n] = G
    A1[:, n] = 1.0
    A_parts.append(A1)
    b_parts.append(np.zeros(K))
    A2 = np.zeros((2 * n, n + 1))
    A2[:n, :n] = np.eye(n)
    A2[n:, :n] = -np.eye(n)
    b2 = np.concatenate([1.0 - h, h])
    A_parts.append(A2)
    b_parts.append(b2)
    A3 = np.zeros((2 * n, n + 1))
    A3[:n, :n] = np.eye(n)
    A3[n:, :n] = -np.eye(n)
    b3 = np.full(2 * n, trust_radius)
    A_parts.append(A3)
    b_parts.append(b3)
    A4 = np.zeros((1, n + 1))
    A4[0, n] = -1.0
    A_parts.append(A4)
    b_parts.append(np.zeros(1))
    A_ub = np.vstack(A_parts)
    b_ub = np.concatenate(b_parts)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    try:
        res = sco.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          method='highs', options={'maxiter': 10000})
    except Exception:
        try:
            res = sco.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              method='interior-point')
        except Exception:
            return None, 0.0
    if not res.success:
        return None, 0.0
    d = res.x[:n].copy()
    theta = float(res.x[n])
    d = d - np.mean(d)
    lb = np.maximum(-h, -trust_radius) + 1e-10
    ub = np.minimum(1.0 - h, trust_radius) - 1e-10
    d = np.clip(d, lb, ub)
    d = d - np.mean(d)
    if np.linalg.norm(d) < 1e-10:
        return None, 0.0
    return d, max(0.0, theta)


def pair_optimal_delta_inplace(h, i, j, n_iter=30):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5_raw(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5_raw(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    best_m = (a + b) / 2.0
    h[i] = hi_v - best_m
    h[j] = hj_v + best_m
    return best_m


def pairwise_polish(h, n_pairs, rng, n_iter=25):
    n = len(h)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        pair_optimal_delta_inplace(h, i, j, n_iter=n_iter)


def full_pair_sweep(h, rng, n_iter=20):
    n = len(h)
    perm = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = int(perm[a]), int(perm[b])
            lo = max(-h[j], h[i] - 1.0)
            hi = min(h[i], 1.0 - h[j])
            if hi - lo < 1e-14:
                continue
            pair_optimal_delta_inplace(h, i, j, n_iter=n_iter)

    if len(active_ells) < 1:

    d, theta = solve_lp(h, active_ells, trust_radius)

    base_c5 = compute_c5_raw(h)

        c5t = compute_c5_raw(h_trial)
        if c5t < best_c5 - 1e-14:
            best_c5 = c5t
            best_h = h_trial.copy()
    improved = best_c5 < base_c5 - 1e-14
    return best_h, best_c5, improved


def gradient_step(h, temp=0.04, step_size=0.008):
    n = len(h)
    off = n - 1
    corr = compute_corr(h)
    raw_max = float(np.max(corr))
    exp_vals = np.exp((corr - raw_max) / temp)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for r in range(len(corr)):
        w = weights[r]
        if w < 1e-12:
            continue
        k = r - off
        for p in range(n):
            g = 0.0
            idx1 = p - k
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]
            idx2 = p + k
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[p] += w * g
    h_trial = h - step_size * grad
    h_trial = np.clip(h_trial, 0.0, 1.0)
    h_trial = project_onto_constraint(h_trial, float(n / 2.0))
    return h_trial


def make_binary_start(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_golden_start(n):
    h = np.zeros(n)
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    for i in range(n):
        h[i] = 1.0 - ((i * inv_phi) % 1.0)
    return project_onto_constraint(h, float(n / 2.0))


def make_cosine_start(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(2.0 * np.pi * x * 0.5)
    return project_onto_constraint(h, float(n / 2.0))


def make_voronoi_start(n, rng):
    h = np.zeros(n)
    centers = rng.choice(n, size=n // 4, replace=False)
    for i in range(n):
        dmin = min(abs(i - c) for c in centers)
        h[i] = 1.0 - min(1.0, dmin / 5.0)
    return project_onto_constraint(h, float(n / 2.0))
