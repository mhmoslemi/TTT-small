            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-9:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _smooth_grad(h, beta):
    scores = _scores(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.max(scores)
        w = np.ones(len(scores)) / len(scores)
    else:
        w = w / ws
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad


def _line_search(h, p, c5_cur, n_coarse=20):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    if alpha_max < 1e-10:
        return 0.0, c5_cur

    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.unique(np.concatenate([
        np.logspace(-4, np.log10(alpha_max), 10),
        np.linspace(0.0, alpha_max, 10)
    ]))
    alphas = alphas[(alphas > 0) & (alphas <= alpha_max + 1e-12)]

    for a in alphas:
        h_t = h + a * p
        if h_t.min() < -1e-10 or h_t.max() > 1.0 + 1e-10:
            continue
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    delta = max(alpha_max * 0.08, 0.003)
    lo_a = max(0.0, best_a - delta)
    hi_a = min(alpha_max, best_a + delta)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(18):
        if hi_a - lo_a < 1e-11:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)

    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _main_optimize(h, deadline, beta_start=20.0, max_iters=10000, use_lp=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    beta = beta_start
    radius = 0.12
    momentum = np.zeros(n)
    mom_rate = 0.7

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break

        scores = _scores(h)
        grad = _smooth_grad(h, beta)
        norm = float(np.max(np.abs(grad)))
        p = -grad / norm if norm > 1e-12 else None

        lp_p = None
        if use_lp and (p is None or it % 5 == 0):
            G = _grad_matrix(h)
            lp_p = _lp_direction(h, scores, G, radius)

        candidates = []
        if p is not None:
            candidates.append(p)
        if lp_p is not None and float(np.max(np.abs(lp_p))) > 1e-12:
            if p is not None:
                lp_p_norm = lp_p / max(float(np.max(np.abs(lp_p))), 1e-12)
            else:
                lp_p_norm = lp_p
            candidates.append(lp_p_norm)

        # Stochastic direction as escape
        if stall > 20 and norm < 1e-8:
            active = np.where(scores > float(np.max(scores)) * 0.95)[0]
            if len(active) > 0:
                idx = active[np.random.randint(len(active))]
                rand_dir = np.random.randn(n)
                rand_dir -= np.mean(rand_dir)
                rand_dir /= max(float(np.max(np.abs(rand_dir))), 1e-12)
                candidates.append(rand_dir)

        best_a = 0.0
        best_new_c5 = cur_c5
        for p_cand in candidates:
            a, c_new = _line_search(h, p_cand, cur_c5, n_coarse=16)
            if c_new < best_new_c5:
                best_new_c5 = c_new
                best_a = a

        if best_new_c5 < cur_c5 - 1e-12 and best_a > 0:
            # Find which direction worked (re-evaluate for the best one)
            # Just use the first candidate for simplicity
            h_new = _project(h + best_a * candidates[0], target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur_c5 - 1e-13:
                momentum = mom_rate * momentum + (1 - mom_rate) * (-grad / max(norm, 1e-12))
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                beta = min(beta * 1.04, 150.0)
                radius = min(radius * 1.08, 0.5)
            else:
                stall += 1
                beta = max(beta * 0.9, 3.0)
                radius *= 0.9
        else:
            stall += 1
            beta = max(beta * 0.88, 3.0)
            radius *= 0.88
            if stall % 15 == 0 and radius < 0.015:
                radius = 0.06

    return best_h, best_c5


def _pair_polish(h, deadline, max_passes=25):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]))
                    else:
                        a_max = min(a_max, h[j])
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]))
                    else:
                        a_max = min(a_max, h[k])
                    if a_max < 1e-10:
                        continue
                    for a in [a_max * t for t in [0.1, 0.3, 0.5, 0.7, 1.0]]:
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Main optimization (smooth gradient + LP)
    deadline1 = t0 + budget_s * 0.42
    h, c5 = _main_optimize(h, deadline1, beta_start=25.0, max_iters=10000, use_lp=True)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Perturbation restarts with varied scales
    deadline2 = t0 + budget_s * 0.75
    scales = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3]
    for scale in scales:
        if time.time() > deadline2:
            break
        if best_c5 < 0.38072:
            break
        for attempt in range(20):
            if time.time() > deadline2:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(deadline2, time.time() + budget_s * 0.035)
            h_try, c5_try = _main_optimize(h_pert, sub_dl, beta_start=50.0, max_iters=600, use_lp=False)
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 3: Structured perturbations (sinusoidal, block shifts)
    if time.time() < t0 + budget_s * 0.84:
        deadline3 = t0 + budget_s * 0.86
        n_pts = np.arange(best_n)
        for freq in [1, 2, 3, 4, 5]:
            if time.time() > deadline3:
                break
            for amp in [0.05, 0.1, 0.2]:
                if time.time() > deadline3:
                    break
                h_sin = best_h + amp * np.sin(2 * np.pi * freq * n_pts / best_n)
                h_sin = _project(h_sin, best_n / 2.0)
                sub_dl = min(deadline3, time.time() + budget_s * 0.02)
                h_s, c5_s = _main_optimize(h_sin, sub_dl, beta_start=40.0, max_iters=400, use_lp=False)
                if c5_s < best_c5 - 1e-10:
                    best_c5 = c5_s
                    best_h = h_s.copy()

    # Phase 4: Pair polish
    if time.time() < t0 + budget_s * 0.90:
        deadline4 = t0 + budget_s * 0.92
        hp, c5p = _pair_polish(best_h, deadline4, max_passes=30)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 5: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.96:
        deadline5 = t0 + budget_s * 0.97
        for dn in [-3, -2, -1, 1, 2, 3, 4, 5, 6, 8, 10]:
            if time.time() > deadline5:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 160:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _main_optimize(h_r, min(deadline5, time.time() + budget_s * 0.02),
                                       beta_start=40.0, max_iters=600, use_lp=True)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final polish: tight LP + pair
    remaining = t0 + budget_s - time.time()
    if remaining > 5.0:
        h_f, c5_f = _main_optimize(best_h, t0 + budget_s * 0.99,
                                    beta_start=80.0, max_iters=2000, use_lp=True)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Last pair polish
    remaining = t0 + budget_s - time.time()
    if remaining > 3.0:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.998, max_passes=10)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
