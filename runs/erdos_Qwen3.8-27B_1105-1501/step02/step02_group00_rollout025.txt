import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(1, n):
        # lag d, index n-1+d
        G[n-1+d, :n-d] += 1.0 - h[d:]
        G[n-1+d, d:] += -h[:n-d]
        # lag -d, index n-1-d  
        G[n-1-d, d:] += 1.0 - h[:n-d]
        G[n-1-d, :n-d] += -h[d:]
    G[n-1, :] = 1.0 - 2.0 * h
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    # Coarse search
    alphas = np.linspace(0.0, alpha_max, 20)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        if a < 1e-10:
            continue
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    idx = int(best_a / alpha_max * 19)
    lo_a = max(0.0, alphas[max(0, idx-1)])
    hi_a = min(alpha_max, alphas[min(19, idx+1)])
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2; c2 = c1; f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1; c1 = c2; f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-10:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.8
            if stall % 10 == 0 and radius < 0.03:
                radius = 0.06
    
    return best_h, best_c5

def _binary_anneal(n, seed, n_iters=200000):
    """Simulated annealing on binary configurations."""
    rng = np.random.default_rng(seed)
    half = n // 2
    # Start with alternating pattern (good initial structure)
    h = np.zeros(n)
    h[::2] = 1.0
    # Ensure exactly half ones
    ones_count = int(h.sum())
    if ones_count > half:
        # Remove some ones
        one_idx = np.where(h == 1.0)[0]
        to_remove = rng.choice(one_idx, ones_count - half, replace=False)
        h[to_remove] = 0.0
    elif ones_count < half:
        zero_idx = np.where(h == 0.0)[0]
        to_add = rng.choice(zero_idx, half - ones_count, replace=False)
        h[to_add] = 1.0
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.05
    T_min = 0.0005
    cool_rate = (T_min / T) ** (1.0 / n_iters)
    
    for it in range(n_iters):
        T *= cool_rate
        if T < T_min:
            T = T_min
        
        # Pick random 1 and 0 to swap
        idx_one = rng.integers(0, n)
        tries = 0
        while h[idx_one] == 0 and tries < 100:
            idx_one = rng.integers(0, n)
            tries += 1
        if h[idx_one] == 0:
            continue
        idx_zero = rng.integers(0, n)
        tries = 0
        while h[idx_zero] == 1 and tries < 100:
            idx_zero = rng.integers(0, n)
            tries += 1
        if h[idx_zero] == 1:
            continue
        
        h[idx_one] = 0.0
        h[idx_zero] = 1.0
        new_c5 = _exact_c5(h)
        
        if new_c5 <= cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[idx_one] = 1.0
            h[idx_zero] = 0.0
    
    return best_h, best_c5

def _subgrad_descent(h, deadline, lr_init=0.02, top_k=5):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    momentum = np.zeros(n)
    
    while time.time() < deadline:
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_s = np.max(scores)
        
        # Get top-k indices
        top_idx = np.argsort(scores)[-top_k:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in top_idx:
            d = idx - (n - 1)
            if d > 0:
                grad[:n-d] += 1.0
                grad[:n-d] += -h[d:]
                grad[d:] += -h[:n-d]
            elif d < 0:
                m = -d
                grad[m:] += (1.0 - h[:n-m])
                grad[:n-m] += -h[m:]
            else:
                grad += (1.0 - 2.0 * h)
        grad /= top_k
        
        # Momentum
        momentum = 0.9 * momentum + 0.1 * grad
        h_new = h - lr * momentum
        h_new = _project(h_new, n / 2.0)
        
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.01, 0.2)
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
    
    return best_h, best_c5
