    old = h[p]
    
    # Compute d[k] for all lags
    d = np.zeros(M)
    for k in range(M):

        ps = p + s
        pm = p - s
        A = (1.0 - h[ps]) if 0 <= ps < n else 0.0
        B = h[pm] if 0 <= pm < n else 0.0
        d[k] = A - B

    c = corr_old - d * old
    
    # Find x in [0,1] minimizing max_k (d[k]*x + c[k])
    # Candidates: 0, 1, and pairwise intersections in (0,1)
    
    # Pairwise intersections
    # x = (c[j] - c[i]) / (d[i] - d[j]) for d[i] != d[j]

    best_val = np.inf
    best_x = old  # default
    
    # Evaluate at x=0 and x=1
    f0 = np.max(c)
    f1 = np.max(d + c)

        best_x = 0.0
