- If s < 0: let t = -s, V_s = dx * sum_{i=t}^{n-1} h[i]*(1-h[i-t])

Let me simplify. For a given j and s:
- Let me define: the lag s correlation sums over i where both i and i+s are in [0, n).
- For s >= 0: i ranges from 0 to n-1-s
- For s < 0: i ranges from -s to n-1

The derivative dV_s/dh[j] = dx * (contribution to the "first factor" - contribution to the "second factor")

First factor: h[j] appears as h[i] when i=j, so we need j in the valid range for i.
Second factor: h[j] appears as h[i+s] when i+s=j, i.e., i=j-s, so we need j-s in the valid range for i.

- i=j-s is valid if 0 <= j-s <= n-1-s, i.e., s <= j <= n-1-s+s = n-1, so s <= j <= n-1

For s < 0 (t = -s > 0):

```python
import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[s:n]
        else:
            t = -s
            grad[0:L] += w * (1.0 - h[t:n])
            grad[t:n] -= w * h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    return w

def coord_slope(h, j):
    """Compute dV_s/dh[j] for all lags s (indexed 0 to 2n-2)."""
    n = len(h)
    dx = 2.0 / n
    a = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        val = 0.0
        if s >= 0:
            if j <= n - 1 - s:
                val += (1.0 - h[j + s])
            if s <= j:
                val -= h[j - s]
        else:
            t = -s
            if t <= j:
                val += (1.0 - h[j - t])
            if j <= n - 1 - t:
                val -= h[j + t]
        a[idx] = dx * val
    return a

def pair_coord_descent(h, max_sweeps=100, order=None):
    """Pair coordinate descent maintaining sum constraint."""
    n = len(h)
    h_cur = h.copy()
    best_f = official_score(h_cur)
    
    if order is None:
        order = list(range(n - 1))
    
    for sweep in range(max_sweeps):
        improved = False
        V = compute_values(h_cur)
        for j in order:
            j2 = j + 1
            s = h_cur[j] + h_cur[j2]
            a = coord_slope(h_cur, j)
            b = coord_slope(h_cur, j2)
            
            slope = a - b
            const = V - a * h_cur[j] + b * s - b * h_cur[j2]
            
            lo_x = max(0.0, s - 1.0)
            hi_x = min(1.0, s)
            
            if hi_x - lo_x < 1e-14:
                continue
            
            # Bisection on subgradient
            for _ in range(45):
                mid_x = (lo_x + hi_x) / 2.0
                vals = slope * mid_x + const
                m = vals.max()
                active = vals >= m - 1e-13
                if not active.any():
                    active = np.zeros(len(slope), dtype=bool)
                    active[np.argmax(vals)] = True
                sg_lo = slope[active].min()
                sg_hi = slope[active].max()
                if sg_lo > 1e-13:
                    hi_x = mid_x
                elif sg_hi < -1e-13:
                    lo_x = mid_x
                else:
                    break
            
            x_new = (lo_x + hi_x) / 2.0
            if abs(x_new - h_cur[j]) < 1e-14:
                continue
            
            old_j = h_cur[j]
            old_j2 = h_cur[j2]
            h_cur[j] = x_new
            h_cur[j2] = s - x_new
            
            f_new = official_score(h_cur)
            if f_new < best_f - 1e-14:
                best_f = f_new
                improved = True
                V = compute_values(h_cur)
            else:
                h_cur[j] = old_j
                h_cur[j2] = old_j2
        
        if not improved:
            break
    
    return h_cur, best_f

def pg_step(h, beta, alpha_max=0.8, max_ls=16):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = fast_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
