def solve_lp_direction(h, active_indices, trust_radius=0.05):
    n = len(h)
    offset = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = np.max(corr)
    
    G = build_active_gradients(h, active_indices)
    slacks = M - corr[active_indices]
    
    d = cp.Variable(n)
    theta = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        d >= -h,
        d <= 1 - h,
        theta >= 0
    ]
    if trust_radius < np.inf:
        constraints.append(d >= -trust_radius)
        constraints.append(d <= trust_radius)
    
    for r in range(len(active_indices)):
        constraints.append(G[r] @ d + theta <= slacks[r])
    
    prob = cp.Problem(cp.Maximize(theta), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return None
    
    if d.value is None or theta.value is None:
        return None
    if theta.value <= 0:
        return None
    
    d_val = d.value
    # Project back to feasibility
    d_val = np.clip(d_val, 0, 1)  # No, d is the direction, not h
    # Sum should be 0
    d_val -= np.mean(d_val)
    return d_val

    theta = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        d >= -h * 0.99,
        d <= (1 - h) * 0.99,
        d >= -trust_radius,
        d <= trust_radius,
        theta >= 0
    ]

        constraints.append(G[r] @ d + theta <= slacks[r])
    
    prob = cp.Problem(cp.Maximize(theta), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)
    except:

    if theta.value is None or theta.value <= 1e-10:
        return None, None

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(25):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

        d <= ub,
        theta >= 0

    except Exception:
