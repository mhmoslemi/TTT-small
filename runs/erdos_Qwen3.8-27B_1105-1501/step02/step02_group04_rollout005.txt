def compute_c5(h):

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0

def build_grad_rows(h, active_indices):
    """Build gradient rows for selected correlation output indices."""
    n = len(h)
    v = 1.0 - h
    offset = n - 1
    G = np.zeros((len(active_indices), n))
    for r, p in enumerate(active_indices):
        s = p - offset  # shift
        # For valid n: 0 <= n < n and 0 <= n+s < n
        n_lo = max(0, -s)
        n_hi = min(n - 1, n - 1 - s)
        if n_lo > n_hi:

        G[r, idxs] -= h[idxs + s]  # -h[n+s] for each term
    return G

    """ LP-based active-lag direction step """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    
    # Select active lags: those within margin of max
    active_mask = corr >= max_corr - margin
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        return None
    # If too many, take top K
    if len(active_indices) > K:
        top_vals = np.sort(corr)[::-1][:K]
        threshold = top_vals[-1]
        active_indices = np.where(corr >= threshold - 1e-12)[0]
    
    active_indices = list(active_indices)
    K_actual = len(active_indices)
    
    # Build gradient matrix
    G = build_grad_rows(h, active_indices)
    
    # Slacks
    slacks = max_corr - corr[active_indices]
    
    # LP: max theta s.t. G[l] @ d + theta <= slacks[l], sum(d) = 0, lb <= d <= ub, theta >= 0
    # Variables: d[0..n-1], theta
    # Total variables: n + 1
    # Objective: minimize -theta (maximize theta)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0  # minimize -theta
    
    # Inequality constraints: G[l] @ d + theta <= slacks[l]
    # A_ub @ x <= b_ub
    A_ub = np.zeros((K_actual, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0  # theta coefficient
    b_ub = slacks.copy()
    
    # Equality constraints: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: for d[i], lb = max(-h[i], -trust_radius), ub = min(1-h[i], trust_radius)
    # For theta: lb = 0, ub = None (or max slack)
    lb = np.maximum(-h, -trust_radius)
    ub = np.minimum(1.0 - h, trust_radius)
    lb = np.concatenate([lb, [0.0]])
    ub = np.concatenate([ub, [max_corr * 0.5]])  # cap theta
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lb, ub)), method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    if np.abs(d).max() < 1e-10:
        return None
    
    return d

import time
from scipy.optimize import linprog


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def build_grad_rows(h, active_indices):
    n = len(h)
    v = 1.0 - h
    offset = n - 1
    G = np.zeros((len(active_indices), n))
    for r, p in enumerate(active_indices):
        k = p - offset
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        if n_lo > n_hi:
            continue
        idxs = np.arange(n_lo, n_hi + 1)
        first_vars = idxs + k
        G[r, first_vars] += v[idxs]
        G[r, idxs] -= h[idxs + k]
    return G

    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    
    active_mask = corr >= max_corr - margin

    if len(active_indices) == 0:
        return None
    if len(active_indices) > K:
        sorted_corr = np.sort(corr)[::-1]
        threshold = sorted_corr[K - 1]
        active_indices = np.where(corr >= threshold - 1e-14)[0]
    
    active_indices = list(active_indices)
    n_act = len(active_indices)
    
    G = build_grad_rows(h, active_indices)
    slacks = max_corr - corr[np.array(active_indices)]
    
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[-1] = -1.0
    
    A_ub = np.zeros((n_act, nvar))

    A_eq = np.zeros((1, nvar))

    return d


def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)

    a, b = lo, hi

def full_pair_sweep(h, rng, max_pairs=None):
    n = len(h)
    if max_pairs is None:
        max_pairs = n * (n - 1) // 2
    perm = rng.permutation(n)

            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
            count += 1
