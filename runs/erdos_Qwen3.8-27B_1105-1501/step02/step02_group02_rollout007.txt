The pairwise coordinate optimization is the key new idea. Here's how it works:

For the current best h with score C:
- For each pair (j, k) where h[j] > 0.5 and h[k] < 0.5:
  - Define h(δ) = h + δ(e_j - e_k) (transfer δ from j to k)
  - Forward range: δ ∈ [-h[j], 1-h[k]]
  - Find δ* that minimizes max_l F_l(h(δ))
  - This is a 1D piecewise-quadratic minimization
  - Approximate with golden section or direct evaluation
- Accept the best improvement

Since each F_l is quadratic in h (more precisely, it's a sum of products of pairs), F_l(h(δ)) is a quadratic in δ:
F_l(h + δ(e_j - e_k)) = a_l * δ^2 + b_l * δ + c_l

where a_l, b_l, c_l can be computed from the current h.

Then we want to minimize max_l (a_l δ^2 + b_l δ + c_l) over δ in the feasible range.

This is a 1D minimax problem with quadratic pieces. We can solve it by:
1. Evaluating at a grid of δ values
2. Finding the grid points where the max is minimized (roughly)
3. Refining with golden section in the best interval

Or more simply: since n is small (57), we can just evaluate on a fine grid of, say, 100 points and take the best.

For each pair and each grid point, evaluating all F_l takes O(n^2) (for the correlation). With ~100 pairs * 100 grid points * O(n^2) = 10000 * 3249 ≈ 32M operations. That's a bit much per iteration.

Better: precompute the correlation at the current point, then for each (pair, δ), compute the change in each F_l analytically (in O(n) per pair since only 2 coordinates change).

Actually, when we change h[j] → h[j]+δ and h[k] → h[k]-δ, the only correlations that change are those involving index j or index k. For index j, it's involved in correlations with lags s such that j+s is in [0,n) or j-s is in [0,n). That's at most 2n lags (but typically much fewer for a given j).

So for each pair (j,k), we can compute the effect on all F_l in O(n) time (since j is involved in at most 2 lags on each side, and same for k, so at most ~4n lags are affected).

Then for a grid of δ values, we can compute max_l F_l(h(δ)) in O(n) per δ (just update the affected lags and take the max).

Total per pair: O(n^2) for the grid. With ~100 pairs: O(n^3) = O(57^3) ≈ 185K. That's very fast!

```python
import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _grad_lags(x, active_idx):
    """Compute weighted subgradient from active lags."""
    n = len(x)
    dx = 2.0 / n
    cv = _corr_vec(x)
    vals = cv[active_idx]
    max_val = vals.max()
    w = np.exp(30.0 * (vals - max_val))
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(len(active_idx), 1.0 / len(active_idx))
    else:
        w = w / ws
    g = np.zeros(n)
    for ii, k in enumerate(active_idx):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - x[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            g[lo2:hi2] += wk * (-x[lo2 - s:hi2 - s])
    return g

def _subgrad_step(h, beta=50, n_iter=5000, alpha0=0.005, decay=0.9999):
    """Direct subgradient method with adaptive tracking."""
    n = len(h)
    cur = h.copy()
    best = h.copy()
    best_s = _score(cur)
    alpha = alpha0
    for it in range(n_iter):
        cv = _corr_vec(cur)
        max_idx = np.argmax(cv)
        # Get top lags
        top_m = min(8, len(cv))
        active = np.argsort(cv)[::-1][:top_m]
        g = _grad_lags(cur, active)
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        # Project the step
        new = cur - alpha * g
        new = _project(new)
        s_new = _score(new)
        if s_new < best_s:
            best_s = s_new
            best = new.copy()
        if s_new <= _score(cur) + 0.5 * alpha:
            cur = new
        else:
            alpha *= 0.7
        alpha = max(alpha * decay, 1e-6)
    return best, best_s

def _pairwise_optimize(h, max_pairs=50, n_grid=80, max_rounds=5):
    """Optimize by pairwise coordinate transfers."""
    n = len(h)
    cur = h.copy()
    cur_s = _score(cur)
    for rnd in range(max_rounds):
        improved = False
        # Sort coordinates: high ones and low ones
        order = np.argsort(cur)
        low_idx = order[:n//2]
        high_idx = order[n//2:][::-1]
        best_delta_s = cur_s
        best_pair = None
        best_delta = 0
        # Try transfers from high to low
        tried = 0
        for ji in high_idx[:20]:
            for ki in low_idx[:10]:
                if cur[ji] - cur[ki] < 0.1:
                    continue
                tried += 1
                if tried > max_pairs:
                    break
                # Try transferring from ji to ki
                # Forward range: delta in [-cur[ji], 1-cur[ki]]
                # We want to try positive delta (transfer ji→ki)
                lo = 0.0
                hi = min(cur[ji], 1.0 - cur[ki])
                if hi <= 0:
                    continue
                # Evaluate on a grid
                deltas = np.linspace(0, hi, n_grid + 1)
                best_d_local = 0
                best_s_local = cur_s
                for d in deltas:
                    if d == 0:
                        continue
                    h_test = cur.copy()
                    h_test[ji] += d
                    h_test[ki] -= d
                    s_test = _score(h_test)
                    if s_test < best_s_local:
                        best_s_local = s_test
                        best_d_local = d
                if best_s_local < best_delta_s:
                    best_delta_s = best_s_local
                    best_pair = (ji, ki)
                    best_delta = best_d_local
            if tried > max_pairs:
                break
        if best_pair is not None and best_delta_s < cur_s - 1e-12:
            ji, ki = best_pair
            cur[ji] += best_delta
            cur[ki] -= best_delta
            cur = _project(cur)
            cur_s = _score(cur)
            improved = True
        if not improved:
            break
    return cur, cur_s

def _slsqp_smooth(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        es = np.sum(np.exp(sh))
        if es < 1e-300:
            return mc
        return mc + np.log(es) / beta
    
    def _obj_grad(x):
        cv = _corr_vec(x)
        mc = cv.max()
        sh = np.clip(beta * (cv - mc), -500, 0)
        ev = np.exp(sh)
        es = ev.sum()
        if es < 1e-300:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = ev / es
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            lo1 = max(0, -s)
            hi1 = min(n, n - s)
            if lo1 < hi1:
                g[lo1:hi1] += wk * (1.0 - x[lo1 + s:hi1 + s])
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            if lo2 < hi2:
                g[lo2:hi2] += wk * (-x[lo2 - s:hi2 - s])
        return g
    
    def _efun(x):
        return x.sum() - n / 2.0
    
    def _ejac(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_obj_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_direct(h0, maxiter=100):
    n = len(h0)
    cv = _corr_vec(h0)
    t0 = min(cv.max() + 0.01, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def _fun(z):
        return z[-1]
    
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr_vec(h)
        return t - cv
    
    def _ieq_jac(z):
        h = z[:-1]
        n = len(h)
        dx = 2.0 / n
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        for k in range(m):
            s = k - (n - 1)
            lo1 = max(0, -s)
            hi1 = min(n, n - s)
            if lo1 < hi1:
                J[k, lo1:hi1] += dx * (1.0 - h[lo1 + s:hi1 + s])
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            if lo2 < hi2:
                J[k, lo2:hi2] += dx * (-h[lo2 - s:hi2 - s])
            J[k, -1] = 1.0
        return J
    
    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0
    
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq, 'jac': _ieq_jac},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))
