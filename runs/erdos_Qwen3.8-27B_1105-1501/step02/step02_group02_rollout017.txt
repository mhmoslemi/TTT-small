```python
def _lse_grad(x, beta):
    n = len(x)
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        s = (n - 1) - j  # mathematical shift
        g += w[j] * _grad_F_s(x, s)
    return g
```
