    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_f_p(h, p):
    """Gradient of the p-th correlation value (times dx) with respect to h."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        if s == 0:
            g[j] = 1.0 - 2.0 * h[j]
        else:
            if 0 <= j + s < n:
                g[j] += -h[j + s]
            if 0 <= j - s < n:
                g[j] += (1.0 - h[j - s])
    return dx * g

def make_feasible(z):

    target = n / 2.0

    z = np.clip(z, -1.0, 2.0)

        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)

    """Solve LP to find a descent direction for the max correlation."""
    n = len(h0)
    f = compute_f(h0)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        h_var >= h0 - step_size,
        h_var <= h0 + step_size,
    ]
    
    for p in active_indices:
        g = gradient_f_p(h0, p)
        constraints.append(f[p] + g @ (h_var - h0) <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT, cp.PDLP]:
        try:

            if prob.status in ['optimal', 'optimal_inaccurate']:

        except Exception:
            continue
    return None

def active_set_lp(h0, max_iter=50, k_active=5, step_sizes=None):

    n = len(h0)
    if step_sizes is None:
        step_sizes = np.geomspace(0.5, 0.02, max_iter)
    
    h = h0.copy()

    stall = 0

        f = compute_f(h)
        top_idx = np.argsort(f)[-k_active:]
        step = step_sizes[min(it, len(step_sizes)-1)]

                    best = sc_try
                    accepted = True
                    break

def binary_elp(n, max_lag=12, time_limit=60):
    """Solve binary ILP to find good structural patterns."""
    x = cp.Variable(n, boolean=True)
    M = cp.Variable()
    
    constraints = [cp.sum(x) == n // 2]
    
    for t in range(1, max_lag + 1):
        # c_t = sum of (1-x[i])*x[i+t] for i in 0..n-1-t
        y = cp.Variable(n - t)
        constraints += [y <= 1 - x[:n-t]]
        constraints += [y <= x[t:]]
        constraints += [y >= x[t:] - x[:n-t]]
        constraints += [y >= 0]
        constraints.append(cp.sum(y) <= M)
        
        # b_t = sum of x[i]*(1-x[i+t])
        z = cp.Variable(n - t)
        constraints += [z <= x[:n-t]]
        constraints += [z <= 1 - x[t:]]
        constraints += [z >= x[:n-t] - x[t:]]
        constraints += [z >= 0]
        constraints.append(cp.sum(z) <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CBC, cp.SCIP, cp.CPLEX]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
                h_bin = np.round(x.value).astype(float)
                h_cont = make_feasible(h_bin)
                return h_cont
        except Exception:
            continue
    return None

def gradient_f_p(h, p):
    n = len(h)

            if 0 <= j + s < n:

                g[j] += (1.0 - h[j - s])
    return dx * g

    n = len(h)
    dx = 2.0 / n

    shifted = np.clip(shifted, -80, 80)

            parent = make_feasible(parent)
            ps = official_score(parent)
            best_h = parent.copy()
            best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.2 * np.sin(2 * np.pi * x / 2.0))
        best_score = official_score(best_h)

    n = len(best_h)

    # Phase 1: Primary LP optimization from parent
    if remaining() > 120:
        h_opt, sc_opt = active_set_lp(best_h, max_iter=50, k_active=5,
                                      step_start=0.3, step_end=0.003, memory_depth=3)
        update(h_opt, sc_opt)

    # Phase 2: QP complement
    if remaining() > 80:
        h_opt, sc_opt = active_set_qp(best_h, max_iter=30, k_active=5,
                                      reg_start=1.0, reg_end=0.002)
        update(h_opt, sc_opt)

    # Phase 3: Different active set sizes with LP
    for k_a in [3, 7, 10, 15]:
        if remaining() < 50:
            break
        h_opt, sc_opt = active_set_lp(best_h, max_iter=35, k_active=k_a,
                                      step_start=0.2, step_end=0.005, memory_depth=2)
        update(h_opt, sc_opt)

    # Phase 4: Binary ILP for structural patterns
    if remaining() > 150:
        h_bin = binary_ilp(n, max_lag=12, time_limit=min(100, int(remaining() * 0.4)))
        if h_bin is not None:
            h_opt, sc_opt = active_set_lp(h_bin, max_iter=35, k_active=5,
                                          step_start=0.4, step_end=0.005)
            update(h_opt, sc_opt)
            h_opt, sc_opt = active_set_qp(h_opt if h_opt is not None else h_bin,
                                          max_iter=20, k_active=5, reg_start=0.5, reg_end=0.002)
            update(h_opt, sc_opt)

    # Phase 5: Golden-ratio binary pattern
    if remaining() > 60:
        h_gold = golden_binary(n, n // 2)
        h_opt, sc_opt = active_set_lp(h_gold, max_iter=30, k_active=5,
                                      step_start=0.4, step_end=0.01)
        update(h_opt, sc_opt)

    # Phase 6: Random structured starts
    for trial in range(6):
        if remaining() < 50:
            break
        h_struct = np.zeros(n)
        offset = int(rng.integers(0, n))
        stride = int(rng.integers(n // 4, n // 2 + 1))
        for i in range(n // 2):
            h_struct[(offset + i * stride) % n] = 1.0
        h_cand = make_feasible(h_struct)
        h_opt, sc_opt = active_set_lp(h_cand, max_iter=25, k_active=5,
                                      step_start=0.4, step_end=0.02)
        update(h_opt, sc_opt)

    # Phase 7: Resolution sweep
    if remaining() > 100:
        n_base = len(best_h)
        n_opts = sorted(set([max(20, n_base - 6), n_base - 3, n_base,
                            n_base + 3, n_base + 7, n_base + 12]))
        for n_opt in n_opts:
            if remaining() < 30 or n_opt < 20 or n_opt > 100:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            h_opt, sc_opt = active_set_lp(h_int, max_iter=25, k_active=5,
                                          step_start=0.25, step_end=0.005)
            update(h_opt, sc_opt)
            if remaining() > 20:
                h_opt2, sc_opt2 = active_set_qp(h_opt if h_opt is not None else h_int,
                                                max_iter=15, k_active=5,
                                                reg_start=0.5, reg_end=0.003)
                update(h_opt2, sc_opt2)

    # Phase 8: Intensive small-step LP polish
    if remaining() > 50:
        h_opt, sc_opt = active_set_lp(best_h, max_iter=40, k_active=5,
                                      step_start=0.05, step_end=0.001, memory_depth=4)
        update(h_opt, sc_opt)

    # Phase 9: Alternating sub-set LP
    if remaining() > 40:
        h_opt, sc_opt = alternating_lp(best_h, max_iter=40, k_active=8, step=0.02)
        update(h_opt, sc_opt)

    # Phase 10: SLSQP smooth-max refinement
    if remaining() > 30:
        for beta in [500, 2000, 10000]:
            if remaining() < 15:
                break
            h_new, sc = try_slsqp(best_h, beta, maxiter=80)
            if h_new is not None:
                update(h_new, sc)

    # Phase 11: Final combined polish
    if remaining() > 30:
        for k_a in [3, 5, 8]:
            if remaining() < 10:
                break
            h_opt, sc_opt = active_set_lp(best_h, max_iter=25, k_active=k_a,
                                          step_start=0.03, step_end=0.0005, memory_depth=3)
            update(h_opt, sc_opt)

    # Phase 12: One more QP pass
    if remaining() > 20:
        h_opt, sc_opt = active_set_qp(best_h, max_iter=20, k_active=5,
                                      reg_start=0.3, reg_end=0.001)
        update(h_opt, sc_opt)

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
