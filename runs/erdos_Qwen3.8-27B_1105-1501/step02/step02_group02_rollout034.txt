import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _optimize_pair(h, i, j):
    n = len(h)
    c = h[i] + h[j]
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    if t_hi - t_lo < 1e-14:
        return h
    h[i] = t_lo
    h[j] = c - t_lo
    corr_lo = np.correlate(h, 1.0 - h, mode='full')
    h[i] = t_hi
    h[j] = c - t_hi
    corr_hi = np.correlate(h, 1.0 - h, mode='full')
    delta = t_hi - t_lo
    slopes = (corr_hi - corr_lo) / delta
    candidates = [t_lo, t_hi]
    lo_b, hi_b = t_lo, t_hi
    for _ in range(35):
        mid = (lo_b + hi_b) / 2.0
        v = corr_lo + (mid - t_lo) * slopes
        g = v.max()
        active = v >= g - 1e-10
        ld = slopes[active].min()
        rd = slopes[active].max()
        if ld > 1e-12:
            hi_b = mid
        elif rd < -1e-12:
            lo_b = mid
        else:
            lo_b = hi_b = mid
            break
    candidates.append((lo_b + hi_b) / 2.0)
    t_v = c / 2.0
    if t_lo < t_v < t_hi:
        candidates.append(t_v)
    best_t = t_lo
    best_g = np.inf
    for t in candidates:
        t = max(t_lo, min(t_hi, t))
        h[i] = t
        h[j] = c - t
        g = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if g < best_g:
            best_g = g
            best_t = t
    h[i] = best_t
    h[j] = c - best_t
    return h

def _pairwise_cd(h, max_sweeps=500, tol=1e-12, rng=None):
    n = len(h)
    h = h.copy()
    best_s = _score(h)
    half = n // 2
    for sweep in range(max_sweeps):
        for i in range(0, n - 1, 2):
            _optimize_pair(h, i, i + 1)
        for i in range(1, n - 1, 2):
            _optimize_pair(h, i, i + 1)
        _optimize_pair(h, n - 1, 0)
        for i in range(half):
            _optimize_pair(h, i, (i + half) % n)
        if rng is not None and sweep % 3 == 0:
            for _ in range(half):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    _optimize_pair(h, i, j)
        cur_s = _score(h)
        if cur_s < best_s:
            best_s = cur_s
        if best_s - cur_s < tol and sweep > 50:
            break
    return h, best_s

def _lse_obj(x, beta):
    cv = np.correlate(x, 1.0 - x, mode='full')
    n = len(x)
    dx = 2.0 / n
    cv = cv * dx
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for p in range(j0, j1):
            g[p] += wk * (1.0 - x[p - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for p in range(p0, p1):
            g[p] += wk * (-x[p + s])
    return g

def _slsqp_smooth(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def _efun(x):
        return x.sum() - n / 2.0
    def _ejac(x):
        return np.ones(n)
    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _slsqp_direct(h0, maxiter=150):
    n = len(h0)
    cv = np.correlate(h0, 1.0 - h0, mode='full') * (2.0 / n)
    t0 = min(cv.max() + 0.01, 0.99)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def _fun(z):
        return z[-1]
    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
        return t - cv
    def _eq(z):
        return z[:-1].sum() - len(z[:-1]) / 2.0
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': _ieq},
                                    {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _fista_subgrad(h, beta, n_iter, alpha_init=0.005, mu=0.88):
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    y = h.copy()
    h_avg = h.copy()
    best_h = h.copy()
    best_s = _score(h)
    t = 1
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        h_new = y - alpha_init * grad
        h_new = _project(h_new)
        t += 1
        t_prev = t - 1
        t_new = 1.5 * t
        y = h_new + (t_prev / t_new) * (h_new - h_prev)
        y = _project(y)
        h_avg += (h_new - h_avg) / t
        h_prev = h_cur
        h_cur = h_new
        if it % 200 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 500 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _binary_start(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    idx = rng.permutation(n)
    h = h[idx]
    return _project(h)

def _periodic_start(n, freq, rng):
    t = np.arange(n) / n
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * t + phase)
    return _project(h)

def _polish(h, deadline):
