def official_score(h):

    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    # Ensure bracket
    while np.sum(np.clip(z + lo, 0, 1)) > target:
        lo -= 1.0
    while np.sum(np.clip(z + hi, 0, 1)) < target:
        hi += 1.0
    for _ in range(80):

        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

    Minimize max_i (A_i*t^2 + B_i*t + C_i) over t in [t_lo, t_hi].

    Returns optimal t and the optimal value.

        f_lo = t_lo
        f_hi = t_hi

            d = c - L_mid  # constant term

                # Linear: b*t + d ≤ 0
                if abs(b) < 1e-15:
                    if d > 0:
                        feasible = False
                        break

                elif b > 0:

                    if upper < f_hi:
                        f_hi = upper
                else:
                    # t ≥ -d/b
                    lower = -d / b
                    if lower > f_lo:
                        f_lo = lower
            elif a > 0:

                disc = b * b - 4 * a * d
                if disc < 0:
                    feasible = False
                    break
                sqrt_disc = np.sqrt(disc)

                if r1 > f_lo:
                    f_lo = r1
                if r2 < f_hi:
                    f_hi = r2
            else:

                disc = b * b - 4 * a * d

                    # No constraint
                    pass
                else:

                    r1 = (-b - sqrt_disc) / (2 * a)  # since a<0, this is the larger root
                    r2 = (-b + sqrt_disc) / (2 * a)  # smaller root
                    # Feasible: t ≤ r2 or t ≥ r1
                    forbidden.append((r2, r1))

                feasible = False
                break

            L_hi = L_mid
            continue

                    break
            if ok:

                break

    f_hi = t_hi

    for p in candidates:

        if v < best_val:

            best_t = p
