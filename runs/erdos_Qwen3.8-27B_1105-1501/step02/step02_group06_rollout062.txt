def refine_kink(A, B, t_approx, n_corr):
    """Given approximate t, find exact kink minimum."""
    val = (A + B * t_approx).max()
    active = np.where(np.abs(A + B * t_approx - val) < 1e-9)[0]
    if len(active) < 2:
        return t_approx
    # Find the pair in active whose intersection is closest to t_approx
    # and is a valid kink (B values straddle)
    B_active = B[active]
    A_active = A[active]
    # Intersection of lines p1 and p2: t = (A[p2]-A[p1])/(B[p1]-B[p2])
    best_t = t_approx
    best_val = val
    for i in range(len(active)):
        for k in range(i+1, len(active)):
            if abs(B_active[i] - B_active[k]) < 1e-15:
                continue
            t_cross = (A_active[k] - A_active[i]) / (B_active[i] - B_active[k])
            if 0 <= t_cross <= 1:
                v = (A + B * t_cross).max()
                if v < best_val - 1e-12:
                    best_val = v
                    best_t = t_cross
    return best_t

import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return h

def compute_gradient_matrix(h):
    n = len(h)
    n_corr = 2 * n - 1
    Grad = np.zeros((n_corr, n))
    for p in range(n_corr):
        s = p - (n - 1)
        if s >= 0:
            idx1 = np.arange(s, n)
            Grad[p, idx1] += 1.0 - h[idx1 - s]
            idx2 = np.arange(0, n - s)
            Grad[p, idx2] -= h[idx2 + s]
        else:
            t = -s
            idx1 = np.arange(0, n - t)
            Grad[p, idx1] += 1.0 - h[idx1 + t]
            idx2 = np.arange(t, n)
            Grad[p, idx2] -= h[idx2 - t]
    return Grad

def pair_descent_sweep(h, n_grid=200, max_rounds=50, tol=1e-12):
    n = len(h)
    n_corr = 2 * n - 1
    dx = 2.0 / n
    h = h.copy()
    best = official_score(h)
    c_raw = compute_f_raw(h)
    Grad = compute_gradient_matrix(h)
    
    for rnd in range(max_rounds):
        improved = False
        best_round_improve = 0.0
        best_update = None
        
        for j in range(n):
            B_col = Grad[:, j]
            A_base = c_raw - B_col * h[j]
            
            B_mat = B_col[None, :] - Grad.T
            A_mat = A_base[None, :] + Grad.T * h[j]
            
            t_grid = np.linspace(0.0, 1.0, n_grid)
            C_mat = np.max(A_mat[:, :, None] + B_mat[:, :, None] * t_grid[None, None, :], axis=1) * dx
            
            # Clamp t to valid range for each j2
            c0_vals = h[j] + h  # c0 = h[j] + h[j2] for each j2
            t_lo = np.maximum(0.0, c0_vals - 1.0)
            t_hi = np.minimum(1.0, c0_vals)
            
            best_idx = np.argmin(C_mat, axis=1)
            best_t = t_grid[best_idx]
            best_t = np.maximum(best_t, t_lo)
            best_t = np.minimum(best_t, t_hi)

                idx_fine = np.argmin(C_fine)

def weighted_topk_descent(h, K=5, n_grid=200, max_rounds=50, tol=1e-12):
    """Minimize weighted sum of top-K correlations instead of max."""
    n = len(h)
    n_corr = 2 * n - 1
    dx = 2.0 / n
    h = h.copy()
    best = official_score(h)
    
    for rnd in range(max_rounds):
        c_raw = compute_f_raw(h)
        c_norm = c_raw * dx
        Grad = compute_gradient_matrix(h)
        
        # Get top-K
        top_idx = np.argsort(c_norm)[-K:]
        c_top = c_norm[top_idx]
        Grad_top = Grad[top_idx, :]
        
        # Weights: emphasize the max
        w = np.exp(10.0 * (c_top - c_top.max() + 0.05))
        w /= w.sum()
        
        improved = False
        for j in range(n):
            B_col = Grad_top[:, j]
            A_base = c_top - B_col * h[j]
            B_mat = B_col[None, :] - Grad_top.T
            A_mat = A_base[None, :] + Grad_top.T * h[j]
            
            t_grid = np.linspace(0.0, 1.0, n_grid)
            # Weighted sum instead of max
            W_mat = (A_mat[:, :, None] + B_mat[:, :, None] * t_grid[None, None, :]) * w[None, None, :]
            S_mat = W_mat.sum(axis=1) * dx
            
            c0_vals = h[j] + h
            t_lo = np.maximum(0.0, c0_vals - 1.0)
            t_hi = np.minimum(1.0, c0_vals)
            
            best_idx = np.argmin(S_mat, axis=1)
            best_t = t_grid[best_idx]
            best_t = np.maximum(best_t, t_lo)
            best_t = np.minimum(best_t, t_hi)
            
            S_clamped = (A_mat + B_mat * best_t[:, None]) * w[None, :]
            S_clamped = S_clamped.sum(axis=1) * dx
            S_clamped[j] = np.inf
            best_j2 = np.argmin(S_clamped)
            
            t_star = best_t[best_j2]
            c0 = h[j] + h[best_j2]
            new_h = h.copy()
            new_h[j] = t_star
            new_h[best_j2] = c0 - t_star
            sc_new = official_score(new_h)
            
            if sc_new < best - tol:
                h = new_h
                best = sc_new
                improved = True
    
    return h, best

def smooth_obj_fn(h, beta, K):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max approximation
    top = np.sort(vals)[-K:]
    max_val = top.max()
    shifted = beta * (top - max_val)
    shifted = np.clip(shifted, -100, 100)
    return max_val + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def slsqp_refine(h0, beta=5000, K=5, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(smooth_obj_fn, h0, args=(beta, K), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

        top_idx = np.argsort(vals)[-k_active:]
        dx = 2.0 / n
        Grad = compute_gradient_matrix(h)
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        for p in top_idx:
            g = Grad[p, :] * dx
            constraints.append(vals[p] + g @ (h_var - h) <= t_var)
        
        obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h))
        prob = cp.Problem(obj, constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.CVXOPT]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        h_new = h_var.value
        if h_new is None or not np.all(np.isfinite(h_new)):
            break
        h_new = make_feasible(h_new)
        sc = official_score(h_new)
        
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.5, 0.25, 0.1]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-14:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 5:
                    break
    
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    def remaining():
        return deadline - time.time()
    
    best_h = None
    best_score = float('inf')
    
    def update(h, sc):
        nonlocal best_h, best_score
        if h is not None and np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
    
    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass
    
    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * x / 2.0) + 0.1 * np.sin(4 * np.pi * x / 2.0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)
    
    update(best_h, best_score)
    
    # Phase 1: Exact pair descent (main engine)
    if remaining() > 30:
        h1, s1 = pair_descent_sweep(best_h, n_grid=200, max_rounds=80, tol=1e-10)
        update(h1, s1)
    
    # Phase 2: More pair descent with finer grid
    if remaining() > 30:
        h2, s2 = pair_descent_sweep(best_h, n_grid=400, max_rounds=40, tol=1e-11)
        update(h2, s2)
    
    # Phase 3: Weighted top-K descent
    if remaining() > 30:
        for K in [3, 5, 8]:
            if remaining() < 15:
                break
            h3, s3 = weighted_topk_descent(best_h, K=K, n_grid=200, max_rounds=30, tol=1e-11)
            update(h3, s3)
    
    # Phase 4: Active-set QP polish
    if remaining() > 40:
        for ka in [4, 6, 8]:
            if remaining() < 20:
                break
            h4, s4 = active_set_qp(best_h, k_active=ka, max_iter=20, reg_start=0.5, reg_end=0.003)
            update(h4, s4)
    
    # Phase 5: SLSQP with increasing beta
    if remaining() > 40:
        for beta in [500, 2000, 8000, 20000]:
            if remaining() < 15:
                break
            h5, s5 = slsqp_refine(best_h, beta=beta, K=5, maxiter=150)
            update(h5, s5)
    
    # Phase 6: Pair descent again after SLSQP
    if remaining() > 30:
        h6, s6 = pair_descent_sweep(best_h, n_grid=300, max_rounds=50, tol=1e-11)
        update(h6, s6)
    
    # Phase 7: Resolution sweep
    if remaining() > 80:
        n_base = len(best_h)
        n_opts = sorted(set([
            max(20, n_base - 10),
            max(20, n_base - 5),
            n_base,
            min(80, n_base + 5),
            min(80, n_base + 10),
        ]))
        for n_opt in n_opts:
            if remaining() < 20:
                break
            if n_opt < 20 or n_opt % 2 != 0:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            h_opt, sc_opt = pair_descent_sweep(h_int, n_grid=200, max_rounds=40, tol=1e-10)
            update(h_opt, sc_opt)
            if remaining() > 20:
                h_qp, sc_qp = active_set_qp(h_opt, k_active=5, max_iter=15, reg_start=0.5, reg_end=0.005)
                update(h_qp, sc_qp)
    
    # Phase 8: Random perturbations + re-optimize
    if remaining() > 60:
        for i in range(20):
            if remaining() < 15:
                break
            sigma = 0.01 + 0.05 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = pair_descent_sweep(h_cand, n_grid=150, max_rounds=30, tol=1e-10)
            update(h_opt, sc_opt)
    
    # Phase 9: Final intensive pair descent
    if remaining() > 30:
        h9, s9 = pair_descent_sweep(best_h, n_grid=500, max_rounds=60, tol=1e-12)
        update(h9, s9)
    
    # Phase 10: Final QP + SLSQP
    if remaining() > 20:
        h_qp, sc_qp = active_set_qp(best_h, k_active=4, max_iter=15, reg_start=0.3, reg_end=0.001)
        update(h_qp, sc_qp)
    if remaining() > 15:
        h_sl, sc_sl = slsqp_refine(best_h, beta=50000, K=4, maxiter=100)
        update(h_sl, sc_sl)
    if remaining() > 15:
        h_fin, s_fin = pair_descent_sweep(best_h, n_grid=300, max_rounds=30, tol=1e-12)
        update(h_fin, s_fin)
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
