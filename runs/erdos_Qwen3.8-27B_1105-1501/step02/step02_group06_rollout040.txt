    n = len(h)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Adjust sum
    diff = target - np.sum(z)
    if abs(diff) < 1e-15:
        return z
    if diff > 0:
        # Need to add mass; add to elements with room
        room = 1.0 - z
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            add = min(remaining, room[idx])
            z[idx] += add
            remaining -= add
            if remaining < 1e-15:
                break
    else:
        # Need to remove mass
        order = np.argsort(z)
        remaining = -diff
        for idx in order:
            sub = min(remaining, z[idx])
            z[idx] -= sub
            remaining -= sub
            if remaining < 1e-15:
                break
    return np.clip(z, 0.0, 1.0)

def active_qp_step(h, reg, threshold=0.05, max_active=30):
    """One active-set QP step."""
    n = len(h)
    dx = 2.0 / n
    vals = compute_all_f(h)
    vmax = vals.max()
    
    # Select active constraints
    active_mask = vals >= vmax - threshold * vmax
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > max_active:
        # Keep top max_active
        sorted_active = np.argsort(-vals[active_idx])[:max_active]
        active_idx = active_idx[sorted_active]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for idx in active_idx:
        offset = idx - (n - 1)
        g = dx * gradient_f_fast(h, offset)
        constraints.append(vals[idx] + g @ (h_var - h) <= t_var)
    
    obj = cp.Minimize(t_var + (reg / 2.0) * cp.sum_squares(h_var - h))
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue

        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)
