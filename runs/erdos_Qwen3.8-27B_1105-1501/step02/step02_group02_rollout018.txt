    for it in range(n_iter):
        g = _lse_grad(y, beta)
        h_new = _project(y - alpha_init * g)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = h_new + ((t_val - 1.0) / t_new) * (h_new - h_cur)
        y = _project(y)
        h_prev = h_cur
        h_cur = h_new
        t_val = t_new
        h_sum += (h_cur - h_sum) / t_val
        if it % 200 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 1000 == 0 and it > 0:
            ha = _project(h_sum)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_sum)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _blur(h, kernel_width=3):
    n = len(h)
    k = np.exp(-0.5 * (np.arange(-kernel_width, kernel_width + 1) / 1.5) ** 2)
    k = k / k.sum()
    h_pad = np.pad(h, kernel_width, mode='reflect')
    return _project(np.convolve(h_pad, k, mode='valid'))

def _binary_push(h, p):
    return _project(h ** p)

def _find_active(h, n_top=12):
    cv = _corr(h)
    mc = cv.max()
    # Include lags within 5% of max
    threshold = mc * 0.95
    active = np.where(cv >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(cv)[::-1][:n_top]
    elif len(active) > 40:
        active = np.argsort(cv)[::-1][:n_top]
    return active

def _polish_coordinate(h, max_iter=200):
    """Coordinate-wise local optimization."""
    n = len(h)
    h = h.copy()
    cur_s = _score(h)
    for idx in range(max_iter):
        improved = False
        for j in range(n):
            cv = _corr(h)
            mc = cv.max()
            # Try small perturbations
            for delta in [0.01, -0.01, 0.005, -0.005]:
                h_test = h.copy()
                h_test[j] += delta
                h_test = _project(h_test)
                if h_test[j] == h[j]:
                    continue
                s_test = _score(h_test)
                if s_test < cur_s - 1e-12:
                    h = h_test
                    cur_s = s_test
                    improved = True
                    break
        if not improved:
            break
    return h, cur_s

def _beta_anneal_slsqp(h0, deadline, betas=None, maxiter_base=120):
    """Progressive beta annealing with SLSQP."""
    if betas is None:
        betas = [20, 50, 120, 300, 700, 1500]
    cur = h0.copy()
    cur_s = _score(cur)
    for beta in betas:
        if time.time() > deadline - 10:
            break
        remaining = max(15, deadline - time.time())
        iters = max(40, min(maxiter_base, int(remaining / 0.05)))
        new = _slsqp_lse(cur, beta, maxiter=iters)
        ns = _score(new)
        if ns < cur_s - 1e-14:
            cur, cur_s = new, ns
    return cur, cur_s

def _active_set_refine(h0, deadline, n_passes=3):
    """Iterative active-set refinement."""
    cur = h0.copy()
    cur_s = _score(cur)
    for pass_i in range(n_passes):
        if time.time() > deadline - 15:
            break
        active = _find_active(cur, n_top=15)
        # Try with regularization
        for rho in [0.0, 1e-4, 5e-4]:
            if time.time() > deadline - 10:
                break
            new = _slsqp_direct(cur, active, rho=rho, center=cur, maxiter=100)
            ns = _score(new)
            if ns < cur_s - 1e-14:
                cur, cur_s = new, ns
                active = _find_active(cur, n_top=15)
    return cur, cur_s

def _coarse_to_fine(h_start, n_target, deadline):
    """Optimize at coarse resolution first, then interpolate up."""
    n_cur = len(h_start)
    h = h_start.copy()
    s = _score(h)
    # Coarse
    if n_target > 30:
        n_coarse = max(16, n_target // 3)
        h_c = _interp(h, n_coarse)
        sc = _score(h_c)
        if time.time() < deadline - 30:
            h_c, sc = _beta_anneal_slsqp(h_c, deadline, betas=[20, 80, 300], maxiter_base=80)
        # Interpolate back
        h_up = _interp(h_c, n_target)
        s_up = _score(h_up)
        if s_up < s:
            h, s = h_up, s_up
        # Compare with coarse interpolated back
        h_c_up = _interp(h_c, n_target)
        s_c_up = _score(h_c_up)
        if s_c_up < s:
            h, s = h_c_up, s_c_up
    return h, s

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(40) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)

    # Phase 1: Optimize parent at its resolution
    if time.time() < deadline - 60:
        h_opt, s_opt = _beta_anneal_slsqp(parent, deadline,
                                           betas=[30, 80, 200, 500, 1200],
                                           maxiter_base=150)
        if s_opt < best_score - 1e-14:
            best_h, best_score, best_n = h_opt, s_opt, len(h_opt)

    # Phase 2: Active set refinement on best
    if time.time() < deadline - 40:
        h_opt, s_opt = _active_set_refine(best_h, deadline, n_passes=4)
        if s_opt < best_score - 1e-14:
            best_h, best_score, best_n = h_opt, s_opt, len(h_opt)

    # Phase 3: Try nearby resolutions
    n_tries = sorted(set([parent_n - 2, parent_n - 1, parent_n, parent_n + 1,
                          parent_n + 2, parent_n + 4, parent_n - 4]))
    n_tries = [n for n in n_tries if 24 <= n <= 100 and n != best_n]

    res_results = []
    for n in n_tries:
        if time.time() > deadline - 40:
            break
        h_interp = _interp(best_h, n)
        s0 = _score(h_interp)
        if time.time() < deadline - 30:
            h_opt, s_opt = _beta_anneal_slsqp(h_interp, deadline,
                                               betas=[50, 200, 600],
                                               maxiter_base=100)
            if s_opt < s0:
                res_results.append((s_opt, n, h_opt))
            else:
                res_results.append((s0, n, h_interp))
        else:
            res_results.append((s0, n, h_interp))
        sc = _score(res_results[-1][2])
        if sc < best_score - 1e-14:
            best_h, best_score, best_n = res_results[-1][2].copy(), sc, n

    # Phase 4: Focused optimization on top resolutions
    res_results.sort(key=lambda x: x[0])
    for s_r, n_r, h_r in res_results[:3]:
        if time.time() > deadline - 30:
            break
        # Active set refinement
        if time.time() < deadline - 25:
            h_opt, s_opt = _active_set_refine(h_r, deadline, n_passes=3)
            if s_opt < best_score - 1e-14:
                best_h, best_score, best_n = h_opt, s_opt, n_r
        # Additional SLSQP
        if time.time() < deadline - 20:
            h_opt2 = _slsqp_lse(best_h, beta=2000, maxiter=100)
            s_opt2 = _score(h_opt2)
            if s_opt2 < best_score - 1e-14:
                best_h, best_score = h_opt2, s_opt2

    # Phase 5: FISTA refinement on global best
    if time.time() < deadline - 30:
        remaining = deadline - time.time()
        n_iter = min(20000, int(remaining / 0.003))
        best_h, best_score = _fista(best_h, beta=500, n_iter=n_iter, alpha_init=0.004)
        best_n = len(best_h)

    # Phase 6: Diverse restarts
    if time.time() < deadline - 50:
        starts = []
        # Mirror
        starts.append(_mirror(best_h))
        # Blur
        starts.append(_blur(best_h, kernel_width=2))
        starts.append(_blur(best_h, kernel_width=4))
        # Binary push
        starts.append(_binary_push(best_h, 2.0))
        starts.append(_binary_push(best_h, 4.0))
        # Sinusoidal perturbation
        for freq in [2, 3, 5]:
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.05 * np.sin(2 * np.pi * freq * np.arange(best_n) / best_n + phase)
            starts.append(_project(best_h + pert))
        # Random
        for sig in [0.05, 0.1]:
            starts.append(_project(best_h + sig * rng.standard_normal(best_n)))

        # Coarse-to-fine
        if best_n > 30 and time.time() < deadline - 60:
            h_cf, s_cf = _coarse_to_fine(best_h, best_n, deadline)
            starts.append(h_cf)

        # Symmetrize
        h_sym = _project((best_h + best_h[::-1]) / 2.0)
        starts.append(h_sym)

        for h0 in starts:
            if time.time() > deadline - 25:
                break
            # Quick optimization
            if time.time() < deadline - 20:
                h_opt, s_opt = _beta_anneal_slsqp(h0, deadline,
                                                   betas=[100, 400, 1000],
                                                   maxiter_base=80)
            else:
                h_opt, s_opt = h0, _score(h0)
            if s_opt < best_score - 1e-14:
                best_h, best_score, best_n = h_opt, s_opt, len(h_opt)
            else:
                # Try active set on this start
                if time.time() < deadline - 15:
                    h_opt2, s_opt2 = _active_set_refine(h_opt, deadline, n_passes=2)
                    if s_opt2 < best_score - 1e-14:
                        best_h, best_score, best_n = h_opt2, s_opt2, len(h_opt2)

    # Phase 7: Intensive polishing
    if time.time() < deadline - 20:
        # Final LSE with high beta
        h_pol = _slsqp_lse(best_h, beta=3000, maxiter=150)
        s_pol = _score(h_pol)
        if s_pol < best_score - 1e-14:
            best_h, best_score = h_pol, s_pol

    if time.time() < deadline - 15:
        # Active set final
        active = _find_active(best_h, n_top=20)
        for rho in [0.0, 1e-5, 5e-5]:
            h_pol = _slsqp_direct(best_h, active, rho=rho, center=best_h, maxiter=120)
            s_pol = _score(h_pol)
            if s_pol < best_score - 1e-14:
                best_h, best_score = h_pol, s_pol

    if time.time() < deadline - 10:
        # Coordinate polish
        best_h, best_score = _polish_coordinate(best_h, max_iter=100)

    # Final FISTA
    if time.time() < deadline - 5:
        best_h, best_score = _fista(best_h, beta=2000, n_iter=3000, alpha_init=0.002)

    # Final feasibility and score
    best_h = _project(best_h)
    final_score = _score(best_h)
    if final_score < best_score:
        best_score = final_score

    return (best_h, final_score, len(best_h))
