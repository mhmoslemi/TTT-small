def symmetrize(h):
    h_sym = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_sym)

def z_to_h_sym(z, N):
    M = N // 2
    u = 1.0 / (1.0 + np.exp(-np.clip(z, -50.0, 50.0)))
    h = np.empty(N)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    return h

def h_to_z_sym(h, N):
    M = N // 2
    u = h[:M].copy()
    u = np.clip(u, 1e-8, 1.0 - 1e-8)
    return np.log(u / (1.0 - u))

def compute_c_values_sym(z, N):
    h = z_to_h_sym(z, N)
    dx = 2.0 / N
    vals = np.zeros(N)
    omh = 1.0 - h
    for k in range(N):
        L = N - k
        vals[k] = dx * np.dot(h[k:k + L], omh[:L])
    return vals

def compute_c_values_full(h):
    N = len(h)
    dx = 2.0 / N
    vals = np.zeros(N)
    omh = 1.0 - h
    for k in range(N):
        L = N - k
        vals[k] = dx * np.dot(h[k:k + L], omh[:L])
    return vals

def compute_grad_h_for_k(h, k, N, dx):
    L = N - k
    grad = np.zeros(N)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        grad[k:] += dx * (1.0 - h[:L])
        grad[:L] -= dx * h[k:k + L]
    return grad

def compute_grad_z_sym(z, N, active_k):
    h = z_to_h_sym(z, N)
    dx = 2.0 / N
    M = N // 2
    u = h[:M]
    su = u * (1.0 - u)
    n_act = len(active_k)
    Gz = np.zeros((n_act, M))
    h_rev = h[N - 1: N - 1 - M: -1]
    for idx, k in enumerate(active_k):
        gh = compute_grad_h_for_k(h, k, N, dx)
        Gz[idx, :] = su * (gh[:M] - gh[N - 1: N - 1 - M: -1])
    return Gz

def smooth_obj_grad_sym(z, N, beta):
    vals = compute_c_values_sym(z, N)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    max_sh = shifted.max()
    logsum = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    obj = cmax + logsum / beta
    w = np.exp(shifted - max_sh)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(N) / N
    else:
        w /= ws
    Gz = compute_grad_z_sym(z, N, np.arange(N))
    grad = w @ Gz
    return obj, grad

def find_active_k(vals, top_frac=0.35, min_count=12, thresh_frac=0.005):
    N = len(vals)
    cmax = vals.max()
    thresh = cmax - max(thresh_frac * cmax, 1e-6)
    active = set(np.where(vals >= thresh)[0].tolist())
    top_k = max(min_count, int(top_frac * N))
    top_idx = np.argsort(vals)[::-1][:min(top_k, N)]
    active.update(top_idx.tolist())
    active.add(0)
    return sorted(active)

def interpolate_h(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

class SmoothProblem:
    def __init__(self, N, beta):
        self.N = N
        self.beta = beta

    def fun(self, z):
        return smooth_obj_grad_sym(z, self.N, self.beta)[0]

    def jac(self, z):
        return smooth_obj_grad_sym(z, self.N, self.beta)[1]

class ActiveSetSym:
    def __init__(self, N, active_k):
        self.N = N
        self.M = N // 2
        self.active_k = np.array(active_k)

    def fun(self, x):
        return x[-1]

    def jac(self, x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def con_fun(self, x):
        z = x[:-1]
        t = x[-1]
        h = z_to_h_sym(z, self.N)
        dx = 2.0 / self.N
        omh = 1.0 - h
        vals = np.zeros(len(self.active_k))
        for idx, k in enumerate(self.active_k):
            L = self.N - k
            vals[idx] = dx * np.dot(h[k:k + L], omh[:L])
        return t - vals

    def con_jac(self, x):
        z = x[:-1]
        Gz = compute_grad_z_sym(z, self.N, self.active_k)
        J = np.zeros((len(self.active_k), len(x)))
        J[:, :-1] = -Gz
        J[:, -1] = 1.0
        return J

class ActiveSetFull:
    def __init__(self, N, active_k):
        self.N = N
        self.active_k = np.array(active_k)

    def fun(self, x):
        return x[-1]

    def jac(self, x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def con_fun(self, x):
        h = x[:-1]
        t = x[-1]
        dx = 2.0 / self.N
        omh = 1.0 - h
        vals = np.zeros(len(self.active_k))
        for idx, k in enumerate(self.active_k):
            L = self.N - k
            vals[idx] = dx * np.dot(h[k:k + L], omh[:L])
        return t - vals

    def con_jac(self, x):
        h = x[:-1]
        N = self.N
        dx = 2.0 / N
        J = np.zeros((len(self.active_k), len(x)))
        for idx, k in enumerate(self.active_k):
            gh = compute_grad_h_for_k(h, k, N, dx)
            J[idx, :N] = -gh
            J[idx, -1] = 1.0
        return J

    def eq_fun(self, x):
        return np.sum(x[:-1]) - self.N / 2.0

    def eq_jac(self, x):
        g = np.zeros(len(x))
        g[:-1] = 1.0
        return g

def run_bfgs_phase(z0, N, time_limit=60.0):
    t_start = time.time()
    cur = z0.copy()
    betas = [30, 100, 300, 1000, 3000, 8000]
    for beta in betas:
        if time.time() - t_start > time_limit:
            break
        sp = SmoothProblem(N, beta)
        try:
            res = minimize(sp.fun, cur, jac=sp.jac, method='L-BFGS-B',
                           bounds=[(-12.0, 12.0)] * (N // 2),
                           options={'maxiter': 2000, 'ftol': 1e-15})
            if res.success and np.all(np.isfinite(res.x)):
                cur = np.clip(res.x, -12.0, 12.0)
        except Exception:
            pass
    return cur

def run_activeset_sym(z0, N, time_limit=60.0):
    t_start = time.time()
    cur = z0.copy()
    for iteration in range(5):
        if time.time() - t_start > time_limit:
            break
        h_cur = z_to_h_sym(cur, N)
        vals = compute_c_values_sym(cur, N)
        active = find_active_k(vals, top_frac=0.4, min_count=10, thresh_frac=0.003)
        active = active[:25]
        cmax = vals.max()
        t0 = cmax + 1e-5
        x0 = np.concatenate([cur, [t0]])
        asp = ActiveSetSym(N, active)
        bounds = [(-12.0, 12.0)] * (N // 2) + [(0.0, 1.0)]
        cons = [{'type': 'ineq', 'fun': asp.con_fun, 'jac': asp.con_jac}]
        try:
            res = minimize(asp.fun, x0, jac=asp.jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                new_z = np.clip(res.x[:-1], -12.0, 12.0)
                new_h = z_to_h_sym(new_z, N)
                new_sc = official_score(new_h)
                old_sc = official_score(h_cur)
                if new_sc < old_sc:
                    cur = new_z
        except Exception:
            pass
    return cur

def run_activeset_full(h0, N, time_limit=40.0):
    t_start = time.time()
    cur = h0.copy()
    for iteration in range(4):
        if time.time() - t_start > time_limit:
            break
        vals = compute_c_values_full(cur)
        active = find_active_k(vals, top_frac=0.35, min_count=10, thresh_frac=0.003)
        active = active[:22]
        cmax = vals.max()
        t0 = cmax + 1e-5
        x0 = np.concatenate([cur, [t0]])
        asf = ActiveSetFull(N, active)
        bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)]
        cons = [
            {'type': 'ineq', 'fun': asf.con_fun, 'jac': asf.con_jac},
            {'type': 'eq', 'fun': asf.eq_fun, 'jac': asf.eq_jac}
        ]
        try:
            res = minimize(asf.fun, x0, jac=asf.jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                new_h = np.clip(res.x[:N], 0.0, 1.0)
                new_h = make_feasible(new_h)
                new_sc = official_score(new_h)
                old_sc = official_score(cur)
                if new_sc < old_sc:
                    cur = new_h
        except Exception:
            pass
    return cur

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-15:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        x = np.arange(n0)
        h_init = 0.5 + 0.25 * np.sin(2 * np.pi * x / n0) + 0.08 * np.cos(4 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    # Symmetrize parent
    h_sym = symmetrize(best_h)
    sc_sym = official_score(h_sym)
    if sc_sym < best_score - 1e-15:
        update(h_sym, sc_sym)
        best_h = h_sym.copy()

    # Phase 1: Try multiple N values with BFGS in symmetric space
    n_base = len(best_h)
    n_candidates = sorted(set([
        max(40, n_base - 6),
        max(42, n_base - 4),
        max(44, n_base - 2),
        n_base,
        min(80, n_base + 2),
        min(82, n_base + 4),
    ]))

    for N_try in n_candidates:
        if time_left() < 40:
            break
        if N_try % 2 != 0:
            N_try += 1  # ensure even for symmetric param
        if N_try == n_base:
            h_start = best_h
        else:
            h_start = interpolate_h(best_h, N_try)
        h_start = symmetrize(h_start)
        sc_start = official_score(h_start)
        if sc_start < best_score - 1e-15:
            update(h_start, sc_start)
        z_start = h_to_z_sym(h_start, N_try)
        tl = min(time_left() * 0.15, 30.0)
        if tl < 5:
            break
        z_opt = run_bfgs_phase(z_start, N_try, time_limit=tl)
        h_opt = z_to_h_sym(z_opt, N_try)
        sc_opt = official_score(h_opt)
        if sc_opt < best_score - 1e-15:
            update(h_opt, sc_opt)
            # Also try activeset on this
            if time_left() > 30:
                z_ref = run_activeset_sym(z_opt, N_try, time_limit=min(time_left() * 0.1, 20.0))
                h_ref = z_to_h_sym(z_ref, N_try)
                sc_ref = official_score(h_ref)
                if sc_ref < best_score - 1e-15:
                    update(h_ref, sc_ref)

    # Phase 2: Intensive BFGS + ActiveSet on the best N
    n_best = len(best_h)
    if time_left() > 60:
        h_cur = symmetrize(best_h)
        if official_score(h_cur) < best_score - 1e-15:
            update(h_cur, official_score(h_cur))
            best_h = h_cur.copy()
        n_work = len(best_h)
        z_work = h_to_z_sym(best_h, n_work)

        # Multi-beta BFGS with higher precision
        for beta in [2000, 5000, 15000]:
            if time_left() < 15:
                break
            sp = SmoothProblem(n_work, beta)
            try:
                res = minimize(sp.fun, z_work, jac=sp.jac, method='L-BFGS-B',
                               bounds=[(-12.0, 12.0)] * (n_work // 2),
                               options={'maxiter': 3000, 'ftol': 1e-16})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    z_work = np.clip(res.x, -12.0, 12.0)
                    h_tmp = z_to_h_sym(z_work, n_work)
                    sc_tmp = official_score(h_tmp)
                    if sc_tmp < best_score - 1e-15:
                        update(h_tmp, sc_tmp)
                        best_h = h_tmp.copy()
            except Exception:
                pass

    # Phase 3: Active set refinement (multiple rounds)
    if time_left() > 40:
        n_act = len(best_h)
        z_act = h_to_z_sym(best_h, n_act)
        z_act = run_activeset_sym(z_act, n_act, time_limit=min(time_left() * 0.25, 80.0))
        h_act = z_to_h_sym(z_act, n_act)
        sc_act = official_score(h_act)
        if sc_act < best_score - 1e-15:
            update(h_act, sc_act)
            best_h = h_act.copy()

    # Phase 4: Full-space SLSQP (breaks symmetry if beneficial)
    if time_left() > 30:
        h_full = run_activeset_full(best_h, len(best_h), time_limit=min(time_left() * 0.2, 40.0))
        sc_full = official_score(h_full)
        if sc_full < best_score - 1e-15:
            update(h_full, sc_full)
            best_h = h_full.copy()

    # Phase 5: Symmetrize again and polish
    if time_left() > 30:
        h_re_sym = symmetrize(best_h)
        sc_rs = official_score(h_re_sym)
        if sc_rs < best_score - 1e-15:
            update(h_re_sym, sc_rs)
            best_h = h_re_sym.copy()
        n_pol = len(best_h)
        z_pol = h_to_z_sym(best_h, n_pol)
        for beta in [10000, 30000]:
            if time_left() < 10:
                break
            sp = SmoothProblem(n_pol, beta)
            try:
                res = minimize(sp.fun, z_pol, jac=sp.jac, method='L-BFGS-B',
                               bounds=[(-12.0, 12.0)] * (n_pol // 2),
                               options={'maxiter': 2000, 'ftol': 1e-16})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    z_pol = np.clip(res.x, -12.0, 12.0)
                    h_tmp = z_to_h_sym(z_pol, n_pol)
                    sc_tmp = official_score(h_tmp)
                    if sc_tmp < best_score - 1e-15:
                        update(h_tmp, sc_tmp)
                        best_h = h_tmp.copy()
            except Exception:
                pass

    # Phase 6: Random restarts in z-space
    if time_left() > 40:
        n_rs = len(best_h)
        z_best = h_to_z_sym(best_h, n_rs)
        n_starts = max(3, min(10, int(time_left() / 12)))
        for i in range(n_starts):
            if time_left() < 10:
                break
            sigma = 0.08 * (i + 1) / n_starts
            z_cand = z_best + rng.normal(0, sigma, size=n_rs // 2)
            z_cand = np.clip(z_cand, -10.0, 10.0)
            h_cand = z_to_h_sym(z_cand, n_rs)
            sc_cand = official_score(h_cand)
            if sc_cand < best_score - 1e-15:
                update(h_cand, sc_cand)
            # Quick BFGS polish
            if time_left() > 8:
                sp = SmoothProblem(n_rs, 3000)
                try:
                    res = minimize(sp.fun, z_cand, jac=sp.jac, method='L-BFGS-B',
                                   bounds=[(-12.0, 12.0)] * (n_rs // 2),
                                   options={'maxiter': 500, 'ftol': 1e-14})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_p = z_to_h_sym(np.clip(res.x, -12, 12), n_rs)
                        sc_p = official_score(h_p)
                        if sc_p < best_score - 1e-15:
                            update(h_p, sc_p)
                except Exception:
                    pass

    # Phase 7: Subgradient descent on the max (polishing)
    if time_left() > 20:
        h_sub = best_h.copy()
        step = 0.05
        for it in range(100):
            if time_left() < 5:
                break
            vals = compute_c_values_full(h_sub)
            cmax = vals.max()
            thresh = cmax - max(0.01 * cmax, 1e-5)
            active = np.where(vals >= thresh)[0]
            dx = 2.0 / len(h_sub)
            g = np.zeros(len(h_sub))
            for k in active:
                g += compute_grad_h_for_k(h_sub, k, len(h_sub), dx)
            g /= len(active)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            h_sub = make_feasible(h_sub - step * g / gnorm)
            sc_sub = official_score(h_sub)
            if sc_sub < best_score - 1e-15:
                update(h_sub, sc_sub)
            step *= 0.95

    # Phase 8: Final active set with broad constraints
    if time_left() > 15:
        n_fin = len(best_h)
        z_fin = h_to_z_sym(best_h, n_fin)
        vals_fin = compute_c_values_sym(z_fin, n_fin)
        active_fin = find_active_k(vals_fin, top_frac=0.5, min_count=15, thresh_frac=0.002)
        active_fin = active_fin[:25]
        cmax_fin = vals_fin.max()
        x0_fin = np.concatenate([z_fin, [cmax_fin + 1e-4]])
        asp_fin = ActiveSetSym(n_fin, active_fin)
        bounds_fin = [(-12.0, 12.0)] * (n_fin // 2) + [(0.0, 1.0)]
        cons_fin = [{'type': 'ineq', 'fun': asp_fin.con_fun, 'jac': asp_fin.con_jac}]
        try:
            res = minimize(asp_fin.fun, x0_fin, jac=asp_fin.jac, method='SLSQP',
                           bounds=bounds_fin, constraints=cons_fin,
                           options={'maxiter': 800, 'ftol': 1e-15})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_fin = z_to_h_sym(np.clip(res.x[:-1], -12, 12), n_fin)
                sc_fin = official_score(h_fin)
                if sc_fin < best_score - 1e-15:
                    update(h_fin, sc_fin)
        except Exception:
            pass

    # Final validation
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = make_feasible(final_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Ensure feasibility
    assert np.all(final_h >= -1e-14), "Negative values"
    assert np.all(final_h <= 1.0 + 1e-14), "Values > 1"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-9, "Sum constraint violated"

    return final_h, final_score + 1e-12, final_n
