def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_V(h, idx):
    """Gradient of V_idx at h."""
    n = len(h)
    s = idx - (n - 1)
    L = n - abs(s)
    if L <= 0:
        return np.zeros(n)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:s+L] += (1.0 - h[0:L])
        grad[0:L] -= h[s:s+L]
    else:
        t = -s
        grad[0:L] += (1.0 - h[t:t+L])
        grad[t:t+L] -= h[0:L]
    return grad

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = np.clip(beta * (V - V_max), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.full(len(V), 1.0 / len(V))
    return w / ws

def pg_step(h, beta, alpha_max=0.5, max_ls=15):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_weights(V, beta)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if abs(w[idx]) > 1e-14:
            g += w[idx] * gradient_V(h, idx)
    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h.copy(), f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def lp_polish(h, eps, trust, max_iters=30):
    """Iterative LP polishing."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    sc_cur = official_score(h_cur)
    
    for _ in range(max_iters):
        V = compute_values(h_cur)
        V_max = V.max()
        active = np.where(V >= V_max - eps)[0]
        if len(active) == 0:
            break
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_rows = []
        b_rows = []
        for p_idx in active:
            gp = gradient_V(h_cur, p_idx)
            row = np.zeros(n + 1)
            row[:n] = dx * gp
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(-V[p_idx])
        
        if not A_rows:
            break
        
        bounds_list = []
        for i in range(n):
            bounds_list.append((max(-h_cur[i], -trust), min(1.0 - h_cur[i], trust)))
        bounds_list.append((0.0, None))
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        
        try:
            res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                         A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
            if res.success and res.x is not None:

                    sc_new = official_score(h_new)

                        sc_cur = sc_new
                        trust = max(trust * 0.7, 0.01)

                else:
