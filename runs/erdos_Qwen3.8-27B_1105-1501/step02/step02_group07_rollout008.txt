def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.0, posinf=5.0, neginf=-5.0)
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def dct_basis(n, d):
    """Create normalized cos basis for n samples, d modes."""
    x = (np.arange(n) + 0.5) / n
    B = np.zeros((n, d))
    for k in range(1, d + 1):
        B[:, k-1] = np.cos(np.pi * k * x)
    # Normalize columns
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-10] = 1.0
    B /= norms
    return B

def coeff_to_h(B, theta):
    z = 0.5 + B @ theta
    return make_feasible(z)

def h_to_coeffs(h, B):
    """Project h onto the cos basis (least squares)."""
    # h ≈ 0.5 + B @ theta => (h - 0.5) ≈ B @ theta
    residual = h - 0.5
    theta, _, _, _ = np.linalg.lstsq(B, residual, rcond=None)
    return theta

def evaluate_theta(B, theta):
    h = coeff_to_h(B, theta)
    return official_score(h), h

def de_search(B, theta0, pop_size, max_gens, f_scale, cr, bounds, seed, deadline_fn):

    d = len(theta0)
    rng = np.random.default_rng(seed)
    
    # Initialize population
    pop = np.zeros((pop_size, d))
    pop[0] = theta0

    # Evaluate
    scores = np.zeros(pop_size)
    best_h_list = [None] * pop_size
    for i in range(pop_size):
        scores[i], best_h_list[i] = evaluate_theta(B, pop[i])
    
    # Track best
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]

    best_h = best_h_list[best_idx].copy()
    
    for gen in range(max_gens):

            break

                break

                a = rng.integers(pop_size)
                b = rng.integers(pop_size)
                c = rng.integers(pop_size)
            v = pop[a] + f_scale * (pop[b] - pop[c])
            v = np.clip(v, bounds[0], bounds[1])

            j_rand = rng.integers(d)

                scores[i] = trial_score

                if trial_score < best_score:

                    best_theta = trial.copy()
                    best_h = trial_h.copy()

def active_set_lp(h, eps=0.005, trust=0.15):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    # Limit active set size
    if len(active) > 30:
        active = active[np.argsort(V[active])[-30:]]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),

        if res.success and res.x is not None:

                sc = official_score(h_new)
                return h_new, sc

    return None, None

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def pg_step(h, beta, alpha_max=0.8, max_ls=16):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w /= ws
    
    grad = np.zeros(n)
    for idx in range(len(w)):
        wv = w[idx]
        if abs(wv) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wv * (1.0 - h[0:L])
            grad[0:L] -= wv * h[s:n]
        else:
            t = -s
            grad[0:L] += wv * (1.0 - h[t:n])
            grad[t:n] -= wv * h[0:L]
    
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.0, posinf=5.0, neginf=-5.0)
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def dct_basis(n, d):
    x = (np.arange(n) + 0.5) / n
    B = np.zeros((n, d))
    for k in range(1, d + 1):
        B[:, k - 1] = np.cos(np.pi * k * x)
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-10] = 1.0
    B /= norms
    return B

def coeff_to_h(B, theta):
    z = 0.5 + B @ theta
    return make_feasible(z)

def h_to_coeffs(h, B):
    residual = h - 0.5
    theta, _, _, _ = np.linalg.lstsq(B, residual, rcond=None)
    return theta

def evaluate_theta(B, theta):
    h = coeff_to_h(B, theta)
    return official_score(h), h

def de_search(B, theta0, pop_size, max_gens, f_scale, cr, lo, hi, seed, deadline_fn):
    d = len(theta0)
    rng = np.random.default_rng(seed)
    pop = np.zeros((pop_size, d))
    pop[0] = theta0
    for i in range(1, pop_size):
        pop[i] = rng.uniform(lo, hi, d)
    scores = np.zeros(pop_size)
    h_list = [None] * pop_size
    for i in range(pop_size):
        scores[i], h_list[i] = evaluate_theta(B, pop[i])
    best_idx = int(np.argmin(scores))
    best_score = scores[best_idx]
    best_h = h_list[best_idx].copy()
    for gen in range(max_gens):
        if deadline_fn() < 15:
            break
        for i in range(pop_size):
            if deadline_fn() < 15:
                break
            perm = rng.permutation(pop_size)
            candidates = [x for x in perm if x != i]
            a, b, c = candidates[0], candidates[1], candidates[2]
            v = pop[a] + f_scale * (pop[b] - pop[c])
            v = np.clip(v, lo, hi)
            trial = pop[i].copy()
            j_rand = int(rng.integers(d))
            for j in range(d):
                if rng.random() < cr or j == j_rand:
                    trial[j] = v[j]
            trial_score, trial_h = evaluate_theta(B, trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                h_list[i] = trial_h
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial_h.copy()
    return best_h, best_score

def golden_pair(h, i, j):
    n = len(h)
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-10:
        return None
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = h.copy()
    fc[i] = c_pt
    fc[j] = S - c_pt
    fc_score = official_score(fc)
    fd = h.copy()
    fd[i] = d_pt
    fd[j] = S - d_pt
    fd_score = official_score(fd)
    for _ in range(20):
        if b - a < 1e-9:
            break
        if fc_score < fd_score:
            b = d_pt
            d_pt = c_pt
            fd_score = fc_score
            c_pt = b - gr * (b - a)
            fc = h.copy()
            fc[i] = c_pt
            fc[j] = S - c_pt
            fc_score = official_score(fc)
        else:
            a = c_pt
            c_pt = d_pt
            fc_score = fd_score
            d_pt = a + gr * (b - a)
            fd = h.copy()
            fd[i] = d_pt
            fd[j] = S - d_pt
            fd_score = official_score(fd)
    x_best = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] = x_best
    h_new[j] = S - x_best
    h_new = np.clip(h_new, 0.0, 1.0)
    res = n / 2.0 - np.sum(h_new)
    if abs(res) > 1e-10:
        h_new = make_feasible(h_new)
    return h_new

def pairwise_exchange(h, n_pairs, seed):
    n = len(h)
    rng = np.random.default_rng(seed)
    best_score = official_score(h)
    best_h = h.copy()
    for _ in range(n_pairs):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        h_new = golden_pair(best_h, i, j)
        if h_new is None:
            continue
        sc = official_score(h_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_new
    return best_h, best_score

def active_set_lp(h, eps, trust):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    if len(active) > 25:
        active = active[np.argsort(V[active])[-25:]]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        bounds_list.append((max(-h[i], -trust), min(1.0 - h[i], trust)))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def pg_step(h, beta, alpha_max, max_ls):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    shifted = beta * (V - f)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w /= ws
    grad = np.zeros(n)
    for idx in range(len(w)):
        wv = w[idx]
        if abs(wv) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wv * (1.0 - h[0:L])
            grad[0:L] -= wv * h[s:n]
        else:
            t = -s
            grad[0:L] += wv * (1.0 - h[t:n])
            grad[t:n] -= wv * h[0:L]
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, f
    grad /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * grad)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f
