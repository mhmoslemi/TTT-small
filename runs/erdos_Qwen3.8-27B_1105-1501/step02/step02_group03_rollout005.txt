    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, s:] += dx * (1.0 - h[:L])
            G[p, :L] += dx * (-h[s:])
        else:
            t = -s
            L = n - t
            G[p, :L] += dx * (1.0 - h[t:])
            G[p, t:] += dx * (-h[:L])
    return G

def smooth_obj_grad_z(z, n, alpha, beta):
    h = build_h_sym(z, n, alpha)
    vals = compute_vals(h)
    G_h = val_grad_h(h)
    m = n // 2
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -700, 700)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= wsum
    y = build_h_sym(z, n, 1.0)
    y_orig = 1.0 / (1.0 + np.exp(-np.clip(alpha * z, -50, 50)))
    y_orig = np.clip(y_orig, 1e-10, 1.0 - 1e-10)
    dydz = alpha * y_orig * (1.0 - y_orig)
    G_z = np.zeros((2 * n - 1, m))
    for j in range(m):
        G_z[:, j] = dydz[j] * G_h[:, j] - dydz[j] * G_h[:, n - 1 - j]
    grad = w @ G_z
    return obj, grad

def lgbfgs_optimize_z(z0, n, alpha, beta_schedule, time_limit=30.0):
    z = z0.copy()
    t0 = time.time()
    for beta in beta_schedule:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(
                lambda zz: smooth_obj_grad_z(zz, n, alpha, beta)[0],
                z,
                jac=lambda zz: smooth_obj_grad_z(zz, n, alpha, beta)[1],
                method='L-BFGS-B',
                bounds=[(-12.0, 12.0)] * len(z),
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10}
            )
            if res.success or res.fun < float('inf'):
                h_new = build_h_sym(res.x, n, alpha)
                sc = official_score(h_new)
                if np.isfinite(sc):
                    z = res.x
        except Exception:
            pass
    return z

def slsqp_sym(z0, n, alpha, active_idx, time_limit=30.0):
    m = len(z0)
    t0 = time.time()

    def obj_func(params):
        return params[-1]

    def obj_grad(params):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(params):
        z = params[:m]
        t = params[-1]
        h = build_h_sym(z, n, alpha)
        vals = compute_vals(h)
        return t - vals[active_idx]

    def ineq_jac(params):
        z = params[:m]
        h = build_h_sym(z, n, alpha)
        G_h = val_grad_h(h)
        m_local = n // 2
        y_orig = 1.0 / (1.0 + np.exp(-np.clip(alpha * z, -50, 50)))
        y_orig = np.clip(y_orig, 1e-10, 1.0 - 1e-10)
        dydz = alpha * y_orig * (1.0 - y_orig)
        G_z = np.zeros((2 * n - 1, m_local))
        for j in range(m_local):
            G_z[:, j] = dydz[j] * G_h[:, j] - dydz[j] * G_h[:, n - 1 - j]
        J = -G_z[active_idx, :]
        return np.hstack([J, np.ones((len(active_idx), 1))])

    sc0 = official_score(build_h_sym(z0, n, alpha))
    z_start = np.concatenate([z0, [sc0 + 1e-4]])
    bounds = [(-12.0, 12.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    try:
        res = minimize(obj_func, z_start, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = build_h_sym(res.x[:m], n, alpha)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return res.x[:m], sc
    except Exception:
        pass
    return None, None

def find_active(vals, frac=0.25, min_count=8):
    cmax = vals.max()
    threshold = cmax - max(0.01 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def symmetrize(h):
    h_sym = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_sym)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    def check_deadline():
        return time.time() >= deadline

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        x = np.arange(n0)
        h_init = 0.5 + 0.2 * np.sin(2 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    n_base = len(best_h)

    # Phase 1: Antisymmetric smooth optimization on base n
    if time_left() > 50:
        for alpha in [1.0, 2.0, 3.0]:
            if check_deadline():
                break
            z0 = h_to_z(best_h, n_base, alpha)
            beta_sched = [20, 50, 100, 300, 1000, 3000]
            z_opt = lgbfgs_optimize_z(z0, n_base, alpha, beta_sched, time_limit=min(time_left() * 0.2, 30.0))
            h_new = build_h_sym(z_opt, n_base, alpha)
            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                update(h_new, sc)

    # Phase 2: SLSQP with active set on antisymmetric
    if time_left() > 40:
        for alpha in [1.0, 2.0]:
            if check_deadline():
                break
            z0 = h_to_z(best_h, n_base, alpha)
            h_cur = build_h_sym(z0, n_base, alpha)
            vals = compute_vals(h_cur)
            active = find_active(vals)
            z_new, sc = slsqp_sym(z0, n_base, alpha, active, time_limit=min(time_left() * 0.25, 25.0))
            if z_new is not None and sc < best_score - 1e-14:
                h_new = build_h_sym(z_new, n_base, alpha)
                h_new = make_feasible(h_new)
                sc2 = official_score(h_new)
                if sc2 < best_score - 1e-14:
                    update(h_new, sc2)
                z0 = z_new
            # Refine with updated active set
            if time_left() > 20:
                h_cur2 = build_h_sym(z0, n_base, alpha)
                vals2 = compute_vals(h_cur2)
                active2 = find_active(vals2, frac=0.35, min_count=12)
                z_new2, sc2 = slsqp_sym(z0, n_base, alpha, active2, time_limit=min(time_left() * 0.2, 20.0))
                if z_new2 is not None and sc2 < best_score - 1e-14:
                    h_new2 = build_h_sym(z_new2, n_base, alpha)
                    h_new2 = make_feasible(h_new2)
                    sc3 = official_score(h_new2)
                    if sc3 < best_score - 1e-14:
                        update(h_new2, sc3)

    # Phase 3: Multi-resolution
    if time_left() > 60:
        n_base = len(best_h)
        n_opts = sorted(set([
            max(40, n_base - 8),
            max(44, n_base - 4),
            n_base + 2,
            n_base + 4,
            min(76, n_base + 8),
            min(80, n_base + 12),
        ]))
        for n_opt in n_opts:
            if check_deadline():
                break
            if n_opt == n_base:
                continue
            h_int = interpolate_to(best_h, n_opt)
            h_sym = symmetrize(h_int)
            sc_sym = official_score(h_sym)
            if sc_sym < best_score - 1e-14:
                update(h_sym, sc_sym)
            for alpha in [1.5, 2.5]:
                if check_deadline():
                    break
                z0 = h_to_z(h_sym, n_opt, alpha)
                beta_sched = [50, 200, 1000, 3000]
                z_opt = lgbfgs_optimize_z(z0, n_opt, alpha, beta_sched, time_limit=min(time_left() * 0.15, 15.0))
                h_new = build_h_sym(z_opt, n_opt, alpha)
                h_new = make_feasible(h_new)
                sc = official_score(h_new)
                if sc < best_score - 1e-14:
                    update(h_new, sc)
            if time_left() > 20:
                z0 = h_to_z(best_h if len(best_h) == n_opt else interpolate_to(best_h, n_opt), n_opt, 2.0)
                h_cur = build_h_sym(z0, n_opt, 2.0)
                vals = compute_vals(h_cur)
                active = find_active(vals)
                z_new, sc = slsqp_sym(z0, n_opt, 2.0, active, time_limit=min(time_left() * 0.15, 12.0))
                if z_new is not None and sc < best_score - 1e-14:
                    h_new = build_h_sym(z_new, n_opt, 2.0)
                    h_new = make_feasible(h_new)
                    sc2 = official_score(h_new)
                    if sc2 < best_score - 1e-14:
                        update(h_new, sc2)

    # Phase 4: Symmetry-breaking perturbation + full optimization
    if time_left() > 40:
        n_cur = len(best_h)
        for sigma in [0.005, 0.01, 0.02, 0.03, 0.05]:
            if check_deadline():
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n_cur))
            sc_pert = official_score(h_pert)
            if sc_pert < best_score - 1e-14:
                update(h_pert, sc_pert)
            # Projected gradient polish
            for step in [0.05, 0.02, 0.01]:
                if check_deadline():
                    break
                h_tmp = h_pert.copy()
                for it in range(100):
                    vals_t = compute_vals(h_tmp)
                    cmax_t = vals_t.max()
                    thr = cmax_t - max(0.01 * cmax_t, 1e-5)
                    act = np.where(vals_t >= thr)[0]
                    G = val_grad_h(h_tmp)
                    g = G[act].mean(axis=0)
                    gnorm = np.linalg.norm(g)
                    if gnorm < 1e-14:
                        break
                    h_new_t = make_feasible(h_tmp - step * g / gnorm)
                    sc_t = official_score(h_new_t)
                    if sc_t < official_score(h_tmp) - 1e-12:
                        h_tmp = h_new_t
                    else:
                        step *= 0.5
                        if step < 1e-8:
                            break
                sc_final = official_score(h_tmp)
                if sc_final < best_score - 1e-14:
                    update(h_tmp, sc_final)
                    h_pert = h_tmp

    # Phase 5: Full SLSQP without symmetry constraint (on best)
    if time_left() > 30:
        n_cur = len(best_h)
        vals_fin = compute_vals(best_h)
        active_fin = find_active(vals_fin, frac=0.3, min_count=10)
        dx = 2.0 / n_cur

        def full_obj(params):
            return params[-1]

        def full_obj_grad(params):
            g = np.zeros(n_cur + 1)
            g[-1] = 1.0
            return g

        def full_ineq(params):
            h = params[:n_cur]
            t = params[-1]
            vals_f = compute_vals(h)
            return t - vals_f[active_fin]

        def full_ineq_jac(params):
            h = params[:n_cur]
            G = val_grad_h(h)
            J = -G[active_fin, :]
            return np.hstack([J, np.ones((len(active_fin), 1))])

        def full_eq(params):
            return np.sum(params[:n_cur]) - n_cur / 2.0

        def full_eq_jac(params):
            g = np.zeros(n_cur + 1)
            g[:n_cur] = 1.0
            return g

        t_init = official_score(best_h) + 2e-5
        z_full = np.concatenate([best_h, [t_init]])
        bounds_full = [(0.0, 1.0)] * n_cur + [(0.0, 2.0)]
        cons_full = [
            {'type': 'ineq', 'fun': full_ineq, 'jac': full_ineq_jac},
            {'type': 'eq', 'fun': full_eq, 'jac': full_eq_jac}
        ]
        try:
            res_full = minimize(full_obj, z_full, jac=full_obj_grad, method='SLSQP',
                                bounds=bounds_full, constraints=cons_full,
                                options={'maxiter': 400, 'ftol': 1e-15})
            if res_full.x is not None and np.all(np.isfinite(res_full.x)):
                h_full = make_feasible(res_full.x[:n_cur])
                sc_full = official_score(h_full)
                if sc_full < best_score - 1e-14:
                    update(h_full, sc_full)
        except Exception:
            pass

    # Phase 6: Final high-beta polish on current best
    if time_left() > 25:
        n_fin = len(best_h)
        h_sym_fin = symmetrize(best_h)
        if official_score(h_sym_fin) < best_score - 1e-14:
            update(h_sym_fin)
            best_h = h_sym_fin
        for alpha in [2.0, 3.0]:
            if check_deadline():
                break
            z_fin = h_to_z(best_h, n_fin, alpha)
            beta_sched = [500, 2000, 5000, 10000]
            z_opt = lgbfgs_optimize_z(z_fin, n_fin, alpha, beta_sched, time_limit=min(time_left() * 0.3, 15.0))
            h_new = build_h_sym(z_opt, n_fin, alpha)
            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                update(h_new, sc)

    # Final validation
    if best_h is None:
        best_h = np.ones(66) * 0.5
        best_score = official_score(best_h)

    final_h = make_feasible(best_h)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
