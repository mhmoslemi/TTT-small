```python
def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    """Convert to binary by thresholding, preserving count."""
    n = len(h)
    target = n // 2
    # Top target values become 1, rest become 0
    order = np.argsort(h)[::-1][:target]
    result = np.zeros(n)
    result[order] = 1.0
    return result

def block_search(h, max_block_size=10, max_passes=20, time_limit=None, t_start=None):
    """Exact block permutation local search for binary sequences."""
    n = len(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for size in range(4, max_block_size + 1, 2):
        improved = True
        passes = 0
        while improved and passes < max_passes:
            passes += 1
            if time_limit is not None and t_start is not None:
                if time.time() - t_start > time_limit:
                    return best, best_score
            improved = False
            for s in range(0, n - size + 1):
                if time_limit is not None and t_start is not None:
                    if time.time() - t_start > time_limit:
                        return best, best_score
                block = best[s:s+size]
                k = int(np.sum(block))
                if k == 0 or k == size:

                # Try all arrangements of k ones in size positions

                best_local = None

                for comb in combinations(ones_positions, k):
                    new_block = np.zeros(size)

                    if np.array_equal(new_block, block):

                    candidate = best.copy()
                    candidate[s:s+size] = new_block
                    sc = exact_score(candidate)
                    if sc < best_local_score - 1e-15:
                        best_local_score = sc
                        best_local = candidate.copy()
                if best_local is not None:
                    best = best_local
                    best_score = best_local_score
                    improved = True
    return best, best_score
```

For a and v both of length n: k ranges from -(n-1) to (n-1), output length 2n-1.
Output index p corresponds to k = p - (n-1).

So `corr[p] = sum_n h[n+k] * (1-h[n])` where k = p - (n-1).

For k ≥ 0 (p ≥ n-1): n ranges from 0 to n-1-k
`corr[p] = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])`

For k < 0 (p < n-1): let m = -k = n-1-p > 0. n ranges from m to n-1
`corr[p] = sum_{n=m}^{n-1} h[n-m] * (1-h[n])`

Let me verify with a small example. h = [a, b, c] (n=3), v = 1-h = [1-a, 1-b, 1-c].
Full correlation has length 5.
- p=0, k=-2: sum_{n=2}^{2} h[n-2] * (1-h[n]) = h[0]*(1-h[2]) = a*(1-c)
- p=1, k=-1: sum_{n=1}^{2} h[n-1] * (1-h[n]) = h[0]*(1-h[1]) + h[1]*(1-h[2]) = a*(1-b) + b*(1-c)
- p=2, k=0: sum_{n=0}^{2} h[n] * (1-h[n]) = a*(1-a) + b*(1-b) + c*(1-c)
- p=3, k=1: sum_{n=0}^{1} h[n+1] * (1-h[n]) = h[1]*(1-h[0]) + h[2]*(1-h[1]) = b*(1-a) + c*(1-b)
- p=4, k=2: sum_{n=0}^{0} h[n+2] * (1-h[n]) = h[2]*(1-h[0]) = c*(1-a)

So the pattern is:
- For k ≥ 0: corr = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
- For k < 0 (m = -k): corr = sum_{n=m}^{n-1} h[n-m] * (1-h[n])

These are NOT the same! For k=1: b*(1-a) + c*(1-b). For k=-1: a*(1-b) + b*(1-c). Different!

So the correlation is NOT symmetric in general (unless h has special structure). Good to know.

```python
import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    target = n // 2
    order = np.argsort(h)[::-1][:target]
    result = np.zeros(n)
    result[order] = 1.0
    return result

def interp_to_n(src, n_dst):
    n_src = len(src)
    if n_src == n_dst:
        return src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, src)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def block_search(h, sizes, max_passes=10, time_limit=None, t_start=None):
    n = len(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for size in sizes:
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
            break
        improved = True
        passes = 0
        while improved and passes < max_passes:
            passes += 1
            improved = False
            for s in range(0, n - size + 1):
                if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
                    return best, best_score
                block = best[s:s+size]
                k = int(np.sum(block) + 0.5)
                if k == 0 or k == size:
                    continue
                best_local_score = best_score
                best_local_block = None
                for comb in combinations(range(size), k):
                    if tuple(comb) == tuple(np.where(block > 0.5)[0]):
                        continue
                    new_block = np.zeros(size)
                    new_block[list(comb)] = 1.0
                    candidate = best.copy()
                    candidate[s:s+size] = new_block
                    sc = exact_score(candidate)
                    if sc < best_local_score - 1e-15:
                        best_local_score = sc
                        best_local_block = new_block
                if best_local_block is not None:
                    best[s:s+size] = best_local_block
                    best_score = best_local_score
                    improved = True
    return best, best_score

def move_search(h, max_passes=10, time_limit=None, t_start=None):
    n = len(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for _ in range(max_passes):
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:

        ones = list(np.where(best > 0.5)[0])
        zeros = list(np.where(best < 0.5)[0])

                    ones = list(np.where(best > 0.5)[0])
                    zeros = list(np.where(best < 0.5)[0])

    return best, best_score

def slsqp_polish(h0, betas=(50, 200), time_limit=None, t_start=None):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best = h.copy()
    best_score = exact_score(best)
    
    for beta in betas:
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
            break
        try:
            res = minimize(
                _lse_obj, h, args=(beta, dx, n),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}],
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if res.success or hasattr(res, 'x'):
                h_new = make_feasible(res['x'])
                sc = exact_score(h_new)
                if sc < best_score - 1e-12:
                    best = h_new
                    best_score = sc
        except Exception:
            pass
    
    return best, best_score

def _lse_obj(x, beta, dx, n):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def _sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def generate_thue_morse(n):
    arr = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Balance: ensure exactly n//2 ones
    n_ones = int(np.sum(arr))
    target = n // 2
    if n_ones > target:
        # Remove excess ones
        one_indices = np.where(arr == 1)[0]
        rng_local = np.random.default_rng(123)
        to_flip = rng_local.choice(one_indices, n_ones - target, replace=False)
        arr[to_flip] = 0
    elif n_ones < target:
        zero_indices = np.where(arr == 0)[0]
        rng_local = np.random.default_rng(123)
        to_flip = rng_local.choice(zero_indices, target - n_ones, replace=False)
        arr[to_flip] = 1
    return arr
```

```python
import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)


def to_binary(h):
    n = len(h)
    target = n // 2
    order = np.argsort(h)[::-1][:target]
    result = np.zeros(n)
    result[order] = 1.0
    return result


def interp_to_n(src, n_dst):
    n_src = len(src)
    if n_src == n_dst:
        return src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, src)


def block_search(h, size, max_passes=20, time_limit=None, t_start=None):
    n = len(h)
    best = h.copy()
    best_score = exact_score(best)
    for _ in range(max_passes):
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
            break
        best_pos = -1
        best_block = None
        best_local_score = best_score
        n_blocks = n - size + 1
        for s in range(n_blocks):
            if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
                break
            block = best[s:s + size]
            k = int(np.sum(block) + 0.5)
            if k == 0 or k == size:
                continue
            current_pat = np.where(block > 0.5)[0]
            candidate = best.copy()
            for comb in combinations(range(size), k):
                if list(comb) == list(current_pat):
                    continue
                candidate[s:s + size] = 0.0
                candidate[list(comb + s)] = 1.0
                sc = exact_score(candidate)
                if sc < best_local_score - 1e-15:
                    best_local_score = sc
                    best_pos = s
                    best_block = candidate[s:s + size].copy()
        if best_block is not None:
            best[best_pos:best_pos + size] = best_block
            best_score = best_local_score
        else:
            break
    return best, best_score


def move_search(h, max_passes=10, time_limit=None, t_start=None):
    n = len(h)
    best = h.copy()
    best_score = exact_score(best)
    for _ in range(max_passes):
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
            break
        ones = list(np.where(best > 0.5)[0])
        zeros = list(np.where(best < 0.5)[0])
        best_i, best_j = -1, -1
        best_local_score = best_score
        candidate = best.copy()
        for i in ones:
            for j in zeros:
                if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
                    if best_i >= 0:

                        best[best_j] = 1.0

                candidate[j] = 1.0

                    best_i, best_j = i, j
        if best_i >= 0:
            best[best_i] = 0.0
            best[best_j] = 1.0
            best_score = best_local_score
        else:
            break
    return best, best_score


def _lse_obj(x, beta, dx, n):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta


def _sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def _sum_jac(x, n):
    return np.ones(n)


def slsqp_polish(h0, betas=(50, 200, 500), time_limit=None, t_start=None):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best = h.copy()
    best_score = exact_score(best)
    for beta in betas:
        if time_limit is not None and t_start is not None and time.time() - t_start > time_limit:
            break
        try:
            res = minimize(
                _lse_obj, h, args=(beta, dx, n),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}],
                options={'maxiter': 60, 'ftol': 1e-12}
            )
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                h_new = make_feasible(res['x'])
                sc = exact_score(h_new)
                if sc < best_score - 1e-12:
                    best = h_new
                    best_score = sc
                    h = h_new
        except Exception:
            pass
    return best, best_score


def greedy_construct(n, seed=0):

            if sc < best_sc:

            h[pos] = 0.0
        h[best_pos] = 1.0
    return h


def thue_morse_balanced(n):
    arr = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    n_ones = int(np.sum(arr))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(arr == 1)[0]
        rng_l = np.random.default_rng(99)
        to_flip = rng_l.choice(one_idx, n_ones - target, replace=False)
        arr[to_flip] = 0.0
    elif n_ones < target:
        zero_idx = np.where(arr == 0)[0]
        rng_l = np.random.default_rng(99)
        to_flip = rng_l.choice(zero_idx, target - n_ones, replace=False)
        arr[to_flip] = 1.0
    return arr
```
