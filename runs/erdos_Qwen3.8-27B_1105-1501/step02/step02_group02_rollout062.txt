def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            h[i] += add
            res -= add
            if abs(res) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, beta):
    cv = _corr(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    cv = _corr(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j0 = max(0, s)
        j1 = min(n, n + s)
        for p in range(j0, j1):
            g[p] += wk * (1.0 - x[p - s])
        p0 = max(0, -s)
        p1 = min(n, n - s)
        for p in range(p0, p1):
            g[p] += wk * (-x[p + s])
    return g

def _slsqp_smooth(h0, beta, maxiter=200):

    bounds = [(0.0, 1.0)] * n

    def _efun(x):

def _coord_descent(h, n_sweeps=100):
    n = len(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            C0 = _corr(h)
            h[i] = 1.0
            C1 = _corr(h)
            d = C1 - C0
            best_x = 0.0
            best_val = np.inf
            xs = np.linspace(0, 1, 33)
            for x in xs:
                val = np.max(C0 + x * d)
                if val < best_val:
                    best_val = val
                    best_x = x
            h[i] = best_x

def _thue_morse(n):
