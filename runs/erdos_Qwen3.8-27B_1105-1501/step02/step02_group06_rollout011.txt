    n = len(h)
    g = np.zeros(n)
    # valid i: max(0,-s) <= i <= min(n-1, n-1-s)
    i_min = max(0, -s)
    i_max = min(n-1, n-1-s)
    if i_min > i_max:
        return g
    for i in range(i_min, i_max+1):
        j = i + s
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return g

import numpy as np
from scipy.optimize import linprog, minimize
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_lag(h, s):
    """Gradient of F_{s+(n-1)} with respect to h, where s is the offset."""
    n = len(h)
    g = np.zeros(n)
    i_min = max(0, -s)
    i_max = min(n-1, n-1-s)
    if i_min > i_max:
        return g
    for i in range(i_min, i_max+1):
        j = i + s
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return g

def make_feasible(z):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    # Bisection on shift
    lo, hi = -max(z.max(), 0) - 1, max(1 - z.min(), 0) + 1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_direction(h, active_indices, tol_bound=1e-7):
    """Solve LP to find feasible direction decreasing all active lags."""
    n = len(h)
    # Build constraint matrix
    # Variables: d (length n) + gamma (scalar) = n+1 variables
    # Minimize: -gamma (i.e., c = [0,...,0, -1])
    # Constraints:
    #   G_p · d + gamma <= 0 for each active p  (A_ub @ x <= b_ub)
    #   sum(d) = 0  (A_eq @ x = b_eq)
    #   d[i] >= 0 if h[i] <= tol_bound  -> -d[i] <= 0
    #   d[i] <= 0 if h[i] >= 1-tol_bound -> d[i] <= 0
    #   -1 <= d[i] <= 1 for all i
    
    A_ub = []
    b_ub = []
    
    for p in active_indices:
        s = p - (n - 1)
        g = gradient_lag(h, s)
        # G_p · d + gamma <= 0
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = 1.0  # coefficient of gamma
        A_ub.append(row)
        b_ub.append(0.0)
    
    # Box constraints: -1 <= d[i] <= 1
    # d[i] <= 1: row with 1 at position i
    # -d[i] <= 1: row with -1 at position i
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0)
        A_ub.append(-row)
        b_ub.append(1.0)
    
    # Bound-specific constraints
    for i in range(n):
        if h[i] <= tol_bound:
            # d[i] >= 0 -> -d[i] <= 0
            row = np.zeros(n + 1)
            row[i] = -1.0
            A_ub.append(row)
            b_ub.append(0.0)
        if h[i] >= 1.0 - tol_bound:
            # d[i] <= 0
            row = np.zeros(n + 1)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(0.0)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    c = np.zeros(n + 1)
    c[n] = -1.0  # minimize -gamma = maximize gamma
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * (n + 1), method='highs')
        if res.status == 0:
            d = res.x[:n]
            gamma = res.x[n]
            if gamma > 1e-6:
                return d, gamma
    except Exception:
        pass
    return None, 0.0

def line_search(h, d, max_alpha=0.5, n_trials=20):
    """Find best alpha in [0, max_alpha] such that h + alpha*d improves score."""
    best_alpha = 0.0
    best_score = official_score(h)
    for alpha in np.linspace(0.01, max_alpha, n_trials):
        h_try = make_feasible(h + alpha * d)
        sc = official_score(h_try)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
    return best_alpha, best_score

    stall = 0

        if alpha > 0 and sc < best_score - 1e-12:
            h = make_feasible(h + alpha * d)
            best_score = sc
            stall = 0
        else:

                    best_score = sc2

def pair_descent(h0, max_passes=20, tol=1e-10):
    """Coordinate pair descent: for each pair (i,j), find best transfer."""
    n = len(h0)
    h = h0.copy()
    best_score = official_score(h)
    
    for path in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                if h[i] <= 0 or h[j] >= 1:
                    continue
                # Try transferring from i to j or j to i
                for direction in [1, -1]:
                    # h[i] += delta * direction, h[j] -= delta * direction
                    for delta in [0.01, 0.05, 0.1, 0.2]:
                        di = delta * direction
                        dj = -delta * direction
                        if h[i] + di < -1e-9 or h[i] + di > 1 + 1e-9:
                            continue
                        if h[j] + dj < -1e-9 or h[j] + dj > 1 + 1e-9:
                            continue
                        h_try = h.copy()
                        h_try[i] += di
                        h_try[j] += dj
                        h_try = make_feasible(h_try)
                        sc = official_score(h_try)
                        if sc < best_score - tol:
                            h = h_try
                            best_score = sc
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_score

    n = len(h)
    dx = 2.0 / n

def compute_f(h):

def gradient_lag(h, s):
    n = len(h)
    g = np.zeros(n)
    i_min = max(0, -s)
    i_max = min(n-1, n-1-s)
    if i_min > i_max:
        return g
    for i in range(i_min, i_max+1):
        j = i + s
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return g

def make_feasible(z):
    n = len(z)

                if add <= 0: break

                residual -= add

    b_ub_list = []
    
    for p in active_indices:
