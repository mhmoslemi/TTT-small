def compute_f(h):

def subgradient_step(h, alpha, seed=42):
    n = len(h)
    vals = compute_f(h)
    p_max = np.argmax(vals)
    s = p_max - (n - 1)
    dx = 2.0 / n
    
    grad = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            grad[i] -= (1.0 - h[i]) * 0.0  # h[i+s]*(1-h[i]): deriv w.r.t h[i] is -h[i+s]
            grad[i + s] += (1.0 - h[i])   # deriv w.r.t h[i+s] is (1-h[i])
        # More carefully: c[p] = sum_i h[i+s](1-h[i]) for s>=0
        # d(c)/d(h[i]) = -h[i+s] (from term h[i+s]*(1-h[i])) + (1-h[i-s]) (from term h[i]*(1-h[i-s]) if i-s valid)
        # Let me redo: c[p] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
        # d/dh[k] = (1-h[k-s]) [if k-s >= 0 and k-s <= n-1-s, i.e., k >= s and k <= n-1] 
        #          - h[k] [if k >= s]  ... wait
        # Actually d/dh[k] of sum_{i=0}^{n-1-s} h[i+s]*(1-h[i]):
        #   If k appears as h[i+s]: i = k-s, valid if 0 <= k-s and k-s <= n-1-s i.e. s <= k <= n-1
        #     contribution: (1-h[k-s])
        #   If k appears as h[i]: i = k, valid if 0 <= k and k <= n-1-s i.e. 0 <= k <= n-1-s
        #     contribution: -h[k+s]
        grad = np.zeros(n)
        for i in range(n - s):
            # term h[i+s]*(1-h[i])
            # d/dh[i+s] = (1-h[i])
            grad[i + s] += (1.0 - h[i])
            # d/dh[i] = -h[i+s]
            grad[i] -= h[i + s]
    else:
        t = -s
        # c[p] = sum_{i=t}^{n-1} h[i-t]*(1-h[i])
        # d/dh[i] where i >= t: -h[i-t] (from h[i-t]*(1-h[i]))
        # d/dh[i-t] where i-t >= 0: (1-h[i]) (from h[i-t]*(1-h[i]))
        grad = np.zeros(n)
        for i in range(t, n):
            grad[i] -= h[i - t]
            grad[i - t] += (1.0 - h[i])
    
    grad *= dx
    h_new = h - alpha * grad
    h_new = make_feasible(h_new)
    return h_new

def active_direction(h, k_active=8, seed=42):
    n = len(h)
    vals = compute_f(h)
    top_idx = np.argsort(vals)[-k_active:]
    p_max_val = vals[top_idx[0]]
    dx = 2.0 / n
    
    # Build gradient for each active constraint
    # Solve LP: minimize t s.t. grad_p . d <= t - (vals[p] - p_max_val) for active p,
    # sum(d) = 0, -h[i]/h[i] <= d[i] <= (1-h[i])/... use box constraints on h+d
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    h0 = h
    for p in top_idx:
        s = p - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            for i in range(n - s):
                g[i + s] += (1.0 - h0[i])
                g[i] -= h0[i + s]
        else:
            t = -s
            for i in range(t, n):
                g[i] -= h0[i - t]
                g[i - t] += (1.0 - h0[i])
        g *= dx
        # Linearization: vals[p] + g.(h_new - h0) <= t_var + (vals[p] - p_max_val + 1e-10)
        # i.e., g.(h_new - h0) <= t_var - (p_max_val - vals[p]) + 1e-10
        # => g.(h_new - h0) + (p_max_val - vals[p]) - 1e-10 <= t_var
        # => g @ h_new - g @ h0 + (p_max_val - vals[p]) <= t_var + 1e-10
        # Simplify: we want the new value <= t. Approximate new value as vals[p] + g.(h_new-h0)
        # So: vals[p] + g @ (h_new - h0) <= t_var
        # => g @ h_new <= t_var - vals[p] + g @ h0
        rhs_const = g @ h0 - vals[p]
        constraints.append(cp.dot(g, h_var) <= t_var + rhs_const)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    if h_var.value is None:
        return None
    return make_feasible(h_var.value)

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))

def try_slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

            freq = 1.0 + 0.5 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            bias = (rng.random() - 0.5) * 0.3
            h_cand = cosine_init(n, freq=freq, phase=phase, bias=bias)
            sc_cand = official_score(h_cand)
            if sc_cand < best_score:
                update(h_cand, sc_cand)
            # Also try perturbing best
            sigma = 0.03 + 0.05 * rng.random()
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n))
            if remaining() > 15:
                h_opt, sc_opt = pair_descent(h_pert, max_rounds=15, grid_size=25,
                                             seed=seed + 300 + trial, time_limit=max(3, remaining() * 0.2))
                update(h_opt, sc_opt)
    
    # Phase 8: Subgradient polish
    if remaining() > 20:
        h_opt, sc_opt = subgradient_optimize(best_h, max_iter=100, seed=seed + 500,
                                             time_limit=max(5, remaining() * 0.3))
        update(h_opt, sc_opt)
    
    # Phase 9: Final intensive pair descent
    if remaining() > 15:
        plt = max(3, remaining() * 0.4)
        h_opt, sc_opt = pair_descent(best_h, max_rounds=50, grid_size=40,
                                     seed=seed + 999, time_limit=plt)
        update(h_opt, sc_opt)
    
    # Phase 10: Final SLSQP
    if remaining() > 10:
        h_new, sc = try_slsqp_smooth(best_h, 20000, maxiter=50)
        if h_new is not None:
            update(h_new, sc)
    
    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = final_score + 1e-13
    return final_h, final_bound, len(final_h)
