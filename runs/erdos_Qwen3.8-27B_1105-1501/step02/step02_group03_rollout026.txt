def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(h):
    """Ensure h is in [0,1] and sum(h) = n/2."""
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    # Bisection to find shift that makes sum correct
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14: break
    return h

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def sym_to_full(y, n):
    """Convert symmetric parameter y (length n/2) to full h of length n."""
    m = n // 2
    y = np.clip(y, 0.0, 1.0)
    h = np.zeros(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def full_to_sym(h, n):
    """Extract symmetric parameter from full h (best fit)."""
    m = n // 2
    # h[:m] and h[m:] should satisfy h[m+j] ≈ 1 - h[m-1-j]
    # Take average
    y = (h[:m] + (1.0 - h[m-1::-1])) / 2.0
    return y

def sym_vals(y, n):
    """Compute correlation values for symmetric h."""
    h = sym_to_full(y, n)
    return compute_vals(h)

def sym_grad(y, n):
    """Compute gradient of all correlation values w.r.t. y parameters."""
    h = sym_to_full(y, n)
    dx = 2.0 / n
    m = n // 2
    # We need dV_p/dy_j for all p, j
    # h_i for i < m: h_i = y_i
    # h_i for i >= m: h_i = 1 - y_{n-1-i}
    # So dh_i/dy_j = 1 if i = j < m, or if i >= m and n-1-i = j, i.e., i = n-1-j
    # So dh_i/dy_j = delta(i,j) + delta(i, n-1-j) for i,j in [0,m-1]
    # But wait: h_{n-1-j} = 1 - y_j, so dh_{n-1-j}/dy_j = -1
    
    # Let me be more careful.
    # h_i = y_i for i = 0, ..., m-1
    # h_i = 1 - y_{n-1-i} for i = m, ..., n-1
    # So for i >= m: h_i = 1 - y_{n-1-i}, meaning dh_i/dy_j = -delta(n-1-i, j)
    # i.e., dh_i/dy_j = -1 if n-1-i = j, i.e., i = n-1-j
    
    # V_p = dx * sum over appropriate range of h_i * (1 - h_{i+s})
    # where s = p - (n-1) is the shift
    
    # Let me just compute via the full gradient and then project
    # Full gradient: for V_p, dV_p/dh_i
    # Then dV_p/dy_j = sum_i (dV_p/dh_i) * (dh_i/dy_j)
    # = dV_p/dh_j + dV_p/dh_{n-1-j} * (-1) [if n-1-j >= m, which it is when j < m]
    
    # Actually let me just use the full gradient approach
    G = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        s = p - (n-1)
        if s >= 0:
            L = n - s
            G[p, :L] -= dx * h[s:]
            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K]
    
    # Project to y: dV/dy_j = G[:,j] - G[:,n-1-j] for j = 0,...,m-1
    # Wait: dh_j/dy_j = 1 (since h_j = y_j for j < m)
    # dh_{n-1-j}/dy_j = -1 (since h_{n-1-j} = 1 - y_j)
    # So dV/dy_j = dV/dh_j * 1 + dV/dh_{n-1-j} * (-1) = G[:,j] - G[:,n-1-j]
    
    Gy = np.zeros((2*n-1, m))
    for j in range(m):
        Gy[:, j] = G[:, j] - G[:, n-1-j]
    return Gy

def smooth_solve_sym(y0, n, beta, time_limit=20.0, max_iter=5000):

    def obj_grad(y):

        vals = sym_vals(y, n)
        cmax = vals.max()

        # Numerical stability
        shifted = np.clip(shifted, -100, 100)
        logsum = np.log(np.sum(np.exp(shifted)))
        obj = cmax + logsum / beta
        w = np.exp(shifted) / np.sum(np.exp(shifted))
        G = sym_grad(y, n)
        grad = w @ G
        return obj, grad
    
    y = y0.copy()

        return y

            break

def slsqp_sym(y0, n, t_init, max_iter=300, time_limit=30.0):
    """SLSQP on symmetric subspace: min t s.t. t >= V_p for all p."""
    m = n // 2
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def ineq(z):
        y = np.clip(z[:m], 1e-10, 1-1e-10)
        t = z[-1]
        vals = sym_vals(y, n)
        return t - vals
    
    def ineq_jac(z):
        y = np.clip(z[:m], 1e-10, 1-1e-10)
        G = sym_grad(y, n)
        # Inequality: t - V_p >= 0, gradient = [-dV_p/dy_j, 1]
        J = np.zeros((2*n-1, m+1))
        J[:, :m] = -G
        J[:, m] = 1.0
        return J
    
    z0 = np.concatenate([y0, [t_init]])
    bounds = [(1e-10, 1-1e-10)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq, 'jac': ineq_jac}]
    
    t0 = time.time()
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if time.time() - t0 > time_limit * 2:
            # SLSQP might not respect time, but let's check
            pass
        if res.x is not None and np.all(np.isfinite(res.x)):
            y_new = np.clip(res.x[:m], 0.0, 1.0)
            h_new = sym_to_full(y_new, n)
            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if np.isfinite(sc):

    except Exception:

    return None, None, None

def find_active(vals, frac=0.3, min_count=5):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-6)
    active = set(np.where(vals >= threshold)[0])
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def interpolate_y(y_src, n_src, n_tgt):
    """Interpolate symmetric parameter to new resolution."""
    m_src = n_src // 2
    m_tgt = n_tgt // 2
    src_c = (np.arange(m_src) + 0.5) * (1.0 / m_src)
    tgt_c = (np.arange(m_tgt) + 0.5) * (1.0 / m_tgt)
    y_int = np.interp(tgt_c, src_c, y_src)
    return np.clip(y_int, 0.0, 1.0)

def project_to_sym(h):
    """Project arbitrary h to symmetric subspace (best fit)."""
    n = len(h)
    m = n // 2
    # h[i] for i < m should equal y[i]
    # h[n-1-i] for i < m should equal 1 - y[i]
    # Best fit: y[i] = (h[i] + 1 - h[n-1-i]) / 2
    y = (h[:m] + 1.0 - h[n-1::-1][:m]) / 2.0
    return np.clip(y, 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return h

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_all(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                G[p, :L] -= dx * h[s:]
                G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            if K > 0:
                G[p, :K] += dx * (1.0 - h[t:])
                G[p, t:] -= dx * h[:K]
    return G

def smooth_solve(h0, beta, time_limit=20.0, max_iter=10000):
    """Projected gradient on smooth log-sum-exp objective."""
    n = len(h0)
    h = h0.copy()
    
    def obj_grad(h):
        vals = compute_vals(h)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        shifted = np.clip(shifted, -500, 500)
        max_shift = shifted.max()
        exp_shifted = np.exp(shifted - max_shift)
        s = exp_shifted.sum()
        if s < 1e-300:
            logsum = max_shift + np.log(len(vals))
            w = np.ones(len(vals)) / len(vals)
        else:
            logsum = max_shift + np.log(s)
            w = exp_shifted / s
        obj = cmax + logsum / beta
        G = grad_all(h)
        grad = w @ G
        return obj, grad
    
    f, g = obj_grad(h)
    gnorm2 = float(g @ g)
    if gnorm2 < 1e-30:
        return h
    alpha = 1.0 / (gnorm2 + 1e-12)
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        f, g = obj_grad(h)
        gnorm2 = float(g @ g)
        if gnorm2 < 1e-30:
            break
        step = alpha
        for _ in range(50):
            y = make_feasible(h - step * g)
            f_trial, _ = obj_grad(y)
            if f_trial <= f - 1e-5 * step * gnorm2:
                break
            step *= 0.4
        if step < 1e-14:
            break
        f_new, _ = obj_grad(y)
        if not np.isfinite(f_new):
            break
        h, f = y, f_new
        if f_new < f - 1e-12:
            alpha = min(alpha * 1.6, 20.0 / (gnorm2 + 1e-12))
    return h

def slsqp_active(h0, active_idx, t_init, max_iter=300, time_limit=30.0):
    """SLSQP with active constraints only."""
    n = len(h0)
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq(z):
        h = z[:n]
        t = z[-1]
        vals = compute_vals(h)
        return t - vals[active_idx]
    
    def ineq_jac(z):
        h = z[:n]
        G = grad_all(h)
        J = np.zeros((len(active_idx), n + 1))
        J[:, :n] = -G[active_idx, :]
        J[:, n] = 1.0
        return J
    
    def eq_fun(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    z0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def find_active(vals, gap=5e-5):
    """Find indices whose values are within 'gap' of the maximum."""
    cmax = vals.max()
    return sorted(np.where(vals >= cmax - gap)[0].tolist())
