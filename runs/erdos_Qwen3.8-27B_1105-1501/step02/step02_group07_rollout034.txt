```python
def pair_descent_step(h, best_score):
    """One pair descent step: pick random pair, find optimal d, update."""
    n = len(h)
    i, j = random pair
    d_min = max(-h[i], h[j] - 1)  # h[i]+d >= 0, h[j]-d <= 1
    d_max = min(1 - h[i], h[j])   # h[i]+d <= 1, h[j]-d >= 0
    if d_min >= d_max: return h, best_score

    V = compute_values(h)  # or pass in
    delta = compute_delta(h, i, j)  # O(n)

    d_opt = find_optimal_d(V, delta, d_min, d_max)
```

```python
def find_optimal_d(V, delta, d_min, d_max):
    """Find d in [d_min, d_max] minimizing max_k (V[k] + d * delta[k])."""
    if d_min > d_max:
        return 0.0
    if d_min == d_max:
        return d_min
    
    # f(d) = max_k (V[k] + d * delta[k]) is convex piecewise linear
    # Use bisection on the subgradient
    
    def subgrad(d):
        val = V + d * delta
        f_val = val.max()
        active = np.where(np.abs(val - f_val) < 1e-12)[0]
        return delta[active].min(), delta[active].max(), f_val
    
    lo, hi = d_min, d_max
    for _ in range(60):
        mid = (lo + hi) / 2.0
        sg_min, sg_max, _ = subgrad(mid)
        if sg_min > 1e-10:
            hi = mid  # f is increasing, go left
        elif sg_max < -1e-10:
            lo = mid  # f is decreasing, go right
        else:
            return mid  # 0 is in subgradient
    return (lo + hi) / 2.0
```

```python
import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_V(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - z.sum()
    if abs(residual) < 1e-14:
        return z
    if residual > 0:
        order = np.argsort(1.0 - z)
    else:
        order = np.argsort(z)
    for idx in order:
        if residual > 0:
            a = min(residual, 1.0 - z[idx])
            z[idx] += a
            residual -= a
        else:
            a = min(-residual, z[idx])
            z[idx] -= a
            residual += a
        if abs(residual) < 1e-15:
            break
    return np.clip(z, 0.0, 1.0)

def compute_delta(h, i, j):

        cj = 0.0
```

```python
import numpy as np
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_V(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    residual = target - z.sum()
    if abs(residual) < 1e-14:
        return z
    if residual > 0:
        order = np.argsort(1.0 - z)
    else:
        order = np.argsort(z)
    for idx in order:
        if residual > 0:
            a = min(residual, 1.0 - z[idx])
            z[idx] += a
            residual -= a
        else:
            a = min(-residual, z[idx])
            z[idx] -= a
            residual += a
        if abs(residual) < 1e-15:
            break
    return np.clip(z, 0.0, 1.0)

def compute_single_delta(h, pos):
    """Compute ∂V/∂h[pos] for all lags."""
    n = len(h)
    d = np.zeros(2 * n - 1)
    for offset in range(2 * n - 1):
        s = offset - (n - 1)
        c = 0.0
        if s >= 0:
            if s <= pos:
                c += (1.0 - h[pos - s])
            if pos + s < n:
                c -= h[pos + s]
        else:
            t = -s
            if pos + t < n:
                c += (1.0 - h[pos + t])
            if pos - t >= 0:
                c -= h[pos - t]
        d[offset] = c
    return d

def compute_delta_pair(h, i, j):
    """delta = ∂V/∂h[i] - ∂V/∂h[j]"""
    return compute_single_delta(h, i) - compute_single_delta(h, j)

def find_optimal_d(V, delta, d_min, d_max):
    """Find d in [d_min, d_max] minimizing max_k (V[k] + d*delta[k])."""
    if d_min >= d_max:
        return d_min
    lo, hi = d_min, d_max
    for _ in range(50):
        mid = (lo + hi) / 2.0
        val = V + mid * delta
        f_val = val.max()
        active = val >= f_val - 1e-12
        sg_min = delta[active].min()
        sg_max = delta[active].max()
        if sg_min > 1e-10:
            hi = mid
        elif sg_max < -1e-10:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

def pair_descent(h, rng, max_iters, deadline):
    """Run pair descent. Returns (best_h, best_score)."""
    n = len(h)
    best_score = official_score(h)
    best_h = h.copy()
    no_improve = 0
    consecutive_no_improve_limit = 300
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d_min = max(-h[i], h[j] - 1.0)
        d_max = min(1.0 - h[i], h[j])
        if d_min >= d_max - 1e-12:
            no_improve += 1
            if no_improve > consecutive_no_improve_limit:
                break
            continue
        V = compute_V(h)
        delta = compute_delta_pair(h, i, j)
        d_opt = find_optimal_d(V, delta, d_min, d_max)
        if abs(d_opt) < 1e-14:
            no_improve += 1
            if no_improve > consecutive_no_improve_limit:
                break
            continue
        h[i] += d_opt
        h[j] -= d_opt
        sc = official_score(h)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] -= d_opt
            h[j] += d_opt
            no_improve += 1
            if no_improve > consecutive_no_improve_limit:
                break
    return best_h, best_score

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_int = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_int)
```
