    a, b = lo, hi
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def build_grad_row(h, s, n):
    g = np.zeros(n)
    p1_lo = max(0, s)
    p1_hi = min(n - 1, s + n - 1)
    if p1_lo <= p1_hi:
        p1 = np.arange(p1_lo, p1_hi + 1)
        g[p1] = (1.0 - h[p1 - s])
    p2_lo = max(0, -s)
    p2_hi = min(n - 1, n - 1 - s)
    if p2_lo <= p2_hi:
        p2 = np.arange(p2_lo, p2_hi + 1)
        g[p2] -= h[p2 + s]
    return g


def lp_descent(h, trust_radius, max_active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = float(np.max(corr))
    off = n - 1
    lags = np.arange(-(n - 1), n)
    margin = 0.015 * max(0.5, max_corr)
    active_mask = corr >= max_corr - margin
    active_indices = np.where(active_mask)[0]
    if len(active_indices) > max_active:
        vals = corr[active_indices]
        order = np.argsort(-vals)[:max_active]
        active_indices = active_indices[order]
    if len(active_indices) == 0:
        return None
    active_lags = lags[active_indices]
    K = len(active_lags)
    slacks = max_corr - corr[active_indices]
    slacks = np.maximum(slacks, 0.0)
    G = np.zeros((K, n))
    for r in range(K):
        G[r] = build_grad_row(h, int(active_lags[r]), n)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.hstack([G, np.ones((K, 1))])
    b_ub = slacks + 1e-9
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo_d = np.maximum(-trust_radius, -h)
    hi_d = np.minimum(trust_radius, 1.0 - h)
    bounds = [(float(lo_d[i]), float(hi_d[i])) for i in range(n)]
    bounds.append((0.0, float(max_corr + 1.0)))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    d = res.x[:n]
    if np.max(np.abs(d)) < 1e-9:
        return None
    return d


def subgradient_step(h, lr, temp=0.05):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = float(np.max(corr))
    exp_vals = np.exp((corr - max_corr) / temp)
    weights = exp_vals / np.sum(exp_vals)
    off = n - 1
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for i, s in enumerate(lags):
        w = weights[off + i]
        if w < 1e-12:
            continue
        grad += w * build_grad_row(h, int(s), n)
    h_new = h - lr * grad
    h_new = project_onto_constraint(h_new, n / 2.0)
    return h_new


def try_line_search(h, d, c5_cur, alphas):
    for a in alphas:
        h_new = h + a * d
        if np.any(h_new < -1e-9) or np.any(h_new > 1.0 + 1e-9):
            continue
        h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-13:
            return h_new, c5_new
    return None, c5_cur


def do_pair_polish(h, rng, n, num_pairs):
    for _ in range(num_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if i > j:
            i, j = j, i
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta
    return h
