import numpy as np

import cvxpy as cp

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_grads(h, p_list):
    """Compute gradients of lag values for given lag indices."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((len(p_list), n))
    for i, p in enumerate(p_list):
        k = p - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] += (-h[k:])
        else:
            q = -k
            g[:n-q] += (1.0 - h[q:])
            g[q:] += (-h[:n-q])
        grads[i] = dx * g
    return grads

def qp_direction(h, top_k=None, mu=100.0, solver='SCS'):
    """Solve QP trust region subproblem."""
    n = len(h)
    dx = 2.0 / n
    npts = 2 * n - 1
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    C = vals.max()
    
    if top_k is not None and top_k < npts:
        idx = np.argsort(vals)[::-1][:top_k]
    else:
        idx = np.arange(npts)
    
    vals_sel = vals[idx]
    grads = compute_grads(h, list(idx))
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [cp.sum(d) == 0, d >= -h + 1e-6, d <= 1 - h - 1e-6]
    # t >= vals[p] + grads[p] @ d  =>  grads[p] @ d - t <= -vals[p]
    for i in range(len(idx)):
        constraints.append(grads[i] @ d - t <= -vals_sel[i])
    
    objective = cp.Minimize(t + mu * cp.sum_squares(d))
    prob = cp.Problem(objective, constraints)
    
    try:

        if d.value is not None and np.all(np.isfinite(d.value)):

    except Exception:
        pass
    return None

    best_alpha = 0.0

        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def get_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grads(h, p_list):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((len(p_list), n))
    for i, p in enumerate(p_list):
        k = int(p) - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nk = n - k
            g[k:] += (1.0 - h[:nk])
            g[:nk] += (-h[k:])
        else:
            q = -k
            nq = n - q
            g[:nq] += (1.0 - h[q:])
            g[q:] += (-h[:nq])
        grads[i] = dx * g
    return grads

def solve_qp_subproblem(h, mu, top_k, solver):
    n = len(h)
    npts = 2 * n - 1
    vals = get_lag_values(h)
    if top_k is not None and top_k < npts:
        idx = np.argsort(vals)[::-1][:top_k]
    else:
        idx = np.arange(npts)
    vals_sel = vals[idx]
    grads = compute_grads(h, list(idx))
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(d) == 0]
    constraints.append(d >= -h + 1e-9)
    constraints.append(d <= 1.0 - h - 1e-9)
    for i in range(len(idx)):
        constraints.append(grads[i] @ d - t <= -vals_sel[i])
    obj = cp.Minimize(t + mu * cp.sum_squares(d))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, max_iters=20000)
        if d.value is not None and np.all(np.isfinite(d.value)):
            return np.asarray(d.value, dtype=float)
    except Exception:
        pass
    return None

def solve_lp_direction(h, top_k):
    n = len(h)
    npts = 2 * n - 1
    vals = get_lag_values(h)
    if top_k is not None and top_k < npts:
        idx = np.argsort(vals)[::-1][:top_k]
    else:
        idx = np.arange(npts)
    vals_sel = vals[idx]
    grads = compute_grads(h, list(idx))
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(d) == 0]
    constraints.append(d >= -h)
    constraints.append(d <= 1.0 - h)
    for i in range(len(idx)):
        constraints.append(grads[i] @ d - t <= -vals_sel[i])
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=20000)
        if d.value is not None and np.all(np.isfinite(d.value)):
            return np.asarray(d.value, dtype=float)
    except Exception:
        pass
    return None

def line_search(h, d, best_sc):
    alphas = [1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.02, 0.005]
    best_h = None
    best_s = best_sc
    for a in alphas:
        h_t = make_feasible(h + a * d)
        sc = official_score(h_t)
        if sc < best_s:
            best_s = sc
            best_h = h_t
    return best_h, best_s

def qp_iterate(h, deadline, max_rounds=40):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    mu_schedule = [200, 80, 30, 10, 4, 1.5, 0.5]
    k_schedule = [25, 15, 30]
    stuck_count = 0
    for it in range(max_rounds):
        if time.time() >= deadline:
            break
        mu = mu_schedule[min(it // 6, len(mu_schedule) - 1)]
        k = k_schedule[it % len(k_schedule)]
        d = solve_qp_subproblem(best, mu, k, cp.SCS)
        if d is None or np.linalg.norm(d) < 1e-12:
            stuck_count += 1
            if stuck_count > 6:
                break
            continue
        h_new, sc = line_search(best, d, best_sc)
        if h_new is not None and sc < best_sc - 1e-13:
            best = h_new
            best_sc = sc
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 8:
                break
    return best, best_sc

def lp_iterate(h, deadline, max_rounds=30):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    k_schedule = [20, 30, 15, 40]
    for it in range(max_rounds):
        if time.time() >= deadline:
            break
        k = k_schedule[it % len(k_schedule)]
        d = solve_lp_direction(best, k)
        if d is None or np.linalg.norm(d) < 1e-12:
            if it > 5:
                break
            continue
        h_new, sc = line_search(best, d, best_sc)
        if h_new is not None and sc < best_sc - 1e-13:
            best = h_new
            best_sc = sc
        else:
            if it > 8:
                break
    return best, best_sc

def two_point_search(h, n_tries, rng, deadline):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if time.time() >= deadline:
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        cands = [(t_lo + t_hi) / 2.0, t_lo + 0.25 * span, t_lo + 0.75 * span,
                 t_lo + 0.15 * span, t_lo + 0.85 * span]
        best_t = None
        best_local = best_sc
        for t in cands:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-12 or va > 1 + 1e-12 or vb < -1e-12 or vb > 1 + 1e-12:
                continue
            tmp = best.copy()
            tmp[a] = va
            tmp[b] = vb
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = official_score(best)
    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_wave(n, f1, a1, f2, a2, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + phase) + a2 * np.cos(2 * np.pi * f2 * x)
    return make_feasible(h)

def symmetrize(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def subgradient_polish(h, n_iter, deadline):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    alpha = 0.05
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        vals = get_lag_values(best)
        cmax = vals.max()
        tol = 0.03
        active = np.where(vals >= cmax - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        w = np.exp(80.0 * (vals[active] - cmax))
        w = w / w.sum()
        grads = compute_grads(best, list(active))
        grad = np.zeros(n)
        for i, p in enumerate(active):
            grad += w[i] * grads[i]
        d = -grad
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25]:
            h_t = make_feasible(best + mult * d)
            sc = official_score(h_t)
            if sc < best_sc - 1e-14:
                best = h_t
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 5e-5:
                break
        else:
            alpha = min(alpha * 1.2, 0.3)
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(25.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, 2, 0.35, 3, 0.12, 0.3)
        best_score = official_score(best_h)

    n = len(best_h)

    # Try symmetrization
    if time.time() < deadline:
        h_sym = symmetrize(best_h)
        sc_sym = official_score(h_sym)
        pool_states = [(best_h, best_score)]
        if sc_sym < best_score:
            best_h, best_score = h_sym, sc_sym
        else:
            pool_states.append((h_sym, sc_sym))

    # Phase 1: QP trust-region optimization
    if time.time() < deadline:
        h1, s1 = qp_iterate(best_h, min(deadline, time.time() + 200), max_rounds=50)
        if s1 < best_score:
            best_h, best_score = h1, s1
        pool_states.append((h1, s1))

    # Phase 2: LP direction finding
    if time.time() < deadline:
        h2, s2 = lp_iterate(best_h, min(deadline, time.time() + 100), max_rounds=25)
        if s2 < best_score:
            best_h, best_score = h2, s2

    # Phase 3: Two-point exchange
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_tries = int(remaining * 3)
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Phase 4: Subgradient polish
    if time.time() < deadline:
        best_h, best_score = subgradient_polish(best_h, 150, min(deadline, time.time() + 60))

    # Phase 5: Resolution search - try multiple n values
    if time.time() < deadline:
        n_best = len(best_h)
        candidates = set()
        for dn in [-10, -6, -4, -2, 2, 4, 6, 10, 14, -14]:
            nt = n_best + dn
            if 24 <= nt <= 120:
                candidates.add(nt)
        for nc in [32, 40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80]:
            if 24 <= nc <= 120:
                candidates.add(nc)

        for nt in sorted(candidates):
            if time.time() >= deadline:
                break
            if nt == len(best_h):
                continue
            h_int = interp_to(best_h, nt)
            sc_int = official_score(h_int)
            if sc_int < best_score:
                best_h, best_score = h_int, sc_int
            # Quick QP at this resolution
            if time.time() < deadline:
                time_per = (deadline - time.time()) / len(candidates - {nt})
                min_t = max(30, time_per)
                h_q, s_q = qp_iterate(h_int, min(deadline, time.time() + min_t), max_rounds=15)
                if s_q < best_score:
                    best_h, best_score = h_q, s_q

    # Phase 6: Back to best resolution, intensive QP
    if time.time() < deadline:
        h3, s3 = qp_iterate(best_h, min(deadline, time.time() + 120), max_rounds=40)
        if s3 < best_score:
            best_h, best_score = h3, s3

    # Phase 7: More two-point
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_tries = max(2000, int(remaining * 4))
        best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    # Phase 8: Final QP + subgradient
    if time.time() < deadline:
        h4, s4 = qp_iterate(best_h, min(deadline, time.time() + 60), max_rounds=20)
        if s4 < best_score:
            best_h, best_score = h4, s4
        if time.time() < deadline:
            h5, s5 = subgradient_polish(best_h, 100, min(deadline, time.time() + 30))
            if s5 < best_score:
                best_h, best_score = h5, s5

    # Phase 9: Try random perturbations
    if time.time() < deadline:
        n_b = len(best_h)
        for sigma in [0.05, 0.1, 0.15]:
            if time.time() >= deadline:
                break
            h_pert = make_feasible(best_h + rng.normal(0, sigma, size=n_b))
            h_new, sc = two_point_search(h_pert, 800, rng, deadline)
            if sc < best_score:
                best_h, best_score = h_new, sc

    # Phase 10: Final intensive two-point + QP
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 30:
            n_tries = int(remaining * 5)
            best_h, best_score = two_point_search(best_h, n_tries, rng, deadline)

    if time.time() < deadline - 10:
        h6, s6 = qp_iterate(best_h, min(deadline, time.time() + 30), max_rounds=10)
        if s6 < best_score:
            best_h, best_score = h6, s6

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(np.isfinite(final_h))
    assert final_h.min() >= -1e-10
    assert final_h.max() <= 1.0 + 1e-10
    assert abs(final_h.sum() - final_n / 2.0) < 1e-10

    return final_h, final_score, final_n
