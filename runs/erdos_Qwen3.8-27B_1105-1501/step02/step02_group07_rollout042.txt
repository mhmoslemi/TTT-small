        hi = min(delta, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            h_new = make_feasible(h + d)
            return h_new, official_score(h_new)
    except Exception:
        pass
    return None, None

def weighted_grad(h, temp):
    n = len(h)
    V = compute_all(h)
    Vm = V.max()
    x = np.clip((V - Vm) / temp, -60.0, 0.0)
    w = np.exp(x)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w /= ws
    G = all_gradients(h)
    return (w[:, None] * G).sum(axis=0)

def nesterov_pg(h, prev_h, t_iter, temp, lr):
    n = len(h)
    if t_iter > 1.0:
        beta = (t_iter - 1.0) / (t_iter + 1.0)
    else:
        beta = 0.0
    y = h + beta * (h - prev_h)
    grad = weighted_grad(y, temp)
    return make_feasible(y - lr * grad)

def coordinate_polish(h, n_steps, rng):
    n = len(h)
    best_sc = official_score(h)
    deltas = np.array([0.003, -0.003, 0.007, -0.007, 0.014, -0.014,
                       0.028, -0.028, 0.005, -0.005])
    for _ in range(n_steps):
        idx = rng.integers(n)
        for d in deltas:
            h_trial = h.copy()
            val = h[idx] + d
            if val < 0.0 or val > 1.0:
                continue
            h_trial[idx] = val
            h_trial = make_feasible(h_trial)
            sc = official_score(h_trial)
            if sc < best_sc - 1e-15:
                h = h_trial
                best_sc = sc
                break
    return h, best_sc

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_x = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_x = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_x, src_x, h_src)
    return make_feasible(h_int)

def trust_region_phase(h_start, budget_s, seed):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    n = len(h_start)
    h_cur = h_start.copy()
    best_h = h_start.copy()
    best_sc = official_score(h_start)
    delta = 0.12
    stuck = 0
    iter_count = 0
    while time.time() - t0 < budget_s:
        h_new, sc_new = trust_region_lp(h_cur, delta)
        iter_count += 1
        if h_new is not None:
            cur_sc = official_score(h_cur)
            if sc_new < best_sc - 1e-14:
                best_h = h_new.copy()
                best_sc = sc_new
                h_cur = h_new
                delta = min(delta * 1.12, 0.35)
                stuck = 0
            elif sc_new < cur_sc - 1e-14:
                h_cur = h_new
                stuck = 0
            else:
                h_cur = h_new
                delta *= 0.55
                stuck += 1
        else:
            delta *= 0.5
            stuck += 1
        if stuck > 25 or delta < 0.0003:
            h_cur = make_feasible(best_h + rng.normal(0, 0.015, n))
            delta = 0.08
            stuck = 0
    return best_h, best_sc

def nesterov_phase(h_start, budget_s, seed):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    n = len(h_start)
    h_cur = h_start.copy()
    prev_h = h_start.copy()
    best_h = h_start.copy()
    best_sc = official_score(h_start)
    t_iter = 1.0
    temp = 8.0
    lr = 0.06
    stall = 0
    while time.time() - t0 < budget_s:
        t_iter += 1.0
        h_new = nesterov_pg(h_cur, prev_h, t_iter, temp, lr)
        prev_h = h_cur
        h_cur = h_new
        sc = official_score(h_cur)
        if sc < best_sc - 1e-13:
            best_h = h_cur.copy()
            best_sc = sc
            stall = 0
        else:
            stall += 1
        temp = min(temp * 1.003, 80.0)
        if stall > 80:
            h_cur = make_feasible(best_h + rng.normal(0, 0.01, n))
            prev_h = h_cur.copy()
            t_iter = 1.0
            temp = 5.0
            lr = 0.05
            stall = 0
    return best_h, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = 12.0
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def time_left():
        return deadline - time.time()

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
    except (NameError, AttributeError):
        h0 = np.full(60, 0.5)

    h0 = make_feasible(h0)
    best_h = h0.copy()
    best_sc = official_score(h0)
    n0 = len(h0)

    # Phase 1: Trust region LP (main optimization)
    phase1_budget = min(350.0, max(100.0, time_left() * 0.35))
    h1, sc1 = trust_region_phase(best_h, phase1_budget, seed)
    if sc1 < best_sc:
        best_h = h1
        best_sc = sc1

    # Phase 2: Nesterov gradient
    if time_left() > 200:
        phase2_budget = min(200.0, max(50.0, time_left() * 0.25))
        h2, sc2 = nesterov_phase(best_h, phase2_budget, seed + 1)
        if sc2 < best_sc:
            best_h = h2
            best_sc = sc2

    # Phase 3: More trust region on the improved solution
    if time_left() > 150:
        phase3_budget = min(180.0, max(30.0, time_left() * 0.2))
        h3, sc3 = trust_region_phase(best_h, phase3_budget, seed + 2)
        if sc3 < best_sc:
            best_h = h3
            best_sc = sc3

    # Phase 4: Coordinate polish
    if time_left() > 80:
        n_steps = 30000
        h4, sc4 = coordinate_polish(best_h, n_steps, rng)
        if sc4 < best_sc:
            best_h = h4
            best_sc = sc4
        # Second round of polish
        if time_left() > 40:
            h4b, sc4b = coordinate_polish(best_h, 20000, rng)
            if sc4b < best_sc:
                best_h = h4b
                best_sc = sc4b

    # Phase 5: Try different n values
    n_best = len(best_h)
    n_candidates = sorted(set([n_best - 8, n_best - 4, n_best + 4,
                               n_best + 8, n_best + 12, n_best - 12]))
    for n_try in n_candidates:
        if time_left() < 50 or n_try < 16:
            break
        h_int = interpolate_to(best_h, n_try)
        sc_int = official_score(h_int)
        if sc_int < best_sc:
            best_h = h_int.copy()
            best_sc = sc_int
            n_best = n_try
        # Quick trust region at this n
        h_cur = h_int.copy()
        delta = 0.08
        stuck = 0
        local_best = h_cur.copy()
        local_sc = sc_int
        for _ in range(300):
            if time_left() < 30:
                break
            h_new, sc_new = trust_region_lp(h_cur, delta)
            if h_new is not None:
                if sc_new < local_sc - 1e-14:
                    local_best = h_new.copy()
                    local_sc = sc_new
                    h_cur = h_new
                    delta = min(delta * 1.1, 0.25)
                    stuck = 0
                else:
                    h_cur = h_new
                    delta *= 0.6
                    stuck += 1
            else:
                delta *= 0.5
                stuck += 1
            if stuck > 20:
                h_cur = make_feasible(local_best + rng.normal(0, 0.01, n_try))
                delta = 0.06
                stuck = 0
        if local_sc < best_sc:
            best_h = local_best.copy()
            best_sc = local_sc

    # Phase 6: Final trust region polish at best n
    if time_left() > 25:
        final_budget = max(5.0, time_left() * 0.5)
        h6, sc6 = trust_region_phase(best_h, final_budget, seed + 10)
        if sc6 < best_sc:
            best_h = h6
            best_sc = sc6

    # Phase 7: Final coordinate polish
    if time_left() > 10:
        h7, sc7 = coordinate_polish(best_h, 15000, rng)
        if sc7 < best_sc:
            best_h = h7
            best_sc = sc7

    final_h = make_feasible(best_h)
    final_sc = official_score(final_h)
    return final_h, final_sc + 1e-12, len(final_h)
