    n = len(h)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0

def _exact_c5(h):
    n = len(h)

    return float(np.max(corr))

def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

            break

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _make_balanced(h_real, n):
    target = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(-h_real)
    h_bin[idx[:target]] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, max_passes=10000):
    n = len(h_bin)
    h = h_bin.copy()
    best_max = _corr_max(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_new_max = best_max
        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nm = _corr_max(h)
                h[i] = 1.0
                h[j] = 0.0
                if nm < best_new_max - 1e-12:
                    best_new_max = nm
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            break
        
        h[best_i] = 0.0
        h[best_j] = 1.0
        best_max = best_new_max
    
    return h, best_max

    """Annealing + local search hybrid."""
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_max = _corr_max(h)
    cur_max = best_max
    T = 5.0
    
    it = 0
    max_it = 50000
    while time.time() < deadline and it < max_it:
        it += 1
        if T < 0.01:
            # Do local search from current state
            h, cur_max = _binary_local_search(h, min(deadline, time.time() + 5), max_passes=2000)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            break
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_max = _corr_max(h)
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= 0.9995
    
    # Final local search from best
    h_final, max_final = _binary_local_search(best_h, deadline, max_passes=5000)
    if max_final < best_max:
        best_max = max_final
        best_h = h_final
    
    return best_h, best_max

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    target = n / 2.0

            if d >= 0:
                for j in range(n - d):
                    grad[j] += (1.0 - h[j + d])
                    grad[j + d] += (-h[j])
            else:
                e = -d
                for j in range(n - e):
                    grad[j + e] += (1.0 - h[j])
                    grad[j] += (-h[j + e])

            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.5)
        else:
            lr = max(lr * 0.97, 0.001)
        
        h = h_new
    
    return best_h, best_c5
