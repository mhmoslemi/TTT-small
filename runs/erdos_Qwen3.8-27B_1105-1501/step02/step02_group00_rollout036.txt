def _project(v, target):

        return np.ones(n)
    lo = float(v.min()) - 2.0

        else:

        h_c, c5_c = _coord_polish(best_h, min(deadline_total - 5, time.time() + time_rem * 0.4))
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
            best_n = len(h_c)
    
    # Phase 7: Final local polish with fine steps
    time_rem = deadline_total - time.time()
    if time_rem > 5:
        h_f, c5_f = _local_loop(best_h, deadline_total,
                                 (50, 120, 250, 500), 0.05, 5000,
                                 np.random.default_rng(seed + 7777))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(h_f)
    
    # Final feasibility check
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
