import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_gradient(h):
    n = len(h)
    offset = n - 1
    G = np.zeros((n, 2 * n - 1))
    for a in range(n):
        # First term: 1-h[a-d] for d in [max(-a, -(n-1)), min(n-1-a, n-1)]
        # i.e., k in [a, a+n-1]
        k_lo = max(0, a)
        k_hi = min(2 * n - 2, a + n - 1)
        if k_lo <= k_hi:
            d = np.arange(k_lo, k_hi + 1) - offset
            idx = a - d  # = a - (k - offset) = a + offset - k
            G[a, k_lo:k_hi + 1] += (1.0 - h[idx])
        
        # Second term: -h[a+d] for d in [max(-a, -(n-1)), min(n-1-a, n-1)]
        # i.e., k in [n-1-a, 2n-2-a]
        k_lo2 = max(0, n - 1 - a)
        k_hi2 = min(2 * n - 2, 2 * n - 2 - a)
        if k_lo2 <= k_hi2:
            d = np.arange(k_lo2, k_hi2 + 1) - offset
            idx = a + d  # = a + (k - offset) = a + k - offset
            G[a, k_lo2:k_hi2 + 1] -= h[idx]
    return G

def line_objective(corr, G_i, G_j, t, offset, k1, k2):
    """Compute max of corr + t*(G_i - G_j) + quadratic corrections"""
    base = corr + t * (G_i - G_j)
    base[offset] += -2.0 * t * t
    base[k1] += t * t
    if k2 != k1:
        base[k2] += t * t
    return float(np.max(base))

def ternary_search_line(corr, G_i, G_j, offset, k1, k2, t_lo, t_hi, n_iters=30):
    """Ternary search to minimize line objective over [t_lo, t_hi]"""
    a, b = t_lo, t_hi
    for _ in range(n_iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = line_objective(corr, G_i, G_j, m1, offset, k1, k2)
        f2 = line_objective(corr, G_i, G_j, m2, offset, k1, k2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    f_opt = line_objective(corr, G_i, G_j, t_opt, offset, k1, k2)
    return t_opt, f_opt

def pair_sweep(h, corr, G, rng, n_iters=30):
    """One full sweep of pair optimization"""
    n = len(h)
    offset = n - 1
    n_best = np.max(corr)
    perm = rng.permutation(n)
    improved = False
    for a in range(n):
        for b in range(a + 1, n):
            i, j = perm[a], perm[b]
            # Transfer: h[i] += t, h[j] -= t
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-12:
                continue
            k1 = i - j + offset
            k2 = j - i + offset
            k1 = max(0, min(2 * n - 2, k1))
            k2 = max(0, min(2 * n - 2, k2))
            t_opt, f_opt = ternary_search_line(corr, G[i], G[j], offset, k1, k2, t_lo, t_hi, n_iters)
            if f_opt < n_best - 1e-10:
                # Accept: apply the move
                h[i] += t_opt
                h[j] -= t_opt
                # Update corr
                corr = np.correlate(h, 1.0 - h, mode="full")
                n_best = np.max(corr)
                # Rebuild G (expensive but safe)
                G = build_gradient(h)
                improved = True
    return h, corr, n_best, improved

def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_quadratic_residue(n, rng):

    # Find smallest prime >= n
    p = n
    while not is_prime(p):
        p += 1
    h = np.zeros(p)

    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == 1:
            h[a] = 1.0
    # If p != n, interpolate down to n

        new_x = np.linspace(0, 2, n)
        h = np.interp(new_x, old_x, h)
        h = project_onto_constraint(h, n / 2.0)
    return h

    if num < 2:
        return False
    if num < 4:
        return True
    if num % 2 == 0 or num % 3 == 0:
        return False
    i = 5
    while i * i <= num:
        if num % i == 0 or num % (i + 2) == 0:
            return False
        i += 6
    return True

def optimize_single(h, rng, budget_s, n_iters=30):
    """Optimize a single h using pair sweeps"""
    n = len(h)
    h = h.copy()
    h = project_onto_constraint(h, n / 2.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    G = build_gradient(h)

                corr = np.correlate(h, 1.0 - h, mode="full")
                G = build_gradient(h)
                stall = 0
        else:
            stall = 0
    c5 = float(np.max(corr) * (2.0 / n))
    return h, c5

Now let me think about the line search more carefully.

For pair (i,j) with transfer h[i]+=t, h[j]-=t:
- k1 = i-j+offset (lag of pair (i,j))
- k2 = j-i+offset (lag of pair (j,i))

The quadratic corrections:
- At k=offset: -2t² (from self-interactions of i and j)
- At k1: +t² (from the cross term in pair (i,j))
- At k2: +t² (from the cross term in pair (j,i))

But wait, what if k1 = offset? That means i-j+offset = offset, so i=j. Excluded.
What if k1 = k2? That means i-j = j-i, so i=j. Excluded.
What if k1 = 2offset - k2? That's always true: k1+k2 = 2offset. So they're symmetric around offset.

Can k1 equal offset? Only if i=j. Can k2 equal offset? Only if i=j. So for i≠j, all three indices (offset, k1, k2) are distinct.

Wait, what if k1 = 0 or k1 = 2n-2 (boundary)? That's fine, we just add t² to that element.

Let me now also handle the edge case where k1 or k2 might be out of bounds [0, 2n-2]. Since i,j ∈ [0,n-1]:
k1 = i-j+offset. Min: 0-(n-1)+(n-1) = 0. Max: (n-1)-0+(n-1) = 2n-2.
So k1 is always in [0, 2n-2]. Similarly for k2. Good.

Now, one more optimization: instead of building the full G matrix (n × (2n-1)), I can compute G[i,:] and G[j,:] on the fly for each pair. This is O(n) per row, so O(n) per pair evaluation. But building the full matrix is O(n²) and I only need to rebuild it when h changes. Since I rebuild after each accepted move, and accepted moves are relatively rare compared to rejected ones, it's better to build the full matrix once per accepted move and then use it for all subsequent pair evaluations until the next acceptance.
