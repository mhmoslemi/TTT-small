def lag_grad(h, p):
    """Compute gradient of lag value at correlation index p w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s  # number of terms
        # First part: (1-h[i-s]) for i in s..n-1
        # Second part: -h[i+s] for i in 0..n-1-s
        grad[s:] += (1.0 - h[:m])  # i-s ranges 0..m-1 as i ranges s..s+m-1=n-1
        grad[:m] -= h[s:]          # i+s ranges s..s+m-1=n-1 as i ranges 0..m-1

        # Second part: -h[i-t] for i in t..n-1

        grad[t:] -= h[:m]               # i-t ranges 0..m-1 as i ranges t..t+m-1=n-1
    return dx * grad

import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).copy().ravel()
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Greedy residual repair
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add > 1e-15:
                    h[idx] += add
                    residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub > 1e-15:
                    h[idx] -= sub
                    residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[s:] += (1.0 - h[:m])
        grad[:m] -= h[s:]
    else:
        t = -s
        m = n - t
        grad[:m] += (1.0 - h[t:])
        grad[t:] -= h[:m]
    return dx * grad

def all_lag_grads(h, active_indices):
    n = len(h)
    G = np.zeros((len(active_indices), n))
    for i, p in enumerate(active_indices):
        G[i] = lag_grad(h, p)
    return G

def solve_lp_direction(h, active_p, active_vals, rho):
    n = len(h)
    K = len(active_p)
    if K == 0:
        return None
    G = all_lag_grads(h, active_p)  # K x n
    
    # Variables: u_0, ..., u_{n-1}, t (total n+1)
    # Objective: minimize t → c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: G @ u - t <= -active_vals
    # A_ub shape: K x (n+1)
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -active_vals
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free (but bounded by constraints)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            u = res.x[:n]
            t_opt = res.x[n]
            if np.all(np.isfinite(u)) and np.linalg.norm(u) > 1e-12:
                return u, t_opt
    except Exception:
        pass
    return None, None

def lp_refine(h_init, time_limit, rng, max_stall=50):
    h = make_feasible(h_init)
    best = h.copy()
    best_score = score_exact(best)
    eta = 0.008
    rho = 0.12
    K_max = 32
    stall = 0
    t_start = time.time()
    
    while time.time() - t_start < time_limit and stall < max_stall:
        n = len(best)
        V = lag_values(best)
        C = V.max()
        if C <= best_score + 1e-15:
            best_score = C
            best = best.copy()
        
        # Select active lags
        threshold = C - eta
        active_mask = V >= threshold
        active_p = np.where(active_mask)[0]
        if len(active_p) == 0:
            active_p = np.array([np.argmax(V)])
        if len(active_p) > K_max:
            # Keep top K by value
            sorted_idx = np.argsort(-V[active_p])
            active_p = active_p[sorted_idx[:K_max]]
        
        active_vals = V[active_p]
        
        # Solve LP
        result = solve_lp_direction(best, active_p, active_vals, rho)
        if result[0] is None:
            stall += 1
            # Try smaller active set
            if len(active_p) > 4:
                active_p_small = active_p[np.argsort(-V[active_p])[:max(4, len(active_p)//2)]]
                active_vals_small = V[active_p_small]
                result = solve_lp_direction(best, active_p_small, active_vals_small, rho * 1.5)
            if result[0] is None:
                rho *= 0.7
                eta *= 1.3
                if rho < 0.005:
                    rho = 0.12
                continue
            active_p = active_p_small
            active_vals = active_vals_small
        
        u, t_opt = result
        # Line search
        found = False
        for alpha in [1.0, 0.75, 0.55, 0.4, 0.28, 0.18, 0.1, 0.05]:
            trial = make_feasible(best + alpha * u)
            sc = score_exact(trial)
            if sc < best_score - 1e-12:
                best = trial
                best_score = sc
                found = True
                break
        
        if found:
            stall = 0
            # Slightly increase trust region
            rho = min(rho * 1.1, 0.25)
            # Slightly tighten active set
            eta = max(eta * 0.9, 0.001)
        else:
            stall += 1
            rho *= 0.85
            eta = min(eta * 1.2, 0.03)
            if rho < 0.005:
                # Reset and try different parameters
                rho = 0.15
                eta = 0.004
                # Try with only the top 5 lags
                top5 = np.argsort(-V)[:5]
                result = solve_lp_direction(best, top5, V[top5], 0.08)
                if result[0] is not None:
                    u2, _ = result
                    for alpha in [0.5, 0.3, 0.2, 0.1]:
                        trial = make_feasible(best + alpha * u2)
                        sc = score_exact(trial)
                        if sc < best_score - 1e-12:
                            best = trial
                            best_score = sc
                            found = True
                            break
    
    return best, best_score

def two_point_exchange(h, n_tries, rng, time_limit):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    t_start = time.time()
    
    for _ in range(n_tries):
        if time.time() - t_start > time_limit:
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        # Try several t values
        for frac in [0.5, 0.25, 0.75, 1.0/2.618, 2.0/2.618, 0.1, 0.9]:
            t = t_lo + frac * span
            tmp = best.copy()
            tmp[a] = best[a] + t
            tmp[b] = best[b] - t
            sc = score_exact(tmp)
            if sc < best_sc - 1e-12:
                best = tmp
                best_sc = sc
                break
    
    return best, best_sc

def binary_state_search(h_cont, n_steps, rng, time_limit):
    n = len(h_cont)
    if n % 2 != 0:
        return None
    target_on = n // 2
    # Initialize from threshold
    h = make_feasible(h_cont)
    # Select which positions are 1: those with highest h values
    order = np.argsort(-h)
    bits = np.zeros(n, dtype=float)
    bits[order[:target_on]] = 1.0
    
    current = bits.copy()
    current_sc = score_exact(current)
    best = bits.copy()
    best_sc = current_sc
    pos1 = np.where(current > 0.5)[0]
    pos0 = np.where(current < 0.5)[0]
    
    T = 2e-4
    t_start = time.time()
    
    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        # Random swap
        i1 = pos1[rng.integers(0, len(pos1))]
        i0 = pos0[rng.integers(0, len(pos0))]
        current[i1] = 0.0
        current[i0] = 1.0
        new_sc = score_exact(current)
        delta = new_sc - current_sc
        if delta < 0 or np.log(rng.random()) < -delta / max(T, 1e-8):
            current_sc = new_sc
            pos1 = np.where(current > 0.5)[0]
            pos0 = np.where(current < 0.5)[0]
            if new_sc < best_sc:
                best = current.copy()
                best_sc = new_sc
        else:
            # Revert
            current[i1] = 1.0
            current[i0] = 0.0
        T *= 0.999
        
    return best, best_sc
