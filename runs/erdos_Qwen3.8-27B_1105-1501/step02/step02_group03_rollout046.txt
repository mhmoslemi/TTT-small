Since h is parameterized by left half values `left[0:M]`, and the full h is:
- h[j] = left[j] for j = 0..M-1

Actually, the cleanest approach is:

1. Define h(z) = antisym function of z (logistic of z)
2. For the objective, compute all K=2n-1 overlap values
3. For gradients, use the fact that d(overlap_p)/d(z_j) = d(overlap_p)/d(h) dot dh/dz_j

Where dh/dz_j is a sparse vector: only positions j and n-1-j are nonzero.
- dh[j]/dz_j = sigmoid(z_j)(1-sigmoid(z_j)) = h[j]*(1-h[j])
- dh[n-1-j]/dz_j = -1 * sigmoid(z_j)(1-sigmoid(z_j)) = -h[j]*(1-h[j]) = -(1-h[j])*h[j]

Wait: h[n-1-j] = 1 - left[j] = 1 - sigmoid(z_j)
d(h[n-1-j])/dz_j = -sigmoid(z_j)(1-sigmoid(z_j)) = -h[j](1-h[j])

So the Jacobian of h w.r.t. z is:
- J[j, j] = h[j](1-h[j])
- J[n-1-j, j] = -h[j](1-h[j])

And the gradient of overlap_p w.r.t. h is:
- For shift D (where D = p - (n-1)):

np.correlate(a, b, mode='full')[p] = sum_i a[i] * b[i + p - (len(b)-1)]

Where the sum is over all i where both indices are valid.

So for a = h, b = 1-h, both of length n:

  Let me be very careful. The sum is S = sum_i h[i]*(1-h[i+D]) over valid (i, i+D).
  
  dS/dh[k]:
  - k appears as the first index h[i] when i = k: valid if 0 <= k < n and 0 <= k+D < n
    Contribution: (1-h[k+D])
  - k appears as the second index h[i+D] when i+D = k, i.e., i = k-D: valid if 0 <= k-D < n and 0 <= k < n
    Contribution: -h[k-D]
  
  So dS/dh[k] = (1-h[k+D]) * [0<=k<n and 0<=k+D<n] - h[k-D] * [0<=k-D<n and 0<=k<n]

Now, the gradient w.r.t. z_j:
dS/dz_j = sum_k (dS/dh[k]) * (dh[k]/dz_j)

Only k = j and k = n-1-j have nonzero dh[k]/dz_j.
- dh[j]/dz_j = h[j](1-h[j])
- dh[n-1-j]/dz_j = -h[j](1-h[j])

So:
dS/dz_j = (dS/dh[j]) * h[j](1-h[j]) + (dS/dh[n-1-j]) * (-h[j](1-h[j]))

def antisym_from_left(left):
    M = len(left)
    n = 2 * M
    h = np.empty(n)
    h[:M] = left
    h[M:] = 1.0 - left[::-1]
    return h

def overlap_grad_full(h):
    """Gradient of each overlap w.r.t. h. Returns array of shape (K, n)."""
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for p in range(K):
        D = p - (n - 1)
        if D >= 0:
            L = n - D
            # dS/dh[k] = (1-h[k+D]) if k < L, minus h[k-D] if k >= D
            G[p, :L] += (1.0 - h[D:D+L])
            G[p, D:] -= h[:L]
        else:
            a = -D
            L = n - a
            # dS/dh[k] = (1-h[k-a]) if k >= a, minus h[k+a] if k < L
            G[p, a:] += (1.0 - h[:L])
            G[p, :L] -= h[a:]
    return G * dx

def antisym_obj_grad(z, beta=None):
    """
    Compute objective and gradient for antisymmetric parameterization.
    z: array of length M (left half logistic inputs)
    Returns (objective, gradient) where gradient is w.r.t. z.
    If beta is None, use exact max (subgradient).
    """
    M = len(z)
    left = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
    h = antisym_from_left(left)
    vals = compute_overlaps(h)
    cmax = vals.max()
    
    if beta is not None:
        shifted = beta * (vals - cmax)
        # For numerical stability
        shifted = np.clip(shifted, -500, 500)
        logsum = np.log(np.sum(np.exp(shifted)))
        obj = cmax + logsum / beta
        w = np.exp(shifted - logsum)
    else:
        # Subgradient: uniform weight on active set
        obj = cmax
        active = np.where(np.abs(vals - cmax) < 1e-10)[0]
        w = np.zeros(len(vals))
        w[active] = 1.0 / len(active)
    
    # Compute gradient of each overlap w.r.t. h
    G_h = overlap_grad_full(h)  # shape (K, n)
    
    # Chain rule: d(overlap_p)/dz_j = sum_k G_h[p,k] * dh[k]/dz_j
    # dh[j]/dz_j = left[j]*(1-left[j]), dh[n-1-j]/dz_j = -left[j]*(1-left[j])
    ds = left * (1.0 - left)  # shape (M,)
    
    # G_z[p, j] = ds[j] * (G_h[p, j] - G_h[p, n-1-j])
    G_z = ds[np.newaxis, :] * (G_h[:, :M] - G_h[:, M:])  # shape (K, M)
    
    grad = w @ G_z
    return obj, grad

def soft_max_opt(z0, beta, time_limit=10.0, max_iter=500):
    """Optimize soft-max objective using L-BFGS-B."""
    def fun_g(z):
        return antisym_obj_grad(z, beta=beta)
    
    def fun(z):
        return fun_g(z)[0]
    
    def grad(z):
        return fun_g(z)[1]
    
    result = minimize(fun, z0, jac=grad, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'maxfun': max_iter * 3})
    return result.x

def active_set_solve(z0, active_idx, max_iter=200):
    """Solve active-set minimax using SLSQP."""
    M = len(z0)
    
    def obj_fun(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(M + 1)
        g[-1] = 1.0
        return g
    
    # Build constraints: t - overlap_p(z) >= 0 for p in active_idx
    def con_fun(x):
        z = x[:M]
        t = x[-1]
        left = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
        h = antisym_from_left(left)
        vals = compute_overlaps(h)
        return t - vals[active_idx]
    
    def con_jac(x):
        z = x[:M]
        left = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
        h = antisym_from_left(left)
        ds = left * (1.0 - left)
        G_h = overlap_grad_full(h)
        G_z = ds[np.newaxis, :] * (G_h[:, :M] - G_h[:, M:])
        J = np.zeros((len(active_idx), M + 1))
        J[:, :M] = -G_z[active_idx, :]
        J[:, M] = 1.0
        return J
    
    h0 = antisym_from_left(1.0 / (1.0 + np.exp(-np.clip(z0, -30, 30))))
    t0 = compute_overlaps(h0).max() + 1e-6

    # Invert to get z0 (should be close to original)
    
    x0 = np.append(z0, t0)
    bounds = [(-15.0, 15.0)] * M + [(0.0, 3.0)]
