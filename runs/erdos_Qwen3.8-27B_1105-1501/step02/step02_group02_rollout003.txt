def _coord_descent(h0, n_passes=5):
    """Coordinate descent with projection."""
    h = h0.copy()
    best_s = _score(h)
    for _ in range(n_passes):
        improved = False
        for p in range(len(h)):
            # Try a few values for h[p]
            best_val = h[p]
            best_local = best_s
            for trial in [0.0, 0.25, 0.5, 0.75, 1.0, h[p] + 0.1, h[p] - 0.1]:
