    for it in range(max_iter):
        rho = rho_start * (rho_end / rho_start) ** (it / max(1, max_iter - 1))
        h_new = soctep(h, k_active, rho)
        if h_new is None:
            break
        h_new = make_feasible(h_new)
        sc = official_score(h_new)
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.7, 0.5, 0.3, 0.15]:
                h_try = make_feasible(h + alpha * (h_new - h))
                sc_try = official_score(h_try)
                if sc_try < best - 1e-14:
                    h = h_try
                    best = sc_try
                    accepted = True
                    break
            if not accepted:
                stall += 1
                if stall >= 10:
                    break
    return h, best

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def antisy(h):
    return make_feasible(1.0 - h[::-1].copy())

def qp_dir_step(h0, k_active, rho):
    n = len(h0)
    vals = compute_f(h0)
    top_idx = np.argsort(vals)[-k_active:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        cp.norm(h_var - h0, 2) <= rho
    ]
    for p in top_idx:
        s = p - (n - 1)
        g = gradient_p(h0, s)
        constraints.append(vals[p] + g @ (h_var - h0) <= t_var)
    obj = cp.Minimize(t_var + 0.01 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    return np.asarray(h_new, dtype=float)

def slsqp_smooth(h0, beta, maxiter=100):
    def obj_fn(x, b):
        n = len(x)
        dx = 2.0 / n
        vals = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = vals.max()
        shifted = b * (vals - cmax)
        shifted = np.clip(shifted, -50, 50)
        return cmax + np.log(np.sum(np.exp(shifted))) / b
    def eq_fun(x):
        return np.sum(x) - len(x) / 2.0
    def eq_jac(x):
        return np.ones(len(x))
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fn, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    best_h = None
    best_score = float('inf')

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x / 2.0))
        best_score = official_score(best_h)

    # Phase 1: Binary local search from parent
    if remaining() > 100:
        h_bin, sc_bin = binary_local_search(best_h, max_epochs=200, seed=seed)
        sc_bin_cont = official_score(make_feasible(h_bin))
        update(make_feasible(h_bin), sc_bin_cont)

    # Phase 2: SOCP optimization from best
    if remaining() > 150:
        h_opt, sc_opt = soctep_optimize(best_h, max_iter=40, k_active=5, rho_start=0.5, rho_end=0.04)
        update(h_opt, sc_opt)

    # Phase 3: SOCP with different k_active
    if remaining() > 120:
        for ka in [3, 7, 10]:
            if remaining() < 30:
                break
            h_opt, sc_opt = soctep_optimize(best_h, max_iter=25, k_active=ka, rho_start=0.4, rho_end=0.03)
            update(h_opt, sc_opt)

    # Phase 4: QP direction step with regularization
    if remaining() > 100:
        for ka in [5, 8]:
            if remaining() < 30:
                break
            n = len(best_h)
            h = best_h.copy()
            for it in range(30):
                if remaining() < 10:
                    break
                rho = 0.3 * (0.1 / 0.3) ** (it / 29.0)
                h_new = qp_dir_step(h, ka, rho)
                if h_new is None:
                    break
                h_new = make_feasible(h_new)
                sc = official_score(h_new)
                if sc < official_score(h) - 1e-14:
                    h = h_new
                else:
                    break
            update(h, official_score(h))

    # Phase 5: Multi-resolution exploration
    if remaining() > 120:
        n_base = len(best_h)
        for n_opt in [n_base - 5, n_base - 10, n_base + 5, n_base + 10]:
            if remaining() < 40 or n_opt < 20:
                break
            h_int = interp_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            h_opt, sc_opt = soctep_optimize(h_int, max_iter=20, k_active=5, rho_start=0.4, rho_end=0.05)
            update(h_opt, sc_opt)
            # Also try binary search at this resolution
            if remaining() > 50:
                h_bin, sc_b = binary_local_search(h_int, max_epochs=100, seed=seed + n_opt)
                h_bin_f = make_feasible(h_bin)
                sc_bf = official_score(h_bin_f)
                update(h_bin_f, sc_bf)
                h_opt2, sc_opt2 = soctep_optimize(h_bin_f, max_iter=15, k_active=5, rho_start=0.3, rho_end=0.04)
                update(h_opt2, sc_opt2)

    # Phase 6: Anti-symmetric transform
    if remaining() > 60:
        h_sym = antisy(best_h)
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        h_opt, sc_opt = soctep_optimize(h_sym, max_iter=25, k_active=5, rho_start=0.4, rho_end=0.03)
        update(h_opt, sc_opt)

    # Phase 7: Random perturbations + SOCP
    if remaining() > 100:
        for trial in range(12):
            if remaining() < 30:
                break
            sigma = 0.03 + 0.08 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = soctep_optimize(h_cand, max_iter=20, k_active=5, rho_start=0.3, rho_end=0.03)
            update(h_opt, sc_opt)
            # Also try from binary of perturbed
            if remaining() > 40:
                h_bin, sc_b = binary_local_search(h_cand, max_epochs=80, seed=seed * 100 + trial)
                h_bin_f = make_feasible(h_bin)
                sc_bf = official_score(h_bin_f)
                h_opt2, sc_opt2 = soctep_optimize(h_bin_f, max_iter=15, k_active=5, rho_start=0.3, rho_end=0.03)
                update(h_opt2, sc_opt2)

    # Phase 8: SLSQP smooth max
    if remaining() > 50:
        for beta in [2000, 8000, 30000]:
            if remaining() < 20:
                break
            h_new, sc = slsqp_smooth(best_h, beta, maxiter=80)
            if h_new is not None:
                update(h_new, sc)

    # Phase 9: Final intensive polish with very small steps
    if remaining() > 40:
        h_opt, sc_opt = soctep_optimize(best_h, max_iter=30, k_active=5, rho_start=0.15, rho_end=0.01)
        update(h_opt, sc_opt)
        h_opt, sc_opt = soctep_optimize(best_h, max_iter=20, k_active=8, rho_start=0.1, rho_end=0.01)
        update(h_opt, sc_opt)
        h_opt, sc_opt = soctep_optimize(best_h, max_iter=20, k_active=3, rho_start=0.1, rho_end=0.005)
        update(h_opt, sc_opt)

    # Phase 10: One more binary refinement at best resolution
    if remaining() > 40:
        h_bin, sc_b = binary_local_search(best_h, max_epochs=150, seed=seed + 999)
        h_bin_f = make_feasible(h_bin)
        sc_bf = official_score(h_bin_f)
        update(h_bin_f, sc_bf)
        if remaining() > 30:
            h_opt, sc_opt = soctep_optimize(h_bin_f, max_iter=20, k_active=5, rho_start=0.2, rho_end=0.02)
            update(h_opt, sc_opt)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    return final_h, final_score + 1e-13, len(final_h)
