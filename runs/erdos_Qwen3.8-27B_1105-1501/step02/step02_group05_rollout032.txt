import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if residual > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(residual, 1.0 - h[idx])
            if add <= 1e-15:
                break
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    elif residual < -1e-12:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            if sub <= 1e-15:
                break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-14:
                break
    return h

def _eq_fun(x):
    return x.sum() - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    m = vals.max()
    shifted = beta * (vals - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    m = vals.max()
    shifted = beta * (vals - m)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-16:
            continue
        s = p - (n - 1)
        wp = w[p] * dx
        if s >= 0:
            mi = n - s
            grad[:mi] += wp * (1.0 - x[s:s + mi])
            grad[s:] -= wp * x[:mi]
        else:
            t = -s
            mi = n - t
            grad[t:] += wp * (1.0 - x[:mi])
            grad[:mi] -= wp * x[t:t + mi]
    return grad

def slsqp_optimize(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = exact_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def two_point_search(h, max_iters, rng, deadline, refiners=200, max_n_ref=3000):
    n = len(h)
    best = h.copy()
    best_sc = exact_score(best)
    pairs = n * (n - 1) // 2
    h_copy = best.copy()
    h_a = best[0]
    n_ref_done = 0
    stall = 0
    cache_score = 0.0

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        if it % 500 == 499 or it % 5000 == 4999:
            if time.time() >= deadline:
                break

        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue

        t_lo = max(-best[a], best[b] - 1.0) + 1e-12
        t_hi = min(1.0 - best[a], best[b]) - 1e-12
        if t_hi - t_lo < 1e-11:
            continue

        h_a = best[a]
        h_b_best = best[b]
        h_copy[a] = h_a
        h_copy[b] = h_b_best

        def _obj(t):
            h_copy[a] = h_a + t
            h_copy[b] = h_b_best - t
            return exact_score(h_copy)

        res = minimize_scalar(_obj, bounds=(t_lo, t_hi), method='bounded',
                              options={'xatol': 1e-7})
        if res.fun < best_sc - 1e-12:
            best[a] = np.clip(h_a + res.x, 0.0, 1.0)
            best[b] = np.clip(h_b_best - res.x, 0.0, 1.0)
            best_sc = exact_score(best)
            stall = 0
        else:
            stall += 1

        if stall > refiners and n_ref_done < max_n_ref:
            if time.time() >= deadline:
                break
            h_ref, sc_ref = slsqp_optimize(best, 200, maxiter=80)
            n_ref_done += 1
            if h_ref is not None and sc_ref < best_sc - 1e-12:
                best = h_ref
                best_sc = sc_ref
            stall = 0

    return best, best_sc

    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x)
    return make_feasible(h)

def anti_sym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)
