import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(n * 10):
        residual = target - np.sum(h)
        if abs(residual) < 1e-13:
            break
        if residual > 0:
            idx = np.argmax(1.0 - h)
            if 1.0 - h[idx] < 1e-15:
                break
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmin(h)
            if h[idx] < 1e-15:
                break
            sub = min(-residual, h[idx])
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        d = p - (n - 1)
        if d == 0:
            G[p] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            L = n - d
            G[p, :L] += dx * (1.0 - h[d:])
            G[p, d:] -= dx * h[:L]
        else:
            a = -d
            L = n - a
            G[p, a:] += dx * (1.0 - h[:L])
            G[p, :L] -= dx * h[a:]
    return G

def sigmoid(x):
    x = np.clip(x, -500.0, 500.0)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def build_h_from_z(z, alpha, n):
    m = n // 2
    y = sigmoid(alpha * z[:m])
    h = np.zeros(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def smooth_obj(z, alpha, beta, n):
    m = n // 2
    h = build_h_from_z(z, alpha, n)
    vals = corr_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_grad(z, alpha, beta, n):
    m = n // 2
    h = build_h_from_z(z, alpha, n)
    vals = corr_values(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws > 1e-300:
        w /= ws
    else:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    G = corr_grad_matrix(h)
    gh = w @ G
    y = sigmoid(alpha * z[:m])
    dydz = alpha * y * (1.0 - y)
    return dydz * (gh[:m] - gh[n - 1::-1][:m])

def h_to_z(h, alpha, n):
    m = n // 2
    y = h[:m].copy()
    y = np.clip(y, 1e-8, 1.0 - 1e-8)
    return np.log(y / (1.0 - y)) / alpha

def project_anti_sym(h, n):
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return np.clip(h_sym, 0.0, 1.0)

def find_active_set(vals, frac=0.3, min_count=6):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-5)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:top_k]
    active.update(top_idx.tolist())
    return sorted(active)

def active_set_solve(h0, alpha, n, active_idx, max_iter=400):
    m = n // 2
    y0 = h0[:m].copy()
    y0 = np.clip(y0, 1e-6, 1.0 - 1e-6)
    z0 = np.log(y0 / (1.0 - y0)) / alpha
    t0 = float(corr_values(h0).max()) + 1e-5
    z_init = np.concatenate([z0, [t0]])

    def fun(z):
        return z[m]

    def jac(z):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

    def con_fun(z):
        h = build_h_from_z(z[:m], alpha, n)
        vals = corr_values(h)
        return z[m] - vals[active_idx]

    def con_jac(z):
        h = build_h_from_z(z[:m], alpha, n)
        G = corr_grad_matrix(h)
        y = sigmoid(alpha * z[:m])
        dydz = alpha * y * (1.0 - y)
        J = np.zeros((len(active_idx), m + 1))
        for k in range(m):
            J[:, k] = -dydz[k] * (G[active_idx, k] - G[active_idx, n - 1 - k])
        J[:, m] = 1.0
        return J

    bounds = [(-20.0, 20.0)] * m + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(fun, z_init, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):

            h_new = make_feasible(h_new)
            sc = official_score(h_new)
            if np.isfinite(sc) and sc < 2.0:
                return h_new, sc
    except Exception:
        pass
    return None, None

def full_space_smooth(h0, beta, max_iter=3000, tol=1e-10):
    """Optimize in full space with projection (no anti-sym constraint)."""
    n = len(h0)
    h = h0.copy()

    def obj_and_grad(h_v):
        h_v = np.clip(h_v, 1e-10, 1.0 - 1e-10)
        vals = corr_values(h_v)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        logsum = np.log(np.sum(np.exp(shifted)))
        obj = cmax + logsum / beta
        w = np.exp(shifted) / w.sum()
        G = corr_grad_matrix(h_v)
        grad = w @ G
        return obj, grad

    try:
        res = minimize(obj_and_grad, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'gtol': tol})
        if res.x is not None:
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

                z0 = h_to_z(h_as, alpha, n_hi)
                for beta in [500, 5000]:
                    if is_deadline() or time_left() < 5:
                        break
                    max_iter = max(100, int(min(time_left() * 0.3, 15.0) * 40))
                    try:
                        res = minimize(smooth_obj, z0,
                                       jac=lambda z: smooth_grad(z, alpha, beta, n_hi),
                                       method='L-BFGS-B',
                                       options={'maxiter': max_iter, 'gtol': 1e-10})
                        if res.x is not None:
                            h_new = build_h_from_z(res.x, alpha, n_hi)
                            h_new = make_feasible(h_new)
                            sc = official_score(h_new)
                            if np.isfinite(sc) and sc < best_score - 1e-14:
                                update(h_new, sc)
                                h_as = project_anti_sym(h_new, n_hi)
                    except Exception:
                        pass

    # Phase 6: Random perturbations + local refine
    if time_left() > 40:
        n_cur = len(best_h)
        n_starts = max(2, min(10, int(time_left() / 12)))
        sigmas = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]
        for i in range(n_starts):
            if is_deadline():
                break
            sigma = sigmas[i % len(sigmas)]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n_cur))
            sc_cand = official_score(h_cand)
            if sc_cand < best_score - 1e-14:
                update(h_cand, sc_cand)
            # Quick L-BFGS-B polish
            if time_left() > 10:
                h_as = project_anti_sym(h_cand, n_cur)
                alpha = 5.0
                z0 = h_to_z(h_as, alpha, n_cur)
                beta = 5000
                max_iter = max(50, int(min(time_left() * 0.2, 10.0) * 30))
                try:
                    res = minimize(smooth_obj, z0,
                                   jac=lambda z: smooth_grad(z, alpha, beta, n_cur),
                                   method='L-BFGS-B',
                                   options={'maxiter': max_iter, 'gtol': 1e-10})
                    if res.x is not None:
                        h_new = build_h_from_z(res.x, alpha, n_cur)
                        h_new = make_feasible(h_new)
                        sc = official_score(h_new)
                        if np.isfinite(sc) and sc < best_score - 1e-14:
                            update(h_new, sc)
                except Exception:
                    pass

    # Phase 7: Final intensive polish on best solution
    if time_left() > 20:
        n_fin = len(best_h)
        # High beta anti-symmetric
        for alpha in [2.0, 10.0, 50.0]:
            if is_deadline():
                break
            h_as = project_anti_sym(best_h, n_fin)
            z0 = h_to_z(h_as, alpha, n_fin)
            for beta in [10000, 50000]:
                if is_deadline() or time_left() < 3:
                    break
                max_iter = max(50, int(min(time_left() * 0.3, 10.0) * 40))
                try:
                    res = minimize(smooth_obj, z0,
                                   jac=lambda z: smooth_grad(z, alpha, beta, n_fin),
                                   method='L-BFGS-B',
                                   options={'maxiter': max_iter, 'gtol': 1e-12})
                    if res.x is not None:
                        h_new = build_h_from_z(res.x, alpha, n_fin)
                        h_new = make_feasible(h_new)
                        sc = official_score(h_new)
                        if np.isfinite(sc) and sc < best_score - 1e-14:
                            update(h_new, sc)
                except Exception:
                    pass
        # Final active set
        if time_left() > 10:
            vals = corr_values(best_h)
            active = find_active_set(vals, frac=0.4, min_count=10)
            h_as = project_anti_sym(best_h, n_fin)
            h_new, sc = active_set_solve(h_as, 10.0, n_fin, active, max_iter=600)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    if not np.all(final_h >= -1e-12) or not np.all(final_h <= 1.0 + 1e-12):
        final_h = np.clip(final_h, 0.0, 1.0)
        final_score = official_score(final_h)

    if abs(np.sum(final_h) - final_n / 2.0) > 1e-8:
        final_h = make_feasible(final_h)
        final_score = official_score(final_h)

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
