def line_search(h, d):
    """Find best step alpha for h + alpha*d."""
    n = len(h)
    best_alpha = 0.0
    best_c5 = compute_c5(h)
    
    # Backtracking line search
    alpha = 1.0
    for _ in range(20):
        h_trial = h + alpha * d
        if np.min(h_trial) < -1e-9 or np.max(h_trial) > 1 + 1e-9:
            alpha *= 0.5
            continue
        h_trial = np.clip(h_trial, 0.0, 1.0)
        c5 = compute_c5(h_trial)
        if c5 < best_c5:
            best_c5 = c5
            best_alpha = alpha
        alpha *= 0.7
    return best_alpha, best_c5

def pair_optimal_delta(h, i, j):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(hi_v - 1.0, -hj_v)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def randomized_pair_polish(h, rng, n_pairs):
    n = len(h)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = pair_optimal_delta(h, i, j)
        if abs(delta) > 1e-12:
            h[i] -= delta
            h[j] += delta

def full_pair_sweep(h, rng):
    n = len(h)
    order = rng.permutation(n)
    for a in range(n):
        for b in range(a + 1, n):
            i, j = order[a], order[b]
            if i > j:
                i, j = j, i
            delta = pair_optimal_delta(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta

import numpy as np

from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_active_gradients(h, lags):
    n = len(h)
    K = len(lags)
    G = np.zeros((K, n))
    for r in range(K):
        lag = lags[r]
        i_lo = max(0, lag)
        i_hi = min(n, n + lag)
        if i_lo < i_hi:
            idx = np.arange(i_lo, i_hi)
            G[r, idx] += (1.0 - h[idx - lag])
        i_lo2 = max(0, -lag)
        i_hi2 = min(n, n - lag)
        if i_lo2 < i_hi2:
            idx2 = np.arange(i_lo2, i_hi2)
            G[r, idx2] -= h[idx2 + lag]
    return G

def active_set_lp_step(h, n_active=15, trust_radius=0.05):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    
    sorted_idx = np.argsort(corr)[::-1]
    active_idx = sorted_idx[:n_active]
    lags = (active_idx - (n - 1)).astype(int)
    
    G = build_active_gradients(h, lags)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([G, -np.ones((len(lags), 1))])
    b_ub = -corr[active_idx].copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lb = np.maximum(-h, -trust_radius)
    ub = np.minimum(1.0 - h, trust_radius)
    t_lb = max_corr - 2.0 * trust_radius
