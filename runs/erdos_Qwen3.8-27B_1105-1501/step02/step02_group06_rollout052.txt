    # Phase 4: Symmetry transform and optimize

        h_sym = anti_sym(best_h)
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        h_opt, sc_opt = lp_optimize(h_sym, n_iter=100, max_step=0.8,
                                    thresh_start=0.02, thresh_end=0.001,
                                    active_start=15, active_end=8,
                                    deadline=deadline)
        update(h_opt, sc_opt)

    # Phase 5: Fine LP with tight threshold and small steps
    if deadline - time.time() > 30:
        h_opt, sc_opt = lp_optimize(best_h, n_iter=200, max_step=0.3,
                                    thresh_start=0.005, thresh_end=0.0005,
                                    active_start=12, active_end=6,
                                    deadline=deadline)
        update(h_opt, sc_opt)

    # Phase 6: Random perturbations + LP
    if deadline - time.time() > 50:
        for i in range(20):
            if deadline - time.time() < 15:
                break
            sigma = 0.01 + 0.05 * rng.random()
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=len(best_h)))
            h_opt, sc_opt = lp_optimize(h_cand, n_iter=50, max_step=0.5,
                                        thresh_start=0.015, thresh_end=0.002,
                                        active_start=12, active_end=6,
                                        deadline=deadline)
            update(h_opt, sc_opt)

    # Phase 7: SLSQP polish
    if deadline - time.time() > 30:
        for beta in [5000, 20000, 50000]:
            if deadline - time.time() < 15:
                break
            h_new, sc = lse_slsqp(best_h, beta, maxiter=200)
            if h_new is not None:
                update(h_new, sc)

    # Phase 8: Coordinate descent polish
    if deadline - time.time() > 20:
        h_cd, sc_cd = coordinate_descent(best_h, max_pairs=3000, deadline=deadline)
        update(h_cd, sc_cd)

    # Phase 9: Final tight LP
    if deadline - time.time() > 20:
        h_opt, sc_opt = lp_optimize(best_h, n_iter=100, max_step=0.2,
                                    thresh_start=0.003, thresh_end=0.0003,
                                    active_start=10, active_end=5,
                                    deadline=deadline)
        update(h_opt, sc_opt)

    # Phase 10: Try a few more resolutions around the best
    if deadline - time.time() > 40:
        n_base = len(best_h)
        for dn in [-4, -3, -2, -1, 1, 2, 3, 4]:
            if deadline - time.time() < 15:
                break
            n_try = n_base + dn
            if n_try < 20 or n_try > 100:
                continue
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            h_opt, sc_opt = lp_optimize(h_int, n_iter=60, max_step=0.5,
                                        thresh_start=0.015, thresh_end=0.001,
                                        active_start=12, active_end=6,
                                        deadline=deadline)
            update(h_opt, sc_opt)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    assert abs(np.sum(final_h) - len(final_h) / 2.0) < 1e-10
    assert np.all(final_h >= -1e-15) and np.all(final_h <= 1.0 + 1e-15)
    final_bound = final_score + 1e-13
    return final_h, final_bound, len(final_h)
