import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add < 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub < 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def subgradient(h, tol=1e-10):
    """Compute min-norm subgradient of max_k v_k."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    active = np.where(vals >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(vals)])
    k = float(len(active))
    grad = np.zeros(n)
    npts = len(vals)
    for p in active:
        s = int(p) - (n - 1)
        if s >= 0:
            m = n - s
            for j in range(m):
                grad[j] += (1.0 - h[j + s])
                grad[j + s] -= h[j]
        else:
            t = -s
            m = n - t
            for j in range(m):
                grad[j + t] += (1.0 - h[j])
                grad[j] -= h[j + t]
    if k > 0:
        grad *= dx / k
    return grad, cmax

def projected_gradient(h0, n_iter, is_dead, tol_active=1e-8):
    """Projected gradient with exact subgradient and backtracking line search."""
    n = len(h0)
    best = h0.copy()
    best_sc = exact_score(best)
    alpha = 0.5
    for it in range(n_iter):
        if is_dead():
            break
        grad, cmax = subgradient(best, tol=tol_active)
        # Project direction onto null space of equality constraint
        d = -grad - np.mean(-grad)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm
        # Backtracking line search
        improved = False
        for mult in [alpha, alpha*0.5, alpha*0.25, alpha*2.0, alpha*0.125]:
            h_trial = make_feasible(best + mult * d)
            sc = exact_score(h_trial)
            if sc < best_sc - 1e-15:
                best = h_trial
                best_sc = sc
                improved = True
                alpha = min(alpha * 1.3, 2.0)
                break
        if not improved:
            alpha *= 0.5
            if alpha < 1e-8:
                break
    return best, best_sc

def grad_component(h, k):
    """Compute dv_s/dh_k for all lags s, returned as array of length 2n-1."""
    n = len(h)
    dx = 2.0 / n
    npts = 2 * n - 1
    result = np.zeros(npts)
    for p in range(npts):
        s = p - (n - 1)
        if s >= 0:
            m = n - s
            if 0 <= k < m:
                result[p] += (1.0 - h[k + s])
            if s <= k < n:
                result[p] -= h[k - s]
        else:
            t = -s
            m = n - t
            if t <= k < n:
                result[p] += (1.0 - h[k - t])
            if 0 <= k < m:
                result[p] -= h[k + t]
    return result * dx

def two_point_pair(h, i, j):
    """Optimize pair (i,j) exchange via 1D search. Returns new h and score."""
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-14:
        return h, exact_score(h)
    # Compute v_s(t) = v_s(0) + t * c_s
    vals0 = np.correlate(h, 1.0 - h, mode='full')
    gi = grad_component(h, i)
    gj = grad_component(h, j)
    c = gi - gj
    # Sample t values (include golden ratio points for convex piecewise linear)
    span = t_hi - t_lo
    n_samples = 30
    ts = np.linspace(t_lo, t_hi, n_samples)
    # Add golden section points
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    ts = np.concatenate([ts, [t_lo + gr*gr*span, t_lo + gr*span]])
    best_t = 0.0
    best_val = np.inf
    for t in ts:
        v = vals0 + t * c
        mx = float(v.max())
        if mx < best_val:
            best_val = mx
            best_t = t
    # Refine around best_t with finer grid
    if best_t > t_lo + 1e-15 and best_t < t_hi - 1e-15:
        local_span = span * 0.05
        ts2 = np.linspace(best_t - local_span, best_t + local_span, 20)
        ts2 = np.clip(ts2, t_lo, t_hi)
        for t in ts2:
            v = vals0 + t * c
            mx = float(v.max())
            if mx < best_val:
                best_val = mx
                best_t = t
    if abs(best_t) < 1e-14:
        return h, exact_score(h)
    h_new = h.copy()
    h_new[i] = h[i] + best_t
    h_new[j] = h[j] - best_t
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new)
    return h_new, exact_score(h_new)

def two_point_round(h, rng, n_random=200, is_dead=None):
    """One round: random pairs + some systematic pairs."""
    n = len(h)
    best = h.copy()
    best_sc = exact_score(best)
    # Try random pairs
    for _ in range(n_random):
        if is_dead is not None and is_dead():
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        h_new, sc = two_point_pair(best, i, j)
        if sc < best_sc - 1e-14:
            best = h_new
            best_sc = sc
    return best, best_sc

def systematic_two_point(h, is_dead, max_pairs=None):
    """Try a larger set of pairs, prioritizing pairs with large gradient difference."""
    n = len(h)
    best = h.copy()
    best_sc = exact_score(best)
    # Compute per-position gradient magnitudes to prioritize
    grad, _ = subgradient(best, tol=1e-8)
    # Sort pairs by |grad[i] - grad[j]| descending (most promising first)
    if max_pairs is None:
        max_pairs = min(n * (n-1) // 2, 5000)
    # Sample promising pairs
    gr = np.abs(grad)
    pairs_scored = []
    for _ in range(max_pairs * 3):
        i = int(np.random.randint(0, n))
        j = int(np.random.randint(0, n))
        if i == j:
            continue
        score = abs(grad[i] - grad[j])
        pairs_scored.append((score, i, j))
    pairs_scored.sort(key=lambda x: -x[0])
    pairs_scored = pairs_scored[:max_pairs]
    for score, i, j in pairs_scored:
        if is_dead():
            break
        if score < 1e-10:
            continue
        h_new, sc = two_point_pair(best, i, j)
        if sc < best_sc - 1e-14:
            best = h_new
            best_sc = sc
    return best, best_sc

def make_wave(n, freqs, amps, phases):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(2 * np.pi * f * x + p)
    return make_feasible(h)

def antisym_map(h):
    h_new = 0.5 * (h + 1.0 - h[::-1].copy())
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(15.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def is_dead():
        return time.time() >= deadline

    best_h = None
    best_score = float('inf')

    def try_update(h_cand):
        nonlocal best_h, best_score
        if h_cand is None:
            return
        sc = exact_score(h_cand)
        if sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = exact_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, Exception):
        pass

    if best_h is None:
        best_h = make_wave(60, [1.0, 2.0, 3.0], [0.25, 0.15, 0.08], [0.0, 0.5, 1.0])
        best_score = exact_score(best_h)

    n = len(best_h)

    # Phase 1: Quick transforms
    if not is_dead():
        h_sym = antisym_map(best_h)
        try_update(h_sym)
        # Complement-reversed
        h_cr = make_feasible(1.0 - best_h[::-1].copy())
        try_update(h_cr)
        # Slight perturbations
        for sig in [0.02, 0.05]:
            if is_dead():
                break
            h_p = make_feasible(best_h + rng.normal(0, sig, size=n))
            try_update(h_p)

    # Phase 2: Projected gradient (main workhorse)
    if not is_dead():
        pg_time_limit = 0.25 * (budget_s - reserve)
        pg_deadline = time.time() + pg_time_limit
        def is_dead_pg():
            return time.time() >= pg_deadline or is_dead()
        best_h, best_score = projected_gradient(best_h, 5000, is_dead_pg, tol_active=1e-8)

    # Phase 3: Systematic two-point exchange
    if not is_dead():
        tp_time_limit = 0.30 * (budget_s - reserve)
        tp_deadline = time.time() + tp_time_limit
        def is_dead_tp():
            return time.time() >= tp_deadline or is_dead()
        best_h, best_score = systematic_two_point(best_h, is_dead_tp, max_pairs=4000)

    # Phase 4: Alternate PG and 2-point
    if not is_dead():
        alt_deadline = time.time() + 0.20 * (budget_s - reserve)
        def is_dead_alt():
            return time.time() >= alt_deadline or is_dead()
        for round_i in range(20):
            if is_dead_alt():
                break
            # PG step
            best_h, best_score = projected_gradient(best_h, 200, is_dead_alt, tol_active=1e-7)
            if is_dead_alt():
                break
            # 2-point step
            best_h, best_score = systematic_two_point(best_h, is_dead_alt, max_pairs=2000)

    # Phase 5: Resolution exploration
    if not is_dead():
        res_deadline = time.time() + 0.10 * (budget_s - reserve)
        def is_dead_res():
            return time.time() >= res_deadline or is_dead()
        n_base = len(best_h)
        n_opts = sorted(set([n_base + d for d in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12]
                              if 30 <= (n_base + d) <= 100]))
        for n_opt in n_opts:
            if is_dead_res():
                break
            if n_opt == n_base:
                continue
            h_int = interp_to(best_h, n_opt)
            sc_int = exact_score(h_int)
            if sc_int < best_score - 1e-14:
                best_h = h_int
                best_score = sc_int
            # Quick optimize at this resolution
            if not is_dead_res():
                best_h_res, _ = projected_gradient(h_int, 100, is_dead_res, tol_active=1e-7)
                sc_res = exact_score(best_h_res)
                if sc_res < best_score - 1e-14:
                    best_h = best_h_res
                    best_score = sc_res

    # Phase 6: Intensive final polishing
    if not is_dead():
        # More PG with tighter tolerance
        best_h, best_score = projected_gradient(best_h, 3000, is_dead, tol_active=1e-10)
    
    # Phase 7: Final 2-point sweep
    if not is_dead():
        best_h, best_score = systematic_two_point(best_h, is_dead, max_pairs=6000)

    # Phase 8: Last PG
    if not is_dead():
        best_h, best_score = projected_gradient(best_h, 2000, is_dead, tol_active=1e-10)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = exact_score(final_h)
    final_n = len(final_h)

    # Safety check
    if not (np.all(np.isfinite(final_h)) and final_h.min() >= -1e-10 and final_h.max() <= 1.0 + 1e-10):
        # Fallback to constant
        final_h = np.full(final_n, 0.5)
        final_score = exact_score(final_h)

    return final_h, final_score, final_n
