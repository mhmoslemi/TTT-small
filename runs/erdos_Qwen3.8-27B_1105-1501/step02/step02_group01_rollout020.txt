import numpy as np
from scipy.optimize import minimize, linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0 - np.max(z), 3.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_grad_lag(h, k_idx):
    n = len(h)
    d = k_idx - (n - 1)
    if d == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if d > 0:
        grad[d:] += (1.0 - h[:n-d])
        grad[:n-d] += (-h[d:])
    else:
        m = -d
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += (-h[:n-m])
    return grad

        max_c = np.max(corr)

        n_active = max(3, min(15, n // 3))

            # Build and solve LP
            c_obj = np.zeros(n + 1)

def active_gradient_descent(h0, max_iter=5000, time_limit=None, t_start=None):
    """Gradient descent on the max using active set averaging."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_score = official_score(h0)
    
    alpha = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        corr = get_correlations(h) * dx
        max_c = np.max(corr)
        
        # Softmax weights
        beta = 20.0
        shifted = beta * (corr - max_c)
        weights = np.exp(np.clip(shifted, -50, 0))
        weights /= np.sum(weights)
        
        # Weighted gradient
        grad = np.zeros(n)
        # Only compute for lags with significant weight
        significant = np.where(weights > 0.01)[0]
        for k_idx in significant:
            g = get_grad_lag(h, k_idx)
            grad += weights[k_idx] * g
        
        # Update with momentum
        v = momentum * v - alpha * grad
        h_new = h + v
        h_new = make_feasible(h_new)
        
        score = official_score(h_new)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            alpha = min(0.3, alpha * 1.01)
        else:
            h = h_new
            alpha = max(0.01, alpha * 0.95)
            if alpha < 0.02 and it > 100:
                break
    
    return best_h, best_score

def targeted_sa(h0, rng, n_iters=20000, time_limit=None, t_start=None):
    """Simulated annealing with targeted moves."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_score = official_score(h0)
    
    T = 0.05
    T_min = 0.001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        # Compute correlations to find worst lag
        corr = get_correlations(h)
        worst_lag = int(np.argmax(corr))
        d = worst_lag - (n - 1)
        
        # Targeted move: find a position i where h[i]=1 and h[i+d]=0 (if valid)
        # This position contributes to the worst lag
        if d > 0:
            candidates = [i for i in range(n - d) if h[i + d] < h[i]]
        elif d < 0:
            m = -d
            candidates = [i for i in range(n - m) if h[i] < h[i + m]]
        else:
            candidates = list(range(n))
        
        if not candidates:
            candidates = list(range(n))
        
        i = candidates[rng.integers(len(candidates))]
        
        # Find a target position to swap with
        j = rng.integers(n)
        if i == j:
            continue
        
        delta = min(h[i], 1.0 - h[j])
        if delta < 0.01:
            continue
        
        swap_amt = delta * rng.uniform(0.5, 1.0)
        h[i] -= swap_amt
        h[j] += swap_amt
        
        new_score = official_score(h)
        if new_score < best_score:
            best_score = new_score
