def make_feasible(h):
    """Project h to be feasible: 0<=h<=1, sum(h)=n/2"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:

                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return h

def slsqp_obj(v, n, active_idx):
    """Objective for SLSQP: return t (last variable)"""
    return v[-1]

def slsqp_jac(v, n, active_idx):
    """Jacobian of objective: [0, 0, ..., 0, 1]"""
    m = n // 2
    J = np.zeros(m + 1)
    J[-1] = 1.0
    return J

def slsqp_constr(v, n, active_idx):
    """Inequality constraints: t - V_k >= 0"""
    m = n // 2
    y = v[:m]
    t = v[-1]
    h = h_from_y(np.clip(y, 0, 1), n)
    V = get_vals(h)
    return t - V[active_idx]

def slsqp_constr_jac(v, n, active_idx):
    """Jacobian of inequality constraints"""
    m = n // 2
    y = v[:m]
    h = h_from_y(np.clip(y, 0, 1), n)
    G = corr_grad(h)
    S = G[active_idx, :m] - G[active_idx, m:n][:, ::-1]
    # S has shape (len(active_idx), m)
    J = np.hstack([-S, np.ones((len(active_idx), 1))])
    return J

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-14:
        order = np.argsort(1.0 - h)
        for idx in order:
            a = min(residual, 1.0 - h[idx])
            if a <= 0: break
            h[idx] += a
            residual -= a
            if abs(residual) < 1e-15: break
    elif residual < -1e-14:
        order = np.argsort(h)
        for idx in order:
            a = min(-residual, h[idx])
            if a <= 0: break
            h[idx] -= a
            residual += a
            if abs(residual) < 1e-15: break
    return h

def get_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_grad(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            if s == 0:
                G[p] = dx * (1.0 - 2.0 * h)
            else:
                L = n - s
                G[p, s:] += dx * (1.0 - h[:L])
                G[p, :L] -= dx * h[s:]
        else:
            a = -s
            L = n - a
            G[p, :L] += dx * (1.0 - h[a:])
            G[p, a:] -= dx * h[:L]
    return G

def sym_from_h(h):
    n = len(h)
    return 0.5 * (h + 1.0 - h[::-1])

def h_from_y(y, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def lbfgsb_sym_fun(y, beta, n):
    m = n // 2
    h = h_from_y(np.clip(y, 0.0, 1.0), n)
    V = get_vals(h)
    Vmax = V.max()
    shifted = beta * (V - Vmax)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = Vmax + logsum / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w /= ws
    G = corr_grad(h)
    S = w @ G
    grad = S[:m] - S[m:n][::-1]
    return obj, grad

def slsqp_obj(v, n, active_idx):
