    """
    Solve LP: minimize t
    subject to: g_p^T u <= t - V[p] for p in active_idx
                sum(u) = 0
                -min(h_i, rho) <= u_i <= min(1-h_i, rho)
    """
    from scipy.optimize import linprog
    
    # Compute gradients for each active lag
    G = np.zeros((len(active_idx), n))
    for i, p in enumerate(active_idx):
        G[i] = lag_gradient(h, p, n)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -V[active_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C))  # t <= C (we want to decrease)

    if res.success:

    return None

    n = len(h)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Greedy residual fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)  # most slack first
        else:
            order = np.argsort(h)  # most room to subtract first
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:

                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    target = n / 2.0
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, p, n):
    """Gradient of dx * correlate(h, 1-h, 'full')[p]."""
    dx = 2.0 / n
    s = p - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        t = s
        if t == 0:
            g[:] = (1.0 - 2.0 * h)
        else:
            m = n - t
            # First term: h[i+t] * (1-h[i]), i in [0, m-1]
            # d/dh[j] = (1-h[j-t]) for j in [t, n-1]
            g[t:] += (1.0 - h[:m])
            # d/dh[j] = -h[j+t] for j in [0, m-1]
            g[:m] -= h[t:]
    else:
        t = -s
        m = n - t
        # h[i] * (1-h[i+t]), i in [0, m-1]
        # d/dh[j] = (1-h[j+t]) for j in [0, m-1]
        g[:m] += (1.0 - h[t:])
        # d/dh[j] = -h[j-t] for j in [t, n-1]
        g[t:] -= h[:m]
    return g * dx

def verify_gradients(h, n, tol=1e-6):
    """Verify gradient computation against finite differences."""
    dx_val = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx_val
    eps = 1e-7
    max_err = 0.0
    for p in [0, n-1, 2*n-2, (n-1)//2]:
        g = lag_grad(h, p, n)
        for j in range(0, n, max(1, n//10)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            V_plus = np.correlate(h_plus, 1.0 - h_plus, mode='full')[p] * dx_val
            V_minus = np.correlate(h_minus, 1.0 - h_minus, mode='full')[p] * dx_val
            fd = (V_plus - V_minus) / (2 * eps)
            max_err = max(max_err, abs(g[j] - fd))
    return max_err

def lp_minimax_step(h, rho, K, eta=0.0):
    """
    Solve one LP-based minimax step.
    Returns (direction, current_max) or (None, current_max).
    """
    n = len(h)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    C = float(V.max())
    
    # Active set selection
    if eta > 0:
        active_mask = V >= (C - eta)
        active = np.where(active_mask)[0]
    else:
        active = np.array([])
    if len(active) > K or len(active) == 0:
        idx = np.argsort(V)[::-1][:K]
        active = idx
    elif len(active) == 1 and V.max() == C and V[np.argmax(V)] == C:
        # Only one max, add a few more for robustness
        idx = np.argsort(V)[::-1][:max(3, K)]
        active = idx
    
    # Ensure the max lag is always in active set
    max_idx = np.argmax(V)
    if max_idx not in active:
        active = np.append(active[:K-1], max_idx)
    
    # Compute gradients
    G = np.zeros((len(active), n))
    for i, p in enumerate(active):
        G[i] = lag_grad(h, p, n)
    
    # Build and solve LP
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([G, -np.ones((len(active), 1))])
    b_ub = -V[active]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((C - rho * 2, C + 1e-10))  # t should not exceed C much
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, C
    
    if not res.success:
        return None, C
    
    u = res.x[:n]
    if not np.all(np.isfinite(u)):
        return None, C
    
    return u, C

def two_point_exchange(h, n_tries, rng, time_limit_fn):
    """Random two-point exchange search."""
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    
    for it in range(n_tries):
        if time_limit_fn():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates = [t_lo + f * span for f in [0.0, 0.25, 0.5, 0.75, 1.0, 
                   1.0/2.618, 2.0/2.618]]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates:
            va = best[a] + t
            vb = best[b] - t
            if va < -1e-13 or va > 1.0 + 1e-13 or vb < -1e-13 or vb > 1.0 + 1e-13:
                continue
            tmp[a] = va
            tmp[b] = vb
            sc = score_exact(tmp)
            if sc < best_local:
                best_local = sc
                best_t = (t, va, vb)
        if best_t is not None:
            best[a] = best_t[1]
            best[b] = best_t[2]
            best_sc = best_local
    
    return best, best_sc

def binary_anneal(h_init, n_steps, rng, T0, cool, time_limit_fn):
    """Simulated annealing on balanced binary strings."""
    n = len(h_init)
    if n % 2 != 0:
        return None
    # Initialize from threshold
    b = (h_init > 0.5).astype(float)
    # Balance
    n_ones = int(np.sum(b))
    target_ones = n // 2
    if n_ones > target_ones:
        # Remove excess ones
        ones_idx = np.where(b == 1)[0]
        # Remove ones with smallest h_init values
        order = np.argsort(h_init[ones_idx])
        for i in range(n_ones - target_ones):
            b[ones_idx[order[i]]] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(b == 0)[0]
        order = np.argsort(-h_init[zeros_idx])
        for i in range(target_ones - n_ones):
            b[zeros_idx[order[i]]] = 1
    
    cur_sc = score_exact(b)
    best_b = b.copy()
    best_sc = cur_sc
    T = T0
    
    for step in range(n_steps):
        if time_limit_fn():
            break
        # Random swap
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        b[i], b[j] = b[j], b[i]
        new_sc = score_exact(b)
        
        if new_sc < cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / max(T, 1e-10)):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]  # revert
        
        T *= cool
    
    return best_b, best_sc
