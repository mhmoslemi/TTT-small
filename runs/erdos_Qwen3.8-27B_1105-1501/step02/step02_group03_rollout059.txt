def grad_full(h):
    """Compute gradient matrix G where G[k, i] = dc_k/dh_i for all shifts k."""
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    G = np.zeros((K, n))
    for k in range(K):
        s = k - (n - 1)  # physical shift
        if s >= 0:
            L = n - s
            # c = sum_{i=0}^{L-1} h[i+s]*(1-h[i])
            G[k, s:s+L] += dx * (1.0 - h[:L])
            G[k, :L] -= dx * h[s:s+L]
        else:
            t = -s
            L = n - t
            # c = sum_{i=0}^{L-1} h[i]*(1-h[i+t])
            G[k, :L] += dx * (1.0 - h[t:t+L])
            G[k, t:t+L] -= dx * h[:L]
    return G

def grad_z_sym(h, n):

def find_active(vals, frac=0.3, min_count=10):

def nlp_full(h0, active_idx, max_iter=300):

def interpolate_to(h_src, n_tgt):

    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)

def bfgs_sym(z0, beta, scale=1.0, time_limit=15.0, max_iter=200):
    """BFGS optimization in z-space with log-sum-exp smoothing."""
    t0 = time.time()
    x = z0.copy()
    f, g = soft_obj_grad_sym_vec(x, beta, scale)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        # Backtracking line search
        alpha = 1.0 / (gn + 1e-10)
        alpha = min(alpha, 3.0 / (np.max(g**2) + 1e-10))
        for _ in range(40):
            x_trial = np.clip(x - alpha * g, -15.0, 15.0)
            f_trial, g_trial = soft_obj_grad_sym_vec(x_trial, beta, scale)
            if f_trial <= f - 1e-4 * alpha * gn:
                x = x_trial
                f = f_trial
                g = g_trial
                break
            alpha *= 0.4
        else:
            break
    return x

        if time.time() - t0 > time_limit:
            break
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        alpha = min(1.0 / (gn + 1e-10), 3.0 / (np.max(g**2) + 1e-10))
        for _ in range(50):
            x_trial = np.clip(x - alpha * g, -15.0, 15.0)
            f_trial, g_trial = soft_obj_grad_sym(x_trial, beta, scale)
            if f_trial <= f - 1e-4 * alpha * gn:
                x, f, g = x_trial, f_trial, g_trial
                break
            alpha *= 0.4
        else:
            break
    return x

def nlp_sym(z0, active_idx, scale=1.0, max_iter=300):
    M = len(z0)

    def obj_fun(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(M + 1)
        g[-1] = 1.0
        return g

    def cons_fun(x):
        z = x[:M]
        t = x[-1]
        h = h_from_z(z, scale)
        vals = compute_vals(h)
        return t - vals[active_idx]

    def cons_jac(x):
        z = x[:M]
        h = h_from_z(z, scale)
        G = grad_full(h)
        u = h[:M]
        du_dz = u * (1.0 - u) * scale
        Gz_act = (G[active_idx, :M] - G[active_idx, M:][:, ::-1]) * du_dz[np.newaxis, :]
        J = -Gz_act
        J = np.hstack([J, np.ones((len(active_idx), 1))])
        return J

    h0 = h_from_z(z0, scale)
    vals0 = compute_vals(h0)
    t0 = vals0[active_idx].max() + 1e-6
    x0 = np.concatenate([z0, [t0]])
    bounds = [(-15.0, 15.0)] * M + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_fun, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            z_new = res.x[:M]
            h_new = h_from_z(z_new, scale)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc, z_new
    except Exception:
        pass
    return None, None, None

def nlp_full(h0, active_idx, max_iter=300):
    n = len(h0)

    def obj_fun(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        vals = compute_vals(h)
        return t - vals[active_idx]

    def ineq_jac(x):
        h = x[:n]
        G = grad_full(h)
        J = -G[active_idx, :]
        J = np.hstack([J, np.ones((len(active_idx), 1))])
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    vals0 = compute_vals(h0)
    t0 = vals0[active_idx].max() + 1e-6
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj_fun, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def coordinate_descent_sym(z, active_idx, scale=1.0, n_passes=3, step_init=0.05):
    """Simple coordinate descent on active constraints in z-space."""
    M = len(z)
    h = h_from_z(z, scale)
    vals = compute_vals(h)
    best_sc = official_score(h)
    step = step_init
    for _ in range(n_passes):
        improved = False
        for j in range(M):
            for direction in [1, -1]:
                z_trial = z.copy()
                z_trial[j] += direction * step
                z_trial = np.clip(z_trial, -15.0, 15.0)
                h_trial = h_from_z(z_trial, scale)
                sc_trial = official_score(h_trial)
                if sc_trial < best_sc - 1e-14:
                    z = z_trial.copy()
                    h = h_trial
                    best_sc = sc_trial
                    improved = True
        if not improved:
            step *= 0.5
            if step < 1e-5:
                break
        else:
            step = min(step * 1.1, 0.1)
    return z, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = 20.0
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    def check_deadline():
        return time.time() >= deadline

    def update(h, sc):
        nonlocal best_h, best_score
        if h is not None and np.isfinite(sc):
            if sc < best_score - 1e-14:
                best_h = h.copy()
                best_score = sc
            elif best_h is not None and len(h) < len(best_h) and abs(sc - best_score) < 1e-12:
                best_h = h.copy()
                best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 64
        x = np.arange(n0)
        h_init = 0.5 + 0.2 * np.sin(2 * np.pi * x / n0) + 0.08 * np.cos(4 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    # Symmetrize and check
    h_sym_init = symmetrize(best_h)
    sc_sym = official_score(h_sym_init)
    if sc_sym < best_score - 1e-14:
        update(h_sym_init, sc_sym)

    # Ensure even length for symmetric work
    n_work = len(best_h)
    if n_work % 2 == 1:
        n_work += 1
    if len(best_h) != n_work:
        h_work = interpolate_to(best_h, n_work)
    else:
        h_work = best_h.copy()
    h_work_sym = symmetrize(h_work)
    sc_work = official_score(h_work_sym)
    if sc_work < best_score - 1e-14:
        update(h_work_sym, sc_work)

    M = len(h_work_sym) // 2
    scale = 1.0
    z_best = z_from_h_sym(h_work_sym, scale)

    # Phase 1: BFGS with increasing beta
    if time_left() > 30:
        betas = [20, 60, 150, 400, 1000, 3000, 8000, 20000]
        z_cur = z_best.copy()
        for beta in betas:
            if check_deadline():
                break
            tl = min(time_left() * 0.2, 18.0)
            if tl < 3:
                break
            z_cur = bfgs_sym(z_cur, beta, scale, time_limit=tl)
            h_trial = h_from_z(z_cur, scale)
            sc = official_score(h_trial)
            if sc < best_score - 1e-14:
                update(h_trial, sc)
                z_best = z_cur.copy()

    # Phase 2: Active-set SLSQP in symmetric space
    if time_left() > 20:
        h_cur = h_from_z(z_best, scale)
        vals = compute_vals(h_cur)
        active = find_active(vals, frac=0.2, min_count=8)
        h_new, sc, z_new = nlp_sym(z_best, active, scale, max_iter=500)
        if h_new is not None:
            if sc < best_score - 1e-14:
                update(h_new, sc)
            if z_new is not None:
                z_best = z_new

    # Phase 3: Iterate active set refinement with varying sizes
    if time_left() > 40:
        for rnd in range(5):
            if check_deadline():
                break
            h_cur = h_from_z(z_best, scale)
            vals = compute_vals(h_cur)
            for frac in [0.1, 0.2, 0.35]:
                if check_deadline():
                    break
                active = find_active(vals, frac=frac, min_count=6)
                h_new, sc, z_new = nlp_sym(z_best, active, scale, max_iter=350)
                if h_new is not None:
                    if sc < best_score - 1e-14:
                        update(h_new, sc)
                    if z_new is not None:
                        z_best = z_new
            # Coordinate descent polish
            if time_left() > 10:
                z_best, sc_cd = coordinate_descent_sym(z_best, active, scale, n_passes=5, step_init=0.03)
                h_cd = h_from_z(z_best, scale)
                sc_cd = official_score(h_cd)
                if sc_cd < best_score - 1e-14:
                    update(h_cd, sc_cd)

    # Phase 4: Full-space (non-symmetric) SLSQP
    if time_left() > 25:
        h_cur = h_from_z(z_best, scale)
        vals = compute_vals(h_cur)
        active = find_active(vals, frac=0.2, min_count=10)
        h_new, sc = nlp_full(h_cur, active, max_iter=400)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)

    # Phase 5: Resolution search
    if time_left() > 60:
        n_base = len(best_h)
        n_opts = sorted(set(n for n in [n_base-4, n_base-2, n_base+2, n_base+4, n_base+6, n_base+8]
                            if 50 <= n <= 88 and n % 2 == 0))
        for n_opt in n_opts:
            if check_deadline():
                break
            h_int = interpolate_to(best_h, n_opt)
            h_int_sym = symmetrize(h_int)
            sc_int = official_score(h_int_sym)
            if sc_int < best_score - 1e-14:
                update(h_int_sym, sc_int)
            M_opt = n_opt // 2
            z_opt = z_from_h_sym(h_int_sym, scale)
            for beta in [400, 3000]:
                if check_deadline():
                    break
                tl = min(time_left() * 0.12, 12.0)
                if tl < 2:
                    break
                z_opt = bfgs_sym(z_opt, beta, scale, time_limit=tl)
                h_trial = h_from_z(z_opt, scale)
                sc = official_score(h_trial)
                if sc < best_score - 1e-14:
                    update(h_trial, sc)
            if time_left() > 12:
                h_trial = h_from_z(z_opt, scale)
                vals = compute_vals(h_trial)
                active = find_active(vals, frac=0.25, min_count=8)
                h_new, sc, z_new = nlp_sym(z_opt, active, scale, max_iter=250)
                if h_new is not None and sc < best_score - 1e-14:
                    update(h_new, sc)

    # Phase 6: Random restarts
    if time_left() > 40:
        n_starts = max(3, min(10, int(time_left() / 18)))
        sigmas = [0.08, 0.15, 0.25, 0.4, 0.12, 0.2, 0.05, 0.3]
        for i in range(n_starts):
            if check_deadline():
                break
            sigma = sigmas[i % len(sigmas)]
            z_cand = np.clip(z_best + rng.normal(0, sigma, size=M), -15.0, 15.0)
            h_cand = h_from_z(z_cand, scale)
            sc_cand = official_score(h_cand)
            if sc_cand < best_score - 1e-14:
                update(h_cand, sc_cand)
            if time_left() > 8:
                z_cand = bfgs_sym(z_cand, 2000, scale, time_limit=min(time_left() * 0.08, 8.0))
                h_cand = h_from_z(z_cand, scale)
                sc = official_score(h_cand)
                if sc < best_score - 1e-14:
                    update(h_cand, sc)
                    z_best = z_cand.copy()
                # Quick active set
                if time_left() > 8:
                    vals = compute_vals(h_cand)
                    active = find_active(vals, frac=0.2, min_count=6)
                    h_new, sc2, z_new = nlp_sym(z_cand, active, scale, max_iter=200)
                    if h_new is not None and sc2 < best_score - 1e-14:
                        update(h_new, sc2)
                        z_best = z_new

    # Phase 7: Final intensive polish
    if time_left() > 20:
        for beta in [10000, 30000, 80000]:
            if check_deadline():
                break
            tl = min(time_left() * 0.25, 10.0)
            if tl < 2:
                break
            z_new = bfgs_sym(z_best, beta, scale, time_limit=tl)
            h_trial = h_from_z(z_new, scale)
            sc = official_score(h_trial)
            if sc < best_score - 1e-14:
                update(h_trial, sc)
                z_best = z_new

    if time_left() > 15:
        h_cur = h_from_z(z_best, scale)
        vals = compute_vals(h_cur)
        active_broad = find_active(vals, frac=0.4, min_count=12)
        h_new, sc, z_new = nlp_sym(z_best, active_broad, scale, max_iter=500)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)
            z_best = z_new if z_new is not None else z_best

    if time_left() > 10:
        h_cur = h_from_z(z_best, scale)
        vals = compute_vals(h_cur)
        active_tight = find_active(vals, frac=0.1, min_count=5)
        h_new, sc, z_new = nlp_sym(z_best, active_tight, scale, max_iter=400)
        if h_new is not None:
            if sc < best_score - 1e-14:
                update(h_new, sc)
            if z_new is not None:
                z_best = z_new

    # Final full-space attempt
    if time_left() > 12:
        h_cur = best_h.copy()
        vals = compute_vals(h_cur)
        active = find_active(vals, frac=0.25, min_count=10)
        h_new, sc = nlp_full(h_cur, active, max_iter=350)
        if h_new is not None and sc < best_score - 1e-14:
            update(h_new, sc)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    if final_score > best_score + 1e-10:
        if (np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
                and abs(np.sum(best_h) - len(best_h) / 2.0) < 1e-10):
            final_h = np.clip(best_h, 0.0, 1.0)
            final_score = official_score(final_h)

    assert np.all(final_h >= -1e-12)
    assert np.all(final_h <= 1.0 + 1e-12)
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-8

    return final_h, final_score + 1e-12, final_n
