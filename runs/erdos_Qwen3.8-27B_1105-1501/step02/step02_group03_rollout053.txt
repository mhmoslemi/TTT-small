    """Extract left-half and convert to logit space."""
    n = len(h)
    M = n // 2
    left = h[:M].copy()
    # Clip to avoid log(0)
    left = np.clip(left, 1e-8, 1.0 - 1e-8)
    z = np.log(left / (1.0 - left))
    return z

    h = np.empty(n)
    h[:M] = u
    h[M:] = 1.0 - u[::-1]
    return h

            G[p, s:s+L] += dx * (1.0 - h[:L])

    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    def check_dl():
        return time.time() >= deadline

    def update(h, sc):
        nonlocal best_h, best_score
        if h is not None and np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        x = np.arange(n0)
        h_init = 0.5 + 0.2 * np.sin(2 * np.pi * x / n0) + 0.05 * np.cos(4 * np.pi * x / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    update(best_h, best_score)
    n_base = len(best_h)

    # Phase 1: Symmetric logistic L-BFGS-B with increasing beta
    if time_left() > 60:
        h_sym = sym_project(best_h)
        sc_s = official_score(h_sym)
        update(h_sym, sc_s)
        n = n_base
        M = n // 2
        z_cur = to_logistic_z(h_sym)
        beta_sch = [30, 80, 200, 500, 1200, 3000, 8000, 20000]
        for beta in beta_sch:
            if check_dl():
                break
            maxit = 600 if beta < 500 else 400
            z_new = lbbf_opt(z_cur, n, beta, maxit)
            h_new = make_feasible(from_logistic_z(z_new, n))
            sc = official_score(h_new)
            update(h_new, sc)
            if sc < best_score + 1e-12:
                h_sym = sym_project(h_new)
                sc2 = official_score(h_sym)
                update(h_sym, sc2)
                z_cur = to_logistic_z(h_sym)

    # Phase 2: Active-set SLSQP in symmetric space, multiple active set sizes
    if time_left() > 40:
        n = n_base
        M = n // 2
        h_cur = sym_project(best_h)
        z_cur = to_logistic_z(h_cur)
        for frac in [0.15, 0.25, 0.4, 0.6, 0.8]:
            if check_dl():
                break
            vals = compute_vals(h_cur)
            active = find_active(vals, frac, 5)
            z_res = slsqp_sym(z_cur, n, active, 400)
            if z_res is not None:
                h_new = make_feasible(from_logistic_z(z_res, n))
                sc = official_score(h_new)
                update(h_new, sc)
                if sc < best_score + 1e-12:
                    h_cur = sym_project(h_new)
                    z_cur = to_logistic_z(h_cur)

    # Phase 3: Full-space SLSQP
    if time_left() > 50:
        n = n_base
        h_cur = best_h.copy()
        for frac in [0.2, 0.35, 0.5]:
            if check_dl():
                break
            vals = compute_vals(h_cur)
            active = find_active(vals, frac, 8)
            h_new = slsqp_full(h_cur, n, active, 500)
            if h_new is not None:
                sc = official_score(h_new)
                update(h_new, sc)
                if sc < best_score + 1e-12:
                    h_cur = h_new

    # Phase 4: Resolution sweep
    if time_left() > 100:
        n_opts = sorted(set([
            max(40, n_base - 6), max(44, n_base - 4), max(48, n_base - 2),
            n_base, min(76, n_base + 2), min(80, n_base + 4), min(86, n_base + 8),
        ]))
        for n_opt in n_opts:
            if check_dl():
                break
            if n_opt == n_base:
                h_int = best_h.copy()
            else:
                h_int = interpolate_to(best_h, n_opt)
            h_sym_o = sym_project(h_int)
            sc_o = official_score(h_sym_o)
            update(h_sym_o, sc_o)
            M_o = n_opt // 2
            z_o = to_logistic_z(h_sym_o)
            for beta in [300, 1500, 8000]:
                if check_dl():
                    break
                z_o = lbbf_opt(z_o, n_opt, beta, 300)
                h_new = make_feasible(from_logistic_z(z_o, n_opt))
                sc = official_score(h_new)
                update(h_new, sc)
                h_sym_o = sym_project(h_new)
                z_o = to_logistic_z(h_sym_o)
            # Active-set at this resolution
            if time_left() > 10:
                vals_o = compute_vals(h_sym_o)
                active_o = find_active(vals_o, 0.35, 6)
                z_res = slsqp_sym(z_o, n_opt, active_o, 350)
                if z_res is not None:
                    h_new = make_feasible(from_logistic_z(z_res, n_opt))
                    sc = official_score(h_new)
                    update(h_new, sc)

    # Phase 5: Random restarts + local refine
    if time_left() > 60:
        n_cur = len(best_h)
        M_cur = n_cur // 2
        n_starts = max(4, min(14, int(time_left() / 12)))
        sigmas = [0.01, 0.02, 0.05, 0.1, 0.2]
        for i in range(n_starts):
            if check_dl():
                break
            sig = sigmas[i % len(sigmas)]
            h_c = make_feasible(best_h + rng.normal(0, sig, size=n_cur))
            h_cs = sym_project(h_c)
            sc_c = official_score(h_cs)
            update(h_cs, sc_c)
            z_c = to_logistic_z(h_cs)
            z_c = lbbf_opt(z_c, n_cur, 5000, 200)
            h_new = make_feasible(from_logistic_z(z_c, n_cur))
            sc = official_score(h_new)
            update(h_new, sc)
            if time_left() > 10:
                vals_c = compute_vals(h_new)
                act_c = find_active(vals_c, 0.3, 5)
                z_r = slsqp_sym(z_c, n_cur, act_c, 200)
                if z_r is not None:
                    h_new2 = make_feasible(from_logistic_z(z_r, n_cur))
                    sc2 = official_score(h_new2)
                    update(h_new2, sc2)

    # Phase 6: Iterative active-set refinement on best
    if time_left() > 50:
        n = len(best_h)
        M = n // 2
        h_cur = sym_project(best_h)
        z_cur = to_logistic_z(h_cur)
        for rnd in range(5):
            if check_dl():
                break
            improved = False
            for frac in [0.1, 0.2, 0.35, 0.55]:
                if check_dl():
                    break
                vals = compute_vals(h_cur)
                active = find_active(vals, frac, 4)
                z_res = slsqp_sym(z_cur, n, active, 400)
                if z_res is not None:
                    h_new = make_feasible(from_logistic_z(z_res, n))
                    sc = official_score(h_new)
                    if sc < best_score - 1e-14:
                        update(h_new, sc)
                        h_cur = sym_project(h_new)
                        z_cur = to_logistic_z(h_cur)
                        improved = True
            if not improved:
                break

    # Phase 7: Full-space active-set on best
    if time_left() > 30:
        n = len(best_h)
        h_cur = best_h.copy()
        for frac in [0.15, 0.3, 0.5]:
            if check_dl():
                break
            vals = compute_vals(h_cur)
            active = find_active(vals, frac, 6)
            h_new = slsqp_full(h_cur, n, active, 500)
            if h_new is not None:
                sc = official_score(h_new)
                update(h_new, sc)
                if sc < best_score + 1e-12:
                    h_cur = h_new

    # Phase 8: Final high-beta polish
    if time_left() > 25:
        n = len(best_h)
        M = n // 2
        h_cur = sym_project(best_h)
        z_cur = to_logistic_z(h_cur)
        for beta in [20000, 60000]:
            if check_dl():
                break
            z_cur = lbbf_opt(z_cur, n, beta, 500)
            h_new = make_feasible(from_logistic_z(z_cur, n))
            sc = official_score(h_new)
            update(h_new, sc)
            h_cur = sym_project(h_new)
            z_cur = to_logistic_z(h_cur)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = make_feasible(final_h)
    final_score = official_score(final_h)

    return final_h, final_score, len(final_h)
