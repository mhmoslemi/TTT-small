def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -6.0, 6.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    ms = shifted.max()
    exp_s = np.exp(shifted - ms)
    logsum = ms + np.log(exp_s.sum())
    obj = cmax + logsum / beta
    w = exp_s / exp_s.sum()
    dx = 2.0 / n
    grad_h = np.zeros(n)

        s = p - (n - 1)
        wp = w[p]

            continue

        n_opts = sorted(set([max(40, n_base - 8), max(44, n_base - 4),
                             n_base, min(76, n_base + 4), min(80, n_base + 8)]))
        for n_opt in n_opts:
            if time_left() < 20:
                break
            h_int = interpolate_to(best_h, n_opt)
            # Anti-sym optimize
            u_opt = get_u_from_h(h_int, n_opt)
            for beta in [1000, 5000]:
                if time_left() < 10:
                    break
                tl = min(time_left() * 0.15, 15.0)
                if tl < 3:
                    break
                u_opt = pg_sym(u_opt, n_opt, beta, time_limit=tl)
            h_cand = build_sym_h(u_opt, n_opt)
            sc = official_score(h_cand)
            update(h_cand, sc)
            # SLSQP
            if time_left() > 15:
                vals = compute_vals(h_cand)
                active = find_active(vals, frac=0.25, min_count=8)
                u_new, sc_new = slsqp_sym(u_opt, n_opt, active, max_iter=200)
                if u_new is not None and sc_new < best_score - 1e-14:
                    update(build_sym_h(u_new, n_opt), sc_new)

    # === PHASE 4: Random restarts in anti-symmetric space ===
    if time_left() > 50:
        n_cur = len(best_h)
        u_best = get_u_from_h(best_h, n_cur)
        n_starts = max(4, min(12, int(time_left() / 20)))
        for i in range(n_starts):
            if time_left() < 15:
                break
            sigma = [0.02, 0.04, 0.06, 0.1, 0.15, 0.03][i % 6]
            u_cand = np.clip(u_best + rng.normal(0, sigma, size=n_cur // 2), 0.0, 1.0)
            u_cand = pg_sym(u_cand, n_cur, 2000, time_limit=min(time_left() * 0.1, 10.0))
            h_cand = build_sym_h(u_cand, n_cur)
            sc = official_score(h_cand)
            update(h_cand, sc)
            if time_left() > 15:
                vals = compute_vals(h_cand)
                active = find_active(vals, frac=0.2, min_count=8)
                u_new, sc_new = slsqp_sym(u_cand, n_cur, active, max_iter=200)
                if u_new is not None and sc_new < best_score - 1e-14:
                    update(build_sym_h(u_new, n_cur), sc_new)

    # === PHASE 5: Intensive final polish ===
    if time_left() > 20:
        # Re-extract anti-symmetric from best
        u_fin = get_u_from_h(best_h, len(best_h))
        n_fin = len(best_h)
        # Escalating beta
        for beta in [5000, 15000, 50000]:
            if time_left() < 5:
                break
            tl = min(time_left() * 0.25, 15.0)
            if tl < 2:
                break
            u_fin = pg_sym(u_fin, n_fin, beta, time_limit=tl)
            h_cand = build_sym_h(u_fin, n_fin)
            sc = official_score(h_cand)
            update(h_cand, sc)

        # Final SLSQP with various active sets
        for frac in [0.15, 0.25, 0.35]:
            if time_left() < 8:
                break
            h_cur = build_sym_h(u_fin, n_fin)
            vals = compute_vals(h_cur)
            active = find_active(vals, frac=frac, min_count=10)
            u_new, sc = slsqp_sym(u_fin, n_fin, active, max_iter=500)
            if u_new is not None and sc < best_score - 1e-14:
                u_fin = u_new
                update(build_sym_h(u_fin, n_fin), sc)

        # Also try full SLSQP on best_h
        if time_left() > 10:
            vals = compute_vals(best_h)
            active = find_active(vals, frac=0.3, min_count=12)
            h_new, sc = slsqp_full(best_h, active, max_iter=400)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)

    # === PHASE 6: Subgradient polishing on full space ===
    if time_left() > 20:
        h_sub = best_h.copy()
        step = 0.05
        for it in range(300):
            if time_left() < 3:
                break
            vals = compute_vals(h_sub)
            cmax = vals.max()
            threshold = cmax - max(0.02 * cmax, 1e-5)
            active_mask = vals >= threshold
            G = grad_all_full(h_sub)
            g = G[active_mask].mean(axis=0)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            h_sub = make_feasible(h_sub - step * g / gnorm)
            sc_sub = official_score(h_sub)
            update(h_sub, sc_sub)
            step *= 0.98

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    assert np.all(final_h >= -1e-15), "Negative values"
    assert np.all(final_h <= 1.0 + 1e-15), "Values > 1"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-10, "Sum constraint violated"

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
