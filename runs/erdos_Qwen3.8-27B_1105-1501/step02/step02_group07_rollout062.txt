    t_valid = t_valid[(t_valid >= 0.0) & (t_valid <= 1.0)]

    vals = C[None, :] + D[None, :] * candidates[:, None]
    max_vals = vals.max(axis=1)
    idx = np.argmin(max_vals)
    return candidates[idx]

    num_k = 2 * n - 1
    C = np.zeros(num_k)
    D = np.zeros(num_k)
    
    for s in range(-(n-1), n):
        k = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        for i in range(i_lo, i_hi + 1):

Actually, you know what, let me try a different angle. The key observation is:

V[k] = dx * sum_i h[i]*(1-h[i+s_k])

= dx * (sum_{i in range_k} h[i] - sum_{i in range_k} h[i]*h[i+s_k])

The first term is just a partial sum of h. The second term is a "correlation" term.

If h is binary (0 or 1), then h[i]*(1-h[i+s]) = 1 iff h[i]=1 and h[i+s]=0. So V[k] = dx * (number of 1s at positions i where i+s is a 0). This is the "overlap" in the discrete sense.

For binary h with sum = n/2, we have exactly n/2 ones and n/2 zeros. V[k] = dx * (number of 1→0 transitions at shift k).

The minimum possible maximum of these transition counts over all binary sequences of length n with n/2 ones... this is related to the "cutting" problem.

For a binary sequence, the sum over all k of V[k] = dx * sum_k (number of 1→0 at shift k) = dx * (number of 1s) * (number of 0s) = dx * (n/2)*(n/2) = dx * n²/4.

And there are 2n-1 shifts, so the average V[k] = dx * n²/4 / (2n-1) ≈ dx * n/8 = (2/n) * n/8 = 1/4.

So the minimum possible maximum is at least 1/4 (by averaging). The current best is about 0.38, which is significantly above 0.25. So there's room for improvement, but it's not trivial.

Wait, that averaging argument gives a lower bound on the maximum. Let me double-check:
sum_k V[k] = dx * sum_k sum_{i in range_k} h[i]*(1-h[i+s_k])

For binary h: h[i]*(1-h[i+s]) = 1 iff h[i]=1 and h[i+s]=0.
So sum_k V[k] = dx * (number of pairs (i,k) such that h[i]=1, h[i+s_k]=0, and both are in range)

def make_feasible(z):

    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0

            lo = mid
        else:
            hi = mid

    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:

                h[idx] += a
                residual -= a
            else:

def coord_optimize_fast(h, m, n):
    num_k = 2 * n - 1
    C = np.zeros(num_k)
    D = np.zeros(num_k)
    for s in range(-(n - 1), n):
        k = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        for i in range(i_lo, i_hi + 1):
            js = i + s
            if i == m:
                D[k] += (1.0 - h[js])
            elif js == m:
                C[k] += h[i]
                D[k] += (-h[i])
            else:
                C[k] += h[i] * (1.0 - h[js])
    D_diff = D[:, None] - D[None, :]
    C_diff = C[None, :] - C[:, None]
    valid = np.abs(D_diff) > 1e-14
    safe_D = np.where(valid, D_diff, 1.0)
    t_mat = C_diff / safe_D
    t_mat = np.where(valid, t_mat, np.nan)

    t_flat = t_flat[(t_flat >= 0.0) & (t_flat <= 1.0)]

        h_new, sc = pg_step(h_cur, beta, alpha_max=0.3, max_ls=10)
        sc = official_score(h_new)
        if sc < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc
        h_cur = h_new
        beta = min(beta * 1.01, 400.0)

    # Phase 8: One more LP round
    if time_left() > 15:
        h_cur = best_h.copy()
        for eps in [0.001, 0.0005]:
            if time_left() < 15:
                break
            for _ in range(15):
                if time_left() < 15:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.08)
                if h_new is not None:
                    if sc < best_score - 1e-14:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
