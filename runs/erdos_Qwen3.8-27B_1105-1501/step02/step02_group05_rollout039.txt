def official_score(h):

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:

                h[idx] += add
                residual -= add
            else:

                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=50, tol=1e-12, time_check=None):
    """Coordinate descent with exact 1D minimization."""
    for sweep in range(max_sweeps):
        if time_check is not None and time_check():
            break
        
        base_c = np.correlate(h, 1.0 - h, mode='full').copy()
        npts = 2 * n - 1
        n_minus_1 = n - 1
        max_change = 0.0
        
        for j in range(n):
            old = h[j]
            
            # Compute coefficients for each lag
            coeffs = np.zeros(npts)
            for p in range(npts):
                s = p - n_minus_1
                c = 0.0
                if s >= 0:
                    if s <= j:
                        c += (1.0 - h[j - s])
                    if j + s < n:
                        c += (-h[j + s])
                else:
                    q = -s
                    if j < n - q:
                        c += (1.0 - h[j + q])
                    if j >= q:
                        c += (-h[j - q])
                coeffs[p] = c
            
            # Minimize max(base_c + (x-old)*coeffs) over x in [0,1]
            # = minimize max(coeffs*u + base_c) over u in [-old, 1-old]
            lo_u = -old
            hi_u = 1.0 - old
            
            best_u, _ = find_min_1d(coeffs, base_c, lo_u, hi_u)
            new_val = old + best_u
            new_val = max(0.0, min(1.0, new_val))
            
            change = abs(new_val - old)
            max_change = max(max_change, change)
            h[j] = new_val
        
        # Project to satisfy sum constraint
        h = make_feasible(h)
        
        if max_change < tol:
            break
    
    return h

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)
