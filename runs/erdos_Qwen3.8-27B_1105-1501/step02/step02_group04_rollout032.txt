import time
from scipy.optimize import minimize


def compute_exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    h = h.astype(float).copy()
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum to be exact
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmin(result)
        else:
            idx = np.argmax(result)
        result[idx] = min(max(result[idx] + diff, 0.0), 1.0)
    return result

    v = 1.0 - h
    r = np.correlate(h, v, mode='full')
    L = float(np.max(r))
    z = np.exp((r - L) / T)
    S = float(np.sum(z))
    return L + T * float(np.log(S))


def smooth_grad(h, T):
    n = len(h)
    v = 1.0 - h
    r = np.correlate(h, v, mode='full')
    L = float(np.max(r))
    z = np.exp((r - L) / T)
    S = float(np.sum(z))
    w = z / S
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        for t in range(n):
            s1 += w[n - 1 + j - t] * v[t]
            s2 += w[n - 1 + t - j] * h[t]
        grad[j] = s1 - s2
    return grad

    return smooth_value(x, T)


def slsqp_jac(x, n, T):
    return smooth_grad(x, T)

    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)

def pair_line_search(h, i, j, n_iter=30):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    best_t = 0.0
    best_f = compute_exact_c5(h)
    # Check endpoints and midpoint
    for test_t in [lo, hi, (lo + hi) / 2.0]:
        h[i] = hi_v - test_t
        h[j] = hj_v + test_t
        f = compute_exact_c5(h)
        if f < best_f:
            best_f = f
            best_t = test_t
        h[i] = hi_v
        h[j] = hj_v
    # Ternary search
    a, b = lo, hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_exact_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_exact_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
        if b - a < 1e-12:
            break
    t_best = (a + b) / 2.0
    # Check exact best
    h[i] = hi_v - t_best
    h[j] = hj_v + t_best
    f_best = compute_exact_c5(h)
    h[i] = hi_v
    h[j] = hj_v

        best_t = t_best
    elif abs(best_t - (lo + hi) / 2.0) < 1e-10:

        best_t = 0.0

import numpy as np
import time
from scipy.optimize import minimize


def compute_exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmin(result)
        else:
            idx = np.argmax(result)
        result[idx] = min(max(result[idx] + diff, 0.0), 1.0)
    return result


def smooth_value(h, T):
    n = len(h)
    v = 1.0 - h
    r = np.correlate(h, v, mode='full')
    L = float(np.max(r))
    z = np.exp((r - L) / T)
    S = float(np.sum(z))
    return L + T * float(np.log(S))


def smooth_grad(h, T):
    n = len(h)
    v = 1.0 - h
    r = np.correlate(h, v, mode='full')
    L = float(np.max(r))
    z = np.exp((r - L) / T)
    S = float(np.sum(z))
    w = z / S
    j_idx = np.arange(n)
    t_idx = np.arange(n)
    M_idx = (n - 1) + j_idx[:, None] - t_idx[None, :]
    N_idx = (n - 1) + t_idx[None, :] - j_idx[:, None]
    M = w[M_idx]
    N = w[N_idx]
    grad = M @ v - N @ h
    return grad

    return smooth_value(x, T)


def slsqp_jac(x, n, T):
    return smooth_grad(x, T)

    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)

            slsqp_obj, x0,
            jac=slsqp_jac,
            args=(n, T),

            bounds=[(0.0, 1.0)] * n,
            constraints=[
                {'type': 'eq', 'fun': sum_con_fun, 'args': (n,),
                 'jac': sum_con_jac, 'args_jac': (n,)}
            ],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        x_out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x_out = x0
    x_out = project_feasible(x_out, n / 2.0)
    return x_out


def pair_line_search(h, i, j, n_iter=25):
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_exact_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_exact_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
        if b - a < 1e-13:
            break
    t_best = (a + b) / 2.0
    h[i] = hi_v - t_best
    h[j] = hj_v + t_best
    f_test = compute_exact_c5(h)
    h[i] = hi_v
    h[j] = hj_v
    f_orig = compute_exact_c5(h)
    if f_test < f_orig - 1e-14:
        return t_best
    return 0.0


def full_pair_sweep(h, rng, deadline=None):
    n = len(h)
    h = h.copy()
    for _ in range(3):
        if deadline is not None and time.time() > deadline:
            break
        perm = rng.permutation(n)
        for a_idx in range(n):
            for b_idx in range(a_idx + 1, n):
                if deadline is not None and time.time() > deadline:
                    return h
                i, j = perm[a_idx], perm[b_idx]
                if i > j:
                    i, j = j, i
                d = pair_line_search(h, i, j, n_iter=20)
                if abs(d) > 1e-12:
                    h[i] -= d
                    h[j] += d
    return h


def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def make_sine_start(n, rng):
    x = np.arange(n, dtype=float)
    num_freqs = 2 + rng.randint(0, 3)
    h = np.full(n, 0.5)
    for k in range(num_freqs):
        freq = 1.0 + k * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3) / (k + 1)
        h += amp * np.sin(2 * np.pi * freq * x / n + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n / 2.0)
    return h


def make_perturbed(base, rng, scale):
    h = base + rng.randn(len(base)) * scale
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, len(base) / 2.0)
    return h


def transfer_solution(h_source, n_target):
    n_src = len(h_source)
    if n_src == n_target:
        return h_source.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_tgt = np.interp(x_tgt, x_src, h_source,
                      left=h_source[0], right=h_source[-1])
    h_tgt = project_feasible(h_tgt, n_target / 2.0)
    return h_tgt


def verify_gradient(h, T, n):
    eps = 1e-6
    g = smooth_grad(h, T)
    for j in range(min(3, n)):
        e = np.zeros(n)
        e[j] = eps
        h_plus = h + e
        h_plus = np.clip(h_plus, 0.0, 1.1)
        h_minus = h - e
        h_minus = np.clip(h_minus, -0.1, 1.0)
        num_grad = (smooth_value(h_plus, T) - smooth_value(h_minus, T)) / (2 * eps)
        if abs(num_grad - g[j]) > 0.1:
            return False
    return True


def anneal_single(h_init, n, deadline, T_fractions=None):
    dx = 2.0 / n
    if T_fractions is None:
        T_fractions = [6.0, 3.0, 1.5, 0.75, 0.4, 0.2, 0.1, 0.05, 0.025]
    cur = h_init.copy()
    cur_c5 = compute_exact_c5(cur)
    for frac in T_fractions:
        if time.time() > deadline:
            break
        T = frac * dx
        if T < 1e-5:
            T = 1e-5
        h_new = run_slsqp(cur, T, n, maxiter=80, ftol=1e-10)
        c5_new = compute_exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            cur = h_new.copy()
            cur_c5 = c5_new
    return cur, cur_c5


def random_pair_polish(h, rng, n, count, deadline):
    h = h.copy()
    for _ in range(count):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = pair_line_search(h, i, j, n_iter=20)
        if abs(d) > 1e-12:
            h[i] -= d
            h[j] += d
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_deadline = t0 + budget_s - 4.0

    # Get parent
    parent = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if (h0.size > 0 and np.all(np.isfinite(h0))
                and np.all(h0 >= -1e-6) and np.all(h0 <= 1.0 + 1e-6)):
            parent = project_feasible(h0, len(h0) / 2.0)
    except (NameError, Exception):
        pass

    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 70

    # Generate diverse candidates
    candidates = []
    if parent is not None:
        candidates.append(parent.copy())

    default_base = parent if parent is not None else np.full(n_base, 0.5)
    default_base = project_feasible(default_base, n_base / 2.0)

    for _ in range(4):
        candidates.append(make_sine_start(n_base, rng))
    candidates.append(make_random_binary(n_base, rng))
    for scale in [0.02, 0.04, 0.07, 0.12]:
        candidates.append(make_perturbed(default_base, rng, scale))

    # Evaluate and select best
    best_h = None
    best_c5 = float('inf')
    for h_c in candidates:
        h_c = project_feasible(h_c, n_base / 2.0)
        c5 = compute_exact_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    if best_h is None:
        best_h = np.full(n_base, 0.5)
        best_c5 = compute_exact_c5(best_h)

    n = n_base

    # Phase 1: Annealed SLSQP on best candidate (and second best)
    phase1_end = t0 + budget_s * 0.42
    sorted_cands = sorted(candidates, key=lambda x: compute_exact_c5(x))
    for ci in range(min(3, len(sorted_cands))):
        if time.time() > phase1_end:
            break
        h_trial = project_feasible(sorted_cands[ci], n / 2.0)
        h_trial, c5_trial = anneal_single(h_trial, n, phase1_end)
        if c5_trial < best_c5 - 1e-14:
            best_c5 = c5_trial
            best_h = h_trial.copy()

    # Phase 2: Basin hopping
    phase2_end = t0 + budget_s * 0.78
    hop = 0
    while time.time() < phase2_end:
        hop += 1
        scale = 0.015 + 0.05 * rng.random() * (1.0 + 0.3 * (hop % 3))
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n / 2.0)

        if time.time() > phase2_end - 15.0:
            break
        hop_deadline = min(phase2_end - 2.0, time.time() + 45.0)
        h_pert, c5_pert = anneal_single(h_pert, n, hop_deadline)
        if c5_pert < best_c5 - 1e-14:
            best_c5 = c5_pert
            best_h = h_pert.copy()

    # Phase 3: Full pair sweep
    pair_deadline = t0 + budget_s - 12.0
    if time.time() < pair_deadline:
        h_polished = full_pair_sweep(best_h, rng, deadline=pair_deadline)
        c5_pol = compute_exact_c5(h_polished)
        if c5_pol < best_c5 - 1e-14:
            best_c5 = c5_pol
            best_h = h_polished.copy()

    # Phase 4: Random pair search with remaining time
    h_final = best_h.copy()
    final_deadline = t0 + budget_s - 4.0
    while time.time() < final_deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = pair_line_search(h_final, i, j, n_iter=25)
        if abs(d) > 1e-12:
            h_final[i] -= d
            h_final[j] += d
            c5_new = compute_exact_c5(h_final)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_final.copy()

    # Final projection
    best_h = project_feasible(best_h, n / 2.0)
    final_c5 = compute_exact_c5(best_h)

    return best_h, final_c5, n
