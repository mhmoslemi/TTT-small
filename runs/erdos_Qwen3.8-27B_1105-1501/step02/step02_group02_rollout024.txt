    n = len(h0)
    dx = 2.0 / n
    cv = _corr_cv(h0)
    t0 = cv.max() + 0.01
    
    def _obj(z):
        return z[-1]
    
    def _obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def _ineq(z):
        h = z[:-1]
        t = z[-1]
        cv_h = _corr_cv(h)
        return t - cv_h[active_indices]
    
    def _ineq_jac(z):
        h = z[:-1]
        m = len(active_indices)
        J = np.zeros((m, len(z)))
        J[:, -1] = 1.0  # derivative w.r.t. t
        for ii, k in enumerate(active_indices):
            s = k - (n - 1)
            if s >= 0:

    def _eq(z):
        return z[:-1].sum() - n / 2.0
    
    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    
    try:

- Contribution from "h[i+s]" terms: j ranges s to n-1, coefficient is +dx*h[j-s] = dx*h[0:l]
  So J[ii, s:n] += dx * h[0:l]

For s < 0, r = -s, l = n - r:
F_s = dx * sum_{i=r}^{n-1} h[i]*(1-h[i-r])

dF_s/dh[j]:
- h[j] appears as h[i] when i=j, contributing dx*(1-h[j-r]) if r <= j <= n-1 (j >= r)
- h[j] appears as h[i-r] when i-r=j, i.e., i=j+r, contributing -dx*h[j+r] if 0 <= j <= n-r-1 (j < n-r = l)

So:
For j = 0, ..., r-1: dF_s/dh[j] = -dx*h[j+r]
For j = r, ..., l-1: dF_s/dh[j] = dx*(1-h[j-r]) - dx*h[j+r]
For j = l, ..., n-1: dF_s/dh[j] = dx*(1-h[j-r])

Jacobian of t - F_s:
For s < 0, r = -s, l = n - r:
J[ii, r:n] += -dx*(1 - h[0:l])  # from h[i] terms: -dx*(1-h[j-r]) for j=r,...,n-1, h[j-r]=h[0:l]
J[ii, 0:l] += dx * h[r:]  # from h[i-r] terms: +dx*h[j+r] for j=0,...,l-1, h[j+r]=h[r:]

def _project(z):

    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

def _active_jac_row(h, k, n, dx):
    """Jacobian row for constraint t - F_k(h) >= 0, w.r.t. h part only."""
    s = k - (n - 1)
    row = np.zeros(n)
    if s >= 0:
        l = n - s
        if l > 0:
            row[:l] += -dx * (1.0 - h[s:s+l])
            row[s:] += dx * h[:l]
    else:
        r = -s
        l = n - r
        if l > 0:
            row[r:] += -dx * (1.0 - h[:l])
            row[:l] += dx * h[r:]
    return row

import numpy as np
from scipy.optimize import minimize
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if res > 1e-12:
        for i in range(n):
            if res <= 0:
                break
            add = min(res, 1.0 - h[i])
            h[i] += add
            res -= add
    elif res < -1e-12:
        for i in range(n):
            if res >= 0:
                break
            sub = min(-res, h[i])
            h[i] -= sub
            res += sub
    return np.clip(h, 0.0, 1.0)


def _corr_cv(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    cv = _corr_cv(h)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -100, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        obj = mc
    else:
        w = ev / es
        obj = mc + np.log(es) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            l = n - s
            if l > 0:
                grad[:l] += wk * dx * (1.0 - h[s:s + l])
                grad[s:] += wk * dx * (-h[:l])
        else:
            r = -s
            l = n - r
            if l > 0:
                grad[r:] += wk * dx * (1.0 - h[:l])
                grad[:l] += wk * dx * (-h[r:])
    return obj, grad


def _lse_slsqp(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _fun(x):
        return _lse_obj_grad(x, beta)[0]

    def _fun_jac(x):
        return _lse_obj_grad(x, beta)[1]

    def _eq(x):
        return x.sum() - n / 2.0

    def _eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(_fun, h0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _active_jac_row(h, k, n, dx):
    s = k - (n - 1)
    row = np.zeros(n)
    if s >= 0:
        l = n - s
        if l > 0:
            row[:l] += -dx * (1.0 - h[s:s + l])
            row[s:] += dx * h[:l]
    else:
        r = -s
        l = n - r
        if l > 0:
            row[r:] += -dx * (1.0 - h[:l])
            row[:l] += dx * h[r:]
    return row


def _active_minimax(h0, maxiter=80, tol=1e-7):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()

    for _outer in range(6):
        cv = _corr_cv(h)
        mx = cv.max()
        active = np.where(cv >= mx - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(cv)])
        m = len(active)

        def _obj(z):
            return z[-1]

        def _obj_jac(z):
            g = np.zeros(len(z))
            g[-1] = 1.0
            return g

        def _ineq(z):
            h_z = z[:-1]
            t = z[-1]
            cv_z = _corr_cv(h_z)
            return t - cv_z[active]

        def _ineq_jac(z):
            h_z = z[:-1]
            J = np.zeros((m, len(z)))
            J[:, -1] = 1.0
            for ii, k in enumerate(active):
                J[ii, :n] = _active_jac_row(h_z, k, n, dx)
            return J

        def _eq(z):
            return z[:-1].sum() - n / 2.0

        def _eq_jac(z):
            g = np.zeros(len(z))
            g[:-1] = 1.0
            return g

        t0 = mx + 0.01
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * (n + 1)

        try:
            res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[
                               {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
                               {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                           ],
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                h_new = _project(res['x'][:-1])
                s_new = _score(h_new)
                s_old = _score(h)
                if s_new < s_old:
                    h = h_new
                cv_new = _corr_cv(h)
                new_active = np.where(cv_new >= cv_new.max() - tol)[0]
                if set(new_active.tolist()) == set(active.tolist()):
                    break
        except Exception:
            break

    return h


def _fista(h0, beta, n_iter=5000, alpha=0.005, mu=0.85):
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)

    for it in range(n_iter):
        _, grad = _lse_obj_grad(y, beta)
        h_new = y + mu * (h_cur - h_prev) - alpha * grad
        h_new = _project(h_new)
        h_prev = h_cur
        h_cur = h_new
        y = h_cur + mu * (h_cur - h_prev)
        y = _project(y)
        h_avg += (h_cur - h_avg) / (it + 1)

        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
        if it % 2000 == 0 and it > 0:
            ha = _project(h_avg)
            sa = _score(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()

    s = _score(h_cur)
    if s < best_s:
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_h = ha.copy()
    return best_h, _score(best_h)


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _mirror(h):
    return _project(h[::-1].copy())


def _complement(h):
    return _project(1.0 - h)


def _chain_optimize(h0, deadline):
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)

    if time.time() < deadline - 20:
        h_new = _lse_slsqp(cur, beta=40, maxiter=80)
        s_new = _score(h_new)
        if s_new < cur_s:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 15:
        h_new = _lse_slsqp(cur, beta=200, maxiter=80)
        s_new = _score(h_new)
        if s_new < cur_s:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 12:
        cur = _active_minimax(cur, maxiter=60, tol=1e-7)
        cur_s = _score(cur)

    if time.time() < deadline - 10:
        h_new = _lse_slsqp(cur, beta=500, maxiter=60)
        s_new = _score(h_new)
        if s_new < cur_s:
            cur, cur_s = h_new, s_new

    if time.time() < deadline - 8:
        cur = _active_minimax(cur, maxiter=50, tol=1e-8)
        cur_s = _score(cur)

    if time.time() < deadline - 5:
        cur, cur_s = _fista(cur, beta=500, n_iter=4000, alpha=0.003, mu=0.88)

    if time.time() < deadline - 3:
        cur = _active_minimax(cur, maxiter=40, tol=1e-8)
        cur_s = _score(cur)

    return cur, cur_s
