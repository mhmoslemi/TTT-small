def _project_onto_simplex(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    threshold = (lo + hi) / 2.0
    x = np.clip(v - threshold, 0.0, 1.0)
    # Fix rounding

    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _gradient(h):
    """Compute gradient of max_d C[d] w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    # Subgradient: average over all active (max) shifts
    active = np.where(corr >= max_val - 1e-10)[0]

    grad = np.zeros(n)

            if 0 <= i - d < n:

def _grad_c(h, k):
    """Gradient of c_k w.r.t. h, for a specific shift k."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g = 1.0 - 2.0 * h
    elif k > 0:
        # c_k = sum_{n=0}^{n-1-k} h[n+k](1-h[n])
        # d/dh[i]: h[i] appears as h[n+k] at n=i-k (if valid) and as h[n] at n=i (if valid)
        if i-k is in [0, n-1-k]: contributes (1-h[i-k])
        if i is in [0, n-1-k]: contributes -h[i+k]
        # Vectorized:
        # For the "h[n+k]" part: g[i] += (1-h[i-k]) for i in [k, n-1]
        #   i.e., g[k:] += (1-h[:n-k])
        # For the "h[n]" part: g[i] += -h[i+k] for i in [0, n-1-k]
        #   i.e., g[:n-k] += -h[k:]
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] += -h[k:]
    else:
        # k < 0, let m = -k > 0
        # c_{-m} = sum_{n=m}^{n-1} h[n-m](1-h[n]) = sum_{j=0}^{n-1-m} h[j](1-h[j+m])
        # d/dh[i]: h[i] appears as h[j] at j=i (if 0≤i≤n-1-m): contributes (1-h[i+m])
        #          h[i] appears as h[j+m] at j=i-m (if m≤i≤n-1): contributes -h[i-m]
        m = -k
        g[:n-m] += (1.0 - h[m:])
        g[m:] += -h[:n-m]
    return g
