- For 0 < k < 1: escape = k (the part [1-k, 1] escapes to the right)
  But wait, we also need x+k ≤ 2, which is always true for x ∈ [0,1] and k < 1.
  And x+k ≥ 0 is always true.
  So escape(k) = k for 0 < k < 1.
- For k = 0: escape = 0
- For -1 < k < 0: escape = |k| (the part [0, 1+k] escapes to the left)

So the maximum escape is 1 (approached as k → 1⁻). That's terrible.

What about a "fat" set? If A is a union of many small intervals spread out, the escape for each shift is distributed.

Actually, I realize the problem is more subtle because h is [0,1]-valued, not just {0,1}. But for the discrete optimization, values in [0,1] are allowed.

Let me think about what the optimal structure looks like. If we want all shifts to have approximately the same escape, we need a set with a very "uniform" difference set. This is related to the concept of a "B-sequence" or "Golay sequence" in combinatorics.

For the discrete problem with n points, the optimal h likely has values that are not just 0 and 1, but take intermediate values to balance the different shifts.

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        room = 1.0 - h
        order = np.argsort(-room)
        for idx in order:
            add = min(residual, room[idx])
            if add < 1e-15:
                break
            h[idx] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    elif residual < -1e-12:
        order = np.argsort(h)

            sub = min(-residual, h[idx])
            if sub < 1e-15:
                break
            h[idx] -= sub
            residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_shifts(h):
    """Compute all shift values (unnormalized)."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    return raw  # length 2n-1

def gradient_shift(h, s):
    """Gradient of f_s = sum h[i]*(1-h[i+s]) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    # h[k] as first factor: contributes (1-h[k+s]) if 0<=k+s<n
    # h[k] as second factor: contributes -h[k-s] if 0<=k-s<n
    ks = np.arange(n)
    # First part
    valid1 = (ks + s >= 0) & (ks + s < n)
    g[valid1] += 1.0 - h[ks[valid1] + s]
    # Second part
    valid2 = (ks - s >= 0) & (ks - s < n)
    g[valid2] -= h[ks[valid2] - s]
    return g

def solve_qp(h0, shift_indices, reg, timeout=30):
    """Solve one QP step.
    h0: current point (feasible)
    shift_indices: indices into the correlate output to constrain
    reg: regularization parameter
    Returns: proposed point or None
    """
    n = len(h0)
    raw = compute_shifts(h0)
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    
    # Get the shifts we want to constrain
    shifts_to_use = []
    for idx in shift_indices:
        s = idx - (n - 1)
        shifts_to_use.append((idx, s, raw[idx]))
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    for idx, s, fval in shifts_to_use:
        g = gradient_shift(h0, s)
        # Linearized: fval + g·(h - h0) <= t
        constraints.append(fval + g @ (h_var - h0) <= t_var)
    
    obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    
    h_new = h_var.value
    if h_new is None or not np.all(np.isfinite(h_new)):
        return None
    
    return make_feasible(h_new)

def line_search(h0, h1, n_eval=10):
    """Line search between h0 and h1. Returns best (h, score)."""
    best_h = h0.copy()
    best_sc = official_score(h0)
    for alpha in np.linspace(0.0, 1.0, n_eval + 1)[1:]:
        h_try = make_feasible(h0 + alpha * (h1 - h0))
        sc = official_score(h_try)
        if sc < best_sc:
            best_sc = sc
            best_h = h_try
    return best_h, best_sc

def smooth_max_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = raw.max()
    shifted = beta * (raw - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_jac(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = raw.max()
    shifted = beta * (raw - cmax)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))

        g = gradient_shift(h, s)
        jac += w * dx * g
