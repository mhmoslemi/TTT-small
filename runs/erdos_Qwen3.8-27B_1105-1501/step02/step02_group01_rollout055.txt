import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -n * 2, n * 2
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def smooth_lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def smooth_lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        s2 = k - (n-1)
        if s2 >= 0:
            if s2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[s2:] += (1.0 - x[:n-s2])
                grad[:n-s2] += (-x[s2:])
        else:
            m2 = -s2
            if m2 == 0:
                grad += (1.0 - x) - x
            else:
                grad[:n-m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n-m2])
    return grad

def do_slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            smooth_lse_obj, h0, args=(beta, dx),

def compute_bc(h, u, i, j, n):

    L = 2 * n - 1

            pj = j - s

                c[k] = -1.0

            if m == 0:
                c[k] = 2.0

def exact_pair_min(h, i, j, dx, current_score):
    """Exact minimization along pair direction (i donates, j receives)."""
    n = len(h)
    u = 1.0 - h
    t_lo = max(h[i] - 1.0, -h[j])
    t_hi = min(h[i], 1.0 - h[j])
    if t_hi - t_lo < 1e-12:
        return None
    
    a = np.correlate(h, u, mode='full')
    b, c = compute_bc(h, u, i, j, n)
    
    candidates = np.array([t_lo, t_hi])
    
    cur_max = np.max(a)
    active = np.where(a >= cur_max - 0.2 * cur_max)[0]
    top_idx = np.argsort(a)[-10:]
    active = np.unique(np.concatenate([active, top_idx]))
    
    inter_points = []
    na = len(active)
    for ii in range(na):
        for jj in range(ii + 1, na):
            ki, kj = active[ii], active[jj]
            alpha = a[ki] - a[kj]
            beta_d = b[ki] - b[kj]
            gamma = -(c[ki] - c[kj])
            if abs(gamma) < 1e-14:
                if abs(beta_d) > 1e-14:
                    ts = -alpha / beta_d
                    if t_lo < ts < t_hi:
                        inter_points.append(ts)
            else:
                disc = beta_d * beta_d - 4.0 * gamma * alpha
                if disc >= 0:
                    sd = np.sqrt(disc)
                    for ts in [(-beta_d + sd) / (2 * gamma), (-beta_d - sd) / (2 * gamma)]:
                        if t_lo < ts < t_hi:
                            inter_points.append(ts)
    
    for ki in active:
        if c[ki] > 1e-14:
            tv = b[ki] / (2.0 * c[ki])
            if t_lo < tv < t_hi:
                inter_points.append(tv)
    
    if inter_points:
        candidates = np.concatenate([candidates, np.array(inter_points)])
    
    candidates = candidates[(candidates >= t_lo) & (candidates <= t_hi)]
    
    best_t = 0.0
    best_val = current_score
    for t in candidates:
        vals = a + b * t - c * t * t
        fval = dx * np.max(vals)
        if fval < best_val - 1e-15:
            best_val = fval
            best_t = t
    
    if best_val >= current_score - 1e-14:
        return None
    
    h_new = h.copy()
    h_new[i] -= best_t
    h_new[j] += best_t
    h_new = make_feasible(h_new)
    exact_val = official_score(h_new)
    if exact_val < current_score - 1e-14:
        return h_new, exact_val
    return None

                di = sorted_idx[ii]
                dj = sorted_idx[jj]
                key = (min(di, dj), max(di, dj))
                if key in seen:
                    continue
                seen.add(key)
                pairs.append((di, dj))
                if len(pairs) >= n_pairs:
                    break
            if len(pairs) >= n_pairs:
                break
        
        # Add some random pairs
        for _ in range(min(30, n * 3)):
            ri = rng_pair = np.random.randint(0, n, 2)
            key = (min(ri[0], ri[1]), max(ri[0], ri[1]))
            if key not in seen and ri[0] != ri[1]:
                pairs.append((ri[0], ri[1]))
                seen.add(key)
        
        improved = False
        for (pi, pj) in pairs:
            if time_limit is not None and t_start is not None:
                if time.time() - t_start > time_limit:
                    break
            result = exact_pair_min(h, pi, pj, dx, cur_score)
            if result is not None:
                h_new, s_new = result
                h = h_new
                improved = True
                cur_score = s_new
                if s_new < best_score:
                    best_score = s_new
                    best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_score

def build_starts(n, parent, rng):
    """Build diverse starting points for dimension n."""
    starts = []
    
    h_interp = make_feasible(interp_to_n(parent, n))
    starts.append(h_interp)
    
    for p in [1.5, 2.0, 3.0, 5.0]:
        starts.append(make_feasible(h_interp ** p))
    
    if n % 2 == 0:
        n_ones = n // 2
        rb = np.zeros(n)
        rb[rng.choice(n, n_ones, replace=False)] = 1.0
        starts.append(rb)
    
    for sigma in [0.03, 0.08, 0.15, 0.3]:
        starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))
    
    starts.append(make_feasible(1.0 - h_interp[::-1]))
    
    # Periodic patterns
    for period in [3, 5, 7]:
        pat = np.zeros(n)
        for i in range(n):
            if (i % period) < (period + 1) // 2:
                pat[i] = 1.0
        pat = make_feasible(pat)
        starts.append(pat)
    
    # Shifted periodic
    for shift in range(1, min(3, n)):
        pat = np.zeros(n)
        for i in range(n):
            if ((i + shift) % 5) < 3:
                pat[i] = 1.0
        starts.append(make_feasible(pat))
    
    # Antisymmetric pairs
    half = n // 2
    asym = np.zeros(n)
    for i in range(half):
        if i % 3 != 0:
            asym[i] = 1.0
    asym[half:] = 1.0 - asym[:half]
    if n % 2 == 1:
        asym[n//2] = 0.5
    starts.append(make_feasible(asym))
    
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.03 * budget_s)
    deadline = t_start + budget_s - reserve
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(52, 0.5)
    
    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    
    rng = np.random.default_rng(seed)
    
    # Determine which n to try
    pn = len(parent)
    n_candidates = sorted(set([pn, pn-2, pn+2, 48, 50, 52, 54, 46, 56]))
    n_candidates = [n for n in n_candidates if 24 <= n <= 80]
    
    if len(n_candidates) > 6:
        n_candidates = n_candidates[:6]
    
    # Process each dimension
    for n in n_candidates:
        if time.time() > deadline - 40:
            break
        
        starts = build_starts(n, parent, rng)
        
        for h0 in starts:
            if time.time() > deadline - 20:
                break
            
            cur_h = h0.copy()
            cur_score = official_score(cur_h)
            
            # Phase 1: Pair descent (main method)
            if time.time() < deadline - 25:
                tl = min(50, (deadline - time.time()) * 0.25)
                ph, ps = pair_descent(cur_h, max_rounds=300, time_limit=tl, t_start=time.time())
                if ps < cur_score - 1e-13:
                    cur_h = ph
                    cur_score = ps
            
            # Phase 2: SLSQP smoothing
            if time.time() < deadline - 20:
                for beta in [80, 300]:
                    if time.time() > deadline - 15:
                        break
                    nh = do_slsqp_smooth(cur_h, beta, maxiter=100)
                    ns = official_score(nh)
                    if ns < cur_score - 1e-13:
                        cur_h = nh
                        cur_score = ns
            
            # Phase 3: More pair descent after SLSQP
            if time.time() < deadline - 15:
                tl = min(30, (deadline - time.time()) * 0.2)
                ph, ps = pair_descent(cur_h, max_rounds=400, time_limit=tl, t_start=time.time())
                if ps < cur_score - 1e-13:
                    cur_h = ph
                    cur_score = ps
            
            # Phase 4: Final SLSQP polish
            if time.time() < deadline - 10:
                for beta in [150, 500]:
                    if time.time() > deadline - 8:
                        break
                    nh = do_slsqp_smooth(cur_h, beta, maxiter=150)
                    ns = official_score(nh)
                    if ns < cur_score - 1e-13:
                        cur_h = nh
                        cur_score = ns
            
            # Phase 5: Final pair pass
            if time.time() < deadline - 10:
                ph, ps = pair_descent(cur_h, max_rounds=500, time_limit=20, t_start=time.time())
                if ps < cur_score - 1e-13:
                    cur_h = ph
                    cur_score = ps
            
            if cur_score < best_score - 1e-13:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
            elif abs(cur_score - best_score) < 1e-12 and len(cur_h) < best_n:
                best_h = cur_h.copy()
                best_score = cur_score
                best_n = len(cur_h)
    
    # Final refinement on best solution with tight pair search
    remaining = deadline - time.time()
    if remaining > 20:
        for trial in range(min(5, int(remaining / 15))):
            if time.time() > deadline - 10:
                break
            h_pert = make_feasible(best_h + 0.02 * rng.standard_normal(best_n))
            ph, ps = pair_descent(h_pert, max_rounds=500, time_limit=12, t_start=time.time())
            if ps < best_score - 1e-13:
                best_h = ph
                best_score = ps
                best_n = len(ph)
    
    # Final SLSQP polish
    if time.time() < deadline - 5:
        fh = do_slsqp_smooth(best_h, 400, maxiter=200)
        fs = official_score(fh)
        if fs < best_score - 1e-13:
            best_h = fh
            best_score = fs
    
    # Final pair pass on best
    if time.time() < deadline - 3:
        ph, ps = pair_descent(best_h, max_rounds=300, time_limit=5, t_start=time.time())
        if ps < best_score - 1e-13:
            best_h = ph
            best_score = ps
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    
    return (best_h, final_score, len(best_h))
