        if stall > 100:

            stall = 0
            if beta < beta_min + 0.5:

            lr = best_lr * 1.1
        else:

            lr *= 0.7
            if lr < 1e-6:

- If all g_l >= 0, minimum is at δ = 0

Finding the exact minimum of a max of linear functions is equivalent to finding the lowest point of the upper envelope, which can be done by linear programming in 1D (or simply by checking all breakpoints).

The breakpoints occur where two lines s_i + δ*g_i = s_j + δ*g_j, i.e., δ = (s_j - s_i)/(g_i - g_j).

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = v.copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_of_smooth_max(h, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*score_d))) w.r.t. h."""
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    exps = np.exp(beta * (scores - max_s))
    w = exps / float(np.sum(exps))
    
    grad = np.zeros(n)
    # d = 0
    grad += w[n-1] * (1.0 - 2.0*h)
    for d in range(1, n):
        m = n - d
        wd = w[n-1+d]
        wm = w[n-1-d]
        # +d: (1-h[j+d]) for j in [0,m-1], -h[j-d] for j in [d,n-1]
        grad[:m] += wd * (1.0 - h[d:d+m])
        grad[d:] += wd * (-h[:m])
        # -d: -h[j+d] for j in [0,m-1], (1-h[j-d]) for j in [d,n-1]
        grad[:m] += wm * (-h[d:d+m])
        grad[d:] += wm * (1.0 - h[:m])
    return grad

def _smooth_descent(h, deadline, beta_init=100.0, use_nesterov=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    beta = beta_init
    lr = 0.3 / beta
    momentum = np.zeros(n)
    y = h.copy()  # Nesterov lookahead
    stall = 0
    momentum_factor = 0.9
    
    for it in range(10000):
        if time.time() > deadline:
            break
        if stall > 200:
            beta = max(5.0, beta * 0.85)
            lr = 0.3 / beta
            momentum *= 0.5
            stall = 0
            if beta < 6.0:
                break
        
        if use_nesterov:
            y = h + momentum_factor * momentum
            grad = _grad_of_smooth_max(y, beta)
        else:
            grad = _grad_of_smooth_max(h, beta)
        
        # Line search
        best_lr_try = 0.0
        best_c5_ls = cur
        for lr_t in [lr, lr*0.5, lr*0.25, lr*1.5, lr*2.0]:
            if use_nesterov:
                new_mom = momentum_factor * momentum - lr_t * grad
                h_t = _project(h + new_mom, target)
            else:
                h_t = _project(h - lr_t * grad, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_ls - 1e-14:
                best_c5_ls = c5_t
                best_lr_try = lr_t
        
        if best_lr_try > 0:
            if use_nesterov:
                new_mom = momentum_factor * momentum - best_lr_try * grad
                h = _project(h + new_mom, target)
                momentum = new_mom
            else:
                h = _project(h - best_lr_try * grad, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = best_lr_try
        else:
            stall += 1
            lr *= 0.8
            momentum *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=50):

    n = len(h)

    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(123)

            break
        improved = False
        order = rng.permutation(n)
        for _ in range(n):
            j = int(order[_])
            k = int(order[(_+1) % n])
            if j == k:
                continue
            # Try direction: h[j] += delta, h[k] -= delta
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if dj > 0:
                    dmax = min(1.0 - h[j], h[k])
                else:
                    dmax = min(h[j], 1.0 - h[k])
                if dmax < 1e-10:
                    continue
                # Binary search for optimal delta
                lo_d, hi_d = 0.0, dmax
                for _ in range(20):
                    mid_d = (lo_d + hi_d) / 2.0
                    h_t = h.copy()
                    h_t[j] += dj * mid_d
                    h_t[k] += dk * mid_d
                    c5_t = _exact_c5(h_t)
                    if c5_t <= cur:
                        lo_d = mid_d
                    else:
                        hi_d = mid_d
                delta = lo_d
                if delta > 1e-9:
                    h[j] += dj * delta
                    h[k] += dk * delta
                    h = _project(h, target)
                    new_c5 = _exact_c5(h)
                    if new_c5 < best_c5 - 1e-14:
                        best_c5 = new_c5
                        best_h = h.copy()
                    cur = _exact_c5(h)
                    improved = True
        if not improved:
            break
    
    return best_h, best_c5

from scipy.optimize import minimize, NonlinearConstraint, LinearConstraint

def _slsqp_refine(h0, deadline, K=20):
    n = len(h0)
    target = n / 2.0
    h0 = h0.copy()
    scores0 = _scores(h0)
    # Select top K active lags
    top_idx = np.argsort(scores0)[::-1][:K]
    z0 = float(np.max(scores0)) + 0.01
    
    x0 = np.concatenate([h0, [z0]])
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def constraint_fun(x):
        h = x[:n]
        z = x[n]
        scores = _scores(h)
        return z - scores[top_idx]
    
    def constraint_jac(x):
        h = x[:n]
        # Compute gradient matrix
        G = np.zeros((2*n-1, n))
        G[n-1] = 1.0 - 2.0*h
        for d in range(1, n):
            m = n - d
            G[n-1+d, :m] += 1.0 - h[d:d+m]
            G[n-1+d, d:] += -h[:m]
            G[n-1-d, :m] += -h[d:d+m]
            G[n-1-d, d:] += 1.0 - h[:m]
        jac = np.zeros((K, n+1))
        for i, idx in enumerate(top_idx):
            jac[i, :n] = -G[idx, :]
            jac[i, n] = -1.0
        return jac
    
    nlcon = NonlinearConstraint(constraint_fun, 0.0, np.inf, jac=constraint_jac)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    lincon = LinearConstraint(A_eq, target, target)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 * 2 + 1.0)]
    
    res = minimize(obj, x0, method='trust-constr', jac=obj_grad,
                   hess='2-point-grad', constraints=[nlcon, lincon],
                   bounds=bounds, options={'maxiter': 200, 'gtol': 1e-8, 'xtol': 1e-10})
    
    if res.success or res.x is not None:
        h_new = _project(res.x[:n], target)
        c5_new = _exact_c5(h_new)
        return h_new, c5_new
    
    return h0, _exact_c5(h0)

One more thought: the "pair descent" idea is actually very powerful because it exactly minimizes the objective along 2D directions. For each pair (j,k), the problem reduces to a 1D problem: minimize max_l (s_l + δ*g_l) over δ. This can be solved exactly by checking the breakpoints.

For the 1D problem: minimize f(δ) = max_l (a_l + b_l * δ) over δ ∈ [0, δ_max]

The function f is piecewise linear and convex. The minimum is at:
- δ = 0, if all b_l <= 0 (f is decreasing)
- δ = δ_max, if all b_l >= 0 (f is increasing)  
- A breakpoint, where two lines with opposite slopes intersect at the minimum

To find the optimal δ:
1. Start at δ = 0
2. Find the active line(s) at the current δ
3. If all active lines have b_l <= 0, the minimum is at the current δ (or we can move right)
4. If all active lines have b_l >= 0, the minimum is at the current δ (or we can move left)
5. Otherwise, find the next breakpoint and move to it

This is essentially the "megapepton" algorithm for 1D min-max. It's O(K) per step where K is the number of lines, and at most K steps.

For our problem, K = 2n-1 = 127 lines. So finding the optimal δ for one pair takes O(127) operations. Very fast!
