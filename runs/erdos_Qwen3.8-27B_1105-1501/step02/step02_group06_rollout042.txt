def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_all_lags(h):
    """Compute all lag values (normalized by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    """Project z onto {h: sum(h) = n/2, 0 <= h <= 1}."""
    n = len(z)
    target = n / 2.0
    h = np.clip(z, 0.0, 1.0)
    # Find alpha such that sum(clip(z + alpha, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return h

def coord_descent_step(h, j, f_vals, n_grid=500):
    """
    For coordinate j, find the value x in [0,1] that minimizes max_p f_p
    where f_p is the p-th lag correlation value.
    Returns the optimal x.
    """
    n = len(h)
    # For each output index p, compute slope and intercept w.r.t. h[j]
    # f_p = slope_p * h[j] + intercept_p
    
    slopes = np.zeros(2*n - 1)
    for p in range(2*n - 1):
        lag = p - (n - 1)
        slope = 0.0
        if lag >= 0:
            # h[j] appears as h[i+lag] when i = j-lag, valid if 0 <= j-lag <= n-1-lag
            if lag <= j <= n-1:
                slope += (1.0 - h[j - lag])
            # h[j] appears as h[i] when i = j, valid if 0 <= j <= n-1-lag
            if 0 <= j <= n-1-lag:
                slope -= h[j + lag]
        else:
            m = -lag
            # h[j] appears as h[i] when i = j, valid if 0 <= j <= n-1-m
            if 0 <= j <= n-1-m:
                slope += (1.0 - h[j + m])
            # h[j] appears as h[i+m] when i = j-m, valid if 0 <= j-m <= n-1-m
            if m <= j <= n-1:
                slope -= h[j - m]
        slopes[p] = slope
    
    # intercepts: f_p - slope_p * h[j]
    intercepts = f_vals - slopes * h[j]
    
    # Find x in [0,1] minimizing max_p (slopes[p]*x + intercepts[p])
    # Use grid search with refinement
    xs = np.linspace(0.0, 1.0, n_grid)
    vals = np.max(slopes[:, None] * xs[None, :] + intercepts[:, None], axis=0)
    i_best = np.argmin(vals)
    
    # Refine around the best point
    if i_best > 0 and i_best < n_grid - 1:
        x_lo = xs[i_best - 1]
        x_hi = xs[i_best + 1]
        xs_fine = np.linspace(x_lo, x_hi, 200)
        vals_fine = np.max(slopes[:, None] * xs_fine[None, :] + intercepts[:, None], axis=0)
        i_best_fine = np.argmin(vals_fine)
        return xs_fine[i_best_fine]
    else:
        return xs[i_best]

def coordinate_descent(h, max_sweeps=100, n_grid=500):
    """Run coordinate descent to minimize the max correlation."""
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        f_vals = compute_all_lags(h)
        improved = False
        for j in range(n):
            x_new = coord_descent_step(h, j, f_vals, n_grid)

                h[j] = x_new
                h = make_feasible(h)
                f_vals = compute_all_lags(h)  # Update after projection
                improved = True
        
        sc = official_score(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        
        if not improved or sc >= best_score - 1e-12:
            break
    
    return best_h, best_score

def block_cd_2d(h, j, k, f_vals):
    """2D coordinate descent for coordinates j and k using cvxpy LP."""
    n = len(h)
    dx = 2.0 / n
    target_sum = h[j] + h[k]
    
    h_var = cp.Variable(2)
    t_var = cp.Variable()
    constraints = [h_var[0] + h_var[1] == target_sum,
                   h_var >= 0, h_var <= 1]
    
    # Build linearizations for top lags
    f_curr = compute_lags(h)
    order = np.argsort(f_curr)[::-1]
    top_lags = order[:20]
    
    for p in top_lags:
        lag = p - (n - 1)
        # f_p = sum of h[i+lag]*(1-h[i]) terms
        # Linearize around current h
        base = f_curr[p]
        g_j = 0.0
        g_k = 0.0
        if lag >= 0:
            if lag <= j <= n - 1:
                g_j += (1.0 - h[j - lag])
            if 0 <= j <= n - 1 - lag:
                g_j -= h[j + lag]
            if lag <= k <= n - 1:
                g_k += (1.0 - h[k - lag])
            if 0 <= k <= n - 1 - lag:
                g_k -= h[k + lag]
        else:
            m = -lag
            if 0 <= j <= n - 1 - m:
                g_j += (1.0 - h[j + m])
            if m <= j <= n - 1:
                g_j -= h[j - m]
            if 0 <= k <= n - 1 - m:
                g_k += (1.0 - h[k + m])
            if m <= k <= n - 1:
                g_k -= h[k - m]
        constraints.append(dx * (base + g_j * (h_var[0] - h[j]) + g_k * (h_var[1] - h[k])) <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS)
        if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
            new_j, new_k = float(h_var.value[0]), float(h_var.value[1])
            if 0 <= new_j <= 1 and 0 <= new_k <= 1:
                h[j] = new_j
                h[k] = new_k
                return True
    except Exception:
        pass
    return False

        improved = False
        for _ in range(n):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if h[i] == h[j] or i == j:
                continue
            h[i], h[j] = h[j], h[i]
            sc = compute_lags(h).max()
            if sc < best - 1e-12:
                best = sc
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h

def smooth_structures(n):
    """Generate various smooth structured initializations."""
    x = (np.arange(n) + 0.5) * 2.0 / n
    structures = []
    # Sine variations
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0)
            structures.append(make_feasible(h))
            h2 = 0.5 + amp * np.cos(2 * np.pi * freq * x / 2.0)
            structures.append(make_feasible(h2))
    # Polynomial bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            h = 0.5 + 0.5 * np.exp(-((x - center * 2) / width) ** 2)
            structures.append(make_feasible(h))
    # Triangular wave
    for period in [2, 4, 6]:
        t = (x * period / 2.0) % 2.0
        h = 0.5 + 0.3 * (1.0 - np.abs(t - 1.0))
        structures.append(make_feasible(h))
    return structures

def subgradient_with_averaging(h, steps=20000, time_limit=None, t_start=None):
    n = len(h)
    h_curr = h.copy()
    h_avg = h.copy()
    best_score = official_score(h_avg)
    best_h = h_avg.copy()
    scale = official_score(h) * n / 2.0 + 1e-6

    for t in range(1, steps + 1):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        f_vals = compute_lags(h_curr)
        p_star = int(np.argmax(f_vals))
        lag = p_star - (n - 1)

        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx + lag] += (1.0 - h_curr[idx])
            grad[idx] -= h_curr[idx + lag]
        else:
            m = -lag
            idx = np.arange(n - m)
            grad[idx] += (1.0 - h_curr[idx + m])
            grad[idx + m] -= h_curr[idx]

        alpha = 2.0 * scale / (np.sqrt(t) * (np.linalg.norm(grad) + 1e-10))
        alpha = min(alpha, 0.1)
        h_curr = make_feasible(h_curr - alpha * grad)

        gamma = 2.0 / (t + 2.0)
        h_avg = (1.0 - gamma) * h_avg + gamma * h_curr

        if t % 500 == 0 or t == steps:
            sc = official_score(h_avg)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_avg.copy()

    sc = official_score(h_avg)
    if sc < best_score:
        best_score = sc
        best_h = h_avg.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')
    best_n = 0

    def time_left():
        return deadline - time.time()

    def update(h, sc):
        nonlocal best_h, best_score, best_n
        if h is not None and np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            update(parent, ps)
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x0 = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(np.pi * x0))
        update(best_h, official_score(best_h))

    # Phase 1: Coordinate descent at parent resolution
    if time_left() > 60:
        tl = min(120, time_left() - 30)
        h_cd, sc_cd = coord_descent(best_h, max_sweeps=300, time_limit=tl, t_start=t_start)
        update(h_cd, sc_cd)

    # Phase 2: Randomized coordinate descent
    if time_left() > 40:
        tl = min(60, time_left() - 20)
        h_r, sc_r = randomized_coord_descent(best_h, max_sweeps=150, time_limit=tl, t_start=t_start, rng=rng)
        update(h_r, sc_r)

    # Phase 3: Resolution sweeping
    if time_left() > 30:
        n_base = len(best_h)
        n_opts = sorted(set([30, 40, 50, 60, 70, 80, 90, 100, 120]))
        n_opts = [n for n in n_opts if n >= 16 and n <= 200]
        for n_opt in n_opts:
            if time_left() < 25:
                break
            if n_opt == n_base:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            # Quick coordinate descent at this resolution
            tl = min(40, time_left() - 15)
            if tl < 10:
                update(h_int, sc_int)
                continue
            h_cd, sc_cd = coord_descent(h_int, max_sweeps=150, time_limit=tl, t_start=t_start)
            update(h_cd, sc_cd)

    # Phase 4: Structured initializations + coordinate descent
    if time_left() > 60:
        n_try = len(best_h)
        for h_struct in smooth_structures(n_try)[:8]:
            if time_left() < 20:
                break
            sc_s = official_score(h_struct)
            tl = min(30, time_left() - 10)
            if tl < 5:
                continue
            h_cd, sc_cd = coord_descent(h_struct, max_sweeps=100, time_limit=tl, t_start=t_start)
            update(h_cd, sc_cd)

    # Phase 5: Binary pattern search + relax
    if time_left() > 40:
        for n_bin in [40, 60, 80]:
            if time_left() < 20:
                break
            h_bin = binary_search_pattern(n_bin, rng)
            h_relax = make_feasible(h_bin + 0.01 * rng.standard_normal(n_bin))
            tl = min(25, time_left() - 10)
            if tl < 5:
                continue
            h_cd, sc_cd = coord_descent(h_relax, max_sweeps=100, time_limit=tl, t_start=t_start)
            update(h_cd, sc_cd)

    # Phase 6: Subgradient with averaging
    if time_left() > 60:
        tl = min(80, time_left() - 20)
        h_sg, sc_sg = subgradient_with_averaging(best_h, steps=15000, time_limit=tl, t_start=t_start)
        update(h_sg, sc_sg)
        if sc_sg < best_score:
            # Polish the subgradient result
            if time_left() > 20:
                tl2 = min(30, time_left() - 10)
                h_pol, sc_pol = coord_descent(h_sg, max_sweeps=100, time_limit=tl2, t_start=t_start)
                update(h_pol, sc_pol)

    # Phase 7: Anti-symmetric transform + refine
    if time_left() > 30:
        h_sym = make_feasible(1.0 - best_h[::-1].copy())
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        if time_left() > 15:
            tl = min(20, time_left() - 8)
            h_cd, sc_cd = coord_descent(h_sym, max_sweeps=80, time_limit=tl, t_start=t_start)
            update(h_cd, sc_cd)

    # Phase 8: Final intensive coordinate descent polish
    if time_left() > 20:
        tl = min(60, time_left() - 5)
        h_pol, sc_pol = coord_descent(best_h, max_sweeps=500, time_limit=tl, t_start=t_start)
        update(h_pol, sc_pol)

    # Phase 9: One more round at best resolution with high precision
    if time_left() > 15:
        tl = min(30, time_left() - 3)
        h_fin, sc_fin = coord_descent(best_h, max_sweeps=300, time_limit=tl, t_start=t_start)
        update(h_fin, sc_fin)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = final_score + 1e-13
    return final_h, final_bound, len(final_h)
