def _project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} approximately."""
    n = len(h)
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:

    # Distribute the difference among interior points

    tol = 0.01 * max_val + 1e-6
    active = np.where(corr >= max_val - tol)[0]

    # Compute subgradient
    sg = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        for j in range(n):
            if d == 0:
                sg[j] += (1.0 - 2.0 * h[j])
            elif d > 0:
                if j < n - d:
                    sg[j] += (1.0 - h[j + d])
                if j >= d:
                    sg[j] += -h[j - d]
            else:
                g = -d
                if j >= g:
                    sg[j] += (1.0 - h[j - g])
                if j < n - g:
                    sg[j] += -h[j + g]
    sg /= len(active)

    # Project step onto feasible set
    h_new = h - step_size * sg
    h_new = _project_sum(h_new, n / 2.0)
    return h_new

def _try_resolution(h_best, best_c5, n_try, deadline, max_passes=80):
    """Resample to new resolution and optimize."""
    n_old = len(h_best)
    if n_try < 16 or n_try > 256:
        return h_best, best_c5, n_old
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_try) + 0.5) / n_try
    h_r = np.interp(x_new, x_old, h_best)
    h_r = np.clip(h_r, 0.0, 1.0)
    h_r = _project_sum(h_r, n_try / 2.0)
    h_r, c5_r = _paired_cd(h_r, max_passes=max_passes, deadline=deadline, n_grid=15)
    if c5_r < best_c5:
        return h_r, c5_r, n_try
    return h_best, best_c5, n_old

def _binary_greedy(n, iterations=5000, seed=42):
    """Construct a good binary solution via greedy/swap."""
    rng = np.random.default_rng(seed)
    target = n // 2
    h = np.zeros(n)
    positions = rng.choice(n, size=target, replace=False)
    h[positions] = 1.0
    best_c5 = _exact_c5(h)

    for _ in range(iterations):
        improved = False
        for _swap in range(200):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline_final = t0 + budget_s * 0.99

    # Start with initial values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 / 2.0
    h = _project_sum(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Coordinate descent on initial point
    deadline1 = t0 + budget_s * 0.25
    h1, c51 = _paired_cd(h, max_passes=400, deadline=deadline1, n_grid=25)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Subgradient pre-optimization + CD
    if time.time() < t0 + budget_s * 0.35:
        h_sg = best_h.copy()
        for i in range(200):
            if time.time() > t0 + budget_s * 0.35:
                break
            h_sg = _subgradient_step(h_sg, step_size=0.015)
        h_sg2, c5_sg = _paired_cd(h_sg, max_passes=100, deadline=t0 + budget_s * 0.38, n_grid=21)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg2.copy()

    # Phase 3: Multi-restart with diverse perturbations
    rng = np.random.default_rng(seed)
    deadline3 = t0 + budget_s * 0.72
    for attempt in range(500):
        if time.time() > deadline3:
            break
        if best_c5 < 0.3806:
            break

        # Diverse perturbation strategies
        strat = attempt % 4
        if strat == 0:
            # Gaussian
            scale = 0.02 + 0.03 * np.sqrt(attempt)
            h_pert = best_h + rng.normal(0, scale, best_n)
        elif strat == 1:
            # Sinusoidal
            freq = rng.integers(1, 7)
            phase = rng.uniform(0, 2 * np.pi)
            h_pert = best_h + 0.08 * np.sin(np.pi * freq * np.arange(best_n) / best_n + phase)
        elif strat == 2:
            # Block shift
            block = rng.integers(2, 8)
            shift = rng.integers(1, block)
            h_pert = np.roll(best_h[~((np.arange(best_n) % block) < shift)], shift)
            # Reconstruct: move elements in blocks
            h_pert = best_h.copy()
            for b in range(0, best_n, block):
                seg = h_pert[b:b+block]
                h_pert[b:b+block] = np.roll(seg, shift)
        else:
            # Binary projection + re-optimization
            h_pert = (best_h > rng.uniform(0.3, 0.7)).astype(float)

        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = _project_sum(h_pert, best_n / 2.0)

        # Quick check before expensive optimization
        c5_pert = _exact_c5(h_pert)
        passes = 150 if c5_pert < best_c5 * 1.1 else 80
        h_try, c5_try = _paired_cd(h_pert, max_passes=passes, deadline=deadline3, n_grid=15)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Resolution sweep
    if time.time() < t0 + budget_s * 0.88:
        for dn in [8, -8, 16, -16, 32, -32, 48, 96]:
            if time.time() > t0 + budget_s * 0.88:
                break
            n_try = best_n + dn
            if n_try < 16 or n_try > 256:
                continue
            h_best, best_c5, best_n = _try_resolution(
                best_h, best_c5, n_try,
                deadline=min(t0 + budget_s * 0.88, time.time() + budget_s * 0.04),
                max_passes=100)

    # Phase 5: Binary construction as alternative start
    if time.time() < t0 + budget_s * 0.93:
        for bn in [best_n, best_n + 8, best_n - 8]:
            if bn < 16 or time.time() > t0 + budget_s * 0.93:
                continue
            hb, c5b = _binary_greedy(bn, iterations=3000, seed=seed + bn)
            # Convert binary to fractional starting point
            h_frac = hb + 0.1 * rng.normal(0, 0.3, bn)
            h_frac = np.clip(h_frac, 0.0, 1.0)
            h_frac = _project_sum(h_frac, bn / 2.0)
            h_frac, c5_frac = _paired_cd(h_frac, max_passes=100,
                                          deadline=min(t0 + budget_s * 0.93, time.time() + 30),
                                          n_grid=15)
            if c5_frac < best_c5:
                best_c5 = c5_frac
                best_h = h_frac.copy()
                best_n = bn

    # Phase 6: Final intensive polish
    if time.time() < deadline_final:
        # Finer grid CD
        h_f, c5_f = _paired_cd(best_h, max_passes=300, deadline=deadline_final, n_grid=41)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
