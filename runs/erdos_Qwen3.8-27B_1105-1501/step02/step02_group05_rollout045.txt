def make_feasible(z):

            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                add = min(residual, 1.0 - h[idx])
                if add > 1e-15:
                    h[idx] += add
                    residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub > 1e-15:
                    h[idx] -= sub
                    residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_lag_grad(h, k):
    """Compute gradient of raw correlation at lag k with respect to h.
    k ranges from -(n-1) to n-1.
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        idx = np.arange(m)
        # h[i+k] * (1-h[i]): d/dh[i+k] = (1-h[i]), d/dh[i] = -h[i+k]
        np.add.at(grad, idx + k, 1.0 - h[idx])
        np.add.at(grad, idx, -h[idx + k])
    else:
        t = -k
        m = n - t
        idx = np.arange(m)
        # h[i] * (1-h[i+t]): d/dh[i] = (1-h[i+t]), d/dh[i+t] = -h[i]
        np.add.at(grad, idx, 1.0 - h[idx + t])
        np.add.at(grad, idx + t, -h[idx])
    return dx * grad

def lp_minimax_step(h, V, active_idx, rho):
    """Solve LP to find direction that reduces active lags.
    Returns direction u (or None if infeasible).
    """
    n = len(h)
    # Objective: minimize t -> c = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # t is last variable
    
    # Inequality constraints: dx*g_p^T u - t <= -V[p] for active p
    # A_ub @ [u; t] <= b_ub
    A_ub = np.zeros((len(active_idx), n + 1))
    b_ub = np.zeros(len(active_idx))
    
    for i, p in enumerate(active_idx):
        k = p - (n - 1)
        g = compute_lag_grad(h, k)  # includes dx
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -V[p]
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h_i <= u_i <= 1-h_i, also |u_i| <= rho
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, V.max()))  # t upper bound
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            u = res.x[:n]
            return u
        elif res.status == 2:  # infeasible
            return None
        else:
            return None
    except Exception:
        return None

    """Gradient of dx * c[k] with respect to h. k is the lag (-(n-1) to n-1)."""

        m = n - k
