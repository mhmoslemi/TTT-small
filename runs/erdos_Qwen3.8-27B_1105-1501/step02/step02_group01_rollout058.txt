    return float(np.max(corr) * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    target = n / 2.0
    lo = -1.5 - float(np.max(z))
    hi = 1.5 - float(np.min(z))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]

            residual -= add

def to_binary(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)

def greedy_construct(n, rng):
    """Greedy: place 1s one at a time to minimize max correlation."""
    h = np.zeros(n)
    ones = n // 2
    remaining = list(range(n))
    for _ in range(ones):
        best_pos = None
        best_score = float('inf')
        for pos in remaining:
            h[pos] = 1.0
            s = exact_score(h)
            h[pos] = 0.0
            if s < best_score:
                best_score = s
                best_pos = pos
        h[best_pos] = 1.0
        remaining.remove(best_pos)
    return h

def block_search(h, widths, max_rounds, rng, time_limit=60.0):
    """Exhaustive block-local search for binary string."""
    n = len(h)
    cur = h.copy()
    cur_score = exact_score(cur)
    t0 = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for w in widths:
            if w > n or w < 4:
                continue
            n_pos = n - w + 1
            for start in rng.permutation(n_pos):
                if time.time() - t0 > time_limit:
                    return cur, cur_score
                block = cur[start:start+w].astype(int)
                k = int(block.sum())
                if k == 0 or k == w:
                    continue
                best_s = cur_score
                best_pat = None
                for combo in combinations(range(w), k):
                    new_block = np.zeros(w, dtype=int)
                    new_block[list(combo)] = 1
                    if np.all(new_block == block):
                        continue
                    cur[start:start+w] = new_block
                    s = exact_score(cur)
                    if s < best_s:
                        best_s = s
                        best_pat = new_block.copy().astype(float)
                    cur[start:start+w] = block
                if best_pat is not None:
                    cur[start:start+w] = best_pat
                    cur_score = best_s
                    improved = True
        if not improved:
            break
    return cur, cur_score

def lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        d = k - (n-1)
        if d >= 0:
            if d == 0:
                grad += 1.0 - 2.0 * x
            else:
                grad[d:] += (1.0 - x[:n-d])
                grad[:n-d] += (-x[d:])
        else:
            m2 = -d
            if m2 == 0:
                grad += 1.0 - 2.0 * x
            else:
                grad[:n-m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n-m2])
    return grad

def slsqp_refine(h0, betas, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    best = h0.copy()
    best_s = exact_score(h0)
    for beta in betas:
        try:
            res = minimize(
                lse_obj, best, args=(beta, dx),
                jac=lambda x: lse_grad(x, beta, dx),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                             'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                new_h = make_feasible(res['x'])
                new_s = exact_score(new_h)
                if new_s < best_s - 1e-12:
                    best = new_h
                    best_s = new_s
        except Exception:
            pass
    return best, best_s

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def periodic_pattern(n, period, ones_in_period):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    for i in range(n):
        if (i % period) in ones_in_period:
            h[i] = 1.0
    # Adjust to have exactly n/2 ones
    target = n // 2
    current = int(h.sum())
    if current > target:
        # Remove some 1s
        excess = current - target
        ones_idx = np.where(h == 1.0)[0]
        for i in ones_idx[:excess]:
            h[i] = 0.0
    elif current < target:
        deficit = target - current
        zeros_idx = np.where(h == 0.0)[0]
        for i in zeros_idx[:deficit]:
            h[i] = 1.0
    return h

                cur[s1:s1+w1] = b1
                cur[s2:s2+w2] = b2
                if s < best_s:
                    best_s = s
                    cur[s1:s1+w1] = nb1
                    cur[s2:s2+w2] = nb2
                    cur_score = best_s
                    found = True
                    improved = True
                    break
            if found:
                break
        if not improved:
            break
    return cur, cur_score

def lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(corr))
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def lse_grad(x, beta, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(corr))
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d == 0:
                grad += 1.0 - 2.0 * x
            else:
                grad[d:] += (1.0 - x[:n-d])
                grad[:n-d] += (-x[d:])
        else:
            m2 = -d
            if m2 == 0:
                grad += 1.0 - 2.0 * x
            else:
                grad[:n-m2] += (1.0 - x[m2:])
                grad[m2:] += (-x[:n-m2])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_refine(h0, betas, maxiter=80):
    n = len(h0)
    dx = 2.0 / n
    best = h0.copy()
    best_s = exact_score(h0)
    for beta in betas:
        try:
            res = minimize(
                lse_obj, best, args=(beta, dx),
                jac=lambda x: lse_grad(x, beta, dx),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
                new_h = make_feasible(res['x'])
                new_s = exact_score(new_h)
                if new_s < best_s - 1e-12:
                    best = new_h
                    best_s = new_s
        except Exception:
            pass
    return best, best_s

def periodic_binary(n, period, ones_frac):
    h = np.zeros(n)
    n_ones_per = max(1, min(period - 1, int(round(ones_frac * period))))
    for i in range(n):
        if (i % period) < n_ones_per:
            h[i] = 1.0
    target = n // 2
    current = int(h.sum())
    if current > target:
        excess = current - target
        idx = np.where(h == 1.0)[0]
        h[idx[:excess]] = 0.0
    elif current < target:
        deficit = target - current
        idx = np.where(h == 0.0)[0]
        h[idx[:deficit]] = 1.0
    return h

def greedy_binary(n, rng):
    h = np.zeros(n)
    ones_needed = n // 2
    remaining = list(range(n))
    for _ in range(ones_needed):
        best_pos = remaining[0]
        best_s = float('inf')
        for pos in remaining:
            h[pos] = 1.0
            s = exact_score(h)
            h[pos] = 0.0
            if s < best_s:
                best_s = s
                best_pos = pos
        h[best_pos] = 1.0
        remaining.remove(best_pos)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.full(50, 0.5)

    best_h = parent.copy()
    best_score = exact_score(parent)

    n_options = [48, 50, 52, 54, 56, 60, 64]
    pn = len(parent)
    if pn % 2 == 0 and 44 <= pn <= 72 and pn not in n_options:
        n_options.append(pn)
    n_options = sorted(set(n_options))

    # Phase 1: Quick scan - find good n and initial configs
    quick_results = []
    for n in n_options:
        if time.time() > deadline - 100:
            break
        candidates = []
        h_interp = make_feasible(interp_to_n(parent, n))
        candidates.append(to_binary(h_interp))
        for period in [6, 8, 10]:
            candidates.append(periodic_binary(n, period, 0.5))
            candidates.append(periodic_binary(n, period, 0.5 + 1.0/period))
        rb = np.zeros(n)
        rb[rng.choice(n, n//2, replace=False)] = 1.0
        candidates.append(rb)
        # Push variants
        for p in [2.0, 3.0]:
            candidates.append(to_binary(make_feasible(h_interp ** p)))

        seen = set()
        unique = []
        for c in candidates:
            s = exact_score(c)
            key = round(s, 3)
            if key not in seen:
                seen.add(key)
                unique.append(c)
        unique.sort(key=lambda x: exact_score(x))
        unique = unique[:4]

        best_for_n = None
        best_s_for_n = float('inf')
        for h0 in unique:
            if time.time() > deadline - 80:
                break
            bh, bs = block_search(h0, [6, 8], max_rounds=5, rng=rng, time_limit=12.0)
            if bs < best_s_for_n:
                best_s_for_n = bs
                best_for_n = bh
        if best_for_n is not None and best_s_for_n < best_score:
            best_h = best_for_n.copy()
            best_score = best_s_for_n
            quick_results.append((best_s_for_n, n, best_for_n.copy()))
        elif best_for_n is not None:
            quick_results.append((best_s_for_n, n, best_for_n.copy()))

    # Phase 2: Focused optimization on best n values
    if quick_results:
        quick_results.sort(key=lambda x: x[0])
        top_entries = quick_results[:3]
    else:
        top_entries = [(best_score, len(parent), to_binary(parent))]

    for score_est, n, h_start in top_entries:
        if time.time() > deadline - 60:
            break
        if time.time() > deadline - 20:
            break

        # Generate more diverse candidates for this n
        more_candidates = [h_start.copy()]
        h_interp = make_feasible(interp_to_n(parent, n))
        more_candidates.append(to_binary(h_interp))
        for period in [5, 7, 9, 11]:
            more_candidates.append(periodic_binary(n, period, 0.5))
        for _ in range(2):
            rb = np.zeros(n)
            rb[rng.choice(n, n//2, replace=False)] = 1.0
            more_candidates.append(rb)
        more_candidates.append(to_binary(make_feasible(1.0 - h_interp[::-1])))
        # Greedy
        if time.time() < deadline - 60:
            more_candidates.append(greedy_binary(n, rng))

        seen = set()
        unique = []
        for c in more_candidates:
            s = exact_score(c)
            key = round(s, 3)
            if key not in seen:
                seen.add(key)
                unique.append(c)
        unique.sort(key=lambda x: exact_score(x))
        unique = unique[:6]

        for h0 in unique:
            if time.time() > deadline - 30:
                break
            cur_h = h0.copy()
            cur_score = exact_score(cur_h)

            # Multi-pass block search with increasing widths
            t_rem = deadline - time.time() - 20
            alloc = max(5.0, min(40.0, t_rem / 3.0))

            cur_h, cur_score = block_search(cur_h, [6, 8], max_rounds=8, rng=rng, time_limit=alloc)
            if time.time() < deadline - 20:
                cur_h, cur_score = block_search(cur_h, [8, 10], max_rounds=5, rng=rng, time_limit=alloc)
            if time.time() < deadline - 20:
                cur_h, cur_score = block_search(cur_h, [10, 12], max_rounds=3, rng=rng, time_limit=min(alloc, 20.0))

            # Multi-block search for harder escapes
            if time.time() < deadline - 20:
                cur_h, cur_score = multi_block_search(cur_h, 4, 4, max_rounds=5, rng=rng, time_limit=10.0)
                cur_h, cur_score = multi_block_search(cur_h, 4, 6, max_rounds=3, rng=rng, time_limit=10.0)

            # SLSQP polish
            if time.time() < deadline - 15:
                cur_h, cur_score = slsqp_refine(cur_h, [80, 200, 500], maxiter=50)

            # Block search after SLSQP (on the binarized result)
            if time.time() < deadline - 15 and cur_score < best_score + 0.005:
                cur_bin = to_binary(cur_h)
                cur_h, cur_score = block_search(cur_bin, [8, 10], max_rounds=5, rng=rng, time_limit=15.0)
                cur_h2, cur_score2 = slsqp_refine(cur_h, [200], maxiter=40)
                if cur_score2 < cur_score:
                    cur_h, cur_score = cur_h2, cur_score2

            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score

    # Phase 3: Final polish and perturbation
    if time.time() < deadline - 10:
        bh, bs = slsqp_refine(best_h, [100, 300, 800], maxiter=60)
        if bs < best_score:
            best_h, best_score = bh, bs

    if time.time() < deadline - 10:
        for _ in range(4):
            if time.time() > deadline - 8:
                break
            pert = to_binary(best_h)
            ones_idx = np.where(pert == 1.0)[0]
            zeros_idx = np.where(pert == 0.0)[0]
            if len(ones_idx) >= 1 and len(zeros_idx) >= 1:
                for _ in range(3):
                    a = rng.integers(len(ones_idx))
                    b = rng.integers(len(zeros_idx))
                    pert[ones_idx[a]] = 0.0
                    pert[zeros_idx[b]] = 1.0
            pert, ps = block_search(pert, [6, 8, 10], max_rounds=5, rng=rng, time_limit=10.0)
            if ps < best_score - 1e-12:
                best_h, best_score = pert, ps

    # Final SLSQP
    if time.time() < deadline - 5:
        bh, bs = slsqp_refine(best_h, [150, 400], maxiter=80)
        if bs < best_score:
            best_h, best_score = bh, bs

    best_h = make_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
