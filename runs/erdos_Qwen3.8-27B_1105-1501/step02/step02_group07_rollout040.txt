```python
    corr = np.correlate(h, 1.0-h, mode='full')  # length 2n-1
    
    # For each output index k, compute the coefficient of h[i]
    # s = k - (n-1)
    m = np.zeros(2*n-1)
    for k in range(2*n-1):
        s = k - (n-1)
        coeff = 0.0
        if 0 <= i+s < n:
            coeff += (1.0 - h[i+s])
        if 0 <= i-s < n:
            coeff -= h[i-s]
        m[k] = coeff

    # corr[n-1] as function of x = h[i]:
    # = -x^2 + x + (corr[n-1] - h[i]*(1-h[i]))
    c_quad = corr[n-1] - h[i]*(1.0-h[i])
    
    # For all other k, corr[k](x) = (corr[k] - m[k]*h[i]) + m[k]*x
    # (linear in x)
    
    # Find x in [0,1] that minimizes max_k corr[k](x)
    # Candidates: 0, 1, 0.5, and all intersection points

    candidates.add(0.0)
    candidates.add(1.0)
    candidates.add(0.5)  # vertex of the quadratic

        if k == n-1:
            continue
        a = -1.0  # quadratic coeff
        b = 1.0 - m[k]  # linear coeff: (1 - m[k])*x
        c = c_quad - (corr[k] - m[k]*h[i])  # constant
        # Solve: -x^2 + (1-m[k])*x + (c_quad - corr[k] + m[k]*h[i]) = 0
        # x^2 - (1-m[k])*x + (corr[k] - m[k]*h[i] - c_quad) = 0

        c2 = corr[k] - m[k]*h[i] - c_quad

            x1 = (-b2 + sqrt_disc) / (2*a2)
            x2 = (-b2 - sqrt_disc) / (2*a2)
            if 0 <= x1 <= 1: candidates.add(x1)
            if 0 <= x2 <= 1: candidates.add(x2)

    # This is O((2n-1)^2) which for n=60 is about 14000 pairs
    # Might be too many, so let's skip this or do a subset

    lines = [(k, m[k], corr[k] - m[k]*h[i]) for k in range(2*n-1) if k != n-1]

            k2, m2, c2 = lines[idx2]
            if abs(m1 - m2) < 1e-15:
                continue
            x_int = (c2 - c1) / (m1 - m2)
            if 0 <= x_int <= 1:
                candidates.add(x_int)
    
    # Evaluate max at each candidate
    best_x = h[i]
    best_val = np.inf

        vals = np.concatenate([vals, [quad_val]])
        val = vals.max()
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Binary search for the right shift
    lo, hi = -1.0, 1.0

        else:
```

```python
def coord_best_fast(h, i, n_grid=500):
    """Find best h[i] using the analytical 1D structure."""
    n = len(h)
    # Precompute current correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # For each output index k, compute the coefficient of h[i] in corr[k]
    # corr[k] = sum of h[j]*(1-h[j+s]) for valid j, where s = k-(n-1)
    # h[i] contributes:
    #   As first factor: (1-h[i+s]) if 0 <= i+s < n
    #   As second factor: -h[i-s] if 0 <= i-s < n
    
    s_vals = np.arange(2*n-1) - (n-1)  # lag for each output index
    coeff1 = np.where((i + s_vals) >= 0) & ((i + s_vals) < n)  # first factor valid
    coeff2 = np.where((i - s_vals) >= 0) & ((i - s_vals) < n)  # second factor valid
    
    m = np.zeros(2*n-1)
    m[coeff1] += (1.0 - h[i + s_vals[coeff1]])
    m[coeff2] -= h[i - s_vals[coeff2]]
    
    # For k = n-1 (lag 0), the relationship is quadratic in h[i]
    # corr[n-1] = sum_j h[j](1-h[j]) = h[i](1-h[i]) + sum_{j!=i} h[j](1-h[j])
    # = -x^2 + x + c where c = sum_{j!=i} h[j](1-h[j])
    c_quad = corr[n-1] - h[i]*(1.0 - h[i])
    
    # For all other k: corr[k](x) = (corr[k] - m[k]*h[i]) + m[k]*x
    intercepts = corr - m * h[i]
    
    # Evaluate f(x) = max(k != n-1: intercepts[k] + m[k]*x, quad_val(x))
    # where quad_val(x) = -x^2 + x + c_quad
    
    def f_of_x(x):
        vals = intercepts + m * x
        vals[n-1] = -x*x + x + c_quad
        return vals.max()
    
    # Check boundaries and vertex
    candidates = [0.0, 1.0, 0.5]
    
    # Grid search
    xs = np.linspace(0, 1, n_grid)
    vals = np.array([f_of_x(x) for x in xs])
    min_idx = np.argmin(vals)
    candidates.append(xs[min_idx])
    
    # Golden section refinement around best grid point
    if 0 < min_idx < n_grid - 1:
        lo = xs[min_idx - 1]
        hi = xs[min_idx + 1]
        gr = (np.sqrt(5) - 1) / 2
        a, b = lo, hi
        c_pt = b - gr*(b-a)
        d_pt = a + gr*(b-a)
        fc = f_of_x(c_pt)
        fd = f_of_x(d_pt)
        for _ in range(50):
            if fc < fd:
                b = d_pt; d_pt = c_pt; fd = fc
                c_pt = b - gr*(b-a); fc = f_of_x(c_pt)
            else:
                a = c_pt; c_pt = d_pt; fc = fd
                d_pt = a + gr*(b-a); fd = f_of_x(d_pt)
        candidates.append((a+b)/2)
    
    # Find best candidate
    best_x = h[i]
    best_val = f_of_x(h[i])
    for x in candidates:
        x = float(np.clip(x, 0, 1))
        v = f_of_x(x)
        if v < best_val:
            best_val = v
            best_x = x
    
    return best_x
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):

        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                residual -= a

            valid1 = (np.arange(n) + s >= 0) & (np.arange(n) + s < n)
            valid2 = (np.arange(n) - s >= 0) & (np.arange(n) - s < n)
            gp[valid1] += (1.0 - h_cur[np.arange(n)[valid1] + s])
            gp[valid2] -= h_cur[np.arange(n)[valid2] - s]
            row = np.zeros(n + 1)
            row[:n] = dx * gp
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(-float(corr[k_idx] * dx))
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []
        for i in range(n):
            lo_i = max(0.0, h_cur[i] - trust)
            hi_i = min(1.0, h_cur[i] + trust)
            bounds.append((lo_i, hi_i))
        bounds.append((0.0, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                         bounds=bounds, method='highs')
            if res.success and res.x is not None:
                d = res.x[:n]
                h_new = make_feasible(h_cur + d)
                sc_new = official_score(h_new)
                if sc_new < best_sc - 1e-13:
                    best_sc = sc_new
                    h_cur = h_new.copy()
                    trust = min(trust * 1.1, 0.5)
                else:
                    trust = max(trust * 0.7, 0.02)
                    if trust < 0.02:
                        break
        except Exception:
            break
    return h_cur, best_sc

def fourier_perturb(h, rng, amplitude=0.1):
    n = len(h)
    x = np.arange(n) / n
    k = rng.integers(1, max(2, n // 4))
    phi = rng.uniform(0, 2 * np.pi)
    amp = amplitude * rng.uniform(0.5, 1.5)
    pert = amp * np.sin(2 * np.pi * k * x + phi)
    h_new = h + pert
    return make_feasible(h_new)

def random_perturb(h, rng, scale=0.05):
    return make_feasible(h + rng.normal(0, scale, len(h)))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.03 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        h_init = make_feasible(h_init)
    except (NameError, AttributeError):
        n_fallback = 60
        x_f = np.arange(n_fallback) / n_fallback
        h_init = 0.5 + 0.15 * np.sin(np.pi * x_f) + 0.05 * np.sin(2 * np.pi * x_f)
        h_init = make_feasible(h_init)

    best_h = h_init.copy()
    best_score = official_score(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: Coordinate descent from initial
    if time_left() > 50:
        h_cd, sc_cd = coord_descent(h_init, n_grid=400, max_sweeps=80)
        if sc_cd < best_score:
            best_h = h_cd.copy()
            best_score = sc_cd

    # Phase 2: LP polish
    if time_left() > 60:
        h_lp, sc_lp = lp_polish(best_h, n_iters=30, trust=0.3)
        if sc_lp < best_score:
            best_h = h_lp.copy()
            best_score = sc_lp

    # Phase 3: Exploration - Fourier perturbations + coordinate descent
    n_base = len(best_h)
    for restart in range(60):
        if time_left() < 60:
            break
        if restart < 10:
            h_try = fourier_perturb(best_h, rng, amplitude=0.08)
        elif restart < 20:
            h_try = fourier_perturb(best_h, rng, amplitude=0.15)
        elif restart < 30:
            h_try = random_perturb(best_h, rng, scale=0.05)
        elif restart < 40:
            h_try = fourier_perturb(best_h, rng, amplitude=0.05)
            h_try = fourier_perturb(h_try, rng, amplitude=0.05)
        else:
            h_try = random_perturb(best_h, rng, scale=0.03)

        sc_try = official_score(h_try)
        if sc_try < best_score + 0.005:
            h_cd, sc_cd = coord_descent(h_try, n_grid=250, max_sweeps=60,
                                        order=rng.permutation(len(h_try)), rng=rng)
            if sc_cd < best_score - 1e-12:
                best_h = h_cd.copy()
                best_score = sc_cd
            if time_left() > 40:
                h_lp2, sc_lp2 = lp_polish(h_cd, n_iters=15, trust=0.2)
                if sc_lp2 < best_score - 1e-12:
                    best_h = h_lp2.copy()
                    best_score = sc_lp2

    # Phase 4: Multi-resolution exploration
    n_best = len(best_h)
    n_list = sorted(set([n for n in [n_best - 4, n_best - 2, n_best, n_best + 2,
                                      n_best + 4, 40, 44, 48, 52, 56, 64, 68, 72, 80]
                         if 24 <= n <= 100 and n % 2 == 0]))
    for n_try in n_list:
        if time_left() < 40:
            break
        if n_try == n_best:
            continue
        h_interp = interpolate_to(best_h, n_try)
        sc_int = official_score(h_interp)
        h_cd, sc_cd = coord_descent(h_interp, n_grid=300, max_sweeps=60)
        if sc_cd < best_score - 1e-12:
            best_h = h_cd.copy()
            best_score = sc_cd

    # Phase 5: Final polish - coordinate descent + LP
    if time_left() > 30:
        h_cd, sc_cd = coord_descent(best_h, n_grid=600, max_sweeps=100,
                                    order=rng.permutation(len(best_h)))
        if sc_cd < best_score:
            best_h = h_cd.copy()
            best_score = sc_cd

    if time_left() > 20:
        h_lp, sc_lp = lp_polish(best_h, n_iters=40, trust=0.15)
        if sc_lp < best_score:
            best_h = h_lp.copy()
            best_score = sc_lp

    # Phase 6: Very fine coordinate descent
    if time_left() > 15:
        h_cd, sc_cd = coord_descent(best_h, n_grid=800, max_sweeps=50)
        if sc_cd < best_score:
            best_h = h_cd.copy()
            best_score = sc_cd

    # Phase 7: Final LP with small trust
    if time_left() > 10:
        h_lp, sc_lp = lp_polish(best_h, n_iters=20, trust=0.05)
        if sc_lp < best_score:
            best_h = h_lp.copy()
            best_score = sc_lp

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
```
