import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(residual) < 1e-14:
                break
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_V(h, shift_idx):
    n = len(h)
    s = shift_idx - (n - 1)
    g = np.zeros(n)
    L = n - abs(s)
    if L <= 0:
        return g
    if s >= 0:
        g[s:n] += (1.0 - h[0:L])
        g[0:L] -= h[s:n]
    else:
        t = -s
        g[0:L] += (1.0 - h[t:n])
        g[t:n] -= h[0:L]
    return g

def fast_weighted_gradient(h, weights):
    n = len(h)
    g = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            g[s:n] += w * (1.0 - h[0:L])
            g[0:L] -= w * h[s:n]
        else:
            t = -s
            g[0:L] += w * (1.0 - h[t:n])
            g[t:n] -= w * h[0:L]
    return g

def softmax_weights(V, beta):
    V_max = V.max()
    shifted = np.clip(beta * (V - V_max), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.full(len(V), 1.0) / len(V)
    return w / ws

    V = compute_values(h)

    g = fast_weighted_gradient(h, w)

    gn = np.linalg.norm(g)

        return h, V.max()

    nbuf = np.linalg.norm(new_buf)

        return h, V.max()
    new_buf /= nbuf
    h_new = make_feasible(h - alpha * new_buf)
    sc = official_score(h_new)
    return h_new, sc, new_buf
