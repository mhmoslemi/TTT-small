    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual among non-saturated coordinates
        if residual > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        if np.any(mask):
            count = np.sum(mask)
            result[mask] += residual / count
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_values_and_grad(h, T_raw, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Stable log-sum-exp
    shifted = (corr - m) / T_raw
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)  # all ≤ 0
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    value = dx * (m + T_raw * log_sum)
    # Weights
    weights = exp_vals / np.sum(exp_vals)
    # Build M1 matrix
    idx = np.arange(n)
    M1 = weights[idx[None, :] - idx[:, None] + n - 1]
    # Gradient
    grad = dx * (M1 @ (1.0 - h) - M1.T @ h)
    return value, grad

def slsqp_step(h, T_raw, n, maxiter=40):
    # Run one SLSQP step
    def obj(x):
        val, _ = smooth_values_and_grad(x, T_raw, n)
        return val
    
    def jac(x):
        _, g = smooth_values_and_grad(x, T_raw, n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or res.fun < float('inf'):
            return np.asarray(res.x, dtype=float)
    except:
        pass
    return h.copy()

def pair_line_search(h, i, j, n_steps=20):
    # Find optimal delta for transferring from i to j
    n = len(h)
    lo = -h[j]
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0
    # Dense sampling
    pts = np.linspace(lo, hi, n_steps)
    best_val = float('inf')
    best_delta = 0.0
    for d in pts:
        h[i] -= d
        h[j] += d
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_delta = d
        h[i] += d
        h[j] -= d
    return best_delta

def run_pair_sweep(h, rng, max_pairs=None):
    n = len(h)
    if max_pairs is None:
        max_pairs = n * (n - 1) // 2
    count = 0
    for i in range(n):
        for j in range(i + 1, n):
            if count >= max_pairs:
                return h
            d = pair_line_search(h, i, j, n_steps=15)
            if abs(d) > 1e-12:
                h[i] -= d
                h[j] += d
            count += 1
    return h
