One more consideration: the `_corr` function using `np.correlate(h, 1-h, mode='full')` gives the cross-correlation. Let me verify the indexing:

`np.correlate(a, v, mode='full')` returns the full discrete cross-correlation. Specifically, for arrays a (length n) and v (length m), the result has length n+m-1, and the k-th element (0-indexed) is sum_i a[i] * v[i + k - (m-1)].

So for a=h (length n) and v=1-h (length n), the result has length 2n-1, and:
result[k] = sum_i h[i] * (1-h[i + k - (n-1)])

For k = n-1: result[n-1] = sum_i h[i] * (1-h[i]) = F_0 (self-overlap)
For k = n-1+1 = n: result[n] = sum_i h[i] * (1-h[i+1]) (shift by 1, but with boundary effects)

Actually, the valid range for shift s is -(n-1) to n-1, where s = k - (n-1).
For shift s: result[n-1+s] = sum_{valid i} h[i] * (1-h[i+s])

This matches our definition of F_s. Good.

Now, one subtle point: `np.correlate` handles the boundary by only summing over valid indices (where both h[i] and h[i+s] are in range). This is exactly what we want - the finite overlap window.

import numpy as np
from scipy.optimize import minimize
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - float(np.max(z))
    hi = 2.0 - float(np.min(z))
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, beta):
    cv = _corr(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta

def _lse_grad(x, beta):
    n = len(x)
    cv = _corr(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n - s] += dx * wk * (1.0 - x[s:])
            g[s:] += dx * wk * (-x[:n - s])
        else:
            r = -s
            g[r:] += dx * wk * (1.0 - x[:n - r])
            g[:n - r] += dx * wk * (-x[r:])
    return g

def _slsqp_smooth(h0, beta, maxiter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0

def _active_set_solve(h0, active_lags, center, rho, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    cv = _corr(h0)
    t0 = cv.max() + 0.001
    n_active = len(active_lags)

    def _obj(z):
        h = z[:-1]
        t = z[-1]
        reg = rho * 0.5 * np.sum((h - center) ** 2)
        return t + reg

    def _obj_jac(z):
        h = z[:-1]
        g = np.zeros(n + 1)
        g[:-1] = 2.0 * rho * (h - center)
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        h_cv = _corr(h)
        return t - h_cv[active_lags]

    def _ieq_jac(z):
        h = z[:-1]
        J = np.zeros((n_active, n + 1))
        for idx, k in enumerate(active_lags):
            s = k - (n - 1)
            J[idx, -1] = 1.0
            if s >= 0:
                J[idx, :n - s] -= dx * (1.0 - h[s:])
                J[idx, s:] += dx * h[:n - s]
            else:
                r = -s
                J[idx, r:] -= dx * (1.0 - h[:n - r])
                J[idx, :n - r] += dx * h[r:]
        return J

    def _eq(z):
        return z[:-1].sum() - n / 2.0

    def _eq_jac(z):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq, 'jac': _ieq_jac},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0

def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))

def _mirror(h):
    return _project(h[::-1].copy())

def _binary_push(h, p):
    return _project(h ** p)

def _symmetric_blend(h):
    return _project((h + h[::-1]) / 2.0)

def _gaussian_smooth(h, width):
    n = len(h)
    if width < 1:
        return h.copy()
    kernel = np.zeros(2 * width + 1)
    for i in range(2 * width + 1):
        d = i - width
        kernel[i] = np.exp(-d * d / (2.0 * width * width))
    kernel /= kernel.sum()
    padded = np.concatenate([h[-width:], h, h[:width]])
    smoothed = np.convolve(padded, kernel, mode='valid')
    return _project(smoothed)
