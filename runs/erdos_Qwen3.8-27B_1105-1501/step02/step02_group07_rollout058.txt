But there's a subtlety: the linearization of the indefinite quadratic form at point h gives:
f_s(h_new) = f_s(h) + grad_s(h)^T (h_new - h) + 0.5 * (h_new-h)^T H_s (h_new-h)

The quadratic term is NOT zero for a general quadratic form (it would be zero for a bilinear form in two different vectors, but here it's a quadratic form in the same vector).

        else:

    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:

        else:

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_gradient(h, weights):
    n = len(h)
    m = len(weights)
    grad = np.zeros(n)
    for idx in range(m):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[0:L] += w * (1.0 - h[s:n])
            grad[s:n] -= w * h[0:L]
        else:
            t = -s
            grad[t:n] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[t:n]
    return grad

def smooth_weights(V, beta):
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(V)) / len(V)
    else:
        w /= ws
    return w

def pg_momentum(h, velocity, beta, alpha=0.3, mom=0.85, max_ls=10):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = smooth_weights(V, beta)
    g = fast_gradient(h, w)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, velocity, f
    g = g / gn
    new_vel = mom * velocity + (1.0 - mom) * g
    best_h = h.copy()
    best_f = f
    step = alpha
    for _ in range(max_ls):
        h_new = make_feasible(h - step * new_vel)
        f_new = official_score(h_new)
        if f_new < best_f:
            best_h = h_new
            best_f = f_new
        step *= 0.6
        if step < 1e-8:
            break
    return best_h, new_vel, best_f

def sca_full(h, trust=0.1):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_rows = []
    b_rows = []
    for p_idx in range(m):
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[0:L] += (1.0 - h[s:n])
            gp[s:n] -= h[0:L]
        else:
            t = -s
            gp[t:n] += (1.0 - h[0:L])
            gp[0:L] -= h[t:n]
        f_val = V_unnorm[p_idx] * dx
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(f_val - dx * np.dot(gp, h))
    if not A_rows:
        return None, None
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    bounds_list = [(max(0.0, h[i] - trust), min(1.0, h[i] + trust)) for i in range(n)]
    bounds_list.append((0.0, None))
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            return h_new, sc
    except Exception:
        pass
    return None, None
