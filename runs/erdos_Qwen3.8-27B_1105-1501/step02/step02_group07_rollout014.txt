import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, idx):
    n = len(h)
    s = idx - (n - 1)
    L = n - abs(s)
    out = np.zeros(n)
    if L <= 0:
        return out
    if s >= 0:
        out[s:] += (1.0 - h[:L])
        out[:L] -= h[s:]
    else:
        t = -s
        out[:L] += (1.0 - h[t:])
        out[t:] -= h[:L]
    return out

def weighted_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        g += w * lag_grad(h, idx)
    return g

def softmax_w(V, beta):
    Vm = V.max()
    x = np.clip(beta * (V - Vm), -500.0, 0.0)
    w = np.exp(x)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(V), 1.0) / len(V)
    return w / s

def pg_step(h, beta, alpha_max, max_ls):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    w = softmax_w(V, beta)
    g = weighted_grad(h, w)
    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    return best_h, best_f

def solve_lp(h, active, trust):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for idx in active:
        g = lag_grad(h, idx)
        row = np.zeros(n + 1)
        row[:n] = dx * g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=[ [1.0] * n + [0.0] ], b_eq=[0.0],
                     bounds=bounds_list, method='highs')
        if res.success and res.x is not None:

def lag_grad(h, idx):
    n = len(h)
    s = idx - (n - 1)
    L = n - abs(s)
    out = np.zeros(n)
    if L <= 0:
        return out
    if s >= 0:
        out[s:] += (1.0 - h[:L])
        out[:L] -= h[s:]
    else:
        t = -s
        out[:L] += (1.0 - h[t:])
        out[t:] -= h[:L]
    return out

def weighted_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    for i in range(len(weights)):
        w = weights[i]
        if abs(w) < 1e-15:
            continue
        g += w * lag_grad(h, i)
    return g

    w = np.exp(x)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(V), 1.0) / len(V)
    return w / s

                h_new, sc = pg_step(h_cur, beta, 0.15, 8)
            else:
                V = compute_values(h_cur)
                Vm = V.max()
                active = np.where(V >= Vm - 0.0008)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(V)])
                h_new, sc = solve_active_lp(h_cur, active, 0.004)
                if h_new is None:
                    h_new, sc = pg_step(h_cur, beta, 0.1, 6)
            if sc < best_score - 1e-15:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.005, 600.0)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Ensure we're at least as good as the initial
    if final_score > official_score(h0) + 1e-12:
        final_h = h0.copy()
        final_score = official_score(h0)
        final_n = len(h0)

    return final_h, final_score, final_n
