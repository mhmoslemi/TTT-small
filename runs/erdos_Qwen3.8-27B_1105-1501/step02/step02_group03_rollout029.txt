            G[p, s:] += dx * (1.0 - h[:L])
        else:
            t = -s
            K = n - t
            G[p, :K] += dx * (1.0 - h[t:])
            G[p, t:] -= dx * h[:K]
    return G

def make_feasible(z, target=None):
    n = len(z)
    if target is None:
        target = n / 2.0
    z = np.asarray(z, dtype=float).copy().ravel()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def logsumexp_obj_grad(h, beta):
    """Compute smooth objective and gradient."""
    n = len(h)
    vals = compute_vals(h)
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = cmax + logsum / beta
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= wsum
    G = grad_vals(h)
    grad = w @ G
    return obj, grad

def pg_optimize(h0, beta, time_limit, max_iter=50000):
    """Projected gradient with backtracking line search."""
    n = len(h0)
    h = h0.copy()
    f, g = logsumexp_obj_grad(h, beta)
    t0 = time.time()
    alpha = 1.0 / (np.dot(g, g) + 1e-15)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gnorm = np.dot(g, g)
        if gnorm < 1e-30:
            break
        step = alpha
        for _ in range(50):
            y = make_feasible(h - step * g)
            f_trial, _ = logsumexp_obj_grad(y, beta)
            if f_trial <= f - 1e-4 * step * gnorm:
                break
            step *= 0.5
        if step < 1e-14:
            break
        f_new, g_new = logsumexp_obj_grad(y, beta)
        if not np.isfinite(f_new):
            break
        improvement = f - f_new
        h, f, g = y, f_new, g_new
        if improvement > 0:
            alpha = min(alpha * 1.5, 2.0 / (gnorm + 1e-15))
        else:
            alpha *= 0.7
    return h

def full_slsqp(h0, max_iter=500, ftol=1e-14):
    """Solve the full minimax NLP using SLSQP."""
    n = len(h0)
    m = 2 * n - 1  # number of correlation lags
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def con_fun(z):
        h = z[:n]
        t = z[-1]
        vals = compute_vals(h)
        return t - vals
    
    def con_jac(z):
        h = z[:n]
        G = grad_vals(h)
        J = np.zeros((m, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    def eq_fun(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    t0_init = official_score(h0) + 1e-5
    z0 = np.concatenate([h0, [t0_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def find_active(vals, frac=0.3, min_count=10):
    cmax = vals.max()
    threshold = cmax - max(0.005 * cmax, 1e-5)
    active = set(np.where(vals >= threshold)[0].tolist())
    top_k = max(min_count, int(frac * len(vals)))
    top_idx = np.argsort(vals)[::-1][:min(top_k, len(vals))]
    active.update(top_idx.tolist())
    return sorted(active)

def symmetrize(h):
    h_sym = 1.0 - h[::-1].copy()
    h_avg = 0.5 * (h + h_sym)
    return make_feasible(h_avg)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)

    h_int = np.interp(tgt_c, src_c, h_src)
