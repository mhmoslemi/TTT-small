def slsqp_optimize(h0, beta, maxiter=80):

    bounds = [(0.0, 1.0)]*n

    res = minimize(lse_obj_and_grad, h0, args=(beta,), jac=True,

        return h_new

import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

def lse_fg(x, beta):

    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(vals)
    ez = np.exp(beta * (vals - m))
    w = ez / ez.sum()
    fval = m + np.log(ez.sum()) / beta
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-13:
            continue
        s = p - (n - 1)
        if s == 0:
            grad += wp * (1.0 - 2.0 * x)

            mm = n - s
            grad[:mm] += wp * (1.0 - x[s:])
            grad[s:] -= wp * x[:mm]
        else:
            t = -s
            mm = n - t
            grad[t:] += wp * (1.0 - x[:mm])
            grad[:mm] -= wp * x[t:]
    grad *= dx
    return fval, grad

def sum_fun(x):
    return np.sum(x) - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(lse_fg, h0, args=(beta,), jac=True,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return None

def project_gradient_descent(h, n_iter=500, tol=1e-10):
    n = len(h)
    best = h.copy()
    best_sc = exact_score(best)
    step = 0.02
    for it in range(n_iter):
        vals = lag_vals(best)
        cmax = vals.max()
        tol_active = max(0.02, 0.05 * cmax)
        active = np.where(vals >= cmax - tol_active)[0]
        w = np.exp(80.0 * (vals[active] - cmax))
        w = w / w.sum()
        grad = np.zeros(n)
        dx = 2.0 / n

            wp = w[idx]
            if wp < 1e-13:

            s = int(p) - (n - 1)

                grad += wp * (1.0 - 2.0 * best)
            elif s > 0:
                mm = n - s
                grad[:mm] += wp * (1.0 - best[s:])
                grad[s:] -= wp * best[:mm]
            else:
                t = -s
                mm = n - t
                grad[t:] += wp * (1.0 - best[:mm])
                grad[:mm] -= wp * best[t:]
        grad *= dx
        d = -grad
        d -= np.mean(d)
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        improved = False
        for mult in [step, step * 0.5, step * 1.5, step * 0.25, step * 2.0]:

    for it in range(n_tries):

        tmp = best.copy()
        for t in ts:
