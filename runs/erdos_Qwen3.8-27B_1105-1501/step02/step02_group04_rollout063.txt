def project_feasible(h):
    """Project h onto {x: 0≤x_i≤1, sum(x) = n/2}"""
    n = len(h)
    target = n / 2.0
    # Bisection on the threshold
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual among non-saturated coordinates
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        if np.any(mask):
            n_free = np.sum(mask)
            adjust = residual / n_free
            result[mask] = np.clip(result[mask] + adjust, 0.0, 1.0)
    return result

def soft_objective(h, T):
    """Compute smooth objective F_T(h) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Stable log-sum-exp
    shifted = (corr - m) / T
    max_shifted = np.max(shifted)  # should be 0
    log_sum_exp = max_shifted + np.log(np.sum(np.exp(shifted)))
    value = dx * (m + T * log_sum_exp)
    
    # Weights
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Gradient
    grad = np.zeros(n)
    for a in range(n):
        g1 = 0.0
        g2 = 0.0
        for p in range(2*n - 1):
            offset = p - (n - 1)
            # First term: (1-h[a-offset]) if 0 ≤ a-offset ≤ n-1
            r = a - offset
            if 0 <= r < n:
                g1 += weights[p] * (1.0 - h[r])
            # Second term: -h[a+offset] if 0 ≤ a+offset ≤ n-1
            s = a + offset
            if 0 <= s < n:
                g2 += weights[p] * h[s]
        grad[a] = dx * (g1 - g2)
    
    return value, grad

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        n_free = int(np.sum(mask))
        if n_free > 0:
            result[mask] = np.clip(result[mask] + residual / n_free, 0.0, 1.0)
    return result


def eq_con_fun(x):
    return np.sum(x) - len(x) / 2.0


def eq_con_jac(x):
    return np.ones(len(x))


def soft_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / T
    max_s = np.max(shifted)
    lse = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    return dx * (m + T * lse)


def soft_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / T
    max_s = np.max(shifted)
    exp_vals = np.exp(shifted - max_s)
    W = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for a in range(n):
        s1 = 0.0
        for p in range(a, a + n):
            r = a + n - 1 - p
            s1 += W[p] * (1.0 - h[r])
        s2 = 0.0
        p_lo = max(0, n - 1 - a)
        p_hi = 2 * n - 2 - a
        for p in range(p_lo, p_hi + 1):
            s = a + p - n + 1
            s2 += W[p] * h[s]
        grad[a] = dx * (s1 - s2)
    return grad


def soft_obj_grad(x, T):
    v = soft_value(x, T)
    g = soft_grad(x, T)
    return v, g


def run_slsqp(h0, T, maxiter=80, ftol=1e-7):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    try:
        result = minimize(soft_obj_grad, h0, args=(T,), jac=True, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if hasattr(result, 'x') and np.all(np.isfinite(result.x)):
            return result.x
    except Exception:
        pass
    return h0.copy()


def pair_line_search(h, i, j):
    """Find optimal transfer from i to j (h[i] -= delta, h[j] += delta)."""
    hi_v = h[i]
    hj_v = h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    # Ternary search
    a, b = lo, hi
    for _ in range(25):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_v - m1
        h[j] = hj_v + m1
        f1 = compute_c5(h)
        h[i] = hi_v - m2
        h[j] = hj_v + m2
        f2 = compute_c5(h)
        h[i] = hi_v
        h[j] = hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0
