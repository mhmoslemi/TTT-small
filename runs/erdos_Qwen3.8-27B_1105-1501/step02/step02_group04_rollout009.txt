import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_D(h):
    """Build derivative matrix D where D[l,j] = d(corr_l)/d(h_j)"""
    n = len(h)
    m = 2 * n - 1
    D = np.zeros((m, n))
    for j in range(n):
        ej = np.zeros(n)
        ej[j] = 1.0
        D[:, j] = np.correlate(ej, 1.0 - h, mode="full") - np.correlate(h, ej, mode="full")
    return D

def lp_solve(h, D, max_corr, trust):
    """Find LP direction to minimize max correlation"""
    n = len(h)
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    slack = max_corr - corr
    
    # Variables: d[0..n-1], theta
    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -theta
    
    A_ub = np.hstack([D, np.ones((m, 1))])
    b_ub = slack
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-trust, trust)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.success and res.x[-1] > 1e-10:
        d = res.x[:n].copy()
        d -= np.mean(d)  # enforce sum=0
        if np.max(np.abs(d)) > 1e-12:
            return d
    return None

def exact_line_search(h, d, corr0):
    """Exact line search using bilinear expansion"""
    n = len(h)
    # Max alpha for feasibility
    alpha_max = 1.0
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    if np.any(pos_mask):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    if np.any(neg_mask):
        alpha_max = min(alpha_max, np.min(-h[neg_mask] / d[neg_mask]))
    if alpha_max < 1e-12:
        return 0.0
    
    lin = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
    quad = np.correlate(d, d, mode="full")
    
    # Grid search
    n_pts = 40
    alphas = np.linspace(0.0, alpha_max, n_pts)
    vals = np.array([np.max(corr0 + a * lin - a * a * quad) for a in alphas])
    best_idx = np.argmin(vals)
    best_alpha = alphas[best_idx]
    best_val = vals[best_idx]
    
    # Golden section refinement
    if 0 < best_idx < n_pts - 1:
        lo = alphas[max(0, best_idx - 1)]
        hi = alphas[min(n_pts - 1, best_idx + 1)]
        for _ in range(30):
            m1 = lo + (hi - lo) * 0.382
            m2 = hi - (hi - lo) * 0.382
            v1 = np.max(corr0 + m1 * lin - m1 * m1 * quad)
            v2 = np.max(corr0 + m2 * lin - m2 * m2 * quad)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        best_alpha = (lo + hi) / 2.0
        best_val = np.max(corr0 + best_alpha * lin - best_alpha * best_alpha * quad)
    
    return best_alpha

def pair_opt_delta(h, i, j):
    """Exact optimal delta for pair (i,j): h[i]-=d, h[j]+=d"""
    hi_v, hj_v = h[i], h[j]
    lo = max(-hj_v, hi_v - 1.0)
    hi = min(hi_v, 1.0 - hj_v)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i], h[j] = hi_v - m1, hj_v + m1

        h[i], h[j] = hi_v - m2, hj_v + m2
        f2 = compute_c5(h)
        h[i], h[j] = hi_v, hj_v
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0
