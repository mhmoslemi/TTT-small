```python
class AsymSolver:
    def __init__(self, n, active):
        self.n = n
        self.m = n // 2
        self.active = np.array(active)
        self.dx = 2.0 / n
    
    def obj(self, z):
        return z[-1]
    
    def obj_jac(self, z):
        g = np.zeros(z.shape)
        g[-1] = 1.0
        return g
    
    def con_fun(self, z):
        u = z[:-1]
        t = z[-1]
        h = build_h_asym(u, self.n)
        vals = self.dx * np.correlate(h, 1.0 - h, mode='full')
        return t - vals[self.active]
    
    def con_jac(self, z):
        u = z[:-1]
        m = self.m
        n = self.n
        h = build_h_asym(u, n)
        dx = self.dx
        k = len(self.active)
        J = np.zeros((k, m + 1))
        for idx, ki in enumerate(self.active):
            s = ki - (n - 1)
            # ∂v_ki/∂h[i] = dx * [(1-h[i+s])*I(0≤i+s<n) - h[i-s]*I(0≤i-s<n)]
            # ∂v_ki/∂u[j] = ∂v_ki/∂h[j] - ∂v_ki/∂h[n-1-j]
            for j in range(m):
                i1 = j
                i2 = n - 1 - j
                g1 = 0.0
                if 0 <= i1 + s < n:
                    g1 += 1.0 - h[i1 + s]
                if 0 <= i1 - s < n:
                    g1 -= h[i1 - s]
                g2 = 0.0
                if 0 <= i2 + s < n:
                    g2 += 1.0 - h[i2 + s]
                if 0 <= i2 - s < n:
                    g2 -= h[i2 - s]
                J[idx, j] = -dx * (g1 - g2)
            J[idx, -1] = 1.0
        return J
```
