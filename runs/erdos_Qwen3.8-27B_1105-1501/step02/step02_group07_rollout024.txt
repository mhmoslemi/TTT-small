    while time.time() - t_start < time_limit * 0.75:
        h_new, sc = pg_step(h_cur, beta, alpha_max=0.4, max_ls=12)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
        h_cur = h_new
        beta = min(beta * 1.02, 200.0)
        if stall > 50:
            break

    # LP
    h_cur = best_h.copy()
    trust = 0.15
    for eps in [0.005, 0.002, 0.001, 0.0005]:
        if time.time() - t_start > time_limit * 0.9:
            break
        for _ in range(15):
            if time.time() - t_start > time_limit * 0.9:
                break
            h_new, sc = active_set_lp(h_cur, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-13:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
            else:
                break

    # Final pairwise
    if time.time() - t_start < time_limit:
        h_new, sc = pair_exchange_pass(best_h, n_coarse=20, n_fine=30, max_passes=3)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(10.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        h = np.asarray(initial_h_values, dtype=float).ravel()
        h = make_feasible(h)
    except (NameError, AttributeError):
        n = 60
        x = np.arange(n) / n
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
        h = make_feasible(h)

    best_h = h.copy()
    best_score = official_score(h)

    def time_left():
        return deadline - time.time()

    # Phase 1: Initial pairwise exchange
    h_cur = best_h.copy()
    while time_left() > 600:
        h_new, sc = pair_exchange_pass(h_cur, n_coarse=16, n_fine=24, max_passes=3)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
            h_cur = h_new
        else:
            h_cur = h_new
            break

    # Phase 2: PG refinement
    h_cur = best_h.copy()
    beta = 25.0
    stall = 0
    while time_left() > 500:
        h_new, sc = pg_step(h_cur, beta, alpha_max=0.4, max_ls=12)
        if sc < best_score - 1e-13:
            best_h = h_new.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
        h_cur = h_new
        beta = min(beta * 1.02, 200.0)
        if stall > 60:
            break

    # Phase 3: LP polishing
    h_cur = best_h.copy()
    trust = 0.2
    for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time_left() < 150:
            break
        for _ in range(30):
            if time_left() < 150:
                break
            h_new, sc = active_set_lp(h_cur, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-13:
                    best_h = h_new.copy()
                    best_score = sc
                    trust = min(trust * 1.1, 0.35)
                h_cur = h_new
            else:
                break

    # Phase 4: Fine pairwise exchange
    while time_left() > 350:
        h_cur = best_h.copy()
        h_new, sc = pair_exchange_pass(h_cur, n_coarse=24, n_fine=36, max_passes=3)
        if sc < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc
        else:
            break

    # Phase 5: Interleave PG and pairwise
    h_cur = best_h.copy()
    beta = 50.0
    stall = 0
    while time_left() > 200:
        h_new, sc = pg_step(h_cur, beta, alpha_max=0.3, max_ls=10)
        if sc < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
        h_cur = h_new
        beta = min(beta * 1.01, 300.0)
        if stall % 20 == 0 and stall > 0:
            h_new2, sc2 = pair_exchange_pass(h_cur, n_coarse=16, n_fine=24, max_passes=2)
            if sc2 < best_score - 1e-14:
                best_h = h_new2.copy()
                best_score = sc2
                h_cur = h_new2
                stall = 0
        if stall > 100:
            h_cur = make_feasible(best_h + rng.normal(0, 0.015, len(best_h)))
            beta = 30.0
            stall = 0

    # Phase 6: LP polish again
    h_cur = best_h.copy()
    trust = 0.1
    for eps in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time_left() < 100:
            break
        for _ in range(25):
            if time_left() < 100:
                break
            h_new, sc = active_set_lp(h_cur, eps=eps, trust=trust)
            if h_new is not None:
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
            else:
                break

    # Phase 7: Try different resolutions
    if time_left() > 80:
        n_base = len(best_h)
        for n_try in [n_base - 2, n_base + 2, n_base - 4, n_base + 4, n_base - 6, n_base + 6]:
            if time_left() < 50 or n_try < 24:
                break
            h_int = interpolate_to(best_h, n_try)
            sc_int = official_score(h_int)
            # Quick local opt at this resolution
            h_cur2 = h_int.copy()
            for _ in range(30):
                if time_left() < 50:
                    break
                h_new2, sc2 = pg_step(h_cur2, 40.0, alpha_max=0.3, max_ls=10)
                if sc2 < sc_int - 1e-15:
                    sc_int = sc2
                h_cur2 = h_new2
            h_cur2, sc_int = pair_exchange_pass(h_cur2, n_coarse=16, n_fine=24, max_passes=2)
            if sc_int < best_score - 1e-14:
                best_h = h_cur2.copy()
                best_score = sc_int

    # Phase 8: Random restarts with local opt
    if time_left() > 120:
        for attempt in range(5):
            if time_left() < 60:
                break
            perturb_scale = 0.03 / (attempt + 1)
            h_try = make_feasible(best_h + rng.normal(0, perturb_scale, len(best_h)))
            h_opt, sc_opt = local_optimize(h_try, time_limit=min(40, time_left() / 3), rng=rng)
            if sc_opt < best_score - 1e-14:
                best_h = h_opt
                best_score = sc_opt

    # Phase 9: Final fine pairwise
    if time_left() > 30:
        h_new, sc = pair_exchange_pass(best_h.copy(), n_coarse=30, n_fine=40, max_passes=5)
        if sc < best_score - 1e-15:
            best_h = h_new
            best_score = sc

    # Phase 10: Final PG + LP
    if time_left() > 15:
        h_cur = best_h.copy()
        beta = 150.0
        for _ in range(100):
            if time_left() < 10:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.2, max_ls=8)
            if sc < best_score - 1e-15:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.01, 400.0)
        # One more LP
        if time_left() > 10:
            for eps in [0.001, 0.0005]:
                if time_left() < 8:
                    break
                h_new, sc = active_set_lp(best_h, eps=eps, trust=0.05)
                if h_new is not None and sc < best_score - 1e-15:
                    best_h = h_new
                    best_score = sc

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score

    return final_h, final_bound, final_n
