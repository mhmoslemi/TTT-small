import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -11.0, 11.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-12:
        # Need to add
        cap = 1.0 - h
        order = np.argsort(-cap)  # largest capacity first
        for idx in order:
            add = min(residual, cap[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-15:
                break
    elif residual < -1e-12:
        # Need to subtract
        order = np.argsort(h)  # smallest values first
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if residual > -1e-15:
                break
    return h

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_matrix(h):
    """Returns G of shape (2n-1, n) where G[p] is grad of vals[p] w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, s:s+L] += dx * (1.0 - h[0:L])
            G[p, 0:L] -= dx * h[s:s+L]
        else:
            t = -s
            L = n - t
            G[p, 0:L] += dx * (1.0 - h[t:t+L])
            G[p, t:t+L] -= dx * h[0:L]
    return G

def slsqp_full_constraints(h0, maxiter=300, all_constraints=True):
    """Solve the NLP with all (or active) constraints."""
    n = len(h0)
    vals0 = compute_vals(h0)
    cmax0 = vals0.max()
    
    if all_constraints:
        active_idx = list(range(2 * n - 1))
    else:
        threshold = cmax0 - 0.02 * cmax0
        active_idx = list(np.where(vals0 >= threshold)[0])
        # Also add top-k
        top_k = max(len(active_idx), 20)
        top = np.argsort(vals0)[::-1][:top_k]
        active_set = set(active_idx)
        active_set.update(top.tolist())
        active_idx = sorted(active_set)
    
    r = len(active_idx)
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(z):
        h = z[:n]
        t = z[-1]
        vals = compute_vals(h)
        return t - vals[active_idx]
    
    def ineq_jac(z):
        h = z[:n]
        G = grad_matrix(h)
        rows = G[active_idx, :]
        J = np.hstack([-rows, np.ones((r, 1))])
        return J
    
    def eq_fun(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    t0 = cmax0 + 1e-6
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]

def pg_logsumexp(h0, beta, max_iter=5000, time_limit=30.0):
    """Projected gradient with log-sum-exp smoothing."""
    n = len(h0)
    h = h0.copy()
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        vals = compute_vals(h)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        # Numerical stability
        w = np.exp(shifted - shifted.max())
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        G = grad_matrix(h)
        grad = w @ G
        gnorm2 = float(grad @ grad)
        if gnorm2 < 1e-20:
            break
        # Backtracking line search
        step = 2.0 / (gnorm2 + 1e-12)
        for _ in range(40):
            y = make_feasible(h - step * grad)
            # Check descent
            vals_y = compute_vals(y)
            cmax_y = vals_y.max()
            shifted_y = beta * (vals_y - cmax_y)
            w_y = np.exp(shifted_y - shifted_y.max())
            ws_y = w_y.sum()
            if ws_y > 1e-300:
                f_y = cmax_y + np.log(ws_y) / beta
            else:
                f_y = cmax_y
            vals_h = vals
            shifted_h = beta * (vals_h - cmax)
            w_h = np.exp(shifted_h - shifted_h.max())
            ws_h = w_h.sum()
            f_h = cmax + np.log(ws_h) / beta
            
            if f_y <= f_h - 1e-4 * step * gnorm2:
                break
            step *= 0.3
        if step < 1e-15:
            break
        h = y
    return h

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -11.0, 11.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if residual > 1e-13:
        cap = 1.0 - h
        order = np.argsort(-cap)
        for idx in order:
            add = min(residual, cap[idx])
            h[idx] += add
            residual -= add
            if residual < 1e-15:
                break
    elif residual < -1e-13:
        order = np.argsort(h)
        for idx in order:
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
            if residual > -1e-15:
                break
    return h

def compute_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[p, s:s + L] += dx * (1.0 - h[0:L])
            G[p, 0:L] -= dx * h[s:s + L]
        else:
            t = -s
            L = n - t
            G[p, 0:L] += dx * (1.0 - h[t:t + L])
            G[p, t:t + L] -= dx * h[0:L]
    return G

def slsqp_solve(h0, maxiter=400, constraint_mode='all'):
    n = len(h0)
    vals0 = compute_vals(h0)
    cmax0 = vals0.max()
    if constraint_mode == 'all':
        active_idx = np.arange(2 * n - 1)
    else:
        threshold = cmax0 - 0.03 * max(cmax0, 1e-6)
        active_idx = np.where(vals0 >= threshold)[0]
        if len(active_idx) < 10:
            active_idx = np.argsort(vals0)[::-1][:10]
    r = len(active_idx)

    def obj(z):
        return z[-1]

    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(z):
        h = z[:n]
        t = z[-1]
        vals
        return t - vals

    def ineq_jac(z):
        h = z[:n]
        G = grad_matrix(h)
        rows = G[active_idx, :]
        return np.hstack([-rows, np.ones((r, 1))])

    def eq_fun(z):
        return np.sum(z[:n]) - n / 2.0

    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    t0 = cmax0 + 5e-6
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)

    return None, None

def pg_logsumexp(h0, beta, max_iter=3000, time_limit=25.0):
    n = len(h0)
    h = h0.copy()
    t0 = time.time()
    f_prev = None
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        vals = compute_vals(h)
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        f_cur = cmax + np.log(ws) / beta
        if f_prev is not None and abs(f_cur - f_prev) < 1e-16:
            break
        f_prev = f_cur
        G = grad_matrix(h)
        grad = w @ G
        gnorm2 = float(grad @ grad)
        if gnorm2 < 1e-22:
            break
        step = 2.0 / (gnorm2 + 1e-12)
        accepted = False
        for _ in range(40):
            y = make_feasible(h - step * grad)
            vals_y = compute_vals(y)
            cmax_y = vals_y.max()
            sh_y = beta * (vals_y - cmax_y)
            sh_y -= sh_y.max()
            w_y = np.exp(sh_y)
            ws_y = w_y.sum()
            f_y = cmax_y + np.log(ws_y) / beta if ws_y > 1e-300 else cmax_y
            if f_y <= f_cur - 1e-4 * step * gnorm2:
                accepted = True
                break
            step *= 0.4
        if not accepted or step < 1e-15:
            break
        h = y
    return h

    if time_left() > 30:
        n_cur = len(best_h)
        if n_cur <= 80:
            h_new, sc = slsqp_solve(best_h, maxiter=300, constraint_mode='all')
            if h_new is not None and sc < best_score - 1e-10:
                update(h_new, sc)

    # Phase 3: PG refinement
    if time_left() > 40:
        for beta in [200, 1000, 5000]:
            if time_left() < 10:
                break
            h_new = pg_logsumexp(best_h, beta, time_limit=min(time_left() * 0.15, 15.0))
            sc = official_score(h_new)
            if sc < best_score - 1e-10:
                update(h_new, sc)

    # Phase 4: Multi-n exploration
    if time_left() > 60:
        n_cur = len(best_h)
        n_options = sorted(set([
            max(40, n_cur - 8), max(42, n_cur - 6), max(44, n_cur - 4),
            max(46, n_cur - 2), n_cur,
            min(80, n_cur + 2), min(82, n_cur + 4), min(84, n_cur + 6)
        ]))
        for n_opt in n_options:
            if time_left() < 20:
                break
            if n_opt == len(best_h):
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            # Try binary search at this n
            if time_left() > 15:
                h_b, sc_b = binary_local_search(h_int, rng, time_limit=min(time_left() * 0.2, 12.0))
                if sc_b < sc_int:
                    h_int = h_b
                    sc_int = sc_b
                if sc_b < best_score - 1e-10:
                    update(h_b, sc_b)
            # Try continuous refinement
            if time_left() > 15:
                h_new = pg_logsumexp(h_int, 1000, time_limit=min(time_left() * 0.2, 10.0))
                sc_new = official_score(h_new)
                if sc_new < sc_int:
                    h_int = h_new
                    sc_int = sc_new
                if sc_new < best_score - 1e-10:
                    update(h_int, sc_new)
            # SLSQP on this candidate
            if sc_int < best_score + 0.005 and time_left() > 10 and n_opt <= 80:
                h_new, sc = slsqp_solve(h_int, maxiter=200, constraint_mode='all')
                if h_new is not None and sc < best_score - 1e-10:
                    update(h_new, sc)
                    h_int = h_new
                    sc_int = sc

    # Phase 5: More binary exploration with different seeds
    if time_left() > 60:
        n_cur = len(best_h)
        for attempt in range(5):
            if time_left() < 15:
                break
            # Different random binary starting points
            h_rand = np.zeros(n_cur)
            positions = rng.permutation(n_cur)[:n_cur // 2]
            h_rand[positions] = 1.0
            h_rand = (h_rand > 0.5).astype(float)
            # Quick hill climb
            h_b, sc_b = binary_hill_climb(h_rand, rng, time_limit=min(time_left() * 0.15, 15.0))
            if sc_b < best_score - 1e-10:
                update(h_b, sc_b)
            # Mixed refine: take binary result, make continuous, optimize
            if time_left() > 15 and sc_b < best_score + 0.01:
                h_mix = h_b.astype(float).copy()
                # Perturb slightly from binary
                h_mix = 0.8 * h_mix + 0.2 * rng.uniform(0, 1, n_cur)
                h_mix = make_feasible(h_mix)
                h_new = pg_logsumexp(h_mix, 2000, time_limit=min(time_left() * 0.15, 10.0))
                sc_new = official_score(h_new)
                if sc_new < best_score - 1e-10:
                    update(h_new, sc_new)
                if time_left() > 10 and n_cur <= 80:
                    h_sls, sc_sls = slsqp_solve(h_new, maxiter=200, constraint_mode='all')
                    if h_sls is not None and sc_sls < best_score - 1e-10:
                        update(h_sls, sc_sls)

    # Phase 6: Final SLSQP with active set refinement
    if time_left() > 20:
        n_cur = len(best_h)
        if n_cur <= 80:
            # Full constraints
            h_new, sc = slsqp_solve(best_h, maxiter=400, constraint_mode='all')
            if h_new is not None and sc < best_score - 1e-10:
                update(h_new, sc)
            # Active set only
            vals = compute_vals(best_h)
            cmax = vals.max()
            threshold = cmax - 0.02 * cmax
            active = np.where(vals >= threshold)[0]
            h_new, sc = slsqp_solve(best_h, maxiter=400, constraint_mode='active')
            if h_new is not None and sc < best_score - 1e-10:
                update(h_new, sc)

    # Phase 7: PG final polish with high beta
    if time_left() > 15:
        for beta in [5000, 20000]:
            if time_left() < 8:
                break
            h_new = pg_logsumexp(best_h, beta, time_limit=min(time_left() * 0.3, 10.0))
            sc = official_score(h_new)
            if sc < best_score - 1e-10:
                update(h_new, sc)

    # Phase 8: One more binary + continuous cycle on best
    if time_left() > 30:
        n_cur = len(best_h)
        h_b, sc_b = binary_hill_climb(best_h, rng, time_limit=min(time_left() * 0.2, 15.0))
        if sc_b < best_score - 1e-10:
            update(h_b, sc_b)
        # Continuous refinement of the best
        h_cont = best_h.astype(float).copy()
        h_cont = pg_logsumexp(h_cont, 3000, time_limit=min(time_left() * 0.2, 10.0))
        sc_cont = official_score(h_cont)
        if sc_cont < best_score - 1e-10:
            update(h_cont, sc_cont)
        if time_left() > 10 and len(best_h) <= 80:
            h_sls, sc_sls = slsqp_solve(best_h, maxiter=300, constraint_mode='all')
            if h_sls is not None and sc_sls < best_score - 1e-10:
                update(h_sls, sc_sls)

    # Final validation
    final_h = make_feasible(best_h)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_score = official_score(final_h)
    final_n = len(final_h)

    # Ensure sum constraint
    s = np.sum(final_h)
    target = final_n / 2.0
    if abs(s - target) > 1e-10:
        final_h = make_feasible(final_h)
        final_score = official_score(final_h)

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
