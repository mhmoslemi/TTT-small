def gradient_lag(h, off):
    n = len(h)
    g = np.zeros(n)
    # First part: (1-h[j-off]) for valid 0 <= j-off < n
    # j ranges from max(0, off) to min(n-1, n-1+off)
    lo1 = max(0, off)
    hi1 = min(n, n + off)  # exclusive
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        g[idx] += (1.0 - h[idx - off])
    # Second part: -h[j+off] for valid 0 <= j+off < n
    # j ranges from max(0, -off) to min(n-1, n-1-off)
    lo2 = max(0, -off)
    hi2 = min(n, n - off)  # exclusive
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        g[idx] += -h[idx + off]
    return g

        t_var <= 0,  # ensure descent (or remove for feasibility check)

    high_bound[at_low] = np.inf

    for j in range(n):

        else:
            constraints.append(d_var[j] >= -1)
            constraints.append(d_var[j] <= 1)

        return None
    if d_var.value is None:
        return None

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def gradient_lag(h, off):
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, off)
    hi1 = min(n, n + off)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        g[idx] += (1.0 - h[idx - off])
    lo2 = max(0, -off)
    hi2 = min(n, n - off)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        g[idx] += -h[idx + off]
    return g

def all_gradients(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    max_val = corr.max()
    thresh = max_val - 0.05 / dx * 0.01
    active = np.where(corr >= thresh)[0]
    grads = []
    for p in active:
        off = p - (n - 1)
        grads.append(gradient_lag(h, off))
    if not grads:
        off = 0
        grads.append(gradient_lag(h, off))
    return np.array(grads) * dx, active

def lp_descent_direction(h, k_active=15, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    S = corr.max()
    top_idx = np.argsort(corr)[-k_active:]
    grads = np.zeros((len(top_idx), n))
    for i, p in enumerate(top_idx):
        off = p - (n - 1)
        grads[i] = gradient_lag(h, off) * dx
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    at_low = h < 1e-8
    at_high = h > 1.0 - 1e-8
    constraints = [cp.sum(d_var) == 0]
    for i in range(len(top_idx)):
        constraints.append(grads[i] @ d_var <= t_var)
    for j in range(n):
        if at_low[j]:
            constraints.append(d_var[j] >= -1e-4)
            constraints.append(d_var[j] <= 1.0)
        elif at_high[j]:
            constraints.append(d_var[j] >= -1.0)
            constraints.append(d_var[j] <= 1e-4)
        else:
            constraints.append(d_var[j] >= -1.0)
            constraints.append(d_var[j] <= 1.0)
    if lam > 0:
        obj = cp.Minimize(t_var + lam * cp.sum_squares(d_var))
    else:
        obj = cp.Minimize(t_var + 1e-10 * cp.sum_squares(d_var))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None, 0.0
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, 0.0
    if d_var.value is None:
        return None, 0.0
    d = d_var.value
    if not np.all(np.isfinite(d)):
        return None, 0.0
    return d, float(t_var.value) if t_var.value is not None else 0.0

def active_set_qp(h0, k_active=8, max_iter=25, reg_start=2.0, reg_end=0.002):
    n = len(h0)
    h = h0.copy()
    best = official_score(h)
    dx = 2.0 / n
    stall = 0
    for it in range(max_iter):
        reg = reg_start * (reg_end / reg_start) ** (it / max(1, max_iter - 1))
        vals = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(vals)[-k_active:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        for p in top_idx:
            off = p - (n - 1)
            g = gradient_lag(h, off) * dx
            constraints.append(vals[p] + g @ (h_var - h) <= t_var)
        obj = cp.Minimize(t_var + reg / 2.0 * cp.sum_squares(h_var - h))
        prob = cp.Problem(obj, constraints)
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_new = make_feasible(h_var.value)
        sc = official_score(h_new)
        if sc < best - 1e-14:
            h = h_new
            best = sc
            stall = 0
        else:
            accepted = False
            for alpha in [0.6, 0.4, 0.2, 0.1]:

                    best = sc_try
                    accepted = True

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))

def slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_max_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

    dx = 2.0 / n
    for rnd in range(max_rounds):

                continue
            best_delta = 0.0
            best_sc = cur_score
            for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.03, -0.03]:

        n_base = len(best_h)
        n_options = []
        for dn in [-10, -6, -4, -2, 0, 2, 4, 6, 8, 10, 12]:
            nn = n_base + dn
            if 16 <= nn <= 100:
                n_options.append(nn)
        n_options = sorted(set(n_options))
        for n_opt in n_options:
            if remaining() < 20:
                break
            h_int = interpolate_to(best_h, n_opt)
            update(h_int, official_score(h_int))
            h_opt, sc_opt = lp_direction_search(h_int, max_iters=100, k_active=12)
            update(h_opt, sc_opt)
            if remaining() > 20:
                h_qp, sc_qp = active_set_qp(h_int, k_active=10, max_iter=15,
                                            reg_start=1.0, reg_end=0.005)
                update(h_qp, sc_qp)

    # Phase 9: Random perturbations + LP
    if remaining() > 80:
        for i in range(20):
            if remaining() < 20:
                break
            sigma = 0.01 + 0.04 * rng.random()
            n = len(best_h)
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = lp_direction_search(h_cand, max_iters=100, k_active=15)
            update(h_opt, sc_opt)

    # Phase 10: Fine-tune with LP at very tight tolerance
    if remaining() > 40:
        h_lp5, sc_lp5 = lp_direction_search(best_h, max_iters=300, k_active=20, tol=1e-11)
        update(h_lp5, sc_lp5)

    # Phase 11: Final QP + LP squeeze
    if remaining() > 30:
        h_opt, sc_opt = active_set_qp(best_h, k_active=15, max_iter=15,
                                      reg_start=0.5, reg_end=0.001)
        update(h_opt, sc_opt)
        if remaining() > 20:
            h_lp6, sc_lp6 = lp_direction_search(best_h, max_iters=200, k_active=25, tol=1e-12)
            update(h_lp6, sc_lp6)

    # Final validation
    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(np.pi * x))
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    assert abs(np.sum(final_h) - len(final_h) / 2.0) < 1e-10, "Sum constraint violated"
    assert np.all(final_h >= -1e-12) and np.all(final_h <= 1.0 + 1e-12), "Box constraint violated"
    return final_h, final_score + 1e-14, len(final_h)
