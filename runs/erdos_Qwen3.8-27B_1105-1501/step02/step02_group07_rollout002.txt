import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10.0, 10.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                a = min(residual, 1.0 - h[idx])
                h[idx] += a
                residual -= a
            else:
                a = min(-residual, h[idx])
                h[idx] -= a
                residual += a
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_single(h, p_idx):
    n = len(h)
    s = p_idx - (n - 1)
    L = n - abs(s)
    gp = np.zeros(n)
    if L <= 0:
        return gp
    if s >= 0:
        gp[s:n] = (1.0 - h[0:L])
        gp[0:L] -= h[s:n]
    else:
        t = -s
        gp[0:L] = (1.0 - h[t:n])
        gp[t:n] -= h[0:L]
    return gp

def smooth_max_and_grad(h, beta):
    n = len(h)
    V = compute_values(h)
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0 / len(V))
    else:
        w /= ws
    g = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] > 1e-10:
            g += w[idx] * gradient_single(h, idx)
    f_val = V_max + np.log(max(ws, 1e-300)) / beta
    return f_val, g

def bfgs_momentum_step(h, g, prev_g, prev_step, max_alpha=0.5, max_ls=12):
    n = len(h)
    g = g - g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, official_score(h)
    g = g / gn
    # BFGS-like direction
    d = g
    if prev_g is not None and prev_step is not None:
        rho = np.dot(g - prev_g, prev_step)
        if abs(rho) > 1e-15:
            beta_rho = rho / (np.dot(prev_step, prev_step) + 1e-15)
            d = g + beta_rho * prev_step
    dn = np.linalg.norm(d)
    if dn < 1e-15:
        d = g
    else:
        d = d / dn
    best_h = h.copy()
    best_f = official_score(h)
    alpha = max_alpha
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * d)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.6
        if alpha < 1e-8:
            break
    step = best_h - h
    return best_h, best_f, step

def active_set_lp(h, eps=0.004, trust=0.25, max_active=80):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    if len(active) > max_active:
        order = np.argsort(V)[::-1]
        active = order[:max_active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        gp = gradient_single(h, p_idx)
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx] + 1e-12)
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        lo_b = max(-h[i], -trust)
        hi_b = min(1.0 - h[i], trust)
        bounds_list.append((lo_b, hi_b))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

    h_int = np.interp(tgt_c, src_c, h_src)
