import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -2.0, 2.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_fvals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient(h, p):
    """Gradient of F_p with respect to h, where p is the correlate index."""
    n = len(h)
    dx = 2.0 / n
    off = p - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        for i in range(n - off):
            g[i] += 1.0 - h[i + off]
            g[i + off] -= h[i]
    else:
        t = -off
        for i in range(t, n):
            g[i] += 1.0 - h[i - t]
            g[i - t] -= h[i]
    return g * dx

def compute_gradients_batch(h, p_indices):
    """Compute gradients for multiple lags."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(p_indices), n))
    for idx, p in enumerate(p_indices):
        off = p - (n - 1)
        g = np.zeros(n)
        if off >= 0:
            for i in range(n - off):
                g[i] += 1.0 - h[i + off]
                g[i + off] -= h[i]
        else:
            t = -off
            for i in range(t, n):
                g[i] += 1.0 - h[i - t]
                g[i - t] -= h[i]
        G[idx] = g * dx
    return G

def lp_direction(h, active_idx, max_F):
    """Solve LP to find direction minimizing max directional derivative on active set."""
    n = len(h)
    G = compute_gradients_batch(h, active_idx)
    # g0_p = G_p · h won't be needed; we just need G_p · d ≤ s
    
    d_var = cp.Variable(n)
    s_var = cp.Variable()
    constraints = []
    
    # Directional derivative constraints
    for i in range(len(active_idx)):
        constraints.append(G[i] @ d_var <= s_var)
    
    # Sum constraint
    constraints.append(cp.sum(d_var) == 0)
    
    # Box sign constraints
    eps = 1e-9
    for k in range(n):
        if h[k] <= eps:
            constraints.append(d_var[k] >= 0)
        elif h[k] >= 1.0 - eps:
            constraints.append(d_var[k] <= 0)
        else:
            constraints.append(d_var[k] >= -1.0)
            constraints.append(d_var[k] <= 1.0)
    
    # Also bound s
    constraints.append(s_var >= -10.0)
    
    obj = cp.Minimize(s_var)
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None
    return d_var.value, prob.value

def line_search(h, d, max_alpha=1.0, n_trials=8):
    """Find best step size along direction d."""
    n = len(h)
    best_alpha = 0.0
    best_score = compute_score(h)
    
    # Try geometric sequence of alpha values
    alphas = max_alpha * (0.5 ** np.arange(n_trials))
    # Also try very small steps
    alphas = np.concatenate([alphas, [0.01, 0.001]])
    
    for alpha in sorted(set(alphas), reverse=True):
        h_try = make_feasible(h + alpha * d)
        sc = compute_score(h_try)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
    
    return best_alpha, best_score, make_feasible(h + best_alpha * d) if best_alpha > 0 else h

def lp_optimize(h0, max_iter=50, tol_active=1e-7, tol_min=1e-10):
    """Main LP direction optimization loop."""
    n = len(h0)
    h = h0.copy()
    best = compute_score(h)
    stall = 0
    
    for it in range(max_iter):
        fvals = compute_fvals(h)
        max_F = fvals.max()
        
        # Active set: all within tol_active of max
        active_mask = fvals >= max_F - tol_active
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = [np.argmax(fvals)]
        
        # Solve LP
        d, s_val = lp_direction(h, active_idx, max_F)
        if d is None or s_val is None:
            break
        if not np.all(np.isfinite(d)):
            break
        if s_val >= -tol_min:
            # No descent direction found; try with larger active set
            tol_active *= 10
            if tol_active > 0.5:
                break
            continue
        
        # Line search
        alpha, sc_new, h_new = line_search(h, d, max_alpha=0.5)
        
        if alpha > 0 and sc_new < best - 1e-14:
            h = h_new
            best = sc_new
            stall = 0
            # Reset tolerance when we make progress
            tol_active = max(1e-7, tol_active * 0.5)
        else:
            stall += 1
            if stall >= 5:
                break
    
    return h, best

def pair_swap_descent(h0, n_rounds=5, tol=1e-12):
    """Coordinate descent via pair swaps."""
    n = len(h0)
    h = h0.copy()
    best = compute_score(h)
    
    for rnd in range(n_rounds):
        improved = False
        fvals = compute_fvals(h)
        max_F = fvals.max()
        active_mask = fvals >= max_F - 1e-6
        active_idx = np.where(active_mask)[0]
        
        for i in range(n):
            for j in range(i+1, n):
                # Try moving delta from i to j
                # Best delta: minimize max over active p of F_p(h + delta*(e_j - e_i))
                # Directional derivative for p: G_p[j] - G_p[i]
                # We want this negative for all active p
                # So we need G_p[j] < G_p[i] for all active p
                # Find the most constraining: max_p (G_p[j] - G_p[i])
                
                # Compute gradients at i and j for active lags
                grad_i = np.array([compute_gradient(h, p)[i] for p in active_idx])
                grad_j = np.array([compute_gradient(h, p)[j] for p in active_idx])
                # If we move delta from i to j: change in F_p = delta * (grad_p[j] - grad_p[i])
                # We want delta * (grad_p[j] - grad_p[i]) < 0 for all p in active
                # If grad_p[j] < grad_p[i] for all p, then delta > 0 works
                # If grad_p[j] > grad_p[i] for all p, then delta < 0 works
                diffs = grad_j - grad_i
                if all(d < 0 for d in diffs):
                    # Move positive delta from i to j
                    # Max delta limited by h[i] (can't go below 0) and h[j] (can't go above 1)
                    max_d = min(h[i], 1 - h[j])
                    for delta in [max_d * 0.5, max_d * 0.25, max_d * 0.1, max_d]:
                        if delta <= 0: continue
                        h_try = h.copy()
                        h_try[i] -= delta
                        h_try[j] += delta
                        h_try = make_feasible(h_try)
                        sc = compute_score(h_try)
                        if sc < best - 1e-14:
                            h = h_try
                            best = sc
                            improved = True
                            break
                elif all(d > 0 for d in diffs):
                    # Move positive delta from j to i
                    max_d = min(h[j], 1 - h[i])
                    for delta in [max_d * 0.5, max_d * 0.25, max_d * 0.1, max_d]:
                        if delta <= 0: continue
                        h_try = h.copy()
                        h_try[j] -= delta
                        h_try[i] += delta
                        h_try = make_feasible(h_try)
                        sc = compute_score(h_try)
                        if sc < best - 1e-14:
                            h = h_try
                            best = sc
                            improved = True
                            break
        
        if not improved:
            break
