def solve_lp_minimax(h, active_indices, delta):
    n = len(h)
    dx = 2.0 / n
    lags = compute_lags(h)
    
    # Get gradients for active lags
    grads = np.array([compute_grad_for_lag(h, p) for p in active_indices])
    active_vals = lags[active_indices]
    
    # LP: minimize t
    # subject to: grads @ d <= t - active_vals (for each active lag)
    #             -delta <= d <= delta (trust region)
    #             h - delta <= h + d <= h + delta (box constraints)
    #             sum(d) = 0
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [cp.sum(d) == 0]
    for i in range(len(active_indices)):
        constraints.append(grads[i] @ d <= t - active_vals[i])
    constraints.append(d >= np.maximum(0, h - delta) - h)  # h + d >= 0 => d >= -h
    constraints.append(d <= np.minimum(1, h + delta) - h)  # h + d <= 1 => d <= 1-h
    constraints.append(d >= -delta)
    constraints.append(d <= delta)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            try:
                prob.solve(solver=cp.CLARABEL)
            except:
                return None, None
    
    if d.value is None or t.value is None:
        return None, None
    
    return np.asarray(d.value).ravel(), float(t.value)

    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_minimax_step(h, tol, delta, max_active=30):
    """Solve LP to find best minimax direction."""
    n = len(h)
    lags = compute_lags(h)
    cmax = lags.max()
    
    # Active set
    active = np.where(lags >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(lags)])
    if len(active) > max_active:
        # Keep only the most critical ones
        active_vals = lags[active]
        order = np.argsort(-active_vals)
        active = active[order[:max_active]]
    
    # Compute gradients for active lags
    grads = np.array([compute_grad_for_lag(h, p) for p in active])
    active_vals = lags[active]
    
    # LP: minimize t
    # s.t. grads[i] @ d <= t - active_vals[i] for each active i
    #      -h <= d <= 1-h (box)
    #      -delta <= d <= delta (trust)
    #      sum(d) = 0
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        d >= np.maximum(-h, -delta),
        d <= np.minimum(1.0 - h, delta),
    ]
    for i in range(len(active)):
        constraints.append(grads[i] @ d <= t - active_vals[i])
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    t_pred = None
    d_val = None
    for solver in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status == 'optimal' and d.value is not None:
                d_val = np.asarray(d.value, dtype=float).ravel()
                t_pred = float(t.value)
                break
        except Exception:
            continue
    
    if d_val is None:
        return None, None
    
    return d_val, t_pred

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [0.0, t_lo, t_hi, (t_lo + t_hi) / 2.0,
                        t_lo + 0.25 * span, t_lo + 0.75 * span]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13 or val_b < -1e-13 or val_b > 1.0 + 1e-13:

            tmp[b] = val_b

                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = official_score(best)
    return best, best_sc

def gradient_step(h, is_dead, n_iter=100):
    """Subgradient method with step size adaptation."""
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    alpha = 0.02
    for it in range(n_iter):
        if is_dead():
            break
        lags = compute_lags(best)
        cmax = lags.max()
        # Weighted gradient over top lags
        tol = 0.03
        active = np.where(lags >= cmax - tol)[0]
        w = np.exp(50.0 * (lags[active] - cmax))
        w /= w.sum()
        grad = np.zeros(n)
        for i, p in enumerate(active):
            grad += w[i] * compute_grad_for_lag(best, p)
        # Project to zero-sum
        d = -grad
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d /= d_norm
        improved = False
        for mult in [alpha, alpha*0.5, alpha*2, alpha*0.25]:
            h_trial = make_feasible(best + mult * d)
            sc = official_score(h_trial)

                best = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-6:
                break
        else:
            alpha = min(alpha * 1.2, 0.1)
    return best, best_sc

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

                best_sc = sc
                best_alpha = a
    return best_alpha, best_sc

def gradient_polish(h, n_iter, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    alpha = 0.02
    for it in range(n_iter):
        if is_dead():
            break
        lags = compute_lags(best)
        cmax = lags.max()
        tol = 0.03
        active = np.where(lags >= cmax - tol)[0]
        w = np.exp(40.0 * (lags[active] - cmax))
        w /= w.sum()
        all_grads = compute_all_grads(best)
        grad = sum(w[i] * all_grads[active[i]] for i in range(len(active)))
        d = -grad
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d /= d_norm
        improved = False
        for mult in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25]:
            h_trial = make_feasible(best + mult * d)
            sc = official_score(h_trial)
            if sc < best_sc - 1e-14:
                best = h_trial
                best_sc = sc
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:
            alpha = min(alpha * 1.15, 0.08)
    return best, best_sc

def two_point_search(h, n_tries, rng, is_dead):
    n = len(h)
    best = h.copy()
    best_sc = official_score(best)
    for it in range(n_tries):
        if is_dead():
            break
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        candidates_t = [t_lo + (i + 1) * span / 8.0 for i in range(7)]
        best_t = None
        best_local = best_sc
        tmp = best.copy()
        for t in candidates_t:
            val_a = best[a] + t
            val_b = best[b] - t
            if val_a < -1e-13 or val_a > 1.0 + 1e-13:
                continue
            if val_b < -1e-13 or val_b > 1.0 + 1e-13:
                continue
            tmp[a] = val_a
            tmp[b] = val_b
            sc = official_score(tmp)
            if sc < best_local:
                best_local = sc
                best_t = t
        if best_t is not None and abs(best_t) > 1e-12:
            best[a] = np.clip(best[a] + best_t, 0, 1)
            best[b] = np.clip(best[b] - best_t, 0, 1)
            best_sc = official_score(best)
    return best, best_sc

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0

    def is_dead():
        return time.time() >= deadline

    rng = np.random.default_rng(seed)

    # Load parent
    best_h = None
    best_score = float('inf')
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()

            best_h = make_feasible(parent)

        pass

    if best_h is None:
        n = 60

        best_score = official_score(best_h)

    # Phase 1: LP-based minimax optimization (main)
    delta = 0.06

    for iteration in range(800):
        if is_dead():
            break
        d_dir = lp_minimax_direction(best_h, delta, max_cp_iter=12)
        if d_dir is None:

                break
            continue
        step, new_sc = dense_line_search(best_h, d_dir, max_alpha=1.0, n_eval=30)
        if step > 0 and new_sc < best_score - 1e-13:
            best_h = make_feasible(best_h + step * d_dir)
            best_score = new_sc
            stall = 0
            if step >= 0.8:
                delta = min(delta * 1.25, 0.12)
            elif step < 0.3:
                delta = max(delta * 0.9, 0.015)
        else:
            stall += 1
            delta = max(delta * 0.8, 0.008)
            if stall > 40:
                break

    # Phase 2: Gradient polishing
    if not is_dead():
        best_h, best_score = gradient_polish(best_h, 300, is_dead)

    # Phase 3: 2-point search
    if not is_dead():
        remaining = deadline - time.time()
        n_tries = max(3000, int(remaining * 2.5))
        best_h, best_score = two_point_search(best_h, n_tries, rng, is_dead)

    # Phase 4: Second LP round with tighter trust region
    if not is_dead():
        delta = 0.03
        stall = 0
        for iteration in range(400):
            if is_dead():
                break
            d_dir = lp_minimax_direction(best_h, delta, max_cp_iter=10)
            if d_dir is None:
                stall += 1
                delta = max(delta * 0.8, 0.005)
                if stall > 25:

                continue
            step, new_sc = dense_line_search(best_h, d_dir, max_alpha=1.0, n_eval=25)
            if step > 0 and new_sc < best_score - 1e-13:
                best_h = make_feasible(best_h + step * d_dir)
                best_score = new_sc
                stall = 0
                delta = min(delta * 1.2, 0.08)
            else:
                stall += 1
                delta = max(delta * 0.85, 0.004)
                if stall > 30:
                    break

    # Phase 5: Multi-resolution exploration
    if not is_dead():
        n_base = len(best_h)
        tried_ns = set([n_base])
        candidate_ns = sorted(set(
            [n_base + dn for dn in [-12, -8, -4, 4, 8, 12, 16] if 24 <= n_base + dn <= 100] +
            [36, 44, 52, 60, 68, 76, 84]
        ))
        for nt in candidate_ns:
            if is_dead():
                break
            if nt in tried_ns:
                continue
            tried_ns.add(nt)
            h_int = interp_to(best_h, nt)
            sc_int = official_score(h_int)
            h_cur = h_int
            sc_cur = sc_int
            # Quick LP on this resolution

                if is_dead():

                d_dir = lp_minimax_direction(h_cur, delta_nr, max_cp_iter=8)

            if sc_cur < best_score - 1e-13:

    if not is_dead():

    final_n = len(final_h)
    final_h = np.clip(final_h, 0.0, 1.0)
