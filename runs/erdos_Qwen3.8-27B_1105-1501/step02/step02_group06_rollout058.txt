```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_all_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_f_p(h, p):
    n = len(h)
    s = p - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] += (-h[i])
    else:
        t = -s
        for i in range(t, n):
            g[i] += (1.0 - h[i - t])
            g[i - t] += (-h[i])
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_direction(h, active_lags, eps_box=1e-8):
    """Find a descent direction using LP.
    active_lags: list of lag indices p (0 to 2n-2)
    Returns (d, t_opt) or (None, None) if infeasible.
    """
    n = len(h)
    dx = 2.0 / n
    
    # Variables: d[0],...,d[n-1], t  (total n+1)
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag p: (g_p * dx) @ d <= t
    # i.e., (g_p * dx) @ d - t <= 0
    m = len(active_lags)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for idx, p in enumerate(active_lags):
        g = grad_f_p(h, p) * dx
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        # g @ d - t <= 0
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds:
    # -1 <= d[k] <= 1,  for all k
    # Additionally: if h[k] <= eps_box, then d[k] >= 0
    #               if h[k] >= 1-eps_box, then d[k] <= 0
    # t is unbounded below, bounded above by 1 (for normalization)
    bounds = []
    for k in range(n):
        lo_k = -1.0
        hi_k = 1.0
        if h[k] <= eps_box:
            lo_k = 0.0
        if h[k] >= 1.0 - eps_box:
            hi_k = 0.0
        bounds.append((lo_k, hi_k))
    bounds.append((-np.inf, 1.0))  # t unbounded below
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    d = result.x[:n]
    t_opt = result.x[n]
    return d, t_opt

def line_search(h, d, max_iter=30):
    """Find best alpha in (0, 1] such that h + alpha*d is feasible and score improves."""
    score0 = official_score(h)
    n = len(h)
    
    # Find max feasible alpha
    alpha_max = 1.0

def coordinate_descent(h, max_iter=50, tol=1e-10):

    best_score = official_score(h)

        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                # Optimize h[i] and h[j] jointly
                # Keep sum fixed: h[i] + h[j] = const
                s = h[i] + h[j]
                # Try values
                for h_i_try in [0.0, 0.25, 0.5, 0.75, 1.0]:
                    h_j_try = s - h_i_try
                    if h_j_try < -1e-9 or h_j_try > 1.0 + 1e-9:
                        continue
                    h_j_try = np.clip(h_j_try, 0, 1)
                    old_i, old_j = h[i], h[j]
                    h[i], h[j] = h_i_try, h_j_try
                    sc = official_score(h)
                    if sc < best_score - tol:
                        best_score = sc
                        improved = True
                    else:
                        h[i], h[j] = old_i, old_j
        if not improved:
            break
    
    return h, best_score

        sc = np.max(vals + t * slopes)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    h[i], h[j] = a + best_t, b - best_t
    return h
```

Wait, this is actually exact! Since F_p is bilinear in h (it's a sum of products h[i]*h[j]), when we change only h[i] and h[j], each F_p becomes linear in t. So the formula F_p(t) = F_p(0) + t*slope_p is exact, and the max of these linear functions is exactly the new score. No need to recompute!

This makes the pair optimization extremely fast: O(n) per pair (to compute slopes), and we just need to find the optimal t among O(n) breakpoints.

For a full pass over all pairs, it's O(n^3) which for n=60 is about 216,000 operations. Very fast!

Let me include this as a key component.

Actually wait, I need to be more careful. The score is max_p F_p * dx. Since F_p is linear in t, the max is also a function that's the max of linear functions (convex piecewise linear). The minimum of a convex piecewise linear function occurs where the left derivative is <= 0 and the right derivative is >= 0, or at a boundary.

The left derivative at t is max over p with F_p(t) = F_max(t) of slope_p... no wait, the derivative of max_p f_p(t) at a point where f_p and f_q are both active is the max of their slopes (for the right derivative) or min (for the left derivative). Actually no, the subgradient of max is the convex hull of the gradients of active functions.

For a 1D problem: the derivative of g(t) = max_p (a_p + b_p * t) at a point where the max is achieved by p and q is:
- If b_p > b_q: the max switches from q to p as t increases past the crossing point
- The subdifferential is [b_q, b_p] (all slopes of active functions)

So g(t) is non-decreasing only when all active slopes are positive, and non-increasing only when all active slopes are negative. The minimum is where the minimum active slope is <= 0 and the maximum active slope is >= 0.

In practice, I can just find all breakpoints and evaluate at each. The number of breakpoints is at most (2n-1) choose 2, which for n=60 is about 7000. That's fine.

But actually, I don't need ALL breakpoints. I only need the breakpoints that are relevant, i.e., where the maximum function changes. In practice, checking the top 10-20 active lags and their pairwise breakpoints is sufficient.

Let me simplify: I'll just do a ternary search on the 1D convex function. Since it's convex, ternary search will find the minimum in O(log(range/eps)) evaluations. Each evaluation is O(n) (compute max of vals + t*slopes). So total is O(n * log(n)) per pair.

Actually, for a piecewise linear convex function, I can do even better: binary search for the point where the derivative changes sign. The derivative at t is the max slope among active functions (right derivative) or min slope (left derivative). The minimum is where min active slope <= 0 <= max active slope.

Let me just use a simple approach: evaluate at the boundaries and at a few key points, pick the best.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_all_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_f_p(h, p):
    n = len(h)
    s = p - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] += (-h[i])
    else:
        t = -s
        for i in range(t, n):
            g[i] += (1.0 - h[i - t])
            g[i - t] += (-h[i])
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_direction(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(n + 1)
    c[n] = 1.0
    m = len(active_lags)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for idx, p in enumerate(active_lags):
        g = grad_f_p(h, p) * dx
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-1.0, 1.0)] * n + [(-5.0, 1.0)]
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.status == 0:
            return result.x[:n], result.x[n]
    except Exception:
        pass
    return None, None

def line_search(h, d, n_eval=20):
    n = len(h)
    score0 = official_score(h)
    alpha_max = 1.0
    for k in range(n):
        if d[k] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[k]) / d[k])
        elif d[k] < -1e-12:
            alpha_max = min(alpha_max, h[k] / (-d[k]))
    alpha_max = min(alpha_max, 1.0)
    if alpha_max <= 1e-12:
        return h, score0
    best_h = h
    best_score = score0
    alphas = np.linspace(0.05, alpha_max, n_eval)
    for a in alphas:
        h_try = np.clip(h + a * d, 0.0, 1.0)
        h_try = make_feasible(h_try)
        sc = official_score(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try
    return best_h, best_score

def lp_subgradient(h, max_iter=150, threshold=0.005):
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    stall = 0
    for it in range(max_iter):
        vals = compute_all_f(h)
        max_val = vals.max()
        active = np.where(vals >= max_val - threshold)[0]
        if len(active) < 3:
            top = np.argsort(vals)[-5:]
            active = np.array(sorted(set(np.concatenate([active, top]))))
        d, t_opt = lp_direction(h, active)
        if d is None or t_opt >= -1e-10:
            if len(active) > 3:
                active2 = np.where(vals >= max_val - threshold * 0.2)[0]
                d, t_opt = lp_direction(h, active2)
                if d is None or t_opt >= -1e-10:
                    stall += 1
                    if stall >= 15:
                        break
                    continue
            else:
                stall += 1
                if stall >= 15:
                    break
                continue
        h_new, sc_new = line_search(h, d)
        if sc_new < best_score - 1e-13:
            h = h_new
            best_score = sc_new
            stall = 0
        else:
            stall += 1
            if stall >= 15:
                break
    return h, best_score

def pair_descent(h, max_passes=20):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_score = official_score(h)
    for _pass in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                a, b = h[i], h[j]
                t_lo = -a
                t_hi = 1.0 - b
                if t_lo >= t_hi - 1e-12:
                    continue
                vals = compute_all_f(h)
                slopes = np.zeros(2 * n - 1)
                for p in range(2 * n - 1):
                    g = grad_f_p(h, p)
                    slopes[p] = dx * (g[i] - g[j])
                candidates = [t_lo, t_hi, 0.0]
                top_idx = np.argsort(vals)[-min(8, 2 * n - 1):]
                for ip in range(len(top_idx)):
                    for iq in range(ip + 1, len(top_idx)):
                        p, q = top_idx[ip], top_idx[iq]
                        ds = slopes[p] - slopes[q]
                        if abs(ds) > 1e-12:
                            t_bp = (vals[q] - vals[p]) / ds
                            if t_lo < t_bp < t_hi:
                                candidates.append(t_bp)
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    candidates.append(t_lo + frac * (t_hi - t_lo))
                best_t = 0.0
                best_sc = vals.max()
                for t in candidates:
                    t = max(t_lo, min(t_hi, t))
                    sc = np.max(vals + t * slopes)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_t = t
                if abs(best_t) > 1e-10:
                    h[i] = a + best_t
                    h[j] = b - best_t
                    actual_sc = official_score(h)
                    if actual_sc < best_score - 1e-12:
                        best_score = actual_sc
                        improved = True
                    else:
                        h[i], h[j] = a, b
        if not improved:
            break
    return h, best_score

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_jac(h):
    return np.ones(len(h))

def try_slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def projected_gradient_smooth(h, beta, n_steps=50):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_score = official_score(h)
    for step in range(n_steps):
        vals = compute_all_f(h)
        v_shifted = beta * (vals - vals.max())
        v_shifted = np.clip(v_shifted, -50, 50)
        weights = np.exp(v_shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for p in range(len(vals)):
            if weights[p] > 1e-12:
                grad += weights[p] * grad_f_p(h, p) * dx
        alpha = 0.1 / (1.0 + step * 0.05)
        h_try = h - alpha * grad
        h_try = make_feasible(h_try)
        sc = official_score(h_try)
        if sc < best_score:
            h = h_try
            best_score = sc
    return h, best_score
