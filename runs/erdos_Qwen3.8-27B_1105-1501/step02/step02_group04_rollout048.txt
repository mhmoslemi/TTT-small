def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} using bisection on offset."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Find coordinates with room
        if residual > 0:
            # Need to increase some coordinates
            free = np.where(result < 1.0 - 1e-15)[0]

            free = np.where(result > 1e-15)[0]

            result = np.clip(result, 0.0, 1.0)

def smooth_value_and_grad(h, n, T):
    """Compute smooth objective value and gradient.
    Returns (value, gradient) where value = F_T(h) and gradient = dF_T/dh.
    """
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    rmax = np.max(corr)
    # Stable computation
    shifted = (corr - rmax) / T
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    value = dx * (rmax + T * np.log(sum_exp))
    
    # Gradient
    grad = np.zeros(n)
    offset = n - 1
    for s in range(-(n-1), n):
        w = weights[s + offset]
        if w < 1e-18:
            continue
        # Term 1: w * (1-h[i-s]) for valid i-s
        i_lo = max(0, s)
        i_hi = min(n-1, s + n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w * (1.0 - h[idx - s])
        # Term 2: -w * h[i+s] for valid i+s
        i_lo2 = max(0, -s)
        i_hi2 = min(n-1, n-1-s)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            grad[idx] -= w * h[idx + s]
    grad *= dx
    return value, grad

        return v

        _, g = smooth_value_and_grad(x, n, T)
        return g
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=constraints,
                      maxiter=maxiter, options={'ftol': ftol})
        return res.x

        return h0.copy()

    return h
