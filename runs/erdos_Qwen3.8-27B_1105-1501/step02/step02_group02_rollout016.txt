        h[positions] = 1

    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    M = len(lag_indices)

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def _ineq(z):
        h = z[:-1]
        t = z[-1]
        cv = _corr(h)
        return t - cv[lag_indices]

    def _ineq_jac(z):
        h = z[:-1]
        J = np.zeros((M, n + 1))
        for ii, k in enumerate(lag_indices):
            s = k - (n - 1)
            if s >= 0:
                J[ii, :n - s] += dx * (1.0 - h[s:])
                J[ii, s:] -= dx * h[:n - s]
            else:
                t2 = -s
                J[ii, t2:] += dx * (1.0 - h[:n - t2])
                J[ii, :n - t2] -= dx * h[t2:]
            J[ii, -1] = 1.0
        return J

    def _eq(z):
        return z[:-1].sum() - n / 2.0

    def _eq_jac(z):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return _project(res.x[:-1])
    except Exception:
        pass
    return h0


def _constraint_gen(h0, max_rounds=15, M_start=5, M_max=25, maxiter=150):
    """Constraint generation: iteratively add active constraints."""
    n = len(h0)
    h = h0.copy()
    best_s = _eval(h)
    for rnd in range(max_rounds):
        M = min(M_start + rnd * 2, M_max, 2 * n - 1)
        cv = _corr(h)
        active = np.argsort(cv)[::-1][:M]
        h_new = _active_solve(h, active, maxiter=maxiter)
        s_new = _eval(h_new)
        if s_new < best_s - 1e-14:
            h = h_new
            best_s = s_new
        else:
            break
    return h, best_s


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _make_binary(n, rng):
    if n % 2 != 0:
        n -= 1
    target = n // 2
    h = np.zeros(n)
    # Golden ratio construction
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    positions = np.sort(np.floor(np.arange(target) * (1.0 / phi)).astype(int) % n)
    h[positions] = 1.0
    return h


def _smooth_binary(h_bin, alpha=0.3):
    """Smooth binary to continuous."""
    n = len(h_bin)
    kernel_size = 5
    kernel = np.ones(kernel_size) / kernel_size
    padded = np.pad(h_bin, kernel_size // 2, mode='wrap')
    smoothed = np.convolve(padded, kernel, mode='valid')[:n]
    h = alpha * smoothed + (1 - alpha) * h_bin
    return _project(h)


def _nelder_mead(h0, maxiter=300):
    n = len(h0)

    def _fun(x):
        return _eval(_project(x))

    try:
        res = minimize(_fun, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9,
                                'fatol': 1e-12, 'adaptive': True})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            h_p = _project(res.x)
            if _eval(h_p) <= _eval(h0) + 1e-12:
                return h_p
    except Exception:
        pass
    return h0


def _fista_subgrad(h0, beta, n_iter, alpha_init=0.005, mu=0.9):
    """FISTA with LSE subgradient and averaging."""
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _eval(h0)
    t = 1
    for it in range(n_iter):
        grad = _lse_grad(y, beta)
        t_prev = t
        t = 1.5 * t
        h_new = y - alpha_init * grad
        h_new = _project(h_new)
        y = h_new + ((t_prev - 1) / t) * (h_new - h_prev)
        y = _project(y)
        h_avg += (h_new - h_avg) / (t + 1)
        h_prev = h_cur
        h_cur = h_new
        if it % 500 == 0:
            s = _eval(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
            ha = _project(h_avg)
            sa = _eval(ha)
            if sa < best_s:
                best_s = sa
                best_h = ha.copy()
    s = _eval(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _eval(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _make_binary(58, rng)
        parent = _smooth_binary(parent)

    best_h = parent.copy()
    best_score = _eval(best_h)
    parent_n = len(parent)

    # Phase 1: Generate and screen candidates
    candidates = []
    # Parent at same and nearby dimensions
    for dn in range(-4, 5):
        n = parent_n + dn
        if 20 <= n <= 80:
            h_i = _interp(parent, n)
            candidates.append((n, h_i, _eval(h_i)))
    # Binary patterns
    for n in range(50, 68, 2):
        h_bin = _make_binary(n, rng)
        candidates.append((n, h_bin.copy(), _eval(h_bin)))
        h_sm = _smooth_binary(h_bin, alpha=0.2)
        candidates.append((n, h_sm, _eval(h_sm)))
        h_sm2 = _smooth_binary(h_bin, alpha=0.5)
        candidates.append((n, h_sm2, _eval(h_sm2)))

    candidates.sort(key=lambda x: x[2])

    # Quick screening: SLSQP with moderate beta
    screened = []
    for n, h0, s0 in candidates:
        if time.time() > deadline - 180:
            break
        if len(screened) >= 15:
            break
        h_opt = _slsqp_lse(h0, beta=100, maxiter=80)
        s_opt = _eval(h_opt)
        screened.append((s_opt, n, h_opt))
        if s_opt < best_score - 1e-14:
            best_h = h_opt.copy()
            best_score = s_opt

    screened.sort(key=lambda x: x[0])

    # Phase 2: Deep refinement of top candidates
    top_k = min(6, len(screened))
    alloc_each = max(20.0, (deadline - t_start - 120) / max(1, top_k + 3))

    for i in range(top_k):
        if time.time() > deadline - 100:
            break
        s0, n, h0 = screened[i]
        alloc = min(alloc_each, deadline - time.time() - 60)
        if alloc < 15:
            break

        # Stage A: SLSQP with increasing beta
        cur = h0.copy()
        for beta in [200, 500, 1000, 2000]:
            if time.time() > deadline - 60:
                break
            new = _slsqp_lse(cur, beta=beta, maxiter=200)
            ns = _eval(new)
            if ns < _eval(cur) - 1e-14:
                cur = new

        # Stage B: Constraint generation
        if time.time() < deadline - 50:
            cur, cs = _constraint_gen(cur, max_rounds=10, M_start=5, M_max=20)

        # Stage C: FISTA
        if time.time() < deadline - 40:
            cur, cs = _fista_subgrad(cur, beta=500, n_iter=5000,
                                     alpha_init=0.004, mu=0.88)

        # Stage D: Final SLSQP
        if time.time() < deadline - 30:
            new = _slsqp_lse(cur, beta=3000, maxiter=200)
            ns = _eval(new)
            if ns < _eval(cur) - 1e-14:
                cur = new

        if _eval(cur) < best_score - 1e-14:
            best_h = cur.copy()
            best_score = _eval(cur)

    # Phase 3: Intensive refinement on best
    remaining = deadline - time.time()
    if remaining > 30:
        # High-beta SLSQP rounds
        for beta in [500, 1500, 4000, 8000]:
            if time.time() > deadline - 20:
                break
            new = _slsqp_lse(best_h, beta=beta, maxiter=250)
            ns = _eval(new)
            if ns < best_score - 1e-14:
                best_h = new
                best_score = ns

    # Constraint generation on best
    if time.time() < deadline - 30:
        best_h, best_score = _constraint_gen(best_h, max_rounds=12,
                                             M_start=8, M_max=30, maxiter=120)

    # FISTA
    if time.time() < deadline - 25:
        n_iter = min(15000, int((deadline - time.time() - 15) * 2000))
        best_h, best_score = _fista_subgrad(best_h, beta=1000, n_iter=n_iter,
                                            alpha_init=0.003, mu=0.9)

    # Nelder-Mead
    if time.time() < deadline - 15:
        nm_h = _nelder_mead(best_h, maxiter=200)
        nm_s = _eval(nm_h)
        if nm_s < best_score - 1e-14:
            best_h = nm_h
            best_score = nm_s

    # Phase 4: Random restarts
    remaining = deadline - time.time()
    n_restarts = min(8, max(1, int(remaining / 25)))
    for _ in range(n_restarts):
        if time.time() > deadline - 12:
            break
        sigma = rng.choice([0.02, 0.04, 0.06, 0.08])
        h_pert = _project(best_h + sigma * rng.standard_normal(len(best_h)))
        # Quick refine
        cur = _slsqp_lse(h_pert, beta=500, maxiter=100)
        cur, cs = _constraint_gen(cur, max_rounds=6, M_start=6, M_max=14, maxiter=80)
        if cs < best_score - 1e-14:
            best_h = cur
            best_score = cs

    # Final polish
    if time.time() < deadline - 5:
        new = _slsqp_lse(best_h, beta=5000, maxiter=200)
        ns = _eval(new)
        if ns < best_score - 1e-14:
            best_h = new
            best_score = ns

    if time.time() < deadline - 3:
        nm_h = _nelder_mead(best_h, maxiter=100)
        nm_s = _eval(nm_h)
        if nm_s < best_score - 1e-14:
            best_h = nm_h
            best_score = nm_s

    # Final
    best_h = _project(best_h)
    final_score = _eval(best_h)
    return (best_h, final_score, len(best_h))
