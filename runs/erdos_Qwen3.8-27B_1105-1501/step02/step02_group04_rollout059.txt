def smooth_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    cmax = np.max(corr)
    w = np.exp((corr - cmax) / T)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(2*n-1); w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        jlo = max(0, -d)

def project_feasible(h, target=None):
    n = len(h)

        target = n / 2.0
    # Water-filling projection
    lo, hi = -1.0, 1.0

        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)

            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to increase: add to entries that are < 1

            # Actually simpler: distribute among non-saturated

def smooth_value_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    cmax = np.max(corr)
    z = np.exp((corr - cmax) / T)
    S = z.sum()
    val = dx * (cmax + T * np.log(S))
    w = z / S  # softmax weights
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        jlo = max(0, -d)
        jhi = min(n - 1, n - 1 - d)
        # Part A: grad[jlo+d : jhi+d+1] += wk * v[jlo:jhi+1]
        grad[jlo + d: jhi + d + 1] += wk * v[jlo: jhi + 1]
        # Part B: grad[jlo:jhi+1] -= wk * h[jlo+d:jhi+d+1]
        grad[jlo: jhi + 1] -= wk * h[jlo + d: jhi + d + 1]
    grad *= dx
    return val, grad

    v, _ = smooth_value_and_grad(x, _gs['T'])
    return v

def slsqp_optimize(h, T, maxiter=80, ftol=1e-10):
    n = len(h)
    _gs['n'] = n
    _gs['T'] = T
    _gs['target'] = n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_constraint_jac}]
    
    result = minimize(_slsqp_obj, h, jac=_slsqp_jac, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})

def pair_line_search(h, i, j, iters=30):
    """Find optimal delta to transfer from i to j (or -delta from j to i)."""
    hi_val = h[i]
    hj_val = h[j]
    # delta: h[i] -= delta, h[j] += delta
    # constraints: 0 <= hi_val - delta <= 1, 0 <= hj_val + delta <= 1
    lo = max(-hj_val, hi_val - 1.0)
    hi = min(hi_val, 1.0 - hj_val)
    if hi - lo < 1e-13:
        return 0.0
    a, b = lo, hi
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i] = hi_val - m1
        h[j] = hj_val + m1
        f1 = compute_c5(h)
        h[i] = hi_val - m2
        h[j] = hj_val + m2
        f2 = compute_c5(h)
        h[i] = hi_val
        h[j] = hj_val
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def pair_sweep(h, rng, n_pairs=None):
    n = len(h)
    if n_pairs is None:
        n_pairs = n * (n - 1) // 2
    else:
        n_pairs = min(n_pairs, n * (n - 1) // 2)
    perm = rng.permutation(n)
    count = 0
    for a_idx in range(n):
        for b_idx in range(a_idx + 1, n):
            if count >= n_pairs:
                return
            i, j = perm[a_idx], perm[b_idx]
            if i > j:
                i, j = j, i
            delta = pair_line_search(h, i, j)
            if abs(delta) > 1e-12:
                h[i] -= delta
                h[j] += delta
            count += 1

    temps = [6.0 * dx, 3.0 * dx, 1.5 * dx, 0.8 * dx, 0.4 * dx,
             0.2 * dx, 0.1 * dx, 0.05 * dx, 0.025 * dx, 0.01 * dx]
    if n_temps < len(temps):
        temps = temps[:n_temps]

    best_c5_local = compute_c5(h)
