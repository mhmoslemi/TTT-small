def lp_cut_step(h, raw, radius, k_active):
    n = len(h)
    n_raw = 2 * n - 1
    dx = 2.0 / n
    vals = raw * dx
    
    top_idx = np.argsort(vals)[-k_active:]
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize tau
    
    A_ub = []
    b_ub = []
    
    for p in top_idx:
        s = p - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            ip = i + s
            if 0 <= ip < n:
                grad[i] += (1.0 - h[ip])
            im = i - s
            if 0 <= im < n:
                grad[i] -= h[im]
        
        row = np.zeros(n + 1)
        row[:n] = dx * grad
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(raw[p] * dx - dx * grad @ h)
    
    if not A_ub:
        return None
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = res.x[:n]
            if np.all(np.isfinite(h_new)):
                return make_feasible(h_new)
    except Exception:
        pass
    return None

def interpolate_to(h_src, n_tgt):

    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def gradient_descent_subgrad(h, max_iter=200, step_start=0.01, time_limit=100.0, t_start=0.0):
    n = len(h)
    h = h.copy()
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    step = step_start
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        raw = compute_raw(h)
        max_val = raw.max()
        active_thresh = max_val - 0.02 * max_val
        active = np.where(raw >= active_thresh)[0]
        
        grad = np.zeros(n)
        for p in active:
            s = p - (n - 1)
            for i in range(n):
                ip = i + s
                if 0 <= ip < n:
                    grad[i] += (1.0 - h[ip])
                im = i - s
                if 0 <= im < n:
                    grad[i] -= h[im]
        
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-10:
            break
        
        h_try = make_feasible(h - step * grad / norm)
        sc_try = official_score(h_try)
        if sc_try < best_score - 1e-12:
            h = h_try
            best_h = h.copy()
            best_score = sc_try
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.7
            if step < 1e-5:
                break
    
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    def remaining():
        return deadline - time.time()
    
    best_h = None
    best_score = float('inf')
    
    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-13:
            best_h = h.copy()
            best_score = sc
    
    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass
    
    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(2.0 * np.pi * x / 2.0))
        best_score = official_score(best_h)
    
    update(best_h, best_score)
    
    # Phase 1: Pairwise coordinate descent (main workhorse)
    if remaining() > 60:
        h_opt, sc_opt = pairwise_descent(best_h, max_sweeps=150,
                                         time_limit=min(250, remaining() - 30),
                                         t_start=t_start, rng=rng)
        update(h_opt, sc_opt)
    
    # Phase 2: LP cutting-plane
    if remaining() > 60:
        h_opt, sc_opt = lp_optimize(best_h, max_iter=25, k_active=12,
                                    radius_start=0.1, time_limit=min(80, remaining() - 20),
                                    t_start=t_start)
        update(h_opt, sc_opt)
    
    # Phase 3: More pairwise from current best
    if remaining() > 40:
        h_opt, sc_opt = pairwise_descent(best_h, max_sweeps=100,
                                         time_limit=min(150, remaining() - 20),
                                         t_start=t_start, rng=rng)
        update(h_opt, sc_opt)
    
    # Phase 4: Subgradient descent
    if remaining() > 40:
        h_opt, sc_opt = gradient_descent_subgrad(best_h, max_iter=150,
                                                 step_start=0.008,
                                                 time_limit=min(60, remaining() - 15),
                                                 t_start=t_start)
        update(h_opt, sc_opt)
        if remaining() > 30:
            h_opt2, sc_opt2 = pairwise_descent(h_opt, max_sweeps=50,
                                               time_limit=min(40, remaining() - 15),
                                               t_start=t_start, rng=rng)
            update(h_opt2, sc_opt2)
    
    # Phase 5: Anti-symmetric transform + refine
    if remaining() > 40:
        h_sym = anti_sym_transform(best_h)
        sc_sym = official_score(h_sym)
        update(h_sym, sc_sym)
        if remaining() > 30:
            h_opt, sc_opt = pairwise_descent(h_sym, max_sweeps=60,
                                             time_limit=min(50, remaining() - 15),
                                             t_start=t_start, rng=rng)
            update(h_opt, sc_opt)
    
    # Phase 6: Resolution exploration
    if remaining() > 80:
        n_base = len(best_h)
        n_opts = sorted(set([max(20, n_base - 6), n_base - 2, n_base,
                             n_base + 2, n_base + 6]))
        for n_opt in n_opts:
            if remaining() < 20:
                break
            if n_opt < 16 or n_opt > 120:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            update(h_int, sc_int)
            if remaining() > 25:
                h_opt, sc_opt = pairwise_descent(h_int, max_sweeps=50,
                                                 time_limit=min(40, remaining() - 15),
                                                 t_start=t_start, rng=rng)
                update(h_opt, sc_opt)
    
    # Phase 7: Random restarts from perturbed best
    if remaining() > 60:
        n = len(best_h)
        for attempt in range(12):
            if remaining() < 20:
                break
            sigma = 0.015 + 0.04 * rng.random()
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n))
            h_opt, sc_opt = pairwise_descent(h_cand, max_sweeps=40,
                                             time_limit=min(30, remaining() - 10),
                                             t_start=t_start, rng=rng)
            update(h_opt, sc_opt)
    
    # Phase 8: Final LP + pairwise polish
    if remaining() > 40:
        h_opt, sc_opt = lp_optimize(best_h, max_iter=20, k_active=8,
                                    radius_start=0.05, time_limit=min(40, remaining() - 15),
                                    t_start=t_start)
        update(h_opt, sc_opt)
    if remaining() > 25:
        h_opt, sc_opt = pairwise_descent(best_h, max_sweeps=80,
                                         time_limit=min(40, remaining() - 10),
                                         t_start=t_start, rng=rng)
        update(h_opt, sc_opt)
    
    # Final: exact repair and validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    if final_score < best_score - 1e-12:
        best_score = final_score
    
    # Add tiny safety epsilon to reported bound
    final_bound = final_score + 1e-12
    return final_h, float(final_bound), len(final_h)
