import numpy as np
from scipy.optimize import minimize
import time

# === Helper functions ===

def safe_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    # Bisection on shift
    zmin, zmax = z.min(), z.max()
    lo = -zmin - 0.001
    hi = 1.0 - zmax + 0.001
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-15:
                break
            h[i] += add
            residual -= add
            if abs(residual) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.clip(np.interp(x_dst, x_src, h_src), 0.0, 1.0)

# === Smooth LSE objective and gradient ===

def lse_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def lse_grad(x, beta, dx):
    n = len(x)
    L = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        weights = np.ones(L) / L
    else:
        weights = e / s
    grad = np.zeros(n)
    for k in range(L):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        s2 = k - (n - 1)
        if s2 == 0:
            grad += 1.0 - 2.0 * x
        elif s2 > 0:
            grad[s2:] += (1.0 - x[:n - s2]) * w
            grad[:n - s2] += (-x[s2:]) * w
        else:
            m2 = -s2
            grad[:n - m2] += (1.0 - x[m2:]) * w
            grad[m2:] += (-x[:n - m2]) * w
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    try:
        res = minimize(
            lse_obj, h0, args=(beta, dx),
            jac=lambda x: lse_grad(x, beta, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0
