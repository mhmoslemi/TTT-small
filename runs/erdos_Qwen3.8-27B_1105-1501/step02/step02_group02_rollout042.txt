```python
def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:

            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)
```

For a fixed j:

Method: Bisection on t = target max value.
For a given t, find if ∃ a ∈ [0,1] such that F_k(a) ≤ t for all k.

For k ≠ n-1 (affine): 
- If coef[k] > 0: a ≤ (t - cv[k]) / coef[k] + old → upper bound on a
- If coef[k] < 0: a ≥ (t - cv[k]) / coef[k] + old → lower bound on a
- If coef[k] = 0: need cv[k] ≤ t

Wait, I need to be more careful. F_k(a) = cv[k] + (a - old) * coef[k]
F_k(a) ≤ t → (a - old) * coef[k] ≤ t - cv[k]
If coef[k] > 0: a - old ≤ (t - cv[k]) / coef[k] → a ≤ old + (t - cv[k]) / coef[k]
If coef[k] < 0: a - old ≥ (t - cv[k]) / coef[k] → a ≥ old + (t - cv[k]) / coef[k]
  (dividing by negative flips inequality)

For k = n-1 (quadratic):
-dx*a² + dx*a + C0 ≤ t
dx*a² - dx*a + (t - C0) ≥ 0
Roots: a = [1 ± sqrt(1 - 4(t-C0)/dx)] / 2
If discriminant < 0: always satisfied (since leading coeff > 0, quadratic is always positive)
  Wait: dx*a² - dx*a + (t-C0) with dx > 0. If discriminant < 0, the quadratic is always positive. So the constraint is always satisfied. ✓
If discriminant ≥ 0: roots r1 ≤ r2. Quadratic ≥ 0 means a ≤ r1 or a ≥ r2.

So for a given t, the feasible set for a is:
[LB, UB] ∩ ([0, r1] ∪ [r2, 1]) (if disc ≥ 0)
or [LB, UB] ∩ [0, 1] (if disc < 0)

where LB = max(0, all lower bounds from affine constraints)
and UB = min(1, all upper bounds from affine constraints)

The feasible set is non-empty if:
[LB, UB] ∩ ([0, r1] ∪ [r2, 1]) ≠ ∅
i.e., (LB ≤ min(UB, r1)) or (max(LB, r2) ≤ UB)

I can binary search on t to find the minimum t for which the feasible set is non-empty. Then the optimal a is at the "boundary" of the feasible set (one of the endpoints LB, UB, r1, r2).

```python
def _coord_opt_j(h, j, cv):
    """Optimize coordinate j. Returns (new_j_value, new_cv, new_max)."""
    n = len(h)
    dx = 2.0 / n
    old = h[j]
    
    # Compute coefficients
    coef = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        s = k - (n - 1)
        c = 0.0
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 <= j < hi1:
            c += dx * (1.0 - h[j + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 <= j < hi2:
            c += -dx * h[j - s]
        coef[k] = c
    
    C0 = cv[n-1] - dx * old * (1.0 - old)
    
    # Separate affine constraints
    # For k != n-1: F_k(a) = cv[k] + (a-old)*coef[k]
    # Upper bounds (coef > 0): a <= old + (t - cv[k]) / coef[k]
    # Lower bounds (coef < 0): a >= old + (t - cv[k]) / coef[k]
    
    pos_k = np.where((coef > 1e-14) & (np.arange(2*n-1) != n-1))[0]
    neg_k = np.where((coef < -1e-14) & (np.arange(2*n-1) != n-1))[0]
    zero_k = np.where((np.abs(coef) <= 1e-14) & (np.arange(2*n-1) != n-1))[0]
    
    # Bisection on t
    t_lo = max(cv.min(), 0.0)
    t_hi = cv.max()
    
    for _ in range(50):
        t_mid = (t_lo + t_hi) / 2.0
        
        # Check zero-coefficient constraints
        ok = True
        if len(zero_k) > 0 and np.any(cv[zero_k] > t_mid):
            ok = False
        
        if not ok:
            t_lo = t_mid
            continue
        
        # Compute LB and UB from affine constraints
        LB = 0.0
        UB = 1.0
        
        if len(pos_k) > 0:
            bounds = old + (t_mid - cv[pos_k]) / coef[pos_k]
            UB = min(UB, np.min(bounds))
        
        if len(neg_k) > 0:
            bounds = old + (t_mid - cv[neg_k]) / coef[neg_k]
            LB = max(LB, np.max(bounds))
        
        # Check s=0 constraint
        disc0 = 1.0 - 4.0 * (t_mid - C0) / dx
        feasible = (LB <= UB)

            sqrt0 = np.sqrt(disc0)
            r1 = (1.0 - sqrt0) / 2.0
            r2 = (1.0 + sqrt0) / 2.0
            # Feasible: a in [LB, UB] AND (a <= r1 OR a >= r2)
            if not (LB <= min(UB, r1) or max(LB, r2) <= UB):
                feasible = False
        
        if feasible:
            t_hi = t_mid
        else:
            t_lo = t_mid

    # Find optimal a at t_opt
    LB = 0.0
    UB = 1.0
    
    if len(pos_k) > 0:
        bounds = old + (t_opt - cv[pos_k]) / coef[pos_k]
        UB = min(UB, np.min(bounds))
    
    if len(neg_k) > 0:
        bounds = old + (t_opt - cv[neg_k]) / coef[neg_k]
        LB = max(LB, np.max(bounds))
    
    LB = max(0.0, LB)
    UB = min(1.0, UB)
    
    # Handle s=0

    if disc0 >= 0:
        sqrt0 = np.sqrt(disc0)
        r1 = (1.0 - sqrt0) / 2.0
        r2 = (1.0 + sqrt0) / 2.0
        # Prefer the side that gives the "tightest" solution
        # Try [max(LB, r2), UB] first (upper side)
        if max(LB, r2) <= UB:
            LB = max(LB, r2)
        else:
            UB = min(UB, r1)
    
    if LB > UB:

        LB, UB = old, old
    
    # Choose a: try LB, UB, midpoint, pick best
    candidates = [LB, UB, (LB + UB) / 2.0]
    best_a = old
    best_m = t_opt
    
    for a in candidates:
        a = max(0.0, min(1.0, a))
        m = 0.0

        if 2*n - 2 > 0:

            aff_max = max(aff_vals.max() if len(aff_vals) > 0 else 0,

    new_cv[n-1] = -dx*best_a*best_a + dx*best_a + C0
    
    return best_a, new_cv, new_cv.max()
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-15:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _lse_obj(x, beta):
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    es = np.sum(np.exp(sh))
    if es < 1e-300:
        return mc
    return mc + np.log(es) / beta


def _lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    cv = _corr_vec(x)
    mc = cv.max()
    sh = np.clip(beta * (cv - mc), -500, 0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            g[lo1:hi1] += wk * dx * (1.0 - x[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            g[lo2:hi2] += wk * dx * (-x[lo2 - s:hi2 - s])
    return g


def _fista_lse(h0, beta_start, n_iter, deadline, alpha_scale=1.0):
    """Nesterov-accelerated LSE with adaptive beta."""
    n = len(h0)
    h_cur = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_s = _score(h0)
    beta = beta_start
    t = 1
    it = 0
    while it < n_iter:
        if time.time() > deadline:
            break
        grad = _lse_grad(y, beta)
        lr = alpha_scale / np.sqrt(t + 1.0)
        h_new = y - lr * grad
        h_new = _project(h_new)
        t_prev = t
        t = 1.5 * (t + 1)
        y = h_new + (t_prev / t) * (h_new - h_prev)
        y = _project(y)
        h_avg += (h_new - h_avg) / (it + 1)
        h_prev = h_cur
        h_cur = h_new
        it += 1
        if it % 2000 == 0:
            beta *= 3.0
        if it % 500 == 0:
            s = _score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
            if it % 2000 == 0:
                ha = _project(h_avg)
                sa = _score(ha)
                if sa < best_s:
                    best_s = sa
                    best_h = ha.copy()
    s = _score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    ha = _project(h_avg)
    sa = _score(ha)
    if sa < best_s:
        best_s = sa
        best_h = ha.copy()
    return best_h, best_s

                r1 = (1.0 - sq) / 2.0
                r2 = (1.0 + sq) / 2.0
                if max(LB, r2) <= UB:
                    LB = max(LB, r2)
                else:
                    UB = min(UB, r1)

            if LB > UB:
                LB, UB = old, old

            candidates = [LB, UB, (LB + UB) / 2.0]
            best_a = old
            best_m = t_hi
            for a in candidates:
                a = max(0.0, min(1.0, a))
                aff_max = float(np.max(cv[:-1] + (a - old) * coef[:-1])) if n > 1 else 0.0
                aff_max2 = float(np.max(cv[n:] + (a - old) * coef[n:])) if n > 1 else 0.0
                aff_max = max(aff_max, aff_max2)
                qv = -dx * a * a + dx * a + C0
                m = max(aff_max, qv)
                if m < best_m - 1e-15:
                    best_m = m
                    best_a = a

            if best_m < best_score - 1e-13:
                h[j] = best_a
                cv = cv + (best_a - old) * coef
                cv[n - 1] = -dx * best_a * best_a + dx * best_a + C0
                best_score = best_m
                improved = True

        if rnd % 10 == 9:
            cv = _corr_vec(h)
            best_score = float(cv.max())
        if not improved:
            break

    h = _project(h)
    final_score = _score(h)
    return h, final_score


def _slsqp_active(h0, top_m=4, maxiter=80):
    """SLSQP with only top-m active lag constraints."""
    n = len(h0)
    cv = _corr_vec(h0)
    top_idx = np.argsort(cv)[::-1][:top_m]
    t0 = cv.max() + 0.002

    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def _fun(z):
        return z[-1]

    def _fun_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def _ieq(z):
        h = z[:-1]
        t = z[-1]
        cvl = _corr_vec(h)
        return t - cvl[top_idx]

    def _eq(z):
        return z[:-1].sum() - n / 2.0

    def _eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g

    try:
        res = minimize(_fun, z0, jac=_fun_jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': _ieq},
                           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'][:-1])
    except Exception:
        pass
    return h0


def _slsqp_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _efun(x):
        return x.sum() - n / 2.0

    def _ejac(x):
        return np.ones(n)

    try:
        res = minimize(_lse_obj, h0, args=(beta,), jac=_lse_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return _project(res['x'])
    except Exception:
        pass
    return h0


def _interp(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _perturb(h, rng, sigma):
    return _project(h + sigma * rng.standard_normal(len(h)))


def _sinusoidal_perturb(h, rng, freq, amp):
    n = len(h)
    phase = rng.uniform(0, 2 * np.pi)
    pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return _project(h + pert)


def _deep_optimize(h0, deadline, rng):
    """Full optimization chain for one starting point."""
    n = len(h0)
    cur = h0.copy()
    cur_s = _score(cur)

    # Stage 1: LSE + Nesterov (moderate beta)
    if time.time() < deadline - 20:
        new, ns = _fista_lse(cur, beta_start=80, n_iter=8000, deadline=deadline - 10)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns

    # Stage 2: Coordinate descent
    if time.time() < deadline - 15:
        cur, cur_s = _coord_descend(cur, n_rounds=40, deadline=deadline - 5)

    # Stage 3: Active-set SLSQP
    if time.time() < deadline - 10:
        new = _slsqp_active(cur, top_m=4, maxiter=60)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns
        cur, cur_s = _coord_descend(cur, n_rounds=30, deadline=deadline - 5)

    # Stage 4: Higher beta LSE
    if time.time() < deadline - 15:
        new, ns = _fista_lse(cur, beta_start=400, n_iter=6000, deadline=deadline - 8)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns
        cur, cur_s = _coord_descend(cur, n_rounds=40, deadline=deadline - 5)

    # Stage 5: Final SLSQP polish
    if time.time() < deadline - 8:
        new = _slsqp_smooth(cur, beta=600, maxiter=80)
        ns = _score(new)
        if ns < cur_s - 1e-12:
            cur, cur_s = new, ns
        cur, cur_s = _coord_descend(cur, n_rounds=50, deadline=deadline - 3)

    # Final coordinate descent
    if time.time() < deadline - 2:
        cur, cur_s = _coord_descend(cur, n_rounds=80, deadline=deadline - 1)

    return cur, cur_s


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.08 * budget_s)
    deadline = t_start + budget_s - reserve

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = _project(parent)
    except Exception:
        parent = _project(np.ones(50) * 0.5)

    best_h = parent.copy()
    best_score = _score(best_h)
    best_n = len(best_h)
    parent_n = len(parent)

    rng = np.random.default_rng(seed)

    # Phase 1: Multi-n screening
    n_range = list(range(max(30, parent_n - 15), min(80, parent_n + 20) + 1))
    # Ensure even and odd are both included, step by 2 for speed
    n_candidates = sorted(set(n_range[::2]) | {parent_n})

    phase1_results = []
    for n in n_candidates:
        if time.time() > deadline - 120:
            break
        h_interp = _interp(parent, n)
        # Quick LSE
        h_opt, s_opt = _fista_lse(h_interp, beta_start=60, n_iter=3000,
                                  deadline=deadline - 60)
        phase1_results.append((s_opt, n, h_opt))
        sc = _score(h_opt)
        if sc < best_score - 1e-12:
            best_h = h_opt.copy()
            best_score = sc
            best_n = n

    # Sort by score, keep top 6
    phase1_results.sort(key=lambda x: x[0])
    top_candidates = []
    seen_ns = set()
    for s, n, h in phase1_results:
        if n not in seen_ns:
            top_candidates.append((s, n, h))
            seen_ns.add(n)
        if len(top_candidates) >= 6:
            break

    # Phase 2: Deep optimization on top candidates
    for s, n, h in top_candidates:
        if time.time() > deadline - 60:
            break
        # Generate diverse starts
        starts = [h.copy()]
        starts.append(_project(h[::-1].copy()))
        starts.append(_perturb(h, rng, 0.1))
        starts.append(_sinusoidal_perturb(h, rng, 2, 0.08))
        starts.append(_perturb(h, rng, 0.2))

        for h0 in starts:
            if time.time() > deadline - 30:
                break
            cur, cur_s = _deep_optimize(h0, deadline, rng)
            if cur_s < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_s
                best_n = len(cur)

    # Phase 3: Basin hopping around best
    remaining = deadline - time.time()
    max_hops = min(12, int(remaining / 40))
    for hop in range(max_hops):
        if time.time() > deadline - 20:
            break
        # Try different n values near best_n
        delta_n = int(rng.choice([-4, -2, -1, 0, 1, 2, 4]))
        n_try = best_n + delta_n
        n_try = max(30, min(80, n_try))
        h0 = _interp(best_h, n_try)
        h0 = _perturb(h0, rng, 0.05)
        cur, cur_s = _deep_optimize(h0, deadline, rng)
        if cur_s < best_score - 1e-12:
            best_h = cur.copy()
            best_score = cur_s
            best_n = len(cur)

    # Phase 4: Final intensive polish
    if time.time() < deadline - 10:
        best_h, best_score = _fista_lse(best_h, beta_start=2000, n_iter=10000,
                                        deadline=deadline - 5)
    if time.time() < deadline - 5:
        best_h, best_score = _coord_descend(best_h, n_rounds=100, deadline=deadline - 2)
    if time.time() < deadline - 3:
        new = _slsqp_active(best_h, top_m=5, maxiter=50)
        ns = _score(new)
        if ns < best_score - 1e-12:
            best_h = new
            best_score = ns
            best_h, best_score = _coord_descend(best_h, n_rounds=60, deadline=deadline - 1)

    # Final feasibility check
    best_h = _project(best_h)
    final_score = _score(best_h)

    return (best_h, final_score, len(best_h))
```
