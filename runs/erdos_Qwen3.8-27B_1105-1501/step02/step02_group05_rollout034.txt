import numpy as np
from scipy.optimize import linprog
import time

def score_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z[np.isnan(z) | np.isinf(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Greedy repair
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def get_active_set(V, top_k=32, threshold=0.02):
    """Select active lags: top K or within threshold of max."""
    C = V.max()
    mask = V >= C - threshold
    active = np.where(mask)[0]
    if len(active) > top_k:
        # Keep top K
        sorted_idx = np.argsort(V)[::-1][:top_k]
        active = sorted_idx
    return active, C

def compute_gradient_for_lag(h, p, n):
    """Compute gradient of V[p] = dx * raw[p] with respect to h."""
    dx = 2.0 / n
    k = p - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # result = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        for i in range(n - k):
            grad[i + k] += dx * (1.0 - h[i])
            grad[i] -= dx * h[i + k]
    else:
        t = -k
        # result = sum_{j=0}^{n-1-t} h[j] * (1-h[j+t])
        for j in range(n - t):
            grad[j] += dx * (1.0 - h[j + t])
            grad[j + t] -= dx * h[j]
    return grad

def compute_gradient_vectorized(h, p, n):
    """Vectorized gradient computation."""
    dx = 2.0 / n
    k = p - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        i = np.arange(m)
        grad[i + k] += dx * (1.0 - h[i])
        grad[i] -= dx * h[i + k]
    else:
        t = -k
        m = n - t
        j = np.arange(m)
        grad[j] += dx * (1.0 - h[j + t])
        grad[j + t] -= dx * h[j]
    return grad

def validate_orientation(h_test):
    """Validate that our gradient convention matches numpy's correlate."""
    n = len(h_test)
    dx = 2.0 / n
    V = np.correlate(h_test, 1.0 - h_test, mode='full')
    # Check a few lags by finite difference
    for p in [0, n-1, 2*n-2]:
        k = p - (n-1)
        g_analytic = compute_gradient_vectorized(h_test, p, n)
        # Finite difference check
        eps = 1e-7
        diff = np.zeros(n)
        for i in range(n):
            h_plus = h_test.copy()
            h_plus[i] += eps
            V_plus = float(np.correlate(h_plus, 1.0 - h_plus, mode='full')[p])
            h_minus = h_test.copy()
            h_minus[i] -= eps
            V_minus = float(np.correlate(h_minus, 1.0 - h_minus, mode='full')[p])
            diff[i] = (V_plus - V_minus) / (2 * eps) * dx
        if np.max(np.abs(g_analytic - diff)) > 1e-5:
            return False
    return True

def lp_minimax_step(h, active, V, C, rho, top_k=32):
    """Solve LP minimax step on active lags."""
    n = len(h)
    # Compute gradients for active lags
    # Use vectorized approach
    G = np.zeros((len(active), n))
    for idx, p in enumerate(active):
        G[idx] = compute_gradient_vectorized(h, p, n)
    
    c = V[active]  # values at active lags
    
    # LP: min t
    # s.t. G @ u - t <= -c  (i.e., G @ u - t + c <= 0)
    #      sum(u) = 0
    #      max(-h_i, -rho) <= u_i <= min(1-h_i, rho)
    
    n_vars = n + 1  # u_0, ..., u_{n-1}, t
    # Objective: minimize t
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0  # coefficient of t
    
    # Inequality constraints: G @ u - t <= -c
    # A_ub @ x <= b_ub
    A_ub = np.zeros((len(active), n_vars))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c
    
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, C))  # t <= C (can't predict improvement greater than current)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            u = res.x[:n]
            if np.max(np.abs(u)) > 1e-10:
                return u
            else:
                return None
        else:
            return None
    except Exception:
        return None

def lp_refine(h, max_iters=200, time_limit=None, t_start=None):
    """LP-based refinement loop."""
    n = len(h)
    dx = 2.0 / n
    h = make_feasible(h.copy())
    best_h = h.copy()
    best_score = score_exact(best_h)
    
    rho = 0.15
    eta = 0.015
    top_k = min(32, 2*n-1)
    stall = 0
    max_stall = 30
    
    for it in range(max_iters):
        if time_limit is not None:
            if t_start is not None and time.time() - t_start > time_limit:
                break
        
        V = np.correlate(h, 1.0 - h, mode='full') * dx
        C = V.max()
        
        active, _ = get_active_set(V, top_k=top_k, threshold=eta)
        if len(active) == 0:
            active = np.array([np.argmax(V)])
        
        u = lp_minimax_step(h, active, V, C, rho)
        
        accepted = False
        if u is not None:
            # Line search
            scales = [1.0, 0.75, 0.5, 0.35, 0.2, 0.1, 0.05]
            for alpha in scales:
                trial = make_feasible(h + alpha * u)
                sc = score_exact(trial)
                if sc < best_score - 1e-12:
                    h = trial
                    best_h = trial.copy()
                    best_score = sc
                    accepted = True
                    break
                elif sc < C - 1e-12:
                    # Accept if it reduces current C even if not below best
                    h = trial
                    if sc < best_score:
                        best_h = trial.copy()
                        best_score = sc
                    accepted = True
                    break
            if accepted:
                stall = 0
                rho = max(0.03, rho * 0.9)
                continue
        
        # Not accepted - adjust
        stall += 1
        if stall >= max_stall:
            # Try different parameters
            rho *= 0.5
            eta *= 0.5
            if rho < 0.005 or eta < 0.001:
                break
            stall = 0
        else:
            rho *= 0.8
            if rho < 0.02:
                rho = 0.15
                eta *= 0.7
            if eta < 0.001:
                eta = 0.015
        
        # Occasionally try perturbation
        if stall % 10 == 0:
            rng = np.random.default_rng(hash((it, n)) % 2**32)
            pert = rng.normal(0, 0.02, size=n)
            pert -= pert.mean()
            h_trial = make_feasible(h + pert)
            sc_trial = score_exact(h_trial)
            if sc_trial < best_score:
                h = h_trial
                best_h = h_trial.copy()
                best_score = sc_trial
                stall = 0
    
    return best_h, best_score

def binary_annealing(h_init, n_steps=3000, time_limit=None, t_start=None, seed=42):
    """Binary simulated annealing."""
    n = len(h_init)
    if n % 2 != 0:
        return None, None
    
    target_ones = n // 2
    rng = np.random.default_rng(seed)
    
    # Initialize from h_init
    h = np.asarray(h_init, dtype=float).ravel()
    # Threshold to binary
    sorted_idx = np.argsort(-h)
    bits = np.zeros(n, dtype=int)
    bits[sorted_idx[:target_ones]] = 1
    
    current_score = score_exact(bits.astype(float))
    best_bits = bits.copy()
    best_score = current_score
    
    T = 5e-4
    cool = 0.999
    
    pos1 = np.where(bits == 1)[0]
    pos0 = np.where(bits == 0)[0]
    
    for it in range(n_steps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        # Random swap
        i = rng.integers(0, len(pos1))
        j = rng.integers(0, len(pos0))
        a = pos1[i]
        b = pos0[j]
        
        bits[a] = 0
        bits[b] = 1
        pos1[i] = b
        pos0[j] = a
        
        trial_score = score_exact(bits.astype(float))
        delta = trial_score - current_score
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_bits = bits.copy()
        else:
            # Revert
            bits[a] = 1
            bits[b] = 0
            pos1[i] = a
            pos0[j] = b
        
        T *= cool
    
    return best_bits.astype(float), best_score

def interp_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def make_wave(n, freqs, amps, phases=None):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.ones(n) * (0.5)
    if phases is None:
        phases = np.zeros(len(freqs))
    for f, a, ph in zip(freqs, amps, phases):
        h += a * np.sin(2 * np.pi * f * x + ph)
    return make_feasible(h)

def two_point_search(h, n_tries, rng):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for it in range(n_tries):
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        span = t_hi - t_lo
        t = t_lo + rng.random() * span
        tmp = best.copy()
        tmp[a] = best[a] + t
        tmp[b] = best[b] - t
        sc = score_exact(tmp)
        if sc < best_sc:
            best = tmp
            best_sc = sc
    return best, best_sc

def score_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    return float(vals.max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    if n == 0:
        return z
    target = n / 2.0
    z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, p, n):
    """Gradient of V[p] = dx * correlate(h,1-h,'full')[p] w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if p >= n - 1:
        s = p - (n - 1)
        m = n - s
        j = np.arange(m)
        grad[j + s] += dx * (1.0 - h[j])
        grad[j] -= dx * h[j + s]
    else:
        t = (n - 1) - p
        m = n - t
        i = np.arange(m)
        grad[i] += dx * (1.0 - h[i + t])
        grad[i + t] -= dx * h[i]
    return grad

def verify_gradients(h, n):
    """Verify gradient computation against finite differences."""
    dx = 2.0 / n
    eps = 1e-7
    for p in [0, n - 1, 2 * n - 2]:
        g = grad_lag(h, p, n)
        fd = np.zeros(n)
        for i in range(n):
            hp = h.copy(); hp[i] += eps
            hm = h.copy(); hm[i] -= eps
            fd[i] = (np.correlate(hp, 1-hp, mode='full')[p] - np.correlate(hm, 1-hm, mode='full')[p]) / (2*eps) * dx
        if np.max(np.abs(g - fd)) > 1e-4:
            return False
    return True

def lp_step(h, active, V, C, rho):
    """Solve LP minimax direction on active lags. Returns direction or None."""
    n = len(h)
    na = len(active)
    if na == 0:
        return None
    # Build gradient matrix
    G = np.zeros((na, n))
    for idx, p in enumerate(active):
        G[idx] = grad_lag(h, int(p), n)
    c = V[active]
    
    # Variables: u_0..u_{n-1}, t  (total n+1)
    nvar = n + 1
    # Objective: min t
    c_obj = np.zeros(nvar)
    c_obj[n] = 1.0
    # Inequalities: G@u - t <= -c  =>  [G, -1] @ [u;t] <= -c
    A_ub = np.zeros((na, nvar))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c.copy()
    # Equality: sum(u) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i] + 1e-9, -rho)
        hi = min(1.0 - h[i] - 1e-9, rho)
        if lo >= hi:
            lo = (h[i] + 1.0 - h[i]) / 2.0  # = 0
            lo = -rho * 0.01
            hi = rho * 0.01
        bounds.append((lo, hi))
    bounds.append((None, C + 1e-9))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            u = res.x[:n].copy()
            if np.linalg.norm(u) > 1e-10:
                return u
    except Exception:
        pass
    return None

def two_point_random(h, n_tries, rng):

    best = h.copy()
    best_sc = score_exact(best)

def two_point_grad(h, n_tries, rng):
    """Gradient-informed 2-point search."""
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    dx = 2.0 / n
    for it in range(n_tries):
        V = np.correlate(best, 1.0 - best, mode='full') * dx
        C = V.max()
        # Pick worst lag
        p_worst = np.argmax(V)
        g = grad_lag(best, p_worst, n)
        # Choose a from high positive gradient, b from high negative
        # We want to decrease V[p_worst], so move against gradient
        # d = -g means we want to decrease h where g>0 and increase where g<0
        # Pick a (decrease) from top g, b (increase) from bottom g
        idx_a = np.argmax(g)  # largest positive gradient -> decrease h[a]
        idx_b = np.argmin(g)  # most negative gradient -> increase h[b]
        if idx_a == idx_b:
            idx_a = (idx_a + 1) % n
        # Also try random
        if rng.random() < 0.3:
            idx_a = rng.integers(0, n)
            idx_b = rng.integers(0, n)
            if idx_a == idx_b:
                idx_b = (idx_b + 1) % n
        
        t_lo = max(-best[idx_a], best[idx_b] - 1.0)
        t_hi = min(1.0 - best[idx_a], best[idx_b])
        if t_hi - t_lo < 1e-12:
            continue
        # We want t > 0 (decrease a, increase b) if g[idx_a] > 0 and g[idx_b] < 0
        candidates = [0.3 * (t_hi - t_lo) + t_lo, 0.7 * (t_hi - t_lo) + t_lo,
                      0.5 * (t_hi + t_lo), t_lo + 0.1*(t_hi-t_lo), t_lo + 0.9*(t_hi-t_lo)]
        best_t = None
        best_local = best_sc

def binary_anneal(h_init, n_steps, time_limit, t_start, seed):
    n = len(h_init)
    if n % 2 != 0:
        return None
    target = n // 2
    rng = np.random.default_rng(seed)
    h = np.asarray(h_init, dtype=float).ravel()
    sorted_idx = np.argsort(-h)
    bits = np.zeros(n, dtype=np.int8)
    bits[sorted_idx[:target]] = 1
    cur_score = score_exact(bits.astype(float))
    best_bits = bits.copy()
    best_score = cur_score
    T = 3e-4
    cool = 0.9992
    pos1 = list(np.where(bits == 1)[0])
    pos0 = list(np.where(bits == 0)[0])
    for it in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        i = rng.integers(0, len(pos1))
        j = rng.integers(0, len(pos0))
        a = pos1[i]
        b = pos0[j]
        bits[a] = 0
        bits[b] = 1
        pos1[i] = b
        pos0[j] = a
        trial_sc = score_exact(bits.astype(float))
        delta = trial_sc - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_score = trial_sc
            if trial_sc < best_score:
                best_score = trial_sc
                best_bits = bits.copy()
        else:
            bits[a] = 1
            bits[b] = 0
            pos1[i] = a
            pos0[j] = b
        T *= cool
    return best_bits.astype(float)

    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    if n == 0:
        return z
    target = n / 2.0
    z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                if add <= 1e-15:
                    continue
                h[idx] += add
                residual -= add
            else:
                sub = min(-residual, h[idx])
                if sub <= 1e-15:
                    continue
                h[idx] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, p, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if p >= n - 1:
        s = p - (n - 1)
        m = n - s
        j = np.arange(m)
        grad[j + s] += dx * (1.0 - h[j])
        grad[j] -= dx * h[j + s]
    else:
        t = (n - 1) - p
        m = n - t
        i = np.arange(m)
        grad[i] += dx * (1.0 - h[i + t])
        grad[i + t] -= dx * h[i]
    return grad

def verify_gradients(h, n):

    eps = 1e-7
    for p in [0, n - 1, 2 * n - 2, n // 2]:
        g = grad_lag(h, p, n)
        fd = np.zeros(n)
        for i in range(n):
            hp = h.copy()
            hp[i] += eps
            hm = h.copy()
            hm[i] -= eps
            fd[i] = (np.correlate(hp, 1 - hp, mode='full')[p] -
                     np.correlate(hm, 1 - hm, mode='full')[p]) / (2 * eps) * dx
        if np.max(np.abs(g - fd)) > 1e-4:
            return False
    return True

def two_point_random(h, n_tries, rng):
    n = len(h)
    best = h.copy()
    best_sc = score_exact(best)
    for _ in range(n_tries):
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best[a], best[b] - 1.0)
        t_hi = min(1.0 - best[a], best[b])
        if t_hi - t_lo < 1e-12:
            continue
        t = t_lo + rng.random() * (t_hi - t_lo)
        tmp = best.copy()
        tmp[a] = best[a] + t
        tmp[b] = best[b] - t
        sc = score_exact(tmp)
        if sc < best_sc:
            best = tmp
            best_sc = sc
    return best, best_sc

def two_point_grad(h, n_tries, rng):
