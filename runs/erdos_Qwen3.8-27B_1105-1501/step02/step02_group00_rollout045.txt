```python
def _grad_matrix(h):
    n = len(h)
    one_minus_h = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        # G[r, j] = (1-h[j+d]) * 1{0<=j+d<n} - h[j-d] * 1{0<=j-d<n}
        # For the first term: j+d in [0,n) means j in [-d, n-1-d)
        # For the second term: j-d in [0,n) means j in [d, n-1+d)
        jp = j+d (first term valid range)
        jm = j-d (second term valid range)
        
        # Vectorized:
        # First term: for j where 0<=j+d<n, G[r,j] += 1-h[j+d]
        # j ranges from max(0,-d) to min(n-1, n-1-d) for d>=0, or 0 to n-1 for d<0... 
        # Actually for d>=0: j+d<n means j<n-d, and j>=0, so j in [0, n-1-d]
        #              j-d>=0 means j>=d, so j in [d, n-1]
        # For d<0: j+d<0 for j<|d|, so j in [|d|, n-1] for first term
        #          j-d<n means j<n+d, and j>=0, so j in [0, n+d-1] for second term
        
        if d >= 0:
            # First term: j in [0, n-1-d], G[r,j] += 1-h[j+d]
            m = n - d
            if m > 0:
                G[r, :m] += one_minus_h[d:]
            # Second term: j in [d, n-1], G[r,j] -= h[j-d]
            if d > 0:
                G[r, d:] -= h[:n-d]
        else:
            gd = -d
            # First term: j in [gd, n-1], G[r,j] += 1-h[j+d] = 1-h[j-gd]
            m = n - gd
            if m > 0:
                G[r, gd:] += one_minus_h[:m]
            # Second term: j in [0, n-1-gd], G[r,j] -= h[j-d] = h[j+gd]
            if gd > 0:
                G[r, :n-gd] -= h[gd:]
    return G
```

```python
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
