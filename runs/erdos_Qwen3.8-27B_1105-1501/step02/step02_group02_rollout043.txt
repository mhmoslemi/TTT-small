    h=np.asarray(h,float).copy()

    best=_raw_max(h)

def _project(z):
    z = np.asarray(z, dtype=float).ravel()
    if z.size == 0:
        return np.empty(0)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    target = 0.5 * n
    lo = -2.0 - float(np.max(z))
    hi = 2.0 - float(np.min(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
            else:
                add = max(res, h[i] - 1.0)
            if abs(add) < 1e-16:
                break
            h[i] += add
            res -= add
    return np.clip(h, 0.0, 1.0)


def _raw_corr_vec(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def _raw_max(h):

def _score_exact(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 1.0
    return _raw_max(h) * (2.0 / n)


def _interp(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float).ravel()
    n_src = len(h_src)
    if n_dst <= 0:
        return np.empty(0)
    if n_src == n_dst:
        return h_src.copy()
    if n_src == 0:
        return _project(np.full(n_dst, 0.5))
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return _project(np.interp(x_dst, x_src, h_src))


def _mirror(h):

def _symm(h):

def _threshold_binary(h):

def _anti_expand(u, n):

def _dF_dh_one(s, h):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[:m] += 1.0 - h[s:s + m]
            g[s:s + m] -= h[:m]
    else:
        r = -s
        m = n - r
        if m > 0:
            g[r:r + m] += 1.0 - h[:m]
            g[:m] -= h[r:r + m]
    return g


def _am_ineq(x):
    v = x[:-1]
    t = x[-1]
    h = _expand_var(v)
    n = _G_n
    out = np.empty(len(_G_shifts))
    for a, s in enumerate(_G_shifts):

        else:
            r = -s
            m = n - r
            out[a] = t - np.dot(h[r:r + m], 1.0 - h[:m])
    return out


def _am_jac(x):
    v = x[:-1]
    h = _expand_var(v)
    n = _G_n
    nv = len(v)
    m = len(_G_shifts)
    J = np.zeros((m, nv + 1))

def _lse_val(x, beta):
    h = np.asarray(x, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(c.max())
    sh = np.clip(float(beta) * (c - m), -700.0, 0.0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        return m
    return m + np.log(es) / float(beta)


def _lse_gradf(x):
    h = np.asarray(x, dtype=float)
    n = len(h)
    if n == 0:
        return np.empty(0)
    dx = 2.0 / n
    p = n - 1
    beta = _LSE_BETA
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(c.max())
    sh = np.clip(beta * (c - m), -700.0, 0.0)
    ev = np.exp(sh)
    es = ev.sum()
    if es < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = ev / es
    wl = w[::-1]
    A = np.convolve(wl, 1.0 - h, mode="full")[p:p + n]
    B = np.convolve(w, h, mode="full")[p:p + n]
    return dx * (A - B)


def _lse_obj(x):
    return _lse_val(x, _LSE_BETA)


def _lse_eq(x):
    return float(np.sum(x) - 0.5 * len(x))

def _lse_slsqp(h0, beta, maxiter):

        if hasattr(res, "x") and res.x is not None and np.all(np.isfinite(res.x)):

    except Exception:
        pass
    return h0


def _random_transfer_local(h0, deadline, rng):
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    if n == 0:
        return h, 0.0
    best = _raw_max(h)
    while time.time() < deadline - 5.0:
        for _ in range(500):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = float(rng.uniform(-0.25, 0.25))
            if d > 0:
                if h[i] + d > 1.0 or h[j] - d < 0.0:
                    continue
            else:
                if h[i] + d < 0.0 or h[j] - d > 1.0:
                    continue
            old_i = h[i]
            old_j = h[j]
            h[i] += d
            h[j] -= d
            cur = _raw_max(h)
            if cur < best - 1e-13:
                best = cur
            else:
                h[i] = old_i
                h[j] = old_j
        if time.time() > deadline - 20.0:
            break
    return h, best * (2.0 / n)

                    best_c = c2
                    best_h = h_new.copy()
                    best_n = n
                dcount += 1

    if time.time() < deadline - 160.0:
        for beta in (180.0, 700.0):
            if time.time() > deadline - 120.0:
                break
            h_lse = _lse_slsqp(best_h, beta, 70)
            h_lse = _project(h_lse)
            c = _score_exact(h_lse)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h_lse.copy()
                best_n = len(h_lse)

    sweeps = (
        (0.06, 14, 20, 3, 50, (0.0,)),
        (0.02, 10, 16, 4, 60, (0.0, 1e-5)),
        (0.008, 12, 18, 4, 60, (0.0,)),
        (0.20, 20, 28, 3, 40, (0.0,)),
    )
    for eps, topk, maxa, rounds, maxit, rhos in sweeps:
        if time.time() > deadline - 80.0:
            break
        var_new, h_new, raw = _active_solve(
            best_h, best_n, 0, deadline - 70.0,
            rounds=rounds, top_k=topk, active_eps=eps,
            max_active=maxa, maxiter=maxit, try_rhos=rhos
        )
        h_new = _project(h_new)
        c = _score_exact(h_new)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
            best_n = len(h_new)

    if best_n // 2 > 0 and time.time() < deadline - 90.0:
        var = best_h[:best_n // 2].copy()
        var_new, h_new, raw = _active_solve(
            var, best_n, 1, deadline - 70.0,
            rounds=4, top_k=12, active_eps=0.03,
            max_active=18, maxiter=50, try_rhos=(0.0, 1e-5)
        )
        h_new = _project(h_new)
        c = _score_exact(h_new)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
            best_n = len(h_new)

    if time.time() < deadline - 60.0:
        h_loc, c_loc = _random_transfer_local(best_h, deadline - 35.0, rng)
        h_loc = _project(h_loc)
        c_loc = _score_exact(h_loc)
        if c_loc < best_c - 1e-14:
            best_c = c_loc
            best_h = h_loc.copy()
            best_n = len(h_loc)

    if time.time() < deadline - 80.0:
        hb = _threshold_binary(best_h)
        hb, cb = _binary_hillclimb(hb, 20000, rng, deadline - 50.0)
        hb = _project(hb)
        cb = _score_exact(hb)
        if cb < best_c - 1e-14:
            best_c = cb
            best_h = hb.copy()
            best_n = len(hb)
        if time.time() < deadline - 40.0:
            var_new, h_new, raw = _active_solve(
                best_h, best_n, 0, deadline - 30.0,
                rounds=2, top_k=12, active_eps=0.02,
                max_active=18, maxiter=40, try_rhos=(0.0,)
            )
            h_new = _project(h_new)
            c = _score_exact(h_new)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h_new.copy()
                best_n = len(h_new)

    if time.time() < deadline - 40.0:
        h_lse = _lse_slsqp(best_h, 1600.0, 50)
        h_lse = _project(h_lse)
        c = _score_exact(h_lse)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_lse.copy()
            best_n = len(h_lse)

    if time.time() < deadline - 25.0:
        var_new, h_new, raw = _active_solve(
            best_h, best_n, 0, deadline - 15.0,
            rounds=2, top_k=12, active_eps=0.006,
            max_active=18, maxiter=35, try_rhos=(0.0,)
        )
        h_new = _project(h_new)
        c = _score_exact(h_new)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
            best_n = len(h_new)

    best_h = _project(best_h)
    best_c = _score_exact(best_h)
    return (best_h, float(best_c), int(len(best_h)))
