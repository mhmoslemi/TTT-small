    scores_c = scores - np.max(scores)
    w = np.exp(beta * scores_c)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(2*n-1)
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    G = _score_grad_j(h)
    grad = G.T @ w
    # Update momentum
    momentum_buf[:] = mu * momentum_buf + (1 - mu) * grad
    d = -momentum_buf
    # Normalize
    norm = np.max(np.abs(d))
    if norm < 1e-15:
        return d, scores
    d /= norm
    return d, scores

def _score_grad_j(h):
    n = len(h)

        G[n-1+tau, tau:] += 1.0 - h[:m]
        G[n-1-tau, tau:] += 1.0 - h[:m]
