import numpy as np
from scipy.optimize import linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_raw(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def gradient_at_shift(h, s):
    """Gradient of raw(p) with respect to h, where p = (n-1) + s."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # First position: h[j] * (1-h[j+s]) for j = 0..n-1-s
        if n - 1 - s >= 0:
            g[:n-s] += (1.0 - h[s:])  # (1-h[j+s]) for j=0..n-1-s
        # Second position: h[j-s] * (1-h[j]) for j = s..n-1, contributes -h[j-s]
        if s <= n - 1:
            g[s:] -= h[:n-s]  # -h[j-s] for j=s..n-1
    else:
        t = -s
        # First position: h[j] * (1-h[j-t]) for j = t..n-1
        if t <= n - 1:
            g[t:] += (1.0 - h[:-t])  # (1-h[j-t]) for j=t..n-1
        # Second position: h[j+t] * (1-h[j]) for j = 0..n-1-t, contributes -h[j+t]
        if t <= n - 1:
            g[:n-t] -= h[t:]  # -h[j+t] for j=0..n-1-t

    """One LP minimax step. Returns (h_new, t_lp) or (None, None)."""
    n = len(h_c)
    total = 2 * n - 1  # number of raw values
    max_raw = raw_vals.max()
    
    # Select active set: top k or within threshold
    threshold = max_raw - 0.01  # include all within 0.01 of max
    active_idx = np.where(raw_vals >= threshold)[0]
    if len(active_idx) < k_active:
        active_idx = np.argsort(raw_vals)[-k_active:]
    
    # Build LP
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub_rows = []
    b_ub_vals = []
    for p in active_idx:
        s = p - (n - 1)
        g = gradient_at_shift(h_c, s)
        rhs = raw_vals[p] - g @ h_c
        A_ub_rows.append(np.append(g, -1.0))
        b_ub_vals.append(rhs)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    lbs = np.maximum(0.0, h_c - radius)
    ubs = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lbs, ubs))
    bounds.append((-1.0, None))  # t

    # Find best

    if scores[idx] >= best_score:
        return h_c.copy(), best_score

    lo = alphas[max(0, idx-1)]
    hi = alphas[min(len(alphas)-1, idx+1)]
    
    # Golden section
