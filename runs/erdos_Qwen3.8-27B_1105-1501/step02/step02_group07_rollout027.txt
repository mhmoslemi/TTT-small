    n = len(h)
    dx = 2.0 / n
    # Compute current V
    V_old = np.correlate(h, 1.0 - h, mode='full')
    score_old = V_old.max() * dx
    
    # Swap
    h[j] = 1.0
    h[i] = 0.0
    V_new = np.correlate(h, 1.0 - h, mode='full')
    score_new = V_new.max() * dx
    
    delta = score_new - score_old
    if delta < 0:
        return True, h, score_new
    else:
        if T > 0 and rng.random() < np.exp(-delta / T):
            return True, h, score_new
        else:
            h[i] = 1.0
            h[j] = 0.0
            return False, h, score_old

def sa_binary(h0, rng, max_iters=50000, T0=0.05, cooling=0.9999, time_limit=30.0):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        one_idx = np.where(h == 1)[0]
        rng.shuffle(one_idx)
        h[one_idx[:ones - target]] = 0
    elif ones < target:
        zero_idx = np.where(h == 0)[0]
        rng.shuffle(zero_idx)
        h[zero_idx[:target - ones]] = 1
    
    V = np.correlate(h, 1.0 - h, mode='full')
    score = V.max() * (2.0 / n)
    best_h = h.copy()
    best_score = score
    
    T = T0
    t_start = time.time()
    one_idx = np.where(h == 1)[0].astype(int)
    zero_idx = np.where(h == 0)[0].astype(int)
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        
        i = one_idx[rng.integers(len(one_idx))]
        j = zero_idx[rng.integers(len(zero_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        V = np.correlate(h, 1.0 - h, mode='full')
        new_score = V.max() * (2.0 / n)
        
        delta = new_score - score
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            score = new_score
            # Update index sets
            one_idx = np.where(h == 1)[0].astype(int)
            zero_idx = np.where(h == 0)[0].astype(int)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cooling
    
    return best_h, best_score

def pg_step(h, beta, alpha_max=0.8, max_ls=14):
    n = len(h)
    V = compute_values(h)
    f = V.max()
    V_max = V.max()
    shifted = beta * (V - V_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(V), 1.0) / len(V)
    else:
        w /= ws
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if abs(wi) < 1e-14:
            continue
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grad[s:n] += wi * (1.0 - h[0:L])
            grad[0:L] -= wi * h[s:n]
        else:
            t = -s
            grad[0:L] += wi * (1.0 - h[t:n])
            grad[t:n] -= wi * h[0:L]
    
    g = grad - grad.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, f
    g /= gn
    
    best_h = h.copy()
    best_f = f
    alpha = alpha_max
    for _ in range(max_ls):
        h_new = make_feasible(h - alpha * g)
        f_new = official_score(h_new)
        if f_new < best_f - 1e-14:
            best_h = h_new
            best_f = f_new
        alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_f

def active_set_lp(h, eps=0.005, trust=0.2):
    n = len(h)
    dx = 2.0 / n
    V = compute_values(h)
    V_max = V.max()
    active = np.where(V >= V_max - eps)[0]
    if len(active) == 0:
        return None, None
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for p_idx in active:
        s = p_idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        gp = np.zeros(n)
        if s >= 0:
            gp[s:n] = 1.0 - h[0:L]
            gp[0:L] -= h[s:n]
        else:
            t = -s
            gp[0:L] = 1.0 - h[t:n]
            gp[t:n] -= h[0:L]
        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    
    if not A_rows:
        return None, None
    
    bounds_list = []
    for i in range(n):
        lo = max(-h[i], -trust)
        hi = min(1.0 - h[i], trust)
        bounds_list.append((lo, hi))
    bounds_list.append((0.0, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)

    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

        row = np.zeros(n + 1)
        row[:n] = dx * gp
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-V[p_idx])
    if not A_rows:
        return None, None
    bounds_list = []
    for i in range(n):
        bounds_list.append((max(-h[i], -trust), min(1.0 - h[i], trust)))
    bounds_list.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=[0.0], bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-10:
                h_new = make_feasible(h + d)
                sc = official_score(h_new)
                return h_new, sc
    except Exception:
        pass
    return None, None

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def pair_local_search(h, rng, max_passes=50, time_limit=30.0):
    n = len(h)
    best_h = h.copy()
    best_score = official_score(h)
    t_start = time.time()
    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            i = order[ii]
            for jj in range(ii + 1, n):
                j = order[jj]
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-10:
                    continue
                k_samples = 12
                for k in range(1, k_samples):
                    x = lo + (hi - lo) * k / k_samples
                    h_try = best_h.copy()
                    h_try[i] = x
                    h_try[j] = s - x
                    if h_try[i] < -1e-9 or h_try[i] > 1 + 1e-9 or h_try[j] < -1e-9 or h_try[j] > 1 + 1e-9:
                        continue
                    h_try = np.clip(h_try, 0.0, 1.0)
                    sc = official_score(h_try)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_h = h_try
                        improved = True
        if not improved:
            break
    return best_h, best_score

def soft_binarize(h, strength=3.0):
    n = len(h)
    centered = h - 0.5
    h_new = 0.5 + np.sign(centered) * np.abs(centered) ** (1.0 / strength)
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(10.0, 0.04 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    def tleft():
        return deadline - time.time()

    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        h_init = make_feasible(h_init)
    except (NameError, AttributeError):
        n = 60
        x = np.arange(n) / n
        h_init = 0.5 + 0.2 * np.sin(np.pi * x) + 0.08 * np.sin(2 * np.pi * x)
        h_init = make_feasible(h_init)

    best_h = h_init.copy()
    best_score = official_score(h_init)

    # Phase 1: Binary SA with diverse starts and multiple n
    if tleft() > 200:
        n_parent = len(h_init)
        n_options = [n_parent]
        for d in [2, 4, 6, 8, 10, 12, 16, 20]:
            if n_parent - d >= 28:
                n_options.append(n_parent - d)
            if n_parent + d <= 120:
                n_options.append(n_parent + d)
        n_options = sorted(set(n for x in n_options if x % 2 == 0 and 28 <= x <= 120))

        for n_try in n_options:
            if tleft() < 150:
                break
            if n_try == n_parent:
                h_start = h_init
            else:
                h_start = interpolate_to(h_init, n_try)
            h_bin = binarize_h(h_start, rng)
            h_sa, sc_sa = sa_binary(h_bin, rng, max_iters=40000, T0=0.05, cooling=0.99995, time_limit=25.0)
            if sc_sa < best_score:
                best_score = sc_sa
                best_h = make_feasible(h_sa)
                best_score = official_score(best_h)
            # Random binary start
            h_rand = rng.random(n_try)
            h_rand_bin = binarize_h(h_rand, rng)
            h_sa2, sc_sa2 = sa_binary(h_rand_bin, rng, max_iters=25000, T0=0.06, cooling=0.9999, time_limit=15.0)
            if sc_sa2 < best_score:
                best_score = sc_sa2
                best_h = make_feasible(h_sa2)
                best_score = official_score(best_h)
            # Structured start: block pattern
            n_b = n_try // 2
            h_block = np.zeros(n_try)
            h_block[:n_b] = 1.0
            h_block_bin = binarize_h(h_block, rng)
            h_sa3, sc_sa3 = sa_binary(h_block_bin, rng, max_iters=20000, T0=0.06, cooling=0.9999, time_limit=10.0)
            if sc_sa3 < best_score:
                best_score = sc_sa3
                best_h = make_feasible(h_sa3)
                best_score = official_score(best_h)

    # Phase 2: PG from best (with periodic soft-binarization kicks)
    if tleft() > 120:
        h_cur = best_h.copy()
        beta = 30.0
        stall = 0
        it_count = 0
        while tleft() > 80:
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.8, max_ls=16)
            it_count += 1
            if sc < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = sc
                stall = 0
                h_cur = h_new
            elif sc < official_score(h_cur) - 1e-12:
                stall = 0
                h_cur = h_new
            else:
                stall += 1
                h_cur = h_new
            beta = min(beta * 1.05, 600.0)
            if stall > 50:
                kick_type = rng.integers(3)
                if kick_type == 0:
                    h_cur = make_feasible(best_h + rng.normal(0, 0.05, len(best_h)))
                elif kick_type == 1:
                    h_cur = soft_binarize(best_h, strength=2.0 + rng.random() * 3.0)
                else:
                    h_bin = binarize_h(best_h, rng)
                    h_cur = make_feasible(h_bin + rng.normal(0, 0.02, len(best_h)))
                beta = 15.0
                stall = 0
            if it_count % 200 == 0:
                h_cur = soft_binarize(h_cur, strength=4.0)

    # Phase 3: Pair local search
    if tleft() > 80:
        h_pl, sc_pl = pair_local_search(best_h, rng, max_passes=30, time_limit=60.0)
        if sc_pl < best_score:
            best_h = h_pl
            best_score = sc_pl

    # Phase 4: LP polishing (multiple rounds with decreasing eps)
    if tleft() > 50:
        h_cur = best_h.copy()
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if tleft() < 30:
                break
            for _ in range(60):
                if tleft() < 30:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.15)
                if h_new is not None:
                    if sc < best_score - 1e-12:
                        best_h = h_new.copy()
                        best_score = sc
                        h_cur = h_new
                    else:
                        h_cur = h_new

    # Phase 5: Multi-n re-optimization
    if tleft() > 60:
        n_base = len(best_h)
        n_candidates = []
        for d in [2, 4, 6, 8]:
            if n_base - d >= 24:
                n_candidates.append(n_base - d)
            if n_base + d <= 120:
                n_candidates.append(n_base + d)
        for n_try in n_candidates:
            if tleft() < 35:
                break
            h_int = interpolate_to(best_h, n_try)
            h_cur = h_int.copy()
            beta = 40.0
            for _ in range(400):
                if tleft() < 35:
                    break
                h_new, sc = pg_step(h_cur, beta, alpha_max=0.5, max_ls=12)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_cur = h_new
                beta = min(beta * 1.02, 400.0)
            for eps in [0.005, 0.002, 0.001]:
                if tleft() < 35:
                    break
                for _ in range(30):
                    if tleft() < 35:
                        break
                    h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.1)
                    if h_new is not None:
                        if sc < best_score - 1e-12:
                            best_h = h_new.copy()
                            best_score = sc
                        h_cur = h_new

    # Phase 6: Final fine PG + LP
    if tleft() > 25:
        h_cur = best_h.copy()
        beta = 300.0
        for _ in range(500):
            if tleft() < 10:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.3, max_ls=10)
            if sc < best_score - 1e-13:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
        h_cur = best_h.copy()
        for eps in [0.001, 0.0005, 0.0002]:
            if tleft() < 10:
                break
            for _ in range(40):
                if tleft() < 10:
                    break
                h_new, sc = active_set_lp(h_cur, eps=eps, trust=0.05)
                if h_new is not None:
                    if sc < best_score - 1e-13:
                        best_h = h_new.copy()
                        best_score = sc
                    h_cur = h_new

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    return final_h, final_score, final_n
