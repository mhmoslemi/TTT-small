def binary_sa(h0, n_swaps=100000, t_start=1.0, t_end=0.01, rng=None):
    """Simulated annealing on binary sequence with swap moves."""
    n = len(h0)
    h = h0.copy()  # binary: 0s and 1s
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    current_score = official_score(h)
    best_score = current_score
    best_h = h.copy()
    
    for it in range(n_swaps):
        t = t_start * (t_end / t_start) ** (it / n_swaps)
        # Pick a random 1 and a random 0 and swap
        i1 = rng.choice(list(ones))
        i0 = rng.choice(list(zeros))
        
        # Compute delta score
        # We need to recompute the score after swap
        h[i1] = 0
        h[i0] = 1
        new_score = official_score(h)
        
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_score = new_score
            ones.remove(i1)
            ones.add(i0)
            zeros.remove(i0)
            zeros.add(i1)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i1] = 1
            h[i0] = 0
    
    return best_h, best_score

import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def minimize_max_linear_vec(a, b, lo, hi):
    """Minimize max_p (a[p] + b[p]*x) over x in [lo, hi]."""
    # Bisection on subgradient
    def subgrad(x):
        vals = a + b * x
        mv = vals.max()
        active = b[np.abs(vals - mv) < 1e-10]
        return active.min(), active.max()
    
    bmin, bmax = subgrad(lo)
    if bmin >= -1e-12:
        return lo
    bmin, bmax = subgrad(hi)
    if bmax <= 1e-12:
        return hi
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        bmin, bmax = subgrad(mid)
        if bmin >= -1e-12:
            hi = mid
        elif bmax <= 1e-12:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

def coord_optimize(h, i):
    """Find optimal value of h[i] given all other coordinates fixed."""
    n = len(h)
    a_val = np.zeros(2 * n - 1)
    b_val = np.zeros(2 * n - 1)
    
    for p in range(2 * n - 1):
        s = p - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        a = 0.0
        b = 0.0
        for j in range(j_lo, j_hi):
            if j == i:
                b += (1.0 - h[j + s])
            elif j + s == i:
                a += h[j]
                b -= h[j]
            else:
                a += h[j] * (1.0 - h[j + s])
        a_val[p] = a
        b_val[p] = b
    
    return minimize_max_linear_vec(a_val, b_val, 0.0, 1.0)

def coord_descent(h, n_sweeps=30, rng=None, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy()
    best_score = official_score(h)
    
    for sweep in range(n_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            x_new = coord_optimize(h, i)
            h[i] = x_new
        sc = official_score(h)
        if sc < best_score:
            best_score = sc
    
    return h, best_score

    # a_p = sum_{j in L_p, j≠i, j+s≠i} h[j]*(1-h[j+s]) + h[i-s]*I[i-s in L_p]
    # b_p = (1-h[i+s])*I[i in L_p] - h[i-s]*I[i-s in L_p]

    # For all p, compute the full sum first, then subtract the i-terms
    
    # Full correlate: g_p = sum_j h[j]*(1-h[j+s]) for valid j
    g = np.correlate(h, 1.0 - h, mode='full')
    
    # But I need a_p and b_p separately.
    # g_p = a_p + b_p * h[i]
    # So: a_p = g_p - b_p * h[i]
    # I need b_p.
    
    # b_p = (1-h[i+s])*I[i in L_p] - h[i-s]*I[i-s in L_p]
    # i in L_p means: 0 <= i < n and 0 <= i+s < n, i.e., -i <= s < n-i
    # i-s in L_p means: 0 <= i-s < n and 0 <= i < n, i.e., 0 <= i-s < n since 0<=i<n always
    #   which is s <= i < n+s, i.e., s <= i and i < n+s
    
    s_vals = np.arange(2*n-1) - (n-1)  # lag values for each p
    # i in L_p: -i <= s < n-i
    cond1 = (s_vals >= -i) & (s_vals < n - i)
    # i-s in L_p: 0 <= i-s < n (and 0<=i<n which is always true)
    # i-s >= 0 → s <= i; i-s < n → s > i-n
    cond2 = (s_vals <= i) & (s_vals > i - n)
    
    # For cond1: h[i+s] is valid, contribution to b is (1-h[i+s])
    # For cond2: h[i-s] is valid, contribution to b is -h[i-s]
    
    b_val = np.zeros(2*n-1)
    # For cond1 indices:
    idx1 = np.where(cond1)[0]
    for idx in idx1:
        s = s_vals[idx]
        b_val[idx] += (1.0 - h[i + s])
    # For cond2 indices:
    idx2 = np.where(cond2)[0]
    for idx in idx2:
        s = s_vals[idx]
        b_val[idx] -= h[i - s]
    
    a_val = g - b_val * h[i]
    
    return minimize_max_linear_vec(a_val, b_val, 0.0, 1.0)

        else:

    return (lo + hi) / 2.0

def coord_optimize(h, i):
    """Find optimal h[i] given all others fixed. Returns the optimal value."""
    n = len(h)
    g = np.correlate(h, 1.0 - h, mode='full')
    s_vals = np.arange(2 * n - 1) - (n - 1)
    
    # cond1: i is in L_p, i.e., 0<=i<n and 0<=i+s<n
    # Since 0<=i<n always, need 0<=i+s<n, i.e., -i<=s<n-i
    cond1 = (s_vals >= -i) & (s_vals < n - i)
    # cond2: i-s is in L_p, i.e., 0<=i-s<n and 0<=i<n
    # Since 0<=i<n always, need 0<=i-s<n, i.e., i-n<s<=i
    cond2 = (s_vals > i - n) & (s_vals <= i)
    
    b_val = np.zeros(2 * n - 1)
    idx1 = np.where(cond1)[0]
    if len(idx1) > 0:
        s1 = s_vals[idx1]
        b_val[idx1] += (1.0 - h[i + s1])
    idx2 = np.where(cond2)[0]
    if len(idx2) > 0:
        s2 = s_vals[idx2]
        b_val[idx2] -= h[i - s2]
    
    a_val = g - b_val * h[i]
    
    return minimize_max_linear_vec(a_val, b_val, 0.0, 1.0)

def coord_descent(h, n_sweeps=100, rng=None, deadline=None):
    """Run coordinate descent."""
    n = len(h)
    h = h.copy()
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            h[i] = coord_optimize(h, i)
