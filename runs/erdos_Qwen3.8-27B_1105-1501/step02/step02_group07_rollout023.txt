    n = len(h)
    n_total = 2 * n - 1
    grads = np.zeros((n_total, n))
    for idx in range(n_total):
        s = idx - (n - 1)
        L = n - abs(s)
        if L <= 0:
            continue
        if s >= 0:
            grads[idx, s:n] += (1.0 - h[0:L])
            grads[idx, 0:L] -= h[s:n]
        else:
            t = -s
            grads[idx, 0:L] += (1.0 - h[t:n])
            grads[idx, t:n] -= h[0:L]
    return grads

            grad[0:L] += w * (1.0 - h[t:n])

    for trust in trust_levels:
        if time.time() > t_limit:
            break
        for eps in eps_levels:
            if time.time() > t_limit:
                break
            iters = max(20, max_iter // len(trust_levels) // len(eps_levels))
            for _ in range(iters):
                if time.time() > t_limit:
                    break
                h_new, sc = active_set_lp(best_h, eps=eps, trust=trust)
                if h_new is not None:
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_h = h_new
                    else:
                        best_h = h_new
    return best_h, best_score

def slsqp_optimize(h_start, beta, max_iter=500):
    n = len(h_start)
    def obj(x):
        h = make_feasible(x)
        V = compute_values(h)
        Vm = V.max()
        sh = np.clip(beta * (V - Vm), -500.0, 0.0)
        return Vm + np.log(np.sum(np.exp(sh)) / len(V)) / beta
    def objg(x):
        h = make_feasible(x)
        V = compute_values(h)
        Vm = V.max()
        sh = np.clip(beta * (V - Vm), -500.0, 0.0)
        w = np.exp(sh)
        w /= w.sum()
        return gradient_weighted(h, w)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, jac=objg, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        h_opt = make_feasible(res.x)
        return h_opt, official_score(h_opt)
    except Exception:
        return h_start, official_score(h_start)

def optimize_at_n(h_start, n_target, t_limit):
    h = h_start.copy()
    if len(h) != n_target:
        h = interpolate_to(h, n_target)
    else:
        h = make_feasible(h)
    best_h = h.copy()
    best_score = official_score(h)

    # PG phase
    beta = 30.0
    pg_time = min(60.0, (t_limit - time.time()) * 0.3)
    pg_end = time.time() + pg_time
    while time.time() < pg_end:
        h, sc = pg_step(h, beta, alpha_max=0.5, max_ls=10)
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
        beta = min(beta * 1.03, 400.0)

    # SLSQP phase
    if time.time() < t_limit - 60:
        h_sl, sc_sl = slsqp_optimize(best_h, beta=100.0, max_iter=300)
        if sc_sl < best_score - 1e-12:
            best_h = h_sl
            best_score = sc_sl

    # LP polish phase
    lp_end = time.time() + max(10, (t_limit - time.time()) * 0.7)
    best_h, best_score = lp_polish(best_h, min(lp_end, t_limit))

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = 15.0
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
        n0 = len(h0)
    except (NameError, AttributeError):
        n0 = 60
        x = np.arange(n0) / n0
        h0 = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)

    h0 = make_feasible(h0)
    best_h = h0.copy()
    best_score = official_score(h0)

    # Phase 1: Optimize at original n
    t1 = time.time() + min(200.0, (deadline - t0) * 0.3)
    h_opt, sc_opt = optimize_at_n(h0, n0, t1)
    if sc_opt < best_score:
        best_h = h_opt
        best_score = sc_opt

    # Phase 2: Try different n values
    n_candidates = []
    for dn in range(-20, 21, 5):
        nt = n0 + dn
        if 40 <= nt <= 120 and nt != n0:
            n_candidates.append(nt)
    # Add some specific values
    for nt in [50, 55, 65, 70, 75, 80, 85, 90]:
        if nt not in n_candidates and 40 <= nt <= 120:
            n_candidates.append(nt)
    n_candidates.sort()

    time_per_n = max(15.0, (deadline - time.time() - 30) / max(len(n_candidates), 1))
    for nt in n_candidates:
        if time.time() > deadline - 30:
            break
        h_try = interpolate_to(best_h, nt)
        sc_try = official_score(h_try)
        t_end = time.time() + time_per_n
        h_opt, sc_opt = optimize_at_n(h_try, nt, t_end)
        if sc_opt < best_score - 1e-12:
            best_h = h_opt
            best_score = sc_opt

    # Phase 3: Re-optimize at best n with more time
    if time.time() < deadline - 30:
        best_n = len(best_h)
        t_end = time.time() + min(100.0, deadline - time.time() - 20)
        h_opt, sc_opt = optimize_at_n(best_h, best_n, t_end)
        if sc_opt < best_score - 1e-13:
            best_h = h_opt
            best_score = sc_opt

    # Phase 4: Final aggressive LP polish
    if time.time() < deadline - 10:
        t_end = time.time() + max(5.0, deadline - time.time() - 10)
        best_h, best_score = lp_polish(best_h, t_end, max_iter=500)

    # Phase 5: Tiny PG steps
    if time.time() < deadline - 5:
        h_cur = best_h.copy()
        beta = 300.0
        for _ in range(300):
            if time.time() > deadline - 5:
                break
            h_new, sc = pg_step(h_cur, beta, alpha_max=0.05, max_ls=6)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc
            h_cur = h_new
            beta = min(beta * 1.005, 800.0)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    final_bound = final_score + 1e-12

    return final_h, final_bound, final_n
