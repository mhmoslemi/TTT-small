    def obj_func(z):
        return z[-1]

    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(z):
        h = z[:n]
        t = z[-1]
        G = grad_all(h)
        vals = compute_vals(h)
        return t - vals[active_idx]

    def ineq_jac(z):
        h = z[:n]
        G = grad_all(h)
        J = np.zeros((M, n + 1))
        J[:, :n] = -G[active_idx, :]
        J[:, n] = 1.0
        return J

    def eq_fun(z):
        return np.sum(z[:n]) - n / 2.0

    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    c0 = official_score(h0) + 1e-6
    z0 = np.concatenate([h0, [c0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.1)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj_func, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': tol, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            sc = official_score(h_new)
            if np.isfinite(sc) and sc < c0:
                return h_new, sc
    except Exception:
        pass
    return None, None


def slsqp_active_refine(h0, max_rounds=5, max_iter=200):
    n = len(h0)
    h = h0.copy()
    sc = official_score(h)
    for rnd in range(max_rounds):
        vals = compute_vals(h)
        active = find_active(vals, frac=0.4, min_count=12)
        h_new, sc_new = slsqp_solve(h, active, max_iter=max_iter)
        if h_new is not None and sc_new < sc - 1e-12:
            h = h_new
            sc = sc_new
        else:
            break
        vals = compute_vals(h)
        violated = np.where(vals > sc * (1 + 1e-10))[0]
        if len(violated) == 0:
            break
        active = list(set(active + violated.tolist()))
        active = sorted(active)[:min(len(active), 80)]
        h_new, sc_new = slsqp_solve(h, active, max_iter=max_iter)
        if h_new is not None and sc_new < sc - 1e-12:
            h = h_new
            sc = sc_new
    return h, sc


def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)


def fourier_init(n, coeffs, phases):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for m, (a, ph) in enumerate(zip(coeffs, phases), start=1):
        h += a * np.cos(m * np.pi * x + ph)
    return make_feasible(h)


def symmetrize(h):
    h_sym = 1.0 - h[::-1].copy()
    h_avg = 0.5 * (h + h_sym)
    return make_feasible(h_avg)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    def time_left():
        return deadline - time.time()

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
        elif abs(sc - best_score) <= 1e-14 and best_h is not None and len(h) < len(best_h):
            best_h = h.copy()
            best_score = sc

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 66
        x = np.arange(n0)
        h_init = 0.5 + 0.2 * np.sin(np.pi * (np.arange(n0) + 0.5) * 2.0 / n0)
        best_h = make_feasible(h_init)
        best_score = official_score(best_h)

    update(best_h, best_score)

    # Phase 1: Projected gradient with annealed beta (using CORRECT gradient)
    if time_left() > 30:
        betas = [30, 80, 200, 500, 1000, 3000, 8000]
        cur = best_h.copy()
        for beta in betas:
            if time_left() < 5:
                break
            tl = min(time_left() * 0.25, 20.0)
            if tl < 3:
                break
            cur = pg_solve(cur, beta, time_limit=tl)
            sc = official_score(cur)
            if sc < best_score - 1e-14:
                update(cur, sc)

    # Phase 2: Symmetrize and optimize
    if time_left() > 30:
        h_sym = symmetrize(best_h)
        sc_sym = official_score(h_sym)
        if sc_sym < best_score - 1e-14:
            update(h_sym, sc_sym)
            cur_sym = h_sym
        else:
            cur_sym = h_sym
        for beta in [500, 2000, 5000]:
            if time_left() < 5:
                break
            tl = min(time_left() * 0.2, 15.0)
            if tl < 3:
                break
            cur_sym = pg_solve(cur_sym, beta, time_limit=tl)
            sc = official_score(cur_sym)
            if sc < best_score - 1e-14:
                update(cur_sym, sc)

    # Phase 3: SLSQP active set refinement
    if time_left() > 30:
        h_new, sc_new = slsqp_active_refine(best_h, max_rounds=4, max_iter=250)
        if sc_new < best_score - 1e-14:
            update(h_new, sc_new)

    # Phase 4: Multi-resolution search
    if time_left() > 60:
        n_base = len(best_h)
        n_opts = sorted(set([
            max(40, n_base - 8),
            max(44, n_base - 4),
            n_base,
            min(72, n_base + 4),
            min(76, n_base + 8),
        ]))
        for n_opt in n_opts:
            if time_left() < 8:
                break
            if n_opt == n_base:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            cur_int = h_int
            for beta in [300, 1500, 5000]:
                if time_left() < 5:
                    break
                tl = min(time_left() * 0.15, 12.0)
                if tl < 2:
                    break
                cur_int = pg_solve(cur_int, beta, time_limit=tl)
                sc = official_score(cur_int)
                if sc < best_score - 1e-14:
                    update(cur_int, sc)
            h_nr, sc_nr = slsqp_active_refine(cur_int, max_rounds=3, max_iter=150)
            if sc_nr < best_score - 1e-14:
                update(h_nr, sc_nr)

    # Phase 5: Fourier-based initializations
    if time_left() > 60:
        n_try = len(best_h)
        for trial in range(6):
            if time_left() < 8:
                break
            M = 4 + trial
            coeffs = rng.uniform(0.05, 0.25, size=M) * (1.0 / np.arange(1, M + 1))
            phases = rng.uniform(0, 2 * np.pi, size=M)
            h_f = fourier_init(n_try, coeffs, phases)
            h_f = pg_solve(h_f, 500, time_limit=min(time_left() * 0.1, 10.0))
            sc_f = official_score(h_f)
            if sc_f < best_score - 1e-14:
                update(h_f, sc_f)
                h_nr, sc_nr = slsqp_active_refine(h_f, max_rounds=2, max_iter=150)
                if sc_nr < best_score - 1e-14:
                    update(h_nr, sc_nr)

    # Phase 6: Random perturbations + local refine
    if time_left() > 40:
        n_cur = len(best_h)
        n_starts = max(3, min(10, int(time_left() / 15)))
        for i in range(n_starts):
            if time_left() < 5:
                break
            sigma = [0.02, 0.04, 0.06, 0.08, 0.1][i % 5]
            h_cand = make_feasible(best_h + rng.normal(0, sigma, size=n_cur))
            h_cand = pg_solve(h_cand, 1500, time_limit=min(time_left() * 0.12, 10.0))
            sc = official_score(h_cand)
            if sc < best_score - 1e-14:
                update(h_cand, sc)
                h_nr, sc_nr = slsqp_active_refine(h_cand, max_rounds=2, max_iter=150)
                if sc_nr < best_score - 1e-14:
                    update(h_nr, sc_nr)

    # Phase 7: Intensive SLSQP with expanding active set
    if time_left() > 30:
        vals = compute_vals(best_h)
        # Try different active set sizes
        for frac, min_c in [(0.2, 8), (0.35, 12), (0.5, 16), (0.7, 20)]:
            if time_left() < 8:
                break
            active = find_active(vals, frac=frac, min_count=min_c)
            h_new, sc_new = slsqp_solve(best_h, active, max_iter=300)
            if h_new is not None and sc_new < best_score - 1e-14:
                update(h_new, sc_new)

    # Phase 8: Final high-beta polish
    if time_left() > 20:
        for beta in [8000, 20000, 50000]:
            if time_left() < 5:
                break
            tl = min(time_left() * 0.3, 12.0)
            if tl < 2:
                break
            h_new = pg_solve(best_h, beta, time_limit=tl)
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                update(h_new, sc)

    # Phase 9: Final SLSQP polish with broad active set
    if time_left() > 15:
        vals_fin = compute_vals(best_h)
        active_fin = find_active(vals_fin, frac=0.6, min_count=20)
        h_final, sc_final = slsqp_solve(best_h, active_fin, max_iter=500)
        if h_final is not None and sc_final < best_score - 1e-14:
            update(h_final, sc_final)

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    assert np.all(final_h >= -1e-15), "Negative values in solution"
    assert np.all(final_h <= 1.0 + 1e-15), "Values exceed 1"
    assert abs(np.sum(final_h) - final_n / 2.0) < 1e-10, "Sum constraint violated"

    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
