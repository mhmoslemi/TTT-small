def swap_improvement(h, i, j, corr):
    """Compute the improvement in max correlation from swapping h[i] and h[j]."""
    n = len(h)
    dx = 2.0 / n
    a, b = h[i], h[j]
    if a == b:
        return 0, corr.copy()
    
    # For a swap: h[i] becomes b, h[j] becomes a
    # Δ_i = b - a, Δ_j = a - b = -Δ_i
    delta = b - a
    
    # Change in correlation at each lag
    # For position p with change Δp:
    # Δc_s = Δp * ((1-h[p-s]) - h[p+s]) [when valid]
    
    # Instead of computing all 2n-1 lags, let's just compute the new corr
    # and find the max
    
    # Actually, let's just recompute from scratch for simplicity
    # but use the formula to speed it up
    
    # New h after swap
    # h_new[i] = b, h_new[j] = a
    
    # Δc at lag s (where s = k - (n-1) for correlation index k):
    # For position i: Δ_i = b - a
    #   Contribution to c_s: Δ_i * ((1-h[i-s]) - h[i+s]) if valid
    # For position j: Δ_j = a - b = -Δ_i
    #   Contribution to c_s: -Δ_i * ((1-h[j-s]) - h[j+s]) if valid

    for s in range(-(n-1), n):

        # i+s valid: 0 <= i+s < n → -i <= s <= n-1-i
        valid_1 = (0 <= i-s < n)

        elif valid_1:
            new_corr[s + n - 1] += delta * (1 - h[i-s])
        elif valid_2:
            new_corr[s + n - 1] += delta * (-h[i+s])
