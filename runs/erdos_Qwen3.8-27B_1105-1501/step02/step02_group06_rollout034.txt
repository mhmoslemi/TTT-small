def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_shift(s, h):
    n = len(h)
    g = np.zeros(n)
    # First factor: h[j] where j+s is valid
    lo1 = max(0, -s)
    hi1 = min(n-1, n-1-s)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1+1)
        g[idx] += (1.0 - h[idx + s])
    # Second factor: h[j] where j-s is valid (i.e., h[i+s] with i=j-s)
    lo2 = max(0, s)
    hi2 = min(n-1, n-1+s)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2+1)
        g[idx] -= h[idx - s]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    # Bisection to find shift that makes sum correct
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:

                if sub <= 0:

def lp_minimax_step(h, k_active, radius):
    """Solve LP: min t s.t. linearized c_k(h') <= t, sum(h')=n/2, box + trust region."""
    n = len(h)
    c_vals = compute_c(h)
    # Select active constraints
    if k_active < len(c_vals):
        top_idx = np.argsort(c_vals)[-k_active:]
    else:
        top_idx = np.arange(len(c_vals))
    
    # Build LP
    # Variables: h'(n) and t(1)
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub_list = []
    b_ub_list = []
    for k in top_idx:
        s = k - (n - 1)
        g = grad_shift(s, h)
        # c[k] + g·(h' - h) <= t  =>  g·h' - t <= c[k] - g·h
        row = np.zeros(nvar)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(c_vals[k] - g @ h)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h[i] - radius)
        hi_i = min(1.0, h[i] + radius)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if result.status != 0:
        return None, None
    
    h_new = result.x[:n]
    t_new = result.x[-1]
    if not np.all(np.isfinite(h_new)):
        return None, None
    
    h_new = make_feasible(h_new)
    return h_new, float(t_new)

    lo1 = max(0, -s)

    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, s)
    hi2 = min(n - 1, n - 1 + s)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        g[idx] -= h[idx - s]
    return g

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx
] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def lp_minimax_step(h, k_active, radius):
    n = len(h)
    c_vals = compute_c(h)
    n_c = len(c_vals)
    if k_active < n_c:
        top_idx = np.argsort(c_vals)[-k_active:]
    else:
        top_idx = np.arange(n_c)
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in top_idx:
        s = k - (n - 1)
        g = grad_shift(s, h)
        row = np.zeros(nvar)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(c_vals[k] - g @ h)
    if not A_ub_list:
        return None, None
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h[i] - radius)
        hi_i = min(1.0, h[i] + radius)
        if lo_i >= hi_i:
            bounds.append((h[i], h[i]))
        else:
            bounds.append((lo_i, hi_i))
    bounds.append((0.0, None))
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None, None
    if result.status != 0 or result.x is None:
        return None, None
    h_new = result.x[:n]
    t_new = float(result.x[-1])
    if not np.all(np.isfinite(h_new)):
        return None, None
    h_new = make_feasible(h_new)
    return h_new, t_new

def lp_feasibility_step(h, k_active, radius, t_target):
    n = len(h)
    c_vals = compute_c(h)
    n_c = len(c_vals)
    if k_active < n_c:
        top_idx = np.argsort(c_vals)[-k_active:]
    else:
        top_idx = np.arange(n_c)
    c_obj = np.zeros(n)
    A_ub_list = []
    b_ub_list = []
    for k in top_idx:
        s = k - (n - 1)
        g = grad_shift(s, h)
        A_ub_list.append(g)
        b_ub_list.append(t_target - c_vals[k] + g @ h)
    if not A_ub_list:
        return None
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h[i] - radius)
        hi_i = min(1.0, h[i] + radius)
        bounds.append((lo_i, hi_i))
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None
    if result.status != 0 or result.x is None:
        return None
    h_new = result.x
    if not np.all(np.isfinite(h_new)):
        return None
    return make_feasible(h_new)

def line_search_full(h, direction, max_alpha, n_samples):
    n = len(h)
    alpha_max = max_alpha
    for i in range(n):
        if direction[i] > 1e-14:
            a = (1.0 - h[i]) / direction[i]
            if a < alpha_max:
                alpha_max = a
        elif direction[i] < -1e-14:
            a = h[i] / (-direction[i])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(0.0, alpha_max)
    if alpha_max < 1e-13:
        return h.copy(), official_score(h)
    base = np.linspace(0.1, 1.0, n_samples)
    alphas = base * alpha_max
    if alpha_max > 1.5:
        alphas = np.concatenate([alphas, [alpha_max * 0.5, alpha_max * 0.8, alpha_max]])
    best_h = h.copy()
    best_sc = official_score(h)
    for alpha in alphas:
        if alpha <= 0 or alpha > alpha_max + 1e-10:
            continue
        h_try = make_feasible(h + alpha * direction)
        sc = official_score(h_try)
        if sc < best_sc:
            best_sc = sc
            best_h = h_try.copy()
    return best_h, best_sc

def lp_minimax_optimize(h0, max_iter, k_active, radius_init, radius_min, radius_max,
                        line_samples, time_limit):
    t0 = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_score = official_score(h)
    radius = radius_init
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        h_new, t_new = lp_minimax_step(h, k_active=k_active, radius=radius)
        if h_new is None:
            radius *= 0.5
            if radius < radius_min:
                break
            stall += 1
            if stall > 12:
                break
            continue
        direction = h_new - h
        dir_norm = np.max(np.abs(direction))
        if dir_norm < 1e-12:
            stall += 1
            radius *= 0.7
            if stall > 10:
                break
            continue
        max_alpha = min(1.0, max(radius / dir_norm, 0.01))
        h_ls, sc_ls = line_search_full(h, direction, max_alpha, line_samples)
        if sc_ls < best_score - 1e-13:
            h = h_ls.copy()
            best_h = h_ls.copy()
            best_score = sc_ls
            stall = 0
            radius = min(radius * 1.25, radius_max)
        else:
            stall += 1
            radius *= 0.6
            if radius < radius_min:
                break
            if stall > 15:
                break
    return best_h, best_score

def binary_push(h, strength):
    n = len(h)
    zb = np.where(h >= 0.5, 1.0, 0.0)
    h_pushed = h * (1.0 - strength) + zb * strength
    return make_feasible(h_pushed)

def frequency_push(h, freq, amp, phase):
    n = len(h)
    x = (np.arange(n) + 0.5) * 2.0 / n
    pert = amp * np.sin(2.0 * np.pi * freq * x + phase)
    return make_feasible(h + pert)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 12.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    best_h = None
    best_score = 1e10

    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size > 2 and np.all(np.isfinite(parent)):
            parent = make_feasible(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
    except (NameError, AttributeError):
        pass

    if best_h is None:
        n0 = 60
        x = (np.arange(n0) + 0.5) * 2.0 / n0
        best_h = make_feasible(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
        best_score = official_score(best_h)

    # Phase 1: Main LP minimax - large k_active for global view
    if remaining() > 30:
        h_opt, sc_opt = lp_minimax_optimize(
            best_h, max_iter=40, k_active=25, radius_init=0.2,
            radius_min=0.005, radius_max=0.6, line_samples=10,
            time_limit=min(remaining() * 0.3, 200))
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    # Phase 2: Medium k_active refinement
    if remaining() > 30:
        h_opt, sc_opt = lp_minimax_optimize(
            best_h, max_iter=35, k_active=15, radius_init=0.1,
            radius_min=0.003, radius_max=0.4, line_samples=10,
            time_limit=min(remaining() * 0.25, 150))
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    # Phase 3: Binary push + re-optimize
    if remaining() > 40:
        n = len(best_h)
        for strength in [0.3, 0.5, 0.7, 0.9]:
            if remaining() < 20:
                break
            h_b = binary_push(best_h, strength)
            h_opt, sc_opt = lp_minimax_optimize(
                h_b, max_iter=15, k_active=20, radius_init=0.15,
                radius_min=0.005, radius_max=0.4, line_samples=8,
                time_limit=min(remaining() * 0.15, 40))
            if sc_opt < best_score:
                best_h = h_opt
                best_score = sc_opt

    # Phase 4: Random perturbations + local optimization
    if remaining() > 60:
        n = len(best_h)
        for i in range(15):
            if remaining() < 25:
                break
            if i % 3 == 0:
                h_cand = make_feasible(best_h + rng.normal(0, 0.04, size=n))
            elif i % 3 == 1:
                h_cand = make_feasible(best_h + rng.normal(0, 0.12, size=n))
            else:
                h_cand = make_feasible(1.0 - best_h + rng.normal(0, 0.03, size=n))
            h_opt, sc_opt = lp_minimax_optimize(
                h_cand, max_iter=12, k_active=15, radius_init=0.08,
                radius_min=0.003, radius_max=0.25, line_samples=6,
                time_limit=min(remaining() * 0.08, 25))
            if sc_opt < best_score:
                best_h = h_opt
                best_score = sc_opt

    # Phase 5: Frequency perturbation + optimize
    if remaining() > 50:
        n = len(best_h)
        for freq in [1, 2, 3, 5]:
            if remaining() < 20:
                break
            for phase in [0.0, np.pi / 2, np.pi]:
                if remaining() < 10:
                    break
                h_cand = frequency_push(best_h, freq, 0.08, phase)
                h_opt, sc_opt = lp_minimax_optimize(
                    h_cand, max_iter=10, k_active=15, radius_init=0.06,
                    radius_min=0.003, radius_max=0.2, line_samples=6,
                    time_limit=min(remaining() * 0.06, 15))
                if sc_opt < best_score:
                    best_h = h_opt
                    best_score = sc_opt

    # Phase 6: Resolution exploration
    if remaining() > 60:
        n_base = len(best_h)
        n_opts = sorted(set([max(24, n_base - 4), n_base - 2, n_base,
                             n_base + 2, n_base + 4, n_base + 8]))
        for n_opt in n_opts:
            if remaining() < 20:
                break
            if n_opt < 20 or n_opt > 120:
                continue
            h_int = interpolate_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score:
                best_h = h_int
                best_score = sc_int
            h_opt, sc_opt = lp_minimax_optimize(
                h_int, max_iter=15, k_active=min(20, 2 * n_opt // 3),
                radius_init=0.1, radius_min=0.003, radius_max=0.3,
                line_samples=8, time_limit=min(remaining() * 0.12, 30))
            if sc_opt < best_score:
                best_h = h_opt
                best_score = sc_opt

    # Phase 7: Small trust region polishing
    if remaining() > 30:
        h_opt, sc_opt = lp_minimax_optimize(
            best_h, max_iter=40, k_active=30, radius_init=0.03,
            radius_min=0.001, radius_max=0.1, line_samples=12,
            time_limit=min(remaining() * 0.3, 80))
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    # Phase 8: Very fine polish with high k_active
    if remaining() > 20:
        n = len(best_h)
        k_big = min(2 * n - 1, 40)
        h_opt, sc_opt = lp_minimax_optimize(
            best_h, max_iter=30, k_active=k_big, radius_init=0.02,
            radius_min=0.0005, radius_max=0.05, line_samples=14,
            time_limit=min(remaining() * 0.3, 60))
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    # Phase 9: Final tiny steps
    if remaining() > 15:
        h_opt, sc_opt = lp_minimax_optimize(
            best_h, max_iter=25, k_active=20, radius_init=0.01,
            radius_min=0.0002, radius_max=0.02, line_samples=15,
            time_limit=min(remaining() * 0.4, 30))
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    # Final validation
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    if not np.isfinite(final_score) or final_h is None:
        final_h = make_feasible(best_h)
        final_score = official_score(final_h)
    return final_h, final_score, len(final_h)
