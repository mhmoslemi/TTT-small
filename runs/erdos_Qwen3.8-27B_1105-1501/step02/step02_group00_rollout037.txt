import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        k = n - 1 + d
        g = abs(d)
        if g == 0:
            G[k, :] = 1.0 - 2.0 * h
        else:
            if d > 0:
                G[k, d:] += (1.0 - h[:n - d])
                G[k, :n - d] -= h[d:]
            else:
                G[k, :n - g] += (1.0 - h[g:])
                G[k, g:] -= h[:n - g]
    return G

def _pair_find_delta(scores, dir_v, Q, d_lo, d_hi):
    """Find delta minimizing max(scores + delta*dir_v + delta^2*Q) on [d_lo, d_hi]."""
    def f(d):
        vals = scores + d * dir_v
        if Q.any():
            vals = vals.copy()
            vals[Q] += d * d
        return float(np.max(vals))
    
    base_val = f(0.0)
    # Coarse grid
    ngrid = 17
    grid = np.linspace(d_lo, d_hi, ngrid)
    vals = np.array([f(d) for d in grid])
    best_idx = int(np.argmin(vals))
    best_d = float(grid[best_idx])
    best_f = float(vals[best_idx])
    
    if best_idx == 0 or best_idx == ngrid - 1:
        return best_d, base_val - best_f, best_f
    
    # Golden section refinement
    lo = grid[best_idx - 1]
    hi = grid[best_idx + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    best_d = (lo + hi) / 2.0
    best_f = f(best_d)
    return best_d, base_val - best_f, best_f

def _pair_descent_pass(h, deadline, max_passes=20):
    n = len(h)
    target = n / 2.0
    h_work = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h_work)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        scores = _scores(h_work)
        G = _grad_matrix(h_work)
        n_improve = 0
        pairs = list(zip(range(n), range(n)))
        np.random.shuffle(pairs)
        for j, k in pairs:
            if j == k or time.time() > deadline:
                continue
            if h_work[j] < 1e-12 or h_work[k] < 1e-12:
                continue
            if h_work[j] > 1.0 - 1e-12 or h_work[k] > 1.0 - 1e-12:
                if h_work[j] > 1.0 - 1e-12 and h_work[k] > 1.0 - 1e-12:
                    continue
            d_lo = max(-h_work[j], h_work[k] - 1.0)
            d_hi = min(1.0 - h_work[j], h_work[k])
            if d_hi - d_lo < 1e-12:
                continue
            
            dir_v = G[:, j] - G[:, k]
            # Quadratic correction
            Q = np.zeros(2 * n - 1)
            g = abs(j - k)
            if g > 0 and g < n:
                if j > k:
                    # h[j] and h[k] paired in lag g (index n-1+g)
                    if j >= g:
                        Q[n - 1 + g] = 1.0
                    # h[k] and h[j] paired in lag -g (index n-1-g)
                    if k + g <= n - 1:
                        Q[n - 1 - g] = 1.0
                else:
                    # k > j, g = k-j
                    if k >= g:
                        Q[n - 1 + g] = 1.0
                    if j + g <= n - 1:
                        Q[n - 1 - g] = 1.0
            
            best_d, improvement, new_f = _pair_find_delta(scores, dir_v, Q, d_lo, d_hi)
            if improvement > 1e-10:
                h_work[j] = np.clip(h_work[j] + best_d, 0.0, 1.0)
                h_work[k] = np.clip(h_work[k] - best_d, 0.0, 1.0)
                cur = _c5(h_work)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h_work.copy()
                n_improve += 1
        
        if n_improve == 0:
            break
    
    if best_c5 < _c5(h):
        return best_h, best_c5
    return h, _c5(h)

def _smooth_gradient(h, beta):
    n = len(h)
    sc = _scores(h)
    sc_max = np.max(sc)
    exp_s = np.exp(beta * (sc - sc_max))
    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    grad = G.T @ w
    return grad

def _smooth_optimize(h, deadline, beta_start=5.0, beta_end=50.0, lr=0.05):
    n = len(h)
    target = n / 2.0
    h_work = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h_work)
    momentum = np.zeros(n)
    mu = 0.9
    
    n_steps = 200
    t_start = time.time()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        grad = _smooth_gradient(h_work, beta)
        momentum = mu * momentum + (1 - mu) * grad
        # Step
        h_new = h_work - lr * momentum
        h_new = _project(h_new, target)
        cur = _c5(h_new)
        if cur < best_c5:
            best_c5 = cur
            best_h = h_new.copy()
            h_work = h_new
        else:
            h_work = h_new  # still move, but don't update best
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_and_optimize(h, rng, scale, deadline, max_passes):
    n = len(h)
    target = n / 2.0
    h_pert = h + rng.normal(0, scale, n)
    h_pert = _project(h_pert, target)
    h_opt, c5_opt = _pair_descent_pass(h_pert, deadline, max_passes)
    return h_opt, c5_opt
