def lp_minimax(h0, gap_thresh=0.05, radius=0.3):
    """Solve LP: minimize t s.t. linearized active constraints <= t, box, sum, trust region."""
    n = len(h0)
    vals = compute_f(h0)
    fmax = vals.max()
    
    # Select constraints: all within gap_thresh of max, plus the max itself
    threshold = fmax - gap_thresh
    active = np.where(vals >= threshold)[0]
    
    # Build LP: minimize t
    # Variables: h[0..n-1], t (index n)
    # Constraints:
    #   For each active j: grad_j dot h - t <= grad_j dot h0 - vals[j]
    #   sum(h) = n/2
    #   0 <= h_i <= 1
    #   |h_i - h0_i| <= radius (trust region)
    
    # Objective: minimize t → c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active j: sum_i grad_j[i] * h_i - t <= grad_j dot h0 - vals[j]
    A_list = []
    b_list = []
    for j in active:
        s = j - (n - 1)
        g = grad_shift(h0, s)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_list.append(row)
        b_list.append(np.dot(g, h0) - vals[j])
    
    if A_list:
        A_ub = np.array(A_list)
        b_ub = np.array(b_list)
    else:
        A_ub = np.array([[]]).reshape(0, n+1)
        b_ub = np.array([])
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    bounds = [(h0[i] - radius, h0[i] + radius) for i in range(n)]
    bounds.append((0.0, None))  # t >= 0, unbounded above
    
    # Clip bounds to [0,1] for h
    bounds = [(max(0, lb), min(1, ub)) for lb, ub in bounds[:n]]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = make_feasible(res.x[:n])
            return h_new
    except Exception:
        pass
    return None
