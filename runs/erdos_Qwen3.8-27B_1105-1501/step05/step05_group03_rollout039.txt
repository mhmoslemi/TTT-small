```python
def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        else:
```

```python
import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _proj(v, target):
    n = len(v)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    r = target - x.sum()
    if abs(r) > 1e-14:
        m = (x < 0.999) if r > 0 else (x > 0.001)
        c = m.sum()
        if c:
            x[m] += r / c
            x = np.clip(x, 0, 1)
    return x

def _grad_w(h, w):
    """Gradient of sum(w_r * s_r) w.r.t. h, where s_r are the scores."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        wd = w[r] * dx
        if wd < 1e-16:
            continue
        if d >= 0:
            g[d:] += (1.0 - h[:n-d]) * wd
            g[:n-d] -= h[d:] * wd
        else:
            gg = -d
            g[:n-gg] += (1.0 - h[gg:]) * wd
            g[gg:] -= h[:n-gg] * wd
    return g
```

```python
import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _proj(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    r = target - float(np.sum(x))
    if abs(r) > 1e-13:
        if r > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += r / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_w(h, w):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    g = np.zeros(n)
    w0 = w[n - 1] * dx
    if w0 > 1e-16:
        g += (omh - h) * w0
    for d in range(1, n):
        wp = w[n - 1 + d] * dx
        wm = w[n - 1 - d] * dx
        if wp > 1e-16:
            g[d:] += omh[:n - d] * wp
            g[:n - d] -= h[d:] * wp
        if wm > 1e-16:
            g[:n - d] += omh[d:] * wm
            g[d:] -= h[:n - d] * wm
    return g

def _softmax_grad(h_init, deadline, beta0=15.0, beta1=400.0, n_steps=10000,
                  step_init=0.03, mu=0.85, max_stall=200):
    n = len(h_init)
    target = n / 2.0
    h = _proj(h_init.copy(), target)
    v = np.zeros(n)
    cur = _c5(h)
    best_h = h.copy()
    best_c5 = cur
    step = step_init
    stall = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(n_steps - 1, 1))
        beta = beta0 * (beta1 / beta0) ** frac
        sc = _scores(h)
        mx = sc.max()
        e = np.exp(beta * (sc - mx))
        w = e / e.sum()
        grad = _grad_w(h, w)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = mu * v - step * grad
        h_new = _proj(h + v, target)
        c5_new = _c5(h_new)
        if c5_new < cur - 1e-14:
            h = h_new
            cur = c5_new
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            step = min(step * 1.01, 0.08)
        else:
            stall += 1
            v *= 0.5
            step *= 0.92
            if stall > max_stall:
                break
            if step < 1e-6:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return _proj(h.copy(), n_new / 2.0)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _golden_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _proj(h.astype(float), n / 2.0)

def _random_structured(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    # Create blocks of random sizes
    pos = 0
    remaining = n_ones
    while pos < n and remaining > 0:
        gap = rng.integers(0, 4)
        pos += gap
        if pos >= n:
            break
        blen = min(rng.integers(1, 6), n - pos, remaining)
        h[pos:pos + blen] = 1.0
        pos += blen
        remaining -= blen
    return _proj(h, n / 2.0)

def _fourier_init(n, rng, K=6):
    x = np.arange(n) / n
    h = np.zeros(n)
    for k in range(1, K + 1):
        amp = rng.uniform(0.1, 0.4) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x + phase)
    h = np.clip(0.5 + h, 0, 1)
    return _proj(h, n / 2.0)

def _pair_transfer_polish(h, deadline, n_iters=5000, rng=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
```
