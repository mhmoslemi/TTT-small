Let me think about the logistic parameterization more carefully:
- h_i = sigma(q_i - mu) where mu is chosen so sum(h) = n/2
- The gradient with respect to q_i accounts for the change in mu
- This keeps h strictly in (0,1) so we don't need box constraints
- We only need bounds on q (e.g., [-14, 14])

For the log-sum-exp smoothing:
- F_beta = M + (1/beta) * log(mean(exp(beta * (S - M))))
- where M = max(S), S = scores
- Gradient: G^T @ w where w = softmax(beta * (S - M))

This is a smooth approximation to max(S), and as beta -> infinity, it approaches the true max.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _balance_exact(x, target):
    """Project x onto {z: 0<=z<=1, sum(z)=target}"""
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _safe_sigma(t):
    """Numerically stable sigmoid"""
    out = np.zeros_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def _solve_mu(q, target):
    """Find mu such that sum(sigma(q - mu)) = target"""
    n = len(q)
    def F(mu):
        return float(np.sum(_safe_sigma(q - mu))) - target
    lo = float(q.min()) - 20.0
    hi = float(q.max()) + 20.0
    # F(lo) should be > 0, F(hi) should be < 0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if F(mid) > 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target):
    """Map q-space to feasible h-space"""
    mu = _solve_mu(q, target)
    h = _safe_sigma(q - mu)
    # Exact balance correction
    h = _balance_exact(h, target)
    return h

def _logit_from_h(h, eps=1e-6):
    """Convert h to logit coordinates"""
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def _grad_matrix(h):
    """Build gradient matrix G where row r corresponds to lag d = r-(n-1)"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _smooth_obj_grad(q, beta, target):
    """Compute smooth objective and gradient in q-space"""
    h = _q_to_h(q, target)
    S = _scores(h)
    M = np.max(S)
    shifted = np.clip(beta * (S - M), -500, 0)
    ex = np.exp(shifted)
    sw = float(np.sum(ex))
    if sw < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w = ex / sw
    obj = M + np.log(float(np.mean(ex))) / beta
    # Raw gradient
    G = _grad_matrix(h)
    g_h = G.T @ w
    # Chain rule through q -> h mapping
    # h_i = sigma(q_i - mu), dh_i/dq_j = s_i * (delta_ij - dmu/dq_j)
    # dmu/dq_j = s_j / sum(s) where s_i = sigma(q_i-mu)*(1-sigma(q_i-mu))
    mu = _solve_mu(q, target)
    s = _safe_sigma(q - mu) * (1.0 - _safe_sigma(q - mu))
    s_sum = float(np.sum(s))
    if s_sum < 1e-30:
        return obj, np.zeros_like(q)
    rho = float(np.sum(s * g_h)) / s_sum
    g_q = s * (g_h - rho)
    if not np.all(np.isfinite(g_q)):
        g_q = np.zeros_like(q)
    return obj, g_q

def _bfgs_smooth(h_init, deadline, beta_schedule, q_wall=14.0):
    """Run L-BFGS-B in q-space with log-sum-exp smoothing"""
    n = len(h_init)
    target = n / 2.0
    q = _logit_from_h(h_init)
    q = np.clip(q, -q_wall, q_wall)
    best_h = _q_to_h(q, target)
    best_c5 = _exact_c5(best_h)
    
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        def obj_fn(qv):
            o, _ = _smooth_obj_grad(qv, beta, target)
            return o if np.isfinite(o) else 1e10
        def grad_fn(qv):
            _, g = _smooth_obj_grad(qv, beta, target)
            return g
        try:
            n_max = max(50, int(300 * np.sqrt(128.0 / n)))
            res = minimize(obj_fn, q, jac=grad_fn, method='L-BFGS-B',
                          bounds=[(-q_wall, q_wall)] * n,
                          options={'maxiter': n_max, 'ftol': 1e-14, 'gtol': 1e-12})
            if np.all(np.isfinite(res.x)):
                q_new = np.clip(res.x, -q_wall, q_wall)
                h_new = _q_to_h(q_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                q = q_new
        except Exception:
            pass
    return best_h, best_c5

def _complement_lp(h_current, deadline):
    """Solve complement LP to find a potentially better profile"""
    n = len(h_current)
    target = n / 2.0
    dx = 2.0 / n
    # Build B matrix: B[r, j] = sum over pairs (i, j=i+d) of h_current[i] * dx
    # For each lag d, score_d = sum_i h_i * (1 - h_{i+d}) * dx
    # In terms of y = 1-h: score_d = sum_i h_i * y_{i+d} * dx
    # So B[r, j] accumulates h_i * dx for each pair where j = i+d (or i = j-d)
    B = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            B[r, d:] += h_current[:m] * dx
        else:
            g = -d
            m = n - g
            B[r, :m] += h_current[g:] * dx
    # LP: min tau s.t. B@y <= tau, sum(y) = target, 0 <= y <= 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = B
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(-1.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            y = np.clip(res.x[:n], 0.0, 1.0)
            h_cand = 1.0 - y
            h_cand = _balance_exact(h_cand, target)
            c5_cand = _exact_c5(h_cand)
            return h_cand, c5_cand
    except Exception:
        pass
    return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _balance_exact(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _balance_exact(h, n / 2.0)

                        h1[k] -= sign * m1

                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2

                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0

                        best_h = h_try.copy()
                        h = h_try.copy()

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _balance_exact(x, target):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _safe_sigma(t):
    out = np.zeros_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def _solve_mu(q, target):
    n = len(q)
    lo = float(q.min()) - 15.0
    hi = float(q.max()) + 15.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_safe_sigma(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target):
    mu = _solve_mu(q, target)
    h = _safe_sigma(q - mu)
    return _balance_exact(h, target)

def _logit_from_h(h, eps=1e-6):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _smooth_obj_grad(q, beta, target):
    h = _q_to_h(q, target)
    S = _scores(h)
    M = float(np.max(S))
    shifted = np.clip(beta * (S - M), -500.0, 0.0)
    ex = np.exp(shifted)
    sw = float(np.sum(ex))
    if sw < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w = ex / sw
    obj = M + np.log(float(np.mean(ex))) / beta
    G = _grad_matrix(h)
    g_h = G.T @ w
    mu = _solve_mu(q, target)
    sig = _safe_sigma(q - mu)
    s = sig * (1.0 - sig)
    s_sum = float(np.sum(s))
    if s_sum < 1e-30:
        return obj, np.zeros_like(q)
    rho = float(np.sum(s * g_h)) / s_sum
    g_q = s * (g_h - rho)
    if not np.all(np.isfinite(g_q)):

    return obj, g_q

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _balance_exact(h_new, n_new / 2.0)

            ht, ct = _lp_local(ht, min(dl6, time.time() + budget_s * 0.02),
                               radius=0.1, max_iters=800, stall_limit=60)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try
            if time.time() < dl6:
                alpha = alphas[n_try % len(alphas)]
                h_qp = _quasi_init(n_try, alpha)
                h_qp += rng.normal(0, 0.06, n_try)
                h_qp = _balance_exact(h_qp, n_try / 2.0)
                sub_dl2 = min(dl6, time.time() + budget_s * 0.025)
                beta_s2 = np.logspace(np.log10(max(12, Lt // 5)), np.log10(min(500, 3 * Lt)), 4)
                ht2, ct2 = _bfgs_smooth(h_qp, sub_dl2, beta_s2)
                ht2, ct2 = _lp_local(ht2, min(dl6, time.time() + budget_s * 0.015),
                                     radius=0.1, max_iters=600, stall_limit=50)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = n_try

    # Phase 7: Perturbation restarts
    if time.time() < t0 + budget_s * 0.90:
        dl7 = t0 + budget_s * 0.90
        scale = 0.012
        for attempt in range(400):
            if time.time() > dl7:
                break
            h_p = best_h + rng.normal(0, scale, best_n)
            h_p = _balance_exact(h_p, best_n / 2.0)
            sub_dl = min(dl7, time.time() + budget_s * 0.006)
            ht, ct = _lp_local(h_p, sub_dl, radius=0.04, max_iters=200, stall_limit=25)
            if ct < best_c5 - 1e-13:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.002, scale * 0.85)
            else:
                scale = min(0.05, scale * 1.06)

    # Phase 8: Final BFGS + LP
    if time.time() < t0 + budget_s * 0.95:
        dl8 = t0 + budget_s * 0.95
        Lb = 2 * best_n - 1
        beta_f = np.logspace(np.log10(max(50, Lb // 2)), np.log10(min(800, 5 * Lb)), 4)
        h8, c5_8 = _bfgs_smooth(best_h, dl8, beta_f)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
        if time.time() < dl8:
            h8b, c5_8b = _lp_local(best_h, t0 + budget_s * 0.96,
                                    radius=0.03, max_iters=1000, stall_limit=60)
            if c5_8b < best_c5:
                best_c5 = c5_8b
                best_h = h8b.copy()

    # Phase 9: Pair polish (final refinement)
    if time.time() < t0 + budget_s * 0.975:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.98, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _balance_exact(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
