                    i = idx

                    j = idx

        if i == j:
            continue
        t = rng.random()
        h_try = h.copy()
        h_try[i] += t * (h[j] - h[i])
        h_try[j] -= t * (h[j] - h[i])
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    return best_h, best_c5

def _block_swap_polish(h, deadline, rng, n_trials=2000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    block_sizes = [2, 3, 4, 5, 6, 8, 10, 12, 16]
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        bs = block_sizes[rng.integers(0, len(block_sizes))]
        i = rng.integers(0, max(1, n - bs))
        j = rng.integers(0, max(1, n - bs))
        if abs(i - j) <= bs:
            continue
        h_try = h.copy()
        tmp = h_try[i:i+bs].copy()
        h_try[i:i+bs] = h_try[j:j+bs]
        h_try[j:j+bs] = tmp
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    return best_h, best_c5

def _coordinate_polish(h, deadline, n_cycles=2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v = 0.0
            hi_v = 1.0
            for _ in range(30):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy(); h1[j] = m1; h1 = _project(h1, target)
                h2 = h.copy(); h2[j] = m2; h2 = _project(h2, target)
                f1 = _exact_c5(h1); f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy(); h_try[j] = v_opt; h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _active_lag_polish(h, deadline, max_rounds=300):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        scores = _scores(h)
        c5 = float(np.max(scores))
        threshold = c5 - 0.003
        active = np.where(scores >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        improved = False
        for lag_idx in active[:8]:
            if time.time() > deadline:
                break
            d = int(lag_idx) - (n - 1)
            if d == 0:
                continue
            if d > 0:
                m = n - d
                contrib = h[:m] * (1.0 - h[d:])
            else:
                g = -d
                m = n - g
                contrib = h[g:] * (1.0 - h[:m])
            order = np.argsort(-contrib)[:6]
            for idx in order:
                if d > 0:
                    i = int(idx); j = i + d
                else:
                    j = int(idx); i = j + (-d)
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-8:
                    continue
                for frac in [0.15, 0.4, 0.7]:
                    delta = delta_max * frac
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved and rnd > 20:
            break
    return best_h, best_c5

def _perturbation_search(h_best, c5_best, n_pts, deadline, rng, n_attempts=200):
    h = h_best.copy()
    best_h = h.copy()
    best_c5 = c5_best
    scale = 0.02
    for att in range(n_attempts):
        if time.time() > deadline:
            break
        h_pert = h + rng.normal(0, scale, n_pts)
        h_pert = _project(h_pert, n_pts / 2.0)
        sub_dl = min(deadline, time.time() + 8.0)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.06, max_iters=400, radius_min=0.005)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.08, scale * 1.12)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Quick LP from parent (10%)
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.15, max_iters=3000, radius_min=0.002)
    if c5_1 < best_c5:
        best_c5 = c5_1; best_h = h1.copy()

    # Phase 2: Quasi-periodic optimization at multiple resolutions (20%)
    dl2 = t0 + budget_s * 0.30
    qp_resolutions = [64, 80, 96, 112, 128, 144, 160, 192]
    for n_try in qp_resolutions:
        if time.time() > dl2:
            break
        dl_sub = min(dl2, time.time() + budget_s * 0.025)
        h_q, c5_q = _qp_optimize(n_try, dl_sub)
        if h_q is not None and c5_q < best_c5:
            best_c5 = c5_q; best_h = h_q.copy(); best_n = n_try

    # Phase 3: Two-frequency quasi-periodic (10%)
    dl3 = t0 + budget_s * 0.40
    for n_try in [n0, 96, 128, 160]:
        if time.time() > dl3:
            break
        dl_sub = min(dl3, time.time() + budget_s * 0.02)
        h_q2, c5_q2 = _qp2_optimize(n_try, dl_sub)
        if h_q2 is not None and c5_q2 < best_c5:
            best_c5 = c5_q2; best_h = h_q2.copy(); best_n = n_try

    # Phase 4: LP from best so far (10%)
    dl4 = t0 + budget_s * 0.50
    h4, c5_4 = _lp_optimize(best_h, dl4, radius_init=0.12, max_iters=3000, radius_min=0.002)
    if c5_4 < best_c5:
        best_c5 = c5_4; best_h = h4.copy()

    # Phase 5: Smoothed gradient (10%)
    dl5 = t0 + budget_s * 0.60
    h5, c5_5 = _smoothed_grad(best_h, dl5, beta_start=15, beta_end=500, n_steps=300, step_init=0.04)
    if c5_5 < best_c5:
        best_c5 = c5_5; best_h = h5.copy()
    if time.time() < t0 + budget_s * 0.65:
        dl5b = t0 + budget_s * 0.65
        h5b, c5_5b = _lp_optimize(best_h, dl5b, radius_init=0.08, max_iters=2000, radius_min=0.002)
        if c5_5b < best_c5:
            best_c5 = c5_5b; best_h = h5b.copy()

    # Phase 6: Multi-resolution fine tuning (10%)
    dl6 = t0 + budget_s * 0.75
    for n_try in [best_n - 8, best_n, best_n + 8, best_n + 16]:
        if time.time() > dl6 or n_try < 32:
            continue
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        dl_sub = min(dl6, time.time() + budget_s * 0.02)
        h_r2, c5_r2 = _lp_optimize(h_r, dl_sub, radius_init=0.10, max_iters=1500, radius_min=0.003)
        if c5_r2 < best_c5:
            best_c5 = c5_r2; best_h = h_r2.copy(); best_n = n_try

    # Phase 7: Active lag polish + swap polish (10%)
    dl7 = t0 + budget_s * 0.85
    h7, c5_7 = _active_lag_polish(best_h, min(dl7, time.time() + budget_s * 0.04), max_rounds=200)
    if c5_7 < best_c5:
        best_c5 = c5_7; best_h = h7.copy()
    if time.time() < dl7:
        h7b, c5_7b = _swap_polish(best_h, min(dl7, time.time() + budget_s * 0.03), rng, n_trials=4000)
        if c5_7b < best_c5:
            best_c5 = c5_7b; best_h = h7b.copy()
    if time.time() < dl7:
        h7c, c5_7c = _block_swap_polish(best_h, min(dl7, time.time() + budget_s * 0.02), rng, n_trials=1500)
        if c5_7c < best_c5:
            best_c5 = c5_7c; best_h = h7c.copy()

    # Phase 8: Perturbation restarts (10%)
    dl8 = t0 + budget_s * 0.95
    h8, c5_8 = _perturbation_search(best_h, best_c5, best_n, dl8, rng, n_attempts=150)
    if c5_8 < best_c5:
        best_c5 = c5_8; best_h = h8.copy()

    # Phase 9: Final coordinate polish + fine LP (5%)
    dl9 = t0 + budget_s * 0.97
    h9, c5_9 = _coordinate_polish(best_h, min(dl9, time.time() + budget_s * 0.02), n_cycles=2)
    if c5_9 < best_c5:
        best_c5 = c5_9; best_h = h9.copy()
    if time.time() < t0 + budget_s * 0.995:
        dl9b = t0 + budget_s * 0.998
        h9b, c5_9b = _lp_optimize(best_h, dl9b, radius_init=0.01, max_iters=2000, radius_min=0.0005)
        if c5_9b < best_c5:
            best_c5 = c5_9b; best_h = h9b.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
