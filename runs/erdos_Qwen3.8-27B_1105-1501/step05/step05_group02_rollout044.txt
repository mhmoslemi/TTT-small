    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(2*n-1)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    
    grad = np.zeros(n)
    # Lag 0
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    
    # Lag d > 0
    for d in range(1, n):
        wp = w[n-1+d] * dx  # positive lag weight
        wn = w[n-1-d] * dx  # negative lag weight
        wc = wp + wn  # combined weight for the "same sign" terms
        m = n - d
        # i >= d: contribution (1-h[i-d]) * wc
        grad[d:] += wc * (1.0 - h[:m])
        # i < m: contribution (-h[i+d]) * wc
        grad[:m] += -wc * h[d:]
    
    # Center to zero mean (for the sum constraint)
    grad -= grad.mean()
    return grad
