import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(0, n):
        m = n - d
        r = n - 1 + d  # positive lag index
        G[r, :m] += (1.0 - h[d:])
        G[r, d:] += -h[:m]
        if d > 0:
            r2 = n - 1 - d  # negative lag index
            G[r2, d:] += (1.0 - h[:m])
            G[r2, :m] += -h[d:]
    return G * dx

def _lp_direction(h, scores, active_idx, radius):
    n = len(h)
    G = _grad_matrix(h)
    G_act = G[active_idx, :]
    s_act = scores[active_idx]
    n_act = len(active_idx)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = -s_act
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, target, max_evals=25):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(max_evals):

        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)

        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))

    opt_h = _project(h + opt_a * p, target)
    opt_c5 = _exact_c5(opt_h)
    best_a = min(c1, c2, key=lambda a: _exact_c5(_project(h + a * p, target)))
    # Actually let's just use the golden section result

    best_c5 = min(_exact_c5(_project(h + a * p, target)) for a in candidates)
    best_a = [a for a in candidates if _exact_c5(_project(h + a * p, target)) == best_c5][0]
    return best_a, best_c5

def _slsqp_optimize(h, deadline, max_iter=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    def objective(x):
        h_test = _project(x, target)
        return _exact_c5(h_test)
    
    def gradient(x):
        h_test = _project(x, target)
        scores = _scores(h_test)
        # Find the max score and its gradient
        max_idx = np.argmax(scores)
        G = _grad_matrix(h_test)
        # Use max function gradient: gradient of the active score
        # For a smoother approximation, use softmax-weighted gradient
        beta = 50.0
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[max_idx] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        return grad
    
    bounds = [(0, 1)] * n
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    ]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        try:
            res = minimize(objective, h, jac=gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    # Try line search from h to res.x
                    p = res.x - h
                    if np.max(np.abs(p)) > 1e-10:
                        _, c5_ls = _line_search(h, p, target)
                        if c5_ls < best_c5 - 1e-14:

                            best_c5 = c5_ls
                            best_h = h_new2.copy()
                            h = h_new2.copy()
                        else:
                            break
                    else:
                        break
            else:
                break
        except Exception:
            break
    
    return best_h, best_c5

    best_h = h.copy()
    best_c5 = cur_c5

        if time.time() > deadline:
            break

            cur_c5 = _exact_c5(h)

                best_h = h.copy()
            stall = 0
            # Gradually increase radius
            radius = min(radius * 1.03, 0.4)
            # Occasionally update top_k
            if it % 20 == 0:
                top_k = min(top_k + 4, 2*n-1)
        else:
            stall += 1
            radius *= 0.7
            # Try fewer active constraints
            if top_k > 8:
                top_k = max(8, top_k - 4)
            if radius < radius_min:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _coord_descent(h, deadline, n_sweeps=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            # Golden section search for h[j]
            lo_v = 0.0
            hi_v = 1.0
            for _ in range(25):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h, deadline, n_passes=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try random pairs
        n_pairs = min(200, n * 4)
        pairs = set()
        for _ in range(n_pairs):
            j = rng.integers(n)
            k = rng.integers(n)
            if j != k:
                pairs.add((min(j,k), max(j,k)))
        
        for j, k in list(pairs):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                continue
            for sign in [1, -1]:
                if sign > 0:
                    a_max = min(1.0 - h[j], h[k])
                else:
                    a_max = min(h[j], 1.0 - h[k])
                if a_max < 1e-10:
                    continue
                # Binary search for optimal transfer
                lo_a, hi_a = 0.0, a_max
                for _ in range(15):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    h1 = h.copy()
                    h1[j] += sign * m1
                    h1[k] -= sign * m1
                    f1 = _exact_c5(h1)
                    h2 = h.copy()
                    h2[j] += sign * m2
                    h2[k] -= sign * m2
                    f2 = _exact_c5(h2)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                a_opt = (lo_a + hi_a) / 2.0
                h_try = h.copy()
                h_try[j] += sign * a_opt
                h_try[k] -= sign * a_opt
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5
