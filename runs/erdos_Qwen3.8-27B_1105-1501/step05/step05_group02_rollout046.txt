def _binary_exchange(h, deadline, rng, n_iters=30000):

    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.01
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            prob = np.exp((cur_c5 - new_c5) / T)
            if rng.random() < prob:
                cur_c5 = new_c5
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                h[i] = 1.0
                h[j] = 0.0
        T *= cool
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0 - 1e-14)
        else:
            mask = (x > 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = float(np.max(scores))
    w = np.exp((scores - max_s) / tau)
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    grad += w[n - 1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[n - 1 + d] * dx
        wn = w[n - 1 - d] * dx
        grad[:m] += wp * (-h[:m]) + wn * (1.0 - h[d:])
        grad[d:] += wp * (1.0 - h[:m]) + wn * (-h[d:])
    return grad

def _pair_transfer(h, i, j, n_grid=25):
    n = len(h)
    ki = h[i]
    kj = h[j]
    t_min = max(-ki, kj - 1.0)
    t_max = min(1.0 - ki, kj)
    if t_max - t_min < 1e-12:
        t_min = max(ki - 1.0, -kj)
        t_max = min(1.0 - kj, ki)
        if t_max - t_min < 1e-12:
            return None, None
        reverse = True
    else:
        reverse = False
    ts = np.linspace(t_min, t_max, n_grid)
    cur_c5_val = _exact_c5(h)
    best_t = None
    best_f = cur_c5_val
    h_trial = h.copy()
    for t in ts:
        if reverse:
            h_trial[i] = ki - t
            h_trial[j] = kj + t
        else:
            h_trial[i] = ki + t
            h_trial[j] = kj - t
        f = _exact_c5(h_trial)
        if f < best_f:
            best_f = f
            best_t = t
    if best_t is None or abs(best_t) < 1e-14:
        return None, None
    h_trial2 = h.copy()
    if reverse:
        h_trial2[i] = ki - best_t
        h_trial2[j] = kj + best_t
    else:
        h_trial2[i] = ki + best_t
        h_trial2[j] = kj - best_t
    best_f = _exact_c5(h_trial2)
    if best_f >= cur_c5_val - 1e-14:
        return None, None
    return h_trial2, best_f

def _pair_transfer_fine(h, i, j, t_lo, t_hi, reverse, n_iters=25):
    ki = h[i]
    kj = h[j]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_lo, t_hi
    h_cand = h.copy()
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    if reverse:
        h_cand[i] = ki - c1
        h_cand[j] = kj + c1
    else:
        h_cand[i] = ki + c1
        h_cand[j] = kj - c1
    f1 = _exact_c5(h_cand)
    if reverse:
        h_cand[i] = ki - c2
        h_cand[j] = kj + c2
    else:
        h_cand[i] = ki + c2
        h_cand[j] = kj - c2
    f2 = _exact_c5(h_cand)
    for _ in range(n_iters):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            if reverse:
                h_cand[i] = ki - c1
                h_cand[j] = kj + c1
            else:
                h_cand[i] = ki + c1
                h_cand[j] = kj - c1
            f1 = _exact_c5(h_cand)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            if reverse:
                h_cand[i] = ki - c2
                h_cand[j] = kj + c2
            else:
                h_cand[i] = ki + c2
                h_cand[j] = kj - c2
            f2 = _exact_c5(h_cand)
    opt_t = (lo + hi) / 2.0
    if reverse:
        h_cand[i] = ki - opt_t
        h_cand[j] = kj + opt_t
    else:
        h_cand[i] = ki + opt_t
        h_cand[j] = kj - opt_t
    f_opt = _exact_c5(h_cand)
    return h_cand, f_opt

def _improve(h, deadline, rng, n_pairs=2500, n_grid=30):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    tau = 0.01
    while time.time() < deadline and stall < 80:
        improved = False
        grad = _weighted_grad(h, tau)
        grad_centered = grad - grad.mean()
        neg_idx = np.where(grad_centered < -0.01)[0]
        pos_idx = np.where(grad_centered > 0.01)[0]
        npairs_passed = 0
        max_pairs_this_pass = n_pairs + int(len(neg_idx) * len(pos_idx) * 0.3)
        # Gradient-directed pairs first
        if len(neg_idx) > 0 and len(pos_idx) > 0:
            n_directed = min(len(neg_idx) * len(pos_idx), 800)
            for _ in range(n_directed):
                if time.time() > deadline:
                    break
                npairs_passed += 1
                i = neg_idx[rng.integers(len(neg_idx))]
                j = pos_idx[rng.integers(len(pos_idx))]
                if i == j:
                    continue
                h_new, c5_new = _pair_transfer(h, i, j, n_grid=n_grid)
                if h_new is not None:
                    h = h_new
                    cur_c5 = c5_new
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        # Random pairs
        n_random = max(0, n_pairs - npairs_passed)
        for _ in range(n_random):
            if time.time() > deadline:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            h_new, c5_new = _pair_transfer(h, i, j, n_grid=n_grid)
            if h_new is not None:
                h = h_new
                cur_c5 = c5_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        # Weighted gradient line search
        if time.time() < deadline - 2:
            d = -grad_centered
            gn = np.linalg.norm(d)
            if gn > 1e-14:
                d /= gn
                alpha_max = 0.5
                for j in range(n):
                    if d[j] > 1e-14:
                        am = (1.0 - h[j]) / d[j]
                        if am < alpha_max:
                            alpha_max = am
                    elif d[j] < -1e-14:
                        am = h[j] / (-d[j])
                        if am < alpha_max:
                            alpha_max = am
                alpha_max = max(alpha_max, 1e-12)
                if alpha_max > 0.01:
                    alphas = np.linspace(0.0, min(alpha_max, 0.4), 12)
                    for a in alphas[1:]:
                        h_cand = _project(h + a * d, target)
                        f = _exact_c5(h_cand)
                        if f < cur_c5 - 1e-14:
                            h = h_cand
                            cur_c5 = f
                            improved = True
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            break
        if improved:
            stall = 0
            tau = max(0.001, tau * 0.92)
        else:
            stall += 1
            tau = min(0.1, tau * 1.3)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_init(n, rng):
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:n // 2]] = 1.0
    return h

def _binary_exchange(h, deadline, rng, n_iters=40000):
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.008
    T_min = 0.00005
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            prob = np.exp((cur_c5 - new_c5) / T)
            if rng.random() < prob:
                cur_c5 = new_c5
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                h[i] = 1.0
                h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _perturb_and_optimize(h, scale, deadline, rng):
    n = len(h)
    h_p = h + rng.normal(0, scale, n)
    h_p = _project(h_p, n / 2.0)
    return _improve(h_p, deadline, rng, n_pairs=1000, n_grid=20)
