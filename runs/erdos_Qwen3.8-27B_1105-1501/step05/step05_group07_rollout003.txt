    radius = radius_init
    stall = 0
    n_active = int(len(_scores(h)) * active_frac) if active_frac < 1.0 else None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, G, scores, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.8
            if stall > 40 and radius < 0.02:
                radius = 0.08
        if it % time_check_interval == 0 and time.time() > deadline:
            break
    return best_h, best_c5

def _smooth_obj_grad(x, beta):
    h = x
    S = _scores(h)
    maxS = np.max(S)
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        ws = np.zeros_like(S)
        ws[int(np.argmax(S))] = 1.0
    else:
        ws = w / wsum
    obj_val = maxS + np.log(float(np.mean(w))) / beta
    G = _grad_matrix(h)
    grad = G.T @ ws
    return obj_val, grad

def _smooth_optimize(h, deadline, beta_list=None, maxiter_per_phase=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    L = 2 * n - 1
    if beta_list is None:
        b_min = max(8.0, 0.15 * L)
        b_max = min(2000.0, max(b_min * 4, 4.0 * L))
        beta_list = np.geomspace(b_min, b_max, 5)
    for beta in beta_list:
        if time.time() > deadline:
            break
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(_smooth_obj_grad_wrapper, h, args=(float(beta),),
                           jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_phase, 'ftol': 1e-14,
                                    'maxfun': maxiter_per_phase * 2})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _smooth_obj_grad_wrapper(x, beta):
    h = _project(x, len(x) / 2.0)
    S = _scores(h)
    maxS = np.max(S)
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        ws = np.zeros_like(S)
        ws[int(np.argmax(S))] = 1.0
    else:
        ws = w / wsum
    obj_val = maxS + np.log(float(np.mean(w))) / beta
    G = _grad_matrix(h)
    grad = G.T @ ws
    return obj_val, grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = np.sin(2.0 * np.pi * alpha * x + 0.5).astype(float)
    h = 0.5 * (1.0 + np.tanh(2.0 * (h - 0.5)))
    return _project(h, n / 2.0)

def _quasi_periodic_init2(n, alpha):
    x = np.arange(n) / n
    t = (x * alpha) % 1.0
    h = (t < 0.5).astype(float)
    h = h + 0.1 * np.sin(4.0 * np.pi * x)
    return _project(h, n / 2.0)

def _perturb_restart(h, rng, scale, deadline, radius=0.08, max_iters=800, stall_limit=60):
    n = len(h)
    h_pert = h + rng.normal(0, scale, n)
    h_pert = _project(h_pert, n / 2.0)
    ht, ct = _lp_optimize(h_pert, deadline, radius_init=radius,
                          max_iters=max_iters, stall_limit=stall_limit)
    return ht, ct

def _pair_polish(h, deadline, max_passes=5, stride=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if stride is None:
        stride = max(1, n // 24)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, stride):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + stride, n, stride):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float, copy=True)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP optimization from initial (n=128)
    dl1 = t0 + budget_s * 0.18
    h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.3, max_iters=15000,
                            stall_limit=250, time_check_interval=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n0

    # Phase 2: Smooth log-sum-exp optimization
    if time.time() < t0 + budget_s * 0.32:
        dl2 = t0 + budget_s * 0.32
        h2, c5_2 = _smooth_optimize(best_h, dl2, maxiter_per_phase=250)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        dl2b = min(t0 + budget_s * 0.36, time.time() + budget_s * 0.04)
        h2b, c5_2b = _lp_optimize(best_h, dl2b, radius_init=0.1,
                                  max_iters=3000, stall_limit=100)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Multi-resolution
    if time.time() < t0 + budget_s * 0.72:
        dl3 = t0 + budget_s * 0.72
        n_tries = [160, 192, 224]
        for idx, n_try in enumerate(n_tries):
            if time.time() > dl3:
                break
            remaining_in_phase = dl3 - time.time()
            if remaining_in_phase < budget_s * 0.05:
                break
            # Resample from best
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl3, time.time() + budget_s * 0.06)
            ht, ct = _lp_optimize(h_init, sub_dl, radius_init=0.2,
                                  max_iters=5000, stall_limit=150)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try
            # Quasi-periodic start
            if time.time() < dl3:
                alphas = [0.6180339887, 0.4142135624, 0.7071067812]
                h_qp = _quasi_periodic_init2(n_try, alphas[idx % 3])
                h_qp = h_qp + rng.normal(0, 0.08, n_try)
                h_qp = _project(h_qp, n_try / 2.0)
                sub_dl2 = min(dl3, time.time() + budget_s * 0.04)
                ht2, ct2 = _lp_optimize(h_qp, sub_dl2, radius_init=0.25,
                                        max_iters=3000, stall_limit=120)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = n_try
                # Smooth polish on best from this resolution
                if time.time() < dl3 and best_n == n_try:
                    dl_sm = min(dl3, time.time() + budget_s * 0.02)
                    ht3, ct3 = _smooth_optimize(best_h, dl_sm, maxiter_per_phase=150)
                    if ct3 < best_c5:
                        best_c5 = ct3
                        best_h = ht3.copy()

    # Phase 4: Perturbation restarts
    if time.time() < t0 + budget_s * 0.88:
        dl4 = t0 + budget_s * 0.88
        scale = 0.03
        n_attempts = 0
        while time.time() < dl4 and n_attempts < 500:
            n_attempts += 1
            sub_dl = min(dl4, time.time() + budget_s * 0.012)
            ht, ct = _perturb_restart(best_h, rng, scale, sub_dl,
                                      radius=0.06, max_iters=600, stall_limit=50)
            if ct < best_c5 - 1e-14:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.005, scale * 0.85)
            else:
                scale = min(0.1, scale * 1.15)

    # Phase 5: Final LP polish
    if time.time() < t0 + budget_s * 0.94:
        dl5 = t0 + budget_s * 0.94
        h5, c5_5 = _lp_optimize(best_h, dl5, radius_init=0.04,
                                max_iters=5000, stall_limit=100, active_frac=0.5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Smooth final
    if time.time() < t0 + budget_s * 0.96:
        dl6 = t0 + budget_s * 0.96
        h6, c5_6 = _smooth_optimize(best_h, dl6,
                                    beta_list=np.array([200.0, 500.0, 1000.0]),
                                    maxiter_per_phase=200)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.978:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.98, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
