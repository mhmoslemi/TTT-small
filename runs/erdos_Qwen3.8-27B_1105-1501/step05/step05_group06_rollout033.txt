import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _beatty_init(n, alpha, rng):
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.08, n)
    return _project(h, n / 2.0)

def _binary_full_sweep(h, max_rounds=300, use_double=False, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(h)
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    ones = int(np.sum(h_bin == 1))
    if ones != target:
        if ones > target:
            idx = np.where(h_bin == 1)[0]
            perm = rng.permutation(len(idx))
            for k in range(ones - target):
                h_bin[idx[perm[k]]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            perm = rng.permutation(len(idx))
            for k in range(target - ones):
                h_bin[idx[perm[k]]] = 1

    cur_c5 = _exact_c5(h_bin)
    best_c5 = cur_c5
    best_h = h_bin.copy()
    stall = 0

    for rd in range(max_rounds):
        best_delta = 0.0
        best_i, best_j = -1, -1
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]

        for i in ones_idx:
            for j in zeros_idx:
                h_bin[i] = 0
                h_bin[j] = 1
                c5 = _exact_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                if c5 - cur_c5 < best_delta - 1e-15:
                    best_delta = c5 - cur_c5
                    best_i, best_j = i, j

        if best_i != -1:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            cur_c5 += best_delta
            stall = 0
        elif use_double and rng.random() < 0.5:
            ones_idx = np.where(h_bin == 1)[0]
            zeros_idx = np.where(h_bin == 0)[0]
            if len(ones_idx) >= 2 and len(zeros_idx) >= 2:
                i1, i2 = rng.choice(ones_idx, 2, replace=False)
                j1, j2 = rng.choice(zeros_idx, 2, replace=False)
                h_bin[i1] = 0; h_bin[i2] = 0
                h_bin[j1] = 1; h_bin[j2] = 1
                c5 = _exact_c5(h_bin)
                if c5 < cur_c5 + 0.002:
                    cur_c5 = c5
                    stall = 0
                else:
                    h_bin[i1] = 1; h_bin[i2] = 1
                    h_bin[j1] = 0; h_bin[j2] = 0
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

        if stall > 15:
            break

    return best_h, best_c5

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lp_step(h, G, radius, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    scores = _scores(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-12)
        hi = min(radius, 1.0 - h[j] - 1e-12)
        if lo >= hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search(h, p, n_evals=40):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)

    alphas = np.linspace(0, alpha_max, 11)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0:
        return 0.0, float(vals[0])
    if best_i == len(alphas) - 1:
        return alpha_max, float(vals[-1])
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(len(alphas) - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _lp_refine(h, deadline, radius_init=0.08, max_iters=5000, top_k_list=None):
    if top_k_list is None:
        top_k_list = [None, 8, 5, 3, 2]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_tk = len(top_k_list)
    ti = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        tk = top_k_list[ti % n_tk]
        G = _build_G(h)
        p, _ = _lp_step(h, G, radius, top_k=tk)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        alpha, _ = _line_search(h, p)
        if alpha < 1e-14:
            stall += 1
            radius *= 0.8
            ti += 1
            continue
        h_cand = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_cand)
        if actual_c5 < cur_c5 - 1e-14:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 10 and radius < 0.01:
                radius = 0.05

    return best_h, best_c5

def _pair_descent_fast(h, deadline, max_passes=100, thresh=1e-5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h[np.clip(h, 0, 1)]
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not pass_improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
