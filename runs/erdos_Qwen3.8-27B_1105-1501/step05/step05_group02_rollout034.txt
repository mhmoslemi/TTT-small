import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _weighted_grad(h, gamma):
    """Compute weighted gradient of C5 at h."""
    n = len(h)
    dx = 2.0 / n
    sc = _scores(h)
    max_sc = np.max(sc)
    w = np.exp(gamma * (sc - max_sc))
    sw = np.sum(w)
    if sw < 1e-300:
        idx = int(np.argmax(sc))
        w = np.zeros(len(sc))
        w[idx] = 1.0
    else:
        w /= sw
    
    grad = np.zeros(n)
    # Central lag (d=0): score = sum h[i](1-h[i])
    w0 = w[n-1] * dx
    grad += w0 * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        # Positive offset d: score = sum_{i=0}^{n-1-d} h[i+d](1-h[i])
        wp = w[n-1+d] * dx
        if wp > 1e-15:
            # d/dh[j] = (1-h[j-d]) * [j>=d] - h[j+d] * [j<n-d]
            if d < n:
                # j from d to n-1: gets (1-h[j-d])
                grad[d:] += wp * (1.0 - h[:n-d])
                # j from 0 to n-1-d: gets -h[j+d]
                grad[:n-d] += -wp * h[d:]
        
        # Negative offset -d: score = sum_{i=0}^{n-1-d} h[i](1-h[i+d])
        wn = w[n-1-d] * dx
        if wn > 1e-15:
            # j from 0 to n-1-d: gets (1-h[j+d])
            grad[:n-d] += wn * (1.0 - h[d:])
            # j from d to n-1: gets -h[j-d]
            grad[d:] += -wn * h[:n-d]
    
    return grad

def _grad_line_search(h, direction, target):
    """Line search along h + alpha*direction, maintaining feasibility."""
    n = len(h)
    # Find max alpha such that h + alpha*d stays in [0,1]
    alpha_max = 1.0
    for j in range(n):
        if direction[j] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-15:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-12)
    
    # Golden section search on [0, alpha_max]
    cur = _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * direction)
    f2 = _exact_c5(h + c2 * direction)
    for _ in range(20):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * direction)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * direction)
    opt_a = (lo + hi) / 2.0
    opt_h = h + opt_a * direction
    opt_h = _project(opt_h, target)
    opt_c5 = _exact_c5(opt_h)
    return opt_h, opt_c5, opt_a

def _pair_transfer(h, i, j, target):
    """Try transferring mass from j to i (or i to j). Return best (h_new, c5)."""
    hi = h[i]
    hj = h[j]
    # Transfer from j to i: h[i] += t, h[j] -= t
    # Feasible: -hj <= t <= 1-hi
    t_min = max(-hj, -1.0 + 0.0)  # h[j]-t >= 0 => t <= hj; h[i]+t <= 1 => t <= 1-hi
    # Actually: t <= hj (so h[j]-t >= 0) and t <= 1-hi (so h[i]+t <= 1)
    # Also: t >= -hj (so h[j]-t <= 1, i.e., t >= hj-1) and t >= -(1-hi) (so h[i]+t >= 0, i.e., t >= -hi)
    t_lo = max(-hi, hj - 1.0)
    t_hi = min(1.0 - hi, hj)
    
    if t_hi - t_lo < 1e-14:
        return h, _exact_c5(h)
    
    # Sample the interval
    best_t = 0.0
    best_c5 = _exact_c5(h)
    
    # Grid sample
    n_grid = 11
    for k in range(n_grid + 1):
        t = t_lo + k * (t_hi - t_lo) / n_grid
        h[i] = hi + t
        h[j] = hj - t
        c5_val = _exact_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_t = t
    
    # Local golden section around best_t
    if 0 < best_t < t_hi - t_lo:
        mid = (t_hi - t_lo) / n_grid
        lo_t = max(t_lo, best_t - mid)
        hi_t = min(t_hi, best_t + mid)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        h[i] = hi + c1; h[j] = hj - c1
        f1 = _exact_c5(h)
        h[i] = hi + c2; h[j] = hj - c2
        f2 = _exact_c5(h)
        for _ in range(15):
            if hi_t - lo_t < 1e-14:
                break
            if f1 < f2:
                hi_t = c2; f2 = f1
                c2 = c1
                c1 = hi_t - gr * (hi_t - lo_t)
                h[i] = hi + c1; h[j] = hj - c1
                f1 = _exact_c5(h)
            else:
                lo_t = c1; f1 = f2
                c1 = c2
                c2 = lo_t + gr * (hi_t - lo_t)
                h[i] = hi + c2; h[j] = hj - c2
                f2 = _exact_c5(h)
        opt_t = (lo_t + hi_t) / 2.0
        h[i] = hi + opt_t; h[j] = hj - opt_t
        opt_c5 = _exact_c5(h)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_t = opt_t
    
    # Restore
    h[i] = hi + best_t
    h[j] = hj - best_t
    return h, best_c5

def _pair_optimize(h, deadline, rng, target=None):
    """Main pair transfer optimizer."""
    n = len(h)
    if target is None:
        target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline:
        if stall > 500:
            break
        # Choose pairs
        n_pairs = min(200, n * 2)
        improved = False
        for _ in range(n_pairs):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            h_new, c5_new = _pair_transfer(h, i, j, target)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if improved:

        else:

def _grad_optimize(h, deadline, rng):
    """Gradient-based optimizer with adaptive gamma."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = 5.0
    stall = 0
    
    while time.time() < deadline:
        if stall > 100:
            break
        grad = _weighted_grad(h, gamma)
        # Center the gradient (remove mean to preserve sum)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        # Try different step sizes
        best_step_h = h
        best_step_c5 = cur_c5
        for scale in [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]:
            direction = -scale * grad / gn
            h_try, c5_try, _ = _grad_line_search(h, direction, target)
            if c5_try < best_step_c5:
                best_step_c5 = c5_try
                best_step_h = h_try
        if best_step_c5 < cur_c5 - 1e-14:
            h = best_step_h
            cur_c5 = best_step_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.02, 200.0)
        else:
            stall += 1
            gamma = min(gamma * 1.05, 200.0)
            if stall % 20 == 0:
                gamma = max(5.0, gamma * 0.5)
    
    return best_h, best_c5

def _binary_sa(n, deadline, rng):
    """Simulated annealing on binary vectors."""
    K = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:K]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.01
    T_min = 0.0005
    n_iters = 50000
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i_pos = ones[i_idx]
        j_pos = zeros[j_idx]
        h[i_pos] = 0.0
        h[j_pos] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones[i_idx] = j_pos
            zeros[j_idx] = i_pos
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i_pos] = 1.0
            h[j_pos] = 0.0
        T *= cool
    
    return best_h, best_c5

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _exact_c5(h):

            cur_c5 = c5_new
            improved = True
    return h, cur_c5, improved

def _grad_step_pair(h, gamma, target, deadline):
    n = len(h)
    grad = _weighted_grad(h, gamma)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, _exact_c5(h), False
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h
    direction = -grad / gn
    for scale in [0.3, 0.15, 0.05, 0.02, 0.01, 0.005]:
        if time.time() > deadline:
            break
        d = scale * direction
        alpha_max = 1.0
        for j in range(n):
            if d[j] > 1e-15:
                alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
            elif d[j] < -1e-15:
                alpha_max = min(alpha_max, h[j] / (-d[j]))
        alpha_max = max(alpha_max, 1e-12)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1a = hi_a - gr * (hi_a - lo_a)
        c2a = lo_a + gr * (hi_a - lo_a)
        h_try = h
        f1 = _exact_c5(h_try + c1a * d)
        f2 = _exact_c5(h_try + c2a * d)
        for _ in range(18):
            if hi_a - lo_a < 1e-14:
                break
            if f1 < f2:
                hi_a = c2a
                f2 = f1
                c2a = c1a
                c1a = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(h_try + c1a * d)
            else:
                lo_a = c1a
                f1 = f2
                c1a = c2a
                c2a = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(h_try + c2a * d)
        opt_a = (lo_a + hi_a) / 2.0
        h_candidate = h + opt_a * direction * scale
        h_candidate = _project(h_candidate, target)
        c5_candidate = _exact_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    if best_c5 < cur_c5 - 1e-14:
        return best_h, best_c5, True
    return h, cur_c5, False

def _main_optimize(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = 10.0
    stall = 0
    pair_stall = 0
    phase = 0
    while time.time() < deadline:
        if stall > 200:
            break
        sub_dl = min(deadline, time.time() + budget_remaining(deadline, t_global))
        if phase % 3 == 0:
            h_new, c5_new, improved = _grad_step_pair(h, gamma, target, sub_dl)
            if improved:
                h = h_new
                cur_c5 = c5_new
                stall = 0
                gamma = min(gamma * 1.03, 300.0)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall += 1
                gamma = min(gamma * 1.04, 300.0)
        else:
            n_pairs = 100
            h_new, c5_new, improved = _greedy_pair_pass(h, n_pairs, min(deadline, time.time() + 2.0), rng)
            if improved:
                h = h_new
                cur_c5 = c5_new
                pair_stall = 0
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                pair_stall += 1
                stall += 1
        phase += 1
        if stall > 50 and phase % 10 == 0:
            gamma = max(5.0, gamma * 0.6)
    return best_h, best_c5

def budget_remaining(deadline, t0):
    return deadline - time.time()

def _binary_sa(n, deadline, rng):
    K = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:K]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.008
    T_min = 0.0003
    n_iters = 40000
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ii = int(rng.integers(len(ones)))
        jj = int(rng.integers(len(zeros)))
        ip = ones[ii]
        jp = zeros[jj]
        h[ip] = 0.0
        h[jp] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones[ii] = jp
            zeros[jj] = ip
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[ip] = 1.0
            h[jp] = 0.0
        T *= cool
    return best_h, best_c5

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def _directed_pair_pass(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    sc = _scores(h)
    max_sc = np.max(sc)
    active = np.where(sc > max_sc - 0.01)[0]
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d == 0:
            grad += (1.0 - 2.0 * h) * (2.0 / n)
        elif d > 0:
            m = n - d
            grad[d:] += (2.0 / n) * (1.0 - h[:m])
            grad[:m] -= (2.0 / n) * h[d:]
        else:
            g = -d
            m = n - g
            grad[:m] += (2.0 / n) * (1.0 - h[g:])
            grad[g:] -= (2.0 / n) * h[:m]
    grad -= grad.mean()
    order = np.argsort(grad)
    n_low = min(20, n // 4)
    n_high = min(20, n // 4)
    low_set = order[:n_low]
    high_set = order[-n_high:]
    improved = False
    n_tries = min(300, len(low_set) * len(high_set))
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        i = int(low_set[rng.integers(len(low_set))])
        j = int(high_set[rng.integers(len(high_set))])
        if i == j:
            continue
        h_new, c5_new = _pair_transfer(h, i, j)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            improved = True
    return h, cur_c5, improved

t_global = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    global t_global
    t0 = time.time()
    t_global = t0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Initial gradient optimization (fast)
    dl1 = t0 + budget_s * 0.06
    h1, c5_1, _ = _grad_step_pair(best_h, 15.0, n0 / 2.0, dl1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    for _ in range(3):
        if time.time() > dl1:
            break
        h1, c5_1, _ = _grad_step_pair(best_h, 20.0, n0 / 2.0, dl1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Main optimization loop
    dl2 = t0 + budget_s * 0.40
    h_work = best_h.copy()
    c5_work = best_c5
    gamma = 10.0
    stall_count = 0
    while time.time() < dl2:
        # Directed pair pass
        sub_dl = min(dl2, time.time() + 3.0)
        h_new, c5_new, imp = _directed_pair_pass(h_work, sub_dl, rng)
        if imp:
            h_work = h_new
            c5_work = c5_new
            stall_count = 0
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
        else:
            stall_count += 1
        # Random pair pass
        sub_dl = min(dl2, time.time() + 2.0)
        h_new, c5_new, imp = _greedy_pair_pass(h_work, 80, sub_dl, rng)
        if imp:
            h_work = h_new
            c5_work = c5_new
            stall_count = 0
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
        else:
            stall_count += 1
        # Gradient step
        sub_dl = min(dl2, time.time() + 1.5)
        h_new, c5_new, imp = _grad_step_pair(h_work, gamma, n0 / 2.0, sub_dl)
        if imp:
            h_work = h_new
            c5_work = c5_new
            stall_count = 0
            gamma = min(gamma * 1.02, 200.0)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
        else:
            gamma = min(gamma * 1.04, 200.0)
            if stall_count % 15 == 0:
                gamma = max(5.0, gamma * 0.5)
                stall_count = 0
        if stall_count > 200:
            break

    # Phase 3: Resolution exploration
    dl3 = t0 + budget_s * 0.70
    res_list = [n0 - 16, n0 - 32, n0 + 16, n0 + 32, n0 - 8, n0 + 8, 96, 92, 120, 136, 144, 160]
    res_list = sorted(set([r for r in res_list if 48 <= r <= 256 and r % 2 == 0]))
    for nr in res_list:
        if time.time() > dl3:
            break
        if nr == best_n:
            continue
        h_res = _resample(best_h, nr)
        c5_res = _exact_c5(h_res)
        sub_dl = min(dl3, time.time() + budget_s * 0.035)
        h_try = h_res.copy()
        c5_try = c5_res
        for _ in range(20):
            if time.time() > sub_dl:
                break
            h_t2, c5_t2, imp = _directed_pair_pass(h_try, min(sub_dl, time.time() + 1.5), rng)
            if imp:
                h_try = h_t2
                c5_try = c5_t2
            else:
                h_t2, c5_t2, imp = _greedy_pair_pass(h_try, 50, min(sub_dl, time.time() + 1.0), rng)
                if imp:
                    h_try = h_t2
                    c5_try = c5_t2
            h_t2, c5_t2, imp = _grad_step_pair(h_try, 15.0, nr / 2.0, min(sub_dl, time.time() + 0.8))
            if imp:
                h_try = h_t2
                c5_try = c5_t2
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = nr

    # Phase 4: Binary search at best and nearby resolutions
    if time.time() < t0 + budget_s * 0.85:
        dl4 = t0 + budget_s * 0.88
        for nb in [best_n, max(64, best_n - 32), min(192, best_n + 32)]:
            if time.time() > dl4:
                break
            if nb % 2 != 0:
                nb -= 1
            sub_dl = min(dl4, time.time() + budget_s * 0.02)
            hb, c5b = _binary_sa(nb, sub_dl, rng)
            if c5b < best_c5 - 1e-10:
                best_c5 = c5b
                best_h = hb.copy()
                best_n = nb
                sub_dl2 = min(dl4, time.time() + budget_s * 0.01)
                hp, c5p, _ = _greedy_pair_pass(hb, 100, sub_dl2, rng)
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = hp

    # Phase 5: Quasi-periodic seeds
    if time.time() < t0 + budget_s * 0.92:
        dl5 = t0 + budget_s * 0.93
        for a in [0.6180339887, 0.4142135624, 0.7071067812]:
            if time.time() > dl5:
                break
            h_qp = _quasi_periodic(best_n, a)
            sub_dl = min(dl5, time.time() + budget_s * 0.012)
            h_try = h_qp.copy()
            c5_try = _exact_c5(h_try)
            for _ in range(10):
                if time.time() > sub_dl:
                    break
                h_t, c5_t, imp = _directed_pair_pass(h_try, min(sub_dl, time.time() + 0.8), rng)
                if imp:
                    h_try = h_t
                    c5_try = c5_t
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.97:
        dl6 = t0 + budget_s * 0.985
        for _ in range(50):
            if time.time() > dl6:
                break
            h_t, c5_t, imp = _directed_pair_pass(best_h, min(dl6, time.time() + 1.0), rng)
            if imp:
                best_h = h_t
                best_c5 = c5_t
            else:
                h_t, c5_t, imp = _greedy_pair_pass(best_h, 30, min(dl6, time.time() + 0.5), rng)
                if imp:
                    best_h = h_t
                    best_c5 = c5_t
        h_t, c5_t, imp = _grad_step_pair(best_h, 30.0, best_n / 2.0, dl6)
        if imp:
            best_h = h_t
            best_c5 = c5_t

    # Final
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
