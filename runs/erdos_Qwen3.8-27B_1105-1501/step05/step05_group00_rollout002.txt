    C = np.correlate(d, d, mode="full")
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust)
    t_hi = min(t_hi, trust)
    if t_hi - t_lo < 1e-14:
        return 0.0, exact_c5(h)

    def fval(a):
        return float(np.max(S0 + a * T - a * a * C)) * dx

    grid = np.linspace(t_lo, t_hi, 48)
    vals = fval(grid[0])
    best_t, best_v = grid[0], vals
    for gv in grid[1:]:
        fv = fval(gv)
        if fv < best_v:
            best_t, best_v = gv, fv
    a, bbr = (t_lo, t_hi) if abs(best_t - t_lo) < 1e-9 or abs(best_t - t_hi) < 1e-9 \
        else (grid[int(np.argmin([fval(x) for x in grid])) - 1] if int(np.argmin([fval(x) for x in grid])) > 0 else t_lo), \
        (grid[int(np.argmin([fval(x) for x in grid])) + 1] if int(np.argmin([fval(x) for x in grid])) < len(grid) - 1 else t_hi)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = bbr - gr * (bbr - a)
    c2 = a + gr * (bbr - a)
    f1, f2 = fval(c1), fval(c2)
    for _ in range(40):
        if bbr - a < 1e-13:
            break
        if f1 < f2:
            bbr, c2, f2 = c2, c1, f1
            c1 = bbr - gr * (bbr - a)
            f1 = fval(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (bbr - a)
            f2 = fval(c2)
    ot = 0.5 * (a + bbr)
    ov = fval(ot)
    if ov < best_v:
        best_t, best_v = ot, ov
    return best_t, best_v


def grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                G[r, :m] += (1.0 - h[gg:])
                G[r, gg:] += -h[:m]
    return G


def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def lp_loop(h0, deadline, radius_init=0.15):
    n = len(h0)
    target = n / 2.0
    h = project(h0.copy(), target)
    dx = 2.0 / n
    cur = exact_c5(h)
    best_h, best_c = h.copy(), cur
    radius = radius_init
    stall = 0
    restarts = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > 50:
            if restarts < 6:
                h = project(h + rng.normal(0, 0.02, n), target)
                cur = exact_c5(h)
                radius = 0.12
                stall = 0
                restarts += 1
                continue
            break
        scores = np.correlate(h, 1.0 - h, mode="full")
        G = grad_matrix(h)
        p = lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c = line_search(h, p, radius)
        if new_c < cur - 1e-13:
            h = project(h + alpha * p, target)
            cur = exact_c5(h)
            if cur < best_c:
                best_c, best_h = cur, h.copy()
            stall = 0
            pred = float(np.max(scores + G @ p)) * dx
            ratio = cur / max(pred, 1e-15)
            if ratio < 0.7:
                radius = min(radius * 1.12, 0.4)
            elif ratio > 1.5:
                radius *= 0.85
            else:
                radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.08
    return best_h, best_c


def softmax_loop(h0, deadline, tau0=0.02):
    n = len(h0)
    target = n / 2.0
    h = project(h0.copy(), target)
    cur = exact_c5(h)
    best_h, best_c = h.copy(), cur
    tau = tau0
    stall = 0
    while time.time() < deadline and stall < 40:
        d = softmax_direction(h, tau)
        nd = float(np.linalg.norm(d))
        if nd < 1e-10:
            break
        d = d / nd * min(0.15, nd) * 6.0
        alpha, new_c = line_search(h, d, 0.4)
        if new_c < cur - 1e-12:
            h = project(h + alpha * d, target)
            cur = exact_c5(h)
            if cur < best_c:
                best_c, best_h = cur, h.copy()
            stall = 0
            tau = min(tau * 1.08, 0.2)
        else:
            stall += 1
            tau *= 0.8
            if tau < 1e-4:
                tau = 0.05
    return best_h, best_c


def binary_sa(h0, deadline, rng):
    n = len(h0)
    target = n // 2
    dx = 2.0 / n
    order = np.argsort(-h0)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    cur = exact_c5(h)
    best_h, best_c = h.copy(), cur
    T, Tmin, cool = 0.008, 1e-5, 0.9997
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    patience, maxp = 0, 15000
    while time.time() < deadline:
        if patience > maxp:
            T = max(Tmin, T * 0.98)
            patience = 0
        if T < Tmin:
            break
        for _ in range(60):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            nc = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
            delta = nc - cur
            if delta < 0 or (T > Tmin and rng.random() < np.exp(-delta / T)):
                cur = nc
                if cur < best_c:
                    best_c, best_h = cur, h.copy()
                    patience = 0
                else:
                    patience += 1
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
            else:
                h[i], h[j] = 1.0, 0.0
        T = max(Tmin, T * cool)
    return best_h, best_c


def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return project(np.interp(xn, xo, h), n_new / 2.0)


def pair_refine(h0, deadline, max_passes=60):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h, best_c = h.copy(), exact_c5(h)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[ii]
            for jj in range(ii + 1, len(interior)):
                j = interior[jj]
                d = np.zeros(n)
                d[i], d[j] = -1.0, 1.0
                alpha, ctry = line_search(h, d, 0.5)
                if ctry < best_c - 1e-13:
                    h = project(h + alpha * d, target)
                    c = exact_c5(h)
                    if c < best_c:
                        best_c, best_h = c, h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c


def structured_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 4 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    return project(h, n / 2.0)


def sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)


def optimize_n(h0, n_target, deadline, rng):
    if n_target < 24 or n_target > 220 or n_target % 2 != 0:
        return h0, exact_c5(h0)
    if n_target == len(h0):
        hc = h0.copy()
    else:
        hc = resample(h0, n_target)
    t_sf = min(deadline, time.time() + 0.25 * (deadline - time.time()) + 1.0)
    hc, c1 = softmax_loop(hc, min(deadline, time.time() + 25), tau0=0.02)
    hc, c2 = lp_loop(hc, min(deadline, time.time() + 30), radius_init=0.12)
    hc, c3 = binary_sa(hc, min(deadline, time.time() + 12), rng)
    if c3 < c2:
        hc, c2 = hc, c3
    hc, c4 = lp_loop(hc, deadline, radius_init=0.07)
    return hc, min(c1, c2, c3, c4, c4 if c4 is not None else c4)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n_f = 122
        h0 = structured_init(n_f, rng)
    h0 = project(h0, len(h0) / 2.0)
    best_h, best_c = h0.copy(), exact_c5(h0)

    def consider(hc, cc):
        nonlocal best_h, best_c
        if cc < best_c - 1e-12 and 0.0 <= float(np.min(hc)) and float(np.max(hc)) <= 1.0:
            best_h, best_c = hc.copy(), cc

    # A: softmax + LP polish on parent
    hA, cA = softmax_loop(best_h, min(deadline, t0 + budget_s * 0.12), tau0=0.015)
    consider(hA, cA)
    hA2, cA2 = lp_loop(best_h, min(deadline, t0 + budget_s * 0.22), radius_init=0.12)
    consider(hA2, cA2)

    # B: diverse starts
    d_B = min(deadline, t0 + budget_s * 0.34)
    n0 = len(best_h)
    for q in range(8):
        if time.time() > d_B:
            break
        if q % 2 == 0:
            hq = sine_init(n0, 1 + (q // 2), q * 0.7)
        else:
            hq = structured_init(n0, rng)
        sub = min(d_B, time.time() + budget_s * 0.012)
        hq, cq = softmax_loop(hq, sub, tau0=0.02)
        hq, cq = lp_loop(hq, sub, radius_init=0.1)
        consider(hq, cq)

    # C: binary SA + polish
    d_C = min(deadline, t0 + budget_s * 0.46)
    hC, cC = binary_sa(best_h, d_C, rng)
    consider(hC, cC)
    hC2, cC2 = lp_loop(best_h, min(deadline, t0 + budget_s * 0.52), radius_init=0.07)
    consider(hC2, cC2)

    # D: multi-resolution
    d_D = min(deadline, t0 + budget_s * 0.80)
    n_best = len(best_h)
    tried = set()
    offsets = list(range(-20, 44, 2))
    off = 0
    for dn in offsets:
        if time.time() > d_D:
            break
        nt = n_best + dn
        if nt < 40 or nt > 200 or nt in tried:
            continue
        tried.add(nt)
        per = min(d_D, time.time() + max(8.0, budget_s * 0.020))
        hc, cc = optimize_n(best_h, nt, per, rng)
        consider(hc, cc)
        if cc < best_c:
            n_best = nt

    # E: perturbation restarts
    d_E = min(deadline, t0 + budget_s * 0.92)
    scale = 0.04
    for _ in range(300):
        if time.time() > d_E:
            break
        if best_c < 0.3806:
            break
        hp = project(best_h + rng.normal(0, scale, len(best_h)), len(best_h) / 2.0)
        sub = min(d_E, time.time() + budget_s * 0.006)
        ht, ct = softmax_loop(hp, sub, tau0=0.012)
        ht, ct = lp_loop(ht, sub, radius_init=0.06)
        if ct < best_c - 1e-12:
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.16, scale * 1.15)

    # F: pair refine + final polish
    d_F = min(deadline - 2.0, t0 + budget_s * 0.97)
    hF, cF = pair_refine(best_h, d_F, max_passes=50)
    consider(hF, cF)
    hF2, cF2 = lp_loop(best_h, deadline - 1.0, radius_init=0.03)
    consider(hF2, cF2)

    best_h = project(best_h, len(best_h) / 2.0)
    final_c = exact_c5(best_h)
    return best_h, final_c, len(best_h)
