def _project(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target} using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))

            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

This is a crucial insight! The 1D problem is to minimize a convex piecewise-linear function. The minimum occurs at a breakpoint (where two adjacent linear pieces meet) or at a boundary.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if abs(w) < 1e-15:
            continue
        k = r - (n - 1)
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * dx * (1.0 - h[:n - k])
            grad[:n - k] += w * dx * (-h[k:])
        else:
            g = -k
            grad[:n - g] += w * dx * (1.0 - h[g:])
            grad[g:] += w * dx * (-h[:n - g])
    return grad

def _eval_pair(h, i, j, t):
    hi_save = h[i]
    hj_save = h[j]
    h[i] = hi_save + t
    h[j] = hj_save - t
    val = _exact_c5(h)
    h[i] = hi_save
    h[j] = hj_save
    return val

def _pair_optimize(h, i, j):
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-13:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = t_min
    hi = t_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _eval_pair(h, i, j, c1)
    f2 = _eval_pair(h, i, j, c2)
    for _ in range(22):
        if hi - lo < 1e-14:
            break
        if f1 <= f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _eval_pair(h, i, j, c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _eval_pair(h, i, j, c2)
    t_opt = (lo + hi) / 2.0
    f_opt = _eval_pair(h, i, j, t_opt)
    return t_opt, f_opt

def _pair_pass(h, rng, n_pairs, deadline, topk_hint=None):
    n = len(h)
    cur_c5 = _exact_c5(h)
    improved = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if topk_hint is not None and topk_hint[0] > 0 and rng.random() < 0.5:
            i = topk_hint[1][rng.integers(topk_hint[0])]
            j = topk_hint[2][rng.integers(topk_hint[0])]
        else:
            i = rng.integers(n)
            j = rng.integers(n)
        if i == j:
            continue
        t_opt, f_opt = _pair_optimize(h, i, j)
        if f_opt < cur_c5 - 1e-13 and abs(t_opt) > 1e-12:
            h[i] += t_opt
            h[j] -= t_opt
            h = np.clip(h, 0.0, 1.0)
            cur_c5 = f_opt
            improved += 1
    h = _project(h, n / 2.0)
    cur_c5 = _exact_c5(h)
    return h, cur_c5, improved

def _grad_step(h, gamma):
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    max_s = float(np.max(scores))
    w = np.exp(gamma * (scores - max_s))
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = _weighted_grad(h, w)
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, _exact_c5(h)
    p = -grad / gn
    cur_c5 = _exact_c5(h)
    best_a = 0.0
    best_c5 = cur_c5
    a_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_max = min(a_max, h[j] / (-p[j]))
    a_max = max(a_max, 1e-12)
    a = min(0.3 * a_max, 0.15)
    for _ in range(25):
        h_new = h + a * p
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_a = a
        a *= 0.7
        if a < 1e-14:
            break
    if best_a > 0:
        h_new = h + best_a * p
        h_new = _project(h_new, target)
        return h_new, _exact_c5(h_new)
    return h, cur_c5

def _grad_optimize(h, deadline, gamma_init=5.0, gamma_end=400.0, max_it=50000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 50:
            if gamma > 200:
                break
            gamma = min(gamma * 1.15, gamma_end)
        h_new, c5_new = _grad_step(h, gamma)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.03, gamma_end)
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_init, n_iters, deadline, rng):
    n = len(h_init)
    h = h_init.astype(float).copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h >= 0.999)[0])
    zeros = list(np.where(h <= 0.001)[0])
    if len(ones) == 0 or len(zeros) == 0:
        h_bin = _project(h_init, n // 2)
        h = np.where(h_bin > 0.5, 1.0, 0.0).astype(float)
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
    T = 0.004
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (delta < 0.004 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cool, T_min)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new % 2 != 0:
        n_new += 1
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    x = (np.arange(n) + 0.5) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _directed_pair_pass(h, rng, n_pairs, deadline):
    """Pair pass using gradient to guide pair selection."""
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    gamma = 20.0
    w = np.exp(gamma * (scores - max_s))
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = _weighted_grad(h, w)
    grad_centered = grad - np.mean(grad)
    cur_c5 = _exact_c5(h)
    improved = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if rng.random() < 0.6:
            idx = np.argsort(grad_centered)
            j = idx[rng.integers(min(30, n))]
            i = idx[-1 - rng.integers(min(30, n))]
        else:
            i = rng.integers(n)
            j = rng.integers(n)
        if i == j:
            continue
        t_opt, f_opt = _pair_optimize(h, i, j)
        if f_opt < cur_c5 - 1e-13 and abs(t_opt) > 1e-12:
            h[i] += t_opt
            h[j] -= t_opt
            h = np.clip(h, 0.0, 1.0)
            cur_c5 = f_opt
            improved += 1
            w = np.exp(gamma * (_scores(h) - float(np.max(_scores(h)))))
            sw2 = float(np.sum(w))
            if sw2 < 1e-300:
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(_scores(h)))] = 1.0
            else:
                w /= sw2
            grad = _weighted_grad(h, w)
            grad_centered = grad - np.mean(grad)
    h = _project(h, n / 2.0)
    cur_c5 = _exact_c5(h)
    return h, cur_c5, improved
