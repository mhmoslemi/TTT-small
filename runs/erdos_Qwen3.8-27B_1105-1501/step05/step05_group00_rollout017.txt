def smoothed_gradient(b, beta):
    """Compute gradient of smoothed objective F_beta(b) = (1/beta)*log(sum exp(beta*R[k]/(2n))) / ... 
    Actually let's work with G_beta = (1/beta) * log(sum exp(beta * R_k)) which approximates max R_k.
    The actual C5 = max R_k / (2n), so G_beta / (2n) approximates C5.
    dG_beta/db_j = (1/beta) * sum_k w_k * dR_k/db_j where w_k = exp(beta*R_k)/sum exp(beta*R_m)
    """
    n = len(b)
    R = signed_row_values(b)
    # For numerical stability, subtract max
    R_shifted = R - R.max()
    log_weights = R_shifted * beta - beta * R_shifted.max()  # already shifted
    # Actually: log_weights[k] = beta * R_shifted[k]
    # w_k = exp(log_weights[k]) / sum exp(log_weights)
    log_w = beta * R_shifted
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    
    # Compute B (prefix sums)
    B = np.zeros(n + 1)
    B[1:] = np.cumsum(b)
    n1 = np.arange(n)
    P = B[n1] + B[n - n1]  # P_k = B[k] + B[n-k]
    sign_P = np.sign(P)
    # sign_P[0]: when k=0, P_0 = B[0]+B[n] = 0, sign is 0. 
    # We can handle this: use 0 for sign when P=0
    
    # A_k values
    bcorr = np.correlate(b, b, mode='full')
    A = bcorr[n-1:2*n-1]
    
    # Gradient: grad[j] = (1/(2n)) * sum_k w_k * dR_k/db_j
    # dR_k/db_j = sign_P[k] * ([j<k] + [j<n-k]) - ([j<=n-k-1]*b[j+k] + [j>=k]*b[j-k])
    
    grad = np.zeros(n)
    for k in range(n):
        if w[k] < 1e-15:
            continue
        wk = w[k]
        # Prefix part: sign_P[k] * ([j<k] + [j<n-k])
        # For j from 0 to n-1:
        # [j<k] + [j<n-k]:
        #   j < k AND j < n-k: 2 (when j < min(k, n-k))
        #   j < k XOR j < n-k: 1
        #   j >= k AND j >= n-k: 0
        if k == 0:
            prefix_grad = np.zeros(n)
        else:
            prefix_grad = np.zeros(n)
            prefix_grad[:min(k, n-k)] = 2.0
            if k != n - k:  # avoid double counting
                lo = min(k, n-k)
                hi = max(k, n-k)
                if lo < hi:
                    prefix_grad[lo:hi] = 1.0
            prefix_grad *= sign_P[k]
        
        # Autocorrelation part: -[j<=n-k-1]*b[j+k] - [j>=k]*b[j-k]
        auto_grad = np.zeros(n)
        if k == 0:
            # A_0 = sum b_i^2, dA_0/db_j = 2*b_j
            auto_grad = -2.0 * b
        else:
            mk = n - k
            # [j <= n-k-1]: j from 0 to n-k-1
            if mk >= 1:
                auto_grad[:mk] = -b[k:mk+k]  # -b[j+k] for j=0..n-k-1
            # [j >= k]: j from k to n-1
            # b[j-k] for j=k..n-1 is b[0]..b[n-k-1]
            auto_grad[k:] += -b[:n-k]  # -b[j-k] for j=k..n-1
        
        grad += wk * (prefix_grad + auto_grad)
    
    grad /= (2.0 * n)
    return grad, float(np.max(R)) / (2.0 * n)

Let me be careful about the line search. Given h and direction d with sum(d)=0:
- h(alpha) = h + alpha*d
- Feasibility: 0 <= h[i] + alpha*d[i] <= 1 for all i
  - If d[i] > 0: alpha <= (1-h[i])/d[i]
  - If d[i] < 0: alpha <= h[i]/(-d[i])
  - alpha >= 0 (we only consider forward steps)
- So alpha_max = min over all i with d[i]!=0 of the appropriate bound

Then phi(alpha) = dx * max_r (S0[r] + alpha*T[r] - alpha^2*Q[r])

I want to minimize phi over alpha in [0, alpha_max].

This is a 1D problem where the objective is the max of (2n-1) downward parabolas. The minimum of the max of parabolas can be found by:
1. Grid search to get a good initial bracket
2. Golden section or ternary search to refine

Actually, since each term is a concave quadratic (downward), the max of concave functions is not necessarily concave. But in practice, the structure is nice enough that golden section works well.

import numpy as np

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def signed_row_values(b):
    n = len(b)
    B = np.zeros(n + 1)
    B[1:] = np.cumsum(b)
    bcorr = np.correlate(b, b, mode='full')
    A = bcorr[n - 1:2 * n - 1]
    k_arr = np.arange(n)
    M = n - k_arr
    P = B[k_arr] + B[n - k_arr]
    R = M + np.abs(P) - A
    return R, A, P

def c5_from_b(b):
    n = len(b)
    R, _, _ = signed_row_values(b)
    return float(np.max(R)) / (2.0 * n)

def line_search_exact(h, d, dx, max_alpha=1.0):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    alpha_hi = max_alpha
    if np.any(pos):
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_hi = min(alpha_hi, float(np.min(h[neg] / (-d[neg]))))
    alpha_hi = max(alpha_hi, 0.0)
    if alpha_hi < 1e-15:
        return 0.0, exact_c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    n_grid = 64
    grid = np.linspace(0.0, alpha_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_alpha = grid[best_i]
    best_val = obj[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b_ = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b_ = 0.0, alpha_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b_ - gr * (b_ - a)
    c2 = a + gr * (b_ - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(50):
        if b_ - a < 1e-14:
            break
        if f1 < f2:
            b_, c2, f2 = c2, c1, f1
            c1 = b_ - gr * (b_ - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b_ - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    opt_a = (a + b_) / 2.0
    opt_v = float(np.max(S0 + opt_a * T - opt_a * opt_a * Q)) * dx
    if opt_v < best_val:
        best_alpha, best_val = opt_a, opt_v
    v0 = float(np.max(S0)) * dx
    if v0 < best_val:
        best_alpha, best_val = 0.0, v0
    return best_alpha, best_val

def smoothed_grad(b, beta):
    n = len(b)
    R, A, P = signed_row_values(b)
    R_max = R.max()
    log_w = beta * (R - R_max)
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    sign_P = np.sign(P)
    grad = np.zeros(n)
    for k in range(n):
        if w[k] < 1e-14:
            continue
        wk = w[k]
        prefix = np.zeros(n)
        if k > 0:
            mn = min(k, n - k)
            mx = max(k, n - k)
            prefix[:mn] = 2.0
            if mn < mx:
                prefix[mn:mx] = 1.0
            prefix *= sign_P[k]
        auto = np.zeros(n)
        if k == 0:
            auto = -2.0 * b
        else:
            mk = n - k
            if mk >= 1:
                auto[:mk] = -b[k:k + mk]
            auto[k:] += -b[:n - k]
        grad += wk * (prefix + auto)
    grad /= (2.0 * n)
    return grad, float(R.max()) / (2.0 * n)

        A_eq = np.ones((1, n + 1))
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 500})
            if not res.success:
                continue
            p = res.x[:n]
        except Exception:
            continue
        if float(np.max(np.abs(p))) < 1e-13:
            break
        alpha, new_c5 = line_search_exact(h, p, dx, radius)
        if new_c5 < best_c5 - 1e-13:
            h_new = project_h(h + alpha * p, target)
            actual = exact_c5(h_new)
            if actual < best_c5:
                best_c5 = actual
                best_h = h_new.copy()
                h = h_new
                scores = np.correlate(h, 1.0 - h, mode='full')
    return best_h, best_c5

def structured_start(n, freq=3, phase=0.0):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_h(h, n / 2.0)

def block_start(n, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, n // 3 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return project_h(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = structured_start(n0, freq=3, phase=0.5)

    h = project_h(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)

    # Phase 1: Quick multi-start exploration (12% time)
    d1 = t0 + budget_s * 0.12
    candidates = [h.copy()]
    for fr in [1, 2, 3, 4, 5, 7]:
        for ph in [0, 1.0, 2.0]:
            candidates.append(structured_start(n0, fr, ph))
    for s in range(5):
        candidates.append(block_start(n0, seed + 100 + s * 7))

    for cand in candidates:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.012)
        h_g, c5_g = grad_descent_step(cand, sub_dl, beta=30)
        b_back = (1.0 + 2.0 * h_g - 1.0) / 2.0
        b_back = project_h(b_back, n0 / 2.0)
        c5_check = exact_c5(b_back)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = b_back.copy()

    # Phase 2: Smoothed gradient descent on best (15% time)
    d2 = t0 + budget_s * 0.27
    h_gd, c5_gd = grad_descent_step(best_h, d2, beta=40)
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd.copy()

    # Phase 3: LP refinement (15% time)
    d3 = t0 + budget_s * 0.42
    h_lp, c5_lp = lp_refine(best_h, d3, radius=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 4: Pair refinement (20% time)
    d4 = t0 + budget_s * 0.62
    h_pr, c5_pr = pair_refine(best_h, d4)
    if c5_pr < best_c5:
        best_c5 = c5_pr
        best_h = h_pr.copy()

    # Phase 5: Quad-pair refinement (10% time)
    d5 = t0 + budget_s * 0.72
    h_qp, c5_qp = quad_pair_refine(best_h, d5)
    if c5_qp < best_c5:
        best_c5 = c5_qp
        best_h = h_qp.copy()

    # Phase 6: Multi-resolution search (15% time)
    d6 = t0 + budget_s * 0.87
    best_n = len(best_h)
    tried = set([best_n])
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20, -24, 24]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = grad_descent_step(h_r, sub_dl, beta=30)
        if c5_r < best_c5 - 1e-6 and time.time() < d6 - 10:
            # Further refine with pairs
            sub_dl2 = min(d6, time.time() + budget_s * 0.01)
            h_r, c5_r = pair_refine(h_r, sub_dl2)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final refinement cycle (10% time)
    d7 = t0 + budget_s * 0.95
    # Alternating: gradient, pair, gradient
    while time.time() < d7:
        rem = d7 - time.time()
        if rem < 3:
            break
        t_chunk = min(rem * 0.3, 15.0)
        sub1 = time.time() + t_chunk
        h_tmp, c5_tmp = grad_descent_step(best_h, sub1, beta=60)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_tmp.copy()
        sub2 = time.time() + min(t_chunk, d7 - time.time())
        h_tmp, c5_tmp = pair_refine(best_h, sub2)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_tmp.copy()

    # Phase 8: Final polish (5% time)
    h_final, c5_final = lp_refine(best_h, deadline, radius=0.05)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    # Verify and return
    best_h = project_h(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, final_c5, len(best_h)
