def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))

        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

    # Softmax weights
    max_s = np.max(scores)
    w = np.exp(alpha * (scores - max_s))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # Lag 0: index n-1 in scores
    w0 = w[n-1]
    grad += w0 * dx * (1.0 - 2.0 * h)
    
    # Positive lags d=1..n-1: score index n-1+d
    for d in range(1, n):
        wd = w[n-1+d]
        if wd < 1e-15:
            continue
        m = n - d
        grad[d:] += wd * dx * (1.0 - h[:m])
        grad[:m] -= wd * dx * h[d:]
    
    # Negative lags -d (d=1..n-1): score index n-1-d
    for d in range(1, n):
        wd = w[n-1-d]
        if wd < 1e-15:
            continue
        m = n - d
        grad[:m] += wd * dx * (1.0 - h[d:])
        grad[d:] -= wd * dx * h[:m]
    
    return grad

def _pair_line_search(h, p, q, n_eval=25):
    """
    Move mass from q to p: h[p] += t, h[q] -= t
    Find t that minimizes C5.
    Returns (new_h, new_c5, improved)
    """
    n = len(h)
    t_min = max(-h[p], h[q] - 1.0)
    t_max = min(1.0 - h[p], h[q])
    if t_max - t_min < 1e-14:
        return h, _exact_c5(h), False
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_min, t_max
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    h_c = h.copy()
    h_d = h.copy()
    h_c[p] += c; h_c[q] -= c
    h_d[p] += d; h_d[q] -= d
    f_c = _exact_c5(h_c)
    f_d = _exact_c5(h_d)
    
    for _ in range(n_eval):
        if b - a < 1e-13:
            break
        if f_c < f_d:
            b = d
            f_d = f_c
            d = c
            c = b - gr * (b - a)
            h_c[p] += c - (h_c[p] - h[p])  # reset properly
            # Actually let me just recompute
            _h_c = h.copy()
            _h_c[p] += c; _h_c[q] -= c
            f_c = _exact_c5(_h_c)
        else:
            a = c
            f_c = f_d
            c = d
            d = a + gr * (b - a)
            _h_d = h.copy()
            _h_d[p] += d; _h_d[q] -= d
            f_d = _exact_c5(_h_d)
    
    t_opt = (a + b) / 2.0
    if abs(t_opt) < 1e-14:
        return h, _exact_c5(h), False
    
    h_new = h.copy()
    h_new[p] += t_opt
    h_new[q] -= t_opt
    # Clip for safety
    h_new[p] = np.clip(h_new[p], 0.0, 1.0)
    h_new[q] = np.clip(h_new[q], 0.0, 1.0)
    
    new_c5 = _exact_c5(h_new)
    old_c5 = _exact_c5(h)
    if new_c5 < old_c5 - 1e-14:
        return h_new, new_c5, True
    return h, old_c5, False

def _gradient_pairs(h, scores, alpha, topk=20):
    """Find top-k gradient-guided pair transfers."""
    n = len(h)
    grad = _weighted_grad(h, scores, alpha)
    # Direction to decrease weighted obj: increase coords with negative grad, decrease coords with positive grad
    # Best pair: p with most negative grad, q with most positive grad
    # grad[p] - grad[q] should be most negative for the move p<-=q (increase p, decrease q)
    # Direction d = e_p - e_q, directional deriv = grad[p] - grad[q]
    # We want grad[p] - grad[q] < 0, i.e., grad[p] < grad[q]
    # So: p = argmin grad (most negative), q = argmax grad (most positive)
    
    pairs = []
    neg_idx = np.argsort(grad)[:n//3]  # candidates for p (increase)
    pos_idx = np.argsort(grad)[::-1][:n//3]  # candidates for q (decrease)
    
    for pi in neg_idx:
        for qi in pos_idx:
            if pi != qi:
                pairs.append((int(pi), int(qi)))
    
    # Deduplicate and sort by expected improvement
    seen = set()
    unique_pairs = []
    for p, q in pairs:
        if (p, q) not in seen:
            seen.add((p, q))
            unique_pairs.append((p, q, grad[q] - grad[p]))  # positive = expected improvement
    
    unique_pairs.sort(key=lambda x: -x[2])
    return [tuple(x[:2]) for x in unique_pairs[:topk]]

def _pair_optimize(h, deadline, n_top=15, n_random=35, alpha=5.0, max_passes=200):

def _binary_sa(h_start, n, deadline, max_iter=20000):
    """Binary simulated annealing."""
    target = n // 2
    if n % 2 != 0:
        target = n // 2
    h = np.zeros(n)
    pos = np.random.choice(n, size=target, replace=False)
    h[pos] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.005
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(max_iter, 1))
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            if np.random.random() < np.exp((cur_c5 - new_c5) / T):
                cur_c5 = new_c5
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                h[i] = 1.0
                h[j] = 0.0
        T *= cool
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    h = (vals < 0.5).astype(float)
    return _project(h, n / 2.0)

def _min_1d_lp(b, a, t_min, t_max, n_iter=60):
    if t_max - t_min < 1e-15:
        return t_min, float(np.max(b))
    lo = float(np.max(np.minimum(b + a * t_max, b + a * t_min)))
    hi = float(np.max(np.maximum(b + a * t_max, b + a * t_min)))

        upper = t_max

            s = b + a * mid_t
            if float(np.max(s)) <= mid:

def _pair_transfer(h, p, q, n_eval=25):
    n = len(h)
    hp, hq = h[p], h[q]
    t_min = max(-hp, hq - 1.0)
    t_max = min(1.0 - hp, hq)
    if t_max - t_min < 1e-13:
        return h, _exact_c5(h), False
    A = _correlation(h)
    dx = 2.0 / n
    b = A * dx
    slopes = _pair_slopes(h, p, q)
    best_t = 0.0
    best_v = float(np.max(b))
    t_opt, v_opt = _min_1d_lp(b, slopes, t_min, t_max)
    if v_opt < best_v:
        best_v = v_opt
        best_t = t_opt
    h_c = h.copy()
    h_c[p] = np.clip(hp + t_opt, 0.0, 1.0)
    h_c[q] = np.clip(hq - t_opt, 0.0, 1.0)
    actual = _exact_c5(h_c)
    if actual < _exact_c5(h) - 1e-14:
        return h_c, actual, True
    return h, _exact_c5(h), False

def _grad_pair_direct(h):
    n = len(h)
    dx = 2.0 / n
    A = _correlation(h)
    scores = A * dx
    c5 = float(np.max(scores))
    active = np.where(scores > c5 - 1e-12)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    w = np.float64([0.0]) * np.zeros(2 * n - 1)
    for idx in active:
        d = int(idx - (n - 1))
        dd = abs(d)
        if dd == 0:
            w += (1.0 - 2.0 * h)
        else:
            if d > 0:
                m = n - dd
                w[dd:] += (1.0 - h[:m])
                w[:m] -= h[dd:]
            else:
                m = n - dd
                w[:m] += (1.0 - h[dd:])
                w[dd:] -= h[:m]
    g = w * dx
    return g, scores

def _pair_optimize(h, deadline, n_top=10, n_rand=30, alpha_sw=5.0, max_passes=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_passes):
        if time.time() > deadline:
            break
        if stall > 35:
            break
        g, scores = _grad_pair_direct(h)
        max_s = np.max(scores)
        w_e = np.exp(-alpha_sw * (max_s - scores))
        w_n = w_e / np.sum(w_e)
        wg = np.zeros(n)
        for idx in range(2 * n - 1):
            wi = w_n[idx]
            if wi < 1e-15:
                continue
            d = idx - (n - 1)
            dd = abs(d)
            if dd == 0:
                wg += wi * (1.0 - 2.0 * h)
            else:
                if d > 0:
                    m = n - dd
                    wg[dd:] += wi * (1.0 - h[:m])
                    wg[:m] -= wi * h[dd:]
                else:
                    m = n - dd
                    wg[:m] += wi * (1.0 - h[dd:])
                    wg[dd:] -= wi * h[:m]
        wg *= (2.0 / n)
        improved = False
        neg_idx = np.argsort(wg)[:n // 3]
        pos_idx = np.argsort(wg)[::-1][:n // 3]
        all_pairs = []

def _binary_sa(n, deadline, max_iter=15000):
    target = n // 2
    h = np.zeros(n)
    pos = np.random.choice(n, size=target, replace=False)
    h[pos] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(max_iter, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        n_c5 = _exact_c5(h)
        if n_c5 <= cur_c5 or np.random.random() < np.exp((cur_c5 - n_c5) / T):
            cur_c5 = n_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _quasi(n, alpha):
    x = np.arange(n) / n
    return _project(((x * alpha) % 1.0 < 0.5).astype(float), n / 2.0)

    return h, c5
