def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

                d = np.zeros(n)
                d[i:i + bs] = 1.0
                d[j:j + bs] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.3)
                if alpha != 0.0 and c5_try < cur_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved_this_pass = True
                    stall = 0
        if not improved_this_pass:
            stall += 1
            if stall > 3:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _block_pattern(n, rng, n_blocks):
    """Generate a pattern with contiguous blocks of 1s."""
    boundaries = sorted(rng.integers(1, n, size=n_blocks - 1).tolist())
    boundaries = [0] + boundaries + [n]
    h = np.zeros(n)
    for bi in range(n_blocks):
        start = boundaries[bi]
        end = boundaries[bi + 1]
        length = end - start
        if rng.random() < 0.5:
            h[start:start + length // 2] = 1.0
    h = _project(h, n / 2.0)
    return h

def _pair_local_search(h, deadline, n_tries=3000, radius=0.2):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(4242)
    stall = 0
    while time.time() < deadline:
        if stall > 400:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if alpha != 0.0 and c5_try < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.01, 0.5)
        else:
            stall += 1
            if stall % 80 == 0:
                radius *= 0.92
    return best_h, best_c5

def _anneal_refine(h, deadline, n_restarts=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    for _rst in range(n_restarts):
        if time.time() > deadline:
            break
        cur = h.copy()
        cur_c5 = _exact_c5(cur)
        T = 0.015
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for _step in range(2000):
            if time.time() > deadline:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            d = np.zeros(n)
            d[ones[i]] = -1.0
            d[zeros[j]] = 1.0
            alpha, c5_try = _line_search(cur, d, dx, 0.4)
            delta = c5_try - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur = cur + alpha * d
                cur = _project(cur, target)
                cur_c5 = _exact_c5(cur)
                ones = np.where(cur > 0.5)[0]
                zeros = np.where(cur <= 0.5)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
            T *= 0.997
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: SLSQP refinement on parent (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _slsqp_refine(h_init, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Block move local search on best (15%)
    d2 = t0 + budget_s * 0.27
    h2 = best_h.copy()
    h2, c5_2 = _block_local_search(h2, d2, block_sizes=[1, 2, 3, 4, 6, 8, 12], max_passes=12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (25%)
    d3 = t0 + budget_s * 0.52
    n_curr = len(best_h)
    offsets = [-6, 6, -12, 12, -18, 18, -24, 24, -32, 32, -4, 4, -8, 8, -16, 16, -40, 40, -48, 48]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        if time.time() > sub_dl:
            continue
        h_r, c5_r = _slsqp_refine(h_r, sub_dl, taus=[0.3, 0.15, 0.08, 0.04])
        rem = d3 - time.time()
        if rem > 6 and c5_r < best_c5 + 0.004:
            sub_dl2 = min(d3, time.time() + max(3, budget_s * 0.008))
            h_r, c5_r = _block_local_search(h_r, sub_dl2, block_sizes=[1, 2, 3, 5, 8], max_passes=6)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Random and pattern starts with SLSQP (20%)
    d4 = t0 + budget_s * 0.72
    n_starts = 10
    for trial in range(n_starts):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        if trial % 3 == 0:
            n_try = n0 + int(rng.integers(-10, 11))
            n_try = max(48, min(256, n_try))
            h_r = _random_binary(n_try, rng)
        elif trial % 3 == 1:
            n_try = n0 + int(rng.integers(-8, 9))
            n_try = max(48, min(256, n_try))
            h_r = _block_pattern(n_try, rng, int(rng.integers(3, 9)))
        else:
            n_try = n0
            h_r = best_h + rng.normal(0, 0.06, n_try)
            h_r = _project(h_r, n_try / 2.0)
        h_r, c5_r = _slsqp_refine(h_r, sub_dl, taus=[0.5, 0.2, 0.1, 0.05])
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Intensive local polish on best (18%)
    d5 = t0 + budget_s * 0.90
    h5 = best_h.copy()
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h5, c5_5 = _slsqp_refine(h5, sub_dl, taus=[0.1, 0.05, 0.02, 0.01])
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h5 = best_h.copy()
    h5, c5_5 = _block_local_search(h5, sub_dl, block_sizes=[1, 2, 3, 5, 8], max_passes=10)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h5 = best_h.copy()
    h5, c5_5 = _pair_local_search(h5, sub_dl, n_tries=4000, radius=0.15)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    sub_dl = min(d5, time.time() + budget_s * 0.03)
    h5 = best_h.copy()
    h5, c5_5 = _slsqp_refine(h5, sub_dl, taus=[0.05, 0.02, 0.01])
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final polish (10%)
    h6 = best_h.copy()
    h6, c5_6 = _pair_local_search(h6, deadline, n_tries=5000, radius=0.10)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    h6 = best_h.copy()
    h6, c5_6 = _slsqp_refine(h6, deadline, taus=[0.02, 0.01])
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation and projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
