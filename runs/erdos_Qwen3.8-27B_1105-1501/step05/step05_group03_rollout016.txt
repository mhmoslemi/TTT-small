```python
import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_at_lag(h, d):
    """Compute overlap score at lag d"""
    n = len(h)
    dx = 2.0 / n
    if d >= 0:
        m = n - d
        if m <= 0:
            return 0.0
        return dx * np.dot(h[:m], 1.0 - h[d:])
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return dx * np.dot(h[g:], 1.0 - h[:m])

def _score_grad(h, d):
    """Gradient of overlap score at lag d w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0:
            return grad
        if d == 0:
            return dx * (1.0 - 2.0 * h)
        grad[:m] += (1.0 - h[d:])
        grad[d:] -= h[:m]
    else:
        g = -d
        m = n - g
        if m <= 0:
            return grad
        grad[g:] += (1.0 - h[:m])
        grad[:m] -= h[g:]
    return dx * grad

def _scores_all(h):
    """Compute all 2n-1 overlap scores"""
    n = len(h)
    scores = np.zeros(2 * n - 1)
    for d in range(n):
        scores[d] = _score_at_lag(h, d)
    for g in range(1, n):
        scores[n + g - 1] = _score_at_lag(h, -g)
    return scores

def _true_c5(h):
    """Exact C5 evaluation using np.correlate"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _active_lags(h, k):
    """Get the k largest lags by score"""
    n = len(h)
    lags = np.arange(-(n-1), n)
    scores = np.array([_score_at_lag(h, d) for d in lags])
    idx = np.argsort(-scores)[:k]
    return lags[idx]

def _resample(h, n_new):
    """Resample h to n_new points"""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

# SLSQP epigraph functions (top-level)
def _epig_obj(x, ctx):
    n_vars = ctx[0]
    return x[-1]

def _epig_obj_grad(x, ctx):
    n_vars = ctx[0]
    g = np.zeros(n_vars + 1)
    g[-1] = 1.0
    return g

def _epig_eq_func(x, ctx):
    n_vars = ctx[0]
    dx = 2.0 / n_vars
    return np.sum(x[:n_vars]) * dx - 1.0

def _epig_eq_grad(x, ctx):
    n_vars = ctx[0]
    dx = 2.0 / n_vars
    g = np.zeros(n_vars + 1)
    g[:n_vars] = dx
    return g

def _epig_ineq_func(x, ctx):
    n_vars = ctx[0]
    active_lags = ctx[1]
    h = x[:n_vars]
    T = x[-1]
    vals = np.zeros(len(active_lags))
    for i, d in enumerate(active_lags):
        vals[i] = T - _score_at_lag(h, d)
    return vals

def _epig_ineq_jac(x, ctx):
    n_vars = ctx[0]
    active_lags = ctx[1]
    h = x[:n_vars]
    J = np.zeros((len(active_lags), n_vars + 1))
    for i, d in enumerate(active_lags):
        J[i, :n_vars] = -_score_grad(h, d)
        J[i, n_vars] = 1.0
    return J

def _solve_epigraph(h0, active_lags, t_init, maxiter=100):
    """Solve the SLSQP epigraph problem"""
    n = len(h0)
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    ctx = (n, active_lags)
    constraints = [
        {'type': 'eq', 'fun': _epig_eq_func, 'jac': _epig_eq_grad},
        {'type': 'ineq', 'fun': _epig_ineq_func, 'jac': _epig_ineq_jac}
    ]
    try:
        res = minimize(_epig_obj, x0, args=ctx, jac=_epig_obj_grad,
                      bounds=bounds, constraints=constraints,
                      method='SLSQP', options={'maxiter': maxiter, 'disp': False})
        return res
    except Exception:
        return None

def _epigraph_round(h0, deadline, k_active=40, max_subrounds=20, k_step=4):
    """Multiple rounds of SLSQP with dynamic active set"""
    n = len(h0)
    h_best = h0.copy()
    c5_best = _true_c5(h0)
    
    for rnd in range(max_subrounds):
        if time.time() > deadline:
            break
        # Compute active set
        scores = _scores_all(h_best)
        max_score = np.max(scores)
        # Top k_active lags
        k_eff = min(k_active, 2*n-1)
        idx = np.argsort(-scores)[:k_eff]
        lags = np.arange(-(n-1), n)
        active = lags[idx]
        
        # Solve
        t_init = max_score + 0.01
        res = _solve_epigraph(h_best, active, t_init, maxiter=80)
        
        if res is not None:
            h_try = res.x[:n]
            h_try = _project(h_try, n / 2.0)
            c5_try = _true_c5(h_try)
            if c5_try < c5_best - 1e-10:
                h_best = h_try
                c5_best = c5_try
            else:
                # Try with expanded active set
                k_eff2 = min(k_active + k_step, 2*n-1)
                idx2 = np.argsort(-scores)[:k_eff2]
                active2 = lags[idx2]
                res2 = _solve_epigraph(h_best, active2, t_init, maxiter=80)
                if res2 is not None:
                    h_try2 = res2.x[:n]
                    h_try2 = _project(h_try2, n / 2.0)

                    if c5_try2 < c5_best - 1e-10:

        # Check convergence
        scores_new = _scores_all(h_best)
        if np.max(scores_new) - c5_best < 1e-8:
            break
    
    return h_best, c5_best

def _subgrad_momentum(h0, deadline, n_steps=200, beta_start=20, beta_end=200, step_init=0.03):
    """Subgradient with momentum and adaptive softmax"""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    c5_cur = _true_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    step = step_init
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        scores = _scores_all(h)
        max_s = np.max(scores)
        # Softmax weights
        w = np.exp((scores - max_s) * beta)
        sw = np.sum(w)
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        
        # Weighted gradient
        lags = np.arange(-(n-1), n)
        grad = np.zeros(n)
        for i, d in enumerate(lags):
            if w[i] > 1e-10:
                grad += w[i] * _score_grad(h, d)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Nesterov-like momentum
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _true_c5(h_new)
        
        if c5_new < c5_cur - 1e-14:
            h = h_new
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            step = min(step * 1.03, 0.1)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    
    return h_best, c5_best

def _binary_swap_search(n, deadline, h_init, rng, n_iters=50000):
    """Binary local search with swaps"""
    target = n // 2
    # Round to binary
    h = np.zeros(n)
    sorted_idx = np.argsort(-h_init)
    h[sorted_idx[:target]] = 1.0
    c5_cur = _true_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    T = 0.02
    T_min = 0.0005
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = _true_c5(h)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    
    return h_best, c5_best
