            continue
        grad += w[g] * _U_grad_single(b, g)
    grad /= (2.0 * n)
    return grad, U

def _U_grad_single(b, g):
    """Gradient of U_g with respect to b."""
    n = len(b)
    grad = np.zeros(n)
    if g == 0:
        grad[:] = -2.0 * b
        return grad
    # |E_g| term
    lo = min(g, n - g)
    hi = max(g, n - g)
    ie = np.zeros(n)
    ie[:lo] = 2.0
    ie[lo:hi] = 1.0
    Eg = float(b[:g].sum() + b[:n-g].sum())
    sgn = 1.0 if Eg > 0 else (-1.0 if Eg < 0 else 0.0)
    grad += sgn * ie
    # -A_g term
    grad[0:n-g] -= b[g:n]
    grad[g:n] -= b[0:n-g]
    return grad

def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < (1.0 - 1e-10)
        else: mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    """Exact line search for direction d from h."""
    n = len(h)
    d = d - np.mean(d)  # ensure zero mean
    
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, -float(np.min(-h[neg] / d[neg])))
    # Wait, for negative d: h + t*d >= 0 means t <= h/(-d) for d < 0

    # Actually if d < 0, then h + t*d >= 0 requires t <= h/(-d)
    # So the upper bound from negative d entries is: t <= min(h[neg] / (-d[neg]))
    # Let me redo this:
    # For d[i] > 0: h[i] + t*d[i] <= 1 → t <= (1-h[i])/d[i]
    # For d[i] < 0: h[i] + t*d[i] >= 0 → t >= -h[i]/d[i] = h[i]/(-d[i])
    # Also t >= 0 (or t >= -trust_radius if we allow backward steps)

        return 0.0, _exact_c5(h)
    
    # Grid search + golden section
    n_grid = 48
    grid = np.linspace(t_min_allowed, t_max_allowed, n_grid)

    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))

            f2 = float(np.max(S0 + c2 * T - c2**2 * C)) * dx

Hmm wait, I realize there's an issue with my C1 computation. Let me re-examine.

`np.correlate(b, w, mode='full')` has length 2n-1.
The formula is: for the 'full' output, the element at index k is:
result[k] = sum_i b[i] * w[i + k - (len(w)-1)]

where the sum is over all i where both b[i] and w[i+k-(len(w)-1)] are defined.

With len(w) = n: result[k] = sum_i b[i] * w[i + k - (n-1)]

For k = n-1-j (where j ranges from 0 to n-1, so k ranges from n-1 to 0):
result[n-1-j] = sum_i b[i] * w[i - j]

The valid range of i: we need 0 <= i < n and 0 <= i-j < n, so j <= i < j+n and 0 <= i < n.
Since j >= 0: i ranges from j to n-1.
result[n-1-j] = sum_{i=j}^{n-1} b[i] * w[i-j] = sum_{g=0}^{n-1-j} b[j+g] * w[g] = C1[j] ✓

So C1 = result[n-1::-1] where result = np.correlate(b, w, 'full').
C1[0] = result[n-1], C1[1] = result[n-2], ..., C1[n-1] = result[0].

That's `result[n-1::-1]` which gives a length-n array. ✓

For C2: `np.convolve(w, b, mode='full')` has length 2n-1.
conv[k] = sum_i w[i] * b[k-i] (valid for 0 <= i < n and 0 <= k-i < n)
For k = j (0 <= j < n):
conv[j] = sum_{i=max(0,j-n+1)}^{min(j, n-1)} w[i] * b[j-i]
Since j < n: j-n+1 < 0, so lower bound is 0.
conv[j] = sum_{i=0}^{j} w[i] * b[j-i] = C2[j] ✓

So C2 = np.convolve(w, b, 'full')[:n]. ✓

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < (1.0 - 1e-10)
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - np.mean(d)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)

    obj = np.max(vals, axis=1) * dx

    best_val = obj[best_i]
    if best_i > 0 and best_i < n_grid - 1:

def _compute_U(b):
    n = len(b)
    bS = np.correlate(b, b, mode='full')
    A = bS[n - 1:n - 1 + n]
    B = np.zeros(n + 1)
    B[1:] = np.cumsum(b)
    M = (n - np.arange(n)).astype(float)
    E = B[0:n] + B[1:n + 1][::-1]
    U = M + np.abs(E) - A
    return U, E


def _lse_gradient(h, tau):
    n = len(h)
    b = 2.0 * h - 1.0
    U, E = _compute_U(b)
    scaled = tau * U / (2.0 * n)
    scaled_max = np.max(scaled)
    exp_vals = np.exp(scaled - scaled_max)
    w = exp_vals / np.sum(exp_vals)
    sgn_E = np.where(E > 0, 1.0, np.where(E < 0, -1.0, 0.0))
    S = w * sgn_E
    SC = np.concatenate([[0.0], np.cumsum(S)])
    total_S = S.sum()
    grad = np.zeros(n)
    for j in range(n):
        term1 = total_S - SC[j + 1]
        idx2 = n - j
        if 1 <= idx2 <= n:
            term1 += total_S - SC[idx2]
        grad[j] = term1
    corr_bw = np.correlate(b, w, mode='full')
    C1 = corr_bw[n - 1::-1]
    conv_wb = np.convolve(w, b, mode='full')
    C2 = conv_wb[:n]
    grad -= (C1 + C2)
    grad /= (2.0 * n)
    grad -= np.mean(grad)
    return grad


def _subgradient(h):
    n = len(h)
    b = 2.0 * h - 1.0
    U, E = _compute_U(b)
    g_active = int(np.argmax(U))
    grad = np.zeros(n)
    if g_active == 0:
        grad[:] = -2.0 * b
    else:
        g = g_active
        lo = min(g, n - g)
        hi = max(g, n - g)
        ie = np.zeros(n)
        ie[:lo] = 2.0
        ie[lo:hi] = 1.0
        Eg = float(E[g])
        sgn = 1.0 if Eg > 0 else (-1.0 if Eg < 0 else 0.0)
        grad += sgn * ie
        grad[0:n - g] -= b[g:n]
        grad[g:n] -= b[0:n - g]
    grad /= (2.0 * n)
    grad -= np.mean(grad)
    return grad

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    h1, c5_1 = _lse_descent(best_h, d1, tau_init=12.0, tau_final=60.0, rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP polish (10% time)
    d2 = t0 + budget_s * 0.22
    h2, c5_2 = _lp_polish(best_h, d2, radius_init=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary 2-swap SA (18% time)
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _binary_2swap_sa(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LSE descent from binary + LP (18% time)
    d4 = t0 + budget_s * 0.58
    h4a, c5_4a = _lse_descent(best_h, min(d4, t0 + budget_s * 0.50),
                              tau_init=8.0, tau_final=50.0, rng=rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_polish(best_h, d4, radius_init=0.08)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-resolution (22% time)
    d5 = t0 + budget_s * 0.80
    offsets = [-6, 6, -4, 4, -8, 8, -10, 10, -12, 12, -14, 14, -18, 18, -22, 22]
    tried = set()
    per_res = budget_s * 0.014
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + per_res)
        h_r, c5_r = _lse_descent(h_r, sub_dl, tau_init=10.0, tau_final=60.0, rng=rng)
        if time.time() < d5 - 2:
            sub_dl2 = min(d5, time.time() + per_res * 0.6)
            h_r, c5_r = _lp_polish(h_r, sub_dl2, radius_init=0.08)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Aggressive perturbation + re-opt (10% time)
    d6 = t0 + budget_s * 0.90
    scale = 0.03
    for attempt in range(300):
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.005)
        h_t, c5_t = _lse_descent(h_pert, sub_dl, tau_init=8.0, tau_final=40.0, rng=rng)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.2)

    # Phase 7: Pair refinement (6% time)
    d7 = t0 + budget_s * 0.96
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=50)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final polish
    h_f, c5_f = _lp_polish(best_h, deadline, radius_init=0.04)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
