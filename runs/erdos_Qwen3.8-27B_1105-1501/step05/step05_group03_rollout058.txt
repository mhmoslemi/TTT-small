            cross_mask = rng.random(q) < CR
            trial[cross_mask] = mutant[cross_mask]
            trial = np.clip(trial, y_lo, y_hi)

                    best_h = np.clip(0.5 + B @ trial, 0, 1)

def _pair_polish(h, deadline, n_passes=10):

        return np.zeros(n)

        return np.ones(n)

        else:
            hi = mid

    x = np.clip(v - thr, 0.0, 1.0)

        else:

    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None:
            stall += 1
            radius *= 0.6
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search(h, p, target)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.4)
        else:
            stall += 1
            radius *= 0.65
            if radius < radius_min:
                break
    return best_h, best_c5

def _smoothed_grad(h, deadline, beta_start=20, beta_end=500, n_steps=300, step_init=0.05):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores)); w[idx] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.02, 0.15)
        else:
            v = 0.5 * v
            step *= 0.82
            if step < 1e-7:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
        h = _project(h, n / 2.0)
    return h

def _subspace_de(h_init, deadline, q=30, n_pop=40, max_gen=150, F=0.7, CR=0.8, rng=None):
    n = len(h_init)
    target = n / 2.0
    x = np.arange(n) / n
    B = np.zeros((n, q))
    k_sin = min(q // 2, 15)
    for k in range(1, k_sin + 1):
        B[:, k-1] = np.cos(2 * np.pi * k * x)
        B[:, k-1] -= B[:, k-1].mean()
    for j in range(k_sin, q):
        bs = int(rng.integers(max(3, n // 12), n // 2 + 1))
        start = int(rng.integers(0, max(1, n - 2 * bs + 1)))
        amp = 1.0 / np.sqrt(2.0 * bs)
        B[start:start+bs, j] = amp
        B[start+bs:start+2*bs, j] = -amp
        B[:, j] -= B[:, j].mean()
    v = h_init - 0.5
    y_init, _, _, _ = np.linalg.lstsq(B, v, rcond=None)
    y_lo, y_hi = -3.0, 3.0
    def obj_fn(y):
        h_try = np.clip(0.5 + B @ y, 0.0, 1.0)
        h_try = _project(h_try, target)
        return _exact_c5(h_try)
    pop_y = np.zeros((n_pop, q))
    pop_y[0] = y_init
    pop_y[1] = y_init + 0.3 * rng.standard_normal(q)
    pop_y[2] = y_init + 0.6 * rng.standard_normal(q)
    if n_pop > 3:
        pop_y[3:] = y_init + 1.5 * rng.standard_normal((n_pop - 3, q))
    pop_y = np.clip(pop_y, y_lo, y_hi)
    pop_obj = np.array([obj_fn(pop_y[i]) for i in range(n_pop)])
    best_i = int(np.argmin(pop_obj))
    best_obj = pop_obj[best_i]
    best_h = np.clip(0.5 + B @ pop_y[best_i], 0, 1)
    best_h = _project(best_h, target)
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        for i in range(n_pop):
            a, b, c = rng.choice(n_pop, 3, replace=False)
            mutant = pop_y[a] + F * (pop_y[b] - pop_y[c])
            trial = pop_y[i].copy()
            mask = rng.random(q) < CR
            trial[mask] = mutant[mask]
            trial = np.clip(trial, y_lo, y_hi)
            try:
                t_obj = obj_fn(trial)
            except Exception:
                continue
            if t_obj <= pop_obj[i]:
                pop_y[i] = trial
                pop_obj[i] = t_obj
                if t_obj < best_obj:
                    best_obj = t_obj
                    best_h = np.clip(0.5 + B @ trial, 0, 1)
                    best_h = _project(best_h, target)
    return best_h, best_obj

def _coordinate_polish(h, deadline, n_cycles=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(28):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy(); h1[j] = m1; h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy(); h2[j] = m2; h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy(); h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h, deadline, n_passes=6, window=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if window is None:
        window = max(8, n // 8)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            k_start = max(0, j - window)
            k_end = min(n, j + window + 1)
            for k in range(k_start, k_end):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        a_max = min(h[j], 1.0 - h[k])
                    else:
                        a_max = min(h[k], 1.0 - h[j])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy(); h1[j] += sign*m1; h1[k] -= sign*m1
                        h1 = np.clip(h1, 0, 1)
                        f1 = _exact_c5(h1)
                        h2 = h.copy(); h2[j] += sign*m2; h2[k] -= sign*m2
                        h2 = np.clip(h2, 0, 1)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy(); h_try[j] += sign*a_opt; h_try[k] -= sign*a_opt
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(n, deadline, rng, n_iters=15000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    def check_time(frac):
        return time.time() < t0 + budget_s * frac
    
    # Phase 1: LP from initial
    if check_time(0.10):
        dl = t0 + budget_s * 0.10
        h1, c5_1 = _lp_optimize(h_start, dl, radius_init=0.2, max_iters=5000, radius_min=0.0003)
        if c5_1 < best_c5:
            best_c5 = c5_1; best_h = h1.copy()
    
    # Phase 2: Smoothed gradient + LP
    if check_time(0.22):
        dl = t0 + budget_s * 0.22
        h2, c5_2 = _smoothed_grad(best_h, dl, beta_start=15, beta_end=400, n_steps=400, step_init=0.05)
        if c5_2 < best_c5:
            best_c5 = c5_2; best_h = h2.copy()
        if check_time(0.28):
            dl2 = t0 + budget_s * 0.28
            h2b, c5_2b = _lp_optimize(best_h, dl2, radius_init=0.08, max_iters=3000, radius_min=0.0003)
            if c5_2b < best_c5:
                best_c5 = c5_2b; best_h = h2b.copy()
    
    # Phase 3: Subspace DE
    if check_time(0.36):
        dl = t0 + budget_s * 0.36
        h3, c5_3 = _subspace_de(best_h, dl, q=30, n_pop=45, max_gen=120, F=0.7, CR=0.85, rng=rng)
        if c5_3 < best_c5:
            best_c5 = c5_3; best_h = h3.copy()
        if check_time(0.40):
            dl3b = t0 + budget_s * 0.40
            h3b, c5_3b = _lp_optimize(best_h, dl3b, radius_init=0.06, max_iters=2000, radius_min=0.0005)
            if c5_3b < best_c5:
                best_c5 = c5_3b; best_h = h3b.copy()
    
    # Phase 4: Multi-resolution (higher res)
    if check_time(0.58):
        dl4 = t0 + budget_s * 0.58
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
        for n_try in [144, 160, 192]:
            if time.time() > dl4:
                break
            h_hi = _resample(best_h, n_try)
            alpha = alphas[(n_try - 144) // 16 % len(alphas)]
            h_qp = _quasi_init(n_try, alpha, rng)
            c5_hi = _exact_c5(h_hi)
            c5_qp = _exact_c5(h_qp)
            h_cur = h_qp if c5_qp < c5_hi else h_hi
            sub_dl = min(dl4, time.time() + budget_s * 0.05)
            h_opt, c5_opt = _lp_optimize(h_cur, sub_dl, radius_init=0.18, max_iters=3000, radius_min=0.002)
            if c5_opt < best_c5:
                best_c5 = c5_opt; best_h = h_opt.copy(); best_n = n_try
            if time.time() < dl4:
                sub_dl2 = min(dl4, time.time() + budget_s * 0.025)
                h_sg, c5_sg = _smoothed_grad(h_opt, sub_dl2, beta_start=25, beta_end=350, n_steps=200, step_init=0.04)
                if c5_sg < best_c5:
                    best_c5 = c5_sg; best_h = h_sg.copy(); best_n = n_try
                if time.time() < dl4:
                    sub_dl3 = min(dl4, time.time() + budget_s * 0.02)
                    h_f, c5_f = _lp_optimize(best_h, sub_dl3, radius_init=0.04, max_iters=1500, radius_min=0.001)
                    if c5_f < best_c5:
                        best_c5 = c5_f; best_h = h_f.copy(); best_n = n_try
    
    # Phase 5: Diversification via perturbation + binary SA
    if check_time(0.70):
        dl5 = t0 + budget_s * 0.70
        scale = 0.025
        for attempt in range(250):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.008)
            h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.05, max_iters=300, radius_min=0.005)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try; best_h = h_try.copy()
                scale = max(0.002, scale * 0.8)
            else:
                scale = min(0.05, scale * 1.08)
        # Binary SA at a smaller resolution for diversity
        if check_time(0.74):
            sub_dl = min(t0 + budget_s * 0.74, time.time() + budget_s * 0.025)
            h_b, c5_b = _binary_sa(96, sub_dl, rng, n_iters=15000)
            if c5_b < best_c5:
                best_c5 = c5_b; best_h = h_b.copy(); best_n = 96
                if check_time(0.76):
                    dl_b2 = t0 + budget_s * 0.76
                    h_br, c5_br = _lp_optimize(best_h, dl_b2, radius_init=0.1, max_iters=1500, radius_min=0.003)
                    if c5_br < best_c5:
                        best_c5 = c5_br; best_h = h_br.copy()
    
    # Phase 6: Coordinate polish
    if check_time(0.82):
        dl6 = t0 + budget_s * 0.82
        hc, c5c = _coordinate_polish(best_h, dl6, n_cycles=4)
        if c5c < best_c5:
            best_c5 = c5c; best_h = hc.copy()
    
    # Phase 7: Pair polish (windowed for efficiency)
    if check_time(0.92):
        dl7 = t0 + budget_s * 0.92
        window = max(6, best_n // 10)
        hp, c5p = _pair_polish(best_h, dl7, n_passes=5, window=window)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()
    
    # Phase 8: Final aggressive LP
    if check_time(0.97):
        dl8 = t0 + budget_s * 0.97
        h_fin, c5_fin = _lp_optimize(best_h, dl8, radius_init=0.015, max_iters=4000, radius_min=0.0001)
        if c5_fin < best_c5:
            best_c5 = c5_fin; best_h = h_fin.copy()
    
    # Phase 9: Final smoothed gradient
    if check_time(0.995):
        dl9 = t0 + budget_s * 0.995
        h_sg, c5_sg = _smoothed_grad(best_h, dl9, beta_start=60, beta_end=800, n_steps=500, step_init=0.02)
        if c5_sg < best_c5:
            best_c5 = c5_sg; best_h = h_sg.copy()
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
