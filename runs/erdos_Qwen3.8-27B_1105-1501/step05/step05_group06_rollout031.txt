    n_lags = 2 * n - 1

        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]

    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

                      bounds=bounds, method='highs')

    except Exception:
        pass
    return None, None
