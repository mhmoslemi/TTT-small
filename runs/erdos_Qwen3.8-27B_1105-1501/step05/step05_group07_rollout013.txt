import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-12:
        return v
    # Binary search for shift
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _sigmoid(x):
    return np.where(x >= 0, 1/(1+np.exp(-x)), np.exp(x)/(1+np.exp(x)))

def _logistic_h(q, lam):
    return _sigmoid(q - lam)

def _find_lambda(q, target):
    # Find lambda such that sum(sigmoid(q - lambda)) = target
    lo = float(np.min(q)) - 20.0
    hi = float(np.max(q)) + 20.0
    # sigmoid(q - lam) is decreasing in lam
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sigmoid(q - mid)))
        if s > target:
            lo = mid  # need to increase lambda to decrease sum
        else:
            hi = mid
    return (lo + hi) / 2.0

def _h_from_q(q, n):
    target = n / 2.0
    lam = _find_lambda(q, target)
    return _sigmoid(q - lam)

def _smoothed_obj_and_grad(q, n, beta):
    h = _h_from_q(q, n)
    S = _scores(h)
    maxS = np.max(S)
    shifted = np.clip(beta * (S - maxS), -500, 0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w /= wsum
    obj = maxS + np.log(wsum) / beta
    
    G = _grad_matrix(h)
    hh = h * (1.0 - h)  # h*(1-h)
    S_hh = float(np.sum(hh))
    if S_hh < 1e-15:
        return obj, np.zeros(n)
    # A_j = sum_k w_k * G[k,j]
    A = G.T @ w  # shape (n,)
    # B = sum_k w_k * sum_l G[k,l] * hh[l]
    Ghh = G @ hh  # shape (2n-1,)
    B = float(w @ Ghh)
    grad = hh * (A - B / S_hh)
    return obj, grad

def _logistic_optimize(q_init, n, deadline, beta_start=5.0, beta_end=500.0, n_phases=8):
    """Optimize using logistic parameterization and L-BFGS-B."""
    best_c5 = _exact_c5(_h_from_q(q_init, n))
    best_q = q_init.copy()

            return _smoothed_obj_and_grad(q, n, beta)[0]

            return _smoothed_obj_and_grad(q, n, beta)[1]
        
        try:

                          bounds=[(-12.0, 12.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-14})
            if np.all(np.isfinite(res.x)):

                    best_c5 = c5_new
                    best_q = res.x.copy()
                q = res.x.copy()
        except Exception:
            pass
    
    return best_q, best_c5

def _h_to_q(h, n):
    """Convert h to q parameterization (with lambda=0)."""
    eps = 1e-8
    h_clip = np.clip(h, eps, 1.0 - eps)
    q = np.log(h_clip / (1.0 - h_clip))
    return q

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _sigmoid(x):
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _find_lambda(q, target):
    lo = float(np.min(q)) - 30.0
    hi = float(np.max(q)) + 30.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sigmoid(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _h_from_q(q, n):
    target = n / 2.0
    lam = _find_lambda(q, target)
    return _sigmoid(q - lam)

def _smoothed_obj_and_grad(q, n, beta):
    h = _h_from_q(q, n)
    S = _scores(h)
    maxS = float(np.max(S))
    shifted = np.clip(beta * (S - maxS), -500.0, 0.0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w /= wsum
    obj = maxS + np.log(max(wsum, 1e-300)) / beta
    G = _grad_matrix(h)
    hh = h * (1.0 - h)
    S_hh = float(np.sum(hh))
    if S_hh < 1e-15:
        return obj, np.zeros(n)
    A = G.T @ w
    Ghh = G @ hh
    B = float(w @ Ghh)
    grad = hh * (A - B / S_hh)
    return obj, grad

def _h_to_q(h, n):
    eps = 1e-10
    h_clip = np.clip(h, eps, 1.0 - eps)

def _quasi_periodic_init(n, alpha):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []

        bounds.append((max(-radius, -h[j] - 1e-12), min(radius, 1.0 - h[j] + 1e-12)))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:

    except Exception:

def _binary_snap(h, n):
    target = n / 2.0
    h_bin = (h > 0.5).astype(float)
    cur_sum = float(np.sum(h_bin))
    if cur_sum > target:
        order = np.argsort(h_bin)
        for idx in order:
            if cur_sum <= target:
                break
            h_bin[idx] = 0.0
            cur_sum -= 1.0
    elif cur_sum < target:
        order = np.argsort(-h_bin)
        for idx in order:
            if cur_sum >= target:
                break
            h_bin[idx] = 1.0
            cur_sum += 1.0
    return h_bin

            h_qp = _project(h_qp, best_n / 2.0)
            q_qp = _h_to_q(h_qp, best_n)
            q_opt, c5_opt = _logistic_optimize(q_qp, best_n, sub_dl,
                                              beta_start=5.0, beta_end=500.0, n_phases=8)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = _h_from_q(q_opt, best_n).copy()
            # Also try from resampled best
            if time.time() < dl4:
                h_res = _resample(best_h, best_n)
                h_res += rng.normal(0, 0.01, best_n)
                h_res = _project(h_res, best_n / 2.0)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                h_lp, c5_lp = _local_optimize_lp(h_res, sub_dl2, radius_init=0.15,
                                                 max_iters=2000, stall_limit=100)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()

    # Phase 5: Multi-resolution
    if time.time() < t0 + budget_s * 0.88:
        dl5 = t0 + budget_s * 0.88
        for n_try in [64, 96, 160, 192, 256]:
            if time.time() > dl5:
                break
            if n_try == best_n:
                continue
            sub_dl = min(dl5, time.time() + budget_s * 0.025)
            h_res = _resample(best_h, n_try)
            q_res = _h_to_q(h_res, n_try)
            q_opt, c5_opt = _logistic_optimize(q_res, n_try, sub_dl,
                                              beta_start=5.0, beta_end=600.0, n_phases=7)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = _h_from_q(q_opt, n_try).copy()
                best_n = n_try
                if time.time() < dl5:
                    sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
                    hp, cp = _local_optimize_lp(best_h, sub_dl2, radius_init=0.08,
                                                max_iters=1500, stall_limit=60)
                    if cp < best_c5:
                        best_c5 = cp
                        best_h = hp.copy()
            # Also try LP directly at this resolution
            if time.time() < dl5:
                sub_dl3 = min(dl5, time.time() + budget_s * 0.015)
                h_lp, c5_lp = _local_optimize_lp(_resample(best_h, n_try), sub_dl3,
                                                 radius_init=0.2, max_iters=2000, stall_limit=80)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = n_try

    # Phase 6: Perturbation restarts
    if time.time() < t0 + budget_s * 0.94:
        dl6 = t0 + budget_s * 0.94
        scale = 0.02
        for attempt in range(300):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.008)
            q_pert = _h_to_q(h_pert, best_n)
            q_opt, c5_opt = _logistic_optimize(q_pert, best_n, sub_dl,
                                              beta_start=40.0, beta_end=300.0, n_phases=3)
            if c5_opt < best_c5 - 1e-13:
                best_c5 = c5_opt
                best_h = _h_from_q(q_opt, best_n).copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.12, scale * 1.15)

    # Phase 7: Final LP polish
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.97
        h7, c5_7 = _local_optimize_lp(best_h, dl7, radius_init=0.04, max_iters=5000, stall_limit=300)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: High-beta SLSQP final
    if time.time() < t0 + budget_s * 0.98:
        h8, c5_8 = _slsqp_smooth(best_h, t0 + budget_s * 0.985,
                                 beta_init=100.0, beta_final=600.0, n_phases=3)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
        # Tiny LP finish
        if time.time() < deadline_total:
            h8b, c5_8b = _local_optimize_lp(best_h, deadline_total, radius_init=0.02,
                                            max_iters=1000, stall_limit=50)
            if c5_8b < best_c5:
                best_c5 = c5_8b
                best_h = h8b.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
