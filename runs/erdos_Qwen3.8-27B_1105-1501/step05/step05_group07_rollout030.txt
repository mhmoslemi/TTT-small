    for it in range(n_iter):
        # Compute weights (softmax over scores)
        beta = 20.0
        shifted = beta * (scores - cur_max)
        shifted = np.clip(shifted, -100, 0)
        w = np.exp(shifted)
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        
        # Compute pressure
        pressure = np.zeros(n)
        for i in range(n):
            p = 0.0
            for idx in range(len(scores)):
                d = idx - (n - 1)
                wi = w[idx]
                if wi < 1e-15:
                    continue
                ip = i + d
                im = i - d
                if 0 <= ip < n:
                    p += wi * (1.0 - h[ip])
                if 0 <= im < n:
                    p -= wi * h[im]
            pressure[i] = p * dx
        
        # Sort indices by pressure
        order = np.argsort(pressure)
        # Try pairs: high pressure gives to low pressure
        improved = False
        for ki in range(max(0, len(order) - 10), len(order)):
            for kj in range(0, min(10, len(order))):
                i = order[ki]
                j = order[kj]
                if i == j:
                    continue
                delta, new_max = _two_coord_solve(h, i, j, scores, dx)
                if abs(delta) > 1e-12 and new_max < cur_max - 1e-12:
                    h[i] += delta
                    h[j] -= delta
                    h[i] = float(np.clip(h[i], 0.0, 1.0))
                    h[j] = float(np.clip(h[j], 0.0, 1.0))
                    scores = _scores(h)
                    cur_max = float(np.max(scores))
                    improved = True
        if not improved:
            break
    
    return h, scores, cur_max

def _lp_direction(h, scores, radius):
    """LP-based steepest descent direction."""
    n = len(h)
    n_lags = len(scores)
    dx = 2.0 / n
    
    # Build G matrix (gradient of each score w.r.t. h)
    G = np.zeros((n_lags, n))
    for idx in range(n_lags):
        d = idx - (n - 1)
        for a in range(n):
            ap = a + d
            am = a - d
            if 0 <= ap < n:
                G[idx, a] += (1.0 - h[ap])
            if 0 <= am < n:
                G[idx, a] -= h[am]
    G *= dx
    
    # LP: min t s.t. scores + G*delta <= [0,0,...,0] (in normalized form)
    # Actually: min t s.t. G*delta <= -scores + t (for each lag)
    # With constraints: sum(delta) = 0, |delta_j| <= radius, 0 <= h_j + delta_j <= 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search_c5(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.99
    # Golden section
    gr = 0.6180339887498949
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    alpha = (lo + hi) / 2.0
    f = _exact_c5(h + alpha * p)
    return alpha, f

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for idx in range(n_lags):
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] += (1.0 - h[d:])  # h[a] * (1-h[a+d]) term: derivative w.r.t. h[a] is (1-h[a+d])
                G[idx, d:] += -h[:m]  # derivative w.r.t. h[a+d] is -h[a]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, g:] += (1.0 - h[:m])
                G[idx, :m] += -h[g:]
    return G * dx

import numpy as np
from scipy.optimize import linprog, minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _two_coord_solve(h, i, j, scores, dx, n):
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    if delta_hi - delta_lo < 1e-14:
        return 0.0, float(np.max(scores))
    
    n_lags = 2 * n - 1
    offset = n - 1
    A = np.zeros(n_lags)
    for idx in range(n_lags):
        d = idx - offset
        a = 0.0
        ip = i + d
        im = i - d
        jp = j + d
        jm = j - d
        if 0 <= ip < n:
            a += (1.0 - h[ip])
        if 0 <= im < n:
            a -= h[im]
        if 0 <= jp < n:
            a -= (1.0 - h[jp])
        if 0 <= jm < n:
            a += h[jm]
        A[idx] = a * dx
    
    cur_max = float(np.max(scores))
    
    gr = 0.6180339887498949
    lo = delta_lo
    hi = delta_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + c1 * A))
    f2 = float(np.max(scores + c2 * A))
    
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + c1 * A))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + c2 * A))
    
    delta_opt = (lo + hi) / 2.0
    best_val = float(np.max(scores + delta_opt * A))
    
    # Check endpoints
    if f1 < best_val:
        delta_opt, best_val = c1, f1
    if f2 < best_val:
        delta_opt, best_val = c2, f2
    f_lo = float(np.max(scores + delta_lo * A))
    f_hi = float(np.max(scores + delta_hi * A))
    if f_lo < best_val:
        delta_opt, best_val = delta_lo, f_lo
    if f_hi < best_val:
        delta_opt, best_val = delta_hi, f_hi
    if cur_max < best_val:
        delta_opt, best_val = 0.0, cur_max
    
    return delta_opt, best_val

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _two_coord_solve(h, i, j, scores, dx, n):

    offset = n - 1

            a += (1.0 - hi_val[ip])

        A[idx] = a * dx
    cur_max = float(np.max(scores))
    gr = 0.6180339887498949
    lo = delta_lo
    hi = delta_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    f2 = float(np.max(scores + c2 * A))
    for _ in range(40):
        if hi - lo < 1e-13:

        if f1 < f2:

        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + c2 * A))
    delta_opt = (lo + hi) / 2.0
    best_val = float(np.max(scores + delta_opt * A))
    if f1 < best_val:
        delta_opt, best_val = c1, f1
    if f2 < best_val:
        delta_opt, best_val = c2, f2
    f_lo = float(np.max(scores + delta_lo * A))
    if f_lo < best_val:
        delta_opt, best_val = delta_lo, f_lo
    f_hi = float(np.max(scores + delta_hi * A))
    if f_hi < best_val:
        delta_opt, best_val = delta_hi, f_hi
    if cur_max <= best_val:

    return delta_opt, best_val

def _compute_pressure(h, scores, n):
    n_lags = 2 * n - 1
    cur_max = float(np.max(scores))
    beta = 25.0
    shifted = beta * (scores - cur_max)
    shifted = np.clip(shifted, -200, 0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(n_lags)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    pressure = np.zeros(n)
    for i in range(n):
        p = 0.0
        for idx in range(n_lags):
            wi = w[idx]
            if wi < 1e-15:
                continue
            d = idx - (n - 1)
            ip = i + d
            if 0 <= ip < n:
                p += wi * (1.0 - h[ip])
            im = i - d
            if 0 <= im < n:
                p -= wi * h[im]
        pressure[i] = p
    return pressure

def _two_coord_optimize(h, deadline, max_outer=500):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()

    scores = _scores(h)

                best_c5 = cur_max
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for idx in range(n_lags):
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] += (1.0 - h[d:])
                G[idx, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, g:] += (1.0 - h[:m])
                G[idx, :m] += -h[g:]
    return G * dx

def _line_search_c5(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.99
    gr = 0.6180339887498949
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break

            hi = c2; f2 = f1; c2 = c1

        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    alpha = (lo + hi) / 2.0
    return alpha, _exact_c5(h + alpha * p)

def _local_optimize_lp(h, deadline, radius_init=0.2, max_iters=2000, stall_limit=80):
    n = len(h)
    target = n / 2.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_polish(h, deadline, beta_init=50.0, beta_final=250.0, n_phases=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac

        def obj(hf):
            S = _scores(hf)
            maxS = float(np.max(S))
            shifted = beta * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            return maxS + float(np.log(np.mean(np.exp(shifted)))) / beta

        def grad(hf):
            S = _scores(hf)
            maxS = float(np.max(S))
            shifted = beta * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            wsum = float(np.sum(w))
            if wsum < 1e-300:
                w = np.zeros(len(S))
                w[int(np.argmax(S))] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w

        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq',
                                         'fun': _con_fun,
                                         'jac': _con_jac}],
                           options={'maxiter': 150, 'ftol': 1e-14})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)
