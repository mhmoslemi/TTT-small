    m = float(np.max(F))
    Z = np.exp((F - m) / _g_beta)
    s = float(np.sum(Z))
    if s < 1e-300:
        return m
    return m + _g_beta * float(np.log(s))

def smooth_grad(x):
    F = lag_values(x)
    m = float(np.max(F))
    Z = np.exp((F - m) / _g_beta)
    s = float(np.sum(Z))
    if s < 1e-300:
        P = np.zeros_like(F)
        P[int(np.argmax(F))] = 1.0
    else:
        P = Z / s
    G = lag_jacobian(x)
    return G.T @ P

def smooth_eq(x):
    return np.sum(x) - _g_target

def smooth_eq_jac(x):
    return np.ones(_g_n)

def _smooth_slsqp(h, beta, deadline, maxiter=150, ftol=1e-10):
    global _g_beta, _g_n, _g_target
    n = len(h)
    _g_beta = beta
    _g_n = n
    _g_target = n / 2.0

def _lp_direction(h, scores, G, radius, n_active=5, relax=0.0):
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    
    # Build active set
    if n_active >= n_lags:
        active = set(range(n_lags))
    else:
        active = set(sorted_idx[:n_active].tolist())
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    dx_val = 2.0 / n
    b_ub = np.full(n_lags, -scores + relax * dx_val)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(60):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_loop(h, deadline, radius_init=0.12, max_stall=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 4
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        G = lag_jacobian(h) / dx  # Use unnormalized for LP, then scale
        
        # Actually, let's use the normalized scores for LP
        scores_norm = scores / dx  # Remove dx factor for LP direction finding
        
        p = _lp_direction(h, scores_norm, G, radius, n_active=n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            # Try with more active constraints
            n_active = min(n_active + 3, 2 * n - 1)
            p = _lp_direction(h, scores_norm, G, radius, n_active=n_active)
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.7
                if radius < 1e-6:
                    break
                continue
        
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:

            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 4
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.05
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def _coord_refine(h, deadline, n_iter=300, rng=None):

    target = n / 2.0

def _smooth_beta_schedule(h, deadline, c5_current):

            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_loop(h, deadline, radius_init=0.12, max_stall=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 5
    while time.time() < deadline:
        if stall > max_stall:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = lag_jacobian(h) / dx
        p = _lp_direction(h, scores, G, radius, n_active=n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            n_active = min(n_active + 4, 2 * n - 1)
            p = _lp_direction(h, scores, G, radius, n_active=n_active)
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.7
                if radius < 1e-6:
                    break
                if stall % 15 == 0:
                    radius = max(radius, 0.04)
                continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 5
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def _coord_refine(h, deadline, n_iter=300, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    if rng is None:
        rng = np.random.default_rng(777)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.4))
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def _smooth_schedule(h, deadline, c5_init, betas=None, maxiter_per=80):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_init
    if betas is None:
        betas = [0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.0015, 0.001, 0.0007, 0.0005]
    for beta in betas:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 25.0)
        h_new, c5_new = _smooth_slsqp(best_h, beta, sub_dl, maxiter=maxiter_per, ftol=1e-10)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5
