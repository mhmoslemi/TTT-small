import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_exact(h):
    """Compute all 2n-1 overlap scores"""
    n = len(h)
    dx = 2.0 / n
    scores = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        idx = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                scores[idx] = dx * np.dot(h[:m], 1.0 - h[d:])
        else:
            g = -d
            m = n - g
            if m > 0:
                scores[idx] = dx * np.dot(h[g:], 1.0 - h[:m])
    return scores

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _score_grad(h, d):
    """Gradient of score for lag d with respect to h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += dx * (1.0 - h[d:])
            grad[d:] -= dx * h[:m]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[g:] += dx * (1.0 - h[:m])
            grad[:m] -= dx * h[g:]
    return grad

def _epigraph_solve(h0, n_active, method='SLSQP', maxiter=200, tol=1e-8):
    """Solve the epigraph problem with top n_active constraints"""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    # Compute all scores and select top n_active
    scores = _scores_exact(h0)
    top_idx = np.argsort(-scores)[:n_active]
    top_scores = scores[top_idx]
    
    # Convert lag indices to actual lags
    lags = top_idx - (n - 1)  # top_idx is 0-based position in scores array
    
    # Initial T
    T0 = float(np.max(scores)) + 0.01
    
    # Variables: [h (n), T (1)]
    x0 = np.concatenate([h0, [T0]])
    
    # Bounds
    bounds = [(0, 1)] * n + [(0, 2)]
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_constraint(x):
        return np.sum(x[:n]) - target
    
    def eq_jacobian(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    
    def ineq_constraints(x):
        h = x[:n]
        T = x[-1]
        vals = np.zeros(len(lags))
        for i, d in enumerate(lags):
            if d >= 0:
                m = n - d
                if m > 0:
                    vals[i] = T - dx * np.dot(h[:m], 1.0 - h[d:])
                else:
                    vals[i] = T
            else:
                g = -d
                m = n - g
                if m > 0:
                    vals[i] = T - dx * np.dot(h[g:], 1.0 - h[:m])
                else:
                    vals[i] = T
        return vals
    
    def ineq_jacobian(x):
        h = x[:n]
        J = np.zeros((len(lags), n + 1))
        for i, d in enumerate(lags):
            J[i, -1] = 1.0
            grad = _score_grad(h, d)
            J[i, :n] = -grad
        return J
    
    constraints = [
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian},
        {'type': 'ineq', 'fun': ineq_constraints, 'jac': ineq_jacobian}
    ]
    
    try:
        res = minimize(objective, x0, method=method, jac=objective_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': tol})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], target)
            T_new = float(res.x[-1])
            c5_new = _c5_exact(h_new)
            c5_from_T = T_new
            # The actual C5 is what matters
            if c5_new < float(np.max(scores)):
                return h_new, c5_new
    except Exception:
        pass
    return None, None
