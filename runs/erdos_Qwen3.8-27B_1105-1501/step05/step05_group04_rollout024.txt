The LP approach:
1. Linearizes F_s at current h: F_s(h + αp) ≈ F_s(h) + α * G_s * p

3. The linearization is only valid for small α

The SLSQP approach:
1. Uses the exact quadratic F_s in the constraints
2. Finds a point h' that truly satisfies all constraints with a lower t
3. Can take larger steps because the constraints are exact

The key advantage is that SLSQP can see the curvature, so it won't overshoot or undershoot the way the LP + line search might.

However, SLSQP is a local method and might get stuck. The LP approach with line search is more robust in some sense because it always makes progress if a descent direction exists.

So the best strategy is to use both: LP for robust local improvement, SLSQP for larger jumps when the LP gets stuck.

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    dx_val = 2.0 / n
    b_ub = np.full(n_lags, -scores + radius * dx_val)
    for k in active:
        b_ub[k] = -scores[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds.append((None, None))

    return None

    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def _multires(h, deadline, rng, max_calls_slsqp=4):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    tried = set()
    dn_list = list(range(-20, 25, 3))
    rng.shuffle(dn_list)
    for dn in dn_list:
        if time.time() > deadline:
            break
        n_try = n + dn
        if n_try < 30 or n_try > 180 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline, time.time() + 3.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1)
        if c5_r < best_c5 - 1e-12:
            c5_r2 = eval_c5(h_r)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r.copy()
    return best_h, best_c5

def _perturb_slsqp(h, deadline, rng, base_scale=0.03, max_rounds=40):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        h_p = best_h + rng.normal(0, scale, n)
        h_p = project_box_sum(h_p, target)
        sub_dl = min(deadline, time.time() + 6.0)
        h_t, c5_t = _lp_loop(h_p, sub_dl, radius_init=0.08)
        sub_dl2 = min(deadline, time.time() + 15.0)
        if sub_dl2 > time.time() + 3:
            h_t, c5_t = _slsqp_epigraph(h_t, sub_dl2, max_calls=2)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.006, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.05)
    return best_h, best_c5

def _coord_refine(h, deadline, n_iter=300, seed=99):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(seed)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.3))
        if c5_try < best_c5 - 1e-13:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: LP descent
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP epigraph
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _slsqp_epigraph(best_h, d2, max_calls=6)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smoothing + SLSQP
    d3 = t0 + budget_s * 0.38
    h3, c5_3 = _smooth_slsqp(best_h, d3, max_calls=5)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Perturbation + LP + SLSQP
    d4 = t0 + budget_s * 0.55
    h4, c5_4 = _perturb_slsqp(best_h, d4, rng, base_scale=0.03, max_rounds=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Second SLSQP pass on best
    d5 = t0 + budget_s * 0.65
    h5, c5_5 = _slsqp_epigraph(best_h, d5, max_calls=5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution exploration
    d6 = t0 + budget_s * 0.78
    h6, c5_6 = _multires(best_h, d6, rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: LP polish on best
    d7 = t0 + budget_s * 0.86
    h7, c5_7 = _lp_loop(best_h, d7, radius_init=0.08)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: SLSQP on refined
    d8 = t0 + budget_s * 0.93
    h8, c5_8 = _slsqp_epigraph(best_h, d8, max_calls=4)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Phase 9: Coordinate refinement
    d9 = t0 + budget_s * 0.98
    h9, c5_9 = _coord_refine(best_h, d9, n_iter=200, seed=42)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 < 0:
        best_h = project_box_sum(np.array(initial_h_values, dtype=float), len(initial_h_values) / 2.0)
        final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
