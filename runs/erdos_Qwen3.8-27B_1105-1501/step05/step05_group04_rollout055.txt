Objective: minimize t

- sum(h) = n/2 (1 equality constraint)

The key insight is that the LP approach linearizes the constraints, which can miss improvements when multiple constraints are simultaneously active and the bilinear nature matters. The SLSQP approach handles the full nonlinear structure.

phi_beta(h) = m + beta * log(sum_l exp((F_l - m)/beta))

This gives a differentiable surrogate that can be optimized with standard methods. The benefit is that it smoothly blends all near-active constraints, potentially finding directions that the LP (which only considers a subset of active constraints) misses.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project onto {0 <= x <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _eval_c5(h)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_descent(h, deadline, radius_init=0.12, n_active_init=5):
    # LP-based directional descent with near-active set
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = n_active_init
    
    while time.time() < deadline:
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        max_score = float(np.max(scores))
        
        # Build gradient matrix for active set
        sorted_idx = np.argsort(scores)[::-1]
        active = sorted_idx[:n_active]
        
        # Build constraint matrix for active lags
        n_lags = len(scores)
        G = np.zeros((n_active, n))
        for idx_active in range(n_active):
            k = active[idx_active]
            L = max(0, k - n + 1)
            U = min(n - 1, k)
            for j in range(n):
                g = 0.0
                if L <= j <= U:
                    g += (1.0 - h[k - j])
                if L <= k - j <= U:
                    g -= h[k - j]
                G[idx_active, j] = g * dx
        
        # LP: min t s.t. G @ p - t <= -scores[active], sum(p) = 0, box constraints
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_active, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[active]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x[-1] < max_score:
                p = res.x[:n]
                if float(np.max(np.abs(p))) > 1e-13:
                    alpha, new_c5 = _line_search(h, p, dx, radius)
                    if new_c5 < cur_c5 - 1e-13:
                        h = h + alpha * p
                        h = _project(h, target)
                        cur_c5 = _eval_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                            stall = 0
                            radius = min(radius * 1.1, 0.35)
                            n_active = max(3, n_active - 1)
                        else:
                            stall += 1
                            radius *= 0.8
                    else:
                        stall += 1
                        radius *= 0.8
                else:
                    stall += 1
                    radius *= 0.8
            else:
                # Try with more active constraints
                n_active = min(n_active + 3, n_lags)
                stall += 1
                if stall % 20 == 0:
                    radius = max(radius, 0.05)
                    n_active = 3
                else:
                    radius *= 0.85
        except Exception:
            stall += 2
            radius *= 0.7
    
    return best_h, best_c5

def _smooth_optimize(h, beta, deadline, maxiter=30):
    # Optimize log-sum-exp smoothed objective using BFGS
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        scores = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(scores)
        z = np.exp((scores - m) / beta)
        val = m + beta * np.log(np.sum(z))
        P = z / np.sum(z)
        # Gradient: sum_l P_l * grad F_l
        grad = np.zeros(n)
        n_lags = len(scores)
        for l in range(n_lags):
            if P[l] < 1e-10:
                continue
            k = l
            L = max(0, k - n + 1)
            U = min(n - 1, k)
            for j in range(n):
                g = 0.0
                if L <= j <= U:
                    g += (1.0 - x[k - j])
                if L <= k - j <= U:
                    g -= x[k - j]
                grad[j] += P[l] * g * dx
        return val, grad
    
    # Use BFGS with projection (implicit)
    x = h.copy()
    try:
        res = minimize(obj_and_grad, x, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'gtol': 1e-6})
        x_new = _project(res.x, target)
        return x_new
    except Exception:
        return h

def _epigraph_slsqp(h, deadline, maxiter=50):
    # SLSQP on epigraph formulation
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    n_lags = 2 * n - 1
    c0 = _eval_c5(h)
    
    # Variable: [h_0, ..., h_{n-1}, t]
    def objective(x):
        return x[-1]
    
    def constraint_fun(x):
        h_loc = np.clip(x[:n], 0.0, 1.0)
        scores = np.correlate(h_loc, 1.0 - h_loc, mode='full') * dx
        return scores - x[-1]
    
    def constraint_jac(x):
        h_loc = np.clip(x[:n], 0.0, 1.0)
        J = np.zeros((n_lags, n + 1))
        for k in range(n_lags):
            L = max(0, k - n + 1)
            U = min(n - 1, k)
            for j in range(n):
                g = 0.0
                if L <= j <= U:
                    g += (1.0 - h_loc[k - j])
                if L <= k - j <= U:
                    g -= h_loc[k - j]
                J[k, j] = g * dx
            J[k, n] = -1.0
        return J
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac},
        {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target,
         'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}
    ]
    
    bounds = [(0.0, 1.0)] * n + [
        (max(0.0, c0 - 0.1), max(1.0, c0 + 0.05))
    ]
    
    x0 = np.concatenate([h, [c0 + 0.01]])
    
    # Time-limited callback
    class Timeout(Exception):
        pass
    
    count = [0]
    def callback(xk):
        count[0] += 1
        if time.time() > deadline:
            raise Timeout()
        if count[0] % 5 == 0:
            h_try = _project(xk[:n], target)
            c_try = _eval_c5(h_try)
            if c_try < c0 - 1e-10:
                nonlocal_best[0] = h_try.copy()
                nonlocal_best[1] = c_try
    
    nonlocal_best = [h.copy(), c0]
    
    # Can't use nonlocal in top-level function, use list
    best_in_call = [h.copy(), c0]
    
    def callback2(xk):
        count[0] += 1
        if time.time() > deadline:
            raise Timeout()
        if count[0] % 3 == 0:
            try:
                h_try = _project(xk[:n], target)
                if np.all((h_try >= -1e-10) & (h_try <= 1.0 + 1e-10)):
                    c_try = _eval_c5(h_try)
                    if c_try < best_in_call[1]:
                        best_in_call[0] = h_try.copy()
                        best_in_call[1] = c_try
            except Exception:
                pass
    
    try:
        res = minimize(objective, x0, method='SLSQP', constraints=constraints,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-8},
                       callback=callback2)
    except Exception:
        pass
    
    # Also try the final iterate
    try:
        h_try = _project(res.x[:n], target)
        c_try = _eval_c5(h_try)
        if c_try < best_in_call[1]:
            best_in_call[0] = h_try.copy()
            best_in_call[1] = c_try
    except Exception:
        pass
    
    return best_in_call[0], best_in_call[1]

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coord_refine(h, deadline, n_iter=500, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    if rng is None:
        rng = np.random.default_rng(42)

        j = interior[rng.integers(0, len(interior))]
        if i == j:
