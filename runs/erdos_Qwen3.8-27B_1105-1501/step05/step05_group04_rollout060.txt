        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

        cnt = int(np.sum(mask))

def lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for s in range(-(n-1), n):
        r = n - 1 + s
        if s >= 0:
            m = n - s
            if m > 0:
                J[r, s:] += dx * (1.0 - h[:m])
                J[r, :m] -= dx * h[s:]
        else:
            a = -s
            m = n - a
            if m > 0:
                J[r, :m] += dx * (1.0 - h[a:])
                J[r, a:] -= dx * h[:m]
    return J

def _sqp_obj(x):
    return x[_STATE['n']]

def _sqp_grad(x):
    n = _STATE['n']
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _sqp_ineq(x):
    # Returns t - F_s(h) for active s, should be >= 0
    n = _STATE['n']
    h = x[:n]
    t = x[n]
    active = _STATE['active_idx']
    scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - scores[active]

def _sqp_ineq_jac(x):
    n = _STATE['n']
    h = x[:n]
    J = lag_jacobian(h)
    active = _STATE['active_idx']
    na = len(active)
    jac = np.zeros((na, n + 1))
    jac[:, :n] = -J[active, :]  # negative because we want t - F_s >= 0
    jac[:, n] = 1.0
    return jac

def _sqp_eq(x):
    n = _STATE['n']
    target = _STATE['target']
    return np.array([float(np.sum(x[:n])) - target])

def _sqp_eq_jac(x):
    n = _STATE['n']
    jac = np.zeros(n + 1)
    jac[:n] = 1.0
    return jac

def sqp_solve(h, deadline, n_active=20, max_iter=100):
    n = len(h)
    target = n / 2.0
    scores = lag_values(h)
    c0 = float(np.max(scores))

    constraints = [
        {'type': 'ineq', 'fun': _sqp_ineq, 'jac': _sqp_ineq_jac},

def smooth_solve(h, deadline, betas=None):
    """Log-sum-exp smoothing with projected gradient + line search."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    if betas is None:
        c0 = best_c5
        betas = [c0 * f for f in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]]
    
    for beta in betas:
        if time.time() > deadline:
            break
        J = lag_jacobian(best_h)
        scores = lag_values(best_h)
        
        # Projected gradient descent with backtracking line search
        h_cur = best_h.copy()
        step_scale = 0.1
        for it in range(30):
            if time.time() > deadline:
                break
            scores = lag_values(h_cur)
            m = float(np.max(scores))
            shifted = np.clip((scores - m) / beta, -50, 50)
            Z = np.exp(shifted)
            S = float(np.sum(Z))
            P = Z / S
            grad = P @ J
            
            # Projected gradient step
            d = -grad
            # Project direction to sum(d) = 0 for feasibility
            d = d - np.mean(d)
            
            # Line search
            best_t = 0.0
            best_f = float(np.inf)
            t = step_scale
            for ls in range(20):
                h_try = project_box_sum(h_cur + t * d, target)
                s_try = lag_values(h_try)
                m_try = float(np.max(s_try))
                sh_try = np.clip((s_try - m_try) / beta, -50, 50)
                Z_try = np.exp(sh_try)
                f_try = m_try + beta * np.log(float(np.sum(Z_try)))
                if f_try < best_f:
                    best_f = f_try
                    best_t = t
                t *= 0.5
            
            if best_t < 1e-10:
                break
            h_cur = project_box_sum(h_cur + best_t * d, target)
            step_scale = min(best_t * 1.5, 0.5)
        
        c_check = eval_c5(h_cur)
        if c_check < best_c5:
            best_c5 = c_check
            best_h = h_cur.copy()
    
    return best_h, best_c5

def lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        # Build gradient matrix
        J = lag_jacobian(h)
        
        # LP: find direction p minimizing max_s (scores[s] + J[s] @ p)
        # s.t. sum(p) = 0, |p_j| bounded
        n_lags = len(scores)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = J
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=-scores, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x[-1] < float(np.max(scores)) - 1e-10:
                p = res.x[:n]
                # Line search
                alpha, new_c5 = _line_search_lp(h, p, 2.0 / n, radius)
                if new_c5 < cur_c5 - 1e-12:
                    h = project_box_sum(h + alpha * p, target)
                    cur_c5 = eval_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.35)
                    continue
        except Exception:
            pass
        
        stall += 1
        radius *= 0.8
        if stall % 20 == 0:
            radius = max(radius, 0.04)
    
    return best_h, best_c5

def _line_search_lp(h, d, dx, trust):
    """Line search for LP direction."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust)
    t_hi = min(t_hi, trust)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(60):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def coord_refine(h, deadline, n_iter=500, seed=12345):

    target = n / 2.0

        i = interior[rng.integers(0, len(interior))]

                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    
    return best_h, best_c5

def perturb_search(h, deadline, rng, base_scale=0.03):

            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for s in range(-(n - 1), n):
        r = n - 1 + s
        if s >= 0:
            m = n - s
            if m > 0:
                J[r, s:] += dx * (1.0 - h[:m])
                J[r, :m] -= dx * h[s:]
        else:
            a = -s
            m = n - a
            if m > 0:
                J[r, :m] += dx * (1.0 - h[a:])
                J[r, a:] -= dx * h[:m]
    return J

def _sqp_obj(x):
    return x[_STATE['n']]

def _sqp_grad(x):
    n = _STATE['n']
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _sqp_ineq(x):
    n = _STATE['n']
    h = x[:n]
    t = x[n]
    active = _STATE['active_idx']
    scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - scores[active]

def _sqp_ineq_jac(x):
    n = _STATE['n']
    h = x[:n]
    J = lag_jacobian(h)
    active = _STATE['active_idx']
    na = len(active)
    jac = np.zeros((na, n + 1))
    jac[:, :n] = -J[active, :]
    jac[:, n] = 1.0
    return jac

def _sqp_eq(x):
    n = _STATE['n']
    target = _STATE['target']
    return np.array([float(np.sum(x[:n])) - target])

def _sqp_eq_jac(x):
    n = _STATE['n']
    jac = np.zeros(n + 1)
    jac[:n] = 1.0
    return jac

def sqp_solve(h, deadline, n_active=25, max_iter=80):
    n = len(h)
    target = n / 2.0
    scores = lag_values(h)
    c0 = float(np.max(scores))
    best_h = h.copy()
    best_c5 = c0

    for na in ([n_active, max(10, n_active // 2), min(50, n_active + 15)] if n_active > 15 else [n_active]):
        if time.time() > deadline:
            break
        tol = max(0.01, 0.04 * c0)
        active_mask = scores >= c0 - tol
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > na:
            active_idx = np.argsort(scores)[::-1][:na]
        if len(active_idx) < 3:
            active_idx = np.argsort(scores)[::-1][:max(3, min(na, len(scores) // 3))]

        _STATE['n'] = n
        _STATE['target'] = target
        _STATE['active_idx'] = active_idx

        x0 = np.zeros(n + 1)
        x0[:n] = h
        x0[n] = c0 + 0.002

        bounds = [(0.0, 1.0)] * n + [(max(0.0, c0 - 0.08), min(1.0, c0 + 0.15))]
        constraints = [
            {'type': 'ineq', 'fun': _sqp_ineq, 'jac': _sqp_ineq_jac},
            {'type': 'eq', 'fun': _sqp_eq, 'jac': _sqp_eq_jac}
        ]
        try:
            res = minimize(_sqp_obj, x0, method='SLSQP', jac=_sqp_grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-9})
            if hasattr(res, 'x') and res.x is not None:
                h_new = project_box_sum(res.x[:n], target)
                c_new = eval_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c5

def smooth_solve(h, deadline, n_inner=40):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    c_base = best_c5

    betas = [c_base * f for f in [0.4, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008]]

    for beta in betas:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        step_scale = 0.15
        for it in range(n_inner):
            if time.time() > deadline:
                break
            scores = lag_values(h_cur)
            m = float(np.max(scores))
            shifted = np.clip((scores - m) / beta, -60, 60)
            Z = np.exp(shifted)
            S = float(np.sum(Z))
            P = Z / S
            J = lag_jacobian(h_cur)
            grad = P @ J
            d = -grad
            d = d - np.mean(d)
            d_norm = float(np.linalg.norm(d))
            if d_norm < 1e-12:
                break

            best_t = 0.0
            best_f = float('inf')
            t = step_scale
            for _ in range(25):
                h_try = project_box_sum(h_cur + t * d, target)
                s_try = lag_values(h_try)
                m_try = float(np.max(s_try))
                sh_try = np.clip((s_try - m_try) / beta, -60, 60)
                f_try = m_try + beta * np.log(float(np.sum(np.exp(sh_try))))
                if f_try < best_f:
                    best_f = f_try
                    best_t = t
                t *= 0.5

            if best_t < 1e-12:
                break
            h_cur = project_box_sum(h_cur + best_t * d, target)
            step_scale = min(max(best_t * 1.3, 0.01), 0.4)

        c_check = eval_c5(h_cur)
        if c_check < best_c5:
            best_c5 = c_check
            best_h = h_cur.copy()

    return best_h, best_c5

def _line_search_lp(h, d, dx, trust):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust)
    t_hi = min(t_hi, trust)
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        J = lag_jacobian(h)
        n_lags = len(scores)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = J
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        found = False
        try:
            res = linprog(c, A_ub=A_ub, b_ub=-scores, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x[-1] < float(np.max(scores)) - 1e-10:
                p = res.x[:n]
                if float(np.max(np.abs(p))) > 1e-13:
                    alpha, new_c5 = _line_search_lp(h, p, 2.0 / n, radius)
                    if new_c5 < cur_c5 - 1e-13:
                        h = project_box_sum(h + alpha * p, target)
                        cur_c5 = eval_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.04, 0.35)
                        found = True
        except Exception:
            pass

        if not found:
            stall += 1
            radius *= 0.78
            if stall % 25 == 0 and radius < 0.03:
                radius = 0.06

    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def coord_refine(h, deadline, n_iter=500, seed=99):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(seed)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search_lp(h, d, dx, min(h[i], 1.0 - h[j], 0.3))
        if c5_try < best_c5 - 1e-13:
            h = project_box_sum(h + alpha * d, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def perturb_search(h, deadline, rng, base_scale=0.03):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    for _ in range(300):
        if time.time() > deadline:
            break
        h_pert = project_box_sum(best_h + rng.normal(0, scale, n), n / 2.0)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: LP from initial (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP epigraph (10%)
    d2 = t0 + budget_s * 0.18
    h2, c5_2 = sqp_solve(best_h, d2, n_active=25, max_iter=80)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smoothing (8%)
    d3 = t0 + budget_s * 0.26
    h3, c5_3 = smooth_solve(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP refinement (8%)
    d4 = t0 + budget_s * 0.34
    h4, c5_4 = lp_loop(best_h, d4, radius_init=0.12)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution exploration (14%)
    d5 = t0 + budget_s * 0.48
    n_cur = len(best_h)
    tried = set()
    for factor in [0.75, 0.65, 0.55, 1.2, 1.35, 1.5, 1.7, 0.85, 1.1]:
        if time.time() > d5:
            break
        n_try = max(28, int(round(n_cur * factor)))
        if n_try in tried or n_try > 280 or n_try < 24:
            continue
        tried.add(n_try)
        h_r = resample(best_h, n_try)
        sub_dl = min(d5, time.time() + 4.0)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.1)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: SLSQP on best (8%)
    d6 = t0 + budget_s * 0.56
    h6, c5_6 = sqp_solve(best_h, d6, n_active=30, max_iter=100)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Smoothing again (6%)
    d7 = t0 + budget_s * 0.62
    h7, c5_7 = smooth_solve(best_h, d7)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Perturbation + LP (16%)
    d8 = t0 + budget_s * 0.78
    best_h, best_c5 = perturb_search(best_h, d8, rng, base_scale=0.025)

    # Phase 9: SLSQP final push (6%)
    d9 = t0 + budget_s * 0.84
    h9, c5_9 = sqp_solve(best_h, d9, n_active=35, max_iter=120)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    # Phase 10: Coordinate refinement (8%)
    d10 = t0 + budget_s * 0.92
    best_h, best_c5 = coord_refine(best_h, d10, n_iter=600)

    # Phase 11: Final LP polish (4%)
    d11 = t0 + budget_s * 0.96
    h11, c5_11 = lp_loop(best_h, d11, radius_init=0.05)
    if c5_11 < best_c5:
        best_c5 = c5_11
        best_h = h11.copy()

    # Phase 12: Final coord pass with remaining time
    rem = t0 + budget_s - 2.0 - time.time()
    if rem > 5:
        best_h, best_c5 = coord_refine(best_h, t0 + budget_s - 2.0, n_iter=200)

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)
    if not np.isfinite(final_c5):
        best_h = project_box_sum(h0, n0 / 2.0)
        final_c5 = eval_c5(best_h)

    return best_h, final_c5, len(best_h)
