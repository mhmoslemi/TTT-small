import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection to {0 <= x <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-6
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_transfer(h, i, j, n_samples=20):
    """Find best t for transferring mass from j to i: h[i]+=t, h[j]-=t"""
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-12:
        return None, None
    # Sample the objective
    ts = np.linspace(t_min, t_max, n_samples + 2)[1:-1]
    best_t = 0.0
    best_c5 = _exact_c5(h)
    for t in ts:
        h[i] += t
        h[j] -= t
        c5 = _exact_c5(h)
        h[i] -= t
        h[j] += t
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    if best_t == 0.0:
        return None, None
    # Local refinement around best_t
    # ... (golden section or denser sampling)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-7
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _weighted_grad(h, gamma=30.0):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = float(np.max(scores))
    exponents = gamma * (scores - max_s)
    exponents = np.clip(exponents, -100.0, 0.0)
    w = np.exp(exponents)
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w -= w.min()
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wd_pos = w[n - 1 + d]
        if wd_pos > 1e-15:
            m = n - d
            grad[:m] += -wd_pos * h[d:]
            grad[d:] += wd_pos * (1.0 - h[:m])
        wd_neg = w[n - 1 - d]
        if wd_neg > 1e-15:
            m = n - d
            grad[:m] += wd_neg * (1.0 - h[d:])
            grad[d:] += -wd_neg * h[:m]
    grad *= dx
    return grad

def _pair_transfer_eval(h, i, j, n_samples=18):
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-12:
        return 0.0, None
    cur_c5 = _exact_c5(h)
    best_t = 0.0
    best_c5 = cur_c5
    ts = np.linspace(t_min, t_max, n_samples + 2)
    for tt in ts:
        if abs(tt) < 1e-13:
            continue
        h[i] += tt
        h[j] -= tt
        c5 = _exact_c5(h)
        h[i] -= tt
        h[j] += tt
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_t = tt
    if best_t == 0.0:
        return 0.0, cur_c5
    frac = (best_t - t_min) / (t_max - t_min)
    idx = int(frac * (n_samples + 1))
    lo = ts[max(0, idx - 1)]
    hi = ts[min(len(ts) - 1, idx + 1)]
    if hi - lo < 1e-13:
        return best_t, best_c5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2_t = lo + gr * (hi - lo)
    h[i] += c1
    h[j] -= c1
    f1 = _exact_c5(h)
    h[i] -= c1
    h[j] += c1
    h[i] += c2_t
    h[j] -= c2_t
    f2 = _exact_c5(h)
    h[i] -= c2_t
    h[j] += c2_t
    for _ in range(12):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2_t
            f2 = f1
            c2_t = c1
            c1 = hi - gr * (hi - lo)
            h[i] += c1
            h[j] -= c1
            f1 = _exact_c5(h)
            h[i] -= c1
            h[j] += c1
        else:
            lo = c1
            f1 = f2
            c1 = c2_t
            c2_t = lo + gr * (hi - lo)
            h[i] += c2_t
            h[j] -= c2_t
            f2 = _exact_c5(h)
            h[i] -= c2_t
            h[j] += c2_t
    opt_t = (lo + hi) / 2.0
    h[i] += opt_t
    h[j] -= opt_t
    opt_c5 = _exact_c5(h)
    h[i] -= opt_t
    h[j] += opt_t
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_t = opt_t
    if best_t != 0.0 and best_c5 < cur_c5 - 1e-15:
        return best_t, best_c5
    return 0.0, cur_c5

def _pair_descent_pass(h, n_pairs, deadline, rng, gamma=30.0, n_samples=18):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    n_accepted = 0
    grad = _weighted_grad(h, gamma)
    grad_c = grad - np.mean(grad)
    order = np.argsort(grad_c)
    neg_set = order[:max(4, n // 5)]
    pos_set = order[-max(4, n // 5):]
    pairs_tried = 0
    for i in neg_set:
        for j in pos_set:
            if int(i) == int(j):
                continue
            if time.time() > deadline or pairs_tried >= n_pairs:
                break
            t, c5 = _pair_transfer_eval(h, int(i), int(j), n_samples)
            if t != 0.0 and c5 is not None and c5 < _exact_c5(h) - 1e-15:
                h[int(i)] += t
                h[int(j)] -= t
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                n_accepted += 1
                if n_accepted % 5 == 0:
                    grad = _weighted_grad(h, gamma)
                    grad_c = grad - np.mean(grad)
                    order = np.argsort(grad_c)
                    neg_set = order[:max(4, n // 5)]
                    pos_set = order[-max(4, n // 5):]
            pairs_tried += 1
        if time.time() > deadline or pairs_tried >= n_pairs:
            break
    n_rand = max(50, n_pairs // 4)
    for _ in range(n_rand):
        if time.time() > deadline or pairs_tried >= n_pairs + n_rand:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        t, c5 = _pair_transfer_eval(h, i, j, n_samples)
        if t != 0.0 and c5 is not None and c5 < _exact_c5(h) - 1e-15:
            h[i] += t
            h[j] -= t
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            n_accepted += 1
        pairs_tried += 1
    return h, best_c5, n_accepted

def _full_sweep(h, deadline, n_samples=12):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    for i in range(n):
        for j in range(i + 1, n):
            if time.time() > deadline:
                return h, best_c5
            t, c5 = _pair_transfer_eval(h, i, j, n_samples)
            if t != 0.0 and c5 is not None and c5 < _exact_c5(h) - 1e-15:
                h[i] += t
                h[j] -= t
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            t2, c52 = _pair_transfer_eval(h, j, i, n_samples)
            if t2 != 0.0 and c52 is not None and c52 < _exact_c5(h) - 1e-15:
                h[j] += t2
                h[i] -= t2
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    return h, best_c5

def _quasi_periodic(n, alpha):
    x = np.arange(n) * alpha
    h = (np.mod(x, 1.0) < 0.5).astype(float)
    return _project(h, n / 2.0)

def _binary_random(n, rng):
    positions = rng.permutation(n)
    h = np.zeros(n)
    h[positions[:n // 2]] = 1.0
    return h

def _binary_sa(h_start, deadline, rng, n_iters=50000):
    n = len(h_start)
    target = n // 2
    h = h_start.copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h))
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1.0)[0]
            remove = rng.choice(ones, n_ones - target, replace=False)
            h[remove] = 0.0
        else:
            zeros = np.where(h == 0.0)[0]
            add = rng.choice(zeros, target - n_ones, replace=False)
            h[add] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    T = 0.003
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones_list[int(rng.integers(len(ones_list)))]
        j = zeros_list[int(rng.integers(len(zeros_list)))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        ones_list.remove(i)
        ones_list.append(j)
        zeros_list.remove(j)
        zeros_list.append(i)
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_list.remove(j)
            ones_list.append(i)
            zeros_list.remove(i)
            zeros_list.append(j)
        T *= cool
    return best_h, best_c5

def _gradient_descent_step(h, gamma, deadline):
    n = len(h)
    target = n / 2.0
    grad = _weighted_grad(h, gamma)
    grad_c = grad - np.mean(grad)
    gn = np.linalg.norm(grad_c)
    if gn < 1e-15:
        return h, _exact_c5(h)
    d = -grad_c / gn
    cur_c5 = _exact_c5(h)
    best_a = 0.0
    best_c5 = cur_c5
    alpha = 0.05
    for _ in range(15):
        if time.time() > deadline:
            break
        h_try = h + alpha * d
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_a = alpha
        alpha *= 0.6
        if alpha < 1e-8:
            break
    if best_a > 0:
        h_new = _project(h + best_a * d, target)
        return h_new, best_c5
    return h, cur_c5
