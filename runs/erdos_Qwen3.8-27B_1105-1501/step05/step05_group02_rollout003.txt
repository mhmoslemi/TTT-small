def _solve_min_cut(n, edge_list, unary_source, unary_sink):
    """
    Solve min-cut problem.
    n: number of variable nodes (0 to n-1)
    edge_list: list of (u, v, capacity) for directed edges between variable nodes
    unary_source: array of length n, capacity of edge source->i
    unary_sink: array of length n, capacity of edge i->sink
    Returns: binary array x where x[i]=1 if node i is on source side
    """
    S = n  # source
    T = n + 1  # sink
    total_nodes = n + 2
    
    # Build graph
    adj = [[] for _ in range(total_nodes)]
    caps = []
    
    def add_edge(u, v, c):
        idx = len(caps)
        caps.append(c)
        adj[u].append([v, idx])
        caps.append(0.0)
        adj[v].append([u, idx + 1])
    
    # Add unary edges
    for i in range(n):
        if unary_source[i] > 1e-15:
            add_edge(S, i, unary_source[i])
        if unary_sink[i] > 1e-15:
            add_edge(i, T, unary_sink[i])
    
    # Add pairwise edges
    for u, v, c in edge_list:
        if c > 1e-15:
            add_edge(u, v, c)
    
    # Dinic's algorithm
    level = [0] * total_nodes
    it = [0] * total_nodes
    
    def bfs():
        for i in range(total_nodes):
            level[i] = -1
        level[S] = 0
        queue = [S]
        for qi in range(len(queue)):
            u = queue[qi]
            for v, eidx in adj[u]:
                if caps[eidx] > 1e-12 and level[v] < 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        return level[T] >= 0
    
    def dfs(u, f):
        if u == T:
            return f
        while it[u] < len(adj[u]):
            v, eidx = adj[u][it[u]]
            if caps[eidx] > 1e-12 and level[v] == level[u] + 1:
                d = dfs(v, min(f, caps[eidx]))
                if d > 0:
                    caps[eidx] -= d
                    caps[eidx + 1] += d
                    return d
            it[u] += 1
        return 0
    
    max_flow = 0.0
    INF = 1e18
    while bfs():
        for i in range(total_nodes):
            it[i] = 0
        while True:
            f = dfs(S, INF)
            if f == 0:
                break
            max_flow += f
    
    # Get min cut: nodes reachable from S in residual graph
    reachable = [False] * total_nodes
    reachable[S] = True
    stack = [S]
    while stack:
        u = stack.pop()
        for v, eidx in adj[u]:
            if caps[eidx] > 1e-12 and not reachable[v]:
                reachable[v] = True
                stack.append(v)
    
    x = np.array([1.0 if reachable[i] else 0.0 for i in range(n)])
    return x, max_flow

def _dinic_solve(n_vars, edges, S, T):
    """
    Solve max-flow min-cut using Dinic's algorithm.
    n_vars: number of variable nodes (0 to n_vars-1)
    edges: list of (u, v, capacity)
    S, T: source and sink node indices
    
    Returns: (min_cut_set, max_flow)
    min_cut_set: boolean array of length n_vars, True if node is on source side
    """

    adj = [[] for _ in range(n_nodes)]
    cap = []
    
    def add_edge(u, v, c):
        idx = len(cap)
        cap.append(c)
        cap.append(0.0)
        adj[u].append([v, idx])
        adj[v].append([u, idx + 1])
