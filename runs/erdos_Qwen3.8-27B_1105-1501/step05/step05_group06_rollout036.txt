def _de_step(pop, fitnesses, F, CR, rng):
    """One step of DE/rand/1/bin"""
    P, n = pop.shape
    for i in range(P):
        # Select three distinct indices different from i
        idx = [j for j in range(P) if j != i]
        a, b, c = rng.choice(idx, 3, replace=False)
        # Mutation
        v = pop[a] + F * (pop[b] - pop[c])
        # Crossover
        j0 = rng.integers(n)
        for j in range(n):
            if rng.random() < CR or j == j0:
                pop[i][j] = v[j]
        # Project
        pop[i] = _project(pop[i], n / 2.0)
        # Selection
        if _exact_c5(pop[i]) < fitnesses[i]:
            fitnesses[i] = _exact_c5(pop[i])
    return pop, fitnesses

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lp_direction(h, G, radius, top_k):
    n = len(h)
    scores = _scores(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_1d(h, p, target):
    """Find best step size along direction p using projected line search."""
    n = len(h)
    # Find max step before violating bounds
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def eval_at(alpha):
        h_try = _project(h + alpha * p, target)
        return _exact_c5(h_try)
    
    f1 = eval_at(c1)
    f2 = eval_at(c2)
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 <= f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = eval_at(c1)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = eval_at(c2)
    opt_a = (lo + hi) / 2.0
    return opt_a

def _pair_descent(h, deadline, max_passes=200):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    target = n / 2.0
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Build G and scores once per pass
        G = _build_G(h)
        scores = _scores(h)
        for j in range(n):
            if time.time() > deadline:
                break
            h_j = h[j]
            if h_j < 1e-10 or h_j > 1 - 1e-10:
                # Still could move (if h_j is near 0, can only increase; if near 1, can only decrease)
                pass
            for k in range(j + 1, n):
                h_k = h[k]
                # Move alpha from j to k: h[j] -= alpha, h[k] += alpha
                # Or from k to j: h[j] += alpha, h[k] -= alpha
                # Net effect: scores change by alpha * (G[:, j] - G[:, k]) or negative
                C = G[:, j] - G[:, k]
                # Try h[j] += a, h[k] -= a for a in [max(-h[j], -(1-h[k])), min(1-h[j], h[k])]
                a_lo = max(-h_j, h_k - 1.0)
                a_hi = min(1.0 - h_j, h_k)
                if a_hi - a_lo < 1e-12:
                    continue
                # Minimize max(scores + a*C) over a in [a_lo, a_hi]
                # Ternary search
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 <= f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > 1e-7 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    # Fix sum
                    h = _project(h, target)
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        G = _build_G(h)
                        scores = _scores(h)
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        h = _project(h, target)
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    best_idx = int(np.argmin(fitnesses))

        for i in range(pop_size):

                fitnesses[i] = f_trial

                    fitnesses[bi] = c5_loc

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lp_direction(h, G, radius, top_k):
    n = len(h)
    scores = _scores(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, target):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h1 = _project(h + c1 * p, target)
    h2 = _project(h + c2 * p, target)
    f1 = _exact_c5(h1)
    f2 = _exact_c5(h2)
    for _ in range(28):
        if hi - lo < 1e-12:
            break
        if f1 <= f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            h1 = _project(h + c1 * p, target)
            f1 = _exact_c5(h1)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - c1)
            h2 = _project(h + c2 * p, target)
            f2 = _exact_c5(h2)
    opt_a = (lo + hi) / 2.0
    h_opt = _project(h + opt_a * p, target)
    return opt_a, h_opt

def _local_opt(h, deadline, top_k_list=None, radius_init=0.1, max_iters=3000):
    if top_k_list is None:
        top_k_list = [None, 10, 6, 4]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_tk = len(top_k_list)
    ti = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        tk = top_k_list[ti % n_tk]
        G = _build_G(h)
        p = _lp_direction(h, G, radius, tk)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.65
            ti += 1
            if radius < 1e-9:
                break
            continue
        alpha, h_cand = _line_search(h, p, target)
        if alpha < 1e-12:
            stall += 1
            radius *= 0.7
            ti += 1
            continue
        actual_c5 = _exact_c5(h_cand)
        if actual_c5 < cur_c5 - 1e-14:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.7
            ti += 1
            if stall > 15 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=200, thresh=1e-7):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    target = n / 2.0
    G = _build_G(h)
    scores = _scores(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            hj = h[j]
            for k in range(j + 1, n):
                hk = h[k]
                if hj < 1e-10 and hk < 1e-10:
                    continue
                if hj > 1 - 1e-10 and hk > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-hj, hk - 1.0)
                a_hi = min(1.0 - hj, hk)
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 <= f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, target)
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        G = _build_G(h)
                        scores = _scores(h)
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        h = _project(h, target)
                        G = _build_G(h)
                        scores = _scores(h)
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _alpha_sweep(n, deadline, rng):
    """Sweep quasi-periodic alphas to find a good starting point."""
    best_h = None
    best_c5 = 1.0
    alphas = np.concatenate([
        np.linspace(0.1, 0.9, 40),
        [1/phi + k/100 for k in range(20)],  # golden ratio based
        [np.sqrt(2)-1+k/200 for k in range(20)],
    ])
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = _quasi_init(n, alpha, rng)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if best_h is None:
        best_h = _quasi_init(n, 0.618, rng)
        best_c5 = _exact_c5(best_h)
    return best_h, best_c5

def _de_phase(h_best, n, deadline, rng, pop_size=16, F=0.35, CR=0.8):
    """DE with periodic local polish of best member."""
    target = n / 2.0
    pop = np.zeros((pop_size, n))
    pop[0] = h_best
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.3:
            pop[i] = h_best + rng.normal(0, 0.04, n)
        elif r < 0.6:
            alpha = 0.1 + 0.8 * rng.random()
            x = np.arange(n) / n
            pop[i] = ((x * alpha) % 1.0) * 2.0
            pop[i] = (pop[i] > 1.0).astype(float)
        elif r < 0.8:
            x = np.arange(n) / n
            pop[i] = 0.5 + 0.5 * np.sin(2 * np.pi * (1 + rng.integers(1, 8)) * x + rng.uniform(0, 6.28))
        else:
            pop[i] = rng.uniform(0, 1, n)
        pop[i] = _project(pop[i], target)
    fitnesses = np.array([_exact_c5(pop[i]) for i in range(pop_size)])
    
    local_every = 25
    count = 0
    while time.time() < deadline:
        for i in range(pop_size):
            if time.time() > deadline:
                break
            idx = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(idx, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            j0 = rng.integers(n)
            mask = rng.random(n) < CR
            mask[j0] = True
            trial = np.where(mask, v, pop[i])
            trial = _project(trial, target)
            f_trial = _exact_c5(trial)
            if f_trial <= fitnesses[i]:
                pop[i] = trial
                fitnesses[i] = f_trial
            count += 1
            if count >= local_every:
                count = 0
                bi = int(np.argmin(fitnesses))
                sub_dl = min(deadline, time.time() + 4)
                h_loc, c5_loc = _local_opt(pop[bi], sub_dl,
                                           top_k_list=[6, 4, None],
                                           radius_init=0.06, max_iters=150)
                if c5_loc < fitnesses[bi]:
                    pop[bi] = h_loc
                    fitnesses[bi] = c5_loc
    bi = int(np.argmin(fitnesses))
    return pop[bi].copy(), fitnesses[bi]

def _grad_step(h, deadline, n_steps=150):
    """Softmax-weighted gradient descent on the max of scores."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        beta = 20.0 + 2.0 * step
        scores = _scores(h)
        max_s = scores.max()
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        G = _build_G(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.05
        for _ in range(20):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            alpha *= 0.6
            if alpha < 1e-10:
                break
    return best_h, best_c5
