import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute the gradient matrix G where G[k,i] = dS_k/dh_i * dx"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _find_lambda(q, n, target):
    """Find lambda such that sum(sigma(q_i - lambda)) = target"""
    def sigma(x):
        # overflow-safe sigmoid
        result = np.zeros_like(x)
        pos = x >= 0
        result[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
        exp_x = np.exp(x[~pos])
        result[~pos] = exp_x / (1.0 + exp_x)
        return result
    
    lo = float(q.min()) - 10.0
    hi = float(q.max()) + 10.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(sigma(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

    z = q - lam
    h = np.zeros_like(z)
    pos = z >= 0
    h[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    exp_z = np.exp(z[~pos])
    h[~pos] = exp_z / (1.0 + exp_z)
    return np.clip(h, 1e-15, 1.0 - 1e-15)

def _q_from_h(h):
    """Convert h to q-space (logit transform), find lambda"""
    h = np.clip(h, 1e-10, 1.0 - 1e-10)
    q = np.log(h / (1.0 - h))
    n = len(h)
    lam = _find_lambda(q, n, n / 2.0)
    return q, lam

def _smooth_obj_and_grad(q, beta, n):
    """Log-sum-exp smooth objective and its gradient in q-space"""
    target = n / 2.0
    lam = _find_lambda(q, n, target)
    h = _h_from_q(q, lam)
    
    S = _scores(h)
    maxS = np.max(S)
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w /= wsum
    
    obj = maxS + np.log(np.sum(np.exp(shifted))) / beta
    
    # Gradient in h-space
    G = _grad_matrix(h)
    g_h = G.T @ w  # gradient of smooth obj w.r.t. h
    
    # Transform to q-space
    # h_i = sigma(q_i - lam), so dh_i/dq_i = h_i*(1-h_i)
    # dlam/dq_j = h_j*(1-h_j) / sum_k h_k*(1-h_k)
    s = h * (1.0 - h)
    s_sum = np.sum(s)
    if s_sum < 1e-15:
        g_q = g_h * s
    else:
        g_q = s * (g_h - np.sum(g_h * s) / s_sum)
    
    return obj, g_q

def _lbfgs_optimize(q, deadline, beta_schedule, max_iter_per_phase=300):
    """Run L-BFGS-B optimization with beta schedule in q-space"""
    n = len(q)
    target = n / 2.0
    best_h = _h_from_q(q, _find_lambda(q, n, target))
    best_c5 = _exact_c5(best_h)
    
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        
        def obj_wrapper(qw):
            o, _ = _smooth_obj_and_grad(qw, beta, n)
            return o
        
        def grad_wrapper(qw):
            _, g = _smooth_obj_and_grad(qw, beta, n)
            return g
        
        try:
            res = minimize(obj_wrapper, q, jac=grad_wrapper,
                          method='L-BFGS-B',
                          options={'maxiter': max_iter_per_phase, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                lam = _find_lambda(res.x, n, target)
                h_new = _h_from_q(res.x, lam)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                q = res.x
        except Exception:
            pass
    
    return best_h, best_c5, q

        if np.all(np.isfinite(x)):
            return x
    except Exception:
        pass
    return q


def _optimize_qspace(h, deadline):
    n = len(h)
    q, _ = _q_from_h(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(best_h)
    betas = [15, 30, 60, 120, 250, 500, 1000]
    for beta in betas:
        if time.time() > deadline:
            break
        q = _lbfgs_phase(q, beta, n, max_iter=200)
        lam = _find_lambda(q, n, target)
        h_new = _h_from_q(q, lam)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5


def _lp_optimize(h, deadline, radius_init=0.2, max_iters=3000, stall_limit=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        n_lags = len(scores)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.8
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        if float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.999
        n_samp = 12
        alphas = np.linspace(0, alpha_max, n_samp)
        vals = np.array([_exact_c5(h + a * p) for a in alphas])
        best_i = int(np.argmin(vals))
        best_tried = vals[best_i]
        best_a = alphas[best_i]
        if best_i not in (0, n_samp - 1):
            lo_a = alphas[best_i - 1]
            hi_a = alphas[best_i + 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_a - gr * (hi_a - lo_a)
            c2 = lo_a + gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
            f2 = _exact_c5(h + c2 * p)
            for _ in range(12):
                if hi_a - lo_a < 1e-14:
                    break
                if f1 < f2:
                    hi_a = c2
                    f2 = f1
                    c2 = c1
                    c1 = hi_a - gr * (hi_a - lo_a)
                    f1 = _exact_c5(h + c1 * p)
                else:
                    lo_a = c1
                    f1 = f2
                    c1 = c2
                    c2 = lo_a + gr * (hi_a - lo_a)
                    f2 = _exact_c5(h + c2 * p)
            opt_a = (lo_a + hi_a) / 2.0
            opt_c5 = _exact_c5(h + opt_a * p)
            if opt_c5 < best_tried:
                best_tried = opt_c5
                best_a = opt_a
        if best_tried < cur_c5 - 1e-14:
            h_new = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.85
            if stall > 15 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quasi_init(n, alpha):
    x = np.arange(n)
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)


def _subgrad_step(h, beta, alpha):
    n = len(h)
    target = n / 2.0
    S = _scores(h)
    maxS = np.max(S)
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w = w / wsum
    G = _grad_matrix(h)
    g = G.T @ w
    g = g - np.sum(g) * (1.0 / n)
    h_new = _project(h - alpha * g, target)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: q-space L-BFGS-B with coarse-to-fine beta
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _optimize_qspace(h_start, dl1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # LP polish
    dl1b = t0 + budget_s * 0.30
    h1b, c5_1b = _lp_optimize(best_h, dl1b, radius_init=0.12, max_iters=4000, stall_limit=120)
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b.copy()

    # Phase 2: High-beta q-space polish
    if time.time() < t0 + budget_s * 0.40:
        dl2 = t0 + budget_s * 0.40
        q2, _ = _q_from_h(best_h)
        n2 = len(q2)
        target2 = n2 / 2.0
        best_h2 = best_h.copy()
        best_c5_2 = best_c5
        for beta in [300, 600, 1200, 2000]:
            if time.time() > dl2:
                break
            q2 = _lbfgs_phase(q2, beta, n2, max_iter=200)
            lam2 = _find_lambda(q2, n2, target2)
            h_try = _h_from_q(q2, lam2)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_2:
                best_c5_2 = c5_try
                best_h2 = h_try.copy()
        if best_c5_2 < best_c5:
            best_c5 = best_c5_2
            best_h = best_h2.copy()

    # Phase 3: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.70:
        dl3 = t0 + budget_s * 0.70
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
        for n_try in [96, 128, 160]:
            if time.time() > dl3:
                break
            # Resample from best
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl3, time.time() + budget_s * 0.035)
            h_t, c5_t = _optimize_qspace(h_init, sub_dl)
            if time.time() < dl3:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.02)
                h_t2, c5_t2 = _lp_optimize(h_t, sub_dl2, radius_init=0.1, max_iters=2000, stall_limit=80)
                if c5_t2 < c5_t:
                    h_t, c5_t = h_t2, c5_t2
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try
            # Quasi-periodic start
            if time.time() < dl3:
                idx = 0
                for ci in range(2):
                    if time.time() > dl3:
                        break
                    h_qp = _quasi_init(n_try, alphas[(idx + ci) % len(alphas)])
                    h_qp = h_qp + rng.normal(0, 0.04, n_try)
                    h_qp = _project(h_qp, n_try / 2.0)
                    sub_dl3 = min(dl3, time.time() + budget_s * 0.025)
                    h_qt, c5_qt = _optimize_qspace(h_qp, sub_dl3)
                    if time.time() < dl3:
                        sub_dl3b = min(dl3, time.time() + budget_s * 0.015)
                        h_qt2, c5_qt2 = _lp_optimize(h_qt, sub_dl3b, radius_init=0.08, max_iters=1500, stall_limit=60)
                        if c5_qt2 < c5_qt:
                            h_qt, c5_qt = h_qt2, c5_qt2
                    if c5_qt < best_c5:
                        best_c5 = c5_qt
                        best_h = h_qt.copy()
                        best_n = n_try
                    idx += 1

    # Phase 4: Perturbation restarts in q-space
    if time.time() < t0 + budget_s * 0.88:
        dl4 = t0 + budget_s * 0.88
        q_best, _ = _q_from_h(best_h)
        scale = 0.4
        for attempt in range(300):
            if time.time() > dl4:
                break
            q_pert = q_best + rng.normal(0, scale, best_n)
            sub_dl = min(dl4, time.time() + budget_s * 0.006)
            q_res = _lbfgs_phase(q_pert, 100, best_n, max_iter=80)
            lam_r = _find_lambda(q_res, best_n, best_n / 2.0)
            h_pert = _h_from_q(q_res, lam_r)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl2 = min(dl4, time.time() + budget_s * 0.003)
            h_lp, c5_lp = _lp_optimize(h_pert, sub_dl2, radius_init=0.05, max_iters=500, stall_limit=40)
            if c5_lp < best_c5 - 1e-14:
                best_c5 = c5_lp
                best_h = h_lp.copy()
                q_best = _q_from_h(best_h)[0]
                scale = max(0.1, scale * 0.7)
            else:
                scale = min(1.5, scale * 1.12)

    # Phase 5: Subgradient diversification
    if time.time() < t0 + budget_s * 0.93:
        dl5 = t0 + budget_s * 0.93
        n_sub = min(best_n, 96)
        h_sub = _resample(best_h, n_sub)
        best_sub_c5 = _exact_c5(h_sub)
        for step in range(2000):
            if time.time() > dl5:
                break
            frac = step / 1999.0
            alpha_step = 0.08 * (1.0 - frac) + 0.01
            h_sub = _subgrad_step(h_sub, 50.0, alpha_step)
            c5_sub = _exact_c5(h_sub)
            if c5_sub < best_sub_c5 - 1e-10:
                best_sub_c5 = c5_sub
        if best_sub_c5 < _exact_c5(best_h):
            h_up = _resample(h_sub, best_n)
            h_up, c5_up = _lp_optimize(h_up, min(dl5, time.time() + budget_s * 0.02),
                                       radius_init=0.1, max_iters=1500, stall_limit=60)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()

    # Phase 6: Final high-beta polish
    if time.time() < t0 + budget_s * 0.97:
        dl6 = t0 + budget_s * 0.97
        q_final, _ = _q_from_h(best_h)
        n_f = len(q_final)
        target_f = n_f / 2.0
        best_h6 = best_h.copy()
        best_c5_6 = best_c5
        for beta in [500, 1000, 2000, 4000]:
            if time.time() > dl6:
                break
            q_final = _lbfgs_phase(q_final, beta, n_f, max_iter=150)
            lam_f = _find_lambda(q_final, n_f, target_f)
            h_f = _h_from_q(q_final, lam_f)
            c5_f = _exact_c5(h_f)
            if c5_f < best_c5_6:
                best_c5_6 = c5_f
                best_h6 = h_f.copy()
        if best_c5_6 < best_c5:
            best_c5 = best_c5_6
            best_h = best_h6.copy()

    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_final, c5_final = _lp_optimize(best_h, t0 + budget_s * 0.992,
                                         radius_init=0.04, max_iters=2000, stall_limit=80)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
