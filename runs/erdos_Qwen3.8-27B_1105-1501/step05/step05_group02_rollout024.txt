import numpy as np
import time
from scipy.optimize import linprog

def _c5(h):
    n = len(h)
    dx = 2.0/n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0/n
    return np.correlate(h, 1.0-h, mode='full') * dx

def _project(v, target):
    # Project v onto {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        s = np.sum(np.clip(v-mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v-(lo+hi)/2, 0, 1)
    # Fix residual
    r = target - x.sum()
    if abs(r) > 1e-12:
        mask = (x < 1.0) if r > 0 else (x > 0.0)
        x[mask] += r / mask.sum()
        x = np.clip(x, 0, 1)
    return x

def _grad_max(h, scores=None):
    """Compute subgradient of max_k g_k(h)."""
    n = len(h)
    if scores is None:
        scores = _scores(h)
    # Find active set (scores near max)
    max_s = scores.max()
    active = np.where(scores >= max_s - 1e-6)[0]
    # Compute gradient of active scores, average them
    g = np.zeros(n)
    for m in active:
        d = m - (n-1)  # shift
        if d == 0:
            g += (1 - 2*h)
        elif d > 0:
            # g_m = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # dg/dh[i] for i in [0, n-1): 
            #   from term i (if i < n-d): (1-h[i+d])
            #   from term i-d (if i>=d and i-d < n-d): -h[i-d]
            g[:n-d] += (1 - h[d:])
            g[d:] += -h[:n-d]
        else:
            # d < 0, let g = -d > 0
            gg = -d
            # g_m = sum_{i=g}^{n-1} h[i]*(1-h[i-g])
            # dg/dh[i]: from term i (if i>=g): (1-h[i-g])
            #            from term i+g (if i+g < n): -h[i+g]
            g[g:] += (1 - h[:n-gg])
            g[:n-gg] += -h[gg:]
    # Average over active set
    g /= len(active)
    # Center (subtract mean) to preserve sum
    g -= g.mean()
    return g

def _line_search_exact(h, d, t_lo, t_hi, n_gauss=40):
    """Exact line search: minimize max_k g_k(h + t*d)."""
    n = len(h)
    one_m_h = 1.0 - h
    scores_now = np.correlate(h, one_m_h, mode='full')
    
    # Precompute quadratic coefficients
    # g_k(h+t*d) = -a[k]*t^2 + b[k]*t + c[k]
    c = scores_now
    b = np.correlate(d, one_m_h, 'full') - np.correlate(h, d, 'full')
    a = np.correlate(d, d, 'full')
    
    def f(t):
        return np.max(-a*t*t + b*t + c)
    
    # Golden section
    gr = (np.sqrt(5)-1)/2
    lo, hi = t_lo, t_hi
    if hi - lo < 1e-15:
        mid = (lo+hi)/2
        return mid, f(mid)
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(n_gauss):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr*(hi-lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr*(hi-lo)
            f2 = f(c2)
    t_opt = (lo+hi)/2
    return t_opt, f(t_opt)

def _feasible_interval(h, d):
    """Find [t_lo, t_hi] such that 0 <= h+t*d <= 1 for all t in [t_lo, t_hi]."""
    nz = np.nonzero(np.abs(d) > 1e-15)[0]
    if len(nz) == 0:
        return 0.0, 0.0
    t_lo = 0.0
    t_hi = 1.0
    for i in nz:
        if d[i] > 1e-15:
            t_lo = max(t_lo, -h[i]/d[i])
            t_hi = min(t_hi, (1.0-h[i])/d[i])
        else:
            t_lo = max(t_lo, (1.0-h[i])/d[i])  # (h[i]-1)/d[i] = (1-h[i])/|d[i]|
            t_hi = min(t_hi, -h[i]/d[i])  # h[i]/|d[i]|
    return t_lo, t_hi
