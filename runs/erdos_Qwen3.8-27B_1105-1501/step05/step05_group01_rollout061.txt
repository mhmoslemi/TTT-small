def _project(v, target):
    # Project v onto {0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0  # wide enough

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

    # Greedy pair-swap local search on binary sequence
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    # Make it exactly binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

        improved = False

            if time.time() > deadline:
                return best_h, best_c5
            oi = ones[i_idx]

def _line_search(h, d, dx, radius):
    # Robust line search with grid sampling
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_v = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q_v = np.correlate(d, d, mode='full')
    
    # Find valid interval
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, -float(np.min(h[neg] / (-d[neg]))))
        # Actually: h[neg] + t*d[neg] >= 0 → t >= -h[neg]/d[neg] = h[neg]/(-d[neg])
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid search
    n_grid = 30
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([float(np.max(S0 + t * T_v - t*t * Q_v)) * dx for t in ts])
    idx_min = np.argmin(vals)
    
    # Refine around best
    lo = max(t_lo, ts[max(0, idx_min-1)])
    hi = min(t_hi, ts[min(n_grid-1, idx_min+1)])
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T_v - c1*c1 * Q_v)) * dx
    f2 = float(np.max(S0 + c2 * T_v - c2*c2 * Q_v)) * dx
    for _ in range(50):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T_v - c1*c1 * Q_v)) * dx
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T_v - c2*c2 * Q_v)) * dx
    
    t_opt = (lo + hi) / 2.0
    v_opt = float(np.max(S0 + t_opt * T_v - t_opt*t_opt * Q_v)) * dx
    return t_opt, v_opt

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

But since the objective is not convex (it's a max of non-convex functions... wait, each f_l is a quadratic in h, which is not convex), this doesn't directly apply.

Actually, f_l(h) = dx * sum_i h[i] * (1-h[i+l]) is a quadratic form in h. Specifically, f_l(h) = dx * (sum_i h[i] - sum_i h[i]*h[i+l]). The second term is a quadratic form. So f_l is not convex.

But the max of non-convex functions is also non-convex. So we can't use convex optimization directly.

However, we can use a "smooth implicit function" approach:
- Use a smooth approximation to the max: max_l f_l ≈ (1/β) * log(sum_l exp(β*f_l))
- Optimize the smooth approximation
- As β → ∞, the smooth approximation → the true max

This is the "log-sum-exp" smoothing. For large β, it's a good approximation of the max, and it's smooth, so we can use gradient-based methods.

Let me implement this:
1. Compute all f_l values
2. Compute the softmax weights: w_l = exp(β*f_l) / sum exp(β*f_l)
3. The gradient of the smooth max is: sum_l w_l * grad(f_l)
4. Take a step in the negative gradient direction
5. Project to feasible set
6. Increase β over time

This is a well-known technique in nonsmooth optimization. Let me implement it.

The gradient of f_l(h) with respect to h[i] is:
- For the term h[i]*(1-h[i+l]): derivative w.r.t. h[i] is (1-h[i+l]), derivative w.r.t. h[i+l] is -h[i]
- So grad(f_l)[i] = dx * sum over relevant lags of appropriate terms

This is exactly what the _grad_matrix function in the previous code computes!

def _smooth_grad_descent(h, deadline, beta_init=50, beta_max=500, lr_init=0.1):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    
    while time.time() < deadline:
        if no_improve > 200:
            break
        
        # Compute scores
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Softmax weights
        beta_clipped = min(beta, 1000)
        exp_scores = np.exp(beta_clipped * (scores - scores.max()))
        w = exp_scores / exp_scores.sum()
        
        # Gradient: G^T @ w where G is the gradient matrix
        # grad_l[i] = derivative of f_l w.r.t. h[i]
        # For lag corresponding to index r in the full correlation:
        #   d >= 0: f_r = sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
        #   d < 0: f_r = sum_{i=-d}^{n-1} h[i]*(1-h[i+d]) where d is negative
        # 
        # grad[f_r] w.r.t. h[i]:
        #   If i is a "first" index (contributing h[i]): += (1-h[i+d])
        #   If i is a "second" index (contributing 1-h[i]): -= h[i-d]
        
        # Build gradient
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-10:
                continue
            d = r - (n - 1)  # signed lag
            if d >= 0:
                m = n - d
                for i in range(m):
                    grad[i] += w[r] * (1.0 - h[i + d])
                    grad[i + d] += w[r] * (-h[i])
            else:
                g = -d
                m = n - g
                for i in range(m):
                    grad[i + g] += w[r] * (1.0 - h[i])
                    grad[i] += w[r] * (-h[i + g])
        
        # Gradient descent step
        h_new = h - lr * grad
        h_new = _project(h_new, target)
        
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            no_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve % 20 == 0:
                beta = min(beta * 1.1, beta_max)
        
        # Occasionally reset learning rate
        if no_improve % 50 == 0:
            lr = lr_init * 0.5
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    Tv = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Qv = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([float(np.max(S0 + t * Tv - t * t * Qv)) * dx for t in ts])
    idx_min = int(np.argmin(vals))
    lo = max(t_lo, ts[max(0, idx_min - 1)])
    hi = min(t_hi, ts[min(n_grid - 1, idx_min + 1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * Tv - c1 * c1 * Qv)) * dx
    f2 = float(np.max(S0 + c2 * Tv - c2 * c2 * Qv)) * dx
    for _ in range(50):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * Tv - c1 * c1 * Qv)) * dx
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * Tv - c2 * c2 * Qv)) * dx
    t_opt = (lo + hi) / 2.0
    v_opt = float(np.max(S0 + t_opt * Tv - t_opt * t_opt * Qv)) * dx
    return t_opt, v_opt

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m - d]) if d > 0 else (1.0 - h[0:m])
                # Fix: G[r, :m] += (1.0 - h[d:d+m]) doesn't work if d>0
                # Let me redo this properly
                pass
        else:
            pass
    # Redo properly
    G = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        # Positive lag d: f = sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
        r = n - 1 + d
        m = n - d
        if m > 0:
            for i in range(m):
                G[r, i] += (1.0 - h[i + d])
                G[r, i + d] += (-h[i])
    for d in range(1, n):
        # Negative lag -d: f = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
        r = n - 1 - d
        m = n - d
        for i in range(m):
            G[r, i + d] += (1.0 - h[i])
            G[r, i] += (-h[i + d])
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad_step(h, beta, lr):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    shift = scores.max()
    exp_s = np.exp(beta * (scores - shift))
    w = exp_s / exp_s.sum()
    G = _grad_matrix(h)
    grad = G.T @ w
    h_new = h - lr * grad
    h_new = _project(h_new, target)
    return h_new

def _continuous_refine(h, deadline, radius_init=0.06, use_lp=True, use_smooth=True):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best = cur
    best_h = h.copy()
    radius = radius_init
    stall = 0
    beta = 80.0
    lr_smooth = 0.08
    use_lp_flag = use_lp
    use_smooth_flag = use_smooth

    while time.time() < deadline:
        if stall > 120:
            break
        improved = False
        # Try LP direction
        if use_lp_flag:
            scores = np.correlate(h, 1.0 - h, mode='full')
            G = _grad_matrix(h)
            p = _lp_direction(h, scores, G, radius)
            if p is not None and float(np.max(np.abs(p))) > 1e-12:
                alpha, c5_try = _line_search(h, p, dx, radius)
                if c5_try < cur - 1e-13:
                    h = h + alpha * p
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    improved = True
                    if cur < best:
                        best = cur
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.03, 0.3)
                    continue
        # Try smooth gradient
        if use_smooth_flag:
            h_sm = _smooth_grad_step(h, beta, lr_smooth)
            c5_sm = _exact_c5(h_sm)
            if c5_sm < cur - 1e-13:
                h = h_sm
                cur = c5_sm
                improved = True
                if cur < best:
                    best = cur
                    best_h = h.copy()
                stall = 0
                continue
        # Random pair perturbation
        idx1 = int(np.random.randint(0, n))
        idx2 = int(np.random.randint(0, n))
        if idx1 != idx2:
            d = np.zeros(n)
            d[idx1] = 1.0
            d[idx2] = -1.0
            alpha, c5_try = _line_search(h, d, dx, 0.15)
            if c5_try < cur - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                improved = True
                if cur < best:
                    best = cur
                    best_h = h.copy()
                stall = 0
        if not improved:
            stall += 1
            radius *= 0.75
            beta = min(beta * 1.05, 300.0)
            lr_smooth *= 0.85
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.06
                lr_smooth = 0.05

    return best_h, best

def _binary_greedy(h_bin, deadline, max_passes=300, n_random=5):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    cur = _exact_c5(h)
    best = cur
    best_h = h.copy()
    rng = np.random.default_rng(12345)

    for p in range(max_passes):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        # First try random pair swaps (fast)
        for _ in range(n_random * len(ones)):
            if time.time() > deadline:
                return best_h, best
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c5_try = _exact_c5(h)
            if c5_try < cur - 1e-15:
                cur = c5_try
                improved = True
                if cur < best:
                    best = cur
                    best_h = h.copy()
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        # If no random improvement, try systematic
        if not improved:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            for i in ones:
                if time.time() > deadline:
                    return best_h, best
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c5_try = _exact_c5(h)
                    if c5_try < cur - 1e-15:
                        cur = c5_try
                        improved = True
                        if cur < best:
                            best = cur
                            best_h = h.copy()
                        break
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
                if improved:
                    break
        if not improved:
            break
    return best_h, best

def _binary_sa(h_bin, deadline, T_init=0.04, T_min=0.0005, max_iter=80000):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    cur = _exact_c5(h)
    best = cur
    best_h = h.copy()
    rng = np.random.default_rng(9999)
    T = T_init

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if T < T_min:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_try = _exact_c5(h)
        delta = c5_try - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = c5_try
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= 0.99985
    return best_h, best

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return _project(h.copy(), n_new / 2.0)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _block_swap_search(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best = cur
    best_h = h.copy()
    rng = np.random.default_rng(55555)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(200):
            if time.time() > deadline:
                return best_h, best
            # Pick a block of 1-3 consecutive positions to move
            block_len = int(rng.integers(1, 4))
            start = int(rng.integers(0, max(1, n - block_len)))
            block = h[start:start + block_len]
            if block.sum() <= 0:
                continue
            # Pick destination
            dest = int(rng.integers(0, max(1, n - block_len)))
            if dest == start:
                continue
            d = np.zeros(n)
            for k in range(block_len):
                d[start + k] = -h[start + k]
                d[dest + k] = h[start + k]
            if np.allclose(d, 0):
                continue
            alpha, c5_try = _line_search(h, d, dx, 0.4)
            if c5_try < cur - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                improved = True
                if cur < best:
                    best = cur
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)

    d1 = t0 + budget_s * 0.08

    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n binary search (42%)
    d2 = t0 + budget_s * 0.50
    n_list = [n0]
    for dn in range(-40, 61, 4):
        nt = n0 + dn
        if 100 <= nt <= 320 and nt != n0:
            n_list.append(nt)
    # Shuffle non-first elements
    rest = n_list[1:]
    rng.shuffle(rest)
    n_list = [n0] + rest

    candidates = []
    for idx_n, n_try in enumerate(n_list):
        if time.time() > d2:
            break
        n_remaining = len(n_list) - idx_n
        time_per_n = max(1.5, (d2 - time.time()) / max(1, n_remaining))
        sub_dl = min(d2, time.time() + time_per_n)

        n_starts = 4 if n_try == n0 else 2
        best_c5_n = 1e9
        best_h_n = None

        for s in range(n_starts):
            if time.time() > sub_dl:
                break
            if s == 0 and n_try == n0:
                h_r = _project(best_h, float(n_try) / 2.0)
                h_bin = (h_r > 0.5).astype(float)
            elif s == 1:
                # Structured: alternating with perturbation
                h_r = np.zeros(n_try)
                h_r[::2] = 1.0
                h_r = _project(h_r, float(n_try) / 2.0)
                h_bin = (h_r > 0.5).astype(float)
            else:
                h_bin = np.zeros(n_try)
                idx_sel = rng.permutation(n_try)[:n_try // 2]
                h_bin[idx_sel] = 1.0

            time_budget = (sub_dl - time.time()) * 0.7
            h_g, c5_g = _binary_greedy(h_bin, time.time() + max(1, time_budget))
            if c5_g < best_c5_n:
                best_c5_n = c5_g
                best_h_n = h_g.copy()

            # Quick SA if time
            if time.time() < sub_dl - 2 and best_c5_n < best_c5 + 0.02:
                h_sa, c5_sa = _binary_sa(h_g, time.time() + 2.0, T_init=0.03, max_iter=20000)
                if c5_sa < best_c5_n:
                    best_c5_n = c5_sa
                    best_h_n = h_sa.copy()

        if best_h_n is not None:
            candidates.append((best_c5_n, best_h_n, n_try))
            if best_c5_n < best_c5:
                best_c5 = best_c5_n
                best_h = best_h_n.copy()

    candidates.sort(key=lambda x: x[0])

    # Phase 3: Continuous refinement of top candidates (25%)
    d3 = t0 + budget_s * 0.75
    top_k = min(10, len(candidates))
    if top_k > 0:
        time_per_cand = max(3.0, (d3 - time.time()) / top_k)
        for idx_c in range(top_k):
            if time.time() > d3:
                break
            c5_c, h_c, n_c = candidates[idx_c]
            if c5_c > best_c5 + 0.002:

            sub_dl = min(d3, time.time() + time_per_cand)

            if c5_r < best_c5:

            # Try block swap

                    best_h = h_bs.copy()

    # Phase 4: Basin hopping (20%)
    d4 = t0 + budget_s * 0.93
    n_bh = len(best_h)
    scale = 0.06
    h_bh = best_h.copy()
    for attempt in range(800):
        if time.time() > d4:
            break
        h_pert = h_bh + rng.normal(0, scale, n_bh)
        h_pert = _project(h_pert, float(n_bh) / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.007)
        h_t, c5_t = _continuous_refine(h_pert, sub_dl, radius_init=0.04, use_lp=True, use_smooth=True)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            h_bh = h_t.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.18, scale * 1.04)

    # Phase 5: Final polish (7%)
    if time.time() < deadline - 3:
        h_f = best_h.copy()
        h_f, c5_f = _continuous_refine(h_f, deadline - 2, radius_init=0.02, use_lp=True, use_smooth=True)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # Block swap polish

            h_bs, c5_bs = _block_swap_search(best_h, deadline - 0.5, max_passes=15)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()

    best_h = _project(best_h, float(len(best_h)) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
