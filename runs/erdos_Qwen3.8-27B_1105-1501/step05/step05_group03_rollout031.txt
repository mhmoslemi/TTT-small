def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _soft_max(scores, beta):
    """Smooth approximation of max."""
    max_s = np.max(scores)
    if beta < 1e6:
        shifted = beta * (scores - max_s)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -500, 500)
        vals = np.exp(shifted)
        return max_s + np.log(np.sum(vals)) / beta
    return max_s

def _soft_max_grad(scores, beta):
    """Gradient of soft max w.r.t. each score."""
    max_s = np.max(scores)
    if beta < 1e6:
        shifted = beta * (scores - max_s)
        shifted = np.clip(shifted, -500, 500)
        vals = np.exp(shifted)
        weights = vals / np.sum(vals)
        return weights
    idx = np.argmax(scores)
    w = np.zeros(len(scores))
    w[idx] = 1.0
    return w

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target} using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _grad_matrix_topk(h, k_max):
    """Compute gradient matrix for only the top k_max constraints."""
    n = len(h)
    scores = _scores(h)
    # Get top k constraints (excluding lag 0 which is always 0 for binary)
    idx = np.argsort(scores)[::-1][:k_max]

        if d == 0:
            continue
        if d >= 0:
            m = n - d
            G[mi, :m] += (1.0 - h[d:])
            G[mi, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            G[mi, :m] += (1.0 - h[g:])
            G[mi, g:] += -h[:m]
    return G * dx, scores[idx]

def _lp_topk(h, k_active, deadline, radius):

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))

    for it in range(n_steps):

        beta = beta_start * (beta_end / beta_start) ** frac
        
        scores = _scores(h)
        weights = _soft_max_grad(scores, beta)

        grad = np.zeros(n)
        for d_idx in range(n_lags):

                m = n - g

                grad[g:] += -w * h[:m]

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        return grad
    if d > 0:
        m = n - d
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[d:d+m])
        grad[d:] += -h[:m]
    else:
        g = -d
        m = n - g
        if m <= 0:

        grad[g:] += (1.0 - h[:m])

    scores = _scores(h)
    idx = np.argsort(scores)[::-1][:k]
    G = np.zeros((k, n))
    for mi, d_idx in enumerate(idx):

        G[mi] = _grad_for_lag(h, d)
    G *= (2.0 / n)
    return G, scores[idx]

def _lp_direction(h, k_active, radius):

    nc = G.shape[0]

    A_ub = np.zeros((nc, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -active_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, target):
    alpha_max = 1e8
    for j in range(len(h)):
        if p[j] > 1e-14:
            a = (1.0 - h[j]) / p[j]
            if a < alpha_max:
                alpha_max = a
        elif p[j] < -1e-14:
            a = h[j] / (-p[j])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(alpha_max, 1e-12) * 0.95

    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(35):

        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    if f1 < f2:

                break
    return best_h, best_c5

def _smax_gradient_descent(h, deadline, beta_start=30, beta_end=800, n_steps=600, step_init=0.02, momentum=0.92):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        shifted = np.clip(beta * (scores - max_s), -400, 400)
        w = np.exp(shifted)
        sw = np.sum(w)
        if sw < 1e-300:
            idx = int(np.argmax(scores))
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        grad = np.zeros(n)
        for d_idx in range(len(scores)):
            wi = w[d_idx]
            if wi < 1e-10:
                continue
            d = int(d_idx) - (n - 1)
            if d == 0:
                continue
            grad += wi * _grad_for_lag(h, d)
        grad *= (2.0 / n)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = momentum * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.02, 0.12)
        else:
            v = 0.5 * v
            step *= 0.82
            if step < 1e-7:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def _pair_polish(h, deadline, n_passes=5):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            if h[j] < 0.05 or h[j] > 0.95:
                continue
            for k in order[:n]:
                if k <= j:
                    continue
                if h[k] < 0.05 or h[k] > 0.95:
                    continue
                amt = min(h[j], 1.0 - h[k])
                if amt < 0.05:
                    continue
                for frac in [1.0, 0.5, 0.25]:
                    a = amt * frac
                    h_try = h.copy()
                    h_try[j] -= a
                    h_try[k] += a
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _subgradient_step(h, scores, n_active, step_size):
    n = len(h)
    target = n / 2.0
    top_idx = np.argsort(scores)[::-1][:n_active]
    grad = np.zeros(n)
    for d_idx in top_idx:
        d = int(d_idx) - (n - 1)
        if d == 0:
            continue
        grad += _grad_for_lag(h, d)
    grad *= (2.0 / n) / n_active
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn > 1e-15:
        grad /= gn
    h_new = _project(h - step_size * grad, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    def remaining():
        return max(0, t0 + budget_s - time.time())

    # Phase 1: Fine LP from start (20% budget)
    dl = t0 + budget_s * 0.20
    if time.time() < dl:
        h1, c5_1 = _lp_optimize(best_h, dl, k_active=25, radius_init=0.04, max_iters=8000, radius_min=0.0008)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Soft-max gradient with high beta (10% budget)
    dl = t0 + budget_s * 0.30
    if time.time() < dl:
        h2, c5_2 = _smax_gradient_descent(best_h, dl, beta_start=100, beta_end=3000, n_steps=800, step_init=0.012, momentum=0.93)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: LP with slightly larger radius (15% budget)
    dl = t0 + budget_s * 0.45
    if time.time() < dl:
        h3, c5_3 = _lp_optimize(best_h, dl, k_active=30, radius_init=0.08, max_iters=6000, radius_min=0.001)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-resolution exploration (20% budget)
    dl = t0 + budget_s * 0.65
    if time.time() < dl:
        resolutions = [160, 128, 192, 96, 128, 160, 256, 128]
        alphas = [0.6180339887, 0.3819660113, 0.7071067812, 0.4142135624, 0.5, 0.6180339887, 0.3819660113, 0.7071067812]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl:
                break
            if best_c5 < 0.3805:
                break
            for mode in range(2):
                if time.time() > dl:
                    break
                if mode == 0 and n_try != best_n:
                    h_init = _resample(best_h, n_try)
                else:
                    alpha = alphas[ri % len(alphas)]
                    h_init = _quasi_init(n_try, alpha)
                    h_init = h_init + rng.normal(0, 0.04, n_try)
                    h_init = _project(h_init, n_try / 2.0)
                sub_dl = min(dl, time.time() + budget_s * 0.02)
                h_try, c5_try = _lp_optimize(h_init, sub_dl, k_active=25, radius_init=0.08, max_iters=2000, radius_min=0.003)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 5: Resample best to larger n and optimize (10% budget)
    dl = t0 + budget_s * 0.75
    if time.time() < dl:
        for n_big in [2 * best_n, 3 * best_n // 2]:
            if time.time() > dl:
                break
            if n_big > 384:
                continue
            h_big = _resample(best_h, n_big)
            sub_dl = min(dl, time.time() + budget_s * 0.03)
            h_bs, c5_bs = _lp_optimize(h_big, sub_dl, k_active=30, radius_init=0.05, max_iters=3000, radius_min=0.002)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()
                best_n = n_big

    # Phase 6: Perturbation restarts (12% budget)
    dl = t0 + budget_s * 0.87
    if time.time() < dl:
        scale = 0.015
        for attempt in range(200):
            if time.time() > dl:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl, time.time() + budget_s * 0.008)
            h_try, c5_try = _lp_optimize(h_pert, sub_dl, k_active=20, radius_init=0.04, max_iters=800, radius_min=0.004)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.002, scale * 0.75)
            else:
                scale = min(0.06, scale * 1.12)

    # Phase 7: Subgradient + LP cycling (8% budget)
    dl = t0 + budget_s * 0.95
    if time.time() < dl:
        for cyc in range(50):
            if time.time() > dl:
                break
            scores = _scores(best_h)
            h_sg = _subgradient_step(best_h, scores, 5, 0.01)
            c5_sg = _exact_c5(h_sg)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
            sub_dl = min(dl, time.time() + budget_s * 0.006)
            h_lp, c5_lp = _lp_optimize(best_h, sub_dl, k_active=30, radius_init=0.03, max_iters=1000, radius_min=0.002)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 8: Final fine LP (4% budget)
    dl = t0 + budget_s * 0.99
    if time.time() < dl:
        h_fin, c5_fin = _lp_optimize(best_h, dl, k_active=40, radius_init=0.015, max_iters=3000, radius_min=0.0003)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
