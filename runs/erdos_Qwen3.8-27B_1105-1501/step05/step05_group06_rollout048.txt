import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pnorm_obj_grad(h, q):
    """Compute Phi_q(h) and its gradient using log-sum-exp for stability."""
    n = len(h)
    F = _scores(h)  # shape (2n-1,)
    # Clip to avoid log(0)
    F_pos = np.maximum(F, 1e-30)
    log_F = np.log(F_pos)
    q_log_F = q * log_F
    # logsumexp
    max_q_log = np.max(q_log_F)
    log_sum = max_q_log + np.log(np.sum(np.exp(q_log_F - max_q_log)))
    log_phi = log_sum / q
    phi = np.exp(log_phi)
    # Weight: p_r = F_r^q / sum(F_r^q) = exp(q*log_F_r - log_sum)
    p = np.exp(q_log_F - log_sum)
    # Gradient: phi * G^T @ p
    G = _build_G(h)
    grad = phi * (G.T @ p)
    return phi, grad

def _slsqp_solve(h0, q, maxiter=300, ftol=1e-12, timeout=None, t0=None):
    """Solve min Phi_q(h) s.t. sum(h)=n/2, 0<=h<=1 using SLSQP."""
    n = len(h0)
    target = n / 2.0
    
    def obj_and_grad(h):
        v, g = _pnorm_obj_grad(h, q)
        return v, g
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    t_start = time.time()
    best_h = h0.copy()
    best_val = _exact_c5(h0)
    
    def callback(xk):
        nonlocal best_h, best_val
        if timeout is not None and time.time() - t_start > timeout:
            raise ValueError("timeout")
        c5 = _exact_c5(xk)
        if c5 < best_val - 1e-14:
            best_val = c5
            best_h = xk.copy()
        return None
    
    try:
        res = minimize(obj_and_grad, h0, method='SLSQP', jac=True,
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol},
                       callback=callback)
    except (ValueError, Exception):
        pass
    
    # Check final result
    if res is not None and res.x is not None:
        h_final = _project(res.x, target)
        c5_final = _exact_c5(h_final)
        if c5_final < best_val:
            best_val = c5_final
            best_h = h_final
    return best_h, best_val

                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
