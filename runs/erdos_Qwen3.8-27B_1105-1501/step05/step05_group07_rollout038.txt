def _exact_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    # Project v to [0,1]^n with sum = target
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -50.0, 50.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -50.0, 50.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G * dx

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    alphas = np.linspace(0, alpha_max, 16)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, 15):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(15, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _local_optimize_lp(h, deadline, radius_init=0.25, max_iters=5000, stall_limit=150):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):

            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, scores, G, radius)

            stall += 1
            radius *= 0.8

                break
            continue

        if new_c5 < cur_c5 - 1e-13:

            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.82
            if stall > 40 and radius < 0.03:

def _logistic_h_and_s(q, lam):
    """Compute h = sigmoid(q - lam) and s = h*(1-h)"""
    z = q - lam
    h = np.empty_like(z)
    pos = z >= 0
    h[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    h[~pos] = ez / (1.0 + ez)
    s = h * (1.0 - h)
    return h, s

def _solve_lambda(q, target):
    lo, hi = float(q.min()) - 30.0, float(q.max()) + 30.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = q - mid
        h_pos = 1.0 / (1.0 + np.exp(-np.clip(z, 0, 700)))
        ez = np.exp(np.clip(z, -700, 0))
        h_neg = ez / (1.0 + ez)
        h = np.where(z >= 0, h_pos, h_neg)
        s = float(np.sum(h))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _logistic_smooth_obj(q, target, beta):
    lam = _solve_lambda(q, target)
    h, s = _logistic_h_and_s(q, lam)
    sc = _scores(h)
    max_sc = np.max(sc)
    shifted = beta * (sc - max_sc)
    shifted = np.clip(shifted, -700, 0)
    return max_sc + np.log(np.mean(np.exp(shifted))) / beta

def _logistic_smooth_grad(q, target, beta):
    lam = _solve_lambda(q, target)
    h, s = _logistic_h_and_s(q, lam)
    sc = _scores(h)
    max_sc = np.max(sc)
    shifted = beta * (sc - max_sc)
    shifted = np.clip(shifted, -700, 0)
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros(len(sc))
        w[int(np.argmax(sc))] = 1.0
    else:
        w /= wsum
    G = _grad_matrix(h)
    g = G.T @ w  # g_i = sum_k w_k * G[k,i]
    S = float(np.sum(s))
    if S < 1e-30:
        return np.zeros(len(q))
    c = float(np.sum(s * g)) / S
    return s * (g + c)

def _logistic_optimize(h_init, deadline, beta_init=20.0, beta_final=200.0, n_phases=5):
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Initialize q from h: q_i = log(h_i/(1-h_i)) + lam
    # Choose lam = 0, so q_i = logit(h_i)
    h_clipped = np.clip(h_init, 1e-10, 1.0 - 1e-10)
    q = np.log(h_clipped / (1.0 - h_clipped))
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        
        def obj(qx):
            return _logistic_smooth_obj(qx, target, beta)
        
        def grad(qx):
            return _logistic_smooth_grad(qx, target, beta)
        
        try:
            res = minimize(obj, q, jac=grad, method='L-BFGS-B',
                          bounds=[(-25, 25)] * n,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
            if np.all(np.isfinite(res.x)):
                lam = _solve_lambda(res.x, target)
                h_new, _ = _logistic_h_and_s(res.x, lam)
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                q = res.x if np.all(np.isfinite(res.x)) else q
                # Re-derive q from new h for next phase
                h_clipped = np.clip(best_h, 1e-10, 1.0 - 1e-10)
                q = np.log(h_clipped / (1.0 - h_clipped))
        except Exception:
            pass
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -50.0, 50.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G * dx

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    alpha_max = 1.0
    for j in range(len(h)):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    alphas = np.linspace(0, alpha_max, 16)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, 15):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(15, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

_LOG_STATE = {'target': 0.0, 'beta': 1.0}

def _sigmoid(z):
    out = np.empty_like(z)
    pos = z >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    out[~pos] = ez / (1.0 + ez)
    return out

def _log_h_and_s(q, lam):
    z = q - lam
    h = _sigmoid(z)
    s = h * (1.0 - h)
    return h, s

def _solve_lambda(q, target):
    lo = float(q.min()) - 40.0
    hi = float(q.max()) + 40.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = _sigmoid(q - mid)
        s = float(np.sum(h))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _log_obj(q):
    target = _LOG_STATE['target']
    beta = _LOG_STATE['beta']
    lam = _solve_lambda(q, target)
    h, _ = _log_h_and_s(q, lam)
    sc = _scores(h)
    mx = float(np.max(sc))
    shifted = beta * (sc - mx)
    shifted = np.clip(shifted, -700, 0)
    return mx + float(np.log(np.mean(np.exp(shifted)))) / beta

def _log_grad(q):
    target = _LOG_STATE['target']
    beta = _LOG_STATE['beta']
    lam = _solve_lambda(q, target)
    h, s = _log_h_and_s(q, lam)
    sc = _scores(h)
    mx = float(np.max(sc))
    shifted = beta * (sc - mx)
    shifted = np.clip(shifted, -700, 0)
    w = np.exp(shifted)
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.zeros(len(sc))
        w[int(np.argmax(sc))] = 1.0
    else:
        w /= ws
    G = _grad_matrix(h)
    g = G.T @ w
    S = float(np.sum(s))
    if S < 1e-30:
        return np.zeros(len(q))
    c = float(np.dot(s, g)) / S
    return s * (g + c)

def _logistic_optimize(h_init, deadline, beta_init=15.0, beta_final=300.0, n_phases=5):
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    h_clipped = np.clip(h_init, 1e-8, 1.0 - 1e-8)
    q = np.log(h_clipped / (1.0 - h_clipped))
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        _LOG_STATE['target'] = target
        _LOG_STATE['beta'] = beta
        try:
            res = minimize(_log_obj, q, jac=_log_grad, method='L-BFGS-B',
                          bounds=[(-25.0, 25.0)] * n,
                          options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
            if np.all(np.isfinite(res.x)):
                lam = _solve_lambda(res.x, target)
                h_new, _ = _log_h_and_s(res.x, lam)
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    q = np.log(np.clip(h_new, 1e-8, 1.0 - 1e-8) / (1.0 - np.clip(h_new, 1e-8, 1.0 - 1e-8)))
        except Exception:
            pass
    return best_h, best_c5

def _subgrad_optimize(h, deadline, n_steps=2000, b_start=12.0, b_end=80.0):

        else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = np.where((x * alpha) % 1.0 < 0.5, 1.0, 0.0)

def _smooth_quasi_init(n, alpha, width):

def _pair_polish(h, deadline, max_passes=3):

        for n_try in [128, 96, 192]:
            if time.time() > dl4:
                break
            # Resample from current best
            h_init_r = _resample(best_h, n_try)
            sub_dl = min(dl4, time.time() + budget_s * 0.05)
            ht, ct = _local_optimize_lp(h_init_r, sub_dl, radius_init=0.25, max_iters=4000, stall_limit=150)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try
            # Quasi-periodic starts
            if time.time() < dl4:
                ai = alphas[0] + 0.1 * n_try
                h_qp = _smooth_quasi_init(n_try, ai % 1.0, 0.08)
                h_qp += rng.normal(0, 0.03, n_try)
                h_qp = _project(h_qp, n_try / 2.0)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.04)
                ht2, ct2 = _local_optimize_lp(h_qp, sub_dl2, radius_init=0.3, max_iters=3000, stall_limit=120)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = n_try
                # Logistic on quasi-periodic
                if time.time() < dl4:
                    sub_dl3 = min(dl4, time.time() + budget_s * 0.03)
                    ht3, ct3 = _logistic_optimize(ht2, sub_dl3, beta_init=15, beta_final=100, n_phases=3)
                    if ct3 < best_c5:
                        best_c5 = ct3
                        best_h = ht3.copy()
                        best_n = n_try

    # Phase 5: Perturbation restarts
    if time.time() < t0 + budget_s * 0.90:
        dl5 = t0 + budget_s * 0.90
        scale = 0.015
        for attempt in range(300):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.01)
            ht, ct = _local_optimize_lp(h_pert, sub_dl, radius_init=0.05, max_iters=600, stall_limit=60)
            if ct < best_c5 - 1e-13:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.002, scale * 0.85)
            else:
                scale = min(0.1, scale * 1.08)

    # Phase 6: Final logistic + LP polish
    if time.time() < t0 + budget_s * 0.95:
        dl6 = t0 + budget_s * 0.95
        h6, c6 = _logistic_optimize(best_h, dl6, beta_init=100.0, beta_final=500.0, n_phases=4)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()
        h6b, c6b = _local_optimize_lp(best_h, min(dl6, time.time() + budget_s * 0.03),
                                       radius_init=0.04, max_iters=3000, stall_limit=150)
        if c6b < best_c5:
            best_c5 = c6b
            best_h = h6b.copy()

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.98:
        hp, cp = _pair_polish(best_h, t0 + budget_s * 0.985, max_passes=4)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
        # Final tiny LP
        h_fin, c_fin = _local_optimize_lp(best_h, deadline, radius_init=0.02, max_iters=2000, stall_limit=80)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
