def _project(v, target):
    """Project v onto {0≤x≤1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

                c -= 1.0
            else:
                c -= (1.0 - h[i + d])
        # k=j as first index (if j+d < n)
        if j + d < n:
            if j + d == i:
                c += 1.0  # old=0, new=1*(1-0)=1
            else:
                c += (1.0 - h[j + d])
        # k=i-d as second index hit i (if i-d >= 0 and i-d+d=i < n)
        if i - d >= 0:
            if i - d == j:
                c += 1.0  # old: h[j]*(1-h[i]) = 0*0 = 0, new: 1*1 = 1
            else:
                c += h[i - d]  # old: h[i-d]*0 = 0, new: h[i-d]*1 = h[i-d]
        # k=j-d as second index hit j (if j-d >= 0)
        if j - d >= 0:
            if j - d == i:
                c -= 1.0  # old: h[i]*(1-h[j]) = 1*1 = 1, new: 0*0 = 0
            else:
                c -= h[j - d]  # old: h[j-d]*1 = h[j-d], new: h[j-d]*0 = 0
        delta[n - 1 + d] = c
    
    # c[n-1-d] for d=1..n-1: sum_{k=d}^{n-1} h[k]*(1-h[k-d])
    for d in range(1, n):
        c = 0.0
        # k=i (if i >= d)
        if i >= d:
            if i - d == j:
                c -= 1.0  # old: 1*(1-0)=1, new: 0*(1-1)=0
            else:
                c -= (1.0 - h[i - d])
        # k=j (if j >= d)
        if j >= d:
            if j - d == i:
                c += 1.0  # old: 0*(1-1)=0, new: 1*(1-0)=1
            else:
                c += (1.0 - h[j - d])
        # k=i+d (if i+d < n): second index is i
        if i + d < n:
            if i + d == j:
                c += 1.0  # old: h[j]*(1-h[i]) = 0*0 = 0, new: 1*1 = 1
            else:
                c += h[i + d]  # old: h[i+d]*0 = 0, new: h[i+d]*1
        # k=j+d (if j+d < n): second index is j
        if j + d < n:
            if j + d == i:
                c -= 1.0  # old: h[i]*(1-h[j]) = 1*1 = 1, new: 0*0 = 0
            else:
                c -= h[j + d]  # old: h[j+d]*1, new: h[j+d]*0
        delta[n - 1 - d] = c
    
    return delta

Now for the continuous refinement, let me use a simple approach:
- At each step, find the max lag
- Compute the gradient of that score with respect to h
- Take a small step in the negative gradient direction
- Project back to the constraint set

The gradient of c[n-1+d] with respect to h[k]:
∂c[n-1+d]/∂h[k] = (1-h[k+d]) [if k+d < n] - h[k-d] [if k-d ≥ 0]

And ∂c[n-1-d]/∂h[k] = (1-h[k-d]) [if k-d ≥ 0] - h[k+d] [if k+d < n]

For the max lag d*, the gradient of scores[d*] is:
If d* ≥ n-1 (right side, lag = d*-(n-1) ≥ 0):
  grad[k] = (1-h[k+lag]) [if valid] - h[k-lag] [if valid]
If d* < n-1 (left side, lag = (n-1)-d* > 0):
  grad[k] = (1-h[k-lag]) [if valid] - h[k+lag] [if valid]

Then I step in direction -grad (to decrease the max score), project, and check improvement.

But this only decreases one score. Other scores might increase. A better approach is to use a weighted combination of the top-k gradients, or to use the LP approach from the previous code.

For the LP refinement, the idea is:
- At point h, with scores s, want to find p such that s + G@p ≤ t (minimize t)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _beatty_seq(n, alpha, rng):
    """Generate a Beatty-type sequence with exactly n/2 ones."""
    x = np.arange(n)
    vals = (x * alpha) % 1.0
    # Add small noise to break ties
    vals += rng.normal(0, 1e-6, n)
    # Find threshold for exactly n/2 ones
    vals_sorted = np.sort(vals)
    threshold = vals_sorted[n // 2 - 1]
    h = (vals <= threshold).astype(float)
    return h

def _binary_sa(h_init, deadline, rng, n_iter=300000, structure_frac=0.6):

    h = (h_init >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h == 1)[0]
            rm = rng.choice(idx, n_ones - n // 2, replace=False)
            h[rm] = 0
        else:
            idx = np.where(h == 0)[0]
            add = rng.choice(idx, n // 2 - n_ones, replace=False)
            h[add] = 1
    
    cur_scores = _scores(h)
    cur_max = float(np.max(cur_scores))
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    T = 0.03
    T_min = 1e-8
    cool = (T_min / T) ** (1.0 / max(1, n_iter))

        if time.time() > deadline:
            break
        
        # Choose swap
        if rng.random() < structure_frac:
            worst = int(np.argmax(cur_scores))
            d = abs(worst - (n - 1))
            if d == 0:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
            elif worst >= n - 1:
                # Right side: c[n-1+d] = sum h[k]*(1-h[k+d])
                cands = [k for k in ones if k + d < n and h[k + d] == 0]
                if cands:

                    j = i + d
                else:

                    j = zeros[rng.integers(len(zeros))]
            else:

                    j = i - d

                    j = zeros[rng.integers(len(zeros))]
        else:

            j = zeros[rng.integers(len(zeros))]

        new_s = _scores(h)

            cur_max = new_max
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_descent(h, deadline, max_passes=500, thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                # Try moving mass from j to k or k to j
                # h[j] + a, h[k] - a (sum preserved)
                C = scores.copy()
                # Compute change in scores for h[j]+=a, h[k]-=a
                # For each lag d:
                # c[n-1+d] change: (1-h[j+d])*a - (1-h[k+d])*(-a) [if j+d<n, k+d<n]
                #   = a*(1-h[j+d]) + a*(1-h[k+d])... no this isn't right
                # Actually for bilinear: c = sum h[i]*(1-h[i+d])
                # dc/da for h[j]+=a, h[k]-=a:
                # = (1-h[j+d]) - h[j-d] + h[k-d] - (1-h[k+d]) [for right side]
                #   (with appropriate boundary checks)
                # This is getting complicated. Let me just do the direct search.
                
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                
                # Ternary search for optimal a
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    h[j] += m1
                    h[k] -= m1
                    f1 = float(np.max(_scores(h)))
                    h[j] += m2 - m1
                    h[k] -= m2 - m1
                    f2 = float(np.max(_scores(h)))
                    h[j] -= m2
                    h[k] += m2
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual = _exact_c5(h)
                    if actual < cur_c5 - thresh:
                        cur_c5 = actual
                        scores = _scores(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    
    return best_h, best_c5

def _lp_refine(h, deadline, top_k=8, max_iters=2000, radius_init=0.1):
    """LP-based refinement: minimize max scores via linearized LP."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        scores = _scores(h)
        # Get top-k scores
        n_lags = 2 * n - 1
        if top_k and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            s_k = scores[idx]
        else:
            idx = np.arange(n_lags)
            s_k = scores
        
        # Build the linearized model:
        # scores ≈ s_k + G_k @ p, where G_k is the Jacobian at h
        # For c[n-1+d]: ∂/∂h[j] = (1-h[j+d]) - h[j-d] (for valid indices)
        # For c[n-1-d]: ∂/∂h[j] = (1-h[j-d]) - h[j+d] (for valid indices)
        
        G = np.zeros((len(idx), n))
        for ii, lag_idx in enumerate(idx):
            if lag_idx >= n - 1:
                d = lag_idx - (n - 1)
                for j in range(n):
                    g = 0.0
                    if j + d < n:
                        g += (1.0 - h[j + d])
                    if j - d >= 0:
                        g -= h[j - d]
                    G[ii, j] = g
            else:
                d = (n - 1) - lag_idx
                for j in range(n):
                    g = 0.0
                    if j - d >= 0:
                        g += (1.0 - h[j - d])
                    if j + d < n:
                        g -= h[j + d]
                    G[ii, j] = g
        
        # Solve: min t s.t. s_k + G@p <= t, sum(p)=0, |p_j| <= radius, h_j + p_j in [0,1]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -s_k
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                # Line search
                if np.max(np.abs(p)) < 1e-14:
                    stall += 1
                    radius *= 0.7
                    continue
                alpha, new_c5 = _line_search_lp(h, p)
                h_new = _project(h + alpha * p, target)
                actual_c5 = _exact_c5(h_new)
                if
