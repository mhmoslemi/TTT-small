    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]

        h_new2 = _slsqp_solve(best_h, target_t2, max_constraints, max_iter, sub_dl2)
        c5_new2 = _exact_c5(h_new2)
        if c5_new2 < best_c5:
            best_h = h_new2
            best_c5 = c5_new2
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Phase 1: LP from initial (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: SLSQP epigraph (20%)
    d2 = t0 + budget_s * 0.35
    best_h, best_c5 = _slsqp_polish(best_h, d2, max_constraints=25, max_iter=100)
    
    # Phase 3: LP refinement (15%)
    d3 = t0 + budget_s * 0.50
    best_h, best_c5 = _lp_loop(best_h, d3, radius_init=0.1)
    
    # Phase 4: Multi-resolution (15%)
    d4 = t0 + budget_s * 0.65
    best_h, best_c5 = _multiresolution(best_h, d4, rng)
    
    # Phase 5: SLSQP on best multi-res (10%)
    d5 = t0 + budget_s * 0.75
    best_h, best_c5 = _slsqp_polish(best_h, d5, max_constraints=20, max_iter=80)
    
    # Phase 6: Gradient descent (10%)
    d6 = t0 + budget_s * 0.85
    best_h, best_c5 = _gradient_descent(best_h, d6, n_steps=200, lr_init=0.03)
    
    # Phase 7: Perturbation search (8%)
    d7 = t0 + budget_s * 0.93
    best_h, best_c5 = _perturb_search(best_h, d7, rng, n_trials=30)
    
    # Phase 8: Coordinate descent + final LP (7%)
    d8 = t0 + budget_s * 0.97
    best_h, best_c5 = _coord_descent(best_h, d8, n_pairs=1500)
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
