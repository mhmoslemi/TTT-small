def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    return np.correlate(h, 1.0 - h, mode='full')

    Returns shape (2n-1, n) where row j is d(output[j])/d(h).
    """
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):

def slsqp_epigraph(h0, deadline, max_iters=200):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    J = lag_jacobian(h0) * dx  # scale by dx
    scores = lag_values(h0) * dx
    t0 = float(np.max(scores)) + 1e-6
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(max(0.0, t0 - 0.1), 1.0)]
    
    best_h = h0.copy()
    best_c5 = eval_c5(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Constraints: f_j(h) - t <= 0 for each j
    # We'll use a single constraint function that returns all values
    # and a Jacobian function
    
    # SLSQP expects constraints as list of dicts or a single dict with 'type': 'ineq'
    # For vector inequality: fun returns array, jac returns 2D array
    
    n_lags = 2 * n - 1
    
    def con_val(x):
        h = x[:n]
        t = x[-1]
        vals = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - vals  # we need t - F_j >= 0, i.e., F_j - t <= 0
        # SLSQP ineq constraint: fun(x) >= 0
        # So we want F_j(h) - t <= 0, equivalently t - F_j(h) >= 0
    
    def con_jac(x):
        h = x[:n]
        Jc = lag_jacobian(h) * dx
        # For constraint t - F_j >= 0:
        # d/dh[m] = -Jc[j, m]
        # d/dt = 1
        jac = np.zeros((n_lags, n + 1))
        jac[:, :n] = -Jc
        jac[:, n] = 1.0
        return jac
    
    cons = [{'type': 'ineq', 'fun': con_val, 'jac': con_jac}]
    
    x0 = np.concatenate([h0, [t0]])
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iters, 'ftol': 1e-8})
        # Evaluate the result

            c5_new = eval_c5(h_new)
            if c5_new < best_c5:

                best_c5 = c5_new
    except Exception:
        pass

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_jacobian(h):
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        k = j - (n - 1)
        if k >= 0:
            m1 = np.arange(k, n)
            J[j, m1] += (1.0 - h[m1 - k])
            m2 = np.arange(0, n - k)
            J[j, m2] += -h[m2 + k]

            J[j, m1] += (1.0 - h[m1 + a])
            m2 = np.arange(a, n)
            J[j, m2] += -h[m2 - a]
