        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _coord_descent(h, deadline, max_sweeps=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_r = np.arange(2 * n - 1) - (n - 1)
    n_grid = 250
    t_grid = np.linspace(0.0, 1.0, n_grid)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    s
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % 50 == 0 and sweep > 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
        max_change = 0.0
        for i in range(n):
            h_i = h[i]
            ik = i + k_r
            ik_neg = i - k_r
            h_ik = np.where((ik >= 0) & (ik < n), h[ik], 0.0)
            h_ikneg = np.where((ik_neg >= 0) & (ik_neg < n), h[ik_neg], 0.0)
            m = (1.0 - h_ik) - h_ikneg
            A = corr - h_i * m
            q_const = corr[n - 1] - h_i * (1.0 - h_i)
            aff = t_grid[:, None] * m[None, :] + A[None, :]
            aff[:, n - 1] = -1e30
            aff_max = aff.max(axis=1)
            quad = q_const + t_grid - t_grid * t_grid
            f_vals = np.maximum(aff_max, quad)
            idx = int(np.argmin(f_vals))
            lo_i = max(0, idx - 4)
            hi_i = min(n_grid - 1, idx + 4)
            t_lo = t_grid[lo_i]
            t_hi = t_grid[hi_i]
            c1 = t_hi - gr * (t_hi - t_lo)
            c2 = t_lo + gr * (t_hi - t_lo)
            v1 = A + c1 * m
            v1_n1 = v1[n - 1]
            f_c1 = max(float(v1.max()), float(q_const + c1 - c1 * c1))
            v2 = A + c2 * m
            f_c2 = max(float(v2.max()), float(q_const + c2 - c2 * c2))
            for _ in range(20):
                if t_hi - t_lo < 1e-12:
                    break
                if f_c1 < f_c2:
                    t_hi = c2
                    c2 = c1
                    f_c2 = f_c1
                    c1 = t_hi - gr * (t_hi - t_lo)
                    vv = A + c1 * m
                    f_c1 = max(float(vv.max()), float(q_const + c1 - c1 * c1))
                else:
                    t_lo = c1
                    c1 = c2
                    f_c1 = f_c2
                    c2 = t_lo + gr * (t_hi - t_lo)
                    vv = A + c2 * m
                    f_c2 = max(float(vv.max()), float(q_const + c2 - c2 * c2))
            t_best = (t_lo + t_hi) / 2.0
            if abs(t_best - h_i) > 1e-11:
                delta = t_best - h_i
                h[i] = t_best
                corr[:n - 1] += delta * m[:n - 1]
                corr[n:] += delta * m[n:]
                corr[n - 1] += t_best * (1.0 - t_best) - h_i * (1.0 - h_i)
                max_change = max(max_change, abs(delta))
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
        if max_change < 1e-10:
            break
    return best_h, best_c5

def _pair_optimize(h, deadline, n_pairs_per_pass=400, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_r = np.arange(2 * n - 1) - (n - 1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if rng is None:
        rng = np.random.default_rng(777)
    n_grid = 150
    stall = 0
    while time.time() < deadline:
        if stall > 500:
            break
        pass_improved = False
        for _ in range(n_pairs_per_pass):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            h_i = h[i]
            h_j = h[j]
            if abs(h_i - h_j) < 1e-6:
                continue
            s_lo = -min(h_i, 1.0 - h_j)
            s_hi = min(1.0 - h_i, h_j)
            if s_hi - s_lo < 1e-8:
                continue
            ik = i + k_r
            ik_neg = i - k_r
            h_ik = np.where((ik >= 0) & (ik < n), h[ik], 0.0)
            h_ikneg = np.where((ik_neg >= 0) & (ik_neg < n), h[ik_neg], 0.0)
            m_i = (1.0 - h_ik) - h_ikneg
            jk = j + k_r
            jk_neg = j - k_r
            h_jk = np.where((jk >= 0) & (jk < n), h[jk], 0.0)
            h_jkneg = np.where((jk_neg >= 0) & (jk_neg < n), h[jk_neg], 0.0)
            m_j = (1.0 - h_jk) - h_jkneg
            m_diff = m_i - m_j
            A = corr.copy()
            A[n - 1] = 0.0
            q_coeff = 2.0 * (h_j - h_i)
            q_quad = -2.0
            A[n - 1] = -1e30
            s_grid = np.linspace(s_lo, s_hi, n_grid)
            aff = s_grid[:, None] * m_diff[None, :] + A[None, :]
            aff[:, n - 1] = -1e30
            aff_max = aff.max(axis=1)
            quad_part = corr[n - 1] + s_grid * q_coeff + s_grid * s_grid * q_quad
            f_vals = np.maximum(aff_max, quad_part)
            idx = int(np.argmin(f_vals))
            lo_i = max(0, idx - 3)
            hi_i = min(n_grid - 1, idx + 3)
            t_lo = s_grid[lo_i]
            t_hi = s_grid[hi_i]
            c1 = t_hi - gr * (t_hi - t_lo)
            c2 = t_lo + gr * (t_hi - t_lo)
            f_c1 = max(float((A + c1 * m_diff).max()), float(corr[n-1] + c1*q_coeff + c1*c1*q_quad))
            f_c2 = max(float((A + c2 * m_diff).max()), float(corr[n-1] + c2*q_coeff + c2*c2*q_quad))
            for _ in range(18):
                if t_hi - t_lo < 1e-12:
                    break
                if f_c1 < f_c2:
                    t_hi = c2
                    c2 = c1
                    f_c2 = f_c1
                    c1 = t_hi - gr * (t_hi - t_lo)
                    f_c1 = max(float((A + c1 * m_diff).max()), float(corr[n-1] + c1*q_coeff + c1*c1*q_quad))
                else:
                    t_lo = c1
                    c1 = c2
                    f_c1 = f_c2
                    c2 = t_lo + gr * (t_hi - t_lo)
                    f_c2 = max(float((A + c2 * m_diff).max()), float(corr[n-1] + c2*q_coeff + c2*c2*q_quad))
            s_best = (t_lo + t_hi) / 2.0
            if abs(s_best) > 1e-10:
                new_h_i = h_i + s_best
                new_h_j = h_j - s_best
                if 0.0 <= new_h_i <= 1.0 and 0.0 <= new_h_j <= 1.0:
                    h[i] = new_h_i
                    h[j] = new_h_j
                    corr[:n-1] += s_best * m_diff[:n-1]
                    corr[n+1:] += s_best * m_diff[n+1:]
                    corr[n-1] += new_h_i*(1-new_h_i) - h_i*(1-h_i) + new_h_j*(1-new_h_j) - h_j*(1-h_j)
                    pass_improved = True
        if pass_improved:
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-13:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                n_pairs_per_pass = max(100, n_pairs_per_pass // 2)
    return best_h, best_c5

def _lp_direction(h, corrections, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = G.copy()
    A_ub = np.hstack([A_ub, -np.ones((2 * n - 1, 1))])
    b_ub = -corrections
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_refine(h, deadline, radius_init=0.1):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 40:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, corr, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        S0 = corr.copy()
        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')
        t_lo = 0.0
        t_hi = 0.0
        pos = p > 1e-14
        neg = p < -1e-14
        if np.any(pos):
            t_hi = float(np.min((1.0 - h[pos]) / p[pos]))
        if np.any(neg):
            t_hi = min(t_hi, float(np.max(h[neg] / (-p[neg]))))
        t_hi = min(t_hi, radius)
        if t_hi < 1e-12:
            stall += 1
            radius *= 0.7
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = t_hi - gr * t_hi
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        c2 = gr * t_hi
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if t_hi - t_lo < 1e-12:
                break
            if f1 < f2:
                t_hi = c2
                c2 = c1
                f2 = f1
                c1 = t_hi - gr * (t_hi - t_lo)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                t_lo = c1
                c1 = c2
                f1 = f2
                c2 = t_lo + gr * (t_hi - t_lo)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        alpha = (t_lo + t_hi) / 2.0
        h_new = h + alpha * p
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < best_c5 - 1e-13:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_from_continuous(h):
    n = len(h)
    half = n // 2
    idx = np.argsort(h)[-half:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Brief LP warmup (5%)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _lp_refine(h_init, d1, radius_init=0.08)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent (35%)
    d2 = t0 + budget_s * 0.40
    h2 = best_h.copy()
    h2, c5_2 = _coord_descent(h2, d2, max_sweeps=5000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair optimization (20%)
    d3 = t0 + budget_s * 0.60
    h3 = best_h.copy()
    h3, c5_3 = _pair_optimize(h3, d3, n_pairs_per_pass=600, rng=rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Interleave coord descent + pair (15%)
    d4 = t0 + budget_s * 0.75
    h4 = best_h.copy()
    while time.time() < d4:
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _coord_descent(h4, sub_dl, max_sweeps=500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h4, c5_4 = _pair_optimize(h4, sub_dl2, n_pairs_per_pass=300, rng=rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Multi-resolution exploration (12%)
    d5 = t0 + budget_s * 0.87
    n_curr = len(best_h)
    offsets = [4, -4, 8, -8, 12, -12, 16, -16, 24, -24, 32, -32, 40, -40, 48, -48]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 300 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = _coord_descent(h_r, sub_dl, max_sweeps=300)
        rem = d5 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.002:
            sub_dl2 = min(d5, time.time() + budget_s * 0.008)
            h_r, c5_r = _pair_optimize(h_r, sub_dl2, n_pairs_per_pass=200, rng=rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 6: Perturbation + re-optimize (6%)
    d6 = t0 + budget_s * 0.93
    scale = 0.06
    for attempt in range(100):
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.005)
        h_t, c5_t = _coord_descent(h_pert, sub_dl, max_sweeps=200)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.1)

    # Phase 7: Final polish (4%)
    d7 = deadline
    h7 = best_h.copy()
    sub_dl = min(d7, time.time() + budget_s * 0.02)
    h7, c5_7 = _coord_descent(h7, sub_dl, max_sweeps=500)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    sub_dl2 = min(d7, time.time() + budget_s * 0.015)
    h7, c5_7 = _pair_optimize(h7, sub_dl2, n_pairs_per_pass=300, rng=rng)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
