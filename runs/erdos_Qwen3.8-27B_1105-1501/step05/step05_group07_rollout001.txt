def _smooth_obj_grad(q, beta, target):
    # Compute h from q
    h, s, mu = _q_to_h(q, target)
    # Compute scores
    dx = 2.0 / len(h)
    scores = _scores(h)
    # Smooth objective
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -700, 0)
    log_sum = np.log(np.sum(np.exp(shifted))) - np.log(len(scores))
    obj = max_s + log_sum / beta
    # Weights
    w = np.exp(shifted)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    # Gradient in h-space
    G = _grad_matrix(h)
    grad_h = G.T @ w
    # Transform to q-space
    # dF/dq[j] = s[j] * (grad_h[j] - rho)
    # where rho = sum(s * grad_h) / sum(s)
    rho = np.sum(s * grad_h) / np.sum(s) if np.sum(s) > 1e-15 else 0.0
    grad_q = s * (grad_h - rho)
    return obj, grad_q

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _balance(x, target):
    # Project x to [0,1] with sum = target
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    # Bisection on threshold
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(x - mu, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += residual / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def _safe_sigmoid(t):
    # Stable sigmoid
    result = np.zeros_like(t)
    pos = t >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    result[~pos] = exp_t / (1.0 + exp_t)
    return result

def _q_to_h(q, target):
    # Find mu such that sum(sigmoid(q - mu)) = target
    # Use bisection
    lo = np.min(q) - 30.0
    hi = np.max(q) + 30.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = _safe_sigmoid(q - mid)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = _safe_sigmoid(q - mu)
    s = h * (1.0 - h)  # sigmoid derivative
    return h, s, mu

S_d^cross = Σ_i h[i] * (1-y[i+d]) (for valid i, same convention as before)

I want: min_y max_d S_d^cross

This is equivalent to:
min τ
s.t. Σ_i h[i] * (1-y[i+d]) ≤ τ for all d
     Σ y[i] = T
     0 ≤ y[i] ≤ 1

Expanding: Σ_i h[i] - Σ_i h[i]*y[i+d] ≤ τ
Let C_d = Σ_i h[i] (over valid range for shift d)
Then: C_d - Σ_i h[i]*y[i+d] ≤ τ
i.e., Σ_i h[i]*y[i+d] ≥ C_d - τ

So the LP is:
min τ
s.t. -Σ_i h[i]*y[i+d] ≤ -(C_d - τ) for all d
     i.e., Σ_i h[i]*y[i+d] + τ ≥ C_d for all d
     i.e., -Σ_i h[i]*y[i+d] - τ ≤ -C_d
     Σ y[i] = T
     0 ≤ y[i] ≤ 1
     τ ≥ 0 (or τ ≥ -inf, but τ should be ≥ 0 since S_d^cross ≥ 0)

In standard linprog form (minimize c^T x s.t. A_ub x ≤ b_ub, A_eq x = b_eq, bounds):
x = [y_0, ..., y_{n-1}, τ]
c = [0, ..., 0, 1]
A_ub: for each d, row is [-h[i+d] terms, -1] ≤ -C_d
A_eq: [1, 1, ..., 1, 0] = T
bounds: y in [0,1], τ in [0, ∞)

```python
def _complement_lp(h, target):
    n = len(h)
    dx = 2.0 / n
    # For each shift d, compute C_d and the coefficient vector
    n_vars = n + 1  # y_0...y_{n-1}, tau
    c = np.zeros(n_vars)
    c[-1] = 1.0  # minimize tau
    
    A_rows = []
    b_vals = []
    
    for d in range(-(n-1), n):
        # C_d = sum of h[i] over valid range * dx (but we can drop dx since it's just scaling)
        # Actually let's keep it consistent
        if d >= 0:
            m = n - d
            if m <= 0: continue
            # S_d = sum_{i=0}^{m-1} h[i] * (1-y[i+d])
            # C_d = sum_{i=0}^{m-1} h[i]
            C_d = np.sum(h[:m])
            # Constraint: C_d - sum_{i=0}^{m-1} h[i]*y[i+d] <= tau
            # => -sum_{i=0}^{m-1} h[i]*y[i+d] - tau <= -C_d
            # In terms of x = [y_0,...,y_{n-1}, tau]:
            # For i=0..m-1: coefficient of y[i+d] is -h[i]
            row = np.zeros(n_vars)
            for i in range(m):
                row[i+d] += -h[i]
            row[-1] = -1.0  # -tau
            A_rows.append(row)
            b_vals.append(-C_d)
        else:
            g = -d
            m = n - g
            if m <= 0: continue
            # S_d = sum_{i=0}^{m-1} h[i+g] * (1-y[i])
            # C_d = sum_{i=0}^{m-1} h[i+g]
            C_d = np.sum(h[g:g+m])
            # Constraint: C_d - sum_{i=0}^{m-1} h[i+g]*y[i] <= tau
            row = np.zeros(n_vars)
            for i in range(m):
                row[i] += -h[i+g]
            row[-1] = -1.0
            A_rows.append(row)
            b_vals.append(-C_d)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            y = res.x[:n]
            y = _balance(y, target)
            return y
    except Exception:
        pass
    return None
```
