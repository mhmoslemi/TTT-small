def _exact_c5(h):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):

    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')

    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15

        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 30
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(25):

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.05):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lbfgs_smooth(h, deadline, beta_list=(30, 60, 150, 400), max_iter_total=200):
    """Use L-BFGS-B to minimize the log-sum-exp objective."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    used_iter = 0
    
    for beta in beta_list:
        if time.time() > deadline or used_iter >= max_iter_total:
            break
        
        def obj_bound(x):
            # Project to feasible set
            hx = _project(x, target)
            s = _scores(hx) * dx
            m = float(np.max(s))
            v = beta * (s - m)
            return m + float(np.log(np.sum(np.exp(v))) / beta)
        
        def grad_bound(x):
            hx = _project(x, target)
            return _lsumexp_grad(hx, beta)
        
        x0 = best_h.copy()
        res = minimize(obj_bound, x0, jac=grad_bound, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': max_iter_total - used_iter, 'gtol': 1e-8})
        used_iter += res.nit
        
        h_new = _project(res.x, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.20, use_penalty=True):

    target = n / 2.0

def _pair_refine(h, deadline, max_passes=50, rng=None):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kick_and_reopt(h, deadline, rng, n_kicks=25):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.015, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 2.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.10, use_penalty=False)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _targeted_lag_reduction(h, deadline, n_active=5):
    """Specifically target the top active lags for reduction."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    
    while time.time() < deadline:
        scores = _scores(h)
        s_max = float(np.max(scores))
        # Get the top active lags
        active_idx = np.argsort(scores)[-n_active:]
        # Build a combined gradient that specifically reduces these lags
        G = _grad_matrix(h)
        # Weighted sum of gradients for active lags
        combined = np.zeros(n)
        for r in active_idx:
            combined += G[r, :]
        combined -= np.mean(combined)
        cn = np.linalg.norm(combined)
        if cn < 1e-14:
            break
        combined /= cn
        
        # Try both directions
        for sign in [1.0, -1.0]:
            alpha, new_c5 = _line_search(h, sign * combined, dx, 0.3)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * sign * combined
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        else:
            # Try LP with heavy penalty on active lags
            G2 = _grad_matrix(h)
            p = _lp_solve(h, scores, G2, 0.15, 
                        penalty_lags=list(active_idx), penalty_w=0.1)
            if p is not None and np.max(np.abs(p)) > 1e-13:

                    h = h + alpha * p
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:

                        best_h = h.copy()
                    continue

def _grad_matrix(h):

    n = len(h)

    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]

            g = -d
            m = n - g
            if m > 0:

    c[-1] = 1.0

def _smooth_lse(h, deadline, betas=(20, 50, 120, 350), max_iter=120):
    """L-BFGS-B on the log-sum-exp surrogate."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    total_it = 0

    for beta in betas:
        if time.time() > deadline or total_it >= max_iter:
            break

        def _obj(x, _n=n, _t=target, _dx=dx, _b=beta):
            hx = _project(x, _t)
            s = _scores(hx) * _dx
            m = float(np.max(s))
            v = _b * (s - m)
            return m + float(np.log(np.sum(np.exp(v))) / _b)

        def _grad(x, _n=n, _t=target, _dx=dx, _b=beta):
            hx = _project(x, _t)
            s = _scores(hx) * _dx
            m = float(np.max(s))
            v = _b * (s - m)
            exp_v = np.exp(v)
            w = exp_v / float(np.sum(exp_v))
            G = _grad_matrix(hx)
            g = (w * _dx) @ G
            g -= np.mean(g)
            return g

        x0 = best_h.copy()
        try:
            res = minimize(_obj, x0, jac=_grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter - total_it, 'gtol': 1e-7})
            total_it += res.nit
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c5

    best_h = h.copy()

    stall = 0
    bi = 0

        if stall > 50:
            bi += 1
            if bi >= len(beta_list) * 4:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = beta_list[bi % len(beta_list)]
        s = _scores(h) * dx
        m = float(np.max(s))
        v = beta * (s - m)
        exp_v = np.exp(v)
        w = exp_v / float(np.sum(exp_v))

        d = (w * dx) @ G
        d -= np.mean(d)

        if dn < 1e-14:
            bi += 1
            continue

        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.78
    return best_h, best_c5

def _targeted_lag(h, deadline, n_active=4):
    """Focus on reducing the top active lags specifically."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    while time.time() < deadline:
        scores = _scores(h)
        s_max = float(np.max(scores))
        active_idx = np.argsort(scores)[-n_active:]
        G = _grad_matrix(h)
        # Combined descent direction for active lags
        combined = np.zeros(n)
        for r in active_idx:
            combined += G[r, :]
        combined -= np.mean(combined)
        cn = np.linalg.norm(combined)
        if cn < 1e-14:
            break
        combined /= cn

        improved = False
        for sign in [1.0, -1.0]:
            alpha, new_c5 = _line_search(h, sign * combined, dx, 0.25)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * sign * combined
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break

        if not improved:
            # Try LP with heavy penalty
            p = _lp_solve(h, scores, G, 0.12,
                          penalty_lags=list(active_idx), penalty_w=0.08)
            if p is not None and np.max(np.abs(p)) > 1e-13:
                alpha, new_c5 = _line_search(h, p, dx, 0.12)
                if new_c5 < cur_c5 - 1e-13:
                    h = h + alpha * p
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    continue
            break
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=50, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = [i for i in range(n) if 1e-8 < h[i] < 1.0 - 1e-8]
        if len(interior) < 2:
            break
        if len(interior) > 35 and rng is not None:
            pairs = set()
            for _ in range(250):
                i = interior[int(rng.integers(0, len(interior)))]
                j = interior[int(rng.integers(0, len(interior)))]
                if i != j:
                    pairs.add((min(i, j), max(i, j)))
        else:
            pairs = set()
            for idx_i, i in enumerate(interior):
                for j in interior[idx_i + 1:]:
                    pairs.add((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kick_and_reopt(h, deadline, rng, n_kicks=20):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.01, 0.05)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 2.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.10, use_penalty=False)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _alternating(h, deadline, rng):
    """Alternate between LP, smooth, and targeted methods."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mode = 0
    stall = 0
    radius = 0.15
    lr = 0.015
    mom = np.zeros(n)

    while time.time() < deadline:
        if stall > 30:
            mode = (mode + 1) % 3
            stall = 0
            radius = 0.15
            lr = 0.015
            mom = np.zeros(n)

        if mode == 0:
            scores = _scores(h)
            G = _grad_matrix(h)
            s_max = float(np.max(scores))
            pl = None
            if stall > 5:
                pl = list(np.where(scores >= s_max * 0.997)[0])
            p = _lp_solve(h, scores, G, radius, pl)
            if p is not None and np.max(np.abs(p)) > 1e-13:
                alpha, new_c5 = _line_search(h, p, dx, radius)
                if new_c5 < cur_c5 - 1e-12:
                    h = h + alpha * p
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.07, 0.45)
                    continue
            stall += 1
            radius *= 0.78

        elif mode == 1:
            for beta in [50, 150]:
                s = _scores(h) * dx
                m = float(np.max(s))
                v = beta * (s - m)
                ev = np.exp(v)
                w = ev / float(np.sum(ev))
                G = _grad_matrix(h)
                d = (w * dx) @ G
                d -= np.mean(d)
                dn = np.linalg.norm(d)
                if dn < 1e-14:
                    continue
                d /= dn
                mom = 0.82 * mom + 0.18 * d
                mn = np.linalg.norm(mom)
                if mn < 1e-14:
                    continue
                mom /= mn
                alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
                if new_c5 < cur_c5 - 1e-12:
                    h = h + alpha * lr * mom
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    lr = min(lr * 1.05, 0.30)
                    break
            else:
                stall += 1
                lr *= 0.80

        else:
            scores = _scores(h)
            active = np.argsort(scores)[-3:]
            G = _grad_matrix(h)
            comb = np.zeros(n)
            for r in active:
                comb += G[r, :]
            comb -= np.mean(comb)
            cn = np.linalg.norm(comb)
            if cn > 1e-14:
                comb /= cn
                for sign in [1.0, -1.0]:
                    alpha, new_c5 = _line_search(h, sign * comb, dx, 0.20)
                    if new_c5 < cur_c5 - 1e-12:
                        h = h + alpha * sign * comb
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        break
            if stall > 0:
                stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: L-BFGS smooth (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _smooth_lse(h, d1, betas=(15, 40, 100, 300), max_iter=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Alternating optimization (20%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _alternating(best_h, d2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP loop (15%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.18, use_penalty=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Targeted lag reduction (10%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _targeted_lag(best_h, d4, n_active=4)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (15%)
    d5 = t0 + budget_s * 0.80
    best_n = len(best_h)
    offsets = []
    for dn in range(-16, 17):
        if dn != 0:
            nt = best_n + dn
            if 72 <= nt <= 200:
                offsets.append(nt)
    offsets = sorted(set(offsets), key=lambda x: abs(x - best_n))
    tried = set()
    for nt in offsets:
        if time.time() > d5:
            break
        if nt in tried:
            continue
        tried.add(nt)
        h_r = _resample(best_h, nt)
        sub_dl = min(d5, time.time() + budget_s * 0.010)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, use_penalty=True)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt

    # Phase 6: Kick (8%)
    d6 = t0 + budget_s * 0.88
    h6, c5_6 = _kick_and_reopt(best_h, d6, rng, n_kicks=15)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final polish (7%)
    d7 = deadline
    h7, c5_7 = _momentum_smooth(best_h, d7, beta_list=(80, 200), lr_init=0.015)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final LP micro-polish
    if time.time() < deadline - 1.0:
        h8, c5_8 = _lp_loop(best_h, deadline, radius_init=0.04, use_penalty=True)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
