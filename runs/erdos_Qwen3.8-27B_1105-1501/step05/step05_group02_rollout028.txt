def _greedy_local(x, deadline, rng, max_passes=50):

    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        # Transfer mass from i to j: h[i] -= t, h[j] += t
        t_max = min(h[i], 1.0 - h[j])
        if t_max < 1e-6:
            continue
        # Try several t values
        best_t = 0.0
        best_c5_trial = cur_c5
        for frac in [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9]:
            t = t_max * frac
            h[i] -= t
            h[j] += t
            c5_tr = _exact_c5(h)
            if c5_tr < best_c5_trial:
                best_c5_trial = c5_tr
                best_t = t
            # restore
            h[i] += t
            h[j] -= t
        if best_t > 0 and best_c5_trial < cur_c5 - 1e-14:
            h[i] -= best_t
            h[j] += best_t
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _refine_binary_sa(x, deadline, rng, max_iters=200000):
    """Second pass SA with lower temperature."""
    n = len(x)
    x = x.astype(int).copy()
    A, B = _counts_init(x, n)
    cur_max = max(max(A[1:]), max(B[1:]))
    cur_c5 = cur_max * (2.0 / n)
    best_x = x.copy()
    best_c5 = cur_c5
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    T = 1.0
    T_min = 0.02
    stall = 0
    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        new_max = _do_swap(x, A, B, i, j, n)
        new_c5 = new_max * (2.0 / n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            cur_max = new_max
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_x = x.copy()
                stall = 0
                T = max(T_min, T * 0.995)
            else:
                stall += 1
                T = max(T_min, T * 0.998)
        else:
            _undo_swap(x, A, B, i, j, n)
            stall += 1
            T = max(T_min, T * 0.998)
        it += 1
        if stall > 5000:
            break
    return best_x.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
        except NameError:
            h0 = None
    
    n0 = 128
    if h0 is not None and len(h0) > 0:
        n0 = len(h0)
        h0 = _project(h0, n0 / 2.0)
        best_h = h0.copy()
        best_c5 = _exact_c5(h0)
    else:
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
        best_h = h0.copy()
        best_c5 = _exact_c5(h0)
    
    best_n = n0
    
    def _record(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        elif c5 < best_c5 + 1e-14 and n < best_n:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    # Phase 1: Binary SA from binarized parent (30% budget)
    dl1 = t0 + budget_s * 0.25
    x_bin = _binarize(h0, n0)
    x_opt, c5_opt = _binary_sa(x_bin, min(dl1, t0 + budget_s * 0.08), rng, max_iters=200000)
    _record(x_opt, c5_opt, n0)
    
    # Greedy polish
    if time.time() < dl1:
        x_g, c5_g = _greedy_local(x_opt, min(dl1, time.time() + budget_s * 0.03), rng)
        _record(x_g, c5_g, n0)
    
    # Phase 2: More binary SA passes with different seeds (20% budget)
    dl2 = t0 + budget_s * 0.45
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
    for a in alphas:
        if time.time() > dl2:
            break
        x_qp = _init_quasi(n0, a)
        dl_sub = min(dl2, time.time() + budget_s * 0.04)
        x_q, c5_q = _binary_sa(x_qp, dl_sub, rng, max_iters=100000)
        if c5_q < best_c5:
            _record(x_q, c5_q, n0)
            # Refine
            if time.time() < dl2:
                dl_ref = min(dl2, time.time() + budget_s * 0.02)
                x_r, c5_r = _refine_binary_sa(x_q, dl_ref, rng, max_iters=100000)
                _record(x_r, c5_r, n0)
    
    # Random seed
    if time.time() < dl2:
        x_rnd = np.zeros(n0)
        x_rnd[rng.permutation(n0)[:n0 // 2]] = 1.0
        dl_sub = min(dl2, time.time() + budget_s * 0.03)
        x_rs, c5_rs = _binary_sa(x_rnd, dl_sub, rng, max_iters=80000)
        _record(x_rs, c5_rs, n0)
    
    # Perturb best and re-optimize
    if time.time() < dl2:
        for _ in range(5):
            if time.time() > dl2:
                break
            x_p = best_h.copy()
            n_p = len(x_p)
            x_p_int = (x_p > 0.5).astype(int)
            # Random perturbation
            for _ in range(3):
                ones_p = np.where(x_p_int == 1)[0]
                zeros_p = np.where(x_p_int == 0)[0]
                if len(ones_p) > 0 and len(zeros_p) > 0:
                    a = ones_p[rng.integers(len(ones_p))]
                    b = zeros_p[rng.integers(len(zeros_p))]
                    x_p_int[a] = 0
                    x_p_int[b] = 1
            dl_sub = min(dl2, time.time() + budget_s * 0.02)
            x_po, c5_po = _binary_sa(x_p_int, dl_sub, rng, max_iters=50000)
            _record(x_po.astype(float), c5_po, n_p)
    
    # Phase 3: Resolution sweep (25% budget)
    dl3 = t0 + budget_s * 0.70
    resolutions = [96, 112, 144, 160, 192, 224]
    for n_try in resolutions:
        if time.time() > dl3:
            break
        if n_try == best_n:
            continue
        h_res = _resample(best_h, n_try)
        x_bin_r = _binarize(h_res, n_try)
        dl_sub = min(dl3, time.time() + budget_s * 0.035)
        x_ro, c5_ro = _binary_sa(x_bin_r, dl_sub, rng, max_iters=80000)
        if c5_ro < best_c5 - 1e-14:
            _record(x_ro, c5_ro, n_try)
            # Continue optimizing at this resolution
            if time.time() < dl3:
                dl_ref = min(dl3, time.time() + budget_s * 0.02)
                x_r2, c5_r2 = _refine_binary_sa(x_ro, dl_ref, rng, max_iters=60000)
                _record(x_r2, c5_r2, n_try)
    
    # Phase 4: Continuous pair polish (10% budget)
    dl4 = t0 + budget_s * 0.80
    if time.time() < dl4:
        h_pol = _pair_polish(best_h, min(dl4, time.time() + budget_s * 0.04), rng, n_trials=300)
        _record(h_pol, _exact_c5(h_pol), len(h_pol))
    
    # Phase 5: More binary SA at best resolution with different params (10% budget)
    dl5 = t0 + budget_s * 0.90
    n_best = best_n
    x_best_bin = (best_h > 0.5).astype(int)
    for trial in range(4):
        if time.time() > dl5:
            break
        # Slightly perturb
        x_trial = x_best_bin.copy()
        pert = rng.integers(1, 5)
        for _ in range(pert):
            ones_t = np.where(x_trial == 1)[0]
            zeros_t = np.where(x_trial == 0)[0]
            if len(ones_t) > 0 and len(zeros_t) > 0:
                a = ones_t[rng.integers(len(ones_t))]
                b = zeros_t[rng.integers(len(zeros_t))]
                x_trial[a] = 0
                x_trial[b] = 1
        dl_sub = min(dl5, time.time() + budget_s * 0.02)
        x_to, c5_to = _binary_sa(x_trial, dl_sub, rng, max_iters=60000)
        _record(x_to.astype(float), c5_to, n_best)
    
    # Phase 6: Final greedy + pair polish (5% budget)
    dl6 = t0 + budget_s * 0.96
    if time.time() < dl6:
        x_fin = (best_h > 0.5).astype(int)
        dl_sub = min(dl6, time.time() + budget_s * 0.02)
        x_fg, c5_fg = _greedy_local(x_fin, dl_sub, rng, max_passes=100)
        if c5_fg < best_c5 - 1e-14:
            _record(x_fg, c5_fg, len(x_fg))
        h_pp = _pair_polish(best_h, min(dl6, time.time() + budget_s * 0.02), rng, n_trials=100)
        c5_pp = _exact_c5(h_pp)
        _record(h_pp, c5_pp, len(h_pp))
    
    # Final: ensure validity
    final_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(final_h)
    
    return final_h, final_c5, best_n
