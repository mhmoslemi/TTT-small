When h[b] goes from 0 to 1:
- As first arg: had h[b]*(1-h[b-k]) = 0. Now (1-h[b-k]) if 0≤b-k<n, else 0. Change: +(1-h[b-k]) if 0≤b-k<n, else 0.
- As second arg: had h[b+k]*(1-h[b]) = h[b+k] if 0≤b+k<n, else 0. Now 0. Change: -h[b+k] if 0≤b+k<n, else 0.

So: delta_corr[k] = -(1-h[a-k])*I(0≤a-k<n) + h[a+k]*I(0≤a+k<n) + (1-h[b-k])*I(0≤b-k<n) - h[b+k]*I(0≤b+k<n)

For a binary h, h[x] is either 0 or 1. So:
- -(1-h[a-k]) is either 0 (if h[a-k]=1) or -1 (if h[a-k]=0)
- h[a+k] is either 0 or 1
- (1-h[b-k]) is either 0 or 1
- -h[b+k] is either 0 or -1

So delta_corr[k] ∈ {-2, -1, 0, 1, 2}

The new C5 = max over k of (corr[k] + delta_corr[k]) * dx

To find the best swap, I need to minimize max_k (corr[k] + delta_corr[k]).

This is a minimax problem over the swaps. For efficiency, I can:
1. Precompute the current corr and identify which lags are "active" (near the max)
2. For each candidate swap, compute delta_corr only for the active lags
3. Estimate the new max as the max of (corr[k] + delta_corr[k]) for active k, and the second-largest inactive corr

But this is getting complicated. Let me just do the full evaluation with np.correlate. For n=128, it's fast enough.

Actually, let me benchmark: np.correlate on two length-128 arrays takes about 3-5 microseconds in C. The Python loop overhead per iteration (two assignments + function call + comparison) is about 2-3 microseconds. Total per swap: ~7-8 microseconds. For 4096 swaps: ~30ms per pass. In 60 seconds, that's 2000 passes. More than enough for convergence.

import numpy as np
import time

def _exact_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        if s > target: lo = mid

    res = target - float(np.sum(x))

        else:

    return _make_binary(h)

def _ls(h, deadline, max_passes=1000):

    n = len(h)
    target = n // 2
    h = _make_binary(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        best_delta = -1.0
        best_i = -1
        best_j = -1
        n_ones = len(ones)
        n_zeros = len(zeros)
        for ii in range(n_ones):
            i = ones[ii]
            for jj in range(n_zeros):
                j = zeros[jj]
                h[i] = 0.0
                h[j] = 1.0
                c = _exact_c5(h)
                d = cur_c5 - c

                    best_delta = d
                    best_i = i
                    best_j = j
                h[i] = 1.0
                h[j] = 0.0

            i, j = best_i, best_j
            h[i] = 0.0
            h[j] = 1.0
            cur_c5 += best_delta
            # Safety: recompute
            cur_c5 = _exact_c5(h)
            idx_i = ones.index(i)
            idx_j = zeros.index(j)

    n = len(h)
    target = n // 2
    h = _make_binary(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

def _gradient_polish(h, deadline, n_steps=300, step_init=0.03):
    """Smoothed gradient with subgradient on active lag."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    v = np.zeros(n)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = scores.max()
        # Softmax weights over lags
        beta = 100.0
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        # Gradient: for lag k, f_k = sum_i h[i+k]*(1-h[i]) * dx
        # df_k/dh[i] = dx * [-(1-h[i+k]) if i+k valid + h[i-k] if i-k valid]
        grad = np.zeros(n)
        for lag_idx in range(2*n-1):
            if w[lag_idx] < 1e-10:
                continue
            k = lag_idx - (n-1)
            if k >= 0:
                m = n - k
                # d/dh[i] of h[i+k]*(1-h[i]) = -(1-h[i+k]) for i in [0, m-1]
                grad[:m] += w[lag_idx] * (-(1.0 - h[k:]))
                # d/dh[i] of h[i+k]*(1-h[i]) = +h[i-k] for i in [k, n-1] (from the h[i+k] part)
                if k > 0:
                    grad[k:] += w[lag_idx] * h[:n-k]
            else:
                g = -k
                m = n - g
                # f = sum_i h[i]*(1-h[i+g]) for valid i
                # d/dh[i]: from (1-h[i+g]) part: -h[i] wait...
                # f_k = sum_i h[i-g]*(1-h[i]) for valid i (where i-g >= 0)
                # = sum_j h[j]*(1-h[j+g]) for j in [0, m-1]
                # d/dh[j] = -(1-h[j+g]) for j in [0,m-1] + h[j-g] for j in [g, n-1]
                grad[:m] += w[lag_idx] * (-(1.0 - h[g:]))
                if g > 0:
                    grad[g:] += w[lag_idx] * h[:n-g]
        grad *= (2.0 / n)
        grad -= grad.mean()  # Remove mean to preserve sum constraint approximately
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        v = 0.9 * v - step * grad / gn
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.02, 0.08)
        else:
            v *= 0.5
            step *= 0.85
            if step < 1e-7:
                break
    return best_h, best_c5
