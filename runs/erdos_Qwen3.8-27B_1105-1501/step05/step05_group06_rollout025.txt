```python
def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection on threshold
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
    for _ in range(60):
        mid = (lo + hi) / 2.0
        f_mid = corr + mid * v
        max_val = np.max(f_mid)
        active = f_mid >= max_val - 1e-12
        min_sub = np.min(v[active])
        max_sub = np.max(v[active])
        
        if min_sub <= 0 <= max_sub:
            # 0 is in subdifferential, this is the minimum
            break
        elif min_sub > 0:
            # Function is increasing, minimum is to the left
            hi = mid
        else:
            # max_sub < 0, function is decreasing, minimum is to the right
            lo = mid
    
    alpha_opt = (lo + hi) / 2.0
    alpha_opt = np.clip(alpha_opt, alpha_lo, alpha_hi)
    
    # Verify against 0 and endpoints
    best_alpha = alpha_opt
    best_val = f(alpha_opt)
    for a in [0.0, alpha_lo, alpha_hi]:
        if alpha_lo <= a <= alpha_hi:
            va = f(a)
            if va < best_val:
                best_val = va
                best_alpha = a
    
    return best_alpha
```

- Then verify with O(n) to compute the actual C₅ (or just use the linear approximation)

```python
def _compute_D(h):
    """Compute D[j, m_index] for all j and all lags.
    D[j, k] = (1-h[j+k])*[j+k<n] - h[j-k]*[j-k>=0] for lag k (where k = -(n-1)..(n-1))
    Actually, let me use a 0-based index for the full correlation vector.
    The full corr vector has length 2n-1, indexed 0 to 2n-2.
    corr[p] corresponds to lag (p - (n-1)).
    """
    n = len(h)
    D = np.zeros((n, 2*n-1))
    for j in range(n):
        # For each lag m (from -(n-1) to n-1):
        # D[j, m+(n-1)] = (1-h[j+m])*[j+m<n] - h[j-m]*[j-m>=0]
        # j+m < n means m < n-j
        # j-m >= 0 means m <= j
        # So for m = -(n-1) to n-1:
        #   [j+m < n] is true when m < n-j, i.e., m_index = m+(n-1) < n-j+n-1 = 2n-1-j
        #   [j-m >= 0] is true when m <= j, i.e., m_index = m+(n-1) <= j+n-1
        
        # Let me just compute it directly
        for p in range(2*n-1):
            m = p - (n-1)  # lag
            val = 0.0
            jm = j + m
            if 0 <= jm < n:
                val += (1.0 - h[jm])
            jmm = j - m
            if 0 <= jmm < n:
                val -= h[jmm]
            D[j, p] = val
    return D
```

This is O(n²) to compute, which is fine.

```python
def _compute_D(h):
    n = len(h)
    L = 2 * n - 1
    D = np.zeros((n, L))
    for j in range(n):
        # D[j, p] = (1-h[j+m])*[0<=j+m<n] - h[j-m]*[0<=j-m<n] where m = p-(n-1)
        # Term 1: (1-h[j+m]) for m = p-(n-1)
        #   j+m = j + p - (n-1) = p + j - n + 1
        #   Valid when 0 <= p + j - n + 1 < n, i.e., n-1-j <= p < 2n-1-j
        #   Value: 1 - h[p + j - n + 1]
        # Term 2: h[j-m] = h[j - p + n - 1] = h[n-1+j-p]
        #   Valid when 0 <= n-1+j-p < n, i.e., j < p-n+1 < j+n ... 
        #   0 <= n-1+j-p < n means: n-1+j-p >= 0 and n-1+j-p < n
        #   => p <= n-1+j and p > j-1
        #   So: j < p <= n-1+j
        #   Value: h[n-1+j-p]
        
        # Let me just use a loop for clarity
        p_range = np.arange(L)
        m = p_range - (n-1)
        jm = j + m  # = j + p - (n+1) ... let me just compute
        term1 = np.zeros(L)
        mask1 = (jm >= 0) & (jm < n)
        term1[mask1] = 1.0 - h[jm[mask1]]
        
        jmm = j - m  # = j - p + n - 1
        term2 = np.zeros(L)
        mask2 = (jmm >= 0) & (jmm < n)
        term2[mask2] = h[jmm[mask2]]
        
        D[j] = term1 - term2
    return D
```

This is O(n²) with vectorized inner loops, so it's fast.
