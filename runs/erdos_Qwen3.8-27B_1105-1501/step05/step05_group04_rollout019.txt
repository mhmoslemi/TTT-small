def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _project_box_sum(v, target):
    n = len(v)
    if target <= 1e-15:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = float(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _h_from_a(a, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = a
    h[n-m:] = 1.0 - a[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def _a_from_h(h):
    n = len(h)
    m = n // 2
    a = np.zeros(m)
    for i in range(m):
        a[i] = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
    return np.clip(a, 0.0, 1.0)

def _lse_full_obj_grad(h, beta, lam_sum):
    n = len(h)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    u = float(np.max(V))
    z = np.clip((V - u) / beta, -50.0, 50.0)
    exp_z = np.exp(z)
    sum_exp = float(np.sum(exp_z))
    w = exp_z / sum_exp
    obj = u + beta * np.log(sum_exp)
    grad = np.zeros(n)
    for r in range(len(V)):
        wr = w[r]
        if abs(wr) < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m2 = n - d
            grad[:m2] += wr * (1.0 - h[d:])
            grad[d:] -= wr * h[:m2]
        else:
            e = -d
            m2 = n - e
            grad[e:] += wr * (1.0 - h[:m2])
            grad[:m2] -= wr * h[e:]
    grad *= dx
    if lam_sum > 0:
        s = float(np.sum(h))
        obj += lam_sum * (s - n / 2.0) ** 2
        grad += 2.0 * lam_sum * (s - n / 2.0)
    return obj, grad
