def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _grad_at_lag(h, d):
    """Gradient of correlation at signed lag d w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] += -h[:m]
    else:
        g2 = -d
        m = n - g2
        if m > 0:
            g[g2:] += (1.0 - h[:m])
            g[:m] += -h[g2:]
    return g

def _softmax_grad(h, beta):
    """Compute gradient of log-sum-exp objective."""
    n = len(h)

def _line_search_robust(h, d, dx, max_step):

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')

    t_hi = max_step

        # Source block
        i = int(rng.integers(0, n - block_size + 1))
        # Target block
        j = int(rng.integers(0, n - block_size + 1))

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _line_search(h, d, dx, max_step):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = max_step
    t_lo = -max_step
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    ns = 60
    ts = np.linspace(t_lo, t_hi, ns)
    vals = np.empty(ns)
    for k in range(ns):
        t = ts[k]
        vals[k] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_k = int(np.argmin(vals))
    if best_k == 0:
        lo, hi = t_lo, ts[min(4, ns - 1)]
    elif best_k == ns - 1:
        lo, hi = ts[max(0, ns - 5)], t_hi
    else:
        lo = ts[max(0, best_k - 3)]
        hi = ts[min(ns - 1, best_k + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if hi - lo < 5e-13:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if vals[best_k] < opt_val:
        opt_t = ts[best_k]
        opt_val = vals[best_k]
    return opt_t, opt_val

def _subgrad_direction(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr))
    tol = max_val - (max_val * 1e-6 + 1e-10)
    active = np.where(corr >= tol)[0]
    g = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:])
                g[d:] += -h[:m]
        else:
            g2 = -d
            m = n - g2
            if m > 0:
                g[g2:] += (1.0 - h[:m])
                g[:m] += -h[g2:]
    n_active = len(active)
    if n_active > 0:
        g /= n_active
    g -= np.mean(g)
    norm = np.sqrt(np.sum(g * g))
    if norm > 1e-15:
        g /= norm
    return g

def _momentum_optimize(h, deadline, max_iter=20000):

    target = n / 2.0
    dx = 2.0 / n

            cur_c5 = _exact_c5(h)

            stall = 0

        else:
            stall += 1
