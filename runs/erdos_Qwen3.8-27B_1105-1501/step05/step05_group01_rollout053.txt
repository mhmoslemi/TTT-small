def _project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

        else:

    # Refine around best with golden section on local interval
    lo_t = ts[max(0, best_idx-2)]

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, t_max, n_grid=40):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    ts = np.linspace(0.0, t_max, n_grid)
    base = S0[None, :]
    tt = ts[:, None]
    vals = np.max(base + tt * T[None, :] - tt * tt * Q[None, :], axis=1) * dx
    best_idx = int(np.argmin(vals))
    best_t = float(ts[best_idx])
    best_val = float(vals[best_idx])
    lo_t = float(ts[max(0, best_idx - 3)])
    hi_t = float(ts[min(n_grid - 1, best_idx + 3)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(30):
        if hi_t - lo_t < 1e-11:
            break
        if f1 < f2:
            hi_t = c2
            c2 = c1
            f2 = f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo_t = c1
            c1 = c2
            f1 = f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (lo_t + hi_t) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * Q)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    if best_val > float(np.max(S0)) * dx - 1e-14:
        return 0.0, float(np.max(S0)) * dx
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[r, :m] += (1.0 - h[d:d + m])
            G[r, d:] += -h[:m]
        else:
            g = -d
            G[r, g:] += (1.0 - h[:m])
            G[r, :m] += -h[g:g + m]
    return G

def _lp_direction(h, G, scores, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores + 0.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = 0.0
            hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgrad_direction(h, G, scores, radius, eps_rel=0.02):
    n = len(h)
    mx = float(np.max(scores))
    active = scores >= mx * (1.0 - eps_rel)
    cnt = int(np.sum(active))
    if cnt == 0:
        return None
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if active[r]:
            g += G[r, :]
    g /= cnt
    g -= np.mean(g)
    norm = float(np.max(np.abs(g)))
    if norm < 1e-15:
        return None
    d = -g / norm
    for j in range(n):
        if d[j] > 0:
            max_t = min(radius, (1.0 - h[j]) / d[j])
        elif d[j] < 0:
            max_t = min(radius, h[j] / (-d[j]))
        else:
            max_t = radius
        d[j] *= 1.0
    d = np.clip(d, -radius, radius)
    d = np.where(h + d > 1.0, h + d - (h + d - 1.0), d)
    d = np.clip(d, -h, 1.0 - h)
    d = np.clip(d, -radius, radius)
    d -= np.mean(d)
    return d

def _targeted_lag_swap(h, dx, target, n_tries=50):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(scores)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(n_tries):
        worst_r = int(np.argmax(scores))
        d_lag = worst_r - (n - 1)
        if d_lag == 0:
            break
        m = n - abs(d_lag)
        if d_lag > 0:
            pairs = [(i, i + d_lag) for i in range(m)]
        else:
            g = -d_lag
            pairs = [(i + g, i) for i in range(m)]
        best_pair = None
        best_diff = 0.0
        for (i, j) in pairs:
            diff = h[i] - h[j]
            if abs(diff) > abs(best_diff):
                best_diff = diff
                best_pair = (i, j)
        if best_pair is None or abs(best_diff) < 1e-8:
            break
        i, j = best_pair
        if h[i] >= h[j]:
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        t_max = min(0.4, abs(h[i] - h[j]) + 1e-6)
        alpha, c5_try = _line_search(h, d, dx, t_max)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            best_c5 = _exact_c5(h)
            best_h = h.copy()
            scores = np.correlate(h, 1.0 - h, mode='full')
        else:
            break
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.12, max_iter=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, G, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            p = _subgrad_direction(h, G, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.65
            if radius < 1e-7:
                break
            continue
        t_max = float(np.min(np.maximum(
            np.where(p > 1e-14, (1.0 - h[p > 1e-14]) / p[p > 1e-14], np.inf),
            np.where(p < -1e-14, h[p < -1e-14] / (-p[p < -1e-14]), np.inf)
        ))) if (np.max(np.abs(p)) > 1e-14) else 0.0
        t_max = min(t_max, radius)
        if t_max < 1e-14:
            stall += 1
            radius *= 0.65
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, t_max)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.72
            if stall % 25 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):

    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _structured_binary(n, rng, kind=0):
    h = np.zeros(n)
    half = n // 2
    if kind == 0:
        idx = rng.permutation(n)[:half]
        h[idx] = 1.0
    elif kind == 1:

        fill = max(1, period // 2)

                h[i] = 1.0

        if excess > 0:
            ones = np.where(h == 1.0)[0]
            to_flip = rng.choice(ones, size=min(excess, len(ones)), replace=False)
            h[to_flip] = 0.0
        elif excess < 0:
            zeros = np.where(h == 0.0)[0]
            to_flip = rng.choice(zeros, size=min(-excess, len(zeros)), replace=False)
            h[to_flip] = 1.0
    elif kind == 2:
        blocks = rng.integers(2, n // 8 + 2)
        block_size = n // blocks
        for b in range(blocks):
            start = b * block_size
            end = min(n, start + block_size)
            if b % 2 == 0:
                h[start:end] = 1.0
        excess = int(np.sum(h)) - half
        if excess > 0:
            ones = np.where(h == 1.0)[0]
            to_flip = rng.choice(ones, size=min(excess, len(ones)), replace=False)
            h[to_flip] = 0.0
        elif excess < 0:
            zeros = np.where(h == 0.0)[0]
            to_flip = rng.choice(zeros, size=min(-excess, len(zeros)), replace=False)
            h[to_flip] = 1.0
    else:
        idx = rng.permutation(n)[:half]
        h[idx] = 1.0
    return h

def _random_pair_swap(h, deadline, n_tries=3000, radius=0.2):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    while time.time() < deadline:
        if stall > 800:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        if h[i] >= h[j]:
            d[i] = -1.0
            d[j] = 1.0
        else:
            d[i] = 1.0
            d[j] = -1.0
        t_max = min(radius, abs(h[i] - h[j]) + 0.1)
        alpha, c5_try = _line_search(h, d, dx, t_max)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                radius *= 0.85
        if radius < 0.01:
            break
    return best_h, best_c5

def _all_pair_swap_very(h, deadline, max_passes=4):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(0, n, 2):
            if time.time() > deadline:
                break
            for j in range(i + 1, n, 3):
                if time.time() > deadline:
                    break
                d = np.zeros(n)
                if h[i] >= h[j]:
                    d[i] = -1.0
                    d[j] = 1.0
                else:
                    d[i] = 1.0
                    d[j] = -1.0
                t_max = 0.3
                alpha, c5_try = _line_search(h, d, dx, t_max)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP warmup from initial (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_loop(h_init, d1, radius_init=0.10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Targeted lag swap refinement (8%)
    d2 = t0 + budget_s * 0.20
    h2 = best_h.copy()
    h2, c5_2 = _targeted_lag_swap(h2, 2.0 / len(h2), len(h2) / 2.0, n_tries=200)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-start exploration (20%)
    d3 = t0 + budget_s * 0.40
    n_curr = len(best_h)
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16, -24, 24, -32, 32, -1, 1, -6, 6, -20, 20]
    tried_ns = set([n0])
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        rem = d3 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.005:
            sub_dl2 = min(d3, time.time() + 3.0)
            h_r, c5_r = _targeted_lag_swap(h_r, 2.0 / len(h_r), len(h_r) / 2.0, n_tries=100)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Also try a few random starts
    d3b = min(d3 + budget_s * 0.05, t0 + budget_s * 0.45)
    for trial in range(6):
        if time.time() > d3b:
            break
        n_try = n_curr
        kind = trial % 4
        h_r = _structured_binary(n_try, rng, kind=kind)
        sub_dl = min(d3b, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Main intensive optimization (30%)
    d4 = t0 + budget_s * 0.75
    h4 = best_h.copy()
    # Long LP pass
    sub_dl = min(d4, time.time() + budget_s * 0.10)
    h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.08)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Targeted swap
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4 = best_h.copy()
    h4, c5_4 = _targeted_lag_swap(h4, 2.0 / len(h4), len(h4) / 2.0, n_tries=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Another LP pass
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4 = best_h.copy()
    h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.06)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Random pair swap
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4 = best_h.copy()
    h4, c5_4 = _random_pair_swap(h4, sub_dl, n_tries=2000, radius=0.15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Final LP in phase 4
    sub_dl = min(d4, time.time() + budget_s * 0.02)
    h4 = best_h.copy()
    h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.04)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Perturbation + re-optimize (13%)
    d5 = t0 + budget_s * 0.88
    scale = 0.03
    for attempt in range(300):
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05)
        if c5_t < best_c5 - 1e-6:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.10, scale * 1.08)

    # Phase 6: Final polish (8%)
    d6 = deadline - 5.0
    h6 = best_h.copy()
    h6, c5_6 = _lp_loop(h6, d6, radius_init=0.03)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    h6 = best_h.copy()
    h6, c5_6 = _targeted_lag_swap(h6, 2.0 / len(h6), len(h6) / 2.0, n_tries=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    h6 = best_h.copy()
    h6, c5_6 = _all_pair_swap_very(h6, deadline - 2.0, max_passes=3)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final LP micro-polish
    h6 = best_h.copy()
    h6, c5_6 = _lp_loop(h6, deadline - 1.0, radius_init=0.02)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
