        else:

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    gap = target - float(np.sum(x))
    if abs(gap) > 1e-12:
        if gap > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += gap / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, dx, r):
    """Gradient of corr_scaled[r] w.r.t. h. Returns array of shape (n,)."""
    n = len(h)
    s = r - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        for i in range(n):
            v = 0.0
            if s <= i < n:
                v += (1.0 - h[i - s])
            if 0 <= i < m:
                v -= h[i + s]
            g[i] = dx * v
    else:
        gg = -s
        m = n - gg
        for i in range(n):
            v = 0.0
            if 0 <= i < m:
                v += (1.0 - h[i + gg])
            if gg <= i < n:
                v -= h[i - gg]
            g[i] = dx * v
    return g

def _grad_rows(h, dx, row_indices):
    """Gradients for multiple rows. Returns array of shape (len(row_indices), n)."""
    n = len(h)
    G = np.zeros((len(row_indices), n))
    for idx, r in enumerate(row_indices):
        G[idx] = _grad_row(h, dx, r)
    return G

def _slsqp_solve(h0, deadline, K=40, max_iter_per_call=300):
    """SLSQP with epigraph formulation using top-K active constraints."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = _proj(h0.copy(), target)
    c5_init = _exact_c5(h)
    
    def objective(x):
        return x[n]
    
    def objective_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def make_constraint(r):
        def confun(x):
            h_x = x[:n]
            corr_r = np.correlate(h_x, 1.0 - h_x, mode='full') * dx
            return corr_r[r] - x[n]
        def conjac(x):
            h_x = x[:n]
            g = np.zeros(n + 1)
            g[:n] = _grad_row(h_x, dx, r)
            g[n] = -1.0
            return g
        return {'type': 'ineq', 'fun': lambda x: -confun(x), 'jac': lambda x: -conjac(x)}
    
    def eqfun(x):
        return float(np.sum(x[:n])) - target
    
    def eqjac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    # Identify active rows
    corr_vals = _corr_scaled(h)
    max_corr = float(np.max(corr_vals))
    threshold = max_corr - 0.0005
    active = np.where(corr_vals >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-corr_vals)[:K]
    if len(active) > K:
        # Keep top K
        active = active[np.argsort(-corr_vals[active])][:K]
    active = sorted(active)
    
    constraints = [make_constraint(r) for r in active]
    constraints.append({'type': 'eq', 'fun': eqfun, 'jac': eqjac})
    
    x0 = np.concatenate([h, [c5_init + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    
    best_h = h.copy()
    best_c5 = c5_init
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter_per_call, 'iprint': 0,
                                'ftol': 1e-14})
        if res.x is not None:
            h_new = res.x[:n]
            h_new = _proj(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    except Exception:
        pass
    
    return best_h, best_c5

def _active_descent(h0, deadline, tol=1e-5, K=20):
    """Active-set steepest descent with exact line search."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = _proj(h0.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    step_scale = 0.5
    
    while time.time() < deadline and stall < 200:
        corr_vals = _corr_scaled(h)
        max_c = float(np.max(corr_vals))
        threshold = max_c - tol
        active = np.where(corr_vals >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-corr_vals)[:K]
        if len(active) > K:
            active = active[np.argsort(-corr_vals[active])][:K]
        
        # Compute gradient of C5 ≈ average of gradients of active rows
        G = _grad_rows(h, dx, active)
        # We want to decrease C5, so direction = -mean of active gradients
        # But we need to be in the null space of sum(h) = target
        # Project: d = -mean(G) + (mean(G).mean())*ones (to make sum(d)=0)
        d_raw = -G.mean(axis=0)
        d = d_raw - np.mean(d_raw)  # Make sum-zero
        
        # Handle box constraints: zero out components that would violate
        # For positive d[i], h[i] can increase; for negative d[i], h[i] can decrease
        mask_pos = d > 1e-16
        mask_neg = d < -1e-16
        # Scale down directions that would immediately violate bounds
        if np.any(mask_pos) and np.any(h[mask_pos] >= 0.999):
            d[mask_pos & (h >= 0.999)] = 0.0
        if np.any(mask_neg) and np.any(h[mask_neg] <= 0.001):
            d[mask_neg & (h <= 0.001)] = 0.0
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            stall += 1
            continue
        
        d = d / d_norm * step_scale
        
        # Safe line search
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        
        # Find max step before box violation
        t_max = 1.0
        pos = d > 1e-14
        neg = d < -1e-14
        if np.any(pos):
            t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_max = max(t_max, float(np.max(h[neg] / d[neg])))
        t_max = min(t_max, 1.0)
        
        if t_max < 1e-12:
            stall += 1
            step_scale *= 0.8
            continue
        
        # Golden section search on [0, t_max]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = 0.0, t_max
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):

            if f1 < f2:

                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (a + b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        if opt_val < f1:
            best_t = opt_t
        elif f1 <= f2:
            best_t = c1
        else:
            best_t = c2
        best_t_val = float(np.max(S0 + best_t * T - best_t * best_t * C)) * dx
        
        if best_t_val < cur_c5 - 1e-14:
            h = h + best_t * d
            h = _proj(h, target)
            cur_c5 = _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _pair_refine(h0, deadline, max_rounds=100):
    """Iterative pair-move refinement."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try all pairs in random order
        pairs = []
        idx = np.arange(n)
        rng.shuffle(idx)
        for ii in range(n):
            for jj in range(ii + 1, n):
                i, j = idx[ii], idx[jj]
                pairs.append((i, j))
        rng.shuffle(pairs)
        
        for i, j in pairs:
            if time.time() > deadline:
                break
            for sign in [1.0, -1.0]:
                alpha, c5_try = _pair_move_line_search(h, i, j, sign)
                if c5_try < best_c5 - 1e-13:
                    h[i] += alpha * sign
                    h[j] -= alpha * sign
                    h = _proj(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

        for signs in sign_options:
            alpha, c5_try = _quadruplet_move(h, i, j, k, l, signs)
            if c5_try < best_c5 - 1e-13:
                h[i] += alpha * signs[0]
                h[j] += alpha * signs[1]
                h[k] += alpha * signs[2]
                h[l] += alpha * signs[3]
                h = _proj(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    stall = 0
        if not improved:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        h0 = np.zeros(n0)
        h0[::2] = 1.0
    
    n0 = len(h0)
    h = _proj(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Phase 1: SLSQP with various K (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _slsqp_multi_k(h, d1, K_list=[20, 40], max_iter=150)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Active-set descent (20% time)
    d2 = min(t0 + budget_s * 0.50, deadline)
    h2, c5_2 = _active_descent(best_h, d2, tol=1e-4, K=15)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: SLSQP again on improved point (15% time)
    d3 = min(t0 + budget_s * 0.65, deadline)
    h3, c5_3 = _slsqp_multi_k(best_h, d3, K_list=[30, 60], max_iter=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Pair refinement (20% time)
    d4 = min(t0 + budget_s * 0.85, deadline)
    h4, c5_4 = _pair_refine(best_h, d4, max_rounds=60)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Quadruplet refinement (5% time)
    d5 = min(t0 + budget_s * 0.90, deadline)
    h5, c5_5 = _quad_refine(best_h, d5, rng, max_rounds=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Multi-resolution (5% time)
    d6 = min(t0 + budget_s * 0.95, deadline)
    n_cur = len(best_h)
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > d6:
            break
        n_try = n_cur + dn
        if n_try < 24 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _active_descent(h_r, min(d6, time.time() + 5), tol=1e-4, K=15)
        if c5_r < best_c5:
            h_r2, c5_r2 = _pair_refine(h_r, min(d6, time.time() + 5), max_rounds=20)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
    
    # Final SLSQP polish
    if time.time() < deadline - 5:
        h_f, c5_f = _slsqp_solve(best_h, deadline - 2, K=40, max_iter_per_call=100)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Ensure validity
    best_h = _proj(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
