```python
    dx = 2.0 / n
    nlags = 2 * n - 1
    G = np.zeros((nlags, n))
    for r in range(nlags):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
```

```python
    n = len(h)
    dx = 2.0 / n
    c0 = eval_c5(h)
    
    # Get initial scores to determine active set
    scores = lag_values(h)
    sorted_idx = np.argsort(scores)[::-1]
    active_idx = sorted_idx[:n_active_lags]
    
    def objective(x):
        return x[-1]
    
    def equality_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def equality_jac(x):
        jac = np.zeros(n + 1)
        jac[:n] = 1.0
        return jac
    
    def inequality_fun(x):
        h_val = x[:n]
        t = x[-1]
        sv = lag_values(h_val)
        return t - sv[active_idx]  # These should be >= 0
    
    def inequality_jac(x):
        h_val = x[:n]
        # Jacobian of (t - F_s(h)) w.r.t. x
        # d/dh_j (t - F_s) = -dF_s/dh_j
        # d/dt (t - F_s) = 1
        G = lag_jacobian(h_val)
        jac = np.zeros((len(active_idx), n + 1))
        jac[:, :n] = -G[active_idx, :]
        jac[:, n] = 1.0
        return jac
    
    constraints = [
        {'type': 'eq', 'fun': equality_fun, 'jac': equality_jac},
        {'type': 'ineq', 'fun': inequality_fun, 'jac': inequality_jac}
    ]
    
    bounds = [(0, 1)] * n + [(0, c0 + 0.1)]
    x0 = np.concatenate([h, [c0 + 1e-4]])
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8})

        h_new = project_box_sum(res.x[:n], n / 2.0)
        c5_new = eval_c5(h_new)

    return h, c0
```

My formulation:

- Inequality: t - F_r(h) >= 0 for each active lag r

The objective is minimize t.

The Jacobian of the ith inequality constraint with respect to x = [h, t]:
- d/dh_j (t - F_i(h)) = -dF_i/dh_j
- d/dt (t - F_i(h)) = 1

```python
def epigraph_solve(h, deadline, n_active, ftol=1e-8, maxiter=150):
    n = len(h)
    c0 = eval_c5(h)
    
    # Determine active lags
    scores = lag_values(h)
    sorted_idx = np.argsort(scores)[::-1]
    active_idx = sorted_idx[:n_active]
    
    def _obj(x):
        return x[-1]
    
    def _eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def _eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    def _ineq_fun(x):
        h_val = x[:n]
        t_val = x[-1]
        sv = lag_values(h_val)
        return t_val - sv[active_idx]
    
    def _ineq_jac(x):
        h_val = x[:n]
        G_active = lag_jacobian_rows(h_val, active_idx)
        j = np.zeros((len(active_idx), n + 1))
        j[:, :n] = -G_active
        j[:, n] = 1.0
        return j
    
    constraints = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}
    ]
    
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    x0 = np.concatenate([h, [c0 + 1e-5]])
    
    try:
        res = minimize(_obj, x0, method='SLSQP', jac=[0.0]*n + [1.0],
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], n / 2.0)
            c5_new = eval_c5(h_new)
            if c5_new < c0:
                return h_new, c5_new
    except Exception:
        pass
    
    return h, c0
```

```python
    G_active = lag_jacobian_rows(h_val, active_idx)
    j = np.zeros((len(active_idx), n + 1))
    j[:, :n] = -G_active
    j[:, n] = 1.0
    return j
```

```python
def lag_values(h):
    n = len(h)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_jacobian_rows(h, indices):
    n = len(h)
    dx = 2.0 / n
    m = len(indices)
    G = np.zeros((m, n))
    for idx in range(m):
        r = indices[idx]
        k = r - (n - 1)
        if k >= 0:
            mm = n - k
            if mm > 0:
                G[idx, k:] += dx * (1.0 - h[:mm])
                G[idx, :mm] += -dx * h[k:]
        else:
            a = -k
            mm = n - a
            if mm > 0:
                G[idx, :mm] += dx * (1.0 - h[a:])
                G[idx, a:] += -dx * h[:mm]
    return G

def lag_jacobian(h):
    n = len(h)
    nlags = 2 * n - 1
    return lag_jacobian_rows(h, list(range(nlags)))

def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])

def quad_line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    
    n_grid = 60
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val
```

def lp_direction(h, scores, radius, n_active):
    n = len(h)
    nlags = 2 * n - 1
    dx = 2.0 / n
    
    # Select active lags
    sorted_idx = np.argsort(scores)[::-1]
    active_set = sorted_idx[:max(n_active, 3)]
    
    G = lag_jacobian_rows(h, active_set)
    
    # LP: minimize u
    # subject to: G_active @ p - u <= -scores[active_set]
    #             sum(p) = 0
    #             -radius*h <= p <= radius*(1-h)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(active_set), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active_set]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-13:
                return p
    except Exception:
        pass
    return None
