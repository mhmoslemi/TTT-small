```python
    if target >= n:

    z = [h_0, ..., h_{n-1}, t]
    Minimize t subject to t >= score_d for all d, sum(h) = n/2.
    """
    h = z[:n]
    t = z[n]
    # Objective: t
    return t

    """Equality constraint: sum(h) - n/2 = 0"""
    return z[:n].sum() - n / 2.0

    """Jacobian of equality constraint."""
    jac = np.zeros(n + 1)
    jac[:n] = 1.0
    return jac

def _compute_scores_raw(h):
    """Compute all raw scores (without dx factor)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr

def _slsqp_ineq_func(z, n):
    """Inequality constraints: t - score_d >= 0 for all d."""
    h = z[:n]
    t = z[n]
    scores = _compute_scores_raw(h)
    return t - scores

def _slsqp_ineq_jac(z, n):
    """Jacobian of inequality constraints.
    For constraint t - score_d >= 0:
    d(t - score_d)/d(h_j) = -d(score_d)/d(h_j)
    d(t - score_d)/dt = 1
    """
    h = z[:n]
    n_scores = 2 * n - 1
    jac = np.zeros((n_scores, n + 1))
    jac[:, n] = 1.0  # dt/dt = 1
    
    # For each score index k (0 to 2n-2), compute gradient w.r.t. h
    # score_k = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i
    # Let d = k - (n-1) be the lag
    # If d >= 0: score = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
    #   d(score)/d(h[j]):
    #     if j in [0, n-1-d]: contribution from h[j]*(1-h[j+d]) term: (1-h[j+d])
    #     if j in [d, n-1]: contribution from h[j-d]*(1-h[j]) term: -h[j-d]
    # If d < 0, let g = -d: score = sum_{i=g}^{n-1} h[i] * (1-h[i-g])
    #   d(score)/d(h[j]):
    #     if j in [g, n-1]: contribution (1-h[j-g])
    #     if j in [0, n-1-g]: contribution -h[j+g]
    
    for k in range(n_scores):
        d = k - (n - 1)
        if d >= 0:
            m = n - d  # number of terms
            if m <= 0:
                continue
            # score = sum_{i=0}^{m-1} h[i] * (1-h[i+d])
            # d(score)/d(h[j]):
            #   from first factor (h[i], i=j): 1-h[j+d] if j < m
            #   from second factor (h[i+d], i+d=j, i=j-d): -h[j-d] if j >= d
            if d == 0:
                # score = sum h[i]*(1-h[i]), d/dh[j] = 1-2h[j]
                jac[k, :n] = 1.0 - 2.0 * h
            else:
                # First factor: j in [0, m-1] = [0, n-1-d]
                jac[k, :m] += (1.0 - h[d:])
                # Second factor: j in [d, n-1], contribution -h[j-d]
                jac[k, d:] += -h[:m]
        else:
            g = -d  # g > 0
            m = n - g
            # score = sum_{i=g}^{n-1} h[i] * (1-h[i-g])
            # Let j = i: j in [g, n-1], contribution (1-h[j-g])
            # Let j = i-g: j in [0, n-1-g], contribution -h[j+g]
            if g == 0:
                jac[k, :n] = 1.0 - 2.0 * h
            else:
                jac[k, g:] += (1.0 - h[:m])
                jac[k, :m] += -h[g:]
    
    return jac
```

```python
    """Compute gradient of score_d w.r.t. h.
    d is the lag (can be negative).
    For d >= 0: score = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
    For d < 0: let g=-d, score = sum_{i=g}^{n-1} h[i] * (1-h[i-g])
    """
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        return 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m <= 0:
            return grad
        # h[i] appears as first factor: i in [0, m-1], contrib: 1-h[i+d]
        grad[0:m] += (1.0 - h[d:n])
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

class _EP:
    """Epigraph SLSQP problem for a given n."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
        self.n_s = 2 * n - 1
    
    def fun(self, z):
        return z[self.n]
    
    def eq_fun(self, z):
        return z[:self.n].sum() - self.target
    
    def eq_jac(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g
    
    def ineq_fun(self, z):
        return z[self.n] - _scores_raw(z[:self.n])
    
    def ineq_jac(self, z):
        h = z[:self.n]
        n = self.n
        n_s = self.n_s
        jac = np.zeros((n_s, n + 1))
        jac[:, n] = 1.0
        for k in range(n_s):
            d = k - (n - 1)
            if d == 0:
                jac[k, :n] = 1.0 - 2.0 * h
            elif d > 0:
                m = n - d
                jac[k, :m] = 1.0 - h[d:n]
                jac[k, d:n] += -h[:m]
            else:
                g = -d
                m = n - g
                jac[k, g:n] = 1.0 - h[:m]
                jac[k, :m] += -h[g:n]
        return jac
    
    def solve(self, h_init, maxiter=300, ftol=1e-10):
        n = self.n
        h0 = _project(h_init, self.target)
        s0 = _scores_raw(h0)
        t0 = float(np.max(s0)) + 0.5

            {'type': 'ineq', 'fun': self.ineq_fun, 'jac': self.ineq_jac}
        ]
        res = minimize(self.fun, z0, method='SLSQP', bounds=bounds,

        if res.x is not None:
            h_opt = _project(res.x[:n], self.target)
            return h_opt, _exact_c5(h_opt)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

class _EP:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
        self.n_s = 2 * n - 1

    def fun(self, z):
        return z[self.n]

    def eq_fun(self, z):
        return z[:self.n].sum() - self.target

    def eq_jac(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

    def ineq_fun(self, z):
        return z[self.n] - _scores_raw(z[:self.n])

    def ineq_jac(self, z):
        h = z[:self.n]
        n = self.n
        n_s = self.n_s
        jac = np.zeros((n_s, n + 1))
        jac[:, n] = 1.0
        for k in range(n_s):
            d = k - (n - 1)
            if d == 0:
                jac[k, :n] = 1.0 - 2.0 * h
            elif d > 0:
                m = n - d
                jac[k, :m] = 1.0 - h[d:n]
                jac[k, d:n] += -h[:m]
            else:
                g = -d
                m = n - g
                jac[k, g:n] = 1.0 - h[:m]
                jac[k, :m] += -h[g:n]
        return jac

    def solve(self, h_init, maxiter=300, ftol=1e-10):
        n = self.n
        h0 = _project(h_init, self.target)
        s0 = _scores_raw(h0)
        t0 = float(np.max(s0)) + 0.3
        z0 = np.concatenate([h0, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        cons = [
            {'type': 'eq', 'fun': self.eq_fun, 'jac': self.eq_jac},
            {'type': 'ineq', 'fun': self.ineq_fun, 'jac': self.ineq_jac}
        ]
        res = minimize(self.fun, z0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_opt = _project(res.x[:n], self.target)
            c5 = _exact_c5(h_opt)
            return h_opt, c5
        return h0, _exact_c5(h0)

def _slsqp_opt(h_init, n, maxiter=300, ftol=1e-10):
    prob = _EP(n)
    return prob.solve(h_init, maxiter, ftol)

def _coord_polish(h_init, n, deadline, n_cycles=2):

def _quasi_periodic_init(n, alpha, noise=0.02, rng=None):

        h7, c5_7 = _slsqp_opt(h_best, n_best, maxiter=400, ftol=1e-12)
        if c5_7 < c5_best:
            c5_best = c5_7
            h_best = h7.copy()
        if remaining() > 15:
            h7b, c5_7b = _smoothed_subgrad(h_best, n_best, dl7, n_steps=300, step_init=0.02, mom=0.88)
            if c5_7b < c5_best:
                c5_best = c5_7b
                h_best = h7b.copy()

    # Phase 8: Coordinate polish
    if remaining() > 20:
        dl8 = t0 + budget_s * 0.95
        h8, c5_8 = _coord_polish(h_best, n_best, dl8, n_cycles=2)
        if c5_8 < c5_best:
            c5_best = c5_8
            h_best = h8.copy()

    # Phase 9: Final SLSQP
    if remaining() > 10:
        dl9 = t0 + budget_s * 0.995
        h9, c5_9 = _slsqp_opt(h_best, n_best, maxiter=500, ftol=1e-13)
        if c5_9 < c5_best:
            c5_best = c5_9
            h_best = h9.copy()

    # Final verification
    h_best = _project(h_best, n_best / 2.0)
    final_c5 = _exact_c5(h_best)
    return h_best, final_c5, n_best
```
