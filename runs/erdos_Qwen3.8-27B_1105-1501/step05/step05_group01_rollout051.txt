    stall = 0
    rng = np.random.default_rng(777)
    for _it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                radius *= 0.92
                if radius < 0.005:
                    radius = 0.04
    return best_h, best_c5

def _local_opt_subgrad(h, deadline, max_iter=5000, radius=0.06):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for _it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        d = _subgrad_dir(h, scores, rel_tol=0.05)
        if d is None:
            break
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            radius *= 0.9
            if radius < 0.003:
                break
    return best_h, best_c5

def _local_opt_block(h, deadline, max_iter=8000, block_sizes=(2, 3, 4, 5, 6, 8), radius=0.06):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(4242)
    for _it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        k = block_sizes[int(rng.integers(0, len(block_sizes)))]
        if k >= n:
            k = 1
        i = int(rng.integers(0, n - k + 1))
        j = int(rng.integers(0, n - k + 1))
        if j <= i + k - 1 and j >= i - k:
            stall += 1
            continue
        d = np.zeros(n)
        d[i:i+k] = 1.0 / k
        d[j:j+k] = -1.0 / k
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                radius *= 0.93
                if radius < 0.004:
                    radius = 0.03
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    h_b = np.zeros(n)
    h_b[idx] = 1.0
    return h_b

def _sinusoidal_perturb(h, freq, amp, rng):
    n = len(h)
    x = np.arange(n) / n
    phase = rng.uniform(0, 2 * np.pi)
    d = amp * np.sin(2 * np.pi * freq * x + phase)
    d -= np.mean(d)
    return h + d

def _polish(h, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    # Alternating subgrad and pair
    for phase in range(6):
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 40)
        best_h, best_c5 = _local_opt_subgrad(best_h, sub_dl, max_iter=2000, radius=0.04)
    for phase in range(6):
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 40)
        best_h, best_c5 = _local_opt_pair(best_h, sub_dl, max_iter=5000, radius=0.05)
    # Block refinement
    for phase in range(4):
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 50)
        best_h, best_c5 = _local_opt_block(best_h, sub_dl, max_iter=3000, radius=0.04)
    # Final subgrad
    sub_dl = min(deadline, time.time() + 30)
    best_h, best_c5 = _local_opt_subgrad(best_h, sub_dl, max_iter=3000, radius=0.03)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick local opt on initial (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _local_opt_subgrad(h_init, d1, max_iter=3000, radius=0.06)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    h1, c5_1 = _local_opt_pair(h_init, d1, max_iter=5000, radius=0.06)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Block moves on best (15%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _local_opt_block(best_h, d2, max_iter=6000, block_sizes=(2, 3, 4, 5, 6, 8), radius=0.05)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    h2, c5_2 = _local_opt_subgrad(best_h, d2, max_iter=2000, radius=0.04)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (25%)
    d3 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, 24, -24, 32, -32, 40, -40, 48, 56, -48]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _local_opt_subgrad(h_r, sub_dl, max_iter=1500, radius=0.05)
        rem = d3 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.005:
            sub_dl2 = min(d3, time.time() + max(3, budget_s * 0.012))
            h_r, c5_r = _local_opt_pair(h_r, sub_dl2, max_iter=4000, radius=0.05)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Basin hopping (20%)
    d4 = t0 + budget_s * 0.75
    h_bh = best_h.copy()
    c5_bh = best_c5
    scale = 0.06
    for attempt in range(400):
        if time.time() > d4:
            break
        n_h = len(h_bh)
        if attempt % 20 == 0:
            # Try a binary restart
            h_pert = _binarize(h_bh)
            # Add sinusoidal perturbation
            freq = int(rng.integers(2, 8))
            h_pert = _sinusoidal_perturb(h_pert, freq, 0.3, rng)
        else:
            h_pert = h_bh + rng.normal(0, scale, n_h)
        h_pert = _project(h_pert, n_h / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_t, c5_t = _local_opt_subgrad(h_pert, sub_dl, max_iter=1500, radius=0.05)
        if time.time() < d4 and c5_t < best_c5 + 0.003:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_t, c5_t = _local_opt_pair(h_t, sub_dl2, max_iter=3000, radius=0.04)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            c5_bh = c5_t
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.05)

    # Phase 5: Intensive polish (20%)
    d5 = deadline
    h5 = best_h.copy()
    h5, c5_5 = _polish(h5, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final: one more round of different n
    if time.time() < deadline - 20:
        n_final = len(best_h)
        for dn in [-2, 2, -6, 6]:
            if time.time() > deadline - 5:
                break
            n_try = n_final + dn
            if n_try < 40 or n_try > 300:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(deadline, time.time() + 10)
            h_r, c5_r = _local_opt_subgrad(h_r, sub_dl, max_iter=1000, radius=0.03)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
