def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _scores(h):

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _sigmoid(x):
    """Numerically stable sigmoid"""
    out = np.empty_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _line_search_full(h, d, dx, trust_radius):
    """Exact line search for h + t*d with box and sum constraints"""
    n = len(h)
    d = d - d.mean()  # zero mean preserves sum

    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')

def _grad_matrix(h):
    """Compute gradient matrix G where G[r,j] = d(score_r)/d(h_j)"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _active_grad(h, scores, top_k=5):
    """Compute supergradient focusing on top-k active lags"""
    n = len(h)
    s_max = float(np.max(scores))
    # Find top-k lags
    idx = np.argsort(scores)[-top_k:]
    G = _grad_matrix(h)
    # Weight by exp(beta * (score - s_max))
    beta = 50.0
    w = np.exp(beta * (scores[idx] - s_max))
    w = w / float(np.sum(w))
    grad = (G[idx] * w[:, None]).sum(axis=0)
    grad -= grad.mean()
    return grad

def _lp_descent(h, scores, G, radius, active_lags=None):
    """LP to find descent direction"""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if active_lags is not None:
        for k in active_lags:
            b_ub[k] -= 0.001  # Push below
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sigmoid(x):

            if c5_r < best_c5 - 1e-12 and time.time() < deadline - 10:
                sub_dl2 = min(deadline, time.time() + 8.0)
                h_r, c5_r = _pair_refine(h_r, sub_dl2, max_passes=40, rng=rng)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
    return best_h, best_c5


def _kick_reopt(h_best, deadline, rng, n_kicks=20):
    n = len(h_best)
    target = n / 2.0
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _active_set_opt(h_k, sub_dl, rng)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(h)
    best_h = h.copy()

    # Phase 1: Active set LP optimization (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _active_set_opt(h, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smooth descent with momentum (15%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _smooth_descent(best_h, d2, lr_init=0.03)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Logit evolution (25%)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _logit_evolution(best_h, d3, rng, pop_size=10)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution search (15%)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = _multi_res_search(best_h, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Kick and reoptimize (5%)
    d5 = t0 + budget_s * 0.88
    h5, c5_5 = _kick_reopt(best_h, d5, rng, n_kicks=15)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final pair refinement and active set polish (12%)
    if time.time() < deadline - 20:
        d6 = min(deadline - 15.0, t0 + budget_s * 0.94)
        h6, c5_6 = _pair_refine(best_h, d6, max_passes=100, rng=rng)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    if time.time() < deadline - 5:
        h7, c5_7 = _active_set_opt(best_h, deadline, rng)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final feasibility and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 1.0:
        best_h = _project(h0, n0 / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
