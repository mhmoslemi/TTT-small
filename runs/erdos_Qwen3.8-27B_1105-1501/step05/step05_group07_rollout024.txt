import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_LQ(i, j, h, n):
    d = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1
    L = np.zeros(n_lags)
    Q = np.zeros(n_lags)
    idx = i + d
    valid = (idx >= 0) & (idx < n)
    L[valid] += (1.0 - h[idx[valid]])
    idx = i - d
    valid = (idx >= 0) & (idx < n)
    L[valid] -= h[idx[valid]]
    idx = j + d
    valid = (idx >= 0) & (idx < n)
    L[valid] -= (1.0 - h[idx[valid]])
    idx = j - d
    valid = (idx >= 0) & (idx < n)
    L[valid] += h[idx[valid]]
    r1 = (j - i) + (n - 1)
    if 0 <= r1 < n_lags:
        Q[r1] = 1.0
    r2 = (i - j) + (n - 1)
    if 0 <= r2 < n_lags:
        Q[r2] = 1.0
    return L, Q

def _eval_max(S0, L, Q, dx, delta):
    return np.max(S0 + dx * (L * delta + Q * delta * delta))

def _golden_search(S0, L, Q, dx, lo, hi, n_iter=28):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _eval_max(S0, L, Q, dx, c1)
    f2 = _eval_max(S0, L, Q, dx, c2)
    for _ in range(n_iter):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _eval_max(S0, L, Q, dx, c1)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _eval_max(S0, L, Q, dx, c2)
    delta = (lo + hi) * 0.5
    return delta, _eval_max(S0, L, Q, dx, delta)

def _two_coord_optimize(h, deadline, max_sweeps=20000, min_improvement=1e-14, stagnation_limit=500):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    stagnant = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        S0 = _scores(h)
        cur_c5 = float(np.max(S0))
        best_imp = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        row = np.arange(n)
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                lo = -h[i]
                hi = 1.0 - h[j]
                if hi - lo < 1e-12:
                    continue
                L, Q = _pair_LQ(i, j, h, n)
                delta, new_c5 = _golden_search(S0, L, Q, dx, lo, hi)
                imp = cur_c5 - new_c5
                if imp > best_imp:
                    best_imp = imp
                    best_i = i
                    best_j = j
                    best_delta = delta
        if best_i < 0 or best_imp < min_improvement:
            stagnant += 1
            if stagnant > stagnation_limit:
                break
            continue
        stagnant = 0
        h[best_i] += best_delta
        h[best_j] -= best_delta
        h[best_i] = min(max(h[best_i], 0.0), 1.0)
        h[best_j] = min(max(h[best_j], 0.0), 1.0)
        new_c5 = _exact_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=3000, stall_limit=100):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        n_lags = len(scores)
        G = np.zeros((n_lags, n))
        d_arr = np.arange(-(n - 1), n)
        for r in range(n_lags):
            d = d_arr[r]
            if d >= 0:
                m = n - d
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:d + m] += -h[:m]
            else:
                g = -d
                m = n - g
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
        G = G * dx
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []
        for j in range(n):
            bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.8
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.8
            continue
        if float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.999
        alphas = np.linspace(0, alpha_max, 15)
        vals = np.array([_exact_c5(h + a * p) for a in alphas])
        best_a = alphas[int(np.argmin(vals))]
        new_c5 = float(np.min(vals))
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + best_a * p, n / 2.0)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.85
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _two_coord_multi_restart(h_best, c5_best, deadline, rng, n_restarts=20, scale_init=0.03):
    n = len(h_best)
    best_h = h_best.copy()
    best_c5 = c5_best
    scale = scale_init
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        sub_dl = min(deadline, time.time() + 8.0)
        ht, ct = _two_coord_optimize(h_pert, sub_dl, max_sweeps=5000, min_improvement=1e-14, stagnation_limit=200)
        if ct < best_c5 - 1e-13:
            best_c5 = ct
            best_h = ht.copy()
            scale = max(0.004, scale * 0.75)
        else:
            scale = min(0.08, scale * 1.2)
    return best_h, best_c5

def _two_coord_different_n(h_cur, c5_cur, deadline, rng):
    best_h = h_cur.copy()
    best_c5 = c5_cur
    best_n = len(h_cur)
    for n_try in [96, 160, 192]:
        if time.time() > deadline:
            break
        h_init = _resample(h_cur, n_try)
        sub_dl = min(deadline, time.time() + 25.0)
        ht, ct = _two_coord_optimize(h_init, sub_dl, max_sweeps=8000, min_improvement=1e-14, stagnation_limit=300)
        if ct < best_c5 - 1e-10:
            best_c5 = ct
            best_h = ht.copy()
            best_n = n_try
        if time.time() > deadline:
            break
        alpha_list = [0.6180339887, 0.4142135624, 0.7071067812]
        for alpha in alpha_list:
            if time.time() > deadline:
                break
            h_qp = _quasi_periodic_init(n_try, alpha)
            h_qp = h_qp + rng.normal(0, 0.03, n_try)
            h_qp = _project(h_qp, n_try / 2.0)
            sub_dl2 = min(deadline, time.time() + 15.0)
            ht2, ct2 = _two_coord_optimize(h_qp, sub_dl2, max_sweeps=5000, min_improvement=1e-14, stagnation_limit=200)
            if ct2 < best_c5 - 1e-10:
                best_c5 = ct2
                best_h = ht2.copy()
                best_n = n_try
    return best_h, best_c5, best_n
