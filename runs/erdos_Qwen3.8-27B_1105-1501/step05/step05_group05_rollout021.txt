def _local_search(h, deadline, t_max=0.5, frac_pairs=0.3):
    """Pair-transfer local search with gradient-guided pair selection"""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    while time.time() < deadline:
        corr0 = np.correlate(h, 1.0 - h, mode='full')
        G = _compute_grad_matrix(h)
        
        # Compute subgradient (average of active lags)
        s_max = float(np.max(corr0))
        active = corr0 >= s_max * 0.999
        s = G[active, :].mean(axis=0)
        
        # Select candidate pairs: high s to low s
        n_candidates = max(5, int(n * frac_pairs))
        top_idx = np.argsort(s)[::-1][:n_candidates]
        bot_idx = np.argsort(s)[:n_candidates]
        
        best_improvement = 0.0
        best_pair = None
        best_t = 0.0
        
        for i in top_idx:
            for j in bot_idx:
                if i == j:
                    continue
                # Only try if there's room for transfer
                if h[i] < 1e-10 or h[j] > 1.0 - 1e-10:
                    continue
                t, obj = _pair_transfer_optimize(h, i, j, G, corr0, dx)
                if t != 0.0:
                    improvement = _exact_c5(h) - obj
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_pair = (i, j)
                        best_t = t
        
        if best_improvement > 1e-12 and best_pair is not None:
            i, j = best_pair
            h[i] -= best_t
            h[j] += best_t
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5
