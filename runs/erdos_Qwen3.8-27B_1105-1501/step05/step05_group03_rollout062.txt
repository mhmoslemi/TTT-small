def _weighted_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    # weights has length 2n-1, indexed from -(n-1) to (n-1)
    # g[j] = dx * sum_k weights[k] * [(1-h[j+k]) * 1_{valid} + (-h[j-k]) * 1_{valid}]
    g = np.zeros(n)
    # Part 1: sum_k w[k] * (1-h[j+k]) for j+k in [0,n-1]
    one_minus_h = 1.0 - h
    # Part 2: -sum_k w[k] * h[j-k] for j-k in [0,n-1]
    # Note: sum_k w[k] * h[j-k] = sum_j' w[j-j'] * h[j'] where j'=j-k
    
    # Let me use a different approach: for each k, add w[k] * (1-h) shifted by k to g
    # and w[k] * (-h) shifted by -k to g
    for k in range(-(n-1), n):
        wk = weights[k + n - 1]
        if abs(wk) < 1e-15:
            continue
        # (1-h[j+k]) valid when j+k in [0,n-1], i.e., j in [-k, n-1-k]
        if k >= 0:

            g[:n-k] += wk * one_minus_h[k:]

    target = n / 2.0

    scores = corr * dx

    w = np.exp(beta * (scores - max_s))
    w /= np.sum(w)
    
    # Compute weighted gradient
    g = np.zeros(n)
    # w is indexed from -(n-1) to (n-1), stored in w[0] to w[2n-2]
    for k in range(-(n-1), n):

        if wk < 1e-15:

    g -= g.mean()
    
    # Step
    v = h - step * g
    h_new = _threshold_project(v, target)

import numpy as np

def _threshold_project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = n - 1 - m
        if k >= 0:
            mm = n - k
            if mm > 0:
                G[m, :mm] += (1.0 - h[k:])
                G[m, k:] += -h[:mm]
        else:
            kk = -k
            mm = n - kk
            G[m, kk:] += (1.0 - h[:mm])
            G[m, :mm] += -h[kk:]
    return G * dx

def _coord_optimize(h, n_cycles=5, deadline=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        scores = _scores_arr(h)
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                break
            a = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                k = n - 1 - m
                val = 0.0
                jp = j + k
                if 0 <= jp < n:
                    val += (1.0 - h[jp])
                jm = j - k
                if 0 <= jm < n:
                    val -= h[jm]
                a[m] = val
            base = scores - a * h[j]
            best_t = h[j]
            best_val = float(np.max(base + a * h[j]))
            g0 = float(np.max(base))
            g1 = float(np.max(base + a))
            if g0 < best_val - 1e-15:
                best_val = g0
                best_t = 0.0
            if g1 < best_val - 1e-15:
                best_val = g1
                best_t = 1.0
            k_star = int(np.argmax(base + a * h[j]))
            for l in range(2 * n - 1):
                if l == k_star:
                    continue
                da = a[k_star] - a[l]
                if abs(da) < 1e-15:
                    continue
                t_try = (base[l] - base[k_star]) / da
                if 0.0 < t_try < 1.0:
                    g_try = float(np.max(base + a * t_try))
                    if g_try < best_val - 1e-15:
                        best_val = g_try
                        best_t = t_try
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                g_try = float(np.max(base + a * frac))
                if g_try < best_val - 1e-15:
                    best_val = g_try
                    best_t = frac
            if abs(best_t - h[j]) > 1e-10:

                scores = _scores_arr(h)
                improved = True

def _subgradient_optimize(h, deadline, beta_start=10.0, beta_end=300.0, step_init=0.1, max_iters=100000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    mom = np.zeros(n)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, 50000))
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores_arr(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        g = np.zeros(n)
        for m in range(2 * n - 1):
            wk = w[m]
            if wk < 1e-15:
                continue
            k = n - 1 - m
            if k >= 0:
                mm = n - k
                if mm > 0:
                    g[:mm] += wk * (1.0 - h[k:])
                    g[k:] += (-wk) * h[:mm]
            else:
                kk = -k
                mm = n - kk
                g[kk:] += wk * (1.0 - h[:mm])
                g[:mm] += (-wk) * h[kk:]
        g *= (2.0 / n)
        g -= g.mean()
        mom = 0.85 * mom - step * g
        h_new = _threshold_project(h + mom, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.005, 0.35)
            stall = 0
        else:
            mom = 0.5 * mom
            step *= 0.93
            stall += 1
            if stall > 300:
                break
    return best_h, best_c5

def _lp_direction(h, radius):
    n = len(h)
    scores = _scores_arr(h)
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, max_evals=25):
    n = len(h)
    target = n / 2.0
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_threshold_project(h + c1 * p, target))
    f2 = _exact_c5(_threshold_project(h + c2 * p, target))
    for _ in range(max_evals):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_threshold_project(h + c1 * p, target))
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_threshold_project(h + c2 * p, target))
    best_a = c1 if f1 < f2 else c2
    best_f = min(f1, f2)
    return best_a, best_f

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=3000, radius_min=0.002):
    n = len(h)

def _pair_polish(h, deadline, n_passes=3):

                if a_max < 1e-10:

                    h2[i] += sign * m2

                    else:
                        lo_a = m1

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _threshold_project(h_new, n_new / 2.0)

def _qp_init(n, alpha, noise=0.0, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    if noise > 0 and rng is not None:
        h += rng.normal(0, noise, n)
    return _threshold_project(h, n / 2.0)

def _multi_alpha_init(n, rng):
    alphas = [0.6180339887, 0.3819660113, 0.7071067812, 0.4142135624, 0.5854101966]
    h = np.zeros(n)
    for a in alphas:
        x = np.arange(n) / n
        h += ((x * a) % 1.0) < 0.5
    h = h / len(alphas)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
    return _threshold_project(h, n / 2.0)
