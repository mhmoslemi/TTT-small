def _bit_scores(bits_int, n):
    """Compute all lag scores for a binary vector represented as a Python int."""
    scores = []
    for d in range(1, n):
        mask = (1 << (n - d)) - 1
        lower = bits_int & mask
        upper = (bits_int >> d) & mask
        # Count positions where lower=1 and upper=0 (positive direction)
        pos_count = (lower & (~upper) & mask).bit_count()
        # Count positions where upper=1 and lower=0 (negative direction)  
        neg_count = (upper & (~lower) & mask).bit_count()
        scores.append(max(pos_count, neg_count))
    return scores

from scipy.optimize import linprog, milp, LinearConstraint, Bounds
from scipy import sparse
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _bit_score(bits, n):
    """Compute max lag count for binary vector as Python integer.
    bits: Python int where bit i corresponds to position i.
    Returns the maximum count over all lags (one direction max with the other)."""
    max_count = 0
    for g in range(1, n):
        mask = (1 << (n - g)) - 1
        A = bits & mask
        B = (bits >> g) & mask
        pos = (A & (~B) & mask).bit_count()  # h[i]=1, h[i+g]=0
        neg = ((~A) & B & mask).bit_count()  # h[i]=0, h[i+g]=1
        c = pos if pos > neg else neg
        if c > max_count:
            max_count = c
    return max_count

def _bits_to_array(bits, n):
    """Convert Python int bits to numpy array."""

def _golden_init(n, alpha=None):
    if alpha is None:
        alpha = (math.sqrt(5) - 1) / 2
    target = n // 2
    h = np.zeros(n)
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    h[vals < 0.5] = 1.0
    # Adjust to have exactly target ones
    ones = np.sum(h)
    if ones > target:
        # Remove excess ones (those closest to threshold 0.5)
        one_idx = np.where(h == 1)[0]
        one_vals = vals[one_idx]
        # Remove the ones with values closest to 0.5
        order = np.argsort(np.abs(one_vals - 0.5))
        n_remove = int(ones - target)
        h[one_idx[order[:n_remove]]] = 0
    elif ones < target:
        # Add missing ones
        zero_idx = np.where(h == 0)[0]
        zero_vals = vals[zero_idx]
        order = np.argsort(np.abs(zero_vals - 0.5))
        n_add = int(target - ones)
        h[zero_idx[order[:n_add]]] = 1
    return h

    idx = rng.permutation(n)

                cur_count = new_count
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

```python
def _grad_row(h, d, n):
    """Compute the gradient of L_d(h) = dx * sum h[i]*(1-h[i+d]) with respect to h.
    Returns a vector of length n."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        for i in range(m):
            grad[i] += dx * (1 - h[i+d])
            grad[i+d] += dx * (-h[i])
    else:
        g = -d
        m = n - g
        for i in range(m):
            grad[i+g] += dx * (1 - h[i])
            grad[i] += dx * (-h[i+g])
    return grad
```

Wait, this is O(n) per lag, and we have O(n) lags, so total O(n²). That's the same as the direct correlation. For the active set approach with K lags, it's O(K*n).

Actually, let me vectorize this:

```python
def _grad_rows(h, active_lags, n):
    """Compute gradient rows for active lags. Returns shape (len(active_lags), n)."""
    dx = 2.0 / n
    G = np.zeros((len(active_lags), n))
    for idx, d in enumerate(active_lags):
        if d >= 0:
            m = n - d
            G[idx, :m] += (1 - h[d:])
            G[idx, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            G[idx, g:] += (1 - h[:m])
            G[idx, :m] += -h[g:]
    G *= dx
    return G
```

This is O(K*n) total. For K=10, n=512: 5120 operations. Very fast.
