    #   d/dh[j] = (1-h[j-g]) for j >= g, minus h[j+g] for j < n-g

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v.astype(float), -10.0, 10.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(corr))
    exps = np.exp((corr - M) / tau)
    total = np.sum(exps)
    soft_val = (M + tau * np.log(total)) * dx
    weights = exps / total
    grad = np.zeros(n)
    w0 = weights[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[n - 1 + d]
        wn = weights[n - 1 - d]
        m = n - d
        if wp > 1e-15:
            grad[:m] += wp * (1.0 - h[d:d + m])
            grad[d:] -= wp * h[:m]
        if wn > 1e-15:
            grad[d:] += wn * (1.0 - h[:m])
            grad[:m] -= wn * h[d:d + m]
    grad *= dx
    return soft_val, grad

def _lbfgs_smooth(h0, tau, time_limit, max_iter=300):
    t_start = time.time()
    n = len(h0)
    state = {'n_eval': 0, 'done': False}

    def obj(x):
        state['n_eval'] += 1
        if time.time() > time_limit:
            state['done'] = True
            return 1e10
        val, _ = _soft_obj_and_grad(x, tau)
        return val

    def grad(x):
        state['n_eval'] += 1
        if time.time() > time_limit:
            state['done'] = True
            return np.zeros(n)
        _, g = _soft_obj_and_grad(x, tau)
        return g

    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'gtol': 1e-7, 'maxls': 30})
        h_res = _project(res.x, n / 2.0)
    except Exception:
        return h0.copy(), _exact_c5(h0)
    return h_res, _exact_c5(h_res)

def _line_search_pair(h, i, j, dx, max_radius=0.5):
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0, -max_radius)
    t_hi = min(h[j], 1.0 - h[i], max_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)

    def eval_t(t):
        hi_val = h[i] + t
        hj_val = h[j] - t
        if hi_val < -1e-12 or hi_val > 1 + 1e-12 or hj_val < -1e-12 or hj_val > 1 + 1e-12:
            return 1e10
        h_new = h.copy()
        h_new[i] = hi_val
        h_new[j] = hj_val

        else:
            t_lo, c1, f1 = c1, c2, f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = eval_t(c2)

def _block_swap_eval(h, a, b, k):

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        best_c5 = c5_5
        best_h = h5.copy()
    # Block swaps
    sub_dl = min(d5, time.time() + budget_s * 0.08)
    h5 = best_h.copy()
    h5, c5_5 = _block_swap_loop(h5, sub_dl, max_passes=500)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # L-BFGS from refined
    sub_dl = min(d5, time.time() + budget_s * 0.03)
    h5 = best_h.copy()
    for tau in [0.04, 0.02, 0.01]:
        if time.time() > sub_dl:
            break
        h5, c5_5 = _lbfgs_smooth(h5, tau, sub_dl, max_iter=150)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Pair swaps
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h5 = best_h.copy()
    h5, c5_5 = _grad_pair_loop(h5, sub_dl, nsamples=500)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Another LP pass
    sub_dl = min(d5, time.time() + budget_s * 0.02)
    h5 = best_h.copy()
    h5, c5_5 = _lp_loop(h5, sub_dl, radius_init=0.04)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Basin hopping with diverse perturbations (10%)
    d6 = t0 + budget_s * 0.93
    scale = 0.03
    for attempt in range(300):
        if time.time() > d6:
            break
        pert_type = attempt % 3
        if pert_type == 0:
            h_p = best_h + rng.normal(0, scale, len(best_h))
        elif pert_type == 1:
            # Block perturbation
            h_p = best_h.copy()
            k = int(rng.integers(3, min(20, len(best_h) // 4)))
            a = int(rng.integers(0, len(best_h) - k))
            h_p[a:a+k] = rng.uniform(0, 1, k)
        else:
            # Swap-based perturbation
            h_p = best_h.copy()
            n_swaps = int(rng.integers(2, 6))
            for _ in range(n_swaps):
                i = int(rng.integers(0, len(h_p)))
                j = int(rng.integers(0, len(h_p)))
                if i != j:
                    h_p[i], h_p[j] = h_p[j], h_p[i]
        h_p = _project(h_p, len(h_p) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.008)
        h_p, c5_p = _lp_loop(h_p, sub_dl, radius_init=0.05)
        rem = d6 - time.time()
        if rem > 2:
            sub_dl2 = min(d6, time.time() + max(1, budget_s * 0.003))
            h_p, c5_p = _block_swap_loop(h_p, sub_dl2, max_passes=50)
        if c5_p < best_c5 - 1e-12:
            best_c5 = c5_p
            best_h = h_p.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.08)

    # Phase 7: Final polish (7%)
    d7 = deadline
    h7 = best_h.copy()
    sub_dl = min(d7, time.time() + budget_s * 0.03)
    h7, c5_7 = _lp_loop(h7, sub_dl, radius_init=0.03)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h7 = best_h.copy()
    sub_dl = min(d7, time.time() + budget_s * 0.02)
    h7, c5_7 = _block_swap_loop(h7, sub_dl, max_passes=200)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h7 = best_h.copy()
    sub_dl = min(d7, time.time() + budget_s * 0.01)
    h7, c5_7 = _grad_pair_loop(h7, sub_dl, nsamples=200)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final projection and validation
    best_h = _project(best_h, len(best_h
                 ) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
