    return obj, grad

                a *= 0.65
                if a < 1e-12:
                    break
            if found:
                lr = min(lr * 1.03, 0.15)
            else:
                lr *= 0.7
                if lr < 1e-10:
                    break
    return best_h, best_c5

def _build_G_full(h):
    return _build_G(h)

def _lp_direction(h, G, radius, top_k=None):
    from scipy.optimize import linprog
    n = len(h)
    n_lags = 2 * n - 1
    scores = _scores(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_exact(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    return opt_a, _exact_c5(h + opt_a * p)

def _local_lp(h, deadline, top_k_list=None, radius_init=0.1, max_iters=3000):
    if top_k_list is None:
        top_k_list = [None, 12, 8, 5, 3]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_tk = len(top_k_list)
    ti = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        tk = top_k_list[ti % n_tk]
        G = _build_G(h)
        p = _lp_direction(h, G, radius, top_k=tk)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search_exact(h, p)
        h_cand = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_cand)
        if actual_c5 < cur_c5 - 1e-13:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 15 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=150, thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _smooth_variant(h, kernel_width):
    n = len(h)
    k = max(1, int(kernel_width))
    kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / k) ** 2)
    kernel /= kernel.sum()
    h_smooth = np.convolve(h, kernel, mode='same')
    return _project(h_smooth, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    # Get parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: p-norm continuation on parent (SLSQP)
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _continue_pnorm(h_start, dl1, q_list=[4, 8, 16, 32], method='slsqp')
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: trust-constr with higher q
    if time.time() < t0 + budget_s * 0.20:
        dl2 = t0 + budget_s * 0.22
        h2, c5_2 = _continue_pnorm(best_h, dl2, q_list=[16, 32, 64, 128], method='trust-constr')
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Softmax gradient descent
    if time.time() < t0 + budget_s * 0.30:
        dl3 = t0 + budget_s * 0.32
        h3, c5_3 = _softmax_grad_step(best_h, dl3, beta_list=[10, 30, 80, 200], max_iters=400)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: LP-based local optimization
    if time.time() < t0 + budget_s * 0.42:
        dl4 = t0 + budget_s * 0.44
        h4, c5_4 = _local_lp(best_h, dl4, top_k_list=[None, 10, 6, 4, 3],
                             radius_init=0.1, max_iters=2000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.65:
        dl5 = t0 + budget_s * 0.67
        resolutions = [192, 256, 320, 160, n0]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.5, 0.3819660113]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl5:
                break
            if best_c5 < 0.3805:
                break
            for s in range(2):
                if time.time() > dl5:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    alpha = alphas[ri % len(alphas)]
                    h_init = _quasi_init(n_try, alpha, rng)
                sub_dl = min(dl5, time.time() + budget_s * 0.035)
                # p-norm
                h_try, c5_try = _continue_pnorm(h_init, sub_dl,
                                                 q_list=[8, 16, 32, 64], method='slsqp')
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                # Fallback: softmax
                if time.time() < dl5:
                    sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
                    h_sf, c5_sf = _softmax_grad_step(h_try, sub_dl2,
                                                      beta_list=[30, 80, 200], max_iters=200)
                    if c5_sf < best_c5:
                        best_c5 = c5_sf
                        best_h = h_sf.copy()
                        best_n = n_try

    # Phase 6: Smoothed variants of best
    if time.time() < t0 + budget_s * 0.75:
        dl6 = t0 + budget_s * 0.76
        for kw in [2, 3, 5]:
            if time.time() > dl6:
                break
            h_sm = _smooth_variant(best_h, kw)
            sub_dl = min(dl6, time.time() + budget_s * 0.02)
            h_r, c5_r = _continue_pnorm(h_sm, sub_dl, q_list=[8, 16, 32], method='slsqp')
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 7: Pair descent polish
    if time.time() < t0 + budget_s * 0.85:
        dl7 = t0 + budget_s * 0.87
        best_h, best_c5 = _pair_descent(best_h, dl7, max_passes=200, thresh=1e-6)
        # Follow with tiny LP
        if time.time() < dl7:
            sub_dl = min(dl7, time.time() + budget_s * 0.02)
            h_lp, c5_lp = _local_lp(best_h, sub_dl, top_k_list=[5, 3],
                                     radius_init=0.03, max_iters=800)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 8: Perturbation restarts
    if time.time() < t0 + budget_s * 0.93:
        dl8 = t0 + budget_s * 0.94
        scale = 0.02
        for attempt in range(300):
            if time.time() > dl8:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl8, time.time() + budget_s * 0.006)
            h_try, c5_try = _local_lp(h_pert, sub_dl, top_k_list=[4, 3],
                                       radius_init=0.04, max_iters=300)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.004, scale * 0.8)
            else:
                scale = min(0.06, scale * 1.08)

    # Phase 9: Final micro-polish
    if time.time() < hard_deadline:
        dl9 = hard_deadline
        h_pd, c5_pd = _pair_descent(best_h, dl9, max_passes=100, thresh=1e-9)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
        if time.time() < dl9 - 3:
            h_fin, c5_fin = _local_lp(best_h, dl9, top_k_list=[3],
                                       radius_init=0.005, max_iters=200)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final projection and verification
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
