Let me also think about what the structure of an optimal solution looks like. For the Erdős minimum overlap problem, the optimal configurations are often (but not always) binary (0/1) sequences with a specific pattern. The minimum overlap is related to the maximum cross-correlation between the set and its complement.

For a binary sequence with n/2 ones and n/2 zeros, the overlap at lag g is:

This counts the number of 1s in position i where position i+g is 0. This is the "inversion count" at lag g.

The goal is to minimize the maximum inversion count over all lags.

This is related to sequencing problems and has connections to difference sets and Golomb rulers.

import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum x_i = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_rows(h):
    """Return the full correlation array (without dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _active_lags(h, threshold=0.001):
    """Find lags where correlation is within threshold of the max."""
    corr = _corr_rows(h)
    mx = corr.max()
    return np.where(corr >= mx - threshold)[0]

def _signed_energy(h):
    """Compute the signed energy vector U_g for g=0..n-1."""
    n = len(h)
    b = 2.0 * h - 1.0
    # B[k] = sum_{s=0}^{k-1} b[s], B[0]=0
    B = np.zeros(n + 1)
    for i in range(n):
        B[i+1] = B[i] + b[i]
    # A_g = sum_{i=0}^{n-g-1} b[i]*b[i+g]
    b_corr = np.correlate(b, b, mode='full')
    U = np.zeros(n)
    for g in range(n):
        M = n - g
        E = B[g] + B[n-g] - B[n]
        A = b_corr[n-1+g]
        U[g] = M + abs(E) - A
    return U / (2.0 * n)

    E_vec = B[n - g] + B[g] - B[n]
    A_vec = b_corr[n - 1 + g]
    U = (M + np.abs(E_vec) - A_vec) / (2.0 * n)
    return U

def _active_set(U, tol=0.002):
    mx = U.max()
    return np.where(U >= mx - tol)[0]

def _signed_gradients(h, active_g):
    """Compute gradient of U_g w.r.t. b for each active lag g."""
    n = len(h)
    b = 2.0 * h - 1.0
    B = np.zeros(n + 1)
    np.cumsum(b, out=B[1:])
    b_corr = np.correlate(b, b, mode='full')
    
    m = len(active_g)
    grads = np.zeros((m, n))
    for idx, g in enumerate(active_g):
        E = B[n-g] + B[g] - B[n]
        sg = 1.0 if E >= 0 else -1.0
        grad = np.zeros(n)
        # ∂U_g/∂b[i] = sg*(1_{i<g} + 1_{i<n-g} - 1) - b[i+g]*1_{i<n-g} - b[i-g]*1_{i>=g}
        if g == 0:
            # ∂U_0/∂b[i] = sg*(1_{i<0}+1_{i<n}-1) - b[i]*1_{i<n} - b[i]*1_{i>=0}
            # = sg*(0+1-1) - b[i] - b[i] = -2*b[i]
            # (since E_0 = B[n]+B[0]-B[n] = 0, so sg is ambiguous; but |0|=0, derivative is 0)
            # Actually for g=0, E_0 = 0 always, so the |E| term has 0 gradient (subgradient)
            # U_0 = n + 0 - A_0 = n - sum b[i]^2
            # ∂U_0/∂b[i] = -2*b[i]
            grad[:] = -2.0 * b
        else:
            # Sign term: sg * (1_{i<g} + 1_{i<n-g} - 1)
            sign_part = np.zeros(n)
            sign_part[:g] += sg       # 1_{i<g}
            sign_part[:n-g] += sg     # 1_{i<n-g}
            sign_part[:] -= sg        # -1
            # A_g terms: -b[i+g] for i<n-g, -b[i-g] for i>=g
            a_part = np.zeros(n)
            a_part[:n-g] = -b[g:]     # b[i+g] for i=0..n-g-1
            a_part[g:] += -b[:n-g]    # b[i-g] for i=g..n-1
            grad = sign_part + a_part
        grads[idx] = grad
    return grads

def _lp_direction(h, active_g, radius):
    """Find direction d in b-space minimizing max directional derivative on active lags."""
    n = len(h)
    grads = _signed_gradients(h, active_g)
    m = len(active_g)
    # min t s.t. grads[k]·d <= t for all k, sum(d)=0, -radius<=d<=radius
    # Variables: d[0..n-1], t
    # Objective: minimize t → c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [( -radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_exact(h, v, max_alpha):
    """Line search: minimize C5(h + alpha*v) for alpha in [0, max_alpha]."""
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = np.correlate(v, v, mode='full')
    
    def phi(alpha):
        return float(np.max(S0 + alpha * T - alpha * alpha * C)) * dx
    
    # Find feasible range
    pos = v > 1e-15
    neg = v < -1e-15
    t_hi = max_alpha
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / v[pos])))
    t_lo = 0.0
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-v[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, phi(0.0)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = t_lo
    b = t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = phi(c1)
    f2 = phi(c2)
    for _ in range(60):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = phi(c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = phi(c2)
    opt = (a + b) / 2.0
    val = phi(opt)
    return opt, val

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=5000):
    """Main LP-based optimization loop using signed energy gradients."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        U = _signed_U(h)
        active = _active_set(U, tol=0.002)
        d = _lp_direction(h, active, radius)
        if d is None or np.max(np.abs(d)) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        # Convert to h-space direction
        v = d / 2.0
        # Ensure sum(v) = 0 (should be from LP constraint)
        v -= np.mean(v)
        alpha, new_c5 = _line_search_exact(h, v, radius)
        if new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha * v
            h = _project(h_new, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Adapt radius
            U_new = _signed_U(h)
            ratio = cur_c5 / max(U_new.max(), 1e-15)
            if ratio < 0.8:
                radius = min(radius * 1.1, 0.4)
            else:
                radius = min(radius * 1.02, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall > 30 and radius < 0.05:
                # Try a perturbation
                rng = np.random.default_rng(it)
                h = _project(h + rng.normal(0, 0.01, n), target)
                cur_c5 = _exact_c5(h)
                stall = 0
                radius = 0.1
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    """Binary SA with focus on active lags."""
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.005
    T_min = 1e-5
    cooling = 0.9998
    patience = 0
    max_patience = 50000
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min:
                break
            T *= 0.98
            patience = 0
        for _ in range(200):
            if not ones or not zeros:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                cur_c5 = new_c5
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5
