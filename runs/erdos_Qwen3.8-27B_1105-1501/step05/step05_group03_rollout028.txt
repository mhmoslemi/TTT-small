def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def _smoothed_grad_optimize(h0, deadline, beta_start=20, beta_end=400, n_steps=500, step_init=0.04):
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target).copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores_all(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        grad = np.zeros(n)
        for k in range(len(scores)):
            d = int(k - (n - 1))
            if w[k] > 1e-10:
                grad += w[k] * _grad_for_lag(h, d)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.02, 0.10)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
    return _project(h, n / 2.0)

def _binary_init(n, h_template, rng=None):
    if h_template is not None and len(h_template) == n:
        h = (h_template > 0.5).astype(float)
    else:
        x = np.arange(n) / n
        h = ((x * 0.6180339887) % 1.0) < 0.5
        h = h.astype(float)
    cnt = int(np.sum(h))
    target = n // 2
    if cnt > target:
        ones = np.where(h == 1.0)[0]
        np.random.seed(0)
        to_remove = np.random.choice(ones, cnt - target, replace=False)
        h[to_remove] = 0.0
    elif cnt < target:
        zeros = np.where(h == 0.0)[0]
        np.random.seed(0)
        to_add = np.random.choice(zeros, target - cnt, replace=False)
        h[to_add] = 1.0
    return h

def _coordinate_polish(h0, deadline, n_cycles=2):
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target).copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(30):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy(); h1[j] = m1; h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy(); h2[j] = m2; h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy(); h_try[j] = v_opt; h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _line_search_dir(h_start, h_dir, target):
    n = len(h_start)
    alpha_max = 0.0
    for j in range(n):
        diff = h_dir[j] - h_start[j]
        if diff > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h_start[j]) / diff) if alpha_max > 0 else (1.0 - h_start[j]) / diff
        elif diff < -1e-14:
            a_j = h_start[j] / (-diff)
            alpha_max = a_j if alpha_max == 0.0 else min(alpha_max, a_j)
    alpha_max = max(alpha_max, 0.01) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_a, hi_a = 0.0, alpha_max
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    h1 = _project(h_start + c1 * h_dir, target)
    f1 = _exact_c5(h1)
    h2 = _project(h_start + c2 * h_dir, target)
    f2 = _exact_c5(h2)
    for _ in range(25):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2; f2 = f1; c2 = c1
            c1 = hi_a - gr * (hi_a - lo_a)
            h1 = _project(h_start + c1 * h_dir, target)
            f1 = _exact_c5(h1)
        else:
            lo_a = c1; f1 = f2; c1 = c2
            c2 = lo_a + gr * (hi_a - lo_a)
            h2 = _project(h_start + c2 * h_dir, target)
            f2 = _exact_c5(h2)
    opt_a = (lo_a + hi_a) / 2.0
    h_opt = _project(h_start + opt_a * h_dir, target)
    f_opt = _exact_c5(h_opt)
    best_a = c1 if f1 < f2 else c2
    best_f = min(f1, f2)
    if f_opt < best_f:
        return opt_a, f_opt
    best_h = _project(h_start + best_a * h_dir, target)
    return best_a, _exact_c5(best_h)

def _perturb_optimize(h0, deadline, rng, n_attempts=200, scale_init=0.03):
    n = len(h0)
    target = n / 2.0
    best_h = _project(h0, target).copy()
    best_c5 = _exact_c5(best_h)
    scale = scale_init
    for _att in range(n_attempts):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        c5_pert = _exact_c5(h_pert)
        sub_deadline = min(deadline, time.time() + 2.0)
        h_opt, c5_opt = _slsqp_optimize(h_pert, sub_deadline, max_active=32, max_rounds=5, initial_top=16)
        if c5_opt < best_c5 - 1e-13:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.12)
    return best_h, best_c5

def _symmetric_init(h0):
    n = len(h0)
    if n % 2 != 0:
        return h0.copy()
    h_sym = (h0 + h0[::-1]) / 2.0
    return _project(h_sym, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 128
        h0 = _quasi_periodic_init(n0, 0.6180339887, rng)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    if budget_s < 30:
        return best_h, best_c5, best_n

    # Phase 1: SLSQP from initial (main workhorse)
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _slsqp_optimize(h_start, dl1, max_active=56, max_rounds=25, initial_top=28)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothed gradient from best
    if time.time() < t0 + budget_s * 0.30:
        dl2 = t0 + budget_s * 0.32
        h2, c5_2 = _smoothed_grad_optimize(best_h, dl2, beta_start=15, beta_end=350, n_steps=300, step_init=0.035)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        # Follow-up SLSQP
        if time.time() < t0 + budget_s * 0.38:
            dl2b = t0 + budget_s * 0.40
            h2b, c5_2b = _slsqp_optimize(best_h, dl2b, max_active=48, max_rounds=15, initial_top=24)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Multi-resolution
    if time.time() < t0 + budget_s * 0.55:
        dl3 = t0 + budget_s * 0.55
        resolutions = [n0, n0 - 16, n0 + 16, n0 - 32, 96]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl3:
                break
            if n_try < 32:
                continue
            for s in range(2):
                if time.time() > dl3:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    alpha = alphas[ri % len(alphas)]
                    h_init = _quasi_periodic_init(n_try, alpha, rng)
                sub_dl = min(dl3, time.time() + budget_s * 0.035)
                h_try, c5_try = _slsqp_optimize(h_init, sub_dl, max_active=40, max_rounds=12, initial_top=20)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 4: Symmetric variant
    if time.time() < t0 + budget_s * 0.62:
        dl4 = t0 + budget_s * 0.64
        h_sym = _symmetric_init(best_h)
        h4, c5_4 = _slsqp_optimize(h_sym, dl4, max_active=40, max_rounds=10, initial_top=20)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Another SLSQP round with larger active set
    if time.time() < t0 + budget_s * 0.72:
        dl5 = t0 + budget_s * 0.74
        h5, c5_5 = _slsqp_optimize(best_h, dl5, max_active=80, max_rounds=20, initial_top=40)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Perturbation restarts
    if time.time() < t0 + budget_s * 0.85:
        dl6 = t0 + budget_s * 0.85
        h6, c5_6 = _perturb_optimize(best_h, dl6, rng, n_attempts=150, scale_init=0.025)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Coordinate polish
    if time.time() < t0 + budget_s * 0.93:
        dl7 = t0 + budget_s * 0.94
        hc, c5c = _coordinate_polish(best_h, dl7, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 8: Final fine SLSQP
    if time.time() < t0 + budget_s * 0.97:
        dl8 = t0 + budget_s * 0.982
        h_fin, c5_fin = _slsqp_optimize(best_h, dl8, max_active=48, max_rounds=10, initial_top=24)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
