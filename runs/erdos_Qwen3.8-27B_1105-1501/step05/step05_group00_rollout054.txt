    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')

def _balance_direction(h, scores, radius):

    n = len(h)
    m = 2 * n - 1

        return None

    avg_grad -= avg_grad.mean()

def _clip_direction(h, d, radius):

    neg = d < 0

def _block_init(n, rng):

def _periodic_init(n, period, duty):

        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        # Fallback
        try:
            h0_fallback = np.array(initial_h_values, dtype=float)
            h0_fallback = _project(h0_fallback, len(h0_fallback) / 2.0)
            return h0_fallback, _exact_c5(h0_fallback), len(h0_fallback)
        except NameError:
            h_fb = _sine_init(120, 3, 0.0)
            return h_fb, _exact_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
