    # = (sum A^q)^(1/q - 1) * G^T @ (A^(q-1))
    
    # Let's compute:
    A = F_pos / M  # in [0, 1]
    Aq = A ** q
    Sq = Aq.sum()
    if Sq < 1e-300:
        return obj, np.zeros(n)
    grad = (Sq ** (1.0/q - 1.0)) * (G.T @ (A ** (q-1)))
    return obj, grad

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_obj_grad(h, q, G=None):
    n = len(h)
    F = _scores(h)
    F_pos = np.maximum(F, 1e-30)
    M = float(F_pos.max())
    if M < 1e-18:
        return 0.0, np.zeros(n)
    if G is None:
        G = _build_G(h)
    A = F_pos / M
    Aq = A ** q
    Sq = float(Aq.sum())
    if Sq < 1e-300:
        return M, np.zeros(n)
    obj = M * Sq ** (1.0 / q)
    w = A ** (q - 1)
    grad = (Sq ** (1.0 / q - 1.0)) * (G.T @ w)
    return obj, grad

def _slsqp_pnorm(h0, q, target, maxiter=200, deadline=None):
    n = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)

    def objfun(x):
        val, _ = _pnorm_obj_grad(x, q)
        return val

    def gradfun(x):
        _, g = _pnorm_obj_grad(x, q)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target),
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objfun, h0, jac=gradfun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if deadline is not None and time.time() > deadline:
            return None
        if res.x is not None:
            x = _project(res.x, target)
            return x
    except Exception:
        pass
    return None

def _pg_pnorm(h0, q, target, max_iter=300, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_pnorm, _ = _pnorm_obj_grad(h, q)

        for _ in range(40):

def _pnorm_obj_val(h, q):

    if M < 1e-18:

    A = F_pos / M
    Sq = float((A ** q).sum())
    return M * Sq ** (1.0 / q)

                best_h = h_pg.copy()
                h = h_pg

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_refine(h, deadline, radius=0.08, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:

        c_vec[-1] = 1.0
        A_ub = np.zeros((len(s_k), n + 1))

        A_ub[:, n] = -1.0
        b_ub = -s_k
        A_eq = np.zeros((1, n + 1))

        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]

        try:

            if not res.success:

                radius *= 0.7
                continue
            p = res.x[:n]

            alpha_max = max(alpha_max, 1e-10)

def _coord_refine(h, deadline, max_passes=50):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    G = _build_G(h)
    scores = _scores(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-12 or h[j] > 1 - 1e-12:
                continue
            C = G[:, j]
            a_lo = -h[j]
            a_hi = 1.0 - h[j]
            for _ in range(30):
                m1 = a_lo + (a_hi - a_lo) / 3.0
                m2 = a_hi - (a_hi - a_lo) / 3.0
                f1 = float(np.max(scores + m1 * C))
                f2 = float(np.max(scores + m2 * C))
                if f1 < f2:
                    a_hi = m2
                else:
                    a_lo = m1
            alpha = (a_lo + a_hi) / 2.0
            if abs(alpha) < 1e-12:

            h[j] += alpha
            actual_c5 = _exact_c5(h)
            if actual_c5 < best_c5 - 1e-14:

                best_h = h.copy()
                scores = _scores(h)
                G = _build_G(h)
                improved = True
            else:
                h[j] -= alpha
        if not improved:
            break
