from scipy.optimize import linprog, minimize, minimize_scalar
import time

_slq_state = {}

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        s = n - 1 - r
        if s >= 0:
            m = n - s
            if m > 0:
                G[r, s:n] += dx * (1.0 - h[0:m])
                G[r, 0:m] -= dx * h[s:n]
        else:
            a = -s
            m = n - a
            if m > 0:
                G[r, 0:m] += dx * (1.0 - h[a:n])
                G[r, a:n] -= dx * h[0:m]
    return G

def line_search_c5(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-13:
        return 0.0, float(np.max(S0)) * dx
    def _obj(t):
        val = S0 + t * T - t * t * C
        return float(np.max(val)) * dx
    try:
        res = minimize_scalar(_obj, bounds=(t_lo, t_hi), method='bounded',
                              options={'xatol': 1e-12})
        best_t = float(res.x)
        best_val = float(res.fun)
    except Exception:
        grid = np.linspace(t_lo, t_hi, 50)
        vals = [float(np.max(S0 + t * T - t*t * C)) * dx for t in grid]
        bi = int(np.argmin(vals))
        best_t, best_val = grid[bi], vals[bi]
    for t_chk in [t_lo, t_hi]:
        v_chk = _obj(t_chk)
        if v_chk < best_val:
            best_val, best_t = v_chk, t_chk
    return best_t, best_val

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
