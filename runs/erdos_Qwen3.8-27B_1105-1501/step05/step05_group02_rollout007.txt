    n = len(v)
    v = np.asarray(v, dtype=float)

        return np.zeros(n)
    if target >= n:

    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _score_jac(h):
    """Jacobian of scores w.r.t. h. Shape: (2n-1, n)"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    G[n-1, :] = dx * (1.0 - 2.0*h)

        G[n-1+d, :m] += -dx * h[d:]
        G[n-1-d, :m] += dx * (1.0 - h[d:])
        G[n-1-d, d:] += -dx * h[:m]
    return G

    max_s = scores.max()
    log_sum = max_s + np.log(np.sum(np.exp(beta * (scores - max_s))))

    G = _score_jac(h)  # (2n-1, n)
