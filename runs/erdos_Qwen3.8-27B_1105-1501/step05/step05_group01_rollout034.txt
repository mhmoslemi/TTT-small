    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi <= t_lo + 1e-14:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    dd = a + gr * (b - a)
    fc = float(np.max(S0 + c * T - c * c * C)) * dx
    fd = float(np.max(S0 + dd * T - dd * dd * C)) * dx
    for _ in range(45):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = dd; dd = c; fd = fc
            c = b - gr * (b - a)
            fc = float(np.max(S0 + c * T - c * c * C)) * dx
        else:
            a = c; c = dd; fc = fd
            dd = a + gr * (b - a)
            fd = float(np.max(S0 + dd * T - dd * dd * C)) * dx
    return (a + b) / 2.0

def _soft_newton_dir(h, K=5, tau=0.01, reg=1e-3):
    n = len(h)
    dx = 2.0 / n
    S = _scores(h)
    top_idx = np.argsort(S)[-K:]
    S_top = S[top_idx].astype(float)
    S_max = float(np.max(S_top))
    exps = np.exp((S_top - S_max) / tau)
    w = exps / float(np.sum(exps))
    grad = np.zeros(n)
    H = np.zeros((n, n))
    for idx, wi in zip(top_idx, w):
        lag = int(idx) - (n - 1)
        g_r = _grad_lag(h, lag, n)
        grad += wi * g_r
        if lag == 0:
            H[np.diag_indices(n)] += wi * (-2.0)
        elif lag > 0:
            H[:n - lag, lag:] += wi * (-1.0)
            H[lag:, :n - lag] += wi * (-1.0)
        else:
            d2 = -lag
            H[d2:, :n - d2] += wi * (-1.0)
            H[:n - d2, d2:] += wi * (-1.0)
        coef = (wi * wi - wi) / tau
        H += coef * np.outer(g_r, g_r)
    grad *= dx
    H *= dx
    H += reg * np.eye(n)
    grad -= np.sum(grad) / n
    try:
        p = np.linalg.solve(H, -grad)
    except Exception:
        p = -grad
    p -= np.sum(p) / n
    return p

def _newton_loop(h, deadline, K=5, tau_init=0.015):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    tau = tau_init
    stall = 0
    while time.time() < deadline:
        if stall > 35:
            break
        p = _soft_newton_dir(h, K, tau)
        pnorm = np.linalg.norm(p)
        if pnorm < 1e-12:
            S = _scores(h)
            top = int(np.argmax(S))
            lag = top - (n - 1)
            g = _grad_lag(h, lag, n)
            g -= np.sum(g) / n
            p = -g
            pnorm = np.linalg.norm(p)
            if pnorm < 1e-12:
                break
        t_max = 0.4 / (np.max(np.abs(p)) / pnorm)
        t_max = min(t_max, 0.5 * pnorm)
        t = _line_search(h, p, t_max)
        if abs(t) < 1e-14:
            S = _scores(h)
            top = int(np.argmax(S))
            lag = top - (n - 1)
            g = _grad_lag(h, lag, n)
            g -= np.sum(g) / n
            p = -g
            pnorm = np.linalg.norm(p)
            if pnorm > 1e-12:
                t_max = 0.2 * pnorm / (np.max(np.abs(p)) + 1e-15)
                t_max = min(t_max, 0.3)
                t = _line_search(h, p, t_max)
            if abs(t) < 1e-14:
                stall += 1
                tau *= 0.75
                K = min(12, K + (1 if stall % 10 == 0 else 0))
                if tau < 0.003:
                    tau = 0.005
                continue
        h_new = h + t * p
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur - 1e-14:
            h = h_new
            cur = new_c5
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            tau = max(0.003, tau * 0.97)
        else:
            stall += 1
            tau *= 0.85
            if tau < 0.004:
                tau = 0.008
                K = min(12, K + 1)
    return best_h, best_c5

def _pair_polish(h, deadline, n_sweeps=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    for _sw in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 0.01:
                    continue
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                t = _line_search(h, d, 0.45)
                if abs(t) > 1e-14:
                    h_new = h + t * d
                    h_new = _project(h_new, target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < cur - 1e-14:
                        h = h_new
                        cur = c5_new
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _random_pair(h, deadline, n_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    rng = np.random.default_rng(777)
    stall = 0
    while time.time() < deadline and stall < 300:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        t = _line_search(h, d, 0.4)
        if abs(t) > 1e-14:
            h_new = h + t * d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur - 1e-14:
                h = h_new
                cur = c5_new
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_flip(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    rng = np.random.default_rng(555)
    stall = 0
    binary = np.round(h)
    binary = _project(binary, target)
    cur_b = _exact_c5(binary)
    h_b = binary.copy()
    best_b = binary.copy()
    best_c5_b = cur_b
    while time.time() < deadline and stall < 200:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if h_b[i] < 0.5 and h_b[j] > 0.5:
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            t = _line_search(h_b, d, 1.0)
            if abs(t) > 1e-14:
                h_new = h_b + t * d
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_b - 1e-14:
                    h_b = h_new
                    cur_b = c5_new
                    if cur_b < best_c5_b:
                        best_c5_b = cur_b
                        best_b = h_b.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
    if best_c5_b < cur:
        h = best_b.copy()
        cur = best_c5_b
        best_h = best_b.copy()
        best_c5 = best_c5_b
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _perturb_reopt(h, deadline, n_pert=30, scale=0.03):
    n = len(h)
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    rng = np.random.default_rng(4242)
    for _ in range(n_pert):
        if time.time() > deadline:
            break
        h_p = h + rng.normal(0, scale, n)
        h_p = _project(h_p, n / 2.0)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = _newton_loop(h_p, sub_dl, K=5, tau_init=0.01)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.1)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Newton optimization on parent (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _newton_loop(h_init, d1, K=5, tau_init=0.012)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair polish + random pair (10%)
    d2 = t0 + budget_s * 0.20
    pa = (d2 - time.time()) / 2.0
    hp, c5p = _pair_polish(best_h, min(d2, time.time() + pa), n_sweeps=2)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()
    hp, c5p = _random_pair(best_h, d2, n_iters=3000)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Phase 3: Multi-resolution exploration (20%)
    d3 = t0 + budget_s * 0.40
    n_curr = len(best_h)
    candidates_n = []
    for dn in [-12, 12, -24, 24, -36, 36, -48, 48, -60, 60, -8, 8, -16, 16, -20, 20, -28, 28, -40, 40, -56, 56, -72, 72]:
        n_try = n_curr + dn
        if n_try < 60 or n_try > 300:
            continue
        candidates_n.append(n_try)
    tried = set()
    for n_try in candidates_n:
        if time.time() > d3:
            break
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _newton_loop(h_r, sub_dl, K=5, tau_init=0.01)
        if c5_r < best_c5 + 0.002:
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_r, c5_r = _random_pair(h_r, sub_dl2, n_iters=1000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Binary-flip search (10%)
    d4 = t0 + budget_s * 0.50
    hb, c5b = _binary_flip(best_h, d4)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()

    # Phase 5: Intensive local optimization (20%)
    d5 = t0 + budget_s * 0.70
    h5 = best_h.copy()
    sub_dl = min(d5, time.time() + budget_s * 0.07)
    h5, c5_5 = _newton_loop(h5, sub_dl, K=7, tau_init=0.008)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h5 = best_h.copy()
    h5, c5_5 = _pair_polish(h5, sub_dl, n_sweeps=3)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h5 = best_h.copy()
    h5, c5_5 = _random_pair(h5, sub_dl, n_iters=5000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h5 = best_h.copy()
    h5, c5_5 = _newton_loop(h5, sub_dl, K=8, tau_init=0.006)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Perturbation + re-optimize (15%)
    d6 = t0 + budget_s * 0.85
    h6, c5_6 = _perturb_reopt(best_h, d6, n_pert=40, scale=0.025)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Try a few random binary starts (8%)
    d7 = t0 + budget_s * 0.93
    for trial in range(6):
        if time.time() > d7:
            break
        n_try = len(best_h)
        h_r = _random_binary(n_try, rng)
        sub_dl = min(d7, time.time() + (d7 - time.time()) / (6 - trial))
        h_r, c5_r = _newton_loop(h_r, sub_dl, K=5, tau_init=0.02)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Final polish (7%)
    d8 = deadline
    h8 = best_h.copy()
    h8, c5_8 = _newton_loop(h8, min(d8, time.time() + budget_s * 0.025), K=6, tau_init=0.005)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    h8 = best_h.copy()
    h8, c5_8 = _pair_polish(h8, min(d8, time.time() + budget_s * 0.02), n_sweeps=2)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    h8 = best_h.copy()
    h8, c5_8 = _newton_loop(h8, d8, K=6, tau_init=0.004)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
