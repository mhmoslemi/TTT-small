def _slsqp_epigraph(h_init, n, active_lags, t_target, deadline):
    # Variables: [h_0, ..., h_{n-1}, t]
    # x has length n+1
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    h = x[:-1]
    scores = np.correlate(h, 1-h, mode='full') * (2.0/n)
    
    # Constraint: sum(h) = n/2
    def eq_con(x):
        return np.sum(x[:-1]) - n/2.0
    
    def eq_grad(x):
        g = np.ones(n+1)
        g[-1] = 0.0
        return g
    
    # Inequality constraints: t - score_s >= 0 for s in active_lags
    # In scipy: constraint value >= 0 means we're in feasible region
    # type='ineq' means fun >= 0
    
    constraints = [{'type': 'eq', 'fun': eq_con, 'jac': eq_grad}]
    for lag_idx in active_lags:
        def make_ineq(lag):
            def ineq_fun(x):
                h_x = x[:-1]
                t_x = x[-1]
                s = _lag_index_to_shift(lag, n)
                score = _single_lag_score(h_x, s, n)
                return t_x - score
            def ineq_grad(x):
                h_x = x[:-1]
                g = np.zeros(n+1)
                g[-1] = 1.0
                # gradient of -score w.r.t. h
                gs = _single_lag_grad(h_x, s, n)
                g[:-1] = -gs
                return g
            return {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_grad}

```python
def _score_at_lag(h, lg):
    n = len(h)
    if lg >= 0:
        m = n - lg
        if m <= 0:
            return 0.0
        return np.dot(h[lg:], 1.0 - h[:m])
    else:
        a = -lg
        m = n - a
        if m <= 0:
            return 0.0
        return np.dot(h[:m], 1.0 - h[a:])

def _grad_at_lag(h, lg):
    n = len(h)
    g = np.zeros(n)
    if lg >= 0:
        m = n - lg
        if m <= 0:
            return g
        # h[i+lg] * (1-h[i]) for i = 0..m-1
        # d/dh[k]:
        # If k >= lg and k <= n-1: from h[k] = h[i+lg] where i = k-lg, contributes (1-h[k-lg])
        # If k < m: from (1-h[k]) where k = i, contributes -h[k+lg]
        # Condition 1: lg <= k <= n-1 (always true for k in [lg, n-1])
        # Condition 2: k <= m-1 = n-1-lg, so k <= n-1-lg
        g[lg:] += (1.0 - h[:m])  # for k = lg to n-1, h[k][k-lg] pair
        g[:m] += -h[lg:]  # for k = 0 to m-1, (1-h[k]) term
    else:
        a = -lg
        m = n - a
        if m <= 0:
            return g
        # h[i] * (1-h[i+a]) for i = 0..m-1
        # d/dh[k]:
        # If k < m: from h[k], contributes (1-h[k+a])
        # If k >= a and k < n: from (1-h[k]) where i+a = k, i = k-a, contributes -h[k-a]
        g[:m] += (1.0 - h[a:a+m])  # for k = 0 to m-1
        g[a:] += -h[:m]  # for k = a to n-1
    return g
```

def _slqp_solve(h_init, n, lags, t_cap, maxiter=200):
    """Solve the SLSQP epigraph problem."""
    _SLP['n'] = n
    _SLP['target'] = n / 2.0
    _SLP['lags'] = lags
    
    x0 = np.zeros(n + 1)
    x0[:n] = h_init

        {'type': 'ineq', 'fun': _slp_ineq_fun, 'jac': _slp_ineq_jac}

    try:

                       constraints=constraints, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})

            c5_new = _exact_c5(h_new)
            if c5_new < _exact_c5(h_init):

    return None, None

        {'type': 'ineq', 'fun': _s_ineq, 'jac': _s_ineq_j}

        pass
    return None, None
