            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5

def _grad_optimize(h, deadline, beta_init=15, beta_max=500, max_iter=30000,
                   step_init=0.04, mom_beta=0.9):
    n = len(h)
    target = n / 2.0
    x = h.copy()
    mom = np.zeros(n)
    step = step_init
    beta = beta_init
    cur_f = _exact_c5(x)
    best_h = x.copy()
    best_f = cur_f
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _smoothed_grad(x, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        mom = mom_beta * mom - step * g
        x_new = _project(x + mom, target)
        f_new = _exact_c5(x_new)
        if f_new < cur_f - 1e-16:
            x = x_new
            cur_f = f_new
            if cur_f < best_f:
                best_f = cur_f
                best_h = x.copy()
            beta = min(beta * 1.008, beta_max)
            step = min(step * 1.015, 0.12)
            stall = 0
        else:
            mom = 0.5 * mom
            step *= 0.82
            beta = max(beta * 0.97, 5.0)
            stall += 1
            if stall > 150:
                x = _project(best_h + 0.015 * np.random.randn(n), target)
                cur_f = _exact_c5(x)
                mom = np.zeros(n)
                step = step_init
                beta = beta_init
                stall = 0
    return best_h, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _qp_init(n, alpha, threshold=0.5, smooth=0.0, rng=None):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    if smooth > 0:
        h = 1.0 / (1.0 + np.exp(-(threshold - vals) / smooth))
    else:
        h = (vals < threshold).astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h = _project(h + rng.normal(0, 0.02, n), n / 2.0)
    return h

def _binary_swap_sa(h, deadline, n_iters=30000, T_start=0.04, T_min=0.0005):
    n = len(h)
    target = n // 2
    h = h.copy()
    h = (h >= 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1.0)[0]
        order = np.argsort(np.random.rand(n_ones))
        for idx in order[:n_ones - target]:
            h[ones[idx]] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        order = np.argsort(np.random.rand(len(zeros)))
        for idx in order[:target - n_ones]:
            h[zeros[idx]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_start
    cool = (T_min / T_start) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_iter = min(n_iters, max(1, int((deadline - time.time()) / 0.002)))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _coord_descent_polish(h, deadline, n_cycles=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(35):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h, deadline, n_passes=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_size = max(1, n // 20)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, max(1, step_size // 4)):
            if time.time() > deadline:
                break
            for k in range(j + 1, n, max(1, step_size // 4)):
                if time.time() > deadline:
                    break
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(10):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _run_single(n, h_init, deadline, rng):
    """Run a full optimization pipeline for a single resolution."""
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    # Phase 1: Gradient optimization
    if time.time() < deadline:
        dl1 = min(deadline, time.time() + (deadline - time.time()) * 0.35)
        h1, c5_1 = _grad_optimize(best_h, dl1, beta_init=10, beta_max=400, max_iter=15000, step_init=0.035)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    # Phase 2: LP optimization
    if time.time() < deadline:
        dl2 = min(deadline, time.time() + (deadline - time.time()) * 0.35)
        h2, c5_2 = _lp_optimize(best_h, dl2, radius_init=0.12, max_iters=3000, radius_min=0.001)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    # Phase 3: Second gradient pass with different params
    if time.time() < deadline:
        dl3 = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h3, c5_3 = _grad_optimize(best_h, dl3, beta_init=30, beta_max=600, max_iter=10000, step_init=0.02)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    # Phase 4: Final LP
    if time.time() < deadline:
        dl4 = deadline
        h4, c5_4 = _lp_optimize(best_h, dl4, radius_init=0.05, max_iters=2000, radius_min=0.0005)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Optimize from parent at native resolution
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _grad_optimize(h_start, dl1, beta_init=12, beta_max=350, max_iter=20000, step_init=0.035)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    dl1b = t0 + budget_s * 0.28
    h1b, c5_1b = _lp_optimize(best_h, dl1b, radius_init=0.12, max_iters=4000, radius_min=0.001)
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b.copy()

    # Phase 2: Multi-resolution exploration with quasi-periodic inits
    dl2_end = t0 + budget_s * 0.55
    resolutions = [64, 80, 96, 112, 128, 144, 160, 192, 64, 96, 128]
    alphas = [0.6180339887, 0.3819660113, 0.6180339887, 0.6180339887, 0.6180339887,
              0.6180339887, 0.6180339887, 0.6180339887, 0.3819660113, 0.6180339887, 0.6180339887]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl2_end:
            break
        if best_c5 < 0.3806:
            break
        alpha = alphas[ri % len(alphas)]
        # Try resampling and QP init
        inits = []
        inits.append(_resample(best_h, n_try))
        inits.append(_qp_init(n_try, alpha, 0.5, 0.0, rng))
        # Also try slightly different alpha
        inits.append(_qp_init(n_try, alpha + 0.01 * rng.standard_normal(), 0.5, 0.0, rng))
        for h_init in inits:
            if time.time() > dl2_end:
                break
            sub_dl = min(dl2_end, time.time() + budget_s * 0.02)
            h_try, c5_try = _run_single(n_try, h_init, sub_dl, rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 3: Continue optimizing the best
    dl3 = t0 + budget_s * 0.68
    h3, c5_3 = _lp_optimize(best_h, dl3, radius_init=0.08, max_iters=4000, radius_min=0.001)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    dl3b = t0 + budget_s * 0.74
    h3b, c5_3b = _grad_optimize(best_h, dl3b, beta_init=25, beta_max=500, max_iter=15000, step_init=0.025)
    if c5_3b < best_c5:
        best_c5 = c5_3b
        best_h = h3b.copy()

    # Phase 4: Perturbation restarts with mixed LP+gradient
    dl4 = t0 + budget_s * 0.85
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl4:
            break
        h_pert = _project(best_h + rng.normal(0, scale, best_n), best_n / 2.0)
        sub_dl = min(dl4, time.time() + budget_s * 0.008)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.05, max_iters=800, radius_min=0.003)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 5: Binary SA at smaller resolutions for diversity
    if time.time() < t0 + budget_s * 0.90:
        for n_bin in [64, 96, 128]:
            if time.time() > t0 + budget_s * 0.90:
                break
            sub_dl = min(t0 + budget_s * 0.90, time.time() + budget_s * 0.015)
            h_b, c5_b = _binary_swap_sa(best_h if len(best_h) == n_bin else _resample(best_h, n_bin),
                                        sub_dl, n_iters=15000, T_start=0.03, T_min=0.0005)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                # Polish the binary solution
                if time.time() < t0 + budget_s * 0.90:
                    sub_dl2 = min(t0 + budget_s * 0.90, time.time() + budget_s * 0.01)
                    h_bp, c5_bp = _lp_optimize(h_b, sub_dl2, radius_init=0.1, max_iters=1000, radius_min=0.003)
                    if c5_bp < best_c5:
                        best_c5 = c5_bp
                        best_h = h_bp.copy()

    # Phase 6: Coordinate descent polish
    if time.time() < t0 + budget_s * 0.95:
        dl6 = t0 + budget_s * 0.95
        hc, c5c = _coord_descent_polish(best_h, dl6, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.975:
        dl7 = t0 + budget_s * 0.975
        hp, c5p = _pair_polish(best_h, dl7, n_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 8: Final fine LP
    if time.time() < deadline:
        h_fin, c5_fin = _lp_optimize(best_h, deadline, radius_init=0.01, max_iters=2000, radius_min=0.0003)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
