def _grad_rows_all(h):
    """Full gradient matrix for all lags."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))

    if tau_init is None:
        tau_init = 0.03 * cur_c5
    taus = [tau_init, tau_init * 0.5, tau_init * 0.25, tau_init * 0.1, tau_init * 0.05]
    for tau in taus:
        if time.time() > deadline:
            break
        lr = 2.0 / (n * max(dx, 0.01))
        for it in range(300):
            if time.time() > deadline:
                break
            scores = _corr_values(h)
            M = float(np.max(scores))
            exponents = (scores - M) / tau
            exponents = np.clip(exponents, -40, 40)
            exps = np.exp(exponents)
            Z = float(np.sum(exps))
            p = exps / Z
            G = _grad_rows(h, np.arange(2 * n - 1))
            grad = (p @ G) * dx
            h_new = h - lr * grad
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            lr *= 0.85
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = np.full(n0, 0.5)
    
    h0 = _project(np.clip(h0, 0.0, 1.0), n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    # Phase 1: Quick multi-start (12%)
    d1 = t0 + budget_s * 0.12
    starts = [h0.copy()]
    for freq in [1, 2, 3, 5]:
        for ph in [0.0, 1.2, 2.4]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(5):
        starts.append(_project(rng.random(n0), n0 / 2.0))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + 5)
        h_s, c5_s = _lp_optimize(s, sub_dl, radius_init=0.08, top_k=8)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 2: Main LP (28%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _lp_optimize(best_h, d2, radius_init=0.18, top_k=14)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Smooth subgradient (8%)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _smooth_subgrad(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    d3b = t0 + budget_s * 0.52
    h3b, c5_3b = _lp_optimize(best_h, d3b, radius_init=0.1, top_k=10)
    if c5_3b < best_c5:
        best_c5 = c5_3b
        best_h = h3b.copy()
    
    # Phase 4: Binary SA + polish (18%)
    d4 = t0 + budget_s * 0.68
    h4a, c5_4a = _binary_sa(best_h, min(d4, t0 + budget_s * 0.60), rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    h4b, c5_4b = _lp_optimize(best_h, d4, radius_init=0.1, top_k=10)
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()
    
    # Phase 5: Pair descent (12%)
    d5 = t0 + budget_s * 0.80
    h5, c5_5 = _pair_descent(best_h, d5, max_passes=150)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Multi-resolution (12%)
    d6 = t0 + budget_s * 0.92
    best_n = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, 8, -8, 10, -10, 12, -12, 14, -14]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 280 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.002:
            sub_dl = min(d6, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.12, top_k=10)
            if c5_r < best_c5 and time.time() < d6 - 3:
                h_r2, c5_r2 = _pair_descent(h_r, min(d6, time.time() + budget_s * 0.008), max_passes=40)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    
    # Phase 7: Quadruple descent + final LP (8%)
    d7 = t0 + budget_s * 0.95
    h7, c5_7 = _quadruple_descent(best_h, d7)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    
    h_f, c5_f = _lp_optimize(best_h, deadline, radius_init=0.04, top_k=8)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    
    return best_h, final_c5, len(best_h)
