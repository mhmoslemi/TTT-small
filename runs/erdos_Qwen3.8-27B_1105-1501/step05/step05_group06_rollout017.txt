    if opt_val < vals[best_i]:
        return opt_a, opt_val
    return float(alphas[best_i]), float(vals[best_i])

def _pair_descent(h, deadline, max_passes=500, verify_thresh=1e-6, max_pairs_per_pass=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    n_pairs = n * (n - 1) // 2
    cur_pass = 0
    while cur_pass < max_passes:
        if time.time() > deadline:
            break
        pass_improved = False
        pair_idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                pair_idx += 1
                if max_pairs_per_pass is not None and pair_idx > max_pairs_per_pass:
                    break
                if h[i] < 1e-10 and h[j] < 1e-10:
                    continue
                if h[i] > 1 - 1e-10 and h[j] > 1 - 1e-10:
                    continue
                pred_alpha, pred_c5 = _pair_opt_alpha(scores, i, j, h, n)
                if cur_c5 - pred_c5 > verify_thresh and abs(pred_alpha) > 1e-10:
                    h[i] += pred_alpha
                    h[j] -= pred_alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[i] -= pred_alpha
                        h[j] += pred_alpha
            if max_pairs_per_pass is not None and pair_idx > max_pairs_per_pass:
                break
        if not pass_improved:
            break
        cur_pass += 1
    return best_h, best_c5

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    center = n - 1
    G[center, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[center + d, :m] += -(h[d:])
        G[center + d, d:] += (1.0 - h[:m])
        G[center - d, :m] += (1.0 - h[d:])
        G[center - d, d:] += -(h[:m])
    G *= dx
    return G

def _lp_solve(h, G, radius, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    scores = _scores(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    n_grid = 13
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0:
        return 0.0, float(vals[0])
    if best_i == n_grid - 1:
        return alpha_max, float(vals[-1])
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(22):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < float(vals[best_i]):
        return opt_a, opt_c5
    return alphas[best_i], float(vals[best_i])
