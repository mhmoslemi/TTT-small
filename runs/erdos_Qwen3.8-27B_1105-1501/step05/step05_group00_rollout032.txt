```python
def _pair_optimize(C, eff, dx, delta_lo, delta_hi, top_k=20):
    """Find delta that minimizes max_d dx*(C[d] + eff[d]*delta) over [delta_lo, delta_hi].
    Only considers the top_k lags at the current point.
    """
    n_lags = len(C)
    # Get top k lags by current value
    if top_k < n_lags:
        idx = np.argpartition(-C, top_k)[:top_k]
    else:
        idx = np.arange(n_lags)
    
    C_sub = C[idx]
    eff_sub = eff[idx]
    
    # Candidate deltas: boundaries + pairwise crossings
    candidates = [delta_lo, delta_hi]
    
    # Pairwise crossings
    for i in range(len(idx)):
        for j in range(i+1, len(idx)):
            denom = eff_sub[j] - eff_sub[i]
            if abs(denom) > 1e-15:
                delta_star = (C_sub[j] - C_sub[i]) / denom
                if delta_lo <= delta_star <= delta_hi:
                    candidates.append(delta_star)
    
    candidates = np.array(candidates)
    # Evaluate at all candidates
    vals = np.max(C_sub[None, :] + candidates[:, None] * eff_sub[None, :], axis=1) * dx
    best_idx = np.argmin(vals)
    return candidates[best_idx], vals[best_idx]
```

```python
def _project(v, target):
    # Water-filling projection onto {x : 0 <= x_i <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _compute_slopes(h):
    """Compute the gradient matrix: S[d_idx, p] = dC_d/dh[p].
    Shape: (2n-1, n)
    """
    n = len(h)
    S = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d > 0:
            m = n - d
            S[r, :m] += omh[d:d + m]   # 1 - h[p+d] for p=0..m-1
            S[r, d:] -= h[:m]           # -h[p-d] for p=d..n-1
        elif d < 0:
            e = -d
            m = n - e
            S[r, e:] += omh[:m]         # 1 - h[p-e] for p=e..n-1
            S[r, :m] -= h[e:e + m]      # -h[p+e] for p=0..m-1
        else:
            S[r, :] = 1.0 - 2.0 * h
    return S

def _pair_best_delta(C, eff, delta_lo, delta_hi, top_k=30):
    """Find delta in [delta_lo, delta_hi] minimizing max(C + eff*delta) over top_k lags.
    Returns (best_delta, best_value)."""
    n_lags = len(C)
    if top_k < n_lags:
        idx = np.argpartition(-C, top_k)[:top_k]
    else:
        idx = np.arange(n_lags)

    e_s = eff[idx]

    n_k = len(idx)

            denom = e_s[j] - e_s[i]
            if abs(denom) > 1e-14:
                ds = (C_s[j] - C_s[i]) / denom
                if delta_lo - 1e-12 <= ds <= delta_hi + 1e-12:
                    candidates = np.append(candidates, ds)
    
    # Also add the point where the max line has zero slope (if exists)
    zero_mask = np.abs(e_s) < 1e-14
    # Evaluate
    vals = np.max(C_s[None, :] + candidates[:, None] * e_s[None, :], axis=1)
    best_i = int(np.argmin(vals))
    return candidates[best_i], vals[best_i]
```
