    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 2.0, v.max() + 2.0

            lo = mid
        else:

    Q = np.correlate(d, d, mode='full')

    # Dense sampling
    n_samples = 80

    # Find best region
    best_idx = np.argmin(vals)
    
    # Refine around best
    lo = max(0, ts[max(0, best_idx - 3)])
    hi = min(t_max, ts[min(n_samples - 1, best_idx + 3)])
    
    # Golden section on the refined interval
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)

def _binary_greedy(h, deadline, max_passes=200):
    """Full greedy: try all swaps, do the best one, repeat."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for _pass in range(max_passes):
        if time.time() > deadline:

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        best_gain = 1e-15
        best_i, best_j = -1, -1

                h_new[i] = 0.0
                h_new[j] = 1.0

        v2 = (s2 >= 0) & (s2 < n)
        di[v2] += h[s2[v2]]
        s3 = j + s_range
        v3 = (s3 >= 0) & (s3 < n)
        dj = np.zeros(2 * n - 1)
        dj[v3] += 1.0 - h[s3[v3]]
        s4 = j - s_range
        v4 = (s4 >= 0) & (s4 < n)
        dj[v4] -= h[s4[v4]]
        new_C = cur_C + di + dj
        new_max = float(new_C.max())
        new_c5 = new_max * dx
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            h[i] = 0.0
            h[j] = 1.0
            cur_C = new_C
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        T *= T_decay
    return best_h, best_c5

def _line_search(h, d, dx, t_max):
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    if t_max < 1e-14:
        return 0.0, float(S.max()) * dx
    n_samp = 64
    ts = np.linspace(0.0, t_max, n_samp)
    vals = np.array([float(np.max(S + t * T - t * t * Q)) * dx for t in ts])
    best_idx = int(np.argmin(vals))
    lo = max(0.0, ts[max(0, best_idx - 2)])
    hi = min(t_max, ts[min(n_samp - 1, best_idx + 2)])
    if hi - lo < 1e-12:
        return ts[best_idx], vals[best_idx]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(25):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S + opt_t * T - opt_t * opt_t * Q)) * dx
    if vals[best_idx] < opt_val:
        opt_t, opt_val = ts[best_idx], vals[best_idx]
    return opt_t, opt_val

def _continuous_refine(h, deadline, max_iters=3000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    radius = 0.1
    stall = 0
    for _iter in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = np.where(corr > corr.max() - 0.5)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        grad = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            for j in range(n):
                c = 0.0
                if 0 <= j + lag < n:
                    c += (1.0 - h[j + lag])
                if 0 <= j - lag < n:
                    c -= h[j - lag]
                grad[j] += c
        grad /= len(active)
        grad -= np.sum(grad) * h / target
        nrm = np.linalg.norm(grad)
        if nrm < 1e-12:
            stall += 1
            if stall > 20:
                break
            continue
        d = -grad / nrm
        pos = d > 1e-12
        neg = d < -1e-12
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = 0.0
        if np.any(neg):
            t_lo = max(t_lo, -float(np.max(h[neg] / d[neg])))
        if t_hi - t_lo < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, d, dx, t_hi)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _continuous_pair(h, deadline, max_iters=5000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(333)
    radius = 0.15
    stall = 0
    for _iter in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        pos = d > 1e-12
        neg = d < -1e-12
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, (1.0 - h[pos]).max() / d[pos].max())
        t_lo = 0.0
        if np.any(neg):
            t_lo = max(t_lo, -h[neg].max() / d[neg].max())
        t_hi = min(t_hi, 1.0 - h[i])
        t_lo = max(t_lo, -h[j])
        if t_hi - t_lo < 1e-14:
            stall += 1
            continue
        alpha, new_c5 = _line_search(h, d, dx, t_hi)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                radius *= 0.85
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _all_pair_swap(h, deadline, max_passes=10):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                t_hi = min(1.0 - h[i], h[j])
                alpha, c5_try = _line_search(h, d, dx, t_hi)
                if c5_try < cur_c5 - 1e-12:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Binary greedy on initial (15% budget)
    d1 = t0 + budget_s * 0.15
    h_b = _to_binary(h_init)
    h_b, c5_b = _binary_greedy(h_b, d1, max_passes=500)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Phase 2: Multiple structured starts with binary greedy (25%)
    d2 = t0 + budget_s * 0.40
    starts = []
    for alpha_try in [0.6180339887, 0.7071067812, 0.5773502692, 0.5320888863]:
        starts.append(_golden_binary(n0, alpha_try))
    starts.append(_arithmetic_binary(n0, 2, 1))
    starts.append(_arithmetic_binary(n0, 4, 2))
    starts.append(_arithmetic_binary(n0, 3, 1))
    starts.append(_to_binary(h_init))
    per_start = (d2 - time.time()) / max(1, len(starts))
    for h_s in starts:
        if time.time() > d2 - 2:
            break
        sub_dl = min(d2, time.time() + max(2, per_start))
        h_s, c5_s = _binary_greedy(h_s, sub_dl, max_passes=300)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Simulated annealing on best binary (15%)
    d3 = t0 + budget_s * 0.55
    h_sb = _to_binary(best_h)
    h_sb, c5_sb = _binary_sa(h_sb, d3, T_init=0.004, T_decay=0.9997, max_steps=50000)
    if c5_sb < best_c5:
        best_c5 = c5_sb
        best_h = h_sb.copy()

    # Phase 4: Continuous refinement (25%)
    d4 = t0 + budget_s * 0.80
    h_cont = best_h.copy()
    # Subgradient refine
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h_cont, c5_cont = _continuous_refine(h_cont, sub_dl, max_iters=2000)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont.copy()
    # Pair direction optimize
    sub_dl = min(d4, time.time() + budget_s * 0.10)
    h_cont = best_h.copy()
    h_cont, c5_cont = _continuous_pair(h_cont, sub_dl, max_iters=4000)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont.copy()
    # Another subgradient pass
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h_cont = best_h.copy()
    h_cont, c5_cont = _continuous_refine(h_cont, sub_dl, max_iters=1000)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont.copy()

    # Phase 5: Multi-n exploration (10%)
    d5 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    offsets = [8, -8, 16, -16, 24, -24, 32, -32, 48, -48]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 320 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        h_rb = _to_binary(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_rb, c5_rb = _binary_greedy(h_rb, sub_dl, max_passes=100)
        if c5_rb < best_c5 + 0.002:
            sub_dl2 = min(d5, time.time() + budget_s * 0.02)
            h_rc, c5_rc = _continuous_refine(h_rb, sub_dl2, max_iters=500)
            if c5_rc < best_c5:
                best_c5 = c5_rc
                best_h = h_rc.copy()
                n_curr = n_try

    # Phase 6: Final polish (10%)
    d6 = deadline
    h6 = best_h.copy()
    sub_dl = min(d6, time.time() + budget_s * 0.04)
    h6, c5_6 = _continuous_refine(h6, sub_dl, max_iters=1000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    sub_dl = min(d6, time.time() + budget_s * 0.04)
    h6 = best_h.copy()
    h6, c5_6 = _continuous_pair(h6, sub_dl, max_iters=3000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
