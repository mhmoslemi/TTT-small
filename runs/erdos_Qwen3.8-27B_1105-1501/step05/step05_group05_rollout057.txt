def _project(v, target):

        return np.zeros(n)

        return np.ones(n)

        else:

def _line_search(h, d, dx, trust_radius):

def _logsumexp_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    v = np.clip(v, -50, 50)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        wm = w[r]
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wm * (1.0 - h[d:d + m])
                grad[d:] += wm * (-h[:m])

            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wm * (-h[g:g + m])
                grad[g:] += wm * (1.0 - h[:m])

def _lp_solve(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _logit_map(h, eps=1e-5):
    """Map h values to logit space."""
    h_clipped = np.clip(h, eps, 1.0 - eps)
    return np.log(h_clipped / (1.0 - h_clipped))

def _logit_to_h(v, target):
    """Map logit values back to h with exact mass constraint."""
    n = len(v)
    v = np.clip(v, -30, 30)
    # Find theta such that sum(sigmoid(v+theta)) = target
    def sig(x):
        x = np.clip(x, -50, 50)
        return 1.0 / (1.0 + np.exp(-x))
    
    lo = -max(float(np.max(v)) + 30, 30)
    hi = max(float(-np.min(v)) + 30, 30)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(sig(v + mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = sig(v + theta)
    # Fix residual
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        if res > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += res / cnt
            h = np.clip(h, 0, 1)
    return h

def _logit_evolution(h, deadline, rng, pop_size=8, n_gens=50):

            # Mutate

        # Keep best

        population = all_inds_sorted[:pop_size]

        h1, c1 = _lp_loop(h, d1, rad0=0.18)
        h = h1
        cur = c1
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
        d2 = min(deadline, time.time() + sub)
        h2, c2 = _grad_descend(h, d2, beta_seq=(50, 150, 400), lr0=0.025)
        h = h2
        cur = c2
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
        d3 = min(deadline, time.time() + sub)
        h3, c3 = _pair_refine(h, d3, rng, n_pairs_per_pass=200)
        h = h3
        cur = c3
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=25):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    per_kick = max(3.0, (deadline - time.time()) / max(n_kicks, 1))
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.08)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + per_kick)
        h_t, c5_t = _combined(h_k, sub_dl, rng)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 122
        h0 = np.full(n0, 0.5)
        for i in range(n0):
            h0[i] = 0.5 + 0.25 * np.sin(2.0 * np.pi * 3.0 * i / n0)
        h0 = _project(h0, n0 / 2.0)

    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick combined local optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c1 = _combined(h, d1, rng)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Logit evolution - main exploration (30%)
    d2 = t0 + budget_s * 0.50
    h2, c2 = _logit_evolve(best_h, d2, rng, pop_size=14)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()

    # Phase 3: Multi-resolution (20%)
    d3 = t0 + budget_s * 0.70
    n_best = len(best_h)
    offsets = sorted(range(-16, 17, 2), key=lambda x: (abs(x), x))
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 220 or n_try == n_best:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.002:
            sub_dl = min(d3, time.time() + budget_s * 0.012)
            h_r, c5_r = _combined(h_r, sub_dl, rng)
            if c5_r < best_c5 - 1e-13:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_best = n_try

    # Phase 4: Another round of evolution from best (15%)
    d4 = t0 + budget_s * 0.85
    h4, c4 = _logit_evolve(best_h, d4, rng, pop_size=10)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4.copy()

    # Phase 5: Kick and reoptimize (10%)
    d5 = t0 + budget_s * 0.95
    h5, c5 = _kick_reopt(best_h, d5, rng, n_kicks=15)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()

    # Phase 6: Final pair refinement
    d6 = t0 + budget_s - 2.0
    h6, c6 = _pair_refine(best_h, d6, rng, n_pairs_per_pass=400)
    if c6 < best_c5:
        best_c5 = c6
        best_h = h6.copy()

    # Final validation and projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
