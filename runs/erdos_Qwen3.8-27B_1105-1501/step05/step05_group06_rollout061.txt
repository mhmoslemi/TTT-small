def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def _smooth_init(h, sigma=0.3, rng=None):
    """Smooth h by adding noise and re-projecting."""
    n = len(h)
    if rng is not None:
        noise = rng.normal(0, sigma, n)
    else:
        noise = np.zeros(n)
    return _project(h + noise, n / 2.0)

Phi_q = (sum_r F_r^q)^(1/q)

dPhi_q/dh_j = (1/q) * (sum F^q)^(1/q - 1) * sum_r q * F_r^(q-1) * dF_r/dh_j
            = (sum F^q)^(1/q - 1) * sum_r F_r^(q-1) * G[r,j]

Let S = sum F_r^q. Then:
(sum F^q)^(1/q - 1) = S^((1-q)/q)

And F_r^(q-1) = (maxF * u_r)^(q-1) where u_r = F_r/maxF.

So dPhi/dh_j = S^((1-q)/q) * sum_r (maxF*u_r)^(q-1) * G[r,j]
            = S^((1-q)/q) * maxF^(q-1) * sum_r u_r^(q-1) * G[r,j]

And S = maxF^q * sum(u_r^q) = maxF^q * sumuq.

So S^((1-q)/q) = (maxF^q * sumuq)^((1-q)/q) = maxF^(1-q) * sumuq^((1-q)/q)

Therefore: dPhi/dh_j = maxF^(1-q) * sumuq^((1-q)/q) * maxF^(q-1) * sum_r u_r^(q-1) * G[r,j]
                  = sumuq^((1-q)/q) * sum_r u_r^(q-1) * G[r,j]

Let w_r = u_r^(q-1). Then:
dPhi/dh_j = sumuq^((1-q)/q) * (G^T @ w)_j

And Phi_q = maxF * sumuq^(1/q)

coeff = sumuq ** ((1.0 - q) / q)
G = _build_G(h)
grad = coeff * (G.T @ w)
phi = maxF * sumuq ** (1.0 / q)

import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt == 0:
            mask = x < 1.0 if res > 0 else x > 0
            cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    omh = 1.0 - h
    for d in range(n):
        r = n - 1 + d
        m = n - d
        if d > 0:
            G[r, :m] += dx * omh[d:]
        else:
            G[r, :n] += dx * omh
        G[r, d:] += -dx * h[:m]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, s:] += dx * omh[:m]
        G[r, :m] += -dx * h[s:]
    return G

def _pnorm_val_grad(h, q):
    n = len(h)
    F = _scores(h)
    F = np.maximum(F, 0.0)
    maxF = float(F.max())
    if maxF < 1e-30:
        return 0.0, np.zeros(n)
    u = F / maxF
    if q <= 2:
        w = u
    else:
        w = u ** (q - 1)
    if q > 2:
        sumuq = float((u ** q).sum())
    else:
        sumuq = float((u ** q).sum())
    if sumuq < 1e-300:
        sumuq = 1e-300
    phi = maxF * sumuq ** (1.0 / q)
    coeff = sumuq ** ((1.0 - q) / q)
    G = _build_G(h)
    grad = coeff * (G.T @ w)
    return phi, grad

def _pnorm_val_only(h, q):
    n = len(h)
    F = _scores(h)
    F = np.maximum(F, 0.0)
    maxF = float(F.max())
    if maxF < 1e-30:
        return 0.0
    u = F / maxF
    sumuq = float((u ** q).sum())
    if sumuq < 1e-300:
        sumuq = 1e-300
    return maxF * sumuq ** (1.0 / q)

def _proj_grad_opt(h0, q, deadline, max_iters=400, alpha_init=0.04):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    val = _pnorm_val_only(h, q)
    alpha = alpha_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        _, grad = _pnorm_val_grad(h, q)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            break
        alpha_eff = min(alpha, 0.3 / gnorm)
        found = False
        a = alpha_eff
        for _bt in range(25):
            h_t = _project(h - a * grad, target)
            v_t = _pnorm_val_only(h_t, q)
            if v_t < val - 1e-16:
                found = True
                break
            a *= 0.45
            if a < 1e-15:
                break
        if not found:
            stall += 1
            alpha = max(0.001, alpha * 0.6)
            if alpha_eff < 1e-14:
                break
            continue
        h = h_t
        alpha = min(alpha * 1.2, 0.4)
        stall = 0
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        val = _pnorm_val_only(h, q)
    return best_h, best_c5

def _continuation(h0, deadline, q_list, alpha_init=0.04):
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    for q in q_list:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        sub_dl = time.time() + remaining * 0.88
        h_new, c5_new = _proj_grad_opt(best_h, q, sub_dl,
                                         max_iters=350, alpha_init=alpha_init)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha_succ, rng):
    x = np.arange(n) / n
    h = ((x * alpha_succ) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.06, n)
    return _project(h, n / 2.0)

def _build_G_lp(h, scores, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    G = _build_G(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        return G[idx], scores[idx]
    return G, scores

def _lp_solve(h, G_k, s_k, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = minimize(lambda x: x[-1], x0=np.zeros(n + 1),
                       method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x[-1],
                                      'jac': lambda x: np.append(np.zeros(n), 1.0)}],
                       options={'maxiter': 100})
        # Actually use linprog-style via scipy
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_local_opt(h, deadline, top_k_list=None, radius_init=0.1, max_iters=2000):
    if top_k_list is None:
        top_k_list = [None, 8, 5, 3]

    target = n / 2.0

            if radius < 1e-9:

        # Line search
        alpha_max = 1.0

            elif p[j] < -1e-14:

def _pair_descent(h, deadline, max_passes=100, thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-12 or h[j] > 1.0 - 1e-12:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-12 or h[k] > 1.0 - 1e-12:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(15):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _slsqp_obj(x, q):
    val, grad = _pnorm_val_grad(x, q)
    return val, grad

_SLQP_n = 0

def _slsqp_cfun(x):
    return np.sum(x) - _SLQP_n * 0.5

def _slsqp_cjac(x):
    return np.ones(_SLQP_n)

def _slsqp_opt(h0, q, maxiter=80):
    global _SLQP_n
    _SLQP_n = len(h0)
