    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:

                # ∂/∂h[i] = (1-h[i+d]) for i in [0, m-1] (as first factor)
                #           -h[i-d] for i in [d, n-1] (as second factor in 1-h)
                G[r, :m] += (1.0 - h[d:d+m])  # as first factor: (1-h[i+d]) for i=0..m-1
                G[r, d:] += -h[:m]  # as second factor: -h[i-d] for i=d..n-1

                #           -h[i+g] for i in [0, n-1-g] (as second factor in 1-h)
                G[r, g:] += (1.0 - h[:m])  # as first factor: (1-h[i-g]) for i=g..n-1
                G[r, :m] += -h[g:g+m]  # as second factor: -h[i+g] for i=0..m-1
