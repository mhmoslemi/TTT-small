def _lse_max(values, beta):
    """Compute log-sum-exp approximation to max."""
    shifted = beta * values - beta * np.max(values)
    return np.max(values) + np.log(np.sum(np.exp(shifted))) / beta

    """Weighted average of subgradients using LSE weights."""
    shifted = beta * values
    max_val = np.max(shifted)
    exp_vals = np.exp(shifted - max_val)
    weights = exp_vals / np.sum(exp_vals)
    return weights

def _hasse_grad(h, beta):

    n = len(h)
    dx = 2.0 / n

    values = corr * dx

Actually, I just realized something important. Let me re-examine the boundary conditions of the correlation.

np.correlate(h, 1-h, mode='full') for arrays of length n gives output of length 2n-1.

The output at index k (0-indexed) is:
out[k] = sum_i h[i] * (1-h[i+k-(n-1)])

for valid i (where both indices are in [0, n-1]).

So for the "shift d" lag (where d = k - (n-1)):
out[n-1+d] = sum_i h[i+d] * (1-h[i])

where i ranges over values where both i and i+d are in [0, n-1].

For d ≥ 0: i ranges from 0 to n-1-d
For d < 0: i ranges from -d to n-1

This is equivalent to:
For d ≥ 0: sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
For d < 0: sum_{i=|d|}^{n-1} h[i-|d|] * (1-h[i])

Note that f_{-d} is related to f_d but not the same (it's the "opposite direction" count).

f_d = # of positions where h goes from 0 to 1 when shifted right by d
f_{-d} = # of positions where h goes from 0 to 1 when shifted left by d = # of positions where h goes from 1 to 0 when shifted right by d

So f_d + f_{-d} = total number of "transitions" at distance d (in either direction).

For a binary sequence, f_d + f_{-d} = 2 * (# of 1s in h[0:n-d]) + 2 * (# of 0s in h[d:n]) - 2 * (# of positions where h[i] = h[i+d])

def _subgrad_method(h, deadline, n_iter=2000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.02
    for i in range(n_iter):
        if time.time() > deadline:
            break
        lags = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = float(np.max(lags))
        n_active = max(1, int(0.05 * len(lags)))
        active = np.argsort(lags)[-n_active:]
        wd = np.zeros(len(lags))
        wd[active] = 1.0 / n_active
        grad = np.zeros(n)
        for r in active:
            d = r - (n - 1)
            wc = wd[r] * dx
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += wc * (1.0 - h[d:])
                    grad[d:] -= wc * h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += wc * (1.0 - h[g:])
                    grad[g:] -= wc * h[:m]
        h = _project(h - step * grad, target)
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        step *= 0.998
    return best_h, best_c5

def _coord_descent(h, deadline, n_pairs=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if len(interior) < 2:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        h_try_1 = h + 0.3 * d
        h_try_1 = _project(h_try_1, target)
        c5_try = _exact_c5(h_try_1)
        if c5_try < best_c5 - 1e-13:
            h = h_try_1
            best_c5 = c5_try
            best_h = h.copy()
            interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        else:
            d[i] = 1.0
            d[j] = -1.0
            h_try_2 = h + 0.3 * d
            h_try_2 = _project(h_try_2, target)
            c5_try2 = _exact_c5(h_try_2)
            if c5_try2 < best_c5 - 1e-13:
                h = h_try_2
                best_c5 = c5_try2
                best_h = h.copy()
                interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    return best_h, best_c5

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    grid = np.linspace(t_lo, t_hi, 80)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < len(grid) - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 2
            radius *= 0.75
            if radius < 1e-6:
                break
            if stall % 15 == 0:
                radius = max(radius, 0.05)
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.8
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=-scores, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    # Relaxed: allow top 3 active lags to be violated slightly
    sorted_idx = np.argsort(scores)[::-1]
    active_set = set(sorted_idx[:3].tolist())
    b_ub_relax = np.full(n_lags, -scores + radius * dx)
    for k in active_set:
        b_ub_relax[k] = -scores[k]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub_relax, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multiresearch(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tried = set()
    for dn in range(-10, 15):
        if time.time() > deadline:
            break
        n_try = n + dn
        if n_try < 24 or n_try > 160 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline, time.time() + 4.0)
        _OPT_STATE['n'] = n_try
        _OPT_STATE['dx'] = 2.0 / n_try
        _OPT_STATE['beta'] = 100
        try:
            res = minimize(_lse_obj_grad, h_r, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n_try,
                          options={'maxiter': 80, 'gtol': 1e-9})
            h_r = _project(res.x, n_try / 2.0)
        except Exception:
            pass
        sub_dl2 = min(deadline, time.time() + 3.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.1)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def _perturb_lse(h, deadline, rng, base_scale=0.05):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = base_scale
    for _ in range(200):
        if time.time() > deadline:
            break
        h_pert = _project(best_h + rng.normal(0, scale, n), target)
        sub_dl = min(deadline, time.time() + 3.0)
        _OPT_STATE['n'] = n
        _OPT_STATE['dx'] = 2.0 / n
        _OPT_STATE['beta'] = 200
        try:
            res = minimize(_lse_obj_grad, h_pert, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 60, 'gtol': 1e-9})
            h_pert = _project(res.x, target)
        except Exception:
            pass
        sub_dl2 = min(deadline, time.time() + 2.0)
        h_t, c5_t = _lp_loop(h_pert, sub_dl2, radius_init=0.06)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.08)
    return best_h, best_c5

def _anneal_binary(h, deadline, n_flips=500):
    """Try binary annealing on the current solution."""
    n = len(h)
    target = n / 2.0
    # Start from binarized version
    h_bin = (h >= 0.5).astype(float)
    h_bin = _project(h_bin, target)
    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)
    rng = np.random.default_rng(99)
    temp = 0.02
    for _ in range(n_flips):
        if time.time() > deadline:
            break
        # Flip a small group
        group_size = rng.integers(1, 5)
        indices = rng.choice(n, size=group_size, replace=False)
        h_try = best_h.copy()
        for idx in indices:
            h_try[idx] = 1.0 - h_try[idx]
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        else:
            delta = c5_try - best_c5
            if rng.random() < np.exp(-delta / temp):
                best_h = h_try.copy()
        temp *= 0.995
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LSE optimization (18% of time)
    d1 = t0 + budget_s * 0.18
    best_h, best_c5 = _lse_optimize(best_h, d1, betas=[20, 50, 100, 200, 400])
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 2: Top-k optimization (10%)
    d2 = t0 + budget_s * 0.28
    best_h, best_c5 = _topk_optimize(best_h, d2)
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 3: LP refinement (18%)
    d3 = t0 + budget_s * 0.46
    best_h, best_c5 = _lp_loop(best_h, d3, radius_init=0.12)
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 4: Multi-n exploration (16%)
    d4 = t0 + budget_s * 0.62
    best_h, best_c5 = _multiresearch(best_h, d4, rng)
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 5: Perturbation + LSE (14%)
    d5 = t0 + budget_s * 0.76
    best_h, best_c5 = _perturb_lse(best_h, d5, rng, base_scale=0.04)
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 6: LP polish (10%)
    d6 = t0 + budget_s * 0.86
    best_h, best_c5 = _lp_loop(best_h, d6, radius_init=0.06)
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 7: Subgradient fine-tuning (6%)
    d7 = t0 + budget_s * 0.92
    best_h, best_c5 = _subgrad_method(best_h, d7, n_iter=1000)
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 8: Coordinate descent final polish (8%)
    d8 = t0 + budget_s * 0.99
    best_h, best_c5 = _coord_descent(best_h, d8, n_pairs=5000)
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)

    # Final validation
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
