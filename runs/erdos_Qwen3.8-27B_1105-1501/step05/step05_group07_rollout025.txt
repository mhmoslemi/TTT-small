import numpy as np
from scipy.optimize import linprog, minimize
import time

_global_beta = 50.0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _sig(z):
    out = np.empty_like(z, dtype=float)
    pos = z >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    out[~pos] = ez / (1.0 + ez)
    return out

def _find_lambda(q):
    n = len(q)
    target = 0.5 * n
    lo = float(np.min(q)) - 30.0
    hi = float(np.max(q)) + 30.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sig(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q):
    lam = _find_lambda(q)
    return _sig(q - lam)

def _h_to_q(h):
    eps = 1e-6
    hc = np.clip(h, eps, 1.0 - eps)
    return np.log(hc / (1.0 - hc))

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m]) * dx
                G[r, :m] += (-h[d:]) * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:]) * dx
                G[r, g:n] += (-h[:m]) * dx
    return G

def _obj_val(q):
    beta = _global_beta
    h = _q_to_h(q)
    S = _scores(h)
    maxS = float(np.max(S))
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500.0, 0.0)
    return maxS + np.log(float(np.mean(np.exp(shifted)))) / beta

def _jac_val(q):
    beta = _global_beta
    h = _q_to_h(q)
    n = len(h)
    S = _scores(h)
    maxS = float(np.max(S))
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w = w / sw
    G = _build_G(h)
    gh = G.T @ w
    hm1h = h * (1.0 - h)
    m = float(np.sum(hm1h))
    if m < 1e-15:
        return np.zeros(n)
    c = float(np.dot(gh, hm1h)) / m
    return hm1h * (gh - c)

def _lbfgs_q(h_init, deadline, beta_list):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    q = _h_to_q(h_init)
    
    for beta in beta_list:
        if time.time() > deadline:
            break
        _global_beta = beta
        try:
            res = minimize(_obj_val, q, jac=_jac_val, method='L-BFGS-B',
                          bounds=[(-50.0, 50.0)] * n,
                          options={'maxiter': 400, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxls': 50})
            h_new = _q_to_h(res.x)
            h_new = _project(h_new, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                q = _h_to_q(h_new)
            else:
                # Still update q to the new point for next iteration
                q = _h_to_q(_project(_q_to_h(res.x), n / 2.0))
        except Exception:
            pass
    
    return best_h, best_c5
