    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))

        else:

            x = np.clip(x, 0.0, 1.0)
    return x

def _build_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    omh = 1.0 - h
    for r in range(L):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, d:] += dx * omh[:m]
            G[r, :m] += -dx * h[d:]
        else:
            m = n + d
            G[r, :m] += dx * omh[-d:]
            G[r, -d:] += -dx * h[:m]
    return G

    M = float(c.max())

        return M, G[idx].copy()

    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _build_jacobian(h)
    scores = _scores(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-12 and h[k] < 1e-12:
                    continue
                if h[j] > 1 - 1e-12 and h[k] > 1 - 1e-12:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(14):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > 1e-9 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_jacobian(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5


def _grad_step(h, deadline, n_steps=100, beta_start=30, beta_end=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _build_jacobian(h)

    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = scores.max()
        if max_s < 1e-30:
            break
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.05
        for _ in range(20):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                G = _build_jacobian(h)
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
    return best_h, best_c5


def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: P-norm continuation at original resolution
    q_list_1 = [8, 16, 32, 64, 128, 256]
    dl1 = t0 + budget_s * 0.22
    h1, c5_1 = _continuation(best_h, q_list_1, dl1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP-based local optimization
    if time.time() < t0 + budget_s * 0.38:
        dl2 = t0 + budget_s * 0.40
        h2, c5_2 = _local_opt_lp(best_h, dl2, top_k=8, radius_init=0.06, max_iters=400)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: High-q p-norm refinement
    if time.time() < t0 + budget_s * 0.50:
        dl3 = t0 + budget_s * 0.52
        q_list_3 = [128, 256, 512]
        h3, c5_3 = _continuation(best_h, q_list_3, dl3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-resolution search
    if time.time() < t0 + budget_s * 0.70:
        dl4 = t0 + budget_s * 0.72
        resolutions = [192, 256, 320]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl4:
                break
            for s in range(2):
                if time.time() > dl4:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    h_init = _quasi_init(n_try, alphas[ri % len(alphas)], rng)
                sub_dl = min(dl4, time.time() + budget_s * 0.04)
                q_try = [16, 64, 128, 256]
                h_try, c5_try = _continuation(h_init, q_try, sub_dl)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                else:
                    sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                    h_lp, c5_lp = _local_opt_lp(h_try, sub_dl2, top_k=6, radius_init=0.06, max_iters=200)
                    if c5_lp < best_c5:
                        best_c5 = c5_lp
                        best_h = h_lp.copy()
                        best_n = n_try

    # Phase 5: Grad step + p-norm on best
    if time.time() < t0 + budget_s * 0.80:
        dl5 = t0 + budget_s * 0.82
        h5, c5_5 = _grad_step(best_h, dl5, n_steps=80, beta_start=40, beta_end=400)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < dl5:
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h5b, c5_5b = _continuation(best_h, [256, 512, 1024], sub_dl)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: Pair descent polish
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.92
        h6, c5_6 = _pair_descent(best_h, dl6, max_passes=120)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final LP + micro p-norm
    if time.time() < hard_deadline - 5:
        dl7 = hard_deadline
        h7, c5_7 = _local_opt_lp(best_h, dl7, top_k=4, radius_init=0.02, max_iters=200)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
        if time.time() < dl7 - 3:
            h7b, c5_7b = _pair_descent(best_h, dl7, max_passes=60)
            if c5_7b < best_c5:
                best_c5 = c5_7b
                best_h = h7b.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
