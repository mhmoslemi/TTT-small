def binary_local_search(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    def eval_c5():
        corr = np.correlate(h, 1-h, mode='full')
        return float(np.max(corr) * dx)
    
    cur_c5 = eval_c5()
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.01
    while time.time() < deadline:
        # Swap a random 1 and 0
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0
        h[j] = 1
        new_c5 = eval_c5()
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        T = max(0.00001, T * 0.9999)
    
    return best_h, best_c5
