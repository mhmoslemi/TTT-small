import numpy as np
from scipy.optimize import linprog
import time

def _c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_J(h):
    """Build the Jacobian of correlation w.r.t. h at point h.
    For correlation at lag d: sum_i h[i]*(1-h[i+d]) + sum_i (1-h[i])*h[i+d] (for the appropriate half)
    Actually let's just build G as in the previous code.
    """
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n - 1, n))
    one_minus_h = 1.0 - h
    for d in range(n):
        r = n - 1 + d
        m = n - d
        # corr[r] = sum_{i=0}^{m-1} h[i+d] * (1-h[i])
        # d/dh[j] for j in [d, n-1]: (1-h[j-d])
        # d/dh[j] for j in [0, m-1]: -h[j+d]
        G[r, d:] += dx * one_minus_h[:m]
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):

        m = n - s

        # d/dh[j] for j in [0, m-1]: (1-h[j+s])
        # d/dh[j] for j in [s, n-1]: -h[j-s]
        G[r, :m] += dx * one_minus_h[s:]
        G[r, s:] += -dx * h[:m]
    return G

def _lp_direction(h, G, scores, top_k, radius):
    """Find best step direction using LP on top_k active constraints."""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    
    # Variables: p (n) + t (1)
    # Minimize t
    # s.t.: G_k @ (h + p) <= t  =>  G_k @ p - t <= -G_k @ h = -s_k
    #       sum(p) = 0
    #       -radius <= p_j <= radius (also clipped by bounds)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G_k, -np.ones((len(s_k), 1))])
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs',
                      options={'time_limit': 10})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _binary_sa(h, deadline, n_iter=50000):
    """Simulated annealing on the binary part of h."""
    n = len(h)
    # Start from binarized version
    h_bin = (h > 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    while len(ones) > n // 2:
        i = ones.pop()
        h_bin[i] = 0
        zeros.append(i)
    while len(ones) < n // 2:
        i = zeros.pop()
        h_bin[i] = 1
        ones.append(i)
    
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    def cost(hb):
        s = _c5_scores(hb)
        return s.max()
    
    cur_cost = cost(h_bin)
    best_cost = cur_cost
    best_h = h_bin.copy()
    
    T = cur_cost * 0.3
    T_min = 1e-6
    cooling = 0.9997
    
    rng = np.random.default_rng(42)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Pick random 1 and 0 to swap
        oi = rng.choice(list(ones))
        zi = rng.choice(list(zeros))
        
        h_bin[oi] = 0
        h_bin[zi] = 1
        new_cost = cost(h_bin)
        delta = new_cost - cur_cost
        
        if delta <= 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_cost = new_cost
            ones.remove(oi)
            ones.add(zi)
            zeros.add(oi)
            zeros.remove(zi)
            if cur_cost < best_cost:
                best_cost = cur_cost
                best_h = h_bin.copy()
        else:
            h_bin[oi] = 1
            h_bin[zi] = 0
        
        T *= cooling
    
    return best_h, best_cost

- For each lag d:
  - Position a: was 1, now 0.

    - a contributes to lag d as "i+d" if a-d >= 0: before h[a]=1 (counts if h[a-d]=0), after h[a]=0 (doesn't count)
  - Position b: was 0, now 1.
    - b contributes to lag d as "i" if b+d < n: before h[b]=0 (counts if h[b+d]=1), after h[b]=1 (doesn't count)
    - b contributes to lag d as "i+d" if b-d >= 0: before h[b]=0 (doesn't count), after h[b]=1 (counts if h[b-d]=0)

delta_d = (new_count - old_count)

Let me denote:
- For position a (1→0):
  - As "i" (a as the 0-position): if a+d<n, gain +1 if h[a+d]=1
  - As "i+d" (a as the 1-position): if a-d>=0, lose 1 if h[a-d]=0
- For position b (0→1):

  - As "i+d" (b as the 1-position): if b-d>=0, gain 1 if h[b-d]=0

import numpy as np
from scipy.optimize import linprog
import time

def _c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_J(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * omh[:m]
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * omh[s:]
        G[r, s:] += -dx * h[:m]
    return G

def _lp_step(h, G, scores, top_k, radius):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'time_limit': 5})
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search_min(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    n_grid = 10
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = np.array([_c5_eval(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0:
        return 0.0, float(vals[0])
    if best_i == n_grid - 1:
        return alpha_max, float(vals[-1])
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5_eval(h + c1 * p)
    f2 = _c5_eval(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _c5_eval(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _c5_eval(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _c5_eval(h + opt_a * p)
    if opt_c5 < float(vals[best_i]):
        return opt_a, opt_c5
    return alphas[best_i], float(vals[best_i])

def _binary_sa(h_start, deadline, max_iter=80000, seed=42):

    rng = np.random.default_rng(seed)
    h = (h_start > 0.5).astype(float)

            h[rng.choice(zeros, need, replace=False)] = 1

def _subgrad_step(h, deadline, n_steps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_eval(h)
    cur_c5 = best_c5
    G = _build_J(h)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _c5_scores(h)
        peak = scores.max()
        active = np.where(scores > peak - 0.02)[0]
        w = scores[active]
        w = w - w.min() + 0.1
        w = w ** 2
        w /= w.sum()
        grad = G[active].T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        for alpha_try in [0.08, 0.04, 0.02, 0.01, 0.005]:
            h_new = _project(h - alpha_try * grad, target)
            c5_new = _c5_eval(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                G = _build_J(h)
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                h_bin2, _ = _binary_sa(best_h, min(dl4, time.time() + budget_s * 0.02),
                                        max_iter=20000, seed=seed + n_try + attempt)
                h_init = _project(h_bin2 + rng.normal(0, 0.04, n_try), n_try / 2.0)
            else:
                # Quasi-random init
                x = np.arange(n_try) / n_try
                alpha = [0.6180339887, 0.4142135624, 0.7071067812, 0.5, 0.5555][attempt % 5]
                h_init = ((x * alpha) % 1.0 < 0.5).astype(float)
                h_init += rng.normal(0, 0.05, n_try)
                h_init = _project(h_init, n_try / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.035)
            h_try, c5_try = _local_opt_lp(h_init, sub_dl,
                                           top_k_list=[None, 10, 6, 4],
                                           radius_init=0.12, max_iters=2000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                sub_dl2 = min(dl4, time.time() + budget_s * 0.012)
                h_sg2, c5_sg2 = _subgrad_step(h_try, sub_dl2, n_steps=80)
                if c5_sg2 < best_c5:
                    best_c5 = c5_sg2
                    best_h = h_sg2.copy()
                    best_n = n_try

    # Phase 5: Pair descent refinement
    if time.time() < t0 + budget_s * 0.75:
        dl5 = t0 + budget_s * 0.75
        h_pd, c5_pd = _pair_descent_fast(best_h, dl5, max_passes=150)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
        # Follow with LP
        if time.time() < dl5 - 5:
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h_lp2, c5_lp2 = _local_opt_lp(best_h, sub_dl,
                                           top_k_list=[5, 4, 3, None],
                                           radius_init=0.04, max_iters=1000)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h_lp2.copy()

    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.90
    scale = 0.03
    for attempt in range(300):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.007)
        h_try, c5_try = _local_opt_lp(h_pert, sub_dl,
                                       top_k_list=[5, 3, None],
                                       radius_init=0.04, max_iters=300)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 7: Final polish
    if time.time() < hard_deadline:
        dl7 = hard_deadline
        h_fin, c5_fin = _pair_descent_fast(best_h, dl7, max_passes=80)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if time.time() < dl7 - 3:
            h_fin2, c5_fin2 = _local_opt_lp(best_h, dl7,
                                             top_k_list=[3, 4, 5],
                                             radius_init=0.008, max_iters=200)
            if c5_fin2 < best_c5:
                best_c5 = c5_fin2
                best_h = h_fin2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5_eval(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
