```python
def lag_jacobian(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        if k >= n-1:
            d = k - (n-1)
            if d == 0:
                G[k] = 1.0 - 2.0*h
            else:
                # Positive lag: F = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
                # h[j] in h[i+d] term: j-d = i, so i = j-d, valid when 0 <= j-d <= n-1-d
                #   i.e., d <= j <= n-1
                # Contribution: (1-h[j-d])
                # h[j] in (1-h[i]) term: i = j, valid when 0 <= j <= n-1-d
                # Contribution: -h[j+d]
                lo1, hi1 = d, n-1
                G[k, lo1:hi1+1] += (1.0 - h[lo1-d:hi1+1-d])
                lo2, hi2 = 0, n-1-d
                G[k, lo2:hi2+1] += -h[lo2+d:hi2+1+d]
        else:
            a = n-1-k
            # Negative lag: F = sum_{i=0}^{n-1-a} h[i]*(1-h[i+a])
            # h[j] in h[i] term: i = j, valid when 0 <= j <= n-1-a
            # Contribution: (1-h[j+a])
            # h[j] in (1-h[i+a]) term: i+a = j, i = j-a, valid when 0 <= j-a <= n-1-a
            #   i.e., a <= j <= n-1
            # Contribution: -h[j-a]
            lo1, hi1 = 0, n-1-a
            G[k, lo1:hi1+1] += (1.0 - h[lo1+a:hi1+1+a])
            lo2, hi2 = a, n-1
            G[k, lo2:hi2+1] += -h[lo2-a:hi2+1-a]
    return G
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_jacobian(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        if k >= n - 1:
            d = k - (n - 1)
            if d == 0:
                G[k] = 1.0 - 2.0 * h
            else:
                lo1, hi1 = d, n - 1
                G[k, lo1:hi1 + 1] += (1.0 - h[lo1 - d:hi1 + 1 - d])
                lo2, hi2 = 0, n - 1 - d
                if lo2 <= hi2:
                    G[k, lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
        else:
            a = n - 1 - k
            lo1, hi1 = 0, n - 1 - a
            G[k, lo1:hi1 + 1] += (1.0 - h[lo1 + a:hi1 + 1 + a])
            lo2, hi2 = a, n - 1
            if lo2 <= hi2:
                G[k, lo2:hi2 + 1] += -h[lo2 - a:hi2 + 1 - a]
    return G

def smooth_obj(x, beta, n, target):
    h = np.clip(x, 0.0, 1.0)
    scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = float(np.max(scores))
    z = np.exp((scores - m) / beta)
    s = float(np.sum(z))
    return m + beta * np.log(s)

def smooth_grad(x, beta, n, target):
    h = np.clip(x, 0.0, 1.0)
    dx_val = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx_val
    m = float(np.max(scores))
    z = np.exp((scores - m) / beta)
    s = float(np.sum(z))
    P = z / s
    G = lag_jacobian(h) * dx_val
    return P @ G

def smooth_optimize(h0, beta, deadline):
    n = len(h0)
    target = n / 2.0
    x0 = h0.copy()
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(lambda x: smooth_obj(x[:n], beta, n, target),
                       x0,
                       jac=lambda x: np.concatenate([smooth_grad(x[:n], beta, n, target), [1.0]]),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-7})
        if res.success or res.x is not None:
            h_new = project_box_sum(res.x[:n], target)
            c5_new = eval_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h0, eval_c5(h0)

def epigraph_solve(h0, c5_cur, deadline, n_constr=-1):
    n = len(h0)
    target = n / 2.0
    dx_val = 2.0 / n
    scores = np.correlate(h0, 1.0 - h0, mode='full') * dx_val
    if n_constr < 0 or n_constr >= len(scores):
        idx = np.arange(len(scores))
    else:
        sorted_idx = np.argsort(scores)[::-1]
        idx = sorted_idx[:n_constr]
    t0 = float(np.max(scores)) + 1e-6
    x0 = np.concatenate([h0, [t0]])
    n_lags = len(idx)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def constr_func(x):
        h = np.clip(x[:n], 0.0, 1.0)
        t = x[-1]
        all_scores = np.correlate(h, 1.0 - h, mode='full') * dx_val
        return t - all_scores[idx]

    def constr_jac(x):
        h = np.clip(x[:n], 0.0, 1.0)
        G = lag_jacobian(h) * dx_val
        J = np.zeros((n_lags, n + 1))
        J[:, :n] = -G[idx, :]
        J[:, n] = 1.0
        return J

    eq_func = lambda x: float(np.sum(x[:n])) - target
    eq_jac = lambda x: np.concatenate([np.ones(n), [0.0]])

    cons = [
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac},
        {'type': 'ineq', 'fun': constr_func, 'jac': constr_jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(max(0.0, t0 - 0.05), 2.0)]

    best_h = h0.copy()
    best_c5 = c5_cur
    t_watch = time.time()
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-9})
        if res.x is not None:
            h_new = project_box_sum(res.x[:n], target)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass
    return best_h, best_c5

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    n_grid = 80
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = max(t_lo, -trust_radius), min(t_hi, trust_radius)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _lp_solve_dir(h, scores, G, radius, n_active):
    n = len(h)
    n_lags = len(scores)
    cur_max = float(np.max(scores))
    tol = 0.02 * cur_max
    active = np.where(scores >= cur_max - tol)[0]
    if len(active) > n_active:
        top_idx = np.argsort(scores)[::-1][:n_active]
        active = top_idx
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active, :]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < cur_max - 1e-12:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def lp_loop(h, deadline, radius_init=0.12, n_active=15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = lag_jacobian(h)
        p, _ = _lp_solve_dir(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            p, _ = _lp_solve_dir(h, scores, G, radius, min(30, 2 * n_active))
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.7
                if radius < 1e-6:
                    break
                if stall % 20 == 0:
                    radius = max(radius, 0.05)
                continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = project_box_sum(h, target)

            if cur_c5 < best_c5:
                best_c5 = cur_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)
```
