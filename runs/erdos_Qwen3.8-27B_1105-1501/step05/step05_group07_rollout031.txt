    n = len(h)

def _scores(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _balance(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # Distribute the difference
    if diff > 0:
        # Need to add mass; give to elements with most room
        room = 1.0 - v
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            add = min(room[idx], remaining)
            v[idx] += add
            remaining -= add
            if remaining < 1e-14:
                break
    else:
        # Need to remove mass; take from elements with most value
        order = np.argsort(v)
        remaining = -diff
        for idx in order:
            rem = min(v[idx], remaining)
            v[idx] -= rem
            remaining -= rem
            if remaining < 1e-14:
                break
    return v

def _sigma(t):
    """Numerically stable sigmoid"""
    result = np.empty_like(t)
    pos = t >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    result[~pos] = exp_t / (1.0 + exp_t)
    return result

def _apply_q(q, target, q_bound=14.0):
    """Map q to feasible h via logistic with offset mu"""
    n = len(q)
    q = np.clip(q, -q_bound, q_bound)
    # Find mu such that sum sigma(q - mu) = target
    lo = q.min() - 10.0
    hi = q.max() + 10.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = _sigma(q - mid).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return _sigma(q - mu)

    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

    """Compute gradient matrix: returns (2n-1, n) matrix where row d gives grad of score d w.r.t. h, scaled by dx"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))

            j2 = k - d
            if 0 <= j2 < n:

            if g != 0.0:

    return G

def _balance(v, target):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-10:
        return v
    if diff > 0:
        room = 1.0 - v
        idx = np.argsort(-room)
        rem = diff
        for i in idx:
            a = min(room[i], rem)
            v[i] += a
            rem -= a
            if rem < 1e-12: break
    else:
        idx = np.argsort(v)
        rem = -diff
        for i in idx:
            a = min(v[i], rem)
            v[i] -= a
            rem -= a
            if rem < 1e-12: break
    return v

def _sigma(t):

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _balance(v, target):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-10:
        return v
    if diff > 0:
        room = 1.0 - v
        idx = np.argsort(-room)
        rem = diff
        for i in idx:
            a = min(room[i], rem)
            v[i] += a
            rem -= a
            if rem < 1e-13:
                break
    else:
        idx = np.argsort(v)
        rem = -diff
        for i in idx:
            a = min(v[i], rem)
            v[i] -= a
            rem -= a
            if rem < 1e-13:
                break
    return v

def _sigma(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    e = np.exp(t[~pos])
    out[~pos] = e / (1.0 + e)
    return out

def _q_to_h(q, target, qb=14.0):
    q = np.clip(q, -qb, qb)
    lo = float(q.min()) - 12.0
    hi = float(q.max()) + 12.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if _sigma(q - mid).sum() > target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    return _sigma(q - mu)

def _logit(h, eps=1e-7):
    hc = np.clip(h, eps, 1.0 - eps)
    return np.log(hc / (1.0 - hc))

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    G[n - 1] = (1.0 - 2.0 * h) * dx
    for d in range(1, n):
        m = n - d
        r = n - 1 + d
        G[r, d:] += (1.0 - h[:m]) * dx
        G[r, :m] += -h[d:] * dx
        r = n - 1 - d
        G[r, :m] += (1.0 - h[d:]) * dx
        G[r, d:] += -h[:m] * dx
    return G

def _smooth_obj_grad(q, beta, target, qb=14.0):
    h = _q_to_h(q, target, qb)
    s = h * (1.0 - h)
    sc = _scores(h)
    max_sc = sc.max()
    shifted = np.clip(beta * (sc - max_sc), -500.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(sc))
        w[int(np.argmax(sc))] = 1.0
    else:
        w /= wsum
    obj = max_sc + np.log(wsum / len(sc)) / beta
    G = _grad_matrix(h)
    g_h = G.T @ w
    s_sum = s.sum()
    if s_sum < 1e-15:
        return obj, np.zeros_like(q)
    rho = float(s @ g_h) / s_sum
    grad_q = s * (g_h - rho)
    if not np.all(np.isfinite(grad_q)):
        grad_q = np.zeros_like(q)
    return float(obj), grad_q

def _lbfgs_q_optimize(q_start, beta, target, deadline, maxiter=300, qb=14.0):
    n = len(q_start)
    best_q = q_start.copy()
    best_obj = _smooth_obj_grad(q_start, beta, target, qb)[0]
    bounds = [(-qb, qb)] * n
    try:
        res = minimize(lambda x: _smooth_obj_grad(x, beta, target, qb)[0],
                       q_start,
                       jac=lambda x: _smooth_obj_grad(x, beta, target, qb)[1],
                       method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if np.all(np.isfinite(res.x)):
            res_obj = _smooth_obj_grad(res.x, beta, target, qb)[0]
            if res_obj < best_obj:
                best_obj = res_obj
                best_q = res.x.copy()
    except Exception:
        pass
    return best_q

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _balance(h_new, n_new / 2.0)

def _complement_lp(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1
    A = np.zeros((L, n))
    hc = h.copy()
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                A[r, :m] = h[d:] * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                A[r, g:] = h[:m] * dx
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(L)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        if time.time() > deadline:
            return None
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            y = _balance(res.x[:n], target)
            return y
    except Exception:
        pass
    return None

def _line_search_c5(h1, h2, n_samp=25):
    n = len(h1)
    alphas = np.linspace(0.0, 1.0, n_samp)
    vals = np.array([_exact_c5((1 - a) * h1 + a * h2) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0 or best_i == n_samp - 1:
        return alphas[best_i], vals[best_i]
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5((1 - c1) * h1 + c1 * h2)
    f2 = _exact_c5((1 - c2) * h1 + c2 * h2)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5((1 - c1) * h1 + c1 * h2)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5((1 - c2) * h1 + c2 * h2)
    a_opt = 0.5 * (lo + hi)
    c5_opt = _exact_c5((1 - a_opt) * h1 + a_opt * h2)
    if c5_opt < vals[best_i]:
        return a_opt, c5_opt
    return alphas[best_i], vals[best_i]

def _pair_polish(h, deadline, max_passes=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 24)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = 0.5 * (lo_a + hi_a)
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5
