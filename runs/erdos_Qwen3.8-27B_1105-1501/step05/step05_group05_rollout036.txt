def _safe_sigmoid(x):
    """Numerically stable sigmoid."""
    out = np.empty_like(x, dtype=float)

    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _score_subgradient(h, dx, active_frac=0.1):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                radius = best_radius * 0.55
                stall = 0
    return best_h, best_c5

def _smooth_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    v = np.clip(v, -500, 500)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-16:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _smooth_descent(h, deadline, beta_seq, lr_init=0.04):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.82
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 100:
            bi += 1
            if bi >= len(beta_seq) * 5:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = beta_seq[bi % len(beta_seq)]
        d = _smooth_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            bi += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.8
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100, use_subset=False, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if use_subset and rng is not None and n > 60:
            pairs = set()
            n_try = min(600, n * 4)
            for _ in range(n_try):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and 1e-10 < h[i] < 1 - 1e-10 and 1e-10 < h[j] < 1 - 1e-10:
                    pairs.add((min(i, j), max(i, j)))
        else:
            pairs = set()
            for i in range(n):
                if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                    continue
                for j in range(i + 1, n):
                    if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                        continue
                    pairs.add((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _logit_es(h_init, deadline, pop_size=8, sigma_init=0.07, rng=None):
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    pop = [best_h.copy() for _ in range(pop_size)]
    sigma = sigma_init
    stall = 0
    while time.time() < deadline:
        if stall > 300:
            break
        offspring = []
        for i in range(pop_size):
            v = _h_to_v(pop[i % len(pop)], eps=1e-5)
            v_mut = v + rng.normal(0, sigma, n)
            h_mut = _v_to_h(v_mut, target)
            c5_mut = _exact_c5(h_mut)
            offspring.append((c5_mut, h_mut))
        offspring.sort(key=lambda x: x[0])
        if offspring[0][0] < best_c5:
            best_c5 = offspring[0][0]
            best_h = offspring[0][1].copy()
            pop[0] = best_h.copy()
            stall = 0
            sigma = max(sigma * 0.97, 0.003)
        else:
            stall += 1
            sigma = min(sigma * 1.04, 0.6)
        for i in range(min(4, pop_size)):
            if i < len(offspring) and offspring[i][0] < best_c5 + 0.005:
                idx = (i + 1) % pop_size
                pop[idx] = offspring[i][1].copy()
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=40):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.1)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 2.5)
        h_t, c5_t = _local_lp(h_k, sub_dl, radius_init=0.1)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _score_directed_moves(h, deadline, rng, n_rounds=30):
    """Try moves directed by the score structure: move mass from high-contribution coords to low."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _round in range(n_rounds):
        if time.time() > deadline:
            break
        scores = np.correlate(best_h, 1.0 - h if False else 1.0 - best_h, mode='full')
        s_max = float(np.max(scores))
        # Compute coordinate pressure from active lags
        active_thresh = s_max * 0.97
        active = scores >= active_thresh
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            break
        pressure = np.zeros(n)
        for r in active_idx:
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    pressure[:m] += (1.0 - best_h[d:d + m])
                    pressure[d:] += -best_h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    pressure[:m] += -best_h[g:g + m]
                    pressure[g:] += (1.0 - best_h[:m])
        # Find high-pressure and low-pressure interior coordinates
        interior = (best_h > 0.05) & (best_h < 0.95)
        if np.sum(interior) < 2:
            break
        int_idx = np.where(interior)[0]
        int_pressure = pressure[int_idx]
        # Sort: high pressure first, low pressure last
        order = np.argsort(int_pressure)
        n_high = max(1, len(order) // 4)
        n_low = max(1, len(order) // 4)
        high_set = set(int_idx[order[-n_high:]].tolist())
        low_set = set(int_idx[order[:n_low]].tolist())
        # Try moves from high to low
        improved_this_round = False
        for i in high_set:
            for j in low_set:
                if time.time() > deadline:
                    break
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(best_h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    best_h = best_h + alpha * d
                    best_h = _project(best_h, target)
                    cur = _exact_c5(best_h)
                    if cur < best_c5:
                        best_c5 = cur
                        improved_this_round = True
        if not improved_this_round:
            # Also try random small kicks
            d = np.zeros(n)
            k1 = int(rng.integers(0, n))
            k2 = int(rng.integers(0, n))
            while k2 == k1 and n > 1:
                k2 = int(rng.integers(0, n))
            d[k1] = 1.0
            d[k2] = -1.0
            alpha, c5_try = _line_search(best_h, d, dx, 0.3)
            if c5_try < best_c5 - 1e-14:
                best_h = best_h + alpha * d
                best_h = _project(best_h, target)
                best_c5 = _exact_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 3.0

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick LP polish (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_lp(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution search (22%)
    d2 = t0 + budget_s * 0.30
    best_n = len(best_h)
    offsets = sorted([0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16,
                      -18, 18, -20, 20, -22, 22, -24, 24, -26, 26, -28, 28, -30, 30],
                     key=lambda x: abs(x))
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = best_n + dn
        if n_try < 80 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try == best_n and dn == 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.018)
        h_r, c5_r = _local_lp(h_r, sub_dl, radius_init=0.12)
        if c5_r < best_c5 - 1e-14 and time.time() < d2 - 10:
            sub_dl2 = min(d2, time.time() + 4.0)
            h_r, c5_r = _pair_refine(h_r, sub_dl2, max_passes=25, use_subset=True, rng=rng)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 3: Score-directed moves on best (15%)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _score_directed_moves(best_h, d3, rng, n_rounds=50)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP with larger radius (15%)
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = _local_lp(best_h, d4, radius_init=0.35)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Smooth descent with annealing beta (15%)
    d5 = t0 + budget_s * 0.75
    h5, c5_5 = _smooth_descent(best_h, d5, beta_seq=[15, 40, 100, 250, 600, 1500], lr_init=0.035)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Logit evolution (10%)
    d6 = t0 + budget_s * 0.85
    h6, c5_6 = _logit_es(best_h, d6, pop_size=10, sigma_init=0.05, rng=rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Kick and reoptimize (5%)
    d7 = t0 + budget_s * 0.90
    h7, c5_7 = _kick_reopt(best_h, d7, rng, n_kicks=25)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final pair refinement + LP polish (10%)
    d8 = t0 + budget_s * 0.97
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=80, use_subset=True, rng=rng)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline:
        h9, c5_9 = _local_lp(best_h, deadline, radius_init=0.06)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
