import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    """Build gradient matrix G: row r = lag d=r-(n-1), G[r,j] = dS_r/dh[j]."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for d in range(0, n):
        m = n - d
        r = n - 1 + d
        G[r, :m] += omh[d:d + m]
        G[r, d:n] += -h[:m]
        if d > 0:
            r2 = n - 1 - d
            G[r2, d:n] += omh[:m]
            G[r2, :m] += -h[d:n]
    return G

def _line_search(h, d, dx, trust_radius):
    """Exact line search along direction d (zero-mean) for objective max(corr(h,1-h))*dx."""
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    t_lo = -trust_radius
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Grid + golden section
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if 0 < best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(35):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_step(h, G, scores, radius, dead_tol=1e-6):
    """Solve trust-region LP: min t s.t. G@p + t >= scores, sum(p)=0, |p_i|<=radius.
    Returns direction p or None."""
    n = len(h)
    # Variables: p_0..p_{n-1}, t
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Constraints: scores[r] - G[r,:]@p - t <= 0  =>  -G[r,:]@p - t <= -scores[r]
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = -G
    A_ub[:, n] = -1.0
    b_ub = -scores + radius * 1e-3  # small tolerance to avoid infeasibility
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.25, max_stall=50):
    """LP-based local optimization loop."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    while time.time() < deadline:
        if stall > max_stall:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _build_G(h)
        p = _lp_step(h, G, scores, radius)
        if p is None:
            stall += 1
            radius *= 0.6
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-12:
            h_new = h + alpha * p
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    best_radius = radius
                stall = 0
                radius = min(radius * 1.08, 0.6)
            else:
                stall += 1
                radius *= 0.75
        else:
            stall += 1
            radius *= 0.7
        if radius < 1e-9:
            radius = best_radius * 0.4
            stall = 0
    return best_h, best_c5

def _smooth_grad(h, dx, beta):
    """Compute smooth gradient using log-sum-exp weights."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _smooth_optimize(h, deadline, betas=(30, 80, 200, 500), lr_init=0.03):
    """Smooth gradient descent with momentum."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.85
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 80:
            bi += 1
            if bi >= len(betas) * 4:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = betas[bi % len(betas)]
        d = _smooth_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            bi += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * lr * mom
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.06, 0.45)
            else:
                stall += 1
                lr *= 0.8
        else:
            stall += 1
            lr *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coord_descent(h, deadline, max_rounds=200):
    """Coordinate descent: try moving each coordinate up or down."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    order = np.arange(n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        np.random.shuffle(order)
        improved = False
        for i in order:
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            # Try moving mass from i to a random other coordinate
            for _ in range(3):
                j = int(np.random.randint(0, n))
                if j == i or h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-13:
                    h_new = h + alpha * d
                    h_new = _project(h_new, target)
                    actual = _exact_c5(h_new)
                    if actual < best_c5 - 1e-14:
                        h = h_new
                        best_c5 = actual
                        best_h = h.copy()
                        improved = True
                        break
        if not improved and rnd > 10:
            break
    return best_h, best_c5

def _kick_reopt(h, deadline, n_kicks=30):
    """Random kick + LP reoptimization."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    for _ in range(n_kicks):
        if time.time() > deadline - 2:
            break
        scale = rng.uniform(0.01, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline - 1, time.time() + 2.0)
        h_t, c5_t = _lp_optimize(h_k, sub_dl, radius_init=0.1, max_stall=15)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _active_equalize(h, deadline, tol_frac=0.01, max_iter=50):
    """Equalize active lags using least-squares on the active gradient system."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        active = np.where(scores >= s_max * (1.0 - tol_frac))[0]
        if len(active) < 2:
            break
        G = _build_G(h)
        G_a = G[active, :]  # (k, n)
        # We want to find p (zero-sum, bounded) such that G_a @ p is as uniform as possible
        # i.e., minimize max(G_a @ p) - min(G_a @ p) subject to sum(p)=0, |p|<=radius
        # Approximate: minimize ||G_a @ p - mean||^2 + penalty for non-zero-sum
        # Use simple gradient step on the active rows
        row_sums = G_a @ np.ones(n)  # not useful directly
        # Compute weighted gradient toward equalizing
        residual = scores[active] - s_max
        # Direction: -sum of active gradients weighted by residual
        d = np.zeros(n)
        for idx in active:
            d -= residual[idx] * G[idx, :]
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        alpha, new_c5 = _line_search(h, d, dx, 0.15)
        if new_c5 < cur_c5 - 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            break
    return best_h, best_c5
