def _grad_matrix(h):
    """Compute G where G[r,i] = d(score[r])/d(h[i])."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for d in range(-(n-1), n):

        if d >= 0:
            m = n - d
            if m > 0:

        else:

            m = n - g

def _safe_sigma(t):
    """Numerically stable sigmoid."""
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

import numpy as np
from scipy.optimize import minimize, linprog
import time

_current_beta = 100.0
_current_target = 64.0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += dx * (1.0 - h[d:])
                G[r, d:] += -dx * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += -dx * h[g:]
                G[r, g:] += dx * (1.0 - h[:m])
    return G

def _safe_sigma(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def _solve_mu(q, target):
    lo = float(q.min()) - 16.0
    hi = float(q.max()) + 16.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_safe_sigma(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target):
    mu = _solve_mu(q, target)
    return _safe_sigma(q - mu)

def _h_to_q(h):
    eps = 1e-5
    hc = np.clip(h, eps, 1.0 - eps)
    return np.log(hc / (1.0 - hc))

def _smooth_obj(x):
    h = _q_to_h(x, _current_target)
    S = _scores(h)
    maxS = float(np.max(S))
    shifted = _current_beta * (S - maxS)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    mean_exp = float(np.mean(exp_s))
    if mean_exp < 1e-300:
        return maxS
    return maxS + float(np.log(mean_exp)) / _current_beta

def _smooth_jac(x):
    h = _q_to_h(x, _current_target)
    S = _scores(h)
    maxS = float(np.max(S))
    shifted = _current_beta * (S - maxS)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = float(np.sum(exp_s))
    if wsum < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w = exp_s / wsum
    G = _grad_matrix(h)
    g_h = G.T @ w
    mu = _solve_mu(x, _current_target)
    t = x - mu
    sig = _safe_sigma(t)
    s = sig * (1.0 - sig)
    S_sum = float(np.sum(s))
    if S_sum < 1e-300:
        return np.zeros_like(x)
    rho = float(np.sum(s * g_h)) / S_sum
    g_q = s * (g_h - rho)
    g_q = np.where(np.isfinite(g_q), g_q, 0.0)
    return g_q

def _balance_exact(x, target):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    res2 = target - float(np.sum(x))
    if abs(res2) > 1e-12:
        idx = int(np.argmax(np.abs(x - 0.5)))
        x[idx] = np.clip(x[idx] + res2, 0.0, 1.0)
    return x

def _lbf_bfgs(h_init, deadline, n_phases=10, beta_lo=None, beta_hi=None):
    global _current_beta, _current_target
    n = len(h_init)
    target = n / 2.0
    _current_target = target
    q = _h_to_q(h_init)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    L = 2 * n - 1
    if beta_lo is None:
        beta_lo = max(6.0, 0.08 * L)
    if beta_hi is None:
        beta_hi = min(800.0, 3.0 * L)
    betas = np.geomspace(beta_lo, beta_hi, n_phases)
    qb = q.copy()
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        _current_beta = betas[phase]
        maxiter = max(50, int(180.0 * np.sqrt(128.0 / n)))
        try:
            res = minimize(_smooth_obj, qb, jac=_smooth_jac, method='L-BFGS-B',
                          bounds=[(-18.0, 18.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
            if np.all(np.isfinite(res.x)):
                qb = res.x
                h_new = _balance_exact(_q_to_h(qb, target), target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def _crystallize(h, threshold_lo=0.05, threshold_hi=0.95):
    n = len(h)
    target = n / 2.0
    hc = h.copy()
    hc[hc < threshold_lo] = 0.0
    hc[hc > threshold_hi] = 1.0
    return _balance_exact(hc, target)

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _balance_exact(h_new, n_new / 2.0)

def _perturb(h, rng, scale):
    n = len(h)
    h_p = h + rng.normal(0, scale, n)
    return _balance_exact(h_p, n / 2.0)

def _coord_descent(h, deadline, n_passes=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    for p in range(n_passes):
        if time.time() > deadline:
            break
        order = np.random.default_rng(42 + p).permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            # Try moving h[i] up or down
            for direction in [1, -1]:
                a_max = 0.1
                if direction > 0:
                    a_max = min(0.1, 1.0 - h[i])
                else:
                    a_max = min(0.1, h[i])
                if a_max < 1e-10:
                    continue
                # Binary search for best
                lo_a, hi_a = 0.0, a_max
                best_a = 0.0
                best_val = _exact_c5(h)
                for _ in range(12):
                    mid = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[i] += direction * mid
                    h_try = np.clip(h_try, 0.0, 1.0)
                    # Adjust sum: find another coordinate to compensate
                    diff = float(np.sum(h) - np.sum(h_try))
                    if abs(diff) > 1e-14:
                        if direction > 0:
                            j = int(np.argmin(h_try))
                            if h_try[j] + diff <= 1.0:
                                h_try[j] += diff
                        else:
                            j = int(np.argmax(h_try))
                            if h_try[j] + diff >= 0.0:
                                h_try[j] += diff
                    h_try = _balance_exact(h_try, target)
                    val = _exact_c5(h_try)
                    if val < best_val:
                        best_val = val
                        best_a = mid
                        lo_a = mid
                    else:
                        hi_a = mid
                if best_a > 1e-10 and best_val < best_c5 - 1e-14:
                    h[i] += direction * best_a
                    h = _balance_exact(h, target)
                    best_c5 = _exact_c5(h)
    return h, best_c5

def _lp_direction_optimize(h, deadline, n_iters=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        S = _scores(h)
        G = _grad_matrix(h)
        maxS = float(np.max(S))
        w = np.exp(S - maxS)
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w /= wsum
        g = G.T @ w
        g -= g.mean()
        # Line search: h - alpha * g, adjusted for sum
        g_no_mean = g - g.mean()
        best_alpha = 0.0
        best_val = best_c5
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            h_try = _balance_exact(h - alpha * g_no_mean, target)
            val = _exact_c5(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 0 and best_val < best_c5 - 1e-14:
            h = _balance_exact(h - best_alpha * g_no_mean, target)
            best_c5 = _exact_c5(h)
        elif it > 20:
            break
    return h, best_c5

            if cp < best_c5 - 1e-13:
                best_c5 = cp
                best_h = hp.copy()
                scale = max(0.004, scale * 0.82)
                n_restarts = 0
            else:
                scale = min(0.08, scale * 1.12)
                n_restarts += 1

    # Phase 6: High-beta final L-BFGS-B polish
    if time.time() < t0 + budget_s * 0.93:
        dl6 = t0 + budget_s * 0.93
        h6, c5_6 = _lbf_bfgs(best_h, dl6, n_phases=8, beta_lo=100, beta_hi=800)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Coordinate descent final polish
    if time.time() < t0 + budget_s * 0.965:
        h7, c5_7 = _coord_descent(best_h, t0 + budget_s * 0.965, n_passes=5)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: One more L-BFGS-B pass
    if time.time() < t0 + budget_s * 0.975:
        h8, c5_8 = _lbf_bfgs(best_h, t0 + budget_s * 0.975, n_phases=4, beta_lo=200, beta_hi=1000)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation
    best_h = _balance_exact(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
