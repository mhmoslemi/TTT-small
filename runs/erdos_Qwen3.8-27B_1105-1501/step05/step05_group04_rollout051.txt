def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad_row(h, s):
    """Gradient of lag s value w.r.t. h, including dx factor."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        # F_s = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
        # d/dh[j]: (1-h[j-s]) if j>=s, minus h[j+s] if j<=n-1-s
        for j in range(s, n):
            grad[j] += (1.0 - h[j-s])
        for j in range(0, n-s):
            grad[j] -= h[j+s]
    else:
        a = -s
        # F_{-a} = sum_{i=0}^{n-1-a} h[i]*(1-h[i+a])
        # d/dh[j]: (1-h[j+a]) if j<=n-1-a, minus h[j-a] if j>=a
        for j in range(0, n-a):
            grad[j] += (1.0 - h[j+a])
        for j in range(a, n):
            grad[j] -= h[j-a]
    return grad * dx

def lag_jacobian(h):
    """Full Jacobian matrix of shape (2n-1, n)."""
    n = len(h)
    J = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        s = r - (n-1)
        J[r] = lag_grad_row(h, s)
    return J

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

    if abs(res) > 1e-12:

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

    sorted_idx = np.argsort(scores)[::-1]

    A_ub[:, :n] = G

    b_ub = np.full(n_lags, -scores + radius * 0.01)
    for k in active_set:
        b_ub[k] = -scores[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def _smooth_objective(h, beta, deadline, state):
    """Log-sum-exp smoothing of max lag value."""
    if time.time() > deadline:
        raise TimeoutException()
    F = lag_values(h)
    m = F.max()
    Z = np.exp((F - m) / beta)
    val = m + beta * np.log(Z.sum())
    # Gradient
    P = Z / Z.sum()
    G = lag_jacobian(h)
    grad = G.T @ P
    return val, grad

def _smooth_solve(h0, beta, deadline):
    """Solve smoothed problem with SLSQP."""
    n = len(h0)
    target = n / 2.0
    
    def obj_gradient(x):
        h = np.clip(x, 0, 1)  # SLSQP might push outside bounds slightly
        val, grad = _smooth_objective(h, beta, deadline, None)
        return val, grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj_gradient, h0, method='SLSQP', jac=True,
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': 40, 'ftol': 1e-10})
        h_new = project_box_sum(np.clip(res.x, 0, 1), target)
        return h_new
    except TimeoutException:

    except Exception:

    for r in range(2 * n - 1):
        s = r - (n - 1)

                J[r, :n-s] -= dx * h[s:]
        else:

            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = f_obj(c1)
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = f_obj(c2)
    t_opt = (a + b) / 2
    f_opt = f_obj(t_opt)
    # Also check endpoints
    f_lo = f_obj(t_lo)
    f_hi = f_obj(t_hi)
    if f_lo < f_opt:
        t_opt, f_opt = t_lo, f_lo
    if f_hi < f_opt:
        t_opt, f_opt = t_hi, f_hi
    return t_opt, f_opt
