def _project(v, target):

def _lp_step(h, radius, penalty_threshold=0.998, penalty_w=0.03):

    s_max = float(np.max(scores))

    G = np.zeros((2*n-1, n))

            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]

            m = n - g

    return None

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = trust_radius
    t_lo = -trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if 0.0 >= t_lo and 0.0 <= t_hi:
        f0 = float(np.max(S0)) * dx
        if f0 < opt_val:
            opt_t, opt_val = 0.0, f0
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lse_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(np.clip(v, -600.0, 0.0))
    w = exp_v / (float(np.sum(exp_v)) + 1e-30)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad, s_max

def _smooth_opt(h, deadline, beta_start=25.0, beta_end=600.0, lr_init=0.035):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    lr = lr_init
    stall = 0
    t_start = time.time()
    t_total = max(deadline - t_start, 1.0)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / t_total)
        beta = beta_start * (beta_end / beta_start) ** frac
        grad, _ = _lse_grad(h, dx, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.82 * mom + 0.18 * grad
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            break
        mom /= mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.04, 0.45)
        else:
            stall += 1
            lr *= 0.8
            if stall > 60:
                break
    return best_h, best_c5

def _lp_opt(h, deadline, radius_init=0.22, penalty_thresh=0.997, penalty_w=0.02):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    while time.time() < deadline:
        if stall > 40:
            stall = 0
            radius = best_radius * 0.5
            if radius < 0.01:
                break
            continue
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        G = _grad_matrix(h)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = -1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores.copy()
        near = scores >= s_max * penalty_thresh
        b_ub[near] -= penalty_w
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            if lo > hi:
                lo = hi = 0.0
            bounds.append((lo, hi))
        bounds.append((0.0, None))
        p = None
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        if p is None or np.max(np.abs(p)) < 1e-12:
            stall += 1
            radius *= 0.78
            if radius < 0.005:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.07, 0.5)
        else:
            stall += 1
            radius *= 0.72
            if stall > 30 and radius < 0.05:
                radius = best_radius * 0.4
                stall = 0
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=200, random_pairs=False, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    no_imp_passes = 0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if random_pairs and rng is not None and n > 40:
            pairs = set()
            n_attempts = min(2000, n * n // 4)
            for _ in range(n_attempts):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and 1e-8 < h[i] < 1.0 - 1e-8 and 1e-8 < h[j] < 1.0 - 1e-8:
                    pairs.add((min(i, j), max(i, j)))
        else:
            interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
            pairs = set()
            for ii in range(len(interior)):
                i = interior[ii]
                for jj in range(ii + 1, len(interior)):
                    j = interior[jj]
                    pairs.add((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if improved:
            no_imp_passes = 0
        else:
            no_imp_passes += 1
            if no_imp_passes >= 3:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _try_symmetries(h, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    candidates = [h.copy(), h[::-1].copy()]
    for shift in range(1, max(2, n // 8)):
        candidates.append(np.roll(h, shift).copy())
        candidates.append(np.roll(h, -shift).copy())
    for hc in candidates:
        if time.time() > deadline:
            break
        hc = _project(hc, n / 2.0)
        c5 = _exact_c5(hc)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = hc.copy()
    return best_h, best_c5

def _triple_move(h, deadline, max_triples=500, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if len(interior) < 3:
        return best_h, best_c5
    tried = 0
    for _ in range(max_triples * 3):
        if time.time() > deadline or tried >= max_triples:
            break
        i = interior[int(rng.integers(0, len(interior)))]
        j = interior[int(rng.integers(0, len(interior)))]
        k = interior[int(rng.integers(0, len(interior)))]
        if len({i, j, k}) < 3:
            continue
        tried += 1
        for sign_i in [1, -1]:
            d = np.zeros(n)
            d[i] = sign_i
            d[j] = -sign_i * 0.5
            d[k] = -sign_i * 0.5
            alpha, c5_try = _line_search(h, d, dx, 0.3)
            if c5_try < best_c5 - 1e-12:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
    return best_h, best_c5
