import numpy as np

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -2.0, 2.0)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pnorm_value_grad(h, q):
    n = len(h)
    dx = 2.0 / n
    F = _scores(h)
    L = 2 * n - 1
    F_max = float(F.max())
    if F_max < 1e-300:
        return 0.0, np.zeros(n)
    r = F / F_max
    r = np.maximum(r, 0.0)
    if q <= 1.0:
        r_q = np.ones(L)
        sum_r_q = float(L)
        r_qm1 = np.ones(L)
    else:
        r_q = r ** q
        sum_r_q = float(r_q.sum())
        if sum_r_q < 1e-300:
            r_qm1 = np.zeros(L)
            r_qm1[int(np.argmax(F))] = 1.0
        else:
            r_qm1 = r ** (q - 1)
    value = F_max * (sum_r_q ** (1.0 / q))
    if q == 1:
        scale = 1.0 / L
    else:
        scale = sum_r_q ** (1.0 / q - 1.0)
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = float(r_qm1[k + n - 1])
        if w < 1e-25:
            continue
        j_lo = max(0, k)
        j_hi = min(n - 1, k + n - 1)
        if j_lo <= j_hi:
            idx1 = np.arange(j_lo - k, j_hi - k + 1)
            grad[j_lo:j_hi + 1] += w * dx * (1.0 - h[idx1])
        j_lo2 = max(0, -k)
        j_hi2 = min(n - 1, n - 1 - k)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2 + k, j_hi2 + k + 1)
            grad[j_lo2:j_hi2 + 1] -= w * dx * h[idx2]
    grad *= scale
    return value, grad

def _slsqp_run(h0, q, deadline, maxiter=150):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    def obj(x):
        v, _ = _pnorm_value_grad(x, q)
        return v

    def jac(x):
        _, g = _pnorm_value_grad(x, q)
        return g

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    if time.time() > deadline:
        return best_h, best_c5
    try:
        res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if time.time() > deadline:
            return best_h, best_c5
        h_new = _project(res.x, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c5

def _resample(h, n_new):

        return h.copy()

    h_new = np.interp(x_new, x_old, h)

            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_dl = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    q_sched1 = [8, 16, 32, 64, 128]
    dl1 = min(hard_dl, t0 + budget_s * 0.12)
    best_h, best_c5 = _continuation(best_h, q_sched1, dl1, method='slsqp')

    if time.time() < hard_dl:
        dl1b = min(hard_dl, time.time() + budget_s * 0.04)
        best_h, best_c5 = _continuation(best_h, [64, 128, 256], dl1b, method='slsqp')

    if time.time() < hard_dl:
        dl2 = min(hard_dl, time.time() + budget_s * 0.06)
        h_ar, c5_ar = _active_refine(best_h, dl2, top_k=5)
        if c5_ar < best_c5:
            best_c5 = c5_ar
            best_h = h_ar

    # Multi-start exploration
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5554878993]
    n_tries = [128, 160, 256, 192, 128]
    dl3 = t0 + budget_s * 0.55
    for i in range(5):
        if time.time() > dl3:
            break
        n_t = n_tries[i % len(n_tries)]
        if i < 2:
            h_init = best_h.copy()
            if n_t != best_n:
                h_init = _resample(h_init, n_t)
        else:
            h_init = _quasi_init(n_t, alphas[i % len(alphas)], rng)
        sub_dl = min(dl3, time.time() + budget_s * 0.05)
        q_list = [16, 32, 64, 128]
        h_try, c5_try = _continuation(h_init, q_list, sub_dl, method='slsqp')
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_t
            sub_dl2 = min(dl3, time.time() + budget_s * 0.02)
            h_ar2, c5_ar2 = _active_refine(best_h, sub_dl2, top_k=4)
            if c5_ar2 < best_c5:
                best_c5 = c5_ar2
                best_h = h_ar2
        else:
            sub_dl2 = min(dl3, time.time() + budget_s * 0.025)
            h_pg, c5_pg = _proj_grad_run(h_try, 64, sub_dl2, n_iter=100)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg
                best_n = len(h_try)
                sub_dl3 = min(dl3, time.time() + budget_s * 0.02)
                h_ar3, c5_ar3 = _active_refine(best_h, sub_dl3, top_k=4)
                if c5_ar3 < best_c5:
                    best_c5 = c5_ar3
                    best_h = h_ar3

    # More diverse sinusoidal starts
    dl4 = t0 + budget_s * 0.72
    for i in range(4):
        if time.time() > dl4:
            break
        k_mode = 3 + i
        phase = rng.uniform(0, 2 * np.pi)
        n_t = 128 if i % 2 == 0 else 160
        h_init = _sinusoid_init(n_t, k_mode, phase, rng)
        sub_dl = min(dl4, time.time() + budget_s * 0.035)
        h_try, c5_try = _continuation(h_init, [16, 32, 64, 128], sub_dl, method='slsqp')
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_t

    # High-q refinement on best
    if time.time() < hard_dl:
        dl5 = min(hard_dl, time.time() + budget_s * 0.08)
        best_h, best_c5 = _continuation(best_h, [128, 192, 256, 512], dl5, method='slsqp')

    # Active set refinement round 2
    if time.time() < hard_dl:
        dl6 = min(hard_dl, time.time() + budget_s * 0.06)
        h_ar, c5_ar = _active_refine(best_h, dl6, top_k=4)
        if c5_ar < best_c5:
            best_c5 = c5_ar
            best_h = h_ar

    # Pair polish
    if time.time() < hard_dl:
        dl7 = min(hard_dl, time.time() + budget_s * 0.10)
        h_pp, c5_pp = _pair_polish(best_h, dl7, n_passes=80)
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp

    # Final SLSQP passes with very high q
    if time.time() < hard_dl:
        dl8 = min(hard_dl, time.time() + budget_s * 0.05)
        for qf in [256, 512]:
            if time.time() > dl8:
                break
            h_try, c5_try = _slsqp_run(best_h, qf, dl8, maxiter=80)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Perturbation restarts
    if time.time() < hard_dl - 10:
        dl9 = min(hard_dl, time.time() + budget_s * 0.06)
        scale = 0.02
        for attempt in range(60):
            if time.time() > dl9:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl9, time.time() + budget_s * 0.005)
            h_try, c5_try = _slsqp_run(h_pert, 128, sub_dl, maxiter=50)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.06, scale * 1.15)

    # Final pair polish
    if time.time() < hard_dl - 5:
        dl10 = min(hard_dl, time.time() + budget_s * 0.03)
        h_pp2, c5_pp2 = _pair_polish(best_h, dl10, n_passes=40)
        if c5_pp2 < best_c5:
            best_c5 = c5_pp2
            best_h = h_pp2

    # Final projection and validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    elif final_c5 > best_c5 + 1e-9:
        # projected version is worse, use pre-projection stored value
        best_c5 = best_c5
    return best_h, best_c5, best_n
