def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))

    Fixed version: correct signs for negative lags."""
    n = len(h)

    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, target, n_eval=25):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            a = (1.0 - h[j]) / p[j]
            if a < alpha_max:
                alpha_max = a
        elif p[j] < -1e-14:
            a = h[j] / (-p[j])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(alpha_max, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(n_eval):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_h = _project(h + opt_a * p, target)
    opt_c5 = _exact_c5(opt_h)
    best_a = min(opt_a, c1, c2, key=lambda a: _exact_c5(_project(h + a * p, target)))
    best_c5 = _exact_c5(_project(h + best_a * p, target))
    return best_a, best_c5

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=5000, radius_min=0.002, top_k=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_k=top_k)
        if p is None:
            stall += 1
            radius *= 0.6
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search(h, p, target)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.65
            if radius < radius_min:
                break
    return best_h, best_c5

def _block_descent(h, deadline, block_size=8, n_cycles=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for start in range(0, n, block_size):
            if time.time() > deadline:
                break
            end = min(start + block_size, n)
            bsz = end - start
            block_sum = h[start:end].sum()
            for _bs_iter in range(20):
                scores = _scores(h)
                G = _grad_matrix(h)
                G_sub = G[:, start:end]
                blk = h[start:end]
                bsum = float(blk.sum())
                best_dir = None
                best_obj = best_c5
                for trial in range(4):
                    if trial == 0:
                        direction = -G_sub.T @ (scores / max(np.max(scores), 1e-10))
                    elif trial == 1:
                        direction = -np.argmax(scores) * np.ones(bsz) + np.arange(bsz)
                        direction = direction - direction.mean()
                    elif trial == 2:
                        direction = np.sin(np.pi * np.arange(bsz) / bsz)
                    else:
                        direction = np.random.randn(bsz)
                    direction -= direction.mean()
                    dn = np.linalg.norm(direction)
                    if dn < 1e-10:
                        continue
                    direction /= dn
                    a_max_p = min([(1.0 - blk[i]) / direction[i] for i in range(bsz) if direction[i] > 1e-10], default=1.0)
                    a_max_m = min([(-blk[i]) / direction[i] for i in range(bsz) if direction[i] < -1e-10], default=1.0)
                    a_max = min(a_max_p, a_max_m, 0.5)
                    if a_max < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[start:end] = np.clip(blk + a_max * 0.5 * direction, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_obj - 1e-12:
                        best_obj = c5_try
                        best_dir = direction
                        best_a = a_max * 0.5
                if best_dir is not None and best_obj < best_c5 - 1e-14:
                    h[start:end] = np.clip(h[start:end] + best_a * best_dir, 0, 1)
                    h = _project(h, target)
                    best_c5 = _exact_c5(h)
                    best_h = h.copy()
                    improved = True
                else:
                    break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0 < 0.5).astype(float)
    return _project(h, n / 2.0)

def _adaptive_lp(h, deadline, radius_init=0.12, max_iters=3000, radius_min=0.001):
    """LP with adaptive top_k: start with all, switch to top-K when stuck."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    top_k = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_k=top_k)
        if p is None:
            stall += 1
            if stall == 3 and top_k is None:
                top_k = max(20, len(scores) // 10)
                stall = 0
                continue
            radius *= 0.6
            if stall > 8 and top_k is not None:
                top_k = max(10, top_k // 2)
                stall = 0
                continue
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search(h, p, target)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5

def _coordinate_polish(h, deadline, n_cycles=2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            lo_v = max(0.0, h[j] - 0.1)
            hi_v = min(1.0, h[j] + 0.1)
            for _ in range(25):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _gradient_descent(h, deadline, n_steps=200, step_init=0.03):
    """Subgradient method with projection for the max objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    momentum = np.zeros(n)
    beta_mom = 0.85
    for it in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = np.max(scores)
        active = np.where(scores >= max_s - 1e-6)[0]
        G = _grad_matrix(h)
        grad = G[active, :].mean(axis=0)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        momentum = beta_mom * momentum - step * grad
        h_new = _project(h + momentum, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.02, 0.1)
        else:
            momentum *= 0.5
            step *= 0.8
            if step < 1e-7:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Adaptive LP from initial (fixed gradient)
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _adaptive_lp(h_start, dl1, radius_init=0.15, max_iters=5000, radius_min=0.001)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient descent + LP refinement
    if time.time() < t0 + budget_s * 0.20:
        dl2 = t0 + budget_s * 0.20
        h2, c5_2 = _gradient_descent(best_h, dl2, n_steps=150, step_init=0.04)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        if time.time() < t0 + budget_s * 0.24:
            dl2b = t0 + budget_s * 0.24
            h2b, c5_2b = _adaptive_lp(best_h, dl2b, radius_init=0.10, max_iters=3000, radius_min=0.001)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.50:
        dl3 = t0 + budget_s * 0.50
        resolutions = [96, 64, 80, 112, 160, 192, 128, 144]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113,
                  0.6180339887, 0.5, 0.4142135624, 0.7071067812]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl3:
                break
            if best_c5 < 0.3805:
                break
            sub_dl = min(dl3, time.time() + budget_s * 0.035)
            if sub_dl <= time.time():
                break
            h_init = _resample(best_h, n_try)
            h_try, c5_try = _adaptive_lp(h_init, sub_dl, radius_init=0.12, max_iters=2000, radius_min=0.003)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            # Also try quasi-periodic init at this resolution
            if time.time() < dl3:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.02)
                if sub_dl2 > time.time():
                    alpha = alphas[ri % len(alphas)]
                    h_qp = _quasi_init(n_try, alpha)
                    h_qp2, c5_qp = _adaptive_lp(h_qp, sub_dl2, radius_init=0.15, max_iters=1500, radius_min=0.005)
                    if c5_qp < best_c5:
                        best_c5 = c5_qp
                        best_h = h_qp2.copy()
                        best_n = n_try

    # Phase 4: Block descent on best
    if time.time() < t0 + budget_s * 0.60:
        dl4 = t0 + budget_s * 0.60
        h4, c5_4 = _block_descent(best_h, dl4, block_size=8, n_cycles=3)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        # Follow up with LP
        if time.time() < t0 + budget_s * 0.65:
            dl4b = t0 + budget_s * 0.65
            h4b, c5_4b = _adaptive_lp(best_h, dl4b, radius_init=0.08, max_iters=2000, radius_min=0.001)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: Perturbation restarts
    dl5 = t0 + budget_s * 0.80
    scale = 0.03
    for attempt in range(200):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.015)
        if sub_dl <= time.time():
            break
        h_try, c5_try = _adaptive_lp(h_pert, sub_dl, radius_init=0.06, max_iters=800, radius_min=0.003)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 6: Coordinate polish
    if time.time() < t0 + budget_s * 0.92:
        dl6 = t0 + budget_s * 0.92
        hc, c5c = _coordinate_polish(best_h, dl6, n_cycles=3)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 7: Final fine LP
    if time.time() < t0 + budget_s * 0.98:
        dl7 = t0 + budget_s * 0.998
        h_fin, c5_fin = _adaptive_lp(best_h, dl7, radius_init=0.01, max_iters=3000, radius_min=0.0005)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
