import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-7
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-13:
        return None, None
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo = t_min
    hi = t_max
    c1 = hi - gr * (hi - lo)

    hc1 = h.copy()
    hc1[i] += c1
    hc1[j] -= c1
    f1 = _exact_c5(hc1)

    hc2[i] += c2

            c2 = c1

            hc1[j] = h[j] - c1
            f1 = _exact_c5(hc1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            hc2[i] = h[i] + c2
            hc2[j] = h[j] - c2

    h_opt = h.copy()
    h_opt[i] += t_opt

    f_opt = _exact_c5(h_opt)

def _weighted_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp((scores - max_s) / tau)
    sw = np.sum(w)
    if sw < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-15:
            continue
        d = r - (n - 1)
        if d == 0:
            grad += wr * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += wr * dx * (-h[d:])
            grad[d:] += wr * dx * (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            grad[:m] += wr * dx * (1.0 - h[g:])
            grad[g:] += wr * dx * (-h[:m])
    grad -= grad.mean()
    return grad


def _grad_line_search(h, grad, deadline):

    pn = np.linalg.norm(p)

    hi = alpha_max

    c2 = lo + gr * (hi - lo)
