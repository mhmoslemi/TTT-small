def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

            lo = mid
        else:
            hi = mid

def _line_search(h, d, dx, trust_radius):

    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15

        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):

    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)

def _active_lp_direction(h, scores, dx, radius, n_active=8):
    """Compute LP direction using only active lags."""
    n = len(h)
    # Find active lags (those closest to maximum)
    M = float(np.max(scores))
    idx = np.argsort(-scores)[:n_active]
    
    # Build constraint matrix for active lags only
    # For each active lag, compute the gradient of the correlation w.r.t. h
    # The correlation at lag k (q = n-1+k) is:
    #   if k >= 0: sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])
    #   if k < 0: sum_{i=0}^{n+k-1} h[i-k]*(1-h[i])  [where k<0]
    
    # Gradient of corr[q] w.r.t. h:
    # For lag k = q - (n-1):
    #   if k >= 0: 
    #     d/dh[i] = dx*(1-h[i+k]) for i in [0, n-k-1]
    #     d/dh[i+k] = -dx*h[i] for i in [0, n-k-1]
    #   if k < 0, d = -k:
    #     d/dh[i] = -dx*h[i+d] for i in [0, n-d-1]
    #     d/dh[i+d] = dx*(1-h[i]) for i in [0, n-d-1]
    
    # Build the LP
    # Variables: d[0..n-1], u (reduction amount)
    # Minimize -u
    # Subject to: grad_q · d - u <= M - scores[q] for each active lag q
    #             sum(d) = 0
    #             -min(d, h[i], 1-h[i]) <= d[i] <= min(radius, ...)
    
    A_ub = np.zeros((len(idx), n + 1))
    b_ub = np.zeros(len(idx))
    for ii, q in enumerate(idx):
        k = q - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            m = n - k
            for i in range(m):
                grad[i] += dx * (1.0 - h[i + k])
                grad[i + k] -= dx * h[i]
        else:
            d = -k
            m = n - d
            for i in range(m):
                grad[i] -= dx * h[i + d]
                grad[i + d] += dx * (1.0 - h[i])
        A_ub[ii, :n] = grad
        A_ub[ii, n] = -1.0
        b_ub[ii] = M - float(scores[q])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]

    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -u = maximize u
    
    try:

def _line_search(h, d, dx, trust_radius):
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(35):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction_active(h, scores, dx, radius, n_active=10):
    n = len(h)
    n_lags = 2 * n - 1
    M = float(np.max(scores))
    # Select active lags
    idx = np.argsort(-scores)[:min(n_active, n_lags)]
    
    A_ub = np.zeros((len(idx), n + 1))
    b_ub = np.zeros(len(idx))
    for ii, q in enumerate(idx):
        k = int(q) - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            m = n - k
            for i in range(m):
                grad[i] += dx * (1.0 - h[i + k])
                grad[i + k] -= dx * h[i]
        else:
            d_ = -k
            m = n - d_
            for i in range(m):
                grad[i] -= dx * h[i + d_]
                grad[i + d_] += dx * (1.0 - h[i])
        A_ub[ii, :n] = grad
        A_ub[ii, n] = -1.0
        b_ub[ii] = M - float(scores[q])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _binary_local(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    
    # Start from rounded binary
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    
    T = 0.003
    T_min = 0.000001
    cooling = 0.9999
    patience = 0
    max_patience = 5000
    
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min:
                break
            T *= 0.9
            patience = 0
        
        # Steepest descent: try all swaps
        ones_pos = np.where(h > 0.5)[0]
        zeros_pos = np.where(h <= 0.5)[0]
        
        best_delta = 0
        best_i, best_j = -1, -1
        
        S0 = np.correlate(h, 1.0 - h, mode='full')
        
        for i in ones_pos:
            for j in zeros_pos:
                # Compute effect of moving 1 from i to j
                # h' = h - e_i + e_j
                # delta_corr = corr(h', 1-h') - corr(h, 1-h)
                # = (corr(δ, 1-h) - corr(h, δ) - corr(δ, δ)) where δ = -e_i + e_j
                # For efficiency, compute directly
                h[i] = 0.0
                h[j] = 1.0
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = float(np.max(new_corr)) * dx
                delta = new_max - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h[i] = 1.0
                h[j] = 0.0
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_c5 = best_delta + cur_c5  # Actually need to recompute
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            T = max(T_min, T * cooling)
        else:
            # No improving swap found; try random swap with SA
            if T > T_min:
                i = ones_pos[rng.integers(len(ones_pos))]
                j = zeros_pos[rng.integers(len(zeros_pos))]
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = _exact_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                else:
                    h[i] = 1.0
                    h[j] = 0.0
                    patience += 1
                T = max(T_min, T * cooling)
            else:
                break
    
    return best_h, best_c5
