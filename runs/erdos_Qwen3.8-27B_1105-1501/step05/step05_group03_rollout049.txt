            m = n - g

                    h_init = _quasi_init(n_try, alpha, rng, noise=0.02)
                sub_dl = min(dl3, time.time() + budget_s * 0.025)
                h_try, c5_try = _lp_optimize(h_init, sub_dl, radius_init=0.15, max_iters=3000, radius_min=0.003)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 4: Re-optimize best with subgrad + LP
    if time.time() < t0 + budget_s * 0.52:
        dl4 = t0 + budget_s * 0.52
        h4, c5_4 = _subgrad_optimize(best_h, dl4, k_active=4, n_steps=400, step_init=0.03)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        if time.time() < t0 + budget_s * 0.58:
            dl4b = t0 + budget_s * 0.58
            h4b, c5_4b = _lp_optimize(best_h, dl4b, radius_init=0.10, max_iters=4000, radius_min=0.0005)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: Evolutionary search
    if time.time() < t0 + budget_s * 0.68:
        dl5 = t0 + budget_s * 0.68
        h_f, c5_f = _evolutionary_search(best_h, best_n, dl5, rng, n_pop=24, n_gen=150, K=14)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if time.time() < t0 + budget_s * 0.72:
            sub_dl = min(t0 + budget_s * 0.72, time.time() + budget_s * 0.02)
            h_fp, c5_fp = _lp_optimize(best_h, sub_dl, radius_init=0.08, max_iters=2000, radius_min=0.002)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 6: Binary SA at different resolutions
    if time.time() < t0 + budget_s * 0.80:
        for n_bin in [64, 96, 128]:
            if time.time() > t0 + budget_s * 0.80:
                break
            sub_dl = min(t0 + budget_s * 0.80, time.time() + budget_s * 0.02)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=12000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(t0 + budget_s * 0.80, time.time() + budget_s * 0.015)
                h_br, c5_br = _lp_optimize(h_b, sub_dl2, radius_init=0.10, max_iters=1500, radius_min=0.003)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()

    # Phase 7: Perturbation restarts
    dl7 = t0 + budget_s * 0.90
    scale = 0.02
    for attempt in range(400):
        if time.time() > dl7:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.01)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.05, max_iters=400, radius_min=0.005)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 8: Coordinate polish
    if time.time() < t0 + budget_s * 0.95:
        dl8 = t0 + budget_s * 0.95
        hc, c5c = _coordinate_polish(best_h, dl8, n_cycles=3)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 9: Pair polish
    if time.time() < t0 + budget_s * 0.97:
        dl9 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, dl9, n_passes=5)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 10: Final fine LP
    if time.time() < t0 + budget_s * 0.995:
        dl10 = t0 + budget_s * 0.998
        h_fin, c5_fin = _lp_optimize(best_h, dl10, radius_init=0.01, max_iters=3000, radius_min=0.0003)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
