def _sigma(t):
    """Numerically stable sigmoid."""
    out = np.empty_like(t)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def _sigma_prime(t):
    s = _sigma(t)
    return s * (1.0 - s)

def _find_mu(q, target, R=20.0):
    """Find mu such that sum(sigma(q - mu)) = target."""
    lo, hi = -R, R
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(_sigma(q - mid))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target):
    mu = _find_mu(q, target)
    return _sigma(q - mu)

def _lp_descent(h, deadline, radius=0.2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    cur_radius = radius
    no_improve = 0
    
    while time.time() < deadline and no_improve < 200:
        scores = _scores(h)  # (2n-1,) array
        G = _grad_matrix(h)  # (2n-1, n) matrix
        n_lags = len(scores)
        
        # LP: min t s.t. G@p - t*1 <= -scores, sum(p)=0, bounds on p
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-cur_radius, -h[j])
            hi = min(cur_radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))  # t is free
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                no_improve += 1
                cur_radius *= 0.9
                if cur_radius < 1e-6: break
                continue
        except:
            no_improve += 1
            cur_radius *= 0.9
            if cur_radius < 1e-6: break
            continue
        
        p = res.x[:n]
        if np.max(np.abs(p)) < 1e-14:
            no_improve += 1
            cur_radius *= 0.85
            if cur_radius < 1e-6: break
            continue
        
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.999
        
        # Golden section line search
        best_alpha, best_val = _line_search(h, p, alpha_max)
        new_h = _balance(h + best_alpha * p, target)
        new_c5 = _exact_c5(new_h)
        
        if new_c5 < cur_c5 - 1e-14:
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
            cur_radius = min(cur_radius * 1.05, 0.5)
        else:
            no_improve += 1
            cur_radius *= 0.85
            if no_improve > 50 and cur_radius < 0.05:
                cur_radius = 0.15  # restart with larger radius
    
    return best_h, best_c5

def _lsm_bfgs(h_init, deadline, beta_start=30, beta_end=500, n_phases=6):

    target = n / 2.0

    best_h = h.copy()

    # Initialize q from h
    eps = 1e-6
    h_clipped = np.clip(h, eps, 1-eps)
    q = np.log(h_clipped / (1.0 - h_clipped))
    
    for phase in range(n_phases):
        if time.time() > deadline: break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Define objective and gradient in q-space
        def obj_q(q_vec):
            h_vec = _q_to_h(q_vec, target)
            scores = _scores(h_vec)
            M = np.max(scores)
            shifted = beta * (scores - M)
            shifted = np.clip(shifted, -500, 0)
            return M + np.log(np.mean(np.exp(shifted))) / beta
        
        def grad_q(q_vec):
            h_vec = _q_to_h(q_vec, target)
            scores = _scores(h_vec)
            M = np.max(scores)
            shifted = beta * (scores - M)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            sw = np.sum(w)
            if sw < 1e-300:
                w = np.zeros(len(scores))
                w[int(np.argmax(scores))] = 1.0
            else:
                w /= sw
            G = _grad_matrix(h_vec)
            g_h = G.T @ w  # gradient wrt h
            # Transform to q-space
            mu = _find_mu(q_vec, target)
            s = _sigma_prime(q_vec - mu)
            S = np.sum(s)
            if S < 1e-30:
                return np.zeros(n)
            rho = np.dot(g_h, s) / S
            return s * (g_h - rho)
        
        try:
            res = minimize(obj_q, q, jac=grad_q, method='L-BFGS-B',
                          bounds=[(-14, 14)] * n,
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-14})
            if res.x is not None:

                    best_h = h_new.copy()
                    h = h_new

            pass
    
    return best_h, best_c5
