def _build_basis(n, n_fourier, n_blocks, rng):
    """Build a sum-zero basis matrix of shape (n, q)."""
    x = np.arange(n) / n
    cols = []
    
    # Fourier modes (sum-zero by construction for k>=1)
    for k in range(1, n_fourier + 1):
        cols.append(np.cos(2 * np.pi * k * x))
        cols.append(np.sin(2 * np.pi * k * x))
    
    # Rectangular block pairs (sum-zero)
    used_blocks = 0
    for _ in range(n_blocks):
        r = rng.integers(2, max(3, n // 4))
        a = rng.integers(0, max(1, n - 2 * r + 1))
        if a + 2 * r > n:
            continue
        col = np.zeros(n)
        col[a:a+r] = 1.0 / np.sqrt(r)
        col[a+r:a+2*r] = -1.0 / np.sqrt(r)
        cols.append(col)
        used_blocks += 1
    
    B = np.column_stack(cols)
    # Ensure each column is sum-zero
    for j in range(B.shape[1]):
        B[:, j] -= B[:, j].mean()
    return B

def _differential_evolution(F, x0, bounds, n_pop, n_gen, F_mut, CR, rng, deadline):
    n_dim = len(x0)
    n_pop_eff = max(n_pop, 2 * n_dim + 1)
    # Initialize population
    pop = np.zeros((n_pop_eff, n_dim))
    pop[0] = x0
    for i in range(1, n_pop_eff):
        pop[i] = rng.uniform(bounds[0], bounds[1], n_dim)
    
    fit = np.array([F(x) for x in pop])
    best_idx = np.argmin(fit)
    best_x = pop[best_idx].copy()
    best_fit = fit[best_idx]
    
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(n_pop_eff):
            # Select three distinct individuals
            while True:
                a = rng.integers(n_pop_eff)
                b = rng.integers(n_pop_eff)
                c = rng.integers(n_pop_eff)
                if len({a, b, c, i}) == 4:
                    break
            v = pop[a] + F_mut * (pop[b] - pop[c])
            v = np.clip(v, bounds[0], bounds[1])
            
            # Binomial crossover
            trial = pop[i].copy()
            j0 = rng.integers(n_dim)
            for j in range(n_dim):
                if j == j0 or rng.random() < CR:
                    trial[j] = v[j]
            
            trial = np.clip(trial, bounds[0], bounds[1])
            f_trial = F(trial)
            if f_trial <= fit[i]:
                pop[i] = trial
                fit[i] = f_trial
                if f_trial < best_fit:
                    best_fit = f_trial
                    best_x = trial.copy()
    
    return best_x, best_fit

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=5000, radius_min=0.001):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None:
            stall += 1
            radius *= 0.6
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search(h, p, target)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.65
            if radius < radius_min:
                break
    return best_h, best_c5

def _build_subspace_basis(n, n_fourier, n_blocks, rng):
    x = np.arange(n) / n
    cols = []
    for k in range(1, n_fourier + 1):
        cols.append(np.cos(2 * np.pi * k * x))
        cols.append(np.sin(2 * np.pi * k * x))
    for _ in range(n_blocks):
        r = rng.integers(2, max(3, n // 4 + 1))
        a = rng.integers(0, max(1, n - 2 * r + 1))
        if a + 2 * r > n:
            continue
        col = np.zeros(n)
        col[a:a+r] = 1.0 / np.sqrt(r)
        col[a+r:a+2*r] = -1.0 / np.sqrt(r)
        cols.append(col)
    if not cols:
        return np.zeros((n, 1))
    B = np.column_stack(cols)
    for j in range(B.shape[1]):
        B[:, j] -= B[:, j].mean()
    return B

def _subspace_eval(y, B, target):
    raw = 0.5 + B @ y
    h = _project(raw, target)
    return _exact_c5(h)

def _differential_evolution(F, x0, bounds_lo, bounds_hi, n_pop, n_gen, F_mut, CR, rng, deadline):
    n_dim = len(x0)
    n_pop_eff = max(n_pop, n_dim + 4)
    # Ensure at least 2*n_dim+1 for DE
    n_pop_eff = max(n_pop_eff, 2 * n_dim + 1)
    
    pop = np.zeros((n_pop_eff, n_dim))
    pop[0] = x0
    for i in range(1, n_pop_eff):
        pop[i] = rng.uniform(bounds_lo, bounds_hi)
    
    fit = np.zeros(n_pop_eff)
    for i in range(n_pop_eff):
        fit[i] = F(pop[i])
    
    best_idx = int(np.argmin(fit))
    best_x = pop[best_idx].copy()
    best_fit = fit[best_idx]
    
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(n_pop_eff):
            # Pick 3 distinct from pop, different from i
            candidates = [j for j in range(n_pop_eff) if j != i]
            a, b, c = rng.choice(candidates, size=3, replace=False)
            v = pop[a] + F_mut * (pop[b] - pop[c])
            v = np.clip(v, bounds_lo, bounds_hi)
            
            trial = pop[i].copy()
            j0 = rng.integers(n_dim)
            for j in range(n_dim):
                if j == j0 or rng.random() < CR:
                    trial[j] = v[j]
            
            f_trial = F(trial)
            if f_trial <= fit[i]:
                pop[i] = trial
                fit[i] = f_trial
                if f_trial < best_fit:
                    best_fit = f_trial
                    best_x = trial.copy()
    return best_x, best_fit

def _subspace_optimize(h, deadline, rng, n_fourier=6, n_blocks=30, n_pop=40, n_gen=300, F_mut=0.7, CR=0.8):
    n = len(h)
    target = n / 2.0
    B = _build_subspace_basis(n, n_fourier, n_blocks, rng)
    q = B.shape[1]
    
    # Initialize y from least squares fit
    v = h - 0.5
    y0 = np.linalg.lstsq(B, v, rcond=None)[0]
    
    bounds_lo = -3.0 * np.ones(q)
    bounds_hi = 3.0 * np.ones(q)
    
    def F(y):
        return _subspace_eval(y, B, target)
    
    y_best, c5_best = _differential_evolution(F, y0, bounds_lo, bounds_hi,
                                               n_pop, n_gen, F_mut, CR, rng, deadline)
    h_best = _project(0.5 + B @ y_best, target)
    c5_final = _exact_c5(h_best)
    return h_best, c5_final

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coordinate_polish(h, deadline, n_cycles=2):

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_arr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, target, max_evals=30):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(max_evals):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    mid = (lo + hi) / 2.0
    f_mid = _exact_c5(_project(h + mid * p, target))
    best_a = min((f1, c1), (f2, c2), (f_mid, mid), key=lambda x: x[0])
    return best_a[0], best_a[1]

            stall = 0

        else:
            stall += 1
            radius *= 0.65
            if radius < radius_min:
                break
    return best_h, best_c5

def _build_basis(n, n_fourier, n_blocks, rng):
    x = np.arange(n) / n
    cols = []
    for k in range(1, n_fourier + 1):
        cols.append(np.cos(2.0 * np.pi * k * x))
        cols.append(np.sin(2.0 * np.pi * k * x))
    for _ in range(n_blocks):
        r = int(rng.integers(2, max(3, n // 4 + 1)))
        a = int(rng.integers(0, max(1, n - 2 * r + 1)))
        if a + 2 * r > n:
            continue
        col = np.zeros(n)
        col[a:a + r] = 1.0 / np.sqrt(r)
        col[a + r:a + 2 * r] = -1.0 / np.sqrt(r)
        cols.append(col)
    if not cols:
        col = np.zeros(n)
        col[n // 4:n // 2] = 1.0
        col[n // 2:3 * n // 4] = -1.0
        cols.append(col)
    B = np.column_stack(cols)
    for j in range(B.shape[1]):
        B[:, j] -= B[:, j].mean()
    return B

def _de_search(F, x0, lo, hi, n_pop, n_gen, Fm, CR, rng, deadline):
    q = len(x0)
    n_pop_eff = max(n_pop, 2 * q + 1)
    pop = np.zeros((n_pop_eff, q))
    pop[0] = x0
    for i in range(1, n_pop_eff):
        pop[i] = rng.uniform(lo, hi)
    fit = np.array([F(pop[i]) for i in range(n_pop_eff)])
    best_i = int(np.argmin(fit))
    best_x = pop[best_i].copy()
    best_f = fit[best_i]
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(n_pop_eff):
            mask = np.ones(n_pop_eff, dtype=bool)
            mask[i] = False
            idxs = np.where(mask)[0]
            a, b, c = rng.choice(idxs, size=3, replace=False)
            v = pop[a] + Fm * (pop[b] - pop[c])
            v = np.clip(v, lo, hi)
            trial = pop[i].copy()
            j0 = int(rng.integers(q))
            for j in range(q):
                if j == j0 or rng.random() < CR:
                    trial[j] = v[j]
            f_t = F(trial)
            if f_t <= fit[i]:
                pop[i] = trial
                fit[i] = f_t
                if f_t < best_f:
                    best_f = f_t
                    best_x = trial.copy()
    return best_x, best_f
