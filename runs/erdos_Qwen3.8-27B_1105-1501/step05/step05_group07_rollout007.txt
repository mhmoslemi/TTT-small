                    # term: h[i-d]*(1-h[i])
                    affected_indices.append(r)
                    affected_coeffs.append(-h[i-d] * dx)
                    affected_consts.append(cur_scores[r] - h[i-d] * (1 - old_hi) * dx)

                non_affected = cur_scores.copy()
                for r in affected_indices:
                    non_affected[r] = -1
                non_affected[n-1] = -1  # self is handled separately

                a_arr = np.array(affected_coeffs)
                b_arr = np.array(affected_consts)

                linear_scores = a_arr[:, None] * grid[None, :] + b_arr[:, None]  # (n_aff, n_grid)

                all_scores = np.maximum(self_scores_2d, non_affected_max)
                max_scores = np.max(all_scores, axis=0)  # (n_grid,)
                
                best_j = np.argmin(max_scores)
                best_t = grid[best_j]
                best_val = max_scores[best_j]
                
                if best_val < cur_scores.max() - 1e-13:
                    h[i] = best_t
                    improved = True
            
            # Project
            h = _project(h, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if not improved:
                break
        
        return best_h, best_c5

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    n_samp = 16
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _local_optimize_lp(h, deadline, radius_init=0.25, max_iters=5000, stall_limit=150):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.85
            if stall > 40 and radius < 0.02:
                radius = 0.1
    return best_h, best_c5

def _coord_descent(h, deadline, n_sweeps=200, n_grid=100):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    grid = np.linspace(0, 1, n_grid)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_hi = h[i]
            cur_scores = _scores(h)
            cur_max = cur_scores.max()
            self_base = cur_scores[n-1] - old_hi*(1.0-old_hi)*dx
            self_a = -dx
            self_b = dx
            self_c = self_base
            
            aff_idx = []
            aff_coeff = []
            aff_const = []
            
            for d in range(1, n - i):
                r = n - 1 + d
                aff_idx.append(r)
                aff_coeff.append((1.0 - h[i+d]) * dx)
                aff_const.append(cur_scores[r] - old_hi * (1.0 - h[i+d]) * dx)
            
            for d in range(1, i + 1):
                r = n - 1 - d
                aff_idx.append(r)
                aff_coeff.append(-h[i-d] * dx)
                aff_const.append(cur_scores[r] - h[i-d] * (1.0 - old_hi) * dx)
            
            if len(aff_idx) == 0:
                a_arr = np.zeros(1)
                b_arr = np.array([0.0])
            else:
                a_arr = np.array(aff_coeff)
                b_arr = np.array(aff_const)
                aff_arr = np.array(aff_idx)
            
            non_aff = cur_scores.copy()
            non_aff[n-1] = -1e10
            if len(aff_idx) > 0:
                non_aff[aff_arr] = -1e10
            non_aff_max = float(np.max(non_aff))
            
            lin = a_arr[:, None] * grid[None, :] + b_arr[:, None]
            self_s = self_a * grid**2 + self_b * grid + self_c
            combined = np.maximum(lin, self_s[None, :])
            combined = np.maximum(combined, non_aff_max)
            maxes = np.max(combined, axis=0)
            
            best_j = int(np.argmin(maxes))
            best_t = grid[best_j]
            best_val = maxes[best_j]
            
            if best_val < cur_max - 1e-14:
                h[i] = best_t
                improved = True
        
        h = _project(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _slsqp_optimize(h, deadline, beta_init=40.0, beta_final=200.0, n_phases=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        
        def obj(hf, b=beta):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = b * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            return maxS + np.log(np.mean(np.exp(shifted))) / b
        
        def grad(hf, b=beta):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = b * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            wsum = np.sum(w)
            if wsum < 1e-300:
                idx = int(np.argmax(S))
                w = np.zeros(len(S))
                w[idx] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w
        
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq',
                                        'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 150, 'ftol': 1e-14})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5
