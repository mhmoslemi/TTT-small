def _solve_epigraph(h0, deadline, max_iter=300):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    h0 = _project(h0, target)
    c5_0 = _exact_c5(h0)
    
    # Objective: minimize t
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Inequality constraints: t - score_r(h) >= 0
    def ineq_constraints(x):
        h = x[:n]
        t = x[-1]
        s = _scores(h)
        return t - s
    
    def ineq_jac(x):
        h = x[:n]
        G = _grad_matrix(h)
        J = np.zeros((n_lags, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    # Equality constraint: sum(h) - target = 0
    def eq_constraint(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, c5_0 + 0.1)]
    
    x0 = np.concatenate([h0, [c5_0]])
    
    constraints = [
        {'type': 'ineq', 'fun': ineq_constraints, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac},
    ]
    
    best_h = h0.copy()
    best_c5 = c5_0
    
    options = {'maxiter': max_iter, 'disp': False}
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                      bounds=bounds, constraints=constraints, options=options)
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass
    
    return best_h, best_c5

        J[:, n] = 1.0
        return J
    
    def eq(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, c5_0 + 0.05)]
    x0 = np.concatenate([h0, [c5_0]])
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac},
    ]
    
    best_h = h0.copy()
    best_c5 = c5_0
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'disp': False})
        h_new = _project(res.x[:n], target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    except Exception:
        pass
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += dx * (1.0 - h[:m])
            G[r, :m] += -dx * h[d:]
        else:
            a = -d
            m = n - a
            G[r, :m] += dx * (1.0 - h[a:])
            G[r, a:] += -dx * h[:m]
    return G

def _line_search(h, p, dx, radius):
    """Find best step in direction p, maintaining feasibility."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
    C = np.correlate(p, p, mode='full')
    
    t_lo, t_hi = 0.0, radius
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / p[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(S0 + c * T - c * c * C)) * dx
    fd = float(np.max(S0 + d * T - d * d * C)) * dx
    
    for _ in range(60):
        if b - a < 1e-14:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = float(np.max(S0 + c * T - c * c * C)) * dx
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = float(np.max(S0 + d * T - d * d * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_direction(h, scores, G, radius, n_active=5):
    """LP to find direction that reduces active lags."""
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:n_active]
    
    # min -delta s.t. G[active] @ p + delta <= 0, sum(p)=0, bounds on p
    # Variables: p (n), delta (1)
    # Objective: minimize -delta → c = [0,...,0,-1]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality: G[active] @ p + delta <= 0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active, :]
    A_ub[:, n] = 1.0
    b_ub = np.zeros(len(active))
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[i]), min(radius, 1.0 - h[i])) for i in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success and -res.x[-1] > 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, n_active=5, radius_init=0.1):
    """Iterative LP direction + line search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, n_active)
        
        if p is None or np.max(np.abs(p)) < 1e-14:
            # Try with more active constraints
            p = _lp_direction(h, scores, G, radius, min(n_active + 3, 2 * n - 1))
            if p is None or np.max(np.abs(p)) < 1e-14:
                stall += 1
                radius *= 0.75
                if radius < 1e-7:
                    break
                continue
        
        alpha, new_c5 = _line_search(h, p, dx, min(radius, 0.5))
        if new_c5 < best_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5
