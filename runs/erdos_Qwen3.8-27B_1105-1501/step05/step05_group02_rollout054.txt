import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _proj(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    S = _scores(h)
    maxS = np.max(S)
    w = np.exp((S - maxS) / tau)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(S))] = 1.0
    else:
        w /= wsum
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d == 0:
            g += w[r] * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            g[d:] += w[r] * dx * (1.0 - h[:m])
            g[:m] -= w[r] * dx * h[d:]
        else:
            e = -d
            m = n - e
            g[:m] += w[r] * dx * (1.0 - h[e:])
            g[e:] -= w[r] * dx * h[:m]
    return g

def _pair_line_search(h, i, j, n_samples):
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-12:
        return None, None
    h_test = h.copy()
    best_t = 0.0
    best_c5 = _exact_c5(h)
    ts = np.linspace(t_min, t_max, n_samples)
    for t in ts:
        h_test[i] = h[i] + t
        h_test[j] = h[j] - t
        c5 = _exact_c5(h_test)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    if best_t == 0.0:
        return None, None
    step = (t_max - t_min) / max(n_samples - 1, 1)
    lo = max(t_min, best_t - 2.0 * step)
    hi = min(t_max, best_t + 2.0 * step)
    if hi - lo > 1e-13:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h_test[i] = h[i] + c1
        h_test[j] = h[j] - c1
        f1 = _exact_c5(h_test)
        h_test[i] = h[i] + c2
        h_test[j] = h[j] - c2
        f2 = _exact_c5(h_test)
        for _ in range(18):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                h_test[i] = h[i] + c1
                h_test[j] = h[j] - c1
                f1 = _exact_c5(h_test)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                h_test[i] = h[i] + c2
                h_test[j] = h[j] - c2
                f2 = _exact_c5(h_test)
        opt_t = (lo + hi) / 2.0
        h_test[i] = h[i] + opt_t
        h_test[j] = h[j] - opt_t
        opt_c5 = _exact_c5(h_test)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_t = opt_t
    if best_t == 0.0:
        return None, None
    return best_t, best_c5

def _pair_optimize(h, deadline, tau_init, n_pairs, max_no_improve, rng, n_samples=15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    no_improve = 0
    while time.time() < deadline:
        if no_improve > max_no_improve:
            break
        g = _weighted_grad(h, tau)
        g_centered = g - np.mean(g)
        order = np.argsort(g_centered)
        improved = False
        kp = min(n_pairs, n // 2)
        for k in range(kp):
            if time.time() > deadline:
                break
            j = order[n - 1 - k]
            i = order[k]
            if i == j:
                continue
            t, new_c5 = _pair_line_search(h, i, j, n_samples)
            if t is not None and new_c5 < cur_c5 - 1e-13:
                h[i] += t
                h[j] -= t
                h = _proj(h, target)
                cur_c5 = _exact_c5(h)
                improved = True
                no_improve = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not improved:
            for _ in range(30):
                if time.time() > deadline:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                t, new_c5 = _pair_line_search(h, i, j, n_samples - 2)
                if t is not None and new_c5 < cur_c5 - 1e-13:
                    h[i] += t
                    h[j] -= t
                    h = _proj(h, target)
                    cur_c5 = _exact_c5(h)
                    improved = True
                    no_improve = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
        if not improved:
            no_improve += 1
            tau *= 0.7
            if tau < 1e-7:
                break
        else:
            tau = min(tau * 1.1, 0.05)
    return best_h, best_c5

def _full_sweep_optimize(h, deadline, n_samples=12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sweep = 0
    while time.time() < deadline:
        improved = False
        g = _weighted_grad(h, 0.003)
        gc = g - np.mean(g)
        order = np.argsort(gc)
        for k in range(n // 2):
            if time.time() > deadline:
                break
            j = order[n - 1 - k]
            i = order[k]
            if i == j:
                continue
            t, new_c5 = _pair_line_search(h, i, j, n_samples)
            if t is not None and new_c5 < cur_c5 - 1e-13:
                h[i] += t
                h[j] -= t
                h = _proj(h, target)
                cur_c5 = _exact_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not improved:
            break
        sweep += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.permutation(n)
    h[pos[:n // 2]] = 1.0
    return h

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _proj(h.astype(float), n / 2.0)

def _perturb_init(h, rng, scale):
    return _proj(h + rng.normal(0, scale, len(h)), len(h) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _proj(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Initial pair optimization at parent resolution
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = _pair_optimize(h, dl1, tau_init=0.005, n_pairs=50,
                              max_no_improve=50, rng=rng, n_samples=15)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1.copy(), len(h1)

    # Phase 2: Full sweep polish
    dl2 = t0 + budget_s * 0.40
    h2, c5_2 = _full_sweep_optimize(best_h, dl2, n_samples=12)
    if c5_2 < best_c5:
        best_c5, best_h, best_n = c5_2, h2.copy(), len(h2)

    # Phase 3: Complement optimization
    dl3 = t0 + budget_s * 0.48
    h_comp = _proj(1.0 - best_h, best_n / 2.0)
    h3, c5_3 = _pair_optimize(h_comp, dl3, tau_init=0.004, n_pairs=45,
                              max_no_improve=40, rng=rng, n_samples=14)
    if c5_3 < best_c5:
        best_c5, best_h, best_n = c5_3, h3.copy(), len(h3)

    # Phase 4: Multi-resolution
    dl4 = t0 + budget_s * 0.68
    n_list = [96, 112, 120, 136, 144, 160, 192]
    for n_try in n_list:
        if time.time() > dl4:
            break
        h_res = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.04)
        h_r, c5_r = _pair_optimize(h_res, sub_dl, tau_init=0.004, n_pairs=35,
                                   max_no_improve=30, rng=rng, n_samples=13)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Quick polish at this resolution
            sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
            h_r2, c5_r2 = _full_sweep_optimize(best_h, sub_dl2, n_samples=11)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()

    # Phase 5: Multi-start exploration
    dl5 = t0 + budget_s * 0.85
    best_n = best_n
    for attempt in range(80):
        if time.time() > dl5:
            break
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        which = attempt % 5
        if which == 0:
            h_start = _binary_init(best_n, rng)
        elif which == 1:
            alpha = 0.5 + 0.15 * float(attempt % 7) / 7.0
            h_start = _quasi_periodic(best_n, alpha)
        elif which == 2:
            h_start = _perturb_init(best_h, rng, 0.06)
        elif which == 3:
            h_start = _perturb_init(1.0 - best_h, rng, 0.06)
        else:
            # Random smooth pattern
            h_start = _proj(rng.random(best_n) * 0.5 + 0.25, best_n / 2.0)
        h_try, c5_try = _pair_optimize(h_start, sub_dl, tau_init=0.008,
                                       n_pairs=25, max_no_improve=15,
                                       rng=rng, n_samples=12)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)

    # Phase 6: Final deep polish
    if time.time() < deadline:
        dl6 = deadline
        h_fin, c5_fin = _full_sweep_optimize(best_h, dl6, n_samples=10)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        # One more pair pass with very low tau
        if time.time() < deadline - 10:
            h_fin2, c5_fin2 = _pair_optimize(best_h, deadline, tau_init=0.001,
                                             n_pairs=60, max_no_improve=10,
                                             rng=rng, n_samples=11)
            if c5_fin2 < best_c5:
                best_c5 = c5_fin2
                best_h = h_fin2.copy()

    # Final validation and return
    best_h = _proj(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
