    # Using KKT: H d + grad + lambda * 1 = 0, sum(d) = 0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G * (2.0 / n)

def _newton_optimize(h, deadline, top_k=5, max_iters=500, damp=0.9):
    """Newton's method on top-k max smoothing."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        # Get top-k scores
        idx = np.argsort(scores)[::-1][:top_k]
        S_top = scores[idx]
        maxS = S_top[0]

    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G * dx

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    alphas = np.linspace(0, alpha_max, 16)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, len(alphas) - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(len(alphas) - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(18):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _newton_optimize(h, deadline, n_phases=5, beta_start=20.0, beta_end=200.0):
    """Newton's method on log-sum-exp smoothed max with KKT constraint."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    betas = np.geomspace(beta_start, beta_end, n_phases)
    for beta in betas:
        if time.time() > deadline:
            break
        max_iters = 80
        for it in range(max_iters):
            if time.time() > deadline:
                break
            scores = _scores(h)
            maxS = np.max(scores)
            shifted = beta * (scores - maxS)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            wsum = np.sum(w)
            if wsum < 1e-300:
                idx = int(np.argmax(scores))
                w = np.zeros(len(scores))
                w[idx] = 1.0
            else:
                w /= wsum
            
            G = _grad_matrix(h)
            grad = G.T @ w
            H = G.T @ (w[:, None] * G)
            
            # Regularize Hessian
            trace_H = np.trace(H)
            mu = max(1e-8, 0.01 * trace_H / n)
            H_reg = H + mu * np.eye(n)
            
            # KKT system for constrained Newton step
            K = np.zeros((n + 1, n + 1))
            K[:n, :n] = H_reg
            K[:n, n] = 1.0
            K[n, :n] = 1.0
            rhs = np.zeros(n + 1)
            rhs[:n] = -grad
            
            try:
                sol = np.linalg.solve(K, rhs)
                d = sol[:n]
            except np.linalg.LinAlgError:
                d = -grad
                d -= d.mean()
            
            # Line search
            alpha = 1.0
            cur_c5 = _exact_c5(h)
            improved = False
            for _ls in range(25):
                h_try = h + alpha * d
                if np.any(h_try < -0.5) or np.any(h_try > 1.5):
                    alpha *= 0.5
                    continue
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-15:
                    h = h_try
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h.copy()
                    improved = True
                    break
                alpha *= 0.6
                if alpha < 1e-12:
                    break
            if not improved:
                # Fallback: gradient step
                grad_tan = grad - grad.mean()
                gn = np.linalg.norm(grad_tan)
                if gn < 1e-14:
                    break
                d2 = -grad_tan / gn
                for a_try in [0.1, 0.05, 0.02, 0.01, 0.005]:
                    h_try = _project(h + a_try * d2, target)
                    c5_try = _exact_c5(h_try)
                    if c5_try < _exact_c5(h) - 1e-15:
                        h = h_try
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_h = h.copy()
                        improved = True
                        break
                if not improved:
                    break
    
    return best_h, best_c5

def _slsqp_optimize(h, deadline, beta_init=40.0, beta_final=300.0, n_phases=4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    def _obj(hf):
        S = _scores(hf)
        b = _beta
        maxS = np.max(S)
        if b < 1e-6:
            return float(maxS)
        shifted = b * (S - maxS)
        shifted = np.clip(shifted, -500, 0)
        return float(maxS + np.log(np.mean(np.exp(shifted))) / b)
    
    def _grad(hf):
        S = _scores(hf)
        b = _beta
        maxS = np.max(S)
        shifted = b * (S - maxS)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            idx = int(np.argmax(S))
            w = np.zeros(len(S))
            w[idx] = 1.0
        else:
            w /= wsum
        G = _grad_matrix(hf)
        return G.T @ w
    
    _beta = beta_init
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        _beta = beta_init * (beta_final / beta_init) ** frac
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq',
                                        'fun': lambda x: float(np.sum(x) - target),
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 300, 'ftol': 1e-15})
            if np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 40)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])

                    lo_a, hi_a = 0.0, a_max
                    for _b in range(6):
                        m1 = lo_a + (hi_a - lo_a) / 3.0

                        f1 = _exact_c5(h1)

                        f2 = _exact_c5(h2)

        if c5_6 < best_c5:
            best_c5 = c5_6; best_h = h6.copy()

    # Phase 7: Perturbation restarts
    if time.time() < t0 + budget_s * 0.88:
        dl7 = t0 + budget_s * 0.88
        scale = 0.015
        for attempt in range(300):
            if time.time() > dl7:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl7, time.time() + budget_s * 0.01)
            ht, ct = _local_optimize_lp(h_pert, sub_dl, radius_init=0.05, max_iters=300, stall_limit=40)
            if ct < best_c5 - 1e-14:
                best_c5 = ct; best_h = ht.copy()
                scale = max(0.002, scale * 0.75)
            else:
                scale = min(0.06, scale * 1.08)

    # Phase 8: Newton final
    if time.time() < t0 + budget_s * 0.93:
        dl8 = t0 + budget_s * 0.93
        h8, c5_8 = _newton_optimize(best_h, dl8, n_phases=3, beta_start=100.0, beta_end=500.0)
        if c5_8 < best_c5:
            best_c5 = c5_8; best_h = h8.copy()

    # Phase 9: LP final
    if time.time() < t0 + budget_s * 0.96:
        h9, c5_9 = _local_optimize_lp(best_h, t0 + budget_s * 0.96,
                                       radius_init=0.03, max_iters=3000, stall_limit=100)
        if c5_9 < best_c5:
            best_c5 = c5_9; best_h = h9.copy()

    # Phase 10: Pair polish
    if time.time() < t0 + budget_s * 0.98:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.98, max_passes=5)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
