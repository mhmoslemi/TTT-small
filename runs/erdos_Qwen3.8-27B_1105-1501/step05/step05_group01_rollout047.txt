import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    """Golden section line search for phi(t) = max_l (S_l + t*T_l - t^2*Q_l) * dx
    with box constraint h + t*d in [0,1]."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Box constraint bounds
    t_lo = 0.0
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        # h + t*d >= 0 => t >= -h/d = h/(-d) for d<0... 
        # Actually h[neg] + t*d[neg] >= 0 => t <= h[neg]/(-d[neg])
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    for _ in range(50):
        if t_hi - t_lo < 5e-13:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * Q)) * dx
    return opt_t, opt_val

    n = len(h)

def _grad_at_lag(h, lag_idx):
    """
    Compute gradient of S_lag_idx w.r.t. h.
    lag_idx is the index in the 'full' correlation output (0 to 2n-2).
    The signed lag is d = lag_idx - (n-1).
    """
    n = len(h)
    d = lag_idx - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            # h[i] appears in position i for i in [0, m-1]
            # contributes (1 - h[i+d])
            g[:m] += (1.0 - h[d:])
            # (1-h[i]) appears in position i+d for i in [0, m-1], i.e., position j = i+d for j in [d, n-1]
            # contributes -h[i] = -h[j-d]
            g[d:] += (-h[:m])
    else:
        g_neg = -d
        m = n - g_neg
        if m > 0:
            # h[i+g_neg] * (1-h[i]) for i in [0, m-1]
            # h[i+g_neg] is at position j = i+g_neg for j in [g_neg, n-1]
            # contributes (1 - h[i]) = (1 - h[j-g_neg])
            g[g_neg:] += (1.0 - h[:m])
            # (1-h[i]) is at position i for i in [0, m-1]
            # contributes -h[i+g_neg]
            g[:m] += (-h[g_neg:])
    return g
