Let me think about what specific improvements could help:

1. The epigraph approach with SLSQP should be more effective than LP linearization because:
   - Score functions S_k(h) = dx * Σ h_i(1-h_{i+d}) are bilinear
   - LP only uses first-order approximation
   - SLSQP can take multiple Newton-like steps in one solve

2. Active set management: Start with top-K active constraints, gradually add more as we get closer to the optimum.

3. The gradient of the score functions is straightforward:
   - For positive lag d: S_d = dx * Σ_{i=0}^{n-d-1} h_i(1-h_{i+d})
   - ∂S_d/∂h_i = dx * [(1-h_{i+d}) for i < n-d] - dx * [h_{i-d} for i ≥ d]
   - More precisely: for variable h_j, it appears in term i=j as source (coefficient 1-h_{j+d}) and in term i=j-d as target (coefficient -h_{j-d})

Let me code this up carefully:

For n points, dx = 2/n, target = n/2.

Score for lag d (0 ≤ d ≤ n-1):
S_d = dx * Σ_{i=0}^{n-d-1} h_i * (1 - h_{i+d})

For lag -g (1 ≤ g ≤ n-1):
S_{-g} = dx * Σ_{i=0}^{n-g-1} h_{i+g} * (1 - h_i)

Note: S_d and S_{-d} are different (they measure different things).

Gradient of S_d with respect to h_j:
- h_j appears as source in term i=j (if j ≤ n-d-1): contribution dx*(1-h_{j+d})
- h_j appears as target in term i=j-d (if j ≥ d and j-d ≤ n-d-1, i.e., j ≥ d): contribution dx*(-h_{j-d})

So: ∂S_d/∂h_j = dx*(1-h_{j+d}) * [j ≤ n-d-1] - dx*h_{j-d} * [j ≥ d]

Gradient of S_{-g} with respect to h_j:
S_{-g} = dx * Σ_{i=0}^{n-g-1} h_{i+g} * (1 - h_i)
- h_j appears as source in term i=j-g (if j ≥ g and j-g ≤ n-g-1, i.e., g ≤ j ≤ n-1): contribution dx*(1-h_{j-g})
- h_j appears as target in term i=j (if j ≤ n-g-1): contribution dx*(-h_{j+g})

So: ∂S_{-g}/∂h_j = dx*(1-h_{j-g}) * [g ≤ j ≤ n-1] - dx*h_{j+g} * [j ≤ n-g-1]

For the epigraph SLSQP problem:
- Variables: h_0, ..., h_{n-1}, T (n+1 variables)
- Objective: minimize T
- Constraints:
  - h_j ∈ [0,1] for all j (bounds)
  - Σ h_j = n/2 (equality)
  - T - S_k(h) ≥ 0 for active lags k (inequality)

The Jacobian of the inequality constraint T - S_k(h):
- ∂/∂h_j = -∂S_k/∂h_j
- ∂/∂T = 1

import numpy as np
from scipy.optimize import minimize
import time

def _scores_all(h):
    """Compute all 2n-1 normalized overlap scores."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    scores = np.zeros(n_lags)
    # np.correlate(h, 1-h, mode='full') gives the scores
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    return scores

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {h: 0<=h<=1, sum(h)=target}."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
        # Second pass for any residual
        res2 = target - float(np.sum(x))
        if abs(res2) > 1e-12:
            if res2 > 0:
                mask2 = x < 1.0
            else:
                mask2 = x > 0.0
            cnt2 = int(np.sum(mask2))
            if cnt2 > 0:
                x[mask2] += res2 / cnt2
                x = np.clip(x, 0.0, 1.0)
    return x

def _score_and_grad_selected(h, ids, dx, n):
    """
    Compute scores and their gradients for selected lag IDs.
    ids: array of lag indices in [0, 2n-2]
    Returns: scores_selected (array), grad_matrix (array of shape (len(ids), n))
    """
    q = len(ids)
    scores_sel = np.zeros(q)
    grad_mat = np.zeros((q, n))
    
    for idx, lag_id in enumerate(ids):
        if lag_id < n:  # positive lag d = lag_id
            d = lag_id
            m = n - d
            if m <= 0:
                continue
            # S_d = dx * sum_{i=0}^{m-1} h[i] * (1-h[i+d])
            scores_sel[idx] = dx * np.dot(h[:m], 1.0 - h[d:])
            # Gradient wrt h[j]:
            # Source: h[j] for j in [0, m-1]: coeff = 1-h[j+d]
            grad_mat[idx, :m] += (1.0 - h[d:])
            # Target: h[j] for j in [d, n-1]: coeff = -h[j-d]
            grad_mat[idx, d:] += -h[:m]
        else:  # negative lag, g = lag_id - n + 1
            g = lag_id - n + 1
            # S_{-g} = dx * sum_{i=0}^{n-g-1} h[i+g] * (1-h[i])
            m = n - g
            if m <= 0:
                continue
            scores_sel[idx] = dx * np.dot(h[g:], 1.0 - h[:m])
            # Gradient wrt h[j]:
            # Source: h[j] for j in [g, n-1]: coeff = 1-h[j-g]
            grad_mat[idx, g:] += (1.0 - h[:m])
            # Target: h[j] for j in [0, m-1]: coeff = -h[j+g]
            grad_mat[idx, :m] += -h[g:]
    
    return scores_sel, grad_mat

def _epig_objective(x, h0, ids, dx, n):
    """Objective: T = x[n]"""
    return x[n]

def _epig_objective_grad(x, h0, ids, dx, n):
    """Gradient of objective."""
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq_constraint(x, n, target):
    """Equality: sum(h) = target"""
    return float(np.sum(x[:n])) - target

def _eq_jacobian(x, n, target):
    """Jacobian of equality constraint."""
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def _ineq_constraints(x, ids, dx, n):
    """Inequality: T - S_k(h) >= 0 for each active lag k."""
    h = x[:n]
    T = x[n]
    scores_sel, _ = _score_and_grad_selected(h, ids, dx, n)
    return T - scores_sel

def _ineq_jacobian(x, ids, dx, n):
    """Jacobian of inequality constraints."""
    h = x[:n]
    q = len(ids)
    _, grad_mat = _score_and_grad_selected(h, ids, dx, n)
    J = np.zeros((q, n + 1))
    J[:, :n] = -grad_mat
    J[:, n] = 1.0
    return J

def _epig_solve(h_start, active_ids, dx, n, t_init, maxiter=100):
    """Solve the epigraph SLSQP problem."""
    target = n / 2.0
    q = len(active_ids)
    
    # Initial x
    x0 = np.zeros(n + 1)
    x0[:n] = h_start
    x0[n] = max(t_init, np.max(_score_and_grad_selected(h_start, active_ids, dx, n)[0]) + 0.01)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    cons = [
        {'type': 'eq', 'fun': lambda x: _eq_constraint(x, n, target),
         'jac': lambda x: _eq_jacobian(x, n, target)},
    ]
    # Add inequality constraints (SLSQP expects g(x) >= 0)
    for idx in range(q):
        # Can't use lambda with loop variable easily, use a different approach
        # Actually let's use a vectorized constraint instead
        pass
    
    # Better: use a single constraint function that returns vector
    # SLSQP doesn't support vector constraints directly in 'ineq' type
    # We need to add them individually or use a workaround
    
    # Actually, scipy SLSQP supports multiple inequality constraints:
    # Each constraint is a dict with type='ineq'
    # But we can't use lambdas in list comprehension effectively
    # Let's build them properly
    
    def ineq_fun(x):
        return _ineq_constraints(x, active_ids, dx, n)
    
    def ineq_jac(x):
        return _ineq_jacobian(x, active_ids, dx, n)
    
    # SLSQP requires individual constraints, not vector
    # So we add each one separately
    cons_list = [
        {'type': 'eq', 'fun': lambda x: _eq_constraint(x, n, target),
         'jac': lambda x: _eq_jacobian(x, n, target)},
    ]
    
    # For SLSQP, we need individual ineq constraints
    # But q can be up to 128, making this slow
    # Alternative: combine into a single constraint with minimum
    # T - max(S_k) >= 0 is equivalent to all T - S_k >= 0
    # But max is non-smooth. Instead, use sum of soft violations.
    # Or: just add all individually (SLSQP can handle it)
    
    # Actually for efficiency, let's use the minimum approach:
    # The binding constraint is the one with max S_k
    # So T - max_k S_k(h) >= 0
    # This is non-smooth but SLSQP often handles it OK
    
    # Better approach: add top constraints individually
    for i in range(q):
        def _make_ieq(i):
            def fun(x):
                h = x[:n]
                T = x[n]
                s, _ = _score_and_grad_selected(h, active_ids[i:i+1], dx, n)
                return T - s[0]
            def jac(x):
                h = x[:n]
                _, gm = _score_and_grad_selected(h, active_ids[i:i+1], dx, n)
                j = np.zeros(n + 1)
                j[:n] = -gm[0]
                j[n] = 1.0
                return j
            return {'type': 'ineq', 'fun': fun, 'jac': jac}
        cons_list.append(_make_ieq(i))
    
    try:
        res = minimize(_epig_objective, x0, args=(h_start, active_ids, dx, n),
                       method='SLSQP', jac=_epig_objective_grad,
                       bounds=bounds, constraints=cons_list,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        return res
    except Exception:
        return None

def _select_top_ids(scores, q, rng=None):
    """Select top q lag IDs by score, with optional random jitter."""
    n_lags = len(scores)
    if q >= n_lags:
        return np.arange(n_lags)
    if rng is not None:
        # Add small jitter to break ties
        jitter = rng.uniform(0, 1e-6, n_lags)
        scores_j = scores + jitter
    else:
        scores_j = scores
    top = np.argsort(-scores_j)[:q]
    return top

def _epig_optimize(h_start, deadline, active_schedule=None, rng=None):
    """
    Run epigraph SLSQP optimization with growing active set.
    Returns (best_h, best_c5).
    """
    n = len(h_start)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h_start, target).copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    if active_schedule is None:
        active_schedule = [16, 24, 32, 48, 64, 96, 128]
    
    for q in active_schedule:
        if time.time() > deadline:
            break
        if q >= 2 * n - 1:
            q = 2 * n - 1
        
        # Select active IDs
        scores = _scores_all(h)
        active_ids = _select_top_ids(scores, q, rng)
        
        # Solve
        t_init = best_c5 + 0.02
        res = _epig_solve(h, active_ids, dx, n, t_init, maxiter=80)
        
        if res is not None and hasattr(res, 'x') and res.x is not None:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return _project(h, n_new / 2.0).copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.03, n)
        h = _project(h, n / 2.0)
    return h

def _coordinate_polish(h, deadline, n_cycles=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v = h[j] - 0.3
            hi_v = h[j] + 0.3
            lo_v = max(0.0, lo_v)
            hi_v = min(1.0, hi_v)
            if hi_v - lo_v < 1e-8:
                continue
            for _ in range(25):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _lp_direction(h, scores, radius, dx, n):
    """LP-based direction finding (from parent, kept as backup)."""
    from scipy.optimize import linprog
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    # Build gradient matrix
    G = np.zeros((n_lags, n))
    for d in range(n):
        r = d  # positive lag
        m = n - d
        if m > 0:
            G[r, :m] += (1.0 - h[d:])
            G[r, d:] += -h[:m]
    for g in range(1, n):
        r = n + g - 1
        m = n - g
        if m > 0:
            G[r, g:] += (1.0 - h[:m])
            G[r, :m] += -h[g:]
    G *= dx
    
    A_ub[:, :n] = G
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize(h_start, deadline, radius_init=0.15, max_iters=3000, radius_min=0.002):
    n = len(h_start)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h_start, target).copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores_all(h)
        p = _lp_direction(h, scores, radius, dx, n)
        if p is None:
            stall += 1
            radius *= 0.65
            if radius < radius_min:
                break
            continue
        # Line search
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.95
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = alpha_max
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h1 = _project(h + c1 * p, target)
        h2 = _project(h + c2 * p, target)
        f1 = _exact_c5(h1)
        f2 = _exact_c5(h2)
        for _ in range(20):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                h1 = _project(h + c1 * p, target)
                f1 = _exact_c5(h1)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                h2 = _project(h + c2 * p, target)
                f2 = _exact_c5(h2)
        
        opt_a = (lo + hi) / 2.0
        opt_h = _project(h + opt_a * p, target)
        opt_c5 = _exact_c5(opt_h)
        
        if f1 < f2:
            best_a, best_trial_c5 = c1, f1
        else:
            best_a, best_trial_c5 = c2, f2
        if opt_c5 < best_trial_c5:
            best_a, best_trial_c5 = opt_a, opt_c5
        
        if best_trial_c5 < cur_c5 - 1e-14:
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    
    return best_h, best_c5
