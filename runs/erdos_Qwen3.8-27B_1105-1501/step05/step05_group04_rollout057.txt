        for i, r in enumerate(active_idx):
            k = r - (n - 1)
            jac[i, :n] = -_grad_lag(h_x, k, dx)
            jac[i, n] = 1.0
        return jac

    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    x0 = np.concatenate([h, [c0 + 0.005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = _project(np.clip(res.x[:n], 0.0, 1.0), n / 2.0)
        c5_new = _eval_c5(h_new)
        if c5_new < c0 - 1e-10:
            return h_new, c5_new
    except Exception:
        pass
    return h, c0

def _lse_slsqp(h, deadline, beta, maxiter=100):
    n = len(h)
    dx = 2.0 / n

    def obj(x):
        lags = _lag_values(x)
        m = np.max(lags)
        z = np.exp((lags - m) / beta)
        return m + beta * np.log(np.sum(z))

    def grad(x):
        lags = _lag_values(x)
        m = np.max(lags)
        z = np.exp((lags - m) / beta)
        p = z / np.sum(z)
        g = np.zeros(n)
        for r in range(2 * n - 1):
            k = r - (n - 1)
            g += p[r] * _grad_lag(x, k, dx)
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        j = np.ones(n)
        return j

    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        h_new = _project(np.clip(res.x, 0.0, 1.0), n / 2.0)
        c5_new = _eval_c5(h_new)
        if c5_new < _eval_c5(h) - 1e-10:
            return h_new, c5_new
    except Exception:
        pass
    return h, _eval_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return _project(h_new, n_new / 2.0)

def _coord_refine(h, deadline, n_iter=500, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    if rng is None:
        rng = np.random.default_rng(9999)
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if len(interior) < 2:
        return best_h, best_c5
    count = 0
    while count < n_iter and time.time() < deadline:
        count += 1
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        max_alpha = min(h[i], 1.0 - h[j], 0.5)
        alpha, c5_try = _line_search(h, d, dx, max_alpha)
        if c5_try < best_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def _systematic_coord(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if len(interior) < 2:
        return best_h, best_c5
    ll = len(interior)
    step = max(1, ll // 200)
    for ii in range(0, ll, step):
        if time.time() > deadline:
            break
        for jj in range(0, ll, step):
            if ii == jj:
                continue
            i = interior[ii]
            j = interior[jj]
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            max_alpha = min(h[i], 1.0 - h[j], 0.3)
            alpha, c5_try = _line_search(h, d, dx, max_alpha)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _eval_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
                    ll = len(interior)
                    if ll < 2:
                        return best_h, best_c5
                    step = max(1, ll // 200)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _eval_c5(h)

    # Phase 1: LP polish (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP epigraph with increasing active sets (22%)
    d2 = t0 + budget_s * 0.30
    for na in [10, 20, 40, 80]:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + (d2 - t0 - (time.time() - t0)) / max(1, 4 - [10,20,40,80].index(na)))
        h2, c5_2 = _slsqp_epigraph(best_h, sub_dl, n_active=na, maxiter=150)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            # Follow up with LP
            sub_dl2 = min(d2, time.time() + 5.0)
            h2b, c5_2b = _lp_loop(best_h, sub_dl2, radius_init=0.08)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Log-sum-exp SLSQP (15%)
    d3 = t0 + budget_s * 0.45
    if time.time() < d3:
        c_base = best_c5
        for beta_frac in [0.5, 0.3, 0.15, 0.08, 0.04]:
            if time.time() > d3 - 5:
                break
            beta = c_base * beta_frac
            if beta < 1e-5:
                beta = 1e-5
            h3, c5_3 = _lse_slsqp(best_h, d3, beta=beta, maxiter=80)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                sub_dl = min(d3, time.time() + 4.0)
                h3b, c5_3b = _lp_loop(best_h, sub_dl, radius_init=0.06)
                if c5_3b < best_c5:
                    best_c5 = c5_3b
                    best_h = h3b.copy()

    # Phase 4: Resolution cycling (25%)
    d4 = t0 + budget_s * 0.70
    n_base = len(best_h)
    tried = set()
    n_candidates = list(range(max(30, n_base - 25), min(150, n_base + 35), 3))
    rng.shuffle(n_candidates)
    n_tried = 0
    for n_try in n_candidates:
        if time.time() > d4 or n_tried >= 20:
            break
        if n_try in tried or n_try < 24:
            continue
        tried.add(n_try)
        n_tried += 1
        h_r = _resample(best_h, n_try)
        c5_r_init = _eval_c5(h_r)
        sub_dl = min(d4, time.time() + 8.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1)
        if c5_r < best_c5 and time.time() < d4 - 3:
            h_r2, c5_r2 = _slsqp_epigraph(h_r, min(d4, time.time() + 3.0), n_active=15, maxiter=80)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: SLSQP epigraph on best (10%)
    d5 = t0 + budget_s * 0.80
    if time.time() < d5:
        for na in [30, 60]:
            if time.time() > d5:
                break
            h5, c5_5 = _slsqp_epigraph(best_h, min(d5, time.time() + 10.0), n_active=na, maxiter=100)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()

    # Phase 6: Coordinate refinement (10%)
    d6 = t0 + budget_s * 0.90
    if time.time() < d6:
        best_h, best_c5 = _systematic_coord(best_h, min(d6, time.time() + 30.0))
    d6b = t0 + budget_s * 0.95
    if time.time() < d6b:
        best_h, best_c5 = _coord_refine(best_h, d6b, n_iter=300, rng=rng)

    # Phase 7: Final LP polish (5%)
    d7 = hard_deadline - 5.0
    if time.time() < d7:
        h7, c5_7 = _lp_loop(best_h, d7, radius_init=0.04)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
        # Quick coordinate pass
        rem = hard_deadline - 2.0
        if time.time() < rem:
            best_h, best_c5 = _coord_refine(best_h, rem, n_iter=100, rng=np.random.default_rng(seed + 1))

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _eval_c5(best_h)
    if not np.isfinite(final_c5):
        best_h = _project(h0, n0 / 2.0)
        final_c5 = _eval_c5(best_h)
    return best_h, final_c5, len(best_h)
