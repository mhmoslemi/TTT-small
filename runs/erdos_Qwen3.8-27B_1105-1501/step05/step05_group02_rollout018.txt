    """Golden section line search for h + alpha*p."""
    n = len(h)
    # Find max alpha
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))

    c2 = lo + gr * (hi - lo)

            c2 = lo + gr * (hi - lo)
            f2 = _c5(h + c2 * p)

    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def _grad_raw(h, idx):
    n = len(h)
    d = idx - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        g[d:] += (1.0 - h[:m])
        g[:m] -= h[d:]
    else:
        gv = -d
        m = n - gv
        g[:m] += (1.0 - h[gv:])
        g[gv:] -= h[:m]
    return g

def _lp_dir(h, active, radius):
    n = len(h)
    dx = 2.0 / n
    sc = _scores(h)
    na = len(active)
    c = np.zeros(n + 1); c[n] = 1.0
    A_ub = np.zeros((na, n + 1))
    b_ub = np.zeros(na)
    for i, idx in enumerate(active):
        A_ub[i, :n] = _grad_raw(h, idx) * dx
        A_ub[i, n] = -1.0
        b_ub[i] = -sc[idx]
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                   bounds=bounds, method='highs')
        if r.success: return r.x[:n]
    except: pass
    return None

def _line_search(h, p, n_eval=25):
    n = len(h)
    amax = 0.5
    for j in range(n):
        if p[j] > 1e-14: amax = min(amax, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14: amax = min(amax, h[j] / (-p[j]))
    amax = max(amax, 1e-10)
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    lo, hi = 0.0, amax
    c1 = hi - gr * (hi - lo); c2 = lo + gr * (hi - lo)
    f1 = _c5(h + c1 * p); f2 = _c5(h + c2 * p)
    for _ in range(n_eval):
        if hi - lo < 1e-14: break
        if f1 < f2:
            hi, f2, c2 = c1, f1, c1
            c1 = hi - gr * (hi - lo); f1 = _c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c1
            c2 = lo + gr * (hi - lo); f2 = _c5(h + c2 * p)
    a = (lo + hi) * 0.5
    return a, _c5(h + a * p)

def _lp_opt(h, deadline, r0=0.3, max_it=5000, topk=20):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    cur = _c5(h)
    bh, bc = h.copy(), cur
    rad = r0
    stall = 0
    for it in range(max_it):
        if time.time() > deadline: break
        if stall > 100: break
        sc = _scores(h)
        if topk < 2 * n - 1:
            active = np.argsort(sc)[-topk:]
        else:
            active = np.arange(2 * n - 1)
        p = _lp_dir(h, active, rad)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1; rad *= 0.7
            if rad < 1e-8: break
            continue
        a, nc = _line_search(h, p)
        if nc < cur - 1e-14:
            h = _project(h + a * p, tgt)
            cur = _c5(h)
            if cur < bc: bc, bh = cur, h.copy()
            stall = 0
            rad = min(rad * 1.1, 0.5)
        else:
            stall += 1; rad *= 0.7
            if stall > 20 and rad < 0.02: rad = 0.12
    return bh, bc

def _coord_descent(h, deadline, n_sweeps=50):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    cur = _c5(h)
    bh, bc = h.copy(), cur
    for sweep in range(n_sweeps):
        if time.time() > deadline: break
        imp = False
        for j in range(n):
            # Try moving h[j] by small amounts
            best_v = h[j]
            best_c = cur
            for dv in [-0.15, -0.08, -0.03, 0.03, 0.08, 0.15]:
                h[j] = np.clip(h[j] + dv, 0.0, 1.0)
                h = _project(h, tgt)
