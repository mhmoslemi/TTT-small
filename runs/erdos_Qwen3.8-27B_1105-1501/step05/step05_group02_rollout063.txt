For a pair (p, q), the feasible interval for t is:
- t >= -h[p] (so h[p] + t >= 0)
- t <= 1 - h[p] (so h[p] + t <= 1)
- t <= h[q] (so h[q] - t >= 0)
- t >= -(1 - h[q]) (so h[q] - t <= 1)

So t_min = max(-h[p], -(1-h[q])) and t_max = min(1-h[p], h[q]).

If t_min >= t_max, the pair is infeasible (both at same boundary).

For the line search, I'll:
1. Sample the objective at, say, 21 points in [t_min, t_max]
2. Find the best sample
3. Do golden section in the neighborhood of the best sample
4. Return the best t and the new h

Score at index n-1 (lag 0): S_0 = sum_i h[i](1-h[i])

- If p < n-d: p is the "second" index (i = p), contribution: -h[p+d] (derivative of (1-h[p]) is -1, times h[p+d])

So dS_d/dh[p] = (1-h[p-d]) * [p >= d] + (-h[p+d]) * [p < n-d]

Similarly for negative lag: S_{-d} = sum_{i=0}^{n-d-1} h[i](1-h[i+d])
dS_{-d}/dh[p]:
- If p < n-d: p is the "first" index, contribution: 1-h[p+d]
- If p >= d: p is the "second" index (i+d = p, i = p-d), contribution: -h[p-d]

Now for the weighted gradient:

This gives the direction to increase the weighted sum of scores. We want to decrease it, so we move in the negative direction.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(h, target):
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, w):
    """Compute weighted gradient of scores. w has length 2n-1."""
    n = len(h)
    grad = np.zeros(n)
    # Center (lag 0): index n-1
    w0 = w[n-1]
    grad += w0 * (1.0 - 2.0 * h)
    # Positive lags d=1..n-1: index n-1+d
    for d in range(1, n):
        wd = w[n-1+d]
        if wd == 0.0:
            continue
        m = n - d
        # h[p] as first index (p = i+d, i=0..m-1, so p = d..n-1): 1-h[p-d]
        grad[d:] += wd * (1.0 - h[:m])
        # h[p] as second index (p = i, i=0..m-1, so p = 0..m-1 = 0..n-d-1): -h[p+d]
        grad[:m] += -wd * h[d:]
    # Negative lags d=1..n-1: index n-1-d
    for d in range(1, n):
        wd = w[n-1-d]
        if wd == 0.0:
            continue
        m = n - d
        # h[p] as first index (p = i, i=0..m-1): 1-h[p+d]
        grad[:m] += wd * (1.0 - h[d:])
        # h[p] as second index (p = i+d, i=0..m-1, so p = d..n-1): -h[p-d]
        grad[d:] += -wd * h[:m]
    return grad

def _softmax_weights(scores, tau):
    s = scores - np.max(scores)
    w = np.exp(s / tau)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    return w

def _line_search_dir(h, p, target, n_samples=15):
    """Line search from h in direction p, maintaining feasibility."""
    n = len(h)
    # Find max alpha such that h + alpha*p stays in [0,1]
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    # Sample
    best_a = 0.0
    best_c5 = _exact_c5(h)
    for i in range(1, n_samples + 1):
        a = alpha_max * i / n_samples
        h_new = _project(h + a * p, target)
        c5 = _exact_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
    # Golden section refinement around best
    lo = max(0.0, best_a - alpha_max / n_samples)
    hi = min(alpha_max, best_a + alpha_max / n_samples)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(20):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_h = _project(h + opt_a * p, target)
    opt_c5 = _exact_c5(opt_h)
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_a = opt_a
    if best_a > 0:
        return _project(h + best_a * p, target), best_c5
    return h, best_c5

def _pair_transfer(h, p, q, target):
    """Exact pair transfer: h[p] += t, h[q] -= t. Returns (h_new, c5) or (h, c5)."""
    n = len(h)
    hp = h[p]
    hq = h[q]
    t_min = max(-hp, -(1.0 - hq))
    t_max = min(1.0 - hp, hq)
    if t_max - t_min < 1e-12:
        return h, _exact_c5(h)
    
    # Sample along the interval
    n_samples = 13
    cur_c5 = _exact_c5(h)
    best_t = 0.0
    best_c5 = cur_c5
    for i in range(1, n_samples + 1):
        t = t_min + (t_max - t_min) * i / n_samples
        h[p] = hp + t
        h[q] = hq - t
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    # Restore and do golden section around best
    h[p] = hp
    h[q] = hq
    
    # Golden section in neighborhood
    lo = max(t_min, best_t - (t_max - t_min) / n_samples)
    hi = min(t_max, best_t + (t_max - t_min) / n_samples)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h[p] = hp + c1; h[q] = hq - c1
    f1 = _exact_c5(h)
    h[p] = hp + c2; h[q] = hq - c2
    f2 = _exact_c5(h)
    for _ in range(15):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            h[p] = hp + c1; h[q] = hq - c1
            f1 = _exact_c5(h)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            h[p] = hp + c2; h[q] = hq - c2
            f2 = _exact_c5(h)
    opt_t = (lo + hi) / 2.0
    h[p] = hp + opt_t; h[q] = hq - opt_t
    opt_c5 = _exact_c5(h)
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_t = opt_t
    
    if best_t != 0.0:
        h[p] = hp + best_t
        h[q] = hq - best_t
        return h, best_c5
    h[p] = hp
    h[q] = hq
    return h, cur_c5

def _pair_optimize(h, deadline, target, n_pairs_per_pass=3000, n_passes_max=500):

    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(n_passes_max):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        # Compute gradient for pair selection
        scores = _scores(h)
        max_s = np.max(scores)
        tau = 0.01 * max_s
        w = _softmax_weights(scores, tau)
        grad = _weighted_grad(h, w)
        grad -= np.mean(grad)
        
        # Select pairs: high positive grad (want to decrease) paired with high negative grad (want to increase)
        # Actually we want to move in -grad direction: decrease where grad is high, increase where grad is low
        # Transfer from high-grad to low-grad
        order = np.argsort(grad)
        # First n/2 have lowest grad (want to increase), last n/2 have highest (want to decrease)
        # Pairs: (low_grad[i], high_grad[j]) means transfer from high_grad[j] to low_grad[i]
        
        n_low = n // 4
        n_high = n // 4
        lows = order[:n_low]
        highs = order[-n_high:]
        
        # Also add some random pairs
        rng_local = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
        n_rand_pairs = max(0, n_pairs_per_pass - n_low * n_high)
        
        improved = False
        for i in range(n_low):
            if time.time() > deadline:
                break
            for j in range(n_high):
                if time.time() > deadline:
                    break
                p = lows[i]  # destination (increase)
                q = highs[j]  # source (decrease)
                if p == q:
                    continue
                h_new, c5_new = _pair_transfer(h, p, q, target)
                if c5_new < cur_c5 - 1e-13:
                    h = h_new
                    # Recompute after projection (pair transfer preserves sum exactly)
                    cur_c5 = c5_new
                    improved = True
                    if cur_c5 < best_c5:

                        best_h = h.copy()
                else:

                    h = h_new  # _pair_transfer restores if no change

        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)
