import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(6):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 140
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx

def _row_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nn = np.arange(0, n - k)
    else:
        nn = np.arange(-k, n)
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g


def _active_direction(h, corr, k_active, tau, rng):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):

    return _project(base, n / 2.0)

    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 4 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        base = _sine_init(n, 1, 0.0, rng)
    n0 = len(base)
    h0 = _project(base, n0 / 2.0)

    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: multi-start (12% of budget)
    d1 = t0 + budget_s * 0.12
    starts = [h0.copy()]
    freq_phase = [(1, 0.0), (1, 1.0), (2, 0.0), (2, 1.0), (3, 0.5), (5, 0.0)]
    for f_, p_ in freq_phase:
        starts.append(_sine_init(n0, f_, p_, rng))
    for _ in range(4):
        starts.append(_block_init(n0, rng))
    for s in starts:
        if time.time() > d1:
            break
        sub = min(d1, time.time() + max(2.0, budget_s * 0.018))
        hh, ff = _optimize(s, sub, rng, k_active=8, tau=0.02)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Phase 2: main deep optimization (68% of budget)
    d2 = t0 + budget_s * 0.80
    hh, ff = _optimize(best_h, d2, rng, pair_frac=0.20, rand_frac=0.08, k_active=10, tau=0.018)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: multi-resolution (12% of budget)
    d3 = t0 + budget_s * 0.92
    bn = len(best_h)
    tried = set()
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 24 or n_try > 240 or n_try != n_try//1:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub = min(d3, time.time() + max(3.0, budget_s * 0.016))
        hr2, fr2 = _optimize(hr, sub, rng, k_active=8, tau=0.02)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 4: final pair-heavy polish (8% of budget)
    hh, ff = _optimize(best_h, deadline, rng, pair_frac=0.45, rand_frac=0.05, k_active=12, tau=0.012)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
