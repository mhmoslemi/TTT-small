def _precompute_contribs(h, n):
    corr_len = 2 * n - 1
    out = np.zeros((n, corr_len))
    inn = np.zeros((n, corr_len))
    for i in range(n):
        lo_k = max(0, n - 1 - i)
        hi_k = min(corr_len - 1, 2 * n - 2 - i)
        if lo_k <= hi_k:
            d_range = np.arange(lo_k, hi_k + 1) - (n - 1)
            h_idx = i + d_range
            out[i, lo_k:hi_k+1] = 1.0 - h[h_idx]
        lo_k2 = max(0, i)
        hi_k2 = min(corr_len - 1, n - 1 + i)
        if lo_k2 <= hi_k2:
            d_range2 = np.arange(lo_k2, hi_k2 + 1) - (n - 1)
            h_idx2 = i - d_range2
            inn[i, lo_k2:hi_k2+1] = h[h_idx2]
    return out, inn

def _binary_greedy(h, deadline, max_iters=20000, stochastic_after=500):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    best_c5 = cur_max * 2.0 / n
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_c5
    rng = np.random.default_rng(777)
    no_improve = 0
    corr_len = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_ones = len(ones)
        n_zeros = len(zeros)
        out, inn = _precompute_contribs(h, n)
        A = corr[None, :] - out[ones] + inn[ones]
        B = out[zeros] - inn[zeros]
        total_size = n_ones * n_zeros * corr_len
        if total_size < 30_000_000:
            M = (A[:, None, :] + B[None, :, :]).max(axis=2)
        else:
            M = np.empty((n_ones, n_zeros))
            chunk = max(1, 30_000_000 // (n_zeros * corr_len))
            for i0 in range(0, n_ones, chunk):
                i1 = min(i0 + chunk, n_ones)
                M[i0:i1, :] = (A[i0:i1, None, :] + B[None, :, :]).max(axis=2)
        use_stoch = (it >= stochastic_after and rng.random() < 0.3)
        if use_stoch:
            i_idx = rng.integers(n_ones)
            j_idx = rng.integers(n_zeros)
        else:
            flat_idx = int(np.argmin(M))
            i_idx = flat_idx // n_zeros
            j_idx = flat_idx % n_zeros
        i = ones[i_idx]
        j = zeros[j_idx]
        new_max_val = float(M[i_idx, j_idx])
        if new_max_val >= cur_max - 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        c5 = cur_max * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        no_improve = 0
    return best_h, best_c5

def _binary_sa(h, deadline, n_iters=50000):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    cur_c5 = cur_max * 2.0 / n
    best_c5 = cur_c5
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T = 0.01
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    rng = np.random.default_rng(42)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_corr))
        new_c5 = new_max * 2.0 / n
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            cur_c5 = new_c5
            corr = new_corr
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _subgradient_optimize(h, deadline, max_iters=5000, step_init=0.15, beta_min=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = float(np.max(scores))
        active = scores > max_s - 0.005
        if not np.any(active):
            active = scores > max_s - 0.02
        if not np.any(active):
            active[np.argmax(scores)] = True
        grad = np.zeros(n)
        corr_len = 2 * n - 1
        for k in np.where(active)[0]:
            d = k - (n - 1)
            for i in range(n):
                g = 0.0
                ip = i + d
                if 0 <= ip < n:
                    g += (1.0 - h[ip])
                im = i - d
                if 0 <= im < n:
                    g -= h[im]
                grad[i] += g
        grad = grad / max(1.0, np.linalg.norm(grad))
        grad -= grad.mean()
        h_new = _project(h - step * grad, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
            step = min(step * 1.05, 0.3)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.001 or no_improve > 50:
                break
    return best_h, best_c5

def _resample_to_n(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish_full(h, deadline, n_passes=5, step_size=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if step_size is None:
        step_size = max(1, n // 16)
    rng = np.random.default_rng(99)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for _ in range(3):
                k = rng.integers(n)
                if k == j or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                signs = [1, -1]
                rng.shuffle(signs)
                for sign in signs:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    corr_len = 2 * n - 1
    G = np.zeros((corr_len, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        na = n - abs(d)
        if d >= 0:
            for i in range(na):
                G[r, i] += (1.0 - h[i+d])
                G[r, i+d] += -h[i]
        else:
            for i in range(na):
                G[r, i] += (1.0 - h[i-d])
                G[r, i-d] += -h[i]
    G *= 2.0 / n
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((corr_len, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.12, max_iters=2000, radius_min=0.002):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, radius)
        if p is None:
            stall += 1
            radius *= 0.6
            if radius < radius_min:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.95
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = alpha_max
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(25):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, target))
        best_a = c1 if f1 < f2 else c2
        best_f = min(f1, f2)
        if best_f < cur_c5 - 1e-14:
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    alphas = [
        (1.0 + np.sqrt(5.0)) / 2.0,
        1.0 / ((1.0 + np.sqrt(5.0)) / 2.0),
        np.sqrt(2.0),
        np.sqrt(3.0),
        np.pi / 2.0,
        np.e,
        2.0 - np.sqrt(2.0),
        (np.sqrt(5.0) - 1.0) / 2.0,
        1.0 / np.sqrt(2.0),
        np.sqrt(13.0) / 2.0,
    ]
    
    resolutions = [64, 80, 96, 112, n0, 128, 160, 192]
    seen = set()
    res_list = []
    for r in resolutions:
        if r >= 4 and r not in seen and r % 2 == 0:
            seen.add(r)
            res_list.append(r)
    
    # Phase 1: Binary greedy search from quasi-periodic inits at multiple resolutions
    phase1_deadline = t0 + budget_s * 0.30
    for n_try in res_list:
        if time.time() > phase1_deadline:
            break
        for alpha in alphas[:6]:
            if time.time() > phase1_deadline:
                break
            for phase in [0.0, 0.25, 0.5]:
                if time.time() > phase1_deadline:
                    break
                h_q = _quasi_binary(n_try, alpha, phase)
                sub_dl = min(phase1_deadline, time.time() + budget_s * 0.02)
                h_b, c5_b = _binary_greedy(h_q, sub_dl, max_iters=8000, stochastic_after=300)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    best_n = n_try
    
    # Also try binarizing the initial
    if time.time() < phase1_deadline:
        h_bin0 = _binarize(h_start, n0)
        sub_dl = min(phase1_deadline, time.time() + budget_s * 0.02)
        h_b0, c5_b0 = _binary_greedy(h_bin0, sub_dl, max_iters=8000, stochastic_after=300)
        if c5_b0 < best_c5:
            best_c5 = c5_b0
            best_h = h_b0.copy()
            best_n = n0
    
    # Phase 2: SA + more greedy from best binary
    phase2_deadline = t0 + budget_s * 0.42
    if time.time() < phase2_deadline:
        h_bin_best = _binarize(best_h, best_n)
        sub_dl = min(phase2_deadline, time.time() + budget_s * 0.025)
        h_sa, c5_sa = _binary_sa(h_bin_best, sub_dl, n_iters=30000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        sub_dl2 = min(phase2_deadline, time.time() + budget_s * 0.02)
        h_g2, c5_g2 = _binary_greedy(best_h, sub_dl2, max_iters=10000, stochastic_after=200)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()
    
    # Phase 3: Multi-resolution binary exploration with more alphas
    phase3_deadline = t0 + budget_s * 0.52
    for n_try in res_list:
        if time.time() > phase3_deadline:
            break
        for alpha in alphas:
            if time.time() > phase3_deadline:
                break
            h_q = _quasi_binary(n_try, alpha, 0.0)
            sub_dl = min(phase3_deadline, time.time() + budget_s * 0.012)
            h_b, c5_b = _binary_greedy(h_q, sub_dl, max_iters=5000, stochastic_after=200)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_try
    
    # Phase 4: Continuous subgradient optimization from best binary
    phase4_deadline = t0 + budget_s * 0.62
    if time.time() < phase4_deadline:
        h_subdl = min(phase4_deadline, time.time() + budget_s * 0.04)
        h_sg, c5_sg = _subgradient_optimize(best_h, h_subdl, max_iters=3000, step_init=0.12)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        # LP from continuous result
        if time.time() < phase4_deadline:
            h_subdl2 = min(phase4_deadline, time.time() + budget_s * 0.03)
            h_lp, c5_lp = _lp_optimize(best_h, h_subdl2, radius_init=0.10, max_iters=1500)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
    
    # Phase 5: More continuous optimization with different parameters
    phase5_deadline = t0 + budget_s * 0.72
    if time.time() < phase5_deadline:
        h_subdl3 = min(phase5_deadline, time.time() + budget_s * 0.03)
        h_sg2, c5_sg2 = _subgradient_optimize(best_h, h_subdl3, max_iters=2000, step_init=0.06)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
        if time.time() < phase5_deadline:
            h_subdl4 = min(phase5_deadline, time.time() + budget_s * 0.03)
            h_lp2, c5_lp2 = _lp_optimize(best_h, h_subdl4, radius_init=0.08, max_iters=1500)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h_lp2.copy()
    
    # Phase 6: Resample to other resolutions and optimize
    phase6_deadline = t0 + budget_s * 0.80
    for n_try in res_list:
        if time.time() > phase6_deadline:
            break
        if n_try == best_n:
            continue
        h_res = _resample_to_n(best_h, n_try)
        # Binary polish at this resolution
        h_bin_r = _binarize(h_res, n_try)
        sub_dl = min(phase6_deadline, time.time() + budget_s * 0.015)
        h_br, c5_br = _binary_greedy(h_bin_r, sub_dl, max_iters=5000, stochastic_after=200)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
            best_n = n_try
        else:
            # Try continuous from resampled
            sub_dl2 = min(phase6_deadline, time.time() + budget_s * 0.01)
            h_lr, c5_lr = _subgradient_optimize(h_res, sub_dl2, max_iters=1000, step_init=0.08)
            if c5_lr < best_c5:
                best_c5 = c5_lr
                best_h = h_lr.copy()
                best_n = n_try
    
    # Phase 7: Pair polish
    phase7_deadline = t0 + budget_s * 0.88
    if time.time() < phase7_deadline:
        h_p, c5_p = _pair_polish_full(best_h, phase7_deadline, n_passes=4)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
    
    # Phase 8: Final LP
    phase8_deadline = t0 + budget_s * 0.96
    if time.time() < phase8_deadline:
        h_fin, c5_fin = _lp_optimize(best_h, phase8_deadline, radius_init=0.05, max_iters=3000, radius_min=0.001)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Final: verify and return
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    # Safety: ensure validity
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
