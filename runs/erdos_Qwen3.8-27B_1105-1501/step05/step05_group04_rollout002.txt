    G = np.zeros(n)
    n_lags = 2 * n - 1
    for r in range(n_lags):
        w = weights[r]
        if abs(w) < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            # grad[j] += w * (1-h[j-d]) for j >= d
            # grad[j] += w * (-h[j+d]) for j < n-d
            G[d:] += w * (1.0 - h[:n-d])
            # Wait, G[j] for j >= d gets (1-h[j-d]), so G[d:] += w*(1-h[:n-d])
            # G[j] for j < n-d gets -h[j+d], so G[:n-d] += w*(-h[d:])
            G[:n-d] += w * (-h[d:])
        else:
            b = -d
            # grad[j] += w * (1-h[j+b]) for j < n-b
            # grad[j] += w * (-h[j-b]) for j >= b
            G[:n-b] += w * (1.0 - h[b:])
            G[b:] += w * (-h[:n-b])
    return G

def _soft_lse_obj_grad(a, n, beta):
    m = n // 2
    h = _h_from_a(a, n)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    n_lags = len(V)
    
    Vmax = float(np.max(V))
    exps = np.exp((V - Vmax) / beta)
    sum_exps = float(np.sum(exps))
    phi = Vmax + beta * np.log(sum_exps)
    w = exps / sum_exps
    
    G_h = _compute_lag_grad_weighted(h, w, n)
    
    G_a = np.zeros(m)
    for i in range(m):
        G_a[i] = G_h[i] - G_h[n-1-i]
    
    return phi, G_a

def _soft_margin_obj_grad(a, n, B, delta):
    m = n // 2
    h = _h_from_a(a, n)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    n_lags = len(V)
    
    z = (V - B) / delta
    obj = 0.0
    w = np.zeros(n_lags)
    for r in range(n_lags):
        zr = z[r]
        if zr > 50:
            w[r] = 1.0
            obj += delta * zr
        elif zr > -20:
            ez = np.exp(zr)
            w[r] = ez / (1.0 + ez)  # sigmoid
            obj += delta * np.log1p(ez)
        else:
            w[r] = 0.0
            obj += 0.0
    
    G_h = _compute_lag_grad_weighted(h, w, n)
    G_a = np.zeros(m)
    for i in range(m):
        G_a[i] = G_h[i] - G_h[n-1-i]
    
    return obj, G_a

def _callback(x):
    """Callback for L-BFGS-B to track best exact solution."""
    if _G['timeout']:
        return True  # signal to stop
    if time.time() > _G['deadline']:
        _G['timeout'] = True
        return True
    try:
        n = _G['n']
        h = _h_from_a(x, n)
        c = _c5(h)
        if c < _G['best_c5']:
            _G['best_c5'] = c
            _G['best_h'] = h.copy()
    except Exception:
        pass
    return False

def _lbfgs_optimize(a_init, n, obj_func, deadline, maxiter=300, **kwargs):
    """Run L-BFGS-B optimization."""
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    options = {
        'maxiter': maxiter,
        'ftol': 1e-10,
        'gtol': 1e-8,
        'disp': False,
    }
    options.update(kwargs)
    
    _G['n'] = n
    _G['deadline'] = deadline
    _G['timeout'] = False
    
    try:
        res = minimize(obj_func, a_init, jac=True, method='L-BFGS-B',
                      bounds=bounds, callback=_callback, options=options)
    except Exception:
        pass
    
    # Final evaluation
    if not _G['timeout']:
        try:
            h = _h_from_a(res.x, n)
            c = _c5(h)
            if c < _G['best_c5']:
                _G['best_c5'] = c
                _G['best_h'] = h.copy()
        except Exception:
            pass
    
    return _G['best_h'], _G['best_c5']

def _resample_sym(h_old, n_new):
    """Resample h to new resolution and project to symmetric form."""
    n_old = len(h_old)
    if n_new == n_old:
        return _a_from_h(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    a_new = _a_from_h(h_new)
    return a_new
