import numpy as np
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    """Exact line search in direction d (zero-mean) for the C5 objective."""
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find valid range for t
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lse_gradient(h, dx, beta):
    """Compute smoothed gradient using log-sum-exp."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    
    # Compute gradient of sum_k w_k * corr_k
    # d corr_k / dh[j] as derived above
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        k = r - (n - 1)  # this is the shift
        wm = w[r]
        # d c_k / dh[j] = (1-h[j]) * I[j in shifted range] - h[j] * I[j in original range]
        # Shifted range: max(k,0) to min(n-1+k, n-1) for the (1-h[j]) term
        # Original range: max(0,-k) to min(n-1, n-1-k) for the -h[j] term
        j_lo_s = max(k, 0)
        j_hi_s = min(n - 1 + k, n - 1)
        if j_lo_s <= j_hi_s:
            grad[j_lo_s:j_hi_s + 1] += wm * (1.0 - h[j_lo_s:j_hi_s + 1])
        j_lo_o = max(0, -k)
        j_hi_o = min(n - 1, n - 1 - k)
        if j_lo_o <= j_hi_o:
            grad[j_lo_o:j_hi_o + 1] -= wm * h[j_lo_o:j_hi_o + 1]
    
    grad -= np.mean(grad)  # zero-mean to preserve mass
    return grad

def _subgrad_descent(h, deadline, beta_list=(50, 150, 400, 1000), lr_init=0.015, momentum=0.85):
    """Projected subgradient descent with momentum and adaptive beta."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    bi = 0
    
    while time.time() < deadline:
        if stall > 50:
            bi += 1
            if bi > len(beta_list) * 4:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
            continue
        
        beta = beta_list[bi % len(beta_list)]
        d = _lse_gradient(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            bi += 1
            continue
        d = d / dn
        mom = momentum * mom + (1.0 - momentum) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.04, 0.3)
        else:
            stall += 1
            lr *= 0.82
    
    return best_h, best_c5

def _pair_refine(h, deadline, random_pairs=0, rng=None):
    """Exhaustive or random pair refinement."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    improved_any = True
    
    while improved_any and time.time() < deadline:
        improved_any = False
        rounds = 0
        while improved_any and rounds < 5 and time.time() < deadline:
            improved_any = False
            rounds += 1
            
            if random_pairs > 0 and rng is not None:
                # Random pairs from interior
                for _ in range(random_pairs):
                    if time.time() > deadline:
                        break
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
                    if not (1e-10 < h[i] < 1 - 1e-10):
                        continue
                    if not (1e-10 < h[j] < 1 - 1e-10):
                        continue
                    if i > j:
                        i, j = j, i
                    d = np.zeros(n)
                    d[i] = -1.0
                    d[j] = 1.0
                    alpha, c5_try = _line_search(h, d, dx, 0.5)
                    if c5_try < best_c5 - 1e-13:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved_any = True
            else:
                # All pairs
                for i in range(n):
                    if time.time() > deadline:
                        return best_h, best_c5
                    if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                        continue
                    for j in range(i + 1, n):
                        if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                            continue
                        d = np.zeros(n)
                        d[i] = -1.0
                        d[j] = 1.0
                        alpha, c5_try = _line_search(h, d, dx, 0.5)
                        if c5_try < best_c5 - 1e-13:
                            h = h + alpha * d
                            h = _project(h, target)
                            cur = _exact_c5(h)
                            if cur < best_c5:
                                best_c5 = cur
                                best_h = h.copy()
                                improved_any = True
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _logistic_map(v, target):

    lo = -np.max(v) - 30

        else:
            hi = mid
    theta = (lo + hi) / 2
    h = sigmoid(v + theta)
    return _project(h, target)

def _les_evolution(h_init, deadline, rng, pop_size=8, n_mut=15):
    """Logistic Evolution Strategy."""
    n = len(h_init)
    target = n / 2.0
    h_init = np.clip(h_init, 1e-8, 1 - 1e-8)
    v_init = np.log(h_init / (1.0 - h_init))
    
    # Initial population
    pop_v = [v_init.copy()]
    pop_h = [h_init.copy()]
    pop_c = [_exact_c5(h_init)]
    
    # Add diverse initializations
    for _ in range(pop_size - 1):
        noise = rng.normal(0, 0.3, n)
        noise -= noise.mean()
        v_new = v_init + noise
        h_new = _logistic_map(v_new, target)
        pop_v.append(v_new)
        pop_h.append(h_new)
        pop_c.append(_exact_c5(h_new))
    
    best_idx = int(np.argmin(pop_c))
    best_h = pop_h[best_idx].copy()
    best_c5 = pop_c[best_idx]
    
    sigma = 0.12
    gen = 0
    no_improve = 0
    
    while time.time() < deadline and gen < 300:
        gen += 1
        # Select elites
        order = np.argsort(pop_c)
        n_elite = max(2, pop_size // 3)
        elites = list(range(n_elite))
        
        # Generate offspring
        offspring_v = []
        offspring_h = []
        offspring_c = []
        
        for _ in range(n_mut):
            # Pick parents
            pa = pop_v[elites[int(rng.integers(0, n_elite))]]
            pb = pop_v[elites[int(rng.integers(0, n_elite))]]
            
            # Crossover + mutation
            if rng.random() < 0.5:
                v_child = 0.5 * pa + 0.5 * pb
            else:
                v_child = pa.copy()
            
            v_child += rng.normal(0, sigma, n)
            v_child -= v_child.mean()
            
            h_child = _logistic_map(v_child, target)
            c_child = _exact_c5(h_child)
            
            offspring_v.append(v_child)
            offspring_h.append(h_child)
            offspring_c.append(c_child)
        
        # Combine and select
        all_v = pop_v + offspring_v
        all_h = pop_h + offspring_h
        all_c = pop_c + offspring_c
        
        order = np.argsort(all_c)
        best_found = all_c[order[0]]
        
        if best_found < best_c5 - 1e-13:
            best_c5 = best_found
            best_h = all_h[order[0]].copy()
            no_improve = 0
        else:
            no_improve += 1
        
        pop_v = [all_v[i] for i in order[:pop_size]]
        pop_h = [all_h[i] for i in order[:pop_size]]
        pop_c = [all_c[i] for i in order[:pop_size]]
        
        # Adaptive sigma
        if no_improve > 15:
            sigma *= 1.3
            no_improve = 0
        elif best_found < best_c5:
            sigma = max(sigma * 0.95, 0.005)
        sigma = min(sigma, 1.0)
        
        # Also try local refinement on best
        if time.time() < deadline and _exact_c5(best_h) >= best_c5:
            sub_dl = min(deadline, time.time() + 2.0)
            h_ref, c5_ref = _pair_refine(best_h, sub_dl, random_pairs=500, rng=rng)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

def _kick_reopt(h, deadline, rng, n_kicks=30):
    """Random kicks followed by local reoptimization."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _ in range(n_kicks):
        if time.time() > deadline:

def _triple_refine(h, deadline, rng, n_triples=5000):
    """Triple-point refinement: move mass from one point to two others."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _ in range(n_triples):
        if time.time() > deadline:
            break
        i = int(rng.integers(0, n))
        if not (1e-10 < h[i] < 1 - 1e-10):
            continue
        j = int(rng.integers(0, n))
        if j == i or not (1e-10 < h[j] < 1 - 1e-10):
            continue
        k = int(rng.integers(0, n))
        if k == i or k == j:
            continue
        if not (1e-10 < h[k] < 1 - 1e-10):
            continue
        
        # Try both: i -> j,k and j,k -> i
        for sign in [1, -1]:
            d = np.zeros(n)
            d[i] = -sign
            d[j] = sign * 0.5
            d[k] = sign * 0.5
            alpha, c5_try = _line_search(h, d, dx, 0.4)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                break
    
    return best_h, best_c5
