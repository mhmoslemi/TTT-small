import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _build_G(h):
    """Build the gradient matrix G where G[r, j] = dS_{lag}/dh_j * dx"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] = (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] = (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G * (2.0 / n)

def _safe_sigma(t):
    """Safe logistic function"""
    result = np.empty_like(t, dtype=float)
    pos = t >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    result[~pos] = exp_t / (1.0 + exp_t)
    return result

def _solve_mu(q, target):

def _complement_lp(h, deadline):
    """Solve LP to find complement profile y"""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Build LP
    # Variables: y_0,...,y_{n-1}, u (total n+1)
    # Minimize u
    # Subject to: A_d · y + u >= b_d for all d
    #             sum(y) = target
    #             0 <= y_j <= 1
    #             u >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize u
    
    # Inequality: -A_d · y - u <= -b_d
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    
    for d in range(-(n-1), n):
        r = n - 1 + d
        # b_d = sum of h[i] over valid i
        if d >= 0:
            m = n - d
            b_d = float(np.sum(h[d:]))
            # A_{d,j} = h[j+d] for 0 <= j <= n-1-d
            A_ub[r, :m] = -h[d:]
        else:
            g = -d
            m = n - g
            b_d = float(np.sum(h[:m]))
            # A_{d,j} = h[j-g] for g <= j <= n-1
            A_ub[r, g:] = -h[:m]
        A_ub[r, n] = -1.0  # coefficient of u
        b_ub[r] = -b_d
    
    # Equality: sum(y) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            y = res.x[:n]
            y = _balance_exact(y, target)
            return y
    except Exception:
        pass

    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):

        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] = (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
