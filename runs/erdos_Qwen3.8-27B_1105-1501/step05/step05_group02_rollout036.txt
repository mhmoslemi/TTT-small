def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _thue_morse(n):
    x = np.zeros(n, dtype=np.int8)
    for i in range(n):
        x[i] = bin(i).count('1') & 1

    target = n // 2
    if n_ones < target:
        zeros = np.where(x == 0)[0]
        x[zeros[:target - n_ones]] = 1
    elif n_ones > target:
        ones = np.where(x == 1)[0]
        x[ones[:n_ones - target]] = 0
    return x

def _golden_seq(n, alpha):

def _init_scores(x):
    n = len(x)
    scores = np.zeros(2 * n - 1, dtype=np.float64)
    for d in range(1, n):
        m = n - d
        scores[n - 1 - d] = float(np.dot(x[:m], 1 - x[d:]))
        scores[n - 1 + d] = float(np.dot(x[d:], 1 - x[:m]))
    return scores

def _flip_update(x, i, scores):
    n = len(x)
    old = x[i]
    if old == 1:
        if i > 0:
            d_left = np.arange(1, i + 1)
            left = x[i - d_left].astype(np.float64)
            scores[n - 1 - d_left] += left
            scores[n - 1 + d_left] -= (1.0 - left)
        if i < n - 1:
            d_right = np.arange(1, n - i)
            right = x[i + d_right].astype(np.float64)
            scores[n - 1 - d_right] -= (1.0 - right)
            scores[n - 1 + d_right] += right
    else:
        if i > 0:
            d_left = np.arange(1, i + 1)
            left = x[i - d_left].astype(np.float64)
            scores[n - 1 - d_left] -= left
            scores[n - 1 + d_left] += (1.0 - left)

            right = x[i + d_right].astype(np.float64)
            scores[n - 1 - d_right] += (1.0 - right)
            scores[n - 1 + d_right] -= right
    x[i] = 1 - old

    for it in range(max_iters):

            break
        if no_improve > stall_threshold:

        i_idx = rng.integers(len(ones))

        _flip_update(x, j, scores)
