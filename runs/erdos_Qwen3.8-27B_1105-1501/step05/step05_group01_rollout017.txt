```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-12
    neg = d < -1e-12
    t_hi = np.inf
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    t_lo = 0.0
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
    t_hi = min(t_hi, radius)
    if t_hi <= t_lo + 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
    gl, gr = t_lo, t_hi
    c1 = gl + gr_phi * (gr - gl)
    c2 = gr - gr_phi * (gr - gl)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    for _ in range(50):
        if gr - gl < 1e-13:
            break
        if f1 < f2:
            gr, c2, f2 = c1, None, None
            c2 = gr - gr_phi * (gr - gl)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
        else:
            gl, c1, f1 = c2, None, None
            c1 = gl + gr_phi * (gr - gl)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
    
    opt_t = (gl + gr) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _random_pair_dir(h, rng):
    n = len(h)
    ones = np.where(h > 0.1)[0]
    zeros = np.where(h < 0.9)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None
    i = int(rng.integers(0, len(ones)))
    j = int(rng.integers(0, len(zeros)))
    if ones[i] == zeros[j]:
        return None
    d = np.zeros(n)
    d[ones[i]] = -1.0
    d[zeros[j]] = 1.0
    return d

def _random_block_dir(h, rng, block_size):
    n = len(h)
    # Pick block_size random positions to decrease and block_size to increase
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    if len(ones) < block_size or len(zeros) < block_size:
        return None
    A = set(rng.choice(ones, size=block_size, replace=False).tolist())
    B = set(rng.choice(zeros, size=block_size, replace=False).tolist())
    overlap = A & B
    if len(overlap) > 0:
        return None
    d = np.zeros(n)
    for i in A:
        d[i] = -1.0
    for j in B:
        d[j] = 1.0
    return d

def _targeted_dir(h, scores, rng, block_size=1):
    n = len(h)
    worst_idx = int(np.argmax(scores))
    lag = worst_idx - (n - 1)
    
    if lag == 0:
        frac = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
        if len(frac) < 2 * block_size:
            return None
        A = set(rng.choice(frac, size=block_size, replace=False).tolist())
        remaining = np.array([x for x in frac if x not in A])
        if len(remaining) < block_size:
            return None
        B = set(rng.choice(remaining, size=block_size, replace=False).tolist())
        d = np.zeros(n)
        for i in A:
            d[i] = 1.0
        for j in B:
            d[j] = -1.0
        return d
    
    k = abs(lag)
    # For |lag| = k:
    # Positive lag k: score = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    #   Problem: h[i+k]=1, h[i]=0 (contributes 1)
    #   Fix: increase h[i] (if safe) or decrease h[i+k] (if safe)
    # Negative lag -k: score = sum_{i=k}^{n-1} h[i-k]*(1-h[i])
    #   Problem: h[i-k]=1, h[i]=0 (contributes 1)
    #   Fix: increase h[i] (if safe) or decrease h[i-k] (if safe)
    
    inc_cands = []
    dec_cands = []
    
    if lag > 0:
        for i in range(n - k):
            if h[i] < 0.3 and h[i + k] > 0.7:
                inc_cands.append(i)
            if h[i + k] > 0.7 and h[i] < 0.3:
                dec_cands.append(i + k)
    else:
        for i in range(k, n):
            if h[i] < 0.3 and h[i - k] > 0.7:
                inc_cands.append(i)
            if h[i - k] > 0.7 and h[i] < 0.3:
                dec_cands.append(i - k)
    
    if len(inc_cands) < block_size or len(dec_cands) < block_size:
        return None
    
    A = set(rng.choice(np.array(dec_cands), size=block_size, replace=False).tolist())
    B = set(rng.choice(np.array(inc_cands), size=block_size, replace=False).tolist())
    if A & B:
        # Remove overlap
        B = B - A
        if len(B) < block_size:
            return None
    
    d = np.zeros(n)
    for i in A:
        d[i] = -1.0
    for j in list(B)[:block_size]:
        d[j] = 1.0
    return d
