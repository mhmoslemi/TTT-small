import numpy as np

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def build_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        r = n - 1 + d
        m = n - d
        if m > 0:
            G[r, :m] += (1.0 - h[d:d + m])
            G[r, d:] += -h[:m]
        r2 = n - 1 - d
        m2 = n - d
        if m2 > 0:
            G[r2, d:] += (1.0 - h[:m2])
            G[r2, :m2] += -h[d:d + m2]
    r0 = n - 1
    G[r0] += 1.0 - 2.0 * h
    return G

def exact_line_search(h, d, S0, dx, max_alpha=None):
    n = len(h)
    A1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    A2 = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    a_lo = -1e15
    a_hi = 1e15
    if np.any(pos):
        a_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        a_lo = float(np.max(h[neg] / d[neg]))
    if a_lo < 0:
        a_lo = max(a_lo, -max_alpha) if max_alpha else a_lo
    if a_hi > 0:
        a_hi = min(a_hi, max_alpha) if max_alpha else a_hi
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    ng = 24
    grid = np.linspace(a_lo, a_hi, ng)
    vals = S0[None, :] + grid[:, None] * A1[None, :] - (grid[:, None] ** 2) * A2[None, :]
    obj = np.max(vals, axis=1) * dx
    bi = int(np.argmin(obj))
    bt = grid[bi]
    bv = obj[bi]
    if 0 < bi < ng - 1:
        a, b = grid[bi - 1], grid[bi + 1]
    else:
        a, b = a_lo, a_hi
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * A1 - c1 * c1 * A2)) * dx
    f2 = float(np.max(S0 + c2 * A1 - c2 * c2 * A2)) * dx
    for _ in range(22):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * A1 - c1 * c1 * A2)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * A1 - c2 * c2 * A2)) * dx
    ot = (a + b) * 0.5
    ov = float(np.max(S0 + ot * A1 - ot * ot * A2)) * dx
    if ov < bv:
        bt, bv = ot, ov
    return bt, bv

def lp_direction(h, S0, G, radius):
    n = len(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1 + n, n + 1))
    A_ub[:2 * n - 1, :n] = G
    A_ub[:2 * n - 1, n] = -1.0
    b_ub = np.full(2 * n - 1, 0.0) - S0
    A_ub[2 * n - 1:2 * n - 1 + n, :n] = np.eye(n)
    A_ub[2 * n - 1:2 * n - 1 + n, n] = 0.0
    b_ub2 = np.full(n, radius)
    A_ub[2 * n - 1 + n:2 * n - 1 + 2 * n, :n] = -np.eye(n)
    b_ub3 = np.full(n, radius)
    full_b_ub = np.concatenate([b_ub, b_ub2, b_ub3])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    res = linprog(c_vec, A_ub=A_ub, b_ub=full_b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] * (n + 1), method='highs')
    if res.success:
        return res.x[:n]
    return None

                break
            continue
        alpha, new_c5 = exact_line_search(h, d, S0, dx, max_alpha=radius)
        if new_c5 < cur_c5 - 1e-13:

            h = project(h, target)

                best_h = h.copy()
            stall = 0

            ratio = cur_c5 / max(abs(pred), 1e-15)
            if ratio < 0.8:

def pair_line_search(h, i, j, S0, dx):

    d[j] = 1.0

def pair_optimize(h, deadline, beam=40):

            ii = int(rng.integers(n))
            jj = int(rng.integers(n))

                continue

    d7 = t0 + budget_s * 0.96
    h7, c5_7 = fine_pair_refine(best_h, d7)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h_f, c5_f = lp_optimize(best_h, deadline, radius_init=0.04)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, final_c5, len(best_h)
