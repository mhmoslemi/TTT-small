def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.5
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)

    G = np.zeros((2 * n - 1, n))
    for s in range(-(n-1), n):

        m = n - abs(s)

        else:
            a = -s
            G[r, :m] += dx * (1.0 - h[a:])
            G[r, a:] += -dx * h[:m]

    n = len(h0)

        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        h = x[:n]
        t = x[-1]
        s = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        return t - s[active]
    
    def ineq_jac(x):

        G = _build_G(h)
        J = np.zeros((top_k, n + 1))
        J[:, :n] = -G[active]
        J[:, n] = 1.0
        return J

        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J

        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},

                       bounds=bounds, constraints=constraints,
