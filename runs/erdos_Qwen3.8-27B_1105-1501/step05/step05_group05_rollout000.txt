def _exact_c5(h):

def _project(v, target):

def _grad_for_lag(h, d_lag):
    n = len(h)
    g = np.zeros(n)
    if d_lag >= 0:
        m = n - d_lag
        if m > 0:
            g[:m] += (1.0 - h[d_lag:d_lag + m])
            g[d_lag:] += -h[:m]
    else:
        gg = -d_lag
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:gg + m])
            g[gg:] += -h[:m]
    return g

def _lsumexp_grad(h, dx, beta, top_k=None):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d_lag = r - (n - 1)
        g = _grad_for_lag(h, d_lag)
        grad += w[r] * g
    grad -= np.mean(grad)
    return grad

def _active_balance_dir(h, dx, top_k=20, tol_frac=0.02):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    thresh = s_max - tol_frac * s_max
    active = np.where(scores >= thresh)[0]
    if len(active) == 0:
        return None
    if len(active) > top_k:
        order = np.argsort(scores[active])[::-1][:top_k]
        active = active[order]
    K = len(active)
    G = np.zeros((K + 1, n))
    for i, r in enumerate(active):
        d_lag = r - (n - 1)
        G[i, :] = _grad_for_lag(h, d_lag)
    G[K, :] = 1.0 / np.sqrt(n)
    b = np.zeros(K + 1)
    b[:K] = -1.0
    lam = 0.1
    A = G.T @ G + lam * np.eye(n)
    rhs = G.T @ b
    try:
        d = np.linalg.solve(A, rhs)
    except np.linalg.LinAlgError:
        d = np.linalg.lstsq(A, rhs, rcond=None)[0]
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return None
    return d / dn

        n_pairs = min(400, n * 3)
        for _ in range(n_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if h_cur[i] <= 1e-10 or h_cur[i] >= 1.0 - 1e-10:
                continue
            if h_cur[j] <= 1e-10 or h_cur[j] >= 1.0 - 1e-10:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h_cur, d, dx, 0.5)
            if c5_try < best_c5 - 1e-13:
                h_cur = h_cur + alpha * d
                h_cur = _project(h_cur, target)
                cur = _exact_c5(h_cur)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h_cur.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kick_reopt(h, deadline, rng, n_kicks=25):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.015, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = _project(best_h + scale * direction, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _combined_opt(h_k, sub_dl, beta_list=(60, 150, 300))
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)

    # Phase 1: Lift to n=256 and optimize (35% budget)
    d1 = t0 + budget_s * 0.35
    h256 = _resample(h_start, 256)
    c5_256 = _exact_c5(h256)
    if c5_256 < best_c5:
        best_c5 = c5_256
        best_h = h256.copy()
    h1, c5_1 = _combined_opt(h256, d1, beta_list=(40, 80, 160, 400), use_active=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP polish on current best (15% budget)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _lp_opt(best_h, d2, radius_init=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (20% budget)
    d3 = t0 + budget_s * 0.70
    n_candidates = [192, 224, 288, 320, 384, 448, 512]
    for nc in n_candidates:
        if time.time() > d3:
            break
        h_r = _resample(best_h, nc)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _combined_opt(h_r, sub_dl, beta_list=(60, 120, 250), use_active=True)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Continue optimization on best resolution (15% budget)
    d4 = t0 + budget_s * 0.85
    h4, c5_4 = _combined_opt(best_h, d4, beta_list=(50, 100, 200, 500), use_active=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Kicks and reoptimization (8% budget)
    d5 = t0 + budget_s * 0.93
    h5, c5_5 = _kick_reopt(best_h, d5, rng, n_kicks=20)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final pair refinement (7% budget)
    d6 = t0 + budget_s - 2.0
    h6, c5_6 = _pair_refine(best_h, d6, rng, max_passes=60)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
