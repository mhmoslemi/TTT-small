    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binary_local_search(h, deadline, max_sweeps=200):
    """Exhaustive binary local search: try all one-zero swaps, keep best."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = 0.0
        best_a = -1
        best_b = -1
        
        for a in ones:
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                new_c5 = _exact_c5(h)
                if new_c5 < best_c5 - 1e-15:
                    best_improve = new_c5
                    best_a = a
                    best_b = b
                h[a] = 1.0
                h[b] = 0.0
        
        if best_a == -1:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        cur_c5 = best_improve
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5

def _binary_local_search_fast(h, deadline, max_sweeps=500):
    """Faster version using precomputed counts and O(n) delta per swap."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    
    # Precompute counts for d=1..n-1
    counts = np.zeros(n - 1)
    for d in range(1, n):
        m = n - d
        counts[d-1] = np.sum(h[:m] * (1.0 - h[d:]))
    
    cur_max = float(np.max(counts)) * dx
    best_h = h.copy()
    best_c5 = cur_max
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h <= 0.5)[0])
        if not ones or not zeros:
            break
        
        best_new_max = cur_max + 1.0
        best_a = -1
        best_b = -1
        
        for a in ones:
            for b in zeros:
                # Compute new max count after swap
                new_max = 0.0
                for d in range(1, n):
                    m = n - d
                    delta = 0.0
                    # p = a
                    if a < m:
                        pd = a + d
                        old_t = h[a] * (1.0 - h[pd])
                        nh_a = 0.0
                        nh_pd = 1.0 if pd == b else h[pd]
                        new_t = nh_a * (1.0 - nh_pd)
                        delta += new_t - old_t
                    # p = a-d
                    if a >= d:
                        p = a - d
                        pd = p + d
                        old_t = h[p] * (1.0 - h[pd])
                        nh_p = 1.0 if p == b else h[p]
                        nh_pd = 0.0
                        new_t = nh_p * (1.0 - nh_pd)
                        delta += new_t - old_t
                    # p = b
                    if b < m:
                        pd = b + d
                        old_t = h[b] * (1.0 - h[pd])
                        nh_b = 1.0
                        nh_pd = 0.0 if pd == a else h[pd]
                        new_t = nh_b * (1.0 - nh_pd)
                        delta += new_t - old_t
                    # p = b-d
                    if b >= d:
                        p = b - d
                        pd = p + d
                        old_t = h[p] * (1.0 - h[pd])
                        nh_p = 0.0 if p == a else h[p]
                        nh_pd = 1.0
                        new_t = nh_p * (1.0 - nh_pd)
                        delta += new_t - old_t
                    
                    new_c = counts[d-1] + delta
                    if new_c > new_max:
                        new_max = new_c
                
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_a = a
                    best_b = b
        
        if best_a == -1 or best_new_max >= cur_max - 1e-15:
            break
        
        # Apply best swap
        h[best_a] = 0.0
        h[best_b] = 1.0
        # Update counts
        for d in range(1, n):
            m = n - d
            delta = 0.0
            if best_a < m:
                pd = best_a + d
                old_t = 1.0 * (1.0 - (1.0 if pd == best_b else (0.0 if h[pd] > 0.5 else 0.0)))
                # Recompute from scratch for safety
            # Just recompute all counts
            break
        counts = np.zeros(n - 1)
        for d in range(1, n):
            m = n - d
            counts[d-1] = np.sum(h[:m] * (1.0 - h[d:]))
        
        cur_max = float(np.max(counts)) * dx
        if cur_max < best_c5:
            best_c5 = cur_max
            best_h = h.copy()
    
    return best_h, best_c5

def _lse_objective(h, tau, target):
    """Log-sum-exp smooth approximation of C5."""
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = (scores - max_s) / tau
    shifted = np.clip(shifted, -500, 500)
    log_sum = max_s + tau * (np.log(np.sum(np.exp(shifted))) - np.log(len(scores)))
    # Correction: we want max + tau * log(sum exp((C_r - C_max)/tau))
    # = max + tau * log(sum exp(shifted)) where shifted = (scores-max)/tau
    # But this isn't quite right. Let me use: max + tau * (logsumexp(shifted) - log(n_lags))
    # Actually the standard log-sum-exp: logsumexp(x) = max + log(sum(exp(x-max)))
    # So phi = logsumexp(scores/tau) * tau = tau * (max(scores)/tau + log(sum(exp((scores-max)/tau))))
    #       = max(scores) + tau * log(sum(exp((scores-max)/tau)))
    # This is NOT the same as max + tau*log(average). It's the soft max.
    return float(max_s + tau * np.log(np.sum(np.exp(shifted))))

def _lse_gradient(h, tau):
    """Analytical gradient of log-sum-exp objective."""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = (scores - max_s) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)  # normalized weights
    
    # Gradient of C_r with respect to h_j
    # For lag d (shift = r - (n-1)):
    # If d >= 0: C_d = dx * sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
    #   dC_d/dh_j = dx * [-(h[j+d]) if j <= n-d-1] + dx * [(1-h[j-d]) if j >= d and j-d <= n-d-1]
    #   Wait, let me be more careful.
    #   C_d = dx * sum_{i} h[i]*(1-h[i+d]) for i=0..n-d-1
    #   dC_d/dh_j: j appears as h[i] when i=j (if j < n-d), contribution dx*(1-h[j+d])
    #              j appears as h[i+d] when i=j-d (if j-d >= 0 and j-d < n-d), contribution dx*(-h[j-d])
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            wr = w[r] * dx
            # j as first index: j in [0, m-1], contributes wr*(1-h[j+d])
            grad[:m] += wr * (1.0 - h[d:d+m])
            # j as second index: j in [d, n-1], j-d in [0, m-1], contributes -wr*h[j-d]
            grad[d:] += -wr * h[:m]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            wr = w[r] * dx
            # For negative lag -g: C_{-g} = dx * sum_{i=0}^{n-g-1} h[i+g]*(1-h[i])
            # j as second index (h[i]): j in [0, m-1], contributes wr*(1-h[j+g])... 
            # Wait, let me use the np.correlate convention.
            # np.correlate(h, 1-h, 'full')[r] = sum_i h[i+r-(n-1)] * (1-h)[i] (roughly)
            # Actually for equal-length inputs, np.correlate(a, b, 'full')[k] 
            # = sum_i a[i+k] * b[i] with appropriate ranges.
            # Let me just use the gradient formula that matches the scores.
            # For d < 0, let g = -d > 0.
            # C_{-g} in our correlation: sum_{i=0}^{n-g-1} h[i+g]*(1-h[i])
            # j appears as h[i+g] when i = j-g (if 0 <= j-g <= n-g-1), contributes wr*(-h[j-g])... 
            # No: d/dh_j [h[i+g]*(1-h[i])] where j = i+g: (1-h[i]) = (1-h[j-g])
            # j appears as h[i] when i = j (if 0 <= j <= n-g-1), contributes wr*(-h[j+g])
            # Wait I'm getting confused. Let me just verify numerically.
            # For now, use the same structure as positive lag but with h and (1-h) swapped roles.
            # C_{-g} = dx * sum_{i=0}^{n-g-1} h[i+g]*1 - h[i+g]*h[i]
            # dC/dh_j: j = i+g => i=j-g, valid if 0<=j-g<=n-g-1 i.e. g<=j<=n-1
            #   contribution: dx*(1 - h[j-g])
            # j = i => i=j, valid if 0<=j<=n-g-1
            #   contribution: -dx*h[j+g]
            grad[g:] += wr * (1.0 - h[:m])
            grad[:m] += -wr * h[g:]
    
    return grad

def _lse_optimize(h0, deadline, target, tau_init=0.01, tau_steps=None):
    """Optimize using log-sum-exp SLSQP."""
    n = len(h0)
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    if tau_steps is None:
        tau_steps = [tau_init, tau_init * 0.4, tau_init * 0.15, tau_init * 0.06, 0.0008]
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    for tau in tau_steps:
        if time.time() > deadline:
            break
        
        def obj(x):
            return _lse_objective(x, tau, target)
        
        def grad(x):
            return _lse_gradient(x, tau)
        
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.x is not None:
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_binary(n, alpha):
    """Generate quasi-periodic binary pattern."""
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1.0)[0]
        for i in range(n_ones - target):
            h[idx[i]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0.0)[0]
        for i in range(target - n_ones):
            h[idx[i]] = 1.0
    return h

def _binary_sa(h_init, deadline, rng, n_iters=50000):
    """Simulated annealing on binary space."""
    n = len(h_init)
    h = h_init.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Ensure exact balance
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1.0)[0]
        rng.shuffle(ones_idx)
        for i in range(n_ones - target):
            h[ones_idx[i]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0.0)[0]
        rng.shuffle(zeros_idx)
        for i in range(target - n_ones):
            h[zeros_idx[i]] = 1.0
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.02
    T_min = 0.0005
    cool_rate = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        T *= cool_rate
    
    return best_h, best_c5

def _refine_continuous(h0, deadline, target):
    """Combined continuous refinement: SLSQP + projected subgradient."""
    n = len(h0)
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Phase A: SLSQP with decreasing tau
    dl_a = min(deadline, time.time() + (deadline - time.time()) * 0.5)
    h_a, c5_a = _lse_optimize(h, dl_a, target, tau_init=0.02)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()
    
    # Phase B: Projected subgradient with softmax weighting
    h = best_h.copy()
    cur_c5 = best_c5
    gamma = 10.0
    step = 0.1
    for it in range(2000):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(gamma * (scores - max_s))
        sw = np.sum(w)
        if sw > 1e-300:
            w /= sw
        else:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        # Gradient
        grad = _lse_gradient(h, 1.0/gamma)  # reuse gradient function
        # Projected step
        new_h = _project(h - step * grad / (np.linalg.norm(grad) + 1e-15), target)
        new_c5 = _exact_c5(new_h)
        if new_c5 < cur_c5 - 1e-14:
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.05, 0.5)
            gamma = min(gamma * 1.01, 200.0)
        else:
            step *= 0.8
            gamma = min(gamma * 1.1, 500.0)
            if step < 1e-6:
                break
    
    return best_h, best_c5

def _full_pipeline(h_start, n, deadline, rng):
    """Run full optimization pipeline on a single resolution."""
    target = n / 2.0
    h = _project(h_start, target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Step 1: Binary local search (exhaustive)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != n // 2:
        # Fix balance
        idx_ones = np.where(h_bin == 1)[0]
        idx_zeros = np.where(h_bin == 0)[0]
        if n_ones > n // 2:
            for i in range(n_ones - n // 2):
                h_bin[idx_ones[i]] = 0
        else:
            for i in range(n // 2 - n_ones):
                h_bin[idx_zeros[i]] = 1
    
    dl_bin = min(deadline, time.time() + (deadline - time.time()) * 0.3)
    h_bl, c5_bl = _binary_local_search(h_bin, dl_bin, max_sweeps=100)
    if c5_bl < best_c5:
        best_c5 = c5_bl
        best_h = h_bl.copy()
    
    # Step 2: Continuous refinement
    dl_cont = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    h_cr, c5_cr = _refine_continuous(best_h, dl_cont, target)
    if c5_cr < best_c5:
        best_c5 = c5_cr
        best_h = h_cr.copy()
    
    # Step 3: Binary search again on refined
    h_bin2 = (best_h > 0.5).astype(float)
    n_ones2 = int(np.sum(h_bin2))
    if n_ones2 != n // 2:
        idx_ones = np.where(h_bin2 == 1)[0]
        idx_zeros = np.where(h_bin2 == 0)[0]
        if n_ones2 > n // 2:
            for i in range(n_ones2 - n // 2):
                h_bin2[idx_ones[i]] = 0
        else:
            for i in range(n // 2 - n_ones2):
                h_bin2[idx_zeros[i]] = 1
    dl_bin2 = min(deadline, time.time() + (deadline - time.time()) * 0.15)
    h_bl2, c5_bl2 = _binary_local_search(h_bin2, dl_bin2, max_sweeps=50)
    if c5_bl2 < best_c5:
        best_c5 = c5_bl2
        best_h = h_bl2.copy()
    
    # Step 4: Final SLSQP polish
    dl_final = deadline
    h_fn, c5_fn = _lse_optimize(best_h, dl_final, target, tau_init=0.005)
    if c5_fn < best_c5:
        best_c5 = c5_fn
        best_h = h_fn.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Start from initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    # Phase 1: Main pipeline at original resolution
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = _full_pipeline(h_start, n0, dl1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Try different resolutions
    resolutions = [n0, 96, 112, 144, 160, 192]
    for n_try in resolutions:
        if time.time() > t0 + budget_s * 0.70:
            break
        if n_try == n0:
            continue
        h_res = _resample(best_h, n_try)
        dl_res = min(t0 + budget_s * 0.70, time.time() + budget_s * 0.08)
        h_try, c5_try = _full_pipeline(h_res, n_try, dl_res, rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
    
    # Phase 3: Quasi-periodic starting points
    if time.time() < t0 + budget_s * 0.80:
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.3, 0.7]
        for n_qp in [n0, 128, 160]:
            if time.time() > t0 + budget_s * 0.80:
                break
            for a in alphas[:4]:
                if time.time() > t0 + budget_s * 0.80:
                    break
                h_qp = _quasi_binary(n_qp, a)
                dl_qp = min(t0 + budget_s * 0.80, time.time() + budget_s * 0.03)
                h_qo, c5_qo = _full_pipeline(h_qp, n_qp, dl_qp, rng)
                if c5_qo < best_c5:
                    best_c5 = c5_qo
                    best_h = h_qo.copy()
                    best_n = n_qp
    
    # Phase 4: Binary SA at best resolution
    if time.time() < t0 + budget_s * 0.90:
        dl_sa = t0 + budget_s * 0.90
        h_bin_init = (best_h > 0.5).astype(float)
        h_sa, c5_sa = _binary_sa(h_bin_init, dl_sa, rng, n_iters=100000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Polish after SA
        if time.time() < t0 + budget_s * 0.95:
            dl_pol = t0 + budget_s * 0.95
            h_pp, c5_pp = _full_pipeline(best_h, len(best_h), dl_pol, rng)
            if c5_pp < best_c5:
                best_c5 = c5_pp
                best_h = h_pp.copy()
    
    # Phase 5: Final perturbation + polish
    if time.time() < t0 + budget_s * 0.97:
        for attempt in range(30):
            if time.time() > t0 + budget_s * 0.97:
                break
            h_pert = best_h + rng.normal(0, 0.03, len(best_h))
            h_pert = _project(h_pert, len(best_h) / 2.0)
            dl_p = min(t0 + budget_s * 0.985, time.time() + budget_s * 0.01)
            h_po, c5_po = _lse_optimize(h_pert, dl_p, len(best_h) / 2.0, tau_init=0.01)
            if c5_po < best_c5:
                best_c5 = c5_po
                best_h = h_po.copy()
    
    # Final: ensure feasibility and get exact value
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
