def _exact_c5(h):

def _scores(h):

def _project(v, target):
    """Project v onto {x: 0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    # Water-filling
    def g(theta):
        return np.sum(np.clip(v - theta, 0.0, 1.0))
    
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if g(mid) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

    n = len(h)

    elif best_idx == 0:

        else:

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
