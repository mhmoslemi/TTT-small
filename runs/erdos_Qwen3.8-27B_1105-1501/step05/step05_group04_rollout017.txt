def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -2.0, 1 + 2.0)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        if s > target:

        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

        else:

def lag_grad(h, s):
    """Gradient of F_s(h) * dx w.r.t. h, for lag index s = r - (n-1)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            # F_s = dx * sum_{i=0}^{m-1} h[i+s]*(1-h[i])
            # d/dh[j]: if j in [0, m-1]: -dx*h[j+s]
            #          if j in [s, n-1]: dx*(1-h[j-s])
            g[:m] += -dx * h[s:]
            g[s:] += dx * (1.0 - h[:m])
    else:
        a = -s
        m = n - a
        if m > 0:
            # F_{-a} = dx * sum_{i=0}^{m-1} h[i]*(1-h[i+a])
            # d/dh[j]: if j in [0, m-1]: dx*(1-h[j+a])
            #          if j in [a, n-1]: -dx*h[j-a]
            g[:m] += dx * (1.0 - h[a:])
            g[a:] += -dx * h[:m]
    return g

def lag_jacobian(h):

    for r in range(2 * n - 1):
        s = r - (n - 1)
        J[r] = lag_grad(h, s)
    return J

def line_search_exact(h, d, dx, trust_radius):

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find feasible interval for t
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15

    t_hi = min(t_hi, trust_radius)

    c2 = a + gr * (b - a)

    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx

def lp_direction(h, scores, G, radius, active_tol=0.001):
    """LP to find descent direction. Returns direction p or None."""
    n = len(h)
    n_lags = len(scores)
    max_score = float(np.max(scores))
    
    # Active set: lags within active_tol of max
    active = np.where(scores >= max_score - active_tol)[0]
    
    # Build LP: min uuid
    # Variables: p[0..n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: G[p; -1] <= -scores for active lags
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on p
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] * 0.95)
        hi = min(radius, (1.0 - h[j]) * 0.95)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < max_score - 1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    
    if t_hi - t_lo < 1e-15:
        return 0.0, eval_c5(h)
    
    # Grid + golden section
    n_grid = 50
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:], axis=1) * dx
    best_i = int(np.argmin(vals))
    best_t = grid[best_i]
    best_val = vals[best_i]
    
    # Refine with golden section
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(40):
        if b - a < 1e-15:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt_t = (a+b)/2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    
    return best_t, best_val

def lp_solve(h, scores, G, radius, active_idx):
    n = len(h)
    if len(active_idx) == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G[active_idx]
    A_ub[:, n] = -1.0
    b_ub = -scores[active_idx]
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -0.9 * h[j])
        hi = min(radius, 0.9 * (1.0 - h[j]))
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            max_s = float(np.max(scores))
            if res.x[-1] < max_s - 1e-12:
                return res.x[:n]
    except Exception:
        pass
    return None
