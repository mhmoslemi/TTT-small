def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

def _log1pexp(x):
    out = np.zeros_like(x)
    pos = x > 0
    out[pos] = x[pos] + np.log1p(np.exp(-x[pos]))
    out[~pos] = np.log1p(np.exp(x[~pos]))
    return out

def _stable_sigmoid(x):
    out = np.zeros_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _run_lse_solve(a0, n, deadline, beta_start, beta_end, max_iter=300):

    a = a0.copy()

    for bi in range(n_beta):

            o, _ = _lse_obj_grad(x, beta, n)
            return o

            return g
        try:

                best_a = a_new.copy()
            a = a_new
        except Exception:
            pass

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=-scores, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)) - 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a_pt, b_pt = grid[best_i - 1], grid[best_i + 1]
    else:
        a_pt, b_pt = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b_pt - gr * (b_pt - a_pt)
    c2 = a_pt + gr * (b_pt - a_pt)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b_pt - a_pt < 1e-14:
            break
        if f1 < f2:
            b_pt = c2; c2 = c1; f2 = f1
            c1 = b_pt - gr * (b_pt - a_pt)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a_pt = c1; c1 = c2; f1 = f2
            c2 = a_pt + gr * (b_pt - a_pt)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a_pt + b_pt) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _lp_refine(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            n_active = 5
            sorted_idx = np.argsort(scores)[::-1]
            active_set = set(sorted_idx[:n_active].tolist())
            c = np.zeros(n + 1)
            c[-1] = 1.0
            n_lags = 2 * n - 1
            A_ub = np.zeros((n_lags, n + 1))
            A_ub[:, :n] = G
            A_ub[:, n] = -1.0
            dx_val = 2.0 / n
            b_ub = np.full(n_lags, -scores + radius * dx_val)
            for k in active_set:
                b_ub[k] = -scores[k]
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
            bounds.append((None, None))
            try:
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                              bounds=bounds, method='highs')
                if res.success and res.x[-1] < float(np.max(scores)) - 1e-10:
                    p = res.x[:n]
                else:
                    p = None
            except Exception:
                p = None
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.7
                if radius < 1e-6:
                    break
                if stall % 30 == 0:
                    radius = max(radius, 0.03)
                continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project_box_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _coord_descent(h, deadline, n_iter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(99999)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = _project_box_sum(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def _perturb_lse(h, deadline, rng, base_scale=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = base_scale
    for _ in range(200):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project_box_sum(h_pert, n / 2.0)
        a_pert = _a_from_h(h_pert)
        sub_dl = min(deadline, time.time() + 8.0)
        a_new, c5_new = _run_lse_solve(a_pert, n, sub_dl, 80.0, 800.0, max_iter=80)
        h_new = _h_from_a(a_new, n)
        c5_exact = _exact_c5(h_new)
        if c5_exact < best_c5 - 1e-11:
            best_c5 = c5_exact
            best_h = h_new.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = _project_box_sum(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: LSE optimization with complementary parameterization (35%)
    d1 = t0 + budget_s * 0.35
    tried_n = set()
    res_list = [n0, n0-2, n0+2, n0-4, n0+4, 90, 96, 100, 84, 104, 88, 92, 98, 102, 86, 106]
    for n_try in res_list:
        if time.time() > d1:
            break
        if n_try < 30 or n_try > 200 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        a_r = _a_from_h(h_r)
        sub_dl = min(d1, time.time() + 20.0)
        a_new, c5_new = _run_lse_solve(a_r, n_try, sub_dl, 50.0, 500.0, max_iter=200)
        h_new = _h_from_a(a_new, n_try)
        c5_e = _exact_c5(h_new)
        if c5_e < best_c5 - 1e-12:
            best_c5 = c5_e
            best_h = h_new.copy()
            res_list.append(n_try - 1)
            res_list.append(n_try + 1)

    # Phase 2: Soft margin targeting (25%)
    d2 = t0 + budget_s * 0.60
    n_cur = len(best_h)
    a_cur = _a_from_h(best_h)
    margins = [2e-5, 5e-5, 1e-4, 2e-4, 5e-4, 1e-3]
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for mi in range(len(margins)):
        if time.time() > d2:
            break
        B = best_c5 - margins[mi]
        if B <= 0:
            continue
        for di in range(min(3, len(deltas))):
            if time.time() > d2:
                break
            delta = deltas[di]
            sub_dl = min(d2, time.time() + 15.0)
            a_new, c5_new = _run_margin_solve(a_cur, n_cur, B, delta, sub_dl, max_iter=100)
            h_new = _h_from_a(a_new, n_cur)
            c5_e = _exact_c5(h_new)
            if c5_e < best_c5 - 1e-12:
                best_c5 = c5_e
                best_h = h_new.copy()
                a_cur = _a_from_h(h_new)

    # Phase 3: Perturbation + LSE (20%)
    d3 = t0 + budget_s * 0.80
    best_h, best_c5 = _perturb_lse(best_h, d3, rng, base_scale=0.04)

    # Phase 4: LP refinement (12%)
    d4 = t0 + budget_s * 0.92
    n_cur = len(best_h)
    h_lp, c5_lp = _lp_refine(best_h, d4, radius_init=0.08)
    if c5_lp < best_c5 - 1e-12:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 5: Multi-round LSE polish (4%)
    d5 = t0 + budget_s * 0.96
    n_cur = len(best_h)
    a_final = _a_from_h(best_h)
    a_final, c5_fin = _run_lse_solve(a_final, n_cur, d5, 200.0, 2000.0, max_iter=150)
    h_fin = _h_from_a(a_final, n_cur)
    c5_fin_e = _exact_c5(h_fin)
    if c5_fin_e < best_c5 - 1e-12:
        best_c5 = c5_fin_e
        best_h = h_fin.copy()

    # Final coordinate descent if time remains
    rem = deadline - time.time()
    if rem > 15:
        best_h, best_c5 = _coord_descent(best_h, t0 + budget_s - 3, n_iter=200)

    # Final validation
    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    n_pts = len(best_h)
    return best_h, final_c5, n_pts
