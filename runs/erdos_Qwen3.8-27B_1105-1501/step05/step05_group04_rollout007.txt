    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def lag_jacobian_active(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    K = len(active_indices)
    G = np.zeros((K, n))
    for idx, i in enumerate(active_indices):
        s = i - (n - 1)
        if s >= 0:
            m = n - s
            G[idx, s:] += dx * (1.0 - h[:m])
            G[idx, :m] -= dx * h[s:]
        elif s < 0:
            a = -s
            m = n - a
            G[idx, :m] += dx * (1.0 - h[a:])
            G[idx, a:] -= dx * h[:m]
    return G


def full_lag_jacobian(h):
    n = len(h)
    active = np.arange(2 * n - 1)
    return lag_jacobian_active(h, active)


def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    n_grid = 80
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val


def _lp_direction(h, scores, G_active, active_idx, radius):
    n = len(h)
    n_active = len(active_idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = -scores[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    tol_frac = 0.02
    while time.time() < deadline:
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_score = float(np.max(scores))
        tol = tol_frac * max_score
        active_mask = scores >= (max_score - tol)
        n_active = int(np.sum(active_mask))
        if n_active < 3:
            idx_sorted = np.argsort(scores)[::-1]
            active_mask = np.zeros(len(scores), dtype=bool)
            active_mask[idx_sorted[:3]] = True
        active_idx = np.where(active_mask)[0]
        n_cap = min(len(active_idx), 25)
        if len(active_idx) > n_cap:
            top_k = np.argsort(scores[active_idx])[::-1][:n_cap]
            active_idx = active_idx[top_k]
        G_active = lag_jacobian_active(h, active_idx)
        p = _lp_direction(h, scores, G_active, active_idx, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 2
            radius *= 0.7
            tol_frac *= 0.8
            if radius < 1e-7:
                break
            if stall % 30 == 0:
                radius = max(radius, 0.03)
                tol_frac = 0.05
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.3)
            tol_frac = max(tol_frac * 1.05, 0.005)
        else:
            stall += 1
            radius *= 0.78
    return best_h, best_c5


def _slsqp_epigraph(h, deadline, n_constr=25):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    cur_max = float(np.max(scores))
    
    tol = 0.03
    active_mask = scores >= (cur_max - tol)
    if int(np.sum(active_mask)) < 5:
        idx_sorted = np.argsort(scores)[::-1]
        active_mask = np.zeros(len(scores), dtype=bool)
        active_mask[idx_sorted[:5]] = True
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > n_constr:
        top_k = np.argsort(scores[active_idx])[::-1][:n_constr]
        active_idx = active_idx[top_k]
    
    def obj_func(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def con_eq(x):
        return np.sum(x[:n]) - target
    
    def con_eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    def con_ineq(x):
        hx = x[:n]
        t = x[-1]
        sc = np.correlate(hx, 1.0 - hx, mode='full') * dx
        return t - sc[active_idx]
    
    def con_ineq_jac(x):
        hx = x[:n]
        G = lag_jacobian_active(hx, active_idx)
        J = np.zeros((len(active_idx), n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    t0 = cur_max + 0.02
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [
        {'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac},
        {'type': 'ineq', 'fun': con_ineq, 'jac': con_ineq_jac}
    ]
    
    try:
        res = minimize(obj_func, x0, jac=obj_grad, constraints=constraints,
                       bounds=bounds, method='SLSQP',
                       options={'maxiter': 300, 'ftol': 1e-10})
        if res.x is not None:
            h_new = np.clip(np.nan_to_num(res.x[:n], nan=0.5, posinf=1.0, neginf=0.0), 0, 1)
            h_new = project_box_sum(h_new, target)
            c5_new = eval_c5(h_new)
            if c5_new < cur_max - 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h, cur_max


def _logsumexp_descent(h, deadline, max_iter=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    cur_c5 = best_c5
    
    max_score_0 = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    betas = [max_score_0 * 0.5, max_score_0 * 0.25, max_score_0 * 0.12,
             max_score_0 * 0.06, max_score_0 * 0.03, max_score_0 * 0.015]
    
    for beta in betas:
        if time.time() > deadline:
            break
        lr = 0.5 / (beta + 1e-10)
        lr = min(lr, 2.0)
        for it in range(max_iter // len(betas)):
            if time.time() > deadline:
                break
            scores = np.correlate(h, 1.0 - h, mode='full')
            max_s = float(np.max(scores))
            exps = np.exp((scores - max_s) / beta)
            probs = exps / float(np.sum(exps))
            G_all = full_lag_jacobian(h)
            grad = probs @ G_all
            # Projected gradient step with backtracking
            step = lr
            improved = False
            for _bt in range(20):
                h_try = h - step * grad
                h_try = project_box_sum(h_try, target)
                s_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                max_try = float(np.max(s_try))
                phi_try = max_try + beta * float(np.log(np.sum(np.exp((s_try - max_try) / beta))))
                s_cur = np.correlate(h, 1.0 - h, mode='full')
                max_cur = float(np.max(s_cur))
                phi_cur = max_cur + beta * float(np.log(np.sum(np.exp((s_cur - max_cur) / beta))))
                if phi_try < phi_cur - 1e-12:
                    h = h_try
                    improved = True
                    break
                step *= 0.5
            if not improved:
                break
            c5_check = eval_c5(h)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h.copy()
    return best_h, best_c5


def _coord_refine(h, deadline, n_iter=400):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(999)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        max_step = min(h[i], 1.0 - h[j], 0.5)
        alpha, c5_try = _line_search(h, d, dx, max_step)
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)


def _perturb_and_opt(h, deadline, rng, base_scale=0.03):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    stagnation = 0
    for _ in range(300):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.06)
        sub_dl2 = min(deadline, time.time() + 4.0)
        h_t2, c5_t2 = _slsqp_epigraph(h_t, sub_dl2, n_constr=20)
        c5_t = min(c5_t, c5_t2)
        h_t = h_t if c5_t2 >= c5_t else h_t2
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
            stagnation = 0
        else:
            scale = min(0.15, scale * 1.08)
            stagnation += 1
            if stagnation > 30:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    # Phase 1: LP from initial point (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: SLSQP epigraph (15%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _slsqp_epigraph(best_h, d2, n_constr=30)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Log-sum-exp descent (15%)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _logsumexp_descent(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: LP + SLSQP combined (15%)
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = _lp_loop(best_h, min(d4, time.time() + budget_s * 0.07), radius_init=0.10)
    h4b, c5_4b = _slsqp_epigraph(h4, d4, n_constr=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()
    
    # Phase 5: Perturbation + optimization (20%)
    d5 = t0 + budget_s * 0.80
    best_h, best_c5 = _perturb_and_opt(best_h, d5, rng, base_scale=0.03)
    
    # Phase 6: Resolution cycling (10%)
    d6 = t0 + budget_s * 0.90
    tried = set()
    for dn in range(-15, 16, 3):
        if time.time() > d6:
            break
        n_try = len(best_h) + dn
        if n_try < 40 or n_try > 130 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + 4.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10)
        sub_dl2 = min(d6, time.time() + 3.0)
        h_r2, c5_r2 = _slsqp_epigraph(h_r, sub_dl2, n_constr=20)
        if c5_r2 < best_c5 - 1e-11:
            best_c5 = c5_r2
            best_h = h_r2.copy()
        elif c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 7: Final polish (10%)
    d7 = t0 + budget_s * 0.96
    h7, c5_7 = _lp_loop(best_h, d7, radius_init=0.05)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    
    # Final coordinate refinement
    d8 = t0 + budget_s * 0.99
    best_h, best_c5 = _coord_refine(best_h, d8, n_iter=200)
    
    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
