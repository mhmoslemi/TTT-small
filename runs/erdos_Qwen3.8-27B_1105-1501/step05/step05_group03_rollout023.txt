        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _coord_polish(h_start, deadline, n_cycles=2):
    n = len(h_start)
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(30):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h_start, deadline, n_passes=3, rng=None):
    n = len(h_start)
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    step_size = max(1, n // 16)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step_size):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(0, n, step_size):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(10):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline_global = t0 + budget_s * 0.98

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.zeros(128)
        h0[:64] = 1.0
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _c5(h_start)
    best_n = n0

    # Phase 1: Smoothed gradient warm-up (fast, gets into good basin)
    if time.time() < t0 + budget_s * 0.08:
        dl1 = t0 + budget_s * 0.08
        h1, c5_1 = _smoothed_grad(h_start, dl1, beta_start=15, beta_end=200,
                                   n_steps=150, step_init=0.04)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Main epigraph optimization (core contribution)
    if time.time() < t0 + budget_s * 0.45:
        dl2 = t0 + budget_s * 0.45
        h2, c5_2 = _epigraph_loop(best_h, dl2, max_active=80, init_K=20, max_rounds=80)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Smoothed gradient + epigraph refinement
    if time.time() < t0 + budget_s * 0.55:
        dl3 = t0 + budget_s * 0.55
        h3, c5_3 = _smoothed_grad(best_h, dl3, beta_start=30, beta_end=500,
                                   n_steps=100, step_init=0.03)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        if time.time() < t0 + budget_s * 0.60:
            dl3b = t0 + budget_s * 0.60
            h3b, c5_3b = _epigraph_loop(best_h, dl3b, max_active=64, init_K=16, max_rounds=40)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.75:
        dl4 = t0 + budget_s * 0.75
        resolutions = []
        for cand in [64, 80, 96, 112, 128, 160]:
            if cand != n0 and cand >= 32 and cand <= 224:
                resolutions.append(cand)
        for n_try in resolutions:
            if time.time() > dl4:
                break
            if best_c5 < 0.3806:
                break
            h_res = _resample(best_h, n_try)
            c5_res = _c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_try
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_ep, c5_ep = _epigraph_loop(h_res, sub_dl, max_active=48, init_K=16, max_rounds=30)
            if c5_ep < best_c5:
                best_c5 = c5_ep
                best_h = h_ep.copy()
                best_n = n_try

    # Phase 5: Binary SA at smaller resolutions
    if time.time() < t0 + budget_s * 0.82:
        dl5 = t0 + budget_s * 0.82
        for n_bin in [64, 96]:
            if time.time() > dl5:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=10000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
                h_br, c5_br = _epigraph_loop(h_b, sub_dl2, max_active=32, init_K=12, max_rounds=20)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()

    # Phase 6: Perturbation restarts with epigraph
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.90
        scale = 0.02
        for attempt in range(200):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.008)
            h_try, c5_try = _epigraph_loop(h_pert, sub_dl, max_active=32, init_K=12, max_rounds=15)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.06, scale * 1.08)

    # Phase 7: Final epigraph with tighter active set
    if time.time() < t0 + budget_s * 0.95:
        dl7 = t0 + budget_s * 0.95
        h7, c5_7 = _epigraph_loop(best_h, dl7, max_active=64, init_K=10, max_rounds=40)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Coordinate polish
    if time.time() < t0 + budget_s * 0.975:
        dl8 = t0 + budget_s * 0.975
        hc, c5c = _coord_polish(best_h, dl8, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Final: re-project and evaluate
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
