from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj_grad(h, tau):
    """Compute LSE objective and gradient for the C5 minimization."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    M = float(np.max(scores))
    z = np.exp((scores - M) / tau)
    sz = float(np.sum(z))
    val = M + tau * np.log(sz)
    w = z / sz
    
    g = np.zeros(n)
    center = n - 1
    for idx in range(len(scores)):
        wi = w[idx] * dx
        if wi < 1e-15:
            continue
        q = idx - center
        if q >= 0:
            m = n - q
            g[q:] += wi * (1.0 - h[:m])
            g[:m] -= wi * h[q:]
        else:
            gshift = -q
            m = n - gshift
            g[:m] += wi * (1.0 - h[gshift:])
            g[gshift:] -= wi * h[:m]
    return val, g

def _slsqp_optimize(h0, tau, deadline, maxiter=200):
    """Run SLSQP on the LSE surrogate."""
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        x = np.clip(x, 0, 1)
        v, _ = _lse_obj_grad(x, tau)
        return v
    
    def grad(x):
        x = np.clip(x, 0, 1)
        _, g = _lse_obj_grad(x, tau)
        return g
    
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h0.copy(), _exact_c5(h0)

def _line_search_exact(h, d, dx, radius):
    """Exact line search: minimize max(dx * (S0 + t*T - t^2*Q)) over feasible t."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Determine feasible interval
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Dense grid search
    n_grid = 81
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([float(np.max(S0 + t * T - t*t * Q)) * dx for t in grid])
    best_idx = int(np.argmin(vals))
    best_t = grid[best_idx]
    best_val = vals[best_idx]
    
    # Refine around best with golden section
    left = grid[max(0, best_idx-1)]
    right = grid[min(n_grid-1, best_idx+1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = right - gr * (right - left)
    c2 = left + gr * (right - left)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    for _ in range(50):
        if right - left < 1e-12:
            break
        if f1 < f2:
            right = c2; c2 = c1; f2 = f1
            c1 = right - gr * (right - left)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        else:
            left = c1; c1 = c2; f1 = f2
            c2 = left + gr * (right - left)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    
    opt_t = (left + right) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * Q)) * dx
    
    # Verify with exact evaluation
    h_test = h + opt_t * d
    h_test = _project(h_test, n / 2.0)
    exact_val = _exact_c5(h_test)
    
    # Also check grid best
    h_grid = h + best_t * d
    h_grid = _project(h_grid, n / 2.0)
    exact_grid = _exact_c5(h_grid)
    
    if exact_grid < exact_val:
        opt_t = best_t
        opt_val = exact_grid
    
    return opt_t, opt_val

def _block_direction(h, n, rng, block_size):
    """Generate a block swap direction."""
    d = np.zeros(n)
    # Choose two non-overlapping contiguous blocks
    for _ in range(50):
        a_start = int(rng.integers(0, n - block_size))
        b_start = int(rng.integers(0, n - block_size))
        if a_start + block_size <= b_start or b_start + block_size <= a_start:
            break
    else:
        a_start = 0
        b_start = n - block_size
    
    a = slice(a_start, a_start + block_size)
    b = slice(b_start, b_start + block_size)
    
    # Transfer mass from block with higher mean to lower mean
    if np.mean(h[a]) > np.mean(h[b]):
        d[a] = -1.0 / block_size
        d[b] = 1.0 / block_size
    else:
        d[a] = 1.0 / block_size
        d[b] = -1.0 / block_size
    
    return d

def _lowdim_optimize(h_init, n, deadline, n_ctrl=14, method='powell'):
    """Optimize a low-dimensional parameterization."""
    target = n / 2.0
    x_ctrl = np.linspace(0, 2, n_ctrl)
    x_grid = (np.arange(n) + 0.5) / n * 2.0
    
    # Initial control values: interpolate h_init
    u0 = np.interp(x_ctrl, x_grid, h_init)
    
    def map_u(u):
        h_raw = np.interp(x_grid, x_ctrl, u)
        return _project(h_raw, target)
    
    def objective(u):
        h = map_u(u)
        return _exact_c5(h)
    
    bounds = [(-2.0, 3.0)] * n_ctrl
    
    try:
        res = minimize(objective, u0, method=method, bounds=bounds,
                       options={'maxiter': 500, 'maxfev': 2000})
        if res.x is not None:
            h_new = map_u(res.x)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    
    # Fallback: Nelder-Mead
    try:
        res = minimize(objective, u0, method='Nelder-Mead',
                       options={'maxiter': 1000, 'maxfev': 2000, 'xatol': 1e-6, 'fatol': 1e-8})
        if res.x is not None:
            h_new = map_u(res.x)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    
    return h_init.copy(), _exact_c5(h_init)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _block_search(h, deadline, n_blocks=200, block_sizes=None):
    """Iterative block-based local search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    
    if block_sizes is None:
        block_sizes = [2, 3, 4, 5, 8]
    
    radius = 0.3
    stall = 0
    
    while time.time() < deadline:
        improved = False
        for _ in range(n_blocks):
            if time.time() > deadline:
                break
            bs = int(rng.choice(block_sizes))
            if bs >= n // 2:
                bs = n // 4
            d = _block_direction(h, n, rng, bs)
            alpha, c5_try = _line_search_exact(h, d, dx, radius)
            if c5_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
            else:
                stall += 1
        if not improved:
            radius *= 0.7
            if radius < 0.01:
                break
        else:
            radius = min(radius * 1.05, 0.5)
    
    return best_h, best_c5

def _multi_tau_slsqp(h, deadline):
    """Try SLSQP with multiple tau values."""
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    taus = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005]
    
    for tau in taus:

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5

def _project(v, target):

        return np.zeros(n)

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

def _lse_obj_grad(h, tau):

    sz = float(np.sum(z))
    if sz < 1e-30:
        sz = 1e-30
    val = M + tau * np.log(sz)
    w = z / sz
    g = np.zeros(n)
    center = n - 1

        if wi < 1e-14:

        q = idx - center

        else:

            m = n - gs
            g[:m] += wi * (1.0 - h[gs:])

def _slsqp_opt(h0, tau, deadline, maxiter=100):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        x = np.clip(x, 0, 1)
        v, _ = _lse_obj_grad(x, tau)
        return v
    def grad(x):
        x = np.clip(x, 0, 1)
        _, g = _lse_obj_grad(x, tau)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h0.copy(), _exact_c5(h0)

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        if res2.x is not None:
            val2 = objective(res2.x)
            if val2 < best_val:
                best_val = val2
                best_u = res2.x.copy()
    except Exception:
        pass
    h_best = map_u(best_u)
    return h_best, _exact_c5(h_best)

def _active_lag_search(h, deadline, n_iter=3000):
    """Focus on reducing the active (maximizing) lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    radius = 0.2
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = float(np.max(corr))
        active = np.where(corr > max_val - 0.005)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        idx = int(rng.choice(active))
        q = idx - (n - 1)
        d = np.zeros(n)
        if q >= 0:
            m = n - q
            for i in range(min(m, 5)):
                j = int(rng.integers(0, m))
                if h[j + q] < h[j]:
                    d[j + q] += 0.5
                    d[j] -= 0.5
        else:
            gs = -q
            m = n - gs
            for i in range(min(m, 5)):
                j = int(rng.integers(0, m))
                if h[j] < h[j + gs]:
                    d[j] += 0.5
                    d[j + gs] -= 0.5
        if np.max(np.abs(d)) < 1e-10:
            stall += 1
            continue
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-15:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                radius *= 0.8
                stall = 0
                if radius < 0.005:
                    break
    return best_h, best_c5

def _multi_tau_grad(h, deadline, taus=None):
    if taus is None:
        taus = [0.04, 0.025, 0.015, 0.01, 0.007, 0.005]
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n = len(h)
    for tau in taus:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + max(30, budget_frac(deadline, t0_ref) * 20))
        lr = max(0.005, 0.03 * tau)
        h_new, c5_new = _proj_grad(best_h, tau, sub_dl, lr_init=lr)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def budget_frac(deadline, t0):
    elapsed = time.time() - t0
    remaining = max(0, deadline - time.time())
    return remaining / max(1.0, deadline - t0)
