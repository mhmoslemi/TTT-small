def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def line_search_exact(h, d, trust_radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def lp_direction(h, scores, G, radius, n_active=5, relaxed=False):

    c = np.zeros(n + 1)

def smooth_obj_and_grad(x, beta, n):
    h = np.clip(x, 0.0, 1.0)
    F = lag_values(h)
    m = np.max(F)
    shifted = (F - m) / beta
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    Z = np.exp(shifted)
    log_sum = np.log(np.sum(Z))
    val = m + beta * log_sum
    P = Z / np.sum(Z)
    J = lag_jacobian(h)
    grad = J.T @ P
    return val, grad

        return g

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def coord_refine(h, deadline, n_iter=500, rng=None):
    n = len(h)
    target = n / 2.0
    if rng is None:
        rng = np.random.default_rng(999)
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        max_alpha = min(h[i], 1.0 - h[j])
        alpha, c5_try = line_search_exact(h, d, max_alpha * 0.99)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog, minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                J[r, d:] += dx * (1.0 - h[:m])
                J[r, :m] += -dx * h[d:]
        else:
            a = -d
            m = n - a
            if m > 0:
                J[r, :m] += dx * (1.0 - h[a:])
                J[r, a:] += -dx * h[:m]
    return J

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def line_search_exact(h, d, trust_radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def lp_direction(h, scores, G, radius, n_active=6, relaxed=False):
    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    if relaxed:
        sorted_idx = np.argsort(scores)[::-1]
        active = set(sorted_idx[:n_active].tolist())
        dx_val = 2.0 / n
        b_ub = -scores + radius * dx_val * 0.5
        for k in active:
            b_ub[k] = -scores[k]
    else:
        b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_loop(h, deadline, radius_init=0.12):

            if p is None or np.max(np.abs(p)) < 1e-13:

                p = lp_direction(h, scores, G, radius, n_active=max(10, len(scores) // 3), relaxed=True)

    return best_h, best_c5

def _smooth_obj_and_grad(x, beta):
    h = np.clip(x, 0.0, 1.0)
    F = lag_values(h)
    m = np.max(F)
    shifted = (F - m) / beta
    shifted = np.clip(shifted, -60.0, 60.0)
    Z = np.exp(shifted)
    sum_Z = np.sum(Z)
    val = m + beta * np.log(sum_Z)
    P = Z / sum_Z
    J = lag_jacobian(h)
    grad = J.T @ P
    return val, grad

def smooth_solve(h, beta, maxiter=40):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = _smooth_obj_and_grad(x, beta)
        return v

    def grad_fn(x):
        _, g = _smooth_obj_and_grad(x, beta)
        return g

    con_fun_eq = lambda x: np.sum(x) - target
    con_jac_eq = lambda x: np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun_eq, 'jac': con_jac_eq}]
    bounds = [(0, 1)] * n
    try:
        res = minimize(obj, h, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = project_box_sum(res.x, target)
        return h_new
    except Exception:
        return h.copy()

def epigraph_solve(h, maxiter=50):
    n = len(h)
    target = n / 2.0
    c5_now = eval_c5(h)
    t_start = c5_now + 0.015
    x0 = np.append(h, t_start)
    n_lags = 2 * n - 1

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def con_fun(x):
        h_c = np.clip(x[:n], 0.0, 1.0)
        t_c = max(x[n], 0.0)
        F = lag_values(h_c)
        return t_c - F

    def con_jac(x):
        h_c = np.clip(x[:n], 0.0, 1.0)
        J = lag_jacobian(h_c)
        Jc = np.zeros((n_lags, n + 1))
        Jc[:, :n] = -J
        Jc[:, n] = 1.0
        return Jc

    def eq_fun(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    cons = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    bounds = [(0, 1)] * n + [(max(0.0, c5_now * 0.3), 1.0)]
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-9})
        h_new = project_box_sum(res.x[:n], target)
        return h_new
    except Exception:
        return h.copy()

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)
