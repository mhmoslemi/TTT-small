        alpha, new_v = _line_search(best_h, lr * mom, dx, lr * 3.0)
        if new_v < best_c5 - 1e-13:
            best_h = _project(best_h + alpha * lr * mom, target)
            best_c5 = _exact_c5(best_h)
            stall = 0
            lr = min(lr * 1.05, 0.3)
        else:
            stall += 1
            lr *= 0.78
    return best_h, best_c5

def _pair_refine(h, deadline, rng, max_pairs=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = np.where((best_h > 1e-8) & (best_h < 1.0 - 1e-8))[0]
    improved = True
    pass_cnt = 0
    while improved and time.time() < deadline:
        improved = False
        pass_cnt += 1
        if max_pairs and len(interior) > max_pairs:
            idx = rng.choice(interior, size=max_pairs, replace=False)
        else:
            idx = interior
        for i_idx in range(len(idx)):
            if time.time() > deadline:
                break
            i = int(idx[i_idx])
            for j_idx in range(i_idx + 1, len(idx)):
                if time.time() > deadline:
                    break
                j = int(idx[j_idx])
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(best_h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-13:
                    new_h = _project(best_h + alpha * d, target)
                    c5_check = _exact_c5(new_h)
                    if c5_check < best_c5:
                        best_c5 = c5_check
                        best_h = new_h
                        improved = True
                    interior = np.where((best_h > 1e-8) & (best_h < 1.0 - 1e-8))[0]
    return best_h, best_c5

def _kick_and_reopt(h, deadline, rng, n_kicks=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.08)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = _project(best_h + scale * direction, target)
        sub_dl = min(deadline, time.time() + 5.0)
        h_t, c5_t = _smooth_phase(h_k, sub_dl, betas=(100, 300))
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

```python
import numpy as np
from scipy.optimize import minimize
import time

_slqp = {}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, max_alpha=1.0):
    n = len(h)
    d = d - d.mean()
    if np.max(np.abs(d)) < 1e-15:
        return 0.0, _exact_c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = 0.0
    a_hi = max_alpha
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.max(h[neg] / d[neg])))
    if a_hi - a_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_g = 25
    grid = np.linspace(a_lo, a_hi, n_g)
    vals = np.max(S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :], axis=1) * dx
    bi = int(np.argmin(vals))
    ba, bv = grid[bi], vals[bi]
    if bi > 0 and bi < n_g - 1:
        a, b = grid[bi - 1], grid[bi + 1]
    else:
        a, b = a_lo, a_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1, c2 = b - gr * (b - a), a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(25):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    oa = (a + b) / 2.0
    ov = float(np.max(S0 + oa * T - oa * oa * C)) * dx
    if ov < bv:
        ba, bv = oa, ov
    return ba, bv

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _slsqp_con_fun(x, offsets, dx, n):
    h = x[:n]
    t = x[n]
    K = len(offsets)
    vals = np.empty(K)
    for k in range(K):

        if d >= 0:
            m = n - d
            vals[k] = t - (dx * float(np.sum(h[d:d + m] * (1.0 - h[:m]))) if m > 0 else 0.0)
        else:
            a = -d
            m = n - a
            vals[k] = t - (dx * float(np.sum(h[:m] * (1.0 - h[a:a + m]))) if m > 0 else 0.0)
    return vals
