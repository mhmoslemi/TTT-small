```python
def _pair_coef(h, p, q):
    n = len(h)
    k = np.arange(-(n-1), n, dtype=np.int64)
    
    # (1-h[p-k]) valid when 0 <= p-k < n
    pk = p - k  # array
    mask1 = (pk >= 0) & (pk < n)
    
    # h[p+k] valid when 0 <= p+k < n
    p2k = p + k
    mask2 = (p2k >= 0) & (p2k < n)
    
    # (1-h[q-k]) valid when 0 <= q-k < n
    qk = q - k
    mask3 = (qk >= 0) & (qk < n)
    
    # h[q+k] valid when 0 <= q+k < n
    q2k = q + k
    mask4 = (q2k >= 0) & (q2k < n)
    
    coef = np.zeros(2*n-1)
    coef[mask1] += 1.0 - h[pk[mask1]]
    coef[mask2] -= h[p2k[mask2]]
    coef[mask3] -= 1.0 - h[qk[mask3]]
    coef[mask4] += h[q2k[mask4]]
    
    return coef
```
