            dn = np.linalg.norm(d)
            if dn < 1e-13:
                stall += 1
                continue
            d = d / dn
            mom = 0.8 * mom + 0.2 * d
            mn = np.linalg.norm(mom)
            if mn < 1e-13:
                stall += 1
                continue
            mom = mom / mn
            alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.5)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * lr * mom
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.05, 0.35)
            else:
                stall += 1
                lr *= 0.8
        else:  # Informed pairs
            h2, c5_2 = _pair_refine_informed(h, min(deadline, time.time() + 5))
            if c5_2 < cur_c5 - 1e-12:
                h = h2.copy()
                cur_c5 = c5_2
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
    return best_h, best_c5

def _random_kick(h, deadline, rng, n_kicks=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline: break
        scale = rng.uniform(0.01, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _combined_local(h_k, sub_dl, rng)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

The main issue is that the Nelder-Mead on a subset might not be effective because:
1. We're only optimizing k coordinates out of n
2. The projection after each step changes all coordinates
3. The objective is non-smooth (max of many functions)

A better approach might be:
1. Use the subgradient to identify the "most constrained" coordinates
2. Do a 1D coordinate search: for each coordinate i, find the best value of h[i] (with mass compensation via a specific pair partner)
3. Cycle through coordinates

1. **Better pair selection**: Use the subgradient to identify which pairs would be most effective
2. **Multiple passes with re-sorting**: After each improvement, recompute the subgradient and try new pairs
3. **Wider search in the multi-resolution phase**: Try more diverse resolutions

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos): t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if any(neg): t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    # Sort of Bernstein / golden section
    ng = 15
    grid = np.linspace(t_lo, t_hi, ng)
    vals = S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:]
    obj = np.max(vals, axis=1) * dx
    bi = int(np.argmin(obj))
    bt, bv = grid[bi], obj[bi]
    if bi > 0 and bi < ng-1:

    else:

    c1 = b - gr*(b-a); c2 = a + gr*(b-a)

            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx

            a, c1, f1 = c1, c2, f2

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            gd = -d
            m = n - gd
            if m > 0:
                G[r, :m] += (1.0 - h[gd:])
                G[r, gd:] += -h[:m]
    return G

def _lp_step(h, radius=0.2, penalty_lags=None):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = _grad_matrix(h)
    c = np.zeros(n+1); c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None:
        for k in penalty_lags:
            b_ub[k] -= 0.03
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0-h[j])
        if lo > hi: lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except:
        pass
    return None

def _smooth_grad(h, beta=80):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    ev = np.exp(v)
    w = ev / float(np.sum(ev))
    G = _grad_matrix(h)
    g = G.T @ w
    g -= g.mean()
    return g
