    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')

    scaled = beta * corr * dx
    scaled -= scaled.max()  # for numerical stability
    w = np.exp(scaled)

    # grad_corr_r for signed lag s = r-(n-1):
    # if s>=0: (1-h[j-s])*[j>=s] - h[j+s]*[j<=n-1-s]

            grad[s:] += wr * (1.0 - h[:n-s])  # j-s ranges 0..n-1-s as j ranges s..n-1

import numpy as np
import time

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 1e-15:
        return np.zeros(n)
    if target >= n - 1e-15:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(x.sum())
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-6
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_max_and_grad(h, beta):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr * dx
    scaled -= scaled.max()
    exp_w = np.exp(scaled)
    w = exp_w / float(exp_w.sum())
    obj = (scaled.max() / beta + float(np.log(exp_w.sum())) / beta)
    # Gradient
    grad = np.zeros(n)
    nm1 = n - 1
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        s = r - nm1
        if s >= 0:
            # grad[s:] += wr * (1 - h[:n-s])
            # grad[:n-s] -= wr * h[s:]
            if n - s > 0:
                grad[s:] += wr * (1.0 - h[:n-s])
                grad[:n-s] -= wr * h[s:]
        else:
            t = -s
            if n - t > 0:
                grad[:n-t] += wr * (1.0 - h[t:])
                grad[t:] -= wr * h[:n-t]
    return obj, grad * dx

def _pair_descent_step(h, j, k, dx):
    """Exact 1D optimization for pair (j,k) maintaining h[j]+h[k]."""
    n = len(h)
    c = h[j] + h[k]
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    if t_hi - t_lo < 1e-14:
        return h, _c5_exact(h)
    
    # Compute base correlations (with h[j]=h[k]=0) and coefficients
    h_j = h[j]
    h_k = h[k]
    h[j] = 0.0
    h[k] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[j] = 1.0
    h[k] = 0.0
    corr_j = np.correlate(h, 1.0 - h, mode='full')
    h[j] = 0.0
    h[k] = 1.0
    corr_k = np.correlate(h, 1.0 - h, mode='full')
    h[j] = h_j
    h[k] = h_k
    
    coef_j = corr_j - corr_base  # gradient of corr w.r.t. h[j]
    coef_k = corr_k - corr_base  # gradient of corr w.r.t. h[k]
    
    # corr_r(t) = corr_base[r] + t*coef_j[r] + (c-t)*coef_k[r]
    #           = corr_base[r] + c*coef_k[r] + t*(coef_j[r] - coef_k[r])
    intercept = corr_base + c * coef_k
    slope = coef_j - coef_k
    
    # f(t) = max_r (intercept[r] + t*slope[r]) * dx
    # Sample to find min
    n_samples = 80
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = np.zeros(n_samples)
    for i, t in enumerate(ts):
        vals[i] = np.max(intercept + t * slope) * dx
    
    best_idx = np.argmin(vals)
    # Refine with a few bisection steps
    lo = ts[max(0, best_idx-1)]
    hi = ts[min(n_samples-1, best_idx+1)]
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(intercept + m1 * slope)) * dx
        f2 = float(np.max(intercept + m2 * slope)) * dx
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) * 0.5
    h[j] = t_opt
    h[k] = c - t_opt
    return h, _c5_exact(h)

def _gradient_descent(h, deadline, beta_init=15.0, lr_init=0.05):
    """Smooth gradient descent with adaptive beta and learning rate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_init
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.8
    
    while time.time() < deadline:
        obj, grad = _smooth_max_and_grad(h, beta)
        # Project gradient to sum=0 hyperplane
        grad -= np.mean(grad)
        # Momentum
        momentum = mu * momentum + (1 - mu) * grad
        # Step
        h_new = h - lr * momentum
        h_new = _project_sum(h_new, target)
        new_c5 = _c5_exact(h_new)
        if new_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.05, 0.3)
            beta = min(beta * 1.02, 200.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return best_h, best_c5

def _coord_pair_descent(h, deadline, n_restarts=10):
    """Cyclic coordinate pair descent with exact 1D optimization."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Precompute which pairs to try
    # Try all adjacent pairs and some random pairs
    pairs = []
    for i in range(n):
        pairs.append((i, (i + 1) % n))
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        improved_any = False
        # Shuffle pairs for this restart
        rng = np.random.default_rng(42 + restart * 100)
        pair_list = pairs.copy()
        rng.shuffle(pair_list)
        # Add some random non-adjacent pairs
        for _ in range(n):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pair_list.append((min(i,j), max(i,j)))
        
        for (i, j) in pair_list:
            if time.time() > deadline:
                break
            h_new, c5_new = _pair_descent_step(h.copy(), i, j, dx)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                improved_any = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not improved_any:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _basin_hopping(h, deadline, n_restarts=50):
    """Basin hopping: perturb + local optimize."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    rng = np.random.default_rng(777)
    scale = 0.05
    
    for _ in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project_sum(h_pert, target)
        # Quick gradient descent
        dl = min(deadline, time.time() + 2.0)
        h_opt, c5_opt = _gradient_descent(h_pert, dl, beta_init=20, lr_init=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.15)
    
    return best_h, best_c5

import time

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 1e-15:
        return np.zeros(n)
    if target >= n - 1e-15:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(x.sum())
    if abs(res) > 1e-12:
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    nm1 = n - 1
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        s = r - nm1
        if s >= 0:
            m = n - s
            if m > 0:
                G[r, s:] += (1.0 - h[:m])
                G[r, :m] -= h[s:]
        else:
            t = -s
            m = n - t
            if m > 0:
                G[r, :m] += (1.0 - h[t:])
                G[r, t:] -= h[:m]
    return G

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr * dx
    scaled -= scaled.max()
    ew = np.exp(scaled)
    w = ew / float(ew.sum())
    G = _grad_matrix(h)
    grad = G.T @ w
    return float(scaled.max() + np.log(ew.sum())), grad * dx

def _gradient_descent(h, deadline, beta=20.0, lr=0.04, max_iter=50000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur
    momentum = np.zeros(n)
    mu = 0.75
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        _, grad = _smooth_grad(h, beta)
        grad -= np.mean(grad)
        momentum = mu * momentum + (1.0 - mu) * grad
        h_new = h - lr * momentum
        h_new = _project_sum(h_new, target)
        new_c5 = _c5_exact(h_new)
        if new_c5 < cur - 1e-15:
            h = h_new
            cur = new_c5
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            lr = min(lr * 1.03, 0.25)
            beta = min(beta * 1.01, 300.0)
        else:
            lr *= 0.72
            if lr < 1e-6:
                break
    return best_h, best_c5

    t0 = h[j]
    delta = G[:, j] - G[:, k]
    f_lo = float(np.max(corr + (t_lo - t0) * delta)) * dx
    f_hi = float(np.max(corr + (t_hi - t0) * delta)) * dx
    f_cur = float(np.max(corr)) * dx
    if f_lo < f_cur - 1e-14:
        return t_lo, f_lo
    if f_hi < f_cur - 1e-14:
        return t_hi, f_hi
    # Try midpoint and a few points
    best_t = t0
    best_f = f_cur
    for frac in [0.25, 0.5, 0.75, 0.33, 0.67]:
        t_try = t_lo + frac * (t_hi - t_lo)
        f_try = float(np.max(corr + (t_try - t0) * delta)) * dx
        if f_try < best_f:
            best_f = f_try
            best_t = t_try
    if best_f < f_cur - 1e-14:
        return best_t, best_f
    return t0, f_cur

def _pair_exact_move(h, j, k, G, corr):
    n = len(h)
    dx = 2.0 / n
    c = h[j] + h[k]
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    if t_hi - t_lo < 1e-14:
        return 0.0, _c5_exact(h)
    t0 = h[j]
    delta = G[:, j] - G[:, k]
    n_samples = 120
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = np.max(corr[None, :] + (ts - t0)[:, None] * delta[None, :], axis=1) * dx
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_samples - 1, best_idx + 1)]
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(corr + (m1 - t0) * delta)) * dx
        f2 = float(np.max(corr + (m2 - t0) * delta)) * dx
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) * 0.5
    f_opt = float(np.max(corr + (t_opt - t0) * delta)) * dx
    return t_opt, f_opt

def _pair_descent_pass(h, G, corr, deadline, n_pairs=500, rng=None):
    n = len(h)
    dx = 2.0 / n
    cur = _c5_exact(h)
    target = n / 2.0
    if rng is None:
        rng = np.random.default_rng(int(time.time() * 1000) % 1000000)
    # Sort indices by value to find promising pairs (high vs low)
    idx_sorted = np.argsort(h)
    lows = idx_sorted[:n // 2]
    highs = idx_sorted[n // 2:]
    n_ones = len(highs)
    n_zeros = len(lows)
    n_tries = min(n_pairs, n_ones * n_zeros)
    improved = False
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        i = highs[int(rng.integers(0, n_ones))]
        j = lows[int(rng.integers(0, n_zeros))]
        t_opt, f_opt = _pair_quick_move(h, i, j, G, corr)
        if f_opt < cur - 1e-14:
            c = h[i] + h[j]
            h[i] = t_opt
            h[j] = c - t_opt
            h = _project_sum(h, target)
            cur = _c5_exact(h)
            G = _grad_matrix(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            improved = True
    return h, cur, improved

def _pair_descent(h, deadline, max_rounds=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(31337)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        G = _grad_matrix(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        h, cur, improved = _pair_descent_pass(h, G, corr, deadline, n_pairs=800, rng=rng)
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
        if not improved:
            # Try exact moves on a few pairs
            G = _grad_matrix(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_sorted = np.argsort(h)
            lows = idx_sorted[:n // 2]
            highs = idx_sorted[n // 2:]
            for _ in range(200):
                if time.time() > deadline:
                    break
                i = highs[int(rng.integers(0, len(highs)))]
                j = lows[int(rng.integers(0, len(lows)))]
                t_opt, f_opt = _pair_exact_move(h, i, j, G, corr)
                if f_opt < cur - 1e-14:
                    c = h[i] + h[j]
                    h[i] = t_opt
                    h[j] = c - t_opt
                    h = _project_sum(h, target)
                    cur = _c5_exact(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    G = _grad_matrix(h)
                    corr = np.correlate(h, 1.0 - h, mode='full')
    return best_h, best_c5

def _binary_search(h, deadline, max_rounds=20):
    n = len(h)
    dx = 2.0 / n
    # Round to binary
    sorted_idx = np.argsort(-h)
    h_bi = np.zeros(n)
    h_bi[sorted_idx[:n // 2]] = 1.0
    cur = _c5_exact(h_bi)
    best_h = h_bi.copy()
    best_c5 = cur
    rng = np.random.default_rng(555)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bi == 1.0)[0]
        zeros = np.where(h_bi == 0.0)[0]
        n_tries = min(2000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[int(rng.integers(0, len(ones)))]
            j = zeros[int(rng.integers(0, len(zeros)))]
            h_bi[i] = 0.0
            h_bi[j] = 1.0
            c5_new = _c5_exact(h_bi)
            if c5_new < cur - 1e-14:
                cur = c5_new
                improved = True
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h_bi.copy()
                ones = np.where(h_bi == 1.0)[0]
                zeros = np.where(h_bi == 0.0)[0]
            else:
                h_bi[i] = 1.0
                h_bi[j] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _c5_exact(h_init)

    # Phase 1: Gradient descent (12%)
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _gradient_descent(h_init, dl1, beta=25, lr=0.05)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair descent (18%)
    dl2 = t0 + budget_s * 0.30
    h2, c5_2 = _pair_descent(best_h, dl2, max_rounds=40)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary search on current (8%)
    dl3 = t0 + budget_s * 0.38
    hb, c5_b = _binary_search(best_h, dl3, max_rounds=15)
    # Refine the binary with gradient descent
    if c5_b < best_c5 + 0.005:
        hr, c5_r = _gradient_descent(hb, dl3, beta=30, lr=0.03)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = hr.copy()

    # Phase 4: Multi-n exploration (30%)
    dl4 = t0 + budget_s * 0.68
    n_curr = len(best_h)
    offsets = [8, -8, 16, -16, 32, -32, 48, -48, 4, -4, 12, -12, 24, -24, 40, -40, 56, -56]
    tried_ns = set()
    for dn in offsets:
        if time.time() > dl4:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 320 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.018)
        h_r, c5_r = _gradient_descent(h_r, sub_dl, beta=20, lr=0.03)
        rem = dl4 - time.time()
        if rem > 5:
            sub_dl2 = min(dl4, time.time() + budget_s * 0.015)
            h_r, c5_r = _pair_descent(h_r, sub_dl2, max_rounds=20)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Basin hopping (12%)
    dl5 = t0 + budget_s * 0.80
    scale = 0.04
    for attempt in range(80):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project_sum(h_pert, len(best_h) / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.008)
        h_t, c5_t = _gradient_descent(h_pert, sub_dl, beta=30, lr=0.02)
        if time.time() < dl5 and c5_t < best_c5 + 0.003:
            sub_dl2 = min(dl5, time.time() + budget_s * 0.006)
            h_t, c5_t = _pair_descent(h_t, sub_dl2, max_rounds=10)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.12)

    # Phase 6: Final polish (10%)
    h6, c5_6 = _gradient_descent(best_h, deadline, beta=80, lr=0.015)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    h6, c5_6 = _pair_descent(best_h, deadline, max_rounds=30)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _c5_exact(best_h)
    return best_h, final_c5, len(best_h)
