def _subgradient_descent(h, deadline, n_active=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.05
    momentum = np.zeros(n)
    mu = 0.7
    
    while time.time() < deadline:
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_score = float(np.max(scores))
        
        # Active lags
        active = np.where(scores >= max_score - 2.0)[0]
        if len(active) > n_active:
            top_idx = np.argsort(scores[active])[::-1][:n_active]
            active = active[top_idx]
        
        # Gradient (negative direction to decrease max)
        grad = np.zeros(n)
        for r in active:
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:])
                    grad[d:] += (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[g:] += (1.0 - h[:m])
                    grad[:m] += (-h[g:])
        
        grad /= float(len(active))
        grad -= np.mean(grad)
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * grad
        direction = -momentum  # negative gradient
        
        # Line search
        alpha, new_c5 = _line_search(h, direction, step)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * direction
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.05, 0.2)
        else:
            step *= 0.7
            if step < 0.001:
                break
    
    return best_h, best_c5

    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -radius
    t_hi = radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(-h[neg] / d[neg])))
        t_hi = min(t_hi, float(np.min((1.0 - h[neg]) / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    n_eval = 20
    ts = np.linspace(t_lo, t_hi, n_eval)
    vals = np.empty(n_eval)
    for i in range(n_eval):
        t = ts[i]
        vals[i] = float(np.max(S0 + t * T - t * t * C))
    idx = int(np.argmin(vals))
    if idx == 0:
        a, b = ts[0], ts[min(2, n_eval - 1)]
    elif idx == n_eval - 1:
        a, b = ts[max(0, n_eval - 3)], ts[n_eval - 1]
    else:
        a, b = ts[idx - 1], ts[idx + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if b - a < 1e-11:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _subgrad_descent(h, deadline, n_active=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.08
    momentum = np.zeros(n)
    mu = 0.6
    stall = 0
    while time.time() < deadline:
        if stall > 40:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_score = float(np.max(scores))
        active = np.where(scores >= max_score - 3.0)[0]
        if len(active) > n_active:
            top_idx = np.argsort(scores[active])[::-1][:n_active]
            active = active[top_idx]
        grad = np.zeros(n)
        for r in active:
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:])
                    if d > 0:
                        grad[d:] += (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[g:] += (1.0 - h[:m])
                    grad[:m] += (-h[g:])
        grad /= float(len(active))
        grad -= np.mean(grad)
        momentum = mu * momentum + (1.0 - mu) * grad
        direction = -momentum
        norm_d = float(np.max(np.abs(direction)))
        if norm_d < 1e-12:
            break
        alpha, new_c5 = _line_search(h, direction, step)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * direction
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step = min(step * 1.03, 0.25)
        else:
            stall += 1
            step *= 0.75
            if step < 0.002:
                break
    return best_h, best_c5

def _pair_swap_pass(h, deadline, random_order=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    while time.time() < deadline:
        if random_order:
            rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            if abs(h[i] - h[j]) < 1e-12:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search(h, d, 0.5)
            if c5_try < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _block_swap_pass(h, deadline, block_size=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    starts_a = list(range(0, n - block_size + 1))
    starts_b = list(range(0, n - block_size + 1))
    while time.time() < deadline:
        improved = False
        for ia in starts_a:
            if time.time() > deadline:
                break
            for ib in starts_b:
                if abs(ia - ib) < block_size:
                    continue
                d = np.zeros(n)
                d[ia:ia + block_size] = 1.0
                d[ib:ib + block_size] = -1.0
                alpha, c5_try = _line_search(h, d, 0.3)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hopping(h, deadline, rng, n_iter=200):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.06
    T_sa = 0.005
    for it in range(n_iter):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        h_opt, c5_opt = _subgrad_descent(h_pert, min(deadline, time.time() + 1.5), n_active=3)
        delta = c5_opt - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T_sa, 1e-10)):
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                scale = max(0.01, scale * 0.85)
            else:
                scale = min(0.15, scale * 1.1)
        else:
            scale = min(0.15, scale * 1.05)
        T_sa *= 0.997
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Subgradient warmup (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _subgrad_descent(h_init, d1, n_active=4)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Exhaustive pair swap iterations (35%)
    d2 = t0 + budget_s * 0.47
    passes = 0
    while time.time() < d2:
        h2 = best_h.copy()
        h2, c5_2 = _pair_swap_pass(h2, min(d2, time.time() + budget_s * 0.18), random_order=(passes % 2 == 1))
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            passes += 1
        else:
            if passes > 3:
                break
            h2 = best_h.copy()
            h2, c5_2 = _subgrad_descent(h2, min(d2, time.time() + budget_s * 0.03), n_active=3)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
            passes += 1
            if passes > 6:
                break

    # Phase 3: Block moves (12%)
    d3 = t0 + budget_s * 0.59
    for bs in [2, 3, 4, 5]:
        if time.time() > d3:
            break
        h3 = best_h.copy()
        h3, c5_3 = _block_swap_pass(h3, min(d3, time.time() + budget_s * 0.03), block_size=bs)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Resolution sweep (22%)
    d4 = t0 + budget_s * 0.81
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -32, 32, -40, 40, -48, 48]
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 300 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _subgrad_descent(h_r, sub_dl, n_active=3)
        if time.time() < d4 - 5:
            sub_dl2 = min(d4 - 2, time.time() + budget_s * 0.012)
            h_r, c5_r = _pair_swap_pass(h_r, sub_dl2, random_order=False)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Basin hopping (12%)
    d5 = t0 + budget_s * 0.93
    h5, c5_5 = _basin_hopping(best_h, d5, rng, n_iter=150)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final polish (7%)
    d6 = deadline
    h6 = best_h.copy()
    h6, c5_6 = _subgrad_descent(h6, min(d6, time.time() + budget_s * 0.025), n_active=2)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    h6 = best_h.copy()
    h6, c5_6 = _pair_swap_pass(h6, min(d6, time.time() + budget_s * 0.045), random_order=False)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < deadline - 2:
        h6 = best_h.copy()
        h6, c5_6 = _pair_swap_pass(h6, deadline, random_order=False)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5 - 1e-12:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
