def _pair_descent(h, deadline, rng, n_pairs_per_iter=200):

    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur = h.copy()
    cur_c5 = best_c5

        improved = False
        for _ in range(n_pairs_per_iter):
            if time.time() > deadline:
                break
            # Random pair
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            # Direction: move from i to j
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            # Line search
            t, c5_new = _line_search(cur, d, 0.5, 0.5)

                cur_c5 = _exact_c5(cur)
                if cur_c5 < best_c5 - 1e-12:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5
