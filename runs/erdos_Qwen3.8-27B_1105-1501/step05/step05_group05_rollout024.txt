import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
import time

# Helper functions (top-level, no closures)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_quad(h, d, dx, max_alpha):
    """Line search for h + alpha*d using quadratic model of the objective."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds on alpha
    t_lo = -max_alpha
    t_hi = max_alpha
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid search
    n_grid = 21
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(35):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_for_lag(h, d_lag, n):
    """Gradient of corr(h,1-h) at lag d_lag with respect to h."""
    g = np.zeros(n)
    if d_lag >= 0:
        m = n - d_lag
        if m > 0:
            g[:m] += (1.0 - h[d_lag:d_lag+m])
            g[d_lag:] += -h[:m]
    else:
        g2 = -d_lag
        m = n - g2
        if m > 0:
            g[:m] += (1.0 - h[g2:g2+m])
            g[g2:] += -h[:m]
    return g

def _lsumexp_grad(h, dx, beta):
    """Smooth approximation of max using log-sum-exp, compute gradient."""
    n = len(h)
    scores = _scores(h) * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    
    # Gradient: sum over lags of w[k] * grad_lag
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d_lag = k - (n - 1)
        gl = _grad_for_lag(h, d_lag, n)
        grad += w[k] * gl
    grad -= np.mean(grad)  # Zero-sum constraint
    return grad

def _smooth_descent(h, deadline, beta_list=(20, 50, 100, 200, 400, 800), lr_init=0.03):
    """Smooth descent with Nesterov momentum and adaptive beta."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    lr = lr_init
    v = np.zeros(n)  # momentum
    beta_idx = 0
    stall = 0
    total_iters = 0
    while time.time() < deadline:
        beta = beta_list[min(beta_idx, len(beta_list)-1)]
        g = _lsumexp_grad(h, dx, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            beta_idx += 1
            stall += 1
            if stall > 30:
                break
            continue
        g = g / gn
        # Nesterov: v = mu*v + (1-mu)*g, direction = v
        mu = 0.85
        v = mu * v + (1.0 - mu) * g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            continue
        v = v / vn
        step = lr * v
        alpha, new_c5 = _line_search_quad(h, step, dx, lr * 3.0)
        total_iters += 1
        if new_c5 < cur - 1e-14:
            h = h + alpha * step
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.3)
            if total_iters % 50 == 0:
                beta_idx += 1  # Gradually increase beta
        else:
            stall += 1
            lr *= 0.8
            if stall > 20:
                v = np.zeros(n)
                beta_idx += 1
                stall = 0
                lr = lr_init * 0.5
                if beta_idx >= len(beta_list) * 2:
                    break
    return best_h, best_c5

def _resample_cubic(h, n_new):
    """Resample using cubic interpolation."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    try:
        f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
        h_new = f(x_new)
    except Exception:
        h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project(h_new, n_new / 2.0)

def _resample_linear(h, n_new):
    """Resample using linear interpolation."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_step(h, scores, radius):
    """LP step to find descent direction."""
    n = len(h)
    target = n / 2.0
    # Build gradient matrix
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d_lag = k - (n-1)
        if d_lag >= 0:
            m = n - d_lag
            if m > 0:
                G[k, d_lag:] += (1.0 - h[:m])
                G[k, :m] += -h[d_lag:]
        else:
            g2 = -d_lag
            m = n - g2
            if m > 0:
                G[k, :m] += (1.0 - h[g2:])
                G[k, g2:] += -h[:m]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _kick_optimize(h, deadline, rng, n_kicks=30):
    """Random kick and re-optimize."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.03, 0.10)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = _project(best_h + scale * direction, target)
        # Quick smooth optimization from the kicked point
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _smooth_descent(h_k, sub_dl, beta_list=(30, 80, 200), lr_init=0.02)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _full_refine(h, deadline):
    """Full refinement: smooth + pair + LP, cycling."""

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_quad(h, d, dx, max_alpha):

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[k, d:] += (1.0 - h[:m])
                G[k, :m] += -h[d:]
        else:
            g2 = -d
            m = n - g2
            if m > 0:
                G[k, :m] += (1.0 - h[g2:])
                G[k, g2:] += -h[:m]
    return G


def _lsumexp_grad(h, dx, beta, G=None):
    n = len(h)
    if G is None:
        G = _grad_matrix(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    v -= v.max()
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _resample_cubic(h, n_new):

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    try:
        f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
        h_new = f(x_new)
    except Exception:
        h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project(h_new, n_new / 2.0)

    h_new = np.interp(x_new, x_old, h)

def _lp_direction(h, scores, radius, penalty_lags=None):
    n = len(h)
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        b_ub[penalty_lags] -= 0.03
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

        elif phase == 'lp':
            h3, c3 = _lp_loop(best_h, dl, radius_init=0.18)
            if c3 < best_c5:
                best_c5, best_h = c3, h3
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase A: Quick combined refinement at original n (12%)
    dA = t0 + budget_s * 0.12
    hA, cA = _refine_cycle(h, dA)
    if cA < best_c5:
        best_c5, best_h = cA, hA

    # Phase B: Nearby resolutions with cubic resampling (22%)
    dB = t0 + budget_s * 0.34
    n_base = len(best_h)
    tried_n = set()
    for dn in sorted(range(-6, 13, 2), key=lambda x: (abs(x), x)):
        if time.time() > dB:
            break
        n_try = n_base + dn
        if n_try < 80 or n_try > n_base + 24:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        if abs(dn) % 4 == 0:
            h_r = _resample_cubic(best_h, n_try)
        else:
            h_r = _resample_linear(best_h, n_try)
        sub_dl = min(dB, time.time() + budget_s * 0.025)
        h_r, c5_r = _refine_cycle(h_r, sub_dl)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r

    # Phase C: Higher resolution exploration (18%)
    dC = t0 + budget_s * 0.52
    n_current = len(best_h)
    for n_hi in [n_current + 40, n_current + 80, n_current + 130, n_current + 200]:
        if time.time() > dC:
            break
        if n_hi > 500:
            break
        h_hi = _resample_cubic(best_h, n_hi)
        sub_dl1 = min(dC, time.time() + budget_s * 0.05)
        h_hi, c5_hi = _smooth_descent(h_hi, sub_dl1, beta_list=(10, 30, 80, 200, 500), lr_init=0.018)
        if time.time() < dC:
            sub_dl2 = min(dC, time.time() + budget_s * 0.03)
            h_hi, c5_hi = _pair_refine(h_hi, sub_dl2, n_passes=25, max_pairs=8000)
        if c5_hi < best_c5:
            best_c5, best_h = c5_hi, h_hi

    # Phase D: Return to best resolution, intensive refinement (22%)
    n_best = len(best_h)
    dD = t0 + budget_s * 0.74
    hD, cD = _refine_cycle(best_h, dD)
    if cD < best_c5:
        best_c5, best_h = cD, hD

    # Phase E: LP with penalty on active lags (10%)
    dE = t0 + budget_s * 0.84
    hE, cE = _lp_loop(best_h, dE, radius_init=0.15)
    if cE < best_c5:
        best_c5, best_h = cE, hE
    if time.time() < dE:
        hE2, cE2 = _pair_refine(best_h, min(dE, time.time() + 15), n_passes=30)
        if cE2 < best_c5:
            best_c5, best_h = cE2, hE2

    # Phase F: Random kicks (6%)
    dF = t0 + budget_s * 0.90
    hF, cF = _kick_optimize(best_h, dF, rng, n_kicks=35)
    if cF < best_c5:
        best_c5, best_h = cF, hF

    # Phase G: Final intensive pair refinement (10%)
    dG = t0 + budget_s - 1.5
    hG, cG = _pair_refine(best_h, dG, n_passes=200, smart=False)
    if cG < best_c5:
        best_c5, best_h = cG, hG

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
