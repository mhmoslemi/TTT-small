    G = np.zeros((2 * n - 1, n))
    one_minus_h = 1.0 - h
    for d in range(n):
        r = n - 1 + d
        # First factor: j in [0, n-1-d], G[r,j] += dx*(1-h[j+d])
        G[r, :n-d] += dx * one_minus_h[d:]
        # Second factor: j in [d, n-1], G[r,j] += -dx*h[j-d]
        G[r, d:] += -dx * h[:n-d]

        m = n - s
        # First factor: j in [s, n-1], G[r,j] += dx*(1-h[j-s])
        G[r, s:] += dx * one_minus_h[:m]
        # Second factor: j in [0, m-1], G[r,j] += -dx*h[j+s]
        G[r, :m] += -dx * h[s:]
    return G

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x > 1e-15) & (x < 1 - 1e-15)
            if not np.any(mask):
                mask = x < 1
        else:
            mask = (x > 1e-15) & (x < 1 - 1e-15)
            if not np.any(mask):
                mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for d in range(n):
        r = n - 1 + d
        G[r, :n-d] += dx * omh[d:]
        G[r, d:] += -dx * h[:n-d]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, s:] += dx * omh[:m]
        G[r, :m] += -dx * h[s:]
    return G

    n = len(h)

def _line_search(h, p, max_iter=30):
    n = len(h)
    target = n / 2.0
    # Find alpha_max
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = float(alpha_max)
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    f0 = _exact_c5(h)
    for _ in range(max_iter):
        if hi - lo < 1e-12 * max(1.0, alpha_max):
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_h = _project(h + opt_a * p, target)
    opt_c5 = _exact_c5(opt_h)
    if opt_c5 > f0:
        return 0.0, f0, h
    return opt_a, opt_c5, opt_h

def _lp_find_direction(h, G, radius, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    scores = _scores(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    # LP: min t s.t. G_k @ delta - t <= -s_k, sum(delta) = 0, bounds
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
