import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    if abs(res) > 1e-12:

    """Local search over binary sequences with exactly n/2 ones."""
    target = n // 2
    # Convert to binary
    h = np.sort(h_init)[-target:]  # take the top n/2 values
    # Actually, let's just threshold
    h_bin = (h_init > 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Turn off some 1s (the ones with lowest h_init value)
        idx_ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_init[idx_ones])
        to_flip = idx_ones[order[:n_ones - target]]
        h_bin[to_flip] = 0
    elif n_ones < target:
        # Turn on some 0s (the ones with highest h_init value)
        idx_zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_init[idx_zeros])
        to_flip = idx_zeros[order[:target - n_ones]]
        h_bin[to_flip] = 1
    
    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)
    cur_h = h_bin.copy()
    cur_c5 = best_c5
    
    ones = set(np.where(cur_h == 1)[0].tolist())
    zeros = set(np.where(cur_h == 0)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        best_swap = None
        best_swap_c5 = cur_c5
        
        # Try all swaps

        zeros_list = sorted(zeros)
        
        # Limit the number of swaps to try per iteration for speed
        # Try a random subset if too many
        max_swaps_per_iter = 200
        if len(ones_list) * len(zeros_list) > max_swaps_per_iter:
            # Random sample
            pairs = [(ones_list[np.random.randint(len(ones_list))], 
                     zeros_list[np.random.randint(len(zeros_list))])
                     for _ in range(max_swaps_per_iter)]
        else:
            pairs = [(i, j) for i in ones_list for j in zeros_list]
        
        for i, j in pairs:
            if cur_h[i] != 1 or cur_h[j] != 0:

            cur_h[j] = 1

            cur_h[j] = 0

                best_swap = (i, j)
        
        if best_swap is not None:

            cur_h[i] = 0
            cur_h[j] = 1
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            cur_c5 = best_swap_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:

            candidates.add(i)  # k=i, k+d = i+d

            candidates.add(i-d)  # k=i-d, k+d = i

            candidates.add(j-d)  # k=j-d, k+d = j

            # Old term: h[k]*(1-h[k+d])
            # New term: h'[k]*(1-h'[k+d])

            new_v = 0.0 if k == i else (1.0 if k == j else h[k])
            old_partner = h[k+d]

            delta += (new_v * (1 - new_partner) - old_v * (1 - old_partner))
        
        deltas[d] = delta * dx
    return deltas
