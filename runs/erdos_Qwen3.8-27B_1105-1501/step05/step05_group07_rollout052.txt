    #              = sum_k w_k * [beta^2 * S'_k_i * S'_k_j + beta * S''_k_ij] - (beta * sum_k w_k * S'_k_i) * (beta * sum_k w_k * S'_k_j)
    #              = beta * sum_k w_k * S''_k_ij + beta^2 * [sum_k w_k * S'_k_i * S'_k_j - (sum_k w_k * S'_k_i)(sum_k w_k * S'_k_j)]
    #              = beta * sum_k w_k * S''_k_ij + beta^2 * Cov_w(S'_k_i, S'_k_j)
    
    return H, grad

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _hessian_los(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    grad = beta * (G.T @ w)
    H1 = np.zeros((n, n))
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-14:
            continue
        d = k_idx - (n - 1)
        if d > 0:
            m = n - d
            for j in range(m):
                H1[j, j + d] += -dx * w[k_idx]
                H1[j + d, j] += -dx * w[k_idx]
        elif d < 0:
            gg = -d
            m = n - gg
            for j in range(m):
                H1[j, j + gg] += -dx * w[k_idx]
                H1[j + gg, j] += -dx * w[k_idx]
    mean_grad = G.T @ w
    H2 = np.zeros((n, n))
    active = w > 1e-14
    for k_idx in np.where(active)[0]:
        diff = beta * (G[k_idx, :] - mean_grad)
        H2 += w[k_idx] * np.outer(diff, diff)
    H = beta * H1 + H2
    return H, grad

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5, max_iter=30):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    if alpha_max > 1.0:
        alpha_max = 1.0
    n_samp = 30
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _local_optimize_lp(h, deadline, radius_init=0.3, max_iters=10000, stall_limit=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p, lp_t = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-15:
            stall += 1
            radius *= 0.85
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.6)
        else:
            stall += 1
            radius *= 0.88
            if stall > 50 and radius < 0.05:
                radius = 0.15
                stall = 20
    return best_h, best_c5

def _newton_refine(h, deadline, beta_start=50.0, beta_end=500.0, n_phases=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        for inner in range(30):
            if time.time() > deadline:
                break
            H, grad = _hessian_los(h, beta)
            lam = 1e-4
            for _ in range(20):
                try:
                    p = np.linalg.solve(H + lam * np.eye(n), -grad)
                    if np.all(np.isfinite(p)):
                        break
                except np.linalg.LinAlgError:
                    pass
                lam *= 10.0
            if not np.all(np.isfinite(p)):
                break
            if np.max(np.abs(p)) < 1e-15:
                break
            alpha, new_c5 = _line_search(h, p, _exact_c5(h))
            if new_c5 < best_c5 - 1e-15:
                h_new = _project(h + alpha * p, target)
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            elif inner > 5:
                break
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def _slsqp_optimize(h, deadline, beta_init=50.0, beta_final=300.0, n_phases=4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        def _obj(hf):
            S = _scores(hf)
            b = beta
            maxS = np.max(S)
            if b < 1e-6:
                return maxS
            shifted = b * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            return maxS + np.log(np.mean(np.exp(shifted))) / b
        def _grad(hf):
            S = _scores(hf)
            b = beta
            maxS = np.max(S)
            shifted = b * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            wsum = np.sum(w)
            if wsum < 1e-300:
                idx = int(np.argmax(S))
                w = np.zeros(len(S))
                w[idx] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w
        try:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-15})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _subgrad_optimize(h, deadline, n_steps=3000, beta_start=20.0, beta_end=80.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.1
        improved = False
        for _ in range(40):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if not improved and step % 100 == 0:
            alpha = 0.01
            h = _project(h - alpha * grad, target)
            cur_c5 = _exact_c5(h)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 48)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _multi_res_optimize(h, deadline):

                    best_c5 = ct
                    best_h = ht.copy()
                if time.time() > dl6:
                    break

    # Phase 7: Newton polish
    if time.time() < t0 + budget_s * 0.93:
        dl7 = t0 + budget_s * 0.93
        h7, c5_7 = _newton_refine(best_h, dl7, beta_start=100.0, beta_end=600.0, n_phases=4)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final LP + SLSQP
    if time.time() < t0 + budget_s * 0.96:
        dl8 = t0 + budget_s * 0.96
        h8, c5_8 = _slsqp_optimize(best_h, dl8, beta_init=100.0, beta_final=500.0, n_phases=3)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
        h8b, c5_8b = _local_optimize_lp(best_h, min(deadline, time.time() + budget_s * 0.02),
                                         radius_init=0.03, max_iters=5000, stall_limit=150)
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()

    # Phase 9: Pair polish
    if time.time() < t0 + budget_s * 0.985:
        hp, c5p = _pair_polish(best_h, deadline, max_passes=5)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
