def dinic_max_flow(adj, source, sink):
    # adj: list of lists of (neighbor, capacity, reverse_edge_index)
    # Returns max flow value
    
    n_nodes = len(adj)
    flow = 0
    while True:
        # BFS to find level graph
        level = [-1] * n_nodes
        level[source] = 0
        queue = [source]
        for u in queue:
            for v, cap, rev in adj[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        if level[sink] < 0:

        it = [0] * n_nodes
        while True:
            f = dfs(source, sink, float('inf'), level, it, adj)
            if f == 0:
                break
            flow += f
    return flow

def dfs(u, sink, f, level, it, adj):

                return d
    it[u] = len(adj[u])
    return 0

                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.03, 0.35)
        else:
            radius *= 0.75
            if radius < 0.01:
                radius = 0.08
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _local_bit_flip(h, deadline, n_trials=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Try swapping h[i] <-> h[j]
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

def _slsqp_iterative(h, deadline, n_rounds=3):
    n = len(h)
    cur_h = h.copy()
    cur_c5 = _exact_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        # Update active constraints based on current scores
        scores = _scores(cur_h)
        max_s = np.max(scores)
        # Include all constraints above 0.95*max
        active_mask = scores >= 0.95 * max_s
        n_act = int(np.sum(active_mask))
        if n_act < 5:
            n_act = min(10, 2 * n - 1)
        
        k_constr = min(n_act + 5, 2 * n - 1)
        sub_dl = min(deadline, time.time() + 30.0)
        cur_h, cur_c5 = _slsqp_minimax(cur_h, sub_dl, n_constr=k_constr)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    n0 = len(initial_h_values)
    best_h = _project(np.array(initial_h_values, dtype=float), n0 / 2.0)
    best_c5 = _exact_c5(best_h)
    best_n = n0
    
    # Phase 1: SLSQP minimax on initial (most promising new approach)
    dl1 = t0 + budget_s * 0.15
    h_sls, c5_sls = _slsqp_iterative(best_h, dl1, n_rounds=3)
    if c5_sls < best_c5:
        best_c5 = c5_sls
        best_h = h_sls.copy()
    
    # Phase 2: Min-cut for structural changes
    dl2 = t0 + budget_s * 0.30
    for beta_try in [3.0, 5.0, 8.0, 12.0]:
        if time.time() > dl2:
            break
        sub_dl = min(dl2, time.time() + budget_s * 0.03)
        h_mc, c5_mc = _mincut_solve(best_h, sub_dl, beta=beta_try)
        if c5_mc < best_c5:
            best_c5 = c5_mc
            best_h = h_mc.copy()
            # Polish with LP
            sub_dl2 = min(dl2, time.time() + budget_s * 0.02)
            h_pol, c5_pol = _lp_optimize(best_h, sub_dl2, radius_init=0.15, max_it=1000, topk=20)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 3: LP refinement
    dl3 = t0 + budget_s * 0.42
    h_lp, c5_lp = _lp_optimize(best_h, dl3, radius_init=0.2, max_it=3000, topk=30)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()
    
    # Phase 4: SLSQP again after LP
    dl4 = t0 + budget_s * 0.55
    h_sls2, c5_sls2 = _slsqp_iterative(best_h, dl4, n_rounds=2)
    if c5_sls2 < best_c5:
        best_c5 = c5_sls2
        best_h = h_sls2.copy()
    
    # Phase 5: Try different resolutions
    dl5 = t0 + budget_s * 0.75
    res_ns = [96, 112, 144, 160, 176, 192]
    for n_try in res_ns:
        if time.time() > dl5:
            break
        if n_try == best_n:
            continue
        h_res = _resample(best_h, n_try)
        c5_res = _exact_c5(h_res)
        sub_dl = min(dl5, time.time() + budget_s * 0.03)
        # Quick LP at this resolution
        h_rq, c5_rq = _lp_optimize(h_res, sub_dl, radius_init=0.15, max_it=1000, topk=20)
        if c5_rq < best_c5:
            best_c5 = c5_rq
            best_h = h_rq.copy()
            best_n = n_try
            # SLSQP polish
            sub_dl2 = min(dl5, time.time() + budget_s * 0.025)
            h_sp, c5_sp = _slsqp_minimax(h_rq, sub_dl2, n_constr=30)
            if c5_sp < best_c5:
                best_c5 = c5_sp
                best_h = h_sp.copy()
    
    # Phase 6: Local bit flips + LP
    if time.time() < t0 + budget_s * 0.85:
        dl6 = t0 + budget_s * 0.90
        h_bf, c5_bf = _local_bit_flip(best_h, dl6, n_trials=3000)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        sub_dl = min(dl6, time.time() + budget_s * 0.015)
        h_bp, c5_bp = _lp_optimize(best_h, sub_dl, radius_init=0.1, max_it=800, topk=15)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()
    
    # Phase 7: Final SLSQP + LP polish
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.985
        h_fin_s, c5_fin_s = _slsqp_minimax(best_h, dl7, n_constr=25)
        if c5_fin_s < best_c5:
            best_c5 = c5_fin_s
            best_h = h_fin_s.copy()
        sub_dl = min(dl7, time.time() + budget_s * 0.01)
        h_fin_l, c5_fin_l = _lp_optimize(best_h, sub_dl, radius_init=0.05, max_it=1000, topk=15)
        if c5_fin_l < best_c5:
            best_c5 = c5_fin_l
            best_h = h_fin_l.copy()
    
    # Final projection
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
