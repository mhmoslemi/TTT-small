1. Use log-sum-exp smoothing with adaptive temperature
2. Use scipy.optimize.minimize with SLSQP (handles equality + inequality constraints)
3. Do active-set balancing after the smooth phase
4. Multi-resolution search
5. Exact line search refinement

Let me think about efficiency. For n=122, the gradient matrix is (243, 122) ≈ 30k elements. Computing it each time is O(n^2) which is fine. The SLSQP optimizer will call the objective+gradient many times, but each call is O(n^2) ≈ 15k operations, which is fast.

The main concern is that SLSQP might be slow if there are many iterations. I'll limit maxiter and use a warm-start strategy.

Let me also think about a key insight: at the optimum, the active lags (those achieving the maximum) should have their gradients pointing in "opposite" directions in some sense, so that no single move can reduce all of them. The smooth log-sum-exp naturally balances this through the weights.

Actually, I realize there might be an issue with the smooth approach: the log-sum-exp can have numerical issues if tau is very small. I'll handle this by:
- Starting with a reasonable tau (maybe 0.01 * C5)
- Decreasing tau geometrically
- At each tau, run SLSQP for a few iterations
- When the improvement becomes too small, decrease tau further

Let me also add a "direct active-set" method as a complement:
- Identify the top m lags (say m=5)
- Solve a small QP: minimize max of the top m quadratics
- This can be done by introducing a variable t and minimizing t subject to F_k(h+alpha*d) <= t for k in active set
- But this is still nonconvex...

Actually, for the active set approach, since each F_k is quadratic in alpha (along a direction d), we can solve:
min_alpha max_{k in A} (S_k + alpha*L_k - alpha^2*A_k)

This is a 1D problem (for a given direction d) and can be solved exactly with the line search. The challenge is choosing d.

For the direction, I'll use the gradient of the smooth surrogate (which is a weighted combination of active constraint gradients). This should point in a direction that reduces the overall "pressure" from all active constraints.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    nlags = 2 * n - 1
    G = np.zeros((nlags, n))
    for r in range(nlags):
        k = r - (n - 1)
        if k >= 0:
            d = k
            m = n - d
            if m > 0:
                G[r, d:d+m] += (1.0 - h[:m]) * dx
                G[r, :m] += -h[d:d+m] * dx
        else:
            g = -k
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:g+m]) * dx
                G[r, g:g+m] += -h[:m] * dx
    return G

def _smooth_obj_grad(h, tau):
    corr = _corr_values(h)
    M = float(np.max(corr))
    shifted = (corr - M) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0, 50.0)
    exps = np.exp(shifted)
    Z = float(np.sum(exps))
    psi = M + tau * np.log(Z)
    weights = exps / Z
    G = _build_grad_matrix(h)
    grad = weights @ G
    return psi, grad

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, -float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Grid search
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _smooth_optimize(h_init, deadline, tau_init, tau_min, n_starts=1):
    """Optimize using log-sum-exp smoothing with decreasing temperature."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    improvement_threshold = 1e-6
    stall = 0
    while time.time() < deadline and tau > tau_min:
        def obj_grad(x):
            x_p = np.clip(x, 0.0, 1.0)
            val, g = _smooth_obj_grad(x_p, tau)
            return val, g
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                        'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_grad, h, jac=True, constraints=constraints,
                          bounds=bounds, method='SLSQP',
                          options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
            if res.success:
                h_try = _project(res.x, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-10:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    stall = 0
                else:
                    h = _project(res.x, target)
                    stall += 1
            else:
                stall += 1
        except Exception:
            stall += 1
        if stall >= 3:
            tau *= 0.6
            stall = 0
        else:
            tau *= 0.85
    return best_h, best_c5

def _active_set_refine(h, deadline):
    """Focus on reducing the active lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    stall = 0
    while time.time() < deadline and stall < 50:
        corr = _corr_values(h)
        M = float(np.max(corr))
        # Active lags: those within 1% of max
        active_tol = max(1e-6, 0.01 * M)
        active = np.where(corr >= M - active_tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        # Get gradients for active lags
        G = _build_grad_matrix(h)
        G_active = G[active, :]  # shape (len(active), n)
        # Try to find direction that reduces all active constraints
        # Use negative gradient of the sum of active constraints
        g_sum = G_active.sum(axis=0)
        # Project out the constraint gradient (sum direction)
        # We need sum(d) = 0
        n_active = len(active)
        # Subtract the mean to make sum zero
        d = -g_sum - g_sum.mean()
        # Also zero out components where h is at bounds
        at_lower = h < 1e-10
        at_upper = h > 1.0 - 1e-10
        d[at_lower & (d < 0)] = 0
        d[at_upper & (d > 0)] = 0
        # Re-project to sum zero
        d = d - d.mean()
        if float(np.max(np.abs(d))) < 1e-14:
            # Try random pair moves
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                stall += 1
                continue

            d[i] = -1.0
            d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, 0.3)

            h_new = h + alpha * d
            h_new = _project(h_new, target)
            cur = _exact_c5(h_new)
            if cur < best_c5:
                best_c5 = cur
                best_h = h_new.copy()
                h = h_new.copy()

            else:

        else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_descent(h, deadline, max_passes=100):
    """Systematic pair coordinate descent with exact line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Get gradient to prioritize moves
        G = _build_grad_matrix(h)
        corr = _corr_values(h)
        M = float(np.max(corr))
        active_tol = 0.02 * M
        active = np.where(corr >= M - active_tol)[0]
        # For each coordinate, compute "pressure" from active lags
        pressure = np.zeros(n)
        for r in active:
            k = r - (n-1)
            pressure += np.abs(G[r, :])
        # Try moves from high-pressure to low-pressure coordinates
        order = np.argsort(-pressure)
        for _iter in range(200):
            if time.time() > deadline:
                break
            i = order[rng.integers(len(order))]
            # Try moving mass from i to a random j
            j = rng.integers(n)
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-12:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                cur = _exact_c5(h_new)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h_new.copy()
                    h = h_new.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _lp_direction(h, scores, G, radius):
    n = len(h)
    nlags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((nlags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
