import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):

    G = np.zeros((2 * n - 1, n))
    for d in range(n):

        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s

def _golden_1d(scores, C, lo, hi, n_iter=35):
    """Minimize max(scores + t*C) for t in [lo, hi] via golden section."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + c1 * C))
    f2 = float(np.max(scores + c2 * C))
    for _ in range(n_iter):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + c1 * C))
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(scores + opt_t * C))
    return opt_t, opt_val

    G = _build_G(h)
    temp = anneal_temp
    stall_count = 0
    for _pass in range(max_passes):

            break
        pass_improved = False
        n_improve = 0

def _subgrad_descent(h, deadline, n_steps=5000, beta=50.0, momentum=0.7):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)  # momentum
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = scores.max()
        # Softmax weights for active constraints
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _build_G(h)
        grad = G.T @ w
        # Project to mass-preserving direction
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        # Nesterov-like momentum
        v = momentum * v - 0.02 * grad
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn
        # Line search
        alpha = 0.05
        improved = False
        for _ in range(20):
            h_new = _project(h + alpha * v, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
        if not improved:
            # Try negative direction
            alpha = 0.05
            for _ in range(10):
                h_new = _project(h - alpha * v, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    v = -v
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
                alpha *= 0.7
                if alpha < 1e-10:
                    break
        if step % 100 == 0 and beta < 200:
            beta *= 1.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _triple_descent(h, deadline, max_passes=200, rng=None):
    """Try moving mass from 2 points to 1 (or 1 to 2) to escape local minima."""
    n = len(h)

        dl6 = hard_deadline
        # Aggressive pair descent with tiny annealing
        h6, c5_6 = _pair_descent(best_h, dl6, max_passes=5000, anneal_temp=0.0001, rng=rng)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
