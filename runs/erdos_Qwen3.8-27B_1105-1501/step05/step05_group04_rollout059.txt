import numpy as np
from scipy.optimize import minimize, linprog
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _lag_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return _project(h_new, n_new / 2.0)

def _compute_lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for k in range(-(n - 1), n):
        row = n - 1 + k
        if k > 0:
            J[row, k:] = dx * (1.0 - h[:n - k])
            J[row, :n - k] += -dx * h[k:]
        elif k < 0:
            a = -k
            J[row, :n - a] = dx * (1.0 - h[a:])
            J[row, a:] += -dx * h[:n - a]
        else:
            J[row, :] = dx * (1.0 - 2.0 * h)
    return J

def _slsqp_solve(h0, max_time, max_iter=300, k_active=None, t_margin=0.01):
    n = len(h0)
    dx = 2.0 / n
    target_n = n / 2.0
    n_lags = 2 * n - 1
    c0 = _eval_c5(h0)
    
    if k_active is None:
        k_active = n_lags  # Use all constraints by default
    
    lags = _lag_values(h0)
    active_idx = np.argsort(lags)[::-1][:k_active]
    k_act = len(active_idx)
    
    def _obj(x):
        return x[n]
    
    def _obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def _eq_fun(x):
        return np.sum(x[:n]) - target_n
    
    def _eq_jac(x):
        j = np.ones(n + 1)
        j[n] = 0.0
        return j
    
    def _ineq_fun(x):
        h_v = x[:n]
        t = x[n]
        corr = np.correlate(h_v, 1.0 - h_v, mode='full')
        return t - corr[active_idx] * dx
    
    def _ineq_jac(x):
        h_v = np.clip(x[:n], 0.0, 1.0)
        G = _compute_lag_jacobian(h_v)
        J = np.zeros((k_act, n + 1))
        J[:, :n] = -G[active_idx, :]
        J[:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    x0 = np.empty(n + 1)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _lag_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return _project(h_new, n_new / 2.0)

def _lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for k in range(-(n - 1), n):
        row = n - 1 + k
        if k > 0:
            J[row, k:] = dx * (1.0 - h[:n - k])
            J[row, :n - k] += -dx * h[k:]
        elif k < 0:
            a = -k
            J[row, :n - a] = dx * (1.0 - h[a:])
            J[row, a:] += -dx * h[:n - a]
        else:
            J[row, :] = dx * (1.0 - 2.0 * h)
    return J

def _slsqp_epigraph(h0, max_time, max_iter=300, k_active=None):
    n = len(h0)
    dx = 2.0 / n
    target_n = n / 2.0
    n_lags = 2 * n - 1
    c0 = _eval_c5(h0)
    if k_active is None or k_active >= n_lags:
        k_active = n_lags
        active_idx = np.arange(n_lags)
    else:
        lags = _lag_values(h0)
        active_idx = np.argsort(lags)[::-1][:k_active]
    k_act = len(active_idx)
    
    def obj(x):
        return x[n]
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def eq_fun(x):
        return float(np.sum(x[:n]) - target_n)
    def eq_jac(x):
        j = np.ones(n + 1)
        j[n] = 0.0
        return j
    def ineq_fun(x):
        corr = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - corr[active_idx] * dx
    def ineq_jac(x):
        h_v = np.clip(x[:n], 0.0, 1.0)
        G = _lag_jacobian(h_v)
        J = np.zeros((k_act, n + 1))
        J[:, :n] = -G[active_idx, :]
        J[:, n] = 1.0
        return J

def _slsqp_smooth(h0, beta, max_time, max_iter=100):

    n = len(h0)

    target_n = n / 2.0

        m = np.max(F)

        # numerical stability

        log_sum = np.log(np.sum(np.exp(shifted)))
        return m + beta * log_sum

        m = np.max(F)
        shifted = (F - m) / beta
        shifted = np.clip(shifted, -500, 500)
        Z = np.exp(shifted)
        P = Z / np.sum(Z)
        G = _lag_jacobian(x)
        return P @ G

def _coord_refine(h, deadline, rng, n_iter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Try moving mass from i to j (and vice versa)
        for sign in (1, -1):
            d = np.zeros(n)
            d[i] = -sign
            d[j] = sign
            # Find valid step range
            if sign > 0:
                t_max = min(h[i], 1.0 - h[j])
            else:
                t_max = min(h[j], 1.0 - h[i])
            if t_max <= 1e-15:
                continue
            # Golden section search
            a, b = 0.0, t_max
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            h1 = h + c1 * d
            h2 = h + c2 * d
            f1 = _eval_c5(h1)
            f2 = _eval_c5(h2)
            for _ in range(30):
                if b - a < 1e-14:
                    break
                if f1 < f2:
                    b, c2, f2 = c1, c1, f1
                    c1 = b - gr * (b - a)
                    h1 = h + c1 * d
                    f1 = _eval_c5(h1)
                else:
                    a, c1, f1 = c2, c2, f2
                    c2 = a + gr * (b - a)
                    h2 = h + c2 * d
                    f2 = _eval_c5(h2)
            t_opt = (a + b) / 2.0
            h_new = h + t_opt * d
            h_new = _project(h_new, target)
            c5_new = _eval_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    return best_h, best_c5
