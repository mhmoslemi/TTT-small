```python
def _softmax_grad(h, beta, top_k=None):
    """Compute gradient of softmax approximation of max score * dx."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Softmax weights
    # For numerical stability, subtract max
    max_s = np.max(scores)
    exp_s = np.exp(beta * (scores - max_s))
    w = exp_s / np.sum(exp_s)
    
    # If top_k, keep only top k
    if top_k is not None and top_k < len(w):
        idx = np.argsort(w)[::-1][:top_k]
        w = np.zeros_like(w)
        w[idx] = 1.0 / top_k
    
    # Gradient: weighted sum of gradients of each score
    # grad of score_d w.r.t. h[j]:
    # For lag d >= 0: (1-h[j+d])*[j<n-d] - h[j-d]*[j>=d]
    # For lag d < 0 (d' = -d): (1-h[j-d'])*[j-d'>=0 and j-d'<n-d'] - h[j+d']*[j<n-d']
    
    # Let me compute this efficiently
    # The gradient of score at lag d (r = n-1+d) w.r.t. all h:
    # Use the _grad_matrix approach but weighted
    
    # Actually, let me compute it as:
    # grad[j] = sum_d w[d] * (partial score_d / partial h[j])
    
    # For efficiency, I'll compute it using the structure:
    # partial score_d / partial h[j] = (1-h[j+d])*[0<=j<n-d] - h[j-d]*[d<=j<n] for d>=0
    # = (1-h[j-d'])*[0<=j-d'<n-d'] - h[j+d']*[0<=j<n-d'] for d=-d'<0
    
    # Let me build the gradient directly
    grad = np.zeros(n)
    for r in range(2*n - 1):
        d = r - (n - 1)  # signed lag
        if abs(w[r]) < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                # (1-h[j+d]) for j in [0, m-1]
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                # -h[j-d] for j in [d, n-1]
                grad[d:] += -w[r] * h[:m]
        else:
            dp = -d
            m = n - dp
            if m > 0:
                # (1-h[j-dp]) for j in [dp, n-1], i.e., j-dp in [0, m-1]
                grad[dp:] += w[r] * (1.0 - h[:m])
                # -h[j+dp] for j in [0, m-1]
                grad[:m] += -w[r] * h[dp:dp+m]
    
    return grad * dx  # The gradient of f = max * dx is dx * grad of max
```

```python
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            # a = (1-h[j+d])*[j <= n-1-d] - h[j-d]*[j >= d]
            val = 0.0
            if j + d < n:  # j <= n-1-d
                val += (1.0 - h[j + d])
            if j >= d:  # j-d >= 0
                val -= h[j - d]
            a[r] = val
        else:
            dp = -d
            val = 0.0
            if j >= dp:  # j-dp >= 0
                val += (1.0 - h[j - dp])
            if j + dp < n:  # j <= n-1-dp
                val -= h[j + dp]
            a[r] = val
    return a
```

And the constant part b_d = score_d - h[j] * a_d.

Then to minimize max_d (b_d + a_d * t) over t in [0, 1]:
- This is a convex piecewise-linear function
- Use bisection on the subgradient

```python
def _coord_descent_step(h, j):
    """Optimize h[j] with others fixed. Returns new value of h[j]."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    a = _coord_slopes(h, j)
    b = scores - h[j] * a  # constant part
    
    # Minimize max_d (b[d] + a[d]*t) over t in [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = b + a * mid
        max_val = np.max(vals)
        active = vals >= max_val - 1e-14
        a_active = a[active]
        if np.min(a_active) > 1e-12:
            hi = mid
        elif np.max(a_active) < -1e-12:
            lo = mid
        else:
            break
    t_opt = (lo + hi) / 2.0
    return t_opt
```

This is O(n) per bisection step, 40 steps, so O(40n) per coordinate. For a full pass over n coordinates: O(40n²). For n=156: about 970,000 operations per pass. Very fast!

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _linesearch(h, d, dx, radius, n_samp=50):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_max = radius
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    if np.any(pos_mask):
        t_max = min(t_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        t_max = min(t_max, float(np.min(h[neg_mask] / (-d[neg_mask]))))
    
    if t_max < 1e-14:
        return 0.0, float(np.max(S)) * dx
    
    ts = np.linspace(0.0, t_max, n_samp)
    vals = S[None, :] + ts[:, None] * T[None, :] - (ts * ts)[:, None] * C[None, :]
    max_vals = np.max(vals, axis=1) * dx
    best_idx = int(np.argmin(max_vals))
    
    # Golden section refinement
    a = ts[max(0, best_idx - 1)]
    b = ts[min(n_samp - 1, best_idx + 1)]
    candidates = [0.0, t_max, ts[best_idx]]
    
    if b - a > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(S + c1 * T - c1*c1 * C)) * dx
        f2 = float(np.max(S + c2 * T - c2*c2 * C)) * dx
        for _ in range(30):
            if b - a < 1e-12:
                break
            if f1 < f2:
                b = c2; c2 = c1; f2 = f1
                c1 = b - gr * (b - a)
                f1 = float(np.max(S + c1 * T - c1*c1 * C)) * dx
            else:
                a = c1; c1 = c2; f1 = f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(S + c2 * T - c2*c2 * C)) * dx
        candidates.append((a + b) / 2.0)
    
    best_t = 0.0
    best_v = float(np.max(S)) * dx
    for t in candidates:
        t = max(0.0, min(t, t_max))
        v = float(np.max(S + t * T - t * t * C)) * dx
        if v < best_v:
            best_v = v
            best_t = t
    return best_t, best_v

def _fw_step(h, deadline, n_iters=200, radius=0.3):
    """Frank-Wolfe (conditional gradient) optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    dx = 2.0 / n
    stall = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Compute gradient of active lag
        scores = np.correlate(h, 1.0 - h, mode='full')
        # Find active lag(s) - use top few for robustness
        top_k = min(3, len(scores))
        top_idx = np.argsort(scores)[::-1][:top_k]
        # Average gradient of top lags
        grad = np.zeros(n)
        for r in top_idx:
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:d+m])
                    grad[d:] += -h[:m]
            else:
                dp = -d
                m = n - dp
                if m > 0:
                    grad[dp:] += (1.0 - h[:m])
                    grad[:m] += -h[dp:dp+m]
        grad /= top_k
        # Remove mean for sum-zero
        grad -= np.mean(grad)
        
        # Frank-Wolfe: find vertex v* minimizing grad^T v
        v_star = np.zeros(n)
        idx = np.argsort(grad)[:int(target)]
        v_star[idx] = 1.0
        direction = v_star - h
        
        if np.max(np.abs(direction)) < 1e-10:
            stall += 1
        else:
            alpha, new_c5 = _linesearch(h, direction, dx, radius)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * direction
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        
        if stall > 30:
            break
    
    return best_h, best_c5

def _coord_descent(h, n_passes=5, deadline=None):
    """Coordinate descent: optimize each coordinate in turn."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for _pass in range(n_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # Compute new h[j]
            scores = np.correlate(h, 1.0 - h, mode='full')
            # Compute slopes a_d for coordinate j
            a = np.zeros(2 * n - 1)
            for r in range(2 * n - 1):
                d = r - (n - 1)
                if d >= 0:
                    val = 0.0
                    if j + d < n:
                        val += (1.0 - h[j + d])
                    if j >= d:
                        val -= h[j - d]
                    a[r] = val
                else:
                    dp = -d
                    val = 0.0
                    if j >= dp:
                        val += (1.0 - h[j - dp])
                    if j + dp < n:
                        val -= h[j + dp]
                    a[r] = val
            b = scores - h[j] * a
            
            # Bisection on subgradient
            lo, hi = 0.0, 1.0
            for _ in range(35):
                mid = (lo + hi) / 2.0
                vals = b + a * mid
                max_val = np.max(vals)
                active = vals >= max_val - 1e-14
                a_act = a[active]
                if np.min(a_act) > 1e-12:
                    hi = mid
                elif np.max(a_act) < -1e-12:
                    lo = mid
                else:
                    break
            t_opt = (lo + hi) / 2.0
            
            if abs(t_opt - h[j]) > 1e-10:
                h[j] = t_opt
                improved = True
        
        # Project to maintain feasibility (coordinate descent should maintain it,
        # but numerical errors can accumulate)
        h = _project(h, target)
        new_c5 = _exact_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c5
