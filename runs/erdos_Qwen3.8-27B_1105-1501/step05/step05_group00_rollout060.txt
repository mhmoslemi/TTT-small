def _swap_delta(h, i, j):
    """Compute the change in correlation when swapping h[i]:1->0 and h[j]:0->1.
    Returns array of length 2n-1 with the changes."""
    n = len(h)
    delta = np.zeros(2*n - 1)
    # For each lag q (from -(n-1) to n-1), index in corr is q + n - 1
    # q = -(n-1), ..., 0, ..., n-1
    # corr index = q + n - 1, so 0 to 2n-2
    
    # Position i: was 1, becomes 0
    # As first factor (h[i] * (1-h[i+q])): index i+q must be in [0,n-1]
    # q range: -i to n-1-i
    q_lo_i = -i
    q_hi_i = n - 1 - i
    for q in range(q_lo_i, q_hi_i + 1):
        delta[q + n - 1] += -(1 - h[i + q])
    
    # As second factor (h[i-q] * (1-h[i])): index i-q must be in [0,n-1]
    # q range: i-(n-1) to i
    q_lo_i2 = i - (n-1)
    q_hi_i2 = i
    for q in range(q_lo_i2, q_hi_i2 + 1):
        delta[q + n - 1] += h[i - q]
    
    # Position j: was 0, becomes 1
    # As first factor: index j+q must be in [0,n-1]
    q_lo_j = -j
    q_hi_j = n - 1 - j
    for q in range(q_lo_j, q_hi_j + 1):
        delta[q + n - 1] += (1 - h[j + q])

    q_hi_j2 = j

def _compute_swap_delta(h, i, j):
    n = len(h)
    # Effect of removing 1 at i (h[i]: 1->0)
    # h[i] as first factor: affects lags q where 0 <= i+q < n, i.e., -i <= q <= n-1-i
    # change: -(1-h[i+q]) for q in [-i, n-1-i]
    # h[i] as second factor: affects lags q where 0 <= i-q < n, i.e., i-(n-1) <= q <= i
    # change: +h[i-q] for q in [i-(n-1), i]
    
    # Effect of adding 1 at j (h[j]: 0->1)
    # h[j] as first factor: change: +(1-h[j+q]) for q in [-j, n-1-j]
    # h[j] as second factor: change: -h[j-q] for q in [j-(n-1), j]
    
    delta_full = np.zeros(2*n-1)  # indexed by q + n - 1
    
    # Remove 1 at i, first factor
    q_start = max(-i, -(n-1))
    q_end = min(n-1-i, n-1)
    indices = q_start + n - 1  # start index in delta_full
    length = q_end - q_start + 1
    delta_full[indices:indices+length] -= (1 - h[i + q_start : i + q_end + 1])
    
    # Remove 1 at i, second factor
    q_start2 = max(i-(n-1), -(n-1))
    q_end2 = min(i, n-1)
    indices2 = q_start2 + n - 1
    length2 = q_end2 - q_start2 + 1
    delta_full[indices2:indices2+length2] += h[i - q_start2 : i - q_end2 + 1][::-1]
    
    # Add 1 at j, first factor
    q_start3 = max(-j, -(n-1))
    q_end3 = min(n-1-j, n-1)
    indices3 = q_start3 + n - 1
    length3 = q_end3 - q_start3 + 1
    delta_full[indices3:indices3+length3] += (1 - h[j + q_start3 : j + q_end3 + 1])
    
    # Add 1 at j, second factor
    q_start4 = max(j-(n-1), -(n-1))
    q_end4 = min(j, n-1)
    indices4 = q_start4 + n - 1
    length4 = q_end4 - q_start4 + 1
    delta_full[indices4:indices4+length4] -= h[j - q_start4 : j - q_end4 + 1][::-1]
    
    return delta_full
```

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

def _line_search(h, d, dx, trust_radius):
    """Exact line search along direction d from h.
    Returns (best_alpha, best_value)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible alpha range
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = trust_radius
    t_lo = -trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Grid search
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    
    return best_t, best_val

    n = len(h)
    G = np.zeros((2*n-1, n))
    one_minus_h = 1.0 - h
    for q in range(-(n-1), n):
        r = n - 1 + q
        # First term: (1-h)[j+q] for j where 0 <= j+q < n
        # j range: max(0, -q) to min(n-1, n-1-q)
        j_lo = max(0, -q)
        j_hi = min(n-1, n-1-q)
        if j_lo <= j_hi:
            G[r, j_lo:j_hi+1] += one_minus_h[j_lo+q : j_hi+q+1]
        # Second term: -h[j-q] for j where 0 <= j-q < n
        # j range: max(0, q) to min(n-1, n-1+q)
        j_lo2 = max(0, q)
        j_hi2 = min(n-1, n-1+q)
        if j_lo2 <= j_hi2:
            G[r, j_lo2:j_hi2+1] -= h[j_lo2-q : j_hi2-q+1]
    return G
```
