def _exact_c5(h):

def _scores(h):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

def _grad_at_lag(h, d):
    """Gradient of f_d(h) = sum_{valid i} h_i * (1-h_{i+d}) with respect to h_i."""
    n = len(h)

    if d >= 0:

        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
    else:
        g2 = -d
        m = n - g2
        if m > 0:
            g[:m] += (1.0 - h[g2:])
            g[g2:] -= h[:m]
    return g

def _subgradient(h, top_m=10):
    """Compute subgradient of C5 at h using top-m active lags."""
    n = len(h)
    scores = _scores(h)
    # Get top-m lags by score
    idx = np.argsort(scores)[-top_m:]
    # Weight by how close to max
    max_s = scores.max()
    min_top = scores[idx].min()
    if max_s - min_top < 1e-10:
        weights = np.ones(len(idx)) / len(idx)
    else:
        weights = np.exp((scores[idx] - max_s) / 0.01)
        weights /= weights.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for i, lag_idx in enumerate(idx):
        d = lag_idx - (n - 1)
        g += weights[i] * _grad_at_lag(h, d)
    return g * dx, idx

        g = _grad_at_lag(h, d)

def _quasi_periodic_init(n, alpha, threshold=0.5):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        return np.ones(n)
    lo = float(v.min()) - 2.0

        if s > target:

        else:
            hi = mid

            best_c5 = c5_2
            best_h = h2.copy()
        if remaining_frac() > 0.10:
            h2b, c5_2b = _lp_active_opt(best_h, dl_frac(0.30), 5, 0.12, 6000, 0.001)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Multi-resolution with quasi-periodic + resampled starts
    if remaining_frac() > 0.22:
        resolutions = [64, 80, 96, 112, n0, 144, 160, 192]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5,
                  0.6180339887, 0.3819660113, 0.7071067812]
        for ri in range(len(resolutions)):
            if time.time() > dl_frac(0.52):
                break
            n_try = resolutions[ri]
            if n_try < 8 or n_try % 2 != 0:
                continue
            for s in range(2):
                if time.time() > dl_frac(0.52):
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    h_init = _quasi_init(n_try, alphas[ri % len(alphas)])
                sub_dl = min(dl_frac(0.52), time.time() + T * 0.025)
                h_try, c5_try = _lp_active_opt(h_init, sub_dl, 6, 0.15, 3000, 0.003)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 4: Full LP polish (all constraints, more conservative)
    if remaining_frac() > 0.12:
        h4, c5_4 = _full_lp_opt(best_h, dl_frac(0.62), 0.06, 3000, 0.001)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Alternating subgrad + active LP
    if remaining_frac() > 0.12:
        for rnd in range(4):
            if time.time() > dl_frac(0.74):
                break
            h5a, c5_5a = _subgrad_opt(best_h, min(dl_frac(0.74), time.time() + T * 0.025),
                                      3, 0.025, 80)
            if c5_5a < best_c5:
                best_c5 = c5_5a
                best_h = h5a.copy()
            if time.time() > dl_frac(0.74):
                break
            h5b, c5_5b = _lp_active_opt(best_h, min(dl_frac(0.74), time.time() + T * 0.02),
                                        4, 0.08, 2000, 0.002)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: Perturbation restarts
    if remaining_frac() > 0.10:
        h6, c5_6 = _perturb_restart(best_h, dl_frac(0.86), rng, 100, 0.025, T * 0.01)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Pair polish
    if remaining_frac() > 0.07:
        h7, c5_7 = _pair_polish(best_h, dl_frac(0.94), n_passes=3)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final ultra-fine LP
    if remaining_frac() > 0.02:
        h_fin, c5_fin = _lp_active_opt(best_h, dl_frac(0.995), 3, 0.015, 5000, 0.0003)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
