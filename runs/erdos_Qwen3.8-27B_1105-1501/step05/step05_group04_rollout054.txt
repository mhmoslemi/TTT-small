from scipy.optimize import minimize, linprog

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        else:
            mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def lag_grad_row(h, r):
    """Gradient of lag r (0-indexed in full correlation) with respect to h.
    r corresponds to shift s = r - (n-1)."""
    n = len(h)
    dx = 2.0 / n
    s = r - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        for i in range(m):
            g[i + s] += dx * (1.0 - h[i])
            g[i] += -dx * h[i + s]
    else:
        a = -s
        m = n - a
        for i in range(m):
            g[i] += dx * (1.0 - h[i + a])
            g[i + a] += -dx * h[i]
    return g

def lag_jacobian(h):
    n = len(h)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for r in range(n_lags):
        J[r] = lag_grad_row(h, r)
    return J

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(F))
    Z = np.exp((F - m) / beta)
    S = float(np.sum(Z))
    P = Z / S
    val = m + beta * np.log(S)
    # gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if P[r] > 1e-15:
            grad += P[r] * lag_grad_row(h, r)
    return val, grad

    n = len(h)
    target = n / 2.0
    h0 = h.copy()
    
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        val, grad = smooth_obj_and_grad(x, beta)
        return val, grad
    
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-8})
    
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_box_sum(h_new, target)
    return h_new, eval_c5(h_new)

def slsqp_epigraph(h, deadline, maxiter=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    n_lags = 2 * n - 1
    c0 = eval_c5(h)
    t0 = c0 + 0.01
    
    x0 = np.concatenate([h, [t0]])
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def cons_fun(x):
        h_part = x[:n]
        t = x[-1]
        F = np.correlate(h_part, 1.0 - h_part, mode='full') * dx
        return F - t
    
    def cons_jac(x):
        h_part = x[:n]
        J = lag_jacobian(h_part) * dx
        Jc = np.zeros((n_lags, n + 1))
        Jc[:, :n] = J
        Jc[:, n] = -1.0
        return Jc
    
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target,
         'jac': lambda x: np.concatenate([np.ones(n), [0.0]])},
        {'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}
    ]
    
    bounds = [(0.0, 1.0)] * n
    bounds.append((0.0, 1.0))
    
    res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9})
    
    if res.success or res.x is not None:
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_box_sum(h_new, target)
        c5_new = eval_c5(h_new)
        if c5_new < c0:
            return h_new, c5_new
    
    return h, c0

def _line_search(h, d, dx, trust_radius):

    neg = d < -1e-15
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    n_grid = 120
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    # Golden section refine
    if best_i > 0 and best_i < n_grid - 1:
        a_g = grid[best_i - 1]
        b_g = grid[best_i + 1]
    else:
        a_g = t_lo
        b_g = t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b_g - gr * (b_g - a_g)
    c2 = a_g + gr * (b_g - a_g)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds.append((None, None))

    return None

                continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 5
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.78
            if stall % 30 == 0 and radius < 0.03:
                radius = 0.08
                n_active = 10
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)

def _multires(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    sizes = []
    for ratio in [0.6, 0.7, 0.8, 0.85, 1.0, 1.1, 1.2, 1.3, 1.5]:
        sz = int(round(n * ratio))
        if sz >= 24 and sz <= 200 and sz not in sizes:
            sizes.append(sz)
    rng.shuffle(sizes)
    for sz in sizes:
        if time.time() > deadline:
            break
        h_r = resample(best_h, sz)
        sub_dl = min(deadline, time.time() + 8.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def _perturb_lp(h, deadline, rng, base_scale=0.05):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    for _ in range(200):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        sub_dl = min(deadline, time.time() + 6.0)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.08)
    return best_h, best_c5

def _coord_refine(h, deadline, max_iter=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(999)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.3))
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def _perturb_smooth(h, deadline, rng, base_scale=0.06):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    c0 = best_c5
    scale = base_scale
    for _ in range(100):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        sub_dl = min(deadline, time.time() + 10.0)
        h_t, c5_t = slsqp_smooth_phase(h_pert, sub_dl, max_beta_ratio=0.3)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.06)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: LP polish (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothing SLSQP (22%)
    d2 = t0 + budget_s * 0.34
    if time.time() < d2:
        h2, c5_2 = slsqp_smooth_phase(best_h, d2, max_beta_ratio=0.4)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Epigraph SLSQP (12%)
    d3 = t0 + budget_s * 0.46
    if time.time() < d3:
        h3, c5_3 = slsqp_epigraph_phase(best_h, d3, topk=10)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-resolution exploration (18%)
    d4 = t0 + budget_s * 0.64
    if time.time() < d4:
        best_h, best_c5 = _multires(best_h, d4, rng)

    # Phase 5: Perturb + LP (12%)
    d5 = t0 + budget_s * 0.76
    if time.time() < d5:
        best_h, best_c5 = _perturb_lp(best_h, d5, rng, base_scale=0.04)

    # Phase 6: Perturb + Smoothing (10%)
    d6 = t0 + budget_s * 0.86
    if time.time() < d6:
        best_h, best_c5 = _perturb_smooth(best_h, d6, rng, base_scale=0.05)

    # Phase 7: Final LP polish (7%)
    d7 = t0 + budget_s * 0.93
    if time.time() < d7:
        h7, c5_7 = _lp_loop(best_h, d7, radius_init=0.06)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Coordinate refinement (5%)
    d8 = t0 + budget_s * 0.98
    if time.time() < d8:
        best_h, best_c5 = _coord_refine(best_h, d8, max_iter=200)

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
