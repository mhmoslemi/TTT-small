import numpy as np
from scipy.optimize import linprog, minimize
import time

R_WALL = 14.0

def _safe_sigma(t):
    # Stable sigmoid
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def _solve_mu(q, target):
    # Find mu such that sum sigma(q_i - mu) = target
    # sum sigma is strictly decreasing in mu
    lo = -R_WALL
    hi = R_WALL
    # Check if target is in range
    s_lo = float(np.sum(_safe_sigma(q - lo)))
    s_hi = float(np.sum(_safe_sigma(q - hi)))
    if target >= s_lo:
        return lo
    if target <= s_hi:
        return hi
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_safe_sigma(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _map_q_to_h(q, target):
    mu = _solve_mu(q, target)
    h = _safe_sigma(q - mu)
    return h

def _balance(x, target):
    # Project x to [0,1]^n with sum = target
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    n = len(x)
    # Bisection on threshold
    lo = -1.0
    hi = 1.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    result = np.clip(x - thresh, 0.0, 1.0)
    # Fix residual
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-14:
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            result[mask] += residual / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def _compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _exact_c5(h):
    return float(np.max(_compute_scores(h)))

def _build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += (-h[:m])
    return G * dx

def _smooth_obj_grad(q, beta, target):
    h = _map_q_to_h(q, target)
    scores = _compute_scores(h)
    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    obj = max_s + np.log(float(np.mean(np.exp(shifted)))) / beta
    
    G = _build_grad_matrix(h)
    g_h = G.T @ w  # gradient wrt h
    
    # Map gradient to q
    mu = _solve_mu(q, target)
    sigma_val = _safe_sigma(q - mu)
    sigma_prime = sigma_val * (1.0 - sigma_val)
    M = float(np.sum(sigma_prime))
    if M < 1e-30:
        return obj, np.zeros_like(q)
    rho = float(np.sum(sigma_prime * g_h)) / M
    grad_q = sigma_prime * (g_h - rho)
    
    if not np.all(np.isfinite(grad_q)):
        grad_q = np.zeros_like(q)
    return obj, grad_q

def _logit_init(h, noise=0.0):
    eps = 0.01
    h_clip = np.clip(h, eps, 1.0 - eps)
    q = np.log(h_clip / (1.0 - h_clip))
    if noise > 0:
        q += noise * np.random.standard_normal(len(q))  # This won't work without rng
    return q

def _logit_init_rng(h, rng, noise=0.0):
    eps = 0.01
    h_clip = np.clip(h, eps, 1.0 - eps)
    q = np.log(h_clip / (1.0 - h_clip))
    if noise > 0:
        q = q + noise * rng.standard_normal(len(q))
    return q

def _lbfgs_manifold(h_send, deadline, rng):
    # Optimize on the feasible manifold using L-BFGS-B
    n = len(h_send)
    target = n / 2.0
    best_h = h_send.copy()
    best_c5 = _exact_c5(best_h)
    
    # Initialize q
    q = _logit_init_rng(best_h, rng, noise=0.0)
    
    # Beta schedule
    L = 2 * n - 1
    beta_vals = np.logspace(np.log10(max(15.0, n * 0.15)),
                            np.log10(min(8000.0, n * 50.0)), 8)
    
    for beta in beta_vals:
        if time.time() > deadline:
            break
        
        def obj_func(q_arr):
            return _smooth_obj_grad(q_arr, beta, target)[0]
        
        def grad_func(q_arr):
            return _smooth_obj_grad(q_arr, beta, target)[1]
        
        maxiter = max(50, int(200 * np.sqrt(128.0 / n)))
        
        try:
            res = minimize(obj_func, q, jac=grad_func, method='L-BFGS-B',
                          bounds=[(-R_WALL, R_WALL)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxfun': maxiter * 3})
            if np.all(np.isfinite(res.x)):
                h_new = _map_q_to_h(res.x, target)
                h_new = _balance(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                q = res.x.copy()
        except Exception:
            pass
    
    # Final exact line search from q back to h
    return best_h, best_c5

def _complement_lp(h_current, deadline):
    # Solve LP to find complementary profile
    n = len(h_current)
    target = n / 2.0
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Build coefficient matrix B where B[r, j] = sum of h_current[i] for pairs (i, i+d) where i+d = j
    # Score_d = sum_i h[i] * y[i+d] where y = 1 - h_new

    # Coefficient of y[j] in score_d: h[j-d] if j-d >= 0 and j-d < n and j+d' valid...
    # Actually: score_d = sum_{i: 0<=i<n, 0<=i+d<n} h[i] * y[i+d]
    # Let j = i+d, so i = j-d. Coefficient of y[j] = h[j-d] if 0 <= j-d < n and 0 <= j < n
    # And we also need 0 <= i+d < n which is 0 <= j < n (already have)
    # And 0 <= i = j-d < n means d <= j < n+d
    
    B = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        for j in range(n):
            i = j - d
            if 0 <= i < n:
                B[r, j] += h_current[i]
    
    # LP: min tau
    # s.t. B @ y <= tau  (for each lag)
    #      sum(y) = target
    #      0 <= y <= 1
    #      tau free (or >= 0)
    
    # Variables: y[0..n-1], tau
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = B
    A_ub[:, n] = -1.0  # B@y - tau <= 0
    b_ub = np.zeros(n_lags)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(-1.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            y = res.x[:n]

            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-14:
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            result[mask] += residual / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def _compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _exact_c5(h):
    return float(np.max(_compute_scores(h)))

def _build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += (-h[:m])
    return G * dx

def _smooth_obj_grad(q, beta, target):
    h = _map_q_to_h(q, target)
    scores = _compute_scores(h)
    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    obj = max_s + float(np.log(float(np.mean(np.exp(shifted)))) / beta)
    G = _build_grad_matrix(h)
    g_h = G.T @ w
    mu = _solve_mu(q, target)
    sv = _safe_sigma(q - mu)
    sp = sv * (1.0 - sv)
    M = float(np.sum(sp))
    if M < 1e-30:
        return obj, np.zeros_like(q)
    rho = float(np.dot(sp, g_h)) / M
    grad_q = sp * (g_h - rho)
    if not np.all(np.isfinite(grad_q)):
        grad_q = np.zeros_like(q)
    return obj, grad_q

def _logit_init(h, rng, noise=0.0):
    eps = 0.01
    hc = np.clip(h, eps, 1.0 - eps)
    q = np.log(hc / (1.0 - hc))
    if noise > 0:
        q = q + noise * rng.standard_normal(len(q))
    return np.clip(q, -R_WALL, R_WALL)

def _lbfgs_manifold(h_seed, deadline, rng):
    n = len(h_seed)
    target = n / 2.0
    best_h = h_seed.copy()
    best_c5 = _exact_c5(best_h)
    q = _logit_init(best_h, rng, noise=0.0)
    beta_start = max(12.0, n * 0.12)
    beta_end = min(6000.0, n * 45.0)
    beta_vals = np.logspace(np.log10(beta_start), np.log10(beta_end), 7)
    maxiter = max(60, int(180 * np.sqrt(128.0 / n)))
    for beta in beta_vals:
        if time.time() > deadline:
            break
        def obj_f(qa):
            return _smooth_obj_grad(qa, beta, target)[0]
        def grad_f(qa):
            return _smooth_obj_grad(qa, beta, target)[1]
        try:
            res = minimize(obj_f, q, jac=grad_f, method='L-BFGS-B',
                          bounds=[(-R_WALL, R_WALL)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
            if np.all(np.isfinite(res.x)):
                h_new = _map_q_to_h(res.x, target)
                h_new = _balance(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                q = np.clip(res.x, -R_WALL, R_WALL)
        except Exception:
            pass
    return best_h, best_c5

def _complement_lp(h_cur, deadline):
    n = len(h_cur)
    target = n / 2.0
    n_lags = 2 * n - 1
    B = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        for j in range(n):
            i = j - d
            if 0 <= i < n:
                B[r, j] += h_cur[i]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = B
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(-1.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            y = res.x[:n]
            h_new = _balance(1.0 - y, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return None, None

def _lin_search_blend(h1, h2, target):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    f0 = _exact_c5(h1)
    f1v = _exact_c5(h2)
    if f1v >= f0:
        return h1.copy(), f0
    for _ in range(22):
        if hi - lo < 1e-7:
            break
        m1 = lo + (hi - lo) * gr
        m2 = hi - (hi - lo) * gr
        hm1 = _balance((1 - m1) * h1 + m1 * h2, target)
        hm2 = _balance((1 - m2) * h1 + m2 * h2, target)
        fm1 = _exact_c5(hm1)
        fm2 = _exact_c5(hm2)
        if fm1 < fm2:
            hi = m2
        else:
            lo = m1
    m_opt = (lo + hi) / 2.0
    h_opt = _balance((1 - m_opt) * h1 + m_opt * h2, target)
    return h_opt, _exact_c5(h_opt)

def _lp_local(h, deadline, radius_init=0.2, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    stall = 0
    for _it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _compute_scores(h)
        G = _build_grad_matrix(h)
        n_lags = len(scores)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
        bounds.append((None, None))
        p = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.999
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(_balance(h + c1 * p, target))
        f2 = _exact_c5(_balance(h + c2 * p, target))
        for _ in range(22):
            if hi_a - lo_a < 1e-14:
                break
            if f1 < f2:
                hi_a = c2; f2 = f1; c2 = c1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(_balance(h + c1 * p, target))
            else:
                lo_a = c1; f1 = f2; c1 = c2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(_balance(h + c2 * p, target))
        opt_a = (lo_a + hi_a) / 2.0
        h_new = _balance(h + opt_a * p, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
            radius = min(radius * 1.12, 0.5)
        else:
            stall += 1
            radius *= 0.82
            if stall > 35 and radius < 0.02:
                radius = 0.1
    return best_h, best_c5

def _coord_descent(h, deadline, n_sweeps=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if h[i] < 1e-10 or h[i] > 1.0 - 1e-10:
                continue
            scores = _compute_scores(h)
            # Compute coefficient of h[i] in each score
            coeff = np.zeros(2 * n - 1)
            for d in range(-(n - 1), n):
                r = n - 1 + d
                c_val = 0.0
                if d >= 0:
                    if i + d < n:
                        c_val += (1.0 - h[i + d])
                    if i >= d:
                        c_val += -h[i - d]
                else:
                    g = -d
                    if i >= g:
                        c_val += (1.0 - h[i - g])
                    if i + g < n:
                        c_val += -h[i + g]
                coeff[r] = c_val * dx
            # f(delta) = max_d (scores[d] + delta * coeff[d])
            # delta in [-h[i], 1-h[i]]
            d_lo = -h[i]
            d_hi = 1.0 - h[i]
            # Sample to find best delta
            n_samp = 25
            deltas = np.linspace(d_lo, d_hi, n_samp)
            best_delta = 0.0
            best_val = _exact_c5(h)
            for delta in deltas:
                if abs(delta) < 1e-15:
                    continue
                new_scores = scores + delta * coeff
                val = float(np.max(new_scores))
                if val < best_val:
                    best_val = val
                    best_delta = delta
            if abs(best_delta) > 1e-12:
                # Refine with golden section
                gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
                # Find bracket around best_delta
                idx = int(np.argmin([_exact_c5(_balance(h + d * np.eye(n)[i], target)) if False else 0 for d in [0]]))
                # Just use the linear approx: refine around best_delta
                lo_d = best_delta - (d_hi - d_lo) / n_samp
                hi_d = best_delta + (d_hi - d_lo) / n_samp
                lo_d = max(lo_d, d_lo)
                hi_d = min(hi_d, d_hi)
                c1 = hi_d - gr_phi * (hi_d - lo_d)
                c2 = lo_d + gr_phi * (hi_d - lo_d)
                # Use linear score for speed
                def score_at(delta_d):
                    return float(np.max(scores + delta_d * coeff))
                f1 = score_at(c1)
                f2 = score_at(c2)
                for _ in range(15):
                    if hi_d - lo_d < 1e-12:
                        break
                    if f1 < f2:
                        hi_d = c2; f2 = f1; c2 = c1
                        c1 = hi_d - gr_phi * (hi_d - lo_d)
                        f1 = score_at(c1)
                    else:
                        lo_d = c1; f1 = f2; c1 = c2
                        c2 = lo_d + gr_phi * (hi_d - lo_d)
                        f2 = score_at(c2)
                best_delta = (lo_d + hi_d) / 2.0
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                # Fix sum by adjusting another coordinate
                residual = float(np.sum(h)) - target
                if abs(residual) > 1e-14:
                    # Spread correction
                    h -= residual / n
                    h = np.clip(h, 0.0, 1.0)
                    h = _balance(h, target)
                c5_new = _exact_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
        if not improved and sweep > 5:
            break
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 24)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        ht1 = h.copy(); ht1[j] += sign * m1; ht1[k] -= sign * m1
                        ht1 = np.clip(ht1, 0.0, 1.0)
                        f1 = _exact_c5(ht1)
                        ht2 = h.copy(); ht2[j] += sign * m2; ht2[k] -= sign * m2
                        ht2 = np.clip(ht2, 0.0, 1.0)
                        f2 = _exact_c5(ht2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    ht = h.copy()
                    ht[j] += sign * a_opt
                    ht[k] -= sign * a_opt
                    ht = np.clip(ht, 0.0, 1.0)
                    c5_try = _exact_c5(ht)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = ht.copy()
                        h = ht.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _balance(h_new, n_new / 2.0)

def _slsqp_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _slsqp_eq_jac(x):
    return np.ones(len(x))

def _slsqp_phase(h, deadline, beta_val=200.0, maxiter=150):
    n = len(h)
    target = n / 2.0
    def obj_f(hf):
        S = _compute_scores(hf)
        maxS = np.max(S)
        shifted = beta_val * (S - maxS)
        shifted = np.clip(shifted, -500, 0)
        return maxS + float(np.log(float(np.mean(np.exp(shifted)))) / beta_val)
    def grad_f(hf):
        S = _compute_scores(hf)
        maxS = np.max(S)
        shifted = beta_val * (S - maxS)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w /= sw
        G = _build_grad_matrix(hf)
        return G.T @ w
    try:
        res = minimize(obj_f, h, jac=grad_f, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _slsqp_eq_fun,
                                    'jac': _slsqp_eq_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success and np.all(np.isfinite(res.x)):
            h_new = _balance(res.x, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h, _exact_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _balance(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    # Phase 1: L-BFGS-B on feasible manifold
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _lbfgs_manifold(best_h, dl1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1; best_h = h1.copy()
    # Phase 2: LP local
    dl2 = t0 + budget_s * 0.33
    h2, c5_2 = _lp_local(best_h, dl2, radius_init=0.25, max_iters=5000)
    if c5_2 < best_c5:
        best_c5 = c5_2; best_h = h2.copy()
    # Phase 3: L-BFGS with perturbations
    if time.time() < t0 + budget_s * 0.48:
        dl3 = t0 + budget_s * 0.48
        for _att in range(6):
            if time.time() > dl3:
                break
            h_p = _balance(best_h + rng.normal(0, 0.025, best_n), best_n / 2.0)
            h3, c5_3 = _lbfgs_manifold(h_p, min(dl3, time.time() + budget_s * 0.035), rng)
            if c5_3 < best_c5:
                best_c5 = c5_3; best_h = h3.copy()
    # Phase 4: Coordinate descent
    if time.time() < t0 + budget_s * 0.57:
        dl4 = t0 + budget_s * 0.57
        h4, c5_4 = _coord_descent(best_h, dl4, n_sweeps=40)
        if c5_4 < best_c5:
            best_c5 = c5_4; best_h = h4.copy()
    # Phase 5: Complement LP
    if time.time() < t0 + budget_s * 0.70:
        dl5 = t0 + budget_s * 0.70
        for _it in range(12):
            if time.time() > dl5:
                break
            h_comp, c5_comp = _complement_lp(best_h, dl5)
            if h_comp is not None:
                h_line, c5_line = _lin_search_blend(best_h, h_comp, best_n / 2.0)
                if c5_line < best_c5:
                    best_c5 = c5_line; best_h = h_line.copy()
                if c5_comp < best_c5:
                    best_c5 = c5_comp; best_h = h_comp.copy()
                # Polish
                h_pol, c5_pol = _lp_local(best_h, min(dl5, time.time() + budget_s * 0.015),
                                          radius_init=0.12, max_iters=800)
                if c5_pol < best_c5:
                    best_c5 = c5_pol; best_h = h_pol.copy()
    # Phase 6: Multi-resolution
    if time.time() < t0 + budget_s * 0.82:
        dl6 = t0 + budget_s * 0.82
        for n_try in [160, 192]:
            if time.time() > dl6:
                break
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl6, time.time() + budget_s * 0.05)
            ht, ct = _lbfgs_manifold(h_init, sub_dl, rng)
            ht2, ct2 = _lp_local(ht, min(dl6, time.time() + budget_s * 0.03),
                                 radius_init=0.2, max_iters=2000)
            if ct2 < best_c5:
                best_c5 = ct2; best_h = ht2.copy(); best_n = n_try
    # Phase 7: Perturbation restarts
    if time.time() < t0 + budget_s * 0.90:
        dl7 = t0 + budget_s * 0.90
        scale = 0.012
        for _att in range(400):
            if time.time() > dl7:
                break
            h_p = _balance(best_h + rng.normal(0, scale, best_n), best_n / 2.0)
            sub_dl = min(dl7, time.time() + budget_s * 0.008)
            ht, ct = _lp_local(h_p, sub_dl, radius_init=0.04, max_iters=400)
            if ct < best_c5 - 1e-13:
                best_c5 = ct; best_h = ht.copy()
                scale = max(0.002, scale * 0.82)
            else:
                scale = min(0.05, scale * 1.08)
    # Phase 8: SLSQP final
    if time.time() < t0 + budget_s * 0.95:
        h8, c5_8 = _slsqp_phase(best_h, t0 + budget_s * 0.95, beta_val=300.0, maxiter=120)
        if c5_8 < best_c5:
            best_c5 = c5_8; best_h = h8.copy()
    # Phase 9: Final LP polish
    if time.time() < t0 + budget_s * 0.97:
        h9, c5_9 = _lp_local(best_h, t0 + budget_s * 0.97, radius_init=0.04, max_iters=2000)
        if c5_9 < best_c5:
            best_c5 = c5_9; best_h = h9.copy()
    # Phase 10: Pair polish
    if time.time() < t0 + budget_s * 0.985:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.99, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()
    # Final
    best_h = _balance(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
