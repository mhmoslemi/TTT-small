def _smoothed_grad_cont(h, deadline, beta_start=10, beta_end=300, n_steps=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = 0.9
    step = 0.03
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        scores = _scores(h)
        max_s = scores.max()
        # Softmax weights
        w = np.exp(beta * (scores - max_s))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= w_sum
        
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()  # keep sum-zero
        
        # Normalize
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        # Momentum step
        v = momentum * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.01, 0.08)
        else:
            v *= 0.5
            step *= 0.8
            if step < 1e-6:

            # Try without momentum

            c5_try = _exact_c5(h_try)

            continue

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

        else:

        cnt = int(np.sum(mask))

        x = np.clip(x, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_lags(h, lag_indices):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lag_indices), n))
    omh = 1.0 - h
    for idx, r in enumerate(lag_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] += omh[d:]
                G[idx, d:n] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += -h[g:]
                G[idx, g:n] += omh[:m]
    return G * dx

def _lp_local(h, deadline, K=10, radius_init=0.20, radius_min=0.005, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for _it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        K_eff = min(len(scores), K)
        top_idx = np.argsort(scores)[-K_eff:]
        G = _grad_lags(h, top_idx)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((K_eff, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -(scores[top_idx] - G @ h)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = []
        for j in range(n):
            lo_b = max(0.0, h[j] - radius)
            hi_b = min(1.0, h[j] + radius)
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5.0})
            if res.success and res.x is not None:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-12:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.06, 0.45)
                else:
                    stall += 1
                    radius *= 0.65
                    if radius < radius_min:
                        break
            else:
                stall += 1
                radius *= 0.6
                if radius < radius_min:
                    break
        except Exception:
            radius *= 0.5
            if radius < radius_min:
                break
    return best_h, best_c5

def _smoothed_grad(h, deadline, beta_start=15, beta_end=400, n_steps=400, step_init=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    momentum = 0.85
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = float(np.max(scores))
        w = np.exp(beta * (scores - max_s))
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_lags(h, np.arange(len(scores)))
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = momentum * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.02, 0.10)
        else:
            h_try = _project(h - step * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                v = -step * grad
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                v *= 0.5
                step *= 0.75
                if step < 1e-7:
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _qperiodic_init(n, alpha):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    h = (vals < 0.5).astype(float)
    return _project(h, n / 2.0)

def _pair_transfer_sa(h, deadline, rng, n_iter=50000, T_init=0.002, T_end=0.00005):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    cool = (T_end / T_init) ** (1.0 / max(n_iter, 1))
    for _it in range(n_iter):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if h[i] > h[j] + 1e-8:
            direction = 1
            max_d = min(h[i], 1.0 - h[j])
        elif h[j] > h[i] + 1e-8:
            direction = -1
            max_d = min(h[j], 1.0 - h[i])
        else:
            direction = 1 if rng.random() < 0.5 else -1
            max_d = 1e-4
        if max_d < 1e-10:
            continue
        delta = rng.uniform(0.005, 0.15) * max_d
        h[i] -= direction * delta
        h[j] += direction * delta
        h = np.clip(h, 0.0, 1.0)
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or (T > 1e-8 and rng.random() < np.exp((cur_c5 - new_c5) / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] += direction * delta
            h[j] -= direction * delta
            h = np.clip(h, 0.0, 1.0)
        T *= cool
    return best_h, best_c5

def _coord_polish(h, deadline, n_cycles=2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(25):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _lp_local_multi_k(h, deadline, K_list=None, radius_init=0.15):
    if K_list is None:
        K_list = [5, 8, 12, 16, 20]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for K in K_list:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (len(K_list) + 1))
        h_try, c5_try = _lp_local(h, sub_dl, K=K, radius_init=radius_init, radius_min=0.008, max_iters=500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            cur_c5 = c5_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    overall_deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Strong LP optimization (20% budget)
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_local(h_start, dl1, K=12, radius_init=0.22, radius_min=0.004, max_iters=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothed gradient continuation (12% budget)
    dl2 = t0 + budget_s * 0.32
    h2, c5_2 = _smoothed_grad(best_h, dl2, beta_start=12, beta_end=500, n_steps=500, step_init=0.05)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Follow-up LP
    if time.time() < t0 + budget_s * 0.36:
        dl2b = t0 + budget_s * 0.36
        h2b, c5_2b = _lp_local(best_h, dl2b, K=10, radius_init=0.12, radius_min=0.003, max_iters=2000)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Multi-resolution exploration (25% budget)
    dl3 = t0 + budget_s * 0.61
    resolutions = [96, 112, 128, 144, 80, 64, 160]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.6180339887, 0.3819660113]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if n_try == best_n and abs(_exact_c5(best_h) - best_c5) < 1e-15:
            pass
        for s in range(3):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                alpha = alphas[ri % len(alphas)]
                h_init = _qperiodic_init(n_try, alpha)
            else:
                alpha = alphas[(ri + 3) % len(alphas)]
                h_init = _qperiodic_init(n_try, alpha)
                h_init += rng.normal(0, 0.05, n_try)
                h_init = _project(h_init, n_try / 2.0)
            sub_dl = min(dl3, time.time() + budget_s * 0.035)
            h_try, c5_try = _lp_local(h_init, sub_dl, K=8, radius_init=0.18, radius_min=0.005, max_iters=1500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 4: Multi-K LP on best (8% budget)
    if time.time() < t0 + budget_s * 0.69:
        dl4 = t0 + budget_s * 0.69
        h4, c5_4 = _lp_local_multi_k(best_h, dl4, K_list=[4, 8, 15, 25], radius_init=0.10)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Pair transfer SA for diversity (12% budget)
    dl5 = t0 + budget_s * 0.81
    h5, c5_5 = _pair_transfer_sa(best_h, dl5, rng, n_iter=80000, T_init=0.003, T_end=0.00003)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # LP refinement after SA
    if time.time() < t0 + budget_s * 0.84:
        dl5b = t0 + budget_s * 0.84
        h5b, c5_5b = _lp_local(best_h, dl5b, K=10, radius_init=0.12, radius_min=0.004, max_iters=1500)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Perturbation restarts (10% budget)
    dl6 = t0 + budget_s * 0.94
    scale = 0.02
    for _attempt in range(500):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.008)
        h_try, c5_try = _lp_local(h_pert, sub_dl, K=8, radius_init=0.06, radius_min=0.005, max_iters=300)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.06, scale * 1.12)

    # Phase 7: Coordinate polish (5% budget)
    if time.time() < t0 + budget_s * 0.99:
        dl7 = min(overall_deadline, t0 + budget_s * 0.99)
        h7, c5_7 = _coord_polish(best_h, dl7, n_cycles=1)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final fine LP
    if time.time() < overall_deadline - 1.0:
        dl8 = overall_deadline
        h8, c5_8 = _lp_local(best_h, dl8, K=6, radius_init=0.015, radius_min=0.001, max_iters=1000)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
