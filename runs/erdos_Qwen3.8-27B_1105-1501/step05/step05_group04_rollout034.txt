```python
def compute_softmax_grad(h, beta):
    N = len(h)
    dx = 2.0 / N
    overlap = np.correlate(h, 1.0 - h, mode='full') * dx
    # softmax weights
    max_ov = np.max(overlap)
    exp_ov = np.exp(beta * (overlap - max_ov))
    w = exp_ov / np.sum(exp_ov)
    
    grad = np.zeros(N)
    for d in range(-(N-1), N):
        if w[d + N - 1] == 0:
            continue
        wd = w[d + N - 1] * dx
        # i-term: j in [max(0,-d), min(N-1, N-1-d)]
        j1_start = max(0, -d)
        j1_end = min(N-1, N-1-d) + 1  # exclusive for numpy slicing
        if j1_end > j1_start:
            grad[j1_start:j1_end] += wd * (1.0 - h[j1_start+d : j1_end+d])
        # i+d-term: j in [max(0,d), min(N-1, N-1+d)]
        j2_start = max(0, d)
        j2_end = min(N-1, N-1+d) + 1
        if j2_end > j2_start:
            grad[j2_start:j2_end] -= wd * h[j2_start-d : j2_end-d]
    
    return grad, (1.0/beta) * (np.log(np.sum(exp_ov)) + max_ov * beta)
```

def project_box_sum(h, target):

    N = len(h)

    target = N / 2.0

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    active_mask = corr > max_val - 1e-10
    
    # Subgradient: average of gradients of active lags
    grad = np.zeros(N)
    active_indices = np.where(active_mask)[0]
    for idx in active_indices:
        d = idx - (N - 1)
        j1s = max(0, -d)
        j1e = min(N, N - d)
        if j1e > j1s:
            grad[j1s:j1e] += dx * (1.0 - h[j1s + d:j1e + d])
        j2s = max(0, d)
        j2e = min(N, N + d)
        if j2e > j2s:
            grad[j2s:j2e] -= dx * h[j2s - d:j2e - d]
    
    grad /= len(active_indices)
    
    # Take step and project
    h_new = h - step_size * grad
    h_new = project_box_sum(h_new, target)
    return h_new
