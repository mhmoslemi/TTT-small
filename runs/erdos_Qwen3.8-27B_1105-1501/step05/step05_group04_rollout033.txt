    """Find argmin of max_i (s_i + g_i * t) over t in [lo, hi]."""
    n = len(s)
    # Quick check: if all g are same sign, minimum is at endpoint
    if np.all(g >= 0):
        return lo, float(np.max(s + lo * g))
    if np.all(g <= 0):
        return hi, float(np.max(s + hi * g))
    
    # Evaluate at endpoints
    f_lo = float(np.max(s + lo * g))
    f_hi = float(np.max(s + hi * g))

    best_f = min(f_lo, f_hi)

The function f(t) = max_i (s_i + g_i * t) is convex in t. Its minimum over [lo, hi] is:
- If f'(t) > 0 for all t in [lo, hi] (i.e., all active g_i > 0 throughout): minimum at lo
- If f'(t) < 0 for all t in [lo, hi]: minimum at hi
- Otherwise: minimum where 0 enters the subgradient

The bisection I described should work. Let me trace through:
- At t = mid, compute vals = s + mid*g, find max_val = max(vals)
- Active set: indices where vals[i] ≈ max_val
- If all active g[i] > 0: f is increasing at mid, so minimum is to the left → hi = mid
- If all active g[i] < 0: f is decreasing at mid, so minimum is to the right → lo = mid
- If mixed: 0 is in the subgradient, we're at the minimum

This is correct for convex piecewise linear functions.

Let me also think about the LP formulation more carefully.

The LP for finding a direction:
maximize t
subject to:
  G*d - t*1 ≤ -s (i.e., s + G*d - t ≤ 0, meaning all lags decrease by at least t)
  sum(d) = 0 (preserve sum constraint)
  l_i ≤ d_i ≤ u_i (box constraints, where l_i = max(-radius, -h[i]), u_i = min(radius, 1-h[i]))

This finds a direction that reduces ALL lag values by at least t. The objective value after the step would be at most max(s + G*d) = max(s) - t (approximately, for unit step).

The "relaxed" version:
maximize t
subject to:
  G[r]*d - t ≤ -s[r] for r in active_set (top k lags)
  G[r]*d ≤ -s[r] + slack for r not in active_set (don't increase other lags by too much)
  sum(d) = 0
  box constraints

This focuses on reducing the top lags while being more lenient on others.

In the previous code, the relaxed version uses `b_ub = -scores + radius * dx` for non-active constraints, which allows them to increase by up to `radius * dx`. This is a reasonable tolerance.

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _all_lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        if not np.any(mask):
            mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        else:
            g = -d
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    return G

def _min_pwl(s, g, lo, hi):
    """Minimize max_i(s_i + g_i*t) over t in [lo, hi]. Returns (t*, f(t*))."""
    if hi - lo < 1e-14:
        return lo, float(np.max(s + lo * g))
    f_lo = float(np.max(s + lo * g))
    f_hi = float(np.max(s + hi * g))
    best_t = lo if f_lo <= f_hi else hi
    best_f = min(f_lo, f_hi)
    a, b = lo, hi
    for _ in range(60):
        if b - a < 1e-13:
            break
        mid = (a + b) / 2.0
        vals = s + mid * g
        f_mid = float(np.max(vals))
        active = (vals - f_mid) > -1e-13
        g_min = float(np.min(g[active]))
        g_max = float(np.max(g[active]))
        if g_min > 1e-12:
            b = mid
        elif g_max < -1e-12:
            a = mid
        else:
            break
    t_best = (a + b) / 2.0
    f_best = float(np.max(s + t_best * g))
    if f_best < best_f:
        best_t, best_f = t_best, f_best
    return best_t, best_f

def _lp_direction(h, scores, G, radius, n_active):
    n = len(h)
    n_lags = 2 * n - 1
    # Full LP: all constraints tight
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
        if res.success and res.x[-1] < float(np.max(scores)) - 1e-10:
            return res.x[:n]
    except Exception:
        pass
    # Relaxed LP: focus on top n_active lags
    sorted_idx = np.argsort(scores)[::-1]
    active_set = set(sorted_idx[:n_active].tolist())
    dx_val = 2.0 / n
    b_ub_rel = -scores + radius * dx_val
    for k in active_set:
        b_ub_rel[k] = -scores[k]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub_rel, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
        if res.success and res.x[-1] < float(np.max(scores)) - 1e-10:
            return res.x[:n]
    except Exception:
        pass
    # Even more relaxed: only top 1
    if len(active_set) > 1:
        top1 = sorted_idx[0]
        b_ub_rel2 = -scores + radius * 2.0 * dx_val
        b_ub_rel2[top1] = -scores[top1]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub_rel2, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 2.0})
            if res.success and res.x[-1] < float(np.max(scores)) - 1e-10:
                return res.x[:n]
        except Exception:
            pass
    return None
