def _lse_objective(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Numerical stability for log-sum-exp
    m = np.max(corr)
    lse = (1.0/beta) * (m + np.log(np.sum(np.exp(beta * (corr - m)))))
    # Penalty for sum constraint
    s = np.sum(h) - target
    return lse + penalty * s * s

def _lse_gradient(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_terms = np.exp(beta * (corr - m))
    weights = exp_terms / np.sum(exp_terms)  # softmax weights
    
    # Gradient of log-sum-exp = sum_k weights[k] * grad_k
    # grad_k is the gradient of the k-th lag value w.r.t. h
    # Need to compute sum_k weights[k] * grad_k efficiently
    
    # For each lag k (corresponding to shift = k - (n-1)):
    # The gradient of the k-th lag (times dx) w.r.t. h[j] can be computed
    # Let's build the full gradient matrix G where G[k, j] = d(correlate[k])/dh[j] * dx
    
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        w = weights[k]
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            m_pts = n - shift
            if m_pts > 0:
                # f = sum_{i=0}^{m_pts-1} h[i]*(1-h[i+shift])
                # df/dh[j] = (1-h[j+shift]) if j+shift < n, plus -h[j-shift] if j >= shift
                g = np.zeros(n)
                # h[j] as first factor: j ranges 0..m_pts-1, contributes (1-h[j+shift])
                g[:m_pts] += (1.0 - h[shift:shift+m_pts])
                # h[j] as second factor (1-h[j]): j ranges shift..shift+m_pts-1 = n-1
                # contributes -h[j-shift]
                g[shift:] += -h[:m_pts]
                grad += w * g
        else:
            s = -shift
            m_pts = n - s
            if m_pts > 0:
                # f = sum_{i=0}^{m_pts-1} h[i+s]*(1-h[i])
                # df/dh[j]: as h[i+s] where j=i+s, i=j-s: contributes (1-h[j-s]) for j in s..s+m_pts-1
                #           as (1-h[i]) where j=i: contributes -h[j+s] for j in 0..m_pts-1
                g = np.zeros(n)
                g[s:] += (1.0 - h[:m_pts])
                g[:m_pts] += -h[s:s+m_pts]
                grad += w * g
    
    # Add penalty gradient
    grad += 2.0 * penalty * (np.sum(h) - target)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_val_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    exp_sum = float(np.sum(exp_vals))
    lse_val = m + np.log(exp_sum) / beta
    weights = exp_vals / exp_sum
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            mp = n - shift
            if mp > 0:
                grad[:mp] += -w * dx * h[shift:shift + mp]
                grad[shift:] += w * dx * (1.0 - h[:mp])
        else:
            s = -shift
            mp = n - s
            if mp > 0:
                grad[:mp] += w * dx * (1.0 - h[s:s + mp])
                grad[s:] += -w * dx * h[:mp]
    s_dev = float(np.sum(h)) - target
    grad += 2.0 * penalty * s_dev
    total = lse_val + penalty * s_dev * s_dev
    return total, grad

def _run_lse_step(h, beta, penalty, max_iter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_lse_val_grad, h, method='L-BFGS-B', jac=True,
                   args=(beta, penalty), bounds=bounds,
                   options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                            'ftol': 1e-14, 'gtol': 1e-9})
    return res.x

def _lse_continuation(h_init, betas, deadline, penalty=100.0):
    h = h_init.copy()
    n = len(h)
    target = n / 2.0
    n_betas = len(betas)
    for idx, beta in enumerate(betas):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iter = max(30, int(remaining / max(n_betas - idx, 1) / 0.03))
        max_iter = min(max_iter, 2000)
        h_new = _run_lse_step(h, beta, penalty, max_iter)
        h_new = _project_box_sum(h_new, target)
        h = h_new
    return h

def _coord_descend(h_init, deadline, n_iter=2000):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:

        if stall > 300:
            break
        i = int(rng.integers(0, n))

        if i == j:

        d = np.zeros(n)
        d[j] = 1.0
        d[i] = -1.0

        best_t = 0.0
        best_v = best_c5

            h_try = h + t * d
            v_try = _exact_c5(h_try)
            if v_try < best_v:
                best_v = v_try
                best_t = t
        if best_t > 0 and best_v < best_c5 - 1e-14:
            h = h + best_t * d
            h = _project_box_sum(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:

def _coord_descend_structured(h_init, deadline):

        best_i = -1

        best_gain = 1e-14

            i = interior[ii]

                continue

                j = interior[jj]
                gj = g[j]

                    best_i = i

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _multi_res_search(h_init, c5_init, deadline, rng):

    for n_try in n_candidates:

        if c5_r < best_c5 - 1e-10:

def _slsqp_polish(h_init, deadline, n_top=5):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    def slsqp_obj(x):
        h_t = x[:n]
        t = x[n]
        corr = np.correlate(h_t, 1.0 - h_t, mode='full') * dx
        active = np.argsort(corr)[::-1][:n_top]
        viol = np.max(corr[active] - t)
        return max(viol, 0.0)
    
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    t0_val = float(np.max(corr))
    x0 = np.append(h, t0_val)
    bounds_s = [(0.0, 1.0)] * n + [(0.0, t0_val + 0.05)]
    res = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds_s,
                   options={'maxiter': 300, 'ftol': 1e-12})
    if res.success:
        h_new = _project_box_sum(res.x[:n], target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5
