    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        row = n - 1 + k
        i_min = max(0, -k)
        i_max = min(n-1, n-1-k)
        m = i_max - i_min + 1
        if m <= 0:
            continue
        # First term: for j in [i_min, i_max], G[row, j] += (1 - h[j+k])
        j1 = np.arange(i_min, i_max+1)
        G[row, j1] += (1.0 - h[j1 + k])
        # Second term: for j in [i_min+k, i_max+k], G[row, j] -= h[j-k]
        j2 = np.arange(i_min+k, i_max+k+1)
        G[row, j2] -= h[j2 - k]
    G *= dx
    return G

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
        else:
            idx = np.where(x > 1e-12)[0]
        if len(idx) > 0:
            x[idx] += res / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        row = n - 1 + k
        i_min = max(0, -k)
        i_max = min(n-1, n-1-k)
        m = i_max - i_min + 1
        if m <= 0:
            continue
        j1 = np.arange(i_min, i_max+1)
        G[row, j1] += (1.0 - h[j1 + k])
        j2 = np.arange(i_min+k, i_max+k+1)
        G[row, j2] -= h[j2 - k]
    G *= dx
    return G

def _lp_direction_active(h, scores, G, radius, n_active=60):
    n = len(h)
    # Select the top n_active lags with highest scores
    idx = np.argsort(scores)[-n_active:]
    G_sub = G[idx, :]
    scores_sub = scores[idx]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -scores_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

        rng_fix = np.random.default_rng(1)

            h[i] = 1

    best_c5 = _exact_c5(h)

    zeros_set = set(np.where(h == 0)[0].tolist())

        sweep += 1
        improved = False
        # Random shuffle of positions

        zeros_list = list(zeros_set)
        rng_s = np.random.default_rng(sweep)
        rng_s.shuffle(ones_list)
        rng_s.shuffle(zeros_list)

                h[i] = 0
                h[j] = 1
                c5_try = _exact_c5(h)
                if c5_try < cur_c5 - 1e-15 or (anneal and c5_try < cur_c5 + T):
                    # Accept

                        # Good - update best
                        improved = True
                        ones_set.discard(i)
                        ones_set.add(j)

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _qp_init(n, alpha, gap=0.0):

    x = np.arange(n) / n
    if gap > 0:
        beta = 0.5 - gap
        h = ((x * alpha) % 1.0 < beta) | ((x * alpha) % 1.0 > 1.0 - gap)
    else:
        h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
        else:
            idx = np.where(x > 1e-12)[0]
        if len(idx) > 0:
            x[idx] += res / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = n - 1 + k
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        m = i_max - i_min + 1
        if m <= 0:
            continue
        j1 = np.arange(i_min, i_max + 1)
        G[row, j1] += (1.0 - h[j1 + k])
        j2 = np.arange(i_min + k, i_max + k + 1)
        G[row, j2] -= h[j2 - k]
    G *= dx
    return G

def _lp_direction_active(h, scores, G, radius, n_active=60):
    n = len(h)
    n_lags = len(scores)
    if n_active < n_lags:
        idx = np.argsort(scores)[-n_active:]
    else:
        idx = np.arange(n_lags)
    G_sub = G[idx, :]
    scores_sub = scores[idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -scores_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(28):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _binary_local_search(h_init, deadline, max_sweeps=400, anneal=True):
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    target = n // 2
    ones_count = int(np.sum(h))
    if ones_count != target:
        if ones_count > target:
            idx = np.where(h == 1)[0]
            rng_f = np.random.default_rng(42)
            rng_f.shuffle(idx)
            for i in idx[:ones_count - target]:
                h[i] = 0
        else:
            idx = np.where(h == 0)[0]
            rng_f = np.random.default_rng(43)
            rng_f.shuffle(idx)
            for i in idx[:target - ones_count]:
                h[i] = 1
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    ones_list = [int(i) for i in np.where(h == 1)[0]]
    zeros_list = [int(i) for i in np.where(h == 0)[0]]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng_s = np.random.default_rng(1000 + sweep)
        rng_s.shuffle(ones_list)
        rng_s.shuffle(zeros_list)
        if anneal:
            frac = max(0.0, 1.0 - sweep / max(1, max_sweeps * 0.7))

                    zeros_list[zeros_list.index(j)] = i
                    cur_c5 = c5_try
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
                if improved and not anneal:
                    break
        if not improved:
            break
    return best_h, best_c5

def _smoothed_grad_step(h, deadline, n_steps=3000, beta_start=20.0, beta_end=200.0):

    target = n / 2.0

    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        shifted = beta * (scores - max_s)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.1
        improved = False
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-13:
                break
        if not improved and step % 100 == 0:
            alpha = 0.01
            h = _project(h - alpha * grad, target)
            cur_c5 = _exact_c5(h)
    return best_h, best_c5

    if time.time() < t0 + budget_s * 0.40:
        dl2 = t0 + budget_s * 0.40
        h2, c5_2 = _smoothed_grad_step(best_h, dl2, n_steps=2000, beta_start=15.0, beta_end=100.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        # Quick LP polish
        if time.time() < t0 + budget_s * 0.44:
            dl2b = t0 + budget_s * 0.44
            h2b, c5_2b = _lp_optimize(best_h, dl2b, radius_init=0.1, max_iters=3000, stall_limit=100, n_active=60)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Binary local search (15%)
    if time.time() < t0 + budget_s * 0.58:
        dl3 = t0 + budget_s * 0.58
        # Check if solution is nearly binary
        frac_count = np.sum((best_h > 0.1) & (best_h < 0.9))
        if frac_count < best_n * 0.3:
            hb, c5b = _binary_local_search(best_h, dl3, max_sweeps=300, anneal=True)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = hb.copy()
        # LP from best binary or current
        if time.time() < t0 + budget_s * 0.62:
            dl3b = t0 + budget_s * 0.62
            h3b, c5_3b = _lp_optimize(best_h, dl3b, radius_init=0.08, max_iters=3000, stall_limit=100, n_active=60)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration (22%)
    dl4 = t0 + budget_s * 0.84
    alpha_vals = [(1 + np.sqrt(5)) / 2 - 1, np.sqrt(2) - 1, 1.0 / np.sqrt(5.0), (1 + np.sqrt(5)) / 2]
    n_vals = [96, 128, 160]
    for n_try in n_vals:
        if time.time() > dl4:
            break
        # Resample from best
        h_init = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.035)
        ht, ct = _lp_optimize(h_init, sub_dl, radius_init=0.2, max_iters=4000, stall_limit=150, n_active=60)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()
            best_n = n_try
        # Quasi-periodic with noise
        if time.time() < dl4:
            for ai in range(2):
                if time.time() > dl4:
                    break
                a = alpha_vals[(n_try + ai) % len(alpha_vals)]
                h_qp = _qp_init(n_try, a)
                h_qp = _project(h_qp + rng.normal(0, 0.04, n_try), n_try / 2.0)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                ht2, ct2 = _lp_optimize(h_qp, sub_dl2, radius_init=0.25, max_iters=3000, stall_limit=120, n_active=60)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = n_try
                # Gap variant
                if time.time() < dl4:
                    h_gap = _qp_gap_init(n_try, a, 0.05)
                    h_gap = _project(h_gap + rng.normal(0, 0.03, n_try), n_try / 2.0)
                    sub_dl3 = min(dl4, time.time() + budget_s * 0.015)
                    ht3, ct3 = _lp_optimize(h_gap, sub_dl3, radius_init=0.25, max_iters=2000, stall_limit=100, n_active=60)
                    if ct3 < best_c5:
                        best_c5 = ct3
                        best_h = ht3.copy()
                        best_n = n_try

    # Phase 5: Binary search on best (8%)
    if time.time() < t0 + budget_s * 0.92:
        dl5 = t0 + budget_s * 0.92
        frac_count = np.sum((best_h > 0.1) & (best_h < 0.9))
        if frac_count < best_n * 0.4:
            hb, c5b = _binary_local_search(best_h, dl5, max_sweeps=400, anneal=True)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = hb.copy()
            # LP polish after binary
            if time.time() < t0 + budget_s * 0.94:
                hbp, c5bp = _lp_optimize(best_h, t0 + budget_s * 0.94, radius_init=0.06, max_iters=2000, stall_limit=80, n_active=50)
                if c5bp < best_c5:
                    best_c5 = c5bp
                    best_h = hbp.copy()

    # Phase 6: Perturbation restarts (4%)
    if time.time() < t0 + budget_s * 0.965:
        dl6 = t0 + budget_s * 0.965
        scale = 0.012
        for attempt in range(400):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.006)
            ht, ct = _lp_optimize(h_pert, sub_dl, radius_init=0.04, max_iters=800, stall_limit=40, n_active=40)
            if ct < best_c5 - 1e-14:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.05, scale * 1.12)

    # Phase 7: Final fine LP polish
    if time.time() < deadline:
        h7, c5_7 = _lp_optimize(best_h, deadline, radius_init=0.03, max_iters=10000, stall_limit=300, n_active=100)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
