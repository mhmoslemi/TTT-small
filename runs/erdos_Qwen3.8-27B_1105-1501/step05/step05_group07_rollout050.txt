```python
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    """Compute gradient rows for specific lags only."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                # For i in [0, m-1]: h[i] contributes (1-h[i+d]), h[i+d] contributes -h[i]
                G[idx, :m] += (1.0 - h[d:d+m])
                G[idx, d:d+m] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, g:g+m] += (1.0 - h[:m])
                G[idx, :m] -= h[g:g+m]
    return G * dx

def _active_lp(h, scores, K, radius):
    """LP using only top K scores."""
    n = len(h)
    top_idx = np.argsort(scores)[-K:]
    top_scores = scores[top_idx]
    G = _grad_rows(h, top_idx)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -top_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, max_evals=30):
    n = len(h)
    alpha_max = 1.0
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, (1.0 - h[pos]).min() / p[pos].max())
    if np.any(neg):
        alpha_max = min(alpha_max, h[neg].min() / (-p[neg]).max())
    alpha_max = max(alpha_max, 1e-12) * 0.999
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    evals = 2
    for _ in range(max_evals - 2):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
        evals += 1
    opt_a = (lo + hi) / 2.0
    return opt_a, _exact_c5(h + opt_a * p)

        spread = top20[0] - top20[-1]
        if spread < cur_c5 * 0.05:

        else:
            K_cur = K
        
        p, _ = _active_lp(h, scores, K_cur, radius)

            radius *= 0.85
            if radius < 1e-6:

        alpha, new_c5 = _line_search(h, p, max_evals=28)

            if cur_c5 < best_c5:

                best_h = h.copy()
            stall = 0

        else:
            stall += 1
            radius *= 0.85
            if stall > 40 and radius < 0.05:
                radius = 0.1
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] += (1.0 - h[d:d+m])
                G[idx, d:d+m] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, g:g+m] += (1.0 - h[:m])
                G[idx, :m] -= h[g:g+m]
    return G * dx

def _active_lp(h, scores, K, radius):
    n = len(h)
    top_idx = np.argsort(scores)[-K:]
    top_scores = scores[top_idx]
    G = _grad_rows(h, top_idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -top_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, max_evals=26):
    alpha_max = 1.0
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, (1.0 - h[pos]).min() / p[pos].max())
    if np.any(neg):
        alpha_max = min(alpha_max, h[neg].min() / (-p[neg]).max())
    alpha_max = max(alpha_max, 1e-12) * 0.999
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(max_evals - 2):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    return opt_a, _exact_c5(h + opt_a * p)

def _active_lp_opt(h, deadline, K=15, radius_init=0.3, max_iters=10000, stall_limit=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline or stall > stall_limit:
            break
        scores = _scores(h)
        ts = np.sort(scores)[-min(20, len(scores)):]
        spread = ts[0] - ts[-1]
        K_cur = max(5, min(K, int(5 + 15 * spread / max(cur_c5, 1e-10))))
        p, _ = _active_lp(h, scores, K_cur, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.85
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, max_evals=24)
        if new_c5 < cur_c5 - 1e-15:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.4)
        else:
            stall += 1
            radius *= 0.87
            if stall > 30 and radius < 0.03:
                radius = 0.08
    return best_h, best_c5

def _swap_move(h, rng):
    n = len(h)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return h
    h2 = h.copy()
    h2[i], h2[j] = h[j], h[i]
    return h2

def _block_move(h, rng):
    n = len(h)
    blen = int(rng.integers(1, max(2, n // 8)))
    i = int(rng.integers(0, max(1, n - blen)))
    j = int(rng.integers(0, max(1, n - blen)))
    if abs(i - j) < blen:
        return h
    h2 = h.copy()
    h2[i:i+blen], h2[j:j+blen] = h[j:j+blen].copy(), h[i:i+blen].copy()
    return h2

def _gauss_move(h, rng, sigma):
    n = len(h)
    pert = rng.normal(0, sigma, n)
    pert -= pert.mean()
    return _project(h + pert, n / 2.0)

def _sa_optimize(h, deadline, rng, T0=0.01, Tend=0.001, n_steps=2000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    steps = min(n_steps, 50000)
    for step in range(steps):
        if time.time() > deadline:
            break
        frac = step / max(1, steps - 1)
        T = T0 * (Tend / T0) ** frac
        move_type = rng.integers(3)
        if move_type == 0:
            h_try = _swap_move(h, rng)
        elif move_type == 1:
            h_try = _block_move(h, rng)
        else:
            h_try = _gauss_move(h, rng, 0.03 * (1 + frac))
        c5_try = _exact_c5(h_try)
        delta = c5_try - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _thue_morse_init(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = (t == 0).astype(float)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_opt(h, deadline, beta_init=50.0, beta_final=300.0, n_phases=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    from scipy.optimize import minimize
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        def obj(hf, b=beta):
            S = _scores(hf)
            maxS = np.max(S)
            sh = b * (S - maxS)
            sh = np.clip(sh, -400, 0)
            return maxS + np.log(np.mean(np.exp(sh))) / b
        def grad(hf, b=beta):
            S = _scores(hf)
            maxS = np.max(S)
            sh = b * (S - maxS)
            sh = np.clip(sh, -400, 0)
            w = np.exp(sh)
            wsum = np.sum(w)
            if wsum < 1e-300:
                w = np.zeros(len(S))
                w[int(np.argmax(S))] = 1.0
            else:
                w /= wsum
            G = _grad_rows(hf, np.arange(len(S)))
            return G.T @ w
        def eq_fun(x):
            return np.sum(x) - target
        def eq_jac(x):
            return np.ones(n)
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                          options={'maxiter': 150, 'ftol': 1e-14})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5
```
