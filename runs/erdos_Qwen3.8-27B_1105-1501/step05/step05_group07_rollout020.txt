        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            # Verify improvement (projection might degrade)
            if cur_c5 > new_c5 + 1e-10:
                # Use the unprojected version if it's better and close to feasible
                h_new = np.clip(h + alpha * p, 0.0, 1.0)
                h_new = _project(h_new, target)
                cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
            # Potentially shrink active set (focus more)
            if it % 20 == 0 and top_k > 5:
                top_k = max(5, int(top_k * 0.8))
            # Expand if inactive lags are now high
            actual_max = float(np.max(_scores(h)))
            if actual_max > t_new + 0.001:
                top_k = min(2 * n - 1, top_k + 10)
        else:
            stall += 1
            radius *= 0.8
            if top_k < 2 * n - 1 and stall % 20 == 0:
                top_k = min(2 * n - 1, top_k + 5)
                radius = max(radius, 0.1)
            if stall > 50 and radius < 0.02:
                # Reset with expanded set
                top_k = 2 * n - 1
                radius = 0.1
                expand_count += 1
                if expand_count > 20:
                    break
    
    return best_h, best_c5

def _lbfgs_minimax(h, deadline, beta_start=50, beta_end=500, n_phases=4):
    """L-BFGS-B on log-sum-exp smooth minimax."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    def obj(x):
        S = _scores(x)
        m = np.max(S)
        shifted = beta * (S - m)
        shifted = np.clip(shifted, -500, 0)
        return m + np.log(np.sum(np.exp(shifted))) / (beta * len(S))
    
    def jac(x):
        S = _scores(x)
        m = np.max(S)
        shifted = beta * (S - m)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w /= wsum
        G = _grad_matrix(x)
        return G.T @ w
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'gtol': 1e-12})
            if np.all(np.isfinite(res.x)):
                h_new = res.x
                # Normalize to satisfy sum constraint
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish_fast(h, deadline, max_passes=3):
    """Fast pair polish: try many random pairs."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(123)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        n_tried = 0
        for _ in range(200):
            if time.time() > deadline:
                break
            j, k = rng.integers(0, n, 2)
            if j == k:
                continue
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8 or h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                continue
            # Try moving mass from j to k and vice versa
            for sign in [1, -1]:
                if sign > 0:
                    a_max = min(1.0 - h[j], h[k]) * 0.99
                else:
                    a_max = min(h[j], 1.0 - h[k]) * 0.99
                if a_max < 1e-10:
                    continue
                # Ternary search for optimal a
                lo_a, hi_a = 0.0, a_max
                for _ in range(10):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    h1 = h.copy()
                    h1[j] += sign * m1
                    h1[k] -= sign * m1
                    f1 = _exact_c5(h1)
                    h2 = h.copy()
                    h2[j] += sign * m2
                    h2[k] -= sign * m2
                    f2 = _exact_c5(h2)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                a_opt = (lo_a + hi_a) / 2.0
                if a_opt > 1e-8:
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        n_tried += 1
    return best_h, best_c5

def _quasi_init(n, alpha):
    """Quasi-periodic initialization."""
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)
