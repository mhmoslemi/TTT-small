def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:

                G[r, d:] += (1.0 - h[:m])

            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.05):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 122
        x = (np.arange(n0) + 0.5) / n0
        h0 = np.where((x * n0) % 2 < 1.0, 1.0, 0.0)
        h0 = _project(h0, n0 / 2.0)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Combined local (LP + SLSQP alternating) - 28%
    d1 = t0 + budget_s * 0.28
    h1, c5_1 = _combined_local(h, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Dedicated SLSQP active set - 22%
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _slsqp_phase(best_h, d2, max_rounds=60)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smooth gradient - 10%
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _smooth_phase(best_h, d3, beta_list=(50, 150, 400), lr_init=0.02)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution - 18%
    d4 = t0 + budget_s * 0.78
    best_n = len(best_h)
    tried = set([best_n])
    dn_list = []
    for dn in range(-4, 25, 2):
        n_try = best_n + dn
        if 64 <= n_try <= 250 and n_try not in tried:
            dn_list.append(n_try)
    dn_list.sort(key=lambda x: abs(x - best_n))
    for n_try in dn_list:
        if time.time() > d4:
            break
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.15)
        if c5_r < best_c5 + 0.0003 and time.time() < d4 - 8:
            sub_dl2 = min(d4, time.time() + 5.0)
            scores = np.correlate(h_r, 1.0 - h_r, mode='full')
            s_max = float(np.max(scores))
            thresh = s_max * 0.997
            active = np.where(scores >= thresh)[0]
            if len(active) > 25:
                active = np.argsort(scores)[-25:]
            if len(active) >= 3:
                h_r, c5_r = _slsqp_solve(h_r, active, radius=0.08, max_iter=20)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Kick and reoptimize - 10%
    d5 = t0 + budget_s * 0.88
    h5, c5_5 = _kick_and_reopt(best_h, d5, rng, n_kicks=25)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair refinement - 7%
    d6 = t0 + budget_s * 0.95
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=100, rng=rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final SLSQP polish - 3%
    d7 = hard_deadline
    if time.time() < d7 - 2:
        h7, c5_7 = _slsqp_phase(best_h, d7, max_rounds=20)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
