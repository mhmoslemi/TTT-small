2. The overlap at lag d is: dx * sum_i h[i] * (1 - h[i+d]) for valid indices.
3. This is a max-min type problem - we want all the cross-correlation values to be as small as possible.

The logistic parameterization h_i = sigma(q_i - mu) where mu is chosen so sum = n/2 is an exact manifold parameterization. This means:
- We never violate the sum constraint
- We optimize in q-space with L-BFGS-B (unconstrained except box on q)
- The gradient through the manifold is well-defined

This could be more effective because:
- BFGS uses curvature information (Hessian approximation)
- The manifold constraint is handled exactly
- We can use smooth surrogates (log-sum-exp) naturally

For n=128, the problem has 128 variables but effectively 127 degrees of freedom (one scalar for the sum constraint is handled by mu). So we're optimizing 128 q-values with L-BFGS-B.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _safe_sigma(t):
    """Numerically stable sigmoid."""
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def _solve_mu(q, target, wall=14.0):
    """Find mu such that sum(sigma(q_i - mu)) = target."""
    # sigma is decreasing in mu, so sum is decreasing in mu
    # At mu = -wall: sum ≈ n (all sigmas ≈ 1)
    # At mu = +wall: sum ≈ 0 (all sigmas ≈ 0)
    lo = -wall
    hi = wall
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_safe_sigma(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target, wall=14.0):
    """Map q to feasible h with exact sum."""
    mu = _solve_mu(q, target, wall)
    h = _safe_sigma(q - mu)
    # Ensure exact balance
    return _project(h, target)

def _h_to_q(h, eps=1e-6):
    """Map h to q (logit transform)."""
    h_c = np.clip(h, eps, 1.0 - eps)
    return np.log(h_c / (1.0 - h_c))

def _grad_matrix(h):
    """Build gradient matrix G where G[r, i] = dh_i/d(F_r)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _smooth_obj_grad(q, beta, target, wall=14.0):
    """Compute smooth log-sum-exp objective and gradient w.r.t. q."""
    mu = _solve_mu(q, target, wall)
    h = _safe_sigma(q - mu)
    
    # Compute scores
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = np.clip(beta * (scores - max_s), -500, 0)
    
    # Objective: max_s + (1/beta) * log(mean(exp(shifted)))
    # Note: this is max_s + (1/beta)*log(1/L * sum(exp(shifted)))
    #      = max_s + (1/beta)*(logsumexp(shifted) - log(L))
    #      = max_s + (1/beta)*(logsumexp(beta*(scores-max_s)) - log(L) - beta*0)
    # Actually: F_beta = max_s + (1/beta) * log( (1/L) * sum_i exp(beta*(s_i - max_s)) )
    #          = max_s + (1/beta) * (logsumexp(beta*(s_i - max_s)) - log(L))
    #          = max_s + (1/beta) * logsumexp(beta*(s_i - max_s)) - (1/beta)*log(L)
    # For gradient purposes, the constant -log(L)/beta doesn't matter.
    
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0
        wsum = 1.0
    else:
        w /= wsum
    
    obj = max_s + float(np.log(wsum)) / beta  # wait this isn't quite right
    # Actually: F = max_s + (1/beta)*log(sum(exp(beta*(s_i - max_s))))/L
    #           = max_s + (1/beta)*(log(sum(exp(shifted))) - log(L))
    # where shifted = beta*(s_i - max_s)
    # So: F = max_s + (1/beta)*log(mean(exp(shifted)))
    #     = max_s + (1/beta)*(logsumexp(shifted) - log(L))
    
    L = len(scores)
    log_mean_exp = float(np.log(np.sum(w))) - np.log(L)  # = log(mean(exp(shifted)))
    obj = max_s + log_mean_exp / beta
    
    # Gradient of F w.r.t. h:
    # dF/dh_i = sum_r w_r * G[r, i]  where G is grad matrix
    G = _grad_matrix(h)
    grad_h = G.T @ w  # shape (n,)
    
    # Chain rule through h = sigma(q - mu)
    # dh_i/dq_j = sigma'(q_i - mu) * (delta_ij - 1/n_eff)
    # where 1/n_eff comes from dmu/dq_j
    s = h * (1.0 - h)  # sigma'(q_i - mu) = sigma * (1-sigma)
    ssum = float(np.sum(s))
    if ssum < 1e-15:
        return 1e10, np.zeros(len(q))
    
    # dmu/dq_j = s_j / ssum
    # dh_i/dq_j = s_i * (delta_ij - s_j/ssum)
    # dF/dq_j = sum_i (dF/dh_i) * (s_i - s_i * s_j / ssum)
    #          = sum_i grad_h_i * s_i - s_j/ssum * sum_i grad_h_i * s_i
    #          = (grad_h * s) dot (e_j) - s_j/ssum * ((grad_h * s) dot 1)
    gs = grad_h * s  # shape (n,)
    gs_sum = float(np.sum(gs))
    grad_q = gs - (s / ssum) * gs_sum
    
    return obj, grad_q

def _bfgs_smooth(h_seed, target, deadline, beta_schedule, wall=14.0):
    """Run L-BFGS-B optimization in q-space with smooth surrogate."""
    q0 = _h_to_q(h_seed)
    best_q = q0.copy()
    best_h = h_seed.copy()
    best_c5 = _exact_c5(h_seed)
    
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        
        def objfun(qv):
            f, g = _smooth_obj_grad(qv, beta, target, wall)
            if not np.isfinite(f):
                return 1e10, np.zeros(len(qv))
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            return f, g
        
        try:
            res = minimize(objfun, q0, jac=True, method='L-BFGS-B',
                          bounds=[(-wall, wall)] * len(q0),
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10})
            if np.all(np.isfinite(res.x)):
                q0 = res.x
            # Check exact objective
            h_check = _q_to_h(q0, target, wall)
            c5_check = _exact_c5(h_check)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h_check
                best_q = q0.copy()
        except Exception:
            pass
    
    return best_h, best_c5, q0

def _lp_direction(h, scores, G, radius):
    """LP-based direction finding."""
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search_c5(h, p):
    """Golden section line search along direction p."""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    h_new = _project(h + opt_a * p, n / 2.0)
    return h_new, _exact_c5(h_new)

def _local_optimize_lp(h, deadline, radius_init=0.25, max_iters=5000, stall_limit=150):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        h_new, new_c5 = _line_search_c5(h, p)
        if new_c5 < cur_c5 - 1e-13:
            cur_c5 = new_c5
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.82
            if stall > 30 and radius < 0.03:
                radius = 0.12
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _complement_lp(h, deadline):
    """Try complement LP to find diverse direction."""
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    dx = 2.0 / n
    
    # Build LP: find y in [0,1]^n, sum(y)=target, minimizing max_d sum_i h[i]*y[i+d]*dx
    # This is: min tau s.t. for each lag d: sum over valid pairs h[i]*y[j]*dx <= tau
    # where j = i+d (valid range)
    
    G = _grad_matrix(h)  # G[r, j] = contribution of y[j] to score r (when h is fixed)
    # Actually G[r, j] already includes the h factor, so G[r,:] dot y gives the score
    # But we want to minimize max(G @ y) where G accounts for h
    
    # LP: min tau
    # s.t. G[r] dot y - tau <= 0 for all r
    #      sum(y) = target
    #      0 <= y <= 1
    
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(-1.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            y = _project(res.x[:n], target)
            return y
    except Exception:
        pass
    return None

def _pair_polish(h, deadline, max_passes=3):
    """Pair-based local polish."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = max(1, n // 32)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(7):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _subgrad_optimize(h, deadline, n_steps=2000, beta_start=10.0, beta_end=50.0):
    """Smooth subgradient method in q-space."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    q = _h_to_q(h)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        _, grad_q = _smooth_obj_grad(q, beta, target)
        
        if not np.all(np.isfinite(grad_q)):
            break
        gn = np.linalg.norm(grad_q)
        if gn < 1e-15:
            break
        grad_q /= gn
        q -= 0.05 * grad_q
        h_new = _q_to_h(q, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return best_h, best_c5
