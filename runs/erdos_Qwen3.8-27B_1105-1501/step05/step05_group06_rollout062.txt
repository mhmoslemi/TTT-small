    n = len(h)
    F = _scores(h)  # length 2n-1
    S = F.max()
    if S < 1e-30:
        return 0.0, np.zeros(n)
    v = F / S  # in [0, 1]
    # Clip for numerical safety
    v = np.clip(v, 0.0, 1.0)
    if q > 1.5:
        w = v ** (q - 1.0)
    else:
        w = np.ones_like(v)
    W = np.sum(w * v)  # = sum(v^q)
    if W < 1e-300:

        idx = np.argmax(F)
        if G is None:
            G = _build_G(h)
        grad = G[idx]
        val = S
        return val, grad
    val = S * W ** (1.0 / q)
    coeff = W ** (1.0 / q - 1.0)
    if G is None:
        G = _build_G(h)
    grad = coeff * (G.T @ w)

        return np.zeros(n)

        return np.ones(n)

    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    if abs(res) > 1e-12:
        if res > 0:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, 0:m] += dx * (1.0 - h[d:d+m])
        G[r, d:d+m] += -dx * h[0:m]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, 0:m] += dx * (1.0 - h[s:s+m])
        G[r, s:s+m] += -dx * h[0:m]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    S = F.max()
    if S < 1e-30:
        return 0.0
    v = np.clip(F / S, 0.0, 1.0)
    if q <= 1.5:

        if W < 1e-300:
            return S
        return S * W ** (1.0 / q)
    w = v ** (q - 1.0)
    W = float(np.dot(w, v))
    if W < 1e-300:
        return S
    return S * W ** (1.0 / q)

def _pnorm_grad(h, q):
    n = len(h)
    F = _scores(h)
    S = F.max()
    if S < 1e-30:

        return G[int(np.argmax(F))].copy()
    coeff = W ** (1.0 / q - 1.0)
    G = _build_G(h)
    return coeff * (G.T @ w)

            h_res = h_try

    # Phase 3: Diverse quasi-random starts
    dl3 = t0 + budget_s * 0.72
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.555]
    for n_try in [128, 192, 256]:
        if time.time() > dl3:
            break
        for alpha in alphas:
            if time.time() > dl3:
                break
            h_init = _quasi_init(n_try, alpha, rng)
            c5_init = _exact_c5(h_init)
            if c5_init < best_c5 + 0.05:
                sub_dl = min(dl3, time.time() + budget_s * 0.02)
                q_list3 = [16, 64, 192]
                for q in q_list3:
                    if time.time() > dl3:
                        break
                    h_try, c5_try = _slsqp_stage(h_init, q, sub_dl, maxiter=80)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        best_n = n_try
                    h_init = h_try

    # Phase 4: Perturbation + SLSQP restarts
    dl4 = t0 + budget_s * 0.88
    scale = 0.025
    for attempt in range(80):
        if time.time() > dl4:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl4, time.time() + budget_s * 0.008)
        h_try, c5_try = _slsqp_stage(h_pert, 128, sub_dl, maxiter=60)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.8)
        else:
            scale = min(0.07, scale * 1.08)

    # Phase 5: LP + pair descent final polish
    if time.time() < t0 + budget_s * 0.94:
        dl5 = t0 + budget_s * 0.95
        h_lp2, c5_lp2 = _lp_stage(best_h, dl5, top_k_list=[6, 4, 3, None],
                                   radius_init=0.04, max_iters=400)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()

    if time.time() < hard_dl:
        h_pd, c5_pd = _pair_descent(best_h, hard_dl, max_passes=80, verify_thresh=1e-7)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    return best_h, final_c5, best_n
