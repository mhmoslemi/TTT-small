def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _project(h, target):
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _smooth_max_obj(h, beta, scores, G):
    """Compute smooth max objective value."""
    b = beta * (scores - np.max(scores))
    b = np.clip(b, -500, 500)
    return (1.0 / beta) * (np.max(scores) + np.log(np.sum(np.exp(b))))

def _smooth_max_grad(h, beta, scores, G):
    """Compute gradient of smooth max."""
    b = beta * (scores - np.max(scores))
    b = np.clip(b, -500, 500)
    w = np.exp(b)
    w /= np.sum(w)
    return G.T @ w

def _slsqp_smooth(h0, beta, maxiter=200, tol=1e-8):
    """SLSQP optimization of smooth max."""
    n = len(h0)
    target = n / 2.0
    h0 = np.array(h0, dtype=float).copy()
    
    # Pre-compute gradient matrix (it changes with h, so we can't pre-compute)
    # SLSQP will call the objective and gradient multiple times

        scores = _scores_vec(x)
        return _smooth_max_obj(x, beta, scores, None)
    
    def grad(x):
        scores = _scores_vec(x)
        G = _grad_matrix(x)
        return _smooth_max_grad(x, beta, scores, G)
    
    def con_eq(x):
        return x.sum() - target
    
    def con_eq_grad(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_grad}]
    bounds = [(0.0, 1.0)] * n

                   bounds=bounds, constraints=constraints,

        return h_new, _c5_exact(h_new)

def _resample(h, n_new):

    if n_new == n:

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _project(h, target):
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _slsqp_smooth(h0, beta, maxiter=150, tol=1e-8):
    n = len(h0)
    target = n / 2.0
    h0 = np.array(h0, dtype=float).copy()

    def obj(x):
        s = _scores_vec(x)
        mx = np.max(s)
        b = np.clip(beta * (s - mx), -500, 500)
        return mx + np.log(np.sum(np.exp(b))) / beta

    def grad(x):
        s = _scores_vec(x)
        mx = np.max(s)
        b = np.clip(beta * (s - mx), -500, 500)
        w = np.exp(b)
        w /= np.sum(w)
        G = _grad_matrix(x)
        return G.T @ w

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None:
            h_new = _project(res.x, target)
            return h_new, _c5_exact(h_new)
    except Exception:

    return h0, _c5_exact(h0)

def _slsqp_minimax(h0, topk=50, maxiter=150, tol=1e-8):
    n = len(h0)
    target = n / 2.0
    h0 = np.array(h0, dtype=float).copy()
    cur_max = _c5_exact(h0)
    x0 = np.concatenate([h0, [cur_max]])

    scores_init = _scores_vec(h0)
    if topk < 2 * n - 1:
        idx = np.argsort(scores_init)[-topk:]
    else:
        idx = np.arange(2 * n - 1)

    def obj(x):
        return x[-1]

    def grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(x):
        return x[-1] - _scores_vec(x[:n])[idx]

    def ineq_jac(x):
        h = x[:n]
        G = _grad_matrix(h)
        jac = np.zeros((len(idx), n + 1))
        for i, k in enumerate(idx):
            jac[i, :n] = G[k]
            jac[i, -1] = -1.0
        return jac

    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': lambda x: x[:n].sum() - target,
         'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None:
            h_new = _project(res.x[:n], target)
            return h_new, _c5_exact(h_new)
    except Exception:
        pass
    return h0, _c5_exact(h0)

def _lp_step(h, radius, topk=None):
    n = len(h)
    scores = _scores_vec(h)
    G = _grad_matrix(h)
    if topk is not None and topk < 2 * n - 1:
        idx = np.argsort(scores)[-topk:]
        sk = scores[idx]
        Gk = G[idx]
    else:
        sk = scores
        Gk = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(sk), n + 1))
    A_ub[:, :n] = Gk
    A_ub[:, n] = -1.0
    b_ub = -sk
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, n_eval=18):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5_exact(h + c1 * p)
    f2 = _c5_exact(h + c2 * p)
    for _ in range(n_eval):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _c5_exact(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _c5_exact(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    return opt_a, _c5_exact(h + opt_a * p)
