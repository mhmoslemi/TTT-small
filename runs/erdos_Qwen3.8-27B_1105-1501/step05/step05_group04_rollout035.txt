Actually, let me think about this more carefully. The key question is: will a smoothed approach actually beat the LP approach?

The LP approach works by:
1. Finding the active lag (maximal score)
2. Solving an LP to find a direction that decreases all scores
3. Line search in that direction

The smoothed approach works by:
1. Computing a gradient that's a weighted average of per-lag gradients (weights concentrated on active lags)
2. Taking a gradient step

These are somewhat similar in spirit. The LP approach is more aggressive (it tries to decrease ALL lags simultaneously), while the smoothed approach only focuses on the weighted average.

However, the smoothed approach might escape local optima that the LP approach gets stuck in, because:
- The LP might find a direction that decreases the active lag but increases others, and the line search only accepts if the max decreases
- The smoothed gradient provides a smoother landscape where different "basins" might be accessible

    n = len(v)

def _projected_gradient_descent(h_init, beta, deadline, lr_init=0.1):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    
    while time.time() < deadline:
        obj, grad = _smoothed_obj_and_grad(h, beta)
        # Project gradient onto sum=0 hyperplane
        grad_proj = grad - np.mean(grad)
        # Line search
        best_t = 0
        best_obj = obj
        for t in [lr, lr*0.5, lr*0.25, lr*0.125, lr*0.0625]:
            h_try = _project(h - t * grad_proj, target)
            obj_try, _ = _smoothed_obj_and_grad(h_try, beta)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
        if best_t > 0:
            h = _project(h - best_t * grad_proj, target)
            c5_try = _exact_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
                lr = min(lr * 1.1, 1.0)
            else:
                lr *= 0.8
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _smoothed_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    shifted -= float(np.max(shifted))
    W = np.exp(shifted)
    W_s = float(np.sum(W))
    W /= W_s
    obj = dx * (max_s + (1.0 / beta) * np.log(W_s))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        w = W[r]
        if abs(w) < 1e-12:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w * (1.0 - h[:m])
                grad[:m] += -w * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] += -w * h[:m]
    grad *= dx
    return obj, grad


def _smoothed_optimize(h_init, betas, deadline):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        lr = 0.5
        stuck = 0
        for it in range(300):
            if time.time() > deadline:
                break
            obj, grad = _smoothed_obj_grad(h, beta)
            gp = grad - np.mean(grad)
            gn = float(np.linalg.norm(gp))
            if gn < 1e-10:
                break
            gp /= gn
            best_t = 0.0
            best_obj = obj
            for t in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
                h_try = _project(h - t * gn * gp, target)
                o_try, _ = _smoothed_obj_grad(h_try, beta)
                if o_try < best_obj - 1e-14:
                    best_obj = o_try
                    best_t = t
            if best_t > 0:
                h = _project(h - best_t * gn * gp, target)
                c5_try = _exact_c5(h)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h.copy()
                    stuck = 0
                    lr = min(lr * 1.05, 3.0)
                else:
                    stuck += 1
                    lr *= 0.85
            else:
                stuck += 1
                lr *= 0.7
            if stuck > 60:
                break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 80
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _lp_direction(h, scores, G, radius, n_active=4):
    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    dx_val = 2.0 / n
    b_ub = np.full(n_lags, -scores + radius * dx_val)
    sorted_idx = np.argsort(scores)[::-1]
    for k in sorted_idx[:n_active]:
        b_ub[k] = -scores[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 4
    while time.time() < deadline:
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            p = _lp_direction(h, scores, G, radius, min(n_active + 4, 2 * n - 2))
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.7
                if radius < 1e-7:
                    break
                if stall % 30 == 0:
                    radius = max(radius, 0.05)
                    n_active = min(n_active + 2, 2 * n - 2)
                continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 4
            radius = min(radius * 1.08, 0.35)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.03:
                radius = 0.08
                n_active = 6
    return best_h, best_c5
