    G_active = _grad_matrix_select(h, active)

    # subject to w >= 0, sum(w) = 1

    # Just use uniform weights for now
    w = np.ones(len(active)) / len(active)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection onto {x: 0<=x_i<=1, sum(x)=target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, lag_idx):
    """Compute gradient of F_k with respect to h, for given lag index in correlate output."""
    n = len(h)
    dx = 2.0 / n
    k = lag_idx - (n - 1)  # actual lag
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        g[:m] += dx * (1.0 - h[k:])  # d/dh_i of h_i*(1-h_{i+k})
        g[k:] += -dx * h[:m]  # d/dh_{i+k} of h_i*(1-h_{i+k})
    else:
        d = -k
        m = n - d
        g[:m] += -dx * h[d:]  # d/dh_i of h_{i+d}*(1-h_i)
        g[d:] += dx * (1.0 - h[:m])  # d/dh_{i+d}
    return g

def _grad_matrix_select(h, lag_indices):
    """Compute gradient matrix for selected lags only."""
    n = len(h)
    G = np.zeros((len(lag_indices), n))
    for i, li in enumerate(lag_indices):
        G[i] = _lag_grad(h, li)
    return G

def _line_search_exact(h, d, dx, max_alpha=1.0):
    """Exact line search for phi(alpha) = max_q (S_q + alpha*L_q - alpha^2*A_q)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = np.correlate(d, 1.0 - h, mode='full') * dx - np.correlate(h, d, mode='full') * dx
    C = np.correlate(d, d, mode='full') * dx
    
    # Safe interval
    t_lo, t_hi = 0.0, max_alpha
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, 0.0)
    t_hi = min(t_hi, max_alpha)
    if t_hi <= t_lo:
        return 0.0, _exact_c5(h)
    
    # Grid search + golden section
    def phi(a):
        return float(np.max(S0 + a * T - a * a * C))
    
    # Coarse grid
    n_grid = 32
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([phi(a) for a in grid])
    best_i = int(np.argmin(vals))
    
    # Refine with golden section
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = phi(c1), phi(c2)
    for _ in range(50):
        if b - a < 1e-14: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = phi(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = phi(c2)
    
    opt_t = (a + b) / 2.0
    opt_val = phi(opt_t)
    if opt_val > vals[best_i]:
        opt_t, opt_val = grid[best_i], vals[best_i]
    
    return float(opt_t), float(opt_val)

def _lp_topk(h, scores, topk, radius):
    """LP direction using only top-K lags."""
    n = len(h)
    idx = np.argsort(-scores)[:topk]
    G = _grad_matrix_select(h, idx)
    S_top = scores[idx]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize u
    A_ub = np.zeros((topk, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -S_top
    A_eq = np.ones((1, n + 1))
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except:
        pass
    return np.zeros(n)

def _smooth_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = scores.max()
    # Stable log-sum-exp
    shifted = (scores - M) / tau
    Z = np.sum(np.exp(shifted))
    return M + tau * np.log(Z)

def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = scores.max()
    shifted = (scores - M) / tau
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    w = exp_s / Z  # weights
    
    # Weighted sum of gradients
    g = np.zeros(n)
    k_max = n - 1
    for q in range(2 * n - 1):
        if w[q] < 1e-10:
            continue
        k = q - k_max
        g += w[q] * _lag_grad(h, q)
    return g

def _resample(h, n_new):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _lag_grad(h, lag_idx, dx):
    n = len(h)

    g = np.zeros(n)

        m = n - k
        if m > 0:
            g[:m] += dx * (1.0 - h[k:])
            g[k:] += -dx * h[:m]
    else:
        d = -k
        m = n - d
        if m > 0:
            g[:m] += -dx * h[d:]
            g[d:] += dx * (1.0 - h[:m])

            continue
        alpha, new_c5, new_scores = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13 and alpha > 1e-14:
            h_new = h + alpha * p
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < best_c5 - 1e-14:
                best_c5 = actual_c5
                best_h = h_new.copy()
            h = h_new
            cur_scores = _scores(h)
            cur_c5 = actual_c5
            stall = 0
            ratio = cur_c5 / max(new_c5, 1e-15)
            if ratio < 0.85:
                radius = min(radius * 1.08, 0.35)
            elif ratio > 1.3:
                radius *= 0.82
            else:
                radius = min(radius * 1.01, 0.30)
        else:
            stall += 1
            radius *= 0.72
            if stall > 50:
                if radius < 0.01:
                    break
                h = h + rng.normal(0, 0.015, n)
                h = _project(h, target)
                cur_scores = _scores(h)
                cur_c5 = float(np.max(cur_scores))
                stall = 0
                radius = 0.1
    return best_h, best_c5

def _pair_refine(h, deadline, seed=42):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 40
    while time.time() < deadline and stall < max_stall:
        interior = np.where((h > 0.005) & (h < 0.995))[0]
        if len(interior) < 2:
            break
        improved = False
        n_pairs = min(200, len(interior) * (len(interior) - 1) // 2)
        indices = np.arange(len(interior))
        rng.shuffle(indices)
        pairs_checked = 0
        for ii in range(len(interior)):
            if time.time() > deadline or pairs_checked >= n_pairs:
                break
            i = interior[ii]
            for jj in range(ii + 1, len(interior)):
                if time.time() > deadline or pairs_checked >= n_pairs:
                    break
                j = interior[jj]
                pairs_checked += 1
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try, _ = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-13 and alpha > 1e-14:
                    h_new = h + alpha * d
                    h_new = _project(h_new, target)
                    actual = _exact_c5(h_new)
                    if actual < best_c5 - 1e-14:
                        best_c5 = actual
                        best_h = h_new.copy()
                        h = h_new
                        improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = scores.max()
    shifted = (scores - M) / tau
    max_sh = shifted.max()
    exp_s = np.exp(shifted - max_sh)
    Z = np.sum(exp_s)
    w = exp_s / Z
    g = np.zeros(n)
    for q in range(2 * n - 1):
        if w[q] < 1e-12:
            continue
        g += w[q] * _lag_grad(h, q, dx)
    val = M + tau * (max_sh + np.log(Z))
    return val, g

def _smooth_optimize(h0, deadline, seed=42):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = 0.03
    alpha = 0.1
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        val, g = _smooth_obj_grad(h, tau)
        g_tan = g - g.mean()
        gn = np.linalg.norm(g_tan)
        if gn < 1e-10:
            break
        g_norm = g_tan / gn
        for _bs in range(30):
            h_try = _project(h - alpha * g_norm, target)
            val_try, _ = _smooth_obj_grad(h_try, tau)
            if val_try < val:
                break
            alpha *= 0.6
        h = _project(h - alpha * g_norm, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
        tau = max(0.003, tau * 0.97)
        if alpha < 1e-6:
            alpha = 0.05
            h = h + rng.normal(0, 0.01, n)
            h = _project(h, target)
        else:
            alpha = min(alpha * 1.03, 0.5)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 4:
        n_new = 4
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, seed=42):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.006
    T_min = 0.000005
    cooling = 0.9998
    rng = np.random.default_rng(seed)
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])
    patience = 0
    while time.time() < deadline and T > T_min:
        if patience > 30000:
            T = max(T_min, T * 0.97)
            patience = 0
        batch = 50
        for _ in range(batch):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = list(np.where(h > 0.5)[0])
                zeros_list = list(np.where(h <= 0.5)[0])
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _structured_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 6))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 122
        h0 = np.zeros(n0)
        h0[::2][:n0 // 2] = 1.0
        h0 = _project(h0, n0 / 2.0)

    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick LP on initial (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(h0, d1, topk=12, radius_init=0.15, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-start exploration (20%)
    d2 = t0 + budget_s * 0.28
    starts = [best_h.copy()]
    x = np.arange(n0)
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        hs = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n0 + 0.7 * freq)
        starts.append(_project(hs, n0 / 2.0))
    for _ in range(4):
        starts.append(_structured_init(n0, rng))
    for s in starts:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.018)
        hs, c5s = _lp_loop(s, sub_dl, topk=12, radius_init=0.12, seed=seed + 1)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = hs.copy()

    # Phase 3: Main LP (28%)
    d3 = t0 + budget_s * 0.56
    h3, c5_3 = _lp_loop(best_h, d3, topk=15, radius_init=0.12, seed=seed + 2)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + polish (12%)
    d4 = t0 + budget_s * 0.68
    h4a, c5_4a = _binary_sa(best_h, min(d4, t0 + budget_s * 0.58), seed=seed + 3)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, topk=10, radius_init=0.08, seed=seed + 4)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Smooth optimization (10%)
    d5 = t0 + budget_s * 0.78
    h5, c5_5 = _smooth_optimize(best_h, d5, seed=seed + 5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < d5 + budget_s * 0.02:
        h5b, c5_5b = _lp_loop(best_h, min(d5 + budget_s * 0.02, time.time() + budget_s * 0.03),
                              topk=12, radius_init=0.06, seed=seed + 6)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Pair refinement (8%)
    d6 = t0 + budget_s * 0.86
    h6, c5_6 = _pair_refine(best_h, d6, seed=seed + 7)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Resolution sweep (6%)
    d7 = t0 + budget_s * 0.92
    bn = len(best_h)
    tried = set()
    for dn in [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20]:
        if time.time() > d7:
            break
        n_try = bn + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, topk=12, radius_init=0.10, seed=seed + n_try)
        if c5_r < best_c5 - 1e-12 and time.time() < d7 - 2:
            h_r, c5_r2 = _pair_refine(h_r, min(d7, time.time() + budget_s * 0.01), seed=seed + n_try + 1)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Final fine LP (remaining)
    d8 = deadline
    h8, c5_8 = _lp_loop(best_h, d8, topk=25, radius_init=0.04, seed=seed + 100)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 < 0:
        final_c5 = best_c5
        best_h = _project(best_h, len(best_h) / 2.0)
    return best_h, final_c5, len(best_h)
