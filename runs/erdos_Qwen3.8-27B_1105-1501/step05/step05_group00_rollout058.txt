    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

    if abs(res) > 1e-13:

        else:

def _line_search(h, d, dx, trust_radius):

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find feasible interval
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-15:
        return 0.0, float(np.max(S0)) * dx
    
    # Golden section with grid initialization
    grid = np.linspace(t_lo, t_hi, 32)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None]**2) * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t, best_val = grid[best_i], obj[best_i]
    
    # Refine with golden section

            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt_t = (a+b)/2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
