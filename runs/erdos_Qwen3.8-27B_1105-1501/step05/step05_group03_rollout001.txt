        best_c5 = mid_c5
    return best_a, best_c5

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=5000, radius_min=0.002):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None:
            radius *= 0.6
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.03, 0.35)
        else:
            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, rng):
    """Convert continuous h to binary with exact count."""
    n = len(h)
    target = n // 2
    # Sort by value, threshold
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def _bit_scores(h_bin):
    """Compute all lag scores for a binary sequence efficiently."""
    n = len(h_bin)
    dx = 2.0 / n
    # Use correlate
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return corr * dx

def _bit_sa(h_init, deadline, n_iters=50000, T_init=0.05, T_min=0.001):
    """Simulated annealing on binary sequences with swap moves."""
    n = len(h_init)
    target = n // 2
    h = _to_binary(h_init, None)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T = T_init
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Swap a 1 and a 0
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or np.random.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _block_move_sa(h_init, deadline, n_iters=20000, block_sizes=(2,3,4,5), T_init=0.03):
    """SA with block moves (inverting contiguous blocks)."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    T_min = 0.0005
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Try a block move: flip a contiguous block
        bs = block_sizes[np.random.randint(len(block_sizes))]
        if n > bs:
            start = np.random.randint(0, n - bs + 1)
            h_new = h.copy()
            h_new[start:start+bs] = 1.0 - h_new[start:start+bs]
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 or np.random.random() < np.exp((cur_c5 - new_c5) / T):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        T *= cool
    
    return best_h, best_c5
