    F = _scores(h)
    # Clip to avoid negative values (shouldn't happen but for safety)

        grad = np.zeros(len(h))
        return val, grad

    G = _build_G(h)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

    F = np.maximum(F, 0.0)
    Fq = F ** q
    s = float(np.sum(Fq))
    if s < 1e-300:
        return 0.0
    return s ** (1.0 / q)

    F = _scores(h)
    F = np.maximum(F, 0.0)
    Fq = F ** q
    s = float(np.sum(Fq))
    if s < 1e-300:
        return np.zeros(len(h))
    Fqm1 = F ** (q - 1)
    G = _build_G(h)
    return (G.T @ Fqm1) * (s ** (1.0 / q - 1.0))

    if Fmax < 1e-300:
        return 0.0
    shifted = beta * (F - Fmax)
    es = np.exp(shifted)
    return Fmax + float(np.log(np.sum(es))) / beta

                    h[k] -= alpha

                        best_c5 = cur_c5
                        best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

_eq_target = 0.0
_cur_q = 4.0
_cur_beta = 20.0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_obj_fn(h, q):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Fq = F ** q
    s = float(np.sum(Fq))
    if s < 1e-300:
        return 0.0
    return s ** (1.0 / q)

def _pnorm_grad_fn(h, q):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Fq = F ** q
    s = float(np.sum(Fq))
    if s < 1e-300:
        return np.zeros(len(h))
    Fqm1 = F ** (q - 1)
    G = _build_G(h)
    return (G.T @ Fqm1) * (s ** (1.0 / q - 1.0))

def _lse_obj_fn(h, beta):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Fmax = float(F.max())
    if Fmax < 1e-300:
        return 0.0
    shifted = beta * (F - Fmax)
    es = np.exp(shifted)
    return Fmax + float(np.log(np.sum(es))) / beta

def _lse_grad_fn(h, beta):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Fmax = float(F.max())
    if Fmax < 1e-300:
        return np.zeros(len(h))
    shifted = beta * (F - Fmax)
    es = np.exp(shifted)
    w = es / float(np.sum(es))
    G = _build_G(h)
    return G.T @ w

def _w_obj_p(x):
    return _pnorm_obj_fn(x, _cur_q)

def _w_grad_p(x):
    return _pnorm_grad_fn(x, _cur_q)

def _w_obj_l(x):
    return _lse_obj_fn(x, _cur_beta)

def _w_grad_l(x):
    return _lse_grad_fn(x, _cur_beta)

def _eq_fun(x):
    return float(np.sum(x) - _eq_target)

def _eq_jac(x):
    return np.ones(len(x))

def _slsqp_pnorm(h0, deadline, q_list, maxiter):
    global _eq_target, _cur_q
    n = len(h0)
    _eq_target = n / 2.0
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for qi in q_list:
        if time.time() > deadline:
            break
        _cur_q = float(qi)
        try:
            res = minimize(_w_obj_p, best_h, jac=_w_grad_p, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, n / 2.0)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def _slsqp_lse(h0, deadline, beta_list, maxiter):

    n = len(h0)
    _eq_target = n / 2.0
    best_h = h0.copy()

                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):

                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-15:

                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def _proj_grad(h0, deadline, q, n_steps, alpha_init):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    h = h0.copy()
    alpha = alpha_init
    stall = 0
    for step in range(n_steps):

        grad = _pnorm_grad_fn(h, q)
        grad = grad - float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        improved = False
        a = alpha

            obj_new = _pnorm_obj_fn(h_new, q)

                improved = True
                break

            alpha *= 0.7

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, noise, rng):
    x = (np.arange(n) + 0.5) / n
    h = ((x * alpha) % 1.0).astype(float)
    h += rng.normal(0, noise, n)
    return _project(h, n / 2.0)

def _sin_init(n, k, phase, noise, rng):
    x = (np.arange(n) + 0.5) / n
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * k * x + phase)
    h += rng.normal(0, noise, n)
    return _project(h, n / 2.0)

def _pair_descent(h, deadline, max_passes, thresh):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                a = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + a * C))
                if cur_c5 - pred > thresh and abs(a) > 1e-10:
                    h[j] += a
                    h[k] -= a
                    h = np.clip(h, 0.0, 1.0)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                    else:
                        h[j] -= a
                        h[k] += a
        if not improved:
            break
    return best_h, best_c5

def _perturb_restart(best_h, best_c5, deadline, rng, n_attempts, max_time_per):
    n = len(best_h)

    for _ in range(n_attempts):

            best_h = h_try.copy()
