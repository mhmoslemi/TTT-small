    """Find direction that equalizes active lags (least-squares)."""
    n = len(h)

    active = np.where(scores >= s_max - tol * (2.0 / n))[0]

    # We want to find d (zero-mean, bounded) that minimizes max_r (S0[r] + d·grad_r)

    if c5_4 < best_c5:

        best_h = h4.copy()

    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-14:
        mask = (x < 1.0 - 1e-14) if res > 0 else (x > 1e-14)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _line_search(h, d, trust_radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0-h, mode='full')
    T = np.correlate(1.0-h, d, mode='full') - np.correlate(d, h, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15; neg = d < -1e-15
    a_hi = trust_radius; a_lo = -trust_radius
    if np.any(pos): a_hi = min(a_hi, float(np.min((1-h[pos])/d[pos])))
    if np.any(neg): a_lo = max(a_lo, float(-np.min(h[neg]/(-d[neg]))))
    if a_hi - a_lo < 1e-13: return 0.0, float(np.max(S0)*dx)
    
    gr = (np.sqrt(5.0)-1.0)/2.0
    a, b = a_lo, a_hi
    c = b - gr*(b-a); dd = a + gr*(b-a)
    fc = float(np.max(S0+c*T-c*c*C)); fd = float(np.max(S0+dd*T-dd*dd*C))
    for _ in range(35):
        if b-a < 1e-11: break
        if fc < fd: b,dd,fd = dd,c,fc; c = b-gr*(b-a); fc = float(np.max(S0+c*T-c*c*C))
        else: a,c,fc = c,dd,fd; dd = a+gr*(b-a); fd = float(np.max(S0+dd*T-dd*dd*C))
    ba=(a+b)/2; bv=float(np.max(S0+ba*T-ba*ba*C))*dx
    for ta in [a_lo,a_hi]:
        tv=float(np.max(S0+ta*T-ta*ta*C))*dx
        if tv<bv: ba,bv=ta,tv
    return ba,bv

def _build_G(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        dl = r-(n-1)
        lo1=max(0,-dl); hi1=min(n,n-dl)
        if hi1>lo1: G[r,lo1:hi1]+=(1.0-h[lo1+dl:hi1+dl])
        lo2=max(0,dl); hi2=min(n,n+dl)
        if hi2>lo2: G[r,lo2:hi2]-=h[lo2-dl:hi2-dl]
    return G

def _lp_dir(h, radius, G=None, penalty_lags=None, pw=0.04):
    n = len(h)
    if G is None: G = _build_G(h)
    scores = np.correlate(h, 1.0-h, mode='full')
    c = np.zeros(n+1); c[-1]=1.0
    A_ub = np.zeros((2*n-1, n+1)); A_ub[:,:n]=G; A_ub[:,n]=-1.0
    b_ub = scores.copy()
    if penalty_lags is not None:
        b_ub[penalty_lags] -= pw
    A_eq = np.zeros((1,n+1)); A_eq[0,:n]=1.0
    bounds=[(max(-radius,-h[i]),min(radius,1-h[i])) for i in range(n)]
    bounds.append((None,None))
    try:
        res = linprog(c,A_ub=A_ub,b_ub=b_ub,A_eq=A_eq,b_eq=[0.0],
                     bounds=bounds,method='highs',options={'time_limit':5})
        if res.success and res.x[:n].std()>1e-14: return res.x[:n]
    except: pass
    return None

def _lse_dir(h, beta):
    n=len(h); dx=2.0/n
    sc=np.correlate(h,1.0-h,mode='full')*dx
    v=beta*(sc-sc.max()); ew=np.exp(v); w=ew/ew.sum()
    g=np.zeros(n)
    for r in range(2*n-1):
        if w[r]<1e-9: continue
        dl=r-(n-1)
        lo1=max(0,-dl); hi1=min(n,n-dl)
        if hi1>lo1: g[lo1:hi1]+=w[r]*(1.0-h[lo1+dl:hi1+dl])
        lo2=max(0,dl); hi2=min(n,n+dl)
        if hi2>lo2: g[lo2:hi2]-=w[r]*h[lo2-dl:hi2-dl]
    g-=g.mean()
    return g

def _optimize(h, deadline, mode='lp', lr_init=0.03, radius_init=0.15):

        if mode in ('lp','alt'):

            if d_dir is None:

        else:
