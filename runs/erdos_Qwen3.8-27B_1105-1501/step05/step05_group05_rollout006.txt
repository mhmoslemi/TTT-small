    K = len(active_idx)
    target = n / 2.0
    scores = score_vector(h)
    t_init = float(np.max(scores)) + 1e-8
    
    # Precompute gradients at initial point for speed
    # But SLSQP will evaluate at different points, so we need the Jacobian function
    
    x0 = np.concatenate([h, [t_init]])
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def con_fun(x):
        h_x = x[:n]
        t_x = x[-1]
        vals = np.empty(K)
        for i, r in enumerate(active_idx):
            vals[i] = t_x - score_at_r(h_x, r)
        return vals
    
    def con_jac(x):
        h_x = x[:n]
        J = np.empty((K, n + 1))
        for i, r in enumerate(active_idx):
            J[i, :n] = -grad_score_r(h_x, r)
            J[i, n] = 1.0
        return J
    
    bounds = [(max(0.0, h[j] - radius), min(1.0, h[j] + radius)) for j in range(n)]
    bounds.append((0.0, t_init + 0.005))
    
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    beq = np.array([target])
    
    constraints = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, x0, jac=obj_jac, bounds=bounds,
                      constraints=constraints, Aeq=Aeq, beq=beq,
                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14})
    except Exception:
        return h.copy(), exact_c5(h)
    
    if hasattr(res, 'x') and res.x is not None:
        h_new = project(res.x[:n], target)
        c5_new = exact_c5(h_new)
        if c5_new < exact_c5(h) - 1e-14:
            return h_new, c5_new
    return h.copy(), exact_c5(h)
