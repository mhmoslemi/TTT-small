import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-10:
        if resid > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
    return np.clip(x, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n-1, :] = (1.0 - 2.0 * h) * dx
    for d in range(1, n):
        m = n - d
        G[n-1-d, :m] += (1.0 - h[d:d+m]) * dx
        G[n-1-d, d:n] += -h[:m] * dx
        G[n-1+d, d:n] += (1.0 - h[:m]) * dx
        G[n-1+d, :m] += -h[d:d+m] * dx
    return G

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_one_step(h, radius, use_all=True, topk=None):
    """One LP step: find direction p and target t to reduce C5."""
    n = len(h)
    scores = _scores(h)
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    
    if not use_all and topk is not None and topk < n_lags:
        idx = np.argsort(scores)[-topk:]
        G_k = G[idx]
        scores_k = scores[idx]
    else:
        G_k = G
        scores_k = scores
    
    n_constraints = len(scores_k)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G_k, -np.ones((n_constraints, 1))])
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        if lo_j > hi_j:
            lo_j = hi_j = 0.0
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search(h, p, n_eval=15):
    """Golden section line search along h + alpha * p."""
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(h + c1 * p)
    f2 = _c5(h + c2 * p)
    for _ in range(n_eval):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
