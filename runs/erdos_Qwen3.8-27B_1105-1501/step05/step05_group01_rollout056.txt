        pass
    return None

def _block_move(h, deadline, block_size=4, n_grid=20):
    """Move mass from a block of positions to another block."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(55555)
    stall = 0
    while time.time() < deadline:
        if stall > 400:
            break
        # Pick two blocks
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            stall += 1
            continue
        # Direction: increase block i, decrease block j (or vice versa)
        direction_sign = 1.0 if rng.random() < 0.5 else -1.0
        d = np.zeros(n)
        d[i:i+block_size] = direction_sign / block_size
        d[j:j+block_size] = -direction_sign / block_size
        alpha, c5_try = _line_search(h, d, dx, 0.8, n_grid=n_grid)
        if c5_try < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _basin_hopping(h, deadline, n_grid=25):
    """Simulated annealing with controlled perturbations."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(4242)
    
    T_start = 0.015
    T_end = 0.001
    n_steps = 5000
    T = T_start
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Perturb: random direction with sum=0
        pert = rng.normal(0, 0.03, n)
        pert -= pert.mean()  # ensure sum = 0
        h_pert = h + pert
        h_pert = _project(h_pert, target)
        
        # Quick LP optimization from perturbed point
        sub_dl = min(deadline, time.time() + 2.0)
        h_new, c5_new = _lp_optimize(h_pert, sub_dl, radius_init=0.08, n_grid=15, max_stall=20)
        
        cur_c5 = _exact_c5(h)
        delta = c5_new - cur_c5
        T = T_start * (T_end / T_start) ** (step / n_steps)
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    h = np.zeros(n)
    h[idx] = 1.0

def _exact_c5(h):
    n = len(h)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius, n_grid=50):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_hi = max(t_hi, 0.0)
    
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid scan
    if n_grid > 2:
        grid_t = np.linspace(0.0, t_hi, n_grid)
        # Vectorized computation
        vals = S0[None, :] + grid_t[:, None] * T[None, :] - (grid_t[:, None]**2) * C[None, :]
        grid_scores = np.max(vals, axis=1) * dx
        best_idx = int(np.argmin(grid_scores))
        best_val = float(grid_scores[best_idx])
        best_t = float(grid_t[best_idx])
        
        # Check endpoint
        val_hi = float(np.max(S0 + t_hi * T - t_hi*t_hi * C)) * dx
        if val_hi < best_val:
            best_val = val_hi
            best_t = t_hi
        
        # Bracket for golden section
        if best_idx == 0:
            lo, hi = 0.0, float(grid_t[min(1, n_grid-1)])
        elif best_idx == n_grid - 1:
            lo, hi = float(grid_t[n_grid-2]), t_hi
        else:
            lo, hi = float(grid_t[best_idx-1]), float(grid_t[best_idx+1])
    else:
        lo, hi = 0.0, t_hi
        best_val = float(np.max(S0) * dx)
        best_t = 0.0
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(35):

        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)

    if opt_val < best_val:
        best_val = opt_val
        best_t = opt_t
    
    return best_t, best_val

    G = np.zeros((2 * n - 1, n))

        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G
