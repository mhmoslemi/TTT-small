def _project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:

    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else:

    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_grad(h, tau, dx):
    """Gradient of softmax-surrogated max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = corr.max()
    e = np.exp(np.clip((corr - M) / tau, -50, 50))
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n-1+k]
        if wk < 1e-12: continue
        if k == 0:
            grad += wk * (1.0 - 2.0*h)
        elif k > 0:
            m = n - k
            grad[:m] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:m]
        else:
            d = -k
            m = n - d
            grad[d:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[d:]
    return dx * grad

def _grad_descent(h, deadline, tau0=3.0, tau1=0.05):
    n = len(h)
    dx = 2.0/n
    target = n/2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    c5_cur = best_c5
    n_iter = 0
    lr = 0.05
    
    while time.time() < deadline:
        n_iter += 1
        tau = tau0 * (tau1/tau0) ** (min(n_iter, 300)/300.0)
        g = _soft_grad(h, tau, dx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10: break
        
        # Backtracking
        step = lr / g_norm
        for _ in range(30):
            h_new = _project(h - step * g, target)
            c5_new = _exact_c5(h_new)
            if c5_new < c5_cur:
                h = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
                lr = min(lr * 1.05, 0.5)
                break
            step *= 0.6
        else:
            lr *= 0.8
            if lr < 1e-6: break
    
    return best_h, best_c5

def _all_pairs_search(h, deadline, max_passes=200):
    """Exhaustive all-pairs swap search for binary solutions."""
    n = len(h)
    target = n/2.0
    h = h.copy()
    # Binarize
    h_bin = _project(h, target)
    # Threshold
    h_bin[h_bin > 0.5] = 1.0
    h_bin[h_bin < 0.5] = 0.0
    # Fix sum
    s = h_bin.sum()
    while s > target:
        idx = int(np.argmax(h_bin * (np.arange(n) % 7 + 1)))  # deterministic
        # Find a 1 to flip
        ones = np.where(h_bin == 1)[0]
        if len(ones) == 0: break
        h_bin[ones[0]] = 0
        s -= 1
    while s < target:
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) == 0: break
        h_bin[zeros[0]] = 1
        s += 1
    
    best_h, best_c5 = h_bin.copy(), _exact_c5(h_bin)
    cur_c5 = best_c5
    
    for _pass in range(max_passes):
        if time.time() > deadline: break
        improved = False
        # Try all pairs (i, j) where h[i]=1, h[j]=0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0: break
        
        best_delta = -1e-14
        best_pair = None
        for i in ones:
            for j in zeros:
                h_bin[i] = 0
                h_bin[j] = 1
                c5_try = _exact_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = cur_c5 - c5_try
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)

            i, j = best_pair
            h_bin[i] = 0
            h_bin[j] = 1
            cur_c5 -= best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            improved = True
