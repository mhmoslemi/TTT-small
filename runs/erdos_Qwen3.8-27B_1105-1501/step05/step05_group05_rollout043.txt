def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    # Maximize t s.t. G*p - t <= -s, sum(p)=0, |p|<=radius
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(np.clip(v, -50, 0))
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])

            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_general(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (5.0**0.5 - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val > float(np.max(S0)) * dx:
        return 0.0, float(np.max(S0)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _lp_direction(h, scores, G, radius, penalty_lags=None):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None:
        for k in penalty_lags:
            b_ub[k] -= 0.05
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.22, max_iter=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        G = _grad_matrix(h)
        penalty = None
        if stall > 8:
            thresh = s_max * 0.99
            penalty = list(np.where(scores >= thresh)[0])
        p = _lp_direction(h, scores, G, radius, penalty)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search_general(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.07, 0.5)
        else:
            stall += 1
            radius *= 0.72
            if stall > 30 and radius < 0.03:
                radius = max(best_radius * 0.4, 0.05)
                stall = 0
    return best_h, best_c5

def _pair_transfer(h, i, j, dx, S0):
    """Compute optimal alpha for transferring mass from position i to position j."""
    max_alpha = min(h[i], 1.0 - h[j])
    if max_alpha < 1e-13:
        return 0.0, float(np.max(S0)) * dx
    d = np.zeros(len(h))
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    gr = (5.0**0.5 - 1.0) / 2.0
    a, b = 0.0, max_alpha
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_a = (a + b) / 2.0
    opt_v = float(np.max(S0 + opt_a * T - opt_a * opt_a * C)) * dx
    if opt_v >= float(np.max(S0)) * dx:
        return 0.0, float(np.max(S0)) * dx
    return opt_a, opt_v

def _exhaustive_pair_refine(h, deadline, max_rounds=5000):
    """Try all (giver, receiver) pairs each round until convergence."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    eps = 1e-8

    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 15:
            break
        S0 = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        
        givers = np.where(h > eps)[0]
        receivers = np.where(h < 1.0 - eps)[0]
        
        for i in givers:
            if time.time() > deadline:
                break
            for j in receivers:
                if i == j:
                    continue
                alpha, c5_try = _pair_transfer(h, i, j, dx, S0)
                if alpha > 1e-11 and c5_try < best_c5 - 1e-13:
                    d = np.zeros(n)
                    d[i] = -1.0
                    d[j] = 1.0
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        S0 = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _priority_pair_refine(h, deadline, max_rounds=3000):
    """Pair refinement prioritizing pairs that affect the most active lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    eps = 1e-8

    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 20:
            break
        
        S0 = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(S0))
        improved = False
        
        # Rank positions by "importance" - how much they affect active lags
        givers = np.where(h > eps)[0]
        receivers = np.where(h < 1.0 - eps)[0]
        
        # Try all pairs
        for i in givers:
            if time.time() > deadline:
                break
            for j in receivers:
                if i == j:
                    continue
                alpha, c5_try = _pair_transfer(h, i, j, dx, S0)
                if alpha > 1e-11 and c5_try < best_c5 - 1e-13:
                    d = np.zeros(n)
                    d[i] = -1.0
                    d[j] = 1.0
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        S0 = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _smooth_descent(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.02
    mom = np.zeros(n)
    decay = 0.8
    stall = 0
    bi = 0
    beta_list = (30, 80, 200, 500)

    while time.time() < deadline:
        if stall > 40:
            bi += 1
            if bi >= len(beta_list) * 4:
                break
            stall = 0
            mom = np.zeros(n)
            lr = 0.02
        beta = beta_list[bi % len(beta_list)]
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        s_max = float(np.max(scores))
        v = beta * (scores - s_max)
        exp_v = np.exp(np.clip(v, -50, 0))
        w = exp_v / float(np.sum(exp_v))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-14:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:d + m])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (-h[g:g + m])
                    grad[g:] += w[r] * (1.0 - h[:m])
        grad -= np.mean(grad)
        dn = np.linalg.norm(grad)
        if dn < 1e-13:
            bi += 1
            continue
        grad = grad / dn
        mom = decay * mom + (1.0 - decay) * grad
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search_general(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.3)
        else:
            stall += 1
            lr *= 0.75
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kick_reopt(h, deadline, rng, n_kicks=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.01, 0.05)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 2.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.10, max_iter=100)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _binary_threshold_search(h, deadline):
    """Try different binary thresholds to find a good binary solution."""
    n = len(h)
    target = n // 2
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    sorted_idx = np.argsort(h)
    for k in range(max(0, target - 8), min(n, target + 8)):
        if time.time() > deadline:
            break
        binary = np.zeros(n)
        binary[sorted_idx[k:]] = 1.0
        cur = _exact_c5(binary)
        if cur < best_c5 - 1e-14:
            best_c5 = cur
            best_h = binary.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 5.0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        x = np.arange(n0) / n0
        h0 = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x) + 0.15 * np.sin(2 * np.pi * 7 * x)

    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LP optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.25, max_iter=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smooth descent (10%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _smooth_descent(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP again (10%)
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.15, max_iter=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Heavy pair refinement (25%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _exhaustive_pair_refine(best_h, d4, max_rounds=5000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution search (20%)
    d5 = t0 + budget_s * 0.85
    best_n = len(best_h)
    tried_n = set()
    n_range = sorted(set(range(best_n - 12, best_n + 13)), key=lambda x: abs(x - best_n))
    for n_try in n_range:
        if time.time() > d5:
            break
        if n_try < 80 or n_try > 180 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + 3.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.15, max_iter=120)
        if c5_r < best_c5 + 0.002 and time.time() < d5 - 2:
            sub_dl2 = min(d5, time.time() + 4.0)
            h_r, c5_r = _exhaustive_pair_refine(h_r, sub_dl2, max_rounds=800)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Kick and reoptimize (7%)
    d6 = t0 + budget_s * 0.92
    h6, c5_6 = _kick_reopt(best_h, d6, rng, n_kicks=25)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final exhaustive pair refinement (8%)
    d7 = deadline
    h7, c5_7 = _exhaustive_pair_refine(best_h, d7, max_rounds=8000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final: binary threshold and tiny LP polish
    if time.time() < deadline - 2:
        h8, c5_8 = _binary_threshold_search(best_h, deadline - 1)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
        if time.time() < deadline:
            h9, c5_9 = _lp_loop(best_h, deadline, radius_init=0.05, max_iter=100)
            if c5_9 < best_c5:
                best_c5 = c5_9
                best_h = h9.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
