import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5, n_samp=12):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(15):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lse_optimize(h, deadline, beta_init=20.0, beta_final=500.0, n_phases=6):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    def make_obj(beta):
        def obj(hf):
            S = _scores(hf)
            m = np.max(S)
            shifted = beta * (S - m)
            shifted = np.clip(shifted, -500, 0)
            return m + np.log(np.mean(np.exp(shifted))) / beta
        return obj
    
    def make_grad(beta):
        def grad(hf):
            S = _scores(hf)
            m = np.max(S)
            shifted = beta * (S - m)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            wsum = np.sum(w)
            if wsum < 1e-300:
                idx = int(np.argmax(S))
                w = np.zeros(len(S))
                w[idx] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w
        return grad
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        obj = make_obj(beta)
        grad = make_grad(beta)
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq',
                                        'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 150, 'ftol': 1e-14})

                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=3):

                        f2 = _exact_c5(h2)

                    a_opt = (lo_a + hi_a) / 2.0

                    h_try = h.copy()
                    h_try[j] += sign * a_opt

                        best_h = h_try.copy()

        h2b, c5_2b = _local_optimize_lp(best_h, dl2b, radius_init=0.1, max_iters=2000, stall_limit=100)
        if c5_2b < best_c5:
            best_c5 = c5_2b; best_h = h2b.copy()

    # Phase 3: LSE optimization with better annealing (15%)
    if time.time() < t0 + budget_s * 0.38:
        dl3 = t0 + budget_s * 0.38
        h3, c5_3 = _lse_optimize(best_h, dl3, beta_init=15.0, beta_final=400.0, n_phases=6)
        if c5_3 < best_c5:
            best_c5 = c5_3; best_h = h3.copy()
        if time.time() < dl3:
            dl3b = min(dl3, time.time() + budget_s * 0.03)
            h3b, c5_3b = _local_optimize_lp(best_h, dl3b, radius_init=0.06, max_iters=1500, stall_limit=80)
            if c5_3b < best_c5:
                best_c5 = c5_3b; best_h = h3b.copy()

    # Phase 4: Multi-resolution with diverse inits (20%)
    if time.time() < t0 + budget_s * 0.58:
        dl4 = t0 + budget_s * 0.58
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.7]
        n_trials = [96, 128, 160, 64, 192, 256]
        trial_count = 0
        for n_try in n_trials:
            if time.time() > dl4:
                break
            trial_count += 1
            # Resample from best
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl4, time.time() + budget_s * 0.035)
            ht, ct = _local_optimize_lp(h_init, sub_dl, radius_init=0.2, max_iters=2500, stall_limit=100)
            if ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try
            # Quasi-periodic
            if time.time() < dl4:
                a_val = alphas[trial_count % len(alphas)]
                h_qp = _quasi_periodic_init(n_try, a_val)
                h_qp = h_qp + rng.normal(0, 0.08, n_try)
                h_qp = _project(h_qp, n_try / 2.0)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.025)
                ht2, ct2 = _local_optimize_lp(h_qp, sub_dl2, radius_init=0.2, max_iters=1500, stall_limit=80)
                if ct2 < best_c5:
                    best_c5 = ct2; best_h = ht2.copy(); best_n = n_try
                # Quick LSE polish
                if time.time() < dl4 and ct2 < best_c5 + 0.005:
                    sub_dl3 = min(dl4, time.time() + budget_s * 0.015)
                    ht3, ct3 = _lse_optimize(ht2, sub_dl3, beta_init=20, beta_final=150, n_phases=3)
                    if ct3 < best_c5:
                        best_c5 = ct3; best_h = ht3.copy(); best_n = n_try
            # Thue-Morse-like
            if time.time() < dl4:
                h_tm = _thue_morse_init(n_try, rng)
                sub_dl4 = min(dl4, time.time() + budget_s * 0.02)
                ht4, ct4 = _local_optimize_lp(h_tm, sub_dl4, radius_init=0.2, max_iters=1200, stall_limit=80)
                if ct4 < best_c5:
                    best_c5 = ct4; best_h = ht4.copy(); best_n = n_try

    # Phase 5: Perturbation restarts (15%)
    if time.time() < t0 + budget_s * 0.73:
        dl5 = t0 + budget_s * 0.73
        scale = 0.015
        for attempt in range(500):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.01)
            ht, ct = _local_optimize_lp(h_pert, sub_dl, radius_init=0.05, max_iters=400, stall_limit=40)
            if ct < best_c5 - 1e-13:
                best_c5 = ct; best_h = ht.copy()
                scale = max(0.002, scale * 0.85)
            else:
                scale = min(0.1, scale * 1.12)

    # Phase 6: LSE final polish (10%)
    if time.time() < t0 + budget_s * 0.85:
        dl6 = t0 + budget_s * 0.85
        h6, c5_6 = _lse_optimize(best_h, dl6, beta_init=50.0, beta_final=600.0, n_phases=4)
        if c5_6 < best_c5:
            best_c5 = c5_6; best_h = h6.copy()
        if time.time() < dl6:
            h6b, c5_6b = _local_optimize_lp(best_h, t0 + budget_s * 0.88,
                                             radius_init=0.03, max_iters=2000, stall_limit=100)
            if c5_6b < best_c5:
                best_c5 = c5_6b; best_h = h6b.copy()

    # Phase 7: Pair polish (5%)
    if time.time() < t0 + budget_s * 0.94:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.95, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()

    # Phase 8: Final LP micro-polish
    if time.time() < t0 + budget_s * 0.975:
        hf, cf = _local_optimize_lp(best_h, t0 + budget_s * 0.98,
                                     radius_init=0.02, max_iters=3000, stall_limit=200)
        if cf < best_c5:
            best_c5 = cf; best_h = hf.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
