    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP at original resolution (brief)
    dl1 = t0 + budget_s * 0.06
    h1, c5_1 = _local_lp(h_start, dl1, top_k_list=[None, 12, 8],
                         radius_init=0.12, max_iters=1500)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Exact pair descent at original resolution
    dl2 = t0 + budget_s * 0.30
    h2, c5_2 = _pair_descent(best_h, dl2, max_passes=80, thresh_init=1e-7, rng=rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Gradient smoothing
    dl3 = t0 + budget_s * 0.35
    h3, c5_3 = _grad_smooth(best_h, dl3, n_steps=80)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: More pair descent after smoothing
    dl4 = t0 + budget_s * 0.50
    h4, c5_4 = _pair_descent(best_h, dl4, max_passes=100, thresh_init=1e-8, rng=rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Try different resolutions
    dl5 = t0 + budget_s * 0.72
    for n_try in [96, 64, 160, 128]:
        if time.time() > dl5:
            break
        if best_c5 < 0.3806:
            break
        h_init = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + budget_s * 0.08)
        h_try, c5_try = _pair_descent(h_init, sub_dl, max_passes=60,
                                      thresh_init=1e-7, rng=rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            # Follow up with LP
            sub_dl2 = min(dl5, time.time() + budget_s * 0.03)
            h_lp, c5_lp = _local_lp(h_try, sub_dl2, top_k_list=[None, 10],
                                    radius_init=0.10, max_iters=800)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 6: Diverse starts
    dl6 = t0 + budget_s * 0.82
    alphas_qp = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
    for n_try in [128, 96, 64]:
        if time.time() > dl6:
            break
        for attempt in range(3):
            if time.time() > dl6:
                break
            if attempt == 0:
                h_init = _quota_init(n_try, rng)
            else:
                alpha = alphas_qp[attempt % len(alphas_qp)]
                h_init = _quasi_init(n_try, alpha, rng)
            sub_dl = min(dl6, time.time() + budget_s * 0.04)
            h_try, c5_try = _pair_descent(h_init, sub_dl, max_passes=50,
                                          thresh_init=1e-6, rng=rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 7: Perturbation restarts on best
    dl7 = t0 + budget_s * 0.92
    scale = 0.02
    for attempt in range(300):
        if time.time() > dl7:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.006)
        h_try, c5_try = _pair_descent(h_pert, sub_dl, max_passes=30,
                                      thresh_init=1e-8, rng=rng)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 8: Final polish
    if time.time() < hard_deadline:
        dl8 = hard_deadline
        # LP polish
        h_lp, c5_lp = _local_lp(best_h, dl8, top_k_list=[5, 8, None],
                                radius_init=0.03, max_iters=500)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
        # Pair polish
        if time.time() < dl8:
            h_pd, c5_pd = _pair_descent(best_h, dl8, max_passes=50,
                                        thresh_init=1e-9, rng=rng)
            if c5_pd < best_c5:
                best_c5 = c5_pd
                best_h = h_pd.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
