    """Compute gradient of weighted sum of correlation components."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for r in range(2*n-1):
        if abs(weights[r]) < 1e-15:
            continue
        d = r - (n-1)
        if d >= 0:
            # f_r = sum_{i=d}^{n-1} h[i](1-h[i-d])

            grad[d:] += one_minus_h[:m]
            grad[:m] += -h[d:]
        else:

            m = n - g
            grad[:m] += one_minus_h[g:]
            grad[g:] += -h[:m]
    return grad

    """Compute gradient matrix G where G[r, j] = d(corr_r)/d(h_j).
    corr = np.correlate(h, 1-h, mode='full')"""
    n = len(h)

def _softmax_gradient(h, beta):
    """Compute gradient of softmax-smoothed C5."""
    n = len(h)

        w /= w.sum()
        
        # Gradient
        grad = G.T @ w * dx
        grad -= grad.mean()  # sum-zero constraint
        
        # Gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step with backtracking
        accepted = False
        for ls in range(20):
            step = lr / (ls + 1)
            h_try = h - step * grad
            h_try = _project(h_try, target)
            new_c5 = _exact_c5(h_try)
            if new_c5 < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = new_c5
                G = _grad_matrix(h)
                accepted = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                break
        if not accepted:
            stall += 1
            lr *= 0.6
            if stall > 100:
                break
    
    return best_h, best_c5

def _pair_sweep(h_init, deadline, max_sweeps=50):
    """Systematic pair coordinate descent."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    best_c5 = cur_c5
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_refine(h_init, deadline, rng, n_perturbs=30):
    """Try random perturbations + quick refinement."""
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    scale = 0.05
    
    for _ in range(n_perturbs):
        if time.time() > deadline:
            break
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, target)
        # Quick LSE optimization
        h_r, c5_r = _lse_optimize(h_p, min(deadline, time.time() + 30), beta_start=100, beta_end=300)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            scale = max(0.01, scale * 0.7)
        else:
            scale = min(0.15, scale * 1.3)
    
    return best_h, best_c5
