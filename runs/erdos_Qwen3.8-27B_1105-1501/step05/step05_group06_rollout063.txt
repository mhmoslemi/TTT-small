        G[r, :m] -= dx * h[d:]
    for s in range(1, n):
        m = n - s
        r = n - 1 - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] -= dx * h[:m]
    return G


def _smooth_value_and_grad(x):
    global _SMOOTH_CACHE_KEY, _SMOOTH_CACHE_VAL, _SMOOTH_CACHE_GRAD
    arr = np.asarray(x, dtype=float).reshape(-1)
    n = arr.size
    if n == 0:
        return np.inf, np.zeros(0, dtype=float)

    key = arr.tobytes()
    if _SMOOTH_CACHE_KEY is not None and key == _SMOOTH_CACHE_KEY and _SMOOTH_CACHE_GRAD is not None:
        return _SMOOTH_CACHE_VAL, _SMOOTH_CACHE_GRAD

    if arr.min() < 0.0 or arr.max() > 1.0:
        arr = np.clip(arr, 0.0, 1.0)

    s = _scores(arr)
    if s.size == 0 or not np.all(np.isfinite(s)):
        _SMOOTH_CACHE_KEY = None
        return np.inf, np.zeros(n, dtype=float)

    smax = float(np.max(s))
    gamma = float(_SMOOTH_GAMMA)

    if gamma <= 1e-12:
        idx = int(np.argmax(s))
        w = np.zeros(s.size, dtype=float)
        w[idx] = 1.0
        val = smax
    else:
        z = (s - smax) / gamma
        z = np.clip(z, -100.0, 0.0)
        w = np.exp(z)
        tot = float(np.sum(w))
        if not np.isfinite(tot) or tot <= 1e-300:
            idx = int(np.argmax(s))
            w = np.zeros(s.size, dtype=float)
            w[idx] = 1.0
            tot = 1.0
        else:
            w /= tot
        val = gamma * (smax + float(np.log(tot)))

    G = _build_G(arr)
    grad = G.T.dot(w)
    if not np.all(np.isfinite(grad)):
        _SMOOTH_CACHE_KEY = None
        return np.inf, np.zeros(n, dtype=float)

    if np.isfinite(val):
        _SMOOTH_CACHE_KEY = key
        _SMOOTH_CACHE_VAL = float(val)
        _SMOOTH_CACHE_GRAD = grad
    else:
        _SMOOTH_CACHE_KEY = None

    return float(val), grad


def _smooth_f(x):
    return _smooth_value_and_grad(x)[0]


def _smooth_g(x):
    return _smooth_value_and_grad(x)[1]


def _eq_fun(x):
    return float(np.sum(x) - _SMOOTH_TARGET)


def _eq_jac(x):
    return np.ones(_SMOOTH_N, dtype=float)


def _slsqp_chunk(h, gamma, maxiter=10):
    global _SMOOTH_GAMMA, _SMOOTH_N, _SMOOTH_TARGET, _SMOOTH_CACHE_KEY, _SMOOTH_CACHE_VAL, _SMOOTH_CACHE_GRAD
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return h, np.inf

    _SMOOTH_GAMMA = float(gamma)
    _SMOOTH_N = n
    _SMOOTH_TARGET = 0.5 * float(n)
    _SMOOTH_CACHE_KEY = None
    _SMOOTH_CACHE_VAL = None
    _SMOOTH_CACHE_GRAD = None

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": _eq_fun, "jac": _eq_jac}]
    mi = int(max(3, min(30, int(maxiter))))

    try:
        res = minimize(
            _smooth_f,
            h,
            jac=_smooth_g,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": mi, "ftol": 1e-10, "disp": False},
        )
        x = np.asarray(res.x, dtype=float).reshape(-1) if hasattr(res, "x") else h
    except Exception:
        x = h

    if not np.all(np.isfinite(x)):
        x = h

    x = _project(x, 0.5 * float(n))
    c5 = _exact_c5_bound(x)
    return x, c5


def _load_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float).reshape(-1)
    except Exception:
        return None
    if p.size < 8:
        return None
    if not np.all(np.isfinite(p)):
        return None
    return p


def _fallback_construction(n=128, seed=42):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / float(n)
    h = 0.5
    h += 0.35 * np.sin(2.0 * np.pi * 2.0 * x)
    h += 0.18 * np.sin(2.0 * np.pi * 3.0 * x + 0.7)
    h += 0.12 * np.sin(2.0 * np.pi * 5.0 * x + 2.1)
    h += rng.normal(0.0, 0.05, n)
    return _project(h, 0.5 * float(n))


def _direction(h, mode, gamma=0.1, k=3):
    h = np.asarray(h, dtype=float).reshape(-1)
    s = _scores(h)
    L = s.size
    if L == 0:
        return None

    if mode == "gn":
        G = _build_G(h)
        n = h.size
        kk = int(k) if k is not None else 4
        if kk <= 0 or kk >= L:
            kk = max(1, min(4, L))
        idx = np.argsort(s)[-kk:]
        Gk = G[idx]
        B = Gk.T.dot(Gk)
        reg = 1e-7 * (float(np.trace(B)) / float(n) + 1.0)
        B = B + reg * np.eye(n, dtype=float)
        rhs = -Gk.T.dot(s[idx])
        try:
            u = np.linalg.solve(B, np.ones(n, dtype=float))
            v = np.linalg.solve(B, rhs)
            su = float(np.sum(u))
            sv = float(np.sum(v))
            lam = sv / su if abs(su) > 1e-14 else 0.0
            g = v - lam * u
        except Exception:
            w = np.zeros(L, dtype=float)
            w[idx] = 1.0 / float(len(idx))
            g = G.T.dot(w)
    else:
        if mode == "top":
            kk = int(k) if k is not None else 5
            if kk <= 0 or kk >= L:
                kk = max(1, min(5, L))
            idx = np.argsort(s)[-kk:]
            w = np.zeros(L, dtype=float)
            w[idx] = 1.0 / float(len(idx))
        else:
            smax = float(np.max(s))
            gg = float(gamma)
            if gg <= 1e-12 or not np.isfinite(gg):
                idx = int(np.argmax(s))
                w = np.zeros(L, dtype=float)
                w[idx] = 1.0
            else:
                z = (s - smax) / gg
                z = np.clip(z, -100.0, 0.0)
                w = np.exp(z)
                tot = float(np.sum(w))
                if not np.isfinite(tot) or tot <= 1e-300:
                    idx = int(np.argmax(s))
                    w = np.zeros(L, dtype=float)
                    w[idx] = 1.0
                else:
                    w /= tot
        G = _build_G(h)
        g = G.T.dot(w)

    g = np.asarray(g, dtype=float).reshape(-1)
    if g.size == 0 or not np.all(np.isfinite(g)):
        return None

    g = g - float(np.mean(g))
    g = g - float(np.mean(g))
    g[-1] -= float(np.sum(g))
    nrm = float(np.max(np.abs(g)))
    if not np.isfinite(nrm) or nrm < 1e-16:
        return None
    return g / nrm


def _line_search_dir(h, p, target, cur_c5):
    h = np.asarray(h, dtype=float).reshape(-1)
    p = np.asarray(p, dtype=float).reshape(-1)
    n = h.size
    if p.size != n:
        return h.copy(), cur_c5
    if not np.all(np.isfinite(p)):
        return h.copy(), cur_c5

    pnorm = float(np.max(np.abs(p)))
    if not np.isfinite(pnorm) or pnorm < 1e-12:
        return h.copy(), cur_c5

    pos = p > 1e-14
    neg = p < -1e-14
    amax = 2.0
    if np.any(pos):
        amax = min(amax, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        amax = min(amax, float(np.min(h[neg] / (-p[neg]))))
    if not np.isfinite(amax) or amax < 1e-11:
        return h.copy(), cur_c5

    amax = min(amax, 2.0) * 0.999
    fracs = np.array([0.0, 0.05, 0.12, 0.25, 0.4, 0.6, 0.8, 1.0], dtype=float)
    alphas = fracs * amax
    vals = np.empty(alphas.size, dtype=float)
    vals[0] = cur_c5
    best_h = h.copy()
    best_c = cur_c5
    best_i = 0

    for i in range(1, alphas.size):
        c = _exact_c5_bound(h + alphas[i] * p)
        vals[i] = c
        if c < best_c - 1e-15:
            best_c = c
            best_h = h + alphas[i] * p
            best_i = i

    if best_i == 0:
        return best_h, best_c

    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(alphas.size - 1, best_i + 1)]
    if hi - lo < 1e-14:
        return best_h, best_c

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5_bound(h + c1 * p)
    f2 = _exact_c5_bound(h + c2 * p)
    for _ in range(14):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, f2 = c2, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5_bound(h + c1 * p)
        else:
            lo, f1 = c1, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5_bound(h + c2 * p)

    opt = 0.5 * (lo + hi)
    f = _exact_c5_bound(h + opt * p)
    if f < best_c - 1e-15:
        best_c = f
        best_h = h + opt * p

    return best_h, best_c


def _run_spec(h, target, spec, time_budget, max_steps=100):
    if time_budget <= 0.0 or max_steps <= 0:
        return _ensure_feasible(h, target), _exact_c5_bound(_ensure_feasible(h, target))

    cur_h = _ensure_feasible(h, target)
    cur_c5 = _exact_c5_bound(cur_h)
    deadline = time.time() + float(time_budget)

    mode = spec[0]
    gamma = 0.1
    kk = 3
    if len(spec) > 1 and spec[1] is not None:
        gamma = float(spec[1])
    if len(spec) > 2 and spec[2] is not None:
        kk = spec[2]

    stall = 0
    for _ in range(int(max_steps)):
        if time.time() > deadline:
            break
        p = _direction(cur_h, mode, gamma, kk)
        if p is None:
            stall += 1
            if stall > 35:
                break
            continue

        h_new, _safe = _line_search_dir(cur_h, p, target, cur_c5)
        hp = _project(h_new, target)
        c = _exact_c5_bound(hp)
        if c < cur_c5 - 1e-14:
            cur_h = hp
            cur_c5 = c
            stall = 0
        else:
            stall += 1
            if stall > 35:
                break

    return cur_h, cur_c5


def _max_shift(scores, C, a):
    return float(np.max(scores + float(a) * C))


def _lp_dir(h, radius, top_k):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return None

    s = _scores(h)
    L = s.size
    G = _build_G(h)

    if top_k is None or top_k >= L:
        idx = np.arange(L)
    else:
        idx = np.argsort(s)[-int(top_k):]

    if idx.size == 0:
        return None

    s_k = s[idx]
    G_k = G[idx]

    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    A_ub = np.zeros((idx.size, n + 1), dtype=float)
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=float)

    bounds = []
    rad = float(radius)
    for j in range(n):
        lo = max(-rad, -float(h[j]))
        hi = min(rad, 1.0 - float(h[j]))
        if lo > hi:
            lo = hi = float(h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, 1.0))

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if res is None or not res.success:
        return None

    p = np.asarray(res.x[:n], dtype=float).reshape(-1)
    if not np.all(np.isfinite(p)):
        return None
    p = p - float(np.mean(p))
    if not np.all(np.isfinite(p)):
        return None
    if float(np.max(np.abs(p))) < 1e-13:
        return None
    return p


def _lp_phase(h, target, time_budget, start_radius=0.15, top_list=None):
    if time_budget <= 0.0:
        hh = _ensure_feasible(h, target)
        return hh, _exact_c5_bound(hh)

    if top_list is None:
        top_list = [12, 8, 5, 3, None, 6, 4, 10, 2]

    base = _ensure_feasible(h, target)
    base_c5 = _exact_c5_bound(base)
    best_h = base.copy()
    best_c5 = base_c5

    radius = float(start_radius)
    deadline = time.time() + float(time_budget)
    ti = 0
    stall = 0

    while time.time() < deadline and stall < 250:
        tk = top_list[ti % len(top_list)]
        ti += 1

        p = _lp_dir(base, radius, tk)
        if p is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue

        h_new, _safe = _line_search_dir(base, p, target, base_c5)
        hp = _project(h_new, target)
        c = _exact_c5_bound(hp)

        if c < base_c5 - 1e-14:
            base = hp
            base_c5 = c
            if c < best_c5 - 1e-15:
                best_c5 = c
                best_h = base.copy()
            stall = 0
            radius = min(0.35, radius * 1.03)
        else:
            stall += 1
            radius *= 0.72
            if stall > 25 and radius < 0.02:
                radius = 0.08

    return best_h, best_c5


def _pair_polish(h, target, deadline, max_passes=25, verify_thresh=1e-9):
    cur = _ensure_feasible(h, target)
    cur_c5 = _exact_c5_bound(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    n = cur.size

    if n < 2:
        return best_h, best_c5

    scores = _scores(cur)
    G = _build_G(cur)

    for _pass in range(int(max_passes)):
        if time.time() >= deadline:
            break
        improved = False
        for j in range(n):
            if time.time() >= deadline:
                break
            for k in range(j + 1, n):
                if time.time() >= deadline:
                    break

                hj = float(cur[j])
                hk = float(cur[k])
                if hj < 1e-10 and hk < 1e-10:
                    continue
                if hj > 1.0 - 1e-10 and hk > 1.0 - 1e-10:
                    continue

                C = G[:, j] - G[:, k]
                a_lo = max(-hj, hk - 1.0)
                a_hi = min(1.0 - hj, hk)
                if a_hi - a_lo < 1e-13:
                    continue

                lo = a_lo
                hi = a_hi
                for _ in range(16):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = _max_shift(scores, C, m1)
                    f2 = _max_shift(scores, C, m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1

                alpha = 0.5 * (lo + hi)
                pred = _max_shift(scores, C, alpha)

                if cur_c5 - pred > verify_thresh and abs(alpha) > 1e-11:
                    oj = cur[j]
                    ok = cur[k]
                    cur[j] = oj + alpha
                    cur[k] = ok - alpha
                    actual = _exact_c5_bound(cur)
                    if actual < cur_c5 - 1e-15:
                        cur_c5 = actual
                        scores = _scores(cur)
                        G = _build_G(cur)
                        improved = True
                        if cur_c5 < best_c5 - 1e-15:
                            best_c5 = cur_c5
                            best_h = cur.copy()
                    else:
                        cur[j] = oj
                        cur[k] = ok

        if not improved:
            break

    return best_h, best_c5


def _append_candidate(list_cands, h, target):
    h = np.asarray(h, dtype=float).reshape(-1)
    h = _project(h, target)
    c = _exact_c5_bound(h)
    list_cands.append((h, c))


def _cand_key(item):
    return item[1]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    if total <= 0.0:
        total = 1.0
    reserve = max(5.0, 0.04 * total)
    if reserve > 0.22 * total:
        reserve = 0.22 * total
    hard = t0 + total - reserve

    rng = np.random.default_rng(seed)
    parent = _load_parent()
    if parent is not None:
        n = parent.size
        h0 = _project(parent, 0.5 * float(n))
    else:
        n = 128
        h0 = _fallback_construction(n, seed)

    target = 0.5 * float(n)
    best_h = h0.copy()
    best_c5 = _exact_c5_bound(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5

    # SLSQP continuation on a smooth maximum surrogate.
    stage1_end = min(hard, t0 + 0.18 * total)
    gamma_schedule = [
        1.0, 0.6, 0.4, 0.25, 0.18, 0.12, 0.08, 0.06,
        0.045, 0.035, 0.028, 0.022, 0.018, 0.014,
        0.011, 0.009, 0.007, 0.0055, 0.0045,
    ]
    slq_calls = 0
    max_slq_calls = 22

    for g in gamma_schedule:
        if time.time() >= stage1_end or slq_calls >= max_slq_calls:
            break
        for _attempt in range(3):
            if time.time() >= stage1_end or slq_calls >= max_slq_calls:
                break
            h_try, c_try = _slsqp_chunk(work_h, g, 10)
            slq_calls += 1
            if not np.isfinite(c_try):
                break
            max_worse = max(0.003, 0.008 * best_c5)
            if c_try < best_c5 - 1e-14:
                best_h = h_try.copy()
                best_c5 = c_try
                work_h = best_h.copy()
                work_c5 = best_c5
            elif c_try < work_c5 and c_try < best_c5 + max_worse:
                work_h = h_try.copy()
                work_c5 = c_try
            else:
                break

    # Directed projected line-search phase.
    stage2_end = min(hard, t0 + 0.45 * total)
    softs = [0.25, 0.15, 0.10, 0.07, 0.05, 0.035, 0.025, 0.018, 0.013, 0.010, 0.008, 0.006]
    tops = [1, 2, 3, 4, 5, 6, 8, 10, 15]
    specs = []
    si = 0
    for kk in tops:
        specs.append(("top", 0.0, kk))
        if si < len(softs):
            specs.append(("soft", softs[si], 0))
            si += 1
    for kk in (2, 4, 6, 10):
        specs.append(("gn", 0.0, kk))

    base_h = best_h.copy()
    base_c5 = best_c5
    for spec in specs:
        if time.time() >= stage2_end:
            break
        rem = stage2_end - time.time()
        if rem < 0.2:
            break
        tlim = min(4.0, 0.6 * rem)
        h_try, c_try = _run_spec(base_h, target, spec, tlim, 80)
        if c_try < best_c5 - 1e-14:
            best_h = h_try.copy()
            best_c5 = c_try
            work_h = best_h.copy()
            work_c5 = best_c5
            base_h = best_h.copy()
            base_c5 = best_c5
        elif c_try < base_c5 - 1e-15:
            base_h = h_try.copy()
            base_c5 = c_try
            if c_try < work_c5:
                work_h = h_try.copy()
                work_c5 = c_try

    if time.time() < stage2_end and work_c5 > best_c5 + 1e-9:
        base_h = work_h.copy()
        base_c5 = work_c5
        for spec in specs:
            if time.time() >= stage2_end:
                break
            rem = stage2_end - time.time()
            if rem < 0.2:
                break
            tlim = min(4.0, 0.6 * rem)
            h_try, c_try = _run_spec(base_h, target, spec, tlim, 80)
            if c_try < best_c5 - 1e-14:
                best_h = h_try.copy()
                best_c5 = c_try
                work_h = best_h.copy()
                work_c5 = best_c5
                base_h = best_h.copy()
                base_c5 = best_c5
            elif c_try < base_c5 - 1e-15:
                base_h = h_try.copy()
                base_c5 = c_try
                if c_try < work_c5:
                    work_h = h_try.copy()
                    work_c5 = c_try

    # Trust-region linearized minimax LP phase.
    stage3_end = min(hard, t0 + 0.65 * total)
    lp_top = [12, 8, 5, 3, None, 6, 4, 10, 2]
    for sl in range(4):
        if time.time() >= stage3_end:
            break
        rem = (stage3_end - time.time()) / float(4 - sl)
        tlim = min(30.0, max(1.0, rem))
        base_h = best_h if (sl % 2 == 0) else work_h
        h_try, c_try = _lp_phase(base_h, target, tlim, 0.15, lp_top)
        if c_try < best_c5 - 1e-14:
            best_h = h_try.copy()
            best_c5 = c_try
            work_h = best_h.copy()
            work_c5 = best_c5
        elif c_try < work_c5 and c_try < best_c5 + max(0.003, 0.008 * best_c5):
            work_h = h_try.copy()
            work_c5 = c_try

    # Alternative transforms and quick local refinements.
    stage4_end = min(hard, t0 + 0.86 * total)
    cands = []
    _append_candidate(cands, np.convolve(best_h, [0.2, 0.6, 0.2], mode="same"), target)
    _append_candidate(cands, np.clip(1.5 * (best_h - 0.5) + 0.5, 0.0, 1.0), target)
    _append_candidate(cands, np.clip(2.2 * (best_h - 0.5) + 0.5, 0.0, 1.0), target)
    for scale in (0.02, 0.05, 0.10):
        _append_candidate(cands, best_h + rng.normal(0.0, scale, n), target)
    cands.sort(key=_cand_key)

    small_specs = [
        ("top", 0.0, 2),
        ("soft", 0.05, 0),
        ("top", 0.0, 4),
        ("soft", 0.02, 0),
        ("gn", 0.0, 4),
    ]

    for cand_h, cand_c5 in cands:
        if time.time() >= stage4_end:
            break
        if not np.isfinite(cand_c5):
            continue
        if cand_c5 > best_c5 + 0.015:
            continue

        h_try = cand_h.copy()
        c_try = cand_c5

        for g in (0.15, 0.08, 0.05, 0.03, 0.02):
            if time.time() >= stage4_end:
                break
            h2, c2 = _slsqp_chunk(h_try, g, 8)
            if c2 < c_try - 1e-14:
                h_try = h2.copy()
                c_try = c2
            if c2 < best_c5 - 1e-14:
                best_h = h_try.copy()
                best_c5 = c_try
                work_h = best_h.copy()
                work_c5 = best_c5
                break

        if time.time() < stage4_end:
            for spec in small_specs:
                if time.time() >= stage4_end:
                    break
                tlim = min(1.5, max(0.2, 0.1 * (stage4_end - time.time())))
                if tlim <= 0.1:
                    break
                h2, c2 = _run_spec(h_try, target, spec, tlim, 40)
                if c2 < c_try - 1e-14:
                    h_try = h2.copy()
                    c_try = c2
                if c2 < best_c5 - 1e-14:
                    best_h = h_try.copy()
                    best_c5 = c_try
                    work_h = best_h.copy()
                    work_c5 = best_c5
                    break

    # Final polishing: pair descent, small LP, small direction searches.
    if time.time() < hard:
        rem = hard - time.time()
        pair_budget = min(45.0, max(5.0, 0.04 * total))
        deadline = min(hard, time.time() + pair_budget)
        if deadline - time.time() > 2.0:
            h_try, c_try = _pair_polish(best_h, target, deadline, max_passes=25, verify_thresh=1e-9)
            if c_try < best_c5 - 1e-14:
                best_h = h_try.copy()
                best_c5 = c_try
                work_h = best_h.copy()
                work_c5 = best_c5

    if time.time() < hard - 2.0:
        rem = hard - time.time()
        lp_budget = min(25.0, max(3.0, 0.025 * total))
        tlim = min(lp_budget, rem - 2.0)
        if tlim > 2.0:
            h_try, c_try = _lp_phase(best_h, target, tlim, 0.04, [4, 3, 2, 6, None])
            if c_try < best_c5 - 1e-14:
                best_h = h_try.copy()
                best_c5 = c_try
                work_h = best_h.copy()
                work_c5 = best_c5

    if time.time() < hard - 1.0:
        final_specs = [
            ("top", 0.0, 1),
            ("top", 0.0, 2),
            ("top", 0.0, 3),
            ("soft", 0.006, 0),
            ("soft", 0.004, 0),
            ("gn", 0.0, 3),
        ]
        for spec in final_specs:
            if time.time() >= hard - 1.0:
                break
            tlim = min(4.0, hard - time.time() - 1.0)
            if tlim <= 0.2:
                break
            h_try, c_try = _run_spec(best_h, target, spec, tlim, 50)
            if c_try < best_c5 - 1e-14:
                best_h = h_try.copy()
                best_c5 = c_try
                work_h = best_h.copy()
                work_c5 = best_c5

    n_final = best_h.size
    target_final = 0.5 * float(n_final)
    best_h = _project(best_h, target_final)
    final_c5 = _exact_c5_bound(best_h)

    if not np.isfinite(final_c5):
        fb = _fallback_construction(n_final, seed)
        return fb, _exact_c5_bound(fb), int(n_final)

    return best_h, float(final_c5), int(n_final)
