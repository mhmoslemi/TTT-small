        if not improved:
            break
    return best_h, best_c5

def _perturb_slsqp(h_best, deadline, rng, n_attempts=60, top_M=30):
    n = len(h_best)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = 0.03
    
    for attempt in range(n_attempts):
        if time.time() > deadline - 5.0:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        sub_dl = min(deadline - 5.0, time.time() + budget_s_global * 0.015)
        h_try, c5_try = _slsqp_active_set(h_pert, sub_dl, max_rounds=2, top_M=top_M, maxiter=150)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.1)
    return best_h, best_c5

budget_s_global = 1100

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s_global
    budget_s_global = budget_s
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    # Phase 1: SLSQP epigraph from parent at multiple resolutions
    resolutions = [64, 80, 96, 128]
    for n_res in resolutions:
        if time.time() > t0 + budget_s * 0.20:
            break
        h_res = _resample(h_start, n_res)
        sub_dl = min(t0 + budget_s * 0.20, time.time() + budget_s * 0.05)
        # Smoothed gradient first to get closer
        h_sm, c5_sm = _smoothed_grad_v2(h_res, sub_dl, beta_schedule=(10, 150), n_steps=200, step_init=0.04)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()
            best_n = n_res
        # Then SLSQP
        sub_dl2 = min(t0 + budget_s * 0.20, time.time() + budget_s * 0.04)
        h_sl, c5_sl = _slsqp_active_set(h_sm, sub_dl2, max_rounds=3, top_M=35, maxiter=250)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            best_n = n_res
    
    # Phase 2: SLSQP with all constraints on best resolution
    if time.time() < t0 + budget_s * 0.35:
        dl2 = t0 + budget_s * 0.35
        h_all, c5_all = _slsqp_epigraph(best_h, dl2, top_M=None, maxiter=400, ftol=1e-14)
        c5_all = _exact_c5(h_all)
        if c5_all < best_c5:
            best_c5 = c5_all
            best_h = h_all.copy()
        # Follow-up with active set
        if time.time() < dl2:
            h_as, c5_as = _slsqp_active_set(best_h, dl2, max_rounds=4, top_M=30, maxiter=200)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
    
    # Phase 3: Smoothed gradient with different parameters
    if time.time() < t0 + budget_s * 0.50:
        dl3 = t0 + budget_s * 0.50
        for bs, be, st in [(5, 100, 0.05), (30, 500, 0.03), (50, 1000, 0.02)]:
            if time.time() > dl3 - 5.0:
                break
            sub_dl = min(dl3 - 5.0, time.time() + budget_s * 0.04)
            h_sm, c5_sm = _smoothed_grad_v2(best_h, sub_dl, beta_schedule=(bs, be), n_steps=300, step_init=st)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()
            sub_dl2 = min(dl3 - 5.0, time.time() + budget_s * 0.025)
            h_sl, c5_sl = _slsqp_active_set(h_sm, sub_dl2, max_rounds=2, top_M=30, maxiter=150)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
    
    # Phase 4: Multi-resolution SLSQP exploration
    if time.time() < t0 + budget_s * 0.65:
        dl4 = t0 + budget_s * 0.65
        for n_res in [64, 80, 96, 64, 80]:
            if time.time() > dl4 - 5.0:
                break
            h_res = _resample(best_h, n_res)
            sub_dl = min(dl4 - 5.0, time.time() + budget_s * 0.03)
            h_sl, c5_sl = _slsqp_active_set(h_res, sub_dl, max_rounds=3, top_M=30, maxiter=200)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n_res
    
    # Phase 5: Perturbation + SLSQP
    if time.time() < t0 + budget_s * 0.80:
        dl5 = t0 + budget_s * 0.80
        best_h, best_c5 = _perturb_slsqp(best_h, dl5, rng, n_attempts=50, top_M=30)
    
    # Phase 6: Coordinate polish
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.90
        hc, c5c = _coord_polish(best_h, dl6, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()
    
    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.95:
        dl7 = t0 + budget_s * 0.95
        hp, c5p = _pair_polish(best_h, dl7, n_passes=2)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
    
    # Phase 8: Final SLSQP
    if time.time() < t0 + budget_s * 0.99:
        dl8 = t0 + budget_s * 0.995
        h_fin, c5_fin = _slsqp_active_set(best_h, dl8, max_rounds=2, top_M=25, maxiter=150)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
