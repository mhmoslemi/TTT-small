    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_weighted(h, w):
    """
    Compute gradient of ∑_k w[k] * S_k where S_k = scores for shift k.
    Uses symmetry: S_k = S_{-k}.
    w has length n (for k=0..n-1).
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        if w[k] < 1e-15:
            continue
        wk = w[k] * dx
        # dS_k/dh[j] = (1-h[j-k]) for j=k..n-1, minus h[j+k] for j=0..n-1-k
        m = n - k
        grad[k:] += wk * (1.0 - h[:m])
        grad[:m] -= wk * h[k:k+m]
    return grad

def _lbgfs_optimize(h, deadline, beta_schedule, mu=500.0, maxiter_per_phase=200):
    """
    L-BFGS-B optimization with LSE smoothing.
    beta_schedule: list of beta values to use in sequence.
    """
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        
        def obj_and_grad(x):
            S = _scores_half(x)
            maxS = np.max(S)
            shifted = np.clip(beta * (S - maxS), -500, 0)
            f = maxS + np.log(np.mean(np.exp(shifted))) / beta
            sv = float(np.sum(x)) - target
            f += mu * sv * sv
            
            w = np.exp(shifted)
            wsum = w.sum()
            if wsum < 1e-300:
                w = np.zeros(n)
                w[int(np.argmax(S))] = 1.0
            else:
                w = w / wsum
            
            grad = _grad_weighted(x, w)
            grad += 2.0 * mu * sv
            return f, grad
        
        try:
            res = minimize(obj_and_grad, h, jac=True, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter_per_phase, 'ftol': 1e-14, 'gtol': 1e-8})
            if all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5 = _exact_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def _scores_half(h):
    """Scores for k=0..n-1 (using symmetry)."""
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full')
    return full[n-1:] * dx

def _sturmian(n, alpha):
    i = np.arange(n)
    return (np.floor((i+1)*alpha) - np.floor(i*alpha)).astype(float)

def _random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def _binary_search(h, deadline, max_rounds=100, rng_seed=42):
    """Greedy swap-based local search on binary sequences."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(rng_seed)
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Shuffle for randomization
        order_o = rng.permutation(len(ones))
        order_z = rng.permutation(len(zeros))
        for io in order_o:
            if time.time() > deadline:
                break
            for iz in order_z:
                if time.time() > deadline:
                    break
                i, j = ones[io], zeros[iz]
                h[i], h[j] = 0.0, 1.0
                new_c5 = _exact_c5(h)
                if new_c5 < cur_c5 - 1e-15:
                    cur_c5 = new_c5
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    # Update sets (i went to zero, j went to one)
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    break  # restart inner loop with new sets
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c5

def _anneal_binary(h, deadline, T_start=0.005, T_end=0.0001, max_iters=30000, rng_seed=77):
    """Simulated annealing on binary sequences."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(rng_seed)
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, max_iters - 1))
        
        io = rng.integers(len(ones))
        iz = rng.integers(len(zeros))
        i, j = ones[io], zeros[iz]
        
        h[i], h[j] = 0.0, 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            ones[io] = j
            zeros[iz] = i
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c5
