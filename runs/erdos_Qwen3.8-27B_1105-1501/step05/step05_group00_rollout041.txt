    a = np.linspace(a_min, a_max, n_grid)
    F = base[None, :] + a[:, None] * L[None, :] + a[:, None]**2 * Q[None, :]
    obj = np.max(F, axis=1)
    best_i = np.argmin(obj)
    best_a = a[best_i]
    best_val = obj[best_i]
    # Refine with golden section
    if best_i > 0 and best_i < n_grid - 2:
