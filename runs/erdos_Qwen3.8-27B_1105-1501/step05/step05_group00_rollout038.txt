```python
def _lp_direction_active(h, active_indices, radius):
    """Solve LP to find best descent direction using only active lags."""
    n = len(h)
    m = len(active_indices)
    
    # Compute gradients for active lags
    # G[p, j] = dC_p/dh[j]
    S0 = np.correlate(h, 1.0 - h, mode='full')
    
    G = np.zeros((m, n))
    for idx, p in enumerate(active_indices):
        r = p - (n - 1)
        if r >= 0:
            m_valid = n - r
            G[idx, :m_valid] = (1.0 - h[r:])
            if r > 0:
                G[idx, r:] += -h[:m_valid]
        else:
            s = -r
            m_valid = n - s
            if s > 0:
                G[idx, s:] = (1.0 - h[:m_valid])
            G[idx, :m_valid] += -h[s:]
    
    # LP: minimize t
    # subject to G @ d - t <= 0  (G @ d <= t)
    #          sum(d) = 0
    #          -radius <= d <= radius (and bounds from h)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0  # G @ d - t <= 0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, -float(np.max(h[neg] / d[neg])), -trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    n_grid = 256
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(80):
        if b - a < 1e-15:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * Q)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _active_lp(h, tol_frac, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(S0))
    threshold = mx * (1.0 - tol_frac)
    active = np.where(S0 >= threshold)[0]
    m = len(active)
    if m == 0:
        return None
    G = np.zeros((m, n))
    for idx, p in enumerate(active):
        r = p - (n - 1)
        if r >= 0:
            mv = n - r
            G[idx, :mv] = (1.0 - h[r:])
            if r > 0:
                G[idx, r:] += -h[:mv]
        else:
            s = -r
            mv = n - s
            if s > 0:
                G[idx, s:] = (1.0 - h[:mv])
            G[idx, :mv] += -h[s:]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            if float(np.max(np.abs(d))) > 1e-12:
                return d
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
```
