    """Find best t for h[i]+=t, h[j]-=t. Returns (t_best, c5_best) or (None, None)."""
    n = len(h)
    if i == j:
        return None, None
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])

    corr = np.correlate(h, 1.0 - h, mode='full')

    L = np.zeros(2*n - 1)

    # h-side of i: lags k where i is valid

    for k in range(0, i+1):

            f1 = f2
            c1 = c2
