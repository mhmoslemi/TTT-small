import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for water level
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    """Compute all 2n-1 correlation scores (unscaled)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _lse_objective(h, tau, target):
    """Log-sum-exp smoothed objective."""
    s = _scores(h)
    M = float(np.max(s))
    z = np.exp((s - M) / tau)
    return float(M + tau * np.log(np.sum(z)))

def _lse_gradient(h, tau, target):
    """Gradient of the LSE smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    s = _scores(h)
    M = float(np.max(s))
    z = np.exp((s - M) / tau)
    w = z / np.sum(z)
    
    # Compute gradient: sum_k w_k * grad S_k
    # For q >= 0 (index n-1+q in the full correlate output):
    #   S_q = sum_{i=0}^{n-q-1} h[i+q](1-h[i])
    #   grad S_q[j] = (1-h[j-q]) if q<=j<n, plus -h[j+q] if j<n-q
    # For q < 0, g=-q (index n-1-g in full correlate output):
    #   S_{-g} = sum_{i=0}^{n-g-1} h[i](1-h[i+g])
    #   grad S_{-g}[j] = (1-h[j+g]) if j<n-g, plus -h[j-g] if g<=j<n
    
    g = np.zeros(n)
    for q in range(-(n-1), n):
        wq = w[n - 1 + q]
        if wq < 1e-15:
            continue
        if q >= 0:
            m = n - q
            if m > 0:
                # (1-h[j-q]) for j in [q, n-1]
                g[q:] += wq * (1.0 - h[:m])
                # -h[j+q] for j in [0, m-1]
                g[:m] -= wq * h[q:]
        else:
            gg = -q
            m = n - gg
            if m > 0:
                # (1-h[j+g]) for j in [0, m-1]
                g[:m] += wq * (1.0 - h[gg:])
                # -h[j-g] for j in [gg, n-1]
                g[gg:] -= wq * h[:m]
    
    return dx * g

def _line_search_quadratic(h, d, dx, radius):
    """Exact line search along direction d using quadratic model."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible interval
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    for _ in range(50):
        if t_hi - t_lo < 1e-12:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * Q)) * dx
    return opt_t, opt_val

def _gradient_direction(h, tau, target):
    """Compute descent direction from LSE gradient, projected to be sum-zero."""
    g = _lse_gradient(h, tau, target)
    # Subtract mean to ensure sum-zero direction
    g = g - np.mean(g)
    # Normalize
    norm = float(np.max(np.abs(g)))
    if norm < 1e-15:
        return None
    return g / norm

def _slsqp_optimize(h, tau, target, deadline):
    """Run SLSQP on the smoothed objective."""
    n = len(h)
    h = h.copy()
    
    def obj(x):
        if time.time() > deadline:
            return 1e10
        return _lse_objective(x, tau, target)
    
    def grad(x):
        if time.time() > deadline:
            return np.zeros(n)
        return _lse_gradient(x, tau, target)
    
    # Constraint: sum(x) = target
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
            'jac': lambda x: np.ones(n)}
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or res.fun < _lse_objective(h, tau, target):
            h_new = _project(res.x, target)
            return h_new
    except Exception:
        pass
    return h

def _active_set_direction(h, top_k):
    """Direction that balances the top-k scores."""
    n = len(h)
    s = _scores(h)
    idx = np.argsort(s)[-top_k:]
    # Average gradient of top-k scores
    g = np.zeros(n)
    for q_idx in idx:
        q = q_idx - (n - 1)
        if q >= 0:
            m = n - q
            if m > 0:
                g[q:] += (1.0 - h[:m])
                g[:m] -= h[q:]
        else:
            gg = -q
            m = n - gg
            if m > 0:
                g[:m] += (1.0 - h[gg:])
                g[gg:] -= h[:m]
    g /= top_k
    g -= np.mean(g)
    norm = float(np.max(np.abs(g)))
    if norm < 1e-15:
        return None
    return g / norm

def _random_direction(n, rng):
    """Random sum-zero direction."""
    d = rng.normal(0, 1, n)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return d / norm

def _block_direction(h, n, rng):
    """Block swap direction."""
    # Pick two blocks and transfer mass
    blen = rng.integers(2, max(3, n // 10))
    i = rng.integers(0, n - blen)
    j = rng.integers(0, n - blen)
    # Ensure blocks don't overlap too much
    if abs(i - j) < blen:
        j = (i + blen + rng.integers(0, n - 2*blen)) % max(1, n - 2*blen)
    
    d = np.zeros(n)
    d[i:i+blen] = 1.0 / blen
    d[j:j+blen] = -1.0 / blen
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return d / norm

def _pair_swap(h, i, j, dx, radius):
    """Try a pair swap and return improved h and c5."""
    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    alpha, c5_try = _line_search_quadratic(h, d, dx, radius)

        h_new = _project(h + alpha * d, n / 2.0)
        return h_new, c5_try, True
    return h, _exact_c5(h), False

def _resample(h, n_new):
    """Resample h to new resolution."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    # Phase 2: Active-set gradient + line search (15%)
    d2 = t0 + budget_s * 0.50
    n = len(best_h)
    dx = 2.0 / n
    cur_h = best_h.copy()
    cur_c5 = best_c5
    radius = 0.3
    stall = 0
    while time.time() < d2:
        if stall > 40:
            break
        improved = False
        for top_k in [1, 2, 3, 5, 8, 12]:
            if time.time() > d2:
                break
            d = _active_direction(cur_h, top_k)
            if d is None:
                continue
            for sign in [1.0, -1.0]:
                alpha, c5_try = _line_search(cur_h, sign * d, dx, radius)
                if c5_try < cur_c5 - 1e-12:
                    cur_h = _project(cur_h + sign * alpha * d, n / 2.0)
                    cur_c5 = _exact_c5(cur_h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur_h.copy()
                    stall = 0
                    radius = min(radius * 1.04, 0.5)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall % 8 == 0:
                radius *= 0.88

    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = cur_h.copy()

    # Phase 3: Random and block directions (12%)
    d3 = t0 + budget_s * 0.62
    n = len(best_h)
    dx = 2.0 / n
    cur_h = best_h.copy()
    cur_c5 = best_c5
    stall = 0
    radius = 0.25
    while time.time() < d3:
        if stall > 60:
            break
        use_block = rng.random() < 0.3
        if use_block:
            d = _block_dir(n, rng)
        else:
            d = _random_dir(n, rng)
        if d is None:
            continue
        improved = False
        for sign in [1.0, -1.0]:
            alpha, c5_try = _line_search(cur_h, sign * d, dx, radius)
            if c5_try < cur_c5 - 1e-12:
                cur_h = _project(cur_h + sign * alpha * d, n / 2.0)
                cur_c5 = _exact_c5(cur_h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()
                stall = 0
                radius = min(radius * 1.03, 0.4)
                improved = True
                break
        if not improved:
            stall += 1
            if stall % 15 == 0:
                radius *= 0.92

    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = cur_h.copy()

    # Phase 4: Targeted pair swaps (15%)
    d4 = t0 + budget_s * 0.77
    cur_h, cur_c5 = _targeted_pairs(best_h, d4, rng)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = cur_h.copy()

    # Phase 5: Multi-resolution (10%)
    d5 = t0 + budget_s * 0.87
    n_curr = len(best_h)
    offsets = [4, -4, 8, -8, 12, -12]
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 48 or n_try > 350:
            continue
        h_try = _resample(best_h, n_try)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 + 0.002:
            dx_try = 2.0 / n_try
            for tau in [0.01, 0.005]:
                h_try, c5_try = _gradient_descend(h_try, tau, dx_try, 0.3,
                                                   max_iter=100,
                                                   deadline=min(d5, time.time() + budget_s * 0.01))
            h_try, c5_try = _pair_loop(h_try, min(d5, time.time() + budget_s * 0.008), rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = _project(h_try, n_try / 2.0)
                n_curr = n_try

    # Phase 6: Final polish (9%)
    d6 = hard_deadline
    n = len(best_h)
    dx = 2.0 / n
    # Gradient refinement
    for tau in [0.003, 0.001]:
        if time.time() > d6:
            break
        h_try, c5_try = _gradient_descend(best_h, tau, dx, 0.15,
                                           max_iter=100, deadline=d6)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    # Pair swap polish
    h_try, c5_try = _pair_loop(best_h, d6, rng, max_iter=3000)
    if c5_try < best_c5:
        best_c5 = c5_try
        best_h = h_try
    # Final gradient step
    if time.time() < hard_deadline - 10:
        h_try, c5_try = _gradient_descend(best_h, 0.001, dx, 0.1,
                                           max_iter=50, deadline=hard_deadline - 10)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
