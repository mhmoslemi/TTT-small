    target = n / 2.0

    x0 = np.concatenate([h0, [target_t]])

        scores = _all_scores(h)

        J = np.zeros((1, n + 1))
        J[0, :n] = 1.0
        return J

    # Time check
    t_start = time.time()

                      bounds=bounds,
