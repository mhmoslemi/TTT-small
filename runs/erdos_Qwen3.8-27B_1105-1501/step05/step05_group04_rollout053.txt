    dx = 2.0 / n
    target = n / 2.0
    n_lags = 2 * n - 1
    
    c0 = eval_c5(h_init)
    t_init = c0 + 1e-4
    
    x0 = np.zeros(n + 1)
    x0[:n] = h_init
    x0[n] = t_init
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, max(1.0, c0 + 0.1))]
    
    # Precompute Jacobian (it depends on h, so we need to recompute)
    # For SLSQP, we provide a function for the Jacobian
    
    deadline_ref = [deadline]
    
    def objective(x):
        if time.time() > deadline:
            raise TimeoutError()
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def constraints(x):
        if time.time() > deadline:
            raise TimeoutError()
        h = x[:n]
        t = x[n]
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - scores  # should be >= 0

def minimize_single_lag(h, s, n):
    """Minimize F_s(h) subject to sum(h)=n/2, 0<=h<=1."""
    dx = 2.0 / n
    target = n / 2.0
    
    # Build the QP: min h^T Q h + c^T h
    # subject to: 1^T h = target, 0 <= h <= 1
    
    # For s >= 0:
    # F_s = dx * sum_{i=0}^{n-1-s} (h[i+s] - h[i+s]*h[i])
    # = dx * sum_{i=0}^{n-1-s} h[i+s] - dx * sum_{i=0}^{n-1-s} h[i+s]*h[i]
    
    # The linear part: dx * sum_{i=0}^{n-1-s} h[i+s] = dx * sum_{j=s}^{n-1} h[j]
    # The quadratic part: -dx * sum_{i=0}^{n-1-s} h[i+s]*h[i]
    
    # So F_s = sum_j (dx * I[j>=s]) h[j] - dx * sum_{i=0}^{n-1-s} h[i]*h[i+s]

        return lag_jacobian(x)[n-1+s]  # gradient of F_s

    tried = set()
    deltas = list(range(-20, 21, 5))
    rng_deltas = [int(rng.integers(-30, 31)) for _ in range(10)]
    all_deltas = deltas + [d for d in rng_deltas if d not in deltas]
    rng.shuffle(all_deltas)
    for dn in all_deltas:
        if time.time() > deadline:
            break
        n_try = n + dn
        if n_try < 30 or n_try > 180 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline, time.time() + 8.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def _coord_refine(h, deadline, n_iter=400):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(777)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def _perturb_search(h, deadline, rng, base_scale=0.05):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    for _ in range(300):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        sub_dl = min(deadline, time.time() + 5.0)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.1)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.08)
    return best_h, best_c5

def _structured_perturb(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    for k in range(20):
        if time.time() > deadline:
            break
        shift = rng.integers(1, max(2, n // 4))
        h_shift = np.roll(h, shift)
        h_shift = project_box_sum(h_shift, target)
        c5_s = eval_c5(h_shift)
        if c5_s < best_c5 - 1e-10:
            best_h = h_shift.copy()
            best_c5 = c5_s
            sub_dl = min(deadline, time.time() + 5.0)
            best_h, best_c5 = _lp_loop(best_h, sub_dl, radius_init=0.1)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).flatten()
    else:
        n0 = 95
        x = 2.0 * (np.arange(n0) + 0.5) / n0
        h0 = 0.5 + 0.45 * np.sin(2.0 * np.pi * x / 2.0 + 0.5)
        h0 = project_box_sum(h0, n0 / 2.0)

    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: LP from initial (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.18)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LSE smoothing (18%)
    d2 = t0 + budget_s * 0.30
    best_h, best_c5 = _lse_optimize(best_h, d2)

    # Phase 3: SLSQP epigraph (15%)
    d3 = t0 + budget_s * 0.45
    h_epi, c5_epi = _slsqp_epigraph(best_h, d3, max_iter=25)
    if c5_epi < best_c5:
        best_c5 = c5_epi
        best_h = h_epi.copy()

    # Phase 4: Resolution cycling (15%)
    d4 = t0 + budget_s * 0.60
    best_h, best_c5 = _resolution_search(best_h, d4, rng)

    # Phase 5: LSE again on best (12%)
    d5 = t0 + budget_s * 0.72
    best_h, best_c5 = _lse_optimize(best_h, d5, betas=[0.03, 0.015, 0.008, 0.004, 0.002])

    # Phase 6: Perturbation + LP (12%)
    d6 = t0 + budget_s * 0.84
    best_h, best_c5 = _perturb_search(best_h, d6, rng, base_scale=0.04)

    # Phase 7: Structured perturbation (5%)
    d7 = t0 + budget_s * 0.89
    best_h, best_c5 = _structured_perturb(best_h, d7, rng)

    # Phase 8: Coordinate refinement (5%)
    d8 = t0 + budget_s * 0.94
    best_h, best_c5 = _coord_refine(best_h, d8, n_iter=300)

    # Final LP polish
    d9 = hard_deadline
    h_f, c5_f = _lp_loop(best_h, d9, radius_init=0.05)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)

    return best_h, final_c5, len(best_h)
