            break
        if f1 < f2:
            b, f2 = a, f1
            a = b - gr * (b - a)
            f1 = float(np.max(S0 + a * T + a * a * C)) * dx
        else:
            a, f1 = b, f2
            b = a + gr * (b - a)
            f2 = float(np.max(S0 + b * T + b * b * C)) * dx
    mid = (a + b) / 2.0
    mid_val = float(np.max(S0 + mid * T + mid * mid * C)) * dx
    if mid_val < best_val:
        best_t, best_val = mid, mid_val
    return best_t, best_val

def _active_descent(h_init, deadline, max_iter=500):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    global dx_local
    dx_local = dx
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.1
    stall = 0
    rng = np.random.default_rng(777)

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 40:
            if radius > 0.05:
                radius = max(0.05, radius * 0.6)
            else:
                h = h + rng.normal(0, 0.015, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.1
                stall = 0
                continue
            if stall > 80:
                break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = cur_c5
        active = np.where(corr > M - 1e-4)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:15]
        elif len(active) > 40:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:40]]

        p = _lp_direction(h, active, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            # Try steepest descent on active set
            G_act = _corr_grads(h)[active]
            grad_sum = G_act.sum(axis=0)
            # Project to sum-zero
            grad_sum -= grad_sum.mean()
            p = -grad_sum
            p_norm = np.linalg.norm(p)
            if p_norm < 1e-12:
                stall += 10
                radius *= 0.7
                continue
            p = p / p_norm * radius

        alpha, new_c5 = _line_search(h, p, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            if abs(alpha) > 0.3 * radius:
                radius = min(radius * 1.1, 0.35)
            else:
                radius = min(radius * 1.02, 0.30)
        else:
            stall += 1
            radius *= 0.75
            if radius < 1e-5:
                break

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def _structured_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 3 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global dx_local
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _structured_init(n0, rng)

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP on initial (15% time)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _slsqp_optimize(h, d1, max_outer=15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Active descent (20% time)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _active_descent(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: SLSQP again after active descent (10% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _slsqp_optimize(best_h, d3, max_outer=15)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Multi-resolution search (30% time)
    d4 = t0 + budget_s * 0.75
    offsets = [-8, 8, -6, 6, -10, 10, -4, 4, -12, 12, -14, 14, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _slsqp_optimize(h_r, sub_dl, max_outer=8)
        if time.time() < d4 - 10:
            sub_dl2 = min(d4, time.time() + budget_s * 0.025)
            h_r, c5_r = _active_descent(h_r, sub_dl2, max_iter=200)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Multi-start with diverse inits (10% time)
    d5 = t0 + budget_s * 0.85
    starts = []
    for freq in [1, 2, 3, 4]:
        for ph in [0, np.pi / 2, np.pi]:
            starts.append(_sine_init(best_n, freq, ph, rng))
    for _ in range(4):
        starts.append(_structured_init(best_n, rng))
    for s in starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_s, c5_s = _slsqp_optimize(s, sub_dl, max_outer=6)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 6: Final SLSQP + active descent polish (12% time)
    d6 = t0 + budget_s * 0.96
    h6, c5_6 = _slsqp_optimize(best_h, min(d6, time.time() + budget_s * 0.05), max_outer=15)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)
    if time.time() < d6:
        h6b, c5_6b = _active_descent(best_h, d6, max_iter=300)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
            best_n = len(h6b)

    # Final SLSQP
    h_f, c5_f = _slsqp_optimize(best_h, deadline, max_outer=10)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(h_f)

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5 - 1e-12:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
