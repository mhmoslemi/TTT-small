def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    # Water-filling projection to [0,1] with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    # Exact quadratic line search
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid + golden section
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    
    return best_t, best_val

    M = float(S.max())
    shifted = (S - M) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_k = min(20, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad_smooth = w[top_idx] @ G
    grad_smooth -= grad_smooth.mean()
    if np.abs(grad_smooth).max() < 1e-12:
        return None
    norm = np.linalg.norm(grad_smooth)
    if norm < 1e-12:
        return None
    return -grad_smooth / norm

def _lp_loop(h, deadline, radius_init=0.12, K=15, use_smooth=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    tau = 0.02 * cur_c5
    while time.time() < deadline:
        if stall > 60:
            if restart_ctr < 10:
                scale = 0.01 + 0.01 * restart_ctr * 0.3
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                tau = 0.02 * cur_c5
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if (p is None or float(np.max(np.abs(p))) < 1e-13) and use_smooth:
            p = _smooth_direction(h, S, n, tau)
            if p is not None:
                p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if use_smooth and stall % 5 == 0:
                tau *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.06, 0.28)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.95, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 12 == 0 and radius < 0.015:
                radius = 0.05
    return best_h, best_c5

def _pair_refine_grad(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(888)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 60)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.006
    T_min = 0.000005
    cooldown = 0.9999
    patience = 0
    max_patience = 15000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 2:
                break
            T = max(T_min, T * 0.9)
            patience = 0
        if T < T_min:
            break
        for _ in range(200):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _multi_pair_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(444)
    stall = 0
    while time.time() < deadline and stall < 40:
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 4:
            break
        idx = rng.choice(len(interior), size=min(4, len(interior)), replace=False)
        i, j, k, l = interior[idx]
        for signs in [(-1, -1, 1, 1), (1, 1, -1, -1)]:
            d = np.zeros(n)
            d[i] = signs[0]
            d[j] = signs[1]
            d[k] = signs[2]
            d[l] = signs[3]
            alpha, c5_try = _line_search(h, d, dx, 0.3)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                break
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (8% time)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7]:
        for ph in [0.0, 0.7, 1.5, 2.3]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(8):
        starts.append(_block_init(n0, rng, rng.integers(3, 15)))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.008)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main LP optimization (30% time)
    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.14, K=15, use_smooth=False)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP with smooth backup (15% time)
    d3 = t0 + budget_s * 0.53
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.08, K=20, use_smooth=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP polish (12% time)
    d4 = t0 + budget_s * 0.65
    h4a, c5_4a = _binary_sa(best_h, t0 + budget_s * 0.58, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, radius_init=0.06, K=15)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Gradient-guided pair refinement (10% time)
    d5 = t0 + budget_s * 0.75
    h5, c5_5 = _pair_refine_grad(best_h, d5, max_passes=200)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (15% time)
    d6 = t0 + budget_s * 0.90
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, 24, -24]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.005:
            sub_dl = min(d6, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=12)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.02:
                sub_dl2 = min(d6, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d6:
                    h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + budget_s * 0.01),
                                         radius_init=0.05, K=12)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Aggressive perturbation + re-opt (6% time)
    d7 = t0 + budget_s * 0.96
    scale = 0.03
    for attempt in range(300):
        if time.time() > d7:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=12)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.2)

    # Phase 8: Final pair + multi-pair polish
    d8 = t0 + budget_s * 0.975
    h8, c5_8 = _pair_refine_grad(best_h, d8, max_passes=50)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline:
        h9, c5_9 = _multi_pair_refine(best_h, deadline)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final LP polish
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.03, K=10)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
