def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = np.clip(x + res / cnt, 0.0, 1.0)

    n = len(h)

    c1 = b_s - gr * (b_s - a_s)

    f1 = float(np.max(scores0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(scores0 + c2 * T - c2*c2 * C)) * dx
