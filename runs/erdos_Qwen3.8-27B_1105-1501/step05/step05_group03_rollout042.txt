import numpy as np
from scipy.optimize import minimize, linprog

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _exact_c5(h):
    return float(np.max(_scores(h)))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999
        else:
            mask = x > 0.001
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    ns = 2 * n - 1
    G = np.zeros((ns, n))
    for d in range(-(n - 1), n):
        p = d + n - 1
        if d >= 0:
            m = n - d
            if m > 0:
                G[p, :m] = (1.0 - h[d:d + m])
                if d > 0:
                    G[p, d:n] += (-h[0:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[p, g:n] = (1.0 - h[0:m])
                if g > 0:
                    G[p, :m] += (-h[g:n])
    return G * dx

def _lse_grad(h, beta):
    n = len(h)
    scores = _scores(h)
    max_s = np.max(scores)
    e = np.exp(beta * (scores - max_s))
    s = np.sum(e)
    if s < 1e-300:
        idx = np.argmax(scores)
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w = e / s
    G = _build_G(h)
    return G.T @ w

def _lse_descent(h, deadline, beta_start=20.0, beta_end=500.0, n_iters=300,
                 step_init=0.05, momentum=0.85, min_step=1e-5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    step = step_init
    stall = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        grad = _lse_grad(h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        grad -= np.mean(grad)
        v = momentum * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur - 1e-15:
            h = h_new
            cur = c5_new
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            step = min(step * 1.03, 0.15)
            stall = 0
        else:
            v = 0.5 * v
            step *= 0.8
            stall += 1
            if step < min_step or stall > 30:
                break
    return best_h, best_c5

def _slsqp_solver(h_init, deadline, max_active=25, max_rounds=8):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    active = np.argsort(_scores(h))[-max_active:]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        scores = _scores(h)
        t_init = float(np.max(scores))
        z0 = np.append(h, t_init)
        G_active = _build_G(h)[active, :]
        M = len(active)
        def obj(z):
            return z[n]
        def obj_jac(z):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        def eq_fun(z):
            return np.sum(z[:n]) - target
        def eq_jac(z):
            g = np.ones(n + 1)
            g[n] = 0.0
            return g
        def ineq_fun(z):
            return z[n] - _scores(z[:n])[active]
        def ineq_jac(z):
            J = np.zeros((M, n + 1))
            hz = z[:n]
            Gh = _build_G(hz)[active, :]
            J[:, :n] = -Gh
            J[:, n] = 1.0
            return J
        cons = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ]
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = minimize(obj, z0, method='SLSQP', jac=obj_jac,
                           constraints=cons, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.success or res.fun < t_init:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                # Update active set
                new_scores = _scores(h)
                active = np.argsort(new_scores)[-max_active:]
                if np.max(new_scores) < res.x[n] - 1e-10:
                    continue
        except Exception:
            break
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    G = _build_G(h)
    ns = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((ns, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, target, max_a=1.0):
    n = len(h)
    for j in range(n):
        if p[j] > 1e-14:
            max_a = min(max_a, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            max_a = min(max_a, h[j] / (-p[j]))
    max_a = max(max_a, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max_a
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(25):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, target))
    if opt_c5 <= min(f1, f2):
        return opt_a, opt_c5
    if f1 <= f2:
        return c1, f1
    return c2, f2

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng=None):
    x = np.arange(n) / n

    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
        h = _project(h, n / 2.0)
    return h

def _fourier_evolve(h, deadline, rng, n_pop=15, K=10, n_gen=100):

    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2 * (k - 1)] = np.cos(2 * np.pi * k * x)
        B[:, 2 * (k - 1) + 1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    a_init = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    mut = 0.08
    pop_a = a_init + mut * rng.standard_normal((n_pop, 2 * K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        ht = np.clip(0.5 + B @ pop_a[i], 0, 1)
        ht = _project(ht, target)
        pop_c5[i] = _exact_c5(ht)
    bi = int(np.argmin(pop_c5))
    best_c5 = pop_c5[bi]
    best_h = _project(np.clip(0.5 + B @ pop_a[bi], 0, 1), target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        nk = max(3, n_pop // 2)
        pop_a[:nk] = pop_a[order[:nk]]
        pop_c5[:nk] = pop_c5[order[:nk]]
        for i in range(nk, n_pop):
            p1 = order[rng.integers(0, nk)]
            p2 = order[rng.integers(0, nk)]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1 - mix) * pop_a[order[p2]]
            pop_a[i] += mut * rng.standard_normal(2 * K)
            ht = np.clip(0.5 + B @ pop_a[i], 0, 1)
            ht = _project(ht, target)
            pop_c5[i] = _exact_c5(ht)
        mut *= 0.99
        cb = int(np.argmin(pop_c5))
        if pop_c5[cb] < best_c5:
            best_c5 = pop_c5[cb]
            best_h = _project(np.clip(0.5 + B @ pop_a[cb], 0, 1), target)

                h_init = _quasi_init(n_try, alphas[ri % len(alphas)], rng)
            sub_dl = min(dl4, time.time() + time_per_res * 0.4)
            ht, ct = _lse_descent(h_init, sub_dl, beta_start=30, beta_end=400,
                                  n_iters=150, step_init=0.04, momentum=0.85)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try
            sub_dl2 = min(dl4, time.time() + time_per_res * 0.3)
            ht2, ct2 = _slsqp_solver(ht, sub_dl2, max_active=15, max_rounds=4)
            if ct2 < best_c5:
                best_c5 = ct2
                best_h = ht2.copy()
                best_n = n_try
            sub_dl3 = min(dl4, time.time() + time_per_res * 0.2)
            ht3, ct3 = _lp_optimize(ht2, sub_dl3, radius_init=0.06,
                                    max_iters=500, radius_min=0.003)
            if ct3 < best_c5:
                best_c5 = ct3
                best_h = ht3.copy()
                best_n = n_try

    # Phase 5: Fourier + perturbations (15% time)
    dl5 = t0 + budget_s * 0.87
    h_f, c5_f = _fourier_evolve(best_h, min(dl5, t0 + budget_s * 0.78), rng,
                                 n_pop=12, K=8, n_gen=80)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    # Perturbation restarts
    scale = 0.02
    for attempt in range(100):
        if time.time() > dl5:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.008)
        ht, ct = _lse_descent(h_p, sub_dl, beta_start=80, beta_end=300,
                              n_iters=40, step_init=0.03, momentum=0.8)
        if ct < best_c5 - 1e-12:
            best_c5 = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.08)

    # Phase 6: SLSQP on best (6% time)
    dl6 = t0 + budget_s * 0.93
    h6, c5_6 = _slsqp_solver(best_h, dl6, max_active=25, max_rounds=5)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final polish (5% time)
    dl7 = t0 + budget_s * 0.96
    hc, c5c = _coord_polish(best_h, dl7, n_cycles=2)
    if c5c < best_c5:
        best_c5 = c5c
        best_h = hc.copy()

    dl8 = t0 + budget_s * 0.985
    hp, c5p = _pair_polish(best_h, dl8, n_passes=2)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Final fine LP
    dl9 = t0 + budget_s * 0.998
    hf, cf = _lp_optimize(best_h, dl9, radius_init=0.005, max_iters=500, radius_min=0.0005)
    if cf < best_c5:
        best_c5 = cf
        best_h = hf.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
