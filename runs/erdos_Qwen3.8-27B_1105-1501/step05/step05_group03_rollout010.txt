def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0 - 1e-15) & (x > 0)
            if np.sum(mask) == 0:
                mask = x < 1.0
        else:
            mask = (x > 1e-15) & (x < 1.0)
            if np.sum(mask) == 0:
                mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
            # Second pass for any residual
            res2 = target - float(np.sum(x))
            if abs(res2) > 1e-14:
                mask2 = (x > 1e-15) & (x < 1.0 - 1e-15)
                if np.sum(mask2) > 0:
                    x[mask2] += res2 / int(np.sum(mask2))
                    x = np.clip(x, 0.0, 1.0)
    return x

def _build_basis(n, q, rng):
    """Build q sum-zero rectangular block basis vectors."""
    B = np.zeros((n, q))
    col = 0
    # Generate widths
    widths = []
    for w in [1, 2, 4, 8, 16, 32, 64]:
        if 2 * w <= n:
            widths.append(w)
    # Repeat widths to fill up to q columns
    while len(widths) < q:
        widths.append(widths[len(widths) % len(set(widths))])
    widths = widths[:q]
    
    for col in range(q):
        w = widths[col]
        # Choose two non-overlapping intervals of length w
        # First interval
        max_start1 = n - w
        start1 = int(rng.integers(0, max_start1 + 1))
        # Second interval must not overlap
        # Available positions for start2: [0, n-w] excluding [start1, start1+w) overlap
        # Actually they can be adjacent, just can't overlap
        valid_starts = np.array([s for s in range(n - w + 1) if 
                                  not (s < start1 + w and s + w > start1)])
        if len(valid_starts) == 0:
            # Fallback: place second right after first or before
            if start1 + 2 * w <= n:
                start2 = start1 + w
            else:
                start2 = max(0, start1 - w)
        else:
            start2 = int(valid_starts[rng.integers(len(valid_starts))])
        
        c = 1.0 / (2.0 * np.sqrt(w))
        B[start1:start1+w, col] = c
        B[start2:start2+w, col] = -c
        
        # Verify sum is zero
        s = B[:, col].sum()
        if abs(s) > 1e-10:
            # Shouldn't happen, but fix
            B[:, col] -= s / n
    
    return B

def _eval_y(B, y, n):
    """Evaluate C5 for parameter vector y in basis B."""
    target = n / 2.0
    raw = 0.5 + B @ y
    h = _project(raw, target)
    return _exact_c5(h), h

def _sturmian_init(n, alpha, rho=0.0):
    """Generate a Sturmian word of length n with slope alpha."""
    # Sturmian word: w(i) = floor((i+1)*alpha + rho) - floor(i*alpha + rho)
    i = np.arange(n)
    w = np.floor((i + 1) * alpha + rho) - np.floor(i * alpha + rho)
    return w.astype(float)

            break

        j = int(rng.integers(n))
        if i == j:
            continue

        if h[i] > 0.5:

                h = h_new.copy()
        else:
            # Try to move from j to i

def _coordinate_descent_y(B, y_best, n, deadline, rng, n_coords_per_iter=3, step_init=0.1):

        else:
            stall = 0
            step = min(step * 1.1, 0.5)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < (1.0 - 1e-14)
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
            res2 = target - float(np.sum(x))
            if abs(res2) > 1e-14:
                mask2 = (x > 1e-14) & (x < 1.0 - 1e-14)
                cnt2 = int(np.sum(mask2))
                if cnt2 > 0:
                    x[mask2] += res2 / cnt2
                    x = np.clip(x, 0.0, 1.0)
    return x

def _sturmian(n, alpha, rho):
    i = np.arange(n)
    w = np.floor((i + 1) * alpha + rho) - np.floor(i * alpha + rho)
    return w.astype(float)

def _sturmian_best(n, alpha):
    best_h = None
    best_c5 = float('inf')
    for rho in np.linspace(0, alpha, 30):
        w = _sturmian(n, alpha, rho)
        w = _project(w, n / 2.0)
        c5 = _exact_c5(w)
        if c5 < best_c5:
            best_c5 = c5
            best_h = w.copy()
    return best_h, best_c5

def _build_basis(n, q, rng):
    B = np.zeros((n, q))
    col = 0
    while col < q:
        w = int(2 ** (rng.integers(1, 5)))
        if 2 * w > n:
            w = max(1, n // 4)
        if 2 * w > n:
            w = 1
        max_s1 = n - w
        s1 = int(rng.integers(0, max_s1 + 1))
        exclude_lo = max(0, s1 + w - (n - w))
        exclude_hi = min(n - w, s1)
        valid = []
        for s2 in range(n - w + 1):
            if not (s2 < s1 + w and s2 + w > s1):
                valid.append(s2)
        if len(valid) == 0:
            s2 = min(s1 + w, n - w)
        else:
            s2 = valid[int(rng.integers(len(valid)))]
        c = 1.0 / (2.0 * np.sqrt(w))
        B[s1:s1 + w, col] = c
        B[s2:s2 + w, col] = -c
        s = B[:, col].sum()
        if abs(s) > 1e-12:
            B[:, col] -= s / n
        col += 1
    return B

def _eval_y(B, y, n):
    target = n / 2.0
    raw = 0.5 + B @ y
    h = _project(raw, target)
    return _exact_c5(h), h

def _diffevo(B, y0, n, max_time, rng, pop_size=36, F=0.5, CR=0.7):
    q = len(y0)
    deadline = time.time() + max_time
    lo_b, hi_b = -3.0, 3.0
    pop = np.zeros((pop_size, q))
    fits = np.zeros(pop_size)
    pop[0] = y0
    fits[0], _ = _eval_y(B, y0, n)
    for i in range(1, pop_size):
        pop[i] = np.clip(y0 + rng.standard_normal(q) * 0.4, lo_b, hi_b)
        fits[i], _ = _eval_y(B, pop[i], n)
    best_i = int(np.argmin(fits))
    best_y = pop[best_i].copy()
    best_c5 = fits[best_i]
    best_h = _project(0.5 + B @ best_y, n / 2.0)
    stall = 0
    cur_F = F
    while time.time() < deadline:
        improved = False
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = int(rng.integers(pop_size))
            r2 = int(rng.integers(pop_size))
            r3 = int(rng.integers(pop_size))
            while r2 == r1:
                r2 = int(rng.integers(pop_size))
            while r3 == r1 or r3 == r2:
                r3 = int(rng.integers(pop_size))
            trial = pop[r1] + cur_F * (pop[r2] - pop[r3])
            jr = int(rng.integers(q))
            for j in range(q):
                if rng.random() >= CR and j != jr:
                    trial[j] = pop[i, j]
            np.clip(trial, lo_b, hi_b, out=trial)
            c5_t, h_t = _eval_y(B, trial, n)
            if c5_t <= fits[i]:
                pop[i] = trial
                fits[i] = c5_t
                improved = True
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_y = trial.copy()
                    best_h = h_t.copy()
        if not improved:
            stall += 1
            if stall > 4:
                nr = max(3, pop_size // 5)
                idx = rng.choice(pop_size, nr, replace=False)

                    pop[i] = np.clip(best_y + rng.standard_normal(q) * 0.25, lo_b, hi_b)

            stall = 0
        else:
            stall = 0
            cur_F = max(cur_F * 0.99, 0.3)
    return best_y, best_c5, best_h

def _coord_descent_y(B, y_best, n, deadline, rng, step_init=0.08):
    q = len(y_best)
    best_y = y_best.copy()
    best_c5, _ = _eval_y(B, best_y, n)
    step = step_init
    stall = 0
    order = list(range(q))
    while time.time() < deadline and stall < 40:
        rng.shuffle(order)
        improved = False
        for c in order:
            if time.time() > deadline:
                break
            for sign in [1, -1]:
                trial = best_y.copy()
                trial[c] += sign * step
                c5_t, _ = _eval_y(B, trial, n)
                if c5_t < best_c5 - 1e-15:
                    best_y = trial
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            step *= 0.7
            stall += 1
        else:
            stall = 0
            step = min(step * 1.15, 0.5)
    return best_y, best_c5, _project(0.5 + B @ best_y, n / 2.0)

def _pair_polish(h, deadline, n_iters, rng):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if h[i] >= h[j]:
            delta_max = min(h[i], 1.0 - h[j])
        else:
            delta_max = min(h[j], 1.0 - h[i])
        if delta_max < 1e-12:
            continue
        delta = delta_max * rng.random() * 0.6
        if delta < 1e-12:
            continue
        h_new = h.copy()
        if h[i] >= h[j]:
            h_new[i] -= delta
            h_new[j] += delta
        else:
            h_new[i] += delta
            h_new[j] -= delta
        np.clip(h_new, 0.0, 1.0, out=h_new)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    return best_h, best_c5

def _block_polish(h, deadline, rng, n_trials=2000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        w = int(rng.integers(1, max(2, n // 4) + 1))
        if 2 * w > n:
            continue
        s1 = int(rng.integers(0, n - w + 1))
        s2 = int(rng.integers(0, n - w + 1))
        if s1 == s2 or (s2 < s1 + w and s1 < s2 + w):
            continue
        h_new = h.copy()
        h_new[s1:s1+w], h_new[s2:s2+w] = 1.0 - h[s2:s2+w].copy(), 1.0 - h[s1:s1+w].copy()
        np.clip(h_new, 0.0, 1.0, out=h_new)
        if abs(float(np.sum(h_new)) - n / 2.0) > 1e-8:
            h_new = _project(h_new, n / 2.0)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def _resample_arr(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
