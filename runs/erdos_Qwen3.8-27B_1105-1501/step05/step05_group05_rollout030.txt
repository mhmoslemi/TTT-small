import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _sigmoid(x):
    """Numerically stable sigmoid."""
    x = np.clip(x, -500, 500)
    result = np.empty_like(x)
    pos = x >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    result[~pos] = ex / (1.0 + ex)
    return result

def _logit_map(v):
    """Map v-vector to feasible h: h_i = sigma(v_i + theta) with sum(h) = n/2."""
    n = len(v)
    target = n / 2.0
    v = np.clip(v, -50.0, 50.0)
    # Bisection for theta
    lo = -float(np.max(v)) - 40.0
    hi = -float(np.min(v)) + 40.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = _sigmoid(v + mid)
        s = float(np.sum(h))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = _sigmoid(v + theta)
    # Fix residual
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        # Adjust interior points
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(interior))
        if cnt > 0:
            adj = res / cnt
            if -1e-6 < adj < 1e-6:
                h[interior] += adj
        h = np.clip(h, 0.0, 1.0)
        # Final exact fix
        h = _project(h, target)
    return h

def _h_to_logit(h, eps=1e-4):
    """Map feasible h to logit v (without the shift theta)."""
    h = np.clip(h, eps, 1.0 - eps)
    v = np.log(h / (1.0 - h))
    # Center: remove mean so that theta=0 approximately
    v -= np.mean(v)
    return v

def _line_search(h, d, dx, trust_radius):
    """Quadratic line search along direction d (zero-mean)."""
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 30
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(25):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _pair_refine(h, deadline, max_passes=30, rng=None, use_random=False):
    """Iterate over pairs of interior coordinates, trying transfers."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    passes = 0
    while passes < max_passes and time.time() < deadline:
        passes += 1
        improved = False
        if use_random and rng is not None and n > 80:
            # Random subset of pairs
            interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
            if len(interior) > 2:
                n_pairs = min(200, len(interior) * (len(interior) - 1) // 2)
                idx_a = rng.integers(0, len(interior), n_pairs)
                idx_b = rng.integers(0, len(interior), n_pairs)
                pairs = set()
                for i in range(n_pairs):
                    a, b = interior[idx_a[i]], interior[idx_b[i]]
                    if a != b:
                        pairs.add((min(a, b), max(a, b)))
        else:
            pairs = set()
            for i in range(n):
                if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                    continue
                for j in range(i + 1, n):
                    if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                        continue
                    pairs.add((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)

                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _smooth_grad(h, dx, beta):
    """Log-sum-exp smooth gradient."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(np.clip(v, -500, 0))
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _smooth_phase(h, deadline, beta_list, lr_init=0.02):
    """Iterate on smooth gradient with momentum."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.85
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 50:
            bi += 1
            if bi >= len(beta_list) * 4:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = beta_list[bi % len(beta_list)]
        d = _smooth_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            bi += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_solve_local(h, radius):
    """LP-based local step to reduce max correlation."""
    from scipy.optimize import linprog
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    # Build gradient matrix G[r, i] = d score_r / d h_i
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    # LP: minimize t s.t. G @ p - t <= -scores, sum(p) = 0, |p_i| <= radius
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.2):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init

        if stall > 80:
            break
        p = _lp_solve_local(h, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.08, 0.55)
        else:
            stall += 1
            radius *= 0.73
            if stall > 30 and radius < 0.04:
                radius = best_radius * 0.5
                stall = 0
    return best_h, best_c5

def _kick_and_reopt(h, deadline, rng, n_kicks=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.08)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.12)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

            pairs = []
            for i in range(n):
                if h[i] <= 5e-9 or h[i] >= 1.0 - 5e-9:
                    continue
                for j in range(i + 1, n):
                    if h[j] <= 5e-9 or h[j] >= 1.0 - 5e-9:
                        continue
                    pairs.append((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.49)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _kick_and_reopt(h, deadline, rng, n_kicks=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.09)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.12)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _les_search(h_start, deadline, n, rng):
    """Logistic Evolution Strategy for global exploration."""
    v0 = _h_to_logit(h_start, eps=1e-4)
    pop_size = 16
    population = []
    for i in range(4):
        v = v0 + rng.normal(0, 0.06, n)
        h_cand = _logit_map(v, n)
        population.append((v, h_cand, _exact_c5(h_cand)))
    v_refl = v0[::-1].copy()
    population.append((v_refl.copy(), _logit_map(v_refl, n), _exact_c5(_logit_map(v_refl, n))))
    for shift in [1, 2, -1, -2]:
        v_sh = np.roll(v0, shift)
        h_sh = _logit_map(v_sh, n)
        population.append((v_sh, h_sh, _exact_c5(h_sh)))
    for i in range(5):
        v_rand = rng.normal(0, 0.6, n)
        h_rand = _logit_map(v_rand, n)
        population.append((v_rand, h_rand, _exact_c5(h_rand)))
    population.sort(key=lambda x: x[2])
    best_v, best_h = population[0][0], population[0][1]
    best_c5 = population[0][2]
    sigma = 0.15
    elite = 3
    no_improve = 0
    gen = 0
    max_gen = 500
    while time.time() < deadline and gen < max_gen:
        gen += 1
        offspring = [(v[0].copy(), v[1].copy(), v[2]) for v in population[:elite]]
        n_off = pop_size - len(offspring)
        for _ in range(n_off):
            p1 = population[rng.integers(0, min(6, len(population)))]
            p2 = population[rng.integers(0, min(6, len(population)))]
            v1, v2 = p1[0], p2[0]
            op = rng.random()
            if op < 0.35:
                v_child = v1 + sigma * rng.normal(0, 1, n)
            elif op < 0.55:
                mask = rng.random(n) > 0.5
                v_child = np.where(mask, v1, v2) + sigma * 0.5 * rng.normal(0, 1, n)
            elif op < 0.7:
                v_child = v1.copy()
                if n > 4:
                    ia = rng.integers(0, n)
                    ib = rng.integers(0, n)
                    v_child[ia], v_child[ib] = v_child[ib], v_child[ia]
                v_child += sigma * 0.3 * rng.normal(0, 1, n)
            elif op < 0.85:
                v_child = v1.copy()
                if n > 6:
                    a = rng.integers(0, n - 4)
                    b = rng.integers(a + 4, n)
                    v_child[a:b] = v2[a:b]
                v_child += sigma * 0.3 * rng.normal(0, 1, n)
            else:
                v_child = v1.copy()
                if n > 6:
                    a = rng.integers(0, n - 4)
                    b = rng.integers(a + 4, n)
                    v_child[a:b] = -v_child[a:b]
                v_child += sigma * 0.2 * rng.normal(0, 1, n)
            v_child = np.clip(v_child, -40.0, 40.0)
            h_child = _logit_map(v_child, n)
            c5_child = _exact_c5(h_child)
            offspring.append((v_child, h_child, c5_child))
        population = sorted(offspring + population, key=lambda x: x[2])[:pop_size]
        if population[0][2] < best_c5 - 1e-12:
            best_c5 = population[0][2]
            best_h = population[0][1].copy()
            best_v = population[0][0].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 12:
            sigma = max(0.02, sigma * 0.88)
        if no_improve > 35:
            for i in range(3):
                v_rand = best_v + rng.normal(0, 0.8, n)
                v_rand = np.clip(v_rand, -40, 40)
                h_rand = _logit_map(v_rand, n)
                population[i + 1] = (v_rand, h_rand, _exact_c5(h_rand))
            population = sorted(population, key=lambda x: x[2])[:pop_size]
            no_improve = 0
            sigma = max(0.06, sigma * 1.5)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2.0
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None
    if h0 is not None:
        n0 = len(h0)
        h = _project(h0, n0 / 2.0)
    else:
        n0 = 128
        x = np.arange(n0)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * 3 * x / n0) + 0.05 * np.sin(2 * np.pi * 7 * x / n0)
        h = _project(h, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n = len(best_h)
    # Phase 1: Initial LP (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.18)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Phase 2: LES global exploration (30%)
    d2 = t0 + budget_s * 0.38
    h_les, c5_les = _les_search(best_h, d2, n, rng)
    if c5_les < best_c5:
        h_les = _project(h_les, n / 2.0)
        c5_les = _exact_c5(h_les)
        if c5_les < best_c5:
            best_c5 = c5_les
            best_h = h_les.copy()
    # Phase 3: Intensive LP (12%)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.20, use_penalty=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Phase 4: Smooth (8%)
    d4 = t0 + budget_s * 0.58
    h4, c5_4 = _smooth_phase(best_h, d4, beta_list=[50, 120, 300, 700], lr_init=0.02)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Phase 5: Multi-resolution (18%)
    d5 = t0 + budget_s * 0.76
    current_n = len(best_h)
    tried = set()
    offsets = sorted(range(-24, 25, 2), key=lambda x: abs(x))
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = current_n + dn
        if n_try < 60 or n_try > 220 or n_try == current_n:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.15)
        if c5_r < best_c5 + 0.0003 and time.time() < d5 - 3:
            sub_dl2 = min(d5, time.time() + 6.0)
            h_r, c5_r = _pair_refine(h_r, sub_dl2, max_passes=25, rng=rng, use_random=True)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            current_n = n_try
    # Phase 6: Kick (8%)
    d6 = t0 + budget_s * 0.84
    h6, c5_6 = _kick_and_reopt(best_h, d6, rng, n_kicks=30)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Phase 7: Pair refinement exhaustive (8%)
    d7 = t0 + budget_s * 0.92
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=60, use_random=False)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    # Phase 8: Final LP polish (5%)
    d8 = deadline
    if time.time() < d8:
        h8, c5_8 = _lp_loop(best_h, d8, radius_init=0.05, use_penalty=True)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    # Final pair pass if time
    if time.time() < deadline - 0.5:
        h9, c5_9 = _pair_refine(best_h, deadline, max_passes=20, use_random=False)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
