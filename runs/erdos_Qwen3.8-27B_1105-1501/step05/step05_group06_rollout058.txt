import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            idxs = np.arange(lo1, hi1 + 1)
            h_idx = idxs + (n - 1 - k)
            G[k, idxs] += dx * (1.0 - h[h_idx])
        lo2 = max(0, n - 1 - k)
        hi2 = min(n - 1, 2 * n - 2 - k)
        if lo2 <= hi2:
            idxs = np.arange(lo2, hi2 + 1)
            h_idx = k + idxs - (n - 1)
            G[k, idxs] += -dx * h[h_idx]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pnorm_value_grad(F, G, q):
    L = len(F)
    M = float(np.max(F))
    if M < 1e-30:
        return 0.0, np.zeros(G.shape[1])
    u = np.clip(F / M, 0.0, 1.0)
    T = float(np.sum(u ** q))
    if T < 1e-30:
        return 0.0, np.zeros(G.shape[1])
    val = M * (T ** (1.0 / q))
    w = u ** (q - 1)
    factor = T ** (1.0 / q - 1.0)
    grad = factor * (G.T @ w)
    return val, grad

def _proj_grad(h, q_list, target, deadline, alpha_init=0.03, max_stall=60):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for q in q_list:
        if time.time() > deadline:
            break
        cur_c5 = _exact_c5(h)
        alpha = alpha_init
        stall = 0
        momentum = np.zeros(n)
        beta_mom = 0.75
        while time.time() < deadline and stall < max_stall:
            F = _scores(h)
            G = _build_G(h)
            _, grad = _pnorm_value_grad(F, G, q)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-14:
                break
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
            direction = -momentum
            found = False
            for trial in range(12):
                a = alpha * (0.5 ** trial)
                h_new = _project(h + a * direction, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    h = h_new
                    cur_c5 = c5_new
                    alpha = min(a * 2.0, alpha_init)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                alpha *= 0.5
    return best_h, best_c5

def _slsqp_optimize(h, q, target, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    def obj(x):
        F = _scores(x)
        M = float(np.max(F))
        if M < 1e-30:
            return 0.0, np.zeros(n)
        u = np.clip(F / M, 0.0, 1.0)
        T = float(np.sum(u ** q))
        if T < 1e-30:
            return 0.0, np.zeros(n)
        val = M * (T ** (1.0 / q))
        w = u ** (q - 1)
        G = _build_G(x)
        factor = T ** (1.0 / q - 1.0)
        return val, factor * (G.T @ w)

    def mass_fun(x):
        return float(np.sum(x)) - target

    def mass_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': mass_fun, 'jac': mass_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=True, bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            x_new = _project(res.x, target)
            c5_new = _exact_c5(x_new)
            if c5_new < best_c5:
                best_h = x_new
                best_c5 = c5_new
    except Exception:
        pass
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=300):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > 1e-6 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
