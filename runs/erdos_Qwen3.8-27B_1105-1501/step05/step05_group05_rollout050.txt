def _lp_solve(h, scores, radius):
    """LP for trust-region direction."""
    n = len(h)
    # Build gradient matrix G where G[r, i] = d(score_r)/d(h_i)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})

        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _line_search(h, d, dx, s0, trust_radius):
    n = len(h)
    d = d - d.mean()
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(s0)) * dx
    n_grid = 21
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = s0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(s0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(s0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(28):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(s0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(s0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(s0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _pair_refine(h, deadline, max_passes=300, use_random=False, rng=None):

    target = n / 2.0
    h = h.copy()

    for _pass in range(max_passes):

            break
        improved = False
        s0 = np.correlate(h, 1.0 - h, mode='full')
        interior = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(interior) < 2:
            break
        if use_random and rng is not None and len(interior) > 60:
            for _ in range(800):
                if time.time() > deadline:
                    break
                ii = int(rng.integers(len(interior)))
                ij = int(rng.integers(len(interior)))
                if ii == ij:

                i, j = interior[ii], interior[ij]
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0

                        best_c5 = _exact_c5(h)
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def _lp_solve(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_local(h, deadline, radius_init=0.20):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_solve(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if radius < 1e-7:
                break
            continue
        s0 = scores
        d = p
        alpha, new_c5 = _line_search(h, d, dx, s0, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.45)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kick_reopt(h, deadline, rng, n_kicks=25):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.01, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-15:
            continue
        direction /= norm
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)

            best_h = h_t.copy()
