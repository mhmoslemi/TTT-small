import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    if np.max(np.abs(d)) < 1e-16:
        return 0.0, _exact_c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(45):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _subgrad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-18:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += w[r] * (1.0 - h[d:d + m])
            grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            grad[:m] += w[r] * (-h[g:g + m])
            grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _optimize_grad(h, deadline, beta_list=(30, 60, 100, 200, 400, 800), lr_init=0.025):

    dx = 2.0 / n
    target = n / 2.0

    best_h = h.copy()

    lr = lr_init

    decay = 0.83
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 60:
            break
        if stall > 25:
            bi += 1
            if bi >= 24:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = beta_list[min(bi, len(beta_list) - 1)]
        d = _subgrad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            bi += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, mom, dx, lr * 3.5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0

    return best_h, best_c5

def _active_balance(h, deadline, rng, max_iter=200):
    """Find direction that reduces the most active lags simultaneously."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _it in range(max_iter):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        active = np.where(scores >= s_max - 0.02 * s_max)[0]
        if len(active) < 2:
            break
        # Build gradient matrix for active lags
        n_act = len(active)
        G = np.zeros((n_act, n))
        for idx, r in enumerate(active):
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                G[idx, :m] += (1.0 - h[d:d + m])
                G[idx, d:] += -h[:m]
            else:
                g = -d
                m = n - g
                G[idx, :m] += (1.0 - h[g:g + m])
                G[idx, g:] += -h[:m]
        # Find min-norm direction that decreases all active scores
        # Solve: minimize ||G d||^2 subject to sum(d)=0, ||d||_inf <= 1
        # Approximate: d = -G^T (G G^T + lambda I)^{-1} 1
        lam = 0.1
        GGt = G @ G.T + lam * np.eye(n_act)
        ones = np.ones(n_act)
        try:
            coeff = np.linalg.solve(GGt, ones)
            d = G.T @ coeff
        except np.linalg.LinAlgError:
            break
        d = -d
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        d = np.clip(d, -1.0, 1.0)
        d -= d.mean()
        alpha, c5_try = _line_search(h, d, dx, 0.15)
        if c5_try < best_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
    return best_h, best_c5
