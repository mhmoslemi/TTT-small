```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    # Binary search for lambda such that sum(clip(v - lambda, 0, 1)) = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    F = _scores(h)  # shape (2n-1,)
```

```python

from scipy.optimize import minimize
import time

def _exact_c5(h):
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):

    if target <= 0:

    if target >= n:
        return np.ones(n)

    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))

    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):

        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_val(h, q):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Smax = float(F.max())
    if Smax < 1e-300:
        return 0.0
    u = F / Smax
    U = float(np.sum(u ** q))
    if U < 1e-300:
        return Smax
    return Smax * U ** (1.0 / q)

def _pnorm_grad(h, q):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Smax = float(F.max())
    if Smax < 1e-300:
        return np.zeros(len(h))
    u = F / Smax
    U = float(np.sum(u ** q))
    if U < 1e-300:
        return np.zeros(len(h))
    uqm1 = u ** (q - 1.0)
    G = _build_G(h)
    return U ** ((1.0 - q) / q) * (G.T @ uqm1)

def _sum_eq(x):
    return float(np.sum(x)) - len(x) / 2.0

def _sum_eq_jac(x):
    return np.ones(len(x))

    def __init__(self, q):
        self.q = q
    def __call__(self, h):
        return _pnorm_val(h, self.q)
    def grad(self, h):
        return _pnorm_grad(h, self.q)

def _slsqp_chain(h0, q_list, deadline, maxiter_per_q=80):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    
    for q in q_list:
        if time.time() > deadline:
            break
        obj = _Obj(q)
        try:
            res = minimize(obj, h, jac=obj.grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_q, 'ftol': 1e-12})
        except Exception:
            break
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_cand = _project(res.x, n / 2.0)
            c5 = _exact_c5(h_cand)
            h = h_cand
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()
        # Also evaluate current h
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def _proj_grad(h0, deadline, q=64, max_iter=5000):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        _, grad = _pnorm_grad(h, q)
        grad_tan = grad - grad.mean()
        gn = np.linalg.norm(grad_tan)
        if gn < 1e-14:
            break
        grad_tan /= gn
        # Backtracking line search
        alpha = 0.1
        f_cur = _pnorm_val(h, q)
        improved = False
        for _ in range(40):
            h_try = _project(h - alpha * grad_tan, target)
            f_try = _pnorm_val(h_try, q)
            if f_try < f_cur - 1e-14:
                h = h_try
                f_cur = f_try
                improved = True
                break
            alpha *= 0.65
        if improved:
            stall = 0
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    h = ((x * alpha + phase) % 1.0) < 0.5
    h = h.astype(float) + rng.normal(0, 0.03, n)

def _pair_polish(h, deadline, max_passes=200):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if cur_c5 - float(np.max(scores + alpha * C)) > 1e-8 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_val(h, q):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Smax = float(F.max())
    if Smax < 1e-300:
        return 0.0
    u = F / Smax
    U = float(np.sum(u ** q))
    if U < 1e-300:
        return Smax
    return Smax * U ** (1.0 / q)

def _pnorm_grad(h, q):
    F = _scores(h)
    F = np.maximum(F, 0.0)
    Smax = float(F.max())
    if Smax < 1e-300:
        return np.zeros(len(h))
    u = F / Smax
    U = float(np.sum(u ** q))
    if U < 1e-300:
        return np.zeros(len(h))
    uqm1 = u ** (q - 1.0)
    G = _build_G(h)
    return U ** ((1.0 - q) / q) * (G.T @ uqm1)

def _sum_eq(x):
    return float(np.sum(x)) - len(x) / 2.0

def _sum_eq_jac(x):
    return np.ones(len(x))

class _Obj:
    def __init__(self, q):
        self.q = q
    def __call__(self, h):
        return _pnorm_val(h, self.q)
    def grad(self, h):
        return _pnorm_grad(h, self.q)

def _slsqp_chain(h0, q_list, deadline, maxiter_per_q=80):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    for q in q_list:
        if time.time() > deadline:
            break
        obj = _Obj(q)
        try:
            res = minimize(obj, h, jac=obj.grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_q, 'ftol': 1e-12})
        except Exception:
            break
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_cand = _project(res.x, n / 2.0)
            c5 = _exact_c5(h_cand)
            h = h_cand
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _proj_grad(h0, deadline, q=64, max_iter=5000):

    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

            break

        grad_tan = grad - grad.mean()
        gn = np.linalg.norm(grad_tan)

                f_cur = f_try

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _golden_init(n, alpha, phase, rng):
    x = (np.arange(n) + 0.5) / n
    h = ((x * alpha + phase) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _sine_init(n, k, phase, rng):

    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * k * x + phase)

def _pair_polish(h, deadline, max_passes=200):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > 1e-9 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _topk_grad_step(h, deadline, n_steps=200, beta=50.0, top_k=10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        if top_k < len(scores):
            idx = np.argsort(scores)[-top_k:]
        else:
            idx = np.arange(len(scores))
        s = scores[idx]
        max_s = float(s.max())
        w = np.exp(beta * (s - max_s))
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w = w / sw
        full_w = np.zeros(len(scores))
        full_w[idx] = w
        G = _build_G(h)
        grad = G.T @ full_w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.05
        improved = False
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if not improved:
            break
    return best_h, best_c5
```
