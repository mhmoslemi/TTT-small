import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj_grad(h, tau, dx):
    """Compute LSE objective and gradient."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(scores)
    z = np.exp((scores - M) / tau)
    W = np.sum(z)
    obj = M + tau * np.log(W)
    w = z / W  # weights summing to 1
    
    # Compute gradient: grad_j = sum_r w_r * dS_r/dh_j
    # S_r = dx * corr_r where corr is the full correlation
    # dS_r/dh_j for lag r:
    #   r >= 0: (1-h[j-r])*1[j>=r] - h[j+r]*1[j<=n-r-1]  (times dx)
    #   r < 0, g=-r: (1-h[j+g])*1[j<=n-g-1] - h[j-g]*1[j>=g]  (times dx)
    
    grad = np.zeros(n)
    # Lag offsets go from -(n-1) to (n-1), indexed as 0 to 2n-2 in the correlation
    # correlation index c corresponds to lag d = c - (n-1)
    for c in range(2 * n - 1):
        d = c - (n - 1)  # lag
        if w[c] < 1e-15:
            continue
        wd = w[c] * dx
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            # For j in [d, n-1]: contribution (1-h[j-d])
            j_start = d
            j_end = n  # exclusive
            grad[j_start:j_end] += wd * (1.0 - h[:m])
            # For j in [0, m-1]: contribution -h[j+d]
            grad[:m] += -wd * h[d:]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            # For j in [0, m-1]: contribution (1-h[j+g])
            grad[:m] += wd * (1.0 - h[g:])
            # For j in [g, n-1]: contribution -h[j-g]
            grad[g:] += -wd * h[:m]
    return obj, grad

def _line_search_exact(h, d, radius, n_grid=80):
    """Exact line search using quadratic model along direction d (sum d = 0)."""
    n = len(h)
    dx = 2.0 / n
    # Normalize direction
    d = d.copy()
    d -= np.mean(d)  # ensure zero sum
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-15:
        return 0.0, _exact_c5(h)
    d = d / max_abs  # normalize so max |d_i| = 1
    
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find box interval
    t_lo, t_hi = -radius, radius
    pos = d > 1e-10
    neg = d < -1e-10
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-12:
        return 0.0, _exact_c5(h)
    
    # Grid search
    ts = np.linspace(t_lo, t_hi, n_grid)
    phis = np.array([dx * np.max(S0 + t * T - t*t * Q) for t in ts])
    best_idx = np.argmin(phis)
    
    # Refine around best
    if best_idx > 0 and best_idx < n_grid - 1:
        t_ref_lo = ts[max(0, best_idx - 2)]
        t_ref_hi = ts[min(n_grid - 1, best_idx + 2)]
    else:
        t_ref_lo = ts[0]
        t_ref_hi = ts[-1]
    ts2 = np.linspace(t_ref_lo, t_ref_hi, 200)
    phis2 = np.array([dx * np.max(S0 + t * T - t*t * Q) for t in ts2])
    best_idx2 = np.argmin(phis2)
    
    best_t = ts2[best_idx2]
    best_phi = phis2[best_idx2]
    return best_t, best_phi
