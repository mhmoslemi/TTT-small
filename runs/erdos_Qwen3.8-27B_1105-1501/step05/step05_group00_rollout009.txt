def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def active_rows_and_weights(h, n_active=8, tau_frac=0.02):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = float(np.max(corr))
    if max_val <= 0:
        return np.array([0], dtype=int), np.array([1.0])
    tau = tau_frac * max_val
    threshold = max_val - tau
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    if len(active) > n_active:
        order = np.argsort(-corr[active])[:n_active]
        active = active[order]
    vals = corr[active]
    weights = np.exp((vals - max_val) / max(tau, 1e-10))
    weights /= weights.sum()
    return active, weights

def weighted_gradient(h, active, weights):
    n = len(h)
    grad = np.zeros(n)
    for r, w in zip(active, weights):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            for i in range(m):
                grad[i] += w * (1.0 - h[i + d])
                grad[i + d] += -w * h[i]
        else:
            g = -d
            m = n - g
            for i in range(m):
                grad[i + g] += w * (1.0 - h[i])
                grad[i] += -w * h[i + g]
    return grad

def line_search_pair_fast(h, i, j, base_corr, dx):
    n = len(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = max(h[i] - 1.0, -h[j])
    t_hi = min(h[i], 1.0 - h[j])
    if t_hi - t_lo < 1e-14:
        return 0.0, c5_exact(h)
    S0 = base_corr
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = float(obj_grid[best_i])
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(35):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val

def resample_to(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def structured_init(n, rng):

    h6, c5_6 = pair_transfer_refine(best_h, d6, top_d=25, top_r=25, max_rounds=20000, tau_shrink=0.9)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    h6b, c5_6b = multi_coord_refine(best_h, d6, n_coords=2, max_iters=5000)
    if c5_6b < best_c5:
        best_c5 = c5_6b
        best_h = h6b.copy()

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
