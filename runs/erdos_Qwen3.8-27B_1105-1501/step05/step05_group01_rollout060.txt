def _line_search_exact(h, d, dx, trust_radius, n_samples=30):

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds
    eps = 1e-15
    pos = d > eps
    neg = d < -eps
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Sample the line
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = []
    for t in ts:
        vals.append(float(np.max(S0 + t * T - t*t * C)) * dx)
    vals = np.array(vals)
    
    # Find best region
    best_idx = int(np.argmin(vals))
    # Get a bracket around the best
    if best_idx == 0:
        bracket_lo, bracket_hi = t_lo, ts[1]
    elif best_idx == n_samples - 1:
        bracket_lo, bracket_hi = ts[-2], t_hi
    else:
        bracket_lo = ts[max(0, best_idx - 2)]
        bracket_hi = ts[min(n_samples-1, best_idx + 2)]
    
    # Golden section within bracket
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = bracket_hi - gr * (bracket_hi - bracket_lo)
    c2 = bracket_lo + gr * (bracket_hi - bracket_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(40):
        if bracket_hi - bracket_lo < 5e-13:
            break
        if f1 < f2:
            bracket_hi = c2; c2 = c1; f2 = f1
            c1 = bracket_hi - gr * (bracket_hi - bracket_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            bracket_lo = c1; c1 = c2; f1 = f2

            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx

        opt_t = ts[best_idx]
        opt_val = vals[best_idx]

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    """Try swapping contiguous blocks of ones with contiguous blocks of zeros."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        # Find runs of 1s and runs of 0s (in binary sense)
        # For continuous h, treat h > 0.5 as "1"
        binary = (h > 0.5).astype(int)
        
        for bs in block_sizes:
            if time.time() > deadline:
                break
            # Find all blocks of bs consecutive 1s and bs consecutive 0s
            ones_blocks = []
            zeros_blocks = []
            for i in range(n - bs + 1):
                if np.all(binary[i:i+bs] == 1):
                    ones_blocks.append(i)
                if np.all(binary[i:i+bs] == 0):
                    zeros_blocks.append(i)
            
            for ib in ones_blocks:
                for zb in zeros_blocks:
                    if abs(ib - zb) <= bs:
                        continue  # Skip overlapping blocks
                    d = np.zeros(n)
                    d[ib:ib+bs] = 1.0
                    d[zb:zb+bs] = -1.0
                    alpha, c5_try = _line_search(h, d, dx, 0.5)

                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
                        if time.time() > deadline:
                            break
        if not improved:
            break
    
    return best_h, best_c5

def _block_swap_pass(h, deadline, block_sizes=(1,2,3), n_passes=5, max_pairs=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        binary = (h > 0.5).astype(int)
        
        for bs in block_sizes:
            if time.time() > deadline:
                break
            # Find start positions of blocks of size bs
            ones_starts = []
            zeros_starts = []
            for i in range(n - bs + 1):
                if np.all(binary[i:i+bs] == 1):
                    ones_starts.append(i)
                if np.all(binary[i:i+bs] == 0):
                    zeros_starts.append(i)
            
            # Generate valid pairs (non-overlapping)
            pairs = []
            for ib in ones_starts:
                for zb in zeros_starts:
                    # Check non-overlap
                    if zb < ib + bs and ib < zb + bs:
                        continue
                    pairs.append((ib, zb))
            
            if len(pairs) > max_pairs:
                pairs = [pairs[i] for i in rng.choice(len(pairs), max_pairs, replace=False)]
            
            for ib, zb in pairs:
                if time.time() > deadline:
                    break
                d = np.zeros(n)
                d[ib:ib+bs] = -1.0
                d[zb:zb+bs] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    
    return best_h, best_c5

def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

            lo = mid
        else:
            hi = mid

        cnt = int(np.sum(mask))

            b_lo = c1; c1 = c2; f1 = f2
            c2 = b_lo + gr * (b_hi - b_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (b_lo + b_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    if vals[best_idx] < opt_val:
        opt_t = ts[best_idx]
        opt_val = vals[best_idx]
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    dx = 2.0 / n
    while time.time() < deadline:
        if stall > 90:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.62
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.40)
        else:
            stall += 1
            radius *= 0.70
            if stall % 12 == 0 and radius < 0.02:
                radius = 0.10
    return best_h, best_c5

def _block_swap_pass(h, deadline, block_sizes=(1, 2, 3), n_passes=5, max_pairs=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        binary = (h > 0.5).astype(np.int8)
        for bs in block_sizes:
            if time.time() > deadline:
                break
            if bs > n // 2:
                continue
            ones_starts = []
            zeros_starts = []
            for i in range(n - bs + 1):
                if binary[i:i+bs].min() == 1:
                    ones_starts.append(i)
                if binary[i:i+bs].max() == 0:
                    zeros_starts.append(i)
            pairs = []
            for ib in ones_starts:
                for zb in zeros_starts:
                    if zb < ib + bs and ib < zb + bs:
                        continue
                    pairs.append((ib, zb))
            if len(pairs) > max_pairs:
                idx_sel = rng.choice(len(pairs), max_pairs, replace=False)
                pairs = [pairs[i] for i in idx_sel]
            for ib, zb in pairs:
                if time.time() > deadline:
                    break
                d = np.zeros(n)
                d[ib:ib+bs] = -1.0
                d[zb:zb+bs] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _all_pair_swap(h, deadline, max_passes=6):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _subgradient_pass(h, deadline, n_steps=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step_size = 0.03
    stall = 0
    for _step in range(n_steps):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(scores)
        active = np.where(scores >= max_val - 1e-10)[0]
        G = _grad_matrix(h)
        grad = G[active, :].mean(axis=0)
        grad -= grad.mean()
        new_h = _project(h - step_size * grad, target)
        new_c5 = _exact_c5(new_h)
        if new_c5 < cur_c5 - 1e-14:
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step_size = min(step_size * 1.03, 0.12)
        else:
            stall += 1
            step_size *= 0.82
            if stall > 60:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP + block swap on initial (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_loop(h_init, min(d1, time.time() + budget_s * 0.06), 0.18)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1
    h1b, c5_1b = _block_swap_pass(_project(best_h, len(best_h)/2), d1, block_sizes=(1,2,3), n_passes=3, max_pairs=150)
    if c5_1b < best_c5:
        best_c5, best_h = c5_1b, h1b
    h1c, c5_1c = _lp_loop(_project(best_h, len(best_h)/2), d1, 0.10)
    if c5_1c < best_c5:
        best_c5, best_h = c5_1c, h1c

    # Phase 2: Independent random starts at multiple n (18%)
    d2 = t0 + budget_s * 0.30
    n_list = sorted(set([n0, n0-8, n0+8, n0-16, n0+16, n0-32, n0+32, 100, 120, 140, 160, 180, 200, 220]))
    n_list = [x for x in n_list if 60 <= x <= 320]
    per_n = (d2 - time.time()) / len(n_list)
    for n_try in n_list:
        if time.time() > d2 - 5:
            break
        sub_dl = min(d2 - 3, time.time() + max(4, per_n))
        h_r = _random_binary(n_try, rng)
        h_r, c5_r = _lp_loop(h_r, min(sub_dl, time.time() + per_n * 0.5), 0.20)
        if time.time() < sub_dl - 2:
            h_r, c5_r = _block_swap_pass(h_r, min(sub_dl - 1, time.time() + per_n * 0.3), block_sizes=(1,2,3), n_passes=2, max_pairs=120)
        if time.time() < sub_dl:
            h_r, c5_r = _lp_loop(_project(h_r, n_try/2), sub_dl, 0.08)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()

    # Phase 3: Intensive block swap + LP on best (18%)
    d3 = t0 + budget_s * 0.48
    h3 = _project(best_h, len(best_h) / 2)
    h3, c5_3 = _block_swap_pass(h3, min(d3, time.time() + budget_s * 0.07), block_sizes=(1,2,3,4), n_passes=8, max_pairs=250)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3
    h3b, c5_3b = _lp_loop(_project(best_h, len(best_h)/2), min(d3, time.time() + budget_s * 0.05), 0.08)
    if c5_3b < best_c5:
        best_c5, best_h = c5_3b, h3b
    h3c, c5_3c = _subgradient_pass(_project(best_h, len(best_h)/2), min(d3, time.time() + budget_s * 0.03), n_steps=200)
    if c5_3c < best_c5:
        best_c5, best_h = c5_3c, h3c
    h3d, c5_3d = _lp_loop(_project(best_h, len(best_h)/2), d3, 0.06)
    if c5_3d < best_c5:
        best_c5, best_h = c5_3d, h3d

    # Phase 4: Multi-n exploration (20%)
    d4 = t0 + budget_s * 0.68
    n_base = len(best_h)
    offsets = [-20, -12, -8, -4, 4, 8, 12, 20, -32, 32, -48, 48, -64, 64, 96, -96, 128, -128, 80, 100]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d4 - 8:
            break
        n_try = n_base + dn
        if n_try < 50 or n_try > 350 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        per_n = min(budget_s * 0.022, (d4 - 5 - time.time()) / 3)
        sub_dl = min(d4 - 5, time.time() + max(3, per_n))
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _lp_loop(h_r, min(sub_dl, time.time() + per_n * 0.5), 0.15)
        rem = d4 - 5 - time.time()
        if rem > 2 and c5_r < best_c5 + 0.004:
            h_r, c5_r = _block_swap_pass(h_r, min(d4 - 5, time.time() + max(2, rem * 0.4)), block_sizes=(1,2,3), n_passes=3, max_pairs=150)
            rem2 = d4 - 5 - time.time()
            if rem2 > 1:
                h_r, c5_r = _lp_loop(_project(h_r, n_try/2), d4 - 5, 0.07)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()
            n_base = n_try

    # Phase 5: Perturbation + re-optimize (15%)
    d5 = t0 + budget_s * 0.83
    scale = 0.05
    for _attempt in range(400):
        if time.time() > d5 - 3:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d5 - 3, time.time() + budget_s * 0.007)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, 0.08)
        if time.time() < d5 - 6 and c5_t < best_c5 + 0.002:
            sub_dl2 = min(d5 - 6, time.time() + budget_s * 0.003)
            h_t, c5_t = _block_swap_pass(h_t, sub_dl2, block_sizes=(2,3), n_passes=2, max_pairs=100)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t.copy()
            scale = max(0.008, scale * 0.84)
        else:
            scale = min(0.14, scale * 1.07)

    # Phase 6: Final polish (17%)
    d6 = deadline
    h6 = _project(best_h, len(best_h) / 2)
    h6, c5_6 = _lp_loop(h6, min(deadline - 12, time.time() + budget_s * 0.06), 0.05)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6
    if time.time() < deadline - 15:
        h6 = _project(best_h, len(best_h) / 2)
        h6, c5_6 = _all_pair_swap(h6, min(deadline - 12, time.time() + budget_s * 0.05), max_passes=4)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6
    if time.time() < deadline - 10:
        h6 = _project(best_h, len(best_h) / 2)
        h6, c5_6 = _block_swap_pass(h6, min(deadline - 10, time.time() + budget_s * 0.03), block_sizes=(1,2,3), n_passes=3, max_pairs=200)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6
    if time.time() < deadline - 3:
        h6 = _project(best_h, len(best_h) / 2)
        h6, c5_6 = _lp_loop(h6, deadline - 1, 0.03)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
