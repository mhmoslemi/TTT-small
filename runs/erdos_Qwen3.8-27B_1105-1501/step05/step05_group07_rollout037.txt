def _two_coord_optimize(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        C = _coeff_matrix(h)  # shape (n, 2n-1)
        S = _scores(h)  # shape (2n-1,)
        improved = False
        
        # Try all pairs (i, j) with i < j
        # For efficiency, we can limit to pairs that are likely to help
        # But for correctness, let's try all pairs with a time limit
        n_tried = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                n_tried += 1
                if n_tried % 500 == 0 and time.time() > deadline:
                    break
                
                δ_lo = max(-h[i], h[j] - 1.0)
                δ_hi = min(1.0 - h[i], h[j])
                if δ_hi - δ_lo < 1e-12:
                    continue
                
                # Compute L_d = C[i,d] - C[j,d]
                L = C[i] - C[j]
                
                # Compute Q_d
                # Q_d = 1 if d = i-j, -1 if d = j-i, else 0
                # In terms of the index into the scores array (which is n-1+d):
                d_ij = i - j  # this is negative since i < j
                d_ji = j - i  # positive
                # d_ij is valid if |d_ij| <= n-1, which it always is
                Q = np.zeros(2*n-1)
                Q[n-1 + d_ji] = -1.0  # for d = j-i
                Q[n-1 + d_ij] = 1.0   # for d = i-j
                
                # Objective: f(δ) = max_d [S[d] + δ*L[d] + δ²*Q[d]]
                # Ternary search on [δ_lo, δ_hi]
                lo = δ_lo
                hi = δ_hi
                best_f = np.inf
                best_delta = 0.0
                
                # Golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(S + c1*L + c1*c1*Q)
                f2 = np.max(S + c2*L + c2*c2*Q)
                best_f = min(f1, f2)
                best_delta = c1 if f1 < f2 else c2
                
                for _ in range(60):
                    if hi - lo < 1e-14:
                        break
                    if f1 < f2:
                        hi = c2
                        f2 = f1
                        c2 = c1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(S + c1*L + c1*c1*Q)
                    else:
                        lo = c1
                        f1 = f2
                        c1 = c2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(S + c2*L + c2*c2*Q)
                    
                    if f1 < best_f:
                        best_f = f1
                        best_delta = c1
                    if f2 < best_f:
                        best_f = f2
                        best_delta = c2
                
                # Check if improvement
                if best_f < S.max() - 1e-13:
                    h[i] += best_delta
                    h[j] -= best_delta
                    h = np.clip(h, 0.0, 1.0)
                    # Re-project to fix any drift
                    h = _project(h, n/2.0)
                    new_c5 = _exact_c5(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improved = True
                    # Recompute C and S after move
                    C = _coeff_matrix(h)
                    S = _scores(h)
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    """C[m, idx] = dS_d/dh[m] where idx = n-1+d."""
    n = len(h)
    C = np.zeros((n, 2*n-1))
    oml = 1.0 - h
    for d in range(-(n-1), n):
        idx = n - 1 + d
        # (1 - h[m+d]) for m+d in [0,n): m in [max(0,-d), min(n, n-d))
        ms = max(0, -d)
        me = min(n, n - d)

        # -h[m-d] for m-d in [0,n): m in [max(0,d), min(n, n+d))
        ms2 = max(0, d)
        me2 = min(n, n + d)
        if me2 > ms2:
            C[ms2:me2, idx] -= h[ms2-d:me2-d]
    return C

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coeff_matrix(h):
    n = len(h)
    C = np.zeros((n, 2*n-1))
    oml = 1.0 - h
    for d in range(-(n-1), n):

        ms = max(0, -d)

        me2 = min(n, n + d)
        if me2 > ms2:

    return C

    """Minimize max(S + d*L + d^2*Q) over d in [d_lo, d_hi] using golden section."""

    lo, hi = d_lo, d_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * L + c1 * c1 * Q))
    f2 = float(np.max(S + c2 * L + c2 * c2 * Q))
    for _ in range(n_iter):
        if hi - lo < 1e-13 * max(1.0, abs(lo), abs(hi)):
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * L + c1 * c1 * Q))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * L + c2 * c2 * Q))
    best_d = (lo + hi) / 2.0
    best_f = float(np.max(S + best_d * L + best_d * best_d * Q))

    f1e = float(np.max(S + d_hi * L + d_hi * d_hi * Q))

Wait, I'm confusing myself. Let me think again.

f(δ) = max(g(δ), h₁(δ), h₂(δ), ...) where g is concave (quadratic with negative leading coefficient) and h_k are linear or convex quadratic.

The minimum of f is where all the "active" constraints are as small as possible. Since g is concave (inverted U), it's large at the endpoints and small in the middle (relative to its peak). So g will tend to be the max near the endpoints of [d_lo, d_hi], and the other terms will be the max in the middle.

The minimum of f will typically be in the "middle" region where the linear/convex terms dominate, making f effectively convex there.

For the golden section to work, f needs to be unimodal (single minimum). Is that guaranteed?

Consider: f(δ) = max(-δ² + 4, 0). This has min = 0 at δ = ±2, and the function is 0 for |δ| ≥ 2 and positive for |δ| < 2. This is NOT unimodal (it has a flat region at the minimum).

But for our problem, the feasible interval [d_lo, d_hi] is determined by the box constraints, and typically the parabola g(δ) won't be the max at both endpoints (since the other linear terms will bind).

In practice, I think the function will be "approximately" unimodal, and the golden section will work well. But to be safe, let me use a different approach:

**Approach: Dense grid + refinement**

1. Evaluate f(δ) on a grid of, say, 20 points in [d_lo, d_hi].
2. Find the grid point with minimum f.
3. Do golden section (or bisection on the derivative) in the bracket around that grid point.

This ensures we don't miss the global minimum due to non-convexity.

Actually, even simpler: since the function is the max of a few pieces (each either linear, convex quadratic, or concave quadratic), I can find the minimum analytically:

The minimum of f(δ) occurs at:
- An endpoint, OR
- A point where two (or more) of the pieces are equal and the "upper envelope" has a cusp pointing downward.

For the linear pieces: the max of linear functions has its minimum at a "kink" where two lines cross, or at an endpoint.
For the concave quadratic piece: it can only be part of the upper envelope in a bounded region.

This analytical approach is complex to implement. Let me just use the grid + refinement approach:

```python
def _minimize_1d_convex_nonconvex(S, L, Q, d_lo, d_hi):
    """Find min of max(S + d*L + d^2*Q) over d in [d_lo, d_hi].
    Handles the fact that Q has one negative entry (making one term concave)."""
    # Dense grid to find approximate location of minimum
    n_grid = 30
    ds = np.linspace(d_lo, d_hi, n_grid)
    fs = np.array([float(np.max(S + d * L + d * d * Q)) for d in ds])
    # Find bracket
    min_idx = int(np.argmin(fs))
    if min_idx == 0:
        lo, hi = d_lo, ds[1]
    elif min_idx == n_grid - 1:
        lo, hi = ds[-2], d_hi
    else:
        lo, hi = ds[min_idx - 1], ds[min_idx + 1]
    
    # Golden section in [lo, hi] (assuming unimodality in this small interval)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_d = ds[min_idx]
    best_f = fs[min_idx]
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * L + c1 * c1 * Q))
    f2 = float(np.max(S + c2 * L + c2 * c2 * Q))
    if f1 < best_f:
        best_f = f1; best_d = c1
    if f2 < best_f:
        best_f = f2; best_d = c2
    
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * L + c1 * c1 * Q))
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * L + c2 * c2 * Q))
        if f1 < best_f:
            best_f = f1; best_d = c1
        if f2 < best_f:
            best_f = f2; best_d = c2
    
    return best_d, best_f
```

This is safer but slightly slower (30 grid evaluations + 40 golden section evaluations = 70 evaluations per pair vs. 50 for pure golden section). Still fast enough.

Actually, I realize I can be smarter. The non-convexity comes from the single concave term (at lag d = j-i). This term is S_{j-i} + δ*L_{j-i} - δ². It's a downward parabola with peak at δ = L_{j-i}/2.

The key insight: this concave term can only be the "active" (maximum) constraint in a bounded region around its peak. Outside that region, the linear/convex terms dominate, and f is convex.

So the minimum of f is either:
1. In the "convex region" (where the concave term is not the max): found by golden section
2. At the boundary of the "convex region" (where the concave term just stops being the max)

Case 2 corresponds to a kink where the concave parabola equals another term. This is harder to find analytically.

For a practical approach, the grid + refinement should work well. Let me use 15 grid points (enough to bracket the minimum) + 40 golden section iterations.
