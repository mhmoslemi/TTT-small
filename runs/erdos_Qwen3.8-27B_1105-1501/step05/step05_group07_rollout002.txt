def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix numerical drift
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute gradient matrix G where G[r, j] = d(score_r)/d(h_j)"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _logsumexp(scores, beta):
    """Compute log-sum-exp of beta*scores / beta, i.e., smooth max"""
    maxs = np.max(scores)
    shifted = beta * (scores - maxs)
    shifted = np.clip(shifted, -500, 0)
    return maxs + np.log(np.mean(np.exp(shifted))) / beta

def _logsumexp_weights(scores, beta):
    """Compute softmax weights for log-sum-exp"""
    maxs = np.max(scores)
    shifted = beta * (scores - maxs)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= s
    return w

def _smoothed_obj(h, beta):
    scores = _scores(h)
    return _logsumexp(scores, beta)

def _smoothed_grad(h, beta):
    scores = _scores(h)
    w = _logsumexp_weights(scores, beta)
    G = _grad_matrix(h)
    return G.T @ w

def _grad_descent_smoothed(h, deadline, beta_start=5.0, beta_end=500.0, 
                            n_phases=8, steps_per_phase=500, lr_init=0.05):
    """Adaptive gradient descent on smoothed objective with projection"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_init * (0.8 ** phase)
        momentum = h.copy() * 0.0  # Nesterov momentum
        
        for step in range(steps_per_phase):
            if time.time() > deadline:
                break
            grad = _smoothed_grad(h, beta)
            # Center gradient (sum constraint)
            grad -= grad.mean()
            
            # Nesterov-style update
            momentum = 0.9 * momentum + grad
            h_new = _project(h - lr * momentum, target)
            
            # Check improvement
            c5_new = _exact_c5(h_new)
            if c5_new < _exact_c5(h) - 1e-15:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                # Backtrack
                lr *= 0.7
                h_new = _project(h - lr * momentum, target)
                if _exact_c5(h_new) < _exact_c5(h):
                    h = h_new
    
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_descent(h, scores, G, radius):
    """Find LP descent direction"""
    n = len(h)
    n_lags = len(scores)
    # Minimize t subject to scores + G@p - t <= 0, sum(p)=0, bounds on p
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def _line_search_c5(h, p):
    """Line search minimizing exact C5 along direction p"""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-15) * 0.999
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    for _ in range(40):
        if hi - lo < 1e-14:

        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)

        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    opt = (lo + hi) / 2.0
    return opt, _exact_c5(_project(h + opt * p, n / 2.0))

def _lp_optimize(h, deadline, radius_init=0.3, max_iters=10000, stall_limit=200):
