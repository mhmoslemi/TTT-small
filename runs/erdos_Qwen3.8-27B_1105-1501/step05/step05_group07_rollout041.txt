def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_A(h):
    """Compute A matrix where A[i, d_idx] = (1-h[i+d] if valid) - (h[i-d] if valid)
    d ranges from -(n-1) to (n-1), d_idx = d + (n-1)"""
    n = len(h)
    A = np.zeros((n, 2*n-1))
    for d_idx in range(2*n-1):
        d = d_idx - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                A[:m, d_idx] = 1.0 - h[d:]
            if d > 0:
                A[d:, d_idx] -= h[:m]
        else:
            d_neg = -d
            m = n - d_neg
            if m > 0:
                A[d_neg:, d_idx] = 1.0 - h[:m]
            A[:m, d_idx] -= h[d_neg:]
    return A

    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Precompute pair indices
    ii_all, jj_all = np.where(np.triu(np.ones((n,n), dtype=bool), 1))
    n_pairs = len(ii_all)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        S = _scores(h)
        cur_c5 = float(S.max())
        
        if cur_c5 <= best_c5 - 1e-13:

            best_h = h.copy()
        
        A = _compute_A(h)
        G = A[ii_all] - A[jj_all]  # (n_pairs, 2n-1)
        
        t_min = np.maximum(-h[ii_all], h[jj_all] - 1.0)
        t_max = np.minimum(1.0 - h[ii_all], h[jj_all])
        
        feasible = t_max > t_min + 1e-10
        if not np.any(feasible):
            break
        
        Gf = G[feasible]
        tmin_f = t_min[feasible]
        tmax_f = t_max[feasible]
        n_f = len(Gf)
        
        # Bisection
        lo = tmin_f.copy()
        hi = tmax_f.copy()
        
        for _ in range(50):
            mid = (lo + hi) / 2.0
            V = S[None, :] + mid[:, None] * Gf
            Vmax = V.max(axis=1)
            active = V >= Vmax[:, None] - 1e-10
            rg = np.where(active, Gf, -np.inf).max(axis=1)
            lg = np.where(active, Gf, np.inf).min(axis=1)
            
            go_r = rg < -1e-12
            go_l = lg > 1e-12
            if not np.any(go_r) and not np.any(go_l):
                break
            lo[go_r] = mid[go_r]
            hi[go_l] = mid[go_l]
            
            if (hi - lo).max() < 1e-13:
                break
        
        t_star = (lo + hi) / 2.0
        V_star = S[None, :] + t_star[:, None] * Gf
        F_star = V_star.max(axis=1)
        
        best_local = int(F_star.argmin())
        best_F = F_star[best_local]
        
        if best_F >= cur_c5 - 1e-12:

        # Make the move

        i = ii_all[local_idx]

        elif new_c5 > cur_c5 + 1e-10:

    if target >= n:

    lo = float(v.min()) - 2.0

    hi_a = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(18):
        if hi_a - lo_a < 1e-13:
            break
        if f1 < f2:
            hi_a = c2
            f2 = f1
            c2 = c1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            f1 = f2
            c1 = c2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _local_optimize_lp(h, deadline, radius_init=0.2, max_iters=3000, stall_limit=120):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.82
            if stall > 40 and radius < 0.02:
                radius = 0.1
    return best_h, best_c5

def _slsqp_optimize(h, deadline, beta_init=40.0, beta_final=200.0, n_phases=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac

        def obj(hf):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = np.clip(beta * (S - maxS), -500, 0)
            return maxS + np.log(np.mean(np.exp(shifted))) / beta

        def grad(hf):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = np.clip(beta * (S - maxS), -500, 0)
            w = np.exp(shifted)
            wsum = np.sum(w)
            if wsum < 1e-300:
                w = np.zeros(len(S))
                w[int(np.argmax(S))] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w

        def con_fun(x):
            return np.sum(x) - target

        def con_jac(x):
            return np.ones(n)

        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': 100, 'ftol': 1e-14})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Two-coordinate main engine (50% budget)
    dl1 = t0 + budget_s * 0.50
    h1, c5_1 = _two_coord_optimize(h_start, dl1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP polish + two-coord
    if time.time() < t0 + budget_s * 0.65:
        dl2 = t0 + budget_s * 0.60
        h2, c5_2 = _local_optimize_lp(best_h, dl2, radius_init=0.15, max_iters=2000, stall_limit=100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        if time.time() < t0 + budget_s * 0.65:
            h2b, c5_2b = _two_coord_optimize(best_h, t0 + budget_s * 0.65)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: SLSQP global exploration
    if time.time() < t0 + budget_s * 0.78:
        dl3 = t0 + budget_s * 0.78
        h3, c5_3 = _slsqp_optimize(best_h, dl3, beta_init=40.0, beta_final=200.0, n_phases=3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        if time.time() < dl3 + budget_s * 0.03:
            h3b, c5_3b = _two_coord_optimize(best_h, min(dl3 + budget_s * 0.03, deadline))
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()

    # Phase 4: Multi-resolution
    if time.time() < t0 + budget_s * 0.88:
        dl4 = t0 + budget_s * 0.88
        for n_try in [96, 160, 192]:
            if time.time() > dl4:
                break
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl4, time.time() + budget_s * 0.04)
            ht, ct = _two_coord_optimize(h_init, sub_dl, max_iters=1500)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try
            if time.time() < dl4:
                h_qp = _quasi_periodic_init(n_try, 0.6180339)
                h_qp += rng.normal(0, 0.04, n_try)
                h_qp = _project(h_qp, n_try / 2.0)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.03)
                ht2, ct2 = _two_coord_optimize(h_qp, sub_dl2, max_iters=1000)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = n_try

    # Phase 5: Perturbation restarts
    if time.time() < t0 + budget_s * 0.94:
        dl5 = t0 + budget_s * 0.94
        scale = 0.015
        for attempt in range(200):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.015)
            ht, ct = _two_coord_optimize(h_pert, sub_dl, max_iters=800)
            if ct < best_c5 - 1e-13:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.06, scale * 1.15)

    # Phase 6: Final polish
    h6, c5_6 = _two_coord_optimize(best_h, deadline - budget_s * 0.005, max_iters=2000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
