def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        else:

        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_jac_matrix(h):
    """Compute (2n-1, n) Jacobian of all correlation values w.r.t. h."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    # Lag 0: S_0 = sum h_i(1-h_i), grad = 1-2h
    G[n-1, :] = 1.0 - 2.0*h
    # For d = 1, ..., n-1:
    # S_{-d} at output index n-1+d
    # S_d at output index n-1-d
    idx = np.arange(n)
    for d in range(1, n):
        m = n - d
        # S_{-d} = sum_{i=0}^{m-1} h_{i+d}(1-h_i)
        # d/dh_j: (1-h_{j-d}) for j in [d, n-1], -h_{j+d} for j in [0, m-1]
        row = np.zeros(n)
        row[d:] += (1.0 - h[:m])
        row[:m] += -h[d:]
        G[n-1+d, :] = row
        # S_d = sum_{i=0}^{m-1} h_i(1-h_{i+d})
        # d/dh_j: (1-h_{j+d}) for j in [0, m-1], -h_{j-d} for j in [d, n-1]
        row = np.zeros(n)
        row[:m] += (1.0 - h[d:])
        row[d:] += -h[:m]
        G[n-1-d, :] = row
    return G

        j[:n] = 1.0

            stall += 1
            radius *= 0.65
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search(h, p, target)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0 < 0.5).astype(float)
    return _project(h, n / 2.0)

def _binary_sa(n, deadline, rng, n_iters=100000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.04
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    t_start = time.time()
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cool
    return best_h, best_c5

def _smoothed_grad_momentum(h, deadline, beta_start=20, beta_end=300, n_steps=300, step_init=0.05):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        G = _c5_jac_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.02, 0.12)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    return best_h, best_c5

def _coordinate_polish(h, deadline, n_cycles=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v = 0.0
            hi_v = 1.0
            for _ in range(35):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _perturb_lp(h_best, best_c5, dl, rng, n_restarts, time_per):
    n = len(h_best)
    best_h = h_best.copy()
    best_c5_local = best_c5
    scale = 0.03
    for _ in range(n_restarts):
        if time.time() > dl:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        sub_dl = min(dl, time.time() + time_per)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.08, max_iters=500, radius_min=0.004)
        if c5_try < best_c5_local - 1e-12:
            best_c5_local = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.80)
        else:
            scale = min(0.10, scale * 1.08)
    return best_h, best_c5_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    def _update(h_try, c5_try, n_try):
        nonlocal best_h, best_c5, best_n
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 1: SLSQP from initial (n=128)
    if time.time() < t0 + budget_s * 0.12:
        dl1 = t0 + budget_s * 0.12
        h1, c5_1 = _slsqp_optimize(h_start, dl1, K_init=15, max_slips=8)
        _update(h1, c5_1, n0)

    # Phase 2: LP from best
    if time.time() < t0 + budget_s * 0.22:
        dl2 = t0 + budget_s * 0.22
        h2, c5_2 = _lp_optimize(best_h, dl2, radius_init=0.15, max_iters=3000, radius_min=0.002)
        _update(h2, c5_2, best_n)

    # Phase 3: SLSQP again (warmed up)
    if time.time() < t0 + budget_s * 0.32:
        dl3 = t0 + budget_s * 0.32
        h3, c5_3 = _slsqp_optimize(best_h, dl3, K_init=20, max_slips=6)
        _update(h3, c5_3, best_n)

    # Phase 4: Multi-resolution exploration with SLSQP + LP
    if time.time() < t0 + budget_s * 0.52:
        dl4 = t0 + budget_s * 0.52
        resolutions = [96, 64, 160, 192, 256, 128, 80, 112]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.6180339887, 0.3819660113, 0.7071067812]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl4:
                break
            if best_c5 < 0.3806:
                break
            for s in range(2):
                if time.time() > dl4:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    alpha = alphas[ri % len(alphas)]
                    h_init = _quasi_periodic_init(n_try, alpha)
                sub_dl_slsqp = min(dl4, time.time() + budget_s * 0.025)
                h_s, c5_s = _slsqp_optimize(h_init, sub_dl_slsqp, K_init=12, max_slips=5)
                sub_dl_lp = min(dl4, time.time() + budget_s * 0.02)
                h_l, c5_l = _lp_optimize(h_s, sub_dl_lp, radius_init=0.12, max_iters=2000, radius_min=0.003)
                _update(h_l, c5_l, n_try)

    # Phase 5: Smoothed gradient momentum
    if time.time() < t0 + budget_s * 0.62:
        dl5 = t0 + budget_s * 0.62
        h5, c5_5 = _smoothed_grad_momentum(best_h, dl5, beta_start=15, beta_end=500, n_steps=300, step_init=0.04)
        _update(h5, c5_5, best_n)
        if time.time() < t0 + budget_s * 0.67:
            dl5b = t0 + budget_s * 0.67
            h5b, c5_5b = _slsqp_optimize(best_h, dl5b, K_init=20, max_slips=5)
            _update(h5b, c5_5b, best_n)

    # Phase 6: Binary SA for diversity
    if time.time() < t0 + budget_s * 0.78:
        for n_bin in [64, 96, 128]:
            if time.time() > t0 + budget_s * 0.78:
                break
            sub_dl = min(t0 + budget_s * 0.78, time.time() + budget_s * 0.03)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=50000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
            sub_dl2 = min(t0 + budget_s * 0.78, time.time() + budget_s * 0.02)
            h_br, c5_br = _slsqp_optimize(h_b, sub_dl2, K_init=15, max_slips=5)
            _update(h_br, c5_br, n_bin)

    # Phase 7: Perturbation restarts
    dl7 = t0 + budget_s * 0.90
    h_p, c5_p = _perturb_lp(best_h, best_c5, dl7, rng, 200, budget_s * 0.01)
    _update(h_p, c5_p, best_n)

    # Phase 8: Coordinate polish
    if time.time() < t0 + budget_s * 0.96:
        dl8 = t0 + budget_s * 0.96
        hc, c5c = _coordinate_polish(best_h, dl8, n_cycles=2)
        _update(hc, c5c, best_n)

    # Phase 9: Final SLSQP + LP
    if time.time() < t0 + budget_s * 0.995:
        dl9 = t0 + budget_s * 0.998
        h_fin, c5_fin = _slsqp_optimize(best_h, dl9, K_init=25, max_slips=4)
        _update(h_fin, c5_fin, best_n)
        if time.time() < t0 + budget_s * 0.999:
            h_fin2, c5_fin2 = _lp_optimize(best_h, t0 + budget_s * 0.999, radius_init=0.03, max_iters=1000, radius_min=0.001)
            _update(h_fin2, c5_fin2, best_n)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
