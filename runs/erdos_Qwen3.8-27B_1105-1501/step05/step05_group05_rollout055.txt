        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]

def _lp_solve(h, scores, G, radius):

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))

        hi = min(radius, 1.0 - h[j])

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):

        if f1 < f2:

    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.20):

    target = n / 2.0

            break

        G = _grad_matrix(h)
        p = _lp_solve(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.08, 0.55)
        else:
            stall += 1
            radius *= 0.72
            if stall > 25 and radius < 0.04:
                radius = best_radius * 0.4
                stall = 0
    return best_h, best_c5

def _pair_refine_exhaustive(h, deadline, max_passes=500):

def _snap_and_reopt(h, deadline, rng, threshold=0.03):

def _resample(h, n_new):
    n = len(h)
    if n_new == n or n_new < 4:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _logit_mutate(h, rng, scale=0.15):
    n = len(h)
    eps = 1e-6
    h_clip = np.clip(h, eps, 1.0 - eps)
    v = np.log(h_clip / (1.0 - h_clip))
    v += rng.normal(0, scale, n)
    v = np.clip(v, -20, 20)
    h_new = 1.0 / (1.0 + np.exp(-v))
    return _project(h_new, n / 2.0)

def _smooth_lse(h, deadline, beta_list=(40, 100, 250, 500), lr_init=0.02):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.80
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 50:
            bi += 1
            if bi >= len(beta_list) * 3:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = beta_list[bi % len(beta_list)]
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        s_max = float(np.max(scores))
        v = beta * (scores - s_max)
        exp_v = np.exp(v)
        w = exp_v / float(np.sum(exp_v))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:d + m])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (-h[g:g + m])
                    grad[g:] += w[r] * (1.0 - h[:m])
        grad -= np.mean(grad)
        dn = np.linalg.norm(grad)
        if dn < 1e-13:
            bi += 1
            continue
        d = grad / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.06, 0.35)
        else:
            stall += 1
            lr *= 0.78
