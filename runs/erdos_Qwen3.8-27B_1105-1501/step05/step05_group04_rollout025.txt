    """Compute correlation values and gradient wrt a (complementary pair vars).
    Returns: V (2n-1,), G (m, 2n-1) where G[i,s] = dV[s]/da[i]
    """
    m = n // 2
    V = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute full gradient H[j, s] = dV[s]/dh[j]
    # Then G[i, :] = H[i, :] - H[n-1-i, :]
    H = np.zeros((n, 2*n-1))
    
    # d=0: H[j, n-1] = 1 - 2*h[j]
    H[:, n-1] = 1.0 - 2.0 * h

    for d in range(1, n):

        H[:n-d, n-1+d] += (1.0 - h[d:n])
        # Second term: for j = d..n-1, contribution -h[j-d] to H[j, n-1-d]
        H[d:, n-1-d] += (-h[:n-d])
    
    # Combine for complementary pairs
    G = H[:m, :] - H[n-1:n-m-1:-1, :]
    
    return V, G
