def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))

    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, active_idx=None):
    n = len(h)
    if active_idx is None:
        active_idx = list(range(2 * n - 1))
    n_act = len(active_idx)
    if n_act == 0:
        return None
    idx = np.array(active_idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G[idx, :]
    A_ub[:, n] = -1.0
    b_ub = -scores[idx].copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lsumexp_grad(h, dx, beta, top_frac=0.3):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    n_act = max(1, int(len(w) * top_frac))
    top_idx = np.argsort(w)[::-1][:n_act]
    w_top = w[top_idx]
    w_top = w_top / float(np.sum(w_top))
    G = _grad_matrix(h)
    grad = G[top_idx, :].T @ w_top
    grad -= np.mean(grad)
    return grad

    h3, c5_3 = _pair_refine(best_h, d3, max_passes=300, use_random=False, rng=rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution search (15%)
    d4 = t0 + budget_s * 0.80
    best_n = len(best_h)
    tried = set()
    offsets = list(range(-15, 16, 1))
    offsets.sort(key=lambda x: abs(x))
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 80 or n_try > 180:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        if c5_r < best_c5 + 0.0003 and time.time() < d4 - 5:
            sub_dl2 = min(d4, time.time() + 5.0)
            h_r, c5_r = _pair_refine(h_r, sub_dl2, max_passes=100, use_random=False, rng=rng)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Random pair refinement + kick (10%)
    d5 = t0 + budget_s * 0.90
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=200, use_random=True, rng=rng, n_random=800)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < deadline:
        h5b, c5_5b = _kick_reopt(best_h, deadline, rng, n_kicks=10)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Final polish (5%)
    d6 = deadline
    if time.time() < d6:
        h6, c5_6 = _pair_refine(best_h, d6, max_passes=200, use_random=False, rng=rng)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if time.time() < d6 - 1:
        h7, c5_7 = _lp_loop(best_h, deadline, radius_init=0.04)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
