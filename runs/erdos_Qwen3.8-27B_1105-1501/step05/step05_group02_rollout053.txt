                    if i == j:
                        continue
            
            h_new, c5_new = _try_pair(h, int(i), int(j), scores, dx, cur_c5, nc)
            if h_new is not None:
                h = h_new
                cur_c5 = c5_new
                scores = _scores(h)
                n_accepted += 1
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        
        if n_accepted == 0:
            stagnant += 1
        else:
            stagnant = 0
    
    return best_h, best_c5

def _binary_init(n, rng):
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def _subgrad_optimize(h, deadline, gamma_init=5, gamma_max=200, max_it=500):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    nc = n - 1
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(gamma * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        # Compute weighted gradient
        grad = np.zeros(n)
        for d in range(1, n):
            wp = w[nc + d]
            wn = w[nc - d]
            m = n - d
            if wp > 1e-15:
                grad[:m] += wp * dx * (1.0 - h[d:])
                grad[d:] -= wp * dx * h[:m]
            if wn > 1e-15:
                grad[d:] += wn * dx * (1.0 - h[:m])
                grad[:m] -= wn * dx * h[d:]
        # Center contribution
        grad += w[nc] * dx * (1.0 - 2.0 * h)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        # Line search
        best_a = 0.0
        best_c5_local = cur_c5
        alpha = 0.1
        for _ in range(20):
            h_cand = _project(h - alpha * grad, target)
            c5_cand = _exact_c5(h_cand)
            if c5_cand < best_c5_local:
                best_c5_local = c5_cand
                best_a = alpha
            alpha *= 0.6
            if alpha < 1e-12:
                break
        if best_a > 0:
            h = _project(h - best_a * grad, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.02, gamma_max)
        else:
            stall += 1
            gamma = min(gamma * 1.05, gamma_max)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Quick LP (15% budget)
    dl1 = t0 + budget_s * 0.15
    h_lp, c5_lp = _lp_optimize(h, dl1, radius_init=0.25, max_it=2000)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()
    
    # Phase 2: Main pair-transfer optimization (55% budget)
    dl2 = t0 + budget_s * 0.70
    h_pt, c5_pt = _pair_transfer_optimize(best_h, dl2, n_pairs_per_round=300, max_rounds=50000)
    if c5_pt < best_c5:
        best_c5 = c5_pt
        best_h = h_pt.copy()
    
    # Phase 3: Subgradient refinement (10% budget)
    if time.time() < t0 + budget_s * 0.82:
        dl3 = t0 + budget_s * 0.82
        h_sg, c5_sg = _subgrad_optimize(best_h, dl3, gamma_init=10, gamma_max=300, max_it=500)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        # LP after subgrad
        dl3b = min(dl3 + budget_s * 0.03, t0 + budget_s * 0.85)
        h_lp2, c5_lp2 = _lp_optimize(best_h, dl3b, radius_init=0.1, max_it=1000)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()
    
    # Phase 4: More pair-transfer rounds (10% budget)
    if time.time() < t0 + budget_s * 0.93:
        dl4 = t0 + budget_s * 0.93
        h_pt2, c5_pt2 = _pair_transfer_optimize(best_h, dl4, n_pairs_per_round=500, max_rounds=50000)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2.copy()
    
    # Phase 5: Try different resolutions (rest)
    if time.time() < deadline - 30:
        for n_try in [96, 160, 224]:
            if time.time() > deadline - 30:
                break
            h_res = _resample(best_h, n_try)
            sub_dl = min(deadline - 10, time.time() + budget_s * 0.03)
            h_r, c5_r = _pair_transfer_optimize(h_res, sub_dl, n_pairs_per_round=100, max_rounds=5000)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    
    # Final
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Projection made it worse; use pre-projection
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
