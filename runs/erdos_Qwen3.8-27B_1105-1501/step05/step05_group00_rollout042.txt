If we decrease h[i+r] (which is in the "h" part) or increase h[i] (which is in the "1-h" part), we reduce corr[r].

So for active lag r:
- Good moves: decrease h[j] for j in [r, n-1] and increase h[j-r] (shift left by r)
- Or: increase h[j] for j in [0, n-1-r] and decrease h[j+r] (shift right by r)

For the pair move (i, j) with h[i]-=δ, h[j]+=δ:
- This reduces corr[r] if: i is in the "h" position and j is in the "1-h" position for lag r.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    for _ in range(5):
        res = target - float(np.sum(x))
        if abs(res) < 1e-12:
            break
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = -radius, radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    ng = 48
    grid = np.linspace(t_lo, t_hi, ng)
    vals = S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:]
    obj = np.max(vals, axis=1) * dx
    bi = int(np.argmin(obj))
    bt, bv = grid[bi], obj[bi]
    if 0 < bi < ng-1:
        a, b = grid[bi-1], grid[bi+1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0)-1.0)/2.0
    c1, c2 = b-gr*(b-a), a+gr*(b-a)
    f1 = float(np.max(S0+c1*T-c1*c1*C))*dx
    f2 = float(np.max(S0+c2*T-c2*c2*C))*dx
    for _ in range(50):

    if ov < bv:

def _gradient(h, lags):
    """Gradient of sum_{r in lags} corr[r] w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    for r in lags:

        elif r > n - 1:

            g[:m] += -h[k:]
        else:

grad[j] += -h[j+k] for j <= n-1-k (j ranges 0 to n-1-k)

In array terms:

```python
def _cyclic_pair_sweep(h, deadline):
    """Try all pairs in a structured order."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline and stall < 100:
        pass_count = 0
        improved = False
        # Try all (i, j) pairs where i < j
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                # Two directions: move mass i->j or j->i
                for sign in [1, -1]:
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    t, val = _line_search(h, d, dx, 0.5)
                    if val < cur_c5 - 1e-13:
                        h_new = h + t * d
                        h_new = _project(h_new, target)
                        new_c5 = _exact_c5(h_new)
                        if new_c5 < cur_c5 - 1e-14:
                            h = h_new
                            cur_c5 = new_c5
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            improved = True
                            break  # Move to next j
                # Early break if too slow
                if time.time() > deadline:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            pass_count += 1
    
    return best_h, best_c5
```
