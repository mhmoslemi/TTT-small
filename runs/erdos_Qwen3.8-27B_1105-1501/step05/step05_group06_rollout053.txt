def _project(v, target):
    # Bisection on Lagrange multiplier
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for _it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        idx = np.argsort(scores)[-top_k:]
        max_s = scores[idx].max()
        mean_s = scores[idx].mean()
        if max_s - mean_s < 1e-8:
            break
        G = _build_G(h)
        # Direction: reduce high scores, this is a gradient step on the max
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        w[idx] /= top_k
        grad = G @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.03
        for _ in range(20):
            h_new = _project(h - alpha * grad, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                break
            alpha *= 0.6
            if alpha < 1e-10:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: SLSQP at original resolution, increasing q
    q_schedule = [8, 16, 32, 64, 128, 256]
    for qi, q in enumerate(q_schedule):
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + budget_s * 0.06)
        h_cur, c5_cur = _slsqp_step(best_h, q, sub_dl, maxiter=250)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n0

    # Phase 2: Pair descent at original resolution
    if time.time() < deadline:
        sub_dl = min(deadline, time.time() + budget_s * 0.08)
        best_h, best_c5 = _pair_descent(best_h, sub_dl, max_passes=200, thresh=1e-7)
        best_n = len(best_h)

    # Phase 3: Equalization step
    if time.time() < deadline:
        sub_dl = min(deadline, time.time() + budget_s * 0.04)
        h_eq, c5_eq = _equalize_step(best_h, sub_dl, top_k=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 4: More SLSQP with higher q after pair descent
    if time.time() < deadline:
        for q in [128, 256, 512]:
            if time.time() > deadline:
                break
            sub_dl = min(deadline, time.time() + budget_s * 0.04)
            h_cur, c5_cur = _slsqp_step(best_h, q, sub_dl, maxiter=300)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = len(best_h)

    # Phase 5: Higher resolution attempts
    resolutions = [192, 256, 320, 384]
    for n_new in resolutions:
        if time.time() > deadline:
            break
        if best_c5 < 0.38075:
            break
        h_hi = _resample(best_h, n_new)
        c5_hi = _exact_c5(h_hi)
        for q in [16, 64, 128, 256]:
            if time.time() > deadline:
                break
            sub_dl = min(deadline, time.time() + budget_s * 0.035)
            h_hi, c5_hi = _slsqp_step(h_hi, q, sub_dl, maxiter=200)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_new
        if time.time() < deadline:
            sub_dl = min(deadline, time.time() + budget_s * 0.03)
            h_hi, c5_hi = _pair_descent(h_hi, sub_dl, max_passes=100, thresh=1e-7)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_new

    # Phase 6: Perturbation restarts at best resolution
    if time.time() < deadline:
        scale = 0.02
        sub_dl = min(deadline, time.time() + budget_s * 0.08)
        for attempt in range(200):
            if time.time() > sub_dl:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            iter_dl = min(sub_dl, time.time() + budget_s * 0.006)
            h_try, c5_try = _slsqp_step(h_pert, 128, iter_dl, maxiter=80)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.005, scale * 0.85)
            else:
                scale = min(0.06, scale * 1.1)

    # Phase 7: Multi-start with different constructions
    if time.time() < deadline:
        sub_dl = min(deadline, time.time() + budget_s * 0.08)
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5]
        for n_try in [best_n, 128, 256]:
            if time.time() > sub_dl:
                break
            for ai in range(3):
                if time.time() > sub_dl:
                    break
                alpha = alphas[ai % len(alphas)]
                h_init = _quasi_init(n_try, alpha, rng)
                c5_init = _exact_c5(h_init)
                if c5_init > best_c5 + 0.05:
                    continue
                iter_dl = min(sub_dl, time.time() + budget_s * 0.025)
                h_try, c5_try = _slsqp_step(h_init, 64, iter_dl, maxiter=150)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                    if time.time() < sub_dl:
                        pd_dl = min(sub_dl, time.time() + budget_s * 0.02)
                        best_h, best_c5 = _pair_descent(best_h, pd_dl, max_passes=80, thresh=1e-7)

    # Phase 8: Final intensive polish
    if time.time() < deadline:
        sub_dl = min(deadline, time.time() + budget_s * 0.04)
        best_h, best_c5 = _pair_descent(best_h, sub_dl, max_passes=200, thresh=1e-9)
        best_n = len(best_h)
        if time.time() < deadline:
            fin_dl = min(deadline, time.time() + budget_s * 0.025)
            h_fin, c5_fin = _slsqp_step(best_h, 512, fin_dl, maxiter=300)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
                best_n = len(best_h)
            if time.time() < deadline:
                pd_dl2 = min(deadline, time.time() + budget_s * 0.015)
                best_h, best_c5 = _pair_descent(best_h, pd_dl2, max_passes=100, thresh=1e-10)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
