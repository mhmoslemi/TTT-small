```python
def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
```

S[r] = corr[h, 1-h][r]
T[r] = corr[d, 1-h][r] - corr[h, d][r]  (this is the linear term)
Q[r] = corr[d, d][r]  (this is the quadratic coefficient, with a minus sign in the formula)

Wait, let me re-derive. S_r(h+td) = sum_i (h[i]+t*d[i]) * (1 - h[i+d] - t*d[i+d])
= sum_i h[i](1-h[i+d]) + t*sum_i d[i](1-h[i+d]) - t*sum_i h[i]*d[i+d] - t^2*sum_i d[i]*d[i+d]
= S_r(h) + t*[corr(d,1-h)[r] - corr(h,d)[r]] - t^2*corr(d,d)[r]

So T[r] = corr(d,1-h)[r] - corr(h,d)[r] and Q[r] = corr(d,d)[r].

The objective is phi(t) = dx * max_r (S[r] + t*T[r] - t^2*Q[r]).

This is the max of (2n-1) quadratic functions of t. Each quadratic is:
f_r(t) = -Q[r]*t^2 + T[r]*t + S[r]

If Q[r] > 0, f_r is concave (max at t = T[r]/(2*Q[r])).
If Q[r] < 0, f_r is convex.
If Q[r] = 0, f_r is linear.

The max of these is a piecewise function that's generally non-concave.

For the line search, I'll:
1. Compute S, T, Q arrays (each of length 2n-1)
2. Determine feasible interval [t_lo, t_hi]
3. Sample phi(t) at uniform grid points in [t_lo, t_hi]
4. Find the global minimum among samples
5. Also find local minima (points lower than both neighbors)
6. For each candidate minimum, do golden section refinement in a small bracket
7. Return the best t found

This is robust to non-unimodality.

```python
def _line_search_global(h, d, dx, t_lo, t_hi, n_samples=50):
    """Global 1D search using the exact quadratic model."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # phi(t) = dx * max(S + t*T - t^2*Q)
    # Sample on grid
    ts = np.linspace(t_lo, t_hi, n_samples)
    # Compute phi for each t: shape (n_samples, 2n-1)
    # phi_j = dx * max_r (S[r] + t_j*T[r] - t_j^2*Q[r])
    vals = S[None, :] + ts[:, None] * T[None, :] - (ts[:, None])**2 * Q[None, :]
    phis = np.max(vals, axis=1) * dx
    
    # Find global min
    best_idx = int(np.argmin(phis))
    best_t = ts[best_idx]
    best_val = phis[best_idx]
    
    # Find local minima
    local_mins = [best_idx]
    for i in range(1, len(ts) - 1):
        if phis[i] < phis[i-1] and phis[i] < phis[i+1]:
            local_mins.append(i)
    
    # Also check endpoints
    local_mins = list(set(local_mins + [0, len(ts)-1]))
    
    # Refine each local minimum with golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for idx in local_mins[:5]:  # Only refine top 5
        if idx == 0:
            a, b = t_lo, ts[1]
        elif idx == len(ts) - 1:
            a, b = ts[-2], t_hi
        else:
            a, b = ts[idx-1], ts[idx+1]
        
        c = b - gr * (b - a)
        dd = a + gr * (b - a)
        fc = float(np.max(S + c*T - c*c*Q)) * dx
        fd = float(np.max(S + dd*T - dd*dd*Q)) * dx
        
        for _ in range(50):
            if b - a < 1e-10:
                break
            if fc < fd:
                b = dd; dd = c; fd = fc
                c = b - gr * (b - a)
                fc = float(np.max(S + c*T - c*c*Q)) * dx
            else:
                a = c; c = dd; fc = fd
                dd = a + gr * (b - a)
                fd = float(np.max(S + dd*T - dd*dd*Q)) * dx
        
        t_opt = (a + b) / 2.0
        v_opt = float(np.max(S + t_opt*T - t_opt*t_opt*Q)) * dx
        if v_opt < best_val:
            best_val = v_opt
            best_t = t_opt
    
    return best_t, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0
_g_n = 0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_objGrad(x, tau):
    n = len(x)
    dx = 2.0 / n
    S = np.correlate(x, 1.0 - x, mode='full')
    M = float(np.max(S))
    if tau < 1e-10:
        idx = int(np.argmax(S))
        grad = _active_grad(x, idx)
        return float(S[idx] * dx), grad * dx
    exps = np.exp((S - M) / tau)
    total = float(np.sum(exps))
    w = exps / total
    val = (M + tau * np.log(total)) * dx
    grad = np.zeros(n)
    for k in range(n):
        r = k + n - 1
        wk = w[r]
        if wk < 1e-12:
            continue
        m = n - k
        grad[:m] += wk * (1.0 - x[k:k+m])
        grad[k:n] += -wk * x[:m]
    for k in range(1, n):
        r = n - 1 - k
        wk = w[r]
        if wk < 1e-12:
            continue
        m = n - k
        grad[k:n] += wk * (1.0 - x[:m])
        grad[:m] += -wk * x[k:n]
    grad *= dx
    return val, grad

def _active_grad(x, idx):
    n = len(x)
    d = idx - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        grad[:m] += (1.0 - x[d:d+m])
        grad[d:n] += -x[:m]
    else:
        g = -d
        m = n - g
        grad[g:n] += (1.0 - x[:m])
        grad[:m] += -x[g:n]
    return grad

def _con_fun(x):
    return float(np.sum(x) - _g_target)

def _con_jac(x):
    return np.ones(_g_n)

def _slsqp_tau(h, target, tau, maxiter=120):
    n = len(h)
    _g_target = target
    _g_n = n
    try:
        res = minimize(_soft_objGrad, h, args=(tau,), jac=True,
                       method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            x = _project(res.x, target)
            return x
    except Exception:
        pass
    return None

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gradient_direction(h):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(S))
    active = S >= M - 0.5 * (M - float(np.min(S))) * 0.1
    if not np.any(active):
        active = S >= M - 1e-10
    idxs = np.where(active)[0]
    g = np.zeros(n)
    for idx in idxs:
        g += _active_grad(h, int(idx))
    g /= float(len(idxs))
    g -= np.sum(g) * h / float(np.sum(h[h > 1e-10])) if np.sum(h[h > 1e-10]) > 0 else 0
    return g

        elif c5_r < c5_best + 0.002:
            tau_short = [1.0, 0.4, 0.15, 0.05]
            sub_dl = min(d5, time.time() + budget_s * 0.012)
            h_r, c5_r = _slsqp_chain(h_r, n_try / 2.0, sub_dl, tau_short, maxiter_base=60)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_r
                n_curr = n_try

    # Phase 6: Final polish (10%)
    d6 = hard_deadline
    h6 = h_best.copy()
    tau_final = [0.08, 0.04, 0.02, 0.01]
    h6, c5_6 = _slsqp_chain(h6, len(h6) / 2.0, d6, tau_final, maxiter_base=100)
    if c5_6 < c5_best:
        c5_best = c5_6
        h_best = h6
    # Final pair sweep
    sub_dl = min(d6, time.time() + max(5, budget_s * 0.03))
    h6, c5_6 = _pair_sweep(h_best, len(h_best) / 2.0, sub_dl, radius=0.15, max_tries=3000)
    if c5_6 < c5_best:
        c5_best = c5_6
        h_best = h6
    # One more SLSQP pass
    sub_dl = min(d6, time.time() + max(3, budget_s * 0.02))
    h6 = h_best.copy()
    h6, c5_6 = _slsqp_chain(h6, len(h6) / 2.0, sub_dl, [0.03, 0.015], maxiter_base=80)
    if c5_6 < c5_best:
        c5_best = c5_6
        h_best = h6

    h_best = _project(h_best, len(h_best) / 2.0)
    final_c5 = _exact_c5(h_best)
    if final_c5 < c5_best:
        c5_best = final_c5
    return h_best, c5_best, len(h_best)
```
