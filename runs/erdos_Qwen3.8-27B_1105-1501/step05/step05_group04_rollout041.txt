- If f is decreasing on [0,1], minimum is at t=1
- If f is increasing on [0,1], minimum is at t=0
- Otherwise, the minimum is where the subdifferential contains 0

import numpy as np

_STATE = {}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _h_from_a(a, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = a
    if n % 2 == 1:
        h[m] = 0.5
        h[m+1:] = 1.0 - a[::-1]
    else:
        h[m:] = 1.0 - a[::-1]
    return h

def _a_from_h(h):
    n = len(h)
    m = n // 2
    if n % 2 == 0:
        a = 0.5 * (h[:m] + (1.0 - h[m:])[::-1])
    else:
        a = 0.5 * (h[:m] + (1.0 - h[m+1:])[::-1])
    return np.clip(a, 0.0, 1.0)

def _compute_lag_grad(h, w):
    n = len(h)
    grad_h = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        k = r - (n - 1)
        wr = w[r]
        if k >= 0:
            cnt = n - k
            grad_h[:cnt] += wr * (1.0 - h[k:cnt+k])
            grad_h[k:cnt+k] -= wr * h[:cnt]
        else:
            g2 = -k
            cnt = n - g2
            grad_h[:cnt] += wr * (1.0 - h[g2:cnt+g2])
            grad_h[g2:cnt+g2] -= wr * h[:cnt]
    return grad_h

        _STATE['deadline'] = deadline
        _STATE['best_c5'] = best_c5
        _STATE['best_h'] = _h_from_a(best_a, n).copy()
        try:
            res = minimize(_margin_sym_obj, a, jac=_margin_sym_grad, method='L-BFGS-B',
                          bounds=bounds, callback=_cb_sym,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
            if hasattr(res, 'x') and res.x is not None:
                a_new = np.clip(res.x, 0.0, 1.0)
                c5_new = _exact_c5(_h_from_a(a_new, n))
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_a = a_new.copy()
                a = a_new
        except (TimeoutError, Exception):
            pass
    if _STATE['best_c5'] < best_c5:
        best_c5 = _STATE['best_c5']
        best_a = _a_from_h(_STATE['best_h'])
    return best_a, best_c5
