    max_s = np.max(scores)
    # For numerical stability
    shifted = scores - max_s
    log_sum = np.log(np.sum(np.exp(beta * shifted))) + beta * max_s

        return x[-1]

        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    def ineq_funs(x):
        h = x[:n]
        t = x[-1]
        scores = _scores(h)
        return t - scores
    
    def ineq_jac(x):
        h = x[:n]
        G = _score_grad_matrix(h)
        # ineq_r = t - score_r, so d/dh = -G, d/dt = 1
        J = np.zeros((2*n-1, n+1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    x0 = np.zeros(n+1)
    x0[:n] = h0
    x0[-1] = t0
    
    # Bounds
    bounds = [(0, 1)] * n + [(0, None)]
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_funs, 'jac': ineq_jac}

        return h0.copy(), t0

        return h_new, t_new

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _score_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        s = r - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                G[r, s:n] += dx * (1.0 - h[0:L])
                G[r, 0:L] += -dx * h[s:s + L]
        else:
            a = -s
            L = n - a
            if L > 0:
                G[r, 0:L] += dx * (1.0 - h[a:n])
                G[r, a:n] += -dx * h[0:L]
    return G


# SLSQP epigraph top-level functions
def _slsqp_obj(x):
    return x[-1]


def _slsqp_obj_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def _slsqp_eq_fun(x):
    n = len(x) - 1
    return np.array([np.sum(x[:n]) - n / 2.0])


def _slsqp_eq_jac(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def _slsqp_ineq_fun(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    return t - _scores(h)


def _slsqp_ineq_jac(x):
    n = len(x) - 1
    h = x[:n]
    G = _score_grad_matrix(h)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def _slsqp_run(h0, t_start, deadline, maxiter=80):
    n = len(h0)
    target = n / 2.0
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[-1] = max(t_start, _exact_c5(h0))
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac},
        {'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac}
    ]
    if time.time() > deadline:
        return h0.copy(), _exact_c5(h0)
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h0.copy(), _exact_c5(h0)


# Log-sum-exp optimization
def _lse_obj(x, beta, lam, target):
    scores = _scores(x)
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    val = np.log(np.sum(np.exp(shifted))) / beta + max_s
    val += lam * (np.sum(x) - target) ** 2
    return val


def _lse_grad(x, beta, lam, target):
    scores = _scores(x)
    n = len(x)
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    G = _score_grad_matrix(x)
    g = G.T @ p
    g += 2.0 * lam * (np.sum(x) - target)
    return g


def _lse_run(h0, deadline, beta_start=40, beta_end=400, n_stages=5, maxiter_per=40):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    lam = 1.0
    for bi in range(n_stages):
        if time.time() > deadline:
            break
        frac = bi / max(1, n_stages - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        cur_h = best_h.copy()
        try:
            res = minimize(_lse_obj, cur_h, method='L-BFGS-B', jac=_lse_grad,
                           args=(beta, lam, target),
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per, 'gtol': 1e-10})
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new
            if abs(np.sum(res.x) - target) > 0.3:
                lam *= 4.0
        except Exception:
            pass
    return best_h, best_c5


# LP direction finding and line search
def _line_search(h, d, dx, t_max):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = t_max
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(0.0, float(np.max(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 80
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val


def _lp_direction(h, scores, G, radius, n_active=5):
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    active_set = sorted_idx[:n_active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.full(n_lags, 0.0)
    for k in range(n_lags):
        if k in [int(x) for x in active_set]:
            b_ub[k] = -scores[k]
        else:
            b_ub[k] = -scores[k] + radius * 0.5
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 3
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _scores(h)
        G = _score_grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            n_active = min(n_active + 2, 2 * n - 1)
            if n_active > n:
                p = _lp_direction(h, scores, G, radius, max(2, n // 3))
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.72
                if radius < 1e-6:
                    break
                if stall % 30 == 0:
                    radius = max(radius, 0.05)
                continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 3
            radius = min(radius * 1.08, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.03:
                radius = 0.08
                n_active = 5
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _coord_descend(h, deadline, n_iter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5


def _make_starts(h, n, rng, count=6):
    starts = [h.copy()]
    x = 2.0 * (np.arange(n) + 0.5) / n
    for freq in range(1, 4):
        phi = rng.uniform(0, 2 * np.pi)
        h_s = 0.5 + 0.35 * np.sin(freq * np.pi * x + phi) + 0.15 * np.sin(2 * freq * np.pi * x + 2 * phi)
        h_s = _project(h_s, n / 2.0)
        starts.append(h_s)
    h_r = _project(rng.uniform(0, 1, n), n / 2.0)
    starts.append(h_r)
    h_blend = _project(0.7 * h + 0.3 * h_r, n / 2.0)
    starts.append(h_blend)
    return starts[:count]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float).copy()
        except Exception:
            h0 = _project(0.5 + 0.3 * np.sin(np.pi * np.linspace(0, 2, 80)), 40.0)
    else:
        h0 = _project(0.5 + 0.3 * np.sin(np.pi * np.linspace(0, 2, 80)), 40.0)

    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def update(h_cand, c5_cand):
        nonlocal best_h, best_c5, best_n
        if c5_cand < best_c5 - 1e-14:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = len(h_cand)

    # Phase 1: SLSQP from parent (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _slsqp_run(h, best_c5, d1, maxiter=60)
    update(h1, c5_1)
    h1b, c5_1b = _slsqp_run(h, best_c5 - 0.0004, d1, maxiter=60)
    update(h1b, c5_1b)

    # Phase 2: LSE optimization (12%)
    d2 = t0 + budget_s * 0.24
    h2, c5_2 = _lse_run(best_h, d2, beta_start=40, beta_end=300, n_stages=4)
    update(h2, c5_2)

    # Phase 3: Multi-resolution exploration (20%)
    d3 = t0 + budget_s * 0.44
    n_base = best_n
    tried = set([n_base])
    for dn in range(0, 25):
        if time.time() > d3:
            break
        for sign in (1, -1):
            if dn == 0 and sign == -1:
                continue
            n_try = n_base + sign * dn
            if n_try < 30 or n_try > 180 or n_try in tried:
                continue
            tried.add(n_try)
            h_r = _resample(best_h, n_try)
            sub_dl = min(d3, time.time() + 4.0)
            h_r, c5_r = _slsqp_run(h_r, _exact_c5(h_r), sub_dl, maxiter=40)
            update(h_r, c5_r)
            if time.time() < d3:
                sub_dl2 = min(d3, time.time() + 3.0)
                h_r2, c5_r2 = _lp_loop(h_r, sub_dl2, radius_init=0.1)
                update(h_r2, c5_r2)

    # Phase 4: Diverse starting points + SLSQP (18%)
    d4 = t0 + budget_s * 0.62
    for n_try in [best_n, max(30, best_n - 10), min(160, best_n + 10)]:
        if time.time() > d4:
            break
        starts = _make_starts(best_h, n_try, rng, count=5)
        for h_s in starts:
            if time.time() > d4:
                break
            sub_dl = min(d4, time.time() + 5.0)
            h_s, c5_s = _slsqp_run(h_s, _exact_c5(h_s) + 0.001, sub_dl, maxiter=50)
            update(h_s, c5_s)

    # Phase 5: LP-based polishing (18%)
    d5 = t0 + budget_s * 0.80
    h5, c5_5 = _lp_loop(best_h, d5, radius_init=0.1)
    update(h5, c5_5)

    # Phase 6: Coordinate descent (10%)
    d6 = t0 + budget_s * 0.90
    h6, c5_6 = _coord_descend(best_h, d6, n_iter=400)
    update(h6, c5_6)

    # Phase 7: Final SLSQP polish (8%)
    d7 = deadline
    h7, c5_7 = _slsqp_run(best_h, best_c5, d7, maxiter=50)
    update(h7, c5_7)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
