def _smooth_grad(h, dx, beta):

    scores = np.correlate(h, 1.0 - h, mode='full') * dx

                grad[:m] += w[r] * (-h[g:g + m])

def _sig(x):
    """Numerically stable sigmoid."""
    out = np.zeros_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _logit_to_h(v, n):

            lo = mid

    """Optimize in logit space using random kicks + line search."""
    target = len(h) / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = _h_to_logit(best_h)
    T = 0.15
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        T *= 0.95
        direction = rng.normal(0, T, len(h))
        v_new = v + direction
        h_new = _logit_to_h(v_new, len(h))
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            v = v_new.copy()
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T * 0.01, 1e-6)):

def _critical_shift_opt(h, deadline, rng):
    """Identify critical shifts and try to reduce them specifically."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    while time.time() < deadline and stall < 50:
        scores = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        s_max = float(np.max(scores))
        # Find critical lags
        critical = np.where(scores >= s_max * 0.998)[0]
        if len(critical) == 0:
            break
        # For each critical lag, try to find a good move
        improved = False
        for r in critical[:5]:
            if time.time() > deadline:
                break
            d_lag = int(r - (n - 1))
            # Build a direction that reduces overlap at this specific lag
            # The gradient with respect to h_i for lag d is:
            # For d >= 0: grad[i] += (1 - h[i+d]) for i < n-d, grad[i+d] += -h[i]
            # For d < 0: let g = -d, grad[i] += -h[i+g] for i < n-g, grad[i+g] += (1 - h[i])
            d_dir = np.zeros(n)
            if d_lag >= 0:
                m = n - d_lag
                if m > 0:
                    d_dir[:m] += (1.0 - best_h[d_lag:d_lag + m])
                    d_dir[d_lag:] += (-best_h[:m])
            else:
                g = -d_lag
                m = n - g
                if m > 0:
                    d_dir[:m] += (-best_h[g:g + m])
                    d_dir[g:] += (1.0 - best_h[:m])
            d_dir -= np.mean(d_dir)
            if np.linalg.norm(d_dir) < 1e-13:
                continue
            d_dir /= np.linalg.norm(d_dir)
            alpha, c5_try = _line_search(best_h, d_dir, dx, 0.3)
            if c5_try < best_c5 - 1e-14:
                best_h = best_h + alpha * d_dir
                best_h = _project(best_h, target)
                best_c5 = _exact_c5(best_h)
                improved = True
        if not improved:
            # Try random perturbation
            perturb = rng.normal(0, 0.03, n)
            perturb -= perturb.mean()
            h_try = _project(best_h + perturb, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_h = h_try
                best_c5 = c5_try
                improved = True
        if not improved:
            stall += 1
    return best_h, best_c5

def _kick_and_reopt(h, deadline, rng, n_kicks=30):
