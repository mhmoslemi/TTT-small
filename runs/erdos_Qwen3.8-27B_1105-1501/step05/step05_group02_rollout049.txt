import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pseudo_grad(h, gamma):
    """Compute weighted pseudo-gradient of C5 at h using softmax over scores."""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(scores)
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    # Accumulate gradient
    grad = np.zeros(n)
    # For each lag d and its weight, add contribution
    for d in range(0, n):
        # Positive lag: index n-1+d, weight w[n-1+d]
        if d == 0:
            wp = w[n - 1]
            grad += wp * dx * (1.0 - 2.0 * h)
        else:
            m = n - d
            # c[n-1+d] = sum_{j=0}^{m-1} h[j+d]*(1-h[j])
            wp = w[n - 1 + d]
            if wp > 1e-15:
                grad[:m] += wp * dx * (-h[d:])
                grad[d:] += wp * dx * (1.0 - h[:m])
            # Negative lag: index n-1-d, weight w[n-1-d]
            wn = w[n - 1 - d]
            if wn > 1e-15:
                grad[:m] += wn * dx * (1.0 - h[d:])
                grad[d:] += wn * dx * (-h[:m])
    # Center (remove mean to preserve sum)
    grad -= grad.mean()
    return grad, scores

def _pair_line_search(h, i, j, n_grid, cur_c5):
    """Exact 1D line search: h[i]+=t, h[j]-=t. Returns (t, new_c5)."""
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-14:
        return 0.0, cur_c5
    # Grid sampling
    ts = np.linspace(t_min, t_max, n_grid + 2)[1:-1]
    best_t = 0.0
    best_c5 = cur_c5
    for t in ts:
        h[i] += t
        h[j] -= t
        c5 = _exact_c5(h)
        h[i] -= t
        h[j] += t
        if c5 < best_c5:
            best_c5 = c5
            best_t = t

        return 0.0, cur_c5
    # Golden section around best sample

    center_idx = int(frac * (n_grid + 1))

    hi = ts[min(len(ts) - 1, center_idx)]

        return best_t, best_c5

    c2 = lo + gr * (hi - lo)
    h[i] += c1; h[j] -= c1

    h[i] += c2; h[j] -= c2
    f2 = _exact_c5(h)

def _cyclic_pair_pass(h, n_grid):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    best_h = h.copy()
    best_c5 = cur_c5
    T = temp_scale * cur_c5
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5
