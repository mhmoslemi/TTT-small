def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binary_eval(bits, n):
    """Evaluate C5 for binary vector represented as integer bits."""
    max_val = 0
    for d in range(1, n):
        m = n - d
        mask = (1 << m) - 1
        # h[i+d]=1, h[i]=0 (positive lag d in correlate convention)
        c1 = ((bits >> d) & ~(bits) & mask).bit_count()
        # h[i]=1, h[i+d]=0
        c2 = (bits & ~(bits >> d) & mask).bit_count()
        if c1 > max_val:
            max_val = c1
        if c2 > max_val:
            max_val = c2
    return max_val * 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad(h, w):
    """Compute gradient of weighted sum of scores."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if wd < 1e-14:
            continue
        if d == 0:
            grad += wd * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += wd * dx * (1.0 - h[d:])
            grad[d:] -= wd * dx * h[:m]
        else:
            g = -d
            m = n - g
            grad[g:] += wd * dx * (1.0 - h[:m])
            grad[:m] -= wd * dx * h[g:]
    return grad

def _smooth_obj_and_grad(x, tau):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    scores = _scores(h)
    max_s = scores.max()
    shifted = (scores - max_s) / tau
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_vals = np.exp(shifted)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(scores)
        w[int(np.argmax(scores))] = 1.0
    else:
        w = exp_vals / w_sum
    obj = max_s + tau * np.log(w_sum)
    grad = _compute_grad(h, w)
    return obj, grad

def _smooth_obj_func(x):
    return _smooth_obj_and_grad(x, _SMOOTH_TAU[0])[0]

def _smooth_grad_func(x):
    return _smooth_obj_and_grad(x, _SMOOTH_TAU[0])[1]

def _eq_fun(x):
    return np.sum(x) - _OPT_TARGET[0]

def _eq_jac(x):
    return np.ones(len(x))

def _smooth_optimize(h0, tau_list, deadline, maxiter=200):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = tuple((0.0, 1.0) for _ in range(n))
    for tau in tau_list:
        if time.time() > deadline:
            break
        _SMOOTH_TAU[0] = tau
        _OPT_TARGET[0] = target
        x0 = _project(h, target)
        try:
            res = minimize(_smooth_obj_func, x0, jac=_smooth_grad_func,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None:
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                elif c5_new < best_c5 + 1e-13:
                    h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def _subgrad_step(h, gamma):
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = gamma * (scores - max_s)
    shifted = np.clip(shifted, -100.0, 100.0)
    w = np.exp(shifted)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = _compute_grad(h, w)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, _exact_c5(h)
    grad /= gn
    cur_c5 = _exact_c5(h)
    best_a = 0.0
    best_c5 = cur_c5
    alpha = 0.05
    for _ in range(25):
        h_new = _project(h - alpha * grad, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_a = alpha
        alpha *= 0.7
        if alpha < 1e-12:
            break
    if best_a > 0:
        return _project(h - best_a * grad, target), best_c5
    return h, cur_c5

def _subgrad_optimize(h, deadline, gamma_init=5, gamma_max=500, max_it=3000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        h_new, c5_new = _subgrad_step(h, gamma)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.02, gamma_max)
        else:
            stall += 1
            gamma = min(gamma * 1.05, gamma_max)
    return best_h, best_c5

def _binary_eval(bits, n):
    max_val = 0
    for d in range(1, n):
        m = n - d
        mask = (1 << m) - 1
        c1 = ((bits >> d) & ~(bits) & mask).bit_count()
        c2 = (bits & ~(bits >> d) & mask).bit_count()
        if c1 > max_val:
            max_val = c1
        if c2 > max_val:
            max_val = c2
    return max_val * 2.0 / n

def _binary_local_search(bits, n, deadline, max_iters=5000000):
    K = n // 2
    if bits.bit_count() != K:
        return bits, _binary_eval(bits, n)
    cur_c5 = _binary_eval(bits, n)
    best_bits = bits
    best_c5 = cur_c5
    T = 0.08
    min_T = 0.002
    cool_rate = (min_T / T) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        bi = (bits >> i) & 1
        bj = (bits >> j) & 1
        if bi == bj:
            continue
        new_bits = bits ^ (1 << i) ^ (1 << j)
        new_c5 = _binary_eval(new_bits, n)
        if new_c5 <= cur_c5 or np.random.random() < math.exp((cur_c5 - new_c5) / T):
            bits = new_bits
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_bits = bits
        T *= cool_rate
    return best_bits, best_c5

def _bits_to_array(bits, n):
    arr = np.zeros(n)
    for i in range(n):
        if (bits >> i) & 1:
            arr[i] = 1.0
    return arr

def _array_to_bits(h):
    bits = 0
    for i in range(len(h)):
        if h[i] > 0.5:
            bits |= (1 << i)
    return bits

def _make_balanced_bits(h, n):
    sorted_idx = np.argsort(-h)
    bits = 0
    for i in range(n // 2):
        bits |= (1 << sorted_idx[i])
    return bits

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        x = (i * alpha) % 1.0
        if x < 0.5:
            h[i] = 1.0
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fw_step(h, gamma):
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = gamma * (scores - max_s)
    shifted = np.clip(shifted, -100.0, 100.0)
    w = np.exp(shifted)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = _compute_grad(h, w)
    order = np.argsort(grad)
    v = np.zeros(n)
    v[order[:int(target)]] = 1.0
    d = v - h
    if np.max(np.abs(d)) < 1e-14:
        return h, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_a > 0:
        h_new = _project(h + opt_a * d, target)
        return h_new, _exact_c5(h_new)
    return h, _exact_c5(h)

def _fw_optimize(h, deadline, gamma_init=10, gamma_max=300, max_it=2000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        h_new, c5_new = _fw_step(h, gamma)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.02, gamma_max)
        else:
            stall += 1
            gamma = min(gamma * 1.06, gamma_max)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Smooth SLSQP from initial (12% budget)
    dl1 = t0 + budget_s * 0.12
    tau_list1 = [0.04, 0.015, 0.006, 0.002, 0.001]
    h1, c5_1 = _smooth_optimize(h_start, tau_list1, dl1, maxiter=120)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient (8%)
    dl2 = t0 + budget_s * 0.20
    h2, c5_2 = _subgrad_optimize(best_h, dl2, gamma_init=3, gamma_max=200, max_it=2000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smooth again (10%)
    dl3 = t0 + budget_s * 0.30
    tau_list3 = [0.015, 0.006, 0.002, 0.001, 0.0004]
    h3, c5_3 = _smooth_optimize(best_h, tau_list3, dl3, maxiter=150)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Frank-Wolfe (8%)
    dl4 = t0 + budget_s * 0.38
    h4, c5_4 = _fw_optimize(best_h, dl4, gamma_init=8, gamma_max=200, max_it=1500)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary local search at current resolution (22%)
    dl5 = t0 + budget_s * 0.60
    bits = _make_balanced_bits(best_h, best_n)
    h_bin, c5_bin = _binary_local_search(bits, best_n, dl5, max_iters=4000000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = _bits_to_array(h_bin, best_n).copy()
    # Smooth polish the binary solution
    if time.time() < dl5 and best_c5 <= 0.381:
        tau_polish = [0.008, 0.003, 0.001, 0.0004]
        h_bp, c5_bp = _smooth_optimize(best_h, tau_polish, dl5, maxiter=100)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()

    # Phase 6: Multi-resolution exploration (20%)
    dl6 = t0 + budget_s * 0.80
    for n_try in [160, 192, 144, 224, 128]:
        if time.time() > dl6:
            break
        if best_c5 < 0.3805:
            break
        h_res = _resample(best_h, n_try)
        sub_dl = min(dl6, time.time() + budget_s * 0.035)
        tau_res = [0.02, 0.008, 0.003, 0.001]
        h_try, c5_try = _smooth_optimize(h_res, tau_res, sub_dl, maxiter=100)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            # Binary refine at this resolution
            if time.time() < dl6:
                bits_t = _make_balanced_bits(best_h, best_n)
                sub_dl2 = min(dl6, time.time() + budget_s * 0.015)
                h_bt, c5_bt = _binary_local_search(bits_t, best_n, sub_dl2, max_iters=2000000)
                if c5_bt < best_c5:
                    best_c5 = c5_bt
                    best_h = _bits_to_array(h_bt, best_n).copy()
                    # Quick smooth polish
                    if time.time() < dl6:
                        tp2 = [0.004, 0.001]
                        h_bp2, c5_bp2 = _smooth_optimize(best_h, tp2, dl6, maxiter=80)
                        if c5_bp2 < best_c5:
                            best_c5 = c5_bp2
                            best_h = h_bp2.copy()

    # Phase 7: Quasi-periodic multi-start (8%)
    if time.time() < t0 + budget_s * 0.88:
        dl7 = t0 + budget_s * 0.88
        alphas = [0.6180339887, 0.4142135624, 0.3819660113, 0.7071067812]
        for n_qp in [128, 160]:
            if time.time() > dl7:
                break
            for a in alphas:
                if time.time() > dl7:
                    break
                h_qp = _quasi_periodic(n_qp, a)
                sub_dl = min(dl7, time.time() + budget_s * 0.012)
                tau_qp = [0.015, 0.005, 0.002]
                h_qo, c5_qo = _smooth_optimize(h_qp, tau_qp, sub_dl, maxiter=80)
                if c5_qo < best_c5:
                    best_c5 = c5_qo
                    best_h = h_qo.copy()
                    best_n = n_qp
                    # Binary refine
                    if time.time() < dl7:
                        bits_q = _make_balanced_bits(best_h, best_n)
                        sub_dl2 = min(dl7, time.time() + budget_s * 0.008)
                        h_bq, c5_bq = _binary_local_search(bits_q, best_n, sub_dl2, max_iters=1000000)
                        if c5_bq < best_c5:
                            best_c5 = c5_bq
                            best_h = _bits_to_array(h_bq, best_n).copy()

    # Phase 8: Perturbation restarts (8%)
    dl8 = t0 + budget_s * 0.96
    scale = 0.03
    for attempt in range(200):
        if time.time() > dl8:
            break
        h_p = best_h + np.random.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        sub_dl = min(dl8, time.time() + budget_s * 0.005)
        tau_p = [0.005, 0.002]
        h_try, c5_try = _smooth_optimize(h_p, tau_p, sub_dl, maxiter=60)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 9: Final aggressive polish (4%)
    if time.time() < t0 + budget_s * 0.97:
        dl9 = t0 + budget_s * 0.99
        tau_final = [0.003, 0.001, 0.0004, 0.0002]
        h_fin, c5_fin = _smooth_optimize(best_h, tau_final, dl9, maxiter=150)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        # One more subgradient pass
        if time.time() < dl9:
            h_sg, c5_sg = _subgrad_optimize(best_h, dl9, gamma_init=20, gamma_max=500, max_it=500)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
