        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _softmax_obj_grad(h, tau, dx):
    """Compute soft-max smoothed objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Stable log-sum-exp
    M = np.max(corr)
    exp_vals = np.exp((corr - M) / tau)
    log_sum = M / tau + np.log(np.sum(exp_vals))
    obj = dx * tau * log_sum
    
    # Weights
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient
    g = np.zeros(n)
    # For each lag r, compute contribution
    for r in range(len(corr)):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)

            if m > 0:

    g *= dx
    return obj, g

def _line_search_exact(h, d, dx, radius, n_grid=51):
    """Exact line search along direction d using quadratic model."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Sample on grid
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S + ts[:, None] * T[None, :] - ts[:, None]**2 * Q[None, :], axis=1) * dx
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Refine with golden section around the best point
    if 0 < best_idx < n_grid - 1:
        gl = ts[best_idx - 1]
        gh = ts[best_idx + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = gh - gr * (gh - gl)
        c2 = gl + gr * (gh - gl)
        f1 = float(np.max(S + c1 * T - c1**2 * Q)) * dx
        f2 = float(np.max(S + c2 * T - c2**2 * Q)) * dx
        for _ in range(30):
            if gh - gl < 1e-12:
                break
            if f1 < f2:
                gh = c2; c2 = c1; f2 = f1
                c1 = gh - gr * (gh - gl)
                f1 = float(np.max(S + c1 * T - c1**2 * Q)) * dx
            else:
                gl = c1; c1 = c2; f1 = f2
                c2 = gl + gr * (gh - gl)
                f2 = float(np.max(S + c2 * T - c2**2 * Q)) * dx
        best_t = (gl + gh) / 2.0
        best_val = float(np.max(S + best_t * T - best_t**2 * Q)) * dx
    
    return best_t, best_val

from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _softmax_obj(corr, tau):
    M = np.max(corr)
    vals = np.exp((corr - M) / tau)
    return M + tau * np.log(np.sum(vals))

def _softmax_grad(h, tau, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    exps = np.exp((corr - M) / tau)
    w = exps / np.sum(exps)
    g = np.zeros(n)
    for r in range(len(corr)):
        if w[r] < 1e-10:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w[r] * (1.0 - h[:m])
                g[:m] -= w[r] * h[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += w[r] * (1.0 - h[gg:])
                g[gg:] -= w[r] * h[:m]
    g *= dx
    return g, corr

        d = np.zeros(n)
        d[a:a+bl] = 1.0 / bl
        d[b:b+bl] = -1.0 / bl
        alpha, c5_try = _line_search_exact(h, d, dx, 0.5)
        if c5_try < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best = h.copy()
    return best, best_c5

def _lowfreq_dir(n, mode, rng):
    x = np.linspace(0, 1, n, endpoint=False)
    d = np.zeros(n)
    d[:] = np.sin(2 * np.pi * mode * x + rng.uniform(0, 2 * np.pi))
    d -= np.mean(d)
    return d

def _lowfreq_opt(h, target, dx, deadline, n_tries=100):
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(333)
    for trial in range(n_tries):
        if time.time() > deadline:
            break
        mode = int(rng.integers(2, max(3, n // 20)))
        d = _lowfreq_dir(n, mode, rng)
        gn = np.linalg.norm(d)
        if gn < 1e-12:
            continue
        d /= gn
        alpha, c5_try = _line_search_exact(h, d, dx, 0.3)
        if c5_try < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best = h.copy()
    return best, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h_init = _project(h0.copy(), target0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: SLSQP on initial (12%)
    d1 = t0 + budget_s * 0.12
    tau_list1 = [0.08, 0.05, 0.03, 0.018, 0.01]
    h1, c5_1 = _slsqp_chain(h_init, target0, 2.0/n0, tau_list1, d1, max_iter=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Gradient descent refinement
    d1b = t0 + budget_s * 0.16
    sigmas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    for tau_g in [0.03, 0.015, 0.008]:
        if time.time() > d1b:
            break
        h_g, c5_g = _grad_step(best_h, target0, 2.0/n0, tau_g, sigmas, n_steps=10)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: Diverse starts with SLSQP (25%)
    d2 = t0 + budget_s * 0.42
    n_trials = 8
    remaining = max(1.0, d2 - time.time())
    per_trial = remaining / n_trials
    for trial in range(n_trials):
        if time.time() > d2:
            break
        if trial == 0:
            h_r = _random_binary(n0, rng)
        elif trial == 1:
            x = np.linspace(0, 1, n0, endpoint=False)
            h_r = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x + 0.5) + 0.2 * np.cos(2 * np.pi * 5 * x + 1.2)
        elif trial == 2:
            h_r = best_h + rng.normal(0, 0.08, n0)
        elif trial == 3:
            h_r = best_h + rng.normal(0, 0.15, n0)
        elif trial == 4:
            h_r = np.flip(best_h).copy()
        elif trial == 5:
            h_r = _random_binary(n0, rng)
        elif trial == 6:
            x = np.linspace(0, 1, n0, endpoint=False)
            h_r = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x) + rng.normal(0, 0.05, n0)
        else:
            h_r = best_h + rng.normal(0, 0.12, n0)
        h_r = _project(h_r, target0)
        sub_dl = min(d2, time.time() + per_trial * 0.9)
        tau_sub = [0.06, 0.03, 0.015]
        h_r, c5_r = _slsqp_chain(h_r, target0, 2.0/n0, tau_sub, sub_dl, max_iter=35)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Intensive local optimization (28%)
    d3 = t0 + budget_s * 0.70
    dx0 = 2.0 / n0
    # Structured pairs
    sub_dl = min(d3, time.time() + budget_s * 0.07)
    h3, c5_3 = _structured_pair_opt(best_h, target0, dx0, sub_dl, n_tries=1500, radius=0.35)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Random pairs
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h3, c5_3 = _random_pair_opt(best_h, target0, dx0, sub_dl, n_tries=2000, radius=0.3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Block moves
    sub_dl = min(d3, time.time() + budget_s * 0.04)
    h3, c5_3 = _block_move_opt(best_h, target0, dx0, sub_dl, n_tries=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Low-freq directions
    sub_dl = min(d3, time.time() + budget_s * 0.04)
    h3, c5_3 = _lowfreq_opt(best_h, target0, dx0, sub_dl, n_tries=80)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # SLSQP polish
    sub_dl = min(d3, time.time() + budget_s * 0.05)
    h3, c5_3 = _slsqp_chain(best_h, target0, dx0, [0.01, 0.005, 0.003], sub_dl, max_iter=40)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (22%)
    d4 = t0 + budget_s * 0.92
    n_curr = len(best_h)
    offsets = [8, -8, 16, -16, 4, -4, 12, -12, 24, -24, 32, -32]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        target_r = n_try / 2.0
        dx_r = 2.0 / n_try
        rem = d4 - time.time()
        if rem < 5:
            break
        sub_dl = min(d4, time.time() + min(rem * 0.7, budget_s * 0.025))
        h_r, c5_r = _slsqp_chain(h_r, target_r, dx_r, [0.05, 0.025, 0.012], sub_dl, max_iter=35)
        if c5_r < best_c5 + 0.002 and time.time() < d4 - 5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.015)
            h_r, c5_r = _structured_pair_opt(h_r, target_r, dx_r, sub_dl2, n_tries=800, radius=0.3)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Final polish (8%)
    h_final = best_h.copy()
    n_f = len(h_final)
    target_f = n_f / 2.0
    dx_f = 2.0 / n_f
    tau_final = [0.008, 0.004, 0.002]
    h_final, c5_f = _slsqp_chain(h_final, target_f, dx_f, tau_final, deadline, max_iter=45)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_final.copy()
    # Final pair polish
    h_final, c5_f = _structured_pair_opt(best_h, target_f, dx_f, deadline, n_tries=600, radius=0.2)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_final.copy()
    # Gradient step
    h_final, c5_f = _grad_step(best_h, target_f, dx_f, 0.005, [0.1, 0.05, 0.02, 0.01], n_steps=8)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_final.copy()

    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        return best_h, final_c5, len(best_h)
    else:
        # Fallback to pre-projection best
        return best_h, final_c5, len(best_h)
