def smooth_descent(h, deadline, lam_init=50.0, lam_min=5.0, lam_decay=0.998):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lam = lam_init
    stall = 0
    step_scale = 0.05
    
    while time.time() < deadline:
        if stall > 100:
            break

        # Project to sum-zero
        grad -= np.mean(grad)

            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:

                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            # Adapt lambda
            if cur_c5 < best_c5 * 0.999:
                lam = max(lam_min, lam * 0.995)
            step_scale = min(0.2, step_scale * 1.01)

            step_scale *= 0.85

def pair_moves(h, deadline, n_trials=50000):

def cooldown_sa(h, deadline, rng, n_calls=30000):

    best_h = h.copy()

    T_min = 1e-6

    zeros = np.where(h <= 0.5)[0]

            break

        j = zeros[rng.integers(len(zeros))]

def resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)

    ba, bv = grid[bi], vals[bi]
    if bi > 0 and bi < ng - 1:
        a, b = grid[bi - 1], grid[bi + 1]
    else:
        a, b = alpha_lo, alpha_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = phi(c1), phi(c2)
    for _ in range(30):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = phi(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = phi(c2)
    oa = (a + b) / 2.0
    ov = phi(oa)
    if ov < bv:
        ba, bv = oa, ov
    return ba, bv

def lse_gradient(h, lam):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    S_max = np.max(S)
    exp_S = np.exp(lam * (S - S_max))
    w = exp_S / np.sum(exp_S)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-9:
            continue
        s = r - (n - 1)
        if s >= 0:
            m = n - s
            if m > 0:
                grad[:m] += wr * (1.0 - h[s:s + m])
                grad[s:s + m] -= wr * h[:m]
        else:
            g = -s
            m = n - g
            if m > 0:
                grad[g:g + m] += wr * (1.0 - h[:m])
                grad[:m] -= wr * h[g:g + m]
    grad *= dx
    return grad

def smooth_descent(h, deadline, lam_init=80.0, lam_min=10.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lam = lam_init
    stall = 0
    step_scale = 0.04
    while time.time() < deadline:
        if stall > 150:
            break
        grad = lse_gradient(h, lam)
        grad -= np.mean(grad)
        d = -grad * step_scale
        max_step = 0.4
        pos = d < -1e-12
        neg = d > 1e-12
        if np.any(pos):
            max_step = min(max_step, float(np.min(h[pos] / (-d[pos]))))
        if np.any(neg):
            max_step = min(max_step, float(np.min((1.0 - h[neg]) / d[neg])))
        if max_step < 1e-12:
            stall += 1
            step_scale *= 0.7
            if step_scale < 1e-5:
                break
            continue
        alpha, new_c5 = line_search(h, d, max_step)
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-15:
            h = h + alpha * d
            h = project_feasible(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
                lam = max(lam_min, lam * 0.997)
                step_scale = min(0.15, step_scale * 1.02)
            else:
                stall += 1
        else:
            stall += 1
            step_scale *= 0.8
            if stall % 40 == 0:
                lam = max(lam_min, lam * 0.95)
                step_scale = 0.05
    return best_h, best_c5

def guided_pair_moves(h, deadline, rng, n_trials=100000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cur_max_raw = float(np.max(S0))
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-6:
            continue
        du = np.zeros(n)
        du[i] = -1.0
        du[j] = 1.0
        T_vec = np.correlate(du, 1.0 - h, mode='full') - np.correlate(h, du, mode='full')
        eps = min(h[i], 1.0 - h[j]) * 0.8
        if eps < 1e-7:
            continue
        new_max = float(np.max(S0 + eps * T_vec))
        if new_max < cur_max_raw - 1e-10:
            h[i] -= eps
            h[j] += eps
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_max_raw = float(np.max(S0))
            cur_c5 = cur_max_raw * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        new_max2 = float(np.max(S0 - eps * T_vec))
        if new_max2 < cur_max_raw - 1e-10:
            h[i] += eps
            h[j] -= eps
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_max_raw = float(np.max(S0))
            cur_c5 = cur_max_raw * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def binary_sa(h, deadline, rng, n_calls=40000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cur_max_raw = float(np.max(S0))
    T_init = 0.004
    T = T_init
    T_min = 1e-7
    cooling = 0.99997
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for _ in range(n_calls):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        new_S = np.correlate(h, 1.0 - h, mode='full')
        new_max_raw = float(np.max(new_S))
        new_c5 = new_max_raw * dx
        delta = new_c5 - cur_c5
        if delta < -1e-14 or (T > T_min and rng.random() < np.exp(-delta / max(T, 1e-10))):
            cur_c5 = new_c5
            cur_max_raw = new_max_raw
            S0 = new_S
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)

def coordinate_refine(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(300):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        span = min(len(interior), 15)
        for ii in range(len(interior)):
            i = interior[ii]
            for jj in range(ii + 1, min(ii + span, len(interior))):
                j = interior[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, trial_c5 = line_search(h, d, 0.4)
                if trial_c5 < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = project_feasible(h, target)
                    c = exact_c5(h)
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def active_set_direction(h, active_lags, max_iter=3):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    G_acc = np.zeros(n)
    for r in active_lags:
        s = r - (n - 1)
        if s >= 0:
            m = n - s
            if m > 0:
                G_acc[:m] += (1.0 - h[s:s + m])
                G_acc[s:s + m] -= h[:m]
        else:
            g = -s
            m = n - g
            if m > 0:
                G_acc[g:g + m] += (1.0 - h[:m])
                G_acc[:m] -= h[g:g + m]
    G_acc -= np.mean(G_acc)
    norm = np.linalg.norm(G_acc)
    if norm < 1e-12:
        return None
    return -G_acc / norm

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = np.clip(0.5 + 0.5 * np.sin(np.arange(n0) * 2 * np.pi * 5 / n0), 0, 1)
        h0 = project_feasible(h0, n0 / 2)

    h = project_feasible(h0.copy(), n0 / 2)
    best_h = h.copy()
    best_c5 = exact_c5(h)

    # Phase 1: LSE gradient descent (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = smooth_descent(h, d1, lam_init=100.0, lam_min=20.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Guided pair moves (15%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = guided_pair_moves(best_h, d2, rng, n_trials=80000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LSE descent round 2 (15%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = smooth_descent(best_h, d3, lam_init=150.0, lam_min=30.0)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA (20%)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = binary_sa(best_h, d4, rng, n_calls=60000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Polish after SA
    rem = deadline - (deadline - d4)
    h4b, c5_4b = smooth_descent(best_h, min(d4 + budget_s * 0.03, deadline - budget_s * 0.10), lam_init=200.0, lam_min=40.0)
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()

    # Phase 5: Multi-resolution (12%)
    d5 = t0 + budget_s * 0.87
    best_n = len(best_h)
    for dn in [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try % 2 != 0 or n_try > 200:
            continue
        h_r = resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = smooth_descent(h_r, sub_dl, lam_init=80.0, lam_min=20.0)
        if c5_r < best_c5 and time.time() < d5 - 3:
            h_r2, c5_r2 = binary_sa(h_r, min(d5, time.time() + budget_s * 0.01), rng, n_calls=8000)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
            h_r, c5_r = smooth_descent(h_r, min(d5, time.time() + budget_s * 0.008), lam_init=120.0, lam_min=25.0)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Active-set directed refinement (5%)
    d6 = t0 + budget_s * 0.92
    target = len(best_h) / 2.0
    n_act = len(best_h)
    cur = best_h.copy()
    for _it in range(200):
        if time.time() > d6:
            break
        S = np.correlate(cur, 1.0 - cur, mode='full') * (2.0 / n_act)
        S_max = float(np.max(S))
        active = np.where(S > S_max - 1e-4)[0]
        if len(active) == 0:
            break
        d = active_set_direction(cur, active)
        if d is None:
            break
        max_step = 0.3
        pos = d < -1e-12
        neg = d > 1e-12
        if np.any(pos):
            max_step = min(max_step, float(np.min(cur[pos] / (-d[pos]))))
        if np.any(neg):
            max_step = min(max_step, float(np.min((1.0 - cur[neg]) / d[neg])))
        if max_step < 1e-10:
            break
        alpha, trial_c5 = line_search(cur, d, max_step)
        if trial_c5 < best_c5 - 1e-14:
            cur = cur + alpha * d
            cur = project_feasible(cur, target)
            c = exact_c5(cur)
            if c < best_c5:
                best_c5 = c
                best_h = cur.copy()
            else:
                cur = best_h.copy()
        else:
            break

    # Phase 7: Coordinate refinement (7%)
    d7 = t0 + budget_s * 0.99
    h7, c5_7 = coordinate_refine(best_h, d7)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final polish
    h_f, c5_f = smooth_descent(best_h, deadline, lam_init=300.0, lam_min=80.0)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, len(best_h)
