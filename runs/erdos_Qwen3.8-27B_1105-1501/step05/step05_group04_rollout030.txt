import numpy as np
import time
from scipy.optimize import minimize

def _h_from_a(a, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = a
    h[n-m:] = 1.0 - a[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def _full_to_a(h):
    n = len(h)
    m = n // 2
    a = 0.5 * (h[:m] + (1.0 - h[n-m:][::-1]))
    return np.clip(a, 0.0, 1.0)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _lse_fg(x, n, beta):
    a = np.clip(x, 0.0, 1.0)
    h = _h_from_a(a, n)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, 'full') * dx
    u = float(V.max())
    z = beta * (V - u)
    z = np.clip(z, -60.0, 60.0)
    ez = np.exp(z)
    esum = float(ez.sum())
    w = ez / esum
    grad_h = np.zeros(n)
    for r in range(len(V)):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        wt = w[r] * dx
        if d >= 0:
            ml = n - d
            if ml > 0:
                grad_h[:ml] += wt * (1.0 - h[d:])
                grad_h[d:] -= wt * h[:ml]
        else:
            s = -d
            ml = n - s
            if ml > 0:
                grad_h[s:] += wt * (1.0 - h[:ml])
                grad_h[:ml] -= wt * h[s:]
    m = n // 2
    grad_a = grad_h[:m] - grad_h[n-m:][::-1]
    obj = u + (1.0 / beta) * np.log(esum)
    return obj, grad_a

def _margin_fg(x, n, B, delta):
    a = np.clip(x, 0.0, 1.0)
    h = _h_from_a(a, n)
    dx = 2.0 / n
    V = np.correlate(h, 1.0 - h, 'full') * dx
    excess = np.maximum(V - B, 0.0)
    obj = float(np.sum(excess ** 2) / (2.0 * delta))
    if obj < 1e-18:
        return obj, np.zeros(len(a))
    grad_h = np.zeros(len(h))
    mask = V > B
    for r in range(len(V)):
        if not mask[r]:
            continue
        d = r - (n - 1)
        coeff = (V[r] - B) / delta * dx
        if d >= 0:
            ml = n - d
            if ml > 0:
                grad_h[:ml] += coeff * (1.0 - h[d:])
                grad_h[d:] -= coeff * h[:ml]
        else:
            s = -d
            ml = n - s
            if ml > 0:
                grad_h[s:] += coeff * (1.0 - h[:ml])
                grad_h[:ml] -= coeff * h[s:]
    m = n // 2
    grad_a = grad_h[:m] - grad_h[n-m:][::-1]
    return obj, grad_a

def _resample_sym(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    a = _full_to_a(h_new)
    return _h_from_a(a, n_new)

def _solve_lse(a, n, beta, deadline, maxiter=300):
    if len(a) == 0:
        return a, _c5(_h_from_a(a, n))
    bounds = [(0.0, 1.0)] * len(a)
    try:
        res = minimize(_lse_fg, a, args=(n, beta), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        a_new = np.clip(res.x, 0.0, 1.0)
    except Exception:
        a_new = a.copy()
    h_new = _h_from_a(a_new, n)
    return a_new, _c5(h_new)

def _solve_margin(a, n, B, delta, deadline, maxiter=300):

        return a
    bounds = [(0.0, 1.0)] * len(a)
    try:
        res = minimize(_margin_fg, a, args=(n, B, delta), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return a.copy()

def _grad_check(a, n, beta, eps=1e-6):
    _, g_analytic = _lse_fg(a, n, beta)
    g_num = np.zeros(len(a))
    for i in range(min(len(a), 5)):
        ap = a.copy(); ap[i] = min(1.0, ap[i] + eps)
        am = a.copy(); am[i] = max(0.0, am[i] - eps)
        fp, _ = _lse_fg(ap, n, beta)
        fm, _ = _lse_fg(am, n, beta)
        g_num[i] = (fp - fm) / (ap[i] - am[i] + 1e-30)
    denom = np.max(np.abs(g_num)) + 1e-15
    return float(np.max(np.abs(g_analytic[:len(g_num)] - g_num)) / denom)
