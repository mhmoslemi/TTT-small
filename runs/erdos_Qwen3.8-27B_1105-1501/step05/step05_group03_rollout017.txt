def _bit_scores(bits, n):
    """Compute all lag scores for a binary sequence stored as Python int (bit 0 = position 0)."""
    max_count = 0
    for d in range(1, n):
        mask = (1 << (n - d)) - 1
        lower = bits & mask
        upper = (bits >> d) & mask
        # forward: h[i]=1, h[i+d]=0
        fwd = (lower & ~upper & mask).bit_count()
        # backward: h[i]=0, h[i+d]=1
        bwd = (upper & ~lower & mask).bit_count()
        if fwd > max_count:
            max_count = fwd
        if bwd > max_count:
            max_count = bwd
    return max_count

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _bit_score_cached(bi, n, masks):
    max_c = 0
    for d in range(1, n):
        mask = masks[d-1]
        lower = bi & mask
        upper = (bi >> d) & mask
        fwd = (lower & (mask ^ upper)).bit_count()
        if fwd > max_c:
            max_c = fwd
        bwd = (upper & (mask ^ lower)).bit_count()
        if bwd > max_c:
            max_c = bwd
    return max_c

def _make_masks(n):
    return [(1 << (n - d)) - 1 for d in range(1, n)]

def _quasi_bits(n, alpha):
    target = n // 2
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    idx = np.argsort(vals)
    bits = np.zeros(n, dtype=int)
    bits[idx[:target]] = 1
    bi = 0
    for i in range(n):
        if bits[i]:
            bi |= (1 << i)
    return bi

def _random_bits(n, rng):
    target = n // 2
    pos = rng.permutation(n)
    bi = 0
    for i in pos[:target]:
        bi |= (1 << i)
    return bi

def _bits_to_onzeros(bi, n):
    ones = []
    zeros = []
    for i in range(n):
        if (bi >> i) & 1:
            ones.append(i)
        else:
            zeros.append(i)
    return ones, zeros

def _bit_sa(n, masks, deadline, rng, bi_init=None, T0=3.0, T1=0.15, n_max_iters=2000000):
    target = n // 2
    if bi_init is None:
        bi_init = _random_bits(n, rng)
    cur_score = _bit_score_cached(bi_init, n, masks)
    best_bi = bi_init
    best_score = cur_score
    
    ones, zeros = _bits_to_onzeros(bi_init, n)
    cool = (T1 / T0) ** (1.0 / max(n_max_iters, 1))
    T = T0
    no_improve = 0
    stall_limit = 50000
    
    for it in range(n_max_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        bi_new = (bi_init & ~(1 << i)) | (1 << j)
        new_score = _bit_score_cached(bi_new, n, masks)
        
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            delta = new_score - cur_score
            if T > 0.001 and rng.random() < np.exp(-delta / T):
                accept = True
        
        if accept:
            # Update lists
            ones[i] = ones[-1]
            ones.pop()
            ones.append(j)
            zeros[j] = zeros[-1]
            zeros.pop()
            zeros.append(i)
            bi_init = bi_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_bi = bi_init
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cool
        
        # Kick mechanism
        if no_improve > stall_limit:
            # Make several random swaps
            for _ in range(5):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                bi_init = (bi_init & ~(1 << i)) | (1 << j)
                ones[i] = ones[-1]; ones.pop(); ones.append(j)
                zeros[j] = zeros[-1]; zeros.pop(); zeros.append(i)
            cur_score = _bit_score_cached(bi_init, n, masks)
            no_improve = 0
            T = max(T, T0 * 0.3)  # Reheat
    
    return best_bi, best_score

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, n_eval_max=30):

    target = n / 2.0
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
