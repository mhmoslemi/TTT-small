def _lp_direction(h, corr, G, radius, K=20):
    """Find a descent direction using LP on active constraints."""
    n = len(h)
    idx = np.argsort(-corr)
    active = idx[:K]
    G_a = G[active]  # (K, n)
    v_a = corr[active]
    
    # Minimize t subject to:
    #   G_a[i] @ d + v_a[i] <= t  for i in active
    #   sum(d) = 0
    #   -radius <= d[j] <= radius
    
    # Variables: d[0..n-1], t (index n)
    # c: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, n] = -1.0
    b_ub = -v_a
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-radius, radius)] * n + [(None, None)]

                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

    cur_c5 = _exact_c5(h)
    best_h = h.copy()

        if time.time() > deadline:
            break
        corr = _corr_values(h)
        top_idx = np.argsort(-corr)[:n_active_target]
        top_vals = corr[top_idx]
        if np.max(top_vals) - np.min(top_vals) < 1e-6:
            break
        G = _grad_all(h)
        G_top = G[top_idx]
        # Try to find d that equalizes: G_top @ d should be inversely proportional to (top_vals - mean)
        target = np.zeros(n_active_target)
        mean_v = float(np.mean(top_vals))
        target = (mean_v - top_vals) * 0.5
        # Solve least squares: G_top @ d ≈ target, subject to sum(d)=0
        # Use ridge: (G_top.T @ G_top + lam*I + (1/n)*11^T) d = G_top.T @ target
        lam = 0.01
        A = G_top.T @ G_top + lam * np.eye(n)
        A[0] += 1.0  # regularization for sum constraint
        A[:, 0] += 1.0
        b_vec = G_top.T @ target
        b_vec[0] += float(np.sum(top_vals) - n_active_target * mean_v)  # pull sum toward 0
        d = np.linalg.solve(A, b_vec)
        d -= float(np.mean(d))  # enforce sum(d) ~ 0
        d = d / (np.max(np.abs(d)) + 1e-15) * 0.05
        t, new_c5 = _line_search(h, d, 0.1)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + t * d, n/2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Try random pairs among interior points
            interior = np.where((h > 0.05) & (h < 0.95))[0]
            if len(interior) >= 2:
                i, j = rng.choice(len(interior), 2, replace=False)
                ii, jj = interior[i], interior[j]
                d2 = np.zeros(n)
                d2[ii] = -1.0
                d2[jj] = 1.0
                t2, new_c5_2 = _line_search(h, d2, 0.1)
                if new_c5_2 < cur_c5 - 1e-12:
                    h = _project(h + t2 * d2, n/2.0)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        h0 = np.zeros(n0)
        h0[::2] = 1.0
    
    n = len(h0)
    h = _project(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Phase 1: Main LP optimization (50% time)
    d1 = t0 + budget_s * 0.50
    h1, c5_1 = _main_optimize(h, d1, rng, max_iter=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Equioscillation refinement (15% time)
    d2 = t0 + budget_s * 0.65
    h2, c5_2 = _equioscillate(best_h, d2, n_active_target=6)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Second LP pass with different params (10% time)
    d3 = t0 + budget_s * 0.75
    h3, c5_3 = _main_optimize(best_h, d3, rng, max_iter=1000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Multi-n exploration (15% time)
    d4 = t0 + budget_s * 0.90
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = len(best_h) + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _main_optimize(h_r, sub_dl, rng, max_iter=500)
        if c5_r < best_c5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r, c5_r = _equioscillate(h_r, sub_dl2, n_active_target=5)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 5: SLSQP polish (5% time)
    d5 = t0 + budget_s * 0.95
    h5, c5_5 = _slsqp_polish(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Final LP squeeze (5% time)
    h_f, c5_f = _main_optimize(best_h, deadline - 1.0, rng, max_iter=300)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
