The integral is over x such that both x and x+k are in [0,2]. For k > 0, this means x ∈ [0, 2-k]. For k < 0, x ∈ [-k, 2].

Wait, actually the problem statement says h: [0,2] → [0,1]. The overlap integral ∫ h(x)(1-h(x+k)) dx is over the domain where both h(x) and h(x+k) are defined, i.e., x ∈ [0,2] ∩ [-k, 2-k].

In the discrete version with np.correlate, the natural truncation is automatic (the correlation only includes terms where both indices are valid). So the discrete version correctly approximates the continuous one.

Let me also add one more optimization trick: after finding a good solution with SLSQP, I'll try a "rounding" step that pushes values toward 0 or 1 (binary). Sometimes the optimal discrete solution is binary or nearly binary. But I'll only accept the rounded version if it's actually better.

import numpy as np
from scipy.optimize import minimize
import time

_slq = {}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _all_grads(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for s in range(-(n - 1), n):
        row = n - 1 + s
        g = abs(s)
        if g == 0:
            G[row] = 1.0 - 2.0 * h
        else:
            m = n - g
            G[row, g:] += (1.0 - h[:m])
            G[row, :m] -= h[g:]
    return G

def _slq_obj(x):
    n = _slq['n']
    h_ref = _slq['h_ref']
    lam = _slq['lam']
    h = x[:n]
    return x[n] + lam * np.sum((h - h_ref) ** 2)

def _slq_grad(x):
    n = _slq['n']
    h_ref = _slq['h_ref']
    lam = _slq['lam']
    g = np.zeros(n + 1)
    g[:n] = 2.0 * lam * (x[:n] - h_ref)
    g[n] = 1.0
    return g

def _slq_con(x):
    n = _slq['n']
    top = _slq['top']
    h = x[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return x[n] - corr[top]

def _slq_con_jac(x):
    n = _slq['n']
    top = _slq['top']
    h = x[:n]
    K = len(top)
    J = np.zeros((K, n + 1))
    for i in range(K):
        idx = top[i]
        s = idx - (n - 1)
        g = abs(s)
        if g == 0:
            J[i, :n] = 2.0 * h - 1.0
        else:
            m = n - g
            J[i, g:] += -(1.0 - h[:m])
            J[i, :m] += h[g:]
        J[i, n] = 1.0
    return J

def _slq_eq(x):
    n = _slq['n']
    return np.sum(x[:n]) - _slq['target']

def _slq_eq_jac(x):
    n = _slq['n']
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _slsqp_step(h, deadline, lam=0.01, K=40, max_iter=150):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_ref = h.copy()
    cur_c5 = _exact_c5(h)
    corr = _corr_vals(h)
    
    # Select top K lags
    top = np.argsort(-corr)[:K]
    
    # Set up global state
    _slq['n'] = n
    _slq['h_ref'] = h_ref
    _slq['lam'] = lam
    _slq['top'] = top
    _slq['target'] = target
    
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = cur_c5 / dx + 1e-5
    
    bounds = [(0.0, 1.0)] * n + [(0.0, cur_c5 / dx + 0.5)]
    
    constraints = [
        {'type': 'ineq', 'fun': _slq_con, 'jac': _slq_con_jac},
        {'type': 'eq', 'fun': _slq_eq, 'jac': _slq_eq_jac},
    ]
    
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Multiple SLSQP attempts with decreasing lambda
    lam_seq = [lam, lam * 0.5, lam * 0.25, lam * 0.1, lam * 0.05]
    
    for lam_i in lam_seq:
        if time.time() > deadline:
            break
        _slq['lam'] = lam_i
        # Re-select active set
        _slq['top'] = np.argsort(-_corr_vals(h))[:K]
        x0[n] = _exact_c5(h) / dx + 1e-4
        
        res = minimize(_slq_obj, x0, method='SLSQP', jac=_slq_grad,
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-9})
        
        if res.success or res.fun < x0[n]:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                h = h_new.copy()
    
    return best_h, best_c5
