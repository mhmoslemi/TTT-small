def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_max_obj_grad(h, tau):
    """Compute soft-max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    # Softmax weights
    shifted = (corr - M) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    # Objective
    val = dx * (M + tau * np.log(np.sum(exp_s)))
    # Gradient using convolution
    # grad[j] = dx * tau * [sum_{m=0}^{j} (1-h[m]) * w[j-m] - sum_{m=j}^{n-1} h[m] * w[m-j]]
    # First term: convolution of (1-h) with w, take first n elements
    # Second term: correlate(h, w) reversed... 
    
    # Term 1[j] = sum_{m=0}^{j} (1-h[m]) * w[j-m]
    # This is the linear convolution of (1-h) and w, evaluated at positions 0..n-1
    # np.convolve(1-h, w, 'full') gives positions 0 to n+(2n-1)-2 = 3n-3
    # We want positions 0 to n-1
    conv1 = np.convolve(1.0 - h, w, mode='full')[:n]
    
    # Term 2[j] = sum_{k=0}^{n-1-j} h[j+k] * w[k]
    # Let B_rev = h[::-1], then Term 2[j] = sum_{k=0}^{n-1-j} B_rev[n-1-j-k] * w[k]
    # = (B_rev * w)[n-1-j] (where * is linear convolution)
    # np.convolve(B_rev, w, 'full') has length n + (2n-1) - 1 = 3n-2

    B_rev = h[::-1]
    conv2_full = np.convolve(B_rev, w, mode='full')
    # conv2_full has length n + 2n-1 - 1 = 3n-2
    # Term 2[j] = conv2_full[n-1-j]
    term2 = conv2_full[n - 1 - np.arange(n)]  # indices n-1, n-2, ..., 0
