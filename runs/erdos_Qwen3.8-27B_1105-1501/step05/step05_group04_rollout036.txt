    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    h = np.empty(n)
    m = n // 2
    if n % 2 == 0:
        h[:m] = a
        h[m:] = 1.0 - a[::-1]
    else:
        h[:m] = a
        h[m] = 0.5
        h[m+1:] = 1.0 - a[::-1]
    return h

    s = _scores(h)
    s_max = np.max(s)
