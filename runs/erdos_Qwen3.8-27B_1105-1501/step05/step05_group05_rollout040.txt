import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    """Quadratic line search for h + t*d"""
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:

            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    # Also check t=0
    f0 = float(np.max(S0)) * dx
    if opt_val < f0:
        return opt_t, opt_val
    return 0.0, f0

def _grad_matrix(h):
    """Compute gradient matrix G where row r is gradient of lag r score w.r.t. h"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.05, top_k=5, top_w=0.15):
    """Solve LP to find direction that reduces max score"""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    # Extra penalty on top-k lags
    if top_k > 0:
        top_indices = np.argsort(scores)[::-1][:top_k]
        for k in top_indices:
            b_ub[k] -= top_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.20, use_penalty=True):
    """LP-based trust region loop"""
    n = len(h)
    target = n / 2.0

        p = _lp_solve(h, scores, G, radius, penalty_lags)

                best_c5 = cur_c5

def _active_lag_gradient(h, dx, top_k=5):
    """Gradient focusing on the top-k active lags equally"""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    # Take top_k lags and average their gradients
    top_idx = np.argsort(scores)[::-1][:top_k]
    G = _grad_matrix(h)
    grad = np.mean(G[top_idx, :], axis=0)
    grad -= np.mean(grad)
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _logit_map(h, eps=1e-5):
    """Map h to logit space"""
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def _logit_unmap(v):
    """Map logit space back to [0,1]"""
    return 1.0 / (1.0 + np.exp(-v))

def _logit_project(v, target):
    """Find theta such that sum(sigmoid(v+theta)) = target, then return h"""
    n = len(v)
    lo, hi = -60.0, 60.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_candidate = _logit_unmap(v + mid)
        s = float(np.sum(h_candidate))
        if s > target:
            hi = mid
        else:
            lo = mid
    theta = (lo + hi) / 2.0
    h = _logit_unmap(v + theta)
    # Fix residual
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += res / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _logit_ea(h_init, deadline, rng, pop_size=10, max_gens=200):

        h_i = _logit_project(v_i, target)

    best_h = h_init.copy()
    best_c5 = c5_init

            break

        elite_scores = [scores[order[i]] for i in range(4)]
        
        best_score = scores[order[0]]
        best_v = pop[order[0]].copy()
        best_h_gen = _logit_project(best_v, target)
        if best_score < best_c5 - 1e-14:
            best_c5 = best_score
            best_h = best_h_gen.copy()
        
        # Generate offspring
        offspring = []
        new_scores = []
        n_off = pop_size
        for _ in range(n_off):
            # Select parent
            parent = elites[rng.integers(0, len(elites))].copy()
            # Mutate
            mutation_type = rng.integers(0, 4)
            if mutation_type == 0:
                # Gaussian mutation
                child = parent + rng.normal(0, sigma, n)
            elif mutation_type == 1:
                # Sub-vector mutation
                mlen = rng.integers(3, max(4, n // 4))
                mstart = rng.integers(0, n - mlen)
                child = parent.copy()
                child[mstart:mstart+mlen] += rng.normal(0, sigma * 1.5, mlen)
            elif mutation_type == 2:
                # Crossover with another elite
                p2 = elites[rng.integers(0, len(elites))].copy()
                cross_point = rng.integers(1, n - 1)
                child = np.concatenate([parent[:cross_point], p2[cross_point:]])
                child += rng.normal(0, sigma * 0.5, n)
            else:
                # Bit-flip style (invert a section in logit space)
                mlen = rng.integers(3, max(4, n // 5))
                mstart = rng.integers(0, n - mlen)
                child = parent.copy()
                child[mstart:mstart+mlen] = -child[mstart:mstart+mlen] + rng.normal(0, sigma*0.3, mlen)
            
            # Clip to reasonable range
            child = np.clip(child, -25, 25)
            h_child = _logit_project(child, target)
            c = _exact_c5(h_child)
            offspring.append(child)
            new_scores.append(c)
        
        # Select: keep best pop_size from elites + offspring
        all_v = elites + offspring
        all_s = elite_scores + new_scores
        order = np.argsort(all_s)
        pop = [all_v[order[i]].copy() for i in range(pop_size)]
        scores = [all_s[order[i]] for i in range(pop_size)]
        
        # Adapt sigma
        if scores[0] < best_c5 + 1e-10:
            stall += 1
            if stall > 10:
                sigma *= 0.85
                stall = 0
            elif stall > 5:
                sigma *= 0.92
        else:
            sigma = min(sigma * 1.1, 0.5)
            stall = 0
        
        if sigma < 0.003:
            break
    
    return best_h, best_c5

        return opt_t, opt_val
    return 0.0, f0

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, active_weights=None):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if active_weights is not None:
        b_ub += active_weights
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.20, use_weighted=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    while time.time() < deadline:
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        G = _grad_matrix(h)
        active_weights = None
        if use_weighted and stall > 2:
            active_weights = np.zeros(2 * n - 1)
            sorted_idx = np.argsort(scores)[::-1]
            for rank, idx in enumerate(sorted_idx[:min(20, 2 * n - 1)]):
                if scores[idx] > s_max * 0.95:
                    active_weights[idx] = 0.08 * (1.0 - 0.05 * rank)
        p = _lp_solve(h, scores, G, radius, active_weights)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.06, 0.50)
        else:
            stall += 1
            radius *= 0.72
            if stall > 25 and radius < 0.03:
                radius = max(best_radius * 0.5, 0.05)
                stall = 0
    return best_h, best_c5

def _active_gradient(h, top_k=5):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(scores)[::-1][:top_k]
    G = _grad_matrix(h)
    grad = np.mean(G[top_idx, :], axis=0)
    grad -= np.mean(grad)
    return grad

def _smooth_phase(h, deadline, top_ks=(3, 5, 8, 12, 20), lr_init=0.015):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.82
    stall = 0
    ki = 0
    while time.time() < deadline:
        if stall > 60:
            ki += 1
            if ki >= len(top_ks) * 4:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        tk = top_ks[ki % len(top_ks)]
        d = _active_gradient(h, tk)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            ki += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.30)
        else:
            stall += 1
            lr *= 0.78
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=200, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        interior = []
        for i in range(n):
            if 1e-8 < h[i] < 1.0 - 1e-8:
                interior.append(i)
        if len(interior) < 2:
            break
        if len(interior) > 50 and rng is not None:
            pairs = set()
            for _ in range(800):
                i = interior[rng.integers(0, len(interior))]
                j = interior[rng.integers(0, len(interior))]
                if i != j:
                    pairs.add((min(i, j), max(i, j)))
        else:
            pairs = [(interior[a], interior[b]) for a in range(len(interior)) for b in range(a+1, len(interior))]
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            for sign in (1, -1):
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _logit_map(h, eps=1e-5):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def _logit_unmap(v):
    v = np.clip(v, -30, 30)
    return 1.0 / (1.0 + np.exp(-v))

def _logit_project(v, target):
    n = len(v)
    lo, hi = -50.0, 50.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        h_c = _logit_unmap(v + mid)
        s = float(np.sum(h_c))
        if s > target:
            hi = mid
        else:
            lo = mid
    theta = (lo + hi) / 2.0
    h = _logit_unmap(v + theta)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += res / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _logit_ea(h_init, deadline, rng, pop_size=10, max_gens=300):
    n = len(h_init)
    target = n / 2.0
    v_init = _logit_map(h_init)
    pop = [v_init.copy()]
    h0 = _logit_project(v_init, target)
    scores = [_exact_c5(h0)]
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    for i in range(pop_size - 1):
        noise = rng.normal(0, 0.2, n)
        noise -= noise.mean()
        v_i = v_init + noise
        h_i = _logit_project(v_i, target)
        pop.append(v_i)
        scores.append(_exact_c5(h_i))
    sigma = 0.12
    stall = 0
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        order = np.argsort(scores)
        n_elite = min(4, pop_size)
        elites = [pop[order[i]].copy() for i in range(n_elite)]
        elite_s = [scores[order[i]] for i in range(n_elite)]
        if scores[order[0]] < best_c5 - 1e-14:
            best_c5 = scores[order[0]]
            best_h = _logit_project(elites[0], target).copy()
        offspring = []
        new_scores = []
        for _ in range(pop_size):
            parent = elites[rng.integers(0, n_elite)].copy()
            mtype = rng.integers(0, 4)
            if mtype == 0:
                child = parent + rng.normal(0, sigma, n)
            elif mtype == 1:
                mlen = rng.integers(3, max(4, n // 4))
                ms = rng.integers(0, n - mlen)
                child = parent.copy()
                child[ms:ms+mlen] += rng.normal(0, sigma * 1.5, mlen)
            elif mtype == 2:
                p2 = elites[rng.integers(0, n_elite)].copy()
                cp = rng.integers(1, n - 1)
                child = np.concatenate([parent[:cp], p2[cp:]])
                child += rng.normal(0, sigma * 0.5, n)
            else:
                mlen = rng.integers(3, max(4, n // 5))
                ms = rng.integers(0, n - mlen)
                child = parent.copy()
                child[ms:ms+mlen] = -child[ms:ms+mlen] + rng.normal(0, sigma * 0.3, mlen)
            child = np.clip(child, -25, 25)
            h_c = _logit_project(child, target)
            offspring.append(child)
            new_scores.append(_exact_c5(h_c))
        all_v = elites + offspring
        all_s = elite_s + new_scores
        ao = np.argsort(all_s)
        pop = [all_v[ao[i]].copy() for i in range(pop_size)]
        scores = [all_s[ao[i]] for i in range(pop_size)]
        if scores[0] < best_c5 + 1e-12:
            stall += 1
            if stall > 8:
                sigma *= 0.88
                stall = 0
        else:
            sigma = min(sigma * 1.1, 0.6)
            stall = 0
        if sigma < 0.004 or min(scores) < 0.379:
            break
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=40):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.03, 0.09)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = _project(best_h + scale * direction, target)
        rem = deadline - time.time()
        if rem < 2.0:
            break
        sub_dl = min(deadline, time.time() + max(2.0, rem * 0.04))
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.10, use_weighted=False)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _structured_kick(h, deadline, rng, n_tries=20):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        h_k = h.copy()
        op = rng.integers(0, 4)
        if op == 0:
            blen = rng.integers(3, max(4, n // 5))
            start = rng.integers(0, n - blen)
            h_k[start:start+blen] = 1.0 - h_k[start:start+blen]
        elif op == 1:
            noise = rng.normal(0, 0.06, n)
            noise -= noise.mean()
            h_k = h + noise
        elif op == 2:
            blen = rng.integers(4, max(5, n // 4))
            start = rng.integers(0, n - blen)
            h_k[start:start+blen] = h_k[start:start+blen][::-1]
        else:
            blen = rng.integers(3, max(4, n // 5))
            s1 = rng.integers(0, n - 2 * blen)
            s2 = s1 + blen + rng.integers(1, n - 2 * blen - blen + 1)
            tmp = h_k[s1:s1+blen].copy()
            h_k[s1:s1+blen] = h_k[s2:s2+blen]
            h_k[s2:s2+blen] = tmp
        h_k = _project(h_k, target)
        rem = deadline - time.time()
        if rem < 2.0:
            break
        sub_dl = min(deadline, time.time() + max(1.5, rem * 0.03))
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.08, use_weighted=False)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _coord_descent(h, deadline, max_sweeps=50):
    """Coordinate-wise optimization: for each i, find best h[i] on a grid."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            h_old = h[i]
            lo = max(0.0, h_old - 0.3)
            hi = min(1.0, h_old + 0.3)
            if hi - lo < 1e-6:
                continue
            n_grid = 15
            x_vals = np.linspace(lo, hi, n_grid)
            best_x = h_old
            best_local = 1e10
            for x in x_vals:
                h[i] = x
                h = _project(h, target)
                c = _exact_c5(h)
                if c < best_local:
                    best_local = c
                    best_x = x
            if best_x != h_old:
                h[i] = best_x
                h = _project(h, target)
                if best_local < best_c5:
                    best_c5 = best_local
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        h0 = np.full(n0, 0.5)
        h0[:n0 // 4] = 1.0
        h0[-(n0 // 4):] = 1.0

    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LP with weighted penalties (22%)
    d1 = t0 + budget_s * 0.22
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.22, use_weighted=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smooth active-lag descent (12%)
    d2 = t0 + budget_s * 0.34
    h2, c5_2 = _smooth_phase(best_h, d2, top_ks=(3, 5, 8, 12, 15, 20), lr_init=0.012)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair refinement (14%)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _pair_refine(best_h, d3, max_passes=200, rng=rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Logit EA for diversification (14%)
    d4 = t0 + budget_s * 0.62
    h4, c5_4 = _logit_ea(best_h, d4, rng, pop_size=8, max_gens=250)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (14%)
    d5 = t0 + budget_s * 0.76
    best_n = len(best_h)
    tried = set()
    offsets = sorted(range(-24, 25, 2), key=lambda x: (abs(x), x))
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 80 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        rem = d5 - time.time()
        if rem < 3.0:
            break
        sub_dl = min(d5, time.time() + max(2.0, rem * 0.018))
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.14)
        if c5_r < best_c5 + 0.001 and time.time() < d5 - 4:
            sub_dl2 = min(d5, time.time() + 4.0)
            h_r, c5_r = _pair_refine(h_r, sub_dl2, max_passes=30, rng=rng)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Kick and re-optimize (8%)
    d6 = t0 + budget_s * 0.84
    h6, c5_6 = _kick_reopt(best_h, d6, rng, n_kicks=30)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Structured perturbation (6%)
    d7 = t0 + budget_s * 0.90
    h7, c5_7 = _structured_kick(best_h, d7, rng, n_tries=15)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final LP polish (6%)
    d8 = t0 + budget_s * 0.96
    h8, c5_8 = _lp_loop(best_h, d8, radius_init=0.06, use_weighted=True)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final pair polish with remaining time
    rem = deadline - time.time()
    if rem > 3.0:
        h9, c5_9 = _pair_refine(best_h, deadline - 0.5, max_passes=60, rng=rng)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
