            h_new, c5_new = _pair_line_search(h, i, j, 2.0 / n, trust=0.3)
            if c5_new < best_c5 - 1e-13:
                h = _project(h_new, target)
                c5_actual = _exact_c5(h)
                if c5_actual < best_c5:
                    best_c5 = c5_actual
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _multi_resolution(best_h, deadline, rng, rng_seed=999):
    n_best = len(best_h)
    best_c5 = _exact_c5(best_h)
    candidates = []
    for dn in [-20, -16, -12, -8, -4, 4, 8, 12, 16, 20, 24, 28]:
        n_try = n_best + dn
        if 60 <= n_try <= 200:
            candidates.append(n_try)
    candidates = sorted(set(candidates), key=lambda x: abs(x - n_best))
    per_candidate = max(3.0, (deadline - time.time()) / max(len(candidates), 1))
    for n_try in candidates:
        if time.time() > deadline:
            break
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r > best_c5 + 0.002:
            continue
        sub_dl = min(deadline, time.time() + per_candidate * 0.6)
        h_r, c5_r = _sa_phase(h_r, sub_dl, rng, T_init=0.004, T_floor=0.0002, n_phases=2)
        if c5_r < best_c5 + 0.0005 and time.time() < deadline - 5:
            sub_dl2 = min(deadline - 2, time.time() + per_candidate * 0.35)
            h_r, c5_r2 = _pair_refine(h_r, sub_dl2, rng, max_passes=30)
            c5_r = c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=20):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    per_kick = max(1.5, (deadline - time.time()) / max(n_kicks, 1))
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.03, 0.10)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-15:
            continue
        direction /= norm
        h_k = _project(best_h + scale * direction, target)
        c5_k = _exact_c5(h_k)
        sub_dl = min(deadline, time.time() + per_kick * 0.5)
        h_r, c5_r = _sa_phase(h_k, sub_dl, rng, T_init=0.003, T_floor=0.0002, n_phases=2)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    try:
        hv = initial_h_values
        if hv is not None and len(hv) > 10:
            h0 = np.array(hv, dtype=float)
        else:
            h0 = None
    except NameError:
        h0 = None
    if h0 is None:
        h0 = np.full(100, 0.5)
    n0 = len(h0)
    n0 = n0 if n0 % 2 == 0 else n0 + 1
    h0 = np.resize(h0, n0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    d1 = t0 + budget_s * 0.06
    h1, c51 = _greedy_descend(h, d1, rng)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    d2 = t0 + budget_s * 0.38
    h2, c52 = _sa_phase(best_h, d2, rng, T_init=0.007, T_floor=0.0001, n_phases=5)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2.copy()

    d3 = t0 + budget_s * 0.58
    h3, c53 = _informed_sa(best_h, d3, rng, T_init=0.004)
    if c53 < best_c5:
        best_c5 = c53
        best_h = h3.copy()

    d4 = t0 + budget_s * 0.76
    h4, c54 = _multi_resolution(best_h, d4, rng)
    if _exact_c5(h4) < best_c5 + 0.001:
        best_h_candidate = h4
        best_c5_check = _exact_c5(h4)
        if best_c5_check < best_c5:
            best_c5 = best_c5_check
            best_h = best_h_candidate.copy()

    d5 = t0 + budget_s * 0.88
    h5, c55 = _kick_reopt(best_h, d5, rng, n_kicks=15)
    if c55 < best_c5:
        best_c5 = c55
        best_h = h5.copy()

    d6 = t0 + budget_s * 0.96
    h6, c56 = _pair_refine(best_h, d6, rng, max_passes=60)
    if c56 < best_c5:
        best_c5 = c56
        best_h = h6.copy()

    if time.time() < t0 + budget_s - 2.0:
        d7 = t0 + budget_s - 1.0
        h7, c57 = _sa_phase(best_h, d7, rng, T_init=0.002, T_floor=0.00005, n_phases=2)
        if c57 < best_c5:
            best_c5 = c57
            best_h = h7.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
