    if target <= 1e-14:

    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(scores))
    z = np.exp((scores - M) / tau)
    sz = float(np.sum(z))
    val = M + tau * float(np.log(sz))
    w = z / sz
    g = np.zeros(n)
    for d in range(-(n-1), n):
        idx = n - 1 + d
        w_r = w[idx]
        if w_r < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w_r * (1.0 - h[:m])
                g[:m] -= w_r * h[d:]
        else:
            gs = -d
            m = n - gs
            if m > 0:
                g[:m] += w_r * (1.0 - h[gs:])
                g[gs:] -= w_r * h[:m]
    g *= dx
    return val, g

def _proj_grad_step(h, tau, dx, target, step_size, deadline, max_iter=500):
    """Projected gradient descent on LSE surrogate."""
    n = len(h)
    cur_val, cur_grad = _lse_grad(h, tau, dx)
    x = h.copy()
    best_x = h.copy()
    best_c5 = _exact_c5(h)
    for it in range(max_iter):

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 1e-14:
        return np.zeros(n)
    if target >= n - 1e-14:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            gs = -d
            m = n - gs
            if m > 0:
                G[r, :m] += (1.0 - h[gs:])
                G[r, gs:] -= h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    # Find feasible interval
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Dense grid + refinement
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([float(np.max(S0 + t * T - t*t * Q)) * dx for t in ts])
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    # Refine around best
    if best_idx > 0 and best_idx < n_grid - 1:
        lo_t = ts[best_idx - 1]
        hi_t = ts[best_idx + 1]
    else:
        lo_t = t_lo
        hi_t = t_hi
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    for _ in range(40):
        if hi_t - lo_t < 1e-13:
            break
        if f1 < f2:
            hi_t = c2
            c2 = c1
            f2 = f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        else:
            lo_t = c1
            c1 = c2
            f1 = f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    opt_t = 0.5 * (lo_t + hi_t)
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * Q)) * dx
    # Compare with grid best
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lse_grad(h, tau, dx):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(scores))
    z = np.exp((scores - M) / tau)
    sz = float(np.sum(z))
    val = M + tau * float(np.log(sz))
    w = z / sz
    g = np.zeros(n)
    for d in range(-(n-1), n):
        idx = n - 1 + d
        w_r = w[idx]
        if w_r < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w_r * (1.0 - h[:m])
                g[:m] -= w_r * h[d:]
        else:
            gs = -d
            m = n - gs
            if m > 0:
                g[:m] += w_r * (1.0 - h[gs:])
                g[gs:] -= w_r * h[:m]
    g *= dx
    return val, g

def _pg_optimize(h, target, deadline, tau_init=0.1, tau_min=0.004, iters_per_tau=300):
    """Projected gradient with tau annealing."""
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    best_x = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    tau = tau_init
    while tau >= tau_min and time.time() < deadline:
        step = 0.15
        for it in range(iters_per_tau):
            if time.time() > deadline:
                break
            _, g = _lse_grad(x, tau, dx)
            x_new = x - step * g
            x_proj = _project(x_new, target)
            new_c5 = _exact_c5(x_proj)
            if new_c5 < cur_c5 - 1e-14:
                x = x_proj
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_x = x.copy()
                step = min(step * 1.03, 0.4)
            else:
                step *= 0.6
                if step < 1e-9:
                    break
        tau *= 0.75
    return best_x, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hopping(h, target, deadline, rng, n_perturb=30, scale_init=0.15):
    """Basin hopping: perturb + local optimize."""
    n = len(h)
    x = h.copy()
    best_x = h.copy()
    best_c5 = _exact_c5(h)
    scale = scale_init
    for k in range(n_perturb):
        if time.time() > deadline:
            break
        # Perturb
        x_pert = x + rng.normal(0, scale, n)
        x_pert = _project(x_pert, target)
        # Local optimize (short LP + PG)
        sub_dl = min(deadline, time.time() + max(2, (deadline - time.time()) / max(n_perturb - k - 1, 1)))
        x_new, c5_new = _lp_optimize(x_pert, sub_dl, radius_init=0.08)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_x = x_new.copy()
            scale = max(0.03, scale * 0.8)
        else:
            scale = min(0.3, scale * 1.15)
        # Also try PG on the perturbed point
        if time.time() < deadline - 5:
            x_pg, c5_pg = _pg_optimize(x_pert, target, min(deadline, time.time() + 5), tau_init=0.05, tau_min=0.01, iters_per_tau=100)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_x = x_pg.copy()
                scale = max(0.03, scale * 0.8)
    return best_x, best_c5
