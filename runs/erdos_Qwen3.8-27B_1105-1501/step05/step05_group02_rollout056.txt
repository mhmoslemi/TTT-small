def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n

def _project(h, target):
    """Project h to [0,1]^n with sum = target."""
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    # Final fix
    res2 = target - float(np.sum(x))
    if abs(res2) > 1e-15:
        # Distribute among non-edge values
        mid_mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        cnt = int(np.sum(mid_mask))
        if cnt > 0:
            x[mid_mask] += res2 / cnt
    return x

def _pair_line_min(h, p, q, n_eval=30):
    """Minimize C5 along the line h[p]+t, h[q]-t. Returns (best_t, best_c5, new_h)."""
    n = len(h)
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-13:
        return 0.0, _exact_c5(h), h
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    h_c1 = h.copy()
    h_c1[p] += c1
    h_c1[q] -= c1
    f1 = _exact_c5(h_c1)
    h_c2 = h.copy()
    h_c2[p] += c2
    h_c2[q] -= c2
    f2 = _exact_c5(h_c2)
    for _ in range(n_eval):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            h_c1[p] = h[p] + c1
            h_c1[q] = h[q] - c1
            f1 = _exact_c5(h_c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            h_c2[p] = h[p] + c2
            h_c2[q] = h[q] - c2
            f2 = _exact_c5(h_c2)
    t_opt = (a + b) / 2.0
    h_opt = h.copy()
    h_opt[p] += t_opt
    h_opt[q] -= t_opt
    f_opt = _exact_c5(h_opt)
    return t_opt, f_opt, h_opt

    n = len(h)
    dx = 2.0 / n
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-13:
        return 0.0, float(np.max(scores)), h

    # Index in scores: r = 0 to 2n-2 corresponds to lag d = r - (n-1)
    g = np.zeros(2*n-1)
    for d in range(1, n):

        # ∂/∂h[x] = (1-h[x-d])*[x>=d] - h[x+d]*[x<=n-1-d]
        dp_pos = (1.0 - h[p-d]) if p >= d else 0.0
        dp_neg = -h[p+d] if p <= n-1-d else 0.0
        dq_pos = (1.0 - h[q-d]) if q >= d else 0.0
        dq_neg = -h[q+d] if q <= n-1-d else 0.0
        g[n-1+d] = dx * ((dp_pos + dp_neg) - (dq_pos + dq_neg))
        
        # corr_{n-1-d} = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # ∂/∂h[x] = (1-h[x+d])*[x<=n-1-d] - h[x-d]*[x>=d]
        dp_pos2 = (1.0 - h[p+d]) if p <= n-1-d else 0.0
        dp_neg2 = -h[p-d] if p >= d else 0.0
        dq_pos2 = (1.0 - h[q+d]) if q <= n-1-d else 0.0
        dq_neg2 = -h[q-d] if q >= d else 0.0
        g[n-1-d] = dx * ((dp_pos2 + dp_neg2) - (dq_pos2 + dq_neg2))

    # The linear coefficient is: dx * (1-2h[p]) - dx * (1-2h[q]) = dx * 2*(h[q]-h[p])
    # Plus quadratic term: -2t²*dx (this makes the d=0 component concave)
    g[n-1] = dx * 2.0 * (h[q] - h[p])

    # the k=n-1 component is actually scores[n-1] + t*g[n-1] - 2*t²*dx

    # quadratic correction and just do max_k (scores[k] + t*g[k]).

    # Breakpoints are where two lines intersect: scores[k1] + t*g[k1] = scores[k2] + t*g[k2]
    # t = (scores[k2] - scores[k1]) / (g[k1] - g[k2])
    
    # But there could be O(n²) breakpoints. Instead, let's just use sampling + golden section.
    # Since the function is convex (piecewise linear), golden section works.
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(scores + c1 * g))
    f2 = float(np.max(scores + c2 * g))
    for _ in range(n_eval):

            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(scores + c1 * g))

            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(scores + c2 * g))

    h_new = h.copy()
    h_new[p] += t_opt
    h_new[q] -= t_opt
    f_exact = _exact_c5(h_new)
    
    # Also check t=0
    f0 = float(np.max(scores))
    
    if f_exact < f0:
        return t_opt, f_exact, h_new
    return 0.0, f0, h

For n=128:

That's a LOT of iterations! The key question is: will the pair optimization converge? Each pair move is a small improvement, and after many passes, the improvements will become very small. But the advantage over LP is that we can do SO many more iterations.

Let me also think about a smarter pair selection strategy. Instead of purely random pairs, I can:
1. Identify the active lags (those with scores near the max)
2. For each active lag d, the score is corr_d * dx
3. To reduce corr_d = sum h[i](1-h[i+d]), I want to either:
   - Decrease h[i] where h[i+d] is large (move mass from high h[i] to low h[i+d])
   - Or increase h[i] where h[i+d] is small

This gives a directed pair selection: source = position with high h contributing to active lag, destination = neighboring position (at distance d) with low h.

Let me implement a "guided" pair selection:
1. Find top K active lags
2. For each active lag d, for each position i where h[i] is high and h[i+d] is low:
   - Consider pair (i+d, i) [move mass from i to i+d]
   - This would decrease the contribution of position i to lag d
3. Add random pairs for exploration

Actually, let me simplify. The most impactful moves are those that reduce the maximum score. The gradient of the max score w.r.t. h[p] tells us which direction to move each coordinate. If the gradient is positive at p, we want to decrease h[p]; if negative, increase h[p]. So good pairs are (p, q) where grad[p] > 0 and grad[q] < 0 (move mass from p to q).

The subgradient of max_k(score_k) at h is: grad = sum_r w_r * ∂score_r/∂h, where w_r are weights on the active constraints (e.g., w_r = 1/max_score for active r, or softmax weights).

So my algorithm should be:
1. Compute scores
2. Compute subgradient (using softmax or uniform weights on active lags)
3. Select pairs (p, q) where grad[p] > 0 and grad[q] < 0
4. For each such pair, do 1D minimization
5. Accept improvements

This is essentially a coordinate descent / block coordinate method with 2-coordinate blocks, which is known to be effective for problems of this type.

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(h, target):
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    res2 = target - float(np.sum(x))
    if abs(res2) > 1e-15:
        mid_mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        cnt = int(np.sum(mid_mask))
        if cnt > 0:
            x[mid_mask] += res2 / cnt
    return x

def _pair_coefficients(h, p, q):
    """Compute g vector: change in scores per unit t for h[p]+=t, h[q]-=t."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)
    nm1 = n - 1
    for d in range(1, n):
        # corr[nm1+d] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # d/dh[x] = (1-h[x-d])*[x>=d] - h[x+d]*[x <= n-1-d]
        vp = ((1.0 - h[p-d]) if p >= d else 0.0) - (h[p+d] if p <= nm1-d else 0.0)
        vq = ((1.0 - h[q-d]) if q >= d else 0.0) - (h[q+d] if q <= nm1-d else 0.0)
        g[nm1+d] = dx * (vp - vq)
        
        # corr[nm1-d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # d/dh[x] = (1-h[x+d])*[x <= n-1-d] - h[x-d]*[x>=d]
        vp2 = ((1.0 - h[p+d]) if p <= nm1-d else 0.0) - (h[p-d] if p >= d else 0.0)
        vq2 = ((1.0 - h[q+d]) if q <= nm1-d else 0.0) - (h[q-d] if q >= d else 0.0)
        g[nm1-d] = dx * (vp2 - vq2)
    
    # d=0 (index nm1): scores[nm1] change = dx*(1-2h[p])*t - dx*(1-2h[q])*(-t) - 2t²*dx
    # Linear part: dx*2*(h[q]-h[p])
    g[nm1] = dx * 2.0 * (h[q] - h[p])
    return g

def _pair_minimize(h, p, q, scores):
    """Minimize C5 along pair direction. Returns (new_h, new_c5) or (h, orig_c5)."""
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-12:
        return h, float(np.max(scores))
    
    g = _pair_coefficients(h, p, q)
    
    # Golden section on f(t) = max_k(scores[k] + t*g[k])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(scores + c1 * g))
    f2 = float(np.max(scores + c2 * g))
    for _ in range(20):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(scores + c1 * g))
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(scores + c2 * g))
    
    t_opt = 0.5 * (a + b)
    h_new = h.copy()
    h_new[p] += t_opt
    h_new[q] -= t_opt
    c5_new = _exact_c5(h_new)
    c5_orig = float(np.max(scores))
    if c5_new < c5_orig - 1e-14:
        return h_new, c5_new
    return h, c5_orig

def _compute_subgrad(h, scores, gamma):
    """Compute weighted subgradient."""
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    # Softmax weights on scores
    max_s = float(np.max(scores))
    w = np.exp(gamma * (scores - max_s))
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    
    # Gradient: sum_r w[r] * d(cor_r)/dh * dx
    # For each coordinate x:
    #   d(cor[nm1])/dh[x] = 1-2h[x]  (d=0)
    #   d(cor[nm1+d])/dh[x] = (1-h[x-d])*[x>=d] - h[x+d]*[x<=n-1-d]  for d=1..n-1
    #   d(cor[nm1-d])/dh[x] = (1-h[x+d])*[x<=n-1-d] - h[x-d]*[x>=d]  for d=1..n-1
    
    grad = np.zeros(n)
    # d=0 contribution
    w0 = w[nm1]
    grad += w0 * dx * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        wp = w[nm1 + d]  # weight for corr[nm1+d] = sum h[i+d]*(1-h[i])
        wn = w[nm1 - d]  # weight for corr[nm1-d] = sum h[i]*(1-h[i+d])
        
        # From corr[nm1+d]: d/dh[x] = (1-h[x-d])*[x>=d] - h[x+d]*[x<=n-1-d]
        # For x >= d: (1-h[x-d])
        # For x <= n-1-d: -h[x+d]
        if d < n:
            # x from d to n-1: contribute (1-h[x-d])
            if n - 1 - d >= 0:
                range1 = slice(d, n)
                range2 = slice(0, n - d)
                gp = wp * dx * (1.0 - h[range2])  # h[x-d] for x in d..n-1
                gn = -wn * dx * h[range1]  # h[x+d] for x in 0..n-1-d
                grad[range1] += gp
                grad[range2] += gn
            
            # From corr[nm1-d]: d/dh[x] = (1-h[x+d])*[x<=n-1-d] - h[x-d]*[x>=d]
            # For x <= n-1-d: (1-h[x+d])
            # For x >= d: -h[x-d]
            if n - 1 - d >= 0:
                range3 = slice(0, n - d)  # x from 0 to n-1-d
                range4 = slice(d, n)  # x from d to n-1
                gp2 = wn * dx * (1.0 - h[n - d + range3])  # h[x+d] for x in 0..n-1-d
                gn2 = -wp * dx * h[range3]  # h[x-d] for x in d..n-1
                grad[range3] += gp2
                grad[range4] += gn2
    
    return grad

Similarly for negative lag:
score_{n-1-d} = dx * sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
∂/∂h[x] = dx * [(1-h[x+d]) * 1_{x <= n-1-d} - h[x-d] * 1_{x >= d}]

And for d=0:
score_{n-1} = dx * sum_x h[x]*(1-h[x])
∂/∂h[x] = dx * (1-2h[x])

So the full subgradient (using weights w_r) is:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    res2 = target - float(np.sum(x))
    if abs(res2) > 1e-15:
        mid_mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        cnt = int(np.sum(mid_mask))
        if cnt > 0:
            x[mid_mask] += res2 / cnt
    return x

def _pair_coefficients(h, p, q):
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    g = np.zeros(2 * n - 1)
    for d in range(1, n):
        vp = ((1.0 - h[p - d]) if p >= d else 0.0) - (h[p + d] if p <= nm1 - d else 0.0)
        vq = ((1.0 - h[q - d]) if q >= d else 0.0) - (h[q + d] if q <= nm1 - d else 0.0)
        g[nm1 + d] = dx * (vp - vq)
        vp2 = ((1.0 - h[p + d]) if p <= nm1 - d else 0.0) - (h[p - d] if p >= d else 0.0)
        vq2 = ((1.0 - h[q + d]) if q <= nm1 - d else 0.0) - (h[q - d] if q >= d else 0.0)
        g[nm1 - d] = dx * (vp2 - vq2)
    g[nm1] = dx * 2.0 * (h[q] - h[p])
    return g

def _pair_minimize(h, p, q, scores):
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-12:
        return None, None
    g = _pair_coefficients(h, p, q)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(scores + c1 * g))
    f2 = float(np.max(scores + c2 * g))
    for _ in range(22):
        if b - a < 1e-15:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(scores + c1 * g))
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(scores + c2 * g))
    t_opt = 0.5 * (a + b)
    if abs(t_opt) < 1e-13:
        return None, None
    h_new = h.copy()
    h_new[p] += t_opt
    h_new[q] -= t_opt
    c5_new = _exact_c5(h_new)
    c5_orig = float(np.max(scores))
    if c5_new < c5_orig - 1e-14:
        return h_new, c5_new
    return None, None

def _compute_subgrad_simple(h, scores, gamma):
    """Compute subgradient using direct loop over coordinates."""
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    max_s = float(np.max(scores))
    w = np.exp(gamma * (scores - max_s))
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    # d=0: grad[x] += w[nm1]*dx*(1-2h[x])
    grad += w[nm1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[nm1 + d]
        wn = w[nm1 - d]
        # corr[nm1+d]: d/dh[x] = (1-h[x-d])*[x>=d] - h[x+d]*[x<=nm1-d]
        # corr[nm1-d]: d/dh[x] = (1-h[x+d])*[x<=nm1-d] - h[x-d]*[x>=d]
        # For x in [0, d-1]: only x<=nm1-d part
        #   from corr[nm1-d]: (1-h[x+d])
        # For x in [d, nm1-d]: both parts
        #   from corr[nm1+d]: (1-h[x-d])
        #   from corr[nm1-d]: -h[x-d]
        #   total: (1-h[x-d])*(wp) + (-h[x-d])*(wn) = (wp+wn)*(1-h[x-d])... 
        #   wait: wp*(1-h[x-d]) - wn*h[x-d]
        # For x in [nm1-d+1, n-1]: only x>=d part
        #   from corr[nm1+d]: -h[x+d]... but x+d > n-1, so this is 0
        #   Actually x <= nm1-d means x+d <= n-1. If x > nm1-d, x+d > n-1, out of range.
        #   from corr[nm1+d]: (1-h[x-d])*[x>=d] - h[x+d]*[x<=nm1-d]
        #   If x > nm1-d: first term (1-h[x-d]), second term 0
        #   from corr[nm1-d]: (1-h[x+d])*[x<=nm1-d] - h[x-d]*[x>=d]
        #   If x > nm1-d: first term 0, second term -h[x-d]
        #   total: wp*(1-h[x-d]) - wn*h[x-d] (same formula!)
        # Actually for ALL x >= d: total = wp*(1-h[x-d]) - wn*h[x-d]
        # And for x >= d, x-d ranges over [0, n-1-d]
        # For x < d: total = wn*(1-h[x+d])
        # x+d ranges over [d, n-1]
        
        # Vectorized:
        # Part 1: x >= d => x-d in [0, n-1-d]
        idx1 = np.arange(d, n)
        src1 = np.arange(0, n - d)
        contrib1 = wp * (1.0 - h[src1]) - wn * h[src1]
        grad[idx1] += dx * contrib1
        
        # Part 2: x < d => x in [0, d-1], x+d in [d, n-1]
        if d > 0 and d < n:
            idx2 = np.arange(0, d)
            src2 = np.arange(d, n)
            contrib2 = wn * (1.0 - h[src2])
            grad[idx2] += dx * contrib2
    return grad

def _weighted_dir_step(h, scores, gamma, target):
    """Try a step along the negative subgradient direction."""
    n = len(h)
    grad = _compute_subgrad_simple(h, scores, gamma)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    p = -grad / gn
    # Find max alpha such that h + alpha*p stays in [0,1]
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-14)
    cur_c5 = float(np.max(scores))
    best_a = 0.0
    best_c5 = cur_c5
    for _ in range(20):
        for a_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a = min(a_try * alpha_max, alpha_max)
            if a < 1e-14:
                continue
            h_try = _project(h + a * p, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_a = a
    if best_a > 0:
        h_new = _project(h + best_a * p, target)
        return h_new, best_c5
    return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_init(n, rng):
    positions = rng.permutation(n)
    h = np.zeros(n)
    h[positions[:n // 2]] = 1.0
    return h

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _pair_pass(h, scores, n, rng, deadline, max_pairs=5000):
    """One pass of pair transfer optimization."""
    target = n / 2.0
    cur_c5 = float(np.max(scores))
    cur_h = h
    count = 0
    for _ in range(max_pairs):
        if time.time() > deadline:
            break
        p = rng.integers(n)
        q = rng.integers(n)
        if p == q:
            continue
        h_new, c5_new = _pair_minimize(cur_h, p, q, scores)
        if h_new is not None:
            cur_h = h_new
            cur_c5 = c5_new
            scores = _scores_vec(cur_h)
            count += 1
            if count % 50 == 0 and time.time() > deadline:
                break
    return cur_h, cur_c5

def _pair_pass_guided(h, scores, n, rng, deadline, max_pairs=3000, gamma=30.0):
    """Guided pair selection using subgradient."""
    target = n / 2.0
    cur_c5 = float(np.max(scores))
    cur_h = h
    count = 0
    
    grad = _compute_subgrad_simple(cur_h, scores, gamma)
    # Good pairs: dest p where grad[p]<0, source q where grad[q]>0
    neg_idx = np.where(grad < 0)[0]
    pos_idx = np.where(grad > 0)[0]
    
    for _ in range(max_pairs):
        if time.time() > deadline:
            break
        if len(neg_idx) > 0 and len(pos_idx) > 0 and rng.random() < 0.7:
            p = neg_idx[rng.integers(len(neg_idx))]
            q = pos_idx[rng.integers(len(pos_idx))]
        else:
            p = rng.integers(n)
            q = rng.integers(n)
            if p == q:
                continue
        h_new, c5_new = _pair_minimize(cur_h, p, q, scores)
        if h_new is not None:
            cur_h = h_new
            cur_c5 = c5_new
            scores = _scores_vec(cur_h)
            count += 1
            if count % 30 == 0:
                grad = _compute_subgrad_simple(cur_h, scores, gamma)
                neg_idx = np.where(grad < 0)[0]
                pos_idx = np.where(grad > 0)[0]
    return cur_h, cur_c5

def _binary_exchange(h, n, rng, deadline, max_iters=10000):
    """Binary exchange search."""
    if n % 2 != 0:
        return None, None
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    if int(np.sum(h_bin)) != target:
        h_bin = _binary_init(n, rng)
    cur_h = h_bin.copy()
    cur_c5 = _exact_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    ones = np.where(cur_h == 1.0)[0].tolist()
    zeros = np.where(cur_h == 0.0)[0].tolist()
    T = 0.003
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(max_iters, 1))
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = _exact_c5(cur_h)
        if new_c5 < cur_c5 or (new_c5 - cur_c5 < T and rng.random() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            ones = np.where(cur_h == 1.0)[0].tolist()
            zeros = np.where(cur_h == 0.0)[0].tolist()
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        T = max(T * cool, T_min)
    # Convert to continuous and refine
    h_cont = best_h.copy()
    return h_cont, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20.0
    rng = np.random.default_rng(seed)
    
    # Initialize
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = _project(h, target)
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    best_n = n
    
    scores = _scores_vec(h)
    
    # Phase 1: Guided pair optimization (main workhorse)
    # Use increasing gamma to focus on active constraints
    phase1_deadline = t0 + budget_s * 0.45
    gamma = 20.0
    no_improve = 0
    while time.time() < phase1_deadline:
        h, c5 = _pair_pass_guided(h, scores, n, rng, min(deadline, time.time() + budget_s * 0.05),
                                   max_pairs=3000, gamma=gamma)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            no_improve = 0
            gamma = min(gamma * 1.05, 500.0)
        else:
            no_improve += 1
            if no_improve > 5:
                gamma = min(gamma * 1.3, 800.0)
            if no_improve > 20:
                break
        scores = _scores_vec(h)
    
    # Phase 2: Random pair search (broader exploration)
    phase2_deadline = t0 + budget_s * 0.60
    while time.time() < phase2_deadline:
        h, c5 = _pair_pass(h, scores, n, rng, min(deadline, time.time() + budget_s * 0.04),
                           max_pairs=5000)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        scores = _scores_vec(h)
    
    # Phase 3: Weighted gradient steps
    phase3_deadline = t0 + budget_s * 0.70
    for g_try in [10, 30, 80, 150, 300, 500]:
        if time.time() > phase3_deadline:
            break
        h_new, c5_new = _weighted_dir_step(best_h, _scores_vec(best_h), g_try, best_n / 2.0)
        if h_new is not None and c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = len(h_new)
    
    # Phase 4: Try different resolutions
    phase4_deadline = t0 + budget_s * 0.82
    for n_try in [n, n + 32, n + 64, n - 32, n + 96, n + 128]:
        if time.time() > phase4_deadline:
            break
        if n_try < 32 or n_try > 320:
            continue
        if n_try % 2 != 0:
            continue
        sub_dl = min(phase4_deadline, time.time() + budget_s * 0.03)
        h_res = _resample(best_h, n_try)
        scores_res = _scores_vec(h_res)
        # Quick pair optimization at new resolution
        h_try, c5_try = _pair_pass_guided(h_res, scores_res, n_try, rng, sub_dl,
                                           max_pairs=2000, gamma=40.0)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
        # Also try at same n but from resampled (different interpolation)
    
    # Phase 5: Binary exchange for diverse seeds
    if time.time() < t0 + budget_s * 0.90:
        sub_dl = min(deadline, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_exchange(best_h, best_n, rng, sub_dl, max_iters=15000)
        if h_bin is not None and c5_bin < best_c5 - 1e-14:
            # Refine the binary solution with pair optimization
            h_ref = h_bin.copy()
            scores_ref = _scores_vec(h_ref)
            h_try, c5_try = _pair_pass_guided(h_ref, scores_ref, best_n, rng,
                                               min(deadline, time.time() + budget_s * 0.02),
                                               max_pairs=2000, gamma=40.0)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = best_n
    
    # Phase 6: Quasi-periodic seeds
    if time.time() < t0 + budget_s * 0.95:
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
        for a in alphas:
            if time.time() > t0 + budget_s * 0.95:
                break
            h_qp = _quasi_periodic(best_n, a)
            scores_qp = _scores_vec(h_qp)
            c5_qp = float(np.max(scores_qp))
            if c5_qp < best_c5 + 0.01:
                sub_dl = min(deadline, time.time() + budget_s * 0.015)
                h_try, c5_try = _pair_pass_guided(h_qp, scores_qp, best_n, rng, sub_dl,
                                                   max_pairs=1500, gamma=40.0)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = best_n
    
    # Phase 7: Final polishing - intensive pair search on best solution
    sub_dl = deadline
    h_final = best_h.copy()
    scores_final = _scores_vec(h_final)
    while time.time() < sub_dl:
        h_final, c5_f = _pair_pass_guided(h_final, scores_final, best_n, rng,
                                           min(sub_dl, time.time() + budget_s * 0.02),
                                           max_pairs=3000, gamma=80.0)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_final.copy()
        else:
            # Try random pairs for diversification
            h_final, c5_f = _pair_pass(h_final, scores_final, best_n, rng,
                                        min(sub_dl, time.time() + budget_s * 0.015),
                                        max_pairs=3000)
            if c5_f < best_c5 - 1e-14:
                best_c5 = c5_f
                best_h = h_final.copy()
            scores_final = _scores_vec(h_final)
    
    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Projection somehow degraded - use previous
        best_h = best_h.copy()
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
