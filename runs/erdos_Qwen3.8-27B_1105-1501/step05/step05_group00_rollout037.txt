def _slsqp_epigraph(h0, deadline, lambda_reg=0.01, n_active=30):
    """SLSQP with epigraph formulation."""
    n = len(h0)
    target = n / 2.0
    
    # Initial t value
    corr0 = np.correlate(h0, 1.0 - h0, mode='full') * (2.0/n)
    t0 = float(np.max(corr0)) + 1e-4
    
    x0 = np.concatenate([h0, [t0]])
    
    def objective(x):
        h = x[:n]
        t = x[n]
        return t + lambda_reg * np.sum((h - h0)**2)
    
    def objective_grad(x):
        h = x[:n]
        t = x[n]
        grad = np.zeros(n + 1)
        grad[:n] = 2 * lambda_reg * (h - h0)
        grad[n] = 1.0
        return grad
    
    constraints = []
    # Equality: sum(h) = target
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - target,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    # Inequality constraints: t - corr_k >= 0
    # Only include a subset for efficiency
    # Start with all, but SLSQP will handle it
    corr = np.correlate(h0, 1.0 - h0, mode='full') * (2.0/n)
    # Sort by value, take top n_active
    idx = np.argsort(-corr)[:n_active]
    
    for r in idx:
        k = r - (n - 1)
        def con_fun(x, k=k):
            h = x[:n]
            t = x[n]
            # Compute corr for shift k
            if k >= 0:
                m = n - k
                val = np.sum(h[k:]* (1.0 - h[:m]))
            else:
                g = -k
                m = n - g
                val = np.sum(h[:m] * (1.0 - h[g:]))
            return t - val * (2.0/n)
        
        def con_jac(x, k=k):
            h = x[:n]
            t = x[n]
            g = _grad_row(h, k)
            jac = -g.copy()
            jac[n] = 1.0
            return jac
        
        constraints.append({'type': 'ineq', 'fun': con_fun, 'jac': con_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.1)]
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:

            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h0.copy(), _exact_c5(h0)

def _binary_sa(h_init, deadline, rng, max_iter=50000):
    """Binary SA with swap moves."""
    n = len(h_init)
    target = n // 2

    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.01
    T_min = 1e-5
    cooling = 0.99985
    
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if T < T_min:
            break
        
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c5

            break
        improved = False
        # Try moving mass from position i to position j
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                # Direction: decrease h[i], increase h[j]
                if h[i] <= 1e-6 or h[j] >= 1 - 1e-6:
                    continue
                # Find max step
                max_t = min(h[i], 1.0 - h[j])
                # Try a few step sizes
                for frac in [0.5, 1.0]:
                    t = max_t * frac
                    h_try = h.copy()
                    h_try[i] -= t
                    h_try[j] += t
                    h_try = _project(h_try, target)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h.copy()
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        idx1 = np.arange(k, n)
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, m)
        g[idx2] -= h[idx2 + k]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        idx1 = np.arange(0, m)
        g[idx1] += (1.0 - h[idx1 + ga])
        idx2 = np.arange(ga, n)
        g[idx2] -= h[idx2 - ga]
    return g * dx


def _smooth_max_gradient(h, alpha, K=40):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(K) / K
    else:
        w /= w_sum
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _grad_row(h, k)
    return grad, corr


def _line_search_armijo(h, direction, target, f0, step0=0.3, max_ls=30):
    n = len(h)
    best_t = 0.0
    best_f = f0
    t = step0
    c1 = 1e-4
    grad_dir = np.dot(direction, -direction)
    for _ in range(max_ls):
        h_try = h + t * direction
        h_try = _project(h_try, target)
        f_try = _exact_c5(h_try)
        if f_try < best_f:
            best_f = f_try
            best_t = t
        if f_try <= f0 + c1 * t * grad_dir:
            break
        t *= 0.6
        if t < 1e-10:
            break
    return best_t, best_f


def _smooth_max_descent(h, deadline, alpha_init=20.0, alpha_mult=1.5,
                        iters_per_stage=40, max_stages=20, K=40, step0=0.25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_init
    step = step0
    stall = 0

    for stage in range(max_stages):
        for _ in range(iters_per_stage):
            if time.time() > deadline:
                return best_h, best_c5

            grad, corr = _smooth_max_gradient(h, alpha, K)
            # Project to sum-zero
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                return best_h, best_c5

            direction = -grad / gn
            f0 = _exact_c5(h)
            best_t, best_f_ls = _line_search_armijo(h, direction, target, f0, step)

            if best_t > 1e-12 and best_f_ls < best_c5 - 1e-13:
                h = h + best_t * direction
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                # Adapt step
                if best_t > 0.5 * step:
                    step = min(step * 1.1, 1.0)
                elif best_t < 0.1 * step:
                    step = max(step * 0.65, 0.01)
            else:
                stall += 1
                step *= 0.7
                if stall > 30:
                    break

        alpha *= alpha_mult
        if alpha > 1000:
            alpha = 1000.0
        if stall > 30:
            step = max(step * 0.5, 0.005)
            stall = 0

    return best_h, best_c5


def _slsqp_epigraph(h, deadline, lambda_reg=0.005, n_active=35, max_calls=5):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for call in range(max_calls):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        t0 = float(np.max(corr)) + 0.001
        x0 = np.concatenate([h, [t0]])

        # Select active constraints
        top_idx = np.argsort(-corr)[:n_active]

        obj_h0 = h.copy()

        def _obj(x):
            return x[n] + lambda_reg * np.dot(x[:n] - obj_h0, x[:n] - obj_h0)

        def _obj_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 2.0 * lambda_reg * (x[:n] - obj_h0)
            g[n] = 1.0
            return g

        constraints = [{'type': 'eq',
                        'fun': lambda x, t=target: np.sum(x[:n]) - t,
                        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]

        for r in top_idx:
            k = int(r) - (n - 1)

            def _cfun(x, k=k):
                hv = x[:n]
                if k >= 0:
                    m = n - k
                    val = np.dot(hv[k:], 1.0 - hv[:m])
                else:
                    ga = -k
                    m = n - ga
                    val = np.dot(hv[:m], 1.0 - hv[ga:])
                return x[n] - val * dx

            def _cjac(x, k=k):
                g = _grad_row(x[:n], k)
                j = -g.copy()
                j[n] = 1.0
                return j

            constraints.append({'type': 'ineq', 'fun': _cfun, 'jac': _cjac})

        bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.05)]

        try:
            res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-12})
            if res.x is not None:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    break
        except Exception:
            break

    return best_h, best_c5


def _binary_sa(h_init, deadline, rng, max_iters=80000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    T = 0.012
    T_min = 5e-6
    cooling = 0.99992
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])

    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quad_line_search(h, d, dx, target, max_t=0.5):
    """Exact quadratic line search for direction d (unit L2)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C_arr = np.correlate(d, d, mode='full')

    # Determine valid t range from box constraints
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = 0.0
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
        t_lo = 0.0  # We're going in -gradient direction, t should be positive
    t_hi = min(t_hi, max_t)
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)

    # Cubic in t: f(t) = max_r (S0[r] + t*T_arr[r] - t^2*C_arr[r]) * dx
    # Grid search + golden section
    n_grid = 50
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T_arr[None, :] - (grid[:, None] ** 2) * C_arr[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]

    # Golden section refine
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
    f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    for _ in range(50):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_arr - opt_t * opt_t * C_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _active_descent_precise(h, deadline, K=25, max_iters=200):
    """Precise active-set descent using quadratic model line search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        # Active set: top K constraints
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        # Exponential weights
        alpha = 1.0 / (n * 0.001)
        w = np.exp(alpha * (top_vals - M))
        w /= np.sum(w)

        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _grad_row(h, k)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break

        direction = -grad / gn
        t, f_try = _quad_line_search(h, direction, dx, target, max_t=step)
        if t < 1e-12:
            step *= 0.7
            continue
        h_new = h + t * direction
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.8 * step:
                step = min(step * 1.05, 1.5)
            else:
                step = max(step * 0.9, 0.01)
        else:
            step *= 0.75
            if step < 0.005:
                break

    return best_h, best_c5


def _perturbation_search(h, deadline, rng, n_trials=200):
    """Try random perturbations and optimize each briefly."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.05

    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        # Quick descent
        h_try = h_pert.copy()
        for _ in range(15):
            if time.time() > deadline:
                break
            grad, _ = _smooth_max_gradient(h_try, 50.0, K=25)
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-12:
                break
            direction = -grad / gn
            t, f_try = _quad_line_search(h_try, direction, 2.0 / n, target, 0.3)
            if t < 1e-12:
                break
            h_try = h_try + t * direction
            h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.2, scale * 1.1)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * x / n0)
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Smooth max gradient descent (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _smooth_max_descent(h, min(d1, time.time() + budget_s * 0.18),
                                   alpha_init=15.0, alpha_mult=1.4,
                                   iters_per_stage=50, max_stages=25, K=35, step0=0.2)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP epigraph (25% time)
    d2 = t0 + budget_s * 0.45
    lam_vals = [0.002, 0.01, 0.05]
    for lam in lam_vals:
        if time.time() > d2:
            break
        h_s, c5_s = _slsqp_epigraph(best_h, min(d2, time.time() + budget_s * 0.06),
                                    lambda_reg=lam, n_active=30, max_calls=4)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            # Follow up with active descent
            h_a, c5_a = _active_descent_precise(best_h, min(d2, time.time() + budget_s * 0.03),
                                                K=20, max_iters=100)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Phase 3: Active descent precise (15% time)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _active_descent_precise(best_h, d3, K=20, max_iters=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA (15% time)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = _binary_sa(best_h, min(d4, time.time() + budget_s * 0.12), rng, max_iters=60000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Polish binary with active descent
    if time.time() < d4:
        h4b, c5_4b = _active_descent_precise(best_h, d4, K=15, max_iters=100)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-resolution (20% time)
    d5 = t0 + budget_s * 0.92
    best_n = len(best_h)
    offsets = [-4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18, -24, 24]
    tried_n = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 256 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_r, c5_r = _active_descent_precise(h_r, sub_dl, K=20, max_iters=50)
        if time.time() < d5 - 5:
            h_r2, c5_r2 = _slsqp_epigraph(h_r, min(d5, time.time() + budget_s * 0.015),
                                          lambda_reg=0.01, n_active=25, max_calls=2)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Perturbation search (5% time)
    d6 = t0 + budget_s * 0.96
    h6, c5_6 = _perturbation_search(best_h, d6, rng, n_trials=150)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final polish
    h_f, c5_f = _active_descent_precise(best_h, deadline - 1.0, K=15, max_iters=50)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
