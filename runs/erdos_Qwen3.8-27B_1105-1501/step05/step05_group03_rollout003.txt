    coeffs_init = np.linalg.lstsq(B, v_centered, rcond=None)[0]
    bounds = [(-0.7, 0.7)] * (2 * K)
    init_c5 = _exact_c5(h_init)

    def objective(coeffs):
        raw = 0.5 + B @ coeffs
        hh = _project(raw, target)
        return _exact_c5(hh)
    try:
        result = differential_evolution(
            objective, bounds, x0=coeffs_init,
            popsize=popsize, maxiter=maxiter, tol=1e-8,
            mutation=(0.4, 0.9), recombination=0.7,
            seed=123, polish=False, init='latinhypercube'
        )
        if result.fun < init_c5:
            h_best = _project(0.5 + B @ result.x, target)
            return h_best, _exact_c5(h_best)
    except Exception:
        pass
    return h_init.copy(), init_c5


def _pair_polish(h, deadline, n_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    target = n / 2.0
    rng = np.random.default_rng(777)
    step_size = max(2, n // 16)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step_size):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step_size, min(j + step_size * 4, n), max(1, step_size)):
                if k >= n:
                    break
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(12):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def _coord_polish(h, deadline, n_cycles=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(30):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5


def _smoothed_hill_climbing(h, deadline, beta_start=10, beta_end=200, n_steps=200, step_init=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        # Compute gradient direction for weighted sum of overlaps
        G = np.zeros(n)
        for sl in range(len(scores)):
            if w[sl] < 1e-6:
                continue
            d = sl - (n - 1)
            if d >= 0:
                m = n - d
                G[:m] += w[sl] * (1.0 - h[d:])
                G[d:] += w[sl] * (-h[:m])
            else:
                g = -d
                m = n - g
                G[:m] += w[sl] * (1.0 - h[g:])
                G[g:] += w[sl] * (-h[:m])
        G -= G.mean()
        G_norm = np.linalg.norm(G)
        if G_norm < 1e-15:
            break
        G /= G_norm
        v = 0.85 * v - step * G
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.02, 0.10)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    return best_h, best_c5


def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0 < 0.5).astype(float)
    h = _project(h, n / 2.0)
    h += rng.normal(0, 0.02, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    deadline = t0 + budget_s

    # Get initial values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP local optimization from parent
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_local(h_start, dl1, radius_init=0.15, max_iters=5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothed hill climbing
    dl2 = t0 + budget_s * 0.20
    h2, c5_2 = _smoothed_hill_climbing(best_h, dl2, beta_start=12, beta_end=300, n_steps=300, step_init=0.035)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Follow-up LP
    if time.time() < t0 + budget_s * 0.25:
        dl2b = t0 + budget_s * 0.25
        h2b, c5_2b = _lp_local(best_h, dl2b, radius_init=0.08, max_iters=3000)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Multi-resolution binary SA + polish
    dl3 = t0 + budget_s * 0.45
    resolutions = [64, 80, 96, 112, 128, 160, 192]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113,
              0.5, 0.6180339887, 0.3819660113]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if best_c5 < 0.3806:
            break
        alpha = alphas[ri % len(alphas)]
        # Try from quasi-periodic init
        h_init = _quasi_init(n_try, alpha, rng)
        sub_dl = min(dl3, time.time() + budget_s * 0.02)
        h_sa, c5_sa = _binary_sa(n_try, sub_dl, rng, n_iters=25000, alpha_init=alpha)
        # Polish the SA result
        sub_dl2 = min(dl3, time.time() + budget_s * 0.012)
        h_pol, c5_pol = _lp_local(h_sa, sub_dl2, radius_init=0.10, max_iters=2000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_try

    # Phase 4: DE on Fourier space for best resolution
    if time.time() < t0 + budget_s * 0.55:
        dl4 = t0 + budget_s * 0.55
        h_de, c5_de = _de_fourier(best_h, dl4, K=12, popsize=25, maxiter=60)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
        # Polish DE result
        if time.time() < t0 + budget_s * 0.58:
            dl4b = t0 + budget_s * 0.58
            h4b, c5_4b = _lp_local(best_h, dl4b, radius_init=0.06, max_iters=2000)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: More binary SA at different resolutions with different alphas
    dl5 = t0 + budget_s * 0.65
    for n_try in [best_n, best_n // 2 * int(best_n / max(1, best_n // 2)) + 16]:
        if time.time() > dl5:
            break
        if n_try < 16 or n_try > 256:
            continue
        for alpha in [0.618, 0.414, 0.354]:
            if time.time() > dl5:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.015)
            h_sa, c5_sa = _binary_sa(n_try, sub_dl, rng, n_iters=30000, T_init=0.03, alpha_init=alpha)
            sub_dl2 = min(dl5, time.time() + budget_s * 0.01)
            h_pol, c5_pol = _lp_local(h_sa, sub_dl2, radius_init=0.08, max_iters=1500)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_try

    # Phase 6: Perturbation restarts with LP
    dl6 = t0 + budget_s * 0.78
    scale = 0.03
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.01)
        h_try, c5_try = _lp_local(h_pert, sub_dl, radius_init=0.05, max_iters=800)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 7: Smoothed hill climbing again
    if time.time() < t0 + budget_s * 0.84:
        dl7 = t0 + budget_s * 0.84
        h7, c5_7 = _smoothed_hill_climbing(best_h, dl7, beta_start=50, beta_end=500, n_steps=200, step_init=0.025)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
        if time.time() < t0 + budget_s * 0.87:
            dl7b = t0 + budget_s * 0.87
            h7b, c5_7b = _lp_local(best_h, dl7b, radius_init=0.04, max_iters=2000)
            if c5_7b < best_c5:
                best_c5 = c5_7b
                best_h = h7b.copy()

    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.92:
        dl8 = t0 + budget_s * 0.92
        hp, c5p = _pair_polish(best_h, dl8, n_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 9: Coordinate descent
    if time.time() < t0 + budget_s * 0.96:
        dl9 = t0 + budget_s * 0.96
        hc, c5c = _coord_polish(best_h, dl9, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 10: Final fine LP
    if time.time() < deadline - 2.0:
        dl10 = deadline - 1.0
        h_fin, c5_fin = _lp_local(best_h, dl10, radius_init=0.01, max_iters=3000, radius_min=0.0005)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    # Ensure feasibility
    if final_c5 >= best_c5 + 1e-10:
        # Projection didn't worsen things significantly; use projected version
        best_c5 = final_c5

    return best_h, best_c5, best_n
