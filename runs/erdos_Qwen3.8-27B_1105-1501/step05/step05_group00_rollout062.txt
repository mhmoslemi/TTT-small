def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    """Exact line search along direction d from h."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find safe interval
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _active_lags(corr, k=8):
    """Return indices of top-k lag values."""
    return np.argsort(-corr)[:k]

def _lp_direction_active(h, corr, active_idx, radius=0.1):
    """LP direction using only active lags."""
    n = len(h)
    # Build gradient rows only for active lags
    rows = []
    b_vals = []
    M = float(np.max(corr))
    for r in active_idx:
        d = r - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += (1.0 - h[d:])
                grad[d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += -h[g:]
                grad[g:] += (1.0 - h[:m])
        rows.append(grad)
        b_vals.append(M - corr[r])
    
    A_ub = np.array(rows)  # shape (k, n)
    b_ub = np.array(b_vals)
    
    # Add the "decrease" variable
    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -u = maximize u (the decrease)
    
    A_full = np.zeros((len(active_idx), n + 1))
    A_full[:, :n] = A_ub
    A_full[:, n] = -1.0  # grad·p - u <= slack → grad·p + (-1)*u <= slack
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((0, None))  # u >= 0
    
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[n] > 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

Actually, for a fixed h, the delta for a swap (i, j) can be computed as:
δC(r) = f_i(r) + g_j(r)

where f_i(r) is the effect of changing h[i] from 1 to 0, and g_j(r) is the effect of changing h[j] from 0 to 1.

For f_i (h[i]: 1→0):
- For lag d (r = n-1+d):
  - If i+d in [0,n-1]: subtract (1-h[i+d])
  - If i-d in [0,n-1]: add h[i-d]

For g_j (h[j]: 0→1):
- For lag d (r = n-1+d):
  - If j+d in [0,n-1]: add (1-h[j+d])
  - If j-d in [0,n-1]: subtract h[j-d]

I can precompute these as vectors for all positions!

For each position i, define:
- F[i, r] = effect of h[i]: 1→0 on lag r
  - F[i, n-1+d] += -(1-h[i+d]) if 0 ≤ i+d < n
  - F[i, n-1+d] += h[i-d] if 0 ≤ i-d < n

- G[i, r] = effect of h[i]: 0→1 on lag r
  - G[i, n-1+d] += (1-h[i+d]) if 0 ≤ i+d < n
  - G[i, n-1+d] += -h[i-d] if 0 ≤ i-d < n

Note: G[i] = -F[i] (since going 0→1 is the reverse of 1→0)!

Wait, let me verify. F[i] is the change when h[i] goes from 1 to 0.
G[i] is the change when h[i] goes from 0 to 1.
These should be negatives of each other: G[i] = -F[i].

Yes! Because the correlation is bilinear in h, so changing h[i] by Δ changes corr by Δ * (partial corr / partial h[i]). Going from 1 to 0 is Δ = -1, going from 0 to 1 is Δ = +1. So G[i] = -F[i].

Therefore, for a swap (i: 1→0, j: 0→1):
ΔC = F[i] - F[j] (since G[j] = -F[j])

And the new correlation is: C' = C + F[i] - F[j]
New max = max(C + F[i] - F[j])

This is a vector operation! O(n) per swap, but vectorized.

Even better: for a fixed i, we can compute the vector V_i = C + F[i] once, and then for each j, the new max is max(V_i - F[j]).

So the algorithm for a fixed i:
1. Compute V_i = C + F[i] (O(n) once)
2. For each j in zeros: compute max(V_i - F[j]) and check if < cur_max

Step 2 can be partially vectorized if we precompute F for all j.

def _swap_delta(h, corr, i, j):
    """Compute new max correlation after swapping h[i]=1→0, h[j]=0→1."""
    n = len(h)
    m = 2 * n - 1
    # Effect of h[i]: 1→0
    # For each lag r, d = r-(n-1):
    #   delta_i[r] = -(1-h[i+d])*[0<=i+d<n] + h[i-d]*[0<=i-d<n]
    # Effect of h[j]: 0→1
    #   delta_j[r] = (1-h[j+d])*[0<=j+d<n] - h[j-d]*[0<=j-d<n]
    
    d = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For position i:
    ip = i + d  # i+d for each lag
    im = i - d  # i-d for each lag
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)
    delta_i = np.zeros(m)
    delta_i[valid_p] = -(1.0 - h[ip[valid_p]])
    delta_i[valid_m] += h[im[valid_m]]
    
    # For position j:
    jp = j + d
    jm = j - d
    valid_pj = (jp >= 0) & (jp < n)
    valid_mj = (jm >= 0) & (jm < n)
    delta_j = np.zeros(m)
    delta_j[valid_pj] = (1.0 - h[jp[valid_pj]])
    delta_j[valid_mj] -= h[jm[valid_mj]]
    
    new_corr = corr + delta_i + delta_j
    return float(np.max(new_corr))

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx

    if f1 < opt_val:

    if f2 < opt_val:

def _compute_swap_deltas(h, i, j, d_range, n):
    m = 2 * n - 1
    ip = i + d_range
    im = i - d_range
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    di = np.zeros(m)
    di[vp] = -(1.0 - h[ip[vp]])
    di[vm] += h[im[vm]]
    jp = j + d_range
    jm = j - d_range
    vjp = (jp >= 0) & (jp < n)
    vjm = (jm >= 0) & (jm < n)
    dj = np.zeros(m)
    dj[vjp] = (1.0 - h[jp[vjp]])
    dj[vjm] -= h[jm[vjm]]
    return di + dj

def _binary_ls(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))

    best_corr = corr.copy()
    best_max = cur_max
    dx = 2.0 / n
    d_range = np.arange(-(n - 1), n)

    def do_swap(i, j):
        h[i] = 0.0
        h[j] = 1.0
        delta = _compute_swap_deltas(h, i, j, d_range, n)
        # delta was computed with h already changed, need to undo and recompute
        # Actually let's just recompute corr from scratch for correctness
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        return corr_new, float(np.max(corr_new))

def _lp_direction_active(h, corr, active_idx, radius):
    n = len(h)
    M = float(np.max(corr))
    rows = []
    b_vals = []
    for r in active_idx:
        d = int(r) - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += (1.0 - h[d:])
                grad[d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += -h[g:]
                grad[g:] += (1.0 - h[:m])
        rows.append(grad)
        b_vals.append(M - corr[r])
    k = len(active_idx)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = np.array(rows)
    A_ub[:, n] = 1.0
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[n] > 1e-8:
            return res.x[:n]
    except Exception:
        pass
    return None

def _pair_refine(h_init, deadline):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _make_start(n, rng):
    h = np.zeros(n)
    pos = 0

        if rng.random() < 0.5:

        pos += bl
    return _project(h, n / 2.0)
