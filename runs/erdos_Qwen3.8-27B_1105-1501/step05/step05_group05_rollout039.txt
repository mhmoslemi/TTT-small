def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):

def _grad_for_lag(h, d):

        if m > 0:

def _grad_all(h):

                G[r, gg:] += -h[:m]

def _line_search(h, d, dx, trust_radius):
    """Exact line search along direction d from h"""
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Bound t for feasibility
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:

def _fw_min_oracle(g, n):

    Solution: set v_i=1 for the n/2 indices with smallest g_i"""

    else:
        target = n / 2.0
    
    if n % 2 == 0:
        idx = np.argsort(g)[:target]
        v = np.zeros(n)

        return v
    else:

        k = n // 2
        idx = np.argsort(g)
        v = np.zeros(n)

        v[idx[k]] = 0.5

    n = len(h)
    v = _fw_min_oracle(g, n)
    d = v - h
    if np.max(np.abs(d)) < 1e-13:
        return h.copy(), _exact_c5(h)
    alpha, new_c5 = _line_search(h, d, dx, 1.0)

def _pnorm_grad(h, dx, p=20):
    """Gradient of the p-norm approximation of the max"""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    # p-norm: (sum S_d^p)^(1/p)
    # Gradient: (1/p) * (sum S^p)^(-1/p+1) * p * S^(p-1) * grad_S
    # = (sum S^p)^(-1/p+1) * S^(p-1) * grad_S
    S_p = float(np.sum(S**p))
    if S_p < 1e-30:
        return np.zeros(n)
    weight = S**(p-1) / (S_p ** (1.0/p - 1.0))
    weight = weight / float(np.sum(weight))  # normalize
    G = _grad_all(h)
    grad = weight @ G
    grad -= np.mean(grad)  # zero-mean for constraint
    return grad

def _softmax_grad(h, dx, beta=100):
    """Gradient using softmax-smoothed max"""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(S))
    v = beta * (S - s_max)
    np.clip(v, -500, 500, out=v)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    G = _grad_all(h)
    grad = w @ G
    grad -= np.mean(grad)
    return grad

def _active_grad(h, tol=1e-4):
    """Gradient from active set (scores within tol of max)"""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    s_max = float(np.max(S))
    active = np.where(S >= s_max - tol)[0]
    G = _grad_all(h)
    grad = G[active].mean(axis=0)
    grad -= np.mean(grad)
    return grad

    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_dir(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _softmax_dir(h, beta):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    s_max = float(np.max(S))
    v = beta * (S - s_max)
    np.clip(v, -400, 400, out=v)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    G = _grad_all(h)
    g = w @ G
    g -= np.mean(g)
    return g

def _active_dir(h, tol):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    s_max = float(np.max(S))
    active = np.where(S >= s_max - tol)[0]
    G = _grad_all(h)
    g = G[active].mean(axis=0)
    g -= np.mean(g)
    return g

def _pair_refine(h, deadline, rng, n_pairs=300, max_passes=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    passes = 0
    while time.time() < deadline:
        passes += 1
        if passes > max_passes:
            break
        improved = False
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search(h, d, dx, 0.45)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved and passes > 3:
            break
    return best_h, best_c5

def _focused_pair(h, deadline, rng):
    """Pair refinement guided by gradient: prioritize pairs with large gradient difference"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    iterations = 0
    while time.time() < deadline and iterations < 3000:
        iterations += 1
        S = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(S))
        active = np.where(S >= s_max - 2e-4)[0]
        G = _grad_all(h)
        g = G[active].mean(axis=0)
        g -= np.mean(g)
        # Score each position by gradient value
        # Best pair: move from low g to high g (i.e., increase h where g is low, decrease where g is high)
        # Actually: direction d has d[i]=1, d[j]=-1. Change in score ~ alpha*(G_d[i] - G_d[j])
        # We want G_d[i] - G_d[j] < 0 for the active d, so we want g[i] < g[j]
        # So: increase h at position i with small g, decrease h at position j with large g
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) < 2:
            break
        # Pick top and bottom gradient positions among interior
        g_int = g[interior]
        idx_sorted = interior[np.argsort(g_int)]
        # Try pairs from bottom (low g) to top (high g)
        found = False
        n_try = min(80, len(interior) * (len(interior) - 1) // 2)
        for _ in range(n_try):
            if time.time() > deadline:
                break
            # Random selection biased toward extreme gradients
            pi = int(rng.integers(0, len(idx_sorted)))
            pj = int(rng.integers(0, len(idx_sorted)))
            if pi == pj:
                continue
            i_pos = idx_sorted[pi]
            j_pos = idx_sorted[pj]
            if g[i_pos] >= g[j_pos]:
                i_pos, j_pos = j_pos, i_pos
            d = np.zeros(n)
            d[i_pos] = 1.0
            d[j_pos] = -1.0
            alpha, c5_try = _line_search(h, d, dx, 0.45)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    found = True
        if not found and iterations > 20:
            # Fall back to random pairs
            for _ in range(50):
                if time.time() > deadline:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and 1e-8 < h[i] < 1 - 1e-8 and 1e-8 < h[j] < 1 - 1e-8:
                    d = np.zeros(n)
                    d[i] = 1.0
                    d[j] = -1.0
                    alpha, c5_try = _line_search(h, d, dx, 0.45)
                    if c5_try < best_c5 - 1e-14:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            found = True
            if not found:
                break
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        S = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_all(h)
        p = _lp_dir(h, S, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.55)
        else:
            stall += 1
            radius *= 0.73
            if stall > 30 and radius < 0.03:
                radius = max(radius_init * 0.4, 0.05)
                stall = 0
    return best_h, best_c5

def _grad_loop(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    method_i = 0
    while time.time() < deadline:
        if stall > 50:
            break
        method_i += 1
        if method_i % 3 == 1:
            d = _active_dir(h, 1e-4)
        elif method_i % 3 == 2:
            d = _softmax_dir(h, 80)
        else:
            d = _active_dir(h, 5e-4)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            stall += 1
            continue
        d = d / dn
        alpha, new_c5 = _line_search(h, d, dx, 0.25)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 10 == 0:
                kick = rng.normal(0, 0.008, n)
                kick -= kick.mean()
                h = _project(h + kick, target)
                cur_c5 = _exact_c5(h)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kick_reopt(h, deadline, rng, n_kicks=15):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.07)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = _project(best_h + scale * direction, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.12)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError, TypeError):
        n = 122
        h0 = np.zeros(n)
        h0[::2] = 1.0
        h0 = _project(h0, n / 2.0)

    n = len(h0)
    h = _project(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LP optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.22)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient optimization (15%)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _grad_loop(best_h, d2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Focused pair refinement (20%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _focused_pair(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP again (10%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Random pair refinement (10%)
    d5 = t0 + budget_s * 0.75
    h5, c5_5 = _pair_refine(best_h, d5, rng, n_pairs=200, max_passes=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (15%)
    d6 = t0 + budget_s * 0.90
    base_n = len(best_h)
    tried = set()
    for dn in range(-12, 13, 2):
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 60 or n_try > 180 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.018)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        if c5_r < best_c5 + 0.0004 and time.time() < d6 - 3:
            sub_dl2 = min(d6, time.time() + 3.0)
            h_r, c5_r = _focused_pair(h_r, sub_dl2, rng)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Kick and reoptimize (5%)
    d7 = t0 + budget_s * 0.95
    h7, c5_7 = _kick_reopt(best_h, d7, rng, n_kicks=10)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final polish (5%)
    d8 = t0 + budget_s - 2.0
    if time.time() < d8:
        h8, c5_8 = _focused_pair(best_h, d8, rng)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
