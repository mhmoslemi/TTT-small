def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

**Key insight**: The problem is related to finding binary sequences with minimum maximum autocorrelation (or cross-correlation). This is a well-studied problem in coding theory (Golay sequences, Barker sequences, etc.).

For our problem, we want to minimize max_k (S_k - A_k) where S_k is the number of 1s in the first n-k positions and A_k is the autocorrelation at lag k.

Since S_k is determined by the arrangement (not just the count), we need to place the 1s strategically.

**Practical approach**: Let me focus on an efficient local search that:
1. Uses the binary domain directly
2. Allows block moves of varying sizes
3. Uses a "conflict reduction" heuristic: at each step, find the move that most reduces the maximum conflict

For the "conflict reduction" heuristic, I can:
1. Compute all corr_k values
2. Find the lag k* with the maximum correlation
3. Find a pair (i, j) where moving a 1 from i to j (i.e., h[i] was 1, h[j] was 0, and after: h[i]=0, h[j]=1) would reduce corr_{k*} the most
4. Accept if it also doesn't increase other lags too much (or if it reduces the overall max)

This is a greedy approach that focuses on the bottleneck lag.

def _compute_contrib(h, n):
    """For each position p, compute the vector of contributions to each lag."""
    # contrib[p, r] where r = n-1+k (index in full correlation output)
    # For lag k: contribution of position p to corr_k is:
    #   if p <= n-1-k: (1-h[p+k]) [p is the "first" element]
    #   if p >= k: h[p-k] [p is the "second" element, i.e., p-k is the first]
    contrib = np.zeros((n, 2*n-1))
    for p in range(n):
        for k in range(1, n):
            # p contributes to corr_k as first element if p <= n-1-k
            if p <= n-1-k:
                contrib[p, n-1+k] += (1.0 - h[p+k])
            # p contributes to corr_k as second element if p >= k
            if p >= k:
                contrib[p, n-1+k] += h[p-k]
            # Same for negative lag -k (which is at index n-1-k in full output)

            # p contributes as second element if p+k < n: h[p+k]
            if p+k < n:
                contrib[p, n-1-k] += h[p+k]
    return contrib

Now for the continuous relaxation: after finding a good binary solution, I can try to improve by allowing fractional values. The idea is:
1. Start from the binary solution
2. Use a subgradient method: at each step, compute the subgradient of the max function (which involves the active lags), take a step in the negative subgradient direction, and project back to the feasible set.

The subgradient of f(h) = max_k c_k(h) with respect to h is:
∂f/∂h[p] = (1/|A|) * sum_{k in A} ∂c_k/∂h[p] for any subgradient, where A is the set of active lags.

A common choice is to use the average over active lags, or the maximum weight (uniform over all lags that are within epsilon of the max).

    for step in range(10000):
        if time.time() > deadline:
            break
        cur_c = np.correlate(cur, 1.0 - cur, mode='full')
        max_val = np.max(cur_c)
        # Active lags: those within 0.5 of max
        active = cur_c >= max_val - 0.5
        n_active = np.sum(active)
        
        # Subgradient: average of gradients of active lags
        grad = np.zeros(n)
        k_vals = np.arange(-(n-1), n)
        for idx, k in enumerate(k_vals):
            if not active[idx]:
                continue
            # ∂c_k/∂h[p] for all p
            # Same formula as before but only for this k
            for p in range(n):

                if 0 <= a < n:
                    grad[p] += (1.0 - cur[a])
                b = p + k
                if 0 <= b < n:
                    grad[p] -= cur[b]
        
        grad /= n_active

        grad -= np.mean(grad)

            if cur_c5 < best_c5:

                best_h = cur.copy()
            lr = min(lr * 1.02, 0.2)
        else:
            lr *= 0.9

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax / logsumexp with temperature tau
    max_c = np.max(corr)
    exp_corr = np.exp((corr - max_c) / tau)
    lse = max_c + tau * np.log(np.sum(exp_corr))
    return lse * dx

def _smooth_gradient(h, tau):

    weights = exp_corr / np.sum(exp_corr)
    
    # Gradient of logsumexp(corr/tau) w.r.t. h
    # = sum_k weights[k] * (1/tau) * d(corr_k)/dh
    # = (1/tau) * sum_k weights[k] * d(corr_k)/dh

            b = p + k

    for _ in range(80):

            lo = mid
        else:
            hi = mid

                best_c5 = cur_c5
                best_h = cur.copy()
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.85
            if lr < 0.001:
                break
    return best_h, best_c5

def _lp_refine(h, deadline, radius_init=0.10):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        n_k = 2 * n - 1
        grad = _compute_grad(cur, n)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((n_k, n + 1))
        A_ub[:, :n] = grad
        A_ub[:, n] = -1.0
        b_ub = -corr
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-radius, -cur[j])
            hi = min(radius, 1.0 - cur[j])
            bounds.append((lo, hi))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                continue
            d = res.x[:n]
            if np.max(np.abs(d)) < 1e-13:
                stall += 1
                radius *= 0.7
                if radius < 1e-6:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = radius
        t_lo = 0.0
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - cur[pos]) / d[pos])))
        if np.any(neg):
            t_lo = max(t_lo, float(np.max(cur[neg] / d[neg])))
        t_lo = max(t_lo, 0.0)
        if t_hi - t_lo < 1e-14:
            break
        S0 = np.correlate(cur, 1.0 - cur, mode='full')
        T = np.correlate(d, 1.0 - cur, mode='full') - np.correlate(cur, d, mode='full')
        C = np.correlate(d, d, mode='full')
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = t_hi - gr * (t_hi - t_lo)
        c2 = t_lo + gr * (t_hi - t_lo)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if t_hi - t_lo < 5e-13:
                break
            if f1 < f2:
                t_hi = c2
                c2 = c1
                f2 = f1
                c1 = t_hi - gr * (t_hi - t_lo)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                t_lo = c1
                c1 = c2
                f1 = f2
                c2 = t_lo + gr * (t_hi - t_lo)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (t_lo + t_hi) / 2.0
        cur_new = cur + opt_t * d
        cur_new = _project(cur_new, target)
        new_c5 = _exact_c5(cur_new)
        if new_c5 < cur_c5 - 1e-14:
            cur = cur_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap_refine(h, deadline, max_passes=10):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        grad = _compute_grad(cur, n)
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for j in zeros:
            if time.time() > deadline:
                break
            deltas = grad[j] - grad[ones]
            cur_c = np.correlate(cur, 1.0 - cur, mode='full')
            new_vals = cur_c[None, :] + deltas
            new_maxes = np.max(new_vals, axis=1)
            bi = int(np.argmin(new_maxes))
            if new_maxes[bi] < np.max(cur_c) - 1e-10:
                cur[ones[bi]] = 0.0
                cur[j] = 1.0
                cur = _project(cur, target)
                new_c5 = _exact_c5(cur)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Binary steepest descent (15%)
    d1 = t0 + budget_s * 0.12
    h_bin, c5_bin = _binary_steepest(h_init, d1, max_sweeps=5000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 2: Binary block moves (10%)
    d2 = t0 + budget_s * 0.22
    h_blk, c5_blk = _binary_block_moves(h_bin, d2, max_bsize=4)
    if c5_blk < best_c5:
        best_c5 = c5_blk
        best_h = h_blk.copy()

    # Phase 3: Binary basin hopping (18%)
    d3 = t0 + budget_s * 0.40
    h_bh, c5_bh = _binary_basin_hopping(h_blk, d3, n_restarts=60)
    if c5_bh < best_c5:
        best_c5 = c5_bh
        best_h = h_bh.copy()

    # Phase 4: Continuous smooth GD (10%)
    d4 = t0 + budget_s * 0.50
    h_sg, c5_sg = _smooth_gd(h_bh, d4, tau=0.5)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()
    h_sg2, c5_sg2 = _smooth_gd(h_sg, min(deadline, t0 + budget_s * 0.55), tau=0.2)
    if c5_sg2 < best_c5:
        best_c5 = c5_sg2
        best_h = h_sg2.copy()

    # Phase 5: LP refinement (12%)
    d5 = t0 + budget_s * 0.67
    h_lp, c5_lp = _lp_refine(h_sg2, d5, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 6: Multi-resolution exploration (15%)
    d6 = t0 + budget_s * 0.82
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -24, 24, -32, 32, -48, 48]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 256 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_bin_r, _ = _binary_steepest(h_r, sub_dl, max_sweeps=500)
        sub_dl2 = min(d6, time.time() + budget_s * 0.012)
        h_sg_r, c5_r = _smooth_gd(h_bin_r, sub_dl2, tau=0.4)
        sub_dl3 = min(d6, time.time() + budget_s * 0.008)
        h_lp_r, c5_r2 = _lp_refine(h_sg_r, sub_dl3, radius_init=0.08)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_lp_r.copy()
            n_curr = n_try

    # Phase 7: Perturbation + re-optimize (8%)
    d7 = t0 + budget_s * 0.90
    rng = np.random.default_rng(seed)
    scale = 0.05
    for attempt in range(300):
        if time.time() > d7:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _smooth_gd(h_pert, sub_dl, tau=0.3)
        sub_dl2 = min(d7, time.time() + budget_s * 0.003)
        h_t2, c5_t2 = _lp_refine(h_t, sub_dl2, radius_init=0.05)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 8: Final polish (10%)
    d8 = deadline - 2
    h_fin = best_h.copy()
    h_fin, c5_fin = _lp_refine(h_fin, d8, radius_init=0.04)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()
    h_fin, c5_fin = _smooth_gd(best_h, min(deadline - 1, t0 + budget_s * 0.97), tau=0.1)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
