def _project(v, target):

    lo = float(v.min()) - 2.0

        if s > target:

        else:

    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:

        else:

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius):

    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:

    best_h = h.copy()
    best_c5 = _exact_c5(h)

        if time.time() > deadline:
            break

        beta = beta_init * (beta_final / beta_init) ** frac
        try:
            def obj(hf):
                S = _scores(hf)
                maxS = float(np.max(S))
                shifted = beta * (S - maxS)
                shifted = np.clip(shifted, -500.0, 0.0)
                return maxS + float(np.log(np.mean(np.exp(shifted)))) / beta
            def grad(hf):
                S = _scores(hf)
                maxS = float(np.max(S))
                shifted = beta * (S - maxS)

                else:
                    w /= wsum
                G = _grad_matrix(hf)
                return G.T @ w

                return float(np.sum(x) - target)

                return np.ones(n)

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.65
            if alpha < 1e-13:
                break
        if not improved and step % 100 == 0:
            h = _project(h - 0.005 * grad, target)
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for pas in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for step in [1, 2, 4]:
            if time.time() > deadline:
                break
            for j in range(0, n, step):
                if time.time() > deadline:
                    break
                if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                    continue
                for k in range(j + step, n, step):
                    if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                        continue
                    for sign in [1, -1]:
                        if sign > 0:
                            a_max = min(1.0 - h[j], h[k])
                        else:
                            a_max = min(h[j], 1.0 - h[k])
                        if a_max < 1e-10:
                            continue
                        lo_a, hi_a = 0.0, a_max
                        for _ in range(6):
                            m1 = lo_a + (hi_a - lo_a) / 3.0
                            m2 = hi_a - (hi_a - lo_a) / 3.0
                            h1 = h.copy()
                            h1[j] += sign * m1
                            h1[k] -= sign * m1
                            f1 = _exact_c5(h1)
                            h2 = h.copy()
                            h2[j] += sign * m2
                            h2[k] -= sign * m2
                            f2 = _exact_c5(h2)
                            if f1 < f2:
                                hi_a = m2
                            else:
                                lo_a = m1
                        a_opt = (lo_a + hi_a) / 2.0
                        if a_opt < 1e-10:
                            continue
                        h_try = h.copy()
                        h_try[j] += sign * a_opt
                        h_try[k] -= sign * a_opt
                        h_try = np.clip(h_try, 0.0, 1.0)
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h = h_try.copy()
                            improved = True
        if not improved:
            break
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Main LP optimization with small radius (25%)
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.06, max_iters=6000, stall_limit=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution (10%)
    if time.time() < t0 + budget_s * 0.38:
        dl2 = t0 + budget_s * 0.38
        for n_try in [96, 192]:
            if time.time() > dl2:
                break
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl2, time.time() + budget_s * 0.05)
            ht, ct = _lp_optimize(h_init, sub_dl, radius_init=0.06, max_iters=3000, stall_limit=200)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try
            else:
                if time.time() < dl2:
                    h_qp = _quasi_periodic_init(n_try, 0.6180339887)
                    h_qp += rng.normal(0, 0.03, n_try)
                    h_qp = _project(h_qp, n_try / 2.0)
                    sub_dl2 = min(dl2, time.time() + budget_s * 0.03)
                    ht2, ct2 = _lp_optimize(h_qp, sub_dl2, radius_init=0.08, max_iters=2000, stall_limit=150)
                    if ct2 < best_c5:
                        best_c5 = ct2
                        best_h = ht2.copy()
                        best_n = n_try

    # Phase 3: SLSQP + top-k subgradient (12%)
    if time.time() < t0 + budget_s * 0.50:
        dl3 = t0 + budget_s * 0.50
        h3, c5_3 = _slsqp_optimize(best_h, dl3, beta_init=50.0, beta_final=400.0, n_phases=5)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        if time.time() < dl3:
            sub_dl = min(dl3, time.time() + budget_s * 0.015)
            h3b, c5_3b = _lp_optimize(best_h, sub_dl, radius_init=0.04, max_iters=2000, stall_limit=100)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()
        if time.time() < t0 + budget_s * 0.53:
            dl3c = t0 + budget_s * 0.53
            h3c, c5_3c = _topk_subgrad_optimize(best_h, dl3c, n_steps=1500, k=3, lr_start=0.06, lr_end=0.005)
            if c5_3c < best_c5:
                best_c5 = c5_3c
                best_h = h3c.copy()

    # Phase 4: Aggressive perturbation restarts (35%)
    if time.time() < t0 + budget_s * 0.88:
        dl4 = t0 + budget_s * 0.88
        scales = [0.01, 0.015, 0.02, 0.03, 0.01, 0.02, 0.007, 0.012]
        scale_idx = 0
        counter = 0
        while time.time() < dl4:
            scale = scales[scale_idx % len(scales)]
            if counter % 10 == 0 and counter > 50:
                h_pert = 1.0 - best_h + rng.normal(0, 0.005, best_n)
            else:
                h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.006)
            ht, ct = _lp_optimize(h_pert, sub_dl, radius_init=0.04, max_iters=400, stall_limit=80)
            if ct < best_c5 - 1e-14:
                best_c5 = ct
                best_h = ht.copy()
                scale_idx = scale_idx % len(scales)
                counter = 0
            else:
                counter += 1
                if counter > 30:
                    scale_idx += 1
                    counter = 0

    # Phase 5: Final SLSQP + LP (8%)
    if time.time() < t0 + budget_s * 0.95:
        dl5 = t0 + budget_s * 0.95
        h5, c5_5 = _slsqp_optimize(best_h, dl5, beta_init=100.0, beta_final=600.0, n_phases=4)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < t0 + budget_s * 0.96:
            h5b, c5_5b = _lp_optimize(best_h, t0 + budget_s * 0.96,
                                       radius_init=0.03, max_iters=3000, stall_limit=150)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: Pair polish (3.5%)
    if time.time() < deadline:
        remaining = max(0.0, deadline - time.time())
        if remaining > 5:
            hp, c5p = _pair_polish(best_h, time.time() + remaining * 0.7, max_passes=4)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = hp.copy()
        if time.time() < deadline:
            h_final, c5_final = _lp_optimize(best_h, deadline, radius_init=0.02,
                                              max_iters=1000, stall_limit=80)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
