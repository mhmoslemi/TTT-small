import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 35
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.05):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.22, use_penalty=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    while time.time() < deadline:
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        G = _grad_matrix(h)
        penalty_lags = None
        if use_penalty and stall > 5:
            thresh = s_max * 0.995
            penalty_lags = list(np.where(scores >= thresh)[0])
        p = _lp_solve(h, scores, G, radius, penalty_lags)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.07, 0.55)
        else:
            stall += 1
            radius *= 0.73
            if stall > 35 and radius < 0.04:
                radius = best_radius * 0.5
                stall = 0
    return best_h, best_c5

def _sigmoid(x):
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _v_to_h(v, n):
    target = n / 2.0
    lo = -float(np.max(np.abs(v))) - 60.0
    hi = float(np.max(np.abs(v))) + 60.0
    lo = max(lo, -80.0)
    hi = min(hi, 80.0)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sigmoid(v + mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = _sigmoid(v + theta)
    return h

def _h_to_v(h, eps=1e-5):
    h_clipped = np.clip(h, eps, 1.0 - eps)
    return np.log(h_clipped / (1.0 - h_clipped))

def _les_step(pop, scores, mu, rng, sigma, deadline):
    """One evolution strategy step. pop is list of v-vectors, scores is list of c5 values."""
    n_pop = len(pop)
    # Sort by score
    order = np.argsort(scores)
    elites = [pop[i].copy() for i in order[:mu]]
    elite_scores = [scores[i] for i in order[:mu]]
    
    # Generate offspring
    children = []
    for i in range(n_pop):
        if time.time() > deadline:
            break
        # Select parents
        pa_idx = int(rng.integers(0, mu))
        if rng.random() < 0.5 and n_pop > 1:
            pb_idx = int(rng.integers(0, mu))
            alpha = rng.random()
            v = alpha * elites[pa_idx] + (1.0 - alpha) * elites[pb_idx]
        else:
            v = elites[pa_idx].copy()
        
        # Mutate
        op = rng.random()
        if op < 0.5:
            # Gaussian mutation
            v = v + sigma * rng.normal(0, 1, len(v))
        elif op < 0.7:
            # Partial mutation (only some coordinates)
            n_mut = max(1, len(v) // 4)
            idx = rng.choice(len(v), size=n_mut, replace=False)
            v[idx] += sigma * rng.normal(0, 1, n_mut)
        elif op < 0.85:
            # Swap two values
            i, j = rng.integers(0, len(v), size=2)
            v[i], v[j] = v[j], v[i]
        else:
            # Interval mutation
            i, j = sorted(rng.integers(0, len(v), size=2))
            v[i:j] += sigma * 0.3 * rng.normal(0, 1, j - i)
        
        v = np.clip(v, -30, 30)
        h = _v_to_h(v, len(v))
        h = _project(h, len(h) / 2.0)
        c5 = _exact_c5(h)
        children.append((v, c5))
    
    if not children:
        return pop, scores
    
    # Combine elites and children
    all_v = [v for v, _ in children] + elites
    all_s = [s for _, s in children] + elite_scores
    
    # Select best n_pop
    order = np.argsort(all_s)
    new_pop = [all_v[i].copy() for i in order[:n_pop]]
    new_scores = [all_s[i] for i in order[:n_pop]]
    return new_pop, new_scores

def _evolve(h, deadline, rng, n_gen=200, pop_size=10, mu=3, sigma_init=0.15):
    """Logit-space evolution strategy."""
    n = len(h)
    v0 = _h_to_v(h)
    pop = [v0.copy() for _ in range(pop_size)]
    # Add some diversity
    for i in range(1, pop_size):
        pop[i] = v0 + sigma_init * 0.5 * rng.normal(0, 1, n)
        pop[i] = np.clip(pop[i], -30, 30)
    
    scores = [_exact_c5(_project(_v_to_h(v, n), n / 2.0)) for v in pop]
    best_h = h.copy()
    best_c5 = min(scores)
    sigma = sigma_init
    
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        pop, scores = _les_step(pop, scores, mu, rng, sigma, deadline)
        
        # Track best
        best_idx = int(np.argmin(scores))
        if scores[best_idx] < best_c5 - 1e-14:
            best_c5 = scores[best_idx]
            best_v = pop[best_idx].copy()
            best_h = _project(_v_to_h(best_v, n), n / 2.0)
        
        # Adaptive sigma
        if gen > 0 and gen % 15 == 0:
            sorted_s = sorted(scores)
            improvement = sorted_s[0]
            if improvement < best_c5 - 0.001:
                sigma *= 0.85
            else:
                sigma *= 1.1
            sigma = max(0.005, min(sigma, 1.0))
    
    return best_h, best_c5

def _smooth_phase(h, deadline, beta_list=(40, 100, 250, 600), lr_init=0.025):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.84
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 60:
            bi += 1
            if bi >= len(beta_list) * 3:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = beta_list[bi % len(beta_list)]
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        s_max = float(np.max(scores))
        v = beta * (scores - s_max)
        exp_v = np.exp(v)
        w = exp_v / float(np.sum(exp_v))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:d + m])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (-h[g:g + m])
                    grad[g:] += w[r] * (1.0 - h[:m])
        grad -= np.mean(grad)
        dn = np.linalg.norm(grad)
        if dn < 1e-13:
            bi += 1
            continue
        grad = grad / dn
        mom = decay * mom + (1.0 - decay) * grad
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.06, 0.40)
        else:
            stall += 1
            lr *= 0.78
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=60, use_random=False, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n_interior = 0
    for i in range(n):
        if 1e-10 < h[i] < 1.0 - 1e-10:
            n_interior += 1
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if use_random and rng is not None and n > 80:
            pairs = set()
            for _ in range(400):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and 1e-10 < h[i] < 1 - 1e-10 and 1e-10 < h[j] < 1 - 1e-10:
                    pairs.add((min(i, j), max(i, j)))
        else:
            pairs = set()
            for i in range(n):
                if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                    continue
                for j in range(i + 1, n):
                    if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                        continue
                    pairs.add((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _kick_and_reopt(h, deadline, rng, n_kicks=40):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.08)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.12, use_penalty=False)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.22, use_penalty=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    while time.time() < deadline:
        if stall > 120:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        G = _grad_matrix(h)
        penalty_lags = None
        if use_penalty and stall > 5:
            thresh = s_max * 0.995
            penalty_lags = list(np.where(scores >= thresh)[0])
        p = _lp_solve(h, scores, G, radius, penalty_lags)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.07, 0.55)
        else:
            stall += 1
            radius *= 0.73
            if stall > 40 and radius < 0.04:
                radius = best_radius * 0.5
                stall = 0
    return best_h, best_c5

def _sigmoid_safe(x):
    out = np.empty_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _v_to_h(v, n):
    target = n / 2.0
    abs_max = float(np.max(np.abs(v)))
    lo = -abs_max - 50.0
    hi = abs_max + 50.0
    lo = max(lo, -80.0)
    hi = min(hi, 80.0)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sigmoid_safe(v + mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = _sigmoid_safe(v + theta)
    return h

def _h_to_v(h, eps=1e-5):
    hc = np.clip(h, eps, 1.0 - eps)
    return np.log(hc / (1.0 - hc))

def _evolve(h, deadline, rng, n_gen=100, pop_size=10, mu=3, sigma_init=0.18):
    n = len(h)
    target = n / 2.0
    v0 = _h_to_v(h)
    pop = [v0.copy() for _ in range(pop_size)]
    for i in range(1, pop_size):
        pop[i] = v0 + sigma_init * 0.4 * rng.normal(0, 1, n)
        pop[i] = np.clip(pop[i], -25, 25)
    scores = []
    for v in pop:
        hh = _project(_v_to_h(v, n), target)
        scores.append(_exact_c5(hh))
    best_h = h.copy()
    best_c5 = min(scores)
    sigma = sigma_init
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(scores)
        elites = [pop[i].copy() for i in order[:mu]]
        children_v = []
        children_s = []
        n_off = pop_size - mu
        for _ in range(n_off):
            if time.time() > deadline:
                break
            pa = elites[int(rng.integers(0, mu))]
            op = rng.random()
            if op < 0.4:
                v = pa + sigma * rng.normal(0, 1, n)
            elif op < 0.6:
                if mu > 1:
                    pb = elites[int(rng.integers(0, mu))]
                    v = 0.5 * (pa + pb) + sigma * 0.5 * rng.normal(0, 1, n)
                else:
                    v = pa + sigma * rng.normal(0, 1, n)
            elif op < 0.8:
                v = pa.copy()
                k = max(2, n // 5)
                idx = rng.choice(n, size=k, replace=False)
                v[idx] += sigma * rng.normal(0, 1, k)
            else:
                v = pa.copy()
                i1, i2 = sorted(rng.integers(0, n, size=2))
                v[i1:i2] += sigma * 0.2 * rng.normal(0, 1, i2 - i1)
            v = np.clip(v, -30, 30)
            hh = _project(_v_to_h(v, n), target)
            cs = _exact_c5(hh)
            children_v.append(v)
            children_s.append(cs)
        all_v = children_v + elites
        all_s = children_s + [scores[i] for i in order[:mu]]
        aorder = np.argsort(all_s)
        pop = [all_v[i].copy() for i in aorder[:pop_size]]
        scores = [all_s[i] for i in aorder[:pop_size]]
        if scores[0] < best_c5 - 1e-14:
            best_c5 = scores[0]
            best_h = _project(_v_to_h(pop[0], n), target)
        if gen > 0 and gen % 12 == 0:
            if scores[0] < best_c5 - 0.002:
                sigma *= 0.82
            else:
                sigma *= 1.15
            sigma = max(0.004, min(sigma, 1.5))
    return best_h, best_c5

def _coord_descent(h, deadline, max_sweeps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            A = np.zeros(2 * n - 1)
            for r in range(2 * n - 1):
                d = r - (n - 1)
                av = 0.0
                ip = i + d
                if 0 <= ip < n:
                    av += (1.0 - h[ip])
                im = i - d
                if 0 <= im < n:
                    av -= h[im]
                A[r] = av
            B = corr - A * h[i]
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - hi_t + lo_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = float(np.max(A * m1 + B))
                f2 = float(np.max(A * m2 + B))
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1
            t_new = (lo_t + hi_t) / 2.0
            if abs(t_new - h[i]) < 1e-10:
                continue
            h[i] = t_new
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                sweep_improved = True
        if not sweep_improved and sweep > 5:
            break
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=60, use_random=False, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if use_random and rng is not None and n > 80:
            pairs = set()
            for _ in range(500):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and 1e-10 < h[i] < 1 - 1e-10 and 1e-10 < h[j] < 1 - 1e-10:
                    pairs.add((min(i, j), max(i, j)))
        else:
            pairs = set()
            for i in range(n):
                if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                    continue
                for j in range(i + 1, n):
                    if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                        continue
                    pairs.add((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5
