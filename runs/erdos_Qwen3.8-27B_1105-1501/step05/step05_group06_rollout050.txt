import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_val_grad(h, q):
    """Compute p-norm surrogate value and gradient."""
    F = _scores(h)
    M = float(np.max(F))
    if M <= 1e-30:
        return M, np.zeros(len(h))
    w = F / M
    wq1 = np.power(w, q - 1)  # w^(q-1), all in [0,1]
    S = float(np.sum(np.power(w, q)))  # sum(w^q)
    if S <= 0:
        return M, np.zeros(len(h))
    Phi = M * np.power(S, 1.0 / q)
    G = _build_G(h)
    grad = np.power(S, 1.0 / q - 1.0) * (G.T @ wq1)
    return Phi, grad

def _pnorm_fun(x, q):
    F = _scores(x)
    M = float(np.max(F))
    if M <= 1e-30:
        return M
    w = F / M
    S = float(np.sum(np.power(w, q)))
    if S <= 0:
        return M
    return M * np.power(S, 1.0 / q)

def _pnorm_jac(x, q):
    F = _scores(x)
    M = float(np.max(F))
    if M <= 1e-30:
        return np.zeros(len(x))
    w = F / M
    wq1 = np.power(w, q - 1)
    S = float(np.sum(np.power(w, q)))
    if S <= 0:
        return np.zeros(len(x))
    G = _build_G(x)
    return np.power(S, 1.0 / q - 1.0) * (G.T @ wq1)

def _slsqp_optimize(h, q, deadline, maxiter=200):
    """Optimize p-norm surrogate using SLSQP."""
    n = len(h)
    target = n / 2.0
    h0 = h.copy()
    
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                  'jac': lambda x: np.ones(n)}
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(_pnorm_fun, h0, jac=_pnorm_jac, args=(q,),
                      method='SLSQP', bounds=bounds, constraints=[constraint],
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x, target)
            return h_new
    except Exception:
        pass
    return h0

def _pair_descent(h, deadline, max_passes=200, verify_thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > verify_thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    np.clip(h, 0, 1, out=h)
                    if abs(np.sum(h) - n / 2.0) > 1e-10:
                        h = _project(h, n / 2.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        np.clip(h, 0, 1, out=h)
        if not pass_improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G


def _pnorm_fun(x, q):
    F = _scores(x)
    M = float(np.max(F))
    if M <= 1e-30:
        return M
    w = F / M
    S = float(np.sum(np.power(w, q)))
    if S <= 0:
        return M
    return M * np.power(S, 1.0 / q)


def _pnorm_jac(x, q):
    F = _scores(x)
    M = float(np.max(F))
    if M <= 1e-30:
        return np.zeros(len(x))
    w = F / M
    wq1 = np.power(w, q - 1)
    S = float(np.sum(np.power(w, q)))
    if S <= 0:
        return np.zeros(len(x))
    G = _build_G(x)
    return np.power(S, 1.0 / q - 1.0) * (G.T @ wq1)


def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def _eq_jac(x):
    return np.ones(len(x))


def _slsqp_solve(h, q, deadline, maxiter=200):
    n = len(h)
    h0 = h.copy()
    constraint = {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    bounds = [(0.0, 1.0)] * n
    try:
        remaining = deadline - time.time()
        if remaining < 5:
            return h0
        res = minimize(_pnorm_fun, h0, jac=_pnorm_jac, args=(q,),
                       method='SLSQP', bounds=bounds, constraints=[constraint],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x, n / 2.0)
            if c5_check(h_new, h0):
                return h_new
    except Exception:
        pass
    return h0


def c5_check(h_new, h_old):
    return _exact_c5(h_new) < _exact_c5(h_old) + 0.01

def _pair_descent(h, deadline, max_passes=300, verify_thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > verify_thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    np.clip(h, 0.0, 1.0, out=h)
                    if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
                        h = _project(h, n / 2.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        np.clip(h, 0.0, 1.0, out=h)
        if not pass_improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def _smooth_parent(h, n_new, bw):
    n = len(h)
    if n_new == n:
        h_new = h.copy()
    else:
        x_old = (np.arange(n) + 0.5) / n
        x_new = (np.arange(n_new) + 0.5) / n_new
        h_new = np.interp(x_new, x_old, h)
    kernel = np.exp(-0.5 * (np.arange(-bw, bw + 1) / 1.0) ** 2)
    kernel /= kernel.sum()
    padded = np.pad(h_new, bw, mode='wrap')
    h_smooth = np.convolve(padded, kernel, mode='valid')
    return _project(h_smooth, n_new / 2.0)


def _lp_direction(h, top_k, radius):

    scores = _scores(h)

    G_k = G[idx]
    s_k = scores[idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),

        if res.success:

    except Exception:
        pass
    return None

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 3: LP-based direction (complementary to gradient approach)
    if time.time() < t0 + budget_s * 0.45:
        dl3 = t0 + budget_s * 0.45
        h_lp, c5_lp = _lp_optimize(best_h, dl3, top_k_list=[None, 8, 5, 3],
                                   radius_init=0.10, max_iters=1500)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 4: Multi-start exploration
    if time.time() < t0 + budget_s * 0.65:
        dl4 = t0 + budget_s * 0.65
        configs = [
            (n0, 0.6180339887, 'quasi'),
            (n0, 0.4142135624, 'quasi'),
            (160, 0.6180339887, 'quasi'),
            (160, 0.7071067812, 'quasi'),
            (128, 0.3819660113, 'quasi'),
            (192, 0.5, 'quasi'),
            (n0, 0, 'smooth'),
        ]
        for n_try, param, stype in configs:
            if time.time() > dl4 - 15:
                break
            if stype == 'quasi':
                h_init = _quasi_init(n_try, param, rng)
            elif stype == 'smooth':
                h_init = _smooth_parent(best_h, n_try, 3)
            else:
                h_init = _resample(best_h, n_try)
            # Quick gradient optimization
            sub_dl = min(dl4 - 15, time.time() + 25)
            h_try, c5_try = _proj_grad(h_init, 64, sub_dl, n_steps=50, lr_init=0.04)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                # Try LP on this start
                sub_dl2 = min(dl4 - 10, time.time() + 15)
                h_try2, c5_try2 = _lp_optimize(h_try, sub_dl2,
                                               top_k_list=[6, 4, None],
                                               radius_init=0.08, max_iters=800)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()
                    best_n = n_try

    # Phase 5: Pair descent polish
    if time.time() < t0 + budget_s * 0.78:
        dl5 = t0 + budget_s * 0.78
        best_h, best_c5 = _pair_descent(best_h, dl5, max_passes=300, verify_thresh=1e-6)

    # Phase 6: Final p-norm + SLSQP refinement
    if time.time() < t0 + budget_s * 0.88:
        dl6 = t0 + budget_s * 0.88
        for q in [128, 256]:
            if time.time() > dl6 - 10:
                break
            h_new, c5_new = _proj_grad(best_h, q, min(dl6 - 10, time.time() + 20),
                                       n_steps=40, lr_init=0.02)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if time.time() > dl6 - 10:
                break
            h_new2 = _slsqp_solve(best_h, q, min(dl6 - 5, time.time() + 20), maxiter=100)
            c5_new2 = _exact_c5(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()

    # Phase 7: Perturbation restarts
    if time.time() < t0 + budget_s * 0.93:
        dl7 = t0 + budget_s * 0.93
        scale = 0.02
        for attempt in range(80):
            if time.time() > dl7:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl7, time.time() + 10)
            h_try, c5_try = _proj_grad(h_pert, 128, sub_dl, n_steps=40, lr_init=0.03)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.05)

    # Phase 8: Final pair descent
    if time.time() < hard_deadline - 3:
        dl8 = hard_deadline - 3
        best_h, best_c5 = _pair_descent(best_h, dl8, max_passes=150, verify_thresh=1e-8)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, final_c5, best_n
