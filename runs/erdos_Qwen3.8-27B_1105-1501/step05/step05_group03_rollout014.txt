Actually, let me think about the subgradient approach more carefully. The objective is:
C₅(h) = max_d S_d(h)

where S_d(h) = (2/n) * sum_{i} h[i]*(1-h[i+d]) (over valid i)

This is the max of smooth (quadratic) functions, so it's convex in h? No, wait - each S_d is not convex in h because it's bilinear (h[i]*(1-h[j]) = h[i] - h[i]*h[j]). So the objective is not convex.

The subgradient of S_d with respect to h[j] is:
∂S_d/∂h[j] = (2/n) * sum over terms involving h[j]

For a fixed lag d, S_d = (2/n) * sum_{i: i and i+d valid} h[i]*(1-h[i+d])

The derivative with respect to h[j] gets contributions from:
- Terms where i=j: (2/n) * (1-h[j+d]) (if j+d is valid)
- Terms where i+d=j, i.e., i=j-d: (2/n) * (-h[j-d]) (if j-d is valid)

So the gradient of S_d at h is:
(∂S_d/∂h)[j] = (2/n) * [(1-h[j+d])_valid - h[j-d]_valid]

where the subscript "valid" means the term is 0 if the index is out of range.

For the max, the subgradient is a convex combination of the gradients of the active lags (those achieving the max).

This gives us a simple subgradient method:
1. Compute all S_d
2. Find the active lags (those near the max, say within epsilon)
3. Average their gradients
4. Take a step in the negative gradient direction
5. Project back to feasible set

import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_top_lags(h, K=8):
    """Compute gradient contribution from top-K active lags (vectorized)."""
    n = len(h)
    scores = _scores(h)
    n_lags = len(scores)
    n_half = n - 1
    max_s = np.max(scores)
    shifted = 40.0 * (scores - max_s)
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros(n_lags)
        w[np.argmax(scores)] = 1.0
    else:
        w /= ws
    top_idx = np.argsort(w)[-K:]
    grad = np.zeros(n)
    for li in top_idx:
        wt = w[li]
        if wt < 1e-15:
            continue
        d = li - n_half
        c = (2.0 / n) * wt
        ad = abs(d)
        if ad == 0:
            continue
        lo_b = 0
        hi_b = n - ad
        grad[lo_b:hi_b] += c * (1.0 - h[ad:n])
        grad[ad:n] += c * (-h[0:hi_b])
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return np.zeros(n), scores
    return -grad / gn, scores

def _build_block_basis(n, rng, q_target=50):
    """Build sum-zero basis from bipolar blocks."""
    cols = []
    widths = []
    for w in [4, 6, 8, 12, 16, 20, 24, 32, 40, 48, 64, 80, 96]:
        if 2 * w <= n:
            widths.append(w)
    for w in widths:
        half = w // 2
        if half < 1:
            continue
        n_starts = n - w + 1
        if n_starts <= 0:
            continue
        n_blk = max(2, n // w)
        starts = np.unique(np.concatenate([
            np.linspace(0, max(0, n - w), n_blk, dtype=int),
            rng.integers(0, n_starts, size=min(6, n_starts))
        ]))
        for p in starts:
            p = int(p)
            if p < 0 or p + w > n:
                continue
            col = np.zeros(n)
            col[p:p + half] = 1.0
            col[p + half:p + w] = -1.0
            col -= col.mean()
            nrm = np.linalg.norm(col)
            if nrm > 1e-10:
                col /= nrm
            cols.append(col)
    if not cols:
        return np.zeros((n, 0))
    arr = np.column_stack(cols)
    q = min(q_target, arr.shape[1])
    if arr.shape[1] > q:
        idx = rng.choice(arr.shape[1], size=q, replace=False)
        arr = arr[:, idx]
    Q, R = np.linalg.qr(arr)
    rank = int(np.sum(np.abs(np.diag(R)) > 1e-6))
    Q = Q[:, :rank]
    for j in range(Q.shape[1]):
        Q[:, j] -= Q[:, j].mean()
    return Q

def _de_optimize(h_start, basis, deadline, pop_size=50, F=0.7, CR=0.8,
                 n_gen_max=500, bounds_lo=-3.0, bounds_hi=3.0,
                 target=None, rng=None):
    """Differential evolution in y-space with block basis."""
    n = len(h_start)
    if target is None:
        target = n / 2.0
    q = basis.shape[1]
    if q < 2:
        return h_start, _exact_c5(h_start)
    # Initial y: least squares fit
    v = h_start - 0.5
    y0, _, _, _ = np.linalg.lstsq(basis, v, rcond=None)
    lb = np.full(q, bounds_lo)
    ub = np.full(q, bounds_hi)
    # Population
    pop = np.zeros((pop_size, q))
    pop[0] = y0
    for i in range(1, pop_size):
        pop[i] = y0 + rng.uniform(-0.5, 0.5, q)
        np.clip(pop[i], lb, ub, out=pop[i])
    fit = np.full(pop_size, np.inf)
    fit[0] = _exact_c5(_project(0.5 + basis @ y0, target))
    best_c5 = fit[0]
    best_h = _project(0.5 + basis @ y0, target)
    for i in range(1, pop_size):
        h_try = _project(0.5 + basis @ pop[i], target)
        fit[i] = _exact_c5(h_try)
        if fit[i] < best_c5:
            best_c5 = fit[i]
            best_h = h_try.copy()
    gen = 0
    while gen < n_gen_max:
        if time.time() > deadline:
            break
        gen += 1
        order = np.argsort(fit)
        best_i = order[0]
        for i in range(pop_size):
            if time.time() > deadline:
                break
            # Mutation
            r1, r2 = rng.choice(pop_size, 2, replace=False)
            while r1 == r2:
                r2 = rng.choice(pop_size)
            while best_i == r1 or best_i == r2:
                r1 = rng.choice(pop_size)
                r2 = rng.choice(pop_size)
                while r1 == r2:
                    r2 = rng.choice(pop_size)
            trial = pop[best_i] + F * (pop[r1] - pop[r2])
            # Crossover
            cross_point = rng.integers(q)
            mask = rng.random(q) < CR
            mask[cross_point] = True
            trial = np.where(mask, trial, pop[i])
            np.clip(trial, lb, ub, out=trial)
            h_try = _project(0.5 + basis @ trial, target)
            f_try = _exact_c5(h_try)
            if f_try <= fit[i]:
                pop[i] = trial
                fit[i] = f_try
                if f_try < best_c5:
                    best_c5 = f_try
                    best_h = h_try.copy()
        # Adaptive F
        F = max(0.3, F * 0.998)
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    """LP to find a descent direction in the feasible set."""
    n = len(h)
    n_lags = len(scores)
    # Build gradient matrix: row per lag, col per h[j]
    # G[r, j] = dScore_r / dh_j
    G = np.zeros((n_lags, n))
    dx = 2.0 / n
    n_half = n - 1
    # Only use top 20 lags for efficiency
    top_k = np.argsort(scores)[-20:]
    for li in top_k:
        d = li - n_half
        ad = abs(d)
        if ad == 0:
            continue
        lo_b = 0
        hi_b = n - ad
        G[li, lo_b:hi_b] += (1.0 - h[ad:n])
        G[li, ad:n] += (-h[0:hi_b])
    G *= dx
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, target, n_eval=25):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(n_eval):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_h = _project(h + opt_a * p, target)
    opt_c5 = _exact_c5(opt_h)
    if opt_c5 < min(f1, f2):
        return opt_a, opt_c5
    if f1 < f2:
        return c1, f1
    return c2, f2

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=3000, radius_min=0.001):
    n = len(h)
    target = n / 2.0

def _subgrad_optimize(h, deadline, n_steps=500, step_init=0.04, beta=80.0, momentum=0.85):
    """Log-sum-exp subgradient with momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        direction, scores = _grad_top_lags(h, K=10)
        if np.linalg.norm(direction) < 1e-15:
            break
        v = momentum * v + step * direction
        v_norm = np.linalg.norm(v)
        if v_norm > 0.1:
            v = v * 0.1 / v_norm
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:

                best_h = h.copy()
            step = min(step * 1.03, 0.1)
        else:
            v *= 0.5
            step *= 0.88
            if step < 1e-7:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, n_eval_max=5000):
    """Random pair mass-transfer polish."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cnt = 0
    while cnt < n_eval_max:
        if time.time() > deadline:
            break
        cnt += 1
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        for sign in (1, -1):
            if sign > 0:
                a_max = min(h[i], 1.0 - h[j])
            else:
                a_max = min(h[j], 1.0 - h[i])
            if a_max < 1e-10:
                continue
            # Try a few delta values
            for frac in (0.5, 0.2, 0.8):
                delta = sign * a_max * frac
                h[i] += delta
                h[j] -= delta
                if h[i] >= -1e-10 and h[i] <= 1 + 1e-10 and h[j] >= -1e-10 and h[j] <= 1 + 1e-10:
                    h[i] = np.clip(h[i], 0, 1)
                    h[j] = np.clip(h[j], 0, 1)
                    c5_try = _exact_c5(h)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h.copy()
                        continue
                # Revert
                h[i] -= delta
                h[j] -= (-delta)
    return best_h, best_c5

def _quasi_init(n, alpha):
    """Quasi-periodic binary initialization."""
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0 < 0.5).astype(float)
    return _project(h, n / 2.0)
