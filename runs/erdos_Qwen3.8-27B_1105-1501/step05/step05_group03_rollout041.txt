import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_w(w, h):
    """Compute sum_d w[d] * dS_d/dh without forming the full matrix.
    w has length 2n-1, indexed by n-1+d for lag d in [-(n-1), n-1].
    h has length n.
    Returns gradient of length n.
    """
    n = len(h)
    grad = np.zeros(n)
    w0 = w[n-1]
    grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[n-1+d]
        wm = w[n-1-d]
        if abs(wp) > 1e-16:
            grad[:m] += wp * (1.0 - h[d:])
            grad[d:] -= wp * h[:m]
        if abs(wm) > 1e-16:
            grad[d:] += wm * (1.0 - h[:m])
            grad[:m] -= wm * h[d:]
    return grad

def _softmax_weights(scores, t):
    max_s = np.max(scores)
    exps = np.exp(t * (scores - max_s))
    sw = np.sum(exps)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
        return w
    return exps / sw

def _smooth_gd(h, deadline, t_start=5.0, t_end=200.0, n_steps=200, step=0.1):
    """Smoothed gradient descent with increasing temperature."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_val = _exact_c5(h)
    best_h = h.copy()
    best_val = cur_val
    v = np.zeros(n)  # momentum
    
    log_t = np.log(t_end / t_start)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        t = t_start * np.exp(frac * log_t)
        
        scores = _all_scores(h)
        w = _softmax_weights(scores, t)
        g = _grad_w(w, h)
        g -= np.mean(g)  # remove mean to respect sum constraint
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Momentum
        v = 0.85 * v - step * g
        h_new = _project(h + v, target)
        
        val_new = _exact_c5(h_new)
        if val_new < cur_val:
            h = h_new
            cur_val = val_new
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            step = min(step * 1.02, 0.15)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-7:
                break
    
    return best_h, best_val

def _slsqp_h(h, deadline, n_active=10, max_outer=5):
    """SLSQP optimization on active constraints."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_val = _exact_c5(h)
    best_h = h.copy()
    best_val = cur_val
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        scores = _all_scores(h)
        # Get top n_active lags
        top_idx = np.argsort(scores)[-n_active:]
        
        # Build constraint functions
        def objective(x):
            return x[n]
        
        cons = []
        for idx in top_idx:
            d = idx - (n - 1)
            def make_ineq(dd):
                def fun(x):
                    hp = x[:n]
                    u = x[n]
                    if dd == 0:
                        s = np.sum(hp * (1.0 - hp))
                    elif dd > 0:
                        m = n - dd
                        s = np.sum(hp[:m] * (1.0 - hp[dd:]))
                    else:
                        g = -dd
                        m = n - g
                        s = np.sum(hp[g:] * (1.0 - hp[:m]))
                    return u - s
                def jacob(x):
                    hp = x[:n]
                    u = x[n]
                    if dd == 0:
                        s = np.sum(hp * (1.0 - hp))
                    elif dd > 0:
                        m = n - dd
                        s = np.sum(hp[:m] * (1.0 - hp[dd:]))
                    else:
                        g = -dd
                        m = n - g
                        s = np.sum(hp[g:] * (1.0 - hp[:m]))
                    j = np.zeros(n + 1)
                    if dd == 0:
                        j[:n] = -(1.0 - 2.0 * hp)
                    elif dd > 0:
                        m = n - dd
                        j[:m] += -(1.0 - hp[dd:])
                        j[dd:] += hp[:m]
                    else:
                        g = -dd
                        m = n - g
                        j[g:] += -(1.0 - hp[:m])
                        j[:m] += hp[g:]
                    j[n] = 1.0
                    return j
                return {'type': 'ineq', 'fun': fun, 'jac': jacob}
            cons.append(make_ineq(d))
        
        # Equality constraint: sum(h) = target
        def eq_fun(x):
            return np.sum(x[:n]) - target
        def eq_jac(x):
            j = np.zeros(n + 1)
            j[:n] = 1.0
            return j
        cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
        
        u0 = cur_val + 0.005
        x0 = np.concatenate([h, [u0]])
        bounds = [(0.0, 1.0)] * n + [(cur_val - 0.01, None)]
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
            if res.x is not None and res.x[n] < cur_val - 1e-12:
                h_new = _project(res.x[:n], target)
                val_new = _exact_c5(h_new)
                if val_new < best_val:
                    best_val = val_new
                    best_h = h_new.copy()
                cur_val = val_new
                h = h_new.copy()
        except Exception:
            break
    
    return best_h, best_val

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0 < 0.5).astype(float)
    return _project(h, n / 2.0)

def _binary_init(n, alpha):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    # Take the n/2 smallest fractional parts as 1s
    threshold = np.sort(vals)[n//2]
    h = (vals < threshold).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > n//2:
        # Flip some 1s to 0s
        idx_1 = np.where(h == 1.0)[0]
        flip = rng_idx = idx_1[:n_ones - n//2]
        h[flip] = 0.0
    elif n_ones < n//2:
        idx_0 = np.where(h == 0.0)[0]
        flip = idx_0[:n//2 - n_ones]
        h[flip] = 1.0
    return h

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_top_k(h, k, t=1.0):

    top_idx = np.argsort(scores)[-kk:]
    top_scores = scores[top_idx]
    max_s = np.max(top_scores)
    w = np.exp(t * (top_scores - max_s))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(kk)
        w[np.argmax(top_scores)] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)

        d = idx - (n - 1)

        if d == 0:

    n = len(h)

    best_h = h.copy()
    best_val = cur_val
    v = np.zeros(n)
    step = step_init
    log_ratio = np.log(t_end / t_start)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        t = t_start * np.exp(frac * log_ratio)
        g = _grad_top_k(h, k, t)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        v = 0.82 * v - step * g
        h_new = _project(h + v, target)
        val_new = _exact_c5(h_new)
        if val_new < cur_val - 1e-15:

            cur_val = val_new

                best_h = h.copy()

            return np.sum(x[:n]) - target

            j = np.zeros(n + 1)

        cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
        
        u0 = cur_val + 0.003

    n = len(h)
    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] = np.clip(h2[j] + sign * m2, 0, 1)
                        h2[k] = np.clip(h2[k] - sign * m2, 0, 1)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] = np.clip(h_try[j] + sign * a_opt, 0, 1)
                    h_try[k] = np.clip(h_try[k] - sign * a_opt, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _optimize_initial(h, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    dl1 = min(deadline, time.time() + (deadline - time.time()) * 0.5)
    h1, c5_1 = _smooth_gd(h, dl1, t_start=5, t_end=200, n_steps=200, step_init=0.06, k=15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    if time.time() < deadline:
        dl2 = deadline
        h2, c5_2 = _slsqp_active(best_h, dl2, n_active=8, max_outer=6)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    return best_h, best_c5
