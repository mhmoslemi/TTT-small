    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, d, n):
    """Gradient of S_d w.r.t. h. d is the lag (can be negative)."""
    g = np.zeros(n)
    ad = abs(d)
    m = n - ad
    if m <= 0:
        return g
    # S_d = sum_{i=0}^{m-1} h[min(i, i+ad depending on sign)] * (1 - h[...])
    # For d >= 0: S_d = sum_{i=0}^{m-1} h[i]*(1-h[i+d])
    #   grad[j] = (1-h[j+d]) for j < m  [j is first index]
    #           + (-h[j-d]) for j >= d   [j is second index]
    # For d < 0 (let g = -d): S_{-g} = sum_{i=0}^{m-1} h[i+g]*(1-h[i])
    #   grad[j] = (1-h[j-g]) for j >= g  [j is first index (i+g)]
    #           + (-h[j+g]) for j < m     [j is second index (i)]
    if d >= 0:
        g[:m] += (1.0 - h[ad:])
        g[ad:] += -h[:m]
    else:
        g[ad:] += (1.0 - h[:m])
        g[:m] += -h[ad:]
    return g

def _hess_vp(v, d, n):
    """Hessian-vector product for S_d."""
    ad = abs(d)
    if ad == 0:
        return -2.0 * v
    m = n - ad
    if m <= 0:
        return np.zeros(n)
    result = np.zeros(n)
    result[:m] += -v[ad:]
    result[ad:] += -v[:m]
    return result
