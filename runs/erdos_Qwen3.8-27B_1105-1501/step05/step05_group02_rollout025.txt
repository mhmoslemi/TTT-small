```python
def _binary_vec_greedy_pass(h, old_corr, flip_L, flip_R, ones, zeros):
    """One greedy pass: find best swap among all (i in ones, j in zeros)."""
    n = len(h)
    dx = 2.0 / n
    
    # old_L: L(d) for d=1..n-1, stored as old_corr[n-2:0:-1] (reversed)
    # Actually: old_corr[n-1-d] = L(d) for d=1..n-1
    # So old_L = old_corr[n-2:0:-1] gives L(1), L(2), ..., L(n-1)
    # And old_R = old_corr[n:] gives R(1), R(2), ..., R(n-1)
    
    old_L = old_corr[n-2:0:-1].astype(float)  # L(1)..L(n-1)
    old_R = old_corr[n:].astype(float)  # R(1)..R(n-1)
    
    n1 = len(ones)
    n0 = len(zeros)
    
    # Build all pair deltas
    # flip_L[i]: shape (n-1,) for each i in ones
    # flip_L[j]: shape (n-1,) for each j in zeros
    # For pair (i, j): delta_L = flip_L[i] + flip_L[j]
    
    # Vectorized: stack all flip vectors
    FL_ones = flip_L[ones]  # shape (n1, n-1)
    FL_zeros = flip_L[zeros]  # shape (n0, n-1)
    FR_ones = flip_R[ones]  # shape (n1, n-1)
    FR_zeros = flip_R[zeros]  # shape (n0, n-1)
    
    # All pairs: (n1, n0, n-1)
    # delta_L[:, :, d] = FL_ones[:, 0, d] + FL_zeros[0, :, d] ... 
    # Use broadcasting
    delta_L = FL_ones[:, np.newaxis, :] + FL_zeros[np.newaxis, :, :]  # (n1, n0, n-1)
    delta_R = FR_ones[:, np.newaxis, :] + FR_zeros[np.newaxis, :, :]  # (n1, n0, n-1)
    
    new_L = old_L[np.newaxis, np.newaxis, :] + delta_L  # (n1, n0, n-1)
    new_R = old_R[np.newaxis, np.newaxis, :] + delta_R
    
    # New max for each pair
    new_maxes = np.maximum(new_L, new_R).max(axis=2) * dx  # (n1, n0)
    
    # Find best
    flat_idx = np.argmin(new_maxes)
    i_idx = flat_idx // n0
    j_idx = flat_idx % n0
    best_c5 = new_maxes.flat[flat_idx]
    best_i = ones[i_idx]
    best_j = zeros[j_idx]
    
    return best_i, best_j, best_c5
```

```python
def _binary_optimize(h_init, deadline, max_passes=1000):
    n = len(h_init)
    h = h_init.copy()
    dx = 2.0 / n
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Precompute
    old_corr = np.correlate(h, 1.0-h, mode='full').astype(float)
    flip_L, flip_R = _precompute_flip_vectors(h)
    
    for it in range(max_passes):
        if time.time() > deadline:
            break
        
        # Find best swap
        n1 = len(ones)
        n0 = len(zeros)
        
        FL_ones = flip_L[ones]  # (n1, n-1)
        FL_zeros = flip_L[zeros]  # (n0, n-1)
        FR_ones = flip_R[ones]
        FR_zeros = flip_R[zeros]
        
        old_L = old_corr[n-2:0:-1].copy()  # L(1)..L(n-1)
        old_R = old_corr[n:].copy()  # R(1)..R(n-1)
        
        # Vectorized: (n1, n0, n-1)
        new_L = old_L[None, None, :] + FL_ones[:, None, :] + FL_zeros[None, :, :]
        new_R = old_R[None, None, :] + FR_ones[:, None, :] + FR_zeros[None, :, :]
        
        new_maxes = np.maximum(new_L, new_R).max(axis=2) * dx
        
        flat_idx = np.argmin(new_maxes)
        best_val = new_maxes.flat[flat_idx]
        
        if best_val >= cur_c5 - 1e-15:
            break  # No improvement
        
        i_idx = flat_idx // n0
        j_idx = flat_idx % n0
        best_i = ones[i_idx]
        best_j = zeros[j_idx]
        
        # Apply swap
        h[best_i] = 0.0
        h[best_j] = 1.0
        cur_c5 = best_val
        
        # Update corr
        old_corr[n-1 - np.arange(1, n)] += FL_ones[i_idx] + FL_zeros[j_idx]
        old_corr[n + np.arange(0, n-1)] += FR_ones[i_idx] + FR_zeros[j_idx]
        
        # Update ones and zeros

        ones = np.append(ones, best_j)
        zeros = zeros[zeros != best_j]
        zeros = np.append(zeros, best_i)
        
        # Recompute flip vectors
        flip_L, flip_R = _precompute_flip_vectors(h)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
```

Now, let me also think about the continuous optimization. After finding a good binary solution, I can try to improve it by allowing fractional values. The key idea is: if a value is at 0 or 1, maybe moving it slightly inward (or outward) helps reduce the maximum overlap.

For continuous h, the objective is:
C(h) = max_k corr(h, 1-h, k) * dx

This is a max of linear functions (in h), so it's convex in h! Wait, is it?

corr(h, 1-h, k) = Σ_i h[i] * (1-h[i+k]) for valid i. This is NOT linear in h - it's quadratic! It's Σ h[i] - Σ h[i]*h[i+k], which is quadratic.

So the objective is the max of quadratic functions, which is not convex in general. This makes continuous optimization harder.

However, around a point where the max is achieved at a unique k*, the problem is approximately: minimize h[k* related quadratic] subject to constraints. This is a quadratic programming problem (if we ignore the max and just minimize the active constraint).

For a simpler approach, I can use projected gradient descent on the active constraint:
1. Find the k* that achieves the max
2. Compute the gradient of corr(h, 1-h, k*) with respect to h
3. Take a step in the negative gradient direction, projected onto the feasible set

The gradient of corr(h, 1-h, k) with respect to h:
∂/∂h[p] Σ_i h[i]*(1-h[i+k]) = (1-h[p+k]) (if p+k is in range) - h[p-k] (if p-k is in range)

Wait, let me be more careful. Let f(h) = Σ_{i: i,i+k in [0,n)} h[i]*(1-h[i+k]).

∂f/∂h[p] = sum over all i where h[p] appears:
- If i=p and p+k in range: ∂/∂h[p] of h[p]*(1-h[p+k]) = (1-h[p+k])
- If i+k=p, i.e., i=p-k, and p-k in range: ∂/∂h[p] of h[p-k]*(1-h[p]) = -h[p-k]

So ∂f/∂h[p] = (1-h[p+k])*[p+k<n] + (-h[p-k])*[p-k>=0]
= (1-h[p+k])*[p+k<n] - h[p-k]*[p>=k]

This is the same as what I had for the binary case! Good.

For the max of multiple such functions, a subgradient is a convex combination of the gradients of the active (maximizing) functions.

def _continuous_subgrad(h, deadline, max_iters=10000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    alpha = 0.1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0-h, mode='full')
        scores = corr * dx
        max_score = np.max(scores)
        
        # Find active constraints (top few)
        threshold = np.percentile(scores, 95)
        active = np.where(scores >= threshold)[0]
        
        # Compute subgradient: average of gradients of active constraints
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n-1)  # shift
            if k == 0:
                continue
            for p in range(n):
                # ∂f_k/∂h[p]
                if k > 0:
                    if p + k < n:
                        grad[p] += (1.0 - h[p+k])
                    if p - k >= 0:
                        grad[p] -= h[p-k]
                elif k < 0:
                    d = -k
                    if p - d >= 0:
                        grad[p] += (1.0 - h[p-d])
