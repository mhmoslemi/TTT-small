Let me think about the mirror construction more carefully.

If N = 2m (even), define h such that h_i + h_{N-1-i} = 1 for i = 0, ..., N-1.
Wait, that means h_i + h_{2m-1-i} = 1. If i ranges over 0,...,2m-1, each pair (i, 2m-1-i) sums to 1.
Sum of all h_i = sum over pairs = m * 1 = m = N/2. ✓

So we can parameterize by m variables a_0, ..., a_{m-1} where:
h_i = a_i for i = 0, ..., m-1
h_{2m-1-i} = 1 - a_i for i = 0, ..., m-1

This gives us m free variables (with a_i ∈ [0,1]) and the sum constraint is automatically satisfied.

For N=95 (odd), we'd have N=2m+1 with m=47. Then:

h_m = h_47 is free but since sum = N/2 = 47.5, and the pairs contribute 47, we need h_47 = 0.5.

So for N=95: m=47 free variables a_0,...,a_46 ∈ [0,1], h_47 = 0.5, and h_{94-i} = 1-a_i.

This is a nice reduction! We go from 95 variables (with one constraint) to 47 free variables.

            current_a,

            method='L-BFGS-B',

    exp_terms = np.exp(beta * (scores - max_s))
    lse = max_s + float(np.log(np.sum(exp_terms))) / beta
    weight_sum = float(np.sum(exp_terms))
    weights = exp_terms / weight_sum
    m = n // 2
    grad = np.zeros(m)
    r_vals = np.arange(2 * n - 1)
    for i in range(m):
        p = i + r_vals - (n - 1)
        q = r_vals - i
        valid_p = (p >= 0) & (p < n)
        valid_q = (q >= 0) & (q < n)
        contrib = np.zeros(2 * n - 1)
        contrib[valid_p] = v[p[valid_p]]
        contrib[valid_q] = -v[q[valid_q]]
        grad[i] = dx * float(np.dot(weights, contrib))
    return lse, grad


def _optimize_mirror(a_init, N, deadline, beta_list):
    m = N // 2
    current_a = a_init.copy()
    best_a = current_a.copy()
    best_val = _c5_of_h(_reconstruct_mirror(current_a, N))
    for beta in beta_list:
        if time.time() > deadline:
            break

        def f(x):
            return _lse_obj_grad_mirror(x, N, beta)[0]

        def g(x):
            return _lse_obj_grad_mirror(x, N, beta)[1]

        try:
            result = minimize(f, current_a, jac=g, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * m,
                              options={'maxiter': 400, 'gtol': 1e-10})
            candidate = np.clip(result.x, 0.0, 1.0)
            c5_cand = _c5_of_h(_reconstruct_mirror(candidate, N))
            if c5_cand < best_val:
                best_val = c5_cand
                best_a = candidate
            current_a = candidate
        except Exception:
            pass
    if _c5_of_h(_reconstruct_mirror(best_a, N)) < _c5_of_h(_reconstruct_mirror(current_a, N)):
        current_a = best_a
    return current_a


def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = np.linspace(0.5 / n_new, 1.0 - 0.5 / n_new, n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project_h(h_new, n_new / 2.0)


def _line_search_vec(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _c5_of_h(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g, b_g = t_lo, t_hi
    c1 = b_g - gr * (b_g - a_g)
    c2 = a_g + gr * (b_g - a_g)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(80):
        if b_g - a_g < 1e-13:
            break
        if f1 < f2:
            b_g = c2
            c2 = c1
            f2 = f1
            c1 = b_g - gr * (b_g - a_g)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a_g = c1
            c1 = c2
            f1 = f2
            c2 = a_g + gr * (b_g - a_g)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a_g + b_g) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    base_val = _c5_of_h(h)
    if base_val < opt_val:
        return 0.0, base_val
    return opt_t, opt_val


def _lp_refine(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = _project_h(h, target)
    cur_c5 = _c5_of_h(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

    while time.time() < deadline and stall < 120:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full')
        n_lags = 2 * n - 1

        r_idx = np.arange(n_lags)[:, None]
        j_idx = np.arange(n)[None, :]
        idx1 = j_idx + r_idx - (n - 1)
        idx2 = j_idx - r_idx + (n - 1)
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        flat1 = np.where(mask1, idx1, 0)
        flat2 = np.where(mask2, idx2, 0)
        G = mask1.astype(float) * (1.0 - h[flat1]) - mask2.astype(float) * h[flat2]

        c_vec = np.concatenate([np.zeros(n), [1.0]])
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                radius *= 0.7
                stall += 1
                if radius < 1e-7:
                    break
                continue
            d = res.x[:n]
            t_opt = res.x[-1]
            if t_opt >= float(np.max(scores)) - 1e-12:
                radius *= 0.7
                stall += 1
                if radius < 1e-7:
                    break
                continue
        except Exception:
            radius *= 0.7
            stall += 1
            if radius < 1e-7:
                break
            continue

        alpha, new_c5 = _line_search_vec(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = _project_h(h + alpha * d, target)
            cur_c5 = _c5_of_h(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.05

    return best_h, best_c5


def _gradient_descent_full(h, deadline, beta_list, lr_init=0.02):
    n = len(h)
    target = n / 2.0
    h = _project_h(h, target)
    best_h = h.copy()
    best_c5 = _c5_of_h(h)
    lr = lr_init

    for beta in beta_list:
        if time.time() > deadline:
            break
        for step in range(500):
            if time.time() > deadline:
                break
            dx = 2.0 / n
            v = 1.0 - h
            corr = np.correlate(h, v, mode='full')
            scores = corr * dx
            max_s = float(np.max(scores))
            exp_terms = np.exp(beta * (scores - max_s))
            weights = exp_terms / float(np.sum(exp_terms))

            n_lags = 2 * n - 1
            r_idx = np.arange(n_lags)[:, None]
            j_idx = np.arange(n)[None, :]
            idx1 = j_idx + r_idx - (n - 1)
            idx2 = j_idx - r_idx + (n - 1)
            mask1 = (idx1 >= 0) & (idx1 < n)
            mask2 = (idx2 >= 0) & (idx2 < n)
            flat1 = np.where(mask1, idx1, 0)
            flat2 = np.where(mask2, idx2, 0)

            grad = np.zeros(n)
            w_expanded = weights
            for j in range(n):
                idx1_j = j + np.arange(n_lags) - (n - 1)
                idx2_j = j - np.arange(n_lags) + (n - 1)
                m1 = (idx1_j >= 0) & (idx1_j < n)
                m2 = (idx2_j >= 0) & (idx2_j < n)
                contrib = np.zeros(n_lags)
                contrib[m1] = v[idx1_j[m1]]
                contrib[m2] = -h[idx2_j[m2]]
                grad[j] = dx * float(np.dot(w_expanded, contrib))

            h_new = _project_h(h - lr * grad, target)
            c5_new = _c5_of_h(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_new <= _c5_of_h(h) + 1e-12:
                h = h_new
                lr = min(lr * 1.1, 0.1)
            else:
                lr *= 0.7
                if lr < 1e-6:
                    break
    return best_h, best_c5


def _sparse_coord(h, deadline, n_iter=300, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_of_h(h)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5

    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = int(interior[rng.integers(0, len(interior))])
        j = int(interior[rng.integers(0, len(interior))])
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search_vec(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-13:
            h = _project_h(h + alpha * d, target)
            cur = _c5_of_h(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
            if len(interior) < 2:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h0 = _project_h(h0, N0 / 2.0)

    best_h = h0.copy()
    best_c5 = _c5_of_h(h0)

    # Phase 1: Mirror-space LSE optimization (15%)
    d1 = t0 + budget_s * 0.15
    a0 = _extract_mirror(h0, N0)
    a_opt = _optimize_mirror(a0, N0, d1, beta_list=[30, 80, 200, 500, 1000, 2000, 5000])
    h_m = _reconstruct_mirror(a_opt, N0)
    h_m = _project_h(h_m, N0 / 2.0)
    c5_m = _c5_of_h(h_m)
    if c5_m < best_c5:
        best_c5 = c5_m
        best_h = h_m.copy()

    # Phase 2: LP refinement (20%)
    d2 = t0 + budget_s * 0.35
    h_lp, c5_lp = _lp_refine(best_h, d2, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: Gradient descent in full space (10%)
    d3 = t0 + budget_s * 0.45
    h_gd, c5_gd = _gradient_descent_full(best_h, d3, beta_list=[50, 200, 500, 2000], lr_init=0.015)
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd.copy()

    # Phase 4: Multi-resolution mirror search (20%)
    d4 = t0 + budget_s * 0.65
    n_base = len(best_h)
    triedNs = set()
    for dn in list(range(-12, 13)) + [15, 20, 25, 30, -15, -20, -25]:
        if time.time() > d4:
            break
        n_try = n_base + dn
        if n_try < 30 or n_try > 220 or n_try in triedNs:
            continue
        triedNs.add(n_try)
        h_r = _resample_h(best_h, n_try)
        a_r = _extract_mirror(h_r, n_try)
        sub_dl = min(d4, time.time() + 6.0)
        a_opt_r = _optimize_mirror(a_r, n_try, sub_dl, beta_list=[50, 200, 500, 2000])
        h_opt_r = _reconstruct_mirror(a_opt_r, n_try)
        h_opt_r = _project_h(h_opt_r, n_try / 2.0)
        c5_r = _c5_of_h(h_opt_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_opt_r.copy()
            n_base = n_try
            # Quick LP polish at new resolution
            sub_dl2 = min(d4, time.time() + 5.0)
            h_lp_r, c5_lp_r = _lp_refine(h_opt_r, sub_dl2, radius_init=0.08)
            if c5_lp_r < best_c5:
                best_c5 = c5_lp_r
                best_h = h_lp_r.copy()

    # Phase 5: Random restarts in mirror space (15%)
    d5 = t0 + budget_s * 0.80
    n_cur = len(best_h)
    m_cur = n_cur // 2
    a_cur = _extract_mirror(best_h, n_cur)
    for restart in range(40):
        if time.time() > d5:
            break
        noise_scale = max(0.03, 0.18 - 0.004 * restart)
        a_pert = a_cur + rng.normal(0, noise_scale, m_cur)
        a_pert = np.clip(a_pert, 0.0, 1.0)
        sub_dl = min(d5, time.time() + 4.0)
        a_opt_p = _optimize_mirror(a_pert, n_cur, sub_dl,
                                   beta_list=[50, 200, 1000, 5000])
        h_p = _reconstruct_mirror(a_opt_p, n_cur)
        h_p = _project_h(h_p, n_cur / 2.0)
        c5_p = _c5_of_h(h_p)
        if c5_p < best_c5 - 1e-12:
            best_c5 = c5_p
            best_h = h_p.copy()
            a_cur = a_opt_p.copy()

    # Phase 6: Final LP + coordinate refinement (15%)
    d6 = t0 + budget_s * 0.95
    h_lp2, c5_lp2 = _lp_refine(best_h, min(d6, deadline - 30.0), radius_init=0.08)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    d7 = deadline - 3.0
    best_h, best_c5 = _sparse_coord(best_h, d7, n_iter=400, rng=rng)

    # Final validation
    best_h = _project_h(best_h, len(best_h) / 2.0)
    final_c5 = _c5_of_h(best_h)

    return best_h, final_c5, len(best_h)
