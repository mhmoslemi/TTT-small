import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_solve(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    G = _grad_matrix(h)
    # Solve: min t s.t. scores + G@delta <= t, sum(delta)=0, |delta_j|<=radius
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _local_optimize(h, deadline, radius_init=0.3, max_iters=10000, stall_limit=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        p, lp_c5 = _lp_solve(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.85
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        h_new = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_new)
        if actual_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.8
            if stall > 40 and radius < 0.02:
                # Try a larger jump
                radius = 0.15
                stall = 0
    return best_h, best_c5

def _slsqp_lse(h, deadline, beta_init=50.0, beta_final=500.0, n_phases=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    def _obj(hf):
        S = _scores(hf)
        maxS = np.max(S)
        shifted = beta * (S - maxS)
        shifted = np.clip(shifted, -600, 0)
        return maxS + np.log(np.mean(np.exp(shifted))) / beta

    def _grad(hf):
        S = _scores(hf)
        maxS = np.max(S)
        shifted = beta * (S - maxS)
        shifted = np.clip(shifted, -600, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w /= wsum
        G = _grad_matrix(hf)
        return G.T @ w

    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        try:
            cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-15})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _cons_fun(x):
    return np.sum(x) - _CONS_TARGET

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp_topk(h, deadline, K=5, maxiter=300):
    """Minimize sum of top-K scores using SLSQP."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    def _obj(hf):
        S = _scores(hf)
        kth_val = np.partition(S, -K)[-K]
        top_mask = S >= kth_val - 1e-15
        return np.sum(S[top_mask])
    
    def _grad(hf):
        S = _scores(hf)
        kth_val = np.partition(S, -K)[-K]
        top_mask = S >= kth_val - 1e-15
        w = np.zeros(len(S))
        w[top_mask] = 1.0 / np.sum(top_mask)
        G = _grad_matrix(hf)
        return G.T @ w
    
    # Update global target
    global _CONS_TARGET
    _CONS_TARGET = target
    
    try:
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res.x)):
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    except Exception:
        pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, pattern='block'):
    """Create structured initial guesses."""
    if pattern == 'block':
        # Two blocks of 1's and 0's
        h = np.zeros(n)
        block_size = n // 4
        h[:block_size] = 1.0
        h[2*block_size:3*block_size] = 1.0
        return _project(h, n / 2.0)
    elif pattern == 'sine':
        x = np.arange(n) / n
        h = 0.5 + 0.5 * np.sin(2 * np.pi * 2 * x)
        return _project(h, n / 2.0)
    elif pattern == 'sawtooth':
        x = np.arange(n) / n
        h = ((x * 3.0) % 1.0)
        h = np.where(h < 0.5, 1.0, 0.0)
        return _project(h, n / 2.0)
    elif pattern == 'random':
        np.random.seed(123)
        h = (np.random.random(n) < 0.5).astype(float)
        return _project(h, n / 2.0)
    return _project(np.full(n, 0.5), n / 2.0)

def _evolve_optimize(h_best, deadline, n_gen=50, pop_size=8, mut_scale=0.03):
    """Evolutionary strategy combined with LP polishing."""
    n = len(h_best)
    target = n / 2.0
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    
    # Initialize population
    pop = [best_h.copy()]
    for i in range(pop_size - 1):
        h_p = best_h + np.random.normal(0, mut_scale, n)
        pop.append(_project(h_p, target))
    
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        # Polish each member
        for i in range(len(pop)):
            sub_dl = min(deadline, time.time() + 2.0)
            h_pol, c5_pol = _local_optimize(pop[i], sub_dl, radius_init=0.05, max_iters=200, stall_limit=30)
            if c5_pol < _exact_c5(pop[i]):
                pop[i] = h_pol
            if _exact_c5(pop[i]) < best_c5:
                best_c5 = _exact_c5(pop[i])
                best_h = pop[i].copy()
        
        # Sort by fitness
        pop.sort(key=lambda x: _exact_c5(x))
        # Keep best half, generate new from best
        keep = max(2, len(pop) // 2)
        elites = pop[:keep]
        new_pop = elites[:]
        while len(new_pop) < pop_size:
            parent = elites[np.random.randint(0, keep)]
            h_new = parent + np.random.normal(0, mut_scale, n)
            new_pop.append(_project(h_new, target))
        pop = new_pop
        # Anneal mutation
        mut_scale *= 0.97
        mut_scale = max(mut_scale, 0.005)
    
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=3, step_stride=8):
    """Polish by adjusting pairs of coordinates."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Sample pairs
        indices = list(range(n))
        np.random.shuffle(indices)
        sampled = indices[::step_stride]
        for j in sampled[:n//step_stride]:
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in sampled[j+1:j+8]:  # Try a few partners
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Golden section
                    lo_a, hi_a = 0.0, a_max
                    c1 = hi_a - gr * (hi_a - lo_a)
                    c2 = lo_a + gr * (hi_a - lo_a)
                    h1 = h.copy(); h1[j] += sign*c1; h1[k] -= sign*c1
                    h2 = h.copy(); h2[j] += sign*c2; h2[k] -= sign*c2
                    f1 = _exact_c5(h1); f2 = _exact_c5(h2)
                    for _ in range(20):
                        if hi_a - lo_a < 1e-12:
                            break
                        if f1 < f2:
                            hi_a = c2; f2 = f1; c2 = c1
                            c1 = hi_a - gr * (hi_a - lo_a)
                            h1 = h.copy(); h1[j] += sign*c1; h1[k] -= sign*c1
                            f1 = _exact_c5(h1)
                        else:
                            lo_a = c1; f1 = f2; c1 = c2
                            c2 = lo_a + gr * (hi_a - lo_a)
                            h2 = h.copy(); h2[j] += sign*c2; h2[k] -= sign*c2
                            f2 = _exact_c5(h2)
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _exact_pair_optimize(h, j, k):
    """Exactly optimize h by varying h[j] and h[k] (keeping sum fixed)."""
    n = len(h)
    dx = 2.0 / n
    a = h[j]
    b = h[k]
    c = a + b  # fixed sum
    
    # Compute d_l = ∂S_l/∂h[j] - ∂S_l/∂h[k] for each lag l
    # ∂S_l/∂h[j] = (1-h[j+l])*dx (if j+l valid) - h[j-l]*dx (if j-l valid)
    # But wait, S_l uses h(x) and h(x+k), both of which may involve index j.
    
    # More carefully: S_l = sum over valid m of h[m]*(1-h[m+lag]) * dx
    # where lag = l - (n-1) (the actual lag)
    
    # Change h[j] by +t, h[k] by -t
    # Effect on term m:
    #   If m == j: h[j]+t replaces h[j], so term becomes (h[j]+t)*(1-h[j+lag])
    #     Change: t*(1-h[j+lag])
    #   If m+lag == j: h[m]*(1-h[j]-t) replaces h[m]*(1-h[j])
    #     Change: h[m]*t = h[j-lag]*t (if j-lag valid)
    #   Similarly for k with sign reversed
    
    # So d_l (coefficient of t in S_{n-1+lag}) = 
    #   dx * [(1-h[j+lag]) - h[j-lag] - (1-h[k+lag]) + h[k-lag]]
    # with appropriate boundary handling (terms are 0 if index out of range)
    
    scores = _scores(h)
    n_lags = 2 * n - 1
    
    d = np.zeros(n_lags)
    for r in range(n_lags):
        lag = r - (n - 1)
        # Contribution from j
        term_j = 0.0
        if 0 <= j + lag < n:
            term_j += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            term_j -= h[j - lag]
        # Contribution from k (with sign flip)
        term_k = 0.0
        if 0 <= k + lag < n:
            term_k += (1.0 - h[k + lag])
        if 0 <= k - lag < n:
            term_k -= h[k - lag]
        d[r] = dx * (term_j - term_k)
    
    # Now S_l'(t) = scores[l] + t * d[l] for all l
    # Feasible range: max(0, -a) <= t <= min(1, 1-b)
    # i.e., 0 <= a+t <= 1 and 0 <= b-t <= 1
    t_min = max(0, a - 1, -(b))  # a+t >= 0, b-t >= 0
    t_max = min(1 - a, b)  # a+t <= 1, b-t <= 1
    
    # Wait: a+t >= 0 => t >= -a; b-t >= 0 => t <= b
    # a+t <= 1 => t <= 1-a; b-t <= 1 => t >= b-1
    t_min = max(-a, b - 1)
    t_max = min(1 - a, b)
    
    if t_min >= t_max:
        return h.copy(), _exact_c5(h)
    
    # Find t in [t_min, t_max] that minimizes max_l (scores[l] + t * d[l])
    # This is a 1D convex piecewise-linear minimization
    
    # The function f(t) = max_l (scores[l] + t*d[l]) is convex in t
    # (max of linear functions)
    # Minimum is where the subgradient contains 0
    
    # Find all breakpoints: t where scores[l] + t*d[l] = scores[m] + t*d[m]
    # i.e., t = (scores[m] - scores[l]) / (d[l] - d[m])
    
    # For efficiency, only consider the "upper envelope"
    # Use a ternary search or find the active set at the optimum
    
    # Simple approach: ternary search on the convex function
    # But piecewise linear → the minimum is at a breakpoint

    # The subgradient of f at t is [min_l d[l] : max_l d[l]] for the active set

2. At each point, find the active line and check if 0 is in the subgradient

Actually, the subgradient of f at t is {b_l : a_l + b_l*t = f(t)} (the set of slopes of active lines). The minimum is where 0 is in this set, i.e., there exist active lines with b_l ≤ 0 and b_l ≥ 0.

1. Start with t = (t_min + t_max) / 2
2. Find f(t) and the active set
3. If all active lines have positive slope, move left
4. If all have negative slope, move right

S_l(h) = sum_{m: both m and m+lag are in [0,n)} h[m] * (1 - h[m+lag]) * dx

where lag = r - (n-1) for score index r.

If I change h[j] → h[j] + t and h[k] → h[k] - t, the change in S_l is:

ΔS_l = dx * [sum of changes due to j] - dx * [sum of changes due to k]

Changes due to h[j] increasing by t:
- As first factor: h[j] appears in term m=j: (h[j]+t)(1-h[j+lag]) vs h[j](1-h[j+lag]), change = t(1-h[j+lag]) if j+lag valid
- As second factor: h[j] appears in term m=j-lag: h[j-lag](1-h[j]-t) vs h[j-lag](1-h[j]), change = -t*h[j-lag] if j-lag valid

So ΔS_l from j = dx * [t(1-h[j+lag]) - t*h[j-lag]] (with 0 if index invalid)
           = t * dx * [(1-h[j+lag]) - h[j-lag]]

Similarly, ΔS_l from k (with h[k] decreasing by t, so the sign flips):
ΔS_l from k = -t * dx * [(1-h[k+lag]) - h[k-lag]]

Total: ΔS_l = t * dx * [(1-h[j+lag]) - h[j-lag] - (1-h[k+lag]) + h[k-lag]]

So d_l = dx * [(1-h[j+lag]) - h[j-lag] - (1-h[k+lag]) + h[k-lag]]

import numpy as np
from scipy.optimize import linprog, minimize
import time

_CT = [64.0]

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_solve(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            a = (1.0 - h[j]) / p[j]
            if a < alpha_max:
                alpha_max = a
        elif p[j] < -1e-14:
            a = h[j] / (-p[j])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(alpha_max, 1e-12) * 0.999
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(40):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    return opt_a, _exact_c5(h + opt_a * p)

def _local_optimize(h, deadline, radius_init=0.25, max_iters=10000, stall_limit=250):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        p, _ = _lp_solve(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.85
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        h_new = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_new)
        if actual_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.45)
        else:
            stall += 1
            radius *= 0.8
            if stall > 60 and radius < 0.01:
                radius = 0.15
                stall = 0
    return best_h, best_c5

def _cons_fun(x):
    return np.sum(x) - _CT[0]

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp_lse(h, deadline, beta_init=50.0, beta_final=600.0, n_phases=5):
    n = len(h)
    target = n / 2.0
    _CT[0] = target
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    def _obj(hf):
        S = _scores(hf)
        maxS = np.max(S)
        shifted = beta * (S - maxS)
        shifted = np.clip(shifted, -600, 0)
        return maxS + np.log(np.mean(np.exp(shifted))) / beta

    def _grad(hf):
        S = _scores(hf)
        maxS = np.max(S)
        shifted = beta * (S - maxS)
        shifted = np.clip(shifted, -600, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w /= wsum
        G = _grad_matrix(hf)
        return G.T @ w

    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                          options={'maxiter': 400, 'ftol': 1e-15})
            if np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _slsqp_pnorm(h, deadline, p_init=10.0, p_final=200.0, n_phases=5):
    n = len(h)
    target = n / 2.0
    _CT[0] = target
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    def _obj(hf):
        S = _scores(hf)
        S_pos = np.maximum(S, 1e-30)
        return np.sum(S_pos ** p) ** (1.0 / p)

    def _grad(hf):
        S = _scores(hf)
        S_pos = np.maximum(S, 1e-30)
        w = S_pos ** (p - 1)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w = w / wsum * p
        G = _grad_matrix(hf)
        return G.T @ w

    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        p = p_init * (p_final / p_init) ** frac
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                          options={'maxiter': 400, 'ftol': 1e-15})
            if np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _exact_pair_optimize(h, j, k):
    """Exactly optimize by varying h[j]+=t, h[k]-=t. Returns (h_new, c5)."""
    n = len(h)
    dx = 2.0 / n
    a = h[j]
    b = h[k]
    scores = _scores(h)
    n_lags = len(scores)

    d = np.zeros(n_lags)
    for r in range(n_lags):
        lag = r - (n - 1)
        tj = 0.0
        if 0 <= j + lag < n:
            tj += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            tj -= h[j - lag]
        tk = 0.0
        if 0 <= k + lag < n:
            tk += (1.0 - h[k + lag])
        if 0 <= k - lag < n:
            tk -= h[k - lag]
        d[r] = dx * (tj - tk)

    t_min = max(-a, b - 1.0)
    t_max = min(1.0 - a, b)
    if t_min >= t_max - 1e-14:
        return h, _exact_c5(h)

    # Convex piecewise linear: f(t) = max_l (scores[l] + t*d[l])
    # Use ternary search (valid since convex)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_min, t_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = np.max(scores + c1 * d)
    f2 = np.max(scores + c2 * d)
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = np.max(scores + c1 * d)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = np.max(scores + c2 * d)
    best_t = (lo + hi) / 2.0
    h_new = h.copy()
    h_new[j] = a + best_t
    h_new[k] = b - best_t
    h_new = np.clip(h_new, 0.0, 1.0)
    c5_val = np.max(scores + best_t * d)
    return h_new, c5_val

def _pair_polish_exact(h, deadline, max_passes=5, stride=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:

        improved = False
        step = max(1, n // stride)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + step, min(j + 20 * step, n), step):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:

                h_try, c5_try = _exact_pair_optimize(h, j, k)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        dl7 = t0 + budget_s * 0.94
        h7, c5_7 = _slsqp_lse(best_h, dl7, beta_init=200.0, beta_final=1000.0, n_phases=3)
        if c5_7 < best_c5:
            best_c5 = c5_7; best_h = h7.copy()

    # Phase 8: Exact pair polish
    if time.time() < t0 + budget_s * 0.975:
        hp, c5p = _pair_polish_exact(best_h, t0 + budget_s * 0.98, max_passes=6, stride=6)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
