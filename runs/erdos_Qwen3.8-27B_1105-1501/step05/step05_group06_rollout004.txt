import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    """Build the matrix G where G[r, j] gives d(score[r])/d(h[j]) * dx"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        # score[r] = dx * sum_{i=0}^{m-1} h[i] * (1-h[i+d])
        # d/dh[j] for j in [0, m-1]: dx * (1-h[j+d])
        # d/dh[j] for j in [d, n-1]: -dx * h[j-d]
        G[r, :m] += dx * (1.0 - h[d:d+m])
        G[r, d:] += -dx * h[:m]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        # score[r] = dx * sum_{i=0}^{m-1} h[i+s] * (1-h[i])
        # d/dh[j] for j in [0, m-1]: -dx * h[j+s]
        # d/dh[j] for j in [s, n-1]: dx * (1-h[j-s])
        G[r, :m] += -dx * h[s:s+m]
        G[r, s:] += dx * (1.0 - h[:m])
    return G

def _local_opt(h, deadline, top_k_list=None, radius_init=0.1, max_iters=3000):
    """LP-based local optimization"""
    if top_k_list is None:
        top_k_list = [None, 10, 6, 4]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_tk = len(top_k_list)
    ti = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        tk = top_k_list[ti % n_tk]
        G = _build_G(h)
        scores = _scores(h)
        n_lags = 2 * n - 1
        if tk is not None and tk < n_lags:
            idx = np.argsort(scores)[-tk:]
            s_k = scores[idx]
            G_k = G[idx]
        else:
            s_k = scores
            G_k = G
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(s_k), n + 1))
        A_ub[:, :n] = G_k
        A_ub[:, n] = -1.0
        b_ub = -s_k
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
            else:
                p = None
        except Exception:
            p = None
        
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-8:
                break
            continue
        
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(25):
            if hi_a - lo_a < 1e-14:
                break
            if f1 < f2:
                hi_a, f2, c2 = c2, f1, c1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo_a, f1, c1 = c2, f2, c2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(_project(h + c2 * p, target))
        opt_a = (lo_a + hi_a) / 2.0
        h_cand = _project(h + opt_a * p, target)
        actual_c5 = _exact_c5(h_cand)
        
        if actual_c5 < cur_c5 - 1e-13:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 15 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=200):
    """Coordinate pair descent"""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        scores = _scores(h)
        G = _build_G(h)
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if abs(alpha) > 1e-10:
                    pred = float(np.max(scores + alpha * C))
                    if cur_c5 - pred > 1e-8:
                        h[j] += alpha
                        h[k] -= alpha
                        actual_c5 = _exact_c5(h)
                        if actual_c5 < cur_c5 - 1e-14:
                            cur_c5 = actual_c5
                            pass_improved = True
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            scores = _scores(h)
                        else:
                            h[j] -= alpha
                            h[k] += alpha
        if not pass_improved:
            break
    return best_h, best_c5

def _de_step(h, deadline, pop_size=30, n_gens=50, F=0.8, CR=0.7):
    """Differential evolution step"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    # Initialize population centered around h
    pop = np.zeros((pop_size, n))
    pop[0] = h
    for i in range(1, pop_size):
        pop[i] = _project(h + 0.1 * np.random.randn(n), target)
    
    pop_c5 = np.array([_exact_c5(pop[i]) for i in range(pop_size)])
    
    for gen in range(n_gens):
        if time.time() > deadline:
            break
        
        # Sort population
        order = np.argsort(pop_c5)
        for rank, i in enumerate(order):
            if rank == 0:
                continue
            # Select 3 distinct random individuals
            candidates = [x for x in order if x != i]
            a, b, c = np.random.choice(len(candidates), 3, replace=False)
            a, b, c = candidates[a], candidates[b], candidates[c]
            
            # Mutation
            v = pop[order[a]] + F * (pop[order[b]] - pop[order[c]])
            # Crossover
            trial = np.copy(pop[order[c]])
            mask = np.random.random(n) < CR
            trial[mask] = v[mask]
            # Ensure at least one gene is mutated
            if not mask.any():
                j = np.random.randint(n)
                trial[j] = v[j]
            trial = _project(trial, target)
            
            trial_c5 = _exact_c5(trial)
            
            if trial_c5 < pop_c5[order[i]]:
                pop[order[i]] = trial
                pop_c5[order[i]] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    cur_c5 = trial_c5
        
        # Replace worst with best (elitism is already maintained by tournament)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_init(n, rng, alpha=None):
    """Generate a random initial solution"""
    if alpha is None:
        alpha = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113][rng.integers(4)]
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)

            # So constraint on upper: delta <= (n-1) * min_{i≠j, h[i]>0} h[i] (if any positive)
            # Actually this is complicated. Let's just use pair moves.
