from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = v.copy()
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj_and_grad(x, tau):
    """Compute log-sum-exp smooth objective and its gradient."""
    n = len(x)
    dx = 2.0 / n
    scores = _scores(x)
    s_max = np.max(scores)
    # log-sum-exp
    shifted = (scores - s_max) / tau
    max_shifted = np.max(shifted)
    log_sum_exp = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    obj = s_max + tau * log_sum_exp
    
    # weights
    weights = np.exp(shifted - max_shifted)
    weights /= np.sum(weights)
    
    # gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0 and weights[r] > 1e-15:

def _bin_c5(b, n):
    """Compute C5 for binary sequence represented as integer b."""
    best = 0
    for d in range(1, n):
        mask = (1 << (n - d)) - 1
        lower = b & mask
        shifted = (b >> d) & mask
        c_pos = (lower & ~shifted).bit_count()
        c_neg = (shifted & ~lower).bit_count()
        mx = max(c_pos, c_neg)
        if mx > best:
            best = mx
    return 2.0 * best / n

    # This is a quick check - just recompute full C5 for simplicity

def _binary_sa(b, n, deadline, rng, n_iters=100000):
    """Simulated annealing on binary sequences."""
    K = n // 2
    cur_c5 = _bin_c5(b, n)
    best_c5 = cur_c5
    best_b = b
    
    T = 0.02
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    ones = set()
    zeros = set()
    for i in range(n):
        if (b >> i) & 1:
            ones.add(i)
        else:
            zeros.add(i)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        new_b = b ^ (1 << i) ^ (1 << j)
        new_c5 = _bin_c5(new_b, n)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            b = new_b
            ones.discard(i)
            zeros.discard(j)
            ones.add(j)
            zeros.add(i)
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b
        T *= cool
    
    # Convert to array
    best_arr = np.zeros(n)
    for i in range(n):
        if (best_b >> i) & 1:
            best_arr[i] = 1.0
    return best_arr, best_c5

def _binary_hill(b, n, deadline, rng, max_iters=50000):

        return np.zeros(n)

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float).copy()
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _bin_c5(b, n):
    best = 0
    for d in range(1, n):
        mask = (1 << (n - d)) - 1
        lower = b & mask
        shifted = (b >> d) & mask
        c_pos = (lower & ~shifted).bit_count()
        c_neg = (shifted & ~lower).bit_count()
        mx = c_pos if c_pos > c_neg else c_neg
        if mx > best:
            best = mx
    return 2.0 * best / n

def _bin_c5_lag(b, n, d, mask_cache):
    mask = mask_cache[d]
    lower = b & mask
    shifted = (b >> d) & mask
    c_pos = (lower & ~shifted).bit_count()
    c_neg = (shifted & ~lower).bit_count()
    return max(c_pos, c_neg)

def _binary_sa(b, n, deadline, rng, n_iters=200000):
    K = n // 2
    rc = b.bit_count()
    if rc != K:
        ones_pos = [i for i in range(n) if (b >> i) & 1]
        zeros_pos = [i for i in range(n) if not (b >> i) & 1]
        if rc > K:
            for _ in range(rc - K):
                i = ones_pos.pop(rng.integers(len(ones_pos)))
                j = zeros_pos.pop(rng.integers(len(zeros_pos)))
                b ^= (1 << i) ^ (1 << j)
        else:
            for _ in range(K - rc):
                i = zeros_pos.pop(rng.integers(len(zeros_pos)))
                j = ones_pos.pop(rng.integers(len(ones_pos)))
                b ^= (1 << i) ^ (1 << j)
    cur_c5 = _bin_c5(b, n)
    best_c5 = cur_c5
    best_b = b
    T = 0.015
    T_min = 0.00005
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    mask_cache = [(1 << (n - d)) - 1 for d in range(n)]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        bit_i = (b >> i) & 1
        bit_j = (b >> j) & 1
        if bit_i == bit_j:
            continue
        new_b = b ^ (1 << i) ^ (1 << j)
        new_c5 = _bin_c5(new_b, n)
        if new_c5 <= cur_c5:
            accept = True
        else:
            delta = new_c5 - cur_c5
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            b = new_b
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b
        T *= cool
    arr = np.zeros(n)
    for i in range(n):
        if (best_b >> i) & 1:
            arr[i] = 1.0
    return arr, best_c5

def _binary_hill(b, n, deadline, rng, max_iters=100000):
    cur_c5 = _bin_c5(b, n)
    best_c5 = cur_c5
    best_b = b
    ones_list = [i for i in range(n) if (b >> i) & 1]
    zeros_list = [i for i in range(n) if not (b >> i) & 1]
    stall = 0
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        if stall > 200:
            break
        best_impr = 0.0
        best_move = None
        n_tries = min(len(ones_list) * len(zeros_list), 100)
        for _ in range(n_tries):
            i = int(ones_list[rng.integers(len(ones_list))])
            j = int(zeros_list[rng.integers(len(zeros_list))])
            new_b = b ^ (1 << i) ^ (1 << j)
            new_c5 = _bin_c5(new_b, n)
            impr = cur_c5 - new_c5
            if impr > best_impr:
                best_impr = impr
                best_move = (i, j, new_b)
        if best_move is not None:
            i, j, new_b = best_move
            b = new_b
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            cur_c5 -= best_impr
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                ni = int(ones_list[rng.integers(len(ones_list))])
                nj = int(zeros_list[rng.integers(len(zeros_list))])
                b ^= (1 << ni) ^ (1 << nj)
                ones_list.remove(ni)
                zeros_list.remove(nj)
                ones_list.append(nj)
                zeros_list.append(ni)
                cur_c5 = _bin_c5(b, n)
    arr = np.zeros(n)
    for i in range(n):
        if (best_b >> i) & 1:
            arr[i] = 1.0
    return arr, best_c5

def _smooth_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    scores = _scores(x)
    s_max = float(np.max(scores))
    shifted = (scores - s_max) / tau
    max_sh = float(np.max(shifted))
    exp_s = np.exp(shifted - max_sh)
    sum_exp = float(np.sum(exp_s))
    if sum_exp < 1e-300:
        w = np.zeros_like(scores)
        w[int(np.argmax(scores))] = 1.0
    else:
        w = exp_s / sum_exp
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        d = r - (n - 1)
        if d == 0:
            grad += wr * dx * (1.0 - 2.0 * x)
        elif d > 0:
            m = n - d
            if m > 0:
                grad[:m] += wr * dx * (1.0 - x[d:d + m])
                grad[d:] -= wr * dx * x[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:g + m] += wr * dx * (1.0 - x[:m])
                grad[:m] -= wr * dx * x[g:g + m]
    return grad

def _smooth_optimize(h, deadline, max_inner=150):
    n = len(h)
    target = n / 2.0
    cur = _exact_c5(h)
    taus = [0.04 * cur, 0.015 * cur, 0.006 * cur, 0.002 * cur, 0.0008 * cur]
    best_h = h.copy()
    best_c5 = cur
    x = h.copy()
    for tau in taus:
        if time.time() > deadline:
            break
        prev_c5 = _exact_c5(x)
        for step in range(max_inner):
            if time.time() > deadline:
                break
            grad = _smooth_grad(x, tau)
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-15:
                break
            alpha = 0.05 / g_norm
            x_try = _project(x - alpha * grad, target)
            c5_try = _exact_c5(x_try)
            if c5_try < prev_c5:
                x = x_try
                prev_c5 = c5_try
                if prev_c5 < best_c5:
                    best_c5 = prev_c5
                    best_h = x.copy()
            else:
                alpha *= 0.5
                x_try2 = _project(x - alpha * grad, target)
                c5_try2 = _exact_c5(x_try2)
                if c5_try2 < prev_c5:
                    x = x_try2
                    prev_c5 = c5_try2
                    if prev_c5 < best_c5:
                        best_c5 = prev_c5
                        best_h = x.copy()
                else:
                    break
    return best_h, best_c5

def _lp_direction(h, scores, n_active, radius):
    n = len(h)
    G = np.zeros((n_active, n))
    for idx in range(n_active):
        r = int(idx)
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    G *= (2.0 / n)
    idx_top = np.argsort(scores)[-n_active:]
    scores_top = scores[idx_top]
    G_top = G[idx_top]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G_top
    A_ub[:, n] = -1.0
    b_ub = -scores_top
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.2, max_it=3000, n_active=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    radius = radius_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, n_active, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        # Line search
        alpha_max = 0.3
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, alpha_max
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(20):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi, f2 = c2, f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo, f1 = c1, f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        opt_a = (lo + hi) / 2.0
        h_new = h + opt_a * p
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _arr_to_bin(arr, n):
    b = 0
    for i in range(n):
        if arr[i] > 0.5:
            b |= (1 << i)
    return b

def _make_balanced(b, n):
    K = n // 2
    rc = b.bit_count()
    if rc == K:
        return b
    ones_pos = [i for i in range(n) if (b >> i) & 1]
    zeros_pos = [i for i in range(n) if not (b >> i) & 1]
    if rc > K:
        for _ in range(rc - K):
            if not ones_pos or not zeros_pos:
                break
            i = ones_pos.pop()
            j = zeros_pos.pop()
            b ^= (1 << i) ^ (1 << j)
    else:
        for _ in range(K - rc):
            if not ones_pos or not zeros_pos:
                break
            i = zeros_pos.pop()
            j = ones_pos.pop()
            b ^= (1 << i) ^ (1 << j)
    return b
