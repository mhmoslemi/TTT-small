                h_try2 = project_box_sum(res2.x[:n], target)
                c_try2 = eval_c5(h_try2)
                if c_try2 < best_c5 - 1e-12:
                    best_c5 = c_try2
                    best_h = h_try2.copy()
        except Exception:
            pass

    return best_h, best_c5

def lp_direction(h, scores, G, radius, n_active=5):
    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_direction_relaxed(h, scores, G, radius, n_active=5):
    n = len(h)
    n_lags = 2 * n - 1
    c0 = float(np.max(scores))
    mask = scores >= c0 - 0.005
    if np.sum(mask) < n_active:
        sorted_idx = np.argsort(scores)[::-1]
        mask[sorted_idx[:n_active]] = True
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    dx_val = 2.0 / n
    b_ub = np.full(n_lags, -scores + radius * dx_val)
    b_ub[mask] = -scores[mask]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None

def exact_line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    n_grid = 80
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 5
    while time.time() < deadline:
        if stall > 60:
            break
        scores = lag_values(h)
        G = lag_jacobian(h)
        p = lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            n_active_use = min(n_active + 5, 2 * n - 2)
            p = lp_direction_relaxed(h, scores, G, radius, n_active_use)
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                n_active = min(n_active + 3, 2 * n - 2)
                if n_active > 2 * n - 3:
                    p = lp_direction_relaxed(h, scores, G, radius, max(3, (2 * n - 1) // 4))
                if p is None or float(np.max(np.abs(p))) < 1e-13:
                    stall += 2
                    radius *= 0.75
                    if radius < 1e-6:
                        break
                    if stall % 15 == 0:
                        radius = max(radius, 0.04)
                    continue
        alpha, new_c5 = exact_line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 5
            radius = min(radius * 1.08, 0.35)
        else:
            stall += 1
            radius *= 0.78
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.08
                n_active = 8
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def multi_resolution(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    tried = set()
    for trial in range(60):
        if time.time() > deadline:
            break
        dn = rng.integers(-20, 25)
        n_try = n + int(dn)
        if n_try < 30 or n_try > 220 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample(best_h, n_try)
        c5_r = eval_c5(h_r)
        sub_dl = min(deadline, time.time() + 4.0)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.1)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def perturb_search(h, deadline, rng, base_scale=0.05):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    for _ in range(400):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.20, scale * 1.08)
    return best_h, best_c5

def coord_refine(h, deadline, n_iter=300, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    if rng is None:
        rng = np.random.default_rng(777)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = int(interior[rng.integers(0, len(interior))])
        j = int(interior[rng.integers(0, len(interior))])
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = exact_line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def budget_remain(deadline):
    return max(0.0, deadline - time.time())

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: Quick epigraph SLSQP from initial point (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = epigraph_slsqp(h, d1, max_iter=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP-based descent (15%)
    d2 = t0 + budget_s * 0.23
    h2, c5_2 = lp_loop(best_h, d2, radius_init=0.15)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smoothing-based exploration (15%)
    d3 = t0 + budget_s * 0.38
    h3, c5_3 = smooth_descent(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.58
    best_h, best_c5 = multi_resolution(best_h, d4, rng)

    # Phase 5: Epigraph SLSQP from best so far (10%)
    d5 = t0 + budget_s * 0.68
    h5, c5_5 = epigraph_slsqp(best_h, d5, max_iter=50)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Perturbation + LP (17%)
    d6 = t0 + budget_s * 0.85
    best_h, best_c5 = perturb_search(best_h, d6, rng, base_scale=0.04)

    # Phase 7: Coordinate refinement (7%)
    d7 = t0 + budget_s * 0.92
    best_h, best_c5 = coord_refine(best_h, d7, n_iter=200, rng=rng)

    # Phase 8: Final epigraph + LP polish (6%)
    d8 = t0 + budget_s * 0.96
    h8, c5_8 = epigraph_slsqp(best_h, d8, max_iter=30)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final polish
    rem = time.time()
    if hard_deadline - rem > 3:
        h_f, c5_f = lp_loop(best_h, hard_deadline - 2, radius_init=0.05)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
