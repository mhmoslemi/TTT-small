def _grad_matrix_fast(h):
    """Compute gradient matrix G where G[r,j] = d(score_r)/d(h_j) * dx."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    # For lag d >= 0: score = sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
    # d/dh[j]: (1-h[j+d]) if j < n-d, minus h[j-d] if j >= d
    for d in range(1, n):
        m = n - d
        # Positive lag
        r_pos = n - 1 + d
        #Indices where j < m: contribute (1-h[j+d])
        G[r_pos, :m] += (1.0 - h[d:])
        # Indices where j >= d: contribute -h[j-d]
        G[r_pos, d:] -= h[:m]
        # Negative lag (same formula by symmetry of the problem structure)
        r_neg = n - 1 - d
        # score_{-d} = sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        # d/dh[j]: (1-h[j-d]) if j >= d, minus h[j+d] if j < m
        G[r_neg, d:] += (1.0 - h[:m])
        G[r_neg, :m] -= h[d:]
    # Lag 0: score = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
    # d/dh[j] = 1 - 2*h[j]
    G[n-1, :] = 1.0 - 2.0 * h
    G *= dx
    return G

def _lp_optimize(h, deadline, radius_init=0.3, max_it=5000, use_topk=None):
    """LP-based local optimizer."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_lags = 2 * n - 1
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = _scores_vec(h)
        G = _grad_matrix_fast(h)
        
        # Optionally use only top-k constraints
        if use_topk and use_topk < n_lags:
            idx = np.argsort(scores)[-use_topk:]
            scores_k = scores[idx]
            G_k = G[idx]
        else:
            scores_k = scores
            G_k = G
        nk = len(scores_k)
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((nk, n + 1))
        A_ub[:, :n] = G_k
        A_ub[:, n] = -1.0
        b_ub = -scores_k
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lb = max(-radius, -h[j])
            ub = min(radius, 1.0 - h[j])
            bounds.append((lb, ub))
        bounds.append((-1.0, 1.0))
        p = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        
        # Line search (golden section)
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max * 0.99, 1e-12)
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, alpha_max
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(30):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, target))
        opt_a = (lo + hi) / 2.0
        h_new = _project(h + opt_a * p, target)
        new_c5 = _exact_c5(h_new)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.45)
        else:
            stall += 1
            radius *= 0.72
    return best_h, best_c5

def _binary_flip_optimize(h, deadline, rng, n_iters=50000, T_init=0.01, T_min=0.0005):
    """Optimize by flipping individual bits (binary case) or small adjustments."""
    n = len(h)
    target = n / 2.0
    # First make it binary if close
    h = h.copy()
    # Threshold to binary
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly target ones
    n_ones = int(np.sum(h_bin))
    if n_ones > int(target):
        # Remove some ones (prefer those with h closest to 0.5)
        ones = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones])  # most "uncertain" first
        for i in range(n_ones - int(target)):
            h_bin[ones[order[i]]] = 0.0
    elif n_ones < int(target):
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])  # most "uncertain" first
        for i in range(int(target) - n_ones):
            h_bin[zeros[order[i]]] = 1.0
    
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    T = T_init
    cool = (T_min / T_init) ** (1.0 / max(n_iters, 1))
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Pair swap: pick one 1 and one 0, swap them
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = _exact_c5(h_bin)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        T *= cool
    return best_h, best_c5

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _subgrad_optimize(h, deadline, gamma_init=10, gamma_max=500, max_it=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = _scores_vec(h)
        max_s = np.max(scores)
        w = np.exp(gamma * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix_fast(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        # Line search
        alpha = 0.1
        cur_c5_val = cur_c5
        best_c5_step = cur_c5_val
        best_h_step = h.copy()
        for _ in range(20):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5_step:
                best_c5_step = c5_new
                best_h_step = h_new
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if best_c5_step < cur_c5 - 1e-13:
            h = best_h_step
            cur_c5 = best_c5_step
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.03, gamma_max)
        else:
            stall += 1
            gamma = min(gamma * 1.06, gamma_max)
    return best_h, best_c5
