def _line_search(h, d, dx, trust_radius):
    """Exact line search for the max-of-quadratics objective along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section + grid
    def phi(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    
    # Grid search to bracket
    n_grid = 32
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([phi(t) for t in grid])
    best_i = int(np.argmin(vals))
    best_t = grid[best_i]
    best_val = vals[best_i]
    
    # Golden section refine
    if best_i > 0: a = grid[max(0, best_i - 1)]
    else: a = t_lo
    if best_i < n_grid - 1: b = grid[min(n_grid - 1, best_i + 1)]
    else: b = t_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    f2 = phi(c2)
    for _ in range(40):

            c1 = b - gr * (b - a)
            f1 = phi(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = phi(c2)
    opt_t = (a + b) / 2.0
    opt_val = phi(opt_t)
    if opt_val < best_val:
