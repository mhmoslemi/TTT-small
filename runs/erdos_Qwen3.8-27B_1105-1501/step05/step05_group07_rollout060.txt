    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

            m = n - g

import numpy as np
from scipy.optimize import linprog, minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:n] += (1.0 - h[:m])
                G[r, :m] += -h[d:n]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:n])
                G[r, g:n] += -h[:m]
    return G * (2.0 / n)


def _lp_dir(h, scores, G, radius, n_active=None):
    n = len(h)
    n_lags = len(scores)
    if n_active is not None and n_active < n_lags:
        idx = np.argsort(-scores)[:n_active]
        G_sub = G[idx, :]
        s_sub = scores[idx]
    else:
        G_sub = G
        s_sub = scores
    ns = len(s_sub)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((ns, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -s_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    n_samp = 15
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_optimize(h, deadline, radius_init=0.2, max_iters=10000, n_active=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_dir(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.82
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.83
            if stall > 50 and radius < 0.02:
                radius = 0.10
                stall = 0
    return best_h, best_c5


def _coord_descend(h, n_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    scores = _scores(h)
    for _pass in range(n_passes):
        G = _grad_matrix(h)
        improved = False
        for j in range(n):
            a = G[:, j]
            b = scores - a * h[j]
            t_lo = float(np.min(b)) - 0.005
            t_hi = float(np.maximum(np.max(b), np.max(a + b))) + 0.005
            x_opt = h[j]
            for _ in range(40):
                if t_hi - t_lo < 1e-12:
                    break
                t_mid = (t_lo + t_hi) / 2.0
                pos = a > 1e-14
                neg = a < -1e-14
                zero = ~(pos | neg)
                if np.any(zero) and float(np.max(b[zero])) > t_mid:
                    t_lo = t_mid
                    continue
                lb = 0.0
                if np.any(neg):
                    lb = max(lb, float(np.max((t_mid - b[neg]) / a[neg])))
                ub = 1.0
                if np.any(pos):
                    ub = min(ub, float(np.min((t_mid - b[pos]) / a[pos])))
                if lb <= ub + 1e-12:
                    t_hi = t_mid
                    x_opt = float(np.clip(h[j], lb, ub))
                else:
                    t_lo = t_mid
            delta = x_opt - h[j]
            if abs(delta) > 1e-11:
                h[j] = x_opt
                scores += a * delta
                improved = True
        if not improved:
            break
    h = _project(h, target)
    return h


def _slsqp_focused(h, deadline, beta_init=80.0, beta_final=350.0, n_phases=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    def make_obj(beta_val):
        def _obj(hf):
            S = _scores(hf)
            b = beta_val
            mx = np.max(S)
            shifted = b * (S - mx)
            shifted = np.clip(shifted, -500, 0)
            return mx + np.log(np.mean(np.exp(shifted))) / b
        return _obj

    def make_grad(beta_val):
        def _grad(hf):
            S = _scores(hf)
            b = beta_val
            mx = np.max(S)
            shifted = b * (S - mx)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            wsum = np.sum(w)
            if wsum < 1e-300:
                w = np.zeros(len(S))
                w[int(np.argmax(S))] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w
        return _grad

    def sum_fun(x):
        return np.sum(x) - target

    def one_vec(x):
        return np.ones(len(x))

    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        try:
            res = minimize(make_obj(beta), h, jac=make_grad(beta),
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': one_vec}],
                           options={'maxiter': 150, 'ftol': 1e-15})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _kronecker_init(n, alpha, density=0.5):
    x = np.arange(n) * alpha
    h = ((x % 1.0) < density).astype(float)
    return _project(h, n / 2.0)


def _pair_polish(h, deadline, max_passes=3):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 40)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(7):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def _subgrad_optimize(h, deadline, n_steps=2000, beta_start=10.0, beta_end=60.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.08
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:

                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break

    return best_h, best_c5

    # Phase 7: Final polish
    if time.time() < t0 + budget_s * 0.96:
        h7, c5_7 = _lp_optimize(best_h, t0 + budget_s * 0.96, radius_init=0.02,
                                 max_iters=3000, n_active=20)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    h_f = _coord_descend(best_h, n_passes=150)
    c5_f = _exact_c5(h_f)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    if time.time() < t0 + budget_s * 0.985:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.985, max_passes=2)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
