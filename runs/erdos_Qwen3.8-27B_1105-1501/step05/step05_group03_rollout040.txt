def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _bit_c5(X, n):
    """Compute C5 for binary sequence encoded as integer X (bit i = h[i])."""
    max_count = 0
    for d in range(1, n):
        mask = (1 << (n - d)) - 1
        A = X & mask
        B = (X >> d) & mask
        c_pos = (A & (mask ^ B)).bit_count()
        c_neg = ((mask ^ A) & B).bit_count()
        m = c_pos if c_pos > c_neg else c_neg
        if m > max_count:
            max_count = m
    return max_count * 2.0 / n

def _bit_c5_fast(X, n, masks, popcache=None):
    """Faster version with precomputed masks."""
    max_count = 0
    for d in range(1, n):
        mask = masks[d]
        A = X & mask
        B = (X >> d) & mask
        c_pos = (A & (mask ^ B)).bit_count()
        if c_pos > max_count:
            max_count = c_pos
        c_neg = ((mask ^ A) & B).bit_count()
        if c_neg > max_count:
            max_count = c_neg
    return max_count * 2.0 / n

def _make_masks(n):

def _bits_to_int(bits, n):

        if bits[i]:
            X |= (1 << i)
    return X

        if time.time() > deadline:
            break

        if restart == 0 and initial_X is not None:
            X = initial_X
        else:
            # Random initialization with quasi-periodic structure
            if rng.random() < 0.5:
                # Quasi-periodic
                alpha = rng.choice([0.6180339887, 0.4142135624, 0.3819660113, 0.7071067812])
                bits = np.zeros(n, dtype=int)
                for i in range(n):
                    bits[i] = 1 if ((i + 0.5) * alpha * 2.0) % 1.0 < 0.5 else 0
                # Adjust to have exactly n/2 ones
                ones = bits.sum()
                if ones > target_ones:
                    # Remove excess ones
                    excess = ones - target_ones
                    one_positions = np.where(bits == 1)[0]
                    to_remove = rng.choice(one_positions, size=excess, replace=False)
                    bits[to_remove] = 0
                elif ones < target_ones:
                    deficit = target_ones - ones
                    zero_positions = np.where(bits == 0)[0]
                    to_add = rng.choice(zero_positions, size=deficit, replace=False)
                    bits[to_add] = 1
                X = _bits_to_int(bits, n)
            else:

                bits = np.zeros(n, dtype=int)
                positions = rng.permutation(n)[:target_ones]
                bits[positions] = 1
                X = _bits_to_int(bits, n)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_X = X

        T_min = 0.001
        n_iters = max(50000, int((deadline - time.time()) * 5000 / n_restarts))
        cool = (T_min / T) ** (1.0 / max(n_iters, 1))
        
        ones_pos = np.where([(X >> i) & 1 for i in range(n)] == 1)[0]
        zeros_pos = np.ones(n, dtype=bool)
        zeros_pos[ones_pos] = False
        zeros_pos = np.where(zeros_pos)[0]

        for it in range(n_iters):
            if time.time() > deadline:
                break

            i = ones_pos[rng.integers(len(ones_pos))]
            j = zeros_pos[rng.integers(len(zeros_pos))]

            X_new = X ^ (1 << i) ^ (1 << j)
            new_c5 = _bit_c5_fast(X_new, n, masks)

                # Update positions

                ones_pos = np.append(ones_pos, j)
                zeros_pos = np.delete(zeros_pos, np.where(zeros_pos == j))
                zeros_pos = np.append(zeros_pos, i)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_X = X
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                stagnation += 1
            
            T *= cool

            if stagnation > 2000 and it % 100 == 0:

                n_rand = min(5, len(ones_pos), len(zeros_pos))

                    zeros_pos = np.append(zeros_pos, ii)
                cur_c5 = _bit_c5_fast(X, n, masks)
                stagnation = 0

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _make_masks(n):
    return [(1 << (n - d)) - 1 for d in range(1, n)]

def _bit_max_count(X, n, masks, threshold=None):
    max_count = 0
    for d in range(1, n):
        mask = masks[d - 1]
        A = X & mask
        B = (X >> d) & mask
        c_pos = (A & (mask ^ B)).bit_count()
        if c_pos > max_count:
            max_count = c_pos
            if threshold is not None and max_count > threshold:
                return max_count
        c_neg = ((mask ^ A) & B).bit_count()
        if c_neg > max_count:
            max_count = c_neg
            if threshold is not None and max_count > threshold:
                return max_count
    return max_count

def _bits_to_X(bits):
    X = 0
    n = len(bits)
    for i in range(n):
        if bits[i]:
            X |= (1 << i)
    return X

def _X_to_bits(X, n):
    return np.array([(X >> i) & 1 for i in range(n)], dtype=float)

def _random_binary(n, rng):
    target = n // 2
    bits = np.zeros(n, dtype=int)
    positions = rng.permutation(n)[:target]
    bits[positions] = 1
    return _bits_to_X(bits)

def _quasi_binary(n, alpha, rng):
    bits = np.zeros(n, dtype=int)
    for i in range(n):
        bits[i] = 1 if ((i + 0.5) / n * alpha * 2.0) % 1.0 < 0.5 else 0
    ones = bits.sum()
    target = n // 2
    if ones > target:
        pos = np.where(bits == 1)[0]
        rem = rng.choice(pos, size=ones - target, replace=False)
        bits[rem] = 0
    elif ones < target:
        pos = np.where(bits == 0)[0]
        add = rng.choice(pos, size=target - ones, replace=False)
        bits[add] = 1
    return _bits_to_X(bits)

def _from_continuous_to_binary(h, rng):
    n = len(h)
    target = n // 2
    scores = np.zeros(n)
    for i in range(n):
        scores[i] = h[i]
    order = np.argsort(-scores)
    bits = np.zeros(n, dtype=int)
    bits[order[:target]] = 1
    return _bits_to_X(bits)

def _local_search(X, n, masks, deadline, max_iters=200):
    cur_count = _bit_max_count(X, n, masks)
    best_X = X
    best_count = cur_count
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        cur_ones = [(X >> i) & 1 for i in range(n)]
        ones = [i for i in range(n) if cur_ones[i]]
        zeros = [i for i in range(n) if not cur_ones[i]]
        best_delta = 0
        best_swap = None
        for i in ones:
            for j in zeros:
                X_try = X ^ (1 << i) ^ (1 << j)
                try_count = _bit_max_count(X_try, n, masks, threshold=cur_count - 1)
                if try_count < cur_count:
                    delta = cur_count - try_count
                    if delta > best_delta:
                        best_delta = delta
                        best_swap = (i, j)
                    if best_delta >= 2:
                        break
            if best_delta >= 2:
                break
        if best_swap is not None:
            i, j = best_swap
            X = X ^ (1 << i) ^ (1 << j)
            cur_count = _bit_max_count(X, n, masks)
            if cur_count < best_count:
                best_count = cur_count
                best_X = X
        else:
            break
    return best_X, best_count

def _sa_phase(n, masks, deadline, rng, init_X=None, n_restarts=4):
    target = n // 2
    best_X = 0
    best_count = 10**9
    alphas = [0.6180339887, 0.4142135624, 0.3819660113, 0.7071067812, 0.5]
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart == 0 and init_X is not None:
            X = init_X
        elif restart < len(alphas):
            X = _quasi_binary(n, alphas[restart % len(alphas)], rng)
        else:
            X = _random_binary(n, rng)
        cur_count = _bit_max_count(X, n, masks)
        if cur_count < best_count:
            best_count = cur_count
            best_X = X
        dl = min(deadline, time.time() + max(5, (deadline - t0_global) / n_restarts))
        T = float(cur_count) * 0.3 + 2.0
        T_min = 0.05
        total_iters = max(1000, int((dl - time.time()) * 20000))
        cool = (T_min / T) ** (1.0 / max(total_iters, 1))
        ones_list = [i for i in range(n) if (X >> i) & 1]
        zeros_list = [i for i in range(n) if not ((X >> i) & 1)]
        stagnation = 0
        local_check = 0
        for it in range(total_iters):
            if time.time() > dl:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            X_new = X ^ (1 << i) ^ (1 << j)
            new_count = _bit_max_count(X_new, n, masks, threshold=cur_count)
            delta = new_count - cur_count
            if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / max(T, 1e-9))):
                X = X_new
                cur_count = new_count
                idx_i = ones_list.index(i)
                ones_list[idx_i] = j
                zeros_list[zeros_list.index(j)] = i
                if cur_count < best_count:
                    best_count = cur_count
                    best_X = X
                    stagnation = 0
                else:
                    stagnation += 1
                local_check += 1
                if local_check >= 500 and stagnation > 200:
                    ls_X, ls_count = _local_search(X, n, masks, min(dl, time.time() + 2))
                    if ls_count < cur_count:
                        X = ls_X
                        cur_count = ls_count
                        if cur_count < best_count:
                            best_count = cur_count
                            best_X = X
                        ones_list = [i for i in range(n) if (X >> i) & 1]
                        zeros_list = [i for i in range(n) if not ((X >> i) & 1)]
                        stagnation = 0
                    local_check = 0
            else:
                stagnation += 1
            T *= cool
    return best_X, best_count

def _lp_refine(h, deadline, radius=0.1, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        G = np.zeros((2 * n - 1, n))
        for d in range(-(n - 1), n):
            r = n - 1 + d
            if d >= 0:
                m = n - d
                if m > 0:
                    G[r, d:] += (1.0 - h[:m])
                    G[r, :m] += -h[d:]
            else:
                g = -d
                m = n - g
                if m > 0:
                    G[r, :m] += (1.0 - h[g:])
                    G[r, g:] += -h[:m]
        G *= (2.0 / n)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            bounds.append((max(-h[j], -radius), min(1.0 - h[j], radius)))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if not res.success:
                radius *= 0.7
                continue
            p = res.x[:n]
            if float(np.max(np.abs(p))) < 1e-14:
                radius *= 0.7
                if radius < 0.001:
                    break
                continue
        except Exception:
            radius *= 0.7
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.95
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(25):
            if hi_a - lo_a < 1e-10:
                break
            if f1 < f2:
                hi_a, f2, c2 = c2, f1, c1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo_a, f1, c1 = c1, f2, c2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(_project(h + c2 * p, target))
        opt_a = (lo_a + hi_a) / 2.0
        opt_h = _project(h + opt_a * p, target)
        opt_c5 = _exact_c5(opt_h)
        best_eval = min(f1, f2)
        if opt_c5 < best_eval:
            new_c5 = opt_c5
        else:
            new_c5 = best_eval
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + opt_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.05, 0.3)
        else:
            radius *= 0.7
            if radius < 0.001:
                break
    return best_h, best_c5

t0_global = 0.0

def _multi_restart_sa(n, masks, budget, rng, init_X=None):
    global t0_global
    best_X = 0
    best_count = 10**9
    n_rounds = 6
    alphas = [0.6180339887, 0.4142135624, 0.3819660113, 0.7071067812, 0.5, 0.2928932188]
    for rd in range(n_rounds):

        if cnt < best_count:
