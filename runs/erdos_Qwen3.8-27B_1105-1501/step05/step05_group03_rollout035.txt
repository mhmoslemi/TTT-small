    """Create sum-zero block basis."""
    B = np.zeros((n, n_blocks-1))
    starts = np.linspace(0, n, n_blocks+1).astype(int)
    # Handle edge case where some blocks might be empty
    for b in range(n_blocks-1):
        s, e = starts[b], starts[b+1]
        if e > s:
            B[s:e, b] = 1.0
    # Last block gets negative weights
    s, e = starts[n_blocks-1], n
    if e > s:
        for b in range(n_blocks-1):
            B[s:e, b] -= (starts[b+1]-starts[b]) / (e-s)
    # Normalize
    for j in range(n_blocks-1):
        norm = np.sqrt(np.sum(B[:,j]**2))
        if norm > 1e-10:
            B[:,j] /= norm
    return B

def _fourier_basis(n, K):
    x = np.arange(n) / float(n)
    B = np.zeros((n, 2*K))
    for k in range(1, K+1):
        B[:, 2*(k-1)] = np.cos(2*np.pi*k*x)
        B[:, 2*(k-1)+1] = np.sin(2*np.pi*k*x)
    return B

def _hybrid_basis(n, n_blocks, K_fourier):
    B1 = _block_basis(n, n_blocks)
    B2 = _fourier_basis(n, K_fourier)
    B = np.hstack([B1, B2])
    # Simple orthonormalization
    q = B.shape[1]
    for j in range(q):
        for i in range(j):
            B[:,j] -= np.dot(B[:,j], B[:,i]) / (np.dot(B[:,i], B[:,i])+1e-15) * B[:,i]
        norm = np.linalg.norm(B[:,j])
        if norm > 1e-10:
            B[:,j] /= norm
        else:
            B[:,j] = 0
    return B

def _basis_de(h_init, n, deadline, rng, n_blocks=16, K_fourier=6, pop_size=30, max_gen=100):
    """DE in hybrid basis space."""
    target = n / 2.0
    B = _hybrid_basis(n, n_blocks, K_fourier)
    q = B.shape[1]
    
    y0 = np.linalg.lstsq(B, h_init - 0.5, rcond=None)[0]
    
    def eval_y(y):
        raw = 0.5 + B @ y
        h = _project(raw, target)
        return _exact_c5(h)
    
    pop = np.zeros((pop_size, q))
    pop[0] = y0
    for i in range(1, pop_size):
        pop[i] = y0 + rng.standard_normal(q) * 0.3
    
    fitness = np.array([eval_y(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_c5 = fitness[best_idx]
    best_y = pop[best_idx].copy()
    best_h = _project(0.5 + B @ best_y, target)
    
    F = 0.5 + 0.3 * rng.random()
    CR = 0.75
    
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        # Adaptive params
        F = max(0.3, F * 0.995)
        CR = min(0.95, CR + 0.001)
        
        for i in range(pop_size):
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(q) < CR
            if not cross.any():
                cross[rng.integers(q)] = True
            trial = np.where(cross, mutant, pop[i])
            trial_f = eval_y(trial)
            if trial_f <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_c5:
                    best_c5 = trial_f
                    best_y = trial.copy()
                    best_h = _project(0.5 + B @ best_y, target)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_starts(h_best, n, rng, n_starts=10):
    """Generate diverse starting points."""
    starts = []
    alpha_vals = [0.6180339887, 0.3819660113, 0.4142135624, 0.5857864376,
                  0.7071067812, 0.2928932188, 0.7548776662, 0.2451223338]
    
    for i in range(n_starts):
        if i < len(alpha_vals):
            x = np.arange(n) / n
            vals = (x * alpha_vals[i]) % 1.0
            h = (vals < 0.5).astype(float)
        else:
            h = h_best + rng.normal(0, 0.1, n)
        h = _project(h, n/2.0)
        starts.append(h)
    return starts

def _pair_swap_polish(h, deadline, rng, n_iters=50000):
    """Random pair swap with exact evaluation."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    cool = 0.9997
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        delta = rng.random() * 0.1
        # Try both directions
        for sign in [1, -1]:
            h[i] += sign * delta
            h[j] -= sign * delta
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                new_c5 = _exact_c5(h)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
                elif rng.random() < np.exp((cur_c5 - new_c5) / T):
                    cur_c5 = new_c5
                    break
        else:
            h[i] -= sign * delta
            h[j] += sign * delta
        T *= cool
    
    # Restore exact sum
    best_h = _project(best_h, n/2.0)
    best_c5 = _exact_c5(best_h)
    return best_h, best_c5

    n = len(h)

    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

            p = res.x[:n]
            if float(np.max(np.abs(p))) < 1e-14:
                stall += 1
                radius *= 0.7
                if radius < radius_min:
                    break
                continue
            a, f_new = _line_search(h, p, target)
            if f_new < cur_c5 - 1e-14:
                h = _project(h + a * p, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.35)
            else:
                stall += 1
                radius *= 0.65
                if radius < radius_min:
                    break
        except Exception:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5


def _block_basis(n, n_blocks):
    B = np.zeros((n, max(1, n_blocks - 1)))
    edges = np.linspace(0, n, n_blocks + 1).astype(int)
    for b in range(n_blocks - 1):
        s, e = edges[b], edges[b + 1]
        sz = e - s
        if sz <= 0:
            continue
        B[s:e, b] = 1.0
        last_s, last_e = edges[n_blocks - 1], n
        last_sz = last_e - last_s
        if last_sz > 0:
            B[last_s:last_e, b] -= sz / last_sz
    for j in range(B.shape[1]):
        norm = np.linalg.norm(B[:, j])
        if norm > 1e-12:
            B[:, j] /= norm
    return B


def _fourier_basis(n, K):
    x = np.arange(n) / float(n)
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2 * (k - 1)] = np.cos(2 * np.pi * k * x)
        B[:, 2 * (k - 1) + 1] = np.sin(2 * np.pi * k * x)
    return B


def _build_basis(n, n_blocks, K_fourier, rng):
    B1 = _block_basis(n, n_blocks)
    B2 = _fourier_basis(n, K_fourier)
    B = np.hstack([B1, B2])
    q = B.shape[1]
    for j in range(q):
        for i in range(j):
            denom = np.dot(B[:, i], B[:, i])
            if denom > 1e-15:
                B[:, j] -= (np.dot(B[:, j], B[:, i]) / denom) * B[:, i]
        norm = np.linalg.norm(B[:, j])
        if norm > 1e-10:
            B[:, j] /= norm
        else:
            B[:, j] = 0.0
    return B


def _basis_de(h_init, n, deadline, rng, n_blocks=14, K_fourier=5, pop_size=28, max_gen=150):
    target = n / 2.0
    B = _build_basis(n, n_blocks, K_fourier, rng)
    q = B.shape[1]
    if q < 2:
        return h_init.copy(), _exact_c5(h_init)
    y0 = np.linalg.lstsq(B, h_init - 0.5, rcond=None)[0]

    def eval_y(y):
        raw = 0.5 + B @ y
        h = _project(raw, target)
        return _exact_c5(h)

    pop = np.zeros((pop_size, q))
    pop[0] = y0
    for i in range(1, pop_size):
        pop[i] = y0 + rng.standard_normal(q) * 0.4
    fitness = np.array([eval_y(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_c5 = fitness[best_idx]
    best_h = _project(0.5 + B @ pop[best_idx], target)

    F = 0.6
    CR = 0.75
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = max(0.3, F * 0.998)
        CR = min(0.95, CR + 0.002)
        for i in range(pop_size):
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            cross = rng.random(q) < CR
            if not cross.any():
                cross[rng.integers(q)] = True
            trial = np.where(cross, mutant, pop[i])
            trial_f = eval_y(trial)
            if trial_f <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_c5:
                    best_c5 = trial_f
                    best_h = _project(0.5 + B @ trial, target)
    return best_h, best_c5


def _basis_pattern_search(h_init, n, deadline, rng, n_blocks=14, K_fourier=5, max_iters=200):
    target = n / 2.0
    B = _build_basis(n, n_blocks, K_fourier, rng)
    q = B.shape[1]
    y = np.linalg.lstsq(B, h_init - 0.5, rcond=None)[0]
    cur_c5 = _exact_c5(_project(0.5 + B @ y, target))
    best_c5 = cur_c5
    best_y = y.copy()
    best_h = _project(0.5 + B @ y, target)
    step = 0.3

    for it in range(max_iters):
        if time.time() > deadline:
            break
        j = rng.integers(q)
        improved = False
        for sign in [1, -1]:
            y_trial = y.copy()
            y_trial[j] += sign * step
            h_trial = _project(0.5 + B @ y_trial, target)
            f_trial = _exact_c5(h_trial)
            if f_trial < cur_c5 - 1e-14:
                y = y_trial
                cur_c5 = f_trial
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_y = y.copy()
                    best_h = h_trial.copy()
                step = min(step * 1.2, 1.5)
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 0.01:
                break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quasi_init(n, alpha, threshold=0.5):
    x = np.arange(n) / float(n)
    vals = (x * alpha) % 1.0
    h = (vals < threshold).astype(float)
    return _project(h, n / 2.0)


def _pair_sa(h, deadline, rng, n_iters=40000, T_init=0.004, T_min=0.0001):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    cool = (T_min / T_init) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.random() * 0.08
        for sign in [1, -1]:
            h[i] += sign * delta
            h[j] -= sign * delta
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                new_c5 = _exact_c5(h)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
                elif rng.random() < np.exp(min(0.0, (cur_c5 - new_c5) / max(T, 1e-9))):
                    cur_c5 = new_c5
                    break
            h[i] -= sign * delta
            h[j] += sign * delta
        T *= cool
    best_h = _project(best_h, n / 2.0)
    best_c5 = _exact_c5(best_h)
    return best_h, best_c5


def _momentum_grad(h, deadline, beta_start=20, beta_end=400, n_steps=300, step_init=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.02, 0.12)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Focused LP warmup
    if time.time() < t0 + budget_s * 0.12:
        dl = t0 + budget_s * 0.12
        h1, c5_1 = _lp_focused(h_start, dl, K=12, radius_init=0.25, max_iters=4000, radius_min=0.001)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Basis DE at current resolution
    if time.time() < t0 + budget_s * 0.30:
        dl2 = t0 + budget_s * 0.30
        sub_dl = min(dl2, time.time() + budget_s * 0.10)
        h_de, c5_de = _basis_de(best_h, best_n, sub_dl, rng, n_blocks=14, K_fourier=5, pop_size=30, max_gen=120)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
        # Pattern search continuation
        if time.time() < dl2:
            sub_dl2 = min(dl2, time.time() + budget_s * 0.05)
            h_ps, c5_ps = _basis_pattern_search(best_h, best_n, sub_dl2, rng, n_blocks=14, K_fourier=5, max_iters=300)
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()
        # LP follow-up
        if time.time() < dl2:
            sub_dl3 = min(dl2, time.time() + budget_s * 0.03)
            h_lp, c5_lp = _lp_focused(best_h, sub_dl3, K=10, radius_init=0.12, max_iters=2000, radius_min=0.002)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 3: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.50:
        dl3 = t0 + budget_s * 0.50
        n_list = [96, 128, 160, n0]
        n_list = list(dict.fromkeys(n_list))
        alphas = [0.6180339887, 0.3819660113, 0.4142135624, 0.5857864376,
                  0.7071067812, 0.2928932188, 0.5, 0.7548776662]
        for ni, n_try in enumerate(n_list):
            if time.time() > dl3:
                break
            sub_dl = min(dl3, time.time() + budget_s * 0.04)
            if ni % 2 == 0:
                h_init = _resample(best_h, n_try)
            else:
                h_init = _quasi_init(n_try, alphas[ni % len(alphas)])
            h_de, c5_de = _basis_de(h_init, n_try, sub_dl, rng, n_blocks=12, K_fourier=4, pop_size=24, max_gen=80)
            # Quick LP
            if time.time() < dl3:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.025)
                h_lp, c5_lp = _lp_focused(h_de, sub_dl2, K=10, radius_init=0.12, max_iters=1500, radius_min=0.003)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = n_try

    # Phase 4: Diverse quasi-periodic starts with LP
    if time.time() < t0 + budget_s * 0.65:
        dl4 = t0 + budget_s * 0.65
        for alpha in [0.6180339887, 0.3819660113, 0.4142135624, 0.5857864376, 0.7548776662]:
            if time.time() > dl4:
                break
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_q = _quasi_init(best_n, alpha)
            h_lp, c5_lp = _lp_focused(h_q, sub_dl, K=12, radius_init=0.20, max_iters=2000, radius_min=0.002)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 5: Momentum + LP alternating
    if time.time() < t0 + budget_s * 0.78:
        dl5 = t0 + budget_s * 0.78
        sub_dl = min(dl5, time.time() + budget_s * 0.04)
        h_m, c5_m = _momentum_grad(best_h, sub_dl, beta_start=25, beta_end=500, n_steps=250, step_init=0.035)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()
        if time.time() < dl5:
            sub_dl2 = min(dl5, time.time() + budget_s * 0.04)
            h_lp, c5_lp = _lp_focused(best_h, sub_dl2, K=10, radius_init=0.10, max_iters=3000, radius_min=0.001)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 6: DE with different basis structure
    if time.time() < t0 + budget_s * 0.85:
        dl6 = t0 + budget_s * 0.85
        for nb, kf in [(20, 3), (8, 8), (16, 4)]:
            if time.time() > dl6:
                break
            sub_dl = min(dl6, time.time() + budget_s * 0.025)
            h_de, c5_de = _basis_de(best_h, best_n, sub_dl, rng, n_blocks=nb, K_fourier=kf, pop_size=25, max_gen=60)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()

    # Phase 7: Full LP refinement
    if time.time() < t0 + budget_s * 0.93:
        dl7 = t0 + budget_s * 0.93
        h7, c5_7 = _lp_full(best_h, dl7, radius_init=0.06, max_iters=5000, radius_min=0.0005)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Pair SA polish
    if time.time() < t0 + budget_s * 0.98:
        dl8 = t0 + budget_s * 0.98
        h8, c5_8 = _pair_sa(best_h, dl8, rng, n_iters=30000, T_init=0.003, T_min=0.00005)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
