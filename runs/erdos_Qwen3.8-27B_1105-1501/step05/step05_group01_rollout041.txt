```python
    idx = np.argmin(f_vals)

    lo_idx = max(0, idx-2)
    hi_idx = min(n_grid-1, idx+2)

    t_lo2 = ts[lo_idx]
    t_hi2 = ts[hi_idx]
    gr = (np.sqrt(5)-1)/2
    c1 = t_hi2 - gr*(t_hi2-t_lo2)
    c2 = t_lo2 + gr*(t_hi2-t_lo2)
    f1 = np.max(S + c1*T - c1*c1*Q)*dx
    f2 = np.max(S + c2*T - c2*c2*Q)*dx
    
    for _ in range(50):
        if t_hi2 - t_lo2 < 1e-12:
            break
        if f1 < f2:
            t_hi2 = c2; c2 = c1; f2 = f1
            c1 = t_hi2 - gr*(t_hi2-t_lo2)
            f1 = np.max(S + c1*T - c1*c1*Q)*dx
        else:

            c2 = t_lo2 + gr*(t_hi2-t_lo2)
            f2 = np.max(S + c2*T - c2*c2*Q)*dx
    
    opt_t = (t_lo2 + t_hi2)/2
    opt_val = np.max(S + opt_t*T - opt_t*opt_t*Q)*dx
    return opt_t, float(opt_val)
```

```python
    # First part: sum_{j=0}^{i} W[n-1+i-j] * (1-h[j])
    # Second part: sum_{k=0}^{n-1} W[n-1+k-i] * h[k]
    
    # These are cross-correlations!
    # First part: (W_rev * (1-h))[i] where W_rev is W reversed... 
    # Actually, sum_{j=0}^{i} W[n-1+i-j] * (1-h[j]) = sum_{j=0}^{min(i, n-1)} W[n-1+(i-j)] * (1-h[j])
    # This is the (i)-th element of the cross-correlation of W (specifically W[n-1:n+n-1]) with (1-h).
    
    # Hmm, this is getting complex. Let me just implement it with a loop over i, computing each grad[i] in O(n) time.
    # Total O(n²) which is fine for n=156.
    
    for i in range(n):
        # First sum
        s1 = 0.0
        for j in range(i + 1):
            idx = n - 1 + i - j
            if 0 <= idx < 2*n - 1:
                s1 += w[idx] * (1.0 - h[j])
        # Second sum
        s2 = 0.0
        for k in range(n):
            idx = n - 1 + k - i
            if 0 <= idx < 2*n - 1:
                s2 += w[idx] * h[k]
        grad[i] = dx * (s1 - s2)
    
    return obj, grad
```

This is O(n²) per gradient evaluation, which is the same as computing the correlations. So the SLSQP cost per gradient evaluation is O(n²), and for n=156, that's ~24K operations. SLSQP does ~200 gradient evals per solve, so ~5M per solve. With 7 solves, ~35M total. Very fast!

```python
def _softmax_obj_grad_vec(h, tau):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1-h, 'full')
    
    M = np.max(S)
    exps = np.exp(tau * (S - M))
    denom = np.sum(exps)
    obj = dx * (M + np.log(denom)) / tau
    w = exps / denom
    
    # grad[i] = dx * (sum_{j=0}^{i} w[n-1+i-j]*(1-h[j]) - sum_{k=0}^{n-1} w[n-1+k-i]*h[k])
    
    # Vectorized: 
    # First term: for each i, sum_{j=0}^{i} w[n-1 + (i-j)] * (1-h[j])
    # This is the element-wise product of a Toeplitz-like structure.
    # I'll use np.convolve or explicit indexing.
    
    # Method: create the matrix M1 where M1[i,j] = w[n-1+i-j] if j<=i, else 0
    # Then first_term[i] = dot(M1[i,:], (1-h))
    
    # Alternative: use broadcasting
    i_idx = np.arange(n)
    j_idx = np.arange(n)
    
    # First term: w[n-1 + i - j] * (1-h[j]) for j <= i
    diff = i_idx[:, None] - j_idx[None, :]  # shape (n, n), entry (i,j) = i-j
    mask = diff >= 0  # j <= i
    w_idx = n - 1 + diff  # shape (n, n), entry (i,j) = n-1+i-j
    # Clip to valid range
    w_idx_clipped = np.clip(w_idx, 0, 2*n-2)
    w_vals = w[w_idx_clipped] * mask  # zero out invalid
    
    first = w_vals @ (1-h)
    
    # Second term: w[n-1 + k - i] * h[k] for all k
    diff2 = j_idx[None, :] - i_idx[:, None]  # shape (n, n), entry (i,k) = k-i
    w_idx2 = n - 1 + diff2
    w_idx2_clipped = np.clip(w_idx2, 0, 2*n-2)
    w_vals2 = w[w_idx2_clipped]
    
    second = w_vals2 @ h
    
    grad = dx * (first - second)
    return obj, grad
```

This uses O(n²) memory (156² = 24K floats = 192KB) and two matrix-vector products. Very fast!

Both are always in [0, 2n-2]. So no clipping needed!
