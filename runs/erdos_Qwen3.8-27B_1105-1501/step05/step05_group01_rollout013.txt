def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _softmax_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    S_shifted = S - np.max(S)
    w = np.exp(S_shifted / K)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[n - 1 + np.argmax(S)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        k = r - (n - 1)
        wr = w[r]
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[lo1:hi1 + 1] += wr * (1.0 - h[idx - k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= wr * h[lo2:hi2 + 1]
    grad *= dx
    return grad, w

def _line_search_exact(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        b = float(np.min(h[neg] / (-d[neg])))
        t_hi = min(t_hi, b)
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if t_hi - t_lo < 5e-13:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _softmax_descent(h, deadline, K_start=4.0, K_end=0.3, max_iter=2000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    log_K_start = np.log(K_start)
    log_K_end = np.log(K_end)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max(max_iter - 1, 1)
        K = np.exp(log_K_start + frac * (log_K_end - log_K_start))
        grad, w = _softmax_grad(h, K)
        d = -(grad - np.mean(grad))
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            stall += 1
            if stall > 50:
                break
            continue
        d = d / d_norm * 0.3
        alpha, new_c5 = _line_search_exact(h, d, dx, 0.4)
        if new_c5 < cur_c5 - 1e-14:

            h = _project(h, target)
            cur_c5 = _exact_c5(h)

                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

                break
    return best_h, best_c5

def _block_move(h, i, j, dx, target):
    n = len(h)
    d = np.zeros(n)
    d[i:j+1] = h[j:i-1:-1] - h[i:j+1] if j > i else np.zeros(n)
    return d

def _reverse_move(h, start, end, dx, target):
    n = len(h)
    d = np.zeros(n)
    seg = h[start:end+1]
    d[start:end+1] = seg[::-1] - seg
    return d

def _shift_move(h, start, length, dest, dx, target):
    n = len(h)
    d = np.zeros(n)
    seg = h[start:start+length].copy()
    d[start:start+length] = -seg
    d[dest:dest+length] = seg
    return d

def _lp_direction_fast(h, scores, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            G[r, lo1:hi1+1] += (1.0 - h[idx - k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            G[r, lo2:hi2+1] -= h[lo2:hi2+1]
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.ones((1, n))
    b_eq = np.array([0.0])
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):

        if n_try < 40 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _softmax_descent(h_r, sub_dl, K_start=3.0, K_end=0.5, max_iter=1200)
        if c5_r < best_c5 + 0.002:
            rem = d3 - time.time()
            if rem > 5:
                sub_dl2 = min(d3, time.time() + max(3, budget_s * 0.008))
                h_r, c5_r = _hill_climb_pairs(h_r, sub_dl2, n_tries=2000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive local polish (25%)
    d4 = t0 + budget_s * 0.82
    # Softmax with small K
    h4 = best_h.copy()
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h4, c5_4 = _softmax_descent(h4, sub_dl, K_start=1.0, K_end=0.2, max_iter=3000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # LP polish
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4 = best_h.copy()
    h4, c5_4 = _lp_descent(h4, sub_dl, radius_init=0.06)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Block move polish
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h4 = best_h.copy()
    h4, c5_4 = _local_polish(h4, sub_dl, max_iters=8000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Pair swap
    sub_dl = min(d4, time.time() + budget_s * 0.07)
    h4 = best_h.copy()
    h4, c5_4 = _hill_climb_pairs(h4, sub_dl, n_tries=5000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Perturbation + re-optimize (12%)
    d5 = t0 + budget_s * 0.94
    scale = 0.05
    for attempt in range(300):
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _softmax_descent(h_pert, sub_dl, K_start=2.0, K_end=0.4, max_iter=800)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 6: Final polish (6%)
    h6 = best_h.copy()
    h6, c5_6 = _lp_descent(h6, deadline, radius_init=0.04)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Final pair climb
    h6 = best_h.copy()
    h6, c5_6 = _hill_climb_pairs(h6, deadline, n_tries=3000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
