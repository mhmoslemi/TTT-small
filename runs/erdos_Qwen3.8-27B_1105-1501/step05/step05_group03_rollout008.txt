    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute scores and find active lag
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * dx
        max_s = scores.max()
        
        # Compute gradient for active lags (top few)
        top_k = min(3, len(scores))
        active_lags = np.argsort(scores)[-top_k:]
        
        grad = np.zeros(n)
        for active in active_lags:
            if scores[active] < max_s - 1e-10:
                break
            d = active - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:])
                    grad[d:] += -h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[g:] += (1.0 - h[:m])
                    grad[:m] += -h[g:]
        
        grad *= dx
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # LP: minimize grad · p subject to |p| <= radius, sum(p) = 0
        # This is a simple linear program
        c = grad.copy()
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bounds = [(max(-h[j], -radius), min(1.0 - h[j], radius)) for j in range(n)]
        
        try:
            res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs',
                         options={'presolve': True})
            if res.success:
                p = res.x
                if np.max(np.abs(p)) > 1e-14:
                    # Line search
                    h_new = _project(h + p, target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = c5_new
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.02, 0.3)
                    else:
                        stall += 1
                        radius *= 0.7
                        if radius < 0.002:
                            break
                    continue
        except Exception:
            pass
        
        # Fallback: steepest descent
        p = -grad / gn * radius
        h_new = _project(h + p, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            radius *= 0.7
            if radius < 0.002:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _targeted_local_search(bits, n, deadline, rng, n_max_ops=100000):
    """Local search targeting the worst lag."""
    target = n // 2
    cur_count = _bit_eval(bits, n)
    best_bits = bits
    best_count = cur_count
    
    ops = 0
    stuck = 0
    
    while ops < n_max_ops and time.time() < deadline:
        # Find worst lag
        counts = _bit_eval_all(bits, n)
        max_idx = np.argmax(counts)
        worst_d = max_idx + 1  # lag is d (1-indexed)
        worst_count = counts[max_idx]
        
        if worst_count <= best_count:
            # Try to improve further
            pass
        
        # Find a swap that reduces the worst count
        # The worst count at lag d is max(pos_d, neg_d)
        m = (1 << (n - worst_d)) - 1
        low = bits & m
        high = (bits >> worst_d) & m
        pos_pairs = low & (m ^ high)  # positions where h[i]=1, h[i+d]=0
        neg_pairs = high & (m ^ low)  # positions where h[i]=0, h[i+d]=1
        
        # Try to fix one of the worst pairs
        # For pos: find i in pos_pairs, j = i + d where h[j]=0
        # Swap h[i] and h[j] (but h[j] is already 0, so we'd be moving the 1)

        #   - Make h[i] = 0 for some i in pos_pairs (move the 1 elsewhere)
        #   - Make h[i+d] = 1 for some i in pos_pairs (move a 1 to position i+d)
        
        # Let's pick a random pair from the worst set and try to fix it
        if pos_pairs.bit_count() >= neg_pairs.bit_count():

                stuck += 1
