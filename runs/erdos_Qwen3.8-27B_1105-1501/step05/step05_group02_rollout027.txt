```python
    shifted = (scores - max_s) / tau
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    logsum = np.log(np.sum(exps))
    obj = max_s + tau * logsum
    
    # Compute gradient
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for r in range(n_lags):
        d = r - (n - 1)
        w = weights[r]
        if w < 1e-14:
            continue
        if d == 0:
            m = n
            grad[:m] += w * dx * (1.0 - x[0:m])
            grad[0:m] += -w * dx * x[0:m]
        elif d > 0:
            m = n - d
            if m > 0:
                grad[:m] += w * dx * (1.0 - x[d:d+m])
                grad[d:d+m] += -w * dx * x[0:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[0:m] += -w * dx * x[g:g+m]
                grad[g:g+m] += w * dx * (1.0 - x[0:m])
    
    return obj, grad

def _slsqp_smooth(h_start, tau, target_sum, maxiter=200):
    """Run SLSQP on the smooth relaxation."""
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x, t=target_sum: np.sum(x) - t,
                    'jac': lambda x: np.ones(n)}]
    
    result = minimize(
        fun=lambda x: _smooth_obj_and_grad(x, tau, target_sum)[0],
        x0=h_start,
        jac=lambda x: _smooth_obj_and_grad(x, tau, target_sum)[1],
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    
    if result.success:
        h_new = _project(result.x, target_sum)
        return h_new, _exact_c5(h_new)
    else:
        # Even if not fully converged, the iterate might be better
        h_new = _project(result.x, target_sum)
        return h_new, _exact_c5(h_new)

def _smooth_anneal(h_start, deadline):
    """Anneal tau from large to small."""
    n = len(h_start)
    target = n / 2.0
    h = h_start.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

def _coordinate_descent(h, deadline, max_passes=50):
    """Cyclic coordinate descent: optimize one coordinate at a time."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Optimize h[i] given all other coordinates fixed
            # The scores are piecewise quadratic in t = h[i]
            # d=0: contributes t(1-t) to score_0 (with weight dx)
            # d>0: contributes (1-h[i+d])*t to score_{n-1+d} and -(h[i-d])*t to score_{n-1-(d)}
            # Actually let me just do a 1D golden section search
            
            old_val = h[i]
            lo_t, hi_t = 0.0, 1.0
            # Adjust bounds to maintain sum constraint:
            # sum(h) - old_val + new_val = target => new_val = target - (sum(h) - old_val)
            # But we're projecting after, so for now just search [0,1]
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_t - gr * (hi_t - lo_t)
            c2 = lo_t + gr * (hi_t - lo_t)
            
            h[i] = c1
            f1 = _exact_c5(h)
            h[i] = c2
            f2 = _exact_c5(h)

                if hi_t - lo_t < 1e-10:
                    break
                if f1 < f2:
                    hi_t = c2
                    f2 = f1
                    c2 = c1
                    c1 = hi_t - gr * (hi_t - lo_t)
                    h[i] = c1
                    f1 = _exact_c5(h)
                else:
                    lo_t = c1
                    f1 = f2
                    c1 = c2
                    c2 = lo_t + gr * (hi_t - lo_t)
                    h[i] = c2
                    f2 = _exact_c5(h)
            
            opt_t = (lo_t + hi_t) / 2.0
            h[i] = opt_t
            
            # Project to maintain sum constraint
            h = _project(h, target)
            
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                improved = True
                if cur_c5 < best_c5:

                    best_h = h.copy()
            else:

                h[i] = old_val
                h = _project(h, target)
                cur_c5 = _exact_c5(h)

            break
    
    return best_h, best_c5

def _quasi_periodic(n, alpha):

    h = ((x * alpha) % 1.0) < 0.5
```

```python
        r = n - 1 - d
        if d <= i:  # h[i] appears as first factor h[i]
            c[r] += dx * (1.0 - h[i - d])
        if d <= n - 1 - i:  # h[i] appears as second factor (1-h[i])
            c[r] += -dx * h[i + d]
    
    return c
```

Then for the 1D evaluation at coordinate i with new value t:
```python
def _eval_1d(scores, c, old, t, n):
    """Evaluate max score after changing h[i] from old to t."""
    dx = 2.0 / n
    delta = t - old
    if abs(delta) < 1e-15:
        return scores.max()
    # Update all scores
    # For r = n-1: quadratic
    # For others: linear
    new_scores = scores + c * delta
    # Fix the lag-0 term (quadratic correction)
    new_scores[n-1] = scores[n-1] + dx * (t - old) * (1.0 - t - old)
    return new_scores.max()
```

This is O(n) per evaluation! Much better than O(n²).

Now the coordinate descent becomes:
- Precompute coefficients for all coordinates: O(n²) total
- For each coordinate, do golden section search with O(n) evaluations: O(n * 30 * n) = O(30n²) per pass
- Project at the end of each pass: O(n)
- Total per pass: O(30n² + n²) ≈ O(31n²)

For n=128: 31 * 16384 ≈ 500,000 operations per pass. Very fast!

But wait, after I change h[i], the coefficients for all other coordinates change too! So I can't precompute all coefficients for the entire pass. I need to update them after each coordinate change.

Option 1: Recompute coefficients after each change. Cost: O(n) per change, O(n²) per pass total. Acceptable.

Option 2: Update coefficients incrementally. When h[i] changes by delta, the coefficient c_j[r] for coordinate j changes only for lags r where both i and j appear. This is complex.

Let me go with Option 1: recompute coefficients for the current coordinate before each 1D search.

Actually, even simpler: since I'm doing cyclic coordinate descent, after I set h[i] to its new value, I move to i+1. The coefficients for coordinate i+1 depend on the current h (which now has the new h[i]). So I just need to compute c_{i+1} from the current h.

Total cost per pass:
- For each i: compute c_i (O(n)) + golden section (30 * O(n)) = O(31n)
- Total: O(31n * n) = O(31n²)
- Project: O(n)

For n=128: 31 * 128 * 128 ≈ 500,000 basic operations per pass. With numpy, this is essentially instantaneous.

Let me implement this properly now.

One more consideration: the golden section search finds the t that minimizes the max score. But we also need t to be in [0,1] and the projection constraint. Since I'm projecting at the end of each pass, I'll just search over [0,1] freely.

Actually, there's a subtlety: if I set h[i] = 1 and all others sum to n/2 - 1, then the projection might significantly change other coordinates. But since I'm doing this for all coordinates in a pass, the net effect should be small. Let me just try it.

```python
import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coeffs_i(h, i, n):
    """Compute linear coefficients for coordinate i."""
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    for d in range(1, n):
        r = n - 1 + d
        if d <= n - 1 - i:
            c[r] += dx * (1.0 - h[i + d])
        if d <= i:
            c[r] += -dx * h[i - d]
        r = n - 1 - d
        if d <= i:
            c[r] += dx * (1.0 - h[i - d])
        if d <= n - 1 - i:
            c[r] += -dx * h[i + d]
    return c

def _cd_one_pass(h, scores, n):
    """One pass of coordinate descent. Returns updated h and scores."""
    dx = 2.0 / n
    target = n / 2.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    improved = False
    
    for i in range(n):
        c = _coeffs_i(h, i, n)
        old = h[i]
        
        # Golden section search on [0, 1]
        lo_t, hi_t = 0.0, 1.0
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        
        def eval_t(t):
            delta = t - old
            if abs(delta) < 1e-15:
                return scores.max()
            ns = scores + c * delta
            ns[n - 1] = scores[n - 1] + dx * delta * (1.0 - t - old)
            return float(np.max(ns))
        
        f1 = eval_t(c1)
        f2 = eval_t(c2)
        for _ in range(25):
            if hi_t - lo_t < 1e-9:
                break
            if f1 < f2:
                hi_t = c2
                f2 = f1
                c2 = c1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = eval_t(c1)
            else:
                lo_t = c1
                f1 = f2
                c1 = c2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = eval_t(c2)
        
        opt_t = (lo_t + hi_t) / 2.0
        if abs(opt_t - old) > 1e-8:
            h[i] = opt_t
            # Update scores
            delta = opt_t - old
            scores += c * delta
            scores[n - 1] += dx * delta * (1.0 - opt_t - old)
            improved = True
    
    if improved:
        h = _project(h, target)
        scores = _scores(h)
    return h, scores
```

Wait, there's a problem. After I change h[i] and update scores, the coefficients for the next coordinate i+1 are based on the updated h. But I'm computing _coeffs_i(h, i+1, n) at the start of the next iteration, which uses the current h. So that's correct!

But the scores update is approximate - I'm using the linear approximation for all lags except lag 0. The linear approximation is exact for lags where h[i] appears linearly. For lag 0, I use the exact quadratic formula. So the scores should be exact after the update!

Wait, is that right? Let me verify. For lag d > 0 (r = n-1+d):
score_r = dx * sum_{j=0}^{n-1-d} h[j]*(1-h[j+d])

If I change h[i] from old to new:
- If i appears as a first factor (j = i, requiring i <= n-1-d): the term h[i]*(1-h[i+d]) changes from old*(1-h[i+d]) to new*(1-h[i+d]). Change = (new-old)*(1-h[i+d]).
- If i appears as a second factor (j+d = i, i.e., j = i-d, requiring i-d >= 0 and i-d <= n-1-d): the term h[i-d]*(1-h[i]) changes from h[i-d]*(1-old) to h[i-d]*(1-new). Change = -h[i-d]*(new-old).

So the total change in score_r is (new-old) * [(1-h[i+d]) if applicable + (-h[i-d]) if applicable] = delta * c[r].

This is exact! Not an approximation. So my scores update is exact.

For lag 0:
score_0 = dx * sum_j h[j](1-h[j])
Change = dx * (new*(1-new) - old*(1-old)) = dx * (new - new² - old + old²) = dx * (new-old)(1-new-old) = dx * delta * (1 - (old+delta) - old) = dx * delta * (1 - 2*old - delta)

And in my code: dx * delta * (1.0 - t - old) where t = new = old + delta. So: 1 - (old+delta) - old = 1 - 2*old - delta. ✓

Great, so the scores update is exact.

Now, one issue: after I do a full pass and project, the scores need to be recomputed (since projection changes all coordinates). I handle this by calling _scores(h) after projection.
