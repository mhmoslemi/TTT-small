        r = n - 1 + d
        e = abs(d)
        if e == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        else:
            # Indices where 0 <= i <= n-e-1
            idx_lo = np.arange(0, n - e)
            # Indices where e <= i <= n-1
            idx_hi = np.arange(e, n)
            if d >= 0:
                G[r, idx_lo] += -dx * h[idx_lo + e]
                G[r, idx_hi] += dx * (1.0 - h[idx_hi - e])
            else:
                G[r, idx_lo] += dx * (1.0 - h[idx_lo + e])
                G[r, idx_hi] += -dx * h[idx_hi - e]
    return G

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

    lse_val = Fmax + np.log(sum_exp) / beta

    grad = G.T @ w
