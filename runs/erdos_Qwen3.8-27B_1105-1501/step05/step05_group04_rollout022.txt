The Erdős minimum overlap problem... The constant C₅ is related to the minimum possible maximum autocorrelation of a function with integral 1 on [0,2]. 

One structural observation: if h is a step function with value 1 on a set A and 0 elsewhere (with |A| = 1 in measure), then the overlap at lag k is |A ∩ (A-k)| (the measure of the intersection). This is the measure of the set of x such that both x and x+k are in A.

For indicator functions, this is maximized when A has a lot of self-overlap at various lags. To minimize this maximum, we want A to be as "spread out" as possible.

But we're allowed to use values in [0,1], not just {0,1}. This gives more flexibility.

Another observation: at the optimum, we expect h to be "smooth" in some sense (not a pure step function), because rounding to {0,1} would generally increase the max overlap.

    dx = 2.0 / n

            m = n - a

    n = len(h)
    n_lags = 2 * n - 1

    sorted_idx = np.argsort(scores)[::-1]

    #      sum(p) = 0

    #      But also |p_i| <= radius for trust region

    except Exception:
        pass

def line_search(h, d, radius):
    """Find best step alpha in [0, radius] (or [-radius, 0] if needed) for h + alpha*d."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    h0 = h
    # Compute correlation components
    S0 = np.correlate(h0, 1.0 - h0, mode='full')
    T = np.correlate(d, 1.0 - h0, mode='full') - np.correlate(h0, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds on alpha
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = radius
    t_lo = -radius
    if np.any(pos):

    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h0[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    
    # Golden section search on [t_lo, t_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1**2 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2**2 * C)) * dx

    h_new = project_box_sum(h0 + opt_t * d, target)

def lp_direction(h, scores, G, radius, n_active):
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    n_act = min(n_active, n_lags)
    active = sorted_idx[:n_act]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G[active, :]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo_b = max(-radius, -h[i] - 1e-10)
        hi_b = min(radius, 1.0 - h[i] + 1e-10)
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = radius
    t_lo = -radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1**2 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2**2 * C)) * dx
    
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; f2 = f1; c2 = c1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1**2 * C)) * dx
        else:
            a = c1; f1 = f2; c1 = c2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2**2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    h_new = project_box_sum(h + opt_t * d, target)
    opt_val = eval_c5(h_new)
    return opt_t, opt_val
