def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Grid search
    n_grid = 32
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = float(grid[best_i])
    best_val = float(obj_grid[best_i])
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = float(grid[best_i - 1]), float(grid[best_i + 1])
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    
    # Zero lag: corr(0) = dx * sum h[i]*(1-h[i])
    # d/dh[i] = dx * (1 - 2*h[i])
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags k = 1, ..., n-1
    for k in range(1, n):
        m = n - k  # number of terms
        # corr(k) = dx * sum_{i=0}^{m-1} h[i+k]*(1-h[i])
        # d/dh[i+k] = dx*(1-h[i])  for i = 0,...,m-1  → positions k,...,n-1
        # d/dh[i] = -dx*h[i+k]   for i = 0,...,m-1  → positions 0,...,m-1
        q = n - 1 + k
        G[q, k:] += dx * (1.0 - h[:m])
        G[q, :m] += -dx * h[k:]
    
    # Negative lags k = -1, ..., -(n-1)
    for k in range(1, n):
        m = n - k
        # corr(-k) = dx * sum_{i=0}^{m-1} h[i]*(1-h[i+k])
        # d/dh[i] = dx*(1-h[i+k])  for i = 0,...,m-1  → positions 0,...,m-1
        # d/dh[i+k] = -dx*h[i]    for i = 0,...,m-1  → positions k,...,n-1
        q = n - 1 - k
        G[q, :m] += dx * (1.0 - h[k:])
        G[q, k:] += -dx * h[:m]
    
    return G

    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = float(grid[best_i])
    best_val = float(obj_grid[best_i])
    if best_i > 0 and best_i < n_grid - 1:
        a, b = float(grid[best_i - 1]), float(grid[best_i + 1])
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        m = n - k
        q = n - 1 + k
        G[q, k:] += dx * (1.0 - h[:m])
        G[q, :m] += -dx * h[k:]
        q2 = n - 1 - k
        G[q2, :m] += dx * (1.0 - h[k:])
        G[q2, k:] += -dx * h[:m]
    return G

def _default_init(n):
    rng = np.random.default_rng(77)
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(1, max(2, n // 5 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_optimize(h_init, deadline, radius_init=0.15):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restarts = 0
    rng = np.random.default_rng(42)
    while time.time() < deadline:
        if stall > 80:
            if restarts < 8:
                h = _project(h + rng.normal(0, 0.03, n), target)
                cur_c5 = _exact_c5(h)
                radius = 0.10
                stall = 0
                restarts += 1
                continue
            else:
                break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            p = res.x[:n] if res.success else None
        except Exception:
            p = None
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            pred = float(np.max(scores + G @ p)) * dx
            ratio = cur_c5 / max(pred, 1e-15)
            if ratio < 0.75:
                radius = min(radius * 1.15, 0.45)
            elif ratio > 1.4:
                radius *= 0.82
            else:
                radius = min(radius * 1.02, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 30 == 0 and radius < 0.04:
                radius = 0.09
    return best_h, best_c5

def _nlp_active_set(h_init, deadline):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _outer in range(40):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_cur = float(np.max(corr))
        tol = max(0.001, 0.05 * (c5_cur - np.median(corr)))
        active = np.where(corr >= c5_cur - tol)[0]
        if len(active) < 1 or len(active) > 50:
            active = np.argsort(-corr)[:10]
        def nlp_obj(v):
            return v[n]
        def nlp_jac(v):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        def nlp_ineq(v):
            h_v = v[:n]
            c = np.correlate(h_v, 1.0 - h_v, mode='full') * dx
            return v[n] - c[active]
        def nlp_ineq_jac(v):
            h_v = v[:n]
            G = _grad_matrix(h_v)
            J = np.zeros((len(active), n + 1))
            J[:, :n] = -G[active, :]
            J[:, n] = 1.0
            return J
        def nlp_eq(v):
            return np.sum(v[:n]) - target
        def nlp_eq_jac(v):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        cons = [
            {'type': 'ineq', 'fun': nlp_ineq, 'jac': nlp_ineq_jac},
            {'type': 'eq', 'fun': nlp_eq, 'jac': nlp_eq_jac},
        ]
        bounds = [(0.0, 1.0)] * n + [(max(0.1, c5_cur - 0.05), 1.0)]
        x0 = np.concatenate([h, [c5_cur]])
        try:
            res = minimize(nlp_obj, x0, jac=nlp_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-10})
        except Exception:
            break
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            elif c5_new < c5_cur - 1e-10:
                h = h_new.copy()
        else:
            top5 = np.argsort(-corr)[:5]
            def nlp_ineq2(v):
                h_v = v[:n]
                c = np.correlate(h_v, 1.0 - h_v, mode='full') * dx
                return v[n] - c[top5]
            def nlp_ineq_jac2(v):
                h_v = v[:n]
                G = _grad_matrix(h_v)
                J = np.zeros((len(top5), n + 1))
                J[:, :n] = -G[top5, :]
                J[:, n] = 1.0
                return J
            cons2 = [
                {'type': 'ineq', 'fun': nlp_ineq2, 'jac': nlp_ineq_jac2},
                {'type': 'eq', 'fun': nlp_eq, 'jac': nlp_eq_jac},
            ]
            try:
                res2 = minimize(nlp_obj, x0, jac=nlp_jac, method='SLSQP',
                                bounds=bounds, constraints=cons2,
                                options={'maxiter': 80, 'ftol': 1e-10})
                if res2.success:
                    h_new = _project(res2.x[:n], target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                    elif c5_new < c5_cur - 1e-10:
                        h = h_new.copy()
            except Exception:
                pass
            break
    return best_h, best_c5

def _gradient_pairwise(h_init, deadline):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    tabu = set()
    while time.time() < deadline and stall < 100:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        g = abs(k_star)
        m = n - g
        if g == 0:
            grad = 2.0 * (h - 0.5)
        elif k_star >= 0:
            grad = np.zeros(n)
            grad[:m] += -h[g:]
            grad[g:] += (1.0 - h[:m])
        else:
            grad = np.zeros(n)
            grad[:m] += (1.0 - h[g:])
            grad[g:] += -h[:m]
        increasable = np.where(h < 1.0 - 1e-9)[0]
        decreasable = np.where(h > 1e-9)[0]
        if len(increasable) == 0 or len(decreasable) == 0:
            break
        top_i = increasable[np.argsort(grad[increasable])[:min(8, len(increasable))]]
        top_j = decreasable[np.argsort(-grad[decreasable])[:min(8, len(decreasable))]]
        tried_any = False
        for i in top_i:
            for j in top_j:
                if i == j or (i, j) in tabu or (j, i) in tabu:
                    continue
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = _project(h + alpha * d, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        stall = 0
                        try:
                            tabu.discard((i, j))
                        except TypeError:
                            pass
                        if len(tabu) > 200:
                            tabu = set(list(tabu)[:100])
                        tabu.add((j, i))
                        tried_any = True
                        break
                else:
                    tabu.add((i, j))
        if not tried_any:
            stall += 1
            if stall > 30:
                corr_all = np.correlate(h, 1.0 - h, mode='full')
                for attempt in range(5):
                    k_rand = int(np.random.default_rng(stall * 31 + attempt).integers(2 * n - 1))
                    kr = k_rand - (n - 1)
                    gr = abs(kr)
                    mr = n - gr
                    if gr == 0:
                        grad_r = 2.0 * (h - 0.5)
                    elif kr >= 0:
                        grad_r = np.zeros(n)
                        grad_r[:mr] += -h[gr:]
                        grad_r[gr:] += (1.0 - h[:mr])
                    else:
                        grad_r = np.zeros(n)
                        grad_r[:mr] += (1.0 - h[gr:])
                        grad_r[gr:] += -h[:mr]
                    inc_r = np.where(h < 1.0 - 1e-9)[0]
                    dec_r = np.where(h > 1e-9)[0]
                    if len(inc_r) == 0 or len(dec_r) == 0:
                        continue
                    ib = inc_r[int(np.argmin(grad_r[inc_r]))]
                    jb = dec_r[int(np.argmax(grad_r[dec_r]))]
                    if ib == jb:
                        continue
                    d = np.zeros(n)
                    d[ib] = 1.0
                    d[jb] = -1.0
                    alpha, c5_try = _line_search(h, d, dx, 0.5)
                    if c5_try < best_c5 - 1e-14:
                        h = _project(h + alpha * d, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            stall = 0
                            break
                if stall > 0:
                    break
    return best_h, best_c5

def _binary_greedy(h_init, deadline):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    best_h = h.copy()
    best_c5 = cur_c5
    max_passes = 500
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_i = -1
        best_j = -1
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                d_corr = np.zeros(2 * n - 1)
                g_shift = min(i, j)
                for lag in range(1, n):
                    m_lag = n - lag
                    hi_pos = max(i, j)
                    lo_pos = min(i, j)
                    delta = 0.0
                    if lo_pos >= lag and lo_pos < m_lag:
                        delta += 1.0 - h[lo_pos - lag] if h[j] == 1.0 else -(1.0 - h[lo_pos - lag])
                    if hi_pos < m_lag:
                        delta += h[hi_pos + lag] if h[i] == 0.0 else -h[hi_pos + lag]
                    q_plus = n - 1 + lag
                    q_minus = n - 1 - lag
                    d_corr[q_plus] += delta
                    d_corr[q_minus] += delta
                new_max = float(np.max(corr + d_corr * dx))
                delta_c5 = new_max - cur_c5
                if delta_c5 < best_delta:
                    best_delta = delta_c5
                    best_i, best_j = i, j
        if best_i >= 0 and best_delta < -1e-14:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_c5 = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def _binary_greedy_fast(h_init, deadline):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _pass in range(300):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-10
        best_i, best_j = -1, -1
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                new_c5 = _exact_c5(h)
                d = new_c5 - cur_c5
                if d < best_delta:
                    best_delta = d
                    best_i, best_j = i, j
                    best_new_c5 = new_c5
                    best_h_trial = h.copy()
                h[j] = 0.0
            h[i] = 1.0
        if best_i >= 0 and best_delta < -1e-12:
            h = best_h_trial.copy()
            cur_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5
