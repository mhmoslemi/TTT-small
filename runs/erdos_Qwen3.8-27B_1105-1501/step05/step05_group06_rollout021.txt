import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1) if res > 0 else (x > 0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _binary_sa(h_bin, n, deadline, rng, max_iters=1000000):
    """Simulated annealing on binary configurations with swap moves."""
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    T_start = 0.02
    T_end = 0.001
    it = 0
    no_improve = 0
    
    while it < max_iters and time.time() < deadline:
        it += 1
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        # Random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones[i], zeros[j] = zeros[j], ones[i]  # swap in index arrays
            if cur_c5 < best_c5:

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]  # revert
            no_improve += 1
        
        if no_improve > 5000:
            # Try a larger T kick
            T = 0.02
            no_improve = 0
    
    return best_h, best_c5

Let me be more careful. For the lag d (positive shift), the contribution is:
sum_{i=0}^{n-1-d} h[i] * (1-h[i+d]) = sum_{i=0}^{n-1-d} h[i] - sum_{i=0}^{n-1-d} h[i]*h[i+d]

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lp_solve(h, G, radius, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    scores = _scores(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    n_grid = 15
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0:
        return 0.0, float(vals[0])
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < float(vals[best_i]):
        return opt_a, opt_c5
    return alphas[best_i], float(vals[best_i])

def _local_opt(h, deadline, top_k_list=None, radius_init=0.12, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 12, 7, 5, 3]

    for _ in range(max_iters):

            break
        if stall > 100:

        tk = top_k_list[ti % n_tk]
        G = _build_G(h)
        p, _ = _lp_solve(h, G, radius, top_k=tk)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        h_cand = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_cand)
        if actual_c5 < cur_c5 - 1e-13:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.7
            ti += 1
            if stall > 15 and radius < 0.012:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_descent(h, deadline, max_passes=300, verify_thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > verify_thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        h = np.clip(h, 0.0, 1.0)
        if not pass_improved:
            break
    return best_h, best_c5

            continue
        n_starts = 3
        for s in range(n_starts):
            if time.time() > dl2:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                h_init = _golden_pattern(n_try, alphas[ri % len(alphas)])
            else:
                h_init = _quasi_pattern(n_try, alphas[(ri + 2) % len(alphas)], 0.37)
            sub_dl = min(dl2, time.time() + budget_s * 0.035)
            h_try, c5_try = _multi_start_lp(h_init, n_try, sub_dl, rng, n_attempts=1)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                sub_dl2 = min(dl2, time.time() + budget_s * 0.012)
                h_pd, c5_pd = _pair_descent(h_try, sub_dl2, max_passes=50, verify_thresh=1e-5)
                if c5_pd < best_c5:
                    best_c5 = c5_pd
                    best_h = h_pd.copy()
                    best_n = n_try

    # Phase 3: Binary SA at best resolution
    if time.time() < deadline:
        dl3 = t0 + budget_s * 0.58
        h_bin = (best_h > 0.5).astype(float)
        # Ensure exactly n/2 ones
        n_ones = int(np.sum(h_bin == 1.0))
        target_ones = best_n // 2
        if n_ones > target_ones:
            one_idx = np.where(h_bin == 1.0)[0]
            to_flip = one_idx[rng.choice(len(one_idx), n_ones - target_ones, replace=False)]
            h_bin[to_flip] = 0.0
        elif n_ones < target_ones:
            zero_idx = np.where(h_bin == 0.0)[0]
            to_flip = zero_idx[rng.choice(len(zero_idx), target_ones - n_ones, replace=False)]
            h_bin[to_flip] = 1.0
        h_sa, c5_sa = _binary_sa(h_bin, min(dl3, time.time() + budget_s * 0.12), rng)
        # Relax the binary result
        sub_dl = min(dl3, time.time() + budget_s * 0.035)
        h_rel, c5_rel = _relax_binary(h_sa, sub_dl, rng)
        if c5_rel < best_c5:
            best_c5 = c5_rel
            best_h = h_rel.copy()
            best_n = len(h_rel)
        elif c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Also try SA on a different resolution
        if time.time() < dl3 - budget_s * 0.05:
            n_sa2 = min(256, max(128, best_n))
            h_res = _resample(best_h, n_sa2)
            h_bin2 = (h_res > 0.5).astype(float)
            n_ones2 = int(np.sum(h_bin2 == 1.0))
            target_ones2 = n_sa2 // 2
            if n_ones2 > target_ones2:
                oi = np.where(h_bin2 == 1.0)[0]
                h_bin2[oi[rng.choice(len(oi), n_ones2 - target_ones2, replace=False)]] = 0.0
            elif n_ones2 < target_ones2:
                zi = np.where(h_bin2 == 0.0)[0]
                h_bin2[zi[rng.choice(len(zi), target_ones2 - n_ones2, replace=False)]] = 1.0
            h_sa2, c5_sa2 = _binary_sa(h_bin2, min(dl3, time.time() + budget_s * 0.08), rng)
            sub_dl2 = min(dl3, time.time() + budget_s * 0.03)
            h_rel2, c5_rel2 = _relax_binary(h_sa2, sub_dl2, rng)
            if c5_rel2 < best_c5:
                best_c5 = c5_rel2
                best_h = h_rel2.copy()
                best_n = len(h_rel2)

    # Phase 4: Intensive pair descent
    if time.time() < deadline:
        dl4 = t0 + budget_s * 0.75
        best_h, best_c5 = _pair_descent(best_h, dl4, max_passes=500, verify_thresh=1e-6)
        # Follow up with LP
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.02)
            h_lp, c5_lp = _local_opt(best_h, sub_dl, top_k_list=[5, 3, None],
                                     radius_init=0.04, max_iters=1000)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
        # More pair descent
        if time.time() < dl4:
            dl4b = min(dl4, time.time() + budget_s * 0.04)
            best_h, best_c5 = _pair_descent(best_h, dl4b, max_passes=200, verify_thresh=1e-7)

    # Phase 5: Diverse restarts at best resolution
    if time.time() < deadline:
        dl5 = t0 + budget_s * 0.88
        n_work = best_n
        for attempt in range(8):
            if time.time() > dl5:
                break
            if attempt == 0:
                h_init = best_h + rng.normal(0, 0.03, n_work)
            elif attempt == 1:
                h_init = _golden_pattern(n_work, 0.6180339887 + rng.normal(0, 0.02))
            elif attempt == 2:
                h_init = _quasi_pattern(n_work, 0.5 + rng.normal(0, 0.03), rng.uniform(0, 1))
            else:
                h_init = best_h + rng.normal(0, 0.05, n_work)
            h_init = _project(h_init, n_work / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.018)
            h_try, c5_try = _local_opt(h_init, sub_dl,
                                       top_k_list=[None, 8, 5, 3],
                                       radius_init=0.12, max_iters=2000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_work
            else:
                sub_dl2 = min(dl5, time.time() + budget_s * 0.008)
                h_pd, c5_pd = _pair_descent(h_try, sub_dl2, max_passes=60, verify_thresh=1e-6)
                if c5_pd < best_c5:
                    best_c5 = c5_pd
                    best_h = h_pd.copy()
                    best_n = n_work

    # Phase 6: Perturbation refinement (annealing loop)
    if time.time() < deadline:
        dl6 = t0 + budget_s * 0.94
        scale = 0.015
        for attempt in range(300):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.006)
            h_try, c5_try = _local_opt(h_pert, sub_dl,
                                       top_k_list=[4, 3, None],
                                       radius_init=0.04, max_iters=300)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.06, scale * 1.12)

    # Phase 7: Final polish
    if time.time() < deadline:
        dl7 = deadline
        h_pd, c5_pd = _pair_descent(best_h, dl7, max_passes=200, verify_thresh=1e-8)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
        if time.time() < dl7 - 3:
            h_fin, c5_fin = _local_opt(best_h, dl7, top_k_list=[3, 4, 5],
                                       radius_init=0.008, max_iters=500)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
