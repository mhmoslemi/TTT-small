import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        G_sub = G[idx, :]
        scores_sub = scores[idx]
    else:
        G_sub = G
        scores_sub = scores
    n_constraints = len(scores_sub)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -scores_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                phase += 1
            stall = 0
            radius = min(radius * 1.05, 0.45)
        else:
            stall += 1
            radius *= 0.85
            if stall > 30:
                radius = max(0.05, radius * 0.5)
                if radius < 0.005:
                    break
    return best_h, best_c5

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = np.where(((x * alpha) % 1.0) < 0.5, 1.0, 0.0)
    return _project(h, n / 2.0)

def _block_init(n, n_blocks):
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(0, n, 2 * block_size):
        end = min(i + block_size, n)
        h[i:end] = 1.0
    return _project(h, n / 2.0)

def _sinusoidal_init(n, freq, amp):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x)
    return _project(h, n / 2.0)

def _random_binary_init(n, rng, frac):
    h = np.zeros(n)
    n_ones = int(round(n * frac))
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=3):
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 40)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(6):
                        m1 = lo_a + (hi_a - lo_a) * 0.33
                        m2 = hi_a - (hi_a - lo_a) * 0.33
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return h, best_c5

def _smooth_refine(h, n_new, deadline, rng):
    h_up = _resample(h, n_new)
    h_up += rng.normal(0, 0.01, n_new)
    h_up = _project(h_up, n_new / 2.0)
    result, c5 = _lp_optimize_adaptive(h_up, deadline, radius_init=0.15)
    return result, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

            ht, ct = _lp_optimize(h_qp, tl, radius_init=0.25, max_iters=2000, stall_limit=60)
            candidates.append((ht, ct, n_try))
            if ct < best_c5:

    for n_try in [64, 96, 128]:

    for n_try in [96, 128]:
        if time.time() > dl1:

            h_s = _sinusoidal_init(n_try, freq, 0.35)

            h_s = _project(h_s, n_try / 2.0)
            tl = min(dl1, time.time() + budget_s * 0.01)

            candidates.append((ht, ct, n_try))
            if ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try

    # Random binary
    for n_try in [96, 128, 160]:
        if time.time() > dl1:
            break
        for trial in range(3):
            if time.time() > dl1:
                break
            h_r = _random_binary_init(n_try, rng, 0.5)
            h_r += rng.normal(0, 0.1, n_try)
            h_r = _project(h_r, n_try / 2.0)
            tl = min(dl1, time.time() + budget_s * 0.012)
            ht, ct = _lp_optimize(h_r, tl, radius_init=0.25, max_iters=1500, stall_limit=50)

    h2, c5_2 = _lp_optimize_adaptive(best_h, dl2, radius_init=0.25)

        best_c5 = c5_2; best_h = h2.copy()

    # Phase 3: Multi-resolution refinement (20% budget)

        h_up = _resample(best_h, n_try)
        h_up += rng.normal(0, 0.015, n_try)
        h_up = _project(h_up, n_try / 2.0)
        tl = min(dl3, time.time() + budget_s * 0.06)
        ht, ct = _lp_optimize_adaptive(h_up, tl, radius_init=0.15)
        if ct < best_c5:
            best_c5 = ct; best_h = ht.copy(); best_n = n_try

    # Try upscaling best to higher resolution
    for n_try in [best_n + 32, best_n + 64]:
        if time.time() > dl3 or n_try > 256:
            continue
        h_up = _resample(best_h, n_try)
        h_up += rng.normal(0, 0.01, n_try)
        h_up = _project(h_up, n_try / 2.0)
        tl = min(dl3, time.time() + budget_s * 0.05)

    dl4 = t0 + budget_s * 0.90
    scale = 0.015
    n_restarts = 0
    while time.time() < dl4 and n_restarts < 500:
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)

            n_restarts = 0
        else:
            n_restarts += 1
            if n_restarts % 10 == 0:
                scale = min(0.06, scale * 1.08)

    # Phase 5: Final aggressive LP polishing (8% budget)
    dl5 = t0 + budget_s * 0.97
    h5, c5_5 = _lp_optimize(best_h, dl5, radius_init=0.08, max_iters=5000,
                            stall_limit=200, top_k=5, radius_decay=0.85)
    if c5_5 < best_c5:
        best_c5 = c5_5; best_h = h5.copy()

    # Very fine LP
    h5b, c5_5b = _lp_optimize(best_h, t0 + budget_s * 0.975, radius_init=0.02,
                               max_iters=5000, stall_limit=300, top_k=3, radius_decay=0.85)

    # Phase 6: Pair polish
    if time.time() < t0 + budget_s * 0.985:

        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
