def _grad_matrix(h):

            g = -d
            m = n - g

                G[r, :m] += (1.0 - h[g:])

    return G

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.05):
    """Solve LP for descent direction."""

    c[-1] = 1.0

    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})

        pass
    return None

def _safe_sigmoid(x):
    x = np.clip(x, -50.0, 50.0)
    out = np.empty_like(x, dtype=float)

    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')

    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15

        t_lo = float(np.max(h[neg] / d[neg]))

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.04):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        if lo_j > hi_j:
            lo_j = hi_j = 0.0
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

    h3, c5_3 = _logit_evolution(best_h, best_c5, d3, rng, pop_size=16, max_gens=6000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP on best (10%)
    d4 = t0 + budget_s * 0.62
    h4, c5_4 = _lp_local(best_h, d4, radius_init=0.18)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (15%)
    d5 = t0 + budget_s * 0.77
    h5, c5_5 = _multi_res(best_h, best_c5, d5, rng)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Gradient smooth (8%)
    d6 = t0 + budget_s * 0.85
    h6, c5_6 = _gradient_smooth(best_h, d6, beta_list=(50, 120, 300, 600), lr_init=0.018)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Kick and reopt (7%)
    d7 = t0 + budget_s * 0.92
    h7, c5_7 = _kick_reopt(best_h, best_c5, d7, rng, n_kicks=12)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Pair refine (5%)
    d8 = t0 + budget_s * 0.97
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=40)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final LP polish
    if time.time() < deadline:
        h9, c5_9 = _lp_local(best_h, deadline, radius_init=0.08)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
