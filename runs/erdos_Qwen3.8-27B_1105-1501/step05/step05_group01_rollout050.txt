```python
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Determine feasible interval

    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14

    if np.any(neg):
        t_lo = max(t_lo, -float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid search for initial bracket

    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([float(np.max(S0 + t * T - t*t * C)) * dx for t in ts])
    best_idx = int(np.argmin(vals))
    
    # Refine with golden section around best grid point
    if best_idx == 0:
        g_lo, g_hi = t_lo, ts[1]
    elif best_idx == n_grid - 1:
        g_lo, g_hi = ts[-2], t_hi
    else:
        g_lo, g_hi = ts[best_idx - 1], ts[best_idx + 1]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = g_hi - gr * (g_hi - g_lo)
    c2 = g_lo + gr * (g_hi - g_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx

    for _ in range(40):

            break
        if f1 < f2:
            g_hi = c2; c2 = c1; f2 = f1
            c1 = g_hi - gr * (g_hi - g_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            g_lo = c1; c1 = c2; f1 = f2
            c2 = g_lo + gr * (g_hi - g_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (g_lo + g_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

            break

        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)

            radius *= 0.7

            h = h + alpha * p

            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:

            stall = 0

        else:

def _greedy_pair_swap(h, deadline, max_rounds=20):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Focus on pairs that might reduce the max
        scores = _scores(h)
        max_score = float(np.max(scores))
        # Try all pairs but prioritize those involving high-value positions
        order = np.argsort(-h)  # Try to move mass from high to low
        for idx_i in range(n):
            if time.time() > deadline:
                break
            i = int(order[idx_i])
            for idx_j in range(n):
                j = int(order[idx_j])
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.3)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break  # Move to next i
        if not improved:
            break
    return best_h, best_c5

def _random_pair_swap(h, deadline, radius=0.25):

def _block_move(h, deadline):
    """Move contiguous blocks of mass."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(4242)
    stall = 0
    while time.time() < deadline:
        if stall > 200:
            break
        # Pick a block to move
        block_len = int(rng.integers(2, min(6, n//4)))
        if block_len >= n:
            block_len = 2
        start = int(rng.integers(0, n - block_len + 1))
        # Find a destination
        dest = int(rng.integers(0, n - block_len + 1))
        if dest == start:
            stall += 1
            continue
        # Ensure no overlap
        if abs(dest - start) < block_len:
            stall += 1
            continue
        # Create direction: move block from start to dest
        d = np.zeros(n)
        d[start:start+block_len] = -1.0
        d[dest:dest+block_len] = 1.0
        alpha, c5_try = _line_search(h, d, dx, 0.3)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _basin_hopping(h, deadline, n_restarts=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    scale = 0.06
    for _ in range(n_restarts):
        if time.time() > deadline - 5:
            break
        # Perturb
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        # Quick LP optimize
        sub_dl = min(deadline - 3, time.time() + max(2.0, (deadline - time.time()) / max(n_restarts - 1, 1) * 0.8))
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.05)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.15)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_binary(n, method=0, rng=None):
    """Generate structured binary patterns."""
    if rng is None:
        rng = np.random.default_rng(0)
    h = np.zeros(n)
    if method == 0:
        # Random
        idx = rng.permutation(n)[:n//2]
        h[idx] = 1.0
    elif method == 1:
        # Alternating with noise
        h[::2] = 1.0
        # Flip some to make it n/2
        ones = np.sum(h)
        if ones != n // 2:
            # Adjust
            if ones > n // 2:
                one_positions = np.where(h == 1.0)[0]
                flip = rng.choice(one_positions, size=int(ones - n//2), replace=False)
                h[flip] = 0.0
            else:
                zero_positions = np.where(h == 0.0)[0]
                flip = rng.choice(zero_positions, size=int(n//2 - ones), replace=False)
                h[flip] = 1.0
    elif method == 2:
        # Block pattern: blocks of size 2
        for i in range(0, n, 4):
            h[i:i+2] = 1.0
        # Adjust to get exactly n/2 ones
        ones = int(np.sum(h))
        if ones != n // 2:
            if ones > n // 2:
                one_positions = np.where(h == 1.0)[0]
                flip = rng.choice(one_positions, size=ones - n//2, replace=False)
                h[flip] = 0.0
            else:
                zero_positions = np.where(h == 0.0)[0]
                flip = rng.choice(zero_positions, size=n//2 - ones, replace=False)
                h[flip] = 1.0
    elif method == 3:
        # Thue-Morse based
        tm = np.array([bin(i).count('1') % 2 for i in range(n)])
        h[tm == 1] = 1.0
        ones = int(np.sum(h))
        if ones != n // 2:
            if ones > n // 2:
                one_pos = np.where(h == 1.0)[0]
                flip = rng.choice(one_pos, size=ones - n//2, replace=False)
                h[flip] = 0.0
            else:
                zero_pos = np.where(h == 0.0)[0]
                flip = rng.choice(zero_pos, size=n//2 - ones, replace=False)
                h[flip] = 1.0
    else:
        # Modular pattern
        p = int(np.sqrt(n)) + 1
        h[(np.arange(n) % p) < p//2] = 1.0
        ones = int(np.sum(h))
        if ones != n // 2:
            if ones > n // 2:
                one_pos = np.where(h == 1.0)[0]
                flip = rng.choice(one_pos, size=ones - n//2, replace=False)
                h[flip] = 0.0
            else:
                zero_pos = np.where(h == 0.0)[0]
                flip = rng.choice(zero_pos, size=n//2 - ones, replace=False)
                h[flip] = 1.0
    return h
```

```python
    n_lags = 2 * n - 1

    threshold = sorted_scores[min(top_k, n_lags-1)]
    active = scores >= threshold - 1e-6
    # Also always include lag 0 (though it's always 0 for feasible h)
    active[n-1] = True
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_active = int(np.sum(active))
    A_ub = np.zeros((n_active, n + 1))
    idx = 0
    for i in range(n_lags):
        if active[i]:
            A_ub[idx, :n] = G[i]
            A_ub[idx, n] = -1.0
            idx += 1
    b_ub = -scores[active]
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    
    bounds = []

        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    except Exception:
```

Let me also think about another approach: **Direct minimization of the max using a smooth approximation**.

The max function can be approximated by the log-sum-exp:
max_i x_i ≈ (1/τ) * log(sum_i exp(τ * x_i))

for large τ. This gives a smooth, differentiable approximation. We can then use gradient descent on this smooth approximation.

The gradient of the log-sum-exp approximation at point h is:
∇_h [(1/τ) log(sum_k exp(τ * score_k(h)))] = sum_k (exp(τ * score_k) / sum_j exp(τ * score_j)) * ∇_h score_k(h)

This is a weighted average of the gradients of individual score functions, weighted by the softmax of the scores. This automatically focuses on the binding constraints!

The gradient of score_k(h) = sum_i h[i]*(1-h[i+k]) is:
∂score_k/∂h[j] = (1-h[j+k]) if j+k is valid, minus h[j-k] if j-k is valid, etc.

This is exactly what _grad_matrix computes!

So the smooth gradient direction is:
d = -sum_k w_k * G[k, :]
where w_k = softmax(τ * scores)

This is a much simpler and potentially more effective direction than the LP!

```python
def _smooth_gradient_direction(h, tau=50.0):
    """Use log-sum-exp approximation to get a smooth gradient direction."""
    scores = _scores(h)
    # Softmax weights
    max_s = float(np.max(scores))
    exps = np.exp(tau * (scores - max_s))
    weights = exps / float(np.sum(exps))
    # Gradient of weighted sum of scores
    G = _grad_matrix(h)
    grad = G.T @ weights
    # Project out the constraint direction (sum = 0)
    mean_grad = float(np.mean(grad))
    grad -= mean_grad
    # Negative gradient for descent
    return -grad
```

Key observation: The previous algorithm uses a fixed sequence of phases. If it gets a good result early, it spends the rest of the time polishing. If it doesn't, it might waste time in suboptimal phases.

A better approach might be:
1. **Adaptive phase allocation**: Allocate more time to promising phases based on intermediate results.
2. **Parallel exploration**: Try multiple strategies simultaneously (conceptually, since we have 1 CPU, we interleave them).
3. **Early termination of unproductive phases**: If a phase isn't making progress, cut it short.

Let me also think about a completely different approach that might break through:

**Approach: Direct binary search with exact evaluation**

For binary h, the objective is:
C₅ = max_d (count_d * 2/n)

where count_d = |{i : h[i]=1, h[i+d]=0}|

For a binary string, we can compute all counts in O(n^2) time using FFT:
- Let A = h, B = 1-h
- The cross-correlation A ⊛ B gives us the counts at each lag

But for binary strings, we can also compute it more efficiently using bit operations (if n is small enough).

The key insight for binary search: we can use a **simulated annealing** approach where:
- State: a binary string with exactly n/2 ones
- Move: pick two positions (one with 1, one with 0) and swap them
- Energy: C₅ of the current state
- Acceptance: accept if energy decreases, or accept with probability exp(-ΔE/T)

This is what the previous code's `_binary_sa` does, but it uses line search which is overkill for binary moves. For binary moves, the energy change can be computed exactly and quickly.

Let me think about the exact energy change for a binary swap:
- Swap positions i (currently 1) and j (currently 0)
- For each lag d, the count changes by:
  - At lag d = j-i: we remove one 1→0 pair (at position i, the 1 was followed by 0 at j) and add one (at position j, now 1, followed by whatever was at j+d)
  - Actually this is more complex because it affects all lags.

The exact change is:
Δcount_d = (new_h[i]*(1-new_h[i+d]) + new_h[i-d]*(1-new_h[i]) + ... ) - (old_h[i]*(1-old_h[i+d]) + ...)

This is O(n) to compute for each of O(n) lags, so O(n^2) total. For n=156, that's about 24000 operations per swap evaluation. Very fast!

So we can do many, many simulations annealing steps. Let me implement this:
