def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(scores))
    exponents = np.exp((scores - M) / tau)
    total = float(np.sum(exponents))
    soft_val = M + tau * np.log(total)
    weights = exponents / total
    # Gradient
    grad = np.zeros(n)
    center = n - 1
    for r in range(len(scores)):
        w = weights[r]
        if w < 1e-6:
            continue
        d = r - center
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] -= w * h[:m]
    grad *= dx
    return soft_val, grad

def _line_search_exact(h, d, dx, t_lo, t_hi, n_samples=80):
    # Exact 1D line search for phi(t) = dx * max(S + t*T - t^2*Q)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def phi(t):
        return float(np.max(S + t * T - t*t * Q)) * dx
    
    # Sample
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = np.array([phi(t) for t in ts])
    
    # Find minimum
    idx = int(np.argmin(vals))
    # Refine around minimum
    if 0 < idx < n_samples - 1:
        lo_t = ts[idx-1]
        hi_t = ts[idx+1]
    else:
        lo_t = ts[max(0, idx-1)]
        hi_t = ts[min(n_samples-1, idx+1)]
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = phi(c1)
    f2 = phi(c2)
    for _ in range(50):
        if hi_t - lo_t < 1e-12:
            break
        if f1 < f2:
            hi_t = c2
            c2 = c1
            f2 = f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = phi(c1)
        else:
            lo_t = c1
            c1 = c2
            f1 = f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = phi(c2)
    
    opt_t = (lo_t + hi_t) / 2.0
    return opt_t, phi(opt_t)

def _feasible_interval(h, d, radius):
    # Find feasible t range for h + t*d staying in [0,1]
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = -radius
    t_hi = radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / (-d[neg]))))
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    return t_lo, t_hi

def _run_slsqp(h, deadline, target, tau_seq, maxiter_max=60):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    
    for tau in tau_seq:
        if time.time() > deadline:

        # SLSQP optimization

                _slsqp_obj, cur_h,
                args=(tau, n),

                    'jac': lambda x, t=target: np.ones(len(x))

                    best_h = candidate.copy()
                    cur_h = candidate.copy()
        except Exception:

```python
import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _soft_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(scores))
    shifted = (scores - M) / tau
    exponents = np.exp(shifted)
    total = float(np.sum(exponents))
    soft_val = M + tau * np.log(total)
    weights = exponents / total
    grad = np.zeros(n)
    center = n - 1
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-5:
            continue
        d = r - center
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w * (1.0 - h[d:])
                grad[d:] -= w * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[g:]
    grad *= dx
    return soft_val, grad


class _SoftProblem:
    def __init__(self, tau, target):
        self.tau = tau
        self.target = target

    def fun(self, x):
        val, _ = _soft_obj_grad(x, self.tau)
        return val

    def jac(self, x):
        _, g = _soft_obj_grad(x, self.tau)
        return g

    def con_fun(self, x):
        return np.array([float(np.sum(x)) - self.target])

    def con_jac(self, x):
        return np.ones((1, len(x)))


def _slsqp_phase(h, target, tau_seq, deadline, maxiter=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur = h.copy()
    if maxiter is None:
        maxiter = 50
    for tau in tau_seq:
        if time.time() > deadline:
            break
        prob = _SoftProblem(tau, target)
        try:
            res = minimize(
                prob.fun, cur, jac=prob.jac,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': prob.con_fun, 'jac': prob.con_jac}],
                options={'maxiter': maxiter, 'ftol': 1e-10}
            )
            if res.success or res.x is not None:
                cand = _project(res.x, target)
                c5 = _exact_c5(cand)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = cand.copy()
                    cur = cand.copy()
        except Exception:
            pass
    return best_h, best_c5


def _line_search(h, d, dx, t_lo, t_hi, n_samples=60):
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def phi(t):
        return float(np.max(S + t * T - t * t * Q)) * dx

    # Check endpoints and t=0
    candidates = [0.0, t_lo, t_hi]
    vals = [phi(t) for t in candidates]
    
    # Dense sampling
    if t_hi - t_lo > 1e-14:
        ts = np.linspace(t_lo, t_hi, n_samples)
        for t in ts:
            candidates.append(t)
            vals.append(phi(t))
    
    # Find best
    best_idx = int(np.argmin(vals))
    best_t = candidates[best_idx]
    best_val = vals[best_idx]
    
    # Golden section refinement around best
    if 0 < best_idx < len(candidates) - 1:
        g_lo = candidates[best_idx - 1]
        g_hi = candidates[best_idx + 1]
    else:
        g_lo = t_lo
        g_hi = t_hi
    
    if g_hi - g_lo > 1e-14:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = g_hi - gr * (g_hi - g_lo)
        c2 = g_lo + gr * (g_hi - g_lo)
        f1 = phi(c1)
        f2 = phi(c2)
        for _ in range(40):
            if g_hi - g_lo < 1e-13:
                break
            if f1 < f2:
                g_hi = c2
                c2 = c1
                f2 = f1
                c1 = g_hi - gr * (g_hi - g_lo)
                f1 = phi(c1)
            else:
                g_lo = c1
                c1 = c2
                f1 = f2
                c2 = g_lo + gr * (g_hi - g_lo)
                f2 = phi(c2)
        new_t = (g_lo + g_hi) / 2.0
        new_val = phi(new_t)
        if new_val < best_val:
            best_t = new_t
            best_val = new_val
    
    return best_t, best_val


def _pair_swap(h, i, j, target, dx, radius=0.3):
    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    # Feasible interval
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    t_lo = -radius
    t_hi = radius
    if np.any(pos_mask):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
        t_lo = max(t_lo, float(np.max(-h[pos_mask] / d[pos_mask])))
    if np.any(neg_mask):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg_mask]) / (-d[neg_mask]))))
        t_hi = min(t_hi, float(np.min(h[neg_mask] / (-d[neg_mask]))))
    if t_hi - t_lo < 1e-14:
        return h, _exact_c5(h)
    t_opt, val = _line_search(h, d, dx, t_lo, t_hi)
    if val < _exact_c5(h) - 1e-13:
        h_new = _project(h + t_opt * d, target)
        c5_new = _exact_c5(h_new)
        if c5_new < _exact_c5(h):
            return h_new, c5_new
    return h, _exact_c5(h)


def _gradient_direction_step(h, target, tau, dx, deadline, max_steps=10):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur = h.copy()
    for step in range(max_steps):
        if time.time() > deadline:
            break
        _, grad = _soft_obj_grad(cur, tau)
        # Negative gradient direction
        d = -grad
        # Zero-mean
        d -= np.mean(d)
        # Normalize
        d_max = float(np.max(np.abs(d)))
        if d_max < 1e-14:
            break
        d /= d_max
        # Feasible interval
        pos = d > 1e-14
        neg = d < -1e-14
        radius = 0.15
        t_lo = -radius
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - cur[pos]) / d[pos])))
            t_lo = max(t_lo, float(np.max(-cur[pos] / d[pos])))
        if np.any(neg):
            t_lo = max(t_lo, float(np.max((1.0 - cur[neg]) / (-d[neg]))))
            t_hi = min(t_hi, float(np.min(cur[neg] / (-d[neg]))))
        if t_hi - t_lo < 1e-14:
            break
        t_opt, val = _line_search(cur, d, dx, t_lo, t_hi, n_samples=50)
        if val < best_c5 - 1e-12:
            cur = _project(cur + t_opt * d, target)
            c5 = _exact_c5(cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cur.copy()
            radius = min(radius * 1.02, 0.4)
        else:
            radius *= 0.7
            if radius < 0.01:
                break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h
