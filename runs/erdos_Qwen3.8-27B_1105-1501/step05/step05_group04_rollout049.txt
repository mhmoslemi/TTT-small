    def obj(x):
        return x[-1]

        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def eq_con(x):
        return np.sum(x[:n]) - n/2.0
    
    def eq_grad(x):
        g = np.zeros(len(x))
        g[:n] = 1.0
        return g

    # (We'll compute which are near-active)
    
    from scipy.optimize import minimize
    
    x0 = np.concatenate([h, [t_init]])

    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for r in range(n_lags):
        s = r - (n - 1)
        if s > 0:
            m = n - s
            # dF/dh[j] for j >= s: dx*(1-h[j-s])
