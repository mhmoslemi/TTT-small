import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n - 1, n - 1 - d)
        # h[i] as h[n+d]: n = i-d, valid if lo <= i-d <= hi
        # i ranges from lo+d to hi+d
        i_lo1 = lo + d
        i_hi1 = hi + d
        if i_lo1 <= i_hi1:
            i1 = np.arange(i_lo1, i_hi1 + 1)
            G[r, i1] += dx * (1.0 - h[i1 - d])
        # h[i] in (1-h[n]): n = i, valid if lo <= i <= hi
        i_lo2 = lo
        i_hi2 = hi
        if i_lo2 <= i_hi2:
            i2 = np.arange(i_lo2, i_hi2 + 1)
            G[r, i2] += -dx * h[i2 + d]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pnorm_obj_grad(h, q, G):
    F = _scores(h)
    L = len(F)
    S = float(F.max())
    if S < 1e-30:
        return 0.0, np.zeros(len(h))
    w = F / S  # w_r = F_r / S, max is 1
    # w_r^(q-1) for the gradient, w_r^q for W
    if q == 1:
        # Phi_1 = mean, gradient = mean of G rows
        return S, (G.T @ np.ones(L)) / L
    wq1 = np.power(np.maximum(w, 0.0), q - 1)  # w^(q-1)
    wq = wq1 * w  # w^q
    W = float(wq.sum())
    if W < 1e-300:
        # Only one term survives approximately
        idx = int(np.argmax(F))
        val = S
        grad = G[idx] * (1.0 / (L ** (1.0/q)))  # approximate
        return val, grad
    Phi_q = S * W ** (1.0 / q)
    # gradient: W^(1/q - 1) * sum_r w_r^(q-1) * G[r, i]
    # = W^((1-q)/q) * G.T @ wq1
    coeff = W ** ((1.0 - q) / q)
    grad = coeff * (G.T @ wq1)
    return Phi_q, grad

def _pnorm_optimizer(h, q, deadline, max_iter=200):
    n = len(h)
    target = n / 2.0
    G = _build_G(h)
    
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        val, _ = _pnorm_obj_grad(x, q, G)
        return val
    
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        val, grad = _pnorm_obj_grad(x, q, G)
        return val, grad

        return np.array([x.sum() - target])
    
    def cons_jac(x):
        return np.array([np.ones(n)])

    bounds = [(0.0, 1.0)] * n

    best_c5 = _exact_c5(best_h)

            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:

                best_c5 = c5_new

        pass

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_descent(h, deadline, max_passes=200, verify_thresh=1e-6):

    h = h.copy()
    best_h = h.copy()

            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > verify_thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0, 1)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True

                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        h = np.clip(h, 0, 1)

            break

def _quasi_init(n, alpha, rng):

    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)

def _smooth_parent(h, kernel_width=2):

    h_smooth = np.convolve(h_padded, kernel, mode='valid')
    return _project(h_smooth, n / 2.0)

F_r = dx * sum_{n=lo}^{hi} h[n+d] * (1-h[n])

This is a sum of products, so:

For the first sum: h[i] = h[n+d] means n = i-d. Valid if lo <= i-d <= hi.
Contribution: dx * (1-h[i-d]) for each valid i (but there's only one n for each i, so just one term).

Contribution: -dx * h[i+d] for each valid i (again, just one term).

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n - 1, n - 1 - d)
        i_lo1 = lo + d
        i_hi1 = hi + d
        if i_lo1 <= i_hi1:
            i1 = np.arange(i_lo1, i_hi1 + 1)
            G[r, i1] += dx * (1.0 - h[i1 - d])
        if lo <= hi:
            i2 = np.arange(lo, hi + 1)
            G[r, i2] += -dx * h[i2 + d]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pnorm_val_grad(h, q):
    F = _scores(h)
    L = len(F)
    n = len(h)
    S = float(F.max())
    if S < 1e-30:
        return 0.0, np.zeros(n)
    w = F / S
    if q <= 1.01:
        return S, (w @ np.ones(L)) / L
    wq1 = np.power(np.maximum(w, 1e-30), q - 1)
    wq = wq1 * w
    W = float(wq.sum())
    if W < 1e-300:
        idx = int(np.argmax(F))
        G = _build_G(h)
        return S, G[idx].copy()
    Phi = S * W ** (1.0 / q)
    G = _build_G(h)
    coeff = W ** ((1.0 - q) / q)
    grad = coeff * (G.T @ wq1)
    return Phi, grad

def _pair_descent(h, deadline, max_passes=200, verify_thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > verify_thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0, 1)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        h = np.clip(h, 0, 1)
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    h_new, c5_new = _lp_local(best_h, dl2, top_k=None, radius_init=0.1, max_iters=5000)
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new.copy()

    # Phase 3: Pair descent
    if time.time() < t0 + budget_s * 0.40:
        dl3 = t0 + budget_s * 0.42
        best_h, best_c5 = _pair_descent(best_h, dl3, max_passes=300, verify_thresh=1e-7)

    # Phase 4: Second p-norm round with higher q
    q_list2 = [128, 256, 512]
    dl4 = t0 + budget_s * 0.52
    for q in q_list2:
        if time.time() > dl4:
            break
        h_new, c5_new = _pnorm_opt(best_h, q, dl4, max_iter=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = len(best_h)

    # Phase 5: LP + pair descent again
    if time.time() < t0 + budget_s * 0.62:
        dl5 = t0 + budget_s * 0.64
        h_new, c5_new = _lp_local(best_h, dl5, top_k=8, radius_init=0.05, max_iters=3000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if time.time() < dl5:
            best_h, best_c5 = _pair_descent(best_h, dl5, max_passes=200, verify_thresh=1e-8)

    # Phase 6: Multi-resolution exploration
    dl6 = t0 + budget_s * 0.75
    for n_try in [160, 192, 256, 128, 320]:
        if time.time() > dl6:
            break
        h_rs = _resample(best_h, n_try)
        sub_dl = min(dl6, time.time() + budget_s * 0.04)
        h_new, c5_new = _pnorm_opt(h_rs, 64, sub_dl, max_iter=200)
        if c5_new > best_c5:
            sub_dl2 = min(dl6, time.time() + budget_s * 0.02)
            h_new, c5_new = _lp_local(h_rs, sub_dl2, top_k=None, radius_init=0.1, max_iters=2000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            # Quick polish
            if time.time() < dl6:
                sub_dl3 = min(dl6, time.time() + budget_s * 0.015)
                best_h, best_c5 = _pair_descent(best_h, sub_dl3, max_passes=100, verify_thresh=1e-7)

    # Phase 7: Diverse quasi-random starts
    dl7 = t0 + budget_s * 0.86
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.555]
    for i, alpha in enumerate(alphas):
        if time.time() > dl7:
            break
        h_init = _quasi_init(best_n, alpha, rng)
        sub_dl = min(dl7, time.time() + budget_s * 0.02)
        h_try, c5_try = _pnorm_opt(h_init, 32, sub_dl, max_iter=200)
        if c5_try >= best_c5:
            sub_dl2 = min(dl7, time.time() + budget_s * 0.015)
            h_try, c5_try = _lp_local(h_try, sub_dl2, top_k=None, radius_init=0.12, max_iters=1500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(best_h)

    # Phase 8: Perturbation restarts
    dl8 = t0 + budget_s * 0.94
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl8:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl8, time.time() + budget_s * 0.007)
        h_try, c5_try = _lp_local(h_pert, sub_dl, top_k=5, radius_init=0.05, max_iters=300)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.12)

    # Phase 9: Final intensive polish
    if time.time() < hard_deadline:
        best_h, best_c5 = _pair_descent(best_h, hard_deadline, max_passes=300, verify_thresh=1e-9)
        if time.time() < hard_deadline - 5:
            h_fin, c5_fin = _lp_local(best_h, hard_deadline, top_k=4, radius_init=0.01, max_iters=2000)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final projection and verification
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        # Use the pre-projection value if projection worsened it
        final_c5 = best_c5
    return best_h, final_c5, best_n
