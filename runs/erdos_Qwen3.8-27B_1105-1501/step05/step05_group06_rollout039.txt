import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h, n):
    """Convert h to binary with exactly n/2 ones"""
    h = h.copy()
    # Sort by value, assign 1 to top n/2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def _build_G(h):
    """Build gradient matrix G where G[r,j] = d(scores[r])/d(h[j])"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lp_direction(h, G, scores, radius, top_k):
    """Find LP direction to reduce max score"""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores.copy()
        G_k = G
    
    # Minimize t s.t. scores + G*p <= t, |p| <= radius, sum(p) = 0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k - 1e-10  # need to reduce below current
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_c5(h, p, max_iter=30):
    """Golden section line search for C5 along direction p"""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):

            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])

        return 0.0, float(vals[0])
    if best_i == n_grid - 1:
        return alpha_max, float(vals[-1])
    
    # Golden section
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c2 = lo + gr * (hi - lo)

        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _pair_descent(h, deadline, max_passes=200):
    """Pair-wise descent: try swapping mass between pairs of positions"""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                # Ternary search for best alpha
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > 1e-6 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            break

    elif pattern == 'quadratic':

        h = np.zeros(n)
        h[idx[:n//2]] = 1.0

def _pair_descent(h, deadline, max_passes=200, thresh=1e-6):

                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:

                        lo_a = m1

                    h[k] -= alpha

                        h[k] += alpha

def _binary_sa(h_init, deadline, n_iters=150000, t_start=0.25, t_end=0.003):

def _steepest_binary(h_init, deadline, max_rounds=500):

def _structured_init(n, kind, rng):

    elif kind == 5:

        n_ones = int(h.sum())

            h[rng.choice(np.where(h == 1)[0], n_ones - n // 2, replace=False)] = 0
        elif n_ones < n // 2:
            h[rng.choice(np.where(h == 0)[0], n // 2 - n_ones, replace=False)] = 1
    else:
        h[:] = 0.5
    return _project(h, n / 2.0)

            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 10 and radius < 0.015:
                radius = 0.05
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=200, thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(14):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, n_iters=150000, t_start=0.25, t_end=0.003):
    n = len(h_init)
    b = _to_binary(h_init, n)
    cur_c5 = _exact_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(int(time.time() * 10000) % 999983)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for step in range(n_iters):
        if time.time() > deadline:
            break
        frac = step / max(1, n_iters - 1)
        T = t_start * (t_end / t_start) ** frac
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = _exact_c5(b)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def _steepest_binary(h_init, deadline, max_rounds=200):
    n = len(h_init)
    b = _to_binary(h_init, n)
    cur_c5 = _exact_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        best_delta = 0.0
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0
                b[j] = 1
                new_c5 = _exact_c5(b)
                b[i] = 1
                b[j] = 0
                delta = new_c5 - cur_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        b[i] = 0
        b[j] = 1
        cur_c5 += best_delta
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_b = b.copy()
    return best_b, best_c5

def _structured_init(n, kind, rng):
    h = np.zeros(n)
    if kind == 0:
        h[::2] = 1.0
    elif kind == 1:
        h[:n // 2] = 1.0
    elif kind == 2:
        x = np.arange(n) / n
        h = (np.sin(2 * np.pi * x * 3) > 0).astype(float)
    elif kind == 3:
        x = np.arange(n) / n
        h = (np.sin(2 * np.pi * x * 5) > 0.3).astype(float)
    elif kind == 4:
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    elif kind == 5:
        h = (np.arange(n) % 3 == 0).astype(float)
        n_ones = int(h.sum())
        if n_ones > n // 2:
            h[rng.choice(np.where(h == 1)[0], n_ones - n // 2, replace=False)] = 0
        elif n_ones < n // 2:
            h[rng.choice(np.where(h == 0)[0], n // 2 - n_ones, replace=False)] = 1
    else:
        h[:] = 0.5
    return _project(h, n / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)
