def _project(v, target):
    # Project v onto {x: 0<=x_i<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    x = np.clip(v - thresh, 0.0, 1.0)
    # Fix rounding
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

Now, for the LP-based direction finding:

At point h, the objective is f(h) = max_l corr[l] * dx. The "active" lags are those where corr[l]*dx is close to the max.

To find a descent direction, I solve:
min p_t (the t-variable)
s.t. corr[l](h + p) ≤ corr[l](h) + p_t for all l in active set
     sum(p) = 0 (to preserve the mass constraint, approximately)
     |p[i]| ≤ radius

But corr[l](h+p) is quadratic in p, not linear. The LP linearizes it:
corr[l](h+p) ≈ corr[l](h) + G[l] @ p + (p quadratic terms)

If I ignore the quadratic terms (which are small for small p), I get the LP. For larger p, the linearization is poor.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        G[n - 1 + d, d:] = dx * (1.0 - h[:m])
        G[n - 1 + d, :m] += -dx * h[d:]
        G[n - 1 - d, :m] = dx * (1.0 - h[d:])
        G[n - 1 - d, d:] += -dx * h[:m]
    return G

def _lp_direction(h, G, radius, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    scores = _all_scores(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _golden_1d(f, lo, hi, n_sec=30):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(n_sec):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, f2, c2 = c2, f1, c1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, f1, c1 = c1, f2, c2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    return (a + b) / 2.0

def _line_search(h, p, target):
    n = len(h)
    a_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_max = min(a_max, h[j] / (-p[j]))
    a_max = max(a_max, 1e-12)
    n_grid = 15
    alphas = np.linspace(0.0, a_max, n_grid)
    vals = np.array([_exact_c5(_project(h + a * p, target)) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0:
        return 0.0, float(vals[0])
    if best_i == n_grid - 1:
        return a_max, float(vals[-1])
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    def f_1d(a):
        return _exact_c5(_project(h + a * p, target))
    opt_a = _golden_1d(f_1d, lo, hi)
    opt_c5 = f_1d(opt_a)
    if opt_c5 < float(vals[best_i]):
        return opt_a, opt_c5
    return alphas[best_i], float(vals[best_i])

def _local_lp(h, deadline, top_k_list=None, radius_init=0.1, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 10, 6, 4, 3]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_tk = len(top_k_list)
    ti = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        tk = top_k_list[ti % n_tk]
        G = _build_G(h)
        p, _ = _lp_direction(h, G, radius, top_k=tk)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.65
            ti += 1
            if radius < 1e-9:
                break
            continue
        alpha, _ = _line_search(h, p, target)
        h_cand = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_cand)
        if actual_c5 < cur_c5 - 1e-14:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.7
            ti += 1
            if stall > 15 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=500, rng=None, subset_frac=1.0):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    pairs = [(j, k) for j in range(n) for k in range(j + 1, n)]
    n_pairs = len(pairs)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if subset_frac < 1.0:
            n_use = max(1, int(n_pairs * subset_frac))
            if rng is not None:
                idx = rng.choice(n_pairs, size=n_use, replace=False)
            else:
                idx = np.linspace(0, n_pairs - 1, n_use, dtype=int)
            use_pairs = [pairs[i] for i in idx]
        else:
            use_pairs = pairs
        pass_improved = False
        for (j, k) in use_pairs:
            if time.time() > deadline:
                break
            if h[j] < 1e-12 and h[k] < 1e-12:
                continue
            if h[j] > 1 - 1e-12 and h[k] > 1 - 1e-12:
                continue
            a_lo = max(-h[j], h[k] - 1.0)
            a_hi = min(1.0 - h[j], h[k])
            if a_hi - a_lo < 1e-12:
                continue
            h_work = h.copy()
            def f_pair(a):
                h_work[j] = h[j] + a
                h_work[k] = h[k] - a
                return _exact_c5(h_work)
            opt_a = _golden_1d(f_pair, a_lo, a_hi, n_sec=25)
            new_c5 = f_pair(opt_a)
            if new_c5 < cur_c5 - 1e-14 and abs(opt_a) > 1e-12:
                h[j] += opt_a
                h[k] -= opt_a
                cur_c5 = new_c5
                pass_improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not pass_improved:
            break
    return best_h, best_c5

def _random_dir_descent(h, deadline, n_dirs=200, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for _ in range(n_dirs):
        if time.time() > deadline:
            break
        d = rng.standard_normal(n)
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            continue
        d /= dn
        a_max = 1.0
        for j in range(n):
            if d[j] > 1e-14:
                a_max = min(a_max, (1.0 - h[j]) / d[j])
            elif d[j] < -1e-14:
                a_max = min(a_max, h[j] / (-d[j]))
        a_max = min(a_max, 0.5)
        if a_max < 1e-10:
            continue
        def f_dir(a):
            return _exact_c5(_project(h + a * d, target))
        opt_a = _golden_1d(f_dir, -a_max, a_max, n_sec=20)
        h_cand = _project(h + opt_a * d, target)

            cur_c5 = new_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_de(h_base, deadline, rng, m_basis=16, pop_size=30, max_gen=60):
    n = len(h_base)
    target = n / 2.0
    x = np.arange(n)
    basis = np.zeros((m_basis, n))
    for k in range(m_basis):
        basis[k] = np.cos(2.0 * np.pi * (k + 1) * x / n)
        basis[k] -= basis[k].mean()
    # Normalize basis vectors
    for k in range(m_basis):
        bn = np.linalg.norm(basis[k])
        if bn > 1e-10:
            basis[k] /= bn
    best_h = h_base.copy()
    best_c5 = _exact_c5(h_base)
    pop = np.zeros((pop_size, m_basis))
    pop_c5 = np.full(pop_size, np.inf)
    for i in range(pop_size):
        pop[i] = rng.normal(0, 0.03, m_basis)
        h_cand = _project(h_base + pop[i] @ basis, target)
        pop_c5[i] = _exact_c5(h_cand)
    best_i = int(np.argmin(pop_c5))
    if pop_c5[best_i] < best_c5:
        best_c5 = pop_c5[best_i]
        best_h = _project(h_base + pop[best_i] @ basis, target)
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            while r1 == r2 or r1 == r3 or r2 == r3:
                r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            mutant = pop[best_i] + 0.5 * (pop[r1] - pop[r2]) + 0.3 * (pop[r3] - pop[best_i])
            trial = pop[i] * 0.6 + mutant * 0.4
            h_cand = _project(h_base + trial @ basis, target)
            c5_trial = _exact_c5(h_cand)
            if c5_trial < pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = c5_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_cand.copy()
                    best_i = i
    return best_h, best_c5

def _block_opt(h, deadline, block_size=4, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    n_blocks = n // block_size
    for _sweep in range(200):
        if time.time() > deadline:
            break
        improved = False
        block_order = np.arange(n_blocks)
        if rng is not None:
            rng.shuffle(block_order)
        for bi in block_order:
            if time.time() > deadline:
                break
            lo_idx = bi * block_size
            hi_idx = lo_idx + block_size
            if hi_idx > n:
                continue
            block = h[lo_idx:hi_idx].copy()
            if np.all(block < 1e-10) or np.all(block > 1 - 1e-10):
                continue
            # Try small perturbations of this block
            for _trial in range(5):
                pert = rng.uniform(-0.1, 0.1, block_size) if rng is not None else np.zeros(block_size)
                pert[0] -= pert[1:].sum()  # try to preserve local sum
                h_trial = h.copy()
                h_trial[lo_idx:hi_idx] = np.clip(block + pert, 0, 1)
                h_trial = _project(h_trial, target)
                c5_trial = _exact_c5(h_trial)
                if c5_trial < cur_c5 - 1e-14:
                    h = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
        if not improved:
            break
    return best_h, best_c5

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _grad_step(h, deadline, n_steps=150, beta_start=20, beta_end=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _all_scores(h)
        max_s = scores.max()
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _build_G(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.05
        for _ in range(20):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Quick LP at original resolution
    dl1 = t0 + budget_s * 0.06
    h1, c5_1 = _local_lp(h_start, dl1, top_k_list=[None, 12, 8, 5, 3],
                         radius_init=0.15, max_iters=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Random direction descent at current resolution
    if time.time() < t0 + budget_s * 0.12:
        dl2 = t0 + budget_s * 0.12
        h2, c5_2 = _random_dir_descent(best_h, dl2, n_dirs=300, rng=rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Multi-resolution LP search
    dl3 = t0 + budget_s * 0.40
    resolutions = [128, 160, 192, 256, 320, 384, 448]
    alphas_qp = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.555, 0.654]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if best_c5 < 0.3803:
            break
        # Start from resampled best
        h_init = _resample(best_h, n_try)
        sub_dl = min(dl3, time.time() + budget_s * 0.04)
        h_try, c5_try = _local_lp(h_init, sub_dl,
                                  top_k_list=[None, 10, 6, 4],
                                  radius_init=0.12, max_iters=2000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
        # Also try quasi-random init
        if time.time() < dl3:
            alpha = alphas_qp[ri % len(alphas_qp)]
            h_init2 = _quasi_init(n_try, alpha, rng)
            sub_dl2 = min(dl3, time.time() + budget_s * 0.03)
            h_try2, c5_try2 = _local_lp(h_init2, sub_dl2,
                                         top_k_list=[None, 8, 5],
                                         radius_init=0.15, max_iters=1500)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()
                best_n = n_try

    # Phase 4: Pair descent (intensive)
    if time.time() < t0 + budget_s * 0.65:
        dl4 = t0 + budget_s * 0.65
        best_h, best_c5 = _pair_descent(best_h, dl4, max_passes=300, rng=rng,
                                         subset_frac=0.5)
        # LP polish
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.02)
            h_lp, c5_lp = _local_lp(best_h, sub_dl, top_k_list=[5, 4, 3, None],
                                     radius_init=0.04, max_iters=1000)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
        # More pair descent
        if time.time() < dl4:
            dl4b = min(dl4, time.time() + budget_s * 0.03)
            best_h, best_c5 = _pair_descent(best_h, dl4b, max_passes=200, rng=rng,
                                             subset_frac=0.7)

    # Phase 5: Fourier DE for global exploration
    if time.time() < t0 + budget_s * 0.75:
        dl5 = t0 + budget_s * 0.75
        h_de, c5_de = _fourier_de(best_h, dl5, rng, m_basis=16, pop_size=25, max_gen=50)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            # Local polish after DE
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_pol, c5_pol = _local_lp(best_h, sub_dl, top_k_list=[4, 3, 5],
                                       radius_init=0.05, max_iters=500)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Gradient steps + random directions
    if time.time() < t0 + budget_s * 0.82:
        dl6 = t0 + budget_s * 0.82
        h_g, c5_g = _grad_step(best_h, min(dl6, time.time() + budget_s * 0.025),
                               n_steps=80, beta_start=30, beta_end=300)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        if time.time() < dl6:
            h_rd, c5_rd = _random_dir_descent(best_h, dl6, n_dirs=400, rng=rng)
            if c5_rd < best_c5:
                best_c5 = c5_rd
                best_h = h_rd.copy()

    # Phase 7: Diverse restarts at various resolutions
    if time.time() < t0 + budget_s * 0.90:
        dl7 = t0 + budget_s * 0.90
        for n_try in [128, 192, 256]:
            if time.time() > dl7:
                break
            for attempt in range(4):
                if time.time() > dl7:
                    break
                # Mix of resampled and random
                if attempt < 2:
                    h_init = _resample(best_h, n_try)
                    h_init = h_init + rng.normal(0, 0.04, n_try)
                    h_init = _project(h_init, n_try / 2.0)
                else:
                    alpha = [0.6180339887, 0.3819660113][attempt - 2]
                    h_init = _quasi_init(n_try, alpha, rng)
                sub_dl = min(dl7, time.time() + budget_s * 0.02)
                h_try, c5_try = _local_lp(h_init, sub_dl,
                                           top_k_list=[None, 8, 5, 3],
                                           radius_init=0.12, max_iters=1200)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 8: Perturbation restarts (iterated local search)
    dl8 = t0 + budget_s * 0.95
    scale = 0.02
    for attempt in range(300):
        if time.time() > dl8:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl8, time.time() + budget_s * 0.005)
        h_try, c5_try = _local_lp(h_pert, sub_dl, top_k_list=[4, 3, None],
                                   radius_init=0.04, max_iters=300)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.07, scale * 1.08)

    # Phase 9: Final intensive pair descent + LP
    if time.time() < hard_deadline:
        dl9 = hard_deadline
        h_pd, c5_pd = _pair_descent(best_h, dl9, max_passes=200, rng=rng,
                                     subset_frac=1.0)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
        if time.time() < dl9 - 1:
            h_fin, c5_fin = _local_lp(best_h, dl9, top_k_list=[3, 4, 5, 6],
                                       radius_init=0.01, max_iters=200)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
