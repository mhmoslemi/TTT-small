def _block_swap_dir(h, rng, block_size):

    stall = 0
    dx = 2.0 / n
    while time.time() < deadline:
        if stall > max_stall:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _dense_line_search(h, p, dx, radius, n_grid=45)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.35)
        else:
            stall += 1
            radius *= 0.72
            if stall % 25 == 0 and radius < 0.012:
                radius = 0.06
    return best_h, best_c5


def _slsqp_lse_optimize(h, deadline):
    n = len(h)
    target = n / 2.0
    h_work = h.copy()
    best_c5 = _exact_c5(h_work)
    best_h = h_work.copy()
    tau_list = [0.08, 0.04, 0.02, 0.01, 0.005]
    for tau in tau_list:
        if time.time() > deadline:
            break
        if deadline - time.time() < 5:
            break
        try:
            res = minimize(_lse_obj, h_work, jac=_lse_grad, args=(target, tau),
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 150, 'maxfun': 400, 'ftol': 1e-10})
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_work = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5


def _lowdim_to_h(ctrl, n):
    x_ctrl = np.linspace(0.0, 2.0, len(ctrl))
    x_grid = (np.arange(n) + 0.5) * (2.0 / n)
    h_raw = np.interp(x_grid, x_ctrl, ctrl)
    return _project(h_raw, n / 2.0)


def _lowdim_obj(ctrl, n):
    h = _lowdim_to_h(ctrl, n)
    return _exact_c5(h)


def _lowdim_powell(h_start, n, deadline, K=14):
    x_ctrl = np.linspace(0.0, 2.0, K)
    x_grid = (np.arange(n) + 0.5) * (2.0 / n)
    ctrl0 = np.interp(x_ctrl, x_grid, h_start)
    bounds = [(-1.0, 2.0)] * K
    best_c5 = _exact_c5(h_start)
    best_ctrl = ctrl0.copy()

    def _wrap_obj(c):
        return _lowdim_obj(c, n)

    try:
        res = powell(_wrap_obj, ctrl0, bounds=bounds, maxiter=300, tol=1e-8,
                     maximize=False)
        h_new = _lowdim_to_h(res.x, n)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_ctrl = res.x.copy()
    except Exception:
        pass
    return _lowdim_to_h(best_ctrl, n), best_c5


def _random_dir(rng, n, dtype='smooth'):
    if dtype == 'smooth':
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.3, 1.0, size=4)
        x = np.linspace(0, 2, n)
        d = np.zeros(n)
        for f, p, a in zip(freqs, phases, amps):
            d += a * np.sin(np.pi * f * x + p)
    elif dtype == 'block':
        d = np.zeros(n)
        nb = rng.integers(3, max(4, n // 8))
        i = int(rng.integers(0, n - nb))
        j = int(rng.integers(0, n - nb))
        if abs(i - j) < nb:
            j = (j + nb // 2 + int(rng.integers(0, n // 2))) % max(1, n - nb)
        d[i:i + nb] = -1.0 / nb
        d[j:j + nb] = 1.0 / nb
    else:
        d = rng.standard_normal(n)
    d -= np.mean(d)
    mx = np.max(np.abs(d))
    if mx < 1e-12:
        return None
    return d / mx


def _diverse_line_search(h, deadline, n_dirs=500, radius=0.3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0

    while time.time() < deadline:
        if stall > 200:
            break
        dtype = rng.choice(['smooth', 'block', 'random'])
        d = _random_dir(rng, n, dtype)
        if d is None:
            stall += 1
            continue
        alpha, new_c5 = _dense_line_search(h, d, dx, radius, n_grid=40)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.6)
        else:
            stall += 1
            if stall % 50 == 0:
                radius *= 0.9
    return best_h, best_c5


def _grad_direction_search(h, deadline, radius=0.35):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    tau = 0.02

    while time.time() < deadline:
        if stall > 40:
            break
        _, grad = _lse_value_grad(h, tau)
        d = -grad
        d -= np.mean(d)
        mx = np.max(np.abs(d))
        if mx < 1e-12:
            break
        d = d / mx
        alpha, new_c5 = _dense_line_search(h, d, dx, radius, n_grid=50)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.5)
        else:
            stall += 1
            radius *= 0.85
            tau *= 1.3
            if tau > 0.2:
                tau = 0.02
                break

    return best_h, best_c5


def _pair_swap_optimize(h, deadline, n_tries=3000, radius=0.3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(4242)
    stall = 0
    while time.time() < deadline and stall < 600:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _dense_line_search(h, d, dx, radius, n_grid=30)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.02, 0.5)
        else:
            stall += 1
            if stall % 100 == 0:
                radius *= 0.9
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _perturb_reoptimize(best_h, deadline, n_pert=50, base_scale=0.05):
    n = len(best_h)
    best_c5 = _exact_c5(best_h)
    best_h_out = best_h.copy()
    rng = np.random.default_rng(31337)
    scale = base_scale
    for _ in range(n_pert):
        if time.time() > deadline:
            break
        h_pert = best_h_out + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.08, max_stall=25)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h_out = h_t.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.15, scale * 1.15)
    return best_h_out, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    # Validate parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick LP warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_optimize(h_init, d1, radius_init=0.12, max_stall=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LSE optimization (18%)
    d2 = t0 + budget_s * 0.26
    h2, c5_2 = _slsqp_lse_optimize(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (18%)
    d3 = t0 + budget_s * 0.44
    n_curr = len(best_h)
    offsets = [8, -8, 16, -16, 24, -24, 4, -4, 12, -12, 32, -32]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.15, max_stall=30)
        rem = d3 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d3, time.time() + max(3.0, budget_s * 0.006))
            h_r, c5_r = _slsqp_lse_optimize(h_r, sub_dl2)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Low-dimensional global search (15%)
    d4 = t0 + budget_s * 0.59
    n_best = len(best_h)
    for K in [10, 14, 18]:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_low, c5_low = _lowdim_powell(best_h, n_best, sub_dl, K=K)
        if c5_low < best_c5:
            best_c5 = c5_low
            best_h = h_low.copy()
    # Also try from different starts
    for K in [12, 16]:
        if time.time() > d4:
            break
        x_ctrl = np.linspace(0.0, 2.0, K)
        x_grid = (np.arange(n_best) + 0.5) * (2.0 / n_best)
        ctrl_rand = np.interp(x_ctrl, x_grid, best_h) + rng.normal(0, 0.1, K)
        h_rand = _lowdim_to_h(ctrl_rand, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.03)

        def _wrap2(c):
            return _lowdim_obj(c, n_best)

        def _wrap2(c):
            return _lowdim_obj(c, n_best)

        try:
            res = powell(_lowdim_obj, ctrl_rand, bounds=[(-1.0, 2.0)] * K,
                         args=(n_best,), maxiter=200, tol=1e-8)
            h_new = _lowdim_to_h(res.x, n_best)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 5: Gradient direction search (12%)
    d5 = t0 + budget_s * 0.71
    h5, c5_5 = _grad_direction_search(best_h, d5, radius=0.35)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Diverse line search (12%)
    d6 = t0 + budget_s * 0.83
    h6, c5_6 = _diverse_line_search(best_h, d6, n_dirs=800, radius=0.3)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Intensive LP + pair swap (10%)
    d7 = t0 + budget_s * 0.93
    h7, c5_7 = _lp_optimize(best_h, min(d7, time.time() + budget_s * 0.04),
                            radius_init=0.06, max_stall=40)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h7, c5_7 = _pair_swap_optimize(best_h, min(d7, time.time() + budget_s * 0.04),
                                   n_tries=2000, radius=0.25)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Perturbation + reoptimize (5%)
    d8 = deadline
    h8, c5_8 = _perturb_reoptimize(best_h, d8, n_pert=60, base_scale=0.04)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final polish
    if time.time() < deadline - 2:
        h_fin, c5_fin = _lp_optimize(best_h, deadline, radius_init=0.04, max_stall=30)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
