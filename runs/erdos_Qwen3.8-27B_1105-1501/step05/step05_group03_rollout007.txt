            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    return (lo + hi) / 2.0

def _active_lp_optimize(h, deadline, n_top=15, radius_init=0.15, max_iters=3000, radius_min=0.001):
    """Optimize using active-set LP."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        s = _scores(h)
        k = min(n_top, len(s))
        idx = np.argpartition(-s, k-1)[:k]
        active_scores = s[idx]
        
        p = _lp_active(h, idx, active_scores, radius, n)
        if p is None:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
            continue
        
        alpha = _line_search(h, p, target)
        h_try = _project(h + alpha * p, target)
        new_c5 = _exact_c5(h_try)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    
    return best_h, best_c5

def _bit_sa(n, deadline, rng, best_c5_init, n_iters=50000):
    """Simulated annealing on binary sequences."""
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    val = 0
    for i in range(n):
        if h[i]:
            val |= (1 << i)
    
    masks = [(1 << (n - d)) - 1 for d in range(1, n)]
    cur_c5 = _bit_c5(val, n)
    best_val = val
    best_c5 = cur_c5
    
    T = 0.5
    T_min = 0.01
    n_total = n_iters
    cool = (T_min / T) ** (1.0 / max(n_total, 1))
    
    ones = rng.permutation(target)  # track which positions are 1
    zeros = rng.permutation(n - target)
    
    for it in range(n_total):
        if time.time() > deadline:
            break
        i = ones[rng.integers(target)]
        j = zeros[rng.integers(n - target)]
        # Swap: set bit i to 0, bit j to 1
        val_new = (val & ~(1 << i)) | (1 << j)
        new_c5 = _bit_c5(val_new, n)
        
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            val = val_new
            ones[i // n] = j  # This isn't right, let me fix
            # Actually, let me just track the set of 1-positions
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_val = val
        else:
            pass  # revert
        T *= cool
    
    h_best = np.zeros(n)
    for i in range(n):
        if (best_val >> i) & 1:
            h_best[i] = 1.0
    return h_best, best_c5

def _bit_sa_v2(n, deadline, rng, n_iters=50000):
    """Better bit SA with proper tracking."""
    target = n // 2
    val = 0
    # Initialize: first 'target' bits set
    for i in range(target):
        val |= (1 << i)
    
    masks = [(1 << (n - d)) - 1 for d in range(1, n)]
    cur_c5, cur_max = _bit_c5_fast(val, n, masks)
    best_val = val
    best_c5 = cur_c5
    
    T = 1.0 / n  # start temperature
    T_min = T * 0.01
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Pick a random 1-bit to clear and 0-bit to set
        i = rng.integers(n)
        j = rng.integers(n)
        if (val >> i) & 1:
            if ((val >> j) & 1) == 0:
                val_new = (val & ~(1 << i)) | (1 << j)
            else:
                continue
        else:
            if ((val >> j) & 1) == 1:
                val_new = (val | (1 << i)) & ~(1 << j)
            else:
                continue
        
        new_c5, new_max = _bit_c5_fast(val_new, n, masks)
        
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            val = val_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_val = val
        T *= cool
    
    h_best = np.zeros(n)
    for i in range(n):
        if (best_val >> i) & 1:
            h_best[i] = 1.0
    return h_best, best_c5

def _quasi_init(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.05, n)
        h = _project(h, n / 2.0)
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _momentum_grad(h, deadline, beta_start=50, beta_end=500, n_steps=200, step_init=0.04):
    """Smoothed gradient with momentum, focusing on top constraints."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        s = _scores(h)
        max_s = np.max(s)
        # Focus on top constraints
        w = np.exp(beta * (s - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(s))
            w[np.argmax(s)] = 1.0
        else:
            w /= sw
        
        # Compute gradient as weighted sum of per-lag gradients
        n_lags = len(s)
        active_idx = np.where(w > 0.01 / n_lags)[0]
        if len(active_idx) == 0:
            active_idx = [np.argmax(s)]
        
        grad = np.zeros(n)
        for idx in active_idx:
            grad += w[idx] * _grad_for_lags(h, [idx], n)
        
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.03, 0.1)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    
    return best_h, best_c5

def _coordinate_polish(h, deadline, n_cycles=2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(30):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    
    return best_h, best_c5

def _pair_polish(h, deadline, n_passes=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_size = max(1, n // 32)
    
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(0, n, step_size):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Start from initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    # Phase 1: Active-set LP optimization (faster than full LP)
    if time.time() < t0 + budget_s * 0.12:
        dl = t0 + budget_s * 0.12
        h1, c5_1 = _active_lp_optimize(h_start, dl, n_top=12, radius_init=0.15, max_iters=5000, radius_min=0.001)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Phase 2: Momentum gradient
    if time.time() < t0 + budget_s * 0.20:
        dl = t0 + budget_s * 0.20
        h2, c5_2 = _momentum_grad(best_h, dl, beta_start=20, beta_end=500, n_steps=300, step_init=0.04)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        # Follow-up LP
        if time.time() < t0 + budget_s * 0.25:
            dl2 = t0 + budget_s * 0.25
            h2b, c5_2b = _active_lp_optimize(best_h, dl2, n_top=10, radius_init=0.08, max_iters=3000, radius_min=0.001)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()
    
    # Phase 3: Bit-level SA at small resolutions to find good patterns
    if time.time() < t0 + budget_s * 0.35:
        dl3 = t0 + budget_s * 0.35
        for n_bin in [64, 96, 128]:
            if time.time() > dl3:
                break
            sub_dl = min(dl3, time.time() + budget_s * 0.03)
            h_b, c5_b = _bit_sa_v2(n_bin, sub_dl, rng, n_iters=80000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                # Polish the bit solution
                if time.time() < dl3:
                    sub_dl2 = min(dl3, time.time() + budget_s * 0.015)
                    h_bp, c5_bp = _active_lp_optimize(h_b, sub_dl2, n_top=10, radius_init=0.10, max_iters=1000, radius_min=0.005)
                    if c5_bp < best_c5:
                        best_c5 = c5_bp
                        best_h = h_bp.copy()
    
    # Phase 4: Multi-resolution with diverse inits
    if time.time() < t0 + budget_s * 0.45:
        dl4 = t0 + budget_s * 0.45
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5]
        for n_try in [96, 128, 128, 96, 64]:
            if time.time() > dl4:
                break
            if best_c5 < 0.3808:
                break
            alpha = alphas[len([a for a in alphas if True]) % len(alphas)]  # just cycle
            # Use different alphas
            alpha = alphas[(n_try + len(alphas)) % len(alphas)]
            h_init = _quasi_init(n_try, alpha, rng)
            # Also try resampling best
            if rng.random() < 0.5:
                h_init = _resample(best_h, n_try)
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_try, c5_try = _active_lp_optimize(h_init, sub_dl, n_top=12, radius_init=0.15, max_iters=2000, radius_min=0.003)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
    
    # Phase 5: More LP + momentum refinement
    if time.time() < t0 + budget_s * 0.55:
        dl5 = t0 + budget_s * 0.55
        h5, c5_5 = _momentum_grad(best_h, dl5, beta_start=50, beta_end=800, n_steps=300, step_init=0.03)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < t0 + budget_s * 0.62:
            dl5b = t0 + budget_s * 0.62
            h5b, c5_5b = _active_lp_optimize(best_h, dl5b, n_top=8, radius_init=0.05, max_iters=3000, radius_min=0.0005)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()
    
    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.80
    scale = 0.02
    for attempt in range(500):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.008)
        h_try, c5_try = _active_lp_optimize(h_pert, sub_dl, n_top=10, radius_init=0.05, max_iters=800, radius_min=0.005)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.05)
    
    # Phase 7: Coordinate polish
    if time.time() < t0 + budget_s * 0.90:
        dl7 = t0 + budget_s * 0.90
        hc, c5c = _coordinate_polish(best_h, dl7, n_cycles=3)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()
    
    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.95:
        dl8 = t0 + budget_s * 0.95
        hp, c5p = _pair_polish(best_h, dl8, n_passes=4)
