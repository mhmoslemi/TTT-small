    out[~pos] = ex / (1.0 + ex)
    return out

        else:
            stall += 1
            if stall > 20:
                sigma = min(0.9, sigma * 1.15)
                if stall > 50:
                    pop[-1] = best_v.copy() + sigma * 1.5 * rng.normal(0, 1, n)
                    np.clip(pop[-1], -35.0, 35.0, out=pop[-1])
        
        if gen % 10 == 0 and len(pop) < pop_size:
            v_rand = best_v.copy() + sigma * 0.8 * rng.normal(0, 1, n)
            np.clip(v_rand, -35.0, 35.0, out=v_rand)
            h_r = _logit_to_h(v_rand, n)
            c5_r = _exact_c5(h_r)
            pop.append(v_rand)
            scores.append(c5_r)
            combined_info = list(zip(scores, pop))
            combined_info.sort(key=lambda x: x[0])
            keep = min(pop_size, len(combined_info))
            pop = [x[1] for x in combined_info[:keep]]
            scores = [x[0] for x in combined_info[:keep]]
        gen += 1
    return best_v, best_h, best_c5

def _multi_res(best_h, deadline, rng):
    n = len(best_h)
    best_c5 = _exact_c5(best_h)
    best_out = best_h.copy()
    offsets = list(range(-16, 17, 2))
    rng.shuffle(offsets)
    for dn in sorted(offsets, key=lambda x: (abs(x), x)):
        if time.time() > deadline:
            break
        n_try = n + dn
        if n_try < 64 or n_try > 200:
            continue
        x_old = (np.arange(n) + 0.5) / n
        x_new = (np.arange(n_try) + 0.5) / n_try
        h_new = np.interp(x_new, x_old, best_h)
        h_new = _project(h_new, n_try / 2.0)
        c5_new = _exact_c5(h_new)
        sub_dl = min(deadline, time.time() + 8.0)
        h_ref, c5_ref = _pair_refine(h_new, sub_dl, rng, max_passes=25)
        if c5_ref < best_c5 - 1e-13:
            best_c5 = c5_ref
            best_out = h_ref.copy()
    return best_out, best_c5

def _kick_refine(h, deadline, rng, n_kicks=25):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.03, 0.10)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _pair_refine(h_k, sub_dl, rng, max_passes=15)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 122
        h0 = np.ones(n0) * 0.5
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick local refinement (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _pair_refine(h, d1, rng, max_passes=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient refinement (10%)
    d2 = t0 + budget_s * 0.20
    h2, c5_2 = _gradient_refine(best_h, d2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Logit Evolution Strategy (40%)
    d3 = t0 + budget_s * 0.60
    v0 = _h_to_logit(best_h, 122, eps=2e-4)
    n_pop = len(best_h)
    v_best, h_es, c5_es = _es_evolve(v0, n_pop, d3, rng, max_stall=100)
    if c5_es < best_c5:
        best_c5 = c5_es
        best_h = h_es.copy()

    # Phase 4: Local pair refinement after ES (15%)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = _pair_refine(best_h, d4, rng, max_passes=60)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Gradient refinement again (5%)
    d5 = t0 + budget_s * 0.80
    h5, c5_5 = _gradient_refine(best_h, d5, rng)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (12%)
    d6 = t0 + budget_s * 0.92
    h6, c5_6 = _multi_res(best_h, d6, rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Kick and refine (5%)
    d7 = t0 + budget_s * 0.97
    h7, c5_7 = _kick_refine(best_h, d7, rng, n_kicks=15)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final: ensure feasibility and re-evaluate
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    
    if not np.isfinite(final_c5) or final_c5 > 0.5:
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
