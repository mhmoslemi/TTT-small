    t_min = max(-h[p], h[q] - 1.0)
    t_max = min(1.0 - h[p], h[q])
    if t_max - t_min < 1e-12:
        return 0.0, _exact_c5(h)
    
    # Coarse sampling
    n_samp = 15
    ts = np.linspace(t_min, t_max, n_samp + 2)[1:-1]
    best_t = 0.0
    best_val = _exact_c5(h)
    
    for t in ts:
        v = _c5_pair_inplace(h, p, q, t)
        if v < best_val:
            best_val = v
            best_t = t
    
    if best_t == 0.0:
        return 0.0, best_val
    
    # Golden section refinement
    frac = (best_t - t_min) / (t_max - t_min)
    idx = int(frac * n_samp)
    lo = ts[max(0, idx - 1)]
    hi = ts[min(n_samp - 1, idx + 1)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5_pair_inplace(h, p, q, c1)
    f2 = _c5_pair_inplace(h, p, q, c2)
    
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _c5_pair_inplace(h, p, q, c1)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _c5_pair_inplace(h, p, q, c2)
    
    opt_t = (lo + hi) / 2.0
    opt_val = _c5_pair_inplace(h, p, q, opt_t)
    
    if best_val < opt_val:
        opt_t, opt_val = best_t, best_val
    
    return opt_t, opt_val

def _pair_local_opt(h_init, deadline, rng, max_stall=80):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline and stall < max_stall:
        scores = _scores(h)
        max_s = float(np.max(scores))
        tau = max(0.003 * max_s, 1e-8)
        w = np.exp((scores - max_s) / tau)
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
        
        G = _grad_matrix(h)
        pressure = G.T @ w
        pressure -= pressure.mean()
        
        # Sort by pressure
        order = np.argsort(pressure)
        n_quart = max(8, n // 4)
        low_idx = order[:n_quart]
        high_idx = order[-n_quart:]
        
        improved = False
        pair_limit = 800
        tried = 0
        
        for i in range(n_quart):
            if improved or time.time() > deadline:
                break
            for j in range(n_quart):
                if tried >= pair_limit:
                    break
                p = low_idx[j]   # increase (low pressure)
                q = high_idx[i]  # decrease (high pressure)
                if p == q:
                    continue
                tried += 1
                t, val = _pair_line_search(h, p, q)
                if val < cur_c5 - 1e-14:
                    h[p] += t
                    h[q] -= t
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        
        if not improved:
            # Try random pairs
            n_rand = 300
            for _ in range(n_rand):
                if time.time() > deadline:
                    break
                p = int(rng.integers(0, n))
                q = int(rng.integers(0, n))
                if p == q:
                    continue
                t, val = _pair_line_search(h, p, q)
                if val < cur_c5 - 1e-14:
                    h[p] += t
                    h[q] -= t
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha, frac):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < frac
    h = h.astype(float)
    return _project(h, n / 2.0)

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.permutation(n)[:n // 2]
    h[pos] = 1.0
    return h

def _binary_sa(h_init, deadline, rng, n_iters=50000):
    n = len(h_init)
    target = n // 2
    h = h_init.copy()
    # Ensure binary
    h = (h >= 0.5).astype(float)
    # Adjust to exactly target ones
    n_ones = int(np.sum(h))
    if n_ones > target:
        ones = np.where(h == 1.0)[0]
        excess = n_ones - target
        flips = rng.choice(ones, size=excess, replace=False)
        h[flips] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        deficit = target - n_ones
        flips = rng.choice(zeros, size=deficit, replace=False)
        h[flips] = 1.0
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    T = 0.003
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = _exact_c5(h)
        
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
        
        T = max(T * cool, T_min)
    
    return best_h, best_c5
