def _exact_c5(h):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):

        if s > target:

def _grad_scores_vec(h):
    """Gradient of scores w.r.t. h, shape (2n-1, n). Computed in O(n^2)."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            # j from 0 to n-1
            # term1: (1-h[j-d]) if j >= d
            # term2: -h[j+d] if j < n-d
            term1 = np.zeros(n)
            term2 = np.zeros(n)
            if d < n:
                term1[d:] = 1.0 - h[:n-d]
                term2[:n-d] = -h[d:]
            G[r] = dx * (term1 + term2)
        else:
            g = -d
            # term1: (1-h[j+g]) if j < n-g
            # term2: -h[j-g] if j >= g
            term1 = np.zeros(n)
            term2 = np.zeros(n)
            if g < n:
