def _grad_smoothed(h, beta):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    
    max_s = scores.max()
    exp_s = np.exp((scores - max_s) / beta)
    p = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for r in range(n_lags):
        k = r - (n - 1)
        w = p[r] * dx
        if w < 1e-18:
            continue
        # (1-h[i+k]) for i where 0<=i+k<n, i.e., i in [max(0,-k), n-1-min(0,k)]
        # Actually: 0<=i<n (always) and 0<=i+k<n means -k<=i<n-k
        # So i in [max(0, -k), min(n-1, n-1-k)]
        lo1 = max(0, -k)
        hi1 = min(n - 1, n - 1 - k)  # inclusive
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1+k:hi1+1+k])
        
        # -h[i-k] for i where 0<=i-k<n, i.e., k<=i<n+k

            grad[lo2:hi2+1] += w * (-h[lo2-k:hi2+1-k])

                radius = 0.08
    return best_h, best_c5

def _pgs_optimize(h, deadline, beta_init=0.01, max_iter=10000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = beta_init
    lr = 0.08
    momentum = 0.8
    m = np.zeros(n)
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if lr < 1e-6:
            beta = max(0.0005, beta * 0.6)
            lr = 0.08
            m *= 0.2
            if beta < 0.0005:
                break
        grad = _grad_smoothed(h, beta)
        m = momentum * m + grad
        v = h - lr * m
        h_new = _project(v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.4)
            if it % 300 == 0 and it > 0:
                beta = max(0.0005, beta * 0.9)
        else:
            no_improve += 1
            lr *= 0.65
            m = momentum * 0.7 * m
            if no_improve > 200:
                beta = max(0.0005, beta * 0.7)
                no_improve = 0
                lr = 0.08
    return best_h, best_c5

def _slsqp_refine(h0, beta, maxiter=200):
    n = len(h0)
    _SLSQP_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_constraint_jac}]
    res = minimize(_slsqp_smoothed_obj, h0, method='SLSQP',
                   jac=lambda x: _slsqp_smoothed_grad(x, beta),
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_out = _project(res.x, n / 2.0)
        return h_out, _exact_c5(h_out)
    return h0.copy(), _exact_c5(h0)

def _slsqp_sharp(h0, maxiter=150):
    n = len(h0)
    _SLSQP_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_constraint_jac}]
    res = minimize(_slsqp_obj, h0, method='SLSQP',
                   jac=_slsqp_obj_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_out = _project(res.x, n / 2.0)
        return h_out, _exact_c5(h_out)
    return h0.copy(), _exact_c5(h0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coord_refine(h, deadline, n_iter=800):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(31415)
    interior = np.where((h > 5e-9) & (h < 1.0 - 5e-9))[0]
    if len(interior) < 2:
        interior = np.arange(n)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j]) + 1e-12)
        if c5_try < best_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 5e-9) & (h < 1.0 - 5e-9))[0]
                if len(interior) < 2:
                    interior = np.arange(n)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LP from initial (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.18)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP with log-sum-exp from multiple starts (20%)
    d2 = t0 + budget_s * 0.32
    starts = [best_h.copy()]
    for _ in range(4):
        if time.time() > d2:
            break
        pert = best_h + rng.normal(0, 0.05, n0)
        starts.append(_project(pert, n0 / 2.0))
    for beta in [0.04, 0.02, 0.01, 0.005, 0.003]:
        if time.time() > d2:
            break
        for s in starts[:3]:
            if time.time() > d2:
                break
            try:
                h_s, c5_s = _slsqp_refine(s, beta, maxiter=200)
                if c5_s < best_c5 - 1e-12:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    starts = [best_h.copy()]
                    start_pert = best_h + rng.normal(0, 0.03, n0)
                    starts.append(_project(start_pert, n0 / 2.0))
            except Exception:
                pass

    # Phase 3: PG with log-sum-exp (15%)
    d3 = t0 + budget_s * 0.47
    h3, c5_3 = _pgs_optimize(best_h, d3, beta_init=0.008, max_iter=8000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution exploration (18%)
    d4 = t0 + budget_s * 0.65
    n_cur = len(best_h)
    tried = set()
    for dn in range(-20, 21):
        if time.time() > d4:
            break
        n_try = n_cur + dn
        if n_try < 28 or n_try > 180 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + 5.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try

    # Phase 5: SLSQP sharp (subgradient-based) (10%)
    d5 = t0 + budget_s * 0.75
    try:
        h5, c5_5 = _slsqp_sharp(best_h, maxiter=200)
        if c5_5 < best_c5 - 1e-12:
            best_c5 = c5_5
            best_h = h5.copy()
    except Exception:
        pass
    if time.time() < d5:
        try:
            pert = best_h + rng.normal(0, 0.02, len(best_h))
            h5b, c5_5b = _slsqp_sharp(_project(pert, len(best_h) / 2.0), maxiter=150)
            if c5_5b < best_c5 - 1e-12:
                best_c5 = c5_5b
                best_h = h5b.copy()
        except Exception:
            pass

    # Phase 6: PG refine (10%)
    d6 = t0 + budget_s * 0.85
    h6, c5_6 = _pgs_optimize(best_h, d6, beta_init=0.004, max_iter=6000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: LP polish (8%)
    d7 = t0 + budget_s * 0.93
    h7, c5_7 = _lp_loop(best_h, d7, radius_init=0.06)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Coordinate refinement (7%)
    d8 = deadline - 2.0
    best_h, best_c5 = _coord_refine(best_h, d8, n_iter=600)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
