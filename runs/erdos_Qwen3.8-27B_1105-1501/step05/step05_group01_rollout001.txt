    #    Contribution: -h[i+d]
    
    for r in range(len(corr)):
        d = r - (n - 1)
        wd = w[r]
        if wd < 1e-8:
            continue
        m = abs(d)
        # Case d >= 0: n ranges from 0 to n-1-d
        #   h[n+d] term: i = n+d, so i ranges from d to n-1. Contribution: (1-h[i-d]) = (1-h[i-d])
        #   (1-h[n]) term: i = n, so i ranges from 0 to n-1-d. Contribution: -h[i+d]
        if d >= 0:
            if d < n:
                # i from d to n-1: grad[i] += wd * (1-h[i-d])
                grad[d:] += wd * (1.0 - h[:n-d])
                # i from 0 to n-1-d: grad[i] -= wd * h[i+d]
                grad[:n-d] -= wd * h[d:]
        else:
            # d < 0, m = -d. n ranges from m to n-1.
            # h[n+d] = h[n-m] term: i = n-m, so i ranges from 0 to n-1-m. Contribution: (1-h[i+m])
            # (1-h[n]) term: i = n, so i ranges from m to n-1. Contribution: -h[i-m]
            if m < n:
                # i from 0 to n-1-m: grad[i] += wd * (1-h[i+m])
                grad[:n-m] += wd * (1.0 - h[m:])
                # i from m to n-1: grad[i] -= wd * h[i-m]
                grad[m:] -= wd * h[:n-m]
    
    return val, grad * dx

def _line_search_exact(h, p, dx, radius, n_samples=50):
    # Exact line search along direction p
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
    Q = np.correlate(p, p, mode='full')
    
    # Feasible interval
    t_lo = 0.0
    t_hi = 0.0
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / p[neg])))  # h[neg] / p[neg] is negative
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    def phi(t):
        return float(np.max(S + t * T - t * t * Q)) * dx
    
    # Dense sampling
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = np.array([phi(t) for t in ts])
    
    # Find best sample
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Refine with golden section around best
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    # Bracket around best
    if best_idx > 0:
        a = ts[best_idx - 1]
    else:
        a = t_lo
    if best_idx < n_samples - 1:
        b = ts[best_idx + 1]
    else:
        b = t_hi
    
    # Also check endpoints
    for ep in [t_lo, t_hi]:
        v = phi(ep)
        if v < best_val:
            best_t = ep
            best_val = v
            a = ts[max(0, best_idx-1)]
            b = ts[min(n_samples-1, best_idx+1)]
    
    # Golden section search in [a, b]
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = phi(c1)
    f2 = phi(c2)
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = phi(c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = phi(c2)
    
    opt_t = (a + b) / 2.0
    opt_val = phi(opt_t)
    
    # Return best of all candidates
    if best_val < opt_val:
        return best_t, best_val
    return opt_t, opt_val

def _fourier_direction(k, n, rng):
    # Generate a low-frequency Fourier direction
    x = (np.arange(n) + 0.5) / n
    phase = rng.uniform(0, 2 * np.pi)
    d = np.sin(2 * np.pi * k * x + phase)
    d -= d.mean()  # Ensure zero mean
    return d

def _slsqp_solve(h, tau, target, maxiter=50):
    # Solve the smooth problem with SLSQP
    n = len(h)
    
    def objective(x):
        return _softmax_obj_and_grad(x, tau)[0]
    
    def gradient(x):
        return _softmax_obj_and_grad(x, tau)[1]
    
    def constraint(x):
        return float(np.sum(x)) - target
    
    def constraint_grad(x):
        return np.ones(n)
    
    result = minimize(
        objective, h, jac=gradient, method='SLSQP',
        constraints=[{'type': 'eq', 'fun': constraint, 'jac': constraint_grad}],
        bounds=[(0, 1)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-8}
    )
    return result.x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap(h, deadline, max_passes=5, n_pairs_per_pass=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n_pairs_per_pass):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search_exact(h, d, dx, 0.5, n_samples=30)
            if c5_try < cur_c5 - 1e-12:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5
