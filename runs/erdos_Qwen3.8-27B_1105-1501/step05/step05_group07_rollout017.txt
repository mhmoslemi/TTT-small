def _project(v, target):

def _slsqp_optimize(h, deadline, beta_init=50.0, beta_final=300.0, n_phases=4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    def _obj(hf, beta):
        S = _scores(hf)
        maxS = np.max(S)
        shifted = np.clip(beta * (S - maxS), -400, 0)
        return maxS + np.log(np.mean(np.exp(shifted))) / beta
    
    def _grad(hf, beta):
        S = _scores(hf)
        maxS = np.max(S)
        shifted = np.clip(beta * (S - maxS), -400, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            idx = int(np.argmax(S))
            w = np.zeros(len(S))
            w[idx] = 1.0
        else:
            w /= wsum
        G = _grad_matrix(hf)
        return G.T @ w
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac

                                        'fun': lambda x: np.sum(x) - target,

                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

def _anneal_step(h, temperature, rng):

        j = rng.integers(n)

            h_new = h.copy()
            h_new[i] -= move
            h_new[j] += move
            return h_new
        else:
            h_new = h.copy()
            h_new[i] += move
            h_new[j] -= move
            return h_new

        k = rng.integers(n)

        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps * 0.5
        h_new[k] -= eps * 0.5

                best_h = h.copy()
        if it % 2000 == 0 and it > 0:
            h_pol, c5_pol = _lp_optimize(h, min(deadline, time.time() + 2.0), radius_init=0.05, max_iters=200, stall_limit=30)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                h = h_pol
                cur_c5 = c5_pol
    return best_h, best_c5

def _coord_descent(h, deadline, max_rounds=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        np.random.shuffle(order)
        for j in order:
            if time.time() > deadline:
                break
            if h[j] < 1e-10 or h[j] > 1.0 - 1e-10:
                continue
            lo_b = max(0.0, h[j] - 0.3)
            hi_b = min(1.0, h[j] + 0.3)
            if hi_b - lo_b < 1e-12:
                continue
            obj_vals = []
            grid = np.linspace(lo_b, hi_b, 15)
            for gj in grid:
                h_t = h.copy()
                diff = gj - h[j]
                h_t[j] = gj
                others = np.arange(n)
                others = others[others != j]
                room = np.zeros(n)
                if diff > 0:
                    room[others] = 1.0 - h[others]
                elif diff < 0:
                    room[others] = h[others]
                else:
                    continue
                tot_room = room.sum()
                if tot_room < abs(diff) * 0.9:
                    continue
                h_t[others] += diff * room / tot_room
                h_t = np.clip(h_t, 0.0, 1.0)
                h_t = _project(h_t, target)
                obj_vals.append(_exact_c5(h_t))
            if len(obj_vals) == 0:
                continue
            best_idx = int(np.argmin(obj_vals))
            if obj_vals[best_idx] < best_c5 - 1e-14:
                h_try = h.copy()
                gj = grid[best_idx]
                diff = gj - h[j]
                h_try[j] = gj
                others = np.arange(n)
                others = others[others != j]
                if diff > 0:
                    room = 1.0 - h[others]
                else:
                    room = h[others]
                tot_room = room.sum()
                if tot_room > 0:
                    h_try[others] += diff * room / tot_room
                h_try = _project(np.clip(h_try, 0.0, 1.0), target)
                c5_new = _exact_c5(h_try)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    except:
        h_init = None
        n_init = 0
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Phase 1: Multi-start with quasi-periodic + LP (40% time)
    dl1 = t0 + budget_s * 0.40
    alphas = [0.6180339887, 0.7071067812, 0.4142135624, 0.3819660113,
              0.5320888863, 0.7360679775, 0.3090169944, 0.8090169944]
    
    for n_try in [64, 96, 128]:
        if time.time() > dl1:
            break
        # Try initial_h_values resampled
        if h_init is not None and time.time() < dl1:
            h_s = _resample(h_init, n_try)
            sub_dl = min(dl1, time.time() + budget_s * 0.03)
            ht, ct = _lp_optimize(h_s, sub_dl, radius_init=0.15, max_iters=3000, stall_limit=80)
            if ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try
        
        # Quasi-periodic starts
        for ai, alpha in enumerate(alphas):
            if time.time() > dl1:
                break
            h_qp = _quasiperiodic(n_try, alpha, phi=rng.uniform(0, 1))
            h_qp += rng.normal(0, 0.03, n_try)
            h_qp = _project(h_qp, n_try / 2.0)
            sub_dl = min(dl1, time.time() + budget_s * 0.02)
            ht, ct = _lp_optimize(h_qp, sub_dl, radius_init=0.15, max_iters=2000, stall_limit=60)
            if ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try
        
        # Random binary start
        if time.time() < dl1:
            h_rand = (rng.random(n_try) < 0.5).astype(float)
            h_rand = _project(h_rand, n_try / 2.0)
            sub_dl = min(dl1, time.time() + budget_s * 0.015)
            ht, ct = _lp_optimize(h_rand, sub_dl, radius_init=0.2, max_iters=1500, stall_limit=60)
            if ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try
    
    # Phase 2: Simulated annealing (25% time)
    if time.time() < t0 + budget_s * 0.65 and best_h is not None:
        dl2 = t0 + budget_s * 0.65
        h_sn, c5_sn = _sim_anneal(best_h, dl2, rng, n_iter=30000)
        if c5_sn < best_c5:
            best_c5 = c5_sn; best_h = h_sn.copy()
        # LP polish after annealing
        if time.time() < dl2 + budget_s * 0.02:
            h_pol, c5_pol = _lp_optimize(best_h, min(deadline, time.time() + budget_s * 0.03),
                                         radius_init=0.08, max_iters=3000, stall_limit=80)
            if c5_pol < best_c5:
                best_c5 = c5_pol; best_h = h_pol.copy()
    
    # Phase 3: SLSQP optimization (20% time)
    if time.time() < t0 + budget_s * 0.85:
        dl3 = t0 + budget_s * 0.85
        h3, c5_3 = _slsqp_optimize(best_h, dl3, beta_init=50.0, beta_final=400.0, n_phases=5)
        if c5_3 < best_c5:
            best_c5 = c5_3; best_h = h3.copy()
        # LP after SLSQP
        if time.time() < deadline:
            h3b, c5_3b = _lp_optimize(best_h, min(deadline, time.time() + budget_s * 0.02),
                                      radius_init=0.05, max_iters=2000, stall_limit=60)
            if c5_3b < best_c5:
                best_c5 = c5_3b; best_h = h3b.copy()
    
    # Phase 4: Coordinate descent + final LP (10% time)
    if time.time() < t0 + budget_s * 0.95:
        dl4 = t0 + budget_s * 0.95
        h4, c5_4 = _coord_descent(best_h, dl4, max_rounds=30)
        if c5_4 < best_c5:
            best_c5 = c5_4; best_h = h4.copy()
    
    # Phase 5: Final intensive LP
    if time.time() < deadline:
        h5, c5_5 = _lp_optimize(best_h, deadline, radius_init=0.03, max_iters=5000, stall_limit=50)
        if c5_5 < best_c5:
            best_c5 = c5_5; best_h = h5.copy()
    
    # Final projection and validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    best_n = len(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
