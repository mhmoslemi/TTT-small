                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slexpert_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.006
    T_min = 0.000005
    cooling = 0.99975
    patience = 0
    max_patience = 15000
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min:
                break
            T *= 0.97
            patience = 0
        for _ in range(200):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if h[i] < 0.5 and h[j] > 0.5:
                i, j = j, i
            if h[i] < 0.5:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 122
        h0 = _project(np.random.default_rng(seed).uniform(0, 1, n0), n0 / 2.0)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: SLSQP with log-sum-exp (30% budget)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _sleaf_slshqp(best_h, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP optimization (25% budget)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _lp_optimize(best_h, d2, radius_init=0.18)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA (15% budget)
    d3 = t0 + budget_s * 0.70
    h3, c5_3 = _slexpert_sa(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3 + budget_s * 0.03:
        d3b = t0 + budget_s * 0.73
        h3b, c5_3b = _lp_optimize(best_h, d3b, radius_init=0.10)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution (15% budget)
    d4 = t0 + budget_s * 0.88
    n_cur = len(best_h)
    tried = set()
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14]
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_cur + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.12)
        if c5_r < best_c5 - 1e-10 and time.time() < d4 - 3:
            h_r2, c5_r2 = _slexpert_sa(h_r, min(d4, time.time() + budget_s * 0.01), rng)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
            if time.time() < d4:
                h_r, c5_r = _lp_optimize(h_r, d4, radius_init=0.08)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try

    # Phase 5: Aggressive perturbation (7% budget)
    d5 = t0 + budget_s * 0.95
    scale = 0.03
    for attempt in range(300):
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.12)

    # Phase 6: Pair refinement (3% budget)
    d6 = t0 + budget_s * 0.98
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=50)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final polish
    h_f, c5_f = _lp_optimize(best_h, deadline, radius_init=0.04)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5):
        best_h = _project(h0, n0 / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
