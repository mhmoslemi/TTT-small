    # Phase 5: Final polish (5%)
    d5 = t0 + budget_s - 1.0
    if time.time() < d5 - 3:
        sa_rng = np.random.default_rng(seed * 9999 + 42)
        h_f, c5_f = _sa_run(best_h, d5 - 3, sa_rng, T_init=0.0003, max_iter=500000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < d5:
        h_f2, c5_f2 = _pair_refine_sub(best_h, d5, n_pairs_per_pass=100, max_passes=20)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, n_final
