import time

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

def _c5_eval(h):
    """Compute exact C5 value"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_all(h):
    """Compute all 2n-1 score values"""
    n = len(h)
    dx = 2.0 / n
    scores = np.zeros(2*n - 1)
    # Positive lags d=0..n-1 at indices 0..n-1
    for d in range(n):
        m = n - d
        scores[d] = dx * float(np.dot(h[:m], 1.0 - h[d:d+m]))
    # Negative lags d=1..n-1 at indices n..2n-2
    for d in range(1, n):
        m = n - d
        scores[n + d - 1] = dx * float(np.dot(h[d:d+m], 1.0 - h[:m]))
    return scores

def _grad_score_pos(h, d):
    """Gradient of positive lag d score"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    m = n - d
    if m <= 0:
        return g
    # Source term: h[j] appears with coefficient (1-h[j+d]) for j < m
    g[:m] += (1.0 - h[d:d+m])
    # Target term: h[j] appears with coefficient -h[j-d] for j >= d
    if d > 0:
        g[d:] -= h[:m]
    return dx * g

def _grad_score_neg(h, d):
    """Gradient of negative lag d score (d>=1)"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    m = n - d
    if m <= 0:
        return g
    # Source: h[j] appears in h[j] (the shifted one) with coeff (1-h[j-d]) for j >= d
    g[d:] += (1.0 - h[:m])
    # Target: h[j] appears in (1-h[j]) with coeff -h[j+d] for j < m
    g[:m] -= h[d:d+m]
    return dx * g

def _score_and_grad_selected(h, ids):
    """Compute scores and gradients for selected lag ids"""
    n = len(h)
    scores = np.zeros(len(ids))
    grads = np.zeros((len(ids), n))
    for k, lid in enumerate(ids):
        if lid < n:
            # Positive lag d = lid
            d = lid
            m = n - d
            if m > 0:
                scores[k] = (2.0/n) * float(np.dot(h[:m], 1.0 - h[d:d+m]))
                grads[k] = _grad_score_pos(h, d)
        else:
            # Negative lag d = lid - n + 1
            d = lid - n + 1
            m = n - d
            if m > 0:
                scores[k] = (2.0/n) * float(np.dot(h[d:d+m], 1.0 - h[:m]))
                grads[k] = _grad_score_neg(h, d)
    return scores, grads
