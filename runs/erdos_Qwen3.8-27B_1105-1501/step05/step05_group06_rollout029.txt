from scipy.optimize import minimize
import time

_g_q = 16.0
_g_dx = 1.0
_g_n = 128

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_scores(h, dx=None):
    n = len(h)
    if dx is None:
        dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _pnorm_value(scores, q):
    # Compute (sum(scores^q))^(1/q) stably
    max_s = np.max(scores)
    if max_s <= 0:
        return 0.0
    # For stability, factor out max_s
    # (sum s^q)^(1/q) = max_s * (sum (s/max_s)^q)^(1/q)
    ratio = scores / max_s
    sum_rq = np.sum(ratio ** q)
    return max_s * sum_rq ** (1.0 / q)

def _pnorm_value_grad(h, q, dx, scores):
    n = len(h)
    # Compute value
    max_s = np.max(scores)
    if max_s <= 0:
        return 0.0, np.zeros(n)
    ratio = scores / max_s
    sum_rq = np.sum(ratio ** q)
    phi = max_s * sum_rq ** (1.0 / q)
    
    # Compute gradient
    # grad[j] = phi * sum_r w_r * dF_r/dh[j]
    # where w_r = F_r^(q-1) / sum(F_r^(q-1))
    # For stability: w_r = (F_r/max_s)^(q-1) / sum((F_r/max_s)^(q-1))
    # (the max_s factors cancel)
    
    pow_q1 = ratio ** (q - 1.0)  # this is (F_r/max_s)^(q-1)
    sum_pow = np.sum(pow_q1)
    if sum_pow < 1e-300:
        # Fallback: use the argmax
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w = pow_q1 / sum_pow
    
    # Now compute grad[j] = phi * sum_r w_r * dF_r/dh[j]
    # dF_r/dh[j] = dx*(1-h[j+d_r]) if 0<=j+d_r<n - dx*h[j-d_r] if 0<=j-d_r<n
    # where d_r = r - (n-1)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-300:
            continue
        wd = w[r] * dx
        d = r - (n - 1)
        # First term: j in [max(0,-d), min(n, n-d))
        a1 = max(0, -d)
        b1 = min(n, n - d)
        if b1 > a1:
            grad[a1:b1] += wd * (1.0 - h[a1 + d:b1 + d])
        # Second term: j in [max(0,d), min(n, n+d))
        a2 = max(0, d)
        b2 = min(n, n + d)
        if b2 > a2:
            grad[a2:b2] -= wd * h[a2 - d:b2 - d]
    
    grad *= phi
    return phi, grad

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _obj_fun(h):
    scores = _compute_scores(h, _g_dx)
    return _pnorm_value(scores, _g_q)

def _obj_grad(h):
    scores = _compute_scores(h, _g_dx)
    _, grad = _pnorm_value_grad(h, _g_q, _g_dx, scores)
    return grad

def _cons_fun(h):
    return np.sum(h) - _g_n / 2.0

def _cons_jac(h):
    return np.ones(_g_n)

def _smooth_optimize(h, deadline, q_list, maxiter_per_q=200):
    global _g_q, _g_dx, _g_n
    n = len(h)
    _g_dx = 2.0 / n

    best_h = h.copy()

            break
        _g_q = float(q)
        h_cur = best_h.copy()
        # Ensure feasibility

        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        
        try:

                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter_per_q, 'ftol': 1e-12})
            if res.success or res.x is not None:

                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _obj_fun(h):
    scores = _compute_scores(h)
    return _pnorm_value(scores, _g_q)


def _obj_grad(h):
    scores = _compute_scores(h)
    _, grad = _pnorm_value_grad(h, _g_q, scores)
    return grad


def _cons_fun(h):
    return np.sum(h) - _g_n / 2.0


def _cons_jac(h):
    return np.ones(_g_n)


def _smooth_optimize(h, deadline, q_list, maxiter_per_q=300):
    global _g_q, _g_dx, _g_n
    n = len(h)
    _g_dx = 2.0 / n
    _g_n = n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    for q in q_list:
        if time.time() > deadline:
            break
        _g_q = float(q)
        h_cur = _project(h, target)
        try:
            res = minimize(_obj_fun, h_cur, jac=_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter_per_q, 'ftol': 1e-14})
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c5


def _projected_gradient(h, q, deadline, n_steps=200):
    global _g_q, _g_n
    n = len(h)
    _g_n = n
    target = n / 2.0
    _g_q = float(q)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.05
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        scores = _compute_scores(h)
        _, grad = _pnorm_value_grad(h, _g_q, scores)
        grad = grad - np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        improved = False
        for _ls in range(12):
            h_try = _project(h - step * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.6
        if not improved:
            step = min(step * 1.5, 0.1)
        else:
            step = min(step * 1.1, 0.08)
    return best_h, best_c5


def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    h = vals.copy()
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)


def _smooth_init(h, kernel_sigma):
    n = len(h)
    k = max(3, int(3 * kernel_sigma))
    if k % 2 == 0:
        k += 1
    offsets = np.arange(-k // 2, k // 2 + 1)
    kernel = np.exp(-0.5 * (offsets / kernel_sigma) ** 2)
    kernel /= kernel.sum()
    h_smooth = np.convolve(h, kernel, mode='same')
    return _project(h_smooth, n / 2.0)


def _binarize_init(h, rng):
    n = len(h)
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    noise = rng.normal(0, 0.1, n)
    h_pert = h_bin + noise
    return _project(h_pert, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_descent(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _compute_scores(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = np.zeros(2 * n - 1)
                for r in range(2 * n - 1):
                    d = r - (n - 1)
                    val = 0.0
                    lo1 = max(0, -d)
                    hi1 = min(n, n - d)
                    if lo1 <= j < hi1:
                        val += dx * (1.0 - h[j + d])
                    lo2 = max(0, d)
                    hi2 = min(n, n + d)
                    if lo2 <= j < hi2:
                        val -= dx * h[j - d]
                    lo1k = max(0, -d)
                    hi1k = min(n, n - d)
                    if lo1k <= k < hi1k:
                        val -= dx * (1.0 - h[k + d])
                    lo2k = max(0, d)
                    hi2k = min(n, n + d)
                    if lo2k <= k < hi2k:
                        val += dx * h[k - d]
                    C[r] = val
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > 1e-7 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _compute_scores(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not pass_improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Main q-continuation on parent (15% budget)
    dl1 = t0 + budget_s * 0.15
    best_h, best_c5 = _smooth_optimize(best_h, dl1,
                                       q_list=[2, 4, 8, 16, 32, 64, 128, 256, 384],
                                       maxiter_per_q=400)
    best_n = len(best_h)

    # Phase 2: Projected gradient refinement at high q (10%)
    if time.time() < t0 + budget_s * 0.25:
        dl2 = t0 + budget_s * 0.25
        for q in [32, 64, 128]:
            if time.time() > dl2:
                break
            h_pg, c5_pg = _projected_gradient(best_h, q, dl2, n_steps=100)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg
                best_n = len(h_pg)

    # Phase 3: Diverse restarts at parent resolution (25%)
    dl3 = t0 + budget_s * 0.50
    n_alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.555, 0.445]
    starts = []
    starts.append(('parent_smooth', _smooth_init(best_h, 2.0)))
    starts.append(('parent_noise', _project(best_h + rng.normal(0, 0.02, best_n), best_n / 2.0)))
    starts.append(('parent_bin', _binarize_init(best_h, rng)))
    for ai in range(min(4, len(n_alphas))):
        starts.append(('quasi%d' % ai, _quasi_init(best_n, n_alphas[ai], rng)))
    starts.append(('noise05', _project(best_h + rng.normal(0, 0.05, best_n), best_n / 2.0)))
    starts.append(('smooth3', _smooth_init(best_h, 3.0)))

    for name, h_init in starts:
        if time.time() > dl3:
            break
        sub_dl = min(dl3, time.time() + budget_s * 0.035)
        h_try, c5_try = _smooth_optimize(h_init, sub_dl,
                                         q_list=[4, 8, 16, 32, 64, 128],
                                         maxiter_per_q=300)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
        else:
            if time.time() < dl3:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.012)
                h_pg, c5_pg = _projected_gradient(h_try, 64, sub_dl2, n_steps=80)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = len(h_pg)

    # Phase 4: Different resolutions (20%)
    dl4 = t0 + budget_s * 0.70
    for n_try in [192, 256, 96]:
        if time.time() > dl4:
            break
        h_res = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.045)
        h_try, c5_try = _smooth_optimize(h_res, sub_dl,
                                         q_list=[4, 16, 64, 128, 256],
                                         maxiter_per_q=300)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            # Polish at this resolution
            if time.time() < dl4:
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                h_pg, c5_pg = _projected_gradient(best_h, 128, sub_dl2, n_steps=100)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()

    # Phase 5: Pair descent for fine-tuning (15%)
    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.85
        best_h, best_c5 = _pair_descent(best_h, dl5, max_passes=80)
        # Follow-up smooth opt
        if time.time() < dl5:
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_s, c5_s = _smooth_optimize(best_h, sub_dl,
                                         q_list=[128, 256, 384],
                                         maxiter_per_q=200)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
        # More pair descent
        if time.time() < dl5:
            best_h, best_c5 = _pair_descent(best_h, dl5, max_passes=50)

    # Phase 6: Perturbation restarts and final polish (10%)
    dl6 = hard_deadline
    scale = 0.015
    for attempt in range(300):
        if time.time() > dl6:
            break
        h_pert = _project(best_h + rng.normal(0, scale, best_n), best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.006)
        h_try, c5_try = _smooth_optimize(h_pert, sub_dl,
                                         q_list=[64, 128, 256],
                                         maxiter_per_q=150)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Final: pair descent + tiny smooth steps
    if time.time() < hard_deadline - 2:
        best_h, best_c5 = _pair_descent(best_h, hard_deadline - 3, max_passes=60)
    if time.time() < hard_deadline - 1:
        sub_dl = hard_deadline
        h_fin, c5_fin = _smooth_optimize(best_h, sub_dl,
                                         q_list=[256, 384, 512],
                                         maxiter_per_q=100)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
