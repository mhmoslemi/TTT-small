    """One subgradient step: compute active lag, take step in negative gradient direction."""
    n = len(h)
    dx = 2.0 / n

    # dscore/dh[j] = (1-h[j-d]) if j >= d, and -h[j+d] if j+d < n

    # For lag_val >= 0: scores = sum_{i=0}^{n-1-lag_val} h[i] * (1-h[i+lag_val])
    #   d/dh[j] = (1-h[j+lag_val]) if j+lag_val < n, minus h[j-lag_val] if j >= lag_val, else 0
    # For lag_val < 0: let g = -lag_val, scores = sum_{i=0}^{n-1-g} h[i+g] * (1-h[i])

        grad[g:] += (1.0 - h[:m])

import numpy as np
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h, target):
    n = len(h)
    h = h.copy()
    n_ones = int(round(target))
    if n_ones <= 0:
        return np.zeros(n)
    if n_ones >= n:
        return np.ones(n)
    idx = np.argsort(-h)[:n_ones]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _line_search(h, d, dx, radius, n_samples=40):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, -float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = np.array([float(np.max(S0 + t * T - t * t * C)) for t in ts])
    bi = int(np.argmin(vals))
    if bi == 0:
        lo_g, hi_g = t_lo, ts[1]
    elif bi == n_samples - 1:
        lo_g, hi_g = ts[-2], t_hi
    else:
        lo_g, hi_g = ts[bi - 1], ts[bi + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_g - gr * (hi_g - lo_g)
    c2 = lo_g + gr * (hi_g - lo_g)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(50):
        if hi_g - lo_g < 1e-12:
            break
        if f1 < f2:
            hi_g = c2; c2 = c1; f2 = f1
            c1 = hi_g - gr * (hi_g - lo_g)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo_g = c1; c1 = c2; f1 = f2
            c2 = lo_g + gr * (hi_g - lo_g)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo_g + hi_g) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

            stall += 1

    return best_h, best_c5

def _subgrad_nesterov(h, deadline, lr0=0.8):
    n = len(h)
    target = n / 2.0
    x = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    k = 0
    while time.time() < deadline:
        t = k + 1
        lr = lr0 / np.sqrt(t)
        scores = np.correlate(y, 1.0 - y, mode='full')
        active = int(np.argmax(scores))
        lag_val = active - (n - 1)
        grad = np.zeros(n)
        if lag_val >= 0:
            m = n - lag_val
            if m > 0:
                grad[:m] += (1.0 - y[lag_val:])
                grad[lag_val:] -= y[:m]
        else:
            g = -lag_val
            m = n - g
            if m > 0:
                grad[g:] += (1.0 - y[:m])
                grad[:m] -= y[g:]
        x_new = _project(y - lr * grad, target)
        alpha_n = 2.0 / (t + 1.5)
        y = alpha_n * x_new + (1.0 - alpha_n) * y
        y = _project(y, target)
        x = x_new
        c5 = _exact_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        k += 1
    return best_h, best_c5

        sub_dl = min(d4, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.08)

    # Phase 5: Final polish (8%)
    d5 = deadline - 0.5
    h7 = best_h.copy()
    h7, c5_7 = _lp_loop(h7, d5, radius_init=0.03)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h8 = best_h.copy()
    rem = d5 - time.time()
    if rem > 0.5:
        h8, c5_8 = _pair_swap_polish(h8, d5, n_tries=3000, radius=0.1)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
