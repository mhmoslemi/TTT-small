import numpy as np
from scipy.optimize import linprog, minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def lag_jacobian(h):
    """Compute Jacobian of lag values w.r.t. h, shape (2n-1, n)"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        r = n - 1 + s
        if s == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            m = n - s
            # F_s = dx * sum_{i=0}^{m-1} h[i+s] * (1-h[i])
            # dF_s/dh[j] for j in [s, s+m-1]: dx * (1 - h[j-s])
            # dF_s/dh[j] for j in [0, m-1]: -dx * h[j+s]
            G[r, s:s+m] += dx * (1.0 - h[:m])
            G[r, :m] += -dx * h[s:s+m]
        else:
            a = -s
            m = n - a
            # F_{-a} = dx * sum_{i=0}^{m-1} h[i] * (1-h[i+a])
            # dF/dh[j] for j in [0, m-1]: dx * (1 - h[j+a])
            # dF/dh[j] for j in [a, a+m-1]: -dx * h[j-a]
            G[r, :m] += dx * (1.0 - h[a:a+m])
            G[r, a:a+m] += -dx * h[:m]
    return G

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def line_search_quad(h, d, dx, trust_radius):
    """Line search for max over line h + t*d using quadratic approximation of each lag"""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def lp_descend(h, deadline, radius_init=0.15):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()

    stall = 0

        cur_max = float(np.max(scores))

        tol = 1e-8
        active = np.where(scores >= cur_max - tol)[0]
        # Also include a few more for robustness
        if len(active) < 4:
            sorted_idx = np.argsort(scores)[::-1]
            active = sorted_idx[:min(4, len(sorted_idx))]
        active = np.unique(active)
        
        # Build Jacobian for active lags only
        G = np.zeros((len(active), n))
        for idx, r in enumerate(active):
            s = r - (n - 1)
            if s == 0:
                G[idx, :] = dx * (1.0 - 2.0 * h)
            elif s > 0:
                m = n - s
                G[idx, s:s+m] += dx * (1.0 - h[:m])
                G[idx, :m] += -dx * h[s:s+m]
            else:
                a = -s
                m = n - a
                G[idx, :m] += dx * (1.0 - h[a:a+m])
                G[idx, a:a+m] += -dx * h[:m]
        
        # LP: minimize max_i (scores[i] + G[i] @ p) subject to sum(p)=0, box/trust
        # Reformulate: minimize t subject to G[i]@p - t <= -scores[i], sum(p)=0
        # Variable: [p (n), t] => n+1 variables
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        n_act = len(active)
        A_ub = np.zeros((n_act, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[active]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                # Try relaxed version (allow slight violation of non-top constraints)
                # Just skip

                    radius *= 0.8
                    if radius < 1e-5:
                        break
                    continue
                p = res.x[:n]
                if np.max(np.abs(p)) < 1e-14:
                    stall += 1
                    radius *= 0.85
                    if stall > 30:
                        break
                    continue
            else:
                stall += 1
                radius *= 0.8
                if radius < 1e-5:
                    break
                continue
        except Exception:
            stall += 2
            radius *= 0.7
            if radius < 1e-5:
                break
            continue
        
        p = res.x[:n]
        if np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.85
            continue
        
        alpha, new_c5 = line_search_quad(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * p
            h_new = project_box_sum(h_new, target)
            actual_c5 = eval_c5(h_new)

                h = h_new
                cur_c5 = actual_c5

                    best_c5 = cur_c5
                    best_h = h.copy()

    return best_h, best_c5

def slsqp_epigraph(h_seed, deadline, n_slsqp_max=8):
    """SLSQP on epigraph formulation: min t s.t. F_s(h) <= t, sum(h)=n/2, 0<=h<=1"""
    n = len(h_seed)
    target = n / 2.0
    h = project_box_sum(h_seed.copy(), target)
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    scores = lag_values(h)
    cur_max = float(np.max(scores))
    
    for attempt in range(n_slsqp_max):
        if time.time() > deadline:
            break
        t_start = cur_max + 0.001 + 0.003 * attempt
        
        def obj(x):
            return x[-1]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        
        def constraint_fun(x):
            h_x = x[:n]
            t_x = x[-1]
            vals = lag_values(h_x)
            return vals - t_x
        
        def constraint_jac(x):
            h_x = x[:n]
            J = lag_jacobian(h_x)  # (2n-1, n)
            full = np.zeros((2*n-1, n+1))
            full[:, :n] = J
            full[:, n] = -1.0
            return full
        
        constraints = [
            {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac},
            {'type': 'eq', 'fun': lambda x: float(np.sum(x[:n])) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}
        ]
        
        bounds = [(0.0, 1.0)] * n + [(max(0.0, cur_max - 0.05), min(1.0, t_start + 0.1))]
        
        x0 = np.append(h, t_start)
        
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.x is not None:
                h_try = project_box_sum(res.x[:n].copy(), target)
                c5_try = eval_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    cur_max = float(np.max(lag_values(h_try)))
        except Exception:
            pass
    
    return best_h, best_c5

def logsumexp_smooth(h_seed, deadline, beta_start=0.05, beta_min=0.002):
    """Optimize log-sum-exp smoothed objective with SLSQP"""
    n = len(h_seed)
    target = n / 2.0
    h = project_box_sum(h_seed.copy(), target)
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    betas = []
    b = beta_start
    while b >= beta_min:
        betas.append(b)
        b *= 0.6
    betas.append(beta_min)
    
    for beta in betas:
        if time.time() > deadline:
            break
        
        def smooth_obj(hv):
            vals = lag_values(hv)
            m = np.max(vals)
            shifted = (vals - m) / beta
            # clip to avoid overflow
            shifted = np.clip(shifted, -50, 50)
            logsumexp = m + beta * np.log(np.sum(np.exp(shifted)))
            return logsumexp
        
        def smooth_grad(hv):
            vals = lag_values(hv)
            m = np.max(vals)
            shifted = (vals - m) / beta
            shifted = np.clip(shifted, -50, 50)
            Z = np.exp(shifted)
            P = Z / np.sum(Z)
            # gradient = sum P_s * grad F_s
            J = lag_jacobian(hv)  # (2n-1, n)
            return P @ J
        
        def eq_con(hv):
            return float(np.sum(hv)) - target
        
        def eq_jac(hv):
            return np.ones(n)
        
        constraints = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
        bounds = [(0.0, 1.0)] * n
        
        try:
            res = minimize(smooth_obj, h, method='SLSQP', jac=smooth_grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 30, 'ftol': 1e-7})
            if res.x is not None:
                h_try = project_box_sum(res.x.copy(), target)
                c5_try = eval_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)

def coord_refine(h, deadline, n_iter=300):
    """Two-coordinate exchange refinement"""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(42)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Pick two random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Try moving weight from i to j (and vice versa)
        for sign in (1, -1):
            d = np.zeros(n)
            d[i] = -sign
            d[j] = sign
            # Determine feasible range
            if sign > 0:
                # decrease h[i], increase h[j]
                t_max = min(h[i], 1.0 - h[j], 0.5)
                if t_max < 1e-10:
                    continue
                # Line search on [0, t_max]
                t_opt, c5_val = line_search_quad(h, d, dx, t_max)
            else:
                t_max = min(h[j], 1.0 - h[i], 0.5)
                if t_max < 1e-10:
                    continue
                t_opt, c5_val = line_search_quad(h, d, dx, t_max)
            
            if c5_val < best_c5 - 1e-13:

                h_new = project_box_sum(h_new, target)
                actual = eval_c5(h_new)
                if actual < best_c5 - 1e-14:
                    best_c5 = actual
                    best_h = h_new.copy()
                    h = h_new
    
    return best_h, best_c5

    t_x = x[-1]
    dx = 2.0 / _gs_n
    all_vals = np.correlate(h_x, 1.0 - h_x, mode='full') * dx

def _lag_jac(h):

    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        r = n - 1 + s
        if s == 0:

        elif s > 0:

            G[r, :m] += -dx * h[s:s+m]

def line_search_quad(h, d, dx, trust_radius):

    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx

            c2 = a + gr * (b - a)

    opt_t = (a + b) / 2.0

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)
