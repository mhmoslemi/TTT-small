    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

# === Gradient computation ===

def _compute_grad_h(h, w, dx):
    """Compute G(h).T @ w without building full G matrix."""
    n = len(h)
    n_lags = len(w)
    grad_h = np.zeros(n)
    for r in range(n_lags):
        wr = w[r]
        if wr < 1e-15:
            continue
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                grad_h[k:] += wr * (1.0 - h[:m])
                grad_h[:m] += -wr * h[k:]
        else:
            g = -k
            m = n - g
            if m > 0:
                grad_h[:m] += wr * (1.0 - h[g:])
                grad_h[g:] += -wr * h[:m]
    return grad_h * dx

# === Logit manifold ===

def _safe_sigmoid(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    et = np.exp(t[~pos])
    out[~pos] = et / (1.0 + et)
    return out

def _logit(h, eps=1e-4):
    return np.log(np.clip(h, eps, 1.0 - eps) / (1.0 - np.clip(h, eps, 1.0 - eps)))

def _solve_mu(q, target):
    n = len(q)
    def f(mu):
        return float(np.sum(_safe_sigmoid(q - mu))) - target
    lo, hi = -30.0, 30.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target):
    mu = _solve_mu(q, target)
    return _safe_sigmoid(q - mu)

def _smooth_obj_grad(q, beta, target):
    n = len(q)
    h = _q_to_h(q, target)
    S = _scores(h)
    maxS = np.max(S)
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500, 0)
    exp_s = np.exp(shifted)
    wsum = float(np.sum(exp_s))
    if wsum < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w = exp_s / wsum
    F = maxS + np.log(wsum) / beta
    dx = 2.0 / n
    grad_h = _compute_grad_h(h, w, dx)
    s = h * (1.0 - h)
    ssum = float(np.sum(s))
    if ssum < 1e-30:
        return F, np.zeros(n)
    rho = float(np.dot(s, grad_h)) / ssum
    grad_q = s * (grad_h - rho)
    # Replace NaN/Inf with 0
    grad_q = np.nan_to_num(grad_q, nan=0.0, posinf=0.0, neginf=0.0)
    return F, grad_q

# === L-BFGS-B optimization on logit manifold ===

def _lbfgs_logit(h_init, deadline, beta_schedule=None):
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    if beta_schedule is None:
        L = 2 * n - 1
        beta_schedule = [10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 500.0]
    
    q = _logit(h_init)
    
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        
        def obj(qv):
            return _smooth_obj_grad(qv, beta, target)[0]
        
        def grad(qv):
            return _smooth_obj_grad(qv, beta, target)[1]
        
        maxiter = max(50, int(300 * (128.0 / n)))
        try:
            res = minimize(obj, q, jac=grad, method='L-BFGS-B',
                          bounds=[(-12.0, 12.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 
                                   'maxfun': maxiter * 3})
        except Exception:
            break
        
        if np.all(np.isfinite(res.x)):
            h_new = _q_to_h(res.x, target)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            q = res.x.copy()  # Continue from this point for next beta
        else:
            break
    
    return best_h, best_c5

# === LP refinement ===

def _lp_direction(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    dx = 2.0 / n
    
    # Build A_ub: for each lag r, G[r,:] @ delta - tau <= -scores[r]
    # But we want to minimize tau subject to G[r,:] @ delta <= -scores[r] + tau
    # i.e., G[r,:] @ delta - tau <= -scores[r]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, n] = -1.0  # -tau column
    b_ub = np.zeros(n_lags)
    
    for r in range(n_lags):
        b_ub[r] = -scores[r]
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                A_ub[r, k:] += (1.0 - h[:m]) * dx
                A_ub[r, :m] += -h[k:] * dx
        else:
            g = -k
            m = n - g
            if m > 0:
                A_ub[r, :m] += (1.0 - h[g:]) * dx
                A_ub[r, g:] += -h[:m] * dx
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _lp_refine(h_init, deadline, max_iters=3000, radius_init=0.2, stall_limit=100):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        scores = _scores(h)
        p, _ = _lp_direction(h, scores, radius)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.85
            if radius < 1e-6:
                break
            continue
        
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12) * 0.999
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, alpha_max
        f_hi = _exact_c5(h + hi * p)
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(25):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        opt_a = (lo + hi) / 2.0
        h_try = _project(h + opt_a * p, target)
        c5_try = _exact_c5(h_try)
        
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.85
            if stall > 50 and radius < 0.02:
                break
    
    return best_h, best_c5

# === Projected gradient (direct h-space) ===

def _proj_grad(h_init, deadline, n_steps=5000, betas=None):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    if betas is None:
        betas = [5.0, 10.0, 20.0, 40.0, 80.0]
    
    for beta in betas:
        if time.time() > deadline:
            break
        cur_c5 = _exact_c5(h)
        for step in range(n_steps // len(betas)):
            if time.time() > deadline:
                break
            scores = _scores(h)
            maxS = np.max(scores)
            shifted = beta * (scores - maxS)
            shifted = np.clip(shifted, -300, 0)
            exp_s = np.exp(shifted)
            wsum = float(np.sum(exp_s))
            if wsum < 1e-300:
                w = np.zeros(len(scores))
                w[int(np.argmax(scores))] = 1.0
            else:
                w = exp_s / wsum
            
            dx = 2.0 / n
            grad = _compute_grad_h(h, w, dx)
            grad -= np.mean(grad)  # maintain sum constraint
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad /= gn
            
            # Backtracking line search
            alpha = min(1.0 / (beta * gn + 1e-8), 0.1)
            improved = False
            for _ in range(40):
                h_new = _project(h - alpha * grad, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    h = h_new
                    cur_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    improved = True
                    break
                alpha *= 0.7
                if alpha < 1e-14:
                    break
            if not improved:
                # Try smaller fixed step
                h = _project(h - 1e-4 * grad, target)
                cur_c5 = _exact_c5(h)
    
    return best_h, best_c5

# === Multi-resolution ===

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

# === Binary/quantized initialization ===

def _binary_init(n, seed_rng):
    """Create a random binary pattern with exactly n/2 ones."""
    target = n // 2
    idx = seed_rng.permutation(n)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

# === Quasi-periodic initialization ===

def _quasi_periodic_init(n, alpha, seed_rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    # Adjust to exact target
    target = n / 2.0
    curr = float(np.sum(h))
    if curr < target:
        # Need to add (target - curr) ones
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            n_add = int(round(target - curr))
            h[zeros[:n_add]] = 1.0
    elif curr > target:
        ones = np.where(h == 1)[0]
        n_rem = int(round(curr - target))
        h[ones[:n_rem]] = 0.0
    return h

# === Pair polish ===

def _pair_polish(h_init, deadline, max_passes=5):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = max(1, n // 24)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Golden section on [0, a_max]
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    lo_a, hi_a = 0.0, a_max
                    h1 = h.copy()
                    h1[j] += sign * hi_a
                    h1[k] -= sign * hi_a
                    f_hi = _exact_c5(np.clip(h1, 0, 1))
                    c1 = hi_a - gr * (hi_a - lo_a)
                    c2 = lo_a + gr * (hi_a - lo_a)
                    h_t = h.copy()
                    h_t[j] += sign * c1
                    h_t[k] -= sign * c1
                    f1 = _exact_c5(np.clip(h_t, 0, 1))
                    h_t = h.copy()
                    h_t[j] += sign * c2
                    h_t[k] -= sign * c2
                    f2 = _exact_c5(np.clip(h_t, 0, 1))
                    for _ in range(10):
                        if hi_a -
