import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        if d >= 0:
            m = n - d
            if m > 0:

            if m > 0:
                rows[idx_in, 0:m] += dx * (1.0 - h[g:n])
                rows[idx_in, g:n] += -dx * h[0:m]

def _slsqp_epigraph(h, deadline, max_rounds=40, active_init=16, active_max=80):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h_cur = h.copy()
    c5_cur = best_c5
    scores = _scores_all(h_cur)
    cur_active = np.argsort(-scores)[:active_init].tolist()
    T0 = float(np.max(scores)) + 0.015
    x0 = np.concatenate([h_cur, [min(T0, 3.0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 4.0)]

    class Context:
        pass

    ctx = Context()
    ctx.n = n
    ctx.target = target

    def obj_fun(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(x):
        h_vec = x[:n]
        T = x[-1]
        s = _scores_all(h_vec)
        return T - s[ctx.active_list]

    def ineq_jac(x):
        h_vec = x[:n]
        G = _grad_rows(ctx.active_list, h_vec, n)
        J = np.zeros((len(ctx.active_list), n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    def eq_fun(x):
        return float(np.sum(x[:n]) - ctx.target)

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

            break
        ctx.active_list = np.array(cur_active)

            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
        )
        try:
            res = minimize(obj_fun, x0, method='SLSQP', jac=obj_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 80, 'ftol': 1e-12})
        except Exception:
            break
        if res.x is None or not np.all(np.isfinite(res.x)):
            break
        h_try = _project(res.x[:n], target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_cur = h_try.copy()
            c5_cur = c5_try

def _lp_direction(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    G = _grad_rows(np.arange(n_lags), h, n)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and float(np.max(np.abs(res.x[:n]))) > 1e-14:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, target, n_eval=25):
    alpha_max = 1.0
    for j in range(len(h)):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h1 = _project(h + c1 * p, target)
    f1 = _exact_c5(h1)
    h2 = _project(h + c2 * p, target)
    f2 = _exact_c5(h2)
    for _ in range(n_eval):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            h1 = _project(h + c1 * p, target)
            f1 = _exact_c5(h1)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            h2 = _project(h + c2 * p, target)
            f2 = _exact_c5(h2)
    best_a = c1 if f1 < f2 else c2
    best_f = min(f1, f2)
    mid_a = (lo + hi) / 2.0
    h_mid = _project(h + mid_a * p, target)
    f_mid = _exact_c5(h_mid)
    if f_mid < best_f:
        return mid_a, f_mid
    return best_a, best_f

            stall = 0

        else:

            radius *= 0.7

def _smooth_grad(h, deadline, beta_start=20, beta_end=300, n_steps=300, step_init=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores_all(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        G = _grad_rows(np.arange(len(scores)), h, n)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.02, 0.12)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _qp_init(n, alpha, rng=None, threshold=0.5):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    h = (vals < threshold).astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
        h = _project(h, n / 2.0)
    return h

def _binary_init(n, alpha, rng):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    order = np.argsort(vals)
    h = np.zeros(n)
    h[order[:n // 2]] = 1.0
    return h

    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.02
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
        T *= cool
    return best_h, best_c5

def _coord_polish(h, deadline, n_cycles=2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(30):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h, deadline, n_passes=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_size = max(1, n // 16)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step_size):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step_size, min(j + 2 * step_size, n)):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: SLSQP epigraph on parent
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.18)
        h1, c5_1 = _slsqp_epigraph(h_start, dl, max_rounds=35, active_init=16, active_max=64)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: LP polish after SLSQP
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.28)
        h2, c5_2 = _lp_optimize(best_h, dl, radius_init=0.10, max_iters=3000, radius_min=0.001)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Smoothed gradient
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.36)
        h3, c5_3 = _smooth_grad(best_h, dl, beta_start=15, beta_end=350, n_steps=300, step_init=0.035)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        if time.time() < deadline:
            dl3b = min(deadline, t0 + budget_s * 0.40)
            h3b, c5_3b = _slsqp_epigraph(best_h, dl3b, max_rounds=20, active_init=16, active_max=48)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration
    if time.time() < deadline:
        dl4 = min(deadline, t0 + budget_s * 0.62)
        resolutions = [64, 80, 96, 112, 128, 144, 160]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113,
                  0.5, 0.6545084972, 0.3273268354, 0.7548776662, 0.4755282581]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl4:
                break
            if best_c5 <= 0.38070:
                break
            n_alphas = min(3, len(alphas))
            for ai in range(n_alphas):
                if time.time() > dl4:
                    break
                alpha = alphas[(ri * 3 + ai) % len(alphas)]
                if ri % 2 == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    h_init = _qp_init(n_try, alpha, rng, threshold=0.5)
                sub_dl = min(dl4, time.time() + budget_s * 0.04)
                h_try, c5_try = _slsqp_epigraph(h_init, sub_dl, max_rounds=15, active_init=12, active_max=40)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                if time.time() > dl4:
                    break
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                h_try2, c5_try2 = _lp_optimize(best_h, sub_dl2, radius_init=0.08, max_iters=1500, radius_min=0.003)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()

    # Phase 5: Binary SA + continuous relaxation
    if time.time() < deadline:
        dl5 = min(deadline, t0 + budget_s * 0.72)
        for n_bin in [64, 96, 128]:
            if time.time() > dl5:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=25000)
            if c5_b < best_c5 - 1e-12:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
            if time.time() > dl5:
                break
            sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
            h_br, c5_br = _slsqp_epigraph(h_b, sub_dl2, max_rounds=12, active_init=12, active_max=36)
            if c5_br < best_c5:
                best_c5 = c5_br
                best_h = h_br.copy()
                best_n = n_bin

    # Phase 6: Perturbation restarts
    if time.time() < deadline:
        dl6 = min(deadline, t0 + budget_s * 0.84)
        scale = 0.03
        for _attempt in range(200):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.012)
            h_try, c5_try = _slsqp_epigraph(h_pert, sub_dl, max_rounds=10, active_init=12, active_max=32)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.004, scale * 0.8)
            else:
                scale = min(0.07, scale * 1.08)

    # Phase 7: Coordinate polish
    if time.time() < deadline:
        dl7 = min(deadline, t0 + budget_s * 0.92)
        hc, c5c = _coord_polish(best_h, dl7, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 8: Pair polish
    if time.time() < deadline:
        dl8 = min(deadline, t0 + budget_s * 0.95)
        hp, c5p = _pair_polish(best_h, dl8, n_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 9: Final SLSQP
    if time.time() < deadline:
        dl9 = min(deadline, t0 + budget_s * 0.98)
        h_fin, c5_fin = _slsqp_epigraph(best_h, dl9, max_rounds=15, active_init=16, active_max=48)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
