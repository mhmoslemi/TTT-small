import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        r = n - 1 + d
        m = n - d
        G[r, 0:m] += (1.0 - h[d:d + m])
        G[r, d:n] += -h[0:m]
    for d in range(1, n):
        r = n - 1 - d
        m = n - d
        G[r, d:n] += (1.0 - h[0:m])
        G[r, 0:m] += -h[d:d + m]
    return G * (2.0 / n)

def _lp_step(h, scores, G, radius, bounds_lo, bounds_hi):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(bounds_lo[j], -radius), min(bounds_hi[j], radius)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5, target, n_samp=30):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(_project(h + a * p, target)) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(25):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, target))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_momentum_optimize(h, deadline, target, radius_init=0.25, momentum=0.6, max_iters=6000):
    n = len(h)
    h = h.copy().astype(np.float64)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    v = np.zeros(n)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        bounds_lo = -h
        bounds_hi = 1.0 - h
        p, _ = _lp_step(h, scores, G, radius, bounds_lo, bounds_hi)
        if p is None or float(np.max(np.abs(p))) < 1e-15:
            stall += 1
            radius *= 0.85
            if radius < 1e-7:
                break
            continue
        v = momentum * v + (1.0 - momentum) * p
        alpha, new_c5 = _line_search(h, v, cur_c5, target)
        h_new = _project(h + alpha * v, target)
        cur_c5_new = _exact_c5(h_new)
        if cur_c5_new < cur_c5 - 1e-14:
            cur_c5 = cur_c5_new
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            v *= 0.5
            radius *= 0.88
            if stall % 40 == 0 and radius < 0.02:
                radius = 0.15
                v *= 0.3
    return best_h, best_c5

def _softmax_grad_optimize(h, deadline, target, n_steps=4000, beta_start=20.0, beta_end=200.0):
    n = len(h)
    h = h.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)
    momentum = 0.7
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        shifted = np.clip(beta * (scores - max_s), -500, 0)
        w = np.exp(shifted)
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        grad /= gn
        v = momentum * v - grad
        improved = False
        alpha = 0.08
        for _ in range(40):
            h_new = _project(h + alpha * v, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.72
            if alpha < 1e-13:
                break
        if not improved:
            v *= 0.5
    return best_h, best_c5

def _slsqp_optimize(h, deadline, target, beta_init=40.0, beta_final=400.0, n_phases=4):
    n = len(h)
    h = h.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    def _obj(hf):
        S = _scores(hf)
        b = beta
        maxS = np.max(S)
        shifted = b * (S - maxS)
        shifted = np.clip(shifted, -500, 0)
        return maxS + np.log(np.mean(np.exp(shifted))) / b

    def _grad(hf):
        S = _scores(hf)
        b = beta
        maxS = np.max(S)
        shifted = b * (S - maxS)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            idx = int(np.argmax(S))
            w = np.zeros(len(S))
            w[idx] = 1.0
        else:
            w /= wsum
        G = _grad_matrix(hf)
        return G.T @ w

    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq',
                                        'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 150, 'ftol': 1e-15})
            if np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _direct_max_attack(h, deadline, target, n_rounds=500):
    n = len(h)
    h = h.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_rounds):
        if time.time() > deadline:
            break
        scores = _scores(h)
        k_star = int(np.argmax(scores))
        lag = k_star - (n - 1)
        G = _grad_matrix(h)
        grad = G[k_star, :]
        grad -= grad.mean()
        alpha = 0.05
        improved = False
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)

                h = h_new
                improved = True
                break

        if not improved:
            break
    return best_h, best_c5
