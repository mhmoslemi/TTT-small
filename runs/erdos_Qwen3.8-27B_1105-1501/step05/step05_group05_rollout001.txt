import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(35):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.1):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lsumexp_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _lp_loop(h, deadline, radius_init=0.20, use_penalty=True, max_penalty_growth=5.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    penalty_w = 0.05
    while time.time() < deadline:
        if stall > 150:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        G = _grad_matrix(h)
        penalty_lags = None
        if use_penalty and stall > 3:
            thresh = s_max * max(0.98, 1.0 - 0.01 * (stall // 10))
            penalty_lags = list(np.where(scores >= thresh)[0])
        p = _lp_solve(h, scores, G, radius, penalty_lags)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.85
            penalty_w = min(penalty_w * 1.3, max_penalty_growth)
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.6)
            penalty_w = max(0.03, penalty_w * 0.95)
        else:
            stall += 1
            radius *= 0.8
            penalty_w = min(penalty_w * 1.15, max_penalty_growth)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_kick(h, rng, scale=0.05):
    """Create a kick that targets the problematic lags."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    s_max = float(np.max(scores))
    # Find the worst lag
    worst_lag = int(np.argmax(scores))
    d_lag = worst_lag - (n - 1)
    # Create a direction that reduces overlap at the worst lag
    # If d_lag >= 0: overlap is sum h[i] * (1-h[i+d_lag])
    # To reduce: increase h where h[i+d_lag] is small, or decrease h where h[i] is large
    direction = np.zeros(n)
    if d_lag >= 0:
        m = n - d_lag
        direction[:m] -= h[:m]        # decrease h where contributing
        direction[d_lag:] += (1.0 - h[d_lag:])  # increase where (1-h) contributes
    else:
        g = -d_lag
        m = n - g
        direction[:m] -= h[g:g+m]      # h shifted
        direction[g:] += (1.0 - h[:m])
    direction -= direction.mean()
    dn = np.linalg.norm(direction)
    if dn > 1e-15:
        direction = direction / dn
    # Mix with random
    rand_dir = rng.normal(0, 1, n)
    rand_dir -= rand_dir.mean()
    rand_dir /= np.linalg.norm(rand_dir) + 1e-15
    mixed = 0.5 * direction + 0.5 * rand_dir
    return scale * mixed

def _kick_and_reopt(h, deadline, rng, n_kicks=50):

    target = n / 2.0

    best_c5 = _exact_c5(h)

        if time.time() > deadline:
            break
        # Vary kick size
        scale = rng.uniform(0.02, 0.12) * (1.0 + 0.5 * (k // 10))
        if k % 3 == 0:
            direction = _structured_kick(h, rng, scale)
        else:
            direction = rng.normal(0, 1, n)
            direction -= direction.mean()
            direction /= np.linalg.norm(direction) + 1e-15
            direction *= scale
        h_k = best_h + direction
        h_k = _project(h_k, target)
        # Quick local refinement
        sub_dl = min(deadline, time.time() + max(1.0, budget_remaining(deadline) * 0.015))
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.10, use_penalty=True)
        if c5_t < best_c5 - 1e-12:

            best_h = h_t.copy()
    return best_h, best_c5
