                a = -d
                vals[k] = dx * np.dot(h_x[:m], 1.0 - h_x[a:a+m]) - t
        return vals

    def jac(x):
        h_x = x[:n]
        J = np.zeros((K, n + 1))
        J[:, n] = -1.0
        for k in range(K):
            d = shifts[k]
            m = m_vals[k]
            if d >= 0:
                J[k, d:n] += dx * (1.0 - h_x[:m])
                J[k, 0:m] += dx * (-h_x[d:d+m])
            else:
                a = -d
                J[k, 0:m] += dx * (1.0 - h_x[a:a+m])
                J[k, a:n] += dx * (-h_x[:m])
        return J

    return fun, jac

def _slsqp_hard_solve(h, active_indices, radius, maxiter):
    n = len(h)
    dx = 2.0 / n
    current_c5 = _exact_c5(h)
    t0_init = current_c5 + 1e-6
    x0 = np.concatenate([h, [t0_init]])
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((0.0, current_c5 + 0.01))
    cons_fun, cons_jac = _make_slsqp_hard(n, dx, active_indices, n, radius)
    constraints = [{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}]
    eq_fun = lambda x: np.sum(x[:n]) - n / 2.0
    eq_jac = np.zeros(n + 1)
    eq_jac[:n] = 1.0
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': lambda x: eq_jac})
    obj_fn = lambda x: x[n]
    obj_grad = lambda x: next(iter(_gen_obj_grad(n)))
    try:
        res = minimize(obj_fn, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or (hasattr(res, 'x') and np.all(np.isfinite(res.x))):
            h_cand = res.x[:n]
            h_cand = _project(h_cand, n / 2.0)
            c5_cand = _exact_c5(h_cand)
            if c5_cand < current_c5 - 1e-10:
                return h_cand, c5_cand
    except Exception:
        pass
    return None, None

def _gen_obj_grad(n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    yield g

def _slsqp_ls_solve(h, radius, beta, maxiter):
    n = len(h)
    dx = 2.0 / n
    current_c5 = _exact_c5(h)
    t0_init = current_c5 + 1e-5
    x0 = np.concatenate([h, [t0_init]])
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((0.0, current_c5 + 0.01))

    def cons_fun(x):
        h_x = x[:n]
        t = x[n]
        scores = np.correlate(h_x, 1.0 - h_x, mode='full') * dx
        s_max = float(np.max(scores))
        v = beta * (scores - s_max)
        v = np.clip(v, -50.0, 50.0)
        lse = s_max + np.log(np.sum(np.exp(v))) / beta
        return t - lse

    def cons_jac(x):
        h_x = x[:n]
        t = x[n]
        scores_raw = np.correlate(h_x, 1.0 - h_x, mode='full')
        scores = scores_raw * dx
        s_max = float(np.max(scores))
        v = beta * (scores - s_max)
        v = np.clip(v, -50.0, 50.0)
        w = np.exp(v)
        w = w / float(np.sum(w))
        grad_h = np.zeros(n)
        for r in range(len(scores_raw)):
            if w[r] < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                grad_h[:m] += w[r] * dx * (1.0 - h_x[d:d+m])
                grad_h[d:] += w[r] * dx * (-h_x[:m])
            else:
                a = -d
                m = n - a
                grad_h[:m] += w[r] * dx * (-h_x[a:a+m])
                grad_h[a:] += w[r] * dx * (1.0 - h_x[:m])
        J = np.zeros(n + 1)
        J[:n] = -grad_h + np.mean(grad_h)
        J[n] = 1.0
        return J

    constraints = [{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}]
    eq_jac = np.zeros(n + 1)
    eq_jac[:n] = 1.0

    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': lambda x: eq_jac})

    def obj_fn(x):
        return x[n]

    try:
        res = minimize(obj_fn, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=lambda x: obj_grad_single(n),
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or (hasattr(res, 'x') and np.all(np.isfinite(res.x))):
            h_cand = res.x[:n]
            h_cand = _project(h_cand, n / 2.0)
            c5_cand = _exact_c5(h_cand)
            if c5_cand < current_c5 - 1e-10:
                return h_cand, c5_cand
    except Exception:
        pass
    return None, None

def _obj_grad_single(n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _smooth_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    v = np.clip(v, -50.0, 50.0)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(len(scores)):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += w[r] * dx * (1.0 - h[d:d+m])
            grad[d:] += w[r] * dx * (-h[:m])
        else:
            a = -d
            m = n - a
            grad[:m] += w[r] * dx * (-h[a:a+m])
            grad[a:] += w[r] * dx * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _active_phase(h, deadline, use_ls=True, use_hard=True):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.15
    stall = 0
    bi = 0
    beta_list = [80, 200, 500, 1000]
    maxiter_ls = [40, 30, 25]
    maxiter_hard = [40, 30, 25]
    while time.time() < deadline:
        if stall > 50:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        s_max = float(np.max(scores))
        improved_this_round = False

        if use_hard:
            for K in [10, 20, 35]:
                if time.time() > deadline:
                    break
                active = sorted(np.argsort(scores)[-K:])
                h_c, c5_c = _slsqp_hard_solve(h, active, radius, maxiter_hard[bi % len(maxiter_hard)])
                if h_c is not None and c5_c < best_c5 - 1e-12:
                    best_h = h_c.copy()
                    best_c5 = c5_c
                    h = h_c.copy()
                    cur_c5 = c5_c
                    stall = 0
                    radius = min(radius * 1.08, 0.45)
                    improved_this_round = True
                    break
            if improved_this_round:
                continue

        if use_ls:
            beta = beta_list[bi % len(beta_list)]
            h_c, c5_c = _slsqp_ls_solve(h, radius, beta, maxiter_ls[bi % len(maxiter_ls)])
            if h_c is not None and c5_c < best_c5 - 1e-12:
                best_h = h_c.copy()
                best_c5 = c5_c
                h = h_c.copy()
                cur_c5 = c5_c
                stall = 0
                radius = min(radius * 1.08, 0.45)
                bi += 1
                continue

        # Fallback: smooth gradient direction
        d = _smooth_grad(h, 2.0 / n, 200)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            stall += 2
            radius *= 0.85
            bi += 1
            continue
        d = d / dn
        alpha, new_c5 = _line_search(h, d, 2.0 / n, 0.3)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.45)
        else:
            stall += 1
            radius *= 0.85
            bi += 1

    return best_h, best_c5

def _pair_refine(h, deadline, rng, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_tries = min(200, n * n // 4)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _smooth_phase(h, deadline, betas=(100, 300, 800)):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.03
    mom = np.zeros(n)
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 50:
            break
        beta = betas[bi % len(betas)]
        d = _smooth_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            bi += 1
            stall += 1
            continue
        d = d / dn
        mom = 0.8 * mom + 0.2 * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            bi += 1
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.0)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.8
            bi += 1
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.03, 0.10)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 2.0)
        h_t, c5_t = _active_phase(h_k, sub_dl, use_ls=True, use_hard=False)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Active set SLSQP at original resolution (35%)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _active_phase(h, d1, use_ls=True, use_hard=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smooth gradient polish (10%)
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _smooth_phase(best_h, d2, betas=(150, 400, 1000))
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More active set SLSQP (15%)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _active_phase(best_h, d3, use_ls=True, use_hard=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution search (20%)
    d4 = t0 + budget_s * 0.80
    best_n = len(best_h)
    n_options = []
    for delta in [-20, -10, 10, 20, 30, -30, 40, -40, 50, -50]:
        nn = best_n + delta
        if nn >= 60 and nn <= 300:
            n_options.append(nn)
    n_options = sorted(set(n_options), key=lambda x: abs(x - best_n))
    for n_try in n_options:
        if time.time() > d4:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = _active_phase(h_r, sub_dl, use_ls=True, use_hard=True)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Kick and reoptimize (7%)
    d5 = t0 + budget_s * 0.87
    h5, c5_5 = _kick_reopt(best_h, d5, rng, n_kicks=25)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair refinement (6%)
    d6 = t0 + budget_s * 0.93
    h6, c5_6 = _pair_refine(best_h, d6, rng, max_passes=60)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final SLSQP polish (4%)
    d7 = t0 + budget_s * 0.97
    h7, c5_7 = _active_phase(best_h, d7, use_ls=True, use_hard=True)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final projection and validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
