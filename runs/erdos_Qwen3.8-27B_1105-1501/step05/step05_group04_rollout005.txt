def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float).copy()
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:

            x = np.clip(x, 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float).copy()
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m]) * dx
            G[r, :m] += -h[d:] * dx
        else:
            a = -d
            m = n - a
            G[r, :m] += (1.0 - h[a:]) * dx
            G[r, a:] += -h[:m] * dx
    return G

def _slsqp_epigraph(h0, t_upper, max_iter=500, ftol=1e-9, deadline=None):
    n = len(h0)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    target = n / 2.0
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - scores
    
    def ineq_jac(x):
        h = x[:n]
        G = _grad_matrix(h)
        J = np.zeros((n_lags, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    x0 = np.concatenate([h0.copy(), [t_upper]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t_upper + 0.005)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        if deadline is not None and time.time() > deadline:
            return None, None
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                      constraints=constraints, method='SLSQP',
                      options={'maxiter': max_iter, 'ftol': ftol})
        if deadline is not None and time.time() > deadline:
            return None, None
        h_new = _project(res.x[:n], target)
        c5_new = _exact_c5(h_new)
        if c5_new < t_upper:
            return h_new, c5_new
        return None, None
    except Exception:
        return None, None

def _lp_solve(h, scores, G, radius, n_active):
    """Find LP direction to reduce top-n_active lags."""
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:min(n_active, n_lags)]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs')
        if res.success:
            q_val = res.x[-1]
            d = res.x[:n]
            if q_val < scores[active][0] - 1e-12 and np.max(np.abs(d)) > 1e-13:
                return d, q_val
    except Exception:
        pass
    return None, None

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1**2 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2**2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1**2 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2**2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t**2 * C)) * dx
    return opt_t, opt_val

def _lp_loop(h, deadline, radius_init=0.10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 3
    
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _all_scores(h)
        G = _grad_matrix(h)
        p, q_val = _lp_solve(h, scores, G, radius, n_active)
        if p is None:
            n_active = min(n_active + 1, 8)
            p, q_val = _lp_solve(h, scores, G, radius * 1.5, n_active)
            if p is None:
                stall += 2
                radius *= 0.8
                if radius < 1e-6:
                    break
                if stall > 80:
                    radius = 0.05
                    n_active = 3
                continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 3
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.82
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
                n_active = 4
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_seeds(h, rng, n_seeds=8):
    """Generate diverse starting points."""
    n = len(h)
    seeds = [h.copy()]
    
    # Gaussian perturbations at various scales
    for scale in [0.02, 0.05, 0.1, 0.15]:
        s = _project(h + rng.normal(0, scale, n), n/2.0)
        seeds.append(s)
    
    # Anti-symmetric perturbation
    a = 0.5 * (h[:n//2] + 1.0 - h[n-1:n//2-1:-1])
    a = a + rng.normal(0, 0.05, n//2)
    h_anti = np.empty(n)
    h_anti[:n//2] = a
    h_anti[n//2:] = 1.0 - a[::-1]
    if n % 2 == 1:
        h_anti[n//2] = 0.5
    seeds.append(_project(h_anti, n/2.0))
    
    # Sinusoidal
    x = np.linspace(0, 2, n)
    for freq in [2, 3]:
        s = 0.5 + 0.3 * np.sin(np.pi * freq * x / 2.0 + rng.uniform(0, 2*np.pi))
        seeds.append(_project(s, n/2.0))
    
    return seeds[:n_seeds]

def _perturb_lp_search(h, deadline, rng, n_trials=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.05
    
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = _project(best_h + rng.normal(0, scale, n), n/2.0)
        sub_dl = min(deadline, time.time() + 8.0)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.08)
    
    return best_h, best_c5

def _sparse_coord_refine(h, deadline, n_iter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0-h[j], 0.3))
        if c5_try < best_c5 - 1e-13:
            h = _project(h + alpha * d, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0-1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def _multiresolution(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tried = set()
    
    candidates = sorted(set([n-6, n-3, n, n+3, n+6, n+10, n+15, 
                              int(n*0.8), int(n*0.9), int(n*1.1), int(n*1.2),
                              int(n*1.3), int(n*1.5)]))
    
    for n_try in candidates:
        if time.time() > deadline:
            break
        if n_try < 30 or n_try > 200 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline, time.time() + 6.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    return best_h, best_c5
