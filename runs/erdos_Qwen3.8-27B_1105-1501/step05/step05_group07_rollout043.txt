            t_hi = mid
        elif sum_a_active < -1e-12:

        else:

def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_G(h):
    n = len(h)
    G = np.zeros((n, 2 * n - 1))
    d_range = np.arange(-(n - 1), n)
    for p in range(n):
        idx1 = p + d_range
        m1 = (idx1 >= 0) & (idx1 < n)
        if np.any(m1):
            G[p, m1] += 1.0 - h[idx1[m1]]
        idx2 = p - d_range
        m2 = (idx2 >= 0) & (idx2 < n)
        if np.any(m2):
            G[p, m2] -= h[idx2[m2]]
    return G

        else:

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = (np.arange(n) + 0.5) / n
    h = ((x * alpha * 2.0) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def _run_two_coord_optimize(h, deadline, max_sweeps=500):
    n = len(h)
    S = _scores(h)
    G = _compute_G(h)
    best_h = h.copy()
    best_c5 = float(np.max(S))
    rng = np.random.default_rng(12345)
    stall = 0
    for sweep_idx in range(max_sweeps):
        if time.time() > deadline:
            break
        h, S, G, cur_c5, n_acc = _sweep(h, S, G, deadline)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if n_acc == 0:
            stall += 2
        if stall > 20:
            scale = 0.005 + 0.01 * min(stall / 50.0, 1.0)
            h = h + rng.normal(0, scale, n)
            h = _project(h, n / 2.0)
            S = _scores(h)
            G = _compute_G(h)
            stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _deadline
    t0 = time.time()
    _deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Two-coordinate exact minimax (main engine, 45% budget)
    dl1 = t0 + budget_s * 0.45
    h1, c5_1 = _run_two_coord_optimize(best_h, dl1, max_sweeps=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: LP polish (10% budget)
    if time.time() < t0 + budget_s * 0.58:
        dl2 = t0 + budget_s * 0.58
        h2, c5_2 = _lp_polish(best_h, dl2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = len(h2)

    # Phase 3: Two-coordinate after LP (7% budget)
    if time.time() < t0 + budget_s * 0.65:
        dl3 = t0 + budget_s * 0.65
        h3, c5_3 = _run_two_coord_optimize(best_h, dl3, max_sweeps=500)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = len(h3)

    # Phase 4: Multi-resolution with two-coordinate (18% budget)
    if time.time() < t0 + budget_s * 0.83:
        dl4 = t0 + budget_s * 0.83
        n_list = [96, 160, 192, 256, 128]
        for n_try in n_list:
            if time.time() > dl4:
                break
            sub_dl = min(dl4, time.time() + budget_s * 0.035)
            h4 = _resample(best_h, n_try)
            h4r, c5r = _run_two_coord_optimize(h4, sub_dl, max_sweeps=300)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h4r.copy()
                best_n = n_try
            # Quasi-periodic start
            if time.time() < dl4:
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5]
                for ai in range(min(len(alphas), 3)):
                    if time.time() > sub_dl2:
                        break
                    hq = _quasi_init(n_try, alphas[ai])
                    hq += rng.normal(0, 0.04, n_try)
                    hq = _project(hq, n_try / 2.0)
                    hqr, c5q = _run_two_coord_optimize(hq, sub_dl2, max_sweeps=200)
                    if c5q < best_c5:
                        best_c5 = c5q
                        best_h = hqr.copy()
                        best_n = n_try

    # Phase 5: LP + two-coord final polish on best (5% budget)
    if time.time() < t0 + budget_s * 0.90:
        dl5 = t0 + budget_s * 0.87
        h5, c5_5 = _lp_polish(best_h, dl5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < t0 + budget_s * 0.90:
            dl5b = t0 + budget_s * 0.90
            h5b, c5_5b = _run_two_coord_optimize(best_h, dl5b, max_sweeps=500)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()
                best_n = len(h5b)

    # Phase 6: Final aggressive two-coordinate sweep (7% budget)
    if time.time() < t0 + budget_s * 0.97:
        dl6 = t0 + budget_s * 0.97
        h6, c5_6 = _run_two_coord_optimize(best_h, dl6, max_sweeps=1000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
            best_n = len(h6)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
