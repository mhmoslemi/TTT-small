import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d == 0:
            G[r, :] = (1.0 - 2.0 * h) * dx
        elif d > 0:
            m = n - d
            G[r, :m] = (1.0 - h[d:d+m]) * dx
            G[r, d:] += -h[:m] * dx
        else:
            g = -d
            m = n - g
            G[r, g:] = (1.0 - h[:m]) * dx
            G[r, :m] += -h[g:n] * dx
    return G

def _lp_direction(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(40):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _local_optimize_lp(h, deadline, radius_init=0.25, max_iters=10000, stall_limit=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        p, _ = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.6)
        else:
            stall += 1
            radius *= 0.8
            if stall > 40 and radius < 0.05:
                radius = 0.15
                stall = 10  # partial reset
    return best_h, best_c5

def _l_bfgs_optimize(h, deadline, beta_init=50.0, beta_final=500.0, n_phases=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        
        def obj(hf):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = np.clip(beta * (S - maxS), -500, 0)
            return maxS + np.log(np.mean(np.exp(shifted))) / beta
        
        def grad(hf):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = np.clip(beta * (S - maxS), -500, 0)
            w = np.exp(shifted)
            wsum = w.sum()
            if wsum < 1e-300:
                w = np.zeros(len(S))
                w[int(np.argmax(S))] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxls': 50})
            if np.all(np.isfinite(res.x)):

                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _anneal_refine(h, deadline, t_start=0.05, t_end=0.001, n_cool_cycles=5):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    
    for cycle in range(n_cool_cycles):
        if time.time() > deadline:
            break
        # Cooling schedule
        frac = cycle / max(1, n_cool_cycles - 1)
        T = t_start * (t_end / t_start) ** frac
        # Number of SA steps
        n_steps = 2000 - 1500 * frac
        cur_c5 = best_c5
        for step in range(int(n_steps)):
            if time.time() > deadline:
                break
            # Random perturbation: flip or adjust a random subset
            k = rng.integers(1, max(2, n // 8))
            idx = rng.choice(n, size=k, replace=False)
            delta = rng.uniform(-0.1, 0.1, size=k)
            h_try = h.copy()
            h_try[idx] += delta
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 or rng.random() < np.exp(-(c5_try - cur_c5) / max(T, 1e-10)):
                h = h_try
                cur_c5 = c5_try

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d == 0:
            G[r, :] = (1.0 - 2.0 * h) * dx
        elif d > 0:
            m = n - d
            G[r, :m] = (1.0 - h[d:d + m]) * dx
            G[r, d:] += -h[:m] * dx
        else:
            g = -d
            m = n - g
            G[r, g:] = (1.0 - h[:m]) * dx
            G[r, :m] += -h[g:n] * dx
    return G

def _lp_direction(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _l_bfgs_optimize(h, deadline, beta_init=50.0, beta_final=800.0, n_phases=6):

    target = n / 2.0

    best_h = h.copy()
    best_c5 = _exact_c5(h)

        if time.time() > deadline:
            break

        beta = beta_init * (beta_final / beta_init) ** frac

        def obj(hf):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = np.clip(beta * (S - maxS), -500, 0)
            return maxS + np.log(np.mean(np.exp(shifted))) / beta

        def grad(hf):
            S = _scores(hf)
            maxS = np.max(S)
            shifted = np.clip(beta * (S - maxS), -500, 0)
            w = np.exp(shifted)

                w[int(np.argmax(S))] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w

                        best_n = n_try

    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.85
        scale = 0.012
        for attempt in range(400):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.008)
            ht, ct = _local_optimize_lp(h_pert, sub_dl, radius_init=0.05, max_iters=500, stall_limit=50)
            if ct < best_c5 - 1e-13:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.002, scale * 0.85)
            else:
                scale = min(0.05, scale * 1.06)

    # Phase 6: Final L-BFGS + LP polish
    if time.time() < t0 + budget_s * 0.95:
        dl6 = t0 + budget_s * 0.95
        h6, c5_6 = _l_bfgs_optimize(best_h, dl6, beta_init=100.0, beta_final=1500.0, n_phases=4)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        h6b, c5_6b = _local_optimize_lp(best_h, t0 + budget_s * 0.97,
                                         radius_init=0.03, max_iters=3000, stall_limit=150)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Pair polish
    if time.time() < deadline:
        hp, c5p = _pair_polish(best_h, deadline, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
