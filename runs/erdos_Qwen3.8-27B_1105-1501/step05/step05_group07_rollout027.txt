   For pair (i,j), the "base" a[d] = (h[j+d mod n boundary] + h[j-d mod n boundary]) - (h[i+d] + h[i-d])
   where out-of-range terms are 0.

2. Vectorize over j for fixed i: compute all a_j[d] at once
   a_j[d] = h[j+d] + h[j-d] - h[i+d] - h[i-d] (with boundary=0)
   For fixed i and all j, d: this is a 2D array computation.
   
   Actually, for fixed i, the term -(h[i+d]+h[i-d]) is just a vector of length 2n-1.
   The term (h[j+d]+h[j-d]) for varying j is like a "sliding sum" which can be computed efficiently.
   
   h[j+d] for varying j and d: this is just h shifted. For d≥0: h[j+d] = h[d:] for j=0, h[d+1:] for j=1, etc.
   This is a Toeplitz-like matrix.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_ab(h, i, j, n):
    """Compute linear and quadratic coefficients for pair swap h[i]+=d, h[j]-=d."""
    n_lags = 2 * n - 1
    a = np.zeros(n_lags)
    b = np.zeros(n_lags)
    hi = h[i]
    hj = h[j]
    did = j - i
    for o in range(-(n - 1), n):
        idx = n - 1 + o
        if o == 0:
            a[idx] = -2.0
            b[idx] = 2.0
        elif o == did:
            # i+o = j: special pair (i,j)
            lin = (1.0 - hj) + hi
            quad = 1.0
            k1 = j + o  # = 2*j - i
            if 0 <= k1 < n and k1 != i:
                lin -= (1.0 - h[k1])
            k2 = i - o  # = 2*i - j
            if 0 <= k2 < n and k2 != j:
                lin -= h[k2]
            a[idx] = lin
            b[idx] = quad
        elif o == -did:
            # j+o = i: special pair (j,i)
            lin = -(1.0 - hi) - hj
            quad = 1.0
            k1 = i + o  # = 2*i - j
            if 0 <= k1 < n and k1 != j:
                lin += (1.0 - h[k1])
            k2 = j - o  # = 2*j - i
            if 0 <= k2 < n and k2 != i:
                lin += h[k2]
            a[idx] = lin
            b[idx] = quad
        else:
            lin = 0.0
            ip = i + o
            if 0 <= ip < n and ip != j:
                lin += (1.0 - h[ip])
            jp = j + o
            if 0 <= jp < n and jp != i:
                lin -= (1.0 - h[jp])
            im = i - o
            if 0 <= im < n and im != j:
                lin -= h[im]
            jm = j - o
            if 0 <= jm < n and jm != i:
                lin += h[jm]
            a[idx] = lin
    return a, b

def _ternary_max(scores, a, b, lo, hi, n_iter=45):
    """Find delta in [lo,hi] minimizing max(scores + a*delta + b*delta^2)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + a * c1 + b * c1 * c1))
    f2 = float(np.max(scores + a * c2 + b * c2 * c2))
    for _ in range(n_iter):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + a * c1 + b * c1 * c1))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + a * c2 + b * c2 * c2))
    opt = (lo + hi) / 2.0
    return opt, f1 if f1 < f2 else f2, float(np.max(scores + a * opt + b * opt * opt))

def _two_coord_optimize(h, deadline, max_passes=200, tol=1e-13):
    """Exact two-coordinate minimax optimizer."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scores = _scores(h)
    cur_max = float(np.max(scores))
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        best_impr = tol
        best_i, best_j, best_d = -1, -1, 0.0
        best_f = cur_max
        
        for i in range(n):
            hi = h[i]
            if hi < 1e-15 or hi > 1.0 - 1e-15:
                continue
            for j in range(n):
                if i == j:
                    continue
                hj = h[j]
                if hj < 1e-15 or hj > 1.0 - 1e-15:
                    continue
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_lo >= d_hi - 1e-12:
                    continue
                # Quick linear check: compute a only (skip b for speed)
                a, b = _compute_ab(h, i, j, n)
                # Check if 0 is interior local min (gradient = 0 at active lag)
                # Just do ternary search
                opt_d, f_bound, f_opt = _ternary_max(scores, a, b, d_lo, d_hi)
                f0 = cur_max
                if f_opt < f0 - best_impr:
                    best_impr = f0 - f_opt
                    best_i, best_j, best_d = i, j, opt_d
                    best_f = f_opt
                elif f_bound < f0 - best_impr:
                    best_impr = f0 - f_bound
                    best_i, best_j = i, j
                    best_f = f_bound
                    # Recompute best_d for boundary
                    if float(np.max(scores + a * d_lo + b * d_lo * d_lo)) < float(np.max(scores + a * d_hi + b * d_hi * d_hi)):
                        best_d = d_lo
                    else:
                        best_d = d_hi
        
        if best_i < 0:
            break
        h[best_i] += best_d
        h[best_j] -= best_d
        h = _project(h, target)
        scores = _scores(h)
        cur_max = float(np.max(scores))
        if cur_max < best_c5:
            best_c5 = cur_max
            best_h = h.copy()
    
    return best_h, best_c5

def _gradient_two_coord(h, deadline, max_passes=50):
    """Faster variant: use gradient to select which pairs to try."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scores = _scores(h)
    cur_max = float(np.max(scores))
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        # Find active lags
        active_thresh = cur_max - 0.02
        active = np.where(scores >= active_thresh)[0]
        if len(active) == 0:
            active = np.where(scores >= cur_max - 0.1)[0]
        
        # Score candidate pairs by potential improvement on active lags
        candidates = []
        for i in range(n):
            if h[i] < 1e-12 or h[i] > 1.0 - 1e-12:
                continue
            for j in range(n):
                if i == j:
                    continue
                if h[j] < 1e-12 or h[j] > 1.0 - 1e-12:
                    continue
                # Quick score: how much does this pair reduce active lags?
                # Use linear approximation on a few active lags
                best_lin = 0
                for ad in active[:8]:
                    o = ad - (n - 1)
                    # Compute a[ad] quickly
                    lin = 0.0
                    if o == 0:
                        lin = -2.0
                    elif o == j - i:
                        lin = (1.0 - h[j]) + h[i]
                        k1 = 2 * j - i
                        if 0 <= k1 < n and k1 != i:
                            lin -= (1.0 - h[k1])
                        k2 = 2 * i - j
                        if 0 <= k2 < n and k2 != j:
                            lin -= h[k2]
                    elif o == i - j:
                        lin = -(1.0 - h[i]) - h[j]
                        k1 = 2 * i - j
                        if 0 <= k1 < n and k1 != j:
                            lin += (1.0 - h[k1])
                        k2 = 2 * j - i
                        if 0 <= k2 < n and k2 != i:
                            lin += h[k2]
                    else:
                        ip = i + o
                        if 0 <= ip < n and ip != j:
                            lin += (1.0 - h[ip])
                        jp = j + o
                        if 0 <= jp < n and jp != i:
                            lin -= (1.0 - h[jp])
                        im = i - o
                        if 0 <= im < n and im != j:
                            lin -= h[im]
                        jm = j - o
                        if 0 <= jm < n and jm != i:
                            lin += h[jm]
                    if lin < best_lin:
                        best_lin = lin
                if best_lin < -0.001:
                    candidates.append((best_lin, i, j))
        
        if not candidates:
            break
        
        candidates.sort()
        best_impr = 1e-13
        best_move = None
        
        for _, i, j in candidates[:80]:
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-12:
                continue
            a, b = _compute_ab(h, i, j, n)
            opt_d, f_bound, f_opt = _ternary_max(scores, a, b, d_lo, d_hi, n_iter=40)
            f = min(f_opt, f_bound)
            if f < cur_max - best_impr:
                best_impr = cur_max - f
                best_move = (i, j, opt_d if f_opt <= f_bound else (d_lo if f_bound == float(np.max(scores + a*d_lo + b*d_lo*d_lo)) else d_hi))
        
        if best_move is None:
            break
        i, j, d = best_move
        h[i] += d
        h[j] -= d
        h = _project(h, target)
        scores = _scores(h)
        cur_max = float(np.max(scores))
        if cur_max < best_c5:
            best_c5 = cur_max
            best_h = h.copy()
    
    return best_h, best_c5

def _lp_warmup(h, deadline, radius_init=0.25, max_iters=2000):
    """Quick LP-based warmup to get close to local optimum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scores = _scores(h)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Find worst lag
        widx = int(np.argmax(scores))
        o = widx - (n - 1)
        # Gradient of score[widx] w.r.t. h
        g = np.zeros(n)
        for k in range(n):
            kp = k + o
            km = k - o
            if 0 <= kp < n:
                g[k] += (1.0 - h[kp])
                if kp != k:
                    g[kp] += -h[k]
            if o != 0 and 0 <= km < n:
                g[k] += -h[km] if km != k else 0
        # Simpler: just use the negative gradient of the worst score
        g = np.zeros(n)
        for k in range(n):
            if 0 <= k + o < n:
                g[k] += (1.0 - h[k + o])
            if o != 0 and 0 <= k - o < n:
                g[k] += -h[k - o]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        # Line search
        best_a = 0
        best_f = best_c5
        for trial in range(20):
            a = min(0.3 / (trial + 1), 0.5)
            h_try = _project(h - a * g, target)
            f = _exact_c5(h_try)
            if f < best_f:
                best_f = f
                best_a = a
                best_h = h_try.copy()
                best_c5 = f
            else:
                break
        if best_a == 0:
            break
        h = best_h
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)
