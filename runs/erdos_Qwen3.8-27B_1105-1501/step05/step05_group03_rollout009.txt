def _project(v, target):
    """Project v to [0,1]^n with sum = target using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection to find threshold t such that sum(clip(v-t, 0, 1)) = target
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:

        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)
    return x

The key new element vs the parent: **binary local search**. This is a well-understood combinatorial optimization that can find good solutions quickly.

For binary local search:
- State: binary vector h with exactly n/2 ones
- Move: swap a 1 at position i with a 0 at position j
- Objective: C5 = max_d score_d
- Accept if C5 doesn't increase (or with simulated annealing)

The number of possible moves is (n/2) * (n/2) = n^2/4. For n=128, that's 4096 moves per iteration. Each move requires recomputing C5, which is O(n^2) with np.correlate. So one full pass costs O(n^4/4) = about 67 million for n=128. That's too slow for many passes.

Better approach: incremental score update. When we swap h[i] from 1 to 0 and h[j] from 0 to 1:

- So actually all scores change. But we can update them incrementally.

def _binary_local_search(h_init, deadline, n_max_iters=50000):
    """Binary local search: swap 1s and 0s to minimize C5."""
    n = len(h_init)
    target = n // 2
    # Ensure binary
    h = h_init.copy()
    # Project to binary: keep the n/2 largest values as 1s
    idx = np.argsort(h)[::-1][:target]
    h = np.zeros(n)
    h[idx] = 1.0
    
    scores = _scores(h)
    cur_c5 = np.max(scores)
    best_h = h.copy()
    best_c5 = cur_c5
    
    G = _grad_matrix(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    it = 0
    while it < n_max_iters:
        if time.time() > deadline:
            break
        it += 1
        # Try all pairs (i in ones, j in zeros) - but that's too many
        # Instead, try a random subset
        # Or: try all moves and pick the best improving one
        best_move = None
        best_new_c5 = cur_c5
        
        # For efficiency, try a random sample of moves
        n_trials = min(len(ones) * len(zeros), 500)
        for _ in range(n_trials):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            # New scores = scores - G[:,i] + G[:,j]  (h[i]: 1->0, h[j]: 0->1)
            new_scores = scores - G[:, i] + G[:, j]
            new_c5 = np.max(new_scores)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_move = (i, j, new_scores)
        
        if best_move is not None:
            i, j, new_scores = best_move
            h[i] = 0.0
            h[j] = 1.0
            scores = new_scores
            cur_c5 = best_new_c5
            # Update G: column i changes (was 1, now 0), column j changes (was 0, now 1)
            # G[r, i] was computed with h[i]=1, now h[i]=0
            # G[r, j] was computed with h[j]=0, now h[j]=1
            # Need to recompute G[:, i] and G[:, j]
            # Actually, G depends on all of h, not just one position
            # G[r, j] = d(score_r)/d(h[j]) which depends on h[j±d] for various d
            # So changing h[i] and h[j] changes many columns of G
            # This is expensive...
            
            # Alternative: just recompute G from scratch (O(n^2))
            G = _grad_matrix(h)
            
            # Update ones and zeros
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break  # Stuck
    
    return best_h, best_c5

Wait, recomputing G from scratch each time is O(n^2) (since _grad_matrix fills a (2n-1) × n matrix, each row computed in O(n), total O(n^2)). For n=128, that's 16384 operations per recompute. With 500 trials per iteration, each recompute happens only when we accept a move. So the total cost per accepted move is 500 * O(n) [for the trials] + O(n^2) [for G recompute] = about 500*128 + 16384 ≈ 80000 operations. That's fast.

But wait, the issue is that the score update formula `new_scores = scores - G[:, i] + G[:, j]` is only valid if G was computed at the current h. After we change h, G changes. So we need to be careful.

Actually, the formula is:
score_r = sum_{valid i} h[i] * (1 - h[i+d])

This is a bilinear form. If we change h[i] by δ_i and h[j] by δ_j, the change in score_r is:
Δscore_r = δ_i * G[r,i] + δ_j * G[r,j] + (cross terms if i and j interact)

The cross terms: if i and j are at distance d (i.e., j = i+d or i = j+d), then there's a cross term. Specifically:
- If d = j - i > 0: the term h[i]*(1-h[j]) appears in score_d. Changing both gives a cross term: δ_i * (-δ_j) = -δ_i * δ_j.
  Wait: originally the term is h[i]*(1-h[j]) = (1+δ_i)*(1-(1+δ_j))... no wait, h[i] goes from 1 to 0 (δ_i = -1) and h[j] goes from 0 to 1 (δ_j = +1).
  
  Original contribution to score_{j-i}: h[i]*(1-h[j]) = 1*1 = 1
  New contribution: h[i]*(1-h[j]) = 0*0 = 0
  Change: -1
  
  Using the formula: δ_i * G[r,i] + δ_j * G[r,j] = (-1) * G[r,i] + (1) * G[r,j]
  
  G[r,i] for r = n-1+(j-i) (i.e., lag = j-i): this is the derivative of score_{j-i} w.r.t. h[i].
  score_{j-i} = sum_{k=0}^{n-(j-i)-1} h[k]*(1-h[k+j-i])
  The term h[i]*(1-h[i+j-i]) = h[i]*(1-h[j]) appears when k=i.
  G[r,i] = (1-h[j]) = 1-0 = 1 (the "way 1" contribution)
  
  G[r,j] for the same r: h[j] appears in score_{j-i} as h[k]*(1-h[k+j-i]) where k+j-i = j, i.e., k = i.
  So G[r,j] = -h[i] = -1 (the "way 2" contribution, negative sign from the (1-h[...]) term)
  
  So: (-1)*1 + (1)*(-1) = -2. But the actual change is -1. So there's a discrepancy of -1, which is the cross term!
  
  The cross term is: -δ_i * δ_j = -(-1)(+1) = +1. So: -2 + 1 = -1. ✓

So the formula `new_scores = scores + G[:, i]*(-1) + G[:, j]*(+1)` is missing the cross term when i and j interact.

The cross term only affects scores where |i-j| is in a specific range. Specifically, if d = |j-i|, then the cross term affects score_d (and possibly score_{-d} due to the non-circular nature).

For a swap (1 at i, 0 at j), the cross term is:
- For d = j - i (if j > i): the term h[i]*(1-h[j]) in score_d changes by -δ_i*δ_j = -(-1)(1) = +1... 

Hmm, let me just handle this carefully. The bilinear form is:
score_d = sum_{k} h[k] * (1 - h[k+d])  (for d >= 0, with appropriate range)

= sum_k h[k] - sum_k h[k]*h[k+d]

The change when h[i] += δ_i and h[j] += δ_j:
Δscore_d = δ_i - (change in sum_k h[k]*h[k+d])

The change in sum_k h[k]*h[k+d]:
= sum_k (h[k]+δ_k)*(h[k+d]+δ_{k+d}) - sum_k h[k]*h[k+d]
= sum_k δ_k + sum_k h[k]*δ_{k+d} + sum_k δ_k*δ_{k+d}

where δ_k = δ_i if k=i, δ_j if k=j, 0 otherwise.

So sum_k δ_k = δ_i + δ_j = -1 + 1 = 0 (for a swap)
sum_k h[k]*δ_{k+d} = h[i-d]*δ_i + h[j-d]*δ_j (if indices valid) + h[i]*δ_j... 

This is getting complicated. Let me just use a simpler approach: compute the new scores directly by evaluating the bilinear form, or use the gradient matrix and accept a small error from the cross term (which only affects O(1) scores out of O(n) total).

Actually, for the purpose of comparing new_c5 vs cur_c5, a small error in a few scores won't matter much if the max is achieved at a different lag. And if the cross term does matter, we'll catch it when we actually apply the move and recompute.

Let me use a hybrid approach:
1. Fast screening: use the linearized score update (ignoring cross terms) to find candidate moves
2. Exact verification: for the best candidate, compute the exact new C5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _softmax_weights(scores, beta):
    max_s = np.max(scores)
    w = np.exp(beta * (scores - max_s))
    sw = np.sum(w)
    if sw < 1e-300:
        idx = np.argmax(scores)
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w /= sw
    return w

def _subgrad_optimize(h, deadline, beta_start=10, beta_end=300, n_steps=400,
                       step_init=0.04, momentum=0.92):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        w = _softmax_weights(scores, beta)
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = momentum * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.02, 0.10)
        else:
            v = 0.5 * v
            step *= 0.82
            if step < 1e-7:
                break
    return best_h, best_c5
