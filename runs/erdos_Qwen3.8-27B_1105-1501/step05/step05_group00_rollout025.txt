import numpy as np

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_box_sum(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Water filling
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_box_sum(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -1.0
    hi = 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_gradient(h, lag):
    """Gradient of correlate(h,1-h) at given lag offset with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return g
        g[lag:] += (1.0 - h[:m])
        g[:m] += -h[lag:]
    else:
        g_abs = -lag
        m = n - g_abs
        if m <= 0:
            return g
        g[g_abs:] += (1.0 - h[:m])
        g[:m] += -h[g_abs:]
    return g

def line_search_pair(h, i, j, base_corr, dx):
    """Exact line search for pair direction d[i]=-1, d[j]=+1."""
    n = len(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    B = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds
    a_lo = max(-h[j], 1.0 - h[i])
    a_hi = min(h[i], 1.0 - h[j])
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(base_corr) * dx)
    
    # Grid search
    ng = 32
    alphas = np.linspace(a_lo, a_hi, ng)
    # Vectorized: shape (ng, 2n-1)
    vals = base_corr[None, :] + alphas[:, None] * B[None, :] - (alphas[:, None] ** 2) * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_a = alphas[best_i]
    best_v = obj[best_i]
    
    # Golden section
    if best_i > 0 and best_i < ng - 1:
        a, b = alphas[best_i - 1], alphas[best_i + 1]
    else:
        a, b = a_lo, a_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(base_corr + c1 * B - c1 * c1 * C)) * dx
    f2 = float(np.max(base_corr + c2 * B - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(base_corr + c1 * B - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(base_corr + c2 * B - c2 * c2 * C)) * dx
    opt_a = (a + b) / 2.0
    opt_v = float(np.max(base_corr + opt_a * B - opt_a * opt_a * C)) * dx
    if opt_v < best_v:
        best_a, best_v = opt_a, opt_v
    return best_a, best_v

def local_search(h, deadline, seed=42, top_k=25, max_dirs=100, perturb_scale=0.03):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    base_corr = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(seed)
    stall = 0
    n_lags = 2 * n - 1

        # Compute scored array
        scored = base_corr * dx
        max_val = float(np.max(scored))

        n_active = min(20, n_lags)
        active_thresh = max_val - 0.02 * max_val  # within 2% of max
        active_idx = np.where(scored >= active_thresh)[0]

            active_idx = np.argsort(scored)[-n_active:]

        weights = (scored[active_idx] - (max_val - 0.02 * max_val) + 0.001)
        wsum = float(np.sum(weights))
        if wsum > 0:
            weights = weights / wsum
        else:

        g_total = np.zeros(n)
