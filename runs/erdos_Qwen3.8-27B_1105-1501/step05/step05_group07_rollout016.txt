import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G * (2.0 / n)

def _grad_matrix_fast(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    ones_h = 1.0 - h
    for lag in range(n):
        if lag == 0:
            r = n - 1
            G[r, :] += ones_h
            G[r, :] += -h
        else:
            r_pos = n - 1 + lag
            r_neg = n - 1 - lag
            m = n - lag
            G[r_pos, :m] += ones_h[lag:]
            G[r_pos, lag:] += -h[:m]
            G[r_neg, lag:] += ones_h[:m]
            G[r_neg, :m] += -h[lag:]
    return G * dx

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility': 1e-9})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    n_samp = 16
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(18):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _local_optimize_lp(h, deadline, radius_init=0.25, max_iters=10000, stall_limit=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix_fast(h)
        p, _ = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.85
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.85
            if stall > 40 and radius < 0.02:
                break
    return best_h, best_c5

def _subgrad_optimize(h, deadline, n_steps=3000, beta_start=10.0, beta_end=80.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    step = 0
    while step < n_steps:
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix_fast(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.15
        improved = False
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-13:
                break
        if not improved:
            alpha = 0.01
            h = _project(h - alpha * grad, target)
            cur_c5 = _exact_c5(h)
        step += 1
    return best_h, best_c5

def _bfgs_smooth(h_init, deadline, beta_init=20.0, beta_final=200.0, n_phases=5):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n_lags = 2 * n - 1
    beta_min = max(6.0, 0.1 * n_lags)
    beta_max = max(beta_min + 30, min(800.0, 2.0 * n_lags))

    def _obj_h(hf):
        S = _scores(hf)
        b = beta
        maxS = np.max(S)
        shifted = np.clip(b * (S - maxS), -500.0, 0.0)
        return maxS + np.log(np.mean(np.exp(shifted))) / b

    def _obj_h_grad(hf):
        S = _scores(hf)
        b = beta
        maxS = np.max(S)
        shifted = np.clip(b * (S - maxS), -500.0, 0.0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w /= wsum
        G = _grad_matrix_fast(hf)
        g = G.T @ w
        return _obj_h(hf), g

    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_min * (beta_max / beta_min) ** frac
        try:
            res = minimize(_obj_h_grad, h, jac=True, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10})
            if np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                else:
                    h = _project(res.x, target)
        except Exception:
            pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_size = max(1, n // 40)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step_size):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step_size, min(n, j + step_size * 15), step_size):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _complement_lp(h_current):
    n = len(h_current)
    T = n / 2.0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                A_ub[r, :m] += h_current[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                A_ub[r, g:] += h_current[:m]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([T])
    bounds = [(0.0, 1.0)] * n + [(-1.0 * T, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            y = res.x[:n]
            z = 1.0 - y
            return _project(z, T)
    except Exception:
        pass
    return None

        if time.time() < dl3:
            dl3b = min(dl3, time.time() + budget_s * 0.04)
            h3b, c5_3b = _local_optimize_lp(best_h, dl3b, radius_init=0.08, max_iters=3000, stall_limit=100)
            if c5_3b < best_c5:
                best_c5 = c5_3b; best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.75:
        dl4 = t0 + budget_s * 0.75
        n_tries = [int(n0 * 4 // 3), n0 * 2, int(n0 * 5 // 4)]
        n_tries = [x for x in n_tries if n0 < x <= 320]
        n_tries = sorted(set(n_tries))
        alphas = [0.6180339887, 0.4142135624, 0.7071067812]
        for idx, n_try in enumerate(n_tries):
            if time.time() > dl4 or n_try > 288:
                break
            # Resample from best
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl4, time.time() + budget_s * 0.045)
            ht, ct = _local_optimize_lp(h_init, sub_dl, radius_init=0.25, max_iters=4000, stall_limit=130)
            if ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try
            # Quasi-periodic start at new resolution
            if time.time() < dl4:
                h_qp = _quasi_periodic_init(n_try, alphas[idx % len(alphas)])
                h_qp += rng.normal(0, 0.06, n_try)
                h_qp = _project(h_qp, n_try / 2.0)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.035)
                ht2, ct2 = _local_optimize_lp(h_qp, sub_dl2, radius_init=0.25, max_iters=3000, stall_limit=100)
                if ct2 < best_c5:
                    best_c5 = ct2; best_h = ht2.copy(); best_n = n_try
                if time.time() < dl4:
                    ht3, ct3 = _subgrad_optimize(ht2, min(dl4, time.time() + budget_s * 0.02),
                                                  n_steps=1000, beta_start=10, beta_end=50)
                    if ct3 < best_c5:
                        best_c5 = ct3; best_h = ht3.copy(); best_n = n_try
                    ht4, ct4 = _local_optimize_lp(best_h, min(dl4, time.time() + budget_s * 0.015),
                                                   radius_init=0.1, max_iters=2000, stall_limit=80)
                    if ct4 < best_c5:
                        best_c5 = ct4; best_h = ht4.copy(); best_n = n_try

    # Phase 5: Complement LP exploration
    if time.time() < t0 + budget_s * 0.82:
        dl5 = t0 + budget_s * 0.82
        for _comp_iter in range(8):
            if time.time() > dl5:
                break
            z = _complement_lp(best_h)
            if z is None:
                break
            c5_z = _exact_c5(z)
            if c5_z < best_c5:
                best_c5 = c5_z; best_h = z.copy()
            # Line search between current and complement
            lo_a, hi_a = 0.0, 1.0
            for _ in range(10):
                m1 = lo_a + (hi_a - lo_a) / 3.0
                m2 = hi_a - (hi_a - lo_a) / 3.0
                hm1 = _project(best_h * (1 - m1) + z * m1, best_n / 2.0)
                hm2 = _project(best_h * (1 - m2) + z * m2, best_n / 2.0)
                f1 = _exact_c5(hm1)
                f2 = _exact_c5(hm2)
                if f1 < f2:
                    hi_a = m2
                else:
                    lo_a = m1
            a_opt = (lo_a + hi_a) / 2.0
            h_blend = _project(best_h * (1 - a_opt) + z * a_opt, best_n / 2.0)
            c5_blend = _exact_c5(h_blend)
            if c5_blend < best_c5:
                best_c5 = c5_blend; best_h = h_blend.copy()
            # Polish the best
            if time.time() < dl5:
                pol_dl = min(dl5, time.time() + budget_s * 0.01)
                hp, cp = _local_optimize_lp(best_h, pol_dl, radius_init=0.06, max_iters=800, stall_limit=60)
                if cp < best_c5:
                    best_c5 = cp; best_h = hp.copy()

    # Phase 6: Perturbation restarts
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.90
        scale = 0.015
        for attempt in range(300):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.008)
            ht, ct = _local_optimize_lp(h_pert, sub_dl, radius_init=0.05, max_iters=600, stall_limit=50)
            if ct < best_c5 - 1e-14:
                best_c5 = ct; best_h = ht.copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.06, scale * 1.08)

    # Phase 7: Final BFGS polish with high beta
    if time.time() < t0 + budget_s * 0.94:
        dl7 = t0 + budget_s * 0.94
        h7, c5_7 = _bfgs_smooth(best_h, dl7, beta_init=100.0, beta_final=500.0, n_phases=3)
        if c5_7 < best_c5:
            best_c5 = c5_7; best_h = h7.copy()
        h7b, c5_7b = _local_optimize_lp(best_h, min(t0 + budget_s * 0.955, time.time() + budget_s * 0.015),
                                         radius_init=0.03, max_iters=3000, stall_limit=100)
        if c5_7b < best_c5:
            best_c5 = c5_7b; best_h = h7b.copy()

    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.975:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.978, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
