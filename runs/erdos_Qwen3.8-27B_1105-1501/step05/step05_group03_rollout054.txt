def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(n, deadline, rng, n_iters=30000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.04
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cool
    return best_h, best_c5

def _fourier_evolve(h, deadline, n_pop=16, K=10, n_gen=150, mutation_scale=0.08, rng=None):
    n = len(h)
    target = n / 2.0
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2 * (k - 1)] = np.cos(2 * np.pi * k * x)
        B[:, 2 * (k - 1) + 1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    a_init = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    pop_a = a_init + mutation_scale * rng.standard_normal((n_pop, 2 * K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
        h_try = _project(h_try, target)
        pop_c5[i] = _exact_c5(h_try)
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = np.clip(0.5 + B @ pop_a[best_i], 0, 1)
    best_h = _project(best_h, target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        n_keep = max(2, n_pop // 2)
        pop_a[:n_keep] = pop_a[order[:n_keep]]
        pop_c5[:n_keep] = pop_c5[order[:n_keep]]
        for i in range(n_keep, n_pop):
            p1 = order[rng.integers(0, n_keep)]
            p2 = order[rng.integers(0, n_keep)]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1.0 - mix) * pop_a[order[p2]]
            pop_a[i] += mutation_scale * rng.standard_normal(2 * K)
            h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
            h_try = _project(h_try, target)
            pop_c5[i] = _exact_c5(h_try)
        mutation_scale *= 0.995
        cur_best = int(np.argmin(pop_c5))
        if pop_c5[cur_best] < best_c5:
            best_c5 = pop_c5[cur_best]
            best_h = np.clip(0.5 + B @ pop_a[cur_best], 0, 1)
            best_h = _project(best_h, target)
    return best_h, best_c5

def _anneal_refine(h, deadline, n_iters=5000, rng=None):
    """Simulated annealing with continuous perturbations."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    if rng is None:
        rng = np.random.default_rng(999)
    T = 0.005
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Perturb a random subset
        n_pert = int(rng.integers(max(3, n // 10), n // 3))
        idx = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idx] += rng.normal(0, 0.05, n_pert)
        h_try = _project(h_try, target)
        new_c5 = _exact_c5(h_try)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_try
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        T *= cool
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP optimization from initial (30% budget)
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_full(h_start, dl1, radius_init=0.20, max_iters=5000, radius_min=0.001)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Top-k LP (more focused) (10%)
    dl2 = t0 + budget_s * 0.22
    h2, c5_2 = _lp_topk(best_h, dl2, k=5, radius_init=0.10, max_iters=5000, radius_min=0.001)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multiplicative weight subgradient (10%)
    dl3 = t0 + budget_s * 0.32
    h3, c5_3 = _mulw_subgrad(best_h, dl3, n_steps=500, eta0=1.5, step0=0.04)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Hard subgradient (focus on single worst lag) (8%)
    dl4 = t0 + budget_s * 0.40
    h4, c5_4 = _hard_subgrad(best_h, dl4, n_steps=400, step0=0.03)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Follow-up LP after subgradient phases (8%)
    dl5 = t0 + budget_s * 0.48
    h5, c5_5 = _lp_topk(best_h, dl5, k=3, radius_init=0.08, max_iters=4000, radius_min=0.001)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution exploration (17%)
    dl6 = t0 + budget_s * 0.65
    resolutions = [64, 80, 96, 112, 128, 144, 160, 192, 256]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5,
              0.6180339887, 0.2763932022, 0.7137976813, 0.5]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl6:
            break
        if best_c5 < 0.3806:
            break
        for s in range(2):
            if time.time() > dl6:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_periodic_init(n_try, alpha, rng)
            sub_dl = min(dl6, time.time() + budget_s * 0.025)
            h_try, c5_try = _lp_topk(h_init, sub_dl, k=5, radius_init=0.15, max_iters=2000, radius_min=0.003)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 7: Block-structured basin hopping (10%)
    dl7 = t0 + budget_s * 0.75
    h7, c5_7 = _block_hops(best_h, dl7, n_attempts=300, rng=rng)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Fourrier evolution + binary SA (12%)
    dl8 = t0 + budget_s * 0.82
    h_f, c5_f = _fourier_evolve(best_h, dl8, n_pop=16, K=10, n_gen=150, mutation_scale=0.08, rng=rng)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(best_h)
    # Binary SA on a couple resolutions
    dl8b = t0 + budget_s * 0.87
    for n_bin in [96, 128]:
        if time.time() > dl8b:
            break
        sub_dl = min(dl8b, time.time() + budget_s * 0.02)
        h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=30000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_bin
            sub_dl2 = min(dl8b, time.time() + budget_s * 0.01)
            h_br, c5_br = _lp_topk(h_b, sub_dl2, k=5, radius_init=0.10, max_iters=1000, radius_min=0.005)
            if c5_br < best_c5:
                best_c5 = c5_br
                best_h = h_br.copy()

    # Phase 9: Anneal refine (5%)
    dl9 = t0 + budget_s * 0.92
    h9, c5_9 = _anneal_refine(best_h, dl9, n_iters=5000, rng=rng)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    # Phase 10: Final polish - LP + coordinate + pair (8%)
    dl10 = t0 + budget_s * 0.95
    h10, c5_10 = _lp_topk(best_h, dl10, k=3, radius_init=0.05, max_iters=3000, radius_min=0.0005)
    if c5_10 < best_c5:
        best_c5 = c5_10
        best_h = h10.copy()

    dl11 = t0 + budget_s * 0.975
    hc, c5c = _coordinate_polish(best_h, dl11, n_cycles=2)
    if c5c < best_c5:
        best_c5 = c5c
        best_h = hc.copy()

    dl12 = t0 + budget_s * 0.995
    hp, c5p = _pair_polish(best_h, dl12, n_passes=2)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Final validity check
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
