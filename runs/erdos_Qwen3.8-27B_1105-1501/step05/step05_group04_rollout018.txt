def project_box_sum(v, target):

def lag_jac_row(s, h):
    """Jacobian row for lag s (offset from n-1 in full correlation)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    # s ranges from -(n-1) to n-1
    # For s >= 0: F_s = dx * sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
    # For s < 0, let a = -s: F_s = dx * sum_{i=0}^{n-1-a} h[i]*(1-h[i+a])
    if s >= 0:
        m = n - s
        if m > 0:
            # dF/dh[i] for i in [0, m): -dx*h[i+s] (from (1-h[i]) term)
            # dF/dh[j] for j in [s, n): +dx*(1-h[j-s]) (from h[j] term)
            g[:m] += -dx * h[s:s+m]  # from (1-h[i]) where i in [0,m)
            g[s:] += dx * (1.0 - h[:m])  # from h[i+s] where i in [0,m)

            g[a:] += -dx * h[:m]
    return g

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)

def coord_refine(h, deadline, n_iter=500):

    target = n / 2.0

        return best_h, best_c5
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]

        return np.zeros(n)

    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for idx in range(n_lags):
        s = idx - (n - 1)

            m = n - s
            if m > 0:

                J[idx, s:] += dx * (1.0 - h[:m])
        else:

            m = n - a
            if m > 0:
                J[idx, :m] += dx * (1.0 - h[a:a + m])
                J[idx, a:] += -dx * h[:m]
    return J

        sub_dl = min(deadline, time.time() + 4.0)

        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def combined_refine(h, deadline):
    """Alternate between smoothing, LP, and coordinate refinement."""
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    global _SLSQP_N, _SLSQP_TARGET
    _SLSQP_N = n
    _SLSQP_TARGET = n / 2.0

    for cycle in range(3):
        if time.time() > deadline:
            break
        # LP step
        d1 = min(deadline, time.time() + (deadline - time.time()) * 0.3)
        h_lp, c5_lp = lp_loop(best_h, d1, radius_init=0.06)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

        # Smoothing step with small beta
        if time.time() > deadline:
            break
        beta = best_c5 * 0.01
        if beta < 1e-6:
            beta = 1e-6
        d2 = min(deadline, time.time() + (deadline - time.time()) * 0.3)

        def _so(x):
            v, _ = smooth_obj_grad(x, beta)
            return v

        def _so_g(x):
            v, g = smooth_obj_grad(x, beta)
            return v, g

        constraints = [{'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_so_g, best_h, method='SLSQP',
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 80, 'ftol': 1e-10, 'iprint': 0})
            if res.x is not None and len(res.x) == n:
                h_new = project_box_sum(np.asarray(res.x, dtype=float), n / 2.0)
                c_new = eval_c5(h_new)
                if c_new < best_c5 - 1e-12:
                    best_c5 = c_new
                    best_h = h_new.copy()
        except Exception:
            pass

        # Coordinate refinement
        if time.time() > deadline:
            break
        d3 = min(deadline, time.time() + 20.0)
        best_h, best_c5 = coord_refine(best_h, d3, n_iter=200)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s - 5.0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: Initial LP (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = lp_loop(h, d1, radius_init=0.12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothing with SLSQP (17%)
    d2 = t0 + budget_s * 0.25
    best_h, best_c5 = smooth_phase(best_h, d2)

    # Phase 3: LP after smoothing (10%)
    d3 = t0 + budget_s * 0.35
    h3, c5_3 = lp_loop(best_h, d3, radius_init=0.08)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Epigraph SLSQP (20%)
    d4 = t0 + budget_s * 0.55
    best_h, best_c5 = epigraph_phase(best_h, d4)

    # Phase 5: LP polish (10%)
    d5 = t0 + budget_s * 0.65
    h5, c5_5 = lp_loop(best_h, d5, radius_init=0.06)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Combined refinement cycle (15%)
    d6 = t0 + budget_s * 0.80
    best_h, best_c5 = combined_refine(best_h, d6)

    # Phase 7: Perturbation search (7%)
    d7 = t0 + budget_s * 0.87
    best_h, best_c5 = perturb_search(best_h, d7, rng, base_scale=0.04)

    # Phase 8: Multi-resolution (4%)
    d8 = t0 + budget_s * 0.91
    best_h, best_c5 = multiresearch(best_h, d8, rng)

    # Phase 9: Coordinate refinement (3%)
    d9 = t0 + budget_s * 0.94
    best_h, best_c5 = coord_refine(best_h, d9, n_iter=400)

    # Phase 10: Final LP polish (4%)
    d10 = t0 + budget_s * 0.98
    h10, c5_10 = lp_loop(best_h, d10, radius_init=0.04)
    if c5_10 < best_c5:
        best_c5 = c5_10
        best_h = h10.copy()

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
