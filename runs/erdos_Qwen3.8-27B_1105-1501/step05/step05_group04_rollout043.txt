        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp((c - c_max) / beta)
    w_s = exp_c.sum()
    lse = beta * (np.log(w_s) + c_max / beta)
    w = exp_c / w_s
    g = np.zeros(n)
    for j in range(n):
        g[j] = -2.0 * dx * np.dot(w[j:j+n], h)
    return lse, g, w, c

def _prox_grad(h, deadline, beta, use_momentum=True, mu=0.85, n_max=200000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = 5.0
    stall = 0
    t0 = time.time()
    for it in range(n_max):
        if time.time() > deadline:
            break
        lse, g, w, c = _grad_smooth(h, beta)
        if use_momentum:
            v = mu * v + (1.0 - mu) * g
            direction = v
        else:
            direction = g
        ng = np.linalg.norm(direction)
        if ng < 1e-15:
            break
        moved = False
        for bt in range(30):
            h_try = _project(h - lr * direction, target)
            lse_try, _, _, _ = _grad_smooth(h_try, beta)
            if lse_try < lse - 1e-15:
                moved = True
                break
            lr *= 0.5
        if not moved:
            stall += 1
            lr = 5.0
            if stall > 300:
                break
            continue
        h = h_try
        lr = min(lr * 1.4, 100.0)
        stall = 0
        if it % 30 == 0:
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
    cur = _exact_c5(h)
    if cur < best_c5:
        best_c5 = cur
        best_h = h.copy()
    return best_h, best_c5

def _pair_cd(h, deadline, n_iter=100000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        beta = max(dx * 0.05, 1e-4)
        w = np.exp((c - c_max) / beta)
        w /= w.sum()
        A = np.zeros(n)
        for j in range(n):
            A[j] = np.dot(w[j:j+n], h)
        i_best = int(np.argmax(A))
        j_best = int(np.argmin(A))
        if i_best == j_best or A[i_best] - A[j_best] < 1e-12:
            break
        d = np.zeros(n)
        d[i_best] = 1.0
        d[j_best] = -1.0
        t_hi = min(1.0 - h[i_best], h[j_best])
        if t_hi < 1e-10:
            break
        n_grid = 25
        ts = np.linspace(0.0, t_hi, n_grid)
        best_t = 0.0
        best_val = best_c5
        for t in ts:
            h_try = _project(h + t * d, target)
            val = _exact_c5(h_try)
            if val < best_val:
                best_val = val
                best_t = t
        if best_t > 0 and best_val < best_c5 - 1e-14:
            h = _project(h + best_t * d, target)
            best_c5 = best_val
            best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_polish(h, deadline, max_iters=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = 0.05
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(scores)
        active = scores >= c_max - 0.02 * dx * n
        n_active = len(active)
        if n_active == 0:
            active = np.zeros(2*n-1, dtype=bool)
            top3 = np.argsort(scores)[-3:]
            active[top3] = True
            n_active = 3
        # Build small LP: minimize t s.t. G_active * d <= t - scores_active, sum(d)=0, |d|<=radius
        n_lags = 2 * n - 1
        G = np.zeros((n_active, n))
        for idx, r in enumerate(np.where(active)[0]):
            lo = max(0, r - n + 1)
            hi = min(n - 1, r)
            for j in range(lo, hi + 1):
                G[idx, j] = -2.0 * h[r - j]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((n_active, n + 1))
        A_ub[:, :n] = G * dx
        A_ub[:, n] = -1.0
        b_ub = -scores[active] * dx
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x[-1] < c_max * dx - 1e-12:
                p = res.x[:n]
                # Line search
                S0 = np.correlate(h, 1.0 - h, mode='full')
                T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
                C = np.correlate(p, p, mode='full')
                pos = p > 1e-12
                neg = p < -1e-12
                t_hi = radius
                if np.any(pos):
                    t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
                t_lo = -radius
                if np.any(neg):
                    t_lo = max(t_lo, float(np.max(h[neg] / p[neg])))
                # Golden section
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = t_lo, t_hi
                best_t = 0.0
                best_v = _exact_c5(h)
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
                for _ in range(40):
                    if b - a < 1e-13:
                        break
                    if f1 < f2:
                        b = c2
                        c2 = c1
                        f2 = f1
                        c1 = b - gr * (b - a)
                        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
                    else:
                        a = c1
                        c1 = c2
                        f1 = f2
                        c2 = a + gr * (b - a)
                        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
                opt_t = (a + b) / 2.0
                opt_v = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
                if opt_v < best_v:
                    best_t = opt_t
                    best_v = opt_v
                if best_v < best_c5 - 1e-13:
                    h = _project(h + best_t * p, target)
                    best_c5 = _exact_c5(h)
                    best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.1, 0.15)
                else:
                    stall += 1
                    radius *= 0.8
            else:
                stall += 1
                radius *= 0.85
        except Exception:
            stall += 1
            radius *= 0.85
        if stall > 50:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Multi-beta smoothed proximal gradient
    d1 = t0 + budget_s * 0.30
    betas = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002]
    for b in betas:
        sub_dl = min(d1, time.time() + (d1 - time.time()) / len(betas))
        h1, c5_1 = _prox_grad(h, sub_dl, beta=b, use_momentum=True, mu=0.85)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            h = h1.copy()

    # Phase 2: Pair coordinate descent
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _pair_cd(h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        h = h2.copy()

    # Phase 3: LP polish
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _lp_polish(h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        h = h3.copy()

    # Phase 4: Multi-resolution exploration
    d4 = t0 + budget_s * 0.80
    tried = set()
    for dn in range(-5, 15):
        if time.time() > d4:
            break
        n_try = n0 + dn
        if n_try < 30 or n_try > 180 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        # Quick proximal gradient on resampled
        sub_dl = min(d4, time.time() + 3.0)
        h_r, c5_r = _prox_grad(h_r, sub_dl, beta=0.005, use_momentum=True, mu=0.85)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Final polishing with multiple methods
    d5 = t0 + budget_s * 0.95
    h5, c5_5 = _pair_cd(best_h, min(d5, t0 + budget_s * 0.87))
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    h6, c5_6 = _lp_polish(best_h, d5)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final: one more prox grad pass with very small beta
    rem = t0 + budget_s - time.time()
    if rem > 5:
        h7, c5_7 = _prox_grad(best_h, t0 + budget_s - 1, beta=0.001, use_momentum=True, mu=0.9)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
