    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _project_box_sum(v, target):

    n = len(v)

                best_h = h.copy()

def _fista_optimize(h_init, deadline, n_phases=12, beta_start=8.0, beta_end=600.0,
                    max_iter_per_phase=500, step_scale=3.0):
    n = len(h_init)
    target = n * 0.5
    h = _project(h_init, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = max(beta, 4.0)

        y = h.copy()
        t_k = 1.0
        step = step_scale / (1.0 + 0.3 * beta)

                break

                    accepted = True

def _lp_direction(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    G = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    G *= (2.0 / n)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None
