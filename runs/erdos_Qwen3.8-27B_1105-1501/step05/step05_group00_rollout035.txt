def _slsqp_epigraph(h_start, target, dx, n, t_init, max_iter=500, ftol=1e-10):
    from scipy.optimize import minimize
    
    def objective(x):
        t = x[-1]
        # Small regularization toward h_start (optional)
        return t
    
    def constraint_fun(x):
        t = x[-1]
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - dx * corr  # should be >= 0 for all
    
    def constraint_jac(x):
        t = x[-1]
        h = x[:n]
        G = _grad_matrix(h)  # shape (2n-1, n)
        jac = np.zeros((2*n-1, n+1))
        jac[:, :n] = -dx * G
        jac[:, n] = 1.0
        return jac
    
    constraints = [{'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    # Equality: sum(h) = target
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        j = np.zeros(n+1)
        j[:n] = 1.0
        return j
    
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    bounds = [(0, 1)] * n + [(0, 2.0)]
    
    x0 = np.zeros(n+1)
    x0[:n] = h_start
    x0[-1] = t_init
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': max_iter, 'ftol': ftol})
    
    if res.success:
        h_new = res.x[:n]
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        return h_new, c5_new

def _active_direction(h, target, dx, n, eta=0.5, K=50):
    """Find a direction that reduces the top K correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    
    # Top K lags (those within eta*max of the maximum)
    threshold = M - eta * M  # or M - some absolute threshold
    active = np.where(corr >= threshold)[0]
    active = active[:K]  # limit to K
    
    # For active lags, we want to reduce corr[r]
    # Direction: minimize max of G[r,:] · d for r in active
    # This is a small LP or can be approximated by ridge
    
    G_active = _grad_matrix(h)[active, :]  # shape (K, n)
    
    # We want: G_active @ d <= 0 (reduce all active components)
    # With: sum(d) = 0 (preserve constraint)
    #        ||d|| <= radius
    
    # Ridge approach: minimize ||G_active @ d + 1||^2 subject to sum(d)=0
    # Or: find d that minimizes the maximum of G_active @ d
    
    # Simplest: project -G_active.T @ ones onto the sum-zero subspace
    b = np.ones(K)
    d_raw = -G_active.T @ b  # direction that decreases all active correlations
    # Project to sum-zero
    d_raw = d_raw - np.mean(d_raw)
    # Also project to stay in bounds (approximately)
    d = d_raw / (np.max(np.abs(d_raw)) + 1e-12)
    
    return d

def _pair_refine(h, target, dx, n, deadline, max_passes=30):
    """Refine using pairs of interior coordinates."""
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[ii]
            for jj in range(ii+1, len(interior)):
                j = interior[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)

                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, _exact_c5(best_h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
