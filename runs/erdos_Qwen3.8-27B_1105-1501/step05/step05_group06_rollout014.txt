```python
import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_obj_grad(h, q):
    """Compute p-norm surrogate value and gradient."""
    n = len(h)
    F = _scores(h)  # length 2n-1, all >= 0
    M = float(F.max())
    if M < 1e-300:
        return 0.0, np.zeros(n)
    u = F / M  # in [0, 1]
    w = u ** (q - 1.0)  # in [0, 1]
    S = float(np.sum(u * w))  # = sum(u^q)
    if S < 1e-300:
        # Fallback: just use the max
        rmax = int(np.argmax(F))
        G = _build_G(h)
        return M, G[rmax]
    val = M * S ** (1.0 / q)
    G = _build_G(h)
    grad = S ** (1.0 / q - 1.0) * (G.T @ w)
    return val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_val_grad(h, q):
    n = len(h)
    F = _scores(h)
    M = float(F.max())
    if M < 1e-300:
        return 0.0, np.zeros(n)
    u = F / M
    if q <= 1.5:
        w = np.ones_like(u)
        S = float(np.sum(u))
    else:
        w = u ** (q - 1.0)
        S = float(np.sum(u * w))
    if S < 1e-300:
        rmax = int(np.argmax(F))
        G = _build_G(h)
        return M, G[rmax].copy()
    val = M * S ** (1.0 / q)
    G = _build_G(h)
    grad = S ** (1.0 / q - 1.0) * (G.T @ w)
    return val, grad

def _obj_func(h, q):
    val, grad = _pnorm_val_grad(h, q)
    return val, grad

def _mass_con_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def _mass_con_jac(h):
    return np.ones(len(h))

def _slsqp_run(h0, q, maxiter=200, ftol=1e-14):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _mass_con_fun, 'jac': _mass_con_jac}]
    try:
        res = minimize(_obj_func, h0.copy(), args=(q,), method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = _project(res.x, target)
            c5 = _exact_c5(h_new)
            if np.all(np.isfinite(h_new)) and c5 < 1.0:
                return h_new, c5
    except Exception:
        pass
    return h0.copy(), _exact_c5(h0)

def _coord_descent(h, deadline, max_sweeps=50, verify_thresh=1e-10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _build_G(h)
    scores = _scores(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            delta = G[:, j]
            # F_r -> F_r + a * delta_r, where a is the step
            # We want to minimize max_r (scores_r + a * delta_r)
            # Subject to 0 <= h[j]+a <= 1
            lo_a = max(-h[j], 0.0) - 1.0
            hi_a = min(1.0 - h[j], 0.0) + 1.0
            lo_a = max(-h[j], lo_a)
            hi_a = min(1.0 - h[j], hi_a)
            if hi_a - lo_a < 1e-14:
                continue
            lo, hi = lo_a, hi_a
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = float(np.max(scores + m1 * delta))
                f2 = float(np.max(scores + m2 * delta))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            a = (lo + hi) / 2.0
            if cur_c5 - float(np.max(scores + a * delta)) < verify_thresh:
                continue
            h[j] += a
            # Re-project
            h_proj = _project(h, target)
            actual = _exact_c5(h_proj)
            if actual < cur_c5 - 1e-14:
                h = h_proj
                cur_c5 = actual
                scores = _scores(h)
                G = _build_G(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[j] -= a
        if not improved:

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_h(h, width=2):
    n = len(h)
    kernel = np.ones(width) / width
    h_pad = np.concatenate([h[-width:], h, h[:width]])
    h_sm = np.convolve(h_pad, kernel, mode='same')[width:-width]
    return _project(h_sm, n / 2.0)

def _binarize(h, rng):
    """Push toward binary, keeping mass."""
    n = len(h)
    target = int(n // 2)
    frac = h - np.floor(h)
    # Sort by h value, keep top `target` as 1
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    if rng is not None and rng.random() < 0.3:
        # Swap a few randomly
        for _ in range(3):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin

        best_c5 = c5_pd
        best_h = h_pd.copy()

    # Phase 4: Multi-resolution SLSQP search (30% time)
    dl4 = t0 + budget_s * 0.60
    resolutions = [128, 192, 256, 160, 128]
    q_sched_multi = [8, 32, 128, 256]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5555]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl4:
            break
        for s in range(2):
            if time.time() > dl4:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                h_init = _quasi_init(n_try, alphas[ri % len(alphas)], rng)
            sub_dl = min(dl4, time.time() + budget_s * 0.05)
            h_try, c5_try, _ = _continue_q(h_init, best_h, best_c5,
                                            q_sched_multi, sub_dl)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 5: Intensive pair descent on current best (15% time)
    dl5 = t0 + budget_s * 0.75
    h_pd2, c5_pd2 = _pair_descent(best_h, dl5, max_passes=300, verify_thresh=1e-10)
    if c5_pd2 < best_c5:
        best_c5 = c5_pd2
        best_h = h_pd2.copy()
    # Follow with small SLSQP
    if time.time() < dl5 - 5:
        sub_dl = min(dl5, time.time() + budget_s * 0.04)
        for q in [64, 128, 256]:
            if time.time() > sub_dl:
                break
            h_ls, c5_ls = _slsqp_run(best_h, q, maxiter=150)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 6: Perturbation restarts (15% time)
    dl6 = t0 + budget_s * 0.90
    scale = 0.02
    n_pert = best_n
    for attempt in range(500):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, n_pert)
        h_pert = _project(h_pert, n_pert / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        if sub_dl - time.time() < 2:
            break
        # Quick SLSQP with high q
        h_ps, c5_ps = _slsqp_run(h_pert, 128, maxiter=80)
        if c5_ps < best_c5 - 1e-13:
            best_c5 = c5_ps
            best_h = h_ps.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.05)
        # Try pair descent on perturbation
        if time.time() < dl6 - 5:
            sub_dl2 = min(dl6, time.time() + budget_s * 0.008)
            h_pd3, c5_pd3 = _pair_descent(h_ps, sub_dl2, max_passes=30, verify_thresh=1e-10)
            if c5_pd3 < best_c5:
                best_c5 = c5_pd3
                best_h = h_pd3.copy()

    # Phase 7: Final polish (remaining time)
    dl7 = hard_deadline
    if time.time() < dl7 - 5:
        h_final_cd, c5_fc = _coord_descent(best_h, dl7, max_sweeps=100, verify_thresh=1e-12)
        if c5_fc < best_c5:
            best_c5 = c5_fc
            best_h = h_final_cd.copy()
    if time.time() < dl7 - 3:
        h_final_pd, c5_fp = _pair_descent(best_h, dl7, max_passes=200, verify_thresh=1e-12)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_final_pd.copy()

    # Final projection and verification
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    # Safety: if final is worse than tracked best (shouldn't happen), use tracked
    if final_c5 > best_c5 + 1e-12:
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = min(final_c5, best_c5)

    return best_h, float(final_c5), int(len(best_h))
```
