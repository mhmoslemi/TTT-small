    if abs(r) > 1e-10:

        else: mask = x > 1e-12

    xn = (np.arange(n_new) + 0.5) / n_new

def _lp_optimize(h, deadline, radius=0.2, max_it=3000, topk=25):
    n = len(h)
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best_c5 = cur
    for _ in range(max_it):
        if time.time() > deadline: break
        sc = _scores(h)
        G = _grad(h)
        if topk < 2*n-1:
            idx = np.argsort(sc)[-topk:]
            Gk, sk = G[idx], sc[idx]
        else:
            Gk, sk = G, sc
        c = np.zeros(n+1); c[-1] = 1.0
        A_ub = np.column_stack([Gk, -np.ones(len(sk))])
        b_ub = -sk
        A_eq = np.zeros((1, n+1)); A_eq[0,:n] = 1.0
        bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                         bounds=bounds, method='highs')
            if not res.success: continue
            p = res.x[:n]
        except: continue
        if np.max(np.abs(p)) < 1e-14: break
        # Line search
        amax = 0.5
        for j in range(n):
            if p[j] > 1e-14: amax = min(amax, (1-h[j])/p[j])
            elif p[j] < -1e-14: amax = min(amax, h[j]/(-p[j]))
        amax = max(amax, 1e-12)
        # Simple line search
        best_a, best_f = 0.0, cur
        for a in np.linspace(0, amax, 15):
            f = _c5(h + a*p)
            if f < best_f: best_f, best_a = f, a
        if best_f < cur - 1e-13:
            h = _project(h + best_a*p, n/2.0)
            cur = _c5(h)
            if cur < best_c5: best_c5, best_h = cur, h.copy()
            radius = min(radius*1.02, 0.3)
        else:
            radius *= 0.7
            if radius < 1e-6: break
    return best_h, best_c5

def _grad(h):
    """Gradient of bottleneck-weighted score."""
    n = len(h)
    dx = 2.0/n
    sc = np.correlate(h, 1.0-h, mode='full') * dx
    # Weight by score (bottleneck-aware)
    smax = np.max(sc)
    w = np.exp(5.0 * (sc - smax))  # softmax weighting
    w /= np.sum(w)
    # Compute gradient
    G = np.zeros(n)
    for d in range(-(n-1), n):
        i = n-1+d
        wd = w[i]
        if abs(wd) < 1e-15: continue
        m = n - abs(d)
        if d >= 0:
            G[:m] += dx * wd * (1.0 - h[d:d+m])
            G[d:d+m] -= dx * wd * h[:m]
        else:
            g = -d
            G[:m] += dx * wd * (1.0 - h[g:g+m])
            G[g:g+m] -= dx * wd * h[:m]
    G -= G.mean()  # preserve sum
    return G

def _bin_sa(x, deadline, rng, n_iters=50000, use_2swap=False, use_block=False):
    n = len(x)
    cur = _c5(x.astype(float))
    best_x = x.copy()
    best_c5 = cur
    ones = list(np.where(x == 1)[0])
    zeros = list(np.where(x == 0)[0])
    T = 0.03
    Tmin = 0.002
    cool = (Tmin/T)**(1.0/max(n_iters,1))
    it = 0
    while it < n_iters:
        if time.time() > deadline: break
        move_type = rng.random()
        if use_block and move_type < 0.15:
            # Block move: pick a block and rotate it
            blen = rng.integers(2, min(n//3, 20))
            start = rng.integers(0, n - blen)
            # Check if block has at least one 1 and one 0
            block = x[start:start+blen]
            if block.sum() == 0 or block.sum() == blen:
                it += 1
                T *= cool
                continue
            # Try all rotations
            best_rot = 0
            best_f = _c5(x.astype(float))
            for rot in range(1, blen):
                xp = x.copy()
                xp[start:start+blen] = np.roll(block, -rot)
                f = _c5(xp.astype(float))
                if f < best_f:
                    best_f = f
                    best_rot = rot
            if best_rot > 0:
                x[start:start+blen] = np.roll(block, -best_rot)
                cur = best_f
                if cur < best_c5: best_c5, best_x = cur, x.copy()
                it += 1
                T *= cool
                continue
            it += 1
            T *= cool
            continue
        elif use_2swap and move_type < 0.35:
            # 2-swap: swap two 1s with two 0s
            if len(ones) < 2 or len(zeros) < 2:
                it += 1; T *= cool; continue
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            x[i1] = 0; x[i2] = 0
            x[j1] = 1; x[j2] = 1
            new_c5 = _c5(x.astype(float))
            if new_c5 < cur or rng.random() < np.exp((cur-new_c5)/T):
                cur = new_c5
                ones = list(np.where(x==1)[0])
                zeros = list(np.where(x==0)[0])
                if cur < best_c5: best_c5, best_x = cur, x.copy()
            else:
                x[i1] = 1; x[i2] = 1
                x[j1] = 0; x[j2] = 0
        else:
            # 1-swap
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            x[i] = 0; x[j] = 1
            new_c5 = _c5(x.astype(float))
            if new_c5 < cur or rng.random() < np.exp((cur-new_c5)/T):
                cur = new_c5
                ones = list(np.where(x==1)[0])
                zeros = list(np.where(x==0)[0])
                if cur < best_c5: best_c5, best_x = cur, x.copy()
            else:
                x[i] = 1; x[j] = 0
        it += 1
        T *= cool
    return best_x, best_c5

def _cont_optimize(h, deadline):
    """Bottleneck-aware continuous optimization."""
    n = len(h)
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best_c5 = cur
    for it in range(2000):

            cur = _c5(h)
            if cur < best_c5: best_c5, best_h = cur, h.copy()
        else:
            break
    return best_h, best_c5

def _quasi(n, alpha, beta=0.5):
    x = np.arange(n) / n * 2.0  # in [0, 2)
    h = ((x * alpha) % 1.0) < beta
    return _project(h.astype(float), n/2.0)

        r = rng.random()
        if use_block and r < 0.12 and n >= 24:
            blen = int(rng.integers(2, min(8, n//4+1)))
            start = int(rng.integers(0, n - blen))
            block = x[start:start+blen].copy()
            bs = int(block.sum())
            if bs == 0 or bs == blen:
                T *= cool; continue
            best_rot, best_f = 0, cur
            for rot in range(1, blen):
                xp = x.copy()
                xp[start:start+blen] = np.roll(block, -rot)
                f = _c5(xp.astype(float))
                if f < best_f: best_f, best_rot = f, rot
            if best_rot > 0:
                x[start:start+blen] = np.roll(block, -best_rot)
                cur = best_f
                if cur < best_c5: best_c5, best_x = cur, x.copy()
                ones = np.where(x==1)[0].tolist()
                zeros = np.where(x==0)[0].tolist()
            T *= cool
        elif use_2swap and r < 0.35 and len(ones)>=2 and len(zeros)>=2:
            i1, i2 = int(rng.integers(len(ones))), int(rng.integers(len(ones)))
            while i2 == i1: i2 = int(rng.integers(len(ones)))
            j1, j2 = int(rng.integers(len(zeros))), int(rng.integers(len(zeros)))
            while j2 == j1: j2 = int(rng.integers(len(zeros)))
            a, b = ones[i1], ones[i2]
            c2, d2 = zeros[j1], zeros[j2]
            x[a]=0; x[b]=0; x[c2]=1; x[d2]=1
            new_f = _c5(x.astype(float))
            if new_f < cur or rng.random() < np.exp((cur-new_f)/T):
                cur = new_f
                ones = np.where(x==1)[0].tolist()
                zeros = np.where(x==0)[0].tolist()
                if cur < best_c5: best_c5, best_x = cur, x.copy()
            else:
                x[a]=1; x[b]=1; x[c2]=0; x[d2]=0
            T *= cool
        else:
            if not ones or not zeros: break
            i = ones[int(rng.integers(len(ones)))]
            j = zeros[int(rng.integers(len(zeros)))]
            x[i]=0; x[j]=1
            new_f = _c5(x.astype(float))
            if new_f < cur or rng.random() < np.exp((cur-new_f)/T):
                cur = new_f
                ones = np.where(x==1)[0].tolist()
                zeros = np.where(x==0)[0].tolist()
                if cur < best_c5: best_c5, best_x = cur, x.copy()
            else:
                x[i]=1; x[j]=0
            T *= cool
    return best_x, best_c5

def _bin_sa_restarts(h_start, deadline, rng, n_restarts=5, n_per=20000):
    n = len(h_start)
    if n % 2 != 0: return None, _c5(h_start)
    k = n // 2
    best_x, best_c5 = None, _c5(h_start)
    for r in range(n_restarts):
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (n_restarts - r))
        if r == 0:
            idx = np.argsort(h_start)[::-1][:k]
            xb = np.zeros(n); xb[idx] = 1.0
        elif r == 1:
            xp = h_start + rng.normal(0, 0.15, n)
            xp = _project(xp, k)
            idx = np.argsort(xp)[::-1][:k]
            xb = np.zeros(n); xb[idx] = 1.0
        else:
            perm = rng.permutation(n)
            xb = np.zeros(n); xb[perm[:k]] = 1.0
        xo, co = _bin_sa(xb, sub_dl, rng, n_iter=n_per, use_2swap=True, use_block=True)
        if co < best_c5: best_c5, best_x = co, xo.copy()
    return best_x, best_c5

def _quasi(n, alpha, beta=0.5):
    x = np.arange(n) * (2.0/n)
    h = ((x * alpha) % 1.0) < beta
    return _project(h.astype(float), n/2.0)

def _sturmian(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (int((i+1)*alpha) - int(i*alpha)) == 1 else 0.0
    s = int(h.sum())
    if s != n//2:
        if s > n//2:
            idx = np.argsort(-h)[:s - n//2]
            h[idx] = 0.0
        else:
            idx = np.argsort(-h)[s:]
            need = n//2 - s
            h[idx[:need]] = 1.0
    return _project(h, n/2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0/2.0)
    best_h = h.copy()
    best_c5 = _c5(h)
    
    # Phase 1: LP from parent (8%)
    dl1 = t0 + budget_s * 0.08
    h1, c51 = _lp_optimize(h, dl1, radius=0.25, max_it=2000, topk=30)
    if c51 < best_c5: best_c5, best_h = c51, h1
    h1, c51 = _cont_grad_opt(best_h, dl1)
    if c51 < best_c5: best_c5, best_h = c51, h1
    
    # Phase 2: Binary SA multi-restart (35%)
    dl2 = t0 + budget_s * 0.43
    bx, bc = _bin_sa_restarts(best_h, dl2, rng, n_restarts=4, n_per=25000)
    if bx is not None and bc < best_c5:
        best_c5, best_h = bc, _project(bx, len(bx)/2.0)
        # Polish the binary result
        sub_dl = min(dl2, time.time() + budget_s*0.02)
        hp, cp = _lp_optimize(best_h, sub_dl, radius=0.15, max_it=1000, topk=20)
        if cp < best_c5: best_c5, best_h = cp, hp
    
    # Phase 3: Continued binary SA on best (12%)
    dl3 = t0 + budget_s * 0.55
    if n0 % 2 == 0:
        k = n0 // 2
        idx = np.argsort(best_h)[::-1][:k]
        xb = np.zeros(n0); xb[idx] = 1.0
        xb, cb = _bin_sa(xb, dl3, rng, n_iter=50000, use_2swap=True, use_block=True)
        if cb < best_c5:
            best_c5 = cb
            best_h = _project(xb, n0/2.0)
            sub_dl = min(dl3, time.time() + budget_s*0.015)
            hp, cp = _lp_optimize(best_h, sub_dl, radius=0.1, max_it=800, topk=20)
            if cp < best_c5: best_c5, best_h = cp, hp
    
    # Phase 4: Multi-resolution (20%)
    dl4 = t0 + budget_s * 0.75
    tried_ns = set()
    for n_try in [n0, n0+16, n0-16, 96, 112, 144, 160, 192, 80]:
        if time.time() > dl4: break
        if n_try < 32 or n_try > 256 or n_try % 2 != 0: continue
        if n_try in tried_ns: continue
        tried_ns.add(n_try)
        hr = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s*0.02)
        ho, co = _lp_optimize(hr, sub_dl, radius=0.15, max_it=1000, topk=20)
        if co < best_c5:
            best_c5, best_h = co, ho
            # Binary SA at this res
            if n_try % 2 == 0:
                sub_dl2 = min(dl4, time.time() + budget_s*0.015)
                k2 = n_try//2
                idx2 = np.argsort(best_h)[::-1][:k2]
                xb2 = np.zeros(n_try); xb2[idx2] = 1.0
                xb2, cb2 = _bin_sa(xb2, sub_dl2, rng, n_iter=20000, use_2swap=True, use_block=True)
                if cb2 < best_c5:
                    best_c5 = cb2
                    best_h = _project(xb2, n_try/2.0)
                    sub_dl3 = min(dl4, time.time() + budget_s*0.008)
                    hp3, cp3 = _lp_optimize(best_h, sub_dl3, radius=0.08, max_it=500, topk=15)
                    if cp3 < best_c5: best_c5, best_h = cp3, hp3
    
    # Phase 5: Diverse starts (10%)
    dl5 = t0 + budget_s * 0.85
    starts = []
    for a in [np.sqrt(5)-2, 0.382, 0.45, 0.35, 0.618-0.2]:
        if time.time() > dl5: break
        for nq in [128, 160]:
            if time.time() > dl5: break
            starts.append(_quasi(nq, a))
    for st in starts:
        if time.time() > dl5: break
        sub_dl = min(dl5, time.time() + budget_s*0.012)
        sq_o, cq_o = _lp_optimize(st, sub_dl, radius=0.2, max_it=600, topk=20)
        if cq_o < best_c5:
            best_c5, best_h = cq_o, sq_o
    
    # Phase 6: Final polish (10%)
    dl6 = t0 + budget_s * 0.96
    hp, cp = _lp_optimize(best_h, dl6, radius=0.06, max_it=2000, topk=15)
    if cp < best_c5: best_c5, best_h = cp, hp
    hp, cp = _cont_grad_opt(best_h, dl6)
    if cp < best_c5: best_c5, best_h = cp, hp
    
    # Final
    nf = len(best_h)
    best_h = _project(best_h, nf/2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, nf
