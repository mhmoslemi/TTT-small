def _lse_gradient(h, tau):

    M = float(np.max(scores))
    z = np.exp((scores - M) / tau)
    z_sum = float(np.sum(z))
    w = z / z_sum  # weights summing to 1
    
    g = np.zeros(n)
    for r in range(2 * n - 1):

        if q >= 0:
            m = n - q
            if m > 0:
                g[q:] += w[r] * (1.0 - h[:m])
                g[:m] -= w[r] * h[q:]
        else:

            m = n - gs
            if m > 0:
                g[:m] += w[r] * (1.0 - h[gs:])
                g[gs:] -= w[r] * h[:m]
