def _project(v, target):

    n = len(v)

        return np.zeros(n)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _pnorm_value(h, q):
    scores = _scores(h)
    M = float(np.max(scores))
    if M <= 1e-15:
        return 0.0
    u = np.clip(scores / M, 0.0, 1.0)
    return M * float(np.sum(u ** q)) ** (1.0 / q)

def _pnorm_grad(h, q):
    scores = _scores(h)
    M = float(np.max(scores))
    if M <= 1e-15:
        return np.zeros(len(h))
    u = np.clip(scores / M, 0.0, 1.0)
    uq = u ** q
    sum_uq = float(np.sum(uq))
    if sum_uq < 1e-300:
        return np.zeros(len(h))
    G = _build_G(h)
    weights = u ** (q - 1)
    grad = G.T @ weights
    grad *= sum_uq ** (-(q - 1.0) / q)
    return grad

def _pnorm_solve(h, q, deadline, maxiter=120):
    n = len(h)
    target = n / 2.0
    call_count = [0]

    def obj_fun(x):
        call_count[0] += 1
        if time.time() > deadline:
            return _pnorm_value(x, q)
        return _pnorm_value(x, q)

    def grad_fun(x):
        call_count[0] += 1
        if time.time() > deadline:
            return np.zeros(n)
        return _pnorm_grad(x, q)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj_fun, h.copy(), jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            h_new = _project(res.x, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h.copy(), _exact_c5(h)

def _pnorm_continuation(h, deadline, q_list, maxiter_per_q=100):

def _lp_direction(h, G, radius, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    scores = _scores(h)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    if best_i == 0:
        return 0.0, float(vals[0])
    if best_i == n_grid - 1:
        return alpha_max, float(vals[-1])
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)

def _pair_descent(h, deadline, max_passes=200):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_start(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def _smoothed_start(h, kernel_width, n_out):
    n = len(h)
    if n_out != n:
        h = _resample(h, n_out)
    kernel = np.exp(-0.5 * (np.arange(-kernel_width, kernel_width + 1) / max(1, kernel_width // 3))**2)
    kernel /= kernel.sum()
    h_smooth = np.convolve(h, kernel, mode='same')
    return _project(h_smooth, n_out / 2.0)

    if time.time() < hard_dl:
        h_fin, c5_fin = _pair_descent(best_h, hard_dl, max_passes=150)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if time.time() < hard_dl - 3:
            h_fin2, c5_fin2 = _pnorm_continuation(best_h, hard_dl, [256, 512], maxiter_per_q=40)
            if c5_fin2 < best_c5:
                best_c5 = c5_fin2
                best_h = h_fin2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
