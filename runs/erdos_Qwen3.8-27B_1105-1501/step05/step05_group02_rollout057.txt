    h[i] += t, h[j] -= t, t in [t_min, t_max].

    hj = h[j]

        elif d > 0:

            # d(c)/d(h[i]) for i in [0, m-1]: (1-h[i+d])
            # d(c)/d(h[i+d]) for i in [0, m-1]: -h[i]

            #   if j in [d, n-1]: contrib += -h[j-d]

def _direction_line_search(h, direction, n_grid=21):
    """Line search along a direction, projecting to stay feasible."""
    n = len(h)
    target = n / 2.0
    # Find max alpha such that h + alpha*d stays in [0,1]
    alphas = []
    for j in range(n):
        if direction[j] > 1e-14:
            alphas.append((1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alphas.append(h[j] / (-direction[j]))
    alpha_max = min(alphas) if alphas else 1.0
    alpha_max = max(alpha_max, 1e-14)
    
    ts = np.linspace(0, alpha_max, n_grid)
    best_a = 0.0
    best_c5 = _exact_c5(h)
    for t in ts[1:]:
        h_new = h + t * direction
        h_new = _project(h_new, target)

            best_a = t

        h_new = _project(h_new, target)
        return h_new, best_c5
    return h, best_c5

def _binary_local_search(h, deadline, rng, max_iters=50000):

def _resample(h, n_new):

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h, rng):

    n = len(h)

    h_bin = (h > 0.5).astype(float)
    cnt = int(np.sum(h_bin))
    if cnt > target:
        ones = np.where(h_bin == 1.0)[0]
        rng.shuffle(ones)
        h_bin[ones[:cnt - target]] = 0.0
    elif cnt < target:
        zeros = np.where(h_bin == 0.0)[0]
        rng.shuffle(zeros)
        h_bin[zeros[:target - cnt]] = 1.0
    return h_bin

def _pair_line_search(h, i, j):
    """Exact line search: h[i]+=t, h[j]-=t. Returns (t, c5)."""
    hp = h[i]
    hj = h[j]
    t_min = max(-hp, hj - 1.0)
    t_max = min(hj, 1.0 - hp)
    if t_max - t_min < 1e-14:
        return 0.0, _exact_c5(h)

    n_pts = 9
    best_t = 0.0

    for k in range(1, n_pts):
        t = t_min + (t_max - t_min) * k / n_pts
        h[i] = hp + t
        h[j] = hj - t
        v = _exact_c5(h)
        if v < best_val:
            best_val = v
            best_t = t
    # Golden section around best
    span = (t_max - t_min) / n_pts
    lo = max(t_min, best_t - span)
    hi = min(t_max, best_t + span)
    if hi - lo > 1e-12:

        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h[i] = hp + c1
        h[j] = hj - c1
        f1 = _exact_c5(h)
        h[i] = hp + c2
        h[j] = hj - c2
        f2 = _exact_c5(h)
        for _ in range(12):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                h[i] = hp + c1
                h[j] = hj - c1
                f1 = _exact_c5(h)
            else:

                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)

                f2 = _exact_c5(h)

        h[j] = hj - opt_t
        opt_val = _exact_c5(h)
        if opt_val < best_val:
            best_val = opt_val
            best_t = opt_t

    h[j] = hj - best_t
    return best_t, best_val

def _grad_weighted(h, tau_frac=0.01):
    """Weighted gradient using softmax over scores."""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = float(np.max(scores))
    tau = max(max_s * tau_frac, 1e-12)
    w = np.exp((scores - max_s) / tau)
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        wr = w[r] * dx
        if d == 0:
            grad += wr * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += wr * (1.0 - h[d:])
            grad[d:] += wr * (-h[:m])
        else:
            g = -d
            m = n - g
            grad[:m] += wr * (-h[g:])
            grad[g:] += wr * (1.0 - h[:m])
    return grad

def _direction_search(h, direction, n_grid=13):
    """Line search along direction with projection."""
    n = len(h)
    target = n / 2.0
    alpha_max = 1.0
    for j in range(n):

            alpha_max = min(alpha_max, h[j] / (-direction[j]))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_binary(n, alpha, rng):
    """Create quasi-periodic binary pattern."""
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    cnt = int(np.sum(h))
    target = n // 2
    if cnt > target:
        ones = np.where(h == 1.0)[0]
        rng.shuffle(ones)
        h[ones[:cnt - target]] = 0.0
    elif cnt < target:
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - cnt]] = 1.0
    return h

        # Also include second and third worst lags
        idx_sorted = np.argsort(scores)[-4:]

            elif d2 > 0:

                g2 = -d2

                c2[:m2] += (-h[g2:])

            c += c2
        c *= 2.0 / n

        # Bounded direction

        for j in range(n):
