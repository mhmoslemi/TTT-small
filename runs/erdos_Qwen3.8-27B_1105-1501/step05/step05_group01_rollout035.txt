import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_hybrid(h, d, dx, radius, n_grid=30):
    # Hybrid line search: coarse grid + local golden
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, -float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Coarse grid
    t_grid = np.linspace(t_lo, t_hi, n_grid)
    f_grid = np.array([float(np.max(S0 + t * T - t*t * C)) * dx for t in t_grid])
    
    # Find best grid point and neighbors
    best_idx = int(np.argmin(f_grid))
    
    # Local golden section around best region
    lo_idx = max(0, best_idx - 1)
    hi_idx = min(n_grid - 1, best_idx + 1)
    t_lo_g = t_grid[lo_idx]
    t_hi_g = t_grid[hi_idx]
    
    if t_hi_g - t_lo_g < 1e-13:
        opt_t = t_grid[best_idx]
        opt_val = f_grid[best_idx]
    else:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a = t_lo_g
        b = t_hi_g
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
        for _ in range(30):
            if b - a < 1e-13:
                break
            if f1 < f2:
                b = c2; c2 = c1; f2 = f1
                c1 = b - gr * (b - a)
                f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
            else:
                a = c1; c1 = c2; f1 = f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
        opt_t = (a + b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    
    # Compare with grid best
    if f_grid[best_idx] < opt_val:
        opt_t = t_grid[best_idx]
        opt_val = f_grid[best_idx]
    
    return opt_t, opt_val

def _binary_from_continuous(h):
    """Convert continuous h to balanced binary."""
    n = len(h)
    target = n // 2
    # Pick top n/2 elements
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _cyclical_binary_swap(h, deadline, max_passes=50):
    """Cyclical pairwise swap on binary string."""
    n = len(h)
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # Precompute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                # Try swapping h[i]=1 to 0 and h[j]=0 to 1
                h_new = h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
                c5_new = float(np.max(corr_new) * dx)
                if c5_new < cur_c5 - 1e-12:
                    h = h_new
                    corr = corr_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h <= 0.5)[0]
                    break
        if not improved:
            break
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

            # grad[i] += -h[i+d] for 0<=i<=n-1-d
            if d > 0:
                if d <= n - 1:
                    grad[d:] += (1.0 - h[:n-d])  # (1-h[i-d]) for i=d..n-1
                if n - 1 - d >= 0:
                    grad[:n-d] += -h[d:]  # -h[i+d] for i=0..n-1-d
            else:
                # d=0: corr = sum h[i]*(1-h[i]), grad = (1-2h[i])
                grad += (1.0 - 2.0 * h)
        else:
            g = -d  # positive
            # corr[k] = sum_{m=0}^{n-1-g} h[m]*(1-h[m+g])
            # grad[i] += (1-h[i+g]) for 0<=i<=n-1-g
            # grad[i] += -h[i-g] for g<=i<=n-1
            if g > 0:
                if n - 1 - g >= 0:
                    grad[:n-g] += (1.0 - h[g:])  # (1-h[i+g]) for i=0..n-1-g
                grad[g:] += -h[:n-g]  # -h[i-g] for i=g..n-1
    grad /= len(active)
    return grad, mx

def _grad_descent(h, deadline, lr=0.01, n_iter=5000):
    """Subgradient descent with projection."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t = 0.05  # step size
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        grad, mx = _subgradient(h)
        # Check if gradient is meaningful
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        # Line search (simple)
        best_alpha = t
        best_val = cur_c5
        for alpha in [t, t*0.5, t*0.25, t*2, t*4]:
            h_try = h - alpha * grad
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = alpha
        h = h - best_alpha * grad
        h = _project(h, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # Decrease step size
        t *= 0.995
    return best_h, best_c5

def _binary_cyclical_swap(h, deadline, max_passes=200):
    """Fast cyclical pair swap on binary string."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    best_corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_mx = float(np.max(best_corr))
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                # Compute new correlation for swap (i:1→0, j:0→1)
                # Incremental update
                h[j] = 1.0
                h[i] = 0.0
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                mx_new = float(np.max(corr_new))
                if mx_new < best_mx - 1e-12:
                    best_mx = mx_new
                    best_corr = corr_new
                    cur_c5 = best_mx * dx
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h <= 0.5)[0]
                    break
                else:
                    # Revert
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return best_h, best_c5

Wait, this is still O(n^2) per pass because for each one, we try all zeros until we find an improvement. In the worst case, we try all (n/2)^2 pairs. But each trial is O(n) for the correlation. So per pass: O(n^3) = O(156^3) ≈ 3.8M operations. At ~100M ops/sec in numpy, that's ~40ms per pass. We can do ~2500 passes in 100 seconds. That's plenty.

But actually, the `break` after finding an improvement means we typically don't try all pairs. The number of improvements per pass is much less than n^2/4.

Let me think about a better approach. Instead of trying all pairs and taking the first improvement, what about:
1. Compute the "delta" for all pairs (i,j) in O(n^2)
2. Find the best improvement
3. Apply it
4. Repeat

For a binary string, the delta in max correlation when swapping (i,j) can be computed as:
- Recompute correlation (O(n))
- Find max (O(n))
- Compare with current max

So for all n^2/4 pairs: O(n^3). Same as before.

Alternative: for a single swap, can we compute the max correlation in O(1)?
No, because the max could be at any of the 2n-1 lags, and the swap affects many lags.

But we CAN compute the max in O(n) if we maintain the correlation array and only update the affected lags. The swap (i,j) affects lags where the window [d, n-1] contains i or j, or [0, n-1-d] contains i or j (for the negative lags). This is at most O(n) lags, and for each affected lag, the change is O(1) (at most 4 terms change).

So the incremental update is O(n) per swap, and checking if the max decreased requires scanning the correlation array (O(n)). Total: O(n) per swap.

For all n^2/4 pairs: O(n^3) total, same as before. But with smaller constants.

Actually, let me just go with the simple approach: recompute correlation from scratch for each candidate swap. For n=156, this is fast enough.

Let me now think about the overall structure more carefully.

The key question: what's the best n to use?

The C5 value for a binary string of length n is: max(corr) * 2/n.
The "natural" scale of max(corr) is about n/8 (from the average argument).
So C5 ≈ (n/8) * (2/n) = 1/4.

But the actual max(corr) depends on the structure. For a "good" arrangement, max(corr) might be closer to n/8 + small correction. The C5 would then be 1/4 + small/n.

So for larger n, we get closer to 1/4 = 0.25. But we need sum(h) = n/2 exactly, which requires n to be even.

Wait, but the previous code got 0.380892 with n=156. That's much higher than 0.25. Why?

Because the max(corr) is NOT n/8 for a good arrangement. The average is n/8, but the max is higher due to the non-uniformity across lags. The small-lag correlations are inherently larger because they have longer windows.

Let me compute more carefully. For lag d, the window has length n-d. The "expected" c[d] for a random arrangement is (n-d)/4. So c[d] decreases with d. The max is at d=1 (or d=-1): c[1] ≈ (n-1)/4.

C5 ≈ ((n-1)/4) * (2/n) ≈ 1/2.

But we can do better than random! By arranging the ones to minimize the concentration at small lags. The question is: how much can we reduce c[1] below (n-1)/4?

For c[1] = sum_{i=0}^{n-2} h[i+1]*(1-h[i]):
This counts the number of "01" transitions in the sequence.
For a balanced binary string with n/2 ones, the number of "01" transitions equals the number of "10" transitions (they differ by at most 1 depending on boundary). Actually, in a circular sequence, they're exactly equal. In a linear sequence, |#01 - #10| <= 1.

The total number of transitions (#01 + #10) is at most n (for alternating) and at least 2 (for a block). We want to MINIMIZE c[1] = #01 (approximately).

But c[1] = #01 and c[-1] = #10. And #01 + #10 = total transitions.
So max(c[1], c[-1]) >= total_transitions / 2.

To minimize this, we want few transitions. But few transitions means the ones are clustered, which makes large-lag correlations large.

The optimal is a trade-off: moderate number of transitions (so small-lag correlations are moderate) but spread out (so large-lag correlations are also moderate).

This is exactly the problem we're trying to solve! And the answer for n=156 gives C5 ≈ 0.38.

6. Returns the best found

Another idea: use a "number system" approach. The golden ratio sequence (Beatty sequence) has good equidistribution properties.

Let me try the golden ratio approach:
h[i] = 1 if {i * α} < 0.5, where α = (1+√5)/2 or some other irrational.

For α = (1+√5)/2 ≈ 1.618:
The sequence {i*α mod 1} is well-distributed, so the ones are spread out.

Actually, for a Beatty sequence, the n-th term is floor(n*α). The characteristic sequence (1 if i is in the Beatty sequence) has good discrepancy properties.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new

def _greedy_construct(n, alpha):
    """Construct a balanced binary string using a Beatty/greedy approach."""
    h = np.zeros(n)
    target = n // 2
    # Use a golden-ratio-like spacing
    positions = [int(i * alpha * n) % n for i in range(target)]
    # Deduplicate
    seen = set()
    count = 0
    for p in positions:
        if p not in seen and count < target:
            h[p] = 1.0
            seen.add(p)
            count += 1
    # Fill remaining if needed
    if count < target:
        avail = np.where(h == 0)[0]
        for p in avail[:target - count]:
            h[p] = 1.0
    return h

    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.02
    for _ in range(10000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(corr))
        active = np.where(corr >= mx - 1e-8)[0]
        grad = np.zeros(n)
        for k in active:
            d = int(k) - (n - 1)
            if d >= 0:
                if d > 0:
                    if d <= n - 1:
                        grad[d:] += (1.0 - h[:n-d])
                    if n - 1 - d >= 0:
                        grad[:n-d] += -h[d:]
                else:
                    grad += (1.0 - 2.0 * h)
            else:

                if g > 0:
                    if n - 1 - g >= 0:
                        grad[:n-g] += (1.0 - h[g:])
                    if n - g >= 1:
                        grad[g:] += -h[:n-g]
        grad /= len(active)

        h_new = _project(h_new, target)
