import numpy as np
from scipy.optimize import linprog, minimize
import time

_g_target = 0.0
_g_beta = 1.0

def _con_eq_fun(x):
    return np.sum(x) - _g_target

def _con_eq_jac(x):
    return np.ones(len(x))

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    offset = (lo + hi) / 2.0
    x = np.clip(v - offset, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_direction(h, scores, G, radius, top_m=None):
    n = len(h)
    if top_m is not None and 0 < top_m < len(scores):
        top_idx = np.argsort(scores)[-top_m:]
        A_sub = G[top_idx]
        b_sub = -scores[top_idx]
    else:
        A_sub = G
        b_sub = -scores
    nrows = A_sub.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((nrows, n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = b_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((-radius, radius))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        pj = p[j]
        if pj > 1e-14:
            am = (1.0 - h[j]) / pj
            if am < alpha_max:
                alpha_max = am
        elif pj < -1e-14:
            am = h[j] / (-pj)
            if am < alpha_max:
                alpha_max = am
    alpha_max = max(alpha_max, 1e-12) * 0.999
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1v = hi - gr * (hi - lo)
    c2v = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1v * p)
    f2 = _exact_c5(h + c2v * p)
    for _ in range(40):
        if hi - lo < 1e-16:
            break
        if f1 < f2:
            hi = c2v; f2 = f1; c2v = c1v
            c1v = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1v * p)
        else:
            lo = c1v; f1 = f2; c1v = c2v
            c2v = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2v * p)
    opt_a = (lo + hi) / 2.0
    return opt_a, _exact_c5(h + opt_a * p)

def _lp_optimize(h, deadline, radius_init=0.3, max_iters=10000, stall_limit=200, top_m=None):

    target = n / 2.0

    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_m=top_m)
        if p is None or float(np.max(np.abs(p))) < 1e-15:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)

                best_h = h.copy()
            stall = 0

def _lsexp_obj(hf):
    S = _scores(hf)
    maxS = np.max(S)
    shifted = _g_beta * (S - maxS)
    shifted = np.clip(shifted, -500.0, 0.0)
    return maxS + np.log(np.mean(np.exp(shifted))) / _g_beta

def _lsexp_grad(hf):
    S = _scores(hf)
    maxS = np.max(S)
    shifted = _g_beta * (S - maxS)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros(len(S))
        w[int(np.argmax(S))] = 1.0
    else:
        w /= wsum
    G = _grad_matrix(hf)
    return G.T @ w

def _slsqp_lsexp(h, deadline, beta_start=30.0, beta_end=400.0, n_phases=5):
    global _g_beta, _g_target
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    _g_target = target
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        _g_beta = beta_start * (beta_end / beta_start) ** frac
        try:
            res = minimize(_lsexp_obj, h, jac=_lsexp_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq',
                                        'fun': _con_eq_fun,
                                        'jac': _con_eq_jac}],
                          options={'maxiter': 400, 'ftol': 1e-16})
            if res.success and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _golden_ratio_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)

def _self_consistent_field(n, eta, n_iter=200, seed=0):
    """Multiplicative weights / self-consistent field approach."""
    rng = np.random.default_rng(seed)
    h = (rng.random(n) < 0.5).astype(float)
    h = _project(h, n / 2.0)
    p = np.ones(2 * n - 1) / (2 * n - 1)
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        # Compute q_i = 1 - sum_k p_k * h[i+k]
        scores_h = _scores(h)
        for k_idx in range(2 * n - 1):
            pass
        # Efficient: use the gradient at uniform weight
        # q_i = 1 - sum_k p_k h[i+k]
        # For simplicity, compute via convolution-like approach
        q = np.ones(n)
        for d in range(-(n - 1), n):
            r = n - 1 + d
            w = p[r]
            if d >= 0:
                q[:n - d] -= w * h[d:]
            else:
                g = -d
                q[g:] -= w * h[:n - g]
        # Set h_new: 1 where q is smallest (n/2 of them)
        thresh_idx = np.argsort(q)
        h_new = np.zeros(n)
        h_new[thresh_idx[:n // 2]] = 1.0
        h_new = _project(h_new, n / 2.0)
        # Update p via multiplicative weights
        sc = _scores(h_new)
        max_sc = np.max(sc)
        p = np.exp(eta * (sc - max_sc))
        p = p / np.sum(p)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _g_target, _g_beta
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Aggressive LP from initial
    dl1 = t0 + budget_s * 0.22
    h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.4, max_iters=15000, stall_limit=300)
    if c5_1 < best_c5:
        best_c5 = c5_1; best_h = h1.copy()

    # Phase 1b: Focused LP on top lags
    if time.time() < t0 + budget_s * 0.27:
        dl1b = t0 + budget_s * 0.27
        h1b, c5_1b = _lp_optimize(best_h, dl1b, radius_init=0.15, max_iters=8000, stall_limit=150, top_m=12)
        if c5_1b < best_c5:
            best_c5 = c5_1b; best_h = h1b.copy()

    # Phase 2: SLSQP log-sum-exp
    if time.time() < t0 + budget_s * 0.38:
        dl2 = t0 + budget_s * 0.38
        h2, c5_2 = _slsqp_lsexp(best_h, dl2, beta_start=40.0, beta_end=500.0, n_phases=5)
        if c5_2 < best_c5:
            best_c5 = c5_2; best_h = h2.copy()
        if time.time() < dl2:
            h2b, c5_2b = _lp_optimize(best_h, t0 + budget_s * 0.40,
                                       radius_init=0.08, max_iters=3000, stall_limit=80)
            if c5_2b < best_c5:
                best_c5 = c5_2b; best_h = h2b.copy()

    # Phase 3: Subgradient from best
    if time.time() < t0 + budget_s * 0.48:
        dl3 = t0 + budget_s * 0.48
        h3, c5_3 = _subgrad_optimize(best_h, dl3, n_steps=2000, beta_start=20.0, beta_end=80.0)
        if c5_3 < best_c5:
            best_c5 = c5_3; best_h = h3.copy()
        if time.time() < dl3:
            h3b, c5_3b = _lp_optimize(best_h, t0 + budget_s * 0.50,
                                       radius_init=0.1, max_iters=3000, stall_limit=100)
            if c5_3b < best_c5:
                best_c5 = c5_3b; best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.68:
        dl4 = t0 + budget_s * 0.68
        n_tries = [96, 128, 160, 192, 256]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.7320507817]
        for idx, n_try in enumerate(n_tries):
            if time.time() > dl4:
                break
            if n_try == best_n:
                continue
            # Resample from best
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl4, time.time() + budget_s * 0.03)
            ht, ct = _lp_optimize(h_init, sub_dl, radius_init=0.25, max_iters=5000, stall_limit=150)
            if ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try
            # Self-consistent field start
            if time.time() < dl4:
                h_sc, c5_sc = _self_consistent_field(n_try, eta=100.0, n_iter=100, seed=seed + idx * 7)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                ht2, ct2 = _lp_optimize(h_sc, sub_dl2, radius_init=0.2, max_iters=3000, stall_limit=100)
                if ct2 < best_c5:
                    best_c5 = ct2; best_h = ht2.copy(); best_n = n_try
            # Golden ratio + noise
            if time.time() < dl4:
                h_qp = _golden_ratio_init(n_try, alphas[idx % len(alphas)])
                h_qp += rng.normal(0, 0.06, n_try)
                h_qp = _project(h_qp, n_try / 2.0)
                sub_dl3 = min(dl4, time.time() + budget_s * 0.02)
                ht3, ct3 = _lp_optimize(h_qp, sub_dl3, radius_init=0.3, max_iters=3000, stall_limit=100)
                if ct3 < best_c5:
                    best_c5 = ct3; best_h = ht3.copy(); best_n = n_try
                if time.time() < dl4:
                    ht4, ct4 = _slsqp_lsexp(ht3, min(dl4, time.time() + budget_s * 0.015),
                                             beta_start=50, beta_end=300, n_phases=3)
                    if ct4 < best_c5:
                        best_c5 = ct4; best_h = ht4.copy(); best_n = n_try

    # Phase 5: Perturbation restarts on best
    if time.time() < t0 + budget_s * 0.82:
        dl5 = t0 + budget_s * 0.82
        scale = 0.03
        for attempt in range(400):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.008)
            ht, ct = _lp_optimize(h_pert, sub_dl, radius_init=0.06, max_iters=600, stall_limit=50)
            if ct < best_c5 - 1e-13:
                best_c5 = ct; best_h = ht.copy()
                scale = max(0.002, scale * 0.82)
            else:
                scale = min(0.1, scale * 1.12)

    # Phase 6: Deep polish
    if time.time() < t0 + budget_s * 0.93:
        dl6 = t0 + budget_s * 0.93
        h6, c5_6 = _slsqp_lsexp(best_h, t0 + budget_s * 0.88,
                                  beta_start=100.0, beta_end=800.0, n_phases=4)
        if c5_6 < best_c5:
            best_c5 = c5_6; best_h = h6.copy()
        h6b, c5_6b = _lp_optimize(best_h, dl6, radius_init=0.02, max_iters=5000,
                                    stall_limit=200, top_m=6)
        if c5_6b < best_c5:
            best_c5 = c5_6b; best_h = h6b.copy()

    # Phase 7: Micro-polish with very small steps
    if time.time() < t0 + budget_s * 0.97:
        h7, c5_7 = _lp_optimize(best_h, t0 + budget_s * 0.97,
                                  radius_init=0.005, max_iters=4000, stall_limit=150, top_m=4)
        if c5_7 < best_c5:
            best_c5 = c5_7; best_h = h7.copy()

    # Phase 8: Final SLSQP micro
    if time.time() < t0 + budget_s * 0.985:
        h8, c5_8 = _slsqp_lsexp(best_h, t0 + budget_s * 0.985,
                                  beta_start=300.0, beta_end=1000.0, n_phases=2)
        if c5_8 < best_c5:
            best_c5 = c5_8; best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
