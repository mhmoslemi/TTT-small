S[m] = sum_i h[i] * (1-h)[i + m - (n-1)]

When h[i] changes by δ and h[j] changes by -δ:
ΔS[m] = δ * (1-h)[i+m-(n-1)] - δ * (1-h)[j+m-(n-1)]
       = δ * (1 - h[i+m-n+1]) - δ * (1 - h[j+m-n+1])  (where indices are valid)
       + second-order terms from h[i]*(1-h[j]) interactions

The second-order terms: if i + m - (n-1) = j, then the term h[i]*(1-h[j]) changes by δ*(1-(h[j]-δ)) - h[i]*(1-h[j]) = δ - δ².
Similarly if j + m - (n-1) = i.

So for most m, the change is linear in δ. For at most 2 values of m (where the cross-term exists), there's a quadratic term.

This means I can precompute the "gradient" of each score with respect to the pair move, and then the new score is just old_score + linear_term * t + quadratic_t
</think>

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pairwise_optimize(h, deadline, max_sweeps=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if no_improve > 40:
            break

        improved = False
        n_updated_this_sweep = 0
        hi_idx = np.where(h > 0.05)[0]
        lo_idx = np.where(h < 0.95)[0]
        if len(hi_idx) == 0 or len(lo_idx) == 0:
            break

        # Limit pairs for speed
        max_i = min(len(hi_idx), 60)
        max_j = min(len(lo_idx), 60)

        for ii in range(max_i):
            if time.time() > deadline:
                break
            i = hi_idx[ii]
            for jj in range(max_j):
                if time.time() > deadline:
                    break
                j = lo_idx[jj]
                if i == j:
                    continue
                t_lo = max(-h[i], h[j] - 1.0)
                t_hi = min(1.0 - h[i], h[j])
                if t_hi - t_lo < 1e-12:
                    continue

                hi_o = h[i]
                hj_o = h[j]
                best_t = 0.0
                best_v = cur

                # Sample: boundaries + interior
                ts = np.concatenate([np.array([t_lo, t_hi]),
                                     np.linspace(t_lo, t_hi, 8)[1:-1]])
                for t in ts:
                    t = float(np.clip(t, t_lo, t_hi))
                    h[i] = hi_o + t
                    h[j] = hj_o - t
                    c = _exact_c5(h)
                    if c < best_v:
                        best_v = c
                        best_t = t

                h[i] = hi_o
                h[j] = hj_o

                if best_v < cur - 1e-13 and abs(best_t) > 1e-12:
                    h[i] = hi_o + best_t
                    h[j] = hj_o - best_t
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    improved = True
                    no_improve = 0
                    n_updated_this_sweep += 1
                    if n_updated_this_sweep >= 8:
                        break
            if n_updated_this_sweep >= 8:
                break

        if not improved:
            no_improve += 1

    return best_h, best_c5

def _binary_counts(x):
    n = len(x)
    A = np.zeros(n - 1, dtype=np.int32)
    B = np.zeros(n - 1, dtype=np.int32)
    for d in range(1, n):
        a = x[:-d] if d < n else np.array([], dtype=int)
        b = x[d:] if d < n else np.array([], dtype=int)
        if len(a) > 0:
            A[d - 1] = np.sum((a == 1) & (b == 0))
            B[d - 1] = np.sum((a == 0) & (b == 1))
    return A, B

def _binary_c5(x):
    n = len(x)
    dx = 2.0 / n
    A, B = _binary_counts(x)
    return float(np.max(np.concatenate([A, B])) * dx)

def _binary_swap_optimize(x, deadline, max_sweeps=2000):
    n = len(x)
    x = x.copy()
    A, B = _binary_counts(x)
    cur_max = int(np.max(np.concatenate([A, B])))
    cur_c5 = float(cur_max * 2.0 / n)
    best_x = x.copy()
    best_c5 = cur_c5
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if no_improve > 100:
            break

        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        best_delta = 0
        best_i = -1
        best_j = -1

        # Try a subset of swaps
        n_try = min(len(ones) * len(zeros), 2000)
        # Stratified sampling
        step_i = max(1, len(ones) * len(zeros) // max(n_try, 1))
        count = 0
        for ii in range(len(ones)):
            if time.time() > deadline:
                break
            if count >= n_try:
                break
            i = ones[ii]
            for jj in range(0, len(zeros), step_i):
                if count >= n_try:
                    break
                j = zeros[jj]
                if i == j:
                    continue
                count += 1

                # Compute delta for swap i(1->0), j(0->1)
                # Step 1: flip i from 1 to 0
                # A[d] changes: i as first (d=1..n-1-i): A[d-1] -= (x[i+d]==0)
                #               i as second (d=1..i-1): A[d-1] += (x[i-d]==1), B[d-1] -= (x[i-d]==0)
                # Step 2: flip j from 0 to 1 (i already 0)
                # A[d] changes: j as first (d=1..n-1-j): A[d-1] += (x[j+d]==0) [but if j+d==i, x[i] is now 0]
                #               j as second (d=1..j-1): A[d-1] -= (x[j-d] with i->0), B[d-1] += (1-x[j-d] with i->0)

                # For speed, just compute new max from scratch for candidate swaps
                # Only do full recompute for promising candidates
                # Quick estimate: check if swap could improve
                # The max can only decrease if the current max pairs are broken

                # Full recompute (expensive but correct for n=128)
                x_swap = x.copy()
                x_swap[i] = 0
                x_swap[j] = 1
                A_s, B_s = _binary_counts(x_swap)
                new_max = int(np.max(np.concatenate([A_s, B_s])))
                if new_max < cur_max:
                    best_delta = cur_max - new_max
                    best_i = i
                    best_j = j
                    improved = True
                    # For speed, break early if we found improvement
                    break
            if improved:
                break

        if improved:
            x[best_i] = 0
            x[best_j] = 1
            A, B = _binary_counts(x)
            cur_max = int(np.max(np.concatenate([A, B])))
            cur_c5 = float(cur_max * 2.0 / n)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_x, best_c5

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gradient_direction(h, gamma):
    n = len(h)
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    # Compute gradient
    grad = np.zeros(n)
    dx = 2.0 / n
    for d in range(0, n):
        m = n - d
        w_idx = n - 1 + d
        if m > 0:
            # score[n-1+d] = sum_{i=0}^{m-1} h[i](1-h[i+d]) dx
            # d/dh[i] for i < m: (1-h[i+d]) dx
            # d/dh[i] for i >= d: -h[i-d] dx
            if d == 0:
                # R(0) = sum h[i](1-h[i]) dx, d/dh[i] = (1-2h[i]) dx
                grad += w[w_idx] * (1.0 - 2.0 * h) * dx
            else:
                # i as first term: i in 0..m-1
                grad[:m] += w[w_idx] * (1.0 - h[d:m + d]) * dx
                # i as second term: i in d..n-1
                grad[d:] += w[w_idx] * (-h[:m]) * dx
        if d > 0:
            # score[n-1-d] = sum_{j=0}^{m-1} h[j+d](1-h[j]) dx, m = n-d
            w_idx2 = n - 1 - d
            # d/dh[i]: h[i] appears as h[j+d] when j = i-d (i >= d): (1-h[i-d]) dx
            # h[i] appears as h[j] when j = i (i < m): -h[i+d] dx
            grad[d:] += w[w_idx2] * (1.0 - h[:m]) * dx
            grad[:m] += w[w_idx2] * (-h[d:d + m]) * dx
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return -grad / gn

def _gradient_line_search(h, p, deadline):
    n = len(h)
    target = n / 2.0
    cur = _exact_c5(h)
    # Find feasible alpha range
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    # Sample
    best_a = 0.0
    best_c5 = cur
    ts = np.linspace(0, alpha_max, 15)
    for t in ts:
        if time.time() > deadline:
            break
        h_new = _project(h + float(t) * p, target)
        c = _exact_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_a = float(t)
    if best_a > 0:
        h_new = _project(h + best_a * p, target)
        return h_new, _exact_c5(h_new)
    return h, cur

def _gradient_optimize(h, deadline, gamma_start=10, gamma_end=200, max_it=1000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    gamma = gamma_start
    no_imp = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if no_imp > 30:
            break
        p = _gradient_direction(h, gamma)
        if p is None:
            break
        h_new, c_new = _gradient_line_search(h, p, deadline)
        if c_new < cur - 1e-13:
            h = h_new
            cur = c_new
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            no_imp = 0
            gamma = min(gamma * 1.05, gamma_end)
        else:
            no_imp += 1
            gamma = min(gamma * 1.08, gamma_end)
    return best_h, best_c5

def _run_pairwise_from_binary(x, deadline):
    """Convert binary to continuous and run pairwise optimization."""
    n = len(x)
    h = x.astype(float)
    h = _project(h, n / 2.0)
    cur = _exact_c5(h)
    h, cur = _pairwise_optimize(h, deadline, max_sweeps=100)
    return h, cur
