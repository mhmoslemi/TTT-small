import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, 0:m] += (1.0 - h[d:d+m]) * dx
                G[r, d:n] += -h[0:m] * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, 0:m] += (1.0 - h[g:g+m]) * dx
                G[r, g:n] += -h[0:m] * dx
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    n_samp = 16
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _aggressive_lp(h, deadline, radius_start=0.15, min_radius=0.005,
                   stall_limit=40, perturb_scale=0.008, max_iters=100000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_start
    stall = 0
    it = 0
    rng_local = np.random.default_rng(7)
    while it < max_iters:
        if time.time() > deadline:
            break
        it += 1
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.85
            if radius < min_radius:
                if stall > stall_limit:
                    h = _project(h + rng_local.normal(0, perturb_scale, n), target)
                    cur_c5 = _exact_c5(h)
                    radius = max(radius, 0.03)
                    perturb_scale *= 0.7
                    stall = 0
                    if perturb_scale < 0.001:
                        break
                    continue
                else:
                    radius = max(radius * 0.85, min_radius * 0.5)
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.02, 0.3)
        else:
            stall += 1
            radius *= 0.8
            if stall > stall_limit:
                h = _project(h + rng_local.normal(0, perturb_scale, n), target)
                cur_c5 = _exact_c5(h)
                radius = max(radius * 1.5, 0.05)
                perturb_scale *= 0.6
                stall = 0
                if perturb_scale < 0.001 and radius < min_radius * 2:
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_smooth(h, deadline, beta_start=40.0, beta_end=400.0, n_phases=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        def obj_func(hf):
            S = _scores(hf)
            b = beta
            maxS = np.max(S)
            shifted = b * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            return maxS + np.log(np.mean(np.exp(shifted))) / b

        def grad_func(hf):
            S = _scores(hf)
            b = beta
            maxS = np.max(S)
            shifted = b * (S - maxS)
            shifted = np.clip(shifted, -500, 0)
            w = np.exp(shifted)
            wsum = np.sum(w)
            if wsum < 1e-300:
                idx = int(np.argmax(S))
                w = np.zeros(len(S))
                w[idx] = 1.0
            else:
                w /= wsum
            G = _grad_matrix(hf)
            return G.T @ w

        try:
            res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq',
                                        'fun': lambda x, t=target: np.sum(x) - t,
                                        'jac': lambda x: np.ones(len(x))}],
                          options={'maxiter': 150, 'ftol': 1e-15})
            if np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _iterated_local_search(h, deadline, n_restarts=300, sigma=0.005,
                           lp_iters_per_restart=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rng_local = np.random.default_rng(123)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = _project(h + rng_local.normal(0, sigma, n), target)
        cur_c5 = _exact_c5(h_pert)
        radius = 0.04
        for it in range(lp_iters_per_restart):
            if time.time() > deadline:
                break
            scores = _scores(h_pert)
            G = _grad_matrix(h_pert)
            p, _ = _lp_direction(h_pert, scores, G, radius)
            if p is None or float(np.max(np.abs(p))) < 1e-14:
                break
            alpha, new_c5 = _line_search(h_pert, p, cur_c5)
            if new_c5 < cur_c5 - 1e-14:
                h_pert = _project(h_pert + alpha * p, target)
                cur_c5 = _exact_c5(h_pert)
                radius = min(radius * 1.03, 0.15)
            else:
                radius *= 0.8
                if radius < 0.005:
                    break
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h_pert.copy()
            sigma = max(0.002, sigma * 0.9)
            h = h_pert.copy()
        else:
            sigma = min(0.03, sigma * 1.15)
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)
