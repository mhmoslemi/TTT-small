def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    m = int(np.log2(L + 1))
    state = 1
    seq = np.zeros(L, dtype=int)
    for i in range(L):
        seq[i] = state & 1
        fb = ((state >> (m-1)) ^ (state >> (m-4))) & 1
        state = (state >> 1) | (fb << (m-1))
    return seq.astype(float)

    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):

            break

                    zeros = np.where(h < 0.5)[0]
                    j = j  # need to find j's new position...

                    break
            else:
                h[j] = 1.0  # No improvement, revert

        if not improved:

    return best_h, best_c5

1. Compute scores
2. Find the top-K lags
3. Compute the weighted subgradient: g = Σ_{k in top-K} w_k * ∇score_k

The gradient of score at lag d (index n-1+d) with respect to h[j]:

This can be computed in O(1) per (d,j) pair. For the top-K lags, the total cost is O(K*n) per subgradient evaluation. Without building the G matrix, this is faster.

```python
def _subgrad(h, top_k=10, beta=20):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    # Get top-k scores
    idx = np.argsort(scores)[-top_k:]
    s_top = scores[idx]
    # Softmax weights
    w = np.exp(beta * (s_top - s_top.max()))
    w /= w.sum()
    # Compute weighted gradient
    grad = np.zeros(n)
    for (d_idx, wi) in zip(idx, w):
        d = d_idx - (n - 1)  # actual lag
        if d == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif d > 0:
            # score = dx * sum_{i=0}^{n-1-d} h[i+d](1-h[i])
            # d/dh[j]: (1-h[j-d]) if j >= d, minus h[j+d] if j < n-d
            if d < n:
                grad[d:] += wi * dx * (1.0 - h[:-d])
                grad[:n-d] += -wi * dx * h[d:]
        else:
            # d < 0, let s = -d
            s = -d
            # score = dx * sum_{i=0}^{n-1-s} h[i](1-h[i+s])
            # d/dh[j]: (1-h[j+s]) if j < n-s, minus h[j-s] if j >= s
            grad[:n-s] += wi * dx * (1.0 - h[s:])
            grad[s:] += -wi * dx * h[:n-s]
    return grad, scores
```
