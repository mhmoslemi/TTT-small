Let me be very careful about the line search. The parent's approach uses a quadratic model of the max function, which is clever:

For direction d and step t:
  h(t) = h + t*d
  F_s(t) = F_s + t * (∂F_s/∂h · d) + t^2 * (∂²F_s/∂h² · d ... ) 

Wait, F_s is bilinear in h, so F_s(h + t*d) is quadratic in t:
  F_s(h + t*d) = F_s(h) + t * (G_s · d) + t^2 * (quadratic term)

The quadratic term for F_s:
  F_s(h) = dx * sum_i h[i+k] * (1-h[i])
  F_s(h + t*d) = dx * sum_i (h[i+k] + t*d[i+k]) * (1 - h[i] - t*d[i])
  = F_s(h) + t * dx * sum_i [d[i+k]*(1-h[i]) - h[i+k]*d[i]]
  + t^2 * dx * sum_i [-d[i+k]*d[i]]

So the linear term is G_s · d (the directional derivative), and the quadratic term is -dx * sum_i d[i+k]*d[i].

In the parent, this is computed as:
  S0 = correlate(h, 1-h, 'full')  # current F values (before dx)
  T = correlate(d, 1-h, 'full') - correlate(h, d, 'full')  # linear term (before dx)
  C = correlate(d, d, 'full')  # quadratic term (before dx)

Then F_s(t) = (S0[s] + t*T[s] - t^2*C[s]) * dx

And max_s F_s(t) = dx * max_s (S0[s] + t*T[s] - t^2*C[s])

The parent then does a grid search + golden section on this 1D function. This is exact (no approximation) because F_s is exactly quadratic in t.

This is a key advantage: the line search is exact for the bilinear structure.

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = 0.0, 1.0
    # Bisection
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_jac(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        r = n-1+k
        m = n - abs(k)
        if m <= 0: continue
        if k >= 0:
            G[r, k:] += (1.0 - h[:m]) * dx
            G[r, :m] += -h[k:] * dx
        else:
            a = -k
            G[r, :m] += (1.0 - h[a:]) * dx
            G[r, a:] += -h[:m] * dx
    return G

# === Line search ===

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0-h, mode='full')
    T = np.correlate(d, 1.0-h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    # Feasible interval
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo, t_hi = -radius, radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Grid + golden
    grid = np.linspace(t_lo, t_hi, 80)
    vals = np.max(S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:], axis=1)
    best_i = int(np.argmin(vals))
    best_t, best_v = grid[best_i], vals[best_i]
    # Golden section refinement
    a, b = t_lo, t_hi
    gr = (np.sqrt(5.0)-1.0)/2.0
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C))
    f2 = float(np.max(S0 + c2*T - c2*c2*C))
    for _ in range(60):
        if b - a < 1e-15: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C))
    t_opt = (a+b)/2.0
    v_opt = float(np.max(S0 + t_opt*T - t_opt*t_opt*C))
    if v_opt < best_v:
        best_t, best_v = t_opt, v_opt
    return best_t, best_v * dx
