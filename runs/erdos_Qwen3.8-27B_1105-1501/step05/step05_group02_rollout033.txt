        w_cut[:half] = 0.0
        w_cut[half+1:] = 0.0
        K = n // 2
        best_round_c5 = c5_cur
        best_round_h = None
        lam_vals = np.array([0.05, 0.15, 0.3, 0.5, 0.8, 1.2, 2.0, 3.0, 5.0, 8.0])
        x_lo = None
        x_hi = None
        cnt_lo = -1
        cnt_hi = -1
        for lam in lam_vals:
            if time.time() > deadline:
                break
            try:
                x = _solve_cut(n, w_cut, float(lam))
            except Exception:
                continue
            cnt = int(np.sum(x))
            if cnt < K and cnt_lo < 0:
                x_lo = x
                cnt_lo = cnt
            if cnt >= K and cnt_hi < 0:
                x_hi = x
                cnt_hi = cnt
            if abs(cnt - K) <= 2:
                x_p = _project(x, target)
                c5_new = _exact_c5(x_p)
                if c5_new < best_round_c5:
                    best_round_c5 = c5_new
                    best_round_h = x_p.copy()
            if x_lo is not None and x_hi is not None:
                break
        if x_lo is not None and x_hi is not None and cnt_hi > cnt_lo:
            alpha = (K - cnt_lo) / float(cnt_hi - cnt_lo)
            x_interp = alpha * x_hi + (1.0 - alpha) * x_lo
            x_interp = _project(x_interp, target)
            c5_interp = _exact_c5(x_interp)
            if c5_interp < best_round_c5:
                best_round_c5 = c5_interp
                best_round_h = x_interp.copy()
        if best_round_h is not None and best_round_c5 < c5_cur - 1e-11:
            h_cur = best_round_h
            c5_cur = best_round_c5
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
            stall = 0
            beta = min(beta * 1.05, 80.0)
        else:
            stall += 1
            if stall % 5 == 0:
                beta = min(beta * 1.3, 100.0)
            else:
                beta = max(1.0, beta * 0.85)
    return best_h, best_c5

def _slsqp_minimax(h, deadline, topk=40, max_iter=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_arr = h.copy()
    scores0 = _scores(h_arr)
    c5_0 = float(np.max(scores0))
    idx = np.argsort(scores0)[-topk:]
    idx = np.sort(idx)
    n_idx = len(idx)
    def obj(z):
        return z[n]
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def eq_con(z):
        return np.sum(z[:n]) - target
    def eq_con_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    def ineq_con(z):
        hz = z[:n]
        t = z[n]
        sc = np.correlate(hz, 1.0 - hz, mode='full') * dx
        return t - sc[idx]
    def ineq_con_jac(z):
        hz = z[:n]
        J = np.zeros((n_idx, n + 1))
        for row, k in enumerate(idx):
            d = k - (n - 1)
            if d == 0:
                J[row, :n] = (1.0 - 2.0 * hz) * dx
            elif d > 0:
                m = n - d
                J[row, :m] -= hz[d:n] * dx
                J[row, d:n] += (1.0 - hz[:m]) * dx
            else:
                da = -d
                m = n - da
                J[row, :m] += (1.0 - hz[da:n]) * dx
                J[row, da:n] -= hz[:m] * dx
            J[row, n] = -1.0
        return J
    bounds = [(0.0, 1.0)] * n + [(c5_0 * 0.5, 1.5)]
    constraints = [
        {'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac},
        {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_con_jac},
    ]
    z0 = np.concatenate([h_arr, [c5_0 + 1e-4]])
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x[n] < c5_0:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < c5_0 - 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h_arr, c5_0

def _lp_step(h, radius, topk):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    scores = _scores(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = (1.0 - 2.0 * h) * dx
    half = n // 2
    for d in range(1, half + 1):
        m = n - d
        G[n - 1 - d, :m] += (1.0 - h[d:n]) * dx
        G[n - 1 - d, d:n] += -h[:m] * dx
        G[n - 1 + d, :m] += -h[d:n] * dx
        G[n - 1 + d, d:n] += (1.0 - h[:m]) * dx
    if topk < 2 * n - 1:
        idx = np.argsort(scores)[-topk:]
        Gk = G[idx]
        sk = scores[idx]
    else:
        Gk = G
        sk = scores
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(sk), n + 1))
    A_ub[:, :n] = Gk
    A_ub[:, n] = -1.0
    b_ub = -sk
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, n_eval=20):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(n_eval):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _lp_optimize(h, deadline, radius_init=0.25, max_it=3000, topk=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        p = _lp_step(h, radius, topk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall > 8 and radius < 0.03:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    if n0 % 2 != 0:
        n0 = n0 - 1
        h0 = h0[:n0]
    h_start = _project(h0.copy(), n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    # Phase 1: LP polish
    dl1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.3, max_it=3000, topk=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)
    # Phase 2: SLSQP polish
    dl2 = t0 + budget_s * 0.15
    h2, c5_2 = _slsqp_minimax(best_h, dl2, topk=40, max_iter=200)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Phase 3: Min-cut exploration at n0
    dl3 = t0 + budget_s * 0.40
    h3, c5_3 = _mincut_optimize(best_h, dl3, beta_init=3.0, max_rounds=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)
    # Phase 4: LP polish after min-cut
    dl4 = t0 + budget_s * 0.48
    h4, c5_4 = _lp_optimize(best_h, dl4, radius_init=0.15, max_it=2000, topk=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Phase 5: Try other resolutions
    dl5 = t0 + budget_s * 0.72
    for n_try in [96, 112, 144, 160, 176, 192, 208]:
        if time.time() > dl5:
            break
        if best_c5 < 0.3806:
            break
        h_res = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + budget_s * 0.025)
        h_try, c5_try = _lp_optimize(h_res, sub_dl, radius_init=0.2, max_it=1500, topk=25)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
            h_try2, c5_try2 = _mincut_optimize(best_h, sub_dl2, beta_init=4.0, max_rounds=50)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()
                best_n = n_try
    # Phase 6: SLSQP on best
    if time.time() < t0 + budget_s * 0.80:
        dl6 = t0 + budget_s * 0.85
        h6, c5_6 = _slsqp_minimax(best_h, dl6, topk=50, max_iter=300)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    # Phase 7: Min-cut at new resolution
    if time.time() < t0 + budget_s * 0.90:
        dl7 = t0 + budget_s * 0.94
        h7, c5_7 = _mincut_optimize(best_h, dl7, beta_init=5.0, max_rounds=100)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    # Phase 8: Final LP polish
    if time.time() < t0 + budget_s * 0.97:
        dl8 = t0 + budget_s * 0.985
        h8, c5_8 = _lp_optimize(best_h, dl8, radius_init=0.05, max_it=2000, topk=20)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
