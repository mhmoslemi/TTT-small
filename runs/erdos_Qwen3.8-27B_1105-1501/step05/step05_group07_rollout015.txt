def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _overlap_grad(h, w):
    """Compute gradient of weighted sum of overlaps.
    w has shape (2n-1,), h has shape (n,).
    Returns gradient of sum_k w_k * O(d_k) w.r.t. h.
    """
    n = len(h)
    grad = np.zeros(n)
    # For each lag d, O(d) = sum_i h[i]*(1-h[i+d])*dx
    # dhot_ O(d)/∂h[j] = dx*[(1-h[j+d]) if valid] - dx*[h[j-d] if valid]
    dx = 2.0 / n
    for d in range(-(n-1), n):
        if abs(w[n-1+d]) < 1e-15:
            continue
        wd = w[n-1+d] * dx
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] += -wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wd * (1.0 - h[g:])
                grad[g:] += -wd * h[:m]
    return grad

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: L-BFGS-B smooth minmax on n0
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _lbfgs_smooth(h_start, dl1,
        betas=[3.0, 8.0, 20.0, 50.0, 120.0, 300.0], maxiter=400)
    if c5_1 < best_c5:
        best_c5 = c5_1; best_h = h1.copy()

    # Phase 2: LP polish
    dl2 = t0 + budget_s * 0.22
    h2, c5_2 = _lp_optimize(best_h, dl2, radius_init=0.15, max_iters=4000, stall_limit=120)
    if c5_2 < best_c5:
        best_c5 = c5_2; best_h = h2.copy()

    # Phase 3: Active-set polish
    dl3 = t0 + budget_s * 0.30
    h3, c5_3 = _active_polish(best_h, dl3, n_iters=800)
    if c5_3 < best_c5:
        best_c5 = c5_3; best_h = h3.copy()

    # Phase 4: Multi-resolution exploration
    dl4 = t0 + budget_s * 0.55
    for n_try in [96, 160, 192]:
        if time.time() > dl4:
            break
        h_init = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.05)
        ht, ct = _lbfgs_smooth(h_init, sub_dl,
            betas=[5.0, 15.0, 50.0, 150.0, 400.0], maxiter=300)
        if ct < best_c5:
            sub_dl2 = min(dl4, time.time() + budget_s * 0.03)
            ht2, ct2 = _lp_optimize(ht, sub_dl2, radius_init=0.1, max_iters=2000, stall_limit=80)
            if ct2 < best_c5:
                best_c5 = ct2; best_h = ht2.copy(); best_n = n_try
            elif ct < best_c5:
                best_c5 = ct; best_h = ht.copy(); best_n = n_try

    # Phase 5: Structured multi-start at various n
    dl5 = t0 + budget_s * 0.72
    alphas = [0.6180339887, 0.4142135624, 0.3819660113, 0.7071067812,
              0.5000000000, 0.3333333333, 0.7500000000]
    for n_try in [128, 96, 160]:
        if time.time() > dl5:
            break
        for ai in range(3):
            if time.time() > dl5:
                break
            h_qp = _quasi_init(n_try, alphas[(n_try + ai) % len(alphas)])
            h_qp += rng.normal(0, 0.08, n_try)
            h_qp = _project(h_qp, n_try / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.025)
            ht, ct = _lbfgs_smooth(h_qp, sub_dl,
                betas=[10.0, 40.0, 150.0, 500.0], maxiter=200)
            if ct < best_c5:
                sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
                ht2, ct2 = _lp_optimize(ht, sub_dl2, radius_init=0.08, max_iters=1500, stall_limit=60)
                if ct2 < best_c5:
                    best_c5 = ct2; best_h = ht2.copy(); best_n = n_try
                else:
                    best_c5 = ct; best_h = ht.copy(); best_n = n_try

    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.88
    scale = 0.015
    for attempt in range(300):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.008)
        ht, ct = _lbfgs_smooth(h_pert, sub_dl,
            betas=[30.0, 100.0, 400.0], maxiter=100)
        if ct < best_c5 - 1e-14:
            sub_dl2 = min(dl6, time.time() + budget_s * 0.004)
            ht2, ct2 = _active_polish(ht, sub_dl2, n_iters=200)
            if ct2 < best_c5:
                best_c5 = ct2; best_h = ht2.copy()
            else:
                best_c5 = ct; best_h = ht.copy()
            scale = max(0.003, scale * 0.7)
        else:
            scale = min(0.06, scale * 1.15)

    # Phase 7: Final combined polish
    dl7 = t0 + budget_s * 0.95
    h7, c5_7 = _lbfgs_smooth(best_h, dl7,
        betas=[100.0, 300.0, 800.0, 2000.0], maxiter=200)
    if c5_7 < best_c5:
        best_c5 = c5_7; best_h = h7.copy()

    h7b, c5_7b = _active_polish(best_h, t0 + budget_s * 0.965, n_iters=1000)
    if c5_7b < best_c5:
        best_c5 = c5_7b; best_h = h7b.copy()

    h7c, c5_7c = _lp_optimize(best_h, t0 + budget_s * 0.98,
        radius_init=0.03, max_iters=3000, stall_limit=100)
    if c5_7c < best_c5:
        best_c5 = c5_7c; best_h = h7c.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
