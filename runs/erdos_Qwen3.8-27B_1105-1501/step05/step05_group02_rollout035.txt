import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < (1.0 - 1e-12)
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp(np.clip((scores - max_s) / tau, -60, 0))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        grad += w0 * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[n - 1 + d]
        wn = w[n - 1 - d]
        if wp < 1e-15 and wn < 1e-15:
            continue
        m = n - d
        if wp > 1e-15:
            grad[d:] += wp * dx * (1.0 - h[:m])
            grad[:m] -= wp * dx * h[d:]
        if wn > 1e-15:
            grad[:m] += wn * dx * (1.0 - h[d:])
            grad[d:] -= wn * dx * h[:m]
    return grad

def _pair_line_search(h, p, q, n_samples=15):
    hp, hq = h[p], h[q]
    t_min = max(-hp, hq - 1.0)
    t_max = min(1.0 - hp, hq)
    if t_max - t_min < 1e-14:
        return 0.0, _exact_c5(h)
    h_work = h.copy()
    ts = np.linspace(t_min, t_max, n_samples)
    best_t = float(ts[0])
    h_work[p] = hp + best_t
    h_work[q] = hq - best_t
    best_c5 = _exact_c5(h_work)
    for i in range(1, n_samples):
        t = float(ts[i])
        h_work[p] = hp + t
        h_work[q] = hq - t
        c5 = _exact_c5(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    step = (t_max - t_min) / (n_samples - 1)
    lo = max(t_min, best_t - step)
    hi = min(t_max, best_t + step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(12):
        if hi - lo < 1e-10:
            break
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h_work[p] = hp + c1
        h_work[q] = hq - c1
        f1 = _exact_c5(h_work)
        h_work[p] = hp + c2
        h_work[q] = hq - c2
        f2 = _exact_c5(h_work)

        else:
            lo = c1
    opt_t = (lo + hi) / 2.0

    h_work[q] = hq - opt_t
    opt_c5 = _exact_c5(h_work)
    if opt_c5 < best_c5:

        best_c5 = opt_c5
    return best_t, best_c5

def _pair_optimize(h, deadline, rng, n_pairs=2000, tau_init=0.005, use_full_dir=True):

def _resample(h, n_new):
