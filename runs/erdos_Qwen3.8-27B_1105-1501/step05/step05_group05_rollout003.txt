import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _line_search(h, d, dx, max_step):
    """Find best step alpha in [0, max_step] along direction d (mean-zero)."""
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible range
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = max_step
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))  # this should be >= 0 since d<0 and h>=0
    # Actually for neg direction: h + t*d >= 0 means t <= h/(-d) = -h/d
    # So t_hi should also consider: t <= h[i]/(-d[i]) for d[i] < 0
    neg_mask = d < -1e-15
    if np.any(neg_mask):
        t_hi = min(t_hi, float(np.min(h[neg_mask] / (-d[neg_mask]))))
    
    t_lo = max(t_lo, 0.0)
    if t_hi <= t_lo:
        return 0.0, float(np.max(S0)) * dx
    
    # Grid search + golden section
    n_grid = 30
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t, best_val = grid[best_i], obj[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(25):
        if b - a < 1e-11:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt_t = (a+b)/2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, radius, top_k=None):
    """Solve LP to find direction that minimizes predicted max score."""
    n = len(h)
    scores = _scores(h)
    
    # Build gradient matrix for all (or top_k) lags
    if top_k is not None:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(2*n - 1)
    
    G = np.zeros((len(idx), n))
    for ii, r in enumerate(idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[ii, :m] += (1.0 - h[d:d+m])
                G[ii, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[ii, :m] += (1.0 - h[g:])
                G[ii, g:] += -h[:m]
    
    # LP: min t
    # s.t. G @ delta - t <= -scores[idx]  (i.e., predicted score <= current)
    #      sum(delta) = 0
    #      -radius <= delta[i] <= radius
    #      h[i]-1 <= delta[i] <= 1-h[i]  (feasibility)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx].copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], float(c[-1] * res.x[-1])
    except Exception:
        pass
    return None, None

def _lse_gradient(h, dx, beta):
    """Compute gradient of log-sum-exp approximation to max score."""
    n = len(h)
    scores = _scores(h) * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    # Clip for numerical stability
    v = np.maximum(v, -50.0)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    
    # Gradient of LSE with respect to h
    # d/dh_i [sum_r s_r] = gradient of s_r at h
    # s_r(h) = sum_j h[j](1-h[j-s_r]) where s_r = r-(n-1)
    # ds_r/dh_i = (1-h[i+s_r]) if 0<=i+s_r<n, plus -h[i-s_r] if 0<=i-s_r<n
    grad = np.zeros(n)
    for r in range(2*n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g+m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _resample(h, n_new):

    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)

def _pair_refine(h, deadline, rng, max_passes=100):
    """Coordinate descent on pairs."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

        if time.time() > deadline:
            break
        improved = False
        # Random pairs for efficiency
        n_pairs = min(500, n * (n-1) // 2)
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            # Skip if both are at bounds
            if (h[i] <= 1e-10 or h[i] >= 1-1e-10) and (h[j] <= 1e-10 or h[j] >= 1-1e-10):

            d[j] = 1.0

            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=30):

        if c5_t < best_c5 - 1e-12:

            best_h = h_t.copy()
    return best_h, best_c5
