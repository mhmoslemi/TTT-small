def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target} using water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        scores = scores - A[i_g, :] + A[j_g, :]
        A = build_A()  # Rebuild (O(n^2))
        cur_max = float(np.max(scores))
        c5 = cur_max * 2.0 / n
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        stall = 0  # We made progress
    
    return best_h, best_c5
