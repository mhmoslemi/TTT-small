    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):

        else:

    res = target - float(np.sum(x))
    if abs(res) > 1e-14:

        else:

        else:

            cnt = int(np.sum(mask))
            if cnt > 0:
                x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))

        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]

    n = len(h)
    F = _scores(h)  # length 2n-1, all >= 0
    S = float(F.max())
    if S < 1e-30:
        return 0.0, np.zeros(n)
    v = F / S  # in [0, 1]
    # For numerical stability: v^(q-1) might underflow, but that's OK (means zero weight)

        W = float(w.sum())

        # Actually for q=1, Phi_1 = sum F_r = S * sum v_r

        return val, grad
    vp1 = v ** (q - 1)  # underflow is fine
    W = float(vp1.sum())
    if W < 1e-300:
        # Only the max element contributes
        idx = int(np.argmax(v))
        val = S
        grad = G[idx] * S
        return val, grad
    val = S * W ** (1.0 / q)

    coeff = W ** (1.0 / q - 1.0)

    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5

    for step in range(max_steps):
        if time.time() > deadline:
            break
        G = _build_G(h)
        grad = _pnorm_grad(h, q, G)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn  # normalize for step size control
        
        # Line search
        improved = False
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:

                cur_c5 = c5_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if not improved:
            # Try smaller alpha or break
            break

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0:
            mask = x < 1.0 if res > 0 else x > 0.0
            cnt = int(np.sum(mask))

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)

        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]

        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

        return G[idx].copy()

def _pair_descent(h, deadline, max_passes=300, verify_thresh=1e-7):
    n = len(h)

            break

                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1

def _quasi_init(n, alpha_val, rng):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_init(h, kernel_width):
    n = len(h)
    k = max(1, int(kernel_width))
    kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / max(1, k // 2)) ** 2)
    kernel /= kernel.sum()

                best_h = h_lp3.copy()

                best_h, best_c5 = _pair_descent(best_h, dl5, max_passes=200, verify_thresh=1e-9)

    # Phase 6: Perturbation restarts
    dl6 = hard_deadline - 5
    scale = 0.02
    for attempt in range(300):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.006)
        h_try, c5_try = _pnorm_descend(h_pert, 16, sub_dl, max_steps=80, alpha_init=0.06)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.8)
        elif attempt > 20:
            scale = min(0.06, scale * 1.08)

    # Final
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
