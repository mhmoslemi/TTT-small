def _softmax_obj(h, tau):
    """Soft-max surrogate of the max correlation, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(S))
    # Avoid overflow
    eS = np.exp((S - M) / tau)
    soft = M + tau * np.log(np.sum(eS))

def _softmax_grad(h, tau):
    """Gradient of soft-max surrogate."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(S))
    eS = np.exp((S - M) / tau)
    w = eS / np.sum(eS)

    # S_r = sum_i h[i] * (1 - h[i + r - (n-1)]) for valid i
    # grad_h[j] = sum_r w[r] * dS_r/dh[j]
    
    G = np.zeros(n)
