import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, target, n_eval=25):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(n_eval):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    best_a = c1 if f1 < f2 else c2
    best_c5 = f1 if f1 < f2 else f2
    mid = (lo + hi) / 2.0
    c5_mid = _exact_c5(_project(h + mid * p, target))
    if c5_mid < best_c5:
        best_a, best_c5 = mid, c5_mid
    return best_a, best_c5

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=3000, radius_min=0.002):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None:
            stall += 1
            radius *= 0.6
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search(h, p, target)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5

def _build_basis(n, q_target, rng):
    cols = []
    widths = []
    w = 1
    while 2 * w <= n:
        widths.append(w)
        w *= 2
    for w in widths:
        stride = max(1, w // 2)
        for start in range(0, n - 2 * w + 1, stride):
            col = np.zeros(n)
            col[start:start+w] = 1.0
            col[start+w:start+2*w] = -1.0
            norm = np.linalg.norm(col)
            if norm > 1e-15:
                cols.append(col / norm)
    if len(cols) < q_target:
        for _ in range(q_target - len(cols)):
            start = rng.integers(0, max(1, n - 4))
            half = rng.integers(1, max(2, n // 4))
            col = np.zeros(n)
            end = min(start + 2 * half, n)
            mid = start + half
            col[start:mid] = 1.0
            col[mid:end] = -1.0
            nm = np.linalg.norm(col)
            if nm > 1e-15:
                cols.append(col / nm)
    cols = np.array(cols).T
    if cols.shape[1] > q_target:
        idx = rng.choice(cols.shape[1], q_target, replace=False)
        cols = cols[:, idx]
    elif cols.shape[1] < q_target:
        extra = np.zeros((n, q_target - cols.shape[1]))
        for j in range(q_target - cols.shape[1]):
            a = rng.integers(0, n)
            b = rng.integers(0, n)
            extra[a, j] = 1.0
            extra[b, j] = -1.0
            nm = np.linalg.norm(extra[:, j])
            if nm > 1e-15:
                extra[:, j] /= nm
        cols = np.hstack([cols, extra])
    cols = cols - cols.mean(axis=0)
    norms = np.linalg.norm(cols, axis=0)
    norms[norms < 1e-15] = 1.0
    cols = cols / norms
    return cols

def _de_fitness(y, B, target):
    raw = 0.5 + B @ y
    h = _project(raw, target)
    return _exact_c5(h)

def _de_search(B, h_start, target, deadline, rng, pop_size=None, max_gen=200):
    n, q = B.shape
    if pop_size is None:
        pop_size = max(24, min(50, 2 * q))
    v = h_start - 0.5
    y0 = np.linalg.lstsq(B, v, rcond=None)[0]
    bounds_lo = -3.0
    bounds_hi = 3.0
    pop = np.zeros((pop_size, q))
    pop[0] = y0
    for i in range(1, pop_size):
        pop[i] = y0 + rng.uniform(0.3, 2.0, q) * rng.standard_normal(q)
    pop = np.clip(pop, bounds_lo, bounds_hi)
    fit = np.array([_de_fitness(pop[i], B, target) for i in range(pop_size)])
    best_i = int(np.argmin(fit))
    best_y = pop[best_i].copy()
    best_f = fit[best_i]
    F = 0.7
    CR = 0.8
    stall = 0
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = max(0.3, F * (0.99 if stall > 10 else 1.0))
        for i in range(pop_size):
            if time.time() > deadline:
                break
            a, b, c = rng.choice(pop_size, 3, replace=False)
            while a == b or a == c or b == c:
                a, b, c = rng.choice(pop_size, 3, replace=False)
            v_trial = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(q) < CR
            if not np.any(mask):
                mask[rng.integers(q)] = True
            trial = np.where(mask, v_trial, pop[i])
            trial = np.clip(trial, bounds_lo, bounds_hi)
            f_trial = _de_fitness(trial, B, target)
            if f_trial <= fit[i]:
                pop[i] = trial
                fit[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_y = trial.copy()
                    stall = 0
            else:
                stall += 1
        if stall > pop_size * 5:
            for i in range(1, pop_size):
                if rng.random() < 0.3:
                    pop[i] = best_y + rng.uniform(0.5, 2.0) * rng.standard_normal(q)
                    pop[i] = np.clip(pop[i], bounds_lo, bounds_hi)
                    fit[i] = _de_fitness(pop[i], B, target)
            stall = 0
    best_h = _project(0.5 + B @ best_y, target)
    return best_h, best_f, best_y

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng=None):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    h = (vals < 0.5).astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
        h = _project(h, n / 2.0)
    return h

def _momentum_optimize(h, deadline, beta_start=20, beta_end=300, n_steps=200, step_init=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.02, 0.12)
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-6:
                break
    return best_h, best_c5

def _pair_polish(h, deadline, n_iters=5000, rng=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        sign = 1 if rng.random() < 0.5 else -1
        if sign > 0:
            a_max = min(1.0 - h[i], h[j])
        else:
            a_max = min(h[i], 1.0 - h[j])
        if a_max < 1e-10:
            continue
        for _attempt in range(5):
            delta = a_max * rng.uniform(0.1, 1.0)
            h_try = h.copy()
            h_try[i] += sign * delta
            h_try[j] -= sign * delta
            h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                break
    return best_h, best_c5

def _worst_lag_target(h, deadline, n_iters=200, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = np.max(scores)
        active = np.where(scores > max_s - 1e-4)[0]
        d_idx = active[rng.integers(len(active))]
        d = d_idx - (n - 1)
        if d >= 0:
            m = n - d
            idx_a = np.arange(d, n)
            idx_b = np.arange(0, m)
        else:
            g = -d
            m = n - g
            idx_a = np.arange(0, m)
            idx_b = np.arange(g, n)
        contrib = h[idx_a] * (1.0 - h[idx_b])
        top_k = min(5, len(contrib))
        top_idx = np.argsort(contrib)[-top_k:]
        for ti in top_idx:
            if time.time() > deadline:
                break
            ia = idx_a[ti]
            ib = idx_b[ti]
            for direction in [1, -1]:
                for delta in [0.05, 0.1, 0.2]:
                    h_try = h.copy()
                    if direction > 0:
                        if h_try[ia] < 1.0 and h_try[ib] > 0.0:
                            h_try[ia] = min(1.0, h_try[ia] + delta)
                            h_try[ib] = max(0.0, h_try[ib] - delta)
                    else:
                        if h_try[ia] > 0.0 and h_try[ib] < 1.0:
                            h_try[ia] = max(0.0, h_try[ia] - delta)
                            h_try[ib] = min(1.0, h_try[ib] + delta)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
    return best_h, best_c5

        h5, c5_5 = _worst_lag_target(best_h, dl, n_iters=300, rng=rng)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < t0 + budget_s * 0.78:
            dl2 = min(t0 + budget_s * 0.78, time.time() + budget_s * 0.04)
            h5b, c5_5b = _lp_optimize(best_h, dl2, radius_init=0.06,
                                       max_iters=1500, radius_min=0.002)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: DE from current best in new basis (different randomization)
    if time.time() < t0 + budget_s * 0.85:
        n_cur = best_n
        target = n_cur / 2.0
        q = max(20, min(50, n_cur // 3))
        B2 = _build_basis(n_cur, q, rng)
        sub_dl = min(t0 + budget_s * 0.85, time.time() + budget_s * 0.05)
        h6, c5_6, _ = _de_search(B2, best_h, target, sub_dl, rng,
                                   pop_size=max(20, min(35, q)), max_gen=100)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        if time.time() < t0 + budget_s * 0.88:
            dl3 = min(t0 + budget_s * 0.88, time.time() + budget_s * 0.02)
            h6b, c5_6b = _lp_optimize(best_h, dl3, radius_init=0.05,
                                       max_iters=1000, radius_min=0.002)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.93:
        dl = t0 + budget_s * 0.93
        h7, c5_7 = _pair_polish(best_h, dl, n_iters=8000, rng=rng)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        dl = t0 + budget_s * 0.995
        h8, c5_8 = _lp_optimize(best_h, dl, radius_init=0.03, max_iters=3000, radius_min=0.0005)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
