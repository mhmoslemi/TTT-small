```python
def _weighted_gradient(h, weights):
    """Compute weighted gradient of sum_r weights[r] * output[r] * dx"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(2*n-1):
        s = weights[r] * dx
        if s < 1e-15:
            continue
        shift = r - n + 1
        if shift >= 0:
            # h[i] in first factor: i in [shift, n-1]
            i_start = shift
            i_end = n - 1
            if i_end >= i_start:
                idx = np.arange(i_start, i_end+1)
                grad[idx] += s * (1.0 - h[idx - shift])
            # h[i] in second factor: i in [0, n-shift-1]
            i_start = 0
            i_end = n - shift - 1
            if i_end >= i_start:
                idx = np.arange(i_start, i_end+1)
                grad[idx] -= s * h[idx + shift]
        else:
            g = -shift
            # h[i] in first factor: i in [0, n-1-g]
            i_end = n - 1 - g
            if i_end >= 0:
                idx = np.arange(0, i_end+1)
                grad[idx] += s * (1.0 - h[idx + g])
            # h[i] in second factor: i in [g, n-1]
            idx = np.arange(g, n)
            grad[idx] -= s * h[idx - g]
    return grad
```

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        s = w[r] * dx
        if s < 1e-15:
            continue
        shift = r - n + 1
        if shift >= 0:
            if shift < n:
                grad[shift:] += s * (1.0 - h[:n - shift])
            end = n - shift
            if end > 0:
                grad[:end] -= s * h[shift:]
        else:
            g = -shift
            end = n - g
            if end > 0:
                grad[:end] += s * (1.0 - h[g:])
            if end > 0:
                grad[g:] -= s * h[:end]
    return grad

def _softmax_weights(scores, gamma):
    m = float(np.max(scores))
    e = np.exp(gamma * (scores - m))
    s = float(np.sum(e))
    if s < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
        return w
    return e / s

def _grad_step(h, cur_c5, gamma, deadline):

    scores = _scores_vec(h)
    w = _softmax_weights(scores, gamma)
    grad = _weighted_grad(h, w)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h, cur_c5
    p = -grad / gn
    t_lo = 0.0
    t_hi = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            t_hi = min(t_hi, (1.0 - h[j]) / p[j])

            t_hi = min(t_hi, h[j] / (-p[j]))
    t_hi = min(t_hi, 0.5)
    if t_hi < 1e-12:
        return h, cur_c5
    h[i] = h[i]  # no-op to avoid closure
    mid_a = 0.5 * t_hi
    orig = h.copy()
    h_new = orig + mid_a * p
    f_mid = _exact_c5(h_new)

        return h, cur_c5
    lo, hi = 0.0, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0

def _binary_sa(n, rng, deadline, max_iter=30000):

        i = ones[rng.integers(len(ones))]

        h[i] = 0.0
        h[j] = 1.0

            zeros.append(int(i))
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool

def _resample(h, n_new):

    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)

def _pair_pass(h, cur_c5, grad, rng, n_pairs, deadline, n_eval=6):

    n_high = min(n // 4, 32)

    highs = order[-n_high:]
    tried = set()
    attempts = 0
    accepted = 0

            break
        if rng.random() < 0.7:

            i = int(rng.integers(n))
            j = int(rng.integers(n))
        if i == j:

        if time.time() > dl3:
            break
        h_res = _resample(best_h, n_try)
        cur_c = _exact_c5(h_res)
        # Brief optimization at new resolution
        scores = _scores_vec(h_res)
        w = _softmax_weights(scores, 40.0)
        grad = _weighted_grad(h_res, w)
        grad -= grad.mean()
        sub_dl = min(dl3, time.time() + budget_s * 0.04)
        h_res, cur_c, _, _ = _pair_pass(h_res, cur_c, grad, rng, 3000, sub_dl, n_eval=6)
        # Gradient polish
        h_res, cur_c = _grad_step(h_res, cur_c, 80.0, sub_dl)
        if cur_c < best_c5 - 1e-8:
            best_c5 = cur_c
            best_h = h_res.copy()
            best_n = n_try

    # Phase 4: Binary seeds
    dl4 = t0 + budget_s * 0.90
    for n_bin in [128, 96]:
        if time.time() > dl4:
            break
        sub_dl = min(dl4, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_sa(n_bin, rng, sub_dl, max_iter=25000)
        if c5_b < best_c5 + 0.005:
            # Refine binary solution continuously
            h_b = h_b.astype(float)
            scores = _scores_vec(h_b)
            w = _softmax_weights(scores, 30.0)
            grad = _weighted_grad(h_b, w)
            grad -= grad.mean()
            sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
            h_b, c5_b2, _, _ = _pair_pass(h_b, c5_b, grad, rng, 3000, sub_dl2, n_eval=6)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b.copy()
                best_n = n_bin

    # Phase 5: Quasi-periodic seeds
    if time.time() < t0 + budget_s * 0.93:
        dl5 = min(t0 + budget_s * 0.93, deadline)
        for alpha in [0.6180339887, 0.4142135624, 0.7071067812]:
            if time.time() > dl5:
                break
            for n_qp in [128, 160]:
                if time.time() > dl5:
                    break
                h_qp = _quasi_construct(n_qp, alpha)
                c5_qp = _exact_c5(h_qp)
                if c5_qp < best_c5 + 0.01:
                    scores = _scores_vec(h_qp)
                    w = _softmax_weights(scores, 30.0)
                    grad = _weighted_grad(h_qp, w)
                    grad -= grad.mean()
                    sub_dl = min(dl5, time.time() + budget_s * 0.02)
                    h_qp, c5_qp, _, _ = _pair_pass(h_qp, c5_qp, grad, rng, 2000, sub_dl, n_eval=6)
                    if c5_qp < best_c5:
                        best_c5 = c5_qp
                        best_h = h_qp.copy()
                        best_n = n_qp

    # Phase 6: Final aggressive polish
    if time.time() < deadline:
        h_work = best_h.copy()
        cur_c5 = best_c5
        n = len(h_work)
        gamma = 100.0
        for _ in range(30):
            if time.time() > deadline:
                break
            scores = _scores_vec(h_work)
            w = _softmax_weights(scores, gamma)
            grad = _weighted_grad(h_work, w)
            grad -= grad.mean()
            h_work, cur_c5, imp, _ = _pair_pass(h_work, cur_c5, grad, rng, 5000, deadline, n_eval=8)
            if not imp:
                h_new, c5_new = _grad_step(h_work, cur_c5, gamma, deadline)
                if c5_new < cur_c5 - 1e-14:
                    h_work = h_new
                    cur_c5 = c5_new
                gamma = min(gamma * 1.2, 500.0)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_work.copy()

    # Final: ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
