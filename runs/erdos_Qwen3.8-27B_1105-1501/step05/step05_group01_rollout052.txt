        if s >= 0:
            m = n - s
            # First term: h[i] for i in [0, m), contribution (1-h[i+s])
            grad[:m] += w * (1.0 - h[s:s+m])
            # Second term: (1-h[i+s]) for i in [0, m), so h[j] appears with j in [s, n)
            # d/d(h[j]) of h[j-s]*(1-h[j]) = -h[j-s]
            grad[s:] += w * (-h[:m])
        else:
            g = -s  # g > 0
            m = n - g
            # corr at lag s: sum_i h[i] * (1-h[i-s]) where i in [g, n)
            # = sum_i h[i] * (1-h[i-g])
            # First term: h[i] for i in [g, n), contribution (1-h[i-g])
            grad[g:] += w * (1.0 - h[:m])
            # Second term: (1-h[i-g]) for i in [g, n), so h[j] appears with j in [0, m)
            # d/d(h[j]) of h[j+g]*(1-h[j]) = -h[j+g]
            grad[:m] += w * (-h[g:g+m])
    
    return -grad  # negative subgradient (descent direction)

def _subgrad_optimize(h, deadline, n_iter=5000, n_active=15, base_step=0.02):
    """Optimize using subgradient method with projection and momentum."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    h_best = h.copy()
    c5_cur = _exact_c5(h_cur)
    c5_best = c5_cur
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    step_scale = base_step
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        scores = _scores(h_cur)
        grad = _subgradient_step(h_cur, scores, n_active=n_active, step_size=1.0)
        
        # Nesterov momentum
        momentum = mu * momentum - step_scale * grad
        h_new = _project(h_cur + momentum, target)
        
        c5_new = _exact_c5(h_new)
        if c5_new < c5_cur - 1e-14:
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()
            step_scale = min(step_scale * 1.01, 0.1)
        else:
            step_scale *= 0.95
            if step_scale < 0.001:
                step_scale = 0.001
    
    return h_best, c5_best

def _coordinate_descent(h, deadline, max_passes=20):
    """Optimize each coordinate while adjusting one other to maintain sum."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _exact_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Find j to pair with (nearest with opposite tendency)
            # Try all j? Too slow. Just try a few.
            # Actually, let's try a simpler approach: just optimize h[i] and project
            # But projection changes all values. Instead, do a two-coordinate update.
            # Pick j = (i+1) % n as partner
            j = (i + 1) % n
            if i == j:
                continue
            
            # We want to find delta such that h[i]+delta and h[j]-delta maintain sum
            # and reduce c5. This is a 1D optimization over delta.
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            t, c5_try = _line_search(h, d, dx, 0.4, n_samples=20)
            if c5_try < c5_cur - 1e-14:
                h = h + t * d
                h = _project(h, target)
                c5_cur = _exact_c5(h)
                if c5_cur < c5_best:
                    c5_best = c5_cur
                    h_best = h.copy()
                improved = True
        if not improved:
            break
    
    return h_best, c5_best

def _block_move(h, deadline, block_size=3, n_tries=3000):
    """Try to move blocks of consecutive elements."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _exact_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    rng = np.random.default_rng(777)
    stall = 0
    
    while time.time() < deadline:
        if stall > 200:
            break
        # Pick a random block and try to swap with another
        start1 = int(rng.integers(0, n - block_size + 1))
        start2 = int(rng.integers(0, n - block_size + 1))
        if abs(start1 - start2) < block_size:
            stall += 1
            continue
        
        d = np.zeros(n)
        for k in range(block_size):
            d[start1 + k] = 1.0  # or -1.0
            d[start2 + k] = -1.0
        
        t, c5_try = _line_search(h, d, dx, 0.5, n_samples=30)
        if c5_try < c5_cur - 1e-14:
            h = h + t * d
            h = _project(h, target)
            c5_cur = _exact_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            stall = 0
        else:
            stall += 1
            # Try the opposite direction
            t, c5_try = _line_search(h, -d, dx, 0.5, n_samples=30)
            if c5_try < c5_cur - 1e-14:

                h = _project(h, target)
                c5_cur = _exact_c5(h)
                if c5_cur < c5_best:
                    c5_best = c5_cur
                    h_best = h.copy()
                stall = 0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_direction(h, scores, radius):
    """Find direction via LP that reduces all scores."""
    n = len(h)
    # Build gradient matrix G (2n-1 x n)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:] += -h[:m]
            # Wait, let me re-derive. For lag d >= 0:
            # score[d] = sum_{i=0}^{n-d-1} h[i] * (1 - h[i+d])
            # d(score)/d(h[j]) = (1-h[j+d]) for j in [0, n-d) [first factor]
            #                   + (-h[j-d]) for j in [d, n) [second factor, i=j-d]
            m = n - d
            G[r, :m] += (1.0 - h[d:d+m])  # (1-h[j+d]) for j=0..m-1
            G[r, d:d+m] += (-h[:m])  # -h[j-d] for j=d..n-1, i.e., -h[i] for i=0..m-1
        else:
            g = -d
            m = n - g
            # score at lag -g: sum_{i=g}^{n-1} h[i] * (1 - h[i-g])
            # d/d(h[j]) = (1-h[j-g]) for j in [g, n) [first factor, i=j]
            #           + (-h[j+g]) for j in [0, n-g) [second factor, i=j+g]
            G[r, g:g+m] += (1.0 - h[:m])  # (1-h[j-g]) for j=g..n-1, j-g in 0..m-1
            G[r, :m] += (-h[g:g+m])  # -h[j+g] for j=0..m-1, j+g in g..n-1
    
    # LP: min t s.t. G @ d <= -scores + t, |d| <= radius, sum(d) = 0
    # Actually: min t s.t. G @ d + scores <= t (for all lags), 0 <= t
    # With sum(d) = 0 and |d_i| <= radius
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores  # G@d - t <= -scores, i.e., G@d + scores <= t
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] * 0.5)  # keep in [0,1] range
        hi = min(radius, (1.0 - h[j]) * 0.5)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()

def _binary_optimize(h, deadline):
    """Convert to binary and optimize via swaps."""
    n = len(h)
    # Round to binary
    idx = np.argsort(-h)[:n//2]

            c5_cur = c5_try
            if c5_cur < c5_best:

def _logsumexp_grad(h, beta=100):
    n = len(h)
    scores = _scores(h) * (2.0 / n)  # scaled scores
    # logsumexp(scores * beta) / beta
    # gradient = softmax(beta * scores) * grad(scores) * (2/n)
    max_s = scores.max()
    exp_s = np.exp(beta * (scores - max_s))
    weights = exp_s / exp_s.sum()
    
    # Compute weighted gradient of scores
    # (same as _subgradient_step but with arbitrary weights)
    grad = np.zeros(n)
    for r in range(len(scores)):
        if weights[r] < 1e-12:
            continue

        if s >= 0:

            grad[s:s+m] += w * (-h[:m])

            g = -s
            m = n - g
            grad[g:g+m] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[g:g+m])
    
    return grad

    dx = 2.0 / n
    S0 = _scores(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_min = -R
    t_max = R
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_min = max(t_min, float(np.max(-h[pos] / d[pos])))
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_min = max(t_min, float(np.max((1.0 - h[neg]) / d[neg])))
        t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
    
    if t_max - t_min < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Sample
    ts = np.linspace(t_min, t_max, n_samples)
    vals = np.array([float(np.max(S0 + t * T - t*t * C)) * dx for t in ts])
    bi = int(np.argmin(vals))
    f_best = vals[bi]
    t_best = ts[bi]
    
    # Golden section
    lo_i = max(0, bi - 3)
    hi_i = min(n_samples - 1, bi + 3)
    t_lo, t_hi = ts[lo_i], ts[hi_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    if f1 < f_best: f_best, t_best = f1, c1
    if f2 < f_best: f_best, t_best = f2, c2
    
    for _ in range(25):
        if t_hi - t_lo < 1e-10: break
        if f1 < f2:
            t_hi, c2, f2 = c2, c1, f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
            if f1 < f_best: f_best, t_best = f1, c1
        else:
            t_lo, c1, f1 = c1, c2, f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
            if f2 < f_best: f_best, t_best = f2, c2
    
    return t_best, f_best

def _logsumexp_optimize(h, deadline, beta=200, base_lr=0.02, momentum=0.85):
    """Optimize using logsumexp smooth approximation with gradient descent."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _exact_c5(h_cur)
    h_best = h.copy()
    c5_best = c5_cur
    vel = np.zeros(n)
    lr = base_lr
    stall = 0
    
    while time.time() < deadline:
        if stall > 500:
            break
        # Compute smooth gradient
        scores = _scores(h_cur)
        scaled = scores * dx  # This is what we're maximizing
        max_s = scaled.max()
        exp_s = np.exp(beta * (scaled - max_s))
        sw = exp_s / exp_s.sum()  # softmax weights
        
        # Gradient of the smooth objective = sum_r sw[r] * grad_r
        # Use the gradient matrix approach but only for active (high-weight) lags
        G = _grad_scores(h_cur)
        # grad of smooth obj = G.T @ sw (but G is for unscaled scores, so multiply by dx)
        grad_smooth = G.T @ (sw * dx)
        
        # Gradient descent with momentum
        vel = momentum * vel + grad_smooth
        h_new = _project(h_cur - lr * vel, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < c5_cur - 1e-14:
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()
            stall = 0
            lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.93
            stall += 1
            if lr < 0.002:
                lr = 0.002
                if stall > 200:
                    break
    
    return h_best, c5_best

def _lp_refine(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _exact_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 35:
            break
        scores = _scores(h)
        G = _grad_scores(h)
        
        # LP: find direction d that minimizes max t s.t. G@d <= -scores + t
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -0.5*h[j]), min(radius, 0.5*(1-h[j]))) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n]
            else:
                d = None
        except Exception:
            d = None
        
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            stall += 1
            radius *= 0.6
            if radius < 1e-5:
                break
            continue
        
        t, c5_try = _line_search(h, d, dx, radius * 10, n_samples=25)
        if c5_try < c5_cur - 1e-13:
            h = h + t * d
            h = _project(h, target)
            c5_cur = _exact_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.72
            if stall % 12 == 0 and radius < 0.02:
                radius = 0.06
    
    return h_best, c5_best

def _pair_swap_optimize(h, deadline, n_trials=5000):
    """Randomized pair swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _exact_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    rng = np.random.default_rng(1234)
    stall = 0
    
    while time.time() < deadline:
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        t, c5_try = _line_search(h, d, dx, 0.5, n_samples=15)
        if c5_try < c5_cur - 1e-14:
            h = h + t * d
            h = _project(h, target)
            c5_cur = _exact_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            stall = 0
        else:
            stall += 1
    
    return h_best, c5_best

def _block_optimize(h, deadline, block_size=3):
    """Block swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _exact_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    rng = np.random.default_rng(5678)
    stall = 0
    
    while time.time() < deadline:
        if stall > 200:
            break
        b = block_size
        if b >= n:
            break
        s1 = int(rng.integers(0, n - b + 1))
        s2 = int(rng.integers(0, n - b + 1))
        if abs(s1 - s2) < b:
            stall += 1
            continue
        d = np.zeros(n)
        d[s1:s1+b] = 1.0
        d[s2:s2+b] = -1.0
        t, c5_try = _line_search(h, d, dx, 0.5, n_samples=20)
        if c5_try < c5_cur - 1e-14:
            h = h + t * d
            h = _project(h, target)
            c5_cur = _exact_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            stall = 0
        else:
            # Try reverse
            t2, c5_try2 = _line_search(h, -d, dx, 0.5, n_samples=20)
            if c5_try2 < c5_cur - 1e-14:
                h = h - t2 * d
                h = _project(h, target)
                c5_cur = _exact_c5(h)
                if c5_cur < c5_best:
                    c5_best = c5_cur
                    h_best = h.copy()
                stall = 0
            else:
                stall += 1
    
    return h_best, c5_best

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
