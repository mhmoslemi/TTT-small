from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
            if np.sum(mask) == 0:
                mask = x < 1.0
        else:
            mask = (x > 0.0) & (x < 1.0)
            if np.sum(mask) == 0:
                mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

    """Compute p-norm objective and gradient."""
    n = len(h)
    F = _scores(h)  # all non-negative
    # Use sum of F^q as objective (same minimizer as (sum F^q)^(1/q))
    if q == 1:

        # gradient of sum F_r = G.T @ ones
        G = _build_G(h)
        grad = G.sum(axis=0)
        return val, grad
    Fq = F ** q

    G = _build_G(h)
    weights = q * F ** (q - 1)
    grad = G.T @ weights
    return val, grad

    n = len(h)
    n_calls = [0]
    deadline_ref = [deadline]

        if time.time() > deadline_ref[0]:
            # Return large value to signal stop
            return 1e10
        val, grad = _pnorm_obj_grad(x, q)
        # Store gradient for SLSQP
        obj.last_grad = grad
        return val

        n_calls[0] += 1
        if time.time() > deadline_ref[0]:
            return 1e10, np.zeros(n)
        val, grad = _pnorm_obj_grad(x, q)
        return val, grad
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    
    try:

                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
        if res.success or (res.fun < 1e10):
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    
    # Fallback: projected gradient
    h_cur = h.copy()
    step = 0.05
    for _ in range(200):
        if time.time() > deadline:
            break
        F = _scores(h_cur)
        G = _build_G(h_cur)
        weights = q * F ** (q - 1)
        grad = G.T @ weights
        grad -= grad.mean()  # project out equality constraint
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        h_new = _project(h_cur - step * grad, target)
        val_new = float(np.sum(_scores(h_new) ** q))
        val_cur = float(np.sum(_scores(h_cur) ** q))
        if val_new < val_cur:
            h_cur = h_new
            step = min(step * 1.1, 0.2)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return h_cur

        return np.zeros(n)

        return np.ones(n)

            x = np.clip(x, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lse_grad(h, beta):
    """Compute gradient of log-sum-exp surrogate."""
    n = len(h)
    F = _scores(h)
    F = np.maximum(F, 0.0)
    max_F = float(F.max())
    shifted = beta * (F - max_F)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = float(exp_s.sum())
    if sum_exp < 1e-300:
        # Uniform weights
        weights = np.ones(len(F)) / len(F)
    else:
        weights = exp_s / sum_exp
    G = _build_G(h)
    return G.T @ weights

def _lse_optimize(h, beta_list, deadline, target, momentum=0.9):
    """LSE gradient descent with continuation on beta."""
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h_cur)
    v = np.zeros(n)  # momentum
    
    for beta in beta_list:
        if time.time() > deadline:
            break
        step = 0.02
        stall = 0
        for it in range(500):
            if time.time() > deadline:
                break
            grad = _lse_grad(h_cur, beta)
            # Project out equality constraint
            grad_flat = float(grad.mean())
            grad -= grad_flat
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            grad /= gn
            
            # Momentum
            v = momentum * v - step * grad
            h_new = _project(h_cur + v, target)
            
            # Check improvement (using exact C5)
            c5_new = _exact_c5(h_new)
            c5_cur = _exact_c5(h_cur)
            
            if c5_new < c5_cur - 1e-12:
                h_cur = h_new
                stall = 0
                step = min(step * 1.05, 0.1)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
            else:
                v = 0.5 * v
                step *= 0.7
                stall += 1
                if stall > 30:
                    break
                if step < 1e-8:
                    break
        # After each beta, check and reset momentum
        v = np.zeros(n)
    
    return best_h, best_c5
