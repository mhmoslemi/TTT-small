def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, 0.0, 1.0)  # initial clip to speed up bisection
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Build G where G[r,j] = d(score_r)/d(h_j). Returns matrix of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    # Zero lag
    G[n-1] = 1.0 - 2.0 * h
    # Positive lags d=1..n-1: score[n-1+d] = sum_{i=0}^{n-1-d} h[i](1-h[i+d])
    for d in range(1, n):
        m = n - d
        G[n-1+d, :m] += (1.0 - h[d:])
        G[n-1+d, d:] += -h[:m]
    # Negative lags d=1..n-1: score[n-1-d] = sum_{i=d}^{n-1} h[i](1-h[i-d])
    for d in range(1, n):
        m = n - d
        G[n-1-d, d:] += (1.0 - h[:m])
        G[n-1-d, :m] += -h[d:]
    return G

    n = len(h)
    target = n / 2.0
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.25
    stall = 0
    
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        scores = _scores_vec(h)
        G = _grad_matrix(h)
        # Use top-k active constraints
        topk = min(30, 2 * n - 1)
        idx = np.argsort(scores)[-topk:]
        scores_k = scores[idx]
        G_k = G[idx]
        
        # LP: minimize t subject to G_k @ p - t <= -scores_k, sum(p)=0, |p_j| <= radius*(1-h_j or h_j)
        # Actually: scores_k + G_k @ (alpha * p) - t <= 0 → G_k @ p - t <= -scores_k
        # With t = -min(G_k @ p + scores_k) = max(-scores_k - G_k @ p)
        # We want to minimize the maximum, so minimize t subject to G_k @ p - t <= -scores_k
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((topk, n + 1))
        A_ub[:, :n] = G_k
        A_ub[:, n] = -1.0
        b_ub = -scores_k
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                t_val = res.x[-1]
            else:
                stall += 1
                radius *= 0.7
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        # Line search
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(15):

            if f1 < f2:
                hi_a = c2; f2 = f1; c2 = c1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo_a = c1; f1 = f2; c1 = c2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(h + c2 * p)
        
        opt_a = (lo_a + hi_a) / 2.0
        new_h = _project(h + opt_a * p, target)
        new_c5 = _exact_c5(new_h)

            cur_c5 = new_c5

            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall > 15 and radius < 0.02:
                radius = 0.08
    
    return best_h, best_c5

def _build_graph_and_solve(n, edge_specs, src, sink):
    """
    n: number of variable nodes (0..n-1)
    Total nodes: n + 2 (src = n, sink = n+1)
    edge_specs: list of (u, v, capacity)
    """
    from collections import deque
    V = n + 2
    SRC = n
    SNK = n + 1
    
    # Adjacency: adj[u] = list of (v, edge_index)
    adj = [[] for _ in range(V)]
    cap = []  # capacity for forward and backward edges
    
    def add_edge(u, v, c):
        if c <= 1e-15:
            return
        idx = len(cap)
        cap.append(c)    # forward
        cap.append(0.0)  # backward
        adj[u].append((v, idx))
        adj[v].append((u, idx + 1))
    
    for u, v, c in edge_specs:
        add_edge(u, v, c)
    
    total_flow = 0.0
    
    while True:
        # BFS
        level = [-1] * V
        level[SRC] = 0
        q = deque([SRC])
        while q:
            u = q.popleft()
            for v, ei in adj[u]:
                if level[v] < 0 and cap[ei] > 1e-15:
                    level[v] = level[u] + 1
                    q.append(v)
        
        if level[SNK] < 0:
            break
        
        # DFS for blocking flow
        iter_arr = [0] * V
        
        def dfs(u, f):
            if u == SNK:
                return f
            for i in range(iter_arr[u], len(adj[u])):
                iter_arr[u] = i
                v, ei = adj[u][i]
                if level[v] > level[u] and cap[ei] > 1e-15:
                    d = dfs(v, min(f, cap[ei]))
                    if d > 1e-15:
                        cap[ei] -= d
                        cap[ei + 1] += d
                        return d
            iter_arr[u] = len(adj[u])
            return 0.0
        
        while True:
            f = dfs(SRC, float('inf'))
            if f < 1e-15:
                break
            total_flow += f

    visited = [False] * V
