    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    q = min(k_init, 2 * n - 1)
    for round_i in range(max_rounds):
        if time.time() > deadline:
            break
        scores = _scores_all(h)
        active_k = np.argsort(-scores)[:q]
        x_res = _epigraph_solve(h, list(active_k), n, dx, deadline, maxiter=150)
        if x_res is None:
            q = max(16, q // 2)
            continue
        h_new = _project(x_res[:n], n / 2.0)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            scores_check = _scores_all(h)
            non_active = np.setdiff1d(np.arange(2 * n - 1), active_k)
            if len(non_active) > 0:
                max_non = float(np.max(scores_check[non_active]))
                if max_non > cur_c5 + 1e-8:
                    q = min(2 * n - 1, q + k_expand)
        else:
            q = max(16, q - 2)
            if q < 16:
                break
        if q >= 2 * n - 1:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return _project(h.copy(), n / 2.0)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h = h + rng.normal(0, 0.05, n)
        h = _project(h, n / 2.0)
    return h

def _perturb_and_optimize(h0, rng, scale, deadline, k_init=24):
    n = len(h0)
    h_pert = h0 + rng.normal(0, scale, n)
    h_pert = _project(h_pert, n / 2.0)
    return _active_set_optimize(h_pert, deadline, k_init=k_init, k_expand=4, max_rounds=15)

def _grad_direction(h, scores, n, dx, top_k=40):
    idx = np.argsort(-scores)[:top_k]
    grad = np.zeros(n)
    for k in idx:
        grad += _score_k_grad(h, k, dx, n)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return -grad / gn

def _grad_descent(h0, deadline, n_steps=200, step_init=0.03):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    v = np.zeros(n)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores_all(h)
        p = _grad_direction(h, scores, n, dx, top_k=64)
        if p is None:
            break
        v = 0.8 * v + step * p
        for mult in [1.0, 0.5, 0.25, 0.1]:
            h_try = _project(h + mult * v, n / 2.0)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                step = min(step * 1.02, 0.08)
                break
        else:
            v = 0.5 * v
            step *= 0.85
            if step < 1e-7:
                break
    return best_h, best_c5

def _coord_polish(h0, deadline, n_cycles=2):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(35):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h0, deadline, n_passes=3):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    step_size = max(1, n // 16)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h0[j] < 1e-8 or h0[j] > 1.0 - 1e-8:
                continue
            for k in range(0, n, step_size):
                if k == j:
                    continue
                if h0[k] < 1e-8 or h0[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h0[j], h0[k])
                    else:
                        a_max = min(h0[j], 1.0 - h0[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(12):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h0.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h0.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h0.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h0 = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_init(n, rng):
    target = n // 2
    pos = rng.permutation(n)
    h = np.zeros(n)
    h[pos[:target]] = 1.0
    return h

def _binary_sa(n, deadline, rng, n_iters=15000):
    target = n // 2
    h = _binary_init(n, rng)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.02
    T_min = 0.0005
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if h[i] == 1.0 and h[j] == 0.0:
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
            T *= cool
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is None or len(h0) == 0 or not np.all(np.isfinite(h0)):
        n0 = 128
        h0 = _quasi_init(n0, 0.6180339887, rng)
    else:
        n0 = len(h0)

    best_h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(best_h)
    best_n = n0

    # Phase 1: SLSQP active-set from initial
    if time.time() < deadline:
        dl1 = min(deadline, t0 + budget_s * 0.20)
        h1, c5_1 = _active_set_optimize(best_h, dl1, k_init=40, k_expand=10, max_rounds=40)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Smoothed gradient descent
    if time.time() < deadline:
        dl2 = min(deadline, t0 + budget_s * 0.30)
        h2, c5_2 = _grad_descent(best_h, dl2, n_steps=250, step_init=0.025)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        if time.time() < deadline:
            dl2b = min(deadline, t0 + budget_s * 0.35)
            h2b, c5_2b = _active_set_optimize(best_h, dl2b, k_init=32, k_expand=8, max_rounds=30)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Multi-resolution
    if time.time() < deadline:
        dl3 = min(deadline, t0 + budget_s * 0.50)
        res_list = []
        for r in [64, 80, 96, 112, 128, 160]:
            if r != n0:
                res_list.append(r)
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5320888862, 0.4755282581]
        for ri, n_try in enumerate(res_list):
            if time.time() > dl3:
                break
            for s in range(2):
                if time.time() > dl3:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    alpha = alphas[ri % len(alphas)]
                    h_init = _quasi_init(n_try, alpha, rng)
                sub_dl = min(dl3, time.time() + budget_s * 0.025)
                h_try, c5_try = _active_set_optimize(h_init, sub_dl, k_init=24, k_expand=6, max_rounds=25)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 4: Another round of gradient + SLSQP
    if time.time() < deadline:
        dl4 = min(deadline, t0 + budget_s * 0.58)
        h4, c5_4 = _grad_descent(best_h, dl4, n_steps=200, step_init=0.02)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        if time.time() < deadline:
            dl4b = min(deadline, t0 + budget_s * 0.63)
            h4b, c5_4b = _active_set_optimize(best_h, dl4b, k_init=48, k_expand=8, max_rounds=35)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: Binary SA for diversity
    if time.time() < deadline:
        dl5 = min(deadline, t0 + budget_s * 0.72)
        for n_bin in [best_n, min(best_n, 96), 64]:
            if time.time() > dl5:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=15000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
                h_br, c5_br = _active_set_optimize(h_b, sub_dl2, k_init=24, k_expand=6, max_rounds=20)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()

    # Phase 6: Perturbation restarts
    if time.time() < deadline:
        dl6 = min(deadline, t0 + budget_s * 0.84)
        scale = 0.03
        for attempt in range(80):
            if time.time() > dl6:
                break
            h_p, c5_p = _perturb_and_optimize(best_h, rng, scale, dl6, k_init=24)
            if c5_p < best_c5 - 1e-12:
                best_c5 = c5_p
                best_h = h_p.copy()
                scale = max(0.005, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.15)

    # Phase 7: Coordinate polish
    if time.time() < deadline:
        dl7 = min(deadline, t0 + budget_s * 0.90)
        hc, c5c = _coord_polish(best_h, dl7, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 8: Pair polish
    if time.time() < deadline:
        dl8 = min(deadline, t0 + budget_s * 0.94)
        hp, c5p = _pair_polish(best_h, dl8, n_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 9: Final SLSQP with many constraints
    if time.time() < deadline:
        dl9 = min(deadline, t0 + budget_s * 0.98)
        h_fin, c5_fin = _active_set_optimize(best_h, dl9, k_init=64, k_expand=16, max_rounds=25)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, float(best_c5), int(best_n)
