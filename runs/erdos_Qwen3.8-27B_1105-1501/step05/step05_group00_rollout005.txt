```python
def _project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

def _line_search(h, d, dx, trust_radius):

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find safe interval
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)

    if t_hi - t_lo < 1e-14:

    # Grid search

    obj_grid = np.max(vals, axis=1) * dx
```

```python
def _min_norm_direction(h, scores, top_k=6):
    n = len(h)
    sorted_idx = np.argsort(-scores)
    top_indices = sorted_idx[:top_k]
    
    # Build gradient matrix
    G = np.zeros((top_k, n))
    for idx, r in enumerate(top_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] += -h[d:d+m]
                G[idx, d:] += (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:g+m])
                G[idx, g:] += -h[:m]
    
    # Solve: G @ p = -1 (minimize ||p||)
    # p = -G^T (G G^T + eps I)^{-1} 1
    one = np.ones(top_k)
    try:
        GGt = G @ G.T + 1e-10 * np.eye(top_k)
        alpha = np.linalg.solve(GGt, one)
        p = -G.T @ alpha
    except np.linalg.LinAlgError:
        return None
    
    # Normalize to have max |p_i| = 1 (will scale later in line search)
    max_abs = np.max(np.abs(p))
    if max_abs > 1e-15:
        p = p / max_abs
    
    return p
```

```python
def _min_norm_direction_zero_sum(h, scores, top_k=8):
    n = len(h)
    sorted_idx = np.argsort(-scores)
    top_indices = sorted_idx[:top_k]
    
    # Build gradient matrix
    G = np.zeros((top_k, n))
    for idx, r in enumerate(top_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] += -h[d:d+m]
                G[idx, d:] += (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:g+m])
                G[idx, g:] += -h[:m]
    
    # Solve KKT system
    one_k = np.ones(top_k)
    G1 = G @ np.ones(n)  # row sums = G @ 1_n
    GGt = G @ G.T  # K x K
    
    # System: [GGt   G1] [alpha]   [one_k]
    #         [G1^T  n ] [lambda] = [0]
    A_sys = np.zeros((top_k + 1, top_k + 1))
    A_sys[:top_k, :top_k] = GGt
    A_sys[:top_k, top_k] = G1
    A_sys[top_k, :top_k] = G1
    A_sys[top_k, top_k] = float(n)
    
    b_sys = np.zeros(top_k + 1)
    b_sys[:top_k] = one_k
    
    try:
        sol = np.linalg.solve(A_sys + 1e-12 * np.eye(top_k + 1), b_sys)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
        except:
            return None
    
    alpha = sol[:top_k]
    lam = sol[top_k]
    
    p = -G.T @ alpha - lam * np.ones(n)
    
    # Verify zero sum (should be very small)
    p -= p.sum() / n
    
    max_abs = np.max(np.abs(p))
    if max_abs > 1e-15:
        p = p / max_abs
    
    return p
```

def _smooth_objective(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    shifted = (corr - M) / tau
    # Numerically stable log-sum-exp
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return M + tau * log_sum
