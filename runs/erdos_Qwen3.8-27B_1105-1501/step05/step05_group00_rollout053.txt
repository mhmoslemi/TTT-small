from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, trust_radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:

    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx

            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2

    if opt_val < best_val:

    return best_t, best_val

def _grad_matrix(h):
    """Compute (2n-1, n) gradient matrix for corr = np.correlate(h, 1-h, 'full')."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        g = abs(d)
        if g == 0:
            G[r, :] = 1.0 - 2.0 * h
        else:
            # For k in [g, n-1]: contribution 1 - h[k-g]
            # For k in [0, n-1-g]: contribution -h[k+g]
            G[r, g:] = 1.0 - h[:n - g]
            G[r, :n - g] += -h[g:]
    return G

def _lp_direction(h, S, G, radius):
    """Find LP direction to reduce max correlation."""
    n = len(h)
    n_lags = 2 * n - 1
    # Variables: d[0..n-1], t
    # minimize t
    # s.t.: G[r] @ d - t <= -S[r] for all r
    #       sum(d) = 0
    #       bounds on d
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: G[r] @ d - t <= -S[r]
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -S  # S[r] + G[r]@d - t <= 0 => G[r]@d - t <= -S[r]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i])
        hi_i = min(radius, 1.0 - h[i])
        bounds.append((lo_i, hi_i))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.1, max_iter=5000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(42)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        S = np.correlate(h, 1.0 - h, mode='full')
        S_max = float(np.max(S))
        G = _grad_matrix(h)
        p = _lp_direction(h, S, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            if stall > 30:
                if restart_ctr < 5:
                    h = h + rng.normal(0, 0.03, n)
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    radius = 0.1
                    stall = 0
                    restart_ctr += 1
                else:
                    break
            continue
        alpha, new_c5 = _line_search(h, p, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Adaptive radius
            S_new = np.correlate(h, 1.0 - h, mode='full')
            S_new_max = float(np.max(S_new))
            predicted = S_max + float(np.max(G @ p))  # linearized prediction
            if predicted < S_new_max * 0.5:
                radius = min(radius * 1.15, 0.3)
            elif S_new_max > S_max + 0.1 * S_max:
                radius *= 0.85
            else:
                radius = min(radius * 1.02, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall > 40:
                break
    return best_h, best_c5
