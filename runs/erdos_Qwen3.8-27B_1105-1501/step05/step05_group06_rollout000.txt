def _c5(h):

def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

    if top_k is not None and top_k < n_lags:

    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)

    A_eq = np.zeros((1, n + 1))

                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > verify_thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, target)
                    actual_c5 = _c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not pass_improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _de_step(h, target, rng, population_size=20, F=0.8, CR=0.7, n_gen=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5(h)
    pop = np.zeros((population_size, n))
    pop[0] = h
    for i in range(1, population_size):
        pop[i] = _project(h + rng.normal(0, 0.15, n), target)
    pop_c5 = np.array([_c5(pop[i]) for i in range(population_size)])
    best_idx = int(np.argmin(pop_c5))
    if pop_c5[best_idx] < best_c5:
        best_c5 = pop_c5[best_idx]
        best_h = pop[best_idx].copy()
    for _ in range(n_gen):
        if time.time() > 99999:
            break
        for i in range(population_size):
            a, b, c = rng.choice(population_size, 3, replace=False)
            if i in (a, b, c):
                a, b, c = (a + 1) % population_size, (b + 1) % population_size, (c + 1) % population_size
            v = pop[a] + F * (pop[b] - pop[c])
            v = np.where(rng.random(n) < CR, v, pop[i])
            v = _project(v, target)
            v_c5 = _c5(v)
            if v_c5 < pop_c5[i]:
                pop[i] = v
                pop_c5[i] = v_c5
                if v_c5 < best_c5:
                    best_c5 = v_c5
                    best_h = v.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h_start = _project(h0, target0)
    best_h = h_start.copy()
    best_c5 = _c5(h_start)
    best_n = n0

    # Phase 1: Quick LP at original resolution
    dl1 = t0 + budget_s * 0.06
    h1, c5_1 = _local_opt(h_start, dl1, target0, top_k_list=[None, 10, 6, 4],
                          radius_init=0.15, max_iters=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Try multiple resolutions with diverse starts
    dl2 = t0 + budget_s * 0.42
    resolutions = [192, 256, 320, 384, 448, 512, 160]
    golden = 0.6180339887
    alphas_list = [golden, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.555, 0.6823278]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl2:
            break
        target_try = n_try / 2.0
        n_starts = 2
        for s in range(n_starts):
            if time.time() > dl2:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas_list[ri % len(alphas_list)]
                h_init = _quasi_init(n_try, alpha, rng)
            sub_dl = min(dl2, time.time() + budget_s * 0.045)
            h_try, c5_try = _local_opt(h_init, sub_dl, target_try,
                                       top_k_list=[None, 10, 6, 4, 3],
                                       radius_init=0.12, max_iters=3000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                # Polish with gradient
                if time.time() < dl2:
                    sub_dl2 = min(dl2, time.time() + budget_s * 0.02)
                    h_g, c5_g = _grad_step(best_h, sub_dl2, best_n / 2.0,
                                           n_steps=80, beta_start=30, beta_end=300)
                    if c5_g < best_c5:
                        best_c5 = c5_g
                        best_h = h_g.copy()
                        best_n = n_try
            else:
                if time.time() < dl2:
                    sub_dl2 = min(dl2, time.time() + budget_s * 0.012)
                    h_g, c5_g = _grad_step(h_try, sub_dl2, target_try,
                                           n_steps=60, beta_start=25, beta_end=200)
                    if c5_g < best_c5:
                        best_c5 = c5_g
                        best_h = h_g.copy()
                        best_n = n_try

    # Phase 3: DE at best resolution
    if time.time() < t0 + budget_s * 0.58:
        dl3 = t0 + budget_s * 0.60
        target_b = best_n / 2.0
        h_de, c5_de = _de_step(best_h, target_b, rng,
                               population_size=16, F=0.7, CR=0.6, n_gen=30)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
        # Quick LP polish on DE result
        if time.time() < dl3:
            h_lp, c5_lp = _local_opt(best_h, dl3, target_b,
                                     top_k_list=[8, 5, None],
                                     radius_init=0.08, max_iters=1500)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 4: Intensive pair descent
    if time.time() < t0 + budget_s * 0.75:
        dl4 = t0 + budget_s * 0.77
        target_b = best_n / 2.0
        best_h, best_c5 = _pair_descent(best_h, dl4, target_b,
                                        max_passes=300, verify_thresh=1e-5)
        # Follow-up LP
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_lp, c5_lp = _local_opt(best_h, sub_dl, target_b,
                                     top_k_list=[6, 4, 3, None],
                                     radius_init=0.04, max_iters=1000)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
        # More pair descent
        if time.time() < dl4:
            dl4b = min(dl4, time.time() + budget_s * 0.025)
            best_h, best_c5 = _pair_descent(best_h, dl4b, target_b,
                                            max_passes=150, verify_thresh=1e-7)

    # Phase 5: Diverse restarts at multiple resolutions
    if time.time() < t0 + budget_s * 0.88:
        dl5 = t0 + budget_s * 0.89
        for n_try in [256, 320, 192, 384]:
            if time.time() > dl5:
                break
            target_try = n_try / 2.0
            for attempt in range(4):
                if time.time() > dl5:
                    break
                alpha = alphas_list[(attempt + ri) % len(alphas_list)] if ri < len(alphas_list) else alphas_list[attempt]
                h_init = _quasi_init(n_try, alpha, rng)
                sub_dl = min(dl5, time.time() + budget_s * 0.03)
                h_try, c5_try = _local_opt(h_init, sub_dl, target_try,
                                           top_k_list=[None, 8, 5, 3],
                                           radius_init=0.15, max_iters=2500)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                    sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
                    h_p, c5_p = _pair_descent(best_h, sub_dl2, best_n / 2.0,
                                              max_passes=100, verify_thresh=1e-6)
                    if c5_p < best_c5:
                        best_c5 = c5_p
                        best_h = h_p.copy()
                else:
                    sub_dl2 = min(dl5, time.time() + budget_s * 0.01)
                    h_g, c5_g = _grad_step(h_try, sub_dl2, target_try,
                                           n_steps=70, beta_start=40, beta_end=400)
                    if c5_g < best_c5:
                        best_c5 = c5_g
                        best_h = h_g.copy()
                        best_n = n_try

    # Phase 6: Perturbation-based escape
    dl6 = t0 + budget_s * 0.945
    target_b = best_n / 2.0
    scale = 0.03
    for attempt in range(300):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, target_b)
        sub_dl = min(dl6, time.time() + budget_s * 0.006)
        h_try, c5_try = _local_opt(h_pert, sub_dl, target_b,
                                   top_k_list=[5, 3, None],
                                   radius_init=0.04, max_iters=300)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.82)
        else:
            scale = min(0.09, scale * 1.12)

    # Phase 7: Final intensive polish
    if time.time() < hard_deadline:
        dl7 = hard_deadline
        target_b = best_n / 2.0
        # Very tight pair descent
        h_pd, c5_pd = _pair_descent(best_h, dl7, target_b,
                                    max_passes=200, verify_thresh=1e-8)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
        # Tiny LP steps
        if time.time() < dl7 - 3:
            h_fin, c5_fin = _local_opt(best_h, dl7, target_b,
                                       top_k_list=[3, 4, 5, 6],
                                       radius_init=0.01, max_iters=500)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        # One more pair descent pass
        if time.time() < dl7 - 1:
            h_pd2, c5_pd2 = _pair_descent(best_h, dl7, target_b,
                                          max_passes=50, verify_thresh=1e-9)
            if c5_pd2 < best_c5:
                best_c5 = c5_pd2
                best_h = h_pd2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
