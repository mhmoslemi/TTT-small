def _grad_rows_selected(h, scores, k_active):
    n = len(h)
    total = 2 * n - 1
    if total > k_active:
        top_idx = np.argsort(-scores)[:k_active]
    else:
        top_idx = np.arange(total)

    n_cons = G.shape[0]

def _subgrad_direction(h, scores, radius):
    """Projected subgradient step on the active set."""
    n = len(h)
    max_val = float(np.max(scores))
    active = np.where(scores > max_val - 0.3 * (max_val - float(np.min(scores))))[0]
    G = np.zeros((len(active), n))
    for i, r in enumerate(active):
        lag = r - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                G[i, lag:] += (1.0 - h[:m])
                G[i, :m] += -h[lag:]
        else:
            g = -lag
            m = n - g
            if m > 0:
                G[i, :m] += (1.0 - h[g:])
                G[i, g:] += -h[:m]
    # Average gradient of active lags
    d = -G.mean(axis=0)
    # Project onto sum=0
    d -= d.mean()
    # Clip to radius
    d = np.clip(d, -radius, radius)
    # Re-project sum
    d -= d.mean()
    if np.max(np.abs(d)) < 1e-14:

                alpha, c5_try = _line_search(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                    break
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _smooth_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = (corr - c_max) / tau
    shifted = np.clip(shifted, -50, 50)
    log_sum = c_max + tau * np.log(np.sum(np.exp(shifted)))
    return log_sum

def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = np.clip((corr - c_max) / tau, -50, 50)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient of correlation w.r.t. h
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        lag = r - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                G[r, lag:] += (1.0 - h[:m])
                G[r, :m] += -h[lag:]
        else:
            g = -lag
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    grad = (weights[:, None] * G * dx).sum(axis=0)
    return grad

def _smooth_optimize(h_init, deadline, tau_init=None):
    """Optimize smooth surrogate using projected gradient."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_smooth = _smooth_obj(h, 0.02)
    tau = tau_init if tau_init else 0.03
    lr = 0.02
    stall = 0
    while time.time() < deadline:
        if tau < 1e-5:
            break
        for _inner in range(50):
            if time.time() > deadline:
                break
            grad = _smooth_grad(h, tau)
            # Project gradient onto sum=0 subspace
            grad -= grad.mean()
            h_new = h - lr * grad
            h_new = _project(h_new, target)
            s_new = _smooth_obj(h_new, tau)
            if s_new < _smooth_obj(h, tau):
                h = h_new
                # Evaluate exact
                c5_exact = _exact_c5(h)
                if c5_exact < best_smooth - 0.001:
                    best_smooth = c5_exact
                    best_h = h.copy()
                stall = 0
            else:
                lr *= 0.85
                stall += 1
        if stall > 20:
            break
        tau *= 0.7
        lr = 0.02
    return best_h, best_smooth

def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def _greedy_binary(n, rng, seed_tries=5):
    """Greedy: add positions to minimize max correlation."""
    best_h = None
    best_c5 = float('inf')
    for _ in range(seed_tries):
        h = np.zeros(n)
        ones = set()
        target = n // 2
        dx = 2.0 / n
        scored = np.zeros(n)
        for _step in range(target):
            # For each empty position, estimate impact
            best_score = float('inf')
            best_pos = -1
            for pos in range(n):
                if pos in ones:
                    continue
                h[pos] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                sc = float(np.max(corr))
                h[pos] = 0.0
                if sc < best_score:
                    best_score = sc
                    best_pos = pos
            h[best_pos] = 1.0
            ones.add(best_pos)
        c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 120
        h0 = _random_binary(n0, rng)

    best_h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: Quick LP on initial (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.15, k_active=60)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Wide multi-resolution search (40%)
    d2 = t0 + budget_s * 0.50
    n_current = len(best_h)
    # Try a range of n values
    n_candidates = []
    for dn in range(-20, 25, 2):
        nt = n_current + dn
        if 36 <= nt <= 160 and nt % 2 == 0:
            n_candidates.append(nt)
    n_candidates.sort(key=lambda x: abs(x - n_current))

    for nt in n_candidates:
        if time.time() > d2:
            break
        if nt == n_current:
            continue
        h_r = _resample(best_h, nt)
        sub_dl = min(d2, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, k_active=45)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Deep LP on best (15%)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.18, k_active=70)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + polish (15%)
    d4 = t0 + budget_s * 0.80
    h4a, c5_4a = _binary_sa(best_h, min(d4, t0 + budget_s * 0.10), rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, radius_init=0.10, k_active=50)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Smooth surrogate exploration (10%)
    d5 = t0 + budget_s * 0.90
    # Try smooth optimization from current best
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h5a, c5_5a = _smooth_optimize(best_h, sub_dl, tau_init=0.02)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    # Polish smooth result
    if time.time() < d5:
        h5b, c5_5b = _lp_loop(best_h, d5, radius_init=0.08, k_active=50)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Perturbation + re-opt (8%)
    d6 = t0 + budget_s * 0.95
    scale = 0.03
    for attempt in range(200):
        if time.time() > d6:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.07, k_active=45)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.12)

    # Phase 7: Pair refinement (5%)
    d7 = deadline
    h7, c5_7 = _pair_refine(best_h, d7, max_rounds=100)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final: one more LP polish
    h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.04, k_active=30)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
