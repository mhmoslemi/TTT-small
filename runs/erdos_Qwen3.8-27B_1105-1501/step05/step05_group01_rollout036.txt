import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj_grad(h, tau):
    """LSE surrogate objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(scores)
    z = np.exp(np.asarray((scores - M) / tau))
    S = np.sum(z)
    obj = (M + tau * np.log(S)) * dx
    w = z / S
    
    grad = np.zeros(n)
    for d in range(0, n):
        w_pos = w[n - 1 + d]
        w_neg = w[n - 1 - d] if d > 0 else 0.0
        if d == 0:
            # scores[n-1] = sum_i h[i]*(1-h[i]) = sum_i h[i] - sum_i h[i]^2
            # d/dh[j] = 1 - 2*h[j]
            if w_pos > 1e-15:
                grad[:] += w_pos * (1.0 - 2.0 * h)
        else:
            m = n - d
            if m <= 0:
                continue
            if w_pos > 1e-15:
                # S_d = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
                grad[d:] += w_pos * (1.0 - h[:m])
                grad[:m] += w_pos * (-h[d:])
            if w_neg > 1e-15:
                # S_{-d} = sum_{i=0}^{m-1} h[i]*(1-h[i+d])
                grad[:m] += w_neg * (1.0 - h[d:])
                grad[d:] += w_neg * (-h[:m])
    
    grad *= dx
    return obj, grad

def _line_search_quad(h, d, dx, radius):
    """Exact line search using quadratic model along direction d with sum(d)=0."""
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible interval
    t_lo, t_hi = -radius, radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Dense grid search
    n_grid = 33
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([float(np.max(S0 + t * T - t*t * Q)) * dx for t in ts])
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Refine with golden section around best
    if best_idx > 0 and best_idx < n_grid - 1:
        a, b = ts[best_idx - 1], ts[best_idx + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    for _ in range(30):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    best_t = (a + b) / 2.0
    best_val = float(np.max(S0 + best_t * T - best_t*best_t * Q)) * dx
    
    return best_t, best_val

def _lse_descent(h, deadline, tau_init=0.02, n_iter_max=500):
    """LSE-based gradient descent with line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        obj, grad = _lse_obj_grad(h, tau)
        
        # Project gradient to feasible subspace (sum = 0)
        grad_dir = grad - np.mean(grad)
        norm = np.max(np.abs(grad_dir))
        if norm < 1e-12:
            break
        
        # Line search with projected steps
        # We need to move along -grad_dir while staying feasible
        # Use projected line search
        d = -grad_dir / norm  # normalize to max abs = 1
        
        alpha, new_c5 = _line_search_quad(h, d, dx, radius=0.5)
        
        if new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                tau = max(0.002, tau * 0.95)
            else:
                # Projected step didn't help, try smaller
                tau = min(0.1, tau * 1.3)
        else:
            tau = min(0.1, tau * 1.2)
            if tau > 0.08:
                break
    
    return best_h, best_c5

def _slsqp_lse(h, deadline, tau=0.015):
    """SLSQP optimization using LSE surrogate."""
    n = len(h)
    target = n / 2.0
    
    def obj_func(x):
        return _lse_obj_grad(np.clip(x, 0, 1), tau)[0]
    
    def obj_grad(x):
        return _lse_obj_grad(np.clip(x, 0, 1), tau)[1]
    
    def cons_func(x):
        return np.sum(x) - target
    
    def cons_grad(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': cons_func, 'jac': cons_grad}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=obj_grad,
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-10})
        if res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    
    cur_c5 = _exact_c5(h)
    return h, cur_c5

def _lp_direction(h, scores, radius):
    """Find best direction via LP (simplified from previous version)."""
    n = len(h)
    dx = 2.0 / n
    
    # Simplified: find direction that most reduces the max score
    # Use the LP formulation
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    
    # LP: min t s.t. G*d - t*ones <= -scores, |d[i]| <= 1, sum(d) = 0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, rng, seed_offset=0):
    """Create structured initial patterns."""
    h = np.zeros(n)
    mode = seed_offset % 5
    if mode == 0:
        # Fourier-based
        freqs = rng.uniform(0.5, 4.0, size=4)
        phases = rng.uniform(0, 2*np.pi, size=4)
        x = np.linspace(0, 2, n)
        for f, p in zip(freqs, phases):
            h += np.sin(2*np.pi*f*x/2 + p)
    elif mode == 1:
        # Periodic pattern
        period = rng.integers(n//4, n//2 + 1)
        pattern = rng.random(period) > 0.5
        reps = n // period + 1
        h = np.tile(pattern, reps)[:n].astype(float)
    elif mode == 2:
        # Greedy spacing
        positions = [int(rng.integers(0, n))]
        for _ in range(n//2 - 1):
            best_pos = 0
            best_min = -1
            for p in range(n):
                if p in positions:
                    continue
                min_dist = min(min(abs(p - q), n - abs(p - q)) for q in positions)
                if min_dist > best_min:
                    best_min = min_dist
                    best_pos = p
            positions.append(best_pos)
        h[positions] = 1.0
    elif mode == 3:
        # Random with smoothing bias
        h = rng.random(n)
        h[h > 0.5] = 1.0
        h[h <= 0.5] = 0.0
    else:
        # Block pattern
        n_blocks = rng.integers(3, 8)
        block_size = n // n_blocks
        for b in range(n_blocks):
            if rng.random() > 0.5:
                h[b*block_size:(b+1)*block_size] = 1.0
    return _project(h, n/2.0)

            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 70:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.68
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search_quad(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.30)
        else:
            stall += 1
            radius *= 0.70
            if stall % 18 == 0 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5

def _lse_descent(h, deadline, tau_init=0.02, n_iter_max=400):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    stall = 0
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        obj, grad = _lse_obj_grad(h, tau)
        grad_dir = grad - np.mean(grad)
        norm = np.max(np.abs(grad_dir))
        if norm < 1e-13:
            break
        d = -grad_dir / norm
        alpha, new_c5 = _line_search_quad(h, d, dx, 0.4)
        if new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                tau = max(0.003, tau * 0.96)
                stall = 0
            else:
                stall += 1
                tau = min(0.08, tau * 1.2)
        else:
            stall += 1
            tau = min(0.08, tau * 1.15)
            if tau > 0.06:
                break
    return best_h, best_c5

def _slsqp_lse(h, deadline, tau=0.015):
    n = len(h)
    target = n / 2.0
    def obj_func(x):
        return _lse_obj_grad(np.clip(x, 0.0, 1.0), tau)[0]
    def obj_grad(x):
        return _lse_obj_grad(np.clip(x, 0.0, 1.0), tau)[1]
    def cons_func(x):
        return np.sum(x) - target
    def cons_grad(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': cons_func, 'jac': cons_grad}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=obj_grad,
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    cur_c5 = _exact_c5(h)
    return h, cur_c5

def _all_pair_swap(h, deadline, max_passes=6):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search_quad(h, d, dx, 0.4)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _random_pair_swap(h, deadline, n_tries=3000, radius=0.3):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    stall = 0
    while time.time() < deadline:
        if stall > 500:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search_quad(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.02, 0.5)
        else:
            stall += 1
            if stall % 100 == 0:
                radius *= 0.9
    return best_h, best_c5

def _block_swap(h, deadline, n_tries=500, block_min=2, block_max=8):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    while time.time() < deadline:
        if stall > 200:
            break
        b1 = int(rng.integers(block_min, block_max + 1))
        b2 = int(rng.integers(block_min, block_max + 1))
        s1 = int(rng.integers(0, max(1, n - b1)))
        s2 = int(rng.integers(0, max(1, n - b2)))
        d = np.zeros(n)
        d[s1:s1+b1] = 1.0 / b1
        d[s2:s2+b2] = -1.0 / b2
        d -= np.mean(d)
        nw = np.max(np.abs(d))
        if nw < 1e-12:
            stall += 1
            continue
        d /= nw
        alpha, c5_try = _line_search_quad(h, d, dx, 0.35)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _structured_init(n, rng, mode):
    h = np.zeros(n)
    if mode % 6 == 0:
        freqs = rng.uniform(0.5, 5.0, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 2, n, endpoint=False)
        for f, p in zip(freqs, phases):
            h += np.sin(2 * np.pi * f * x / 2.0 + p)
    elif mode % 6 == 1:
        period = int(rng.integers(n // 5, n // 2 + 1))
        pattern = (rng.random(period) > 0.5).astype(float)
        h = np.tile(pattern, n // period + 1)[:n]
    elif mode % 6 == 2:
        positions = [int(rng.integers(0, n))]
        for _ in range(n // 2 - 1):
            best_pos = 0
            best_min = -1
            for p in range(n):
                if p in positions:
                    continue
                min_dist = min(min(abs(p - q), n - abs(p - q)) for q in positions)
                if min_dist > best_min:
                    best_min = min_dist
                    best_pos = p
            positions.append(best_pos)
        h[positions] = 1.0
    elif mode % 6 == 3:
        h = (rng.random(n) > 0.5).astype(float)
    elif mode % 6 == 4:
        n_blocks = int(rng.integers(3, 9))
        block_size = n // n_blocks
        for b in range(n_blocks):
            if rng.random() > 0.5:
                h[b * block_size:(b + 1) * block_size] = 1.0
    else:
        h = np.sin(np.linspace(0, 2 * np.pi * 3, n, endpoint=False))
        h = (h > 0).astype(float)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP warmup (5%)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _lp_loop(h_init, d1, radius_init=0.10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LSE gradient descent (12%)
    d2 = t0 + budget_s * 0.17
    h2 = best_h.copy()
    h2, c5_2 = _lse_descent(h2, d2, tau_init=0.02, n_iter_max=300)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP with LSE (13%)
    d3 = t0 + budget_s * 0.30
    for tau_attempt in [0.03, 0.015, 0.008]:
        if time.time() > d3:
            break
        h3 = best_h.copy()
        sub_dl = min(d3, time.time() + budget_s * 0.05)
        h3, c5_3 = _slsqp_lse(h3, sub_dl, tau=tau_attempt)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-start exploration (20%)
    d4 = t0 + budget_s * 0.50
    n_trials = 10
    for trial in range(n_trials):
        if time.time() > d4:
            break
        if trial < 6:
            h_r = _structured_init(n0, rng, trial)
        else:
            h_r = _random_binary(n0, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.022)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.15)
        rem = d4 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.008:
            sub_dl2 = min(d4, time.time() + max(3, rem * 0.4))
            h_r, c5_r = _lse_descent(h_r, sub_dl2, tau_init=0.012, n_iter_max=80)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Multi-resolution (15%)
    d5 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    for dn in [8, -8, 16, -16, 24, -24, 32, -32, 48, -48, 40, -40, 12, -12, 20, -20]:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 48 or n_try > 280:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.018)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        rem = d5 - time.time()
        if rem > 4 and c5_r < best_c5 + 0.004:
            sub_dl2 = min(d5, time.time() + max(3, rem * 0.4))
            h_r, c5_r = _lse_descent(h_r, sub_dl2, tau_init=0.008, n_iter_max=60)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 6: Intensive polish (20%)
    d6 = t0 + budget_s * 0.85
    sub_dl = min(d6, time.time() + budget_s * 0.05)
    h6 = best_h.copy()
    h6, c5_6 = _lp_loop(h6, sub_dl, radius_init=0.05)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    sub_dl = min(d6, time.time() + budget_s * 0.05)
    h6 = best_h.copy()
    h6, c5_6 = _all_pair_swap(h6, sub_dl, max_passes=5)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    sub_dl = min(d6, time.time() + budget_s * 0.04)
    h6 = best_h.copy()
    h6, c5_6 = _block_swap(h6, sub_dl, n_tries=800)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    sub_dl = min(d6, time.time() + budget_s * 0.03)
    h6 = best_h.copy()
    h6, c5_6 = _random_pair_swap(h6, sub_dl, n_tries=2000, radius=0.12)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    sub_dl = min(d6, time.time() + budget_s * 0.03)
    h6 = best_h.copy()
    h6, c5_6 = _lse_descent(h6, sub_dl, tau_init=0.006, n_iter_max=150)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Perturbation search (10%)
    d7 = t0 + budget_s * 0.95
    scale = 0.03
    for attempt in range(300):
        if time.time() > d7:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.1, scale * 1.08)

    # Phase 8: Final polish (5%)
    d8 = deadline
    h8 = best_h.copy()
    h8, c5_8 = _lp_loop(h8, d8, radius_init=0.03)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    h8 = best_h.copy()
    h8, c5_8 = _all_pair_swap(h8, d8, max_passes=3)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
