def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _true_obj(h, d, dx):
    scores = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(scores) * dx)

def _line_search(h, d, dx, max_step):

        t_hi = min(t_hi, t_lo_cand)

            best_t = t

        else:

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] += (1.0 - h[d:])
            G[r, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            G[r, g:] += (1.0 - h[:m])
            G[r, :m] += (-h[g:])
    return G

def _lp_dir(h, G, scores, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([G, -np.ones(2 * n - 1)])
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] * 0.999)
        hi = min(radius, (1.0 - h[j]) * 0.999)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgrad_dir(h, scores, k_active, radius):
    n = len(h)
    d = k_active - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        g[:m] += (1.0 - h[d:])
        g[d:] += (-h[:m])
    else:
        gg = -d
        m = n - gg
        g[gg:] += (1.0 - h[:m])
        g[:m] += (-h[gg:])
    g -= np.mean(g)
    norm = np.linalg.norm(g)
    if norm < 1e-12:
        return None
    d = -radius * g / norm
    lo = np.maximum(-radius, -h * 0.999)
    hi = np.minimum(radius, (1.0 - h) * 0.999)
    d = np.clip(d, lo, hi)
    d -= np.mean(d)
    d = np.clip(d, lo, hi)
    return d

def _resample(h, n_new):

def _thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        bits = 0
        x = i
        while x > 0:
            bits += x & 1
            x >>= 1
        h[i] = 1.0 if bits % 2 == 0 else 0.0
    return _project(h, n / 2.0)
