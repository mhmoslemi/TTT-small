    def func(x):
        h = np.clip(x, 0.0, 1.0)
        return _pnorm_obj(h, q)

    def jac(x):
        h = np.clip(x, 0.0, 1.0)
        _, grad = _pnorm_score(h, q)
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(func, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project(res.x, target)
    except Exception:
        pass
    return h0

def _pair_descent(h, deadline, max_passes=300, thresh=1e-7):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores_vec(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    np.clip(h, 0, 1, out=h)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores_vec(h)
                        G = _build_G(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not pass_improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _pnorm_chain(h, deadline, q_list=None):
    if q_list is None:
        q_list = [8, 16, 32, 64, 128]
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for q in q_list:
        if time.time() > deadline:
            break
        h = _slsqp_opt(best_h, q, min(deadline, time.time() + 60), maxiter=200)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() > deadline:
            break
        h2 = _proj_grad_opt(best_h, q, min(deadline, time.time() + 40), max_iters=300)
        c5_2 = _exact_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
        except NameError:
            h0 = None
    else:
        h0 = None

    if h0 is None or len(h0) == 0 or not np.all(np.isfinite(h0)):
        h0 = np.linspace(0, 1, 128)
        h0 = (h0 < 0.5).astype(float)
        h0 += rng.normal(0, 0.1, 128)
        h0 = _project(h0, 64.0)

    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: P-norm continuation on parent
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _pnorm_chain(best_h, dl1, q_list=[8, 16, 32, 64, 128])
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair descent on current best
    dl2 = t0 + budget_s * 0.25
    h2, c5_2 = _pair_descent(best_h, dl2, max_passes=200, thresh=1e-7)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More p-norm refinement after pair descent
    dl3 = t0 + budget_s * 0.35
    h3, c5_3 = _pnorm_chain(best_h, dl3, q_list=[32, 64, 128, 256])
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution and diverse starts
    dl4 = t0 + budget_s * 0.65
    resolutions = [160, 192, 256, 128, 320]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl4:
            break
        if best_c5 <= 0.38084:
            break
        for s in range(2):
            if time.time() > dl4:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_init(n_try, alpha, rng)
            sub_dl = min(dl4, time.time() + budget_s * 0.04)
            h_try, c5_try = _pnorm_chain(h_init, sub_dl, q_list=[16, 48, 128])
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                h_pd, c5_pd = _pair_descent(best_h, sub_dl2, max_passes=80, thresh=1e-7)
                if c5_pd < best_c5:
                    best_c5 = c5_pd
                    best_h = h_pd.copy()
            else:
                sub_dl2 = min(dl4, time.time() + budget_s * 0.015)
                h_g, c5_g = _proj_grad_opt(h_try, 64, sub_dl2, max_iters=200)
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = h_g.copy()
                    best_n = n_try

    # Phase 5: Pair descent intensive
    if time.time() < t0 + budget_s * 0.80:
        dl5 = t0 + budget_s * 0.82
        h5, c5_5 = _pair_descent(best_h, dl5, max_passes=400, thresh=1e-8)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.92
    scale = 0.02
    for attempt in range(300):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.006)
        h_try = _slsqp_opt(h_pert, 64, sub_dl, maxiter=100)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 7: Final polish
    if time.time() < hard_deadline:
        dl7 = hard_deadline
        h7, c5_7 = _pair_descent(best_h, dl7, max_passes=200, thresh=1e-9)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
        if time.time() < dl7 - 5:
            h8, c5_8 = _pnorm_chain(best_h, dl7, q_list=[128, 256])
            if c5_8 < best_c5:
                best_c5 = c5_8
                best_h = h8.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
