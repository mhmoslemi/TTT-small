def _project(v, target):

    s = _scores(h)

        if d >= 0:
            m = n - d
            G[i, :m] += (1.0 - h[d:d+m])
            G[i, d:] += -h[:m]
        else:

            m = n - g_
            G[i, :m] += (1.0 - h[g_:g_+m])
            G[i, g_:] += -h[:m]

    # But we actually want G @ d to be negative (decrease all active lags)

    # This is: (G^T G + rho*I) d = -G^T * 1, with the constraint sum(d)=0
    # Use Lagrange: (G^T G + rho*I) d - lambda * 1 = -G^T * 1
    # And sum(d) = 0 → 1^T d = 0
    rho = 0.01

    b = -(G.T @ np.ones(len(active)))
    # Solve with constraint: augment with constraint row
    A_aug = np.vstack([A, np.ones(n)])
    b_aug = np.concatenate([b, [0.0]])
    try:
        d = np.linalg.lstsq(A_aug, b_aug, rcond=None)[0]
    except:
        return None
    d -= d.mean()  # ensure zero sum
    dn = np.linalg.norm(d)
    if dn < 1e-12:
        return None
    return d / dn

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_grad(h, dx, beta):
    n = len(h)
    s = _scores(h) * dx
    s_max = float(np.max(s))
    v = beta * (s - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += w[r] * (1.0 - h[d:d+m])
                g[d:] += -w[r] * h[:m]
        else:
            g_ = -d
            m = n - g_
            if m > 0:
                g[:m] += -w[r] * h[g_:g_+m]
                g[g_:] += w[r] * (1.0 - h[:m])
    g -= g.mean()

def _line_search(h, d, dx, trust_r):

    neg = d < -1e-14
    t_hi = trust_r
    t_lo = -trust_r

    if np.any(neg):

    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    ng = 25
    grid = np.linspace(t_lo, t_hi, ng)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx

    best_t = grid[best_i]
    best_val = obj_grid[best_i]

        else:

    best_h = h.copy()
    best_c5 = cur

    lr = 0.15
    stall = 0
    bi = 0
    for it in range(max_iter):

            break
        frac = it / max_iter
        beta = beta_lo + (beta_hi - beta_lo) * frac
        g = _smooth_grad(h, dx, beta)

        mn = np.linalg.norm(mom)

            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.04, 0.6)
        else:
            stall += 1
            lr *= 0.82
            if stall > 50:
                bi += 1
                if bi > 20:
                    break
                stall = 0
                mom = np.zeros(n)
                lr = 0.10
    return best_h, best_c5

def _pair_line(h, i, j, dx):
    n = len(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    a, v = _line_search(h, d, dx, min(h[i], 1.0 - h[j]))
    if v < _exact_c5(h) - 1e-13:
        h_new = h + a * d
        h_new = _project(h_new, n / 2.0)
        return h_new, _exact_c5(h_new)
    d2 = -d
    a2, v2 = _line_search(h, d2, dx, min(1.0 - h[i], h[j]))
    if v2 < _exact_c5(h) - 1e-13:
        h_new = h + a2 * d2
        h_new = _project(h_new, n / 2.0)
        return h_new, _exact_c5(h_new)
    return h, _exact_c5(h)

def _pairwise_opt(h, deadline, rng, max_rounds=300, randomfrac=1.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(interior) < 2:
            break
        if len(interior) > 40 and randomfrac < 1.0:
            idx = rng.choice(len(interior), size=min(40, len(interior)), replace=False)
            interior = interior[idx]
        for i in interior:
            for j in interior:
                if i == j:
                    continue
                h, new_c5 = _pair_line(h, i, j, dx)
                if new_c5 < cur - 1e-13:
                    cur = new_c5
                    improved = True
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def _active_balance(h, dx, top_k=10, tol_frac=0.08):
    n = len(h)
    s = _scores(h)
    s_max = float(np.max(s))
    if s_max < 1e-12:
        return None
    thresh = s_max * (1.0 - tol_frac)
    active = np.where(s >= thresh)[0]
    if len(active) < 2:
        return None
    active = active[:top_k]
    G = np.zeros((len(active), n))
    for i, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[i, :m] += (1.0 - h[d:d+m])
            G[i, d:] += -h[:m]
        else:
            g_ = -d
            m = n - g_
            G[i, :m] += (1.0 - h[g_:g_+m])
            G[i, g_:] += -h[:m]
    targets = -s[active] + s_max
    rho = 0.1
    A = G.T @ G + rho * np.eye(n)
    b = G.T @ targets
    A[0, :] += 0.0
    A_aug = np.vstack([A, np.ones(n)])
    b_aug = np.concatenate([b, [0.0]])
    try:
        d = np.linalg.lstsq(A_aug, b_aug, rcond=None)[0]
    except np.linalg.LinAlgError:
        return None
    d -= d.mean()
    dn = np.linalg.norm(d)
    if dn < 1e-12:
        return None
    return d / dn

def _combined(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    phase = 0
    stall = 0
    momentum = np.zeros(n)
    lr = 0.08
    while time.time() < deadline:
        if stall > 40:
            phase = 1 - phase
            stall = 0
            momentum = np.zeros(n)
            lr = 0.08
        if phase == 0:
            beta = 50 + 50 * (stall % 10)
            g = _smooth_grad(h, dx, beta)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                stall += 1
                continue
            g = g / gn
            momentum = 0.85 * momentum + 0.15 * g
            mn = np.linalg.norm(momentum)
            if mn < 1e-14:
                stall += 1
                continue
            momentum = momentum / mn
            alpha, new_c5 = _line_search(h, lr * momentum, dx, lr * 3.0)
            if new_c5 < cur - 1e-13:
                h = h + alpha * lr * momentum
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.05, 0.5)
            else:
                stall += 1
                lr *= 0.85
        else:
            d_bal = _active_balance(h, dx, top_k=8, tol_frac=0.10)
            if d_bal is not None:
                alpha, new_c5 = _line_search(h, 0.06 * d_bal, dx, 0.15)
                if new_c5 < cur - 1e-13:
                    h = h + alpha * 0.06 * d_bal
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    stall = 0
                    continue
            stall += 1
            if stall > 10:
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) >= 2:
                    idx = rng.integers(0, len(interior), size=min(20, len(interior)))
                    pairs = [(interior[idx[a]], interior[idx[b]]) for a in range(len(idx)) for b in range(a+1, len(idx)) if interior[idx[a]] != interior[idx[b]]]
                    for (i, j) in pairs[:30]:
                        h, new_c5 = _pair_line(h, i, j, dx)
                        if new_c5 < cur - 1e-13:
                            cur = new_c5
                            if cur < best_c5:
                                best_c5 = cur
                                best_h = h.copy()
                            stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Combined local at original resolution (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _combined(h, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Multi-resolution exploration (55%)
    d2 = t0 + budget_s * 0.67
    base_n = len(best_h)
    target_ns = sorted(set([base_n, base_n+40, base_n*2, base_n*2+16, base_n*3, 256, 320, 384, 448, 512]))
    target_ns = [x for x in target_ns if 80 <= x <= 560]
    for n_try in target_ns:
        if time.time() > d2:
            break
        if n_try == base_n:
            h_r = best_h.copy()
        else:
            h_r = _resample(best_h, n_try)
        cur_r = _exact_c5(h_r)
        rem = max(0, d2 - time.time())
        if rem < 10:
            continue
        t_sub1 = min(d2, time.time() + rem * 0.5)
        h_r, c5_r = _combined(h_r, t_sub1, rng)
        if time.time() < d2 - 5:
            t_sub2 = min(d2, time.time() + (d2 - time.time()) * 0.4)
            h_r, c5_r2 = _pairwise_opt(h_r, t_sub2, rng, max_rounds=80, randomfrac=0.8)
            if c5_r2 < c5_r:
                c5_r = c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r

    # Phase 3: Random kicks + reoptimize (13%)
    d3 = t0 + budget_s * 0.80
    n_best = len(best_h)
    for _ in range(30):
        if time.time() > d3:
            break
        scale = rng.uniform(0.02, 0.09)
        direction = rng.normal(0, 1, n_best)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, n_best / 2.0)
        sub_dl = min(d3, time.time() + 4.0)
        h_k, c5_k = _combined(h_k, sub_dl, rng)
        if c5_k < best_c5 - 1e-12:
            best_c5 = c5_k
            best_h = h_k

    # Phase 4: Upsample best and optimize (8%)
    d4 = t0 + budget_s * 0.88
    if time.time() < d4:
        n_up = min(len(best_h) * 2, 560)
        if n_up > len(best_h):
            h_up = _resample(best_h, n_up)
            h_up, c5_up = _combined(h_up, d4, rng)
            if c5_up < best_c5 - 1e-12:
                best_c5 = c5_up
                best_h = h_up

    # Phase 5: Final pairwise polish (10%)
    d5 = t0 + budget_s * 0.98
    h5, c5_5 = _pairwise_opt(best_h, d5, rng, max_rounds=150, randomfrac=0.7)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5

    # Final polish
    d6 = t0 + budget_s - 1.0
    if time.time() < d6:
        h6, c5_6 = _subgrad_opt(best_h, d6, beta_lo=100, beta_hi=800, max_iter=3000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
