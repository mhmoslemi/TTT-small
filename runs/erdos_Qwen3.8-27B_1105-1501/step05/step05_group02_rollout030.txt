def _exact_c5(h):

    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_transfer_line(h, i, j, n_grid=16):
    """Find best transfer from j to i: h[i]+=t, h[j]-=t"""
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-12:
        return None, None
    
    best_t = 0.0
    best_c5 = _exact_c5(h)
    
    # Sample the line
    ts = np.linspace(t_min, t_max, n_grid)
    h_work = h.copy()
    for t in ts:
        h_work[i] = h[i] + t
        h_work[j] = h[j] - t
        c = _exact_c5(h_work)
        if c < best_c5:
            best_c5 = c
            best_t = t
    
    if best_t == 0.0:
        return None, None
    
    # Golden section refine around best_t
    half_width = (t_max - t_min) / n_grid
    lo = max(t_min, best_t - half_width)
    hi = min(t_max, best_t + half_width)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_apply_transfer(h, i, j, c1))
    f2 = _exact_c5(_apply_transfer(h, i, j, c2))
    for _ in range(12):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_apply_transfer(h, i, j, c1))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_apply_transfer(h, i, j, c2))
    opt_t = (lo + hi) / 2.0
    opt_c5 = _exact_c5(_apply_transfer(h, i, j, opt_t))
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_t = opt_t
    
    return best_t, best_c5

def _apply_transfer(h, i, j, t):
    h_new = h.copy()
    h_new[i] = h[i] + t
    h_new[j] = h[j] - t

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgrad(h, topk=15):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = float(np.max(scores))
    sorted_s = np.sort(scores)[::-1]
    k = min(topk, len(scores))
    gap = max(1e-10, max_s - float(sorted_s[k - 1]))
    tau = gap / 2.0
    w = np.exp((scores - max_s) / tau)
    sw = float(np.sum(w))
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    g = np.zeros(n)
    for d in range(-n + 1, n):
        wd = w[n - 1 + d]
        if wd < 1e-10:
            continue
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            g[d:] += wd * dx * (1.0 - h[:m])
            g[:m] += -wd * dx * h[d:]
        else:
            s2 = -d
            g[:m] += wd * dx * (1.0 - h[s2:])
            g[s2:] += -wd * dx * h[:m]
    return g

def _pair_step(h, i, j, cur_c5):
    """Golden section: find best t for h[i]+=t, h[j]-=t. In-place on h. Returns (t, new_c5, accepted)."""
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-12:
        return 0.0, cur_c5, False
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_min, t_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h[i] += c1; h[j] -= c1
    f1 = _exact_c5(h)
    h[i] -= c1; h[j] += c1
    h[i] += c2; h[j] -= c2
    f2 = _exact_c5(h)
    h[i] -= c2; h[j] += c2
    for _ in range(13):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            h[i] += c1; h[j] -= c1
            f1 = _exact_c5(h)
            h[i] -= c1; h[j] += c1
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            h[i] += c2; h[j] -= c2
            f2 = _exact_c5(h)
            h[i] -= c2; h[j] += c2
    opt_t = (lo + hi) / 2.0
    if abs(opt_t) < 1e-14:
        return 0.0, cur_c5, False
    h[i] += opt_t; h[j] -= opt_t
    opt_c5 = _exact_c5(h)
    if opt_c5 < cur_c5 - 1e-14:
        return opt_t, opt_c5, True
    # Revert
    h[i] -= opt_t; h[j] += opt_t
    return 0.0, cur_c5, False

    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for it in range(iters):
        if time.time() > deadline:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        h[i] = 0.0; h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            cur_c5 = new_c5
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
        else:
            h[i] = 1.0; h[j] = 0.0
        T *= 0.9999
    
    return best_h, best_c5
