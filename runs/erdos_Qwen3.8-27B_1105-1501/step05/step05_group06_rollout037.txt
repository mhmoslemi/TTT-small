Now for the pair descent algorithm:

Given:
- h: current solution, length N
- Active set: indices d_1, ..., d_k of the top-k scores
- F: sensitivity matrix, shape (k, N), where F[i, j] = d(score[d_i])/d(h[j])

For pair (j, k) with alpha moved from position k to position j:
Change in score[i] = alpha * (F[i, j] - F[i, k])

We want to minimize: max_i (score[i] + alpha * (F[i, j] - F[i, k]))

subject to: alpha_lo <= alpha <= alpha_hi where:
alpha_lo = max(-h[j], h[k] - 1)
alpha_hi = min(1 - h[j], h[k])

This is a 1D convex piecewise-linear minimization. With k active lags, the function has at most k "kinks" (where the argmax changes). The minimum is at one of these kinks or at the boundary.

For small k (say k <= 15), I can just use golden section search or ternary search.

Actually, since it's piecewise linear and convex, I can find the minimum more efficiently:
- The slope of g(alpha) = max_i (a_i + b_i * alpha) at a point alpha is max of {b_i : i in argmax_i (a_i + b_i * alpha)}
- At the minimum, 0 is in the subdifferential, meaning the slope changes from ≤ 0 to ≥ 0

But for simplicity and correctness, I'll just use a fine grid + golden section. With k ≤ 20 and ~50 function evaluations, it's fast enough.

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    res = target - x.sum()
    if abs(res) > 1e-12:
        mask = (x < 1) if res > 0 else (x > 0)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _sensitivity_mat(h, lags):
    """ Compute sensitivity matrix for given lag indices.
    lags: array of indices into the (2N-1) correlation array.
    Returns: (len(lags), N) matrix.
    """
    N = len(h)
    K = len(lags)
    F = np.zeros((K, N))
    for idx, m in enumerate(lags):
        sens = np.zeros(N)
        d = m - (N - 1)
        if d >= 0:
            # Positive lag d
            if d == 0:
                sens[:] = 1.0 - 2.0 * h
            else:
                sens[d:] += (1.0 - h[:N-d])
                sens[:N-d] -= h[d:]
        else:
            sl = -d
            if sl == 0:
                sens[:] = 1.0 - 2.0 * h
            else:
                sens[:N-sl] += (1.0 - h[sl:])
                sens[sl:] -= h[:N-sl]
        F[idx] = sens
    return F
```

def _pair_descent(h, k_active, max_passes, deadline, threshold=1e-10):
    N = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        scores = _all_scores(h)
        cur_max = float(scores.max())
        # Get top-k_active lags
        top_idx = np.argsort(scores)[-k_active:]
        F = _sensitivity_mat(h, top_idx)
        top_scores = scores[top_idx]
        
        n_accepted = 0
        for j in range(N):
            for k in range(j+1, N):
                if time.time() > deadline:
                    return best_h, best_c5
                # Feasibility bounds
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-14:
                    continue
                # Direction vectors for active lags
                b = F[:, j] - F[:, k]  # shape (k_active,)
                # g(alpha) = max_i (top_scores[i] + alpha * b[i])
                # Find alpha in [a_lo, a_hi] minimizing g(alpha)
                # Use golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                lo_a, hi_a = a_lo, a_hi
                c1 = hi_a - gr * (hi_a - lo_a)
                c2 = lo_a + gr * (hi_a - lo_a)
                f1 = np.max(top_scores + c1 * b)
                f2 = np.max(top_scores + c2 * b)
                for _ in range(50):
                    if hi_a - lo_a < 1e-12:
                        break
                    if f1 < f2:
                        hi_a, c2, f2 = c2, c1, f1
                        c1 = hi_a - gr * (hi_a - lo_a)
                        f1 = np.max(top_scores + c1 * b)
                    else:
                        lo_a, c1, f1 = c1, c2, f2
                        c2 = lo_a + gr * (hi_a - lo_a)
                        f2 = np.max(top_scores + c2 * b)
                alpha = (lo_a + hi_a) / 2.0
                pred_max = float(np.max(top_scores + alpha * b))
                if pred_max < cur_max - threshold and abs(alpha) > 1e-12:
                    # Apply move
                    h[j] += alpha
                    h[k] -= alpha
                    # Verify with full evaluation
                    new_scores = _all_scores(h)
                    actual_max = float(new_scores.max())
                    if actual_max < cur_c5 - threshold:
                        cur_c5 = actual_max
                        scores = new_scores
                        n_accepted += 1
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        # Update F (sensitivity changed)
                        if n_accepted % 5 == 0:
                            top_idx = np.argsort(scores)[-k_active:]
                            F = _sensitivity_mat(h, top_idx)
                            top_scores = scores[top_idx]
                        else:
                            F = _sensitivity_mat(h, top_idx)
                            top_scores = scores[top_idx]
                    else:
                        # Revert
                        h[j] -= alpha
                        h[k] += alpha
        if n_accepted == 0:
            break
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    res = target - x.sum()
    if abs(res) > 1e-12:
        mask = (x < 1) if res > 0 else (x > 0)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _sensitivity_mat(h, corr_indices):
    N = len(h)
    K = len(corr_indices)
    F = np.zeros((K, N))
    for idx in range(K):
        m = int(corr_indices[idx])
        d = m - (N - 1)
        sens = np.zeros(N)
        if d > 0:
            sens[d:] += (1.0 - h[:N - d])
            sens[:N - d] -= h[d:]
        elif d < 0:
            sl = -d
            sens[:N - sl] += (1.0 - h[sl:])
            sens[sl:] -= h[:N - sl]
        else:
            sens[:] = 1.0 - 2.0 * h
        F[idx] = sens
    return F

def _pair_descent(h, k_act, max_passes, deadline, thresh=1e-10):
    N = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    nr = 2 * N - 1
    for p in range(max_passes):
        if time.time() > deadline:
            break
        scores = _all_scores(h)
        cur_max = float(scores.max())
        top_idx = np.argsort(scores)[-k_act:]
        F = _sensitivity_mat(h, top_idx)
        top_s = scores[top_idx]
        n_acc = 0
        for j in range(N):
            for k in range(j + 1, N):
                if time.time() > deadline:
                    return best_h, best_c5
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-14:
                    continue
                b = F[:, j] - F[:, k]
                bmax = float(b.max())
                bmin = float(b.min())
                if a_lo <= 0.0 <= a_hi:
                    if bmin <= 0.0 <= bmax:
                        continue
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                lo_a, hi_a = a_lo, a_hi
                c1 = hi_a - gr * (hi_a - lo_a)
                c2 = lo_a + gr * (hi_a - lo_a)
                f1 = float(np.max(top_s + c1 * b))
                f2 = float(np.max(top_s + c2 * b))
                for _ in range(45):
                    if hi_a - lo_a < 1e-13:
                        break
                    if f1 < f2:
                        hi_a, f2, c2 = c2, f1, c1
                        c1 = hi_a - gr * (hi_a - lo_a)
                        f1 = float(np.max(top_s + c1 * b))
                    else:
                        lo_a, f1, c1 = c1, f2, c2
                        c2 = lo_a + gr * (hi_a - lo_a)
                        f2 = float(np.max(top_s + c2 * b))
                alpha = (lo_a + hi_a) / 2.0
                if abs(alpha) < 1e-13:
                    continue
                pred = float(np.max(top_s + alpha * b))
                if pred >= cur_max - thresh:
                    continue
                h[j] += alpha
                h[k] -= alpha
                ns = _all_scores(h)
                am = float(ns.max())
                if am < cur_c5 - thresh:
                    cur_c5 = am
                    scores = ns
                    n_acc += 1

                        best_c5 = cur_c5
                        best_h = h.copy()
                    top_idx = np.argsort(scores)[-k_act:]
                    F = _sensitivity_mat(h, top_idx)
                    top_s = scores[top_idx]
                else:
                    h[j] -= alpha
                    h[k] += alpha
        if n_acc == 0:
            break
    return best_h, best_c5

def _binary_swaps(h, max_iters, deadline, rng):
    N = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        j = ones[rng.integers(len(ones))]
        k = zeros[rng.integers(len(zeros))]
        h[j] = 0.0
        h[k] = 1.0
        c5_new = _exact_c5(h)
        if c5_new < cur_c5 - 1e-14:
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[j] = 1.0
            h[k] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_refine(h, deadline, n_steps=200, beta=50.0):
    N = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _all_scores(h)
        max_s = scores.max()
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        F_all = _sensitivity_mat(h, np.arange(2 * N - 1))
        grad = F_all.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.05
        improved = False
        for _ in range(20):
            h_new = _project(h - alpha * grad, N / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.6
            if alpha < 1e-10:
                break
        if not improved:
            break
    return best_h, best_c5
