def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

            x = np.clip(x, 0.0, 1.0)

For the p-norm with large q, F_r^q will underflow unless F_r is close to 1. Since F_r ≤ 0.5, for q > 2, F_r^q ≤ 0.5^q which gets very small. This means:
- For large q, only the largest F_r will contribute to the p-norm
- This is exactly what we want (approaching the max)
- But numerically, we need to handle the underflow

For numerical stability, I'll use:

Then grad_j = sum_r w_r * dF_r/dh_j

    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _continuation_pnorm(h, deadline, q_list=None, method='slsqp'):
    if q_list is None:
        q_list = [8, 16, 32, 64, 128, 256]
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for q in q_list:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 30.0)
        if method == 'slsqp':
            h_new, c5_new = _slsqp_pnorm(best_h, q, sub_dl, maxiter=200)
        else:
            h_new, c5_new = _proj_grad_pnorm(best_h, q, sub_dl, lr_init=0.01, max_steps=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            h_new2, c5_new2 = _proj_grad_pnorm(best_h, q, min(deadline, time.time() + 15.0),
                                               lr_init=0.005, max_steps=200)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
    return best_h, best_c5

def _continuation_softmax(h, deadline, beta_list=None):
    if beta_list is None:
        beta_list = [10, 30, 60, 120, 200, 400]
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for beta in beta_list:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 25.0)
        h_new, c5_new = _proj_grad_softmax(best_h, beta, sub_dl, lr_init=0.01, max_steps=250)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_dl = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: p-norm SLSQP continuation on parent
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _continuation_pnorm(h_start, dl1, q_list=[16, 32, 64, 128], method='slsqp')
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Soft max projected gradient
    if time.time() < t0 + budget_s * 0.22:
        dl2 = t0 + budget_s * 0.24
        h2, c5_2 = _continuation_softmax(best_h, dl2, beta_list=[20, 50, 100, 200, 400])
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: p-norm projected gradient (faster, more steps)
    if time.time() < t0 + budget_s * 0.32:
        dl3 = t0 + budget_s * 0.35
        h3, c5_3 = _continuation_pnorm(best_h, dl3, q_list=[32, 64, 128, 256], method='pg')
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-resolution search
    if time.time() < t0 + budget_s * 0.55:
        dl4 = t0 + budget_s * 0.58
        resolutions = [96, 128, 160, 200]
        alphas_q = [0.618034, 0.414214, 0.707107, 0.5]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl4:
                break
            if best_c5 < 0.38050:
                break
            for s in range(2):
                if time.time() > dl4:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    h_init = _quasi_init(n_try, alphas_q[ri % len(alphas_q)], rng)
                sub_dl = min(dl4, time.time() + budget_s * 0.04)
                h_try, c5_try = _continuation_pnorm(h_init, sub_dl,
                                                    q_list=[16, 48, 96], method='pg')
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 5: Pair descent polishing
    if time.time() < t0 + budget_s * 0.72:
        dl5 = t0 + budget_s * 0.74
        best_h, best_c5 = _pair_descent(best_h, dl5, max_passes=200, thresh=1e-8)
        if time.time() < dl5:
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_sq, c5_sq = _slsqp_pnorm(best_h, 64, sub_dl, maxiter=100)
            if c5_sq < best_c5:
                best_c5 = c5_sq
                best_h = h_sq.copy()
        if time.time() < dl5:
            dl5b = min(dl5, time.time() + budget_s * 0.02)
            best_h, best_c5 = _pair_descent(best_h, dl5b, max_passes=100, thresh=1e-9)

    # Phase 6: Perturbation restarts
    if time.time() < t0 + budget_s * 0.88:
        dl6 = t0 + budget_s * 0.90
        scale = 0.02
        for attempt in range(300):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.006)
            h_try, c5_try = _proj_grad_pnorm(h_pert, 64, sub_dl, lr_init=0.01, max_steps=100)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.06, scale * 1.08)

    # Phase 7: Final polish
    if time.time() < hard_dl:
        dl7 = hard_dl
        h_pd, c5_pd = _pair_descent(best_h, dl7, max_passes=100, thresh=1e-10)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
        if time.time() < dl7 - 3:
            h_fin, c5_fin = _proj_grad_softmax(best_h, 500, min(dl7, time.time() + 10),
                                               lr_init=0.005, max_steps=100)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
