def _grad_matrix(h):

        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_step_full(h, radius, penalty_lags=None, penalty_w=0.08):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        b_ub[penalty_lags] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_optimize(h, deadline, radius_init=0.22, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        use_full = (stall > 10) or (stall % 20 == 15)
        if use_full:
            scores = np.correlate(h, 1.0 - h, mode='full')
            s_max = float(np.max(scores))
            penalty_lags = None
            if stall > 8:
                thresh = s_max - 0.005 * s_max
                penalty_lags = np.where(scores >= thresh)[0]
                if len(penalty_lags) == 0:
                    penalty_lags = None
            p = _lp_step_full(h, radius, penalty_lags=penalty_lags, penalty_w=0.06)
        else:
            top_k = min(15, 2 * n - 1)
            if stall > 5:
                p = _lp_step_focused(h, radius, top_k=top_k, penalty_frac=0.01, penalty_w=0.05)
            else:
                p = _lp_step_focused(h, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if radius < 1e-9:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if alpha == 0.0:
            stall += 1
            radius *= 0.8
            continue
        h_new = h + alpha * p
        h_new = _project(h_new, target)
        real_c5 = _exact_c5(h_new)
        if real_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = real_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.55)
        else:
            stall += 1
            radius *= 0.78
    return best_h, best_c5


def _smooth_optimize(h, deadline, beta_list=(60, 150, 400, 800), lr_init=0.025):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.84
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 50:
            bi += 1
            if bi >= len(beta_list) * 3:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = beta_list[min(bi, len(beta_list) - 1)]
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        s_max = float(np.max(scores))
        v = beta * (scores - s_max)
        exp_v = np.exp(v)
        w = exp_v / float(np.sum(exp_v))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:d + m])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (-h[g:g + m])
                    grad[g:] += w[r] * (1.0 - h[:m])
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            bi += 1
            continue
        d = grad / gn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.4)
        else:
            stall += 1
            lr *= 0.8
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_refine(h, deadline, max_passes=60, max_pairs_per_pass=8000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = int(np.sum(interior))
        if n_int < 2:
            break
        cnt = 0
        for i in range(n):
            if not interior[i]:
                continue
            for j in range(i + 1, n):
                if not interior[j]:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                cnt += 1
                if cnt >= max_pairs_per_pass:
                    break
            if cnt >= max_pairs_per_pass:
                break
        if not improved:
            break
    return best_h, best_c5


def _random_pair_refine(h, deadline, rng, max_trials_per_pass=1500, max_passes=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(max_trials_per_pass):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.45)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def _kick_reopt(h, deadline, rng, n_kicks=25):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.015, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _lp_optimize(h_k, sub_dl, radius_init=0.12, max_iters=25)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5


def _logit_seed(rng, n):
    v = rng.normal(0, 0.6, n)
    target = n / 2.0
    lo, hi = -50.0, 50.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        x = v + mid
        sig = np.zeros(n)
        pos = x >= 0
        sig[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
        ex = np.exp(np.minimum(x[~pos], 0.0))
        sig[~pos] = ex / (1.0 + ex)
        s = float(np.sum(sig))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = v + (lo + hi) / 2.0
    sig = np.zeros(n)
    pos = x >= 0
    sig[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(np.minimum(x[~pos], 0.0))
    sig[~pos] = ex / (1.0 + ex)
    return _project(sig, target)


def _logit_evolve(h_best, c5_best, deadline, rng, n_rounds=200):
    n = len(h_best)
    best_h = h_best.copy()
    best_c5 = c5_best
    pop = [h_best.copy()]
    c5s = [c5_best]
    for _ in range(6):
        hs = _logit_seed(rng, n)
        cs = _exact_c5(hs)
        pop.append(hs)
        c5s.append(cs)
    order = np.argsort(c5s)
    pop = [pop[i] for i in order]
    c5s = [c5s[i] for i in order]
    if c5s[0] < best_c5:
        best_c5 = c5s[0]
        best_h = pop[0].copy()
    sigma = 0.12
    no_improve = 0
    gen = 0
    while time.time() < deadline and gen < n_rounds:
        gen += 1
        if no_improve > 40:
            break
        offspring = []
        oc5s = []
        n_off = 8
        for _ in range(n_off):
            i = int(rng.integers(0, min(3, len(pop))))
            j = int(rng.integers(0, min(3, len(pop))))
            alpha = rng.uniform(0.3, 0.7)
            child = alpha * pop[i] + (1.0 - alpha) * pop[j]
            eps = 1e-5
            hc = np.clip(child, eps, 1.0 - eps)
            vc = np.log(hc / (1.0 - hc))
            vc += rng.normal(0, sigma, n)
            target = n / 2.0
            lo_t, hi_t = -50.0, 50.0
            for _ in range(45):
                mid = (lo_t + hi_t) / 2.0
                x = vc + mid
                sig = np.zeros(n)
                pos_m = x >= 0
                sig[pos_m] = 1.0 / (1.0 + np.exp(-x[pos_m]))
                ex2 = np.exp(np.minimum(x[~pos_m], 0.0))
                sig[~pos_m] = ex2 / (1.0 + ex2)
                s = float(np.sum(sig))
                if s > target:
                    lo_t = mid
                else:
                    hi_t = mid
            x = vc + (lo_t + hi_t) / 2.0
            sig = np.zeros(n)
            pos_m = x >= 0
            sig[pos_m] = 1.0 / (1.0 + np.exp(-x[pos_m]))
            ex2 = np.exp(np.minimum(x[~pos_m], 0.0))
            sig[~pos_m] = ex2 / (1.0 + ex2)
            child_h = _project(sig, target)
            c5c = _exact_c5(child_h)
            offspring.append(child_h)
            oc5s.append(c5c)
        all_h = pop + offspring
        all_c5 = c5s + oc5s
        k_sel = min(5, len(all_h))
        order2 = np.argsort(all_c5)
        pop = [all_h[i] for i in order2[:k_sel]]
        c5s = [all_c5[i] for i in order2[:k_sel]]
        if c5s[0] < best_c5 - 1e-14:
            best_c5 = c5s[0]
            best_h = pop[0].copy()
            no_improve = 0
            sigma = min(sigma * 1.08, 0.5)
        else:
            no_improve += 1
            sigma = max(0.008, sigma * 0.93)
    return best_h, best_c5


def _combined_local(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.20
    stall = 0
    mode = 0
    while time.time() < deadline:
        if stall > 35:
            mode = 1 - mode
            stall = 0
            radius = 0.20
            if mode == 1:
                lr = 0.02
                mom = np.zeros(n)
        if mode == 0:
            top_k = min(15, 2 * n - 1)
            if stall > 8:
                p = _lp_step_focused(h, radius, top_k=top_k, penalty_frac=0.015, penalty_w=0.06)
            else:
                p = _lp_step_focused(h, radius, top_k=top_k)
            if p is not None and np.max(np.abs(p)) > 1e-13:
                alpha, new_c5 = _line_search(h, p, dx, radius)
                if new_c5 < cur_c5 - 1e-14:
                    h = h + alpha * p
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.50)
                    continue
            stall += 1
            radius *= 0.82
        else:
            beta = 80
            scores = np.correlate(h, 1.0 - h, mode='full') * dx
            s_max = float(np.max(scores))
            v = beta * (scores - s_max)
            exp_v = np.exp(v)
            w = exp_v / float(np.sum(exp_v))
            grad = np.zeros(n)
            for r in range(2 * n - 1):
                if w[r] < 1e-15:
                    continue
                d = r - (n - 1)
                if d >= 0:
                    m = n - d
                    if m > 0:
                        grad[:m] += w[r] * (1.0 - h[d:d + m])
                        grad[d:] += w[r] * (-h[:m])
                else:
                    g = -d
                    m = n - g
                    if m > 0:
                        grad[:m] += w[r] * (-h[g:g + m])
                        grad[g:] += w[r] * (1.0 - h[:m])
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-13:
                stall += 1
                continue
            d = grad / gn
            if 'mom' not in dir():
                mom = np.zeros(n)
            mom = 0.82 * mom + 0.18 * d
            mn = np.linalg.norm(mom)
            if mn < 1e-13:
                stall += 1
                continue
            mom = mom / mn
            lr = 0.02
            alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * lr * mom
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                break
            else:
                stall += 1
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 3.0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is not None and len(h0) > 10:
        n0 = len(h0)
        h = _project(h0, n0 / 2.0)
    else:
        n0 = 122
        h = _project(0.5 * np.ones(n0), n0 / 2.0)

    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Combined local optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _combined_local(h, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Focused LP (10%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _lp_optimize(best_h, d2, radius_init=0.22, max_iters=300)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair refinement (12%)
    d3 = t0 + budget_s * 0.42
    h3, c5_3 = _pair_refine(best_h, d3, max_passes=40, max_pairs_per_pass=6000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution search (28%)
    d4 = t0 + budget_s * 0.70
    n_base = len(best_h)
    tried = set()
    resolutions = []
    for dn in range(-40, 41, 2):
        nt = n_base + dn
        if 80 <= nt <= 230 and nt % 2 == 0:
            resolutions.append(nt)
    resolutions.sort(key=lambda x: abs(x - n_base))
    for nt in resolutions:
        if time.time() > d4:
            break
        if nt in tried:
            continue
        tried.add(nt)
        h_r = _resample(best_h, nt)
        sub_dl = min(d4, time.time() + budget_s * 0.010)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.18, max_iters=40)
        if c5_r <= best_c5 + 0.002 and time.time() < d4 - 15:
            sub_dl2 = min(d4, time.time() + 6.0)
            h_r, c5_r = _random_pair_refine(h_r, sub_dl2, rng, max_trials_per_pass=800, max_passes=60)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_base = len(best_h)

    # Phase 5: Logit evolution (8%)
    d5 = t0 + budget_s * 0.78
    h5, c5_5 = _logit_evolve(best_h, best_c5, d5, rng, n_rounds=150)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: LP refinement (6%)
    d6 = t0 + budget_s * 0.84
    h6, c5_6 = _lp_optimize(best_h, d6, radius_init=0.15, max_iters=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Smooth (4%)
    d7 = t0 + budget_s * 0.88
    h7, c5_7 = _smooth_optimize(best_h, d7, beta_list=(80, 200, 500), lr_init=0.02)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Kick and reopt (4%)
    d8 = t0 + budget_s * 0.92
    h8, c5_8 = _kick_reopt(best_h, d8, rng, n_kicks=15)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Phase 9: Final pair refinement (6%)
    d9 = t0 + budget_s * 0.98
    h9, c5_9 = _random_pair_refine(best_h, d9, rng, max_trials_per_pass=1200, max_passes=200)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    # Final polish
    final_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(final_h)
    return final_h, final_c5, len(final_h)
