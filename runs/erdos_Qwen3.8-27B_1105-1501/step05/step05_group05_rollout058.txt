def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):

    G[n - 1, :] = 1.0 - 2.0 * h
    for d in range(1, n):

def _line_search(h, d, trust_radius):
    n = len(h)
    dx = 2.0 / n
    d = d - d.mean()
    dn = np.linalg.norm(d)
    if dn < 1e-15:
        return 0.0, _exact_c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-13:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(45):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val > max(f1, f2):
        opt_t = c1 if f1 < f2 else c2
        opt_val = min(f1, f2)
    return opt_t, opt_val

def _lp_solve(h, scores, G, radius, penalty_lags=None, penalty_w=0.1):
    n = len(h)
    num_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        b_ub[penalty_lags] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-15)
        hi = min(radius, 1.0 - h[j] - 1e-15)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    pen_level = 0.0

    while time.time() < deadline:
        if stall > 150:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        G = _grad_matrix(h)

        penalty_lags = None
        if stall > 3:
            pen_level = min(0.3, pen_level + 0.02)
            thresh = s_max - pen_level * (s_max / max(n, 10))
            penalty_lags = list(np.where(scores >= thresh)[0])
            if len(penalty_lags) > (2 * n - 1) * 0.4:
                thresh = s_max * 0.9995
                penalty_lags = list(np.where(scores >= thresh)[0])

        p = _lp_solve(h, scores, G, radius, penalty_lags, penalty_w=max(0.05, pen_level))
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.82
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            pen_level = max(0.0, pen_level - 0.05)
            radius = min(radius * 1.04, 0.60)
        else:
            stall += 1
            radius *= 0.77
            if stall > 25 and radius < 0.04:
                radius = best_radius * 0.55
                stall = 0
    return best_h, best_c5

        step = lr * mom
        alpha, new_c5 = _line_search(h, step, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * step
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.78
            if lr < 0.001:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(interior) < 2:
            break
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[ii]
            for jj in range(ii + 1, len(interior)):
                j = interior[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=40):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.01, 0.05)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= np.linalg.norm(direction) + 1e-15
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = _lp_loop(h_k, sub_dl, radius_init=0.12)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 122
        x = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
        h0 = np.clip(0.5 + 0.3 * np.sin(4 * np.pi * x / 2.0) + 0.1 * np.sin(7 * np.pi * x / 2.0), 0, 1)
        h0 = _project(h0, n0 / 2.0)

    n0 = len(h0)
    best_h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: Quick LP refinement (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution sweep (30%)
    d2 = t0 + budget_s * 0.45
    base_n = len(best_h)
    tried = set()
    n_candidates = sorted(set(range(max(80, base_n - 30), min(200, base_n + 30) + 1, 2)),
                          key=lambda x: abs(x - base_n))
    for n_try in n_candidates:
        if time.time() > d2:
            break
        if n_try in tried or n_try < 60:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.20)
        if c5_r < best_c5 - 1e-12 or (c5_r < best_c5 + 0.001 and c5_r < best_c5):
            best_c5 = min(best_c5, c5_r)
            if c5_r < best_c5 + 1e-12:
                best_h = h_r.copy()
                # Quick subgradient polish
                sub_dl2 = min(d2, time.time() + 4.0)
                h_r2, c5_r2 = _subgrad_loop(best_h, sub_dl2)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase 3: Subgradient refinement (15%)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _subgrad_loop(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP refinement on best (15%)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Pair refinement (15%)
    d5 = t0 + budget_s * 0.90
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final LP + pair polish (10%)
    d6 = deadline - 2.0
    if time.time() < d6:
        h6, c5_6 = _lp_loop(best_h, min(d6, time.time() + budget_s * 0.04), radius_init=0.06)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        if time.time() < d6:
            h6b, c5_6b = _pair_refine(best_h, d6, max_passes=100)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
