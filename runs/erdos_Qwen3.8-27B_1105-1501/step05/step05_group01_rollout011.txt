        half_width = (ts[min(ci+1, len(ts)-1)] - ts[max(ci-1, 0)]) * 1.5

        f2 = _line_obj(c2, S, T, Q, dx)

            if f1 < f2:
                b = c2
                c2 = c1
                f2 = f1
                c1 = b - gr * (b - a)
                f1 = _line_obj(c1, S, T, Q, dx)
            else:
                a = c1
                c1 = c2
                f1 = f2
                c2 = a + gr * (b - a)
                f2 = _line_obj(c2, S, T, Q, dx)
        opt_t = (a + b) / 2.0
        opt_v = _line_obj(opt_t, S, T, Q, dx)
        if opt_v < best_v:
            best_v = opt_v
            best_t = opt_t
    
    return best_t, best_v

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_direction(h, tau, dx, n):
    """Compute negative soft-max gradient as a direction."""
    scores = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(scores))
    shifted = np.clip((scores - M) / tau, -50.0, 50.0)
    exps = np.exp(shifted)
    w = exps / float(np.sum(exps))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-7:
            continue
        d = r - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            grad[d:] += w[r] * (1.0 - h[:m])
            grad[:m] -= w[r] * h[d:]
        else:
            g = -d
            grad[:m] += w[r] * (1.0 - h[g:])
            grad[g:] -= w[r] * h[:m]
    # Direction is -grad (descent), zero-sum
    d = -grad
    d -= np.mean(d)
    return d

def _line_search_run(h, d, dx, radius, n_grid=45):
    """Run multi-bracket line search along direction d."""
    # Normalize d
    dmax = float(np.max(np.abs(d)))
    if dmax < 1e-14:
        return 0.0, _exact_c5(h)
    d = d / dmax
    t_lo = -radius
    t_hi = radius
    return _multi_bracket_line_search(h, d, dx, t_lo, t_hi, n_grid)

def _gpsearch_direction(h, dx, n):
    """Generate a gradient-descent direction using soft-max with adaptive tau."""
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(scores))
    # Adaptive tau based on the gap between top scores
    sorted_scores = np.sort(scores * dx)[::-1]
    # tau proportional to the gap between 1st and 5th
    gap = max(0.01, (sorted_scores[0] - sorted_scores[min(4, len(sorted_scores)-1)]) * 5)
    tau = max(0.005, min(0.05, gap))
    return _grad_direction(h, tau, dx, n), tau
