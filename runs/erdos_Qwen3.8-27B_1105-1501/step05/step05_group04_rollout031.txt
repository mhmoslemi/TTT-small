    b_ub = np.full(n_lags, -scores + 0.5 * radius * dx_val)
    for k in active_set:
        b_ub[k] = -scores[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j] * 0.99), min(radius, (1.0 - h[j]) * 0.99)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(scores.max()) - 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 3
    while time.time() < deadline:
        if stall > 100:
            break
        scores = lag_values(h)
        G = lag_jacobian(h)
        p = lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            n_active = min(n_active + 2, len(scores) - 1)
            if n_active > len(scores) // 3:
                p = lp_direction(h, scores, G, radius, max(3, len(scores) // 4))
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.72
                if radius < 1e-6:
                    break
                if stall % 30 == 0:
                    radius = max(radius, 0.03)
                continue
        alpha, new_c5 = line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 3
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.78
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.05
                n_active = 5
    return best_h, best_c5

# SLSQP epigraph functions (top-level)
def _sqp_obj(x):
    return x[-1]

def _sqp_ineq_fun(x):
    n_h = len(x) - 1
    return x[-1] - lag_values(x[:n_h])

def _sqp_ineq_jac(x):
    n_h = len(x) - 1
    h = x[:n_h]
    G = lag_jacobian(h)
    J = np.empty((2 * n_h - 1, n_h + 1))
    J[:, :n_h] = -G
    J[:, n_h] = 1.0
    return J

def _sqp_eq_fun(x):
    n_h = len(x) - 1
    return x[:n_h].sum() - n_h / 2.0

def _sqp_eq_jac(x):
    n_h = len(x) - 1
    J = np.zeros(n_h + 1)
    J[:n_h] = 1.0
    return J

def slsqp_epigraph(h_seed, deadline, max_iter=200):
    n = len(h_seed)
    c0 = eval_c5(h_seed)
    x0 = np.concatenate([h_seed, [c0 + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, max(1.0, c0 + 0.2))]
    constraints = [
        {'type': 'ineq', 'fun': _sqp_ineq_fun, 'jac': _sqp_ineq_jac},
        {'type': 'eq', 'fun': _sqp_eq_fun, 'jac': _sqp_eq_jac}
    ]
    try:
        res = minimize(_sqp_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-8})
        h_new = project_box_sum(res.x[:n], n / 2.0)
        c5_new = eval_c5(h_new)
        if c5_new < c0:
            return h_new, c5_new
    except Exception:
        pass
    return h_seed, c0

# Log-sum-exp smoothing
def _lse_obj(x):
    h = x
    F = lag_values(h)
    m = float(F.max())
    beta = _lse_beta
    Z = np.exp((F - m) / beta)
    return m + beta * float(np.log(Z.sum()))

def _lse_grad(x):
    h = x
    F = lag_values(h)
    m = float(F.max())
    beta = _lse_beta
    Z = np.exp((F - m) / beta)
    P = Z / Z.sum()
    G = lag_jacobian(h)
    return P @ G

def _lse_eq_fun(x):
    return x.sum() - len(x) / 2.0

def _lse_eq_jac(x):
    return np.ones(len(x))

def lse_smoothing(h_seed, deadline, betas=None):
    if betas is None:
        c0 = eval_c5(h_seed)
        betas = [c0 * f for f in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]]
    h = h_seed.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        _lse_beta = beta
        h0 = h.copy()
        bounds = [(0.0, 1.0)] * len(h)
        constraints = [{'type': 'eq', 'fun': _lse_eq_fun, 'jac': _lse_eq_jac}]
        try:
            res = minimize(_lse_obj, h0, method='SLSQP', jac=_lse_grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-9})
            h_new = project_box_sum(res.x, len(h) / 2.0)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def coord_refine(h, deadline, n_iter=300, seed=123):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(seed)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5

def perturb_search(h, deadline, rng, base_scale=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    for _ in range(300):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, n / 2.0)
        sub_dl = min(deadline, time.time() + 5.0)
        h_t, c5_t = lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.008, scale * 0.88)
        else:
            scale = min(0.18, scale * 1.06)
    return best_h, best_c5

def multires(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    tried = set()
    for dn in range(0, 40):
        for sign in (1, -1):
            if dn == 0 and sign == -1:
                continue
            if time.time() > deadline:
                break
            n_try = n + sign * dn
            if n_try < 25 or n_try > 250 or n_try in tried:

            if c5_r < best_c5 - 1e-11:

                best_h = h_r.copy()

import numpy as np
from scipy.optimize import minimize, linprog
import time

_lse_beta = 0.05
_deadline = 0.0


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def lag_values(h):
    return np.correlate(h, 1.0 - h, mode='full')


def lag_jacobian(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def line_search(h, d, dx, max_step):
    S0 = lag_values(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = max_step
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_lo = 0.0
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(55):
        if b - a < 1e-15:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val


def lp_direction(h, scores, G, radius, n_active):
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    active_set = set(sorted_idx[:n_active].tolist())
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    dx_val = 2.0 / n
    b_ub = np.full(n_lags, -scores + 0.5 * radius * dx_val)
    for k in active_set:
        b_ub[k] = -scores[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j] * 0.99), min(radius, (1.0 - h[j]) * 0.99)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(scores.max()) - 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None


def lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 4
    while time.time() < deadline:
        if stall > 120:
            break
        scores = lag_values(h)
        G = lag_jacobian(h)
        p = lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            n_active = min(n_active + 3, len(scores) - 1)
            if n_active > len(scores) // 3:
                p = lp_direction(h, scores, G, radius, max(4, len(scores) // 5))
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.72
                if radius < 1e-6:
                    break
                if stall % 30 == 0:
                    radius = max(radius, 0.04)
                    n_active = 5
                continue
        alpha, new_c5 = line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 4
            radius = min(radius * 1.05, 0.28)
        else:
            stall += 1
            radius *= 0.77
            if stall % 25 == 0 and radius < 0.02:
                radius = 0.06
                n_active = 6
    return best_h, best_c5


def _sqp_obj(x):
    if time.time() > _deadline:
        raise TimeoutError("hdl")
    return x[-1]


def _sqp_ineq_fun(x):
    if time.time() > _deadline:
        raise TimeoutError("hdl")
    n_h = len(x) - 1
    return x[-1] - lag_values(x[:n_h])


def _sqp_ineq_jac(x):
    n_h = len(x) - 1
    h = x[:n_h]
    G = lag_jacobian(h)
    J = np.empty((2 * n_h - 1, n_h + 1))
    J[:, :n_h] = -G
    J[:, n_h] = 1.0
    return J


def _sqp_eq_fun(x):
    n_h = len(x) - 1
    return x[:n_h].sum() - n_h / 2.0


def _sqp_eq_jac(x):
    n_h = len(x) - 1
    J = np.zeros(n_h + 1)
    J[:n_h] = 1.0
    return J


def slsqp_epigraph(h_seed, deadline, max_iter=250):
    global _deadline
    n = len(h_seed)
    c0 = eval_c5(h_seed)
    x0 = np.concatenate([h_seed, [c0 + 0.008]])
    bounds = [(0.0, 1.0)] * n + [(0.0, max(1.0, c0 + 0.15))]
    constraints = [
        {'type': 'ineq', 'fun': _sqp_ineq_fun, 'jac': _sqp_ineq_jac},
        {'type': 'eq', 'fun': _sqp_eq_fun, 'jac': _sqp_eq_jac}
    ]
    _deadline = deadline
    try:
        res = minimize(_sqp_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        h_new = project_box_sum(res.x[:n], n / 2.0)
        c5_new = eval_c5(h_new)
        if c5_new < c0 - 1e-12:
            return h_new, c5_new
    except (TimeoutError, Exception):
        pass
    _deadline = 0.0
    return h_seed, c0


def _lse_obj(x):
    if time.time() > _deadline:
        raise TimeoutError("hdl")
    h = x
    F = lag_values(h)
    m = float(F.max())
    beta = _lse_beta
    Z = np.exp((F - m) / beta)
    return m + beta * float(np.log(Z.sum()))


def _lse_grad(x):
    h = x
    F = lag_values(h)
    m = float(F.max())
    beta = _lse_beta
    Z = np.exp((F - m) / beta)
    P = Z / Z.sum()
    G = lag_jacobian(h)
    return P @ G


def _lse_eq_fun(x):
    return x.sum() - len(x) / 2.0


def _lse_eq_jac(x):
    return np.ones(len(x))


def lse_smoothing(h_seed, deadline, betas=None):
    global _lse_beta, _deadline
    if betas is None:
        c0 = eval_c5(h_seed)
        betas = [c0 * f for f in [0.4, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008]]
    h = h_seed.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    _deadline = deadline
    for beta in betas:
        if time.time() > deadline:
            break
        _lse_beta = beta
        h0 = h.copy()
        bounds = [(0.0, 1.0)] * len(h)
        constraints = [{'type': 'eq', 'fun': _lse_eq_fun, 'jac': _lse_eq_jac}]
        try:
            res = minimize(_lse_obj, h0, method='SLSQP', jac=_lse_grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-10})
            h_new = project_box_sum(res.x, len(h) / 2.0)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except (TimeoutError, Exception):
            pass
    _deadline = 0.0
    return best_h, best_c5


def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)


def coord_refine(h, deadline, n_iter=300, seed=123):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(seed)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-12:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5


def perturb_search(h, deadline, rng, base_scale=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scale = base_scale
    for _ in range(400):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, n / 2.0)
        sub_dl = min(deadline, time.time() + 5.0)
        h_t, c5_t = lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.006, scale * 0.87)
        else:
            scale = min(0.2, scale * 1.08)
    return best_h, best_c5


def multires(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    tried = set()
    order = list(range(0, 50))
    rng.shuffle(order)
    for dn in order:
        for sign in (1, -1):
            if dn == 0 and sign == -1:
                continue
            if time.time() > deadline:
                break
            n_try = n + sign * dn
            if n_try < 22 or n_try > 250 or n_try in tried:
                continue
            tried.add(n_try)
            h_r = resample(best_h, n_try)
            sub_dl = min(deadline, time.time() + 4.0)
            h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.1)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 80
        x = 2.0 * (np.arange(n0) + 0.5) / n0
        h0 = 0.5 + 0.35 * np.sin(np.pi * x)
        h0 = project_box_sum(h0, n0 / 2.0)

    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: LP polish (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LSE smoothing (18%)
    d2 = t0 + budget_s * 0.36
    h2, c5_2 = lse_smoothing(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP epigraph (18%)
    d3 = t0 + budget_s * 0.54
    h3, c5_3 = slsqp_epigraph(best_h, d3, max_iter=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (14%)
    d4 = t0 + budget_s * 0.68
    best_h, best_c5 = multires(best_h, d4, rng)

    # Phase 5: LSE + SLSQP again (12%)
    d5 = t0 + budget_s * 0.78
    h5, c5_5 = lse_smoothing(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    d5b = t0 + budget_s * 0.88
    h5b, c5_5b = slsqp_epigraph(best_h, d5b, max_iter=150)
    if c5_5b < best_c5:
        best_c5 = c5_5b
        best_h = h5b.copy()

    # Phase 6: Perturbation (6%)
    d6 = t0 + budget_s * 0.94
    best_h, best_c5 = perturb_search(best_h, d6, rng, base_scale=0.035)

    # Phase 7: Final LP + coord (6%)
    d7 = t0 + budget_s * 0.97
    h7, c5_7 = lp_loop(best_h, d7, radius_init=0.05)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    rem = t0 + budget_s - time.time()
    if rem > 4:
        best_h, best_c5 = coord_refine(best_h, t0 + budget_s - 1, n_iter=150)

    # Final validation
    n = len(best_h)
    best_h = project_box_sum(best_h, n / 2.0)
    final_c5 = eval_c5(best_h)
    return best_h, final_c5, n
