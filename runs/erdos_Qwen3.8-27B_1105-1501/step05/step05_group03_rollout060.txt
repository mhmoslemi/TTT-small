def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.2
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def pair_move_score(h, j, k, delta):

    h_new = np.clip(h_new, 0.0, 1.0)

    if rng is None:
        rng = np.random.default_rng(77)
    stagnant = 0
    tabu_keys = []

    for it in range(max_iters):
        if time.time() > deadline:
            break
        m_star = int(np.argmax(corr))
        s = m_star - (n - 1)
        g = _g_arr(h, s)
        sorted_desc = np.argsort(g)[::-1]
        sorted_asc = np.argsort(g)
        j, k = -1, -1
        found = False
        for a in range(min(12, len(sorted_desc))):
            for b in range(min(12, len(sorted_asc))):
                jj, kk = int(sorted_desc[a]), int(sorted_asc[b])
                if jj == kk:
                    continue
                key = 2 * n * min(jj, kk) + max(jj, kk)
                if key in tabu_keys:
                    continue
                found = True
                j, k = jj, kk
                break
            if found:
                break
        if not found:
            tabu_keys = tabu_keys[-20:]
            j = int(np.argmax(g))
            k = int(np.argmin(g))
            if j == k:
                break
        tabu_keys.append(2 * n * min(j, k) + max(j, k))
        if len(tabu_keys) > 40:
            tabu_keys = tabu_keys[-40:]
        od = _pair_opt(h, j, k)
        if abs(od) > 1e-12:
            h[j] += od
            h[k] -= od
            h = np.clip(h, 0.0, 1.0)
            nc = _ec5(h)
            if nc < cur - 1e-15:
                cur = nc
                corr = np.correlate(h, 1.0 - h, mode='full')
                stagnant = 0
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                h[j] -= od
                h[k] += od
                stagnant += 1
        else:
            stagnant += 1
        if stagnant >= stag_limit:
            sz = 0.015 + 0.015 * (stagnant // (2 * stag_limit))
            h = h + rng.normal(0, sz, n)
            h = _proj(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur = _ec5(h)
            stagnant = 0
            tabu_keys = []
    return best_h, best_c5

def _qp_init(n, alpha, rng=None):
    x = np.arange(n)
    v = ((x / n * alpha) % 1.0) < 0.5
    h = v.astype(float)
    h = _proj(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
        h = _proj(h, n / 2.0)
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _random_binary(n, rng):
    pos = rng.permutation(n)
    h = np.zeros(n)
    h[pos[:n // 2]] = 1.0
    return _proj(h, n / 2.0)

def _smooth_init(n, p, amp, rng=None):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2 * np.pi * p * x)
    if rng is not None:
        h += rng.normal(0, 0.01, n)
    return _proj(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.995
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    start_h = _proj(h0, n0 / 2.0)
    best_h = start_h.copy()
    best_c5 = _ec5(best_h)
    best_n = n0

    # Phase 1: Steepest descent from initial (40% budget)
    dl1 = t0 + budget_s * 0.40
    h1, c51 = _steepest(start_h, dl1, max_iters=100000, stag_limit=250, rng=rng)
    if c51 < best_c5:
        best_c5, best_h, best_n = c51, h1, n0

    # Phase 2: Multi-resolution / multi-init exploration (30% budget)
    dl2 = t0 + budget_s * 0.70
    alphas = [0.6180339887, 0.3819660113, 0.7071067812, 0.4142135624, 0.6545084972, 0.5]
    n_list = [64, 80, 96, 112, 128, 144, 160, 192]
    inits_per_n = 3
    for n_try in n_list:
        if time.time() > dl2:
            break
        for init_idx in range(inits_per_n):
            if time.time() > dl2:
                break
            if init_idx == 0:
                h_init = _resample(best_h, n_try)
            elif init_idx == 1:
                alpha = alphas[init_idx % len(alphas)]
                h_init = _qp_init(n_try, alpha, rng)
            else:
                h_init = _random_binary(n_try, rng)
            sub_dl = min(dl2, time.time() + budget_s * 0.018)
            ht, ct = _steepest(h_init, sub_dl, max_iters=20000, stag_limit=120, rng=rng)
            if ct < best_c5:
                best_c5, best_h, best_n = ct, ht, n_try

    # Phase 3: Continue optimizing best (20% budget)
    dl3 = t0 + budget_s * 0.90
    h3, c53 = _steepest(best_h, dl3, max_iters=100000, stag_limit=200, rng=rng)
    if c53 < best_c5:
        best_c5, best_h = c53, h3

    # Phase 4: Random restarts + quick optimize (remaining budget)
    dl4 = deadline
    scale = 0.03
    n_restarts = 0
    while time.time() < dl4 - 5.0:
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _proj(h_pert, best_n / 2.0)
        rem = max(1.0, dl4 - time.time() - 2.0)
        sub_dl = min(dl4, time.time() + min(8.0, rem * 0.3))
        hp, cp = _steepest(h_pert, sub_dl, max_iters=10000, stag_limit=80, rng=rng)
        n_restarts += 1
        if cp < best_c5 - 1e-12:
            best_c5, best_h = cp, hp
            scale = max(0.005, scale * 0.75)
        else:
            scale = min(0.08, scale * 1.2)

    # Final: fine steepest from best
    if time.time() < deadline - 2.0:
        hf, cf = _steepest(best_h, deadline, max_iters=50000, stag_limit=100, rng=rng)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # Ensure feasibility
    best_h = _proj(best_h, best_n / 2.0)
    final_c5 = _ec5(best_h)
    return best_h, final_c5, best_n
