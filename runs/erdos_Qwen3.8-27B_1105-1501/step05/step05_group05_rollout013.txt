    # Standard form: A_ub @ x <= b_ub
    # x = [d, t]
    # G[k]@d + 1*t <= scores[k]
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = scores.copy()
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad(h, dx, beta):
    """Compute smoothed gradient of C5 using log-sum-exp."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v - np.max(v))  # numerical stability
    w = exp_v / float(np.sum(exp_v))
    
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= np.mean(grad)  # zero mean to preserve sum constraint
    return grad

def _nesterov_smooth(h, deadline, beta_schedule=(30, 80, 200, 500), lr_init=0.03):
    """Smooth optimization with Nesterov momentum."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    y = h.copy()  # Nesterov lookahead
    v_mom = np.zeros(n)
    mu = 0.9  # momentum
    lr = lr_init
    bi = 0
    stall = 0
    
    while time.time() < deadline:
        if stall > 50:
            bi += 1
            if bi > 20:
                break
            stall = 0
            v_mom = np.zeros(n)
            lr = lr_init
            y = h.copy()
        
        beta = beta_schedule[bi % len(beta_schedule)]
        
        # Compute gradient at lookahead point
        g = _smooth_grad(y, dx, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            bi += 1
            continue
        g = g / gn
        
        # Nesterov update
        new_mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        v_mom = new_mu * v_mom + (1.0 - new_mu) * g
        vn = np.linalg.norm(v_mom)
        if vn < 1e-14:
            continue
        v_mom = v_mom / vn
        
        # Line search
        alpha, new_c5 = _line_search(h, lr * v_mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * v_mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # Update Nesterov state
            y = h + ((mu - 1.0) / new_mu) * (h - h + alpha * lr * v_mom)  # simplified
            # Actually simpler: just reset y to h after a step
            y = h.copy()
            mu = new_mu
            stall = 0
            lr = min(lr * 1.05, 0.30)
        else:
            stall += 1
            lr *= 0.8
            v_mom *= 0.9
            mu = max(mu * 0.95, 0.5)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50, rng=None):
    """Pair refinement with gradient-based pair selection."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        
        # Select pairs based on gradient magnitude
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        interior_idx = np.where(interior)[0]
        
        if len(interior_idx) < 2:
            break
        
        # Try pairs: for each interior point, try transferring to its neighbors
        # and to points far away (based on correlation structure)
        tried_pairs = set()
        
        # Gradient-based: compute which pairs most reduce the objective
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        
        for i in interior_idx:
            if time.time() > deadline:
                break
            # Try transferring from i to nearby and far points
            # Neighbor pairs
            for di in [-3, -2, -1, 1, 2, 3, 5, -5, 10, -10]:
                j = i + di
                if 0 <= j < n and interior[j] and (i, j) not in tried_pairs:
                    tried_pairs.add((min(i,j), max(i,j)))
                    d = np.zeros(n)
                    d[i] = 1.0
                    d[j] = -1.0
                    alpha, c5_try = _line_search(h, d, dx, 0.5)
                    if c5_try < best_c5 - 1e-13:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def _kick_reopt(h, deadline, rng, n_kicks=30):
