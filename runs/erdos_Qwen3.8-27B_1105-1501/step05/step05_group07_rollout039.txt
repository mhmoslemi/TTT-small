def _G_matrix(h):
    n = len(h)
    n_lags = 2 * n - 1
    d_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each i, we need h[i + d_vals] and h[i - d_vals] with 0 for out of range
    # i ranges from 0 to n-1
    # i + d_vals ranges from -(n-1) to 2n-2
    # i - d_vals ranges from -(n-1) to 2n-2
    
    # Create index arrays
    i_idx = np.arange(n)[:, None]  # shape (n, 1)
    d_arr = d_vals[None, :]  # shape (1, 2n-1)
    
    ip = i_idx + d_arr  # shape (n, 2n-1), values in [-(n-1), 2n-2]
    im = i_idx - d_arr  # shape (n, 2n-1), values in [-(n-1), 2n-2]
    
    # Clip to valid range, use 0 for out of range
    ip_clipped = np.clip(ip, 0, n-1)
    im_clipped = np.clip(im, 0, n-1)
    
    ip_valid = (ip >= 0) & (ip < n)
    im_valid = (im >= 0) & (im < n)
    
    G = np.where(ip_valid, h[ip_clipped], 0.0) + np.where(im_valid, h[im_clipped], 0.0)
    return G

    im_valid = (im >= 0) & (im < n)
    G = np.where(ip_valid, h[ip_clipped], 0.0) + np.where(im_valid, h[im_clipped], 0.0)
    return G
