        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(45):
        if t_hi - t_lo < 5e-13:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 70:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.68
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.28)
        else:
            stall += 1
            radius *= 0.70
            if stall % 18 == 0 and radius < 0.015:
                radius = 0.05
    return best_h, best_c5

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n//2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _periodic_pattern(n, period, fill):
    """Create a periodic pattern with given period and fill ratio."""
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < fill:
            h[i] = 1.0
    # Project to get exactly n/2 ones
    return _project(h, n/2.0)

def _block_pattern(n, block_size):
    """Create a block pattern: alternating filled/empty blocks."""
    h = np.zeros(n)
    for i in range(n):
        if (i // block_size) % 2 == 0:
            h[i] = 1.0
    return _project(h, n/2.0)

def _sinusoidal_pattern(n, freq, phase=0.0):
    """Create a sinusoidal threshold pattern."""
    x = np.arange(n) / n
    h = (np.sin(2 * np.pi * freq * x + phase) > 0).astype(float)
    return _project(h, n/2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap_optimize(h, deadline, n_tries=5000, radius=0.3):

    target = n / 2.0

def _sa_optimize(h, deadline, T_init=0.05, T_min=1e-4):
    """Simulated annealing on the continuous level."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 2000:
            break
        # Pick a move type
        move_type = rng.random()
        if move_type < 0.5:
            # Single coordinate perturbation (then project)
            i = int(rng.integers(0, n))
            delta = rng.normal(0, 0.15)
            h_new = h.copy()
            h_new[i] += delta
            h_new = _project(h_new, target)
        elif move_type < 0.8:
            # Pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                stall += 1
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search(h, d, dx, 0.3)
            h_new = h + alpha * d
            h_new = _project(h_new, target)
        else:
            # Block move: shift a small block
            block_len = int(rng.integers(2, max(3, n // 10)))
            start = int(rng.integers(0, n - block_len))
            shift = int(rng.integers(-n//4, n//4 + 1))
            h_new = h.copy()
            h_new[start + shift:start + shift + block_len] = h_new[start:start + block_len]
            # This doesn't preserve sum... just use perturbation
            h_new = h + rng.normal(0, 0.1, n)
            h_new = _project(h_new, target)
        
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        T = max(T * 0.9995, T_min)
    
    return best_h, best_c5

def _multi_tau_slsqp(h, deadline):
    """Try SLSQP with multiple tau values."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    taus = [0.5, 1.0, 2.0, 5.0, 10.0, 0.2, 0.1]
    for tau in taus:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 2:
            break
        max_iter = min(150, int(remaining * 30))
        h_t, c5_t = _slsqp_optimize(best_h, tau, deadline, max_iter=max_iter)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
    return best_h, best_c5

    dx = 2.0 / n

    # The directional derivative is grad[i] - grad[j]
    # We want this to be as negative as possible

        # i should have low grad, j should have high grad
        i = int(np.argmin(grad))
        j = int(np.argmax(grad))

            sorted_idx = np.argsort(grad)

def _soft_c5_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exps = np.exp((corr - max_c) / tau)
    s = float(np.sum(exps))
    val = dx * (max_c + tau * np.log(s))
    w = exps / s
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        d = (n - 1) - r
        if d == 0:
            grad += wr * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += wr * (1.0 - h[d:])
            grad[d:] += wr * (-h[:m])
        else:
            g = -d
            m = n - g
            grad[g:] += wr * (1.0 - h[:m])
            grad[:m] += wr * (-h[g:])
    grad *= dx
    return val, grad
