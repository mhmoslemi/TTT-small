```python
def _random_subgrad(h, topk=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Sample from top-k with softmax weights
    idx = np.argsort(corr)[::-1][:topk]
    f_k = corr[idx]
    # Use softmax for sampling weights
    shifted = (f_k - np.max(f_k)) * 50  # temperature
    exps = np.exp(shifted)
    probs = exps / np.sum(exps)
    # Sample one
    r = idx[np.random.choice(len(idx), p=probs)]
    d = r - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        grad[:m] += (1.0 - h[d:])
        grad[d:] -= h[:m]
    else:
        g = -d
        m = n - g
        grad[g:] += (1.0 - h[:m])
        grad[:m] -= h[g:]
    return -grad * dx  # negative for descent
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < 0.999999
        else: mask = x > 1e-7
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj_grad(h, beta, topk=None):
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode='full') * dx
    if topk is not None and topk < len(f):
        idx = np.argsort(f)[::-1][:topk]
    else:
        idx = np.arange(len(f))
    fk = f[idx]
    maxf = np.max(fk)
    shifted = np.clip(beta * (fk - maxf), -500, 500)
    exps = np.exp(shifted)
    total = float(np.sum(exps))
    obj = maxf + np.log(total) / beta
    weights = exps / total
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-12: continue
        r = idx[i]
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w * (1.0 - h[d:])
                grad[d:] -= w * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[g:]
    grad *= dx
    return obj, grad

def _smooth_descent(h, deadline, beta_init=200.0, topk_init=8):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = beta_init
    topk = topk_init
    rng_local = np.random.default_rng(777)
    no_improve = 0
    while time.time() < deadline:
        obj, grad = _smooth_obj_grad(h, beta, topk)
        d = -grad
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        best_alpha = 0.0
        best_val = _exact_c5(h)
        for alpha in np.concatenate([np.linspace(0.005, 0.3, 15), np.linspace(0.3, 1.5, 8)]):
            h_try = _project(h + alpha * d, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val - 1e-13:
                best_val = c5_try
                best_alpha = alpha
        if best_alpha > 0:
            h = _project(h + best_alpha * d, target)
            c5 = _exact_c5(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
                beta = min(beta * 1.15, 5000.0)
                topk = min(topk + 1, 2 * n - 1)
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                beta = max(beta_init * 0.5, beta * 0.6)
                topk = max(3, topk - 2)
                h_pert = _project(h + rng_local.normal(0, 0.05, n), target)
                if _exact_c5(h_pert) < best_c5:
                    h = h_pert
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, n_active=3):
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    active_set = set(sorted_idx[:n_active].tolist())
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    dx_val = 2.0 / n
    b_ub = np.full(n_lags, -scores + radius * dx_val)
    for k in active_set:
        b_ub[int(k)] = -scores[int(k)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x[-1] < float(np.max(scores)):
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    grid = np.linspace(t_lo, t_hi, 60)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < 59:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13: break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 3
    while time.time() < deadline:
        if stall > 60: break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            n_active = min(n_active + 2, 2 * n - 2)
            p = _lp_direction(h, scores, G, radius * 1.5, n_active)
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 2
                radius *= 0.7
                if radius < 1e-7: break
                if stall % 20 == 0:
                    radius = max(radius, 0.05)
                continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 3
            radius = min(radius * 1.05, 0.35)
        else:
            stall += 1
            radius *= 0.8
            if stall % 15 == 0 and radius < 0.03:
                radius = 0.08
                n_active = 5
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coord_refine(h, deadline, n_iter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(99999)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(interior) < 2:
        return best_h, best_c5
    for it in range(n_iter):
        if time.time() > deadline: break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j: continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1.0 - h[j], 0.5))
        if c5_try < best_c5 - 1e-13:
            h = _project(h + alpha * d, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                if len(interior) < 2: break
    return best_h, best_c5

def _basin_jump(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for attempt in range(200):
        if time.time() > deadline: break
        pert_type = rng.integers(0, 4)
        if pert_type == 0:
            h_pert = _project(best_h + rng.normal(0, 0.15, n), target)
        elif pert_type == 1:
            L = rng.integers(3, max(4, n // 4))
            start = rng.integers(0, n - L)
            h_pert = best_h.copy()
            h_pert[start:start+L] = 1.0 - h_pert[start:start+L]
            h_pert = _project(h_pert, target)
        elif pert_type == 2:
            shift = rng.integers(1, max(2, n // 3))
            h_pert = np.roll(best_h, shift)
            h_pert = _project(h_pert, target)
        else:
            h_pert = _project(best_h + rng.uniform(-0.2, 0.2, n), target)
        if np.allclose(h_pert, best_h): continue
        sub_dl = min(deadline, time.time() + 3.0)
        h_candidate, c5_cand = _lp_loop(h_pert, sub_dl, radius_init=0.1)
        if c5_cand < best_c5 - 1e-11:
            best_c5 = c5_cand
            best_h = h_candidate.copy()
    return best_h, best_c5

def _multires(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
```
