    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    c = n - 1
    for d in range(n):
        m = n - d
        G[c + d, d:] += dx * (1.0 - h[:m])
        G[c + d, :m] += -dx * h[d:]
    for s in range(1, n):
        m = n - s
        G[c - s, :m] += dx * (1.0 - h[s:])
        G[c - s, s:] += -dx * h[:m]
    return G

def _pair_descent(h, deadline, grid_k=40, max_passes=1000, thresh=1e-10):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    dx = 2.0 / n
    c = n - 1
    n_lags = 2 * n - 1
    
    # Precompute q vectors
    q_by_d = [None] * n
    for d in range(n):
        q = np.zeros(n_lags)
        q[c] = -2.0
        if d > 0:
            if c + d < n_lags:
                q[c + d] = 1.0
            if c - d >= 0:
                q[c - d] = 1.0
        q_by_d[d] = q
    
    stall = 0
    for p in range(max_passes):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        pass_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                if hi < 1e-12 and hj < 1e-12:
                    continue
                if hi > 1 - 1e-12 and hj > 1 - 1e-12:
                    continue
                a_lo = max(-hi, hj - 1.0)
                a_hi = min(1.0 - hi, hj)
                if a_hi - a_lo < 1e-13:
                    continue
                d = j - i
                C = G[:, i] - G[:, j]
                q = q_by_d[d]
                
                # Quick check: is there any potential improvement?
                # The best possible improvement is bounded by the change in the max score
                # Skip if the gradient difference is too small
                c_max = float(np.max(np.abs(C)))
                if c_max < 1e-15:
                    continue
                
                # Vectorized grid search
                alphas = np.linspace(a_lo, a_hi, grid_k)
                aa = (dx * alphas * alphas)[:, None]
                vals = np.max(scores[None, :] + alphas[:, None] * C[None, :] + aa * q[None, :], axis=1)
                best_idx = int(np.argmin(vals))
                best_val = float(vals[best_idx])
                
                if best_val < cur_c5 - thresh:
                    # Refine around best
                    lo_r = alphas[max(0, best_idx - 1)]
                    hi_r = alphas[min(grid_k - 1, best_idx + 1)]
                    fine = np.linspace(lo_r, hi_r, 15)
                    faa = (dx * fine * fine)[:, None]
                    fvals = np.max(scores[None, :] + fine[:, None] * C[None, :] + faa * q[None, :], axis=1)
                    fidx = int(np.argmin(fvals))
                    best_a = float(fine[fidx])
                    best_val = float(fvals[fidx])
                    
                    if best_val < cur_c5 - thresh:
                        h[i] = hi + best_a
                        h[j] = hj - best_a
                        if h[i] < 0: h[i] = 0.0
                        elif h[i] > 1: h[i] = 1.0
                        if h[j] < 0: h[j] = 0.0
                        elif h[j] > 1: h[j] = 1.0
                        # Update hi for next j in this i-loop
                        hi = h[i]
                        cur_c5 = _exact_c5(h)
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True
                        stall = 0
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                else:
                    stall += 1
    return best_h, best_c5

def _grad_descent(h, deadline, n_steps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _build_G(h)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = float(scores.max())
        # Softmax weights on top scores
        beta = 50.0
        w = np.exp(beta * (scores - max_s))
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        # Gradient direction
        grad = G.T @ w
        grad -= grad.mean()  # maintain sum constraint approximately
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        grad /= gn
        
        # Line search
        best_lr = 0.0
        best_new_c5 = cur_c5
        lr = 0.1
        for _ in range(20):
            h_new = _project(h - lr * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_new_c5:
                best_new_c5 = c5_new
                best_lr = lr
                h_new_copy = h_new.copy()
            lr *= 0.7
            if lr < 1e-8:
                break
        
        if best_lr > 0 and best_new_c5 < cur_c5 - 1e-12:
            h = h_new_copy
            cur_c5 = best_new_c5
            G = _build_G(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0 < 0.5).astype(float)
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _binary_init(n, rng):
    # Random binary with exactly n/2 ones
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
