def _project(v, target):

    n_lags = len(scores)
    # Build gradient matrix G (n_lags x n)
    G = np.zeros((n_lags, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])  # +(1-h[i+d]) for h[i]
                G[r, d:] += (-h[:m])  # -h[i] for h[i+d]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])  # +(1-h[i]) for h[i+g]... 
                # Actually let me redo this
                # For negative lag d=-g: c_d = sum_{i=g}^{n-1} h[i]*(1-h[i-g])
                # = sum_{j=0}^{n-g-1} h[j+g]*(1-h[j])
                # d_c/dh[j+g] += (1-h[j]) for j=0..n-g-1
                # d_c/dh[j] += -h[j+g] for j=0..n-g-1
                G[r, g:n] += (1.0 - h[:m])  # derivative w.r.t. h[j+g]
                G[r, :m] += (-h[g:n])  # derivative w.r.t. h[j]
    G *= (2.0 / n)
    
    # LP: minimize t s.t. G@p <= -scores, t >= |p_j|, sum p = 0
    # Actually: minimize t s.t. scores + G@p <= max_score (or similar)
    # Let me use the formulation from the previous code
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None
