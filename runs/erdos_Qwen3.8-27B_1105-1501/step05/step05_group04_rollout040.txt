For each lag r, score_r(h + t*d) = score_r(h) + t * L_r + t² * Q_r

Let me compute Q_r more carefully.

For lag s ≥ 0 (r = n-1+s):
score_s = sum_{j=0}^{n-1-s} (h[j+s] + t*d[j+s]) * (1 - h[j] - t*d[j])
= sum_{j=0}^{n-1-s} [h[j+s](1-h[j]) + t*h[j+s]*(-d[j]) + t*d[j+s]*(1-h[j]) + t²*(-d[j+s]*d[j])]

So L_s = sum_j [h[j+s]*(-d[j]) + d[j+s]*(1-h[j])]

Q_s = -sum_j d[j+s]*d[j]

For lag -a (r = n-1-a):
score_{-a} = sum_{j=a}^{n-1} (h[j-a] + t*d[j-a]) * (1 - h[j] - t*d[j])
= sum_{j=a}^{n-1} [h[j-a](1-h[j]) + t*h[j-a]*(-d[j]) + t*d[j-a]*(1-h[j]) + t²*(-d[j-a]*d[j])]

L_{-a} = sum_{j=a}^{n-1} [d[j-a]*(1-h[j]) - h[j-a]*d[j]]

G.T @ w where G is (2n-1) × n and w is (2n-1) vector.
This is equivalent to: for each h[i], compute sum over all lags r of w[r] * G[r, i].

G[r, i] depends on the relationship between r and i. For a fixed i, which lags r have G[r, i] ≠ 0?

For lag s (r = n-1+s, s ≥ 0):
G[r, i] = (1-h[i-s]) * [s ≤ i] - h[i+s] * [i ≤ n-1-s]
So G[r, i] ≠ 0 when s ≤ i (first term) or i ≤ n-1-s (second term), i.e., s ≤ i or s ≤ n-1-i.
So s ranges from 0 to max(i, n-1-i).

For lag -a (r = n-1-a, a ≥ 1):
G[r, i] = (1-h[i+a]) * [i ≤ n-1-a] - h[i-a] * [i ≥ a]
So G[r, i] ≠ 0 when i ≤ n-1-a or i ≥ a.

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for s in range(n):
        r = n - 1 + s
        if s == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        else:
            m = n - s
            G[r, s:] += dx * (1.0 - h[:m])
            G[r, :m] += -dx * h[s:]
    for a in range(1, n):
        r = n - 1 - a
        m = n - a
        G[r, :m] += dx * (1.0 - h[a:])
        G[r, a:] += -dx * h[:m]
    return G

        return float(np.max(S0 + t * L + t * t * Q))
    n_grid = 80

    else:

    gr = (np.sqrt(5.0) - 1.0) / 2.0

def _lp_direction(h, scores, G, k=15, radius=0.1):

    c[-1] = 1.0
    top_k = np.argsort(scores)[::-1][:k]
    A_ub = np.zeros((k, n + 1))
    for i in range(k):
        A_ub[i, :n] = G[top_k[i], :]
        A_ub[i, -1] = -1.0
    b_ub = np.zeros(k)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:

def _softmax_direction(h, G, beta=50, radius=0.1):
    n = len(h)
    scores = _scores(h)
    m = scores.max()
    w = np.exp(beta * (scores - m))
    w = w / np.sum(w)
    g = G.T @ w
    d = -g
    d = d - np.mean(d)
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-15:
        return None
    d = d * (radius / max_abs)
    d = np.clip(d, -h, 1.0 - h)
    d = d - np.mean(d)
    if np.max(np.abs(d)) < 1e-14:
        return None
    return d

        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)
    return best_h, best_c5

def _multiresolve(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    tried = set()
    for dn in range(0, 20):
        for sign in (1, -1):
            if dn == 0 and sign == -1:
                continue
            if time.time() > deadline:
                break
            n_try = n + sign * dn
            if n_try < 30 or n_try > 180 or n_try in tried:
                continue
            tried.add(n_try)
            h_r = _resample(best_h, n_try)
            sub_dl = min(deadline, time.time() + 5.0)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
    return best_h, best_c5

def _slsqp_polish(h, deadline, n_sub=5):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    for i in range(n_sub):
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 30.0)
        beta = 60 + 40 * i
        h_new, c5_new = _lfbfgs_lse(best_h, sub_dl, beta=beta)
        if h_new is not None and c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def _symmetric_init(n):
    half = n // 2
    h = np.empty(n)
    x = (np.arange(half) + 0.5) / half
    h[:half] = 0.5 + 0.3 * np.sin(np.pi * 2 * x) + 0.15 * np.sin(np.pi * 4 * x + 0.7)
    if n % 2 == 1:
        h[half] = 0.5
        h[half + 1:] = 1.0 - h[half - 1::-1]
    else:
        h[half:] = 1.0 - h[half - 1::-1]
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
        if n0 < 10 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n0 = 95
        h0 = _symmetric_init(n0)

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5_exact(h)

    # Phase 1: Quick LP polish from parent
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Main LP optimization
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.15)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: L-BFGS with log-sum-exp
    d3 = t0 + budget_s * 0.62
    h3, c5_3 = _slsqp_polish(best_h, d3, n_sub=6)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution
    d4 = t0 + budget_s * 0.74
    best_h, best_c5 = _multiresolve(best_h, d4, rng)

    # Phase 5: Perturbation + LP
    d5 = t0 + budget_s * 0.86
    best_h, best_c5 = _perturb_refine(best_h, d5, rng, base_scale=0.03)

    # Phase 6: Multi-resolution again
    d6 = t0 + budget_s * 0.92
    best_h, best_c5 = _multiresolve(best_h, d6, rng)

    # Phase 7: Coordinate descent
    d7 = t0 + budget_s * 0.96
    best_h, best_c5 = _coord_descent(best_h, d7, n_iter=400)

    # Phase 8: Final LP polish
    d8 = deadline
    h_f, c5_f = _lp_loop(best_h, d8, radius_init=0.06)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    n_f = len(best_h)
    best_h = _project(best_h, n_f / 2.0)
    final_c5 = _c5_exact(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
