def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for s in range(-(n-1), n):
        r = n - 1 + s
        if s >= 0:
            if s == 0:
                J[r, :] = dx * (1.0 - 2.0 * h)
            else:
                m = n - s
                J[r, s:] += dx * (1.0 - h[:m])
                J[r, :m] += -dx * h[s:]
        else:
            a = -s
            m = n - a
            J[r, :m] += dx * (1.0 - h[a:])
            J[r, a:] += -dx * h[:m]
    return J

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    obj_grid = np.max(vals, axis=1) * dx

    if len(combined) > 30:
        combined = active
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(combined), n + 1))
    A_ub[:, :n] = G[combined, :]
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    b_ub = -scores[combined]
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))

            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-12:

        pass
    return None

def _smooth_opt(h, beta, deadline, maxiter=30):

        Z = np.exp((F - m) / beta)
        P = Z / np.sum(Z)
        G = _lag_jacobian(x)
        return P @ G

    c5_try = _eval_c5(h_try)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    scale = 0.03
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, target)
        sub_dl = min(deadline, time.time() + 3.0)
        h_t, c5_t = _lp_loop(h_p, sub_dl, radius_init=0.08, n_active_init=4)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)
    return best_h, best_c5

class TimeoutError(Exception):
    pass

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    h = np.array(initial_h_values, dtype=float)
    n0 = len(h)
    h = _project(h, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    
    # Phase 1: LP optimization from parent (30%)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.15, n_active_init=6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: SLSQP epigraph (15%)
    d2 = t0 + budget_s * 0.45
    if time.time() < d2 - 5:
        h2, c5_2 = _slsqp_epigraph_safe(best_h, d2, rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Phase 3: Multi-resolution exploration (20%)
    d3 = t0 + budget_s * 0.65
    tried = set()
    n_current = len(best_h)
    for _ in range(40):
        if time.time() >= d3:
            break
        dn = int(rng.integers(-15, 20))
        n_try = n_current + dn
        if n_try < 30 or n_try > 180 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _eval_c5(h_r)
        sub_dl = min(d3, time.time() + 4.0)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1, n_active_init=5)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 4: Perturbation + LP (15%)
    d4 = t0 + budget_s * 0.80
    if time.time() < d4 - 5:
        best_h, best_c5 = _perturb_lp(best_h, d4, rng, n_trials=40)
    
    # Phase 5: SLSQP again from improved point (10%)
    d5 = t0 + budget_s * 0.90
    if time.time() < d5 - 5:
        h5, c5_5 = _slsqp_epigraph_safe(best_h, d5, rng)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    
    # Phase 6: Final LP polish (5%)
    d6 = t0 + budget_s * 0.95
    if time.time() < d6 - 3:
        h6, c5_6 = _lp_loop(best_h, d6, radius_init=0.06, n_active_init=8)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Phase 7: Coordinate refinement (remaining)
    rem = deadline - 2.0 - time.time()
    if rem > 5:
        best_h, best_c5 = _coord_refine(best_h, deadline - 2.0, n_iter=500, rng=rng)
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _eval_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    
    return best_h, final_c5, len(best_h)
