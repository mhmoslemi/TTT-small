import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1) if res > 0 else (x > 0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lp_direction(h, G, scores, radius, top_k):
    n = len(h)
    n_lags = 2 * n - 1
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    n_cons = len(s_k)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.array([[1.0] * n + [0.0]])
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, deadline):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    ng = 15
    alphas = np.linspace(0, alpha_max, ng)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0:
        return 0.0, float(vals[0])
    if best_i == ng - 1:
        return alpha_max, float(vals[-1])
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(ng - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-16 or time.time() > deadline:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < float(vals[best_i]):
        return opt_a, opt_c5
    return alphas[best_i], float(vals[best_i])

def _local_opt(h, deadline, top_k_list=None, radius_init=0.15, max_iters=10000):
    if top_k_list is None:
        top_k_list = [None, 15, 10, 6, 4, None]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_tk = len(top_k_list)
    ti = 0
    G = _build_G(h)
    scores = _scores(h)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        tk = top_k_list[ti % n_tk]
        p = _lp_direction(h, G, scores, radius, tk)
        if p is None or np.max(np.abs(p)) < 1e-15:
            stall += 1
            radius *= 0.65
            ti += 1
            if radius < 1e-9:
                break
            continue
        alpha, new_c5 = _line_search(h, p, deadline)
        if alpha < 1e-14:
            stall += 1
            radius *= 0.7
            ti += 1
            continue
        h_cand = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_cand)
        if actual_c5 < cur_c5 - 1e-14:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.4)
            G = _build_G(h)
            scores = _scores(h)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 15 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _anneal_local(h, deadline, n_iter=2000, T_start=0.01, T_end=0.001):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    T = T_start
    for i in range(n_iter):
        if time.time() > deadline:
            break
        frac = i / max(1, n_iter - 1)
        T = T_start + (T_end - T_start) * frac
        idx = rng.integers(0, n)
        jdx = rng.integers(0, n)
        if idx == jdx:
            continue
        amt = rng.uniform(-T, T)
        h2 = h.copy()
        h2[idx] += amt
        h2[jdx] -= amt
        if h2[idx] < 0 or h2[idx] > 1 or h2[jdx] < 0 or h2[jdx] > 1:
            continue
        c5_new = _exact_c5(h2)
        if c5_new < cur_c5 or rng.random() < np.exp((cur_c5 - c5_new) / max(T, 1e-10)):
            h = h2
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def _pair_snip(h, deadline, max_passes=100):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    active_pairs = [(i, j) for i in range(n) for j in range(i+1, n)
                    if 1e-8 < h[i] < 1-1e-8 and 1e-8 < h[j] < 1-1e-8]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(hash((_pass, n)) % (2**31))
        order = list(range(len(active_pairs)))
        rng.shuffle(order)
        for oi in order:
            if time.time() > deadline:
                break
            i, j = active_pairs[oi]
            C = G[:, i] - G[:, j]
            C_norm = np.linalg.norm(C)
            if C_norm < 1e-14:
                continue
            a_lo = max(-h[i], h[j] - 1.0)
            a_hi = min(1.0 - h[i], h[j])
            if a_hi - a_lo < 1e-12:
                continue
            lo_a, hi_a = a_lo, a_hi
            for _ in range(16):
                m1 = lo_a + (hi_a - lo_a) / 3.0
                m2 = hi_a - (hi_a - lo_a) / 3.0
                f1 = np.max(scores + m1 * C)
                f2 = np.max(scores + m2 * C)
                if f1 < f2:
                    hi_a = m2
                else:
                    lo_a = m1
            alpha = (lo_a + hi_a) / 2.0
            if abs(alpha) < 1e-10:
                continue
            h[i] += alpha
            h[j] -= alpha
            actual_c5 = _exact_c5(h)
            if actual_c5 < cur_c5 - 1e-14:
                cur_c5 = actual_c5
                scores = _scores(h)
                G = _build_G(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] -= alpha
                h[j] += alpha
        if not improved:
            break
    return best_h, best_c5

def _subgrad_step(h, deadline, n_steps=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    G = _build_G(h)
    scores = _scores(h)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        max_s = scores.max()
        active = scores > max_s - 0.02
        if not np.any(active):
            active = scores >= max_s
        w = np.zeros(len(scores))
        w[active] = 1.0
        w /= w.sum()
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        lr = 0.08
        for _ in range(20):
            h_new = _project(h - lr * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                G = _build_G(h)
                scores = _scores(h)
                break
            lr *= 0.65
            if lr < 1e-10:
                break
    return best_h, best_c5
