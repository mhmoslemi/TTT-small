import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.5
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        d = k - (n - 1)
        if d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        else:
            a = abs(d)
            m = n - a
            if m <= 0:
                continue
            G[k, a:] += dx * (1.0 - h[:m])
            G[k, :m] += -dx * h[a:]
    return G

def _lp_refine(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline and stall < 120:
        scores = _scores(h)
        G = _grad_matrix(h)
        n_lags = 2 * n - 1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((0, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'time_limit': 2})
            if not res.success:
                stall += 1
                radius *= 0.8
                continue
            d = res.x[:n]
        except Exception:

            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        if float(np.max(np.abs(d))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:

            continue
        t_hi = radius
        t_lo = 0.0

        if t_hi - t_lo < 1e-14:
            stall += 1
            radius *= 0.8
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_g, b_g = t_lo, t_hi
        c1 = b_g - gr * (b_g - a_g)
        c2 = a_g + gr * (b_g - a_g)

                break

                b_g = c2
                c2 = c1
                f2 = f1
                c1 = b_g - gr * (b_g - a_g)

            else:
                a_g = c1
                c1 = c2
                f1 = f2
                c2 = a_g + gr * (b_g - a_g)

def _softmax_refine(h, deadline, T_init=40.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    lr = 0.008

    beta = 0.85
    for _ in range(2000):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = float(np.max(scores))
        shifted = T * (scores - max_s)
        shifted = np.clip(shifted, -500, 500)
        exps = np.exp(shifted)
        w = exps / float(np.sum(exps))
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        momentum = beta * momentum + (1 - beta) * grad
        h_new = _project(h - lr * momentum, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new

        else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_epigraph(h, deadline):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    
    def obj_and_jac(x):
        h_vec = x[:n]
        t_val = x[n]
        h_clip = np.clip(h_vec, 0, 1)
        scores = _scores(h_clip)
        f = t_val
        # Gradient of objective w.r.t. x
        g = np.zeros(n + 1)
        g[n] = 1.0
        # Constraint: t - scores >= 0
        G = _grad_matrix(h_clip)
        c = t_val - scores
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        # Equality: sum(h) - target = 0
        eq_val = float(np.sum(h_vec)) - target
        J_eq = np.zeros(n + 1)
        J_eq[:n] = 1.0
        # Combined for SLSQP
        all_c = np.concatenate([c, [eq_val]])
        all_J = np.vstack([J, J_eq])
        return f, g, all_c, all_J
    
    # Use SLSQP with the epigraph formulation
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = cur_c5 + 0.001
    
    def sobj(x):
        return x[n]
    
    def sobj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def constraints_func(x):
        h_clip = np.clip(x[:n], 0, 1)
        scores = _scores(h_clip)
        G = _grad_matrix(h_clip)
        ineq = x[n] - scores
        ineq_J = np.zeros((2 * n - 1, n + 1))
        ineq_J[:, :n] = -G
        ineq_J[:, n] = 1.0
        eq = float(np.sum(x[:n])) - target
        eq_J = np.zeros(n + 1)
        eq_J[:n] = 1.0
        return ineq, ineq_J, eq, eq_J
    
    def wrap_constraints(x):
        ineq, ineq_J, eq, eq_J = constraints_func(x)
        return {
            'ineq1': {'fun': lambda: ineq, 'jac': lambda: ineq_J},
            'eq1': {'fun': lambda: eq, 'jac': lambda: eq_J}
        }
    
    # SLSQP requires dict form or func form
    # Let's use the older scipy interface
    def cons_fun(x):
        h_clip = np.clip(x[:n], 0, 1)
        scores = _scores(h_clip)
        ineq = x[n] - scores
        eq = float(np.sum(x[:n])) - target
        return np.concatenate([ineq, [eq]])
    
    def cons_jac(x):
        h_clip = np.clip(x[:n], 0, 1)
        G = _grad_matrix(h_clip)
        J = np.zeros((2 * n, n + 1))
        J[:2*n-1, :n] = -G
        J[:2*n-1, n] = 1.0
        J[2*n-1, :n] = 1.0
        return J
    
    bounds = [(0, 1)] * n + [(cur_c5 - 0.01, 1.0)]
    
    try:
        res = minimize(sobj, x0, jac=sobj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints={'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac},
                      options={'maxiter': 300, 'ftol': 1e-12})
        if res.success:
            h_sol = np.clip(res.x[:n], 0, 1)
            h_sol = _project(h_sol, target)
            c5_sol = _exact_c5(h_sol)
            if c5_sol < cur_c5 - 1e-14:
                return h_sol, c5_sol
    except Exception:
        pass
    return h, cur_c5

def _coord_descent(h, deadline, n_iter=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if len(interior) < 2:
            break
        i = interior[rng.integers(len(interior))]
        j = interior[rng.integers(len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha_max = min(h[i], 1.0 - h[j], 0.5)
        # Simple line search
        best_alpha = 0.0
        best_val = best_c5
        for frac in [0.25, 0.5, 0.75, 1.0]:
            a = alpha_max * frac
            h_try = h + a * d
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = a
        if best_alpha > 0:
            h = _project(h + best_alpha * d, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    return best_h, best_c5

def _perturb_search(h, deadline, rng, base_scale=0.03):

    scale = base_scale
    while time.time() < deadline:

            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)
    return best_h, best_c5
