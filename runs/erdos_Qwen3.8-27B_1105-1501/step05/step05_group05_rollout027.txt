def _subgrad_direction(h, dx, beta):
    """Compute a subgradient direction using log-sum-exp weights."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    # Gradient computation
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d_lag = r - (n - 1)
        if d_lag >= 0:
            m = n - d_lag
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d_lag:d_lag + m])
                grad[d_lag:] += w[r] * (-h[:m])
        else:
            g = -d_lag
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _lp_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgrad_dir(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-16:
            continue
        d_lag = r - (n - 1)
        if d_lag >= 0:
            m = n - d_lag
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d_lag:d_lag + m])
                grad[d_lag:] += w[r] * (-h[:m])
        else:
            g = -d_lag
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
                        any_improve = True
                        interior = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
                        break
                if time.time() > deadline:
                    break
            if not improved and not any_improve:
                break

    return best_h, best_c5

def _lp_budget(h, deadline, radius_init=0.18):
    """LP trust region for smaller problems."""
    n = len(h)
    if n > 350:
        return h.copy(), _exact_c5(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

    while time.time() < deadline:
        if stall > 40:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = np.zeros((2 * n - 1, n))
        for d_lag in range(-(n - 1), n):
            r = n - 1 + d_lag
            if d_lag >= 0:
                m = n - d_lag
                if m > 0:
                    G[r, d_lag:] += (1.0 - h[:m])
                    G[r, :m] += -h[d_lag:]
            else:
                g = -d_lag
                m = n - g
                if m > 0:
                    G[r, :m] += (1.0 - h[g:])
                    G[r, g:] += -h[:m]

        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores.copy()
        A_eq = np.array([[1.0] * n + [0.0]])
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        for j in range(n):
            if bounds[j][0] > bounds[j][1]:
                bounds[j] = (0.0, 0.0)
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if not res.success:
                stall += 1
                radius *= 0.75
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.75
            continue

        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
            continue

        alpha, _ = _line_search(h, p, 2.0 / n, radius)
        h_new = _project(h + alpha * p, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.50)
        else:
            stall += 1
            radius *= 0.72
            if stall > 20 and radius < 0.03:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick optimization at original resolution (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _hi_res_optimize(h, d1, rng)
    if c5_1 < best_c5:
        best_h, best_c5 = h1.copy(), c5_1

    # Phase 2: Multi-resolution search (55%)
    # Try increasing resolutions, keep the best
    n_trials = sorted(set([int(n0 * 1.4), int(n0 * 1.7), int(n0 * 2.0),
                           int(n0 * 2.4), int(n0 * 2.8), int(n0 * 3.3)]))
    n_trials = [x for x in n_trials if 140 <= x <= 500]
    # Sort: try closer to original first (cheaper resampling)
    n_trials.sort(key=lambda x: abs(x - n0))

    for n_t in n_trials:
        if time.time() > deadline - 80:
            break
        remaining = deadline - 80 - time.time()
        if remaining < 15:
            break
        # Allocate time: more for higher resolutions
        alloc = min(remaining * 0.35, budget_s * 0.13)
        sub_dl = time.time() + alloc

        h_t = _resample(best_h, n_t)
        c5_init_t = _exact_c5(h_t)

        # If resampling already gives a good starting point, optimize
        h_opt, c5_opt = _hi_res_optimize(h_t, sub_dl, rng)

        # If promising and we have time, add LP polish
        if c5_opt < best_c5 + 0.002 and time.time() < deadline - 30:
            lp_dl = min(deadline - 30, time.time() + min(budget_s * 0.06, 30.0))
            if len(h_opt) <= 300:
                h_lp, c5_lp = _lp_budget(h_opt, lp_dl, radius_init=0.12)
                if c5_lp < c5_opt:
                    h_opt, c5_opt = h_lp, c5_lp

        if c5_opt < best_c5 - 1e-10:
            best_h, best_c5 = h_opt.copy(), c5_opt

    # Phase 3: Focused pair refinement at best resolution (20%)
    d3 = t0 + budget_s * 0.85
    if time.time() < d3:
        h3, c5_3 = _pair_refine_focused(best_h, d3, rng)
        if c5_3 < best_c5:
            best_h, best_c5 = h3.copy(), c5_3

    # Phase 4: Final polish (rest)
    if time.time() < deadline - 5:
        if len(best_h) <= 300:
            h4, c5_4 = _lp_budget(best_h, deadline, radius_init=0.08)
            if c5_4 < best_c5:
                best_h, best_c5 = h4.copy(), c5_4
        else:
            h4, c5_4 = _pair_refine_focused(best_h, deadline, rng)
            if c5_4 < best_c5:
                best_h, best_c5 = h4.copy(), c5_4

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
