def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

            lo = mid
        else:

def line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    def obj_at(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = obj_at(c1)
    f2 = obj_at(c2)
    for _ in range(50):

            break

            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = obj_at(c1)

            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = obj_at(c2)

    # Also check endpoints
    f_lo = obj_at(t_lo)
    f_hi = obj_at(t_hi)
    best_t, best_val = opt_t, opt_val
    if f_lo < best_val:
        best_t, best_val = t_lo, f_lo
    if f_hi < best_val:
        best_t, best_val = t_hi, f_hi
    return best_t, best_val

def grad_lag(h, k):
    """Gradient of F_k(h) with respect to h.
    k is the shift: for k>=0, F_k = sum h[i+k]*(1-h[i])
    for k<0, let j=-k, F_k = sum h[i]*(1-h[i-j])
    """

    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            # h[i+k]*(1-h[i]) for i=0..m-1
            # d/dh[m] for m in [k, n-1]: (1-h[m-k])
            # d/dh[m] for m in [0, m-1]: -h[m+k]
            g[k:] += (1.0 - h[:m])
            g[:m] += -h[k:]
    else:

            # d/dh[m] for m in [j, n-1]: (1-h[m-j])
            # d/dh[m] for m in [0, m-1]: -h[m+j]
            g[j:] += (1.0 - h[:m])
            g[:m] += -h[j:]
    return g

def smooth_grad(h, dx, beta):
    """Log-sum-exp smoothed gradient of max of scores."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    # Clip to prevent overflow
    v = np.clip(v, -500, 0)
    exp_v = np.exp(v)
    total = float(np.sum(exp_v))
    if total < 1e-300:
        return np.zeros(n)
    w = exp_v / total
    grad = np.zeros(n)

        if w[r] < 1e-15:
            continue
        k = r - (n - 1)
        grad += w[r] * grad_lag(h, k)
    grad -= np.mean(grad)
    return grad

        return x[n]

        return g

        def fun(x):

            Fk = float(np.dot(grad_lag(h_x, k), h_x))  # This is wrong, let me compute F_k directly
