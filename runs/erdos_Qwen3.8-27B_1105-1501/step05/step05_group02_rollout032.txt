        count = c_pos + c_neg

    return best

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = np.max(scores)
    shifted = np.clip((scores - max_s) / tau, -50, 50)
    exp_vals = np.exp(shifted)
    logsum = max_s + tau * np.log(np.sum(exp_vals))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for d in range(0, n):
        r_pos = n - 1 + d
        r_neg = n - 1 - d
        m = n - d
        if m > 0:
            grad[d:d+m] += w[r_pos] * dx * (1.0 - h[0:m])
            grad[0:m] += w[r_pos] * dx * (-1.0) * h[d:d+m]
            if d > 0:
                grad[0:m] += w[r_neg] * dx * (1.0 - h[d:d+m])
                grad[d:d+m] += w[r_neg] * dx * (-1.0) * h[0:m]
    return logsum, grad

def _smooth_optimize(h0, deadline, tau_scale=0.06, n_stages=6):
    n = len(h0)
    target = n / 2.0
    cur_c5 = _exact_c5(h0)
    best_h = h0.copy()
    best_c5 = cur_c5
    h_cur = h0.copy()
    tau = cur_c5 * tau_scale
    tau_min = max(0.001, cur_c5 * 0.003)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    for stage in range(n_stages):
        if time.time() > deadline:
            break
        if tau < tau_min:
            break
        def obj(x):
            val, _ = _smooth_obj_grad(x, tau)
            return val
        def grad_fn(x):
            _, g = _smooth_obj_grad(x, tau)
            return g
        try:
            res = minimize(obj, h_cur, jac=grad_fn, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 80, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
        tau *= 0.5
    return best_h, best_c5

def _binary_sa(h0, deadline, rng, max_iters=200000, T_init=0.012, cool=0.99995):
    n = len(h0)
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    T = T_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i_idx = int(rng.integers(len(ones_list)))
        j_idx = int(rng.integers(len(zeros_list)))
        i = ones_list[i_idx]
        j = zeros_list[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones_list[i_idx] = j
            zeros_list[j_idx] = i
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _to_binary(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin
