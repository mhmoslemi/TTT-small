def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _compute_all_counts(bits, n):

def _continuous_polish(h, deadline, n_steps=500, step_size=0.05):
    """Polish a binary solution by allowing fractional values"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Compute subgradient of active lags
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * dx
        max_score = np.max(scores)
        
        # Find active lags (within 5% of max)
        threshold = max_score * 0.95
        active = np.where(scores >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for lag_idx in active:
            d = lag_idx - (n - 1)  # actual lag
            if d == 0:
                continue
            # ∂c_d/∂h[i]
            if d > 0:
                m = n - d
                # For i in [0, m-1]: contribution (1 - h[i+d])
                grad[:m] += (1.0 - h[d:])
                # For i in [d, n-1]: contribution -h[i-d]
                grad[d:] += -h[:m]
            else:
                g = -d
                m = n - g
                # For i in [0, m-1]: contribution (1 - h[i+g])
                grad[:m] += (1.0 - h[g:])
                # For i in [g, n-1]: contribution -h[i-g]
                grad[g:] += -h[:m]
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Take step and project
        h_new = _project(h - step_size * grad, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step_size = min(step_size * 1.03, 0.15)
        else:
            step_size *= 0.85
            if step_size < 1e-6:
                break
    
    return best_h, best_c5

import numpy as np
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _bits_to_arr(bits, n):
    arr = np.zeros(n)
    for i in range(n):
        if (bits >> i) & 1:
            arr[i] = 1.0
    return arr

def _arr_to_bits(arr):
    n = len(arr)
    bits = 0
    for i in range(n):
        if arr[i] > 0.5:
            bits |= (1 << i)
    return bits

def _max_count(bits, n):
    best = 0
    for d in range(1, n):
        m = (1 << (n - d)) - 1
        cd = ((bits & m) & ~(bits >> d)).bit_count()
        cmd = ((m & ~bits) & (bits >> d)).bit_count()
        v = cd if cd > cmd else cmd
        if v > best:
            best = v
    return best

def _counts_pnorm(bits, n, p):
    total = 0.0
    best = 0
    for d in range(1, n):
        m = (1 << (n - d)) - 1
        cd = ((bits & m) & ~(bits >> d)).bit_count()
        cmd = ((m & ~bits) & (bits >> d)).bit_count()
        v = cd if cd > cmd else cmd
        total += v ** p
        if v > best:
            best = v
    return best, total ** (1.0 / p)

def _sa_run(n, deadline, rng, initial_bits=None, T0=5.0, T1=0.05):
    target = n // 2
    if initial_bits is not None:
        bits = initial_bits
    else:
        pos = rng.choice(n, size=target, replace=False)
        bits = 0
        for i in pos:
            bits |= (1 << i)
    
    ones = [i for i in range(n) if (bits >> i) & 1]
    zeros = [i for i in range(n) if not ((bits >> i) & 1)]
    
    cur_K, cur_pn = _counts_pnorm(bits, n, 12)
    best_K = cur_K
    best_bits = bits
    best_c5 = cur_K * 2.0 / n
    
    T = T0
    no_improve = 0
    total = 0
    
    while time.time() < deadline:
        # Single swap move
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        bits_new = bits ^ (1 << i) | (1 << j)
        
        K_new, pn_new = _counts_pnorm(bits_new, n, 12)
        
        if K_new < cur_K:
            bits = bits_new
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            cur_K = K_new; cur_pn = pn_new
            no_improve = 0
            if K_new < best_K:
                best_K = K_new
                best_bits = bits
                best_c5 = K_new * 2.0 / n
        elif pn_new < cur_pn * (1 - 1e-10):
            bits = bits_new
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            cur_K = K_new; cur_pn = pn_new
            no_improve += 1
        else:
            delta = pn_new - cur_pn
            if delta <= 0 or (T > T1 and rng.random() < math.exp(-delta / T)):
                bits = bits_new
                ones.remove(i); zeros.append(i)
                zeros.remove(j); ones.append(j)
                cur_K = K_new; cur_pn = pn_new
            no_improve += 1
        
        T *= 0.99998
        total += 1
        
        if no_improve > 20000:
            # Reheat and shake
            T = max(T, T0 * 0.5)
            no_improve = 0
            for _ in range(rng.integers(3, 12)):
                ii = ones[rng.integers(len(ones))]
                jj = zeros[rng.integers(len(zeros))]
                bits ^= (1 << ii) | (1 << jj)
                ones.remove(ii); zeros.append(ii)
                zeros.remove(jj); ones.append(jj)
            cur_K, cur_pn = _counts_pnorm(bits, n, 12)
    
    return _bits_to_arr(best_bits, n), best_c5

def _continuous_refine(h, deadline, n_iter=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    step = 0.03
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * dx
        mx = np.max(scores)
        active = np.where(scores >= mx * 0.97)[0]
        
        grad = np.zeros(n)
        for a in active:
            d = a - (n - 1)
            if d == 0:
                continue
            if d > 0:
                m = n - d
                grad[:m] += (1.0 - h[d:])
                grad[d:] -= h[:m]
            else:
                g = -d
                m = n - g
                grad[:m] += (1.0 - h[g:])
                grad[g:] -= h[:m]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        h_new = _project(h - step * grad, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur - 1e-15:
            h = h_new
            cur = c5_new
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.88
            if step < 1e-7:
                break
    
    return best_h, best_c5

def _golden_init(n, alpha=None):
    if alpha is None:
        alpha = 0.6180339887498949
    target = n // 2
    vals = np.array([(i * alpha) % 1.0 for i in range(n)])
    idx = np.argsort(vals)
    pos = idx[:target]
    bits = 0
    for i in pos:
        bits |= (1 << int(i))
    return bits
