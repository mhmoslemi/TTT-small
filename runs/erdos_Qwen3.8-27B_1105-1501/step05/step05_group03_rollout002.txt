from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha, threshold=0.5):
    x = (np.arange(n) + 0.5) / n * 2.0  # positions in [0,2]
    vals = (x * alpha) % 1.0
    h = (vals < threshold).astype(float)
    return _project(h, n / 2.0)

def _build_haar_basis(n, rng, q):
    """Build q sum-zero basis vectors using Haar-like and random blocks."""
    B = np.zeros((n, q))
    col_idx = 0
    
    # Add Haar wavelets at various levels
    for level in range(1, min(6, int(np.log2(n)) + 1)):
        block = 2 ** (level - 1)
        total = 2 * block
        n_positions = n - total + 1
        if n_positions <= 0:
            continue
        # Choose some positions
        n_to_add = min(max(1, q // (4 * level)), max(1, n_positions // max(1, n_positions // 3)))
        positions = rng.choice(n_positions, size=min(n_to_add, n_positions), replace=False)
        for pos in positions:
            if col_idx >= q:
                break
            B[pos:pos+block, col_idx] = 1.0 / block
            B[pos+block:pos+2*block, col_idx] = -1.0 / block
            col_idx += 1
    
    # Fill remaining with random block pairs
    while col_idx < q:
        w = int(rng.integers(2, max(3, n // 6)))
        if 2 * w > n:
            w = n // 2
        s = int(rng.integers(0, n - 2 * w + 1))
        B[s:s+w, col_idx] = 1.0 / w
        B[s+w:s+2*w, col_idx] = -1.0 / w
        col_idx += 1
    
    return B[:, :col_idx]

def _y_to_h(y, B, n):
    raw = 0.5 + B @ y
    return _project(raw, n / 2.0)

def _h_to_y(h, B):
    """Least-squares fit of h-0.5 onto B columns."""
    residual = h - 0.5
    # Subtract mean to ensure we're in the right subspace
    y, _, _, _ = np.linalg.lstsq(B, residual, rcond=None)
    return y

def _de_search(B, h_init, deadline, rng, n, q, pop_size=40, n_gen=300, F=0.7, CR=0.8):
    """Differential evolution in the basis space."""
    target = n / 2.0
    y_best = _h_to_y(h_init, B)
    h_best = _y_to_h(y_best, B, n)
    c5_best = _exact_c5(h_best)
    
    # Also try the original h_init
    c5_init = _exact_c5(h_init)
    if c5_init < c5_best:
        c5_best = c5_init
        h_best = h_init.copy()
        y_best = _h_to_y(h_best, B)
    
    # Initialize population
    bounds_lo = np.full(q, -2.0)
    bounds_hi = np.full(q, 2.0)
    
    pop = np.zeros((pop_size, q))
    pop[0] = y_best
    for i in range(1, pop_size):
        pop[i] = y_best + rng.uniform(-0.5, 0.5, q)
        pop[i] = np.clip(pop[i], bounds_lo, bounds_hi)
    
    pop_c5 = np.zeros(pop_size)
    for i in range(pop_size):
        h_try = _y_to_h(pop[i], B, n)
        pop_c5[i] = _exact_c5(h_try)
    
    best_i = int(np.argmin(pop_c5))
    if pop_c5[best_i] < c5_best:
        c5_best = pop_c5[best_i]
        h_best = _y_to_h(pop[best_i], B, n)
        y_best = pop[best_i].copy()
    
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        
        # Occasional rejuvenation
        if gen > 50 and gen % 80 == 0:
            for i in range(pop_size // 2):
                idx = rng.integers(1, pop_size)
                pop[idx] = y_best + rng.uniform(-1.0, 1.0, q)
                pop[idx] = np.clip(pop[idx], bounds_lo, bounds_hi)
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            # DE/rand/1
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.clip(mutant, bounds_lo, bounds_hi)
            
            # Crossover
            trial = pop[i].copy()
            for j in range(q):
                if rng.random() < CR:
                    trial[j] = mutant[j]
            
            h_trial = _y_to_h(trial, B, n)
            c5_trial = _exact_c5(h_trial)
            
            if c5_trial <= pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = c5_trial
                if c5_trial < c5_best:
                    c5_best = c5_trial
                    h_best = h_trial.copy()
                    y_best = trial.copy()
    
    return h_best, c5_best

def _lp_direction(h, scores, radius):
    """Find LP direction to reduce all scores."""
    n = len(h)
    n_lags = len(scores)
    
    # Gradient of score_d with respect to h_j
    # score_d = (2/n) * sum_i h[i] * (1 - h[i+d]) for lag d
    # d score_d / d h_j = (2/n) * [(1-h[j-d]) - h[j+d]] (where valid)
    
    # Build gradient matrix efficiently
    G = np.zeros((n_lags, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    G *= (2.0 / n)
    
    # LP: minimize t s.t. G @ p - t <= -scores, sum p = 0, |p_j| <= radius, bounds
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    A_ub = np.zeros((n_lags, n_vars))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, target):
    """Golden section line search along direction p."""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    
    for _ in range(25):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))

            lo, f1, c2 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    
    opt_a = (lo + hi) / 2.0
    opt_h = _project(h + opt_a * p, target)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    h_new = np.interp(x_new, x_old, h)
