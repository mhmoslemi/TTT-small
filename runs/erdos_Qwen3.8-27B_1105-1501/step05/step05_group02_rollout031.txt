```python
def _pair_optimize_1d(S, D, t_min, t_max):
    """Find t in [t_min, t_max] minimizing max_r(S[r] + t*D[r])."""
    # Bisection on v
    # Upper bound on v*: max(f(t_min), f(t_max))
    f_lo = float(np.max(S + t_min * D))
    f_hi = float(np.max(S + t_max * D))
    v_hi = min(f_lo, f_hi)
    
    # Lower bound on v*: for each r, min over t of S[r]+t*D[r] on [t_min,t_max]
    # = S[r] + t_min*D[r] if D[r]>0, S[r]+t_max*D[r] if D[r]<0, S[r] if D[r]=0
    lb = np.where(D > 0, S + t_min * D, np.where(D < 0, S + t_max * D, S))
    v_lo = float(np.max(lb))
    
    if v_hi - v_lo < 1e-14:
        # The minimum is essentially at a boundary
        if f_lo <= f_hi:
            return t_min
        else:
            return t_max
    
    # Bisection
    for _ in range(60):
        v_mid = (v_lo + v_hi) / 2.0
        # Check feasibility
        mask_pos = D > 1e-15
        mask_neg = D < -1e-15
        mask_zero = ~(mask_pos | mask_neg)
        
        if np.any(S[mask_zero] > v_mid):
            v_lo = v_mid
            continue
        
        hi_bound = t_max
        if np.any(mask_pos):
            hi_bound = min(hi_bound, float(np.min((v_mid - S[mask_pos]) / D[mask_pos])))
        
        lo_bound = t_min
        if np.any(mask_neg):
            lo_bound = max(lo_bound, float(np.min((v_mid - S[mask_neg]) / D[mask_neg])))
        
        if lo_bound <= hi_bound:
            v_hi = v_mid
        else:
            v_lo = v_mid
    
    # Now v_hi is approximately v*. Find the corresponding t.
    v = v_hi
    mask_pos = D > 1e-15
    mask_neg = D < -1e-15
    
    hi_bound = t_max
    if np.any(mask_pos):
        hi_bound = min(hi_bound, float(np.min((v - S[mask_pos]) / D[mask_pos])))
    
    lo_bound = t_min
    if np.any(mask_neg):
        lo_bound = max(lo_bound, float(np.min((v - S[mask_neg]) / D[mask_neg])))
    
    if lo_bound <= hi_bound:
        return (lo_bound + hi_bound) / 2.0
    else:
        # Numerical issue, return boundary
        if f_lo <= f_hi:
            return t_min
        else:
            return t_max
```

For the LP:

def _lp_step(h, S, G, radius):
    n = len(h)
    n_lags = 2 * n - 1
    # Variables: p[0..n-1], z
    # Minimize z
    # Constraints: G[r, :] @ p - z <= -S[r] for all r
    #              sum(p) = 0
    #              -radius <= p[i] <= radius for all i
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize z
    
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -S
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(None, None)]
    for i in range(n):
        bounds[i] = (-radius, radius)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except:
        pass
    return None, None
