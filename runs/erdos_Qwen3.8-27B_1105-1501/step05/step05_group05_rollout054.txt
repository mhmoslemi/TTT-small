def _project(v, target):

def _grad_matrix(h):

            g = -d
            m = n - g

                G[r, :m] += (1.0 - h[g:])

    return G

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(45):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, top_k=None):
    n = len(h)
    nlags = 2 * n - 1
    if top_k is None or top_k >= nlags:
        idx = np.arange(nlags)
        nk = nlags
    else:
        idx = np.argsort(-scores)[:top_k]
        nk = top_k
    c_full = np.zeros(n + 1)
    c_full[-1] = 1.0
    A_ub = np.zeros((nk, n + 1))
    A_ub[:, :n] = G[idx]
    A_ub[:, n] = -1.0
    b_ub = -scores[idx].copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_full, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.20, use_relax=False, relax_k=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    best_radius = radius_init
    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        if use_relax and relax_k is not None and stall > 10:
            p = _lp_solve(h, scores, G, radius, top_k=relax_k)
        else:
            p = _lp_solve(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_radius = radius
            stall = 0
            radius = min(radius * 1.06, 0.55)
        else:
            stall += 1
            radius *= 0.72
            if stall > 30 and radius < 0.05:
                radius = best_radius * 0.4
                stall = 0
    return best_h, best_c5

def _pair_refine(h, deadline, rng, max_passes=200, systematic=True):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        int_list = list(np.where(interior)[0])
        if len(int_list) < 2:
            break
        if systematic and len(int_list) <= 45:
            pairs = []
            for ii in range(len(int_list)):
                for jj in range(ii + 1, len(int_list)):
                    pairs.append((int_list[ii], int_list[jj]))
        else:
            pairs = []
            n_try = min(400, len(int_list) * 10)
            for _ in range(n_try):
                i = int(int_list[rng.integers(len(int_list))])
                j = int(int_list[rng.integers(len(int_list))])
                if i != j:

            pairs = list(set(pairs))
        for (i, j) in pairs:
            if time.time() > deadline:

            for sign in [1, -1]:
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                if sign > 0:

                else:
                    max_step = min(1.0 - h[i], h[j])

                    cur = _exact_c5(h)
                    if cur < best_c5:

                        best_h = h.copy()

        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _smooth_opt(h, deadline, beta_list=(60, 150, 400), lr_init=0.03):

            best_h = h6.copy()
    if time.time() < d6:
        h6, c5_6 = _pair_refine(best_h, d6, rng, max_passes=200)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final validation and projection
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, n_final
