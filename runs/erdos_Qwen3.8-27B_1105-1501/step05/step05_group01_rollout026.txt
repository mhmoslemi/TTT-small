    n = len(h)
    
    def obj_fn(x):
        val, _ = _softmax_obj_and_grad(x, tau)
        return val
    
    def grad_fn(x):
        _, g = _softmax_obj_and_grad(x, tau)
        return g

        return float(np.sum(x)) - target
    
    def cons_grad(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': cons_val, 'jac': cons_grad}]
    bounds = [(0.0, 1.0)] * n

               'disp': 0, 'iprint': 0}

def _line_search_multi(h, d, dx, radius, n_grid=51):
    """Multi-bracket exact line search along direction d."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Feasible interval
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = -radius
    t_hi = radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-13:
        return 0.0, _exact_c5(h)
    
    def phi(t):
        return float(np.max(S + t * T - t * t * Q)) * dx
    
    # Dense grid
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([phi(t) for t in ts])
    
    # Find local minima
    brackets = []
    # Endpoints
    brackets.append((t_lo, ts[1]))
    brackets.append((ts[-2], t_hi))
    # Interior local minima
    for i in range(1, n_grid - 1):
        if vals[i] < vals[i-1] and vals[i] <= vals[i+1]:
            brackets.append((ts[i-1], ts[i+1]))
    
    best_t = 0.0
    best_val = phi(0.0)
    
    # Evaluate at grid points too
    min_grid_idx = int(np.argmin(vals))
    best_t = float(ts[min_grid_idx])
    best_val = float(vals[min_grid_idx])

    for (a, b) in brackets:
        if b - a < 1e-12:

            else:
                a = c1; c1 = c2; f1 = f2

                f2 = phi(c2)

        f_opt = phi(t_opt)
        if f_opt < best_val - 1e-14:

            best_t = t_opt

def _fourier_directions(n, max_mode=8):
    """Generate low-frequency Fourier directions (zero mean)."""
    x = (np.arange(n) + 0.5) / n
    dirs = []
    for m in range(1, max_mode + 1):
        s = np.sin(2 * np.pi * m * x)
        c = np.cos(2 * np.pi * m * x)
        # Make zero mean
        s -= np.mean(s)
        c -= np.mean(c)
        norm_s = np.max(np.abs(s))
        norm_c = np.max(np.abs(c))
        if norm_s > 1e-10:
            dirs.append(s / norm_s)
        if norm_c > 1e-10:
            dirs.append(c / norm_c)
    return dirs

def _block_direction(h, n_blocks=4, rng=None):
    """Generate a block move direction."""
    n = len(h)
    if rng is None:
        rng = np.random.default_rng()
    block_size = max(2, n // (2 * n_blocks))
    # Pick two random blocks
    b1 = int(rng.integers(0, n - 2 * block_size))
    b2 = int(rng.integers(b1 + block_size, n - block_size))
    d = np.zeros(n)
    d[b1:b1+block_size] = 1.0 / block_size
    d[b2:b2+block_size] = -1.0 / block_size
    # Zero mean already by construction
    return d

def _random_pair_dir(n, rng):
    """Generate a random pair direction."""
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

    x = (np.arange(n) + 0.5) / n
    dirs = []
    for m in range(1, max_mode + 1):

        s -= np.mean(s)
        c -= np.mean(c)
        ns = np.max(np.abs(s))
        nc = np.max(np.abs(c))
        if ns > 1e-10:

    d[b1:b1 + block_size] = 1.0 / block_size
    d[b2:b2 + block_size] = -1.0 / block_size

    j = int(rng.integers(0, n))

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_optimize(h, target, tau, max_iter, deadline):
    n = len(h)
    def obj(x):
        return _softmax_obj_grad(x, tau)[0]
    def jac(x):
        return _softmax_obj_grad(x, tau)[1]
    def con_eq(x):
        return float(np.sum(x)) - target
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    options = {'maxiter': max_iter, 'ftol': 1e-12, 'maxls': 25, 'disp': 0}
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints, options=options)
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return _project(x, target)
    except Exception:
        pass
    return h.copy()

def _try_update(h_cand, best_state):
    h_cand = _project(h_cand, len(h_cand) / 2.0)
    if not np.all(np.isfinite(h_cand)):
        return h_cand, best_state[1], False
    c5 = _exact_c5(h_cand)
    if c5 < best_state[1] - 1e-13:
        return h_cand.copy(), c5, True
    return h_cand, best_state[1], False

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h_init = _project(h0.copy(), target0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Soft-max projected gradient with decreasing tau (45%)
    d1 = t0 + budget_s * 0.45
    tau_schedule = [0.08, 0.05, 0.03, 0.018, 0.01, 0.006, 0.0035, 0.002]
    h_curr = best_h.copy()
    for tau in tau_schedule:
        if time.time() > d1:
            break
        h_curr, c5_curr = _proj_grad_step(h_curr, target0, tau, 0.3, 150, d1)
        h_curr = _project(h_curr, target0)
        c5_curr = _exact_c5(h_curr)
        if c5_curr < best_c5 - 1e-13:
            best_c5 = c5_curr
            best_h = h_curr.copy()
        # Line search cleanup in gradient direction
        _, grad = _softmax_obj_grad(h_curr, tau)
        grad -= np.mean(grad)
        gnorm = np.max(np.abs(grad))
        if gnorm > 1e-12:
            d_dir = -grad / gnorm
            dx0 = 2.0 / n0
            dt, _ = _line_search_multi(h_curr, d_dir, dx0, 0.8, n_grid=41)
            if abs(dt) > 1e-12:
                h_new = _project(h_curr + dt * d_dir, target0)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_curr = best_h.copy()

    # Phase 2: SLSQP polishing (20%)
    d2 = t0 + budget_s * 0.65
    h_curr = best_h.copy()
    n_curr = len(h_curr)
    target_curr = n_curr / 2.0
    dx0 = 2.0 / n_curr
    for tau in [0.015, 0.008, 0.004, 0.002]:
        if time.time() > d2:
            break
        h_curr = _slsqp_optimize(h_curr, target_curr, tau, 60, d2)
        h_curr = _project(h_curr, target_curr)
        c5_curr = _exact_c5(h_curr)
        if c5_curr < best_c5 - 1e-13:
            best_c5 = c5_curr
            best_h = h_curr.copy()
        # Gradient cleanup
        _, grad = _softmax_obj_grad(h_curr, tau)
        grad -= np.mean(grad)
        gnorm = np.max(np.abs(grad))
        if gnorm > 1e-12:
            d_dir = -grad / gnorm
            dt, _ = _line_search_multi(h_curr, d_dir, dx0, 0.6, n_grid=41)
            if abs(dt) > 1e-12:
                h_new = _project(h_curr + dt * d_dir, target_curr)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_curr = best_h.copy()

    # Phase 3: Fourier + random direction line search (15%)
    d3 = t0 + budget_s * 0.80
    h_curr = best_h.copy()
    n_curr = len(h_curr)
    target_curr = n_curr / 2.0
    dx0 = 2.0 / n_curr

    fourier_dirs = _fourier_dirs(n_curr, max_mode=12)
    for d_dir in fourier_dirs:
        if time.time() > d3:
            break
        d_dir = d_dir - np.mean(d_dir)
        nm = np.max(np.abs(d_dir))
        if nm < 1e-10:
            continue
        d_dir /= nm
        dt, c5_try = _line_search_multi(h_curr, d_dir, dx0, 1.0, n_grid=49)
        if c5_try < best_c5 - 1e-13:
            h_new = _project(h_curr + dt * d_dir, target_curr)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = best_h.copy()

    # Random pair directions
    stall = 0
    radius = 0.8
    while time.time() < d3 and stall < 80:
        d_dir = _pair_dir(n_curr, rng)
        dt, c5_try = _line_search_multi(h_curr, d_dir, dx0, radius, n_grid=31)
        if c5_try < best_c5 - 1e-13:
            h_new = _project(h_curr + dt * d_dir, target_curr)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = best_h.copy()
                stall = 0
                radius = min(radius * 1.04, 1.5)
        else:
            stall += 1
            if stall % 20 == 0:
                radius *= 0.7
                if radius < 0.04:
                    break

    # Block moves
    stall = 0
    radius = 0.6
    while time.time() < d3 and stall < 80:
        bs = max(2, n_curr // 24)
        d_dir = _block_dir(n_curr, bs, rng)
        dt, c5_try = _line_search_multi(h_curr, d_dir, dx0, radius, n_grid=37)
        if c5_try < best_c5 - 1e-13:
            h_new = _project(h_curr + dt * d_dir, target_curr)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = best_h.copy()
                stall = 0
                radius = min(radius * 1.03, 1.2)
        else:
            stall += 1
            if stall % 20 == 0:
                radius *= 0.7
                if radius < 0.03:
                    break

    # Phase 4: Multi-resolution (10%)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    offsets = [-8, 8, -16, 16, -4, 4, -12, 12, -24, 24, -32, 32, -20, 20, -28, 28]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 48 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        for tau in [0.015, 0.007]:
            if time.time() > d4:
                break
            h_r, _ = _proj_grad_step(h_r, n_try / 2.0, tau, 0.2, 60, d4)
            h_r = _project(h_r, n_try / 2.0)
            c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Final polish (5%)
    d5 = deadline
    h_curr = best_h.copy()
    n_final = len(h_curr)
    target_final = n_final / 2.0
    dx_final = 2.0 / n_final

    # Final projected gradient
    for tau in [0.005, 0.003, 0.002]:
        if time.time() > d5:
            break
        h_curr, c5_c = _proj_grad_step(h_curr, target_final, tau, 0.15, 80, d5)
        h_curr = _project(h_curr, target_final)
        c5_c = _exact_c5(h_curr)
        if c5_c < best_c5 - 1e-13:
            best_c5 = c5_c
            best_h = h_curr.copy()

    # Final SLSQP
    if time.time() < d5 - 5:
        h_curr = _slsqp_optimize(best_h.copy(), target_final, 0.003, 50, d5)
        h_curr = _project(h_curr, target_final)
        c5_c = _exact_c5(h_curr)
        if c5_c < best_c5 - 1e-13:
            best_c5 = c5_c
            best_h = h_curr.copy()

    # Final random pair polish
    h_curr = best_h.copy()
    stall = 0
    radius = 0.4
    while time.time() < d5 - 1 and stall < 40:
        d_dir = _pair_dir(n_final, rng)
        dt, c5_try = _line_search_multi(h_curr, d_dir, dx_final, radius, n_grid=29)
        if c5_try < best_c5 - 1e-13:
            h_new = _project(h_curr + dt * d_dir, target_final)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = best_h.copy()
                stall = 0
        else:
            stall += 1

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
