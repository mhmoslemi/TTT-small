def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _line_search(h, d, trust_radius, n_samples=40):

    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15

def _lp_direction(h, scores, radius):
    n = len(h)
    # Build G: gradient of each lag's correlation w.r.t. h
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        l = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[l, :m] = (1.0 - h[d:])
                G[l, d:n] += -h[:m]
        else:
            gd = -d
            m = n - gd
            if m > 0:
                G[l, gd:n] = (1.0 - h[:m])
                G[l, :m] += -h[gd:]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgrad_direction(h):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_score = np.max(scores)
    # Find active set (within tolerance)
    active = scores >= max_score - 1e-8
    # Compute gradient at active lags and average
    g = np.zeros(n)
    for l in range(len(active)):
        if not active[l]:
            continue
        d = l - (n - 1)
        if d >= 0:
            m = n - d
            g[:m] += (1.0 - h[d:])
            g[d:n] += -h[:m]
        else:
            gd = -d
            m = n - gd
            g[gd:n] += (1.0 - h[:m])
            g[:m] += -h[gd:]
    g /= np.sum(active)
    # Remove mean (to maintain sum constraint approximately)
    g -= np.mean(g)
    return g

def _compute_all_lags(h):
    """Compute S(d) for all d from -(n-1) to n-1"""
    n = len(h)
    # S(d) for d >= 0: sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
    # S(d) for d < 0: let d' = -d, S(-d') = sum_{i=0}^{n-1-d'} h[i+d'] * (1-h[i])
    # Note: S(-d') = sum_{i} h[i+d']*(1-h[i]) = sum_j h[j]*(1-h[j-d']) where j = i+d'
    #       = sum_{j=d'}^{n-1} h[j]*(1-h[j-d'])
    # Which is the same form as S(d') but with roles swapped... 
    # Actually S(d) for the correlation is:
    # corr_full[k] = sum_i h[i] * (1-h[i + k - (n-1)])
    # Let me just use numpy and index properly.
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has 2n-1 elements, indexed 0 to 2n-2
    # corr[n-1+d] corresponds to lag d
    return corr  # 2n-1 array

def _swap_delta(h, a, b):
    """Compute change in each lag's correlation when h[a]:1->0, h[b]:0->1"""
    n = len(h)
    # For each lag d (corresponding to index n-1+d in corr):
    # The change comes from terms involving position a or b
    
    # Position a was 1, now 0:
    # In S(d) = sum_i h[i]*(1-h[i+d]) for d>=0:
    #   a contributes as h[i] if i=a: term was 1*(1-h[a+d]) = 1-h[a+d], now 0
    #     Valid if a+d < n, i.e., d < n-a... wait, i goes from 0 to n-1-d
    #     So a <= n-1-d, i.e., d <= n-1-a
    #   a contributes as h[i+d] if i+d=a: term was h[a-d]*(1-1)=0, now h[a-d]*(1-0)=h[a-d]
    #     Valid if 0 <= a-d and a <= n-1-d, i.e., d <= a and d >= a-(n-1-d) which is automatic
    #     Actually: i = a-d, need 0 <= a-d <= n-1-d, i.e., d <= a and a-d <= n-1-d → a <= n-1 (always)
    #     So valid if 0 <= a-d, i.e., d <= a
    
    # This is getting complicated. Let me just compute it numerically.
    
    # Alternative: just compute the full correlate with the swap applied
    # This is O(n^2) per swap which is 24000 ops for n=156
    # For 6000 swaps: 144M ops... too slow in Python
    
    # Better: compute delta analytically
    delta = np.zeros(2*n - 1)
    
    for d in range(-(n-1), n):
        idx = n - 1 + d
        # S(d) = sum of h[i]*(1-h[i+d]) over valid i
        # The valid i range depends on d:
        # If d >= 0: i ranges from 0 to n-1-d
        #   h[i] and h[i+d] must both be in [0, n-1]
        # If d < 0: let dd = -d, i ranges from 0 to n-1-dd
        #   h[i+dd] and h[i] must both be in [0, n-1]
        #   Equivalent: sum_{j=dd}^{n-1} h[j]*(1-h[j-dd])
        
        # Change due to a (was 1, now 0):
        # a appears in the "first" slot (h[i]) when i = a
        #   and in the "second" slot (h[i+d] or h[i-dd]) when i+d = a or i-dd = a
        
        # For d >= 0:
        if d >= 0:
            # h[i] slot: i = a, need 0 <= a <= n-1-d → a <= n-1-d → d <= n-1-a
            if a <= n - 1 - d:
                # old contribution: 1 * (1 - h[a+d])
                delta[idx] += -(1.0 - h[a + d])
            # h[i+d] slot: i+d = a → i = a-d, need 0 <= a-d and a <= n-1-d
            # i = a-d, need a-d >= 0 → d <= a, and a-d <= n-1-d → a <= n-1 (always true)
            if d <= a:
                # old contribution: h[a-d] * (1 - 1) = 0
                # new contribution: h[a-d] * (1 - 0) = h[a-d]
                delta[idx] += h[a - d]
        else:
            dd = -d
            # S(-dd) = sum_{i=0}^{n-1-dd} h[i+dd]*(1-h[i])
            # h[i+dd] slot: i+dd = a → i = a-dd, need 0 <= a-dd <= n-1-dd → dd <= a <= n-1
            # h[i] slot: i = a, need 0 <= a <= n-1-dd → a <= n-1-dd → dd <= n-1-a
            if a - dd >= 0 and a <= n - 1:  # a-dd >= 0 → dd <= a
                # old: h[a]*(1-h[a-dd]) = 1*(1-h[a-dd]) = 1-h[a-dd]
                # new: 0*(1-h[a-dd]) = 0
                delta[idx] += -(1.0 - h[a - dd])
            if a <= n - 1 - dd:  # a + dd <= n-1
                # h[i] slot: i = a
                # old: h[i+dd]*(1-h[a]) = h[a+dd]*(1-1) = 0
                # new: h[a+dd]*(1-0) = h[a+dd]
                delta[idx] += h[a + dd]
        
        # Change due to b (was 0, now 1):
        if d >= 0:
            if b <= n - 1 - d:
                # h[i] slot: i = b
                # old: 0*(1-h[b+d]) = 0
                # new: 1*(1-h[b+d]) = 1-h[b+d]
                delta[idx] += (1.0 - h[b + d])
            if d <= b:
                # h[i+d] slot: i+d = b → i = b-d
                # old: h[b-d]*(1-0) = h[b-d]
                # new: h[b-d]*(1-1) = 0
                delta[idx] += -h[b - d]
        else:
            dd = -d
            if b - dd >= 0:
                # h[i+dd] slot: i+dd = b → i = b-dd
                # old: h[b]*(1-h[b-dd]) = 0
                # new: 1*(1-h[b-dd]) = 1-h[b-dd]
                delta[idx] += (1.0 - h[b - dd])
            if b <= n - 1 - dd:
                # h[i] slot: i = b
                # old: h[b+dd]*(1-h[b]) = h[b+dd]*(1-0) = h[b+dd]
                # new: h[b+dd]*(1-1) = 0
                delta[idx] += -h[b + dd]
    
    return delta

- p contributes to "h[i+dd]" when i+dd = p, i.e., i = p-dd, valid if p-dd >= 0 and p-dd <= n-1-dd (i.e., dd <= p)

  Contribution: h[p+dd] * (1-v)

So for a position p with value v, its total contribution to lag d is:

- For d < 0 (dd = -d): v*(1-h[p-dd]) [if dd <= p] + h[p+dd]*(1-v) [if p+dd <= n-1]

When h[p] changes from v_old to v_new, the change in contribution at lag d is:

    n = len(h)
    # E_p[d] for p in 0..n-1, d in -(n-1)..n-1
    # For binary h: E_p[d] = (1-h[p+d])*[valid] - h[p-d]*[valid] for the "h[i]" and "h[i+d]" terms
    # Actually, for a position p with value v, the contribution to lag d (d>=0) is:
    #   v*(1-h[p+d]) [if p+d < n] + h[p-d]*(1-v) [if p-d >= 0]
    # The effect of changing v from 0 to 1:
    #   (1-h[p+d]) [if p+d < n] - h[p-d] [if p-d >= 0]
    
    # Let me define E_p[d] as the change in corr[n-1+d] when h[p] goes from 0 to 1
    # (all other h fixed)
    # For d >= 0:

                    val -= h[p - d]

            else:

                if p - dd >= 0:

                if p + dd < n:
                    val -= h[p + dd]
                E[p, idx] = val
    return E
