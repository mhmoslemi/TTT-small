        wn = w[idx_neg]

        # Positive lag
        if abs(wp) >= 1e-15:
            grad[d:] += wp * dx * (1.0 - h[:n-d])
            grad[:n-d] += -wp * dx * h[d:]
        # Negative lag
        if abs(wn) >= 1e-15:
            grad[:n-d] += wn * dx * (1.0 - h[d:])
            grad[d:] += -wn * dx * h[:n-d]

    if abs(w0) >= 1e-15:
        grad += w0 * dx * (1.0 - 2.0 * h)
    
    # Center (remove mean to preserve sum)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return -grad / gn  # descent direction

    n = len(h)

def _scores(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _pair_optimize(h, i, j, n_samples=9):
    """Optimize along the line h[i]+=t, h[j]-=t."""
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(h[j], 1.0 - h[i])

        return 0.0, _exact_c5(h)

def _grad_direction(h, gamma=50.0):
    """Smoothing-weighted gradient."""
    n = len(h)

    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = float(np.sum(w))

        return None
    w /= sw
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for d in range(n):
        if d == 0:
            if abs(w[n-1]) > 1e-15:
                grad += w[n-1] * dx * (1.0 - 2.0 * h)
            continue
        m = n - d
        wp = w[n - 1 + d]
        wn = w[n - 1 - d]
        if abs(wp) > 1e-15:
            grad[d:] += wp * dx * (1.0 - h[:m])
            grad[:m] += -wp * dx * h[d:]

    gn = np.linalg.norm(grad)

            G[r, :m] += -h[d:]

            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    G *= 2.0 / n
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _pair_line_search(h, i, j, n_samples=7, n_golden=10):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(h[j], 1.0 - h[i])
    if t_hi - t_lo < 1e-12:
        return 0.0
    best_t = 0.0
    best_c5 = _exact_c5(h)
    span = t_hi - t_lo
    fracs = [0.2, 0.35, 0.5, 0.65, 0.8]
    for f in fracs:
        t = t_lo + f * span
        h[i] += t
        h[j] -= t
        c5 = _exact_c5(h)
        h[i] -= t
        h[j] += t
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    if best_t > 0:
        lo = max(t_lo, best_t * 0.4)
        hi = min(t_hi, best_t * 1.6)
    elif best_t < 0:
        lo = max(t_lo, best_t * 1.6)
        hi = min(t_hi, best_t * 0.4)
    else:
        lo, hi = t_lo, t_hi
    if hi - lo > 1e-11:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h[i] += c1; h[j] -= c1
        f1 = _exact_c5(h)
        h[i] -= c1; h[j] += c1
        h[i] += c2; h[j] -= c2
        f2 = _exact_c5(h)
        h[i] -= c2; h[j] += c2
        if f1 < best_c5:
            best_c5 = f1; best_t = c1
        if f2 < best_c5:
            best_c5 = f2; best_t = c2
        for _ in range(n_golden):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi = c2; f2 = f1; c2 = c1
                c1 = hi - gr * (hi - lo)
                h[i] += c1; h[j] -= c1
                f1 = _exact_c5(h)
                h[i] -= c1; h[j] += c1
            else:
                lo = c1; f1 = f2; c1 = c2
                c2 = lo + gr * (hi - lo)
                h[i] += c2; h[j] -= c2
                f2 = _exact_c5(h)
                h[i] -= c2; h[j] += c2
            if f1 < best_c5:
                best_c5 = f1; best_t = c1
            if f2 < best_c5:
                best_c5 = f2; best_t = c2
        opt_t = (lo + hi) / 2.0
        h[i] += opt_t; h[j] -= opt_t
        oc = _exact_c5(h)
        h[i] -= opt_t; h[j] += opt_t
        if oc < best_c5:
            best_c5 = oc; best_t = opt_t
    return best_t

def _pair_pass(h, n_tries, rng, deadline, use_gradient_bias=True):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n_acc = 0
    grad_cache = None
    bias_interval = 500
    for it in range(n_tries):
        if time.time() > deadline:
            break
        if use_gradient_bias and it % bias_interval == 0:
            grad_cache = _grad_dir(h, 80.0)
        if use_gradient_bias and grad_cache is not None:
            neg_mask = grad_cache < -0.01
            pos_mask = grad_cache > 0.01
            neg_idx = np.where(neg_mask)[0]
            pos_idx = np.where(pos_mask)[0]
            if len(neg_idx) > 0 and len(pos_idx) > 0:
                i = neg_idx[rng.integers(len(neg_idx))]
                j = pos_idx[rng.integers(len(pos_idx))]
            elif len(pos_idx) > 1:
                i = pos_idx[rng.integers(len(pos_idx))]
                j = pos_idx[rng.integers(len(pos_idx))]
                if i == j:
                    j = (j + 1) % n
            else:
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    j = (j + 7) % n
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                j = (j + 7) % n
        t = _pair_line_search(h, i, j)
        if abs(t) > 1e-10:
            h[i] += t
            h[j] -= t
            c5_now = _exact_c5(h)
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
                best_h = h.copy()
                n_acc += 1
            else:
                h[i] -= t
                h[j] += t
    return best_h, best_c5, n_acc

def _grad_dir(h, gamma):
    n = len(h)
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = float(np.sum(w))
    if sw < 1e-300:
        return None
    w /= sw
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(n):
        if d == 0:
            if abs(w[n - 1]) > 1e-15:
                grad += w[n - 1] * dx * (1.0 - 2.0 * h)
            continue
        m = n - d
        wp = w[n - 1 + d]
        wn = w[n - 1 - d]
        if abs(wp) > 1e-15:
            grad[d:] += wp * dx * (1.0 - h[:m])
            grad[:m] += -wp * dx * h[d:]
        if abs(wn) > 1e-15:
            grad[:m] += wn * dx * (1.0 - h[d:])
            grad[d:] += -wn * dx * h[:m]
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return grad / gn

def _grad_step(h, direction):
    n = len(h)
    alpha_max = np.inf
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(alpha_max, 0.4)
    if alpha_max < 1e-12:
        return h, _exact_c5(h)
    cur_c5 = _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * direction)
    f2 = _exact_c5(h + c2 * direction)
    for _ in range(14):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * direction)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * direction)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * direction)
    if opt_c5 >= cur_c5:
        return h, cur_c5
    h_new = h + opt_a * direction
    h_new = _project(h_new, n / 2.0)
    return h_new, _exact_c5(h_new)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_polish(h, radius=0.2):
    n = len(h)
    scores = _scores(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    G *= 2.0 / n
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-10:
                alpha_max = np.inf
                for j in range(n):
                    if p[j] > 1e-14:
                        alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                    elif p[j] < -1e-14:
                        alpha_max = min(alpha_max, h[j] / (-p[j]))
                alpha_max = min(alpha_max, 0.5)
                if alpha_max > 1e-12:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    lo, hi = 0.0, alpha_max
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = _exact_c5(h + c1 * p)
                    f2 = _exact_c5(h + c2 * p)
                    for _ in range(14):
                        if hi - lo < 1e-13:
                            break
                        if f1 < f2:
                            hi = c2; f2 = f1; c2 = c1
                            c1 = hi - gr * (hi - lo)
                            f1 = _exact_c5(h + c1 * p)
                        else:
                            lo = c1; f1 = f2; c1 = c2
                            c2 = lo + gr * (hi - lo)
                            f2 = _exact_c5(h + c2 * p)
                    opt_a = (lo + hi) / 2.0
                    h_new = h + opt_a * p
                    return _project(h_new, n / 2.0)
    except Exception:
        pass
    return h

def _binary_swap_search(h_init, n_iters, rng, deadline):
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h >= 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_c5
    T = 0.003 * cur_c5
    T_min = 0.0003 * cur_c5
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        pi = rng.integers(len(ones))
        pj = rng.integers(len(zeros))
        i = ones[pi]
        j = zeros[pj]
        old_i, old_j = h[i], h[j]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones[pi] = j
            zeros[pj] = i
        else:
            h[i] = old_i
            h[j] = old_j
        T = max(T * cool, T_min)
    return best_h, best_c5

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)
