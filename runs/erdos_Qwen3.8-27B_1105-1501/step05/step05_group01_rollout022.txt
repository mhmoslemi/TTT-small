    n = _g_n
    dx = _g_dx
    tau = _g_tau
    scores = np.correlate(x, 1.0 - x, mode='full') * dx
    M = np.max(scores)
    exps = np.exp((scores - M) / tau)
    w = exps / np.sum(exps)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-8:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wr * (1.0 - x[:m])
                g[:m] -= wr * x[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += wr * (1.0 - x[gg:])
                g[gg:] -= wr * x[:m]
    return g * dx

def _sm_eq(x):
    return np.array([np.sum(x) - _g_target])

def _sm_eq_jac(x):
    return np.ones((1, _g_n))

def _slsqp_round(h, tau, deadline):
    global _g_tau, _g_n, _g_dx, _g_target
    _g_tau = tau
    _g_n = len(h)
    _g_dx = 2.0 / _g_n
    _g_target = _g_n / 2.0
    try:
        res = minimize(
            _sm_obj, h.copy(), jac=_sm_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * len(h),
            constraints=[{'type': 'eq', 'fun': _sm_eq, 'jac': _sm_eq_jac}],
            options={'maxiter': 80, 'ftol': 1e-9}
        )
        if res.success and res.x is not None:
            candidate = _project(res.x, _g_n / 2.0)
            return candidate
    except Exception:
        pass
    return h  # return original if failed

def _slsqp_chain(h, deadline, tau_list=None):
    if tau_list is None:
        tau_list = [0.06, 0.035, 0.02, 0.012, 0.007, 0.004, 0.0025]
    n = len(h)
    best = h.copy()
    best_c5 = _exact_c5(h)
    for tau in tau_list:
        if time.time() > deadline:
            break
        cand = _slsqp_round(best, tau, deadline)
        if cand is not None:
            c5 = _exact_c5(cand)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best = cand.copy()
    return best, best_c5

def _coord_descent(h, n_passes=5, n_samples=50, deadline=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for _pass in range(n_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            # Compute the 1D objective for changing h[i]
            # Precompute scores with h[i] set to 0 and 1
            old_val = h[i]
            # When h[i] = t:
            # For d=0: S_0 = S_0_rest + t - t^2 where S_0_rest = sum_{j!=i} h[j](1-h[j])
            # For d!=0: S_r = a_r + b_r*t where b_r = dS_r/dh[i] (constant)
            
            # Compute current scores
            scores = np.correlate(h, 1.0 - h, mode='full')
            
            # Compute b_r = dS_r/dh[i]
            b = np.zeros(2 * n - 1)
            for r in range(2 * n - 1):
                d = r - (n - 1)
                if d >= 0:
                    m = n - d
                    if m > 0:
                        # contribution of h[i] to S_r
                        if d <= i <= n - 1:
                            b[r] += (1.0 - h[i - d])
                        if 0 <= i <= m - 1:
                            b[r] -= h[i + d]
                else:
                    gg = -d
                    m = n - gg
                    if m > 0:
                        if 0 <= i <= m - 1:
                            b[r] += (1.0 - h[i + gg])
                        if gg <= i <= n - 1:
                            b[r] -= h[i - gg]
            
            # For d=0, the term is quadratic: t - t^2
            # S_0_rest = sum_{j!=i} h[j](1-h[j])
            S_0_rest = float(np.sum(h * (1.0 - h))) - old_val * (1.0 - old_val)
            # The center index is n-1
            center = n - 1
            
            # Now find best t in [0,1]
            # f(t) = dx * max(max_{r != center}(scores[r] - old_val*b[r] + t*b[r]), S_0_rest + t - t^2)
            # = dx * max(max_{r != center}(a_r + b_r*t), S_0_rest + t - t^2)
            # where a_r = scores[r] - old_val * b[r]
            
            a = scores - old_val * b
            a[center] = S_0_rest  # override for the center
            
            # Sample
            ts = np.linspace(0.0, 1.0, n_samples)
            # Compute f(t) for each t
            # For non-center: a[r] + b[r]*t (vectorized: a[non_center] + b[non_center]*t)
            # For center: S_0_rest + t - t^2
            best_t = 0.0
            best_val = _exact_c5(h)  # current value
            for t in ts:
                non_center_scores = a[:center] + b[:center] * t
                non_center_scores2 = a[center+1:] + b[center+1:] * t
                center_score = S_0_rest + t - t * t
                # Combine
                if len(non_center_scores) > 0:
                    mc = max(np.max(non_center_scores), np.max(non_center_scores2) if len(non_center_scores2) > 0 else 0)
                else:
                    mc = 0.0
                val = dx * max(mc, center_score)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            if abs(best_t - old_val) > 1e-10:
                h[i] = best_t
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5 - 1e-13:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5
