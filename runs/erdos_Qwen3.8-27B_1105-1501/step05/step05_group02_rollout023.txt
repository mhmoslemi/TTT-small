import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _bits_c5(bits, n):
    """Compute C5 for a binary sequence represented as a bitmask."""
    K = n // 2
    # For each lag d, compute b_d = #{i: h[i]=1, h[i+d]=0}, i in [0, n-d-1]
    max_b = 0
    for d in range(1, n):
        lm = (1 << (n - d)) - 1
        # h[i]=1 and h[i+d]=0: bit i in bits, bit i+d NOT in bits
        # bits & lm gives bits in positions 0..n-d-1 that are 1
        # (bits >> d) & lm gives bits in positions 0..n-d-1 of (bits shifted right by d)
        # i.e., bit i of (bits>>d) = bit i+d of bits
        ones_left = bits & lm  # positions 0..n-d-1 where h=1
        ones_shifted = (bits >> d) & lm  # positions 0..n-d-1 where h[i+d]=1
        b_d = ones_left.bit_count() - (ones_left & ones_shifted).bit_count()
        if b_d > max_b:
            max_b = b_d
    return max_b * (2.0 / n)

def _arr_to_bits(h, n):
    """Convert binary array to bitmask."""
    bits = 0
    for i in range(n):
        if h[i] > 0.5:
            bits |= (1 << i)
    return bits

def _bits_to_arr(bits, n):
    """Convert bitmask to binary array."""
    h = np.zeros(n)
    for i in range(n):
        if (bits >> i) & 1:
            h[i] = 1.0
    return h

    """Generate random balanced bitmask."""
    pos = rng.choice(n, size=K, replace=False)
    bits = 0
    for p in pos:
        bits |= (1 << p)
    return bits

def _binary_sa(n, rng, deadline, initial_bits=None, iters_per_temp=1000):
    K = n // 2
    if initial_bits is not None:
        bits = initial_bits
    else:
        bits = _random_balanced_bits(n, K, rng)
    
    cur_c5 = _bits_c5(bits, n)
    best_bits = bits
    best_c5 = cur_c5
    
    T = 0.01  # initial temperature (in units of C5, which is ~0.4)
    T_min = 0.0001
    total_iters = 0
    
    while time.time() < deadline:
        # Do iters_per_temp swaps at this temperature
        ones = []
        zeros = []
        for i in range(n):
            if (bits >> i) & 1:
                ones.append(i)
            else:
                zeros.append(i)
        
        for _ in range(iters_per_temp):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            # Swap: set bit i to 0, set bit j to 1
            new_bits = (bits & ~(1 << i)) | (1 << j)
            new_c5 = _bits_c5(new_bits, n)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                bits = new_bits
                cur_c5 = new_c5
                # Update ones/zeros lists
                ones[ones.index(i)] = j  # This is O(n), inefficient
                zeros[zeros.index(j)] = i
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_bits = bits
            total_iters += 1
        
        T *= 0.95
        if T < T_min:
            break
    
    return best_bits, best_c5

    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    # Take the K smallest vals
    order = np.argsort(vals)
    bits = 0
    for k in range(K):
        bits |= (1 << order[k])
    return bits

    """Local search: try all single swaps, take the best improving one."""
    K = n // 2
    cur_c5 = _bits_c5(bits, n)
    best_c5 = cur_c5
    best_bits = bits
    
    ones = np.array([i for i in range(n) if (bits >> i) & 1])
    zeros = np.array([i for i in range(n) if not ((bits >> i) & 1)])
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_swap = None
        best_val = cur_c5
        # Try a subset of swaps (full search if n is small)
        n_tries = min(len(ones) * len(zeros), 2000)

            j = zeros[rng.integers(len(zeros))]

                best_swap = (i, j, new_bits)

        cur_c5 = best_val

            best_c5 = cur_c5
            best_bits = bits
        # Update ones/zeros
        ones = np.array([ii for ii in range(n) if (bits >> ii) & 1])
        zeros = np.array([ii for ii in range(n) if not ((bits >> ii) & 1)])

Let me also think about the continuous refinement more carefully.

For the continuous problem, the objective is:
f(h) = max_k C_k(h) where C_k(h) = (2/n) * Σ_i h[i] * (1-h[i+k])

This is the max of bilinear functions. The problem is non-convex.

One approach: **semidefinite programming (SDP) relaxation**. The bilinear term h[i]*h[j] can be relaxed using a SDP. But this is expensive and might not give practical improvements.

Another approach: **alternating minimization**. Fix all variables except one, optimize that variable (which is a convex problem in one variable), then move to the next. This is coordinate descent.

For variable h[i]: the objective as a function of h[i] is:
f_i(t) = max_k [(2/n) * (t * A_k(i) + B_k(i))]

where A_k(i) and B_k(i) are the coefficients of t and the constant term in C_k when we isolate h[i].

This is the max of linear functions in t, so it's convex in t! And we minimize it subject to 0 ≤ t ≤ 1.

The minimum of a convex piecewise-linear function on [0,1] is at one of the breakpoints or at the boundary. We can find it by evaluating at the critical points.

Actually, this is a great insight! **Coordinate descent with exact 1D minimization** should work well.

For each coordinate i, the function f_i(t) is convex piecewise-linear. Its minimum can be found by:
1. Computing the slope at t=0 and t=1
