                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _smoothed_grad_step(h, beta, step_size):
    n = len(h)
    target = n / 2.0
    scores = _scores_all(h)
    max_s = np.max(scores)
    w = np.exp(beta * (scores - max_s))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    for i, k in enumerate(np.where(w > 1e-8)[0]):
        grad += w[i] * _score_grad(h, int(k))
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h.copy()
    grad /= gn
    h_new = _project(h - step_size * grad, target)
    return h_new

def _momentum_optimize(h, deadline, n_steps=200, beta_start=15, beta_end=300, step_init=0.05):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores_all(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        grad = np.zeros(n)
        for i, k in enumerate(np.where(w > 1e-8)[0]):
            grad += w[i] * _score_grad(h, int(k))
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.85 * v - step * grad
        h_new = _project(h + v, n / 2.0)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            step = min(step * 1.03, 0.12)
        else:
            v = 0.5 * v
            step *= 0.88
            if step < 1e-6:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(h_start)
    best_h = h_start.copy()
    best_n = n0

    def try_update(h_cand, c5_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        if c5_cand < best_c5 - 1e-14:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            best_n = n_cand
            return True
        return False

    # Phase 1: SLSQP active set at parent resolution
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.18)
        h1, c5_1 = _active_set_optimize(h_start, dl, K_init=20, K_max=100, max_rounds=25)
        try_update(h1, c5_1, n0)

    # Phase 2: Momentum optimization from best
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.32)
        h2, c5_2 = _momentum_optimize(best_h, dl, n_steps=250, beta_start=12, beta_end=350, step_init=0.04)
        try_update(h2, c5_2, len(h2))
        # Follow-up SLSQP
        if time.time() < deadline:
            dl2 = min(deadline, t0 + budget_s * 0.38)
            h2b, c5_2b = _active_set_optimize(h2, dl2, K_init=24, K_max=110, max_rounds=20)
            try_update(h2b, c5_2b, len(h2b))

    # Phase 3: Multi-resolution exploration
    if time.time() < deadline:
        dl3 = min(deadline, t0 + budget_s * 0.58)
        resolutions = [96, 128, 160, 80, 112, 192]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.6180339887]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl3:
                break
            if best_c5 < 0.3807:
                break
            for s in range(2):
                if time.time() > dl3:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    h_init = _quasi_init(n_try, alphas[ri % len(alphas)], rng)
                sub_dl = min(dl3, time.time() + budget_s * 0.035)
                h_try, c5_try = _active_set_optimize(h_init, sub_dl, K_init=16, K_max=80, max_rounds=20)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    best_n = n_try

    # Phase 4: Additional SLSQP from best with larger active set
    if time.time() < deadline:
        dl4 = min(deadline, t0 + budget_s * 0.70)
        h4, c5_4 = _active_set_optimize(best_h, dl4, K_init=32, K_max=best_n * 2 - 1, max_rounds=40)
        try_update(h4, c5_4, len(h4))
        # Momentum polish
        if time.time() < deadline:
            dl4b = min(deadline, t0 + budget_s * 0.76)
            h4b, c5_4b = _momentum_optimize(h4, dl4b, n_steps=150, beta_start=40, beta_end=500, step_init=0.02)
            try_update(h4b, c5_4b, len(h4b))
            if time.time() < deadline:
                dl4c = min(deadline, t0 + budget_s * 0.80)
                h4c, c5_4c = _active_set_optimize(h4b, dl4c, K_init=32, K_max=128, max_rounds=20)
                try_update(h4c, c5_4c, len(h4c))

    # Phase 5: Perturbation restarts
    if time.time() < deadline:
        dl5 = min(deadline, t0 + budget_s * 0.88)
        scale = 0.02
        for attempt in range(50):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.01)
            h_try, c5_try = _active_set_optimize(h_pert, sub_dl, K_init=16, K_max=64, max_rounds=10)
            if c5_try < best_c5 - 1e-13:
                best_h = h_try.copy()
                best_c5 = c5_try
                scale = max(0.003, scale * 0.80)
            else:
                scale = min(0.06, scale * 1.15)

    # Phase 6: Coordinate polish
    if time.time() < deadline:
        dl6 = min(deadline, t0 + budget_s * 0.95)
        hc, c5c = _coordinate_polish(best_h, dl6, n_cycles=2)
        try_update(hc, c5c, len(hc))

    # Phase 7: Pair polish
    if time.time() < deadline:
        dl7 = min(deadline, t0 + budget_s * 0.975)
        hp, c5p = _pair_polish(best_h, dl7, n_passes=2)
        try_update(hp, c5p, len(hp))

    # Final: fine SLSQP
    if time.time() < deadline:
        dl_final = min(deadline, t0 + budget_s * 0.99)
        h_fin, c5_fin = _active_set_optimize(best_h, dl_final, K_init=32, K_max=128, max_rounds=15)
        try_update(h_fin, c5_fin, len(h_fin))

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
