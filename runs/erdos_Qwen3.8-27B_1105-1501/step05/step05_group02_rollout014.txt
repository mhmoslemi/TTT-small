```python
def _binary_c5(B, n):
    """Compute C5 for binary bitmask B of n bits."""
    best = 0
    for d in range(1, n):
        mask = (1 << (n - d)) - 1
        c_pos = (B >> d) & ~B & mask  # 0->1 pairs
        c_neg = B & ~(B >> d) & mask  # 1->0 pairs
        val = max(c_pos.bit_count(), c_neg.bit_count())
        if val > best:

    return best * (2.0 / n)
```

```python
        mask = (1 << (n - d)) - 1
        c_pos = ((B >> d) & ~B) & mask  # 0->1 pairs
        c_neg = (B & ~(B >> d)) & mask  # 1->0 pairs
        val = max(c_pos.bit_count(), c_neg.bit_count())
        if val > best:
            best = val
    return best * (2.0 / n)

def _binary_greedy(B, n, max_rounds=20):

    cur = _binary_c5(B, n)
    for _ in range(max_rounds):
        improved = False
        ones = [i for i in range(n) if (B >> i) & 1]
        zeros = [i for i in range(n) if not ((B >> i) & 1)]

            cur = best_c
            improved = True
        if not improved:
            break
    return B, cur
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binary_c5(B, n):
    best = 0
    for d in range(1, n):
        mask = (1 << (n - d)) - 1
        c_pos = ((B >> d) & ~B) & mask
        c_neg = (B & ~(B >> d)) & mask
        val = c_pos.bit_count()
        val2 = c_neg.bit_count()
        v = val if val > val2 else val2
        if v > best:
            best = v
    return best * (2.0 / n)

def _to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[-target:]
    B = 0
    for i in idx:
        B |= (1 << int(i))
    return B

def _binary_to_arr(B, n):
    h = np.zeros(n)
    for i in range(n):
        if (B >> i) & 1:
            h[i] = 1.0
    return h

def _binary_hill_sa(B_init, n, deadline, rng, restarts=5):
    best_B = B_init
    best_c = _binary_c5(B_init, n)
    B = B_init
    cur = best_c
    T = 0.06
    for restart in range(restarts):
        if time.time() > deadline:
            break
        no_improve = 0
        max_no_improve = 8000
        T = 0.06
        while time.time() < deadline and no_improve < max_no_improve:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            bi = (B >> i) & 1
            bj = (B >> j) & 1
            if bi == bj:
                no_improve += 1
                continue
            new_B = B ^ (1 << i) ^ (1 << j)
            new_c = _binary_c5(new_B, n)
            if new_c < cur or rng.random() < math.exp((cur - new_c) / T):
                B = new_B
                cur = new_c
                no_improve = 0
                if cur < best_c:
                    best_c = cur
                    best_B = B
            else:
                no_improve += 1
            T *= 0.9999
            if T < 0.002:
                T = 0.002
        if time.time() > deadline:
            break
        # Perturb for next restart
        for _ in range(8):
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and ((B >> i) & 1) != ((B >> j) & 1):
                B ^= (1 << i) ^ (1 << j)
    # Final greedy polish
    cur = _binary_c5(B, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if not ((B >> i) & 1):
                continue
            for j in range(n):
                if (B >> j) & 1:
                    continue
                new_B = B ^ (1 << i) ^ (1 << j)
                new_c = _binary_c5(new_B, n)
                if new_c < cur - 1e-15:
                    B = new_B
                    cur = new_c
                    improved = True
                    if cur < best_c:
                        best_c = cur
                        best_B = B
                    break
    B = best_B
    return B, _binary_c5(B, n)

def _smooth_obj_grad(x, tau):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    exps = np.exp((scores - max_s) / tau)
    phi = max_s + tau * math.log(float(np.sum(exps)))
    w = exps / float(np.sum(exps))
    # Compute gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        wr_dx = w[r] * dx
        if d >= 0:
            m = n - d
            if m > 0:
                # c[d] = dx * sum_{i=0}^{m-1} h[i+d]*(1-h[i])
                # d/dc h[j] = wr_dx * [(1-h[j-d]) if d<=j<n  - h[j+d] if j<m]
                lo = d
                hi = n
                # Positive part: (1-h[j-d]) for j in [d, n-1]
                if lo < n:
                    src = h[lo - d:hi - d]  # h[j-d] for j in [d, n-1]
                    grad[lo:hi] += wr_dx * (1.0 - src)
                # Negative part: -h[j+d] for j in [0, m-1]
                if m > 0:
                    src2 = h[:m + d]  # h[j+d] for j in [0, m-1]
                    grad[:m] += wr_dx * (-src2)
        else:
            e = -d
            m = n - e
            if m > 0:
                # c[-d] = dx * sum_{i=0}^{m-1} h[i]*(1-h[i+e])
                # d/dc h[j] = wr_dx * [(1-h[j+e]) if j<m  - h[j-e] if j>=e]
                if m > 0:
                    src = h[:m + e]  # h[j+e] for j in [0, m-1]
                    grad[:m] += wr_dx * (1.0 - src)
                lo = e
                hi = n
                if lo < n:
                    src2 = h[lo - e:hi - e]  # h[j-e] for j in [e, n-1]
                    grad[lo:hi] += wr_dx * (-src2)
    return phi, grad

def _smooth_optimize(h_init, deadline, tau_starts=None):
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    if tau_starts is None:
        tau_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    for tau in tau_starts:
        if time.time() > deadline:
            break
        if time.time() > deadline - 10:
            break
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        
        def obj_grad(x):
            phi, g = _smooth_obj_grad(x, tau)
            return phi, g
        
        res = minimize(obj_grad, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8})
        if hasattr(res, 'x') and res.x is not None:
            h_new = _project(res.x, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        # Also try projected gradient steps as backup
        for _ in range(30):
            if time.time() > sub_dl:
                break
            phi, g = _smooth_obj_grad(h, tau)
            g = g - g.mean()  # Move tangentially to sum constraint
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g = g / gn
            alpha = 0.1
            cur = _exact_c5(h)
            best_a = 0
            best_phi = cur
            for _ in range(15):
                h_try = _project(h - alpha * g, n / 2.0)
                phi_try = _exact_c5(h_try)
                if phi_try < best_phi:
                    best_phi = phi_try
                    best_a = alpha
                alpha *= 0.6
                if alpha < 1e-10:
                    break
            if best_a > 0:
                h = _project(h - best_a * g, n / 2.0)
                c5_new = _exact_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    return best_h, best_c5

def _lp_optimize(h, deadline, radius=0.15, max_it=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = _scores(h)
        G = np.zeros((2 * n - 1, n))
        for d in range(-(n - 1), n):
            r = n - 1 + d
            if d >= 0:
                m = n - d
                if m > 0:
                    G[r, d:] += (1.0 - h[:m])
                    G[r, :m] += -h[d:]
            else:
                g = -d
                m = n - g
                if m > 0:
                    G[r, :m] += (1.0 - h[g:])
                    G[r, g:] += -h[:m]
        G *= (2.0 / n)
        # Top-k constraints
        topk = min(2 * n - 1, 40)
        idx = np.argsort(scores)[-topk:]
        Gk = G[idx]
        sk = scores[idx]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((topk, n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:

                          options={'maxiter': 100})
            if res.success:

                alpha_max = 0.5
                for j in range(n):

                lo_a = 0.0
                hi_a = alpha_max

                c2 = lo_a + gr * (hi_a - lo_a)
                f1 = _exact_c5(h + c1 * p)

                        f2 = f1
                        c2 = c1
                        c1 = hi_a - gr * (hi_a - lo_a)
                        f1 = _exact_c5(h + c1 * p)
                    else:
                        lo_a = c1
                        f1 = f2
                        c1 = c2
                        c2 = lo_a + gr * (hi_a - lo_a)
                        f2 = _exact_c5(h + c2 * p)
                opt_a = (lo_a + hi_a) / 2.0
                h_new = h + opt_a * p
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-13:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.3)

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_binary(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    mask = ((x * alpha) % 1.0) < 0.5
    h[mask] = 1.0
    # Fix count to exactly n/2
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        # Remove excess ones
        excess = np.where(h == 1.0)[0]
        to_remove = excess[rng_quasi.permutation(len(excess))[:ones - target]]
        h[to_remove] = 0.0
    elif ones < target:
        deficit = np.where(h == 0.0)[0]
        to_add = deficit[rng_quasi.permutation(len(deficit))[:target - ones]]
        h[to_add] = 1.0
    return h

        best_h = h_s.copy()
    
    # Phase 4: LP polish after smooth (10% time)
    if time.time() < t0 + budget_s * 0.75:
        dl4 = t0 + budget_s * 0.82
        h4, c5_4 = _lp_optimize(best_h, dl4, radius=0.1, max_it=3000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        # Try binary again from improved continuous
        B_4 = _to_binary(best_h, best_n)
        sub_dl = min(dl4, time.time() + budget_s * 0.02)
        B_4_opt, c5_4b = _binary_hill_sa(B_4, best_n, sub_dl, rng, restarts=3)
        h_4b = _binary_to_arr(B_4_opt, best_n)
        c5_4b_np = _exact_c5(h_4b)
        if c5_4b_np < best_c5:
            best_c5 = c5_4b_np
            best_h = h_4b.copy()
    
    # Phase 5: Multi-start exploration (10% time)
    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.92
        # Quasi-periodic starts
        for alpha in [0.6180339887, 0.4142135624, 0.7071067812]:
            if time.time() > dl5:
                break
            n_qp = best_n
            h_qp = _quasi_binary(n_qp, alpha)
            B_qp = 0
            for i in range(n_qp):
                if h_qp[i] > 0.5:
                    B_qp |= (1 << i)
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            B_qp_opt, c5_qp = _binary_hill_sa(B_qp, n_qp, sub_dl, rng, restarts=3)
            h_qp_arr = _binary_to_arr(B_qp_opt, n_qp)
            c5_qp_np = _exact_c5(h_qp_arr)
            if c5_qp_np < best_c5:
                best_c5 = c5_qp_np
                best_h = h_qp_arr.copy()
        
        # Random binary starts
        for _ in range(5):
            if time.time() > dl5:
                break
            B_rand = 0
            positions = rng.choice(best_n, size=best_n // 2, replace=False)
            for p in positions:
                B_rand |= (1 << int(p))
            sub_dl = min(dl5, time.time() + budget_s * 0.015)
            B_rand_opt, c5_rand = _binary_hill_sa(B_rand, best_n, sub_dl, rng, restarts=3)
            h_rand_arr = _binary_to_arr(B_rand_opt, best_n)
            c5_rand_np = _exact_c5(h_rand_arr)
            if c5_rand_np < best_c5:
                best_c5 = c5_rand_np
                best_h = h_rand_arr.copy()
    
    # Phase 6: Final polish (5% time)
    if time.time() < t0 + budget_s * 0.95:
        dl6 = t0 + budget_s * 0.985
        # Final LP
        h_fin, c5_fin = _lp_optimize(best_h, dl6, radius=0.05, max_it=2000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        # Final binary check
        B_fin = _to_binary(best_h, best_n)
        sub_dl = min(dl6, time.time() + 5)
        B_fin_opt, c5_fin_b = _binary_hill_sa(B_fin, best_n, sub_dl, rng, restarts=2)
        h_fin_arr = _binary_to_arr(B_fin_opt, best_n)
        c5_fin_np = _exact_c5(h_fin_arr)
        if c5_fin_np < best_c5:
            best_c5 = c5_fin_np
            best_h = h_fin_arr.copy()
    
    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
```
