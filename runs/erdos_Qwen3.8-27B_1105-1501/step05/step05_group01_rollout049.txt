def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _score_unscaled(h):
    """Return unscaled correlation scores."""
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto [0,1]^n with sum = target using bisection on water level."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_robust(h, d, dx, radius, max_grid=60):
    """Robust 1D line search: grid evaluation + golden on best brackets."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid evaluation
    grid_t = np.linspace(t_lo, t_hi, max_grid)
    grid_f = np.empty(max_grid)
    for idx, t in enumerate(grid_t):
        grid_f[idx] = float(np.max(S0 + t * T - t*t * C)) * dx
    
    # Find best grid point
    best_idx = int(np.argmin(grid_f))
    
    # Golden section on the bracket around the best point
    # Also try near t_lo and t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    # Try golden on [t_lo, best bracket] and [best bracket, t_hi]
    # Define the bracket as [grid_t[max(0,best_idx-2)], grid_t[min(max_grid-1,best_idx+2)]]
    lo_b = grid_t[max(0, best_idx - 3)]
    hi_b = grid_t[min(max_grid - 1, best_idx + 3)]
    
    def _phi(t):
        return float(np.max(S0 + t * T - t*t * C)) * dx
    
    # Golden section search
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = _phi(c1)
    f2 = _phi(c2)
    for _ in range(50):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b = c2
            c2 = c1
            f2 = f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = _phi(c1)
        else:
            lo_b = c1
            c1 = c2
            f1 = f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = _phi(c2)
    
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = _phi(opt_t)
    
    # Also check the endpoints and grid best
    candidates = [(opt_t, opt_val), (t_lo, _phi(t_lo)), (t_hi, _phi(t_hi)),
                  (grid_t[best_idx], grid_f[best_idx])]
    best_candidate = min(candidates, key=lambda x: x[1])
    return best_candidate[0], best_candidate[1]

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    m_scores = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m_scores, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -float(np.max(scores)) + 1e-6  # Try to reduce the max score
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.10, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search_robust(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.04
    return best_h, best_c5

def _binary_block_move(h_bin, deadline, block_size=3):

        # Find a random block of 1s
        ones = np.where(cur > 0.5)[0]
        if len(ones) < block_size:
            break
        start_idx = int(rng.integers(0, len(ones) - block_size + 1))
        block_ones = ones[start_idx:start_idx + block_size]
        # Check if contiguous
        if not np.all(np.diff(block_ones) == 1):
            stall += 5
            continue
        # Find a random position with at least block_size consecutive zeros
        zeros = np.where(cur <= 0.5)[0]
        if len(zeros) < block_size:
            break
        # Look for consecutive zeros
        found = False
        for _ in range(20):
            z_start = int(rng.integers(0, len(zeros) - block_size + 1))
            block_zeros = zeros[z_start:z_start + block_size]
            if np.all(np.diff(block_zeros) == 1):
                # Move the block
                h_try = cur.copy()
                h_try[block_ones] = 0.0
                h_try[block_zeros] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:

                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    stall = 0
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    """Convert to binary by taking top n/2 values."""
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturbation(h, n, amplitude, rng):
    """Add smooth (low-frequency) perturbation."""
    x = np.arange(n)
    # Random low-frequency sinusoidal perturbation
    freq = rng.uniform(0.5, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    perturb = amplitude * np.sin(2 * np.pi * freq * x / n + phase)
    # Add small noise
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius, n_grid=50):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    m_scores = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m_scores, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_perturb(h, amplitude, rng):

        h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.03, stall_limit=60)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping with smooth perturbations (15%)
    d5 = t0 + budget_s * 0.90
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 and n_restarts < 120:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, stall_limit=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.06)

    # Phase 6: Final polish (10%)
    d6 = deadline
    # Try a couple more n values
    n_best = len(best_h)
    for dn in [-3, 3, -6, 6]:
        if time.time() > d6 - 8:
            break
        n_try = n_best + dn
        if n_try < 48 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.010)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.05, stall_limit=25)
        if time.time() < d6 - 4:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d6, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=8000, T_init=0.001)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Final LP
    h6 = _project(best_h.copy(), len(best_h) / 2.0)
    h6, c5_6 = _lp_loop(h6, d6, radius_init=0.02, stall_limit=40)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Final pair swap
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_loop(h6, d6, n_pairs=2000, radius=0.15, seed=555)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
