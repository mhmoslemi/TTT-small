        rng = np.random.default_rng(0)

        base[::2] = 1.0

            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.08
    stall = 0

    while time.time() < deadline:
        if stall > 50:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _safe_line_search(h, p, dx, radius, n_grid=30)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.2)
        else:
            stall += 1
            radius *= 0.72

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick local search on initial (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_search(h_init, d1, max_k=3, use_guided=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration with structured starts (32%)
    d2 = t0 + budget_s * 0.40
    n_values = [n0, n0 - 8, n0 + 8, n0 - 16, n0 + 16, n0 - 24, n0 + 24,
                n0 - 32, n0 + 32, 80, 100, 120, 144, 168, 180, 96, 112, 128, 140, 160]
    tried_ns = set()
    for n_try in n_values:
        if time.time() > d2:
            break
        if n_try < 40 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        starts = []
        starts.append(_resample(best_h, n_try))
        starts.append(_thue_morse_binary(n_try))
        if n_try >= 6:
            starts.append(_periodic_binary(n_try, 4, 2, seed_val=7))
        if n_try >= 10:
            starts.append(_periodic_binary(n_try, 6, 3, seed_val=13))
        starts.append(_random_binary(n_try, rng))

        for h_start in starts:
            if time.time() > d2:
                break
            h_start = _project(h_start, n_try / 2.0)
            sub_dl = min(d2, time.time() + budget_s * 0.015)
            h_s, c5_s = _local_search(h_start, sub_dl, max_k=3, use_guided=True)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 3: Basin hopping (20%)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _basin_hopping(best_h, d3, n_attempts=35)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP refinement (15%)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = _lp_refine(best_h, d4)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Intensive block search (12%)
    d5 = t0 + budget_s * 0.87
    h5, c5_5 = _local_search(best_h, d5, max_k=2, use_guided=True)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final LP polish (7%)
    d6 = deadline
    h6, c5_6 = _lp_refine(best_h, d6)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
