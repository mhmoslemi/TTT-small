    bounds = [(max(-h[j], -radius), min(1.0 - h[j], radius)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-13:
                return p
    except Exception:
        pass
    return None

def _lp_main(h, deadline, radius_init=0.12, k_top=None, max_stall=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(12345)
    it = 0
    while time.time() < deadline:
        it += 1
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_dir(h, scores, G, radius, k_top)
        if p is None:
            stall += 1
            radius *= 0.72
            if radius < 0.0008:
                break
            if stall >= max_stall:
                h = h + rng.normal(0, 0.015, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = radius_init * 0.45
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        h_new = np.clip(h + alpha_max * p, 0.0, 1.0)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.025, 0.28)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.125, 0.0625]:
                a = alpha_max * frac
                h_try = np.clip(h + a * p, 0.0, 1.0)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-14:
                    h = h_try
                    cur_c5 = c5_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.01, 0.28)
                    improved = True
                    break
            if not improved:
                stall += 1
                radius *= 0.7
                if radius < 0.0008:
                    break
                if stall >= max_stall:
                    h = h + rng.normal(0, 0.012, n)
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = radius_init * 0.4
    return best_h, best_c5

def _smooth_grad(h, deadline, beta_start=20, beta_end=800, n_steps=400, step_init=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.015, 0.10)
        else:
            v = 0.5 * v
            step *= 0.88
            if step < 1e-7:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coord_polish(h, deadline, n_cycles=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(35):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h, deadline, n_passes=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_size = max(1, n // 20)
    rng = np.random.default_rng(777)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        idx = list(range(n))
        rng.shuffle(idx)
        for j in idx:
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k_off in range(step_size, n, step_size):
                if time.time() > deadline:
                    break
                k = (j + k_off) % n
                if k == j:
                    continue
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(12):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _basis_matrix(n, q, rng):
    B = np.zeros((n, q))
    widths = []
    p = 4
    while 2 * p <= n:
        widths.append(p)
        p *= 2
    widths.append(1)
    widths.append(2)
    col = 0
    tries = 0
    while col < q and tries < q * 20:
        tries += 1
        r = widths[tries % len(widths)]
        total_w = 2 * r
        if total_w > n:
            continue
        start = int(rng.integers(0, n - total_w + 1))
        c = rng.choice([-1.0, 1.0])
        B[start:start+r, col] = c
        B[start+r:start+2*r, col] = -c
        norm = np.sqrt(float(np.sum(B[:, col]**2)))
        if norm > 1e-10:
            B[:, col] /= norm
            col += 1
    return B[:, :col]

def _basis_search(h, deadline, q=None, n_pop=40, n_gen=150):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(42)
    if q is None:
        q = min(50, n // 3)
    B = _basis_matrix(n, q, rng)
    actual_q = B.shape[1]
    if actual_q < 4:
        return h.copy(), _exact_c5(h)
    y0 = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    bounds_y = [(-2.0, 2.0)] * actual_q
    pop = np.zeros((n_pop, actual_q))
    pop[0] = y0
    for i in range(1, n_pop):
        pop[i] = y0 + 0.3 * rng.standard_normal(actual_q)
        pop[i] = np.clip(pop[i], -2.0, 2.0)
    fit = np.zeros(n_pop)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for i in range(n_pop):
        h_try = _project(0.5 + B @ pop[i], target)
        fit[i] = _exact_c5(h_try)
        if fit[i] < best_c5:
            best_c5 = fit[i]
            best_h = h_try.copy()
    F = 0.7
    CR = 0.8
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(fit)
        best_i = order[0]
        for i in range(n_pop):
            if time.time() > deadline:
                break
            ia = rng.integers(n_pop)
            ib = rng.integers(n_pop)
            ic = rng.integers(n_pop)
            while ib == ia or ic == ia or ib == ic:
                ib = rng.integers(n_pop)
                ic = rng.integers(n_pop)
            trial = pop[ia] + F * (pop[ib] - pop[ic])
            cross_mask = rng.random(actual_q) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(actual_q)] = True
            trial[cross_mask] = pop[ia][cross_mask]
            trial = np.clip(trial, -2.0, 2.0)
            h_try = _project(0.5 + B @ trial, target)
            f_try = _exact_c5(h_try)
            if f_try <= fit[i]:
                pop[i] = trial
                fit[i] = f_try
                if f_try < best_c5:
                    best_c5 = f_try
                    best_h = h_try.copy()
        F = max(0.4, F - 0.002)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Intensive LP with adaptive k (45% budget)
    dl1 = t0 + budget_s * 0.45
    # Start with top-20 constraints for speed, then expand
    h1, c5_1 = _lp_main(best_h, min(dl1, t0 + budget_s * 0.15), radius_init=0.15, k_top=20, max_stall=15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Expand to top-50
    h1b, c5_1b = _lp_main(best_h, min(dl1, t0 + budget_s * 0.30), radius_init=0.10, k_top=50, max_stall=20)
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b.copy()
    # Full constraints for final precision
    h1c, c5_1c = _lp_main(best_h, dl1, radius_init=0.06, k_top=None, max_stall=25)
    if c5_1c < best_c5:
        best_c5 = c5_1c
        best_h = h1c.copy()

    # Phase 2: Smoothed gradient with continuation (15% budget)
    dl2 = t0 + budget_s * 0.60
    h2, c5_2 = _smooth_grad(best_h, dl2, beta_start=10, beta_end=600, n_steps=350, step_init=0.035)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Quick LP polish after gradient
    if time.time() < t0 + budget_s * 0.65:
        h2p, c5_2p = _lp_main(best_h, t0 + budget_s * 0.65, radius_init=0.05, k_top=30, max_stall=10)
        if c5_2p < best_c5:
            best_c5 = c5_2p
            best_h = h2p.copy()

    # Phase 3: Multi-resolution exploration (15% budget)
    dl3 = t0 + budget_s * 0.80
    res_list = [64, 80, 96, 112, 160]
    for n_try in res_list:
        if time.time() > dl3:
            break
        if best_c5 < 0.3805:
            break
        h_init = _resample(best_h, n_try)
        sub_dl = min(dl3, time.time() + budget_s * 0.025)
        h_try, c5_try = _lp_main(h_init, sub_dl, radius_init=0.12, k_top=20, max_stall=12)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 4: Basis search for diversity (10% budget)
    dl4 = t0 + budget_s * 0.90
    if best_c5 > 0.3806:
        h4, c5_4 = _basis_search(best_h, min(dl4, t0 + budget_s * 0.85), q=40, n_pop=30, n_gen=100)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Perturbation restarts
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl4:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl4, time.time() + budget_s * 0.008)
        h_try, c5_try = _lp_main(h_pert, sub_dl, radius_init=0.05, k_top=15, max_stall=8)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 5: Coordinate polish (5% budget)
    dl5 = t0 + budget_s * 0.95
    hc, c5c = _coord_polish(best_h, dl5, n_cycles=2)
    if c5c < best_c5:
        best_c5 = c5c
        best_h = hc.copy()

    # Phase 6: Pair polish (3% budget)
    dl6 = t0 + budget_s * 0.98
    hp, c5p = _pair_polish(best_h, dl6, n_passes=3)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Final LP with tiny radius
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _lp_main(best_h, t0 + budget_s * 0.998, radius_init=0.008, k_top=None, max_stall=30)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
