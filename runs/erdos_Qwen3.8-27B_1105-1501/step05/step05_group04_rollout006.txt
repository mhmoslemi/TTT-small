Actually, I want to think more carefully about whether SLSQP will actually help here. The problem is:

minimize max_r score_r(h)
subject to 0 ≤ h ≤ 1, sum(h) = n/2

This is a nonconvex problem (max of bilinear functions). The SLSQP epigraph formulation is:

minimize t
subject to score_r(h) ≤ t for all r
         0 ≤ h ≤ 1
         sum(h) = n/2

The KKT conditions at a local minimum require:
- There exist multipliers λ_r ≥ 0 for the inequality constraints
- There exists a multiplier ν for the equality constraint
- The gradient of the Lagrangian is zero

The Lagrangian is:
L = t + Σ_r λ_r (score_r(h) - t) + ν (sum(h) - n/2)

∂L/∂t = 1 - Σ_r λ_r = 0, so Σ_r λ_r = 1
∂L/∂h_j = Σ_r λ_r * ∂score_r/∂h_j + ν = 0

This means that at a local minimum, the weighted sum of the score gradients (with weights summing to 1) is constant across all coordinates. This is the same condition that the LP method identifies (the LP finds a direction that reduces the active constraints).

So if the LP method has converged (no improving direction exists), the SLSQP will also be at a KKT point and won't improve. The difference is in the path taken: SLSQP might take a different path through the constraint surface and find a different KKT point.

The key insight is that SLSQP with multiple starting points can explore different basins of attraction. The LP method from a single starting point only explores one basin.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            a = -d
            m = n - a
            if m > 0:
                G[r, :m] += (1.0 - h[a:])
                G[r, a:] -= h[:m]
    return G

def _slsqp_optimize(h0, max_iter=300, ftol=1e-12, deadline=None):
    """Solve the epigraph SLSQP problem starting from h0.
    Returns (h_sol, t_sol) or None on failure."""
    n = len(h0)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    t0 = _exact_c5(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Equality: sum(h) = n/2
    def eq(x):
        return np.array([np.sum(x[:-1]) - n / 2.0])
    
    def eq_grad(x):
        g = np.ones(n + 1)
        g[-1] = 0.0
        return g
    
    # Inequality: t - score_r(h) >= 0
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        s = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - s
    
    def ineq_grad(x):
        h = x[:-1]
        G = _grad_matrix(h) * dx
        J = np.zeros((n_lags, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    constraints = [
        {'type': 'eq', 'fun': eq, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_grad}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if deadline is not None and time.time() > deadline:
            return None
        if res.success or (res.x[-1] < t0 and np.all(res.x[:-1] >= -1e-8) and np.all(res.x[:-1] <= 1 + 1e-8)):
            h_sol = np.clip(res.x[:-1], 0.0, 1.0)
            h_sol = _project(h_sol, n / 2.0)
            t_act = _exact_c5(h_sol)
            return h_sol, t_act
    except Exception:
        pass
    return None

def _lp_direction(h, scores, G, radius):
    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=-scores, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)) - 1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None
