def eval_c5(h):

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

        else:

def lag_values(h):
    """Compute all lag values F_s(h) * dx matching np.correlate ordering."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_jacobian(h):
    """Compute Jacobian of lag values w.r.t. h. Shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                # F_d = dx * sum_{i=0}^{m-1} h[i+d] * (1-h[i])
                # dF_d/dh[j]: 
                #   if j >= d: contribution from h[j] in first factor: dx*(1-h[j-d])
                #   if j <= m-1: contribution from h[j] in (1-h[j]) factor of pair (j+d, j): -dx*h[j+d]
                G[r, d:] += dx * (1.0 - h[:m])
                G[r, :m] -= dx * h[d:d+m]
        else:
            g = -d  # g > 0
            m = n - g
            if m > 0:
                # F_{-g} = dx * sum_{i=0}^{m-1} h[i] * (1-h[i+g])
                # dF_{-g}/dh[j]:
                #   if j <= m-1: dx*(1-h[j+g]) from first factor
                #   if j >= g: -dx*h[j-g] from (1-h[j-g]) in pair (j-g, j)
                G[r, :m] += dx * (1.0 - h[g:])
                G[r, g:] -= dx * h[:m]
    return G

def smooth_max(h, beta):
    """Log-sum-exp smoothed max of lag values."""
    F = lag_values(h)
    m = float(np.max(F))
    if beta < 1e-12:
        return m
    Z = np.exp((F - m) / beta)
    s = float(np.sum(Z))
    return m + beta * np.log(s)

def smooth_max_grad(h, beta):
    """Gradient of log-sum-exp smoothed max."""
    F = lag_values(h)
    m = float(np.max(F))
    if beta < 1e-12:
        # Return gradient of the max (just the active lag)
        idx = np.argmax(F)
        G = lag_jacobian(h)
        return G[idx, :]
    Z = np.exp((F - m) / beta)
    P = Z / float(np.sum(Z))
    G = lag_jacobian(h)
    return G.T @ P

def lp_direction(h, scores, G, radius, n_active, b_shift):

    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))

    A_ub[:, n] = -1.0
    b_ub = np.full(n_lags, -scores + b_shift)
    # For active set, use tighter constraint
    sorted_idx = np.argsort(scores)[::-1]
    for k in sorted_idx[:n_active]:
        b_ub[k] = -scores[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)) - 1e-14:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, d, dx, trust_radius):
    """Line search with quadratic model."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

            return x[-1]

            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g

            t = x[-1]

            J[:, n] = -1.0

import numpy as np
from scipy.optimize import linprog, minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += dx * (1.0 - h[:m])
                G[r, :m] -= dx * h[d:d + m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += dx * (1.0 - h[g:])
                G[r, g:] -= dx * h[:m]
    return G

def lp_direction(h, scores, G, radius, n_active, b_shift):
    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.full(n_lags, -scores + b_shift)
    sorted_idx = np.argsort(scores)[::-1]
    for k in sorted_idx[:n_active]:
        b_ub[k] = -scores[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)) - 1e-14:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, eval_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def lp_loop(h, deadline, radius_init=0.15, n_active_init=6):

    target = n / 2.0

    radius = radius_init
    stall = 0
    n_active = n_active_init
    iters = 0

        iters += 1

        G = lag_jacobian(h)
        b_shift = radius * (2.0 / n) * 0.3

            h = h_new

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)

def smooth_max_obj(h, beta):
    F = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = float(np.max(F))
    if beta < 1e-10:
        return m
    Z = np.exp((F - m) / beta)
    return m + beta * np.log(float(np.sum(Z)))

def smooth_max_grad(h, beta):
    n = len(h)
    F = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = float(np.max(F))
    Z = np.exp((F - m) / beta)
    P = Z / float(np.sum(Z))
    G = lag_jacobian(h)
    return G.T @ P

def smooth_optimize(h, deadline, betas):

    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0, 1)] * n
        try:
            res = minimize(best_h, lambda x: smooth_max_obj(x, beta),
                          jac=lambda x: smooth_max_grad(x, beta),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 40, 'ftol': 1e-10, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_cand = project_box_sum(res.x, target)
                c5_cand = eval_c5(h_cand)
                if c5_cand < best_c5 - 1e-13:

        except Exception:

    return best_h, best_c5

def epigraph_slsqp(h, deadline, rng, max_attempts=4):

            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        
        def cons_ineq(x):
            hp = x[:n]
            return np.correlate(hp, 1.0 - hp, mode='full') * (2.0 / n) - x[-1]
        
        def cons_ineq_jac(x):

            G = lag_jacobian(hp)
            J = np.zeros((n_lags, n + 1))

            J[:, n] = -1.0
            return J
        
        cons = [{'type': 'ineq', 'fun': cons_ineq, 'jac': cons_ineq_jac},
                {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target,
                 'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
        bounds = [(0, 1)] * n + [(0.01, 1.0)]
        try:
            res = minimize(x0, obj, jac=obj_grad, method='SLSQP',

def multiresearch(h, deadline, rng):

def coord_refine(h, deadline, n_iter=600):

    for _ in range(n_iter):

        i = interior[rng.integers(0, len(interior))]
