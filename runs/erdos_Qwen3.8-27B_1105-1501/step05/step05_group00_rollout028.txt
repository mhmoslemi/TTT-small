def project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target} in L2."""
    n = len(v)

        return np.zeros(n)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):

        if s > target:
            lo = mid
        else:
            hi = mid

        cnt = int(np.sum(mask))
