import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Intensive exact pair descent at original resolution
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _pair_descent_exact(h_start, dl1, max_passes=300, thresh_init=1e-7)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient polishing
    if time.time() < t0 + budget_s * 0.38:
        dl2 = t0 + budget_s * 0.38
        h_g, c5_g = _gradient_step(best_h, dl2, n_steps=60, beta_start=50, beta_end=500)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        # Follow with pair descent
        sub_dl = min(dl2, time.time() + budget_s * 0.02)
        h_p, c5_p = _pair_descent_exact(best_h, sub_dl, max_passes=30, thresh_init=1e-8)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 3: Diverse starts at original resolution
    if time.time() < t0 + budget_s * 0.55:
        dl3 = t0 + budget_s * 0.55
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
        for i in range(len(alphas)):
            if time.time() > dl3:
                break
            h_init = _quasi_init(best_n, alphas[i], rng)
            sub_dl = min(dl3, time.time() + budget_s * 0.04)
            h_try, c5_try = _pair_descent_exact(h_init, sub_dl, max_passes=40, thresh_init=1e-7)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 4: Perturbation restarts
    if time.time() < t0 + budget_s * 0.72:
        dl4 = t0 + budget_s * 0.72
        scale = 0.025
        for attempt in range(80):
            if time.time() > dl4:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.012)
            h_try, c5_try = _pair_descent_exact(h_pert, sub_dl, max_passes=25, thresh_init=1e-8,
                                                 shuffle=True, rng=rng)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.004, scale * 0.7)
            else:
                scale = min(0.06, scale * 1.12)

    # Phase 5: Other resolutions
    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.85
        for n_try in [96, 160, 192]:
            if time.time() > dl5:
                break
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl5, time.time() + budget_s * 0.05)
            h_try, c5_try = _pair_descent_exact(h_init, sub_dl, max_passes=40, thresh_init=1e-7)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 6: Binary and Fejer starts
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.90
        h_b = _binary_init(best_n, rng)
        sub_dl = min(dl6, time.time() + budget_s * 0.02)
        h_try, c5_try = _pair_descent_exact(h_b, sub_dl, max_passes=30, thresh_init=1e-7)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        if time.time() < dl6:
            h_f = _fejer_init(best_n, rng)
            sub_dl2 = min(dl6, time.time() + budget_s * 0.015)
            h_try2, c5_try2 = _pair_descent_exact(h_f, sub_dl2, max_passes=25, thresh_init=1e-7)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase 7: Final intensive polish
    if time.time() < hard_deadline:
        h_fin, c5_fin = _pair_descent_exact(best_h, hard_deadline, max_passes=150, thresh_init=1e-10)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
