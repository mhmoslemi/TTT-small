def _lp_direction(h, G_active, radius):
    n = len(h)
    # min t + epsilon * sum(d_i^2) approximation
    # min t s.t. G_a @ d <= t, -radius <= d_i <= radius, sum(d) = 0, box constraints
    # Variables: [d_1, ..., d_n, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # G_active @ d <= t  =>  G_active @ d - t <= 0
    A_ub = np.zeros((G_active.shape[0], n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = np.zeros(G_active.shape[0])
    
    # Box constraints on d: -min(h, radius) <= d_i <= min(1-h, radius)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0, 10*C))  # t >= 0, upper bound loose
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except:
        pass
    return None

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _exact_c5(h):
    return float(np.max(_corr_vals(h)))

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))

def _lp_dir(h, G_act, radius):
    """Solve LP: min t s.t. G_act @ d <= t, box + sum constraints on d."""
    n = len(h)
    m = G_act.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.array([[1.0] * n + [0.0]])
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        bounds.append((max(-radius, -h[i]), min(radius, 1.0 - h[i])))
    bounds.append((0.0, 2.0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

            else:
                stall += 5
                if stall > max_stall:
                    break
                radius *= 0.5
                if radius < 1e-6:

        if np.any(neg):

            if t_min > 0:
                t_max = min(t_max, t_min)  # can't go below 0
        t_max = max(t_max, 1e-6)

            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                # Adapt radius

                    radius = min(radius * 1.15, 0.5)
                elif ratio > 1.3:
                    radius *= 0.8
            else:

                radius *= 0.75
        else:
            stall += 1
            radius *= 0.75

            stall = 0
        if stall > max_stall:

        if radius < 1e-7:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=100):
    """Try all pairs of interior coordinates."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[ii]
            for jj in range(ii + 1, len(interior)):
                j = interior[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                F = _corr_vals(h)
                Gd = _grad_matrix(h) @ d
                Q2 = _quad_terms(d)
                t_max = min(h[i], 1.0 - h[j])
                if t_max < 1e-12:
                    continue
                t_s, v = _line_search_exact(h, d, F, Gd, Q2, t_max)
                if t_s > 1e-12:
                    h_try = h + t_s * d
                    h_try = _project(h_try, target)
                    c5_t = _exact_c5(h_try)
                    if c5_t < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_t
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    """Binary SA with swaps."""
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.006
    T_min = 0.00001
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    patience = 0
    
    while time.time() < deadline and T > T_min:
        if patience > 15000:
            T = max(T_min, T * 0.98)
            patience = 0
        for _ in range(200):
            if time.time() > deadline:
                break
            if not ones or not zeros:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            cov = _corr_vals(h)
            nc = float(np.max(cov))
            delta = nc - cur_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                cur_c5 = nc
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones = [i2 for i2 in ones if i2 != i]
                ones.append(j)
                zeros = [i2 for i2 in zeros if i2 != j]
                zeros.append(i)
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * 0.9998)
    
    return best_h, best_c5

def _structured_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 5))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    return _project(h, n / 2.0)
