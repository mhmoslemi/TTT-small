    max_count = 0

                    h_bin[to_flip] = 0
                elif ones < K:
                    idx_zeros = np.where(h_bin == 0)[0]
                    to_flip = idx_zeros[:K - ones]
                    h_bin[to_flip] = 1
                for i in range(n_bin):
                    if h_bin[i] > 0.5:
                        seed_bits |= (1 << i)
            sub_dl = min(dl4, time.time() + budget_s * 0.03)
            bits_best, c5_bits = _binary_sa(n_bin, sub_dl, rng, seed_bits=seed_bits,
                                           n_iters=int(budget_s * 200))
            if c5_bits < best_c5:
                h_conv = _bits_to_h(bits_best, n_bin)
                best_c5 = c5_bits
                best_h = h_conv
                best_n = n_bin
                # Smooth polish from binary solution
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                if sub_dl2 > time.time() + 5:
                    h_sp, c5_sp = _smooth_optimize(h_conv, sub_dl2,
                                                  taus=[0.01 * base, 0.003 * base], rng=rng)
                    if c5_sp < best_c5:
                        best_c5 = c5_sp
                        best_h = h_sp.copy()

    # Phase 5: Multi-resolution smooth optimization
    dl5 = t0 + budget_s * 0.80
    for n_try in [n0, n0 - 16, n0 + 32, 96, 160, 112]:
        if time.time() > dl5:
            break
        if n_try < 32 or n_try % 2 != 0:
            continue
        if n_try == best_n:
            continue
        h_res = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + budget_s * 0.035)
        base_try = max(_exact_c5(h_res), 0.1)
        taus_try = [0.03 * base_try, 0.01 * base_try, 0.003 * base_try]
        h_try, c5_try = _smooth_optimize(h_res, sub_dl, taus=taus_try, rng=rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            # LP polish
            sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
            h_try2, c5_try2 = _lp_optimize(h_try, sub_dl2, radius_init=0.08, max_it=1000, topk=20)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase 6: Quasi-periodic + binary
    dl6 = t0 + budget_s * 0.88
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
    for n_qp in [best_n, best_n - 16, 96]:
        if time.time() > dl6:
            break
        if n_qp < 32 or n_qp % 2 != 0:
            continue
        for a in alphas[:3]:
            if time.time() > dl6:
                break
            h_qp = _quasi_periodic(n_qp, a)
            sub_dl = min(dl6, time.time() + budget_s * 0.02)
            base_qp = max(_exact_c5(h_qp), 0.1)
            h_qo, c5_qo = _smooth_optimize(h_qp, sub_dl,
                                          taus=[0.03 * base_qp, 0.008 * base_qp, 0.002 * base_qp],
                                          rng=rng)
            if c5_qo < best_c5:
                best_c5 = c5_qo
                best_h = h_qo.copy()
                best_n = n_qp

    # Phase 7: Final polish
    dl7 = t0 + budget_s * 0.97
    base_final = max(best_c5, 0.1)
    h_fin, c5_fin = _smooth_optimize(best_h, dl7,
                                     taus=[0.002 * base_final, 0.0005 * base_final], rng=rng)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()
    # Final LP
    sub_dl = min(dl7, time.time() + budget_s * 0.02)
    h_fin2, c5_fin2 = _lp_optimize(best_h, sub_dl, radius_init=0.04, max_it=2000, topk=20)
    if c5_fin2 < best_c5:
        best_c5 = c5_fin2
        best_h = h_fin2.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
