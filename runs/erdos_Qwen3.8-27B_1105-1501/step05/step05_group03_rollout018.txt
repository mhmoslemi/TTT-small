    """Gradient of raw score for shift s (where s = k - (n-1), k is output index).
    s ranges from -(n-1) to n-1. Returns gradient of the raw sum (before *dx)."""
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            # (1-h[j+s]) for j in [0, m-1]
            g[:m] += (1.0 - h[s:s+m])
            # -h[j-s] for j in [s, n-1]

        m = n - g_val

            g[g_val:] += (1.0 - h[:m])
            # -h[j+g_val] for j in [0, m-1]
            g[:m] += -h[g_val:n]
    return g

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _scores_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _score_grad_raw(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[:m] += (1.0 - h[s:s + m])
            g[s:] += -h[:m]
    else:
        gv = -s
        m = n - gv
        if m > 0:
            g[gv:] += (1.0 - h[:m])
            g[:m] += -h[gv:n]
    return g

def _slsqp_step(h, active_indices, max_iter=60):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    na = len(active_indices)
    if na == 0:
        return h, None
    scores = _scores_all(h)
    T_init = float(np.max(scores[active_indices])) + 0.002
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = T_init

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def eq_f(x):
        return float(np.sum(x[:n])) - target

    def eq_j(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    def ineq_f(x):
        hx = x[:n]
        sc = _scores_all(hx)
        return x[n] - sc[active_indices]

    def ineq_j(x):
        hx = x[:n]
        J = np.zeros((na, n + 1))
        for i, idx in enumerate(active_indices):
            d = idx - (n - 1)
            grad = _score_grad_raw(hx, d, n) * dx
            J[i, :n] = -grad
            J[i, n] = 1.0
        return J

    cons = [
        {'type': 'eq', 'fun': eq_f, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_f, 'jac': ineq_j}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(obj, x0, jac=obj_jac, constraints=cons,
                       bounds=bounds, method='SLSQP',
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            new_h = _project(res.x[:n], target)
            return new_h, float(res.x[n])
    except Exception:
        pass
    return h, None

def _slsqp_optimize(h, deadline, active_counts=None, max_iter=60, max_inner=6):
    n = len(h)
    target = n / 2.0
    if active_counts is None:

            break

        active = order[:count]

            if time.time() > deadline:

                    best_h = new_h
                    best_c5 = new_c5
                    scores = _scores_all(best_h)
                    order = np.argsort(-scores)
                    active = order[:count]
                else:

    return best_h, best_c5

def _lp_direction(h, active_indices, radius):
    n = len(h)
    dx = 2.0 / n
    na = len(active_indices)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((na, n + 1))
    scores = _scores_all(h)
    for i, idx in enumerate(active_indices):
        d = idx - (n - 1)
        grad = _score_grad_raw(h, d, n) * dx
        A_ub[i, :n] = grad
        A_ub[i, n] = -1.0
    b_ub = -scores[active_indices]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and float(np.max(np.abs(res.x[:n]))) > 1e-14:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, direction, target):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    if alpha_max < 1e-10:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_a = 0.0
    hi_a = alpha_max
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(25):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2
            f2 = f1
            c2 = c1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo_a = c1
            f1 = f2
            c1 = c2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    opt_a = (lo_a + hi_a) / 2.0
    opt_h = _project(h + opt_a * direction, target)
    opt_c5 = _exact_c5(opt_h)
    best_a = c1 if f1 < f2 else c2
    best_c5 = min(f1, f2)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=2000):
    n = len(h)
    target = n / 2.0
    cur_h = h.copy()

    for _ in range(max_iters):

            continue
        alpha, new_c5 = _line_search(cur_h, p, target)
        if new_c5 < cur_c5 - 1e-13:

            cur_c5 = _exact_c5(cur_h)
            if cur_c5 < best_c5:

            stop = 0

            order = np.argsort(-scores)
        else:

            if radius < 0.001 or stop > 5:

def _resample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(n, deadline, rng, n_iters=15000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.02
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _coordinate_polish(h, deadline, n_cycles=2):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(best_h)
    for _cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v = 0.0
            hi_v = 1.0
            for _ in range(30):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = best_h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = best_h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = best_h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()

            sub_dl = min(dl, time.time() + budget_s * 0.02)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=12000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(dl, time.time() + budget_s * 0.01)
                h_br, c5_br = _slsqp_optimize(h_b, sub_dl2, active_counts=[12, 24], max_iter=30, max_inner=3)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()

    # Phase 7: Perturbation + SLSQP restarts
    dl7 = min(deadline, t0 + budget_s * 0.86)
    scale = 0.02
    for _attempt in range(200):
        if time.time() > dl7:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.012)
        h_try, c5_try = _slsqp_optimize(h_pert, sub_dl, active_counts=[12, 24, 48], max_iter=30, max_inner=3)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 8: LP final polish
    if time.time() < deadline:
        dl8 = min(deadline, t0 + budget_s * 0.92)
        h8, c5_8 = _lp_optimize(best_h, dl8, radius_init=0.05, max_iters=3000)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: Coordinate polish
    if time.time() < deadline:
        dl9 = min(deadline, t0 + budget_s * 0.96)
        hc, c5c = _coordinate_polish(best_h, dl9, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Final SLSQP
    if time.time() < deadline:
        dl10 = min(t0 + budget_s * 0.985, time.time() + 20)
        h10, c5_10 = _slsqp_optimize(best_h, dl10, active_counts=[24, 48, 96], max_iter=40, max_inner=3)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
