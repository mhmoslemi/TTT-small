        weights[idx] = np.exp(-gap / (lambda_sm * dx + 1e-10))
    w_sum = float(np.sum(weights))

        w_sum = 1.0

    h_new = np.interp(x_new, x_old, h)

def _subgradient_descent(h, deadline, rng, temp_init=0.04, max_iter=800):

        s_max = float(np.max(scores))
        v = (scores - s_max) / (temp + 1e-15)
        exp_v = np.exp(v)
        w = exp_v / float(np.sum(exp_v))
        grad = np.zeros(n)
        for r in range(2 * n - 1):

                continue
            wd = w[r]

                    grad[:m] += wd * (-h[g:g+m])
                    grad[g:] += wd * (1.0 - h[:m])

            if temp < 1e-6:
                break
            continue
        grad = grad / gn
        mom = decay * mom + (1.0 - decay) * grad
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            break
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
            temp *= 0.993
        else:
            stall += 1
            lr *= 0.75
            temp *= 0.94
            if stall > 60:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kick_and_reopt(h, rng, scale, deadline):
    n = len(h)
    target = n / 2.0
    direction = rng.normal(0, 1, n)
    direction -= direction.mean()
    direction /= (np.linalg.norm(direction) + 1e-15)
    h_k = _project(h + scale * direction, target)
    h_k, c5_k = _lp_loop(h_k, deadline, radius_init=0.12, use_penalty=False)
    return h_k, c5_k

def _combined_phase(h, deadline, rng):
    """Alternate LP and gradient descent for robust local opt."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.18
    lr = 0.04
    mom = np.zeros(n)
    stall = 0
    mode = 0
    while time.time() < deadline:
        if stall > 50:
            mode = 1 - mode
            stall = 0
            radius = 0.18
            lr = 0.04
            mom = np.zeros(n)
        if mode == 0:
            scores = np.correlate(h, 1.0 - h, mode='full')
            G = _grad_matrix(h)
            p = _lp_solve(h, scores, G, radius)
            if p is not None and np.max(np.abs(p)) > 1e-13:
                alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
                if new_c5 < cur_c5 - 1e-12:
                    h = h + alpha * p
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.06, 0.45)
                    continue
            stall += 1
            radius *= 0.8
        else:
            scores = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
            s_max = float(np.max(scores))
            v = (scores - s_max) / 0.05
            exp_v = np.exp(v)
            w = exp_v / float(np.sum(exp_v))
            grad = np.zeros(n)
            for r in range(2 * n - 1):
                if w[r] < 1e-14:
                    continue
                wd = w[r]
                d = r - (n - 1)
                if d >= 0:
                    m = n - d
                    if m > 0:
                        grad[:m] += wd * (1.0 - h[d:d+m])
                        grad[d:] += wd * (-h[:m])
                else:
                    g = -d
                    m = n - g
                    if m > 0:
                        grad[:m] += wd * (-h[g:g+m])
                        grad[g:] += wd * (1.0 - h[:m])
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                stall += 1
                continue
            grad = grad / gn
            mom = 0.7 * mom + 0.3 * grad
            mn = np.linalg.norm(mom)
            if mn < 1e-14:
                stall += 1
                continue
            mom = mom / mn
            alpha, new_c5 = _line_search(h, lr * mom, 2.0/n, lr * 3.0)
            if new_c5 < cur_c5 - 1e-12:
                h = h + alpha * lr * mom
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.05, 0.35)
            else:
                stall += 1
                lr *= 0.8
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 4.0

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        n0 = 122
        h0 = 0.5 + 0.3 * np.sin(np.linspace(0, 3.0 * np.pi, n0))
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0.copy(), len(h0) / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick LP optimization (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.22, use_penalty=False)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair refinement (8%)
    d2 = t0 + budget_s * 0.20
    h2, c5_2 = _pair_refine(best_h, d2, max_passes=150, rng=rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Wide multi-resolution search (25%)
    d3 = t0 + budget_s * 0.45
    n_best = len(best_h)
    n_range = []
    for dn in range(-50, 51, 2):
        n_try = n_best + dn
        if 64 <= n_try <= 220 and n_try % 2 == 0:
            n_range.append(n_try)
    # Sort by distance from current best
    n_range.sort(key=lambda x: (abs(x - n_best), x))
    for n_try in n_range:
        if time.time() > d3:
            break
        if n_try == n_best:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _combined_phase(h_r, sub_dl, rng)
        if c5_r < best_c5 - 1e-7 and time.time() < d3 - 10:
            sub_dl2 = min(d3, time.time() + budget_s * 0.03)
            h_r, c5_r2 = _pair_refine(h_r, sub_dl2, max_passes=80, rng=rng)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r.copy()
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: LP with penalty on best (12%)
    d4 = t0 + budget_s * 0.57
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.20, use_penalty=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Triple refinement (8%)
    d5 = t0 + budget_s * 0.65
    h5, c5_5 = _triple_refine(best_h, d5, rng, max_trials=10000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Subgradient descent (10%)
    d6 = t0 + budget_s * 0.75
    h6, c5_6 = _subgradient_descent(best_h, d6, rng, temp_init=0.03)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Kick and reoptimize (15%)
    d7 = t0 + budget_s * 0.90
    kick_count = 0
    while time.time() < d7:
        kick_count += 1
        frac = kick_count / (kick_count + 10)
        scale = 0.06 + 0.06 * (1.0 - frac)
        sub_dl = min(d7, time.time() + budget_s * 0.035)
        h_k, c5_k = _kick_and_reopt(best_h, rng, scale, sub_dl)
        if c5_k < best_c5 - 1e-12:
            best_c5 = c5_k
            best_h = h_k.copy()

    # Phase 8: Final pair + LP polish (10%)
    d8 = deadline
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=200, rng=rng)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline - 5:
        h9, c5_9 = _lp_loop(best_h, deadline, radius_init=0.06, use_penalty=True)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
