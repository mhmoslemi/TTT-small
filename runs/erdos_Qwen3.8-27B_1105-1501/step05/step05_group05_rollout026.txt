def smooth_grad(h, dx, beta):

        if len(active) > 60:
            active = np.argsort(scores)[-60:]
        result = slsqp_step(h, active, dx, s_max + 1e-6, radius, maxiter=30)
        if result is not None:
            new_h, new_c5 = result
            h = project_feasible(new_h, target)
            actual = exact_c5(h)
            if actual < best_c5 - 1e-12:
                best_c5 = actual
                best_h = h.copy()
                stall = 0
                consecutive_fail = 0
                radius = min(radius * 1.05, 0.40)
            else:
                stall += 1
                radius *= 0.8
                if stall >= 4:
                    h = best_h.copy()
                    radius = max(radius, 0.08)
                    stall = 0
        else:
            consecutive_fail += 1
            radius *= 0.75
            if consecutive_fail >= 8:
                break
            if radius < 0.02:
                break
    return best_h, best_c5


def run_gradient_phase(h, deadline, beta_list=(40, 100, 250, 500)):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.02
    mom = np.zeros(n)
    decay = 0.8
    stall = 0
    bi = 0

    while time.time() < deadline and stall < 50:
        beta = beta_list[bi % len(beta_list)]
        d = smooth_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            bi += 1
            stall += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            bi += 1
            continue
        mom = mom / mn
        res = try_direction(h, lr * mom, dx, cur_c5, target)
        if res is not None:
            h, new_c5 = res
            if new_c5 < best_c5 - 1e-13:
                best_c5 = new_c5
                best_h = h.copy()
            cur_c5 = new_c5
            stall = 0
            lr = min(lr * 1.05, 0.30)
            if stall == 0 and (bi % 15 == 0):
                bi += 1
            bi += 1
        else:
            stall += 1
            lr *= 0.8
            if stall % 10 == 0:
                bi += 1
    return best_h, best_c5


def run_multi_res(h_best, c5_best, deadline, rng):
    n_best = len(h_best)
    best_h = h_best.copy()
    best_c5 = c5_best
    tried = set()
    offsets = []
    for k in range(-15, 16):
        if k != 0:
            offsets.append(k)
    offsets.sort(key=lambda x: (abs(x), x))
    try_n_list = []
    for mult in [1.3, 1.5, 1.7, 2.0, 0.7, 0.8, 1.1]:
        n_try = int(round(n_best * mult))
        if n_try >= 50 and n_try <= 350 and n_try not in tried:
            try_n_list.append(n_try)
    for n_try in try_n_list:
        if time.time() > deadline:
            break
        tried.add(n_try)
        h_r = resample_h(h_best, n_try)
        c5_r = exact_c5(h_r)
        sub_dl = min(deadline, time.time() + 25)
        h_r, c5_r2 = run_slsqp_phase(h_r, sub_dl, max_iters=30)
        h_r, c5_r3 = run_gradient_phase(h_r, min(deadline, time.time() + 10), beta_list=(60, 150))
        c5_final = min(c5_r2, c5_r3)
        if c5_final < best_c5 - 1e-12:
            if c5_r2 < c5_r3:
                best_h = h_r.copy() if c5_r2 == c5_final else best_h
            best_c5 = c5_final
            h_best = best_h.copy()
    return best_h, best_c5


def run_kick(h_best, c5_best, deadline, rng, n_kicks=15):
    n = len(h_best)
    target = n / 2.0
    best_h = h_best.copy()
    best_c5 = c5_best
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.02, 0.06)
        d = rng.normal(0, 1, n)
        d -= d.mean()
        d /= (np.linalg.norm(d) + 1e-15)
        h_k = project_feasible(best_h + scale * d, target)
        sub_dl = min(deadline, time.time() + 8)
        h_t, c5_t = run_slsqp_phase(h_k, sub_dl, max_iters=20)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_dl = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = project_feasible(h, n / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)

    # Phase 1: SLSQP epigraph (largest fraction)
    d1 = t0 + budget_s * 0.42
    h1, c5_1 = run_slsqp_phase(h, d1, max_iters=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient with varying beta
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = run_gradient_phase(best_h, d2, beta_list=(40, 100, 250, 500, 800))
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Another SLSQP round (fresher active set)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = run_slsqp_phase(best_h, d3, max_iters=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = run_multi_res(best_h, best_c5, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Kick + reoptimize
    d5 = t0 + budget_s * 0.90
    h5, c5_5 = run_kick(best_h, best_c5, d5, rng, n_kicks=12)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final SLSQP polish
    d6 = t0 + budget_s * 0.97
    h6, c5_6 = run_slsqp_phase(best_h, d6, max_iters=80)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 >= best_c5 - 1e-14:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
