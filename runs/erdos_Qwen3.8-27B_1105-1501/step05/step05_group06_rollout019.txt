def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -10.0, 10.0)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    x = np.clip(v - mid, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    onemh = 1.0 - h
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * onemh[:m]
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * onemh[s:]
        G[r, s:] += -dx * h[:m]
    return G

def _lp_direction(h, G, scores, top_k, radius):
    n = len(h)
    n_lags = 2 * n - 1
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-radius, radius)] * n
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, n_eval=35):

    for j in range(len(p)):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, 1.0 / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, 1.0 / (-p[j]))
    alpha_max = min(alpha_max, 1.0)
    if alpha_max < 1e-12:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(n_eval):

            c2 = lo + gr * (hi - lo)

    f = _exact_c5(h + alpha * p)

    best_h = pop[best_idx].copy()
    best_c5 = fits[best_idx]
    max_gens = int(100 + (deadline - time.time()) / max(0.01, budget_s * 0.01))
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            while True:
                r1 = int(rng.integers(pop_size))
                r2 = int(rng.integers(pop_size))
                if r1 != i and r2 != i and r1 != r2:
                    break
            v = pop[best_idx] + F * (pop[r1] - pop[r2])
            v = _project(v, target)
            u = pop[i].copy()
            jrand = int(rng.integers(n))
            for j in range(n):
                if rng.random() < CR or j == jrand:
                    u[j] = v[j]
            f_u = _exact_c5(u)
            if f_u <= fits[i]:
                pop[i] = u
                fits[i] = f_u
                if f_u < best_c5:
                    best_c5 = f_u
                    best_h = u.copy()
                    best_idx = i
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Intensive LP at original resolution
    dl1 = t0 + budget_s * 0.14
    h1, c5_1 = _local_lp_opt(h_start, dl1, top_k_list=[2, 3, 4, 5, None],
                             radius_init=0.12, max_iters=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient + LP at original resolution
    if time.time() < t0 + budget_s * 0.22:
        dl2 = t0 + budget_s * 0.24
        h2, c5_2 = _subgrad_opt(best_h, dl2, n_steps=200)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        if time.time() < dl2:
            h2b, c5_2b = _local_lp_opt(best_h, dl2, top_k_list=[3, 4, 5],
                                       radius_init=0.06, max_iters=3000)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Multi-resolution search
    dl3 = t0 + budget_s * 0.52
    resolutions = [96, 64, 128, 160, 128, 80]
    alphas_list = [0.6180339887, 0.4142135624, 0.7071067812,
                   0.3819660113, 0.5, 0.5556]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if best_c5 < 0.3805:
            break
        for s in range(2):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                h_init = _quasi_init(n_try, alphas_list[ri % len(alphas_list)], rng)
            sub_dl = min(dl3, time.time() + budget_s * 0.04)
            h_try, c5_try = _local_lp_opt(h_init, sub_dl,
                                          top_k_list=[2, 3, 5, 8],
                                          radius_init=0.1, max_iters=3000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                if time.time() < dl3:
                    sub_dl2 = min(dl3, time.time() + budget_s * 0.015)
                    h_de, c5_de = _de_evolve(h_try, sub_dl2, pop_size=18)
                    if c5_de < best_c5:
                        best_c5 = c5_de
                        best_h = h_de.copy()
                        best_n = n_try
                        sub_dl3 = min(dl3, time.time() + budget_s * 0.01)
                        h_pol, c5_pol = _local_lp_opt(h_de, sub_dl3,
                                                       top_k_list=[3, 4],
                                                       radius_init=0.04, max_iters=1500)
                        if c5_pol < best_c5:
                            best_c5 = c5_pol
                            best_h = h_pol.copy()

    # Phase 4: Random pair descent on best
    if time.time() < t0 + budget_s * 0.70:
        dl4 = t0 + budget_s * 0.72
        best_h, best_c5 = _random_pair_descent(best_h, dl4, max_passes=4000, n_pairs=500)
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.018)
            h_lp, c5_lp = _local_lp_opt(best_h, sub_dl,
                                        top_k_list=[2, 3, 4],
                                        radius_init=0.02, max_iters=2500)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 5: Perturbation restarts
    if time.time() < t0 + budget_s * 0.84:
        dl5 = t0 + budget_s * 0.86
        scale = 0.025
        for attempt in range(400):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.005)
            h_try, c5_try = _local_lp_opt(h_pert, sub_dl,
                                           top_k_list=[3, 4, 5],
                                           radius_init=0.03, max_iters=600)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.004, scale * 0.82)
            else:
                scale = min(0.07, scale * 1.06)

    # Phase 6: Final intensive polish
    if time.time() < hard_deadline:
        best_h, best_c5 = _random_pair_descent(best_h, hard_deadline,
                                                max_passes=5000, n_pairs=400)
        if time.time() < hard_deadline - 3:
            h_fin, c5_fin = _local_lp_opt(best_h, hard_deadline,
                                           top_k_list=[2, 3],
                                           radius_init=0.005, max_iters=2000)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
