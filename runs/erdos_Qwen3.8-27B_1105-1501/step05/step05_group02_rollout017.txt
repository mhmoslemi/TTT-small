def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _score_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_score(d, h):
    """Gradient of score at lag d (np.correlate index offset) w.r.t. h, times dx."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif d > 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m]) * dx
            grad[:m] += (-h[d:]) * dx
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:]) * dx
            grad[g:] += (-h[:m]) * dx
    return grad

def _lp_step(h, radius):
    """One LP step: find direction that decreases C5."""
    n = len(h)
    scores = _score_vector(h)
    # Build gradient matrix
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        G[r] = _grad_score(d, h)
    
    # LP: min t subject to G@delta + scores <= t, sum(delta)=0, |delta|<=radius
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _slsqp_polish(h, deadline, topk=40):
    """SLSQP minimax polish."""
    n = len(h)
    target = n / 2.0
    
    # Get current scores and select top-k lags
    scores = _score_vector(h)
    c_max = float(np.max(scores))
    # Select lags to constrain
    idx = np.argsort(scores)[-topk:]
    
    # Define the optimization problem
    def objective(x):
        return x[-1]  # minimize t
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    constraints = []
    # Equality: sum(h) = n/2
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - target,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    # Inequality: t - C_d(h) >= 0 for each selected lag
    for lag_idx in idx:
        d = lag_idx - (n - 1)
        def constr(x, d=d):
            # Compute C_d at x[:n]
            h_x = x[:n]
            if d == 0:
                return x[-1] - np.sum(h_x * (1.0 - h_x)) * (2.0/n)
            elif d > 0:
                m = n - d
                return x[-1] - np.sum(h_x[d:] * (1.0 - h_x[:m])) * (2.0/n)
            else:
                g = -d
                m = n - g
                return x[-1] - np.sum(h_x[:m] * (1.0 - h_x[g:])) * (2.0/n)
        
        def constr_jac(x, d=d):
            gx = np.zeros(n + 1)
            gx[-1] = 1.0  # d(x[-1])/dx[-1]
            h_x = x[:n]
            gx[:n] = -_grad_score(d, h_x)  # negative because constraint is t - C_d >= 0
            return gx
        
        constraints.append({'type': 'ineq', 'fun': constr, 'jac': constr_jac})
    
    # Initial point
    x0 = np.concatenate([h.copy(), [c_max + 0.01]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    best_result = None
    best_c5 = _exact_c5(h)
    
    while time.time() < deadline:
        try:
            res = minimize(objective, x0, method='SLSQP',
                         jac=objective_grad,
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    h = h_new.copy()
                    # Update initial point for next iteration
                    x0 = np.concatenate([h_new.copy(), [res.x[-1]]])
                    # Re-select constraints
                    scores = _score_vector(h)
                    c_max = float(np.max(scores))
                    idx = np.argsort(scores)[-topk:]
                    break  # Rebuild constraints with new active set
                else:
                    break
            else:
                break
        except Exception:
            break
    
    return h, best_c5
