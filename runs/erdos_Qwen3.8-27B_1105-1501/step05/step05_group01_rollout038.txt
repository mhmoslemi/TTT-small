                alpha, c5_try = _line_search_quadratic(h, d, dx, 0.15)
                if c5_try < cur_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
    return best_h, best_c5

    # Phase 1: Annealed soft-max gradient on initial (12%)
    d1 = t0 + budget_s * 0.12
