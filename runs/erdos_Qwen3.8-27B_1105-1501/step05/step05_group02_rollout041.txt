def _weighted_grad_direction(h, deadline, gamma):
    """Move along negative weighted gradient direction."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    scores = _scores_array(h)
    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    # Compute weighted gradient
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        k = n - 1 + d
        wk = w[k]
        if wk < 1e-15:
            continue
        if d >= 0:
            m = n - d
            grad[:m] += wk * dx * (1.0 - h[d:])
            grad[d:] += wk * dx * (-h[:m])
        else:
            g = -d
            m = n - g
            grad[:m] += wk * dx * (1.0 - h[g:])
            grad[g:] += wk * dx * (-h[:m])
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, _exact_c5(h)
    grad /= gn
    # Line search
    cur_c5 = _exact_c5(h)
    best_a = 0.0
    best_c5 = cur_c5
    alpha = 0.15
    for _ in range(20):
        h_new = _project(h - alpha * grad, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_a = alpha
        alpha *= 0.7
        if alpha < 1e-10:
            break
    if best_a > 0:
        h_new = _project(h - best_a * grad, target)
        return h_new, _exact_c5(h_new)
    return h, cur_c5

def _resample(h, n_new):

The key issue: I derived the coefficients assuming h[p] += t and h[q] -= t (i.e., mass flows from q to p). The feasible range for t is:
- t_min such that h[p] + t >= 0 and h[q] - t <= 1: t >= -h[p] and t >= h[q] - 1
- t_max such that h[p] + t <= 1 and h[q] - t >= 0: t <= 1 - h[p] and t <= h[q]

So t_min = max(-h[p], h[q]-1) and t_max = min(1-h[p], h[q]).

This is always valid as long as h[p] + t ∈ [0,1] and h[q] - t ∈ [0,1], which is guaranteed for t ∈ [t_min, t_max].

The issue is that I was trying to handle both directions (p to q and q to p) with a single parameterization, which got confusing. Let me just always use the convention: "transfer t from q to p" means h[p] += t, h[q] -= t, and the feasible range is as above.
