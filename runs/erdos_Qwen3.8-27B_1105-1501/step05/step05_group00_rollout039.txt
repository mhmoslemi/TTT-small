import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - float(np.sum(x))
    if abs(r) > 1e-12:
        mask = (x < 1.0) if r > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += r / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _softmax_weights(corr, dx, tau):
    s = corr * dx
    s_shift = s - s.max()
    w = np.exp(tau * s_shift)
    w /= w.sum()
    return w

def _grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    w0 = weights[n - 1]
    grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[n - 1 + d]
        if wp > 1e-12:
            grad[:n - d] += wp * (1.0 - h[d:])
            grad[d:] -= wp * h[:n - d]
        wn = weights[n - 1 - d]
        if wn > 1e-12:
            grad[d:] += wn * (1.0 - h[:n - d])
            grad[:n - d] -= wn * h[d:]
    return grad

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_min = 0.0

    grid = np.linspace(a_min, a_max, 40)
    vals = np.array([float(np.max(S0 + t * T - t * t * Q)) * dx for t in grid])
    best_i = int(np.argmin(vals))
    best_alpha, best_val = grid[best_i], vals[best_i]
    if 0 < best_i < len(grid) - 1:
        lo, hi = grid[best_i - 1], grid[best_i + 1]
    else:
        lo, hi = a_min, a_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    dd = lo + gr * (hi - lo)
    fc = float(np.max(S0 + c * T - c * c * Q)) * dx
    fd = float(np.max(S0 + dd * T - dd * dd * Q)) * dx
    for _ in range(50):
        if hi - lo < 1e-13:
            break
        if fc < fd:

            c = hi - gr * (hi - lo)

        else:

    opt = (lo + hi) / 2.0

    v0 = float(np.max(S0)) * dx

def _grad_desc(h, deadline, tau=30.0, max_iter=20000, eta_init=0.03):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    eta = eta_init
    momentum = np.zeros(n)
    beta = 0.5
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = _corrs(h)
        weights = _softmax_weights(corr, 2.0 / n, tau)
        grad = _grad_weighted(h, weights)
        grad -= grad.mean()
        at_lo = h < 1e-10
        at_hi = h > 1.0 - 1e-10
        grad[at_lo & (grad < 0)] = 0.0
        grad[at_hi & (grad > 0)] = 0.0
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        d = -grad / gnorm
        momentum = beta * momentum + (1 - beta) * d
        momentum -= momentum.mean()
        d_search = momentum
        mn = np.linalg.norm(d_search)
        if mn < 1e-14:
            d_search = d
        else:
            d_search = d_search / mn
        alpha, new_c5 = _line_search(h, d_search, eta)
        if alpha > 1e-14 and new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d_search
            h = _project(h, target)
            actual = _exact_c5(h)
            if actual < cur_c5 - 1e-15:
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                eta = min(eta * 1.05, 0.15)

                    tau *= 1.3
                    beta = min(beta + 0.02, 0.8)
            else:
                h = h + 0.3 * alpha * d_search
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                eta *= 0.8
                stall += 1
        else:
            eta *= 0.75
            stall += 1

                tau *= 0.8
                beta = max(beta - 0.05, 0.3)
                eta = max(eta * 1.3, 0.01)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=30):

def _binary_sa(h_init, deadline, rng):

    sorted_idx = np.argsort(-h_init)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
