import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_samples = 50
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = np.array([np.max(S0 + t * T - t * t * C) * dx for t in ts])
    min_idx = int(np.argmin(vals))
    if min_idx == 0:
        br_lo, br_hi = t_lo, ts[2]
    elif min_idx == n_samples - 1:
        br_lo, br_hi = ts[-3], t_hi
    else:
        br_lo = ts[max(0, min_idx - 1)]
        br_hi = ts[min(n_samples - 2, min_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = br_hi - gr * (br_hi - br_lo)
    c2 = br_lo + gr * (br_hi - br_lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if br_hi - br_lo < 1e-12:
            break
        if f1 < f2:
            br_hi = c2
            c2 = c1
            f2 = f1
            c1 = br_hi - gr * (br_hi - br_lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            br_lo = c1
            c1 = c2
            f1 = f2
            c2 = br_lo + gr * (br_hi - br_lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (br_lo + br_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.65
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.70
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def _pair_swap_search(h, deadline, max_passes=6, block_sizes=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    if block_sizes is None:
        block_sizes = [1, 2, 3, 4, 8]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for blk_len in block_sizes:
            for i in range(0, n - blk_len + 1, max(1, blk_len // 2)):
                if time.time() > deadline:
                    break
                for j in range(i + blk_len, n - blk_len + 1, max(1, blk_len // 2)):
                    if np.all(h[i:i+blk_len] == h[j:j+blk_len]):
                        continue
                    d = np.zeros(n)
                    d[i:i+blk_len] = h[j:j+blk_len] - h[i:i+blk_len]
                    d[j:j+blk_len] = h[i:i+blk_len] - h[j:j+blk_len]
                    if np.max(np.abs(d)) < 1e-14:
                        continue
                    alpha, c5_try = _line_search(h, d, dx, 0.4)
                    if c5_try < cur_c5 - 1e-14:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h, deadline, n_restarts=4, max_steps=5000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    for _restart in range(n_restarts):
        if time.time() > deadline:
            break
        cur = (h > 0.5).astype(float)
        cur_c5 = _exact_c5(cur)
        T = 0.03
        cooled = False
        for _step in range(max_steps):
            if time.time() > deadline:
                break
            if cooled:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or cur[i] == cur[j]:
                continue
            d = np.zeros(n)
            d[i] = cur[j] - cur[i]
            d[j] = cur[i] - cur[j]
            alpha, c5_try = _line_search(cur, d, dx, 0.49)
            if c5_try < cur_c5:
                cur[i], cur[j] = cur[j], cur[i]
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                T = max(0.001, T * 0.995)
            elif rng.random() < np.exp(-max(0, c5_try - cur_c5) / max(T, 1e-10)):
                cur[i], cur[j] = cur[j], cur[i]
                cur_c5 = c5_try
                T = max(0.001, T * 0.998)
            else:
                T = max(0.0005, T * 0.999)
            if T < 0.001:
                cooled = True
    return best_h, best_c5

def _block_sa(h, deadline, n_restarts=3, max_steps=3000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(4444)
    max_blk = min(8, n // 8)
    for _restart in range(n_restarts):
        if time.time() > deadline:
            break
        cur = (h > 0.5).astype(float)
        cur_c5 = _exact_c5(cur)
        T = 0.04
        for _step in range(max_steps):
            if time.time() > deadline:
                break
            blk_len = int(rng.integers(1, max_blk + 1))
            if n - 2 * blk_len < 1:
                continue
            i = int(rng.integers(0, n - 2 * blk_len + 1))
            j = i + blk_len + int(rng.integers(0, n - 2 * blk_len - i))
            if j + blk_len > n:
                j = n - blk_len
                i = int(rng.integers(0, n - 2 * blk_len + 1))
                if i + blk_len > j:
                    continue
            if np.array_equal(cur[i:i+blk_len], cur[j:j+blk_len]):
                continue
            d = np.zeros(n)
            d[i:i+blk_len] = cur[j:j+blk_len] - cur[i:i+blk_len]
            d[j:j+blk_len] = cur[i:i+blk_len] - cur[j:j+blk_len]
            if np.max(np.abs(d)) < 1e-14:
                continue
            alpha, c5_try = _line_search(cur, d, dx, 0.9)
            delta = c5_try - cur_c5
            if delta < 0 or rng.random() < np.exp(-max(0, delta) / max(T, 1e-10)):
                cur[i:i+blk_len], cur[j:j+blk_len] = cur[j:j+blk_len].copy(), cur[i:i+blk_len].copy()
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
            T = max(0.001, T * 0.9985)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):

    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _structured_binary(n, rng, pattern):
    h = np.zeros(n)
    if pattern == 0:
        for i in range(n):
            if i % 2 == 0:
                h[i] = 1.0
        if np.sum(h) > n / 2:
            excess = int(np.sum(h)) - n // 2
            ones = np.where(h == 1)[0]
            removals = rng.choice(ones, size=min(excess, len(ones)), replace=False)
            h[removals] = 0
    elif pattern == 1:
        block_size = max(2, n // 16)
        for i in range(0, n, block_size * 2):
            h[i:i + block_size] = 1.0
        s = np.sum(h)
        if s > n / 2:
            ones = np.where(h == 1)[0]
            removals = rng.choice(ones, size=int(s) - n // 2, replace=False)
            h[removals] = 0
        elif s < n / 2:
            zeros = np.where(h == 0)[0]
            additions = rng.choice(zeros, size=n // 2 - int(s), replace=False)
            h[additions] = 1
    elif pattern == 2:
        m = n // 2
        for i in range(m):
            h[i] = 1.0
            h[i + m] = 0.0
    elif pattern == 3:
        for i in range(n):
            h[i] = 1.0 if ((i * 7) % n) < n // 2 else 0.0
        if int(np.sum(h)) != n // 2:
            s = int(np.sum(h))
            if s > n // 2:
                ones = np.where(h == 1)[0]
                removals = rng.choice(ones, size=s - n // 2, replace=False)
                h[removals] = 0
            else:
                zeros = np.where(h == 0)[0]
                additions = rng.choice(zeros, size=n // 2 - s, replace=False)
                h[additions] = 1
    else:
        h = _random_binary(n, rng)
    return h

def _refine_candidate(h, deadline, lp_time_frac=0.5, sa_time_frac=0.3):
    n = len(h)
    target = n / 2.0
    h = _project(h, target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t_lp_end = time.time() + lp_time_frac * (deadline - time.time())
    t_sa_end = time.time() + (lp_time_frac + sa_time_frac) * (deadline - time.time())
    if time.time() < t_lp_end:
        h_lp, c5_lp = _lp_loop(h, t_lp_end, radius_init=0.15)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    if time.time() < t_sa_end:
        h_sa, c5_sa = _binary_sa(best_h, t_sa_end, n_restarts=2, max_steps=3000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        h_blk, c5_blk = _block_sa(best_h, deadline, n_restarts=1, max_steps=1500)
        if c5_blk < best_c5:
            best_c5 = c5_blk
            best_h = h_blk
    if time.time() < deadline:
        h_lp2, c5_lp2 = _lp_loop(best_h, deadline, radius_init=0.08)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick LP on parent (5%)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _lp_loop(h_init, d1, radius_init=0.10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Diverse initializations with SA+LP (30%)
    d2 = t0 + budget_s * 0.35
    n_candidates = 16
    per_cand = (d2 - time.time()) / max(n_candidates, 1) * 0.85
    for trial in range(n_candidates):
        if time.time() > d2 - 2:
            break
        r = rng.random()
        if r < 0.4:
            h_cand = _random_binary(n0, rng)
        elif r < 0.6:
            h_cand = _structured_binary(n0, rng, int(rng.integers(0, 5)))
        elif r < 0.8:
            h_cand = _resample(best_h, n0) + rng.normal(0, 0.05, n0)
            h_cand = _project(h_cand, n0 / 2.0)
            h_cand = (h_cand > rng.random(n0)).astype(float)
            s = int(np.sum(h_cand))
            if s > n0 // 2:
                ones = np.where(h_cand == 1)[0]
                h_cand[rng.choice(ones, s - n0 // 2, replace=False)] = 0
            elif s < n0 // 2:
                zeros = np.where(h_cand == 0)[0]
                h_cand[rng.choice(zeros, n0 // 2 - s, replace=False)] = 1
        else:
            h_cand = (h_init + rng.normal(0, 0.3, n0) > 0.5).astype(float)
            s = int(np.sum(h_cand))
            if s > n0 // 2:
                ones = np.where(h_cand == 1)[0]
                h_cand[rng.choice(ones, s - n0 // 2, replace=False)] = 0
            elif s < n0 // 2:
                zeros = np.where(h_cand == 0)[0]
                h_cand[rng.choice(zeros, n0 // 2 - s, replace=False)] = 1
        sub_dl = min(d2 - 2, time.time() + per_cand)
        h_r, c5_r = _refine_candidate(h_cand, sub_dl, 0.45, 0.4)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Multi-resolution (25%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    offset_list = [-8, 8, -16, 16, -24, 24, -32, 32, -48, 48, -4, 4, -12, 12, -20, 20, -40, 40, -56, 56, -64, 64, -80, 80]
    tried_ns = set([n_curr])
    for dn in offset_list:
        if time.time() > d3 - 3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 300 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3 - 3, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        rem = d3 - 3 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.005:
            sub_dl2 = min(d3 - 3, time.time() + max(3, rem * 0.4))
            h_r, c5_r = _binary_sa(h_r, sub_dl2, n_restarts=1, max_steps=2000)
            if time.time() < d3 - 3:
                sub_dl3 = min(d3 - 3, time.time() + max(2, (d3 - 3 - time.time()) * 0.5))
                h_r, c5_r = _lp_loop(h_r, sub_dl3, radius_init=0.08)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization on best (20%)
    d4 = t0 + budget_s * 0.80
    h4 = best_h.copy()
    n4 = len(h4)
    # LP refinement
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.06)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Block swap local search
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4 = best_h.copy()
    h4, c5_4 = _pair_swap_search(h4, sub_dl, max_passes=4, block_sizes=[1, 2, 3])
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Binary SA on current best
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4 = best_h.copy()
    h4, c5_4 = _binary_sa(h4, sub_dl, n_restarts=2, max_steps=4000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Block SA
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4 = best_h.copy()
    h4, c5_4 = _block_sa(h4, sub_dl, n_restarts=2, max_steps=2000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # LP final
    sub_dl = min(d4, time.time() + budget_s * 0.03)
    h4 = best_h.copy()
    h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.04)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Perturbation + re-optimize (12%)
    d5 = t0 + budget_s * 0.92
    scale = 0.06
    for attempt in range(300):
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        r = rng.random()
        if r < 0.3:
            h_pert = (h_pert > rng.random(len(best_h))).astype(float)
            s = int(np.sum(h_pert))
            if s > len(best_h) // 2:
                ones = np.where(h_pert == 1)[0]
                h_pert[rng.choice(ones, s - len(best_h)//2, replace=False)] = 0
            elif s < len(best_h) // 2:
                zeros = np.where(h_pert == 0)[0]
                h_pert[rng.choice(zeros, len(best_h)//2 - s, replace=False)] = 1
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 6: Final polish (8%)
    d6 = deadline
    h6 = best_h.copy()
    h6, c5_6 = _lp_loop(h6, d6, radius_init=0.03)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    h6 = best_h.copy()
    h6, c5_6 = _pair_swap_search(h6, d6, max_passes=3, block_sizes=[1, 2])
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6 - 1:
        h6 = best_h.copy()
        h6, c5_6 = _lp_loop(h6, d6, radius_init=0.02)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
