import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_obj(x, tau, n):
    dx = 2.0 / n
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = scores.max()
    log_sum = s_max / tau + np.log(np.exp((scores - s_max) / tau).sum())
    return dx * tau * log_sum

def _soft_jac(x, tau, n):
    dx = 2.0 / n
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = scores.max()
    exps = np.exp((scores - s_max) / tau)
    w = exps / exps.sum()
    G = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-8:
            continue
        d = r - (n - 1)
        ww = w[r]
        if d >= 0:
            m = n - d
            if m > 0:
                G[d:] += ww * (1.0 - x[:m])
                G[:m] -= ww * x[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[:m] += ww * (1.0 - x[g:])
                G[g:] -= ww * x[:m]
    return dx * G

def _line_search_dense(h, d, dx, radius, n_samples=50):
    """Dense line search along direction d with step size bounded by radius."""
    n = len(h)
    target = n / 2.0
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Dense sampling
    ts = np.linspace(t_lo, t_hi, n_samples)
    vals = np.array([float(np.max(S0 + t * T - t*t * Q)) * dx for t in ts])
    
    # Find local minima
    candidates = [0, len(ts) - 1]
    for i in range(1, len(ts) - 1):
        if vals[i] < vals[i-1] and vals[i] < vals[i+1]:
            candidates.append(i)
    
    # Also add the global min
    candidates.append(int(np.argmin(vals)))
    
    best_t = 0.0
    best_val = float('inf')
    
    # Check global min and neighbors
    for ci in set(candidates):
        if ci < len(ts) - 1:
            bracket = (ts[max(0, ci-1)], ts[min(len(ts)-1, ci+1)])
        else:
            bracket = (ts[max(0, ci-1)], ts[ci])
        # Golden section within bracket
        a, b = bracket
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
        for _ in range(30):
            if b - a < 1e-12:
                break
            if f1 < f2:
                b = c2; c2 = c1; f2 = f1
                c1 = b - gr * (b - a)
                f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
            else:
                a = c1; c1 = c2; f1 = f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
        opt_t = (a + b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * Q)) * dx
        if opt_val < best_val:
            best_val = opt_val
            best_t = opt_t
    
    # Also check the endpoints
    for t_check in [t_lo, t_hi]:
        v_check = float(np.max(S0 + t_check * T - t_check*t_check * Q)) * dx
        if v_check < best_val:
            best_val = v_check
            best_t = t_check
    
    return best_t, best_val

def _slsqp_solve(h, tau, maxiter=60):
    """Run SLSQP to minimize soft-max objective."""
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (target,)}]
    
    try:
        res = minimize(_soft_obj, h, jac=_soft_jac, args=(tau, n),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            x = res.x
            x = _project(x, target)
            return x
    except Exception:
        pass
    return h

def _eq_fun(x, target):
    return float(np.sum(x) - target)

def _eq_jac(x, target):
    n = len(x)
    return np.ones(n)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_dir(n, mode, phase=0.0):
    """Generate a zero-mean Fourier direction."""
    x = np.arange(n) / n
    d = np.sin(2 * np.pi * mode * x + phase)
    d -= d.mean()  # ensure zero mean
    norm = np.max(np.abs(d))
    if norm < 1e-10:
        return np.zeros(n)
    return d / norm

def _random_binary(n, rng):

    h = np.zeros(n)
    h[idx] = 1.0
    return h

        else:

def _soft_obj(x, tau, n):
    dx = 2.0 / n
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = scores.max()
    log_sum_exps = np.log(np.exp((scores - s_max) / tau).sum())
    return dx * tau * (s_max / tau + log_sum_exps)


def _soft_jac(x, tau, n):
    dx = 2.0 / n
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = scores.max()
    exps = np.exp((scores - s_max) / tau)

    G = np.zeros(n)

        d = r - (n - 1)

                G[:m] -= ww * x[d:]

            m = n - g

            alpha, c5_try = _line_search_dense(h_ls, d_dir, dx, 0.15)
            if c5_try < c5_ls - 1e-13:
                h_ls = h_ls + alpha * d_dir
                h_ls = _project(h_ls, n_ls / 2.0)
                c5_ls = _exact_c5(h_ls)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    # Random directions
    stall = 0
    radius_rand = 0.2
    for _ in range(5000):
        if time.time() > d3:
            break
        d_rand = rng.normal(0, 1, n_ls)
        d_rand -= d_rand.mean()
        nm = np.max(np.abs(d_rand))
        if nm < 1e-10:
            continue
        d_rand /= nm
        alpha, c5_try = _line_search_dense(h_ls, d_rand, dx, radius_rand)
        if c5_try < c5_ls - 1e-13:
            h_ls = h_ls + alpha * d_rand
            h_ls = _project(h_ls, n_ls / 2.0)
            c5_ls = _exact_c5(h_ls)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
            stall = 0
            radius_rand = min(radius_rand * 1.03, 0.4)
        else:
            stall += 1
            if stall % 100 == 0:
                radius_rand = max(0.05, radius_rand * 0.85)
            if stall > 400:
                break

    # Phase 4: SLSQP fine-tuning (12%)
    d4 = t0 + budget_s * 0.85
    h_ref = best_h.copy()
    for tau_r in [0.004, 0.0025, 0.0015]:
        if time.time() > d4:
            break
        h_ref = _slsqp_solve(h_ref, tau_r, maxiter=70)
        c5_r = _exact_c5(h_ref)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_ref.copy()

    # Phase 5: Pair swaps on best (7%)
    d5 = t0 + budget_s * 0.93
    h_ps = best_h.copy()
    c5_ps = _exact_c5(h_ps)
    n_ps = len(h_ps)
    dx_ps = 2.0 / n_ps
    stall_ps = 0
    radius_ps = 0.4
    for _ in range(10000):
        if time.time() > d5:
            break
        i = int(rng.integers(0, n_ps))
        j = int(rng.integers(0, n_ps))
        if i == j:
            continue
        d_pair = np.zeros(n_ps)
        d_pair[i] = 1.0
        d_pair[j] = -1.0
        alpha, c5_try = _line_search_dense(h_ps, d_pair, dx_ps, radius_ps, n_samples=25)
        if c5_try < c5_ps - 1e-13:
            h_ps = h_ps + alpha * d_pair
            h_ps = _project(h_ps, n_ps / 2.0)
            c5_ps = _exact_c5(h_ps)
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()
            stall_ps = 0
            radius_ps = min(radius_ps * 1.02, 0.6)
        else:
            stall_ps += 1
            if stall_ps % 200 == 0:
                radius_ps = max(0.1, radius_ps * 0.85)
            if stall_ps > 2000:
                break

    # Phase 6: Multi-resolution + final polish (10%)
    d6 = deadline
    n_curr = len(best_h)
    offsets = [4, -4, 8, -8, 12, -12, 16, -16]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 300:
            continue
        h_m = _resample(best_h, n_try)
        h_m = _slsqp_solve(h_m, 0.01, maxiter=30)
        h_m = _project(h_m, n_try / 2.0)
        c5_m = _exact_c5(h_m)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()
            n_curr = n_try

    # Final SLSQP polish
    if time.time() < d6 - 2:
        h_fin = _slsqp_solve(best_h, 0.002, maxiter=40)
        h_fin = _project(h_fin, len(h_fin) / 2.0)
        c5_fin = _exact_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation and return
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
