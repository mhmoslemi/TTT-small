def compute_delta(h, i, j, n):
    """Compute delta in scores for all lags k=1..n-1 when swapping h[i]:0->1, h[j]:1->0"""
    k = np.arange(1, n)  # lags 1 to n-1
    delta = np.zeros(n-1)
    
    # Pair (i, i+k) for k where i+k < n, i.e., k < n-i
    mask1 = (i + k) < n  # k < n-i
    delta[mask1] -= (h[i + k[mask1]] == 1).astype(float)
    
    # Pair (i-k, i) for k where i-k >= 0, i.e., k <= i
    mask2 = k <= i
    delta[mask2] += (h[i - k[mask2]] == 0).astype(float)
    
    # Pair (j, j+k) for k where j+k < n, i.e., k < n-j
    mask3 = (j + k) < n
    delta[mask3] += (h[j + k[mask3]] == 1).astype(float)
    
    # Pair (j-k, j) for k where j-k >= 0, i.e., k <= j
    mask4 = k <= j
    delta[mask4] -= (h[j - k[mask4]] == 0).astype(float)
    
    # Overlap corrections
    # Case 1: j == i+k for k = j-i (if 1 <= j-i <= n-1)

    dj2 = i - j
    if 1 <= dj2 <= n-1:
        delta[dj2-1] += 1  # correction for case 2
    
    return delta

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    # Round to nearest 0/1
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Need to flip (n_ones - target) ones to zeros
        one_positions = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h values
        order = np.argsort(h[one_positions])
        to_flip = order[:n_ones - target]
        h_bin[one_positions[to_flip]] = 0
    elif n_ones < target:
        # Need to flip (target - n_ones) zeros to ones
        zero_positions = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_positions])
        to_flip = order[:target - n_ones]
        h_bin[zero_positions[to_flip]] = 1
    return h_bin

def delta_scores(h, i, j, n):

def _proj(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

        b_ub = np.concatenate([-scores, -scores])
        A_eq = np.ones((1, n))

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
            if res.success:
                p = res.x
                if np.max(np.abs(p)) > 1e-14:

                        elif p[j2] < -1e-14:
                            a_max = min(a_max, h[j2] / (-p[j2]))
                    a_max = max(a_max, 1e-12) * 0.95
                    h_try = _proj(h + a_max * p, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < cur_c5 - 1e-14:
                        h = h_try
                        cur_c5 = c5_try
                        if cur_c5 < best_c5:
                            best_c5 = c5_try
                            best_h = h.copy()
                        continue
        except Exception:
            pass
        break
    return best_h, best_c5

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return to_binary(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = to_binary(h0, n0).copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Resolution list
    resolutions = sorted(set([48, 64, 80, 96, 112, 128, 144, 160, 192, 256, n0]))
    resolutions = [r for r in resolutions if r >= 32 and r % 2 == 0]

    # Phase 1: SA from parent at n0
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.25)
        h1, c5_1 = binary_sa(best_h, best_n, dl, rng, T_init=0.10, T_min=0.0005, reheats=3)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: SA from other initializations at n0
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.30)
        for init_name in ['thue', 'golden']:
            if time.time() > dl:
                break
            if init_name == 'thue':
                h_init = thue_morse_bin(n0)
            else:
                h_init = golden_bin(n0)
            sub_dl = min(dl, time.time() + budget_s * 0.08)
            h_try, c5_try = binary_sa(h_init, n0, sub_dl, rng, T_init=0.08, T_min=0.0005, reheats=2)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n0

    # Phase 3: Multi-resolution exploration
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.55)
        for n_try in resolutions:
            if time.time() > dl:
                break
            if n_try == best_n:
                continue
            # Resample from best
            h_res = resample_h(best_h, n_try)
            sub_dl = min(dl, time.time() + budget_s * 0.06)
            h_try, c5_try = binary_sa(h_res, n_try, sub_dl, rng, T_init=0.07, T_min=0.0005, reheats=2)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            # Also try Thue-Morse at this resolution
            if time.time() < dl and n_try <= 192:
                h_tm = thue_morse_bin(n_try)
                sub_dl2 = min(dl, time.time() + budget_s * 0.04)
                h_try2, c5_try2 = binary_sa(h_tm, n_try, sub_dl2, rng, T_init=0.06, T_min=0.0005, reheats=2)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()
                    best_n = n_try

    # Phase 4: Greedy local search
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.70)
        h_g, c5_g = greedy_local(best_h, best_n, dl, rng, max_rounds=300)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 5: More SA with better targeting
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.82)
        h_s, c5_s = binary_sa(best_h, best_n, dl, rng, T_init=0.04, T_min=0.0002, reheats=4, target_frac=0.6)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Greedy again
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.88)
        h_g2, c5_g2 = greedy_local(best_h, best_n, dl, rng, max_rounds=200)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    # Phase 7: LP continuous polish
    if time.time() < deadline and best_n <= 192:
        dl = min(deadline, t0 + budget_s * 0.95)
        h_lp, c5_lp = lp_polish(best_h, dl, radius=0.05, max_iters=500)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 8: Final greedy
    if time.time() < deadline:
        dl = min(deadline, t0 + budget_s * 0.98)
        h_f, c5_f = greedy_local(best_h, best_n, dl, rng, max_rounds=100)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
