import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    """Golden section line search along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_lo + gr * (t_hi - t_lo)
    c2 = t_hi - gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(50):
        if t_hi - t_lo < 1e-12:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_lo + gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_hi - gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_lo + gr * (t_hi - t_lo)
    c2 = t_hi - gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(50):
        if t_hi - t_lo < 1e-12:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_lo + gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_hi - gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _build_eff_matrices(h):
    """Build effect matrices for binary flips.
    eff10[i] = change in corr_full when flipping h[i] from 1 to 0
    eff01[j] = change in corr_full when flipping h[j] from 0 to 1
    """
    n = len(h)
    r_idx = np.arange(2 * n - 1)
    d = r_idx - (n - 1)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    eff10 = np.zeros((len(ones), 2*n-1))
    eff01 = np.zeros((len(zeros), 2*n-1))
    for k, i in enumerate(ones):
        i_d = i + d
        i_md = i - d
        mask1 = (i_d >= 0) & (i_d < n)
        mask2 = (i_md >= 0) & (i_md < n)
        eff10[k] = (-(1.0 - h[i_d]) * mask1 + h[i_md] * mask2).astype(float)
        eff10[k][~mask1] += 0.0
        i_d_c = np.clip(i_d, 0, n-1)
        i_md_c = np.clip(i_md, 0, n-1)
        eff10[k] = -(1.0 - h[i_d_c]) * mask1 + h[i_md_c] * mask2
    for k, j in enumerate(zeros):
        j_d = j + d
        j_md = j - d
        mask1 = (j_d >= 0) & (j_d < n)
        mask2 = (j_md >= 0) & (j_md < n)
        j_d_c = np.clip(j_d, 0, n-1)
        j_md_c = np.clip(j_md, 0, n-1)
        eff01[k] = (1.0 - h[j_d_c]) * mask1 - h[j_md_c] * mask2
    return ones, zeros, eff10, eff01

def _binary_descent(h_bin, deadline):

    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    curr = np.correlate(h, 1.0 - h, mode='full')
    best_val = float(np.max(curr))
    no_improve = 0
    while time.time() < deadline and no_improve < 30:
        ones, zeros, eff10, eff01 = _build_eff_matrices(h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_oi = -1
        best_zj = -1
        for oi in range(len(ones)):
            temp = (curr[None, :] + eff10[oi][None, :] + eff01).max(axis=1)
            local_min = float(temp.min())
            if local_min < best_val - 1e-15 + best_delta:
                if best_delta <= 0 or local_min < best_val + best_delta:
                    best_delta = local_min - best_val
                    best_oi = oi
                    best_zj = int(np.argmin(temp))
        if best_oi >= 0:
            i = ones[best_oi]
            j = zeros[best_zj]
            h[i] = 0.0
            h[j] = 1.0

            new_val = float(np.max(curr))
            if new_val < best_val - 1e-15:
                best_val = new_val
                no_improve = 0
            else:

def _subgradient(h, deadline, eta0=0.03):
    """Subgradient method focusing on worst lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    cur_c5 = _exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    eta = eta0
    k_iter = 0
    while time.time() < deadline:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_corr = float(np.max(corr))
        cur_c5 = max_corr * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_cur.copy()
        # Find top worst lags
        top_k = min(3, len(corr))
        worst_idx = np.argsort(corr)[-top_k:]
        # Build subgradient
        grad = np.zeros(n)
        r_idx = np.arange(2 * n - 1)
        d = r_idx - (n - 1)
        for r in worst_idx:
            dd = d[r]
            # grad of corr[r] w.r.t. h[j]
            # For lag dd: corr = sum h[a]*(1-h[a+dd])
            # d/dh[j] = (1-h[j+dd])*[0<=j+dd<n] + (-h[j-dd])*[0<=j-dd<n]
            j_d = np.arange(n) + dd
            j_md = np.arange(n) - dd
            m1 = (j_d >= 0) & (j_d < n)
            m2 = (j_md >= 0) & (j_md < n)
            g1 = np.zeros(n)
            g1[m1] = 1.0 - h_cur[j_d[m1]]
            g2 = np.zeros(n)
            g2[m2] = -h_cur[j_md[m2]]
            grad += g1 + g2
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        grad /= grad_norm
        # Remove mean to preserve sum constraint
        grad -= np.mean(grad)
        h_new = h_cur - eta * grad
        h_new = _project(h_new, target)
        h_cur = h_new
        k_iter += 1
        eta = eta0 / (1.0 + 0.01 * k_iter)
        if eta < 1e-6:
            break
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.10):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        # Build gradient matrix
        r_idx = np.arange(2 * n - 1)
        d_vec = r_idx - (n - 1)
        G = np.zeros((2*n-1, n))
        for r, dd in enumerate(d_vec):
            j = np.arange(n)
            jd = j + dd
            jmd = j - dd
            m1 = (jd >= 0) & (jd < n)
            m2 = (jmd >= 0) & (jmd < n)
            jd_c = np.clip(jd, 0, n-1)
            jmd_c = np.clip(jmd, 0, n-1)
            G[r, m1] += (1.0 - h[jd_c[m1]])
            G[r, m2] += (-h[jmd_c[m2]])
        # LP: minimize t s.t. G@p - t*1 <= -scores, sum(p)=0, |p_j| <= radius
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.72
    return best_h, best_c5

def _pair_swap_refine(h, deadline, n_trials=3000, radius=0.2):
    """Random pair swap with line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    while time.time() < deadline:
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.5)
        else:
            stall += 1
            if stall % 100 == 0:
                radius *= 0.85
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    """Convert to exact binary with n/2 ones using top-n/2 selection."""
    n = len(h)
    idx = np.argsort(h)[::-1][:n//2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Binary descent on parent (12%)
    d1 = t0 + budget_s * 0.12
    hb = _to_binary(h_init)
    hb, c5_b = _binary_descent(hb, d1)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = hb.copy()

    # Phase 2: Continuous optimization on best (18%)
    d2 = t0 + budget_s * 0.30
    h2 = best_h.copy()
    # Subgradient
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h2, c5_2 = _subgradient(h2, sub_dl, eta0=0.04)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # LP
    sub_dl = min(d2, time.time() + budget_s * 0.06)
    h2, c5_2 = _lp_loop(h2, sub_dl, radius_init=0.08)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Binary descent on projected result
    hb2 = _to_binary(best_h)
    hb2, c5_2b = _binary_descent(hb2, min(d2, time.time() + 5))
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = hb2.copy()

    # Phase 3: Multi-n exploration (30%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    offsets = list(range(-64, 65, 4))
    offsets.sort(key=lambda x: abs(x))
    tried_ns = set([n_curr])
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 320 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        # Quick binary descent
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        hb_r = _to_binary(h_r)
        hb_r, c5_r = _binary_descent(hb_r, sub_dl)
        # Quick continuous
        rem = d3 - time.time()
        if rem > 3 and c5_r < best_c5 + 0.002:
            sub_dl2 = min(d3, time.time() + max(2, budget_s * 0.008))
            h_r2, c5_r2 = _lp_loop(hb_r, sub_dl2, radius_init=0.08)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                hb_r = h_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = hb_r.copy()
            n_curr = n_try

    # Phase 4: Intensive refinement on best (20%)
    d4 = t0 + budget_s * 0.80
    # Binary descent
    hb4 = _to_binary(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    hb4, c5_4 = _binary_descent(hb4, sub_dl)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = hb4.copy()
    # Subgradient
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4, c5_4 = _subgradient(best_h, sub_dl, eta0=0.02)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # LP
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4, c5_4 = _lp_loop(best_h, sub_dl, radius_init=0.05)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Pair swap
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4, c5_4 = _pair_swap_refine(best_h, sub_dl, n_trials=2000, radius=0.15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.95
    scale = 0.05
    for attempt in range(300):
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        # Quick optimization
        hb_p = _to_binary(h_pert)
        sub_dl = min(d5, time.time() + budget_s * 0.004)
        hb_p, c5_p = _binary_descent(hb_p, sub_dl)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = hb_p.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 6: Final polish (5%)
    h6 = _to_binary(best_h)
    h6, c5_6 = _binary_descent(h6, deadline)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    h6, c5_6 = _subgradient(best_h, deadline, eta0=0.01)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    h6, c5_6 = _lp_loop(best_h, deadline, radius_init=0.03)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
