        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            iters += 1
            # Adapt radius based on achieved step
            step_size = abs(alpha)
            if step_size > 0.5 * radius:
                radius = min(radius * 1.15, 0.45)
            elif step_size < 0.1 * radius:
                radius = max(radius * 0.8, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.04:
                radius = 0.1
    return best_h, best_c5

def _pair_cycle(h, deadline, max_rounds=20):
    """Systematically cycle through all pairs of interior coordinates."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[ii]
            for jj in range(ii + 1, len(interior)):
                if time.time() > deadline:
                    break
                j = interior[jj]
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _random_pair_search(h, deadline, n_attempts=5000):
    """Random pair moves with exact line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    stall = 0
    for _ in range(n_attempts):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, 0.5)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _four_pair_search(h, deadline, n_attempts=3000):
    """4-coordinate moves for escaping local optima."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    stall = 0
    for _ in range(n_attempts):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        idx = rng.choice(n, size=4, replace=False)
        signs = rng.choice([-1, 1], size=4)
        if abs(int(np.sum(signs))) > 1:
            signs[3] = -signs[0] - signs[1] - signs[2]
            if abs(signs[3]) > 1.5:
                continue
            signs[3] = np.sign(signs[3])
        d = np.zeros(n)
        for m in range(4):
            d[idx[m]] = signs[m]
        alpha, c5_try = _line_search(h, d, dx, 0.4)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_restart(h, rng, scale=0.03):
    """Perturb and re-project to escape local optima."""
    n = len(h)
    h_p = h + rng.normal(0, scale, n)
    return _project(h_p, n / 2.0)

def _structured_starts(n, rng):
    """Generate diverse structured starting points."""
    starts = []
    # Sine-based
    for freq in [1, 2, 3, 4, 5, 7]:
        for ph in [0, np.pi / 4, np.pi / 2, 3 * np.pi / 4]:
            x = np.arange(n)
            h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + ph)
            starts.append(_project(h, n / 2.0))
    # Block patterns
    x = np.arange(n)
    for period in [3, 4, 5, 6, 7, 8]:
        pattern_len = period
        duty = pattern_len // 2
        pattern = np.zeros(pattern_len)
        pattern[:duty] = 1.0
        h = np.tile(pattern, n // pattern_len + 1)[:n]
        starts.append(_project(h, n / 2.0))
    # Random blocks
    for _ in range(8):
        h = np.zeros(n)
        pos = 0
        while pos < n:
            blen = max(1, int(rng.integers(1, n // 5 + 1)))
            if rng.random() < 0.5:
                h[pos:pos + blen] = 1.0
            pos += blen
        starts.append(_project(h, n / 2.0))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / n0)
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick multi-start (5% time)
    d1 = t0 + budget_s * 0.05
    starts = _structured_starts(n0, rng)
    starts.append(h.copy())
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + 4.0)
        h_s, c5_s = _lp_loop_fast(s, sub_dl, radius_init=0.12, n_active=15)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main LP optimization (30% time)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _lp_loop_fast(best_h, d2, radius_init=0.18, n_active=20)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair cycle polish (10% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _pair_cycle(best_h, d3, max_rounds=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP again after pair polish (10% time)
    d4 = t0 + budget_s * 0.55
    h4, c5_4 = _lp_loop_fast(best_h, d4, radius_init=0.12, n_active=22)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Random pair + 4-pair search (15% time)
    d5 = t0 + budget_s * 0.70
    h5a, c5_5a = _random_pair_search(best_h, min(d5, t0 + budget_s * 0.60), n_attempts=8000)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5:
        h5b, c5_5b = _four_pair_search(best_h, d5, n_attempts=4000)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: LP polish after pair search (10% time)
    d6 = t0 + budget_s * 0.80
    h6, c5_6 = _lp_loop_fast(best_h, d6, radius_init=0.10, n_active=25)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Multi-resolution (15% time)
    d7 = t0 + budget_s * 0.95
    best_n = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, 8, -8, -10, 10, -12, 12, 14, -14, 16, -16, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        # Quick LP
        sub_dl = min(d7, time.time() + budget_s * 0.02)
        h_r, c5_r = _lp_loop_fast(h_r, sub_dl, radius_init=0.15, n_active=18)
        # Pair polish
        if time.time() < d7 - 2:
            sub_dl2 = min(d7, time.time() + budget_s * 0.015)
            h_r2, c5_r2 = _pair_cycle(h_r, sub_dl2, max_rounds=15)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        # More LP
        if time.time() < d7 - 1:
            h_r, c5_r = _lp_loop_fast(h_r, d7, radius_init=0.08, n_active=20)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final aggressive perturbation + re-opt
    d8 = deadline
    scale = 0.02
    for attempt in range(300):
        if time.time() > d8:
            break
        if best_c5 < 0.3805:
            break
        h_pert = _perturb_restart(best_h, rng, scale)
        sub_dl = min(d8, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop_fast(h_pert, sub_dl, radius_init=0.08, n_active=20)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.2)

    # Final pair polish
    if time.time() < deadline - 2:
        h_f, c5_f = _pair_cycle(best_h, deadline, max_rounds=10)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
