def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    if alpha_max < 1e-14:

    n_samp = 15
    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _local_lp(h, deadline, radius_init=0.25, max_iters=10000, stall_limit=200, active_frac=0.2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    prev_p = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        p, t_new = _lp_direction(h, scores, radius, active_frac)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
            continue
        if prev_p is not None and float(np.linalg.norm(p)) > 1e-10:
            p_blend = 0.6 * p + 0.4 * prev_p
            pn = np.linalg.norm(p_blend)
            if pn > 1e-14:
                p = p_blend / pn * np.linalg.norm(p)
        prev_p = p.copy()
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.8
            prev_p = None
            if stall > 40 and radius < 0.02:
                radius = 0.1
    return best_h, best_c5

def _smooth_optimize(h, deadline, beta_start=50.0, beta_end=300.0, n_phases=4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        def obj(x):
            S = _scores(x)
            mS = np.max(S)
            sh = beta * (S - mS)
            sh = np.clip(sh, -500, 0)
            return mS + np.log(np.mean(np.exp(sh))) / beta
        def grad(x):
            S = _scores(x)
            mS = np.max(S)
            sh = np.clip(beta * (S - mS), -500, 0)
            w = np.exp(sh)
            sw = np.sum(w)
            if sw < 1e-300:
                w = np.zeros(len(S))
                w[int(np.argmax(S))] = 1.0
            else:
                w /= sw
            G = _grad_matrix(x)
            return G.T @ w
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 150, 'ftol': 1e-15})
            if np.all(np.isfinite(res.x)):
                h_new = _project(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _qp_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def _sin_init(n, freq, phase, amp):
    x = np.arange(n) / n * 2.0
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return _project(h, n / 2.0)

def _block_init(n, block_len):
    h = np.zeros(n)
    for i in range(0, n, 2 * block_len):
        h[i:i + block_len] = 1.0
    return _project(h, n / 2.0)

def _fourier_perturb(h, rng, scale):
    n = len(h)
    f = np.zeros(n)
    k_max = max(1, n // 8)
    for k in range(1, k_max + 1):
        f += scale * (rng.standard_normal() * np.cos(2.0 * np.pi * k * np.arange(n) / n) +
                      rng.standard_normal() * np.sin(2.0 * np.pi * k * np.arange(n) / n))
    return _project(h + f, n / 2.0)

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = max(1, n // 40)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + step, n, step):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in (1, -1):
                    a_max = min(1.0 - h[j], h[k]) if sign > 0 else min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Deep LP from initial (40%)
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = _local_lp(h_start, dl1, radius_init=0.3, max_iters=20000, stall_limit=250, active_frac=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smooth optimization (10%)
    if time.time() < t0 + budget_s * 0.45:
        dl2 = t0 + budget_s * 0.45
        h2, c5_2 = _smooth_optimize(best_h, dl2, beta_start=40.0, beta_end=250.0, n_phases=4)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        dl2b = min(t0 + budget_s * 0.47, time.time() + budget_s * 0.02)
        h2b, c5_2b = _local_lp(best_h, dl2b, radius_init=0.1, max_iters=3000, stall_limit=100, active_frac=0.15)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Multi-resolution (20%)
    if time.time() < t0 + budget_s * 0.67:
        dl3 = t0 + budget_s * 0.65
        n_tries = [96, 128, 160, 192]
        for n_try in n_tries:
            if time.time() > dl3:
                break
            # From best resampled
            h_init = _resample(best_h, n_try)
            sub_dl = min(dl3, time.time() + budget_s * 0.035)
            ht, ct = _local_lp(h_init, sub_dl, radius_init=0.2, max_iters=5000, stall_limit=150, active_frac=0.15)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try
            # Quasi-periodic
            if time.time() < dl3:
                alphas = [0.6180339887, 0.3819660113, 0.7071067812, 0.4142135624]
                a = alphas[len(n_tries[:list(n_tries).index(n_try) if n_try in n_tries else 0]) % len(alphas)]
                h_qp = _qp_init(n_try, a)
                h_qp = _project(h_qp + rng.normal(0, 0.03, n_try), n_try / 2.0)
                sub_dl2 = min(dl3, time.time() + budget_s * 0.025)
                ht2, ct2 = _local_lp(h_qp, sub_dl2, radius_init=0.2, max_iters=3000, stall_limit=120, active_frac=0.15)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = n_try

    # Phase 4: Diverse multi-start (20%)
    if time.time() < t0 + budget_s * 0.87:
        dl4 = t0 + budget_s * 0.87
        n_curr = best_n
        attempts = 0
        scale = 0.04
        while time.time() < dl4 and attempts < 300:
            attempts += 1
            # Choose perturbation type
            ptype = attempts % 4
            if ptype == 0:
                h_pert = best_h + rng.normal(0, scale, n_curr)
                h_pert = _project(h_pert, n_curr / 2.0)
            elif ptype == 1:
                freq = 1 + int(rng.integers(0, n_curr // 4))
                ph = rng.uniform(0, 2 * np.pi)
                h_pert = _sin_init(n_curr, freq, ph, 0.3)
                blend = 0.7 * best_h + 0.3 * h_pert
                h_pert = _project(blend, n_curr / 2.0)
            elif ptype == 2:
                h_pert = _fourier_perturb(best_h, rng, scale)
            else:
                bl = int(rng.integers(2, max(3, n_curr // 8)))
                h_pert = _block_init(n_curr, bl)
                blend = 0.5 * best_h + 0.5 * h_pert
                h_pert = _project(blend, n_curr / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.01)
            ht, ct = _local_lp(h_pert, sub_dl, radius_init=0.08, max_iters=800, stall_limit=80, active_frac=0.2)
            if ct < best_c5 - 1e-12:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.005, scale * 0.85)
            elif ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
            else:
                scale = min(0.1, scale * 1.05)

    # Phase 5: Final polish (8%)
    if time.time() < deadline - 5:
        dl5 = deadline - 5
        h5, c5_5 = _local_lp(best_h, min(dl5 - 3, time.time() + budget_s * 0.04),
                             radius_init=0.05, max_iters=5000, stall_limit=200, active_frac=0.1)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        hp, c5p = _pair_polish(best_h, dl5, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
