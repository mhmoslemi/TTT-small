```python
def _binary_sa(h_binary, deadline, T_start=0.5, T_end=0.01, n_iter_max=500000, seed=42):
    n = len(h_binary)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    
    h = h_binary.copy().astype(float)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    cur_c5 = best_c5
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    T = T_start
    cool_rate = (T_end / T_start) ** (1.0 / max(1, n_iter_max))
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        # Pick random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        # Make swap
        h[i] = 0
        h[j] = 1
        
        # Compute new C5
        new_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
        
        # Accept/reject
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # Update ones and zeros
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
        else:
            # Reject: undo swap
            h[i] = 1
            h[j] = 0
        
        T = max(T * cool_rate, T_end)
    
    return best_h, best_c5
```

```python
def _optimal_coord(h, i, corr, dx):
    n = len(h)
    # Compute a_m for each m
    # a_m = (1 - 2*h[i+m-(n-1)]) if 0 <= i+m-(n-1) <= n-1, else 0
    # m ranges from 0 to 2n-2
    
    partner = i + np.arange(2*n-1) - (n-1)  # i + m - (n-1) for m = 0,...,2n-2
    valid = (partner >= 0) & (partner <= n-1)
    a = np.zeros(2*n-1)
    a[valid] = 1 - 2*h[partner[valid]]
    
    # f(t) = max(corr + a*t) where t = v - h[i], t in [-h[i], 1-h[i]]
    t_lo = -h[i]
    t_hi = 1 - h[i]
    
    # Since f is convex piecewise-linear, use golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_g = t_lo + gr * (t_hi - t_lo)
    b_g = t_lo + (1-gr) * (t_hi - t_lo)
    fa = np.max(corr + a * a_g)
    fb = np.max(corr + a * b_g)
    
    for _ in range(60):
        if t_hi - t_lo < 1e-14:
            break
        if fa < fb:
            t_hi = b_g
            b_g = a_g
            fb = fa
            a_g = t_lo + gr * (t_hi - t_lo)
            fa = np.max(corr + a * a_g)
        else:
            t_lo = a_g
            a_g = b_g
            fa = fb
            b_g = t_lo + (1-gr) * (t_hi - t_lo)
            fb = np.max(corr + a * b_g)
    
    t_opt = (t_lo + t_hi) / 2
    v_opt = h[i] + t_opt
    v_opt = np.clip(v_opt, 0, 1)
    return v_opt
```

minimize max(max_{m affected} (c[m] + a_m * t), max_{m unaffected} c[m])

where C_unaffected = max of c[m] over unaffected lags.

This simplifies the problem: we're minimizing the max of (a set of linear functions) and a constant. The minimum is at the point where the envelope of the linear functions just touches the constant, or at the minimum of the linear envelope if it's always below the constant.

```python
def _project(v, target):
    """Project v to [0,1]^n with sum = target using bisection."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(50):

        else:
```

```python
def _lp_direction(h, top_k=10):
    """Find a direction to decrease the top-k lags simultaneously using LP."""
    from scipy.optimize import linprog
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    scores = corr * dx
    
    # Get top k lags
    idx = np.argsort(scores)[-top_k:]
    s_k = scores[idx]
    
    # Build the Jacobian for these lags
    # d(c_k)/d(h_i) = (1-2*h[partner]) if partner valid, else 0
    # where partner = i + m - (n-1)
    
    # We want to find direction p such that:
    # s_k + G_k @ p ≤ s_k - epsilon for all k in top_k
    # i.e., G_k @ p ≤ -epsilon
    # subject to |p_i| ≤ 1 and sum(p) = 0 (to preserve the constraint)
    
    # Actually, let's formulate as:
    # minimize t
    # subject to G_k @ p + t ≥ 0 for all k (i.e., the new score is ≤ current)
    # |p_i| ≤ radius
    # sum(p) = 0
    
    # Wait, I want to DECREASE the scores. So:
    # s_k + dx * (G_k @ p) ≤ s_k for all k
    # G_k @ p ≤ 0 for all k
    # But that's too restrictive. Let me allow some increase:
    # G_k @ p ≤ t for all k, minimize t
    
    # G_k[i, l] = d(corr[l])/d(h_i) for i=0..n-1, l in idx
    # d(corr[m])/d(h_i) = (1-2*h[i+m-(n-1)]) if 0 ≤ i+m-(n-1) ≤ n-1, else 0
    
    # This is the same "a_vec" structure.
    # G[m, i] = 1-2*h[i+m-(n-1)] if valid, else 0
    # Note: G[m, i] = 1-2*h[partner] where partner = i+m-(n-1)
    # Also G is "symmetric" in a sense: G[m, i] relates to h at position partner.
    
    # Actually, let's just compute the full (2n-1) x n matrix and select rows.
    # G[m, i] = 1-2*h[i+m-(n-1)] if 0 ≤ i+m-(n-1) ≤ n-1, else 0
    # But we also have the -delta² correction at m=n-1, which is second order, so ignore for direction finding.
    
    # For the LP, we use the first-order approximation.
    
    # Efficient construction:
    # For each lag m, partner_m(i) = i + m - (n-1)
    # G[m, i] = 1-2*h[i+m-(n-1)] if 0 ≤ i+m-(n-1) ≤ n-1, else 0
    
    # Let's build this as a banded matrix.
    # For a given m, the valid i range is: 0 ≤ i+m-(n-1) ≤ n-1, i.e., n-1-m ≤ i ≤ 2n-2-m
    # But i must also be in [0, n-1], so: max(0, n-1-m) ≤ i ≤ min(n-1, 2n-2-m)
    
    # This is getting complex. Let me just build the full matrix for n=128.
    # (2n-1) x n = 255 x 128 ≈ 33K entries. That's fine.
    
    G = np.zeros((2*n-1, n))
    for i in range(n):
        m_lo = n - 1 - i
        m_hi = 2*n - 2 - i
        G[m_lo:m_hi+1, i] = 1.0 - 2.0 * h
    
    # Select top k rows
    G_k = G[idx]  # shape (top_k, n)
    
    # LP: minimize t
    # subject to: G_k @ p + t ≥ 0 for all k  →  -G_k @ p + t ≤ 0  →  G_k @ p - t ≤ 0
    # |p_i| ≤ radius
    # sum(p) = 0 (to preserve sum(h) = n/2 to first order)
    
    radius = 0.1
    # Variables: p_0, ..., p_{n-1}, t (total n+1 variables)
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: G_k @ p - t ≤ 0
    A_ub = np.hstack([G_k, -np.ones((top_k, 1))])
    b_ub = np.zeros(top_k)
    
    # sum(p) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None
```
