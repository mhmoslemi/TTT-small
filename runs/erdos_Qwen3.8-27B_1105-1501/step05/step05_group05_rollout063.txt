def _lp_step(h, S0, radius=0.15):
    """One LP-based trust region step."""
    from scipy.optimize import linprog
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    h_comp = 1.0 - h
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += h_comp[:m]
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += -h[g:]
                G[r, g:] += h_comp[:m]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -S0.astype(float)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for jj in range(n):
        lo = max(-radius, -h[jj])
        hi = min(radius, 1.0 - h[jj])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    h_comp = 1.0 - h
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * h_comp[d:d + m]
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * h_comp[:m]
    grad -= grad.mean()
    return grad

        t_hi = 1.0
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / dd[pos])))
        t_lo = -1.0
        if np.any(neg):
            t_lo = max(t_lo, float(np.max(h[neg] / dd[neg])))
        if t_hi - t_lo < 1e-14:
            stall += 1
            continue
        ng = 20
        grid = np.linspace(t_lo, t_hi, ng)
        vals = np.array([float(np.max(S0 + g * T - g * g * C)) * dx for g in grid])
        bi2 = int(np.argmin(vals))
        bt, bv = grid[bi2], vals[bi2]
        if bv < cur_c5 - 1e-12:
            h = h + bt * dd
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.4)
        else:
            stall += 1
            lr *= 0.8
    return best_h, best_c5

def _kick_opt(h, deadline, rng, n_kicks=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.03, 0.10)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = _project(best_h + scale * direction, target)
        # Quick optimization
        sub_dl = min(deadline, time.time() + 4.0)
        h_t, c5_t = _pair_transfer_fast(h_k, sub_dl, max_passes=50)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    # Get initial solution
    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick LP + smooth optimization (8%)
    d1 = t0 + budget_s * 0.08
    if time.time() < d1:
        h1, c5_1 = _lp_optimize(h, min(d1, time.time() + budget_s * 0.04))
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    if time.time() < d1:
        h1, c5_1 = _smooth_opt(best_h, d1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Intensive pair transfer on main resolution (35%)
    d2 = t0 + budget_s * 0.43
    h2, c5_2 = _pair_transfer_fast(best_h, d2, max_passes=500)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution search (25%)
    d3 = t0 + budget_s * 0.68
    n_best = len(best_h)
    offsets_to_try = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, -10, 10]:
        n_try = n_best + dn
        if 80 <= n_try <= 220:
            offsets_to_try.append(n_try)
    for n_try in offsets_to_try:
        if time.time() > d3:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + max(5, budget_s * 0.025))
        h_r, c5_r = _pair_transfer_fast(h_r, sub_dl, max_passes=200)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Kicks for diversity (12%)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = _kick_opt(best_h, d4, rng, n_kicks=20)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Re-try multi-resolution with new best (8%)
    d5 = t0 + budget_s * 0.88
    n_best = len(best_h)
    for dn in [-4, -2, 2, 4]:
        if time.time() > d5:
            break
        n_try = n_best + dn
        if 80 <= n_try <= 220:
            h_r = _resample(best_h, n_try)
            sub_dl = min(d5, time.time() + max(3, budget_s * 0.02))
            h_r, c5_r = _pair_transfer_fast(h_r, sub_dl, max_passes=100)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Final pair transfer polish (10%)
    d6 = deadline - 1.0
    if time.time() < d6:
        h6, c5_6 = _pair_transfer_fast(best_h, d6, max_passes=200)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    # Make sure final_c5 is reasonable
    if not np.isfinite(final_c5) or final_c5 > 0.5:
        best_h = _project(h0.copy(), n0 / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
