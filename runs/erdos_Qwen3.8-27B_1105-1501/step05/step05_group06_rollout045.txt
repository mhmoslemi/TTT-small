def _pnorm_obj_grad(h, q):
    h = h.reshape(-1)
    F = _scores(h)
    # Numerically stable p-norm
    M = np.max(F)
    if M <= 1e-30:
        return 0.0, np.zeros(len(h))
    a = F / M  # in [0, 1]
    # For large q, a^q underflows for a < 1, which is fine (those terms are negligible)
    aq = a ** q  # this might underflow, which is OK
    S = np.sum(aq)
    if S <= 0:
        return M, np.zeros(len(h))
    phi = M * S ** (1.0 / q)
    # Gradient: d(phi)/d(h_j) = S^(1/q - 1) * sum_r a_r^(q-1) * dF_r/dh_j
    # = S^(1/q - 1) * (G^T @ w)_j where w_r = a_r^(q-1)
    w = a ** (q - 1)  # in [0, 1]
    G = _build_G(h)
    g = G.T @ w
    grad = S ** (1.0 / q - 1.0) * g
    return phi, grad

    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for it in range(200):
        if time.time() > deadline:
            break
        F = _scores(h)
        M = np.max(F)
        if M <= 1e-30:
            break
        a = F / M
        aq_1 = a ** (q - 1)  # weights
        S = np.sum(a ** q)
        if S <= 0:
            break
        G = _build_G(h)
        grad = G.T @ aq_1
        # Normalize gradient (remove component in feasible direction)
        grad = grad - grad.mean()  # project onto sum=0 subspace
        gn = np.linalg.norm(grad)
        if gn < 1e-15:

        grad /= gn
        
        # Line search
        best_alpha = 0
        best_val = cur_c5
        alpha = 0.5
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)

    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(1 - n, n):
        r = n - 1 + d
        # First term: j such that 0 <= j-d < n => max(0,d) <= j < min(n, n+d)
        j1_lo = max(0, d)
        j1_hi = min(n, n + d)
        if j1_hi > j1_lo:
            G[r, j1_lo:j1_hi] += dx * (1.0 - h[j1_lo - d:j1_hi - d])
        # Second term: j such that 0 <= j+d < n => max(0,-d) <= j < min(n, n-d)
        j2_lo = max(0, -d)
        j2_hi = min(n, n - d)
        if j2_hi > j2_lo:
            G[r, j2_lo:j2_hi] -= dx * h[j2_lo + d:j2_hi + d]
    return G

def _pnorm_obj(x, q):
    F = _scores(x)
    M = float(np.max(F))
    if M <= 1e-30:
        return 0.0
    a = F / M
    S = float(np.sum(a ** q))
    if S <= 0:
        return M
    return M * S ** (1.0 / q)

def _pnorm_grad(x, q):
    F = _scores(x)
    M = float(np.max(F))
    n = len(x)
    if M <= 1e-30:
        return np.zeros(n)
    a = F / M
    w = a ** (q - 1)
    S = float(np.sum(a ** q))
    if S <= 0:
        return np.zeros(n)
    G = _build_G(x)
    g = G.T @ w
    return S ** (1.0 / q - 1.0) * g

def _mass_eq(x):
    return float(np.sum(x)) - len(x) / 2.0

def _mass_eq_jac(x):
    return np.ones(len(x))

def _pnorm_slsqp(h, q, maxiter=150, deadline=None):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _mass_eq, 'jac': _mass_eq_jac}]
    try:
        res = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(q,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or (hasattr(res, 'x') and res.x is not None):
            x = _project(res.x, n / 2.0)
            return x
    except Exception:
        pass
    return h

def _pnorm_continuation(h, q_list, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    for q in q_list:
        if time.time() > deadline:
            break
        # Multiple SLSQP passes per q for robustness
        for _ in range(2):
            if time.time() > deadline:
                break
            cur_h = _pnorm_slsqp(cur_h, q, maxiter=120, deadline=deadline)
            c5_now = _exact_c5(cur_h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = cur_h.copy()
        # Verify and project
        cur_h = _project(cur_h, n / 2.0)
    return best_h, best_c5

def _lp_local(h, deadline, top_k=None, radius=0.1, max_iters=500):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = _scores(h)
        G = _build_G(h)
        L = len(scores)
        if top_k is not None and top_k < L:
            idx = np.argsort(scores)[-top_k:]
            s_k = scores[idx]
            G_k = G[idx]
        else:
            s_k = scores
            G_k = G
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(s_k), n + 1))
        A_ub[:, :n] = G_k
        A_ub[:, n] = -1.0
        b_ub = -s_k
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x is not None:
                p = res.x[:n]
                if np.max(np.abs(p)) > 1e-14:
                    # Line search
                    alpha_max = 1.0
                    for j in range(n):
                        if p[j] > 1e-14:
                            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                        elif p[j] < -1e-14:
                            alpha_max = min(alpha_max, h[j] / (-p[j]))
                    alpha_max = max(alpha_max, 1e-10)
                    best_a = 0.0
                    best_v = best_c5
                    alphas = np.linspace(0.0, alpha_max, 11)
                    for a in alphas:
                        h_try = _project(h + a * p, target)
                        v = _exact_c5(h_try)
                        if v < best_v:
                            best_v = v
                            best_a = a
                    if best_a > 0 and best_v < best_c5 - 1e-13:
                        h = _project(h + best_a * p, target)
                        if best_v < best_c5:
                            best_c5 = best_v
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.05, 0.3)
                    else:
                        stall += 1
                        radius *= 0.7
                        if radius < 1e-7:
                            break
                else:
                    stall += 1
            else:
                stall += 1
        except Exception:
            stall += 1
    return best_h, best_c5

def _pair_descent(h, deadline, max_passes=100):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > 1e-7 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smoothing_step(h, N=5):
    n = len(h)
    w = np.exp(-0.5 * (np.arange(-2, 3) / 1.5) ** 2)
    w /= w.sum()
    h_s = np.convolve(h, w, mode='same')
    return _project(h_s, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    # Get initial from parent
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = _quasi_init(128, 0.6180339887, rng)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: p-norm continuation on parent (main innovation)
    dl1 = t0 + budget_s * 0.30
    q_list_main = [8, 16, 32, 64, 128, 256, 512]
    h_pn, c5_pn = _pnorm_continuation(h_start, q_list_main, dl1)
    if c5_pn < best_c5:
        best_c5 = c5_pn
        best_h = h_pn.copy()

    # Phase 2: LP local optimization on current best
    if time.time() < t0 + budget_s * 0.40:
        dl2 = t0 + budget_s * 0.42
        tk_list = [None, 8, 5, 3]
        radius_cur = 0.12
        h_lp = best_h.copy()
        for tk in tk_list:
            if time.time() > dl2:
                break
            sub_dl = min(dl2, time.time() + budget_s * 0.04)
            h_lp, c5_lp = _lp_local(h_lp, sub_dl, top_k=tk, radius=radius_cur, max_iters=300)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
            else:
                break

    # Phase 3: Multi-start exploration with p-norm
    if time.time() < t0 + budget_s * 0.60:
        dl3 = t0 + budget_s * 0.62
        resolutions = [128, 160, 192, 256]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5]
        qi = 0
        for n_try in resolutions:
            if time.time() > dl3:
                break
            n_starts = 2 if n_try <= 160 else 1
            for s in range(n_starts):
                if time.time() > dl3:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    h_init = _quasi_init(n_try, alphas[qi % len(alphas)], rng)
                    qi += 1
                sub_dl = min(dl3, time.time() + budget_s * 0.05)
                q_sub = [16, 32, 64, 128]
                h_try, c5_try = _pnorm_continuation(h_init, q_sub, sub_dl)
                # Also try LP
                if time.time() < dl3:
                    sub_dl2 = min(dl3, time.time() + budget_s * 0.025)
                    h_try2, c5_try2 = _lp_local(h_try, sub_dl2, top_k=5, radius=0.08, max_iters=200)
                    if c5_try2 < c5_try:
                        h_try, c5_try = h_try2, c5_try2
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 4: High-q refinement on best (fine-tuning with very sharp p-norm)
    if time.time() < t0 + budget_s * 0.75:
        dl4 = t0 + budget_s * 0.77
        q_fine = [256, 512, 1024]
        h_fine, c5_fine = _pnorm_continuation(best_h, q_fine, dl4)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
        # Pair descent polish
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.04)
            h_pd, c5_pd = _pair_descent(best_h, sub_dl, max_passes=80)
            if c5_pd < best_c5:
                best_c5 = c5_pd
                best_h = h_pd.copy()

    # Phase 5: Perturbation restarts around best
    if time.time() < t0 + budget_s * 0.90:
        dl5 = t0 + budget_s * 0.92
        scale = 0.02
        n_pert = best_n
        for attempt in range(300):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, n_pert)
            h_pert = _project(h_pert, n_pert / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.006)
            # Quick p-norm step
            h_try, c5_try = _pnorm_continuation(h_pert, [64, 128], sub_dl)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.06, scale * 1.05)

    # Phase 6: Final polish
    if time.time() < hard_deadline:
        # Very high q SLSQP
        q_ult = [512, 1024, 2048]
        h_ult, c5_ult = _pnorm_continuation(best_h, q_ult, hard_deadline)
        if c5_ult < best_c5:
            best_c5 = c5_ult
            best_h = h_ult.copy()
        # Tiny LP
        if time.time() < hard_deadline - 2:
            h_fin, c5_fin = _lp_local(best_h, hard_deadline, top_k=4, radius=0.01, max_iters=100)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final validation
    best_n = len(best_h)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Use saved best (before final projection which might have perturbed)
        final_c5 = best_c5
    return best_h, final_c5, best_n
