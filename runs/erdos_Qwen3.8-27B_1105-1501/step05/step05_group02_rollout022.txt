import numpy as np
import time
from itertools import combinations

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binarize(h, n_ones):
    """Binarize h by keeping the top n_ones values as 1."""
    n = len(h)
    idx = np.argsort(-h)[:n_ones]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _enumerate_periodic(p):
    """Enumerate all C(p, p/2) binary patterns of length p with p/2 ones."""
    from itertools import combinations
    k = p // 2
    results = []
    for combo in combinations(range(p), k):
        b = np.zeros(p)
        b[list(combo)] = 1.0
        results.append(b)
    return results

    """Score a periodic pattern tiled to length n."""
    p = len(pattern)
    k = n // p
    tile = np.tile(pattern, k)
