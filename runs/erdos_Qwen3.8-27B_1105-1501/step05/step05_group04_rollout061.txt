        return np.zeros(n)

        return np.ones(n)

            G[r, 0:m] -= dx * h[s:s + m]
        elif s == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        else:
            a = -s
            m = n - a
            G[r, 0:m] += dx * (1.0 - h[a:a + m])
            G[r, a:a + m] -= dx * h[:m]
    return G


def line_search_quad(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_lo = 0.0
    if t_hi <= t_lo + 1e-15:
        return 0.0, eval_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val


def _lp_direction(h, scores, G, radius, n_active):
    n = len(h)
    n_lags = 2 * n - 1
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:n_active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < float(np.max(scores)) - 1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None


def lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_active = 5
    while time.time() < deadline:
        if stall > 100:
            break
        scores = lag_values(h)
        G = lag_jacobian(h)
        p = _lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            n_active = min(n_active + 3, 2 * n - 2)
            p = _lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 2
            radius *= 0.7
            if radius < 1e-5:
                break
            if stall % 30 == 0:
                radius = max(radius, 0.04)
            continue
        alpha, new_c5 = line_search_quad(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project_box_sum(h, target)
            cur_c5 = eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            n_active = 5
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.78
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.06
                n_active = 4
    return best_h, best_c5


def smooth_obj(v, n, beta):
    h_c = np.clip(v, 0.0, 1.0)
    F = lag_values(h_c)
    m = F.max()
    shifted = (F - m) / beta
    shifted = shifted - shifted.max()
    Z = np.exp(shifted)
    Zs = Z.sum()
    return m + beta * (np.log(Zs) + shifted.max() + (F.max() - m) / beta * 0)


def smooth_obj_grad(v, n, beta):
    h_c = np.clip(v, 0.0, 1.0)
    F = lag_values(h_c)
    m = F.max()
    shifted = (F - m) / beta
    shifted = shifted - shifted.max()
    Z = np.exp(shifted)
    P = Z / Z.sum()
    J = lag_jacobian(h_c)
    return P @ J


def _eq_fun(v, n, target):
    return np.sum(v) - target


def _eq_jac(v, n, target):
    return np.ones(len(v))


def smooth_slsqp(h, deadline):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    beta_schedule = [0.1, 0.05, 0.025, 0.01, 0.005, 0.002]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n, target)}]
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        maxit = max(20, int(rem * 5))
        try:
            res = minimize(smooth_obj, best_h.copy(), args=(n, beta),
                           jac=smooth_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-9})
            h_new = project_box_sum(res.x, target)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5


def _epi_obj(v, n):
    return v[n]


def _epi_obj_grad(v, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _epi_ineq_fun(v, n, active_idx):
    h = v[:n]
    t = v[n]
    F = lag_values(h)
    return t - F[active_idx]


def _epi_ineq_jac(v, n, active_idx):
    h = v[:n]
    J = lag_jacobian(h)
    rows = -J[active_idx]
    return np.hstack([rows, np.ones((len(active_idx), 1))])


def _epi_eq_fun(v, n, target):
    return np.sum(v[:n]) - target


def _epi_eq_jac(v, n, target):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j


def epigraph_solve(h, deadline, max_rounds=8):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    F = lag_values(h)
    active_idx = np.argsort(F)[::-1][:20]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        if rem < 5:
            break
        t_start = best_c5 + 0.005
        x0 = np.concatenate([best_h, [t_start]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons_eq = {'type': 'eq', 'fun': _epi_eq_fun, 'jac': _epi_eq_jac, 'args': (n, target)}
        cons_ineq = {'type': 'ineq', 'fun': _epi_ineq_fun, 'jac': _epi_ineq_jac,
                     'args': (n, active_idx)}
        maxit = max(50, int(rem * 3))
        maxit = min(maxit, 300)
        try:
            res = minimize(_epi_obj, x0, args=(n,),
                           jac=_epi_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=[cons_eq, cons_ineq],
                           options={'maxiter': maxit, 'ftol': 1e-10})
            h_new = project_box_sum(res.x[:n], target)
            c5_new = eval_c5(h_new)
            improved = False
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                improved = True
            F_new = lag_values(h_new)
            t_val = float(res.x[n]) if np.isfinite(res.x[n]) else best_c5
            violated = np.where(t_val - F_new < -1e-7)[0]
            if len(violated) > 0:
                new_active = np.unique(np.concatenate([active_idx, violated[:10]]))
                active_idx = new_active[:40]
            elif improved:
                F_sorted = np.argsort(F_new)[::-1]
                active_idx = F_sorted[:max(10, len(active_idx) // 2)]
            else:
                if not improved and rnd > 2:
                    break
        except Exception:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(h_new, n_new / 2.0)


def resolution_cycling(h, deadline, rng):
    n0 = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    tried = set()
    deltas = list(range(1, 15))
    rng.shuffle(deltas)
    for sign in (1, -1):
        for d in deltas:
            if time.time() > deadline:
                return best_h, best_c5
            n_try = n0 + sign * d
            if n_try < 30 or n_try > 200 or n_try in tried:
                continue
            tried.add(n_try)
            h_r = _resample(best_h, n_try)
            c5_r = eval_c5(h_r)
            if c5_r < best_c5 - 1e-6:
                sub_dl = min(deadline, time.time() + max(3.0, (deadline - time.time()) * 0.15))
                h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.08)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
    return best_h, best_c5


def coord_refine(h, deadline, n_iter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(42)
    interior = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if len(interior) < 2:
        return best_h, best_c5
    scores = lag_values(h)
    max_s = scores.max()
    active_lags = np.where(scores > max_s - 0.02)[0]
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = interior[rng.integers(0, len(interior))]
        j = interior[rng.integers(0, len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        max_step = min(h[i], 1.0 - h[j], 0.5)
        alpha, c5_try = line_search_quad(h, d, dx, max_step)
        if c5_try < best_c5 - 1e-13:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
                if len(interior) < 2:
                    break
    return best_h, best_c5


def systematic_coord_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    scores = lag_values(h)
    max_s = scores.max()
    active_lags = np.where(scores > max_s - 0.03)[0]
    G = lag_jacobian(h)
    active_grad = G[active_lags]
    interior = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if len(interior) < 2:
        return best_h, best_c5
    n_int = len(interior)
    max_pairs = min(n_int * (n_int - 1) // 2, 5000)
    pair_i = 0
    it = 0
    while pair_i < max_pairs and time.time() < deadline:
        if it % 100 == 0 and time.time() > deadline:
            break
        i_idx = pair_i // max(1, n_int - 1)
        j_remainder = pair_i % max(1, n_int - 1)
        if i_idx >= n_int:
            break
        j_idx = j_remainder
        if j_idx >= i_idx:
            j_idx += 1
        if j_idx >= n_int:
            pair_i += n_int - 1
            continue
        i = interior[i_idx]
        j = interior[j_idx]
        pair_i += 1
        it += 1
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        max_step = min(h[i], 1.0 - h[j], 0.5)
        alpha, c5_try = line_search_quad(h, d, dx, max_step)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = project_box_sum(h, target)
            cur = eval_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                interior = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
                n_int = len(interior)
                if n_int < 2:
                    break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0_arr = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0_arr)
    h = project_box_sum(h0_arr.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    # Phase 1: LP polish (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = lp_loop(h, d1, radius_init=0.12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Epigraph SLSQP (22%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = epigraph_solve(best_h, d2, max_rounds=6)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smooth minimization (18%)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = smooth_slsqp(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Second epigraph pass (18%)
    d4 = t0 + budget_s * 0.66
    h4, c5_4 = epigraph_solve(best_h, d4, max_rounds=8)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Resolution cycling (14%)
    d5 = t0 + budget_s * 0.80
    best_h, best_c5 = resolution_cycling(best_h, d5, rng)

    # Phase 6: Systematic coordinate refinement (8%)
    d6 = t0 + budget_s * 0.88
    best_h, best_c5 = systematic_coord_refine(best_h, d6)

    # Phase 7: Final LP + coord polish (7%)
    d7 = t0 + budget_s * 0.95
    h7, c5_7 = lp_loop(best_h, d7, radius_init=0.06)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    best_h, best_c5 = coord_refine(best_h, t0 + budget_s * 0.98, n_iter=300)

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = eval_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 > 1.0:
        best_h = project_box_sum(h0_arr.copy(), n0 / 2.0)
        final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
