    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d

            m = n - d

            G[r, :m] += (1.0 - h[d:d+m])

            G[r, :m] += (-h[g:g+m])

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import expit
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _balance_exact(x, target):
    n = len(x)
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    cur = float(np.sum(x))
    if abs(cur - target) < 1e-13:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    y = np.clip(x - t, 0.0, 1.0)
    res = target - float(np.sum(y))
    if abs(res) > 1e-14:
        if res > 0:
            mask = y < 1.0 - 1e-15
        else:
            mask = y > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            y[mask] += res / cnt
            y = np.clip(y, 0.0, 1.0)
    return y

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import expit
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _balance_exact(x, target):
    n = len(x)
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    cur = float(np.sum(x))
    if abs(cur - target) < 1e-13:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    y = np.clip(x - t, 0.0, 1.0)
    res = target - float(np.sum(y))
    if abs(res) > 1e-14:
        if res > 0:
            mask = y < 1.0 - 1e-15
        else:
            mask = y > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            y[mask] += res / cnt
            y = np.clip(y, 0.0, 1.0)
    return y

def _solve_mu(q, target):
    lo = float(q.min()) - 20.0
    hi = float(q.max()) + 20.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = float(np.sum(expit(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target):
    mu = _solve_mu(q, target)
    h = expit(q - mu)
    return _balance_exact(h, target)

def _logit_h(h):
    eps = 1e-6
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def _grad_scores_jac(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m]) * dx
                G[r, d:] += (-h[:m]) * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m]) * dx
                G[r, :m] += (-h[g:g + m]) * dx
    return G

def _log_obj(q, target, beta):
    h = _q_to_h(q, target)
    scores = _scores(h)
    M = float(np.max(scores))
    shifted = beta * (scores - M)
    shifted = np.clip(shifted, -500.0, 0.0)
    wsum = float(np.sum(np.exp(shifted)))
    if wsum < 1e-300:
        return M
    return M + float(np.log(wsum / len(scores))) / beta

def _log_grad(q, target, beta):
    h = _q_to_h(q, target)
    scores = _scores(h)
    M = float(np.max(scores))
    shifted = beta * (scores - M)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    G = _grad_scores_jac(h)
    grad_h = G.T @ w
    s = h * (1.0 - h)
    S = float(np.sum(s))
    if S < 1e-15:
        return np.zeros(len(q))
    rho = float(np.dot(s, grad_h)) / S
    grad_q = s * (grad_h - rho)
    return np.nan_to_num(grad_q, nan=0.0, posinf=0.0, neginf=0.0)

def _logistic_optimize(h0, target, deadline, betas=None, R=14.0, maxiter=150):
    n = len(h0)
    h = _balance_exact(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    q = _logit_h(h)
    bounds = [(-R, R)] * n
    if betas is None:
        betas = [5.0, 15.0, 30.0, 60.0, 120.0, 250.0, 500.0]
    for beta in betas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_log_obj, q.copy(), args=(target, beta),
                           jac=_log_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _q_to_h(res.x, target)
                h_new = _balance_exact(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    q = res.x.copy()
                elif np.all(np.isfinite(res.x)):
                    h_new = _q_to_h(res.x, target)
                    q = res.x.copy()
        except Exception:
            pass
    return best_h, best_c5

def _thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if (cnt % 2 == 0) else 0.0
    return h

def _quasi_periodic(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0).astype(float)
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    return h

def _van_der_corput(n):
    h = np.zeros(n)
    for i in range(1, n + 1):
        x = 0.0
        f = 0.5
        ii = i
        while ii > 0:
            if ii % 2 == 1:
                x += f
            ii //= 2
            f *= 0.5
        h[i - 1] = 1.0 if x < 0.5 else 0.0
    return h

                        f2 = f1

                    else:

                        f1 = f2
