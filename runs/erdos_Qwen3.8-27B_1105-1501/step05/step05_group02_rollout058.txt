def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _scores(h):

def _project(v, target):

        if s > target:
            lo = mid
        else:
            hi = mid

        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)
    return x

    final = _exact_c5(h)
    if final < best_c5:

        best_h = h.copy()
    return best_h, best_c5

def _quasi_periodic(n, alpha):

def _multi_coord_search(h, deadline, n_coords=4, n_attempts=500, rng=None):
    """Move mass along n_coords positions simultaneously."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for att in range(n_attempts):
        if time.time() > deadline:
            break
        # Pick n_coords source and n_coords destination positions
        # Move mass from sources to destinations
        idx = rng.choice(n, size=2 * n_coords, replace=False)
        src = idx[:n_coords]
        dst = idx[n_coords:]
        
        # For each, move some mass t from one src to one dst
        # Simple: t = 0.1 for each pair
        t = 0.05
        valid = True
        h_new = h.copy()
        for s, d in zip(src, dst):
            if h_new[s] < t or h_new[d] > 1.0 - t:
                valid = False
                break
            h_new[s] -= t
            h_new[d] += t
        if not valid:
            continue
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    v = np.array(v, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_line_search(h, i, j, n_grid=21):
    hi_val = h[i]
    hj_val = h[j]
    t_min = max(-hi_val, hj_val - 1.0)
    t_max = min(1.0 - hi_val, hj_val)
    if t_max - t_min < 1e-14:
        return None
    cur_val = _exact_c5(h)
    best_t = 0.0
    best_val = cur_val
    ts = np.linspace(t_min, t_max, n_grid)

        h[i] = hi_val + t
        h[j] = hj_val - t
        val = _exact_c5(h)
        if val < best_val:

            best_t = t
    if best_t == 0.0:
        return None
    span = (t_max - t_min) / n_grid

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)

    f1 = _exact_c5(h)

def _hot_pairs(h, scores, n_pairs, rng, explore_frac=0.3):
    n = len(h)
    center = n - 1
    pairs = []
    n_hot = int(n_pairs * (1.0 - explore_frac))
    n_rand = n_pairs - n_hot
    topk = min(5, 2 * n - 2)
    top_idx = np.argsort(scores)[-topk:]
    for lag_idx in top_idx:
        d = lag_idx - center
        if d == 0:
            frac_mask = (h > 0.05) & (h < 0.95)
            frac_idx = np.where(frac_mask)[0]
            for k in range(min(4, len(frac_idx))):
                i = frac_idx[k]
                if h[i] > 0.5:

                else:
                    cands = np.where(h > 0.6)[0]
                if len(cands) > 0:
                    j = cands[rng.integers(len(cands))]
                    if j != i:
                        pairs.append((i, j))
        else:
            ad = abs(d)
            m = n - ad
            if m <= 0:
                continue
            contrib = h[:m] * (1.0 - h[ad:])
            top_c = np.argsort(contrib)[-min(6, m):]
            for k in top_c:
                i = k + ad
                j = k
                if i != j:
                    pairs.append((i, j))
    seen = set()
    unique_pairs = []
    for p in pairs:
        key = (min(p[0], p[1]), max(p[0], p[1]))
        if key not in seen:
            seen.add(key)
            unique_pairs.append(p)
    pairs = unique_pairs[:n_hot]
    for _ in range(n_rand):
        i = rng.integers(n)
        j = rng.integers(n)
        if i != j:
            pairs.append((int(i), int(j)))
    return pairs[:n_pairs]

def _weighted_grad(h, gamma):
    n = len(h)
    center = n - 1
    scores = _scores(h)
    max_s = np.max(scores)
    z = np.exp(gamma * (scores - max_s))
    sw = np.sum(z)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w = z / sw
    grad = w[center] * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[center + d]
        wn = w[center - d]
        if wp < 1e-15 and wn < 1e-15:
            continue
        grad[:m] += -wp * h[d:] + wn * (1.0 - h[d:])
        grad[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    grad -= grad.mean()
    return grad, w

def _grad_step(h, gamma, deadline=None):

    grad, w = _weighted_grad(h, gamma)
    gn = np.linalg.norm(grad)

        return None
    direction = -grad / gn
    alpha_max = 0.5
    for j in range(n):

            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:

    alpha_max = max(alpha_max, 1e-12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
