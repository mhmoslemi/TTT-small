def _compute_subgrad_weights(scores, beta=50):
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    return w

def _grad_from_weights(w, h):
    """Compute G^T w where G is the gradient matrix of scores w.r.t. h."""
    n = len(h)
    # G[r, j] = partial S_r / partial h_j
    # S_r corresponds to lag d = r - (n-1)
    # S_d = sum_k h[k]*(1-h[k+d])
    # partial S_d / partial h[j] = (1-h[j+d]) - h[j-d] (with boundary=0)
    
    n_lags = 2*n - 1
    grad = np.zeros(n)
    for r in range(n_lags):
        if w[r] < 1e-15:
            continue
        d = r - (n-1)
        for j in range(n):
            # partial S_d / partial h_j
            term = 0.0
            # h[j] appears as h[k] when k=j: contributes (1-h[j+d])
            if 0 <= j+d < n:
                term += (1.0 - h[j+d])
            # h[j] appears as h[k+d] when k=j-d: contributes -h[j-d]
            if 0 <= j-d < n:
                term -= h[j-d]
            grad[j] += w[r] * term
    return grad
