```python
def grad_lag(h, d, dx):
    """Gradient of F_d(h) w.r.t. h, where d is signed lag.
    d > 0: F_d = dx * sum_j h[j+d]*(1-h[j]), j=0..n-1-d
    d < 0: let a=-d, F_d = dx * sum_j h[j]*(1-h[j+a]), j=0..n-1-a
    d = 0: F_0 = dx * sum_j h[j]*(1-h[j])
    """
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        m = n - d
        # h[j+d] term: for i in d..n-1, contrib = dx*(1-h[i-d])
        g[d:] += dx * (1.0 - h[:m])
        # h[j] term: for i in 0..m-1, contrib = -dx*h[i+d]
        g[:m] += -dx * h[d:]
    else:
        a = -d
        m = n - a
        # h[j] term: for i in 0..m-1, contrib = dx*(1-h[i+a])
        g[:m] += dx * (1.0 - h[a:a+m])
        # h[j+a] term: for i in a..n-1, contrib = -dx*h[i-a]
        g[a:] += -dx * h[:m]
    return g
```

```python
        g[-1] = 1.0
        return g
    
    def constraint_fun(self, x):
        h = x[:-1]
        t = x[-1]
        result = np.zeros(len(self.active_lags))
        for i, d in enumerate(self.active_lags):

    def constraint_jac(self, x):
```

```python
    for i, d in enumerate(active_lags):
```

Wait, one more thought. The strategy mentions "active set equalizing" as a key new direction. Let me make sure I implement this well. The idea is:

Given the top K lags with scores s_1, s_2, ..., s_K (all close to the maximum), I want to find a direction d (with sum(d)=0) such that all K lags are reduced approximately equally.

Mathematically: find d minimizing max_i (∂s_i/∂h · d) subject to sum(d) = 0, ||d|| = 1.

This is a Chebyshev approximation problem. A simpler approximation is to find d minimizing ||G·d||² where G is the matrix of gradients (rows), with the constraint sum(d) = 0. But this minimizes the average, not the maximum.

A better heuristic: find d that makes G·d as uniform (and negative) as possible. This can be approximated by:
1. Find the minimum-norm d such that G·d = -δ·1 (reduce all by δ)
2. Use this as the search direction

The minimum-norm solution is d = G^T (G G^T)^{-1} (-δ·1), projected onto the zero-mean subspace.

Let me implement this properly:

```python
def equalize_dir(h, active_lags, dx, delta=0.005):
    n = len(h)
    K = len(active_lags)
    if K == 0:
        return None
    
    G = np.zeros((K, n))
    for i, d in enumerate(active_lags):
        G[i] = grad_lag(h, d, dx)
    
    # Project G onto zero-mean subspace
    G_proj = G - G.mean(axis=1, keepdims=True)
    
    # Find d = G_proj^T @ y where G_proj @ d = -delta
    # => G_proj @ G_proj^T @ y = -delta * ones
    A = G_proj @ G_proj.T + 1e-10 * np.eye(K)
    b = -delta * np.ones(K)
    
    try:
        y = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        y = np.linalg.lstsq(A, b, rcond=None)[0]
    
    d = G_proj.T @ y
    d -= d.mean()
    
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    d /= norm
    return d
```

This gives a direction that tries to reduce all active lags by approximately `delta` (in unscaled units). The actual step size is then determined by line search.

```python
class SLSQPWrapper:
    def __init__(self, h0, active_lags, dx, target, radius):
        self.h0 = h0
        self.active_lags = active_lags
        self.dx = dx
        self.target = target
        self.radius = radius
        self.n = len(h0)

    def objective(self, x):
        return x[-1]

    def objective_grad(self, x):
        g = np.zeros(x.shape)
        g[-1] = 1.0
        return g

    def ineq_fun(self, x):
        h = x[:-1]
        t = x[-1]
        result = np.empty(len(self.active_lags))
        for i, d in enumerate(self.active_lags):
            result[i] = t - score_lag(h, d, self.dx)
        return result

    def ineq_jac(self, x):
        h = x[:-1]
        n = self.n
        K = len(self.active_lags)
        J = np.zeros((K, n + 1))
        for i, d in enumerate(self.active_lags):
            J[i, :-1] = -grad_lag(h, d, self.dx)
            J[i, -1] = 1.0
        return J

    def eq_fun(self, x):
        return np.array([np.sum(x[:-1]) - self.target])

    def eq_jac(self, x):
        J = np.zeros((1, self.n + 1))
        J[0, :self.n] = 1.0
        return J

def slsqp_step(h, active_lags, dx, target, radius, maxiter=40):
    n = len(h)
    t_init = exact_c5(h) + 0.002
    wrapper = SLSQPWrapper(h, active_lags, dx, target, radius)
    x0 = np.zeros(n + 1)
    x0[:-1] = h
    x0[-1] = t_init
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        if lo > hi:
            lo = hi = h[i]
        bounds.append((lo, hi))
    bounds.append((0.0, t_init + 0.01))
    constraints = [
        {'type': 'ineq', 'fun': wrapper.ineq_fun, 'jac': wrapper.ineq_jac},
        {'type': 'eq', 'fun': wrapper.eq_fun, 'jac': wrapper.eq_jac}
    ]
    try:
        res = minimize(
            wrapper.objective, x0, method='SLSQP',
            jac=wrapper.objective_grad,
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-11}
        )
    except Exception:
        return None
    if res.success and res.x is not None:
        return res.x[:-1]
    return None

def select_active_lags(scores, n, k_max=20, thresh_frac=0.995):
    s_max = float(np.max(scores))
    thresh = s_max * thresh_frac
    idx = np.where(scores >= thresh)[0]
    if len(idx) > k_max:
        sorted_idx = np.argsort(scores)[::-1]
        idx = sorted_idx[:k_max]
    return [(int(i) - (n - 1)) for i in idx.tolist()]

def equalize_dir(h, active_lags, dx, delta=0.004):
    n = len(h)
    K = len(active_lags)
    if K == 0:
        return None
    G = np.zeros((K, n))
    for i, d in enumerate(active_lags):
        G[i] = grad_lag(h, d, dx)

    A = G_proj @ G_proj.T + 1e-10 * np.eye(K)

    d_vec -= d_vec.mean()

def smooth_grad(h, dx, beta=80.0):

            continue
        d = r - (n - 1)
        grad += w[r] * grad_lag(h, d, dx)
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return grad / norm

            if c5_try < best_c5 - 1e-11:
                h_new = h + alpha * d
                h_new = project(h_new, target)
                c5_new = exact_c5(h_new)
                if c5_new < best_c5 - 1e-11:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    improved = True
                    if improved:
                        break

        if improved:
            stall = 0
            continue

        stall += 1
        if stall > 150:
            break

    return best_h, best_c5

def multi_res_phase(best_h, best_c5, deadline):
    n_best = len(best_h)
    c_best = best_c5
    h_best = best_h.copy()
    tried = set()

    offsets = []
    for dn in range(-30, 31, 4):
        if dn != 0:
            offsets.append(dn)
    # Sort by absolute value
    offsets.sort(key=lambda x: abs(x))

    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 250:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)

        h_r = resample_h(h_best, n_try)
        c5_r = exact_c5(h_r)

        # Quick SLSQP pass
        dx_r = 2.0 / n_try
        target_r = n_try / 2.0
        scores_r = np.correlate(h_r, 1.0 - h_r, mode='full') * dx_r
        active_r = select_active_lags(scores_r, n_try, k_max=15, thresh_frac=0.996)
        sub_deadline = min(deadline, time.time() + 15.0)

        h_r2, c5_r2 = slsqp_phase(h_r, sub_deadline)
        if c5_r2 < c_best - 1e-10:
            c_best = c5_r2
            h_best = h_r2.copy()
            n_best = n_try

    return h_best, c_best

def kick_reopt(h, deadline, rng, n_kicks=25):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = exact_c5(h)

    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(0.015, 0.06)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = best_h + scale * direction
        h_k = project(h_k, target)

        # Quick line search refinement
        scores_k = np.correlate(h_k, 1.0 - h_k, mode='full') * dx
        active_k = select_active_lags(scores_k, n, k_max=10, thresh_frac=0.997)

        sub_dl = min(deadline, time.time() + 8.0)
        h_t, c5_t = slsqp_phase(h_k, sub_dl)
        if c5_t < best_c5 - 1e-10:
            best_c5 = c5_t
            best_h = h_t.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)

    # Get initial values
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)

    # Phase 1: SLSQP active set optimization (45%)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = slsqp_phase(best_h, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution (20%)
    d2 = t0 + budget_s * 0.65
    h2, c5_2 = multi_res_phase(best_h, best_c5, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Line search refinement (15%)
    d3 = t0 + budget_s * 0.80
    h3, c5_3 = line_refine_phase(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Kick and reoptimize (10%)
    d4 = t0 + budget_s * 0.90
    h4, c5_4 = kick_reopt(best_h, d4, rng, n_kicks=20)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Final SLSQP polish (10%)
    d5 = hard_deadline
    h5, c5_5 = slsqp_phase(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if not np.all(np.isfinite(best_h)) or np.any(best_h < -1e-10) or np.any(best_h > 1.0 + 1e-10):
        best_h = project(best_h, len(best_h) / 2.0)
        final_c5 = exact_c5(best_h)
    return best_h, final_c5, len(best_h)
```
