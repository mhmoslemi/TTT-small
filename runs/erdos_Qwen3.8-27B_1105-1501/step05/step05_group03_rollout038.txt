def _slsqp_optimize(h, deadline, n_active=8, max_rounds=40):

    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n_act = n_active

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        raw = _raw_corr(h)
        scores = raw * dx
        max_s = float(np.max(scores))

        top_idx = np.argsort(scores)[::-1][:n_act]
        eq_c = _EqConstraint(n, target)
        constraints = [
            {'type': 'eq', 'fun': eq_c.fun, 'jac': eq_c.jac}
        ]
        ineq_objs = []
        for r in top_idx:
            c_obj = _IneqConstraint(n, int(r), dx)
            ineq_objs.append(c_obj)
            constraints.append({'type': 'ineq', 'fun': c_obj.fun, 'jac': c_obj.jac})

        x0 = np.concatenate([h, [max_s * 1.01]])
        bounds = [(0.0, 1.0)] * n + [(max_s * 0.3, None)]

        res = _slsqp_step(_epigraph_obj, x0, bounds, constraints)

        if res is not None and res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                if best_c5 < 0.3805:
                    break
            else:
                if n_act > 3:
                    n_act = max(3, n_act - 1)
                else:
                    break
        else:
            if n_act > 3:
                n_act = max(3, n_act - 2)
            else:
                break

    return best_h, best_c5

def _lp_direction_reduced(h, scores, radius, n_top):
    n = len(h)
    n_lags = len(scores)
    k = min(n_top, n_lags)
    top_idx = np.argsort(scores)[::-1][:k]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    for i, r in enumerate(top_idx):
        d = int(r - (n - 1))
        A_ub[i, :n] = _grad_row(h, d, 1.0)
        A_ub[i, n] = -1.0
    b_ub = -scores[top_idx]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and float(np.max(np.abs(res.x[:n]))) > 1e-14:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_direction_all(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    for i in range(n_lags):
        d = i - (n - 1)
        A_ub[i, :n] = _grad_row(h, d, 1.0)
        A_ub[i, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -radius)
        hi = min(1.0 - h[j], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and float(np.max(np.abs(res.x[:n]))) > 1e-14:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, target, n_eval=35):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.95
    if alpha_max > 5.0:
        alpha_max = 5.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(n_eval):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_h = _project(h + opt_a * p, target)
    opt_c5 = _exact_c5(opt_h)
    best_c5 = min(f1, f2, opt_c5)
    if opt_c5 <= best_c5:
        return opt_h, opt_c5
    if f1 <= f2:
        return _project(h + c1 * p, target), f1
    return _project(h + c2 * p, target), f2

def _lp_optimize_reduced(h, deadline, radius_init=0.15, n_top=5, max_iters=3000, radius_min=0.001):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        p = _lp_direction_reduced(h, scores, radius, n_top)
        if p is None:
            p = _lp_direction_all(h, scores, radius * 0.5)
        if p is None:
            stall += 1
            radius *= 0.6
            if radius < radius_min:
                break
            continue
        h_new, c5_new = _line_search(h, p, target)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
    return best_h, best_c5

def _subgrad_descent(h, deadline, n_iters=300, step_init=0.8, tol=1e-9):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    momentum = np.zeros(n)
    beta = 0.85
    for it in range(n_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = float(np.max(scores))
        mask = scores >= max_s - 1e-7
        active = np.where(mask)[0]
        grad = np.zeros(n)
        for r in active:
            d = int(r - (n - 1))
            grad += _grad_row(h, d, dx)
        grad /= len(active)
        momentum = beta * momentum - step * grad
        h_new = _project(h + momentum, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.03, 3.0)
        else:
            momentum = 0.5 * momentum
            step *= 0.7
            if step < 1e-7:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.02, n)
        h = _project(h, n / 2.0)
    return h

def _modulated_init(n, alpha, beta_f, rng=None):
    x = np.arange(n) / n
    vals = (x * alpha) % 1.0
    thresh = 0.5 + 0.1 * np.sin(2 * np.pi * beta_f * x)
    h = (vals < thresh).astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.03, n)
        h = _project(h, n / 2.0)
    return h

def _smoothed_grad(h, deadline, beta_start=10, beta_end=200, n_steps=200, step_init=0.05):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            idx = np.argmax(scores)
            w = np.zeros(len(scores))
            w[idx] = 1.0
        else:
            w /= sw
        grad = np.zeros(n)
        for d in range(-(n - 1), n):
            r = n - 1 + d
            grad += w[r] * _grad_row(h, d, 1.0)
        grad -= grad * (np.sum(grad) / n)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.02, 0.15)
        else:
            v = 0.5 * v
            step *= 0.83
            if step < 1e-7:
                break
    return best_h, best_c5

def _coordinate_refine(h, deadline, n_cycles=2):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            best_j = h[j]
            best_f = best_c5
            for trial in range(5):
                t = (trial + 1) / 6.0
                h_try = h.copy()
                h_try[j] = t
                h_try = _project(h_try, target)
                f_try = _exact_c5(h_try)
                if f_try < best_f:
                    best_f = f_try
                    best_j = t
            if best_f < best_c5 - 1e-13:
                h[j] = best_j
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Reduced-constraint LP from parent (top 5)
    if time.time() < t0 + budget_s * 0.12:
        dl = t0 + budget_s * 0.12
        h1, c5_1 = _lp_optimize_reduced(h_start, dl, radius_init=0.2, n_top=5, max_iters=5000, radius_min=0.001)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: SLSQP epigraph polishing
    if time.time() < t0 + budget_s * 0.22:
        dl = t0 + budget_s * 0.22
        h2, c5_2 = _slsqp_optimize(best_h, dl, n_active=8, max_rounds=30)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Subgradient + LP refinement
    if time.time() < t0 + budget_s * 0.32:
        dl = t0 + budget_s * 0.28
        h3, c5_3 = _subgrad_descent(best_h, dl, n_iters=200, step_init=0.5)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        if time.time() < t0 + budget_s * 0.35:
            dl3b = t0 + budget_s * 0.35
            h3b, c5_3b = _lp_optimize_reduced(best_h, dl3b, radius_init=0.1, n_top=4, max_iters=3000, radius_min=0.001)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.55:
        dl4 = t0 + budget_s * 0.55
        resolutions = [96, 128, 80, 64, 128, 96]
        alphas = [0.6180339887, 0.6180339887, 0.4142135624, 0.6180339887, 0.5, 0.3819660113]
        for ri, (n_try, alpha) in enumerate(zip(resolutions, alphas)):
            if time.time() > dl4:
                break
            if best_c5 < 0.3806:
                break
            sub_dl = min(dl4, time.time() + budget_s * 0.035)
            if n_try != best_n:
                if ri % 2 == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    h_init = _quasi_periodic_init(n_try, alpha, rng)
            else:
                h_init = best_h + rng.normal(0, 0.03, n_try)
                h_init = _project(h_init, n_try / 2.0)
            h_try, c5_try = _lp_optimize_reduced(h_init, sub_dl, radius_init=0.15, n_top=5, max_iters=2000, radius_min=0.003)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 5: SLSQP on best (possibly different resolution)
    if time.time() < t0 + budget_s * 0.65:
        dl5 = t0 + budget_s * 0.65
        h5, c5_5 = _slsqp_optimize(best_h, dl5, n_active=6, max_rounds=30)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        # Follow with LP
        if time.time() < t0 + budget_s * 0.70:
            dl5b = t0 + budget_s * 0.70
            h5b, c5_5b = _lp_optimize_reduced(best_h, dl5b, radius_init=0.06, n_top=4, max_iters=3000, radius_min=0.001)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: Smoothed gradient exploration
    if time.time() < t0 + budget_s * 0.78:
        dl6 = t0 + budget_s * 0.78
        h6, c5_6 = _smoothed_grad(best_h, dl6, beta_start=10, beta_end=150, n_steps=200, step_init=0.04)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        if time.time() < t0 + budget_s * 0.82:
            dl6b = t0 + budget_s * 0.82
            h6b, c5_6b = _slsqp_optimize(best_h, dl6b, n_active=5, max_rounds=20)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    # Phase 7: Perturbation restarts
    dl7 = t0 + budget_s * 0.92
    scale = 0.03
    for attempt in range(400):
        if time.time() > dl7:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.01)
        h_try, c5_try = _lp_optimize_reduced(h_pert, sub_dl, radius_init=0.05, n_top=4, max_iters=800, radius_min=0.005)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 8: Modulated seeds at different resolutions
    if time.time() < t0 + budget_s * 0.94:
        dl8 = t0 + budget_s * 0.94
        for n_m in [128, 96]:
            if time.time() > dl8:
                break
            for beta_f in [1.0, 2.0, 3.0]:
                if time.time() > dl8:
                    break
                h_m = _modulated_init(n_m, 0.6180339887, beta_f, rng)
                sub_dl = min(dl8, time.time() + budget_s * 0.012)
                h_try, c5_try = _lp_optimize_reduced(h_m, sub_dl, radius_init=0.12, n_top=5, max_iters=1500, radius_min=0.005)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_m

    # Phase 9: Final SLSQP + coordinate refine
    if time.time() < t0 + budget_s * 0.97:
        dl9 = t0 + budget_s * 0.97
        h9, c5_9 = _slsqp_optimize(best_h, dl9, n_active=5, max_rounds=20)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Phase 10: Coordinate refine
    if time.time() < t0 + budget_s * 0.995:
        dl10 = t0 + budget_s * 0.995
        h10, c5_10 = _coordinate_refine(best_h, dl10, n_cycles=2)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
