def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

def _subgradient(h, n, lags, weights):
    """Compute weighted subgradient of scores at given lags."""
    grad = np.zeros(n)
    for k, w in zip(lags, weights):

            if m > 0:
                grad[g:] += w * (1.0 - h[:m])

def _best_pair_move(h, n):
    """Find the best single pair move."""
    target = n / 2.0
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    cur_max = np.max(scores) * dx
    
    best_score = cur_max
    best_move = None
    
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            # Try moving mass from i to j (h[i] decreases, h[j] increases)
            # and from j to i
            for sign in [1, -1]:
                if sign > 0:
                    # h[i] decreases, h[j] increases
                    t_max = min(h[i], 1-h[j], 0.5)
                else:
                    # h[i] increases, h[j] decreases
                    t_max = min(1-h[i], h[j], 0.5)
                
                if t_max < 0.01:
                    continue
                
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    t = t_max * frac
                    if sign > 0:
                        h_try = h.copy()
                        h_try[i] -= t
                        h_try[j] += t
                    else:
                        h_try = h.copy()
                        h_try[i] += t
                        h_try[j] -= t
                    # Clip to [0,1]
                    h_try = np.clip(h_try, 0, 1)
                    new_max = np.max(np.correlate(h_try, 1-h_try, mode='full')) * dx

                        best_move = (i, j, sign, t)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval_pair(h, n, i, j, t):
    """Evaluate C5 after moving t from i to j (h[i]-=t, h[j]+=t)."""
    hi = h[i] - t
    hj = h[j] + t
    if hi < -1e-12 or hj > 1 + 1e-12 or hi > 1 + 1e-12 or hj < -1e-12:
        return 1e9
    h2 = h.copy()
    h2[i] = hi
    h2[j] = hj
    return float(np.max(_scores(h2)))

def _delta_for_pair(h, n, i, j):
    """Compute delta_k for changing h[i]+t, h[j]-t. Linear in t for k!=0."""
    scores = _scores(h)
    delta = np.zeros(2 * n - 1)
    n1 = n - 1
    for r in range(2 * n - 1):
        k = r - n1
        # Contribution from position i (increasing by t)
        val_i = 0.0
        if 0 <= i + k < n:
            val_i += (1.0 - h[i + k])
        if 0 <= i - k < n:
            val_i -= h[i - k]
        # Contribution from position j (decreasing by t)
        val_j = 0.0
        if 0 <= j + k < n:
            val_j += (1.0 - h[j + k])
        if 0 <= j - k < n:
            val_j -= h[j - k]
        delta[r] = val_i - val_j
    return scores, delta

def _optimal_t(scores, delta, t_min, t_max, n_samples=12):
    """Find best t in [t_min, t_max] minimizing max(scores + t*delta)."""
    if t_max - t_min < 1e-12:
        return 0.0, float(np.max(scores))
    ts = np.linspace(t_min, t_max, n_samples)
    best_t = 0.0
    best_val = float(np.max(scores))
    for t in ts:
        val = float(np.max(scores + t * delta))
        if val < best_val:
            best_val = val
            best_t = t
    # Refine with golden section around best
    lo = max(t_min, best_t - (t_max - t_min) / n_samples)
    hi = min(t_max, best_t + (t_max - t_min) / n_samples)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

def _subgrad_step(h, n, top_m=4, alpha=0.03):
    """One subgradient step focusing on top lags."""
    target = n / 2.0
    scores = _scores(h)
    idx = np.argsort(scores)[::-1][:top_m]
    peak = scores[idx]
    base = peak[-1]
    weights = (peak - base) + 1e-8
    weights = weights / np.sum(weights)
    
    lags = idx - (n - 1)
    grad = np.zeros(n)
    for r, w in zip(idx, weights):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                grad[:m] += w * (1.0 - h[k:])
            if k < n:
                grad[k:] += -w * h[:n - k]
        else:
            g = -k
            m = n - g
            if m > 0:
                grad[g:] += w * (1.0 - h[:m])
            if g < n:
                grad[:m] += -w * h[g:]
    
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, _exact_c5(h)
    grad = grad / gn
    h_new = _project(h - alpha * grad * gn, target)
    return h_new, _exact_c5(h_new)

def _lp_direction(h, n, radius=0.15):
    """LP-based direction finding."""
    scores = _scores(h)
    # Build gradient matrix
    G = np.zeros((2 * n - 1, n))
    n1 = n - 1
    for r in range(2 * n - 1):
        k = r - n1
        if k >= 0:
            m = n - k
            if m > 0:
                G[r, :m] += (1.0 - h[k:])
            if k < n:
                G[r, k:] += -h[:n - k]
        else:
            g = -k
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
            if g < n:
                G[r, :m] += -h[g:]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, trust_radius):

    S0 = _scores(h)
    T = _scores(d) - _scores(h) * 0  # placeholder

    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        n_best = len(best_h)
    
    # Block moves
    sub_dl = min(d4, time.time() + budget_s * 0.03)
    h4 = best_h.copy()
    h4, c5_4 = _block_move_pass(h4, n_best, sub_dl, block_size=2)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        n_best = len(best_h)
    
    # More LP
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4 = best_h.copy()
    h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.04)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        n_best = len(best_h)
    
    # Perturbation
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4 = best_h.copy()
    h4, c5_4 = _perturb_reopt(h4, sub_dl, rng, scale_init=0.03, n_attempts=100)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        n_best = len(best_h)
    
    # Phase 5: Try random starts at smaller n (10%)
    d5 = t0 + budget_s * 0.90
    for trial in range(8):
        if time.time() > d5:
            break
        n_try = max(64, n_best - 40 - int(rng.integers(0, 30)))
        h_r = _random_binary(n_try, rng)
        sub_dl = min(d5, time.time() + max(2, (d5 - time.time()) / max(8 - trial, 1)))
        h_r, c5_r = _best_pair_pass(h_r, n_try, sub_dl, focus_peak=True)
        if time.time() < d5 and c5_r < best_c5 + 0.005:
            sub_dl2 = min(d5, time.time() + 3)
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.1)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 6: Final polish (10%)
    d6 = deadline
    n_final = len(best_h)
    sub_dl = min(d6, time.time() + budget_s * 0.04)
    h6 = best_h.copy()
    h6, c5_6 = _best_pair_pass(h6, n_final, sub_dl, focus_peak=False)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    sub_dl = min(d6, time.time() + budget_s * 0.03)
    h6 = best_h.copy()
    h6, c5_6 = _lp_loop(h6, sub_dl, radius_init=0.03)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    sub_dl = min(d6, time.time() + budget_s * 0.02)
    h6 = best_h.copy()
    h6, c5_6 = _perturb_reopt(h6, sub_dl, rng, scale_init=0.015, n_attempts=50)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
