        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute the matrix G such that scores = G @ h (for the bilinear form).
    Actually, scores[d] = sum_i h[i]*(1-h[i+d]) * dx for valid i.
    The gradient of scores[d] w.r.t. h[j] is:
    - h[j+d]*dx if j is in the overlap for the first factor
    + (1-h[j-d])*dx if j is in the overlap for the second factor
    ... this is getting complicated. Let me just compute it directly.
    """
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] += (1.0 - h[d:]) * dx
            G[r, d:] += (-h[:m]) * dx
        else:
            g = -d
            m = n - g
            G[r, g:] += (1.0 - h[:m]) * dx
            G[r, :m] += (-h[g:]) * dx
    return G

def _soft_score(h, beta):
    """Compute the log-sum-exp approximation to max(scores)."""
    s = _scores(h)
    m = np.max(s)
    if beta < 1e-6:
        return m
    exps = np.exp(beta * (s - m))
    return m + np.log(np.sum(exps)) / beta

def _soft_grad(h, beta):
    """Compute gradient of the soft max score."""
    s = _scores(h)
    m = np.max(s)
    if beta < 1e-6:
        # Subgradient: gradient of the max score
        idx = np.argmax(s)
        G = _grad_matrix(h)
        return G[idx]
    exps = np.exp(beta * (s - m))
    w = exps / np.sum(exps)
    G = _grad_matrix(h)
    return G.T @ w

def _smooth_optimize(h, deadline, beta_start=50, beta_end=500, n_steps=200, step_init=0.05):
    """Smooth gradient descent with momentum on the log-sum-exp objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_score = _soft_score(h, beta_start)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = step_init
    n_beta_steps = max(1, n_steps // 10)
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        grad = _soft_grad(h, beta)
        grad -= grad.mean()  # maintain sum constraint
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = 0.9 * v - step * grad
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Accept if soft score improved or exact C5 improved
        soft_new = _soft_score(h_new, beta)
        if soft_new < cur_score and c5_new <= best_c5 + 0.01:
            h = h_new
            try:
                cur_score = soft_new
            except:
                cur_score = _soft_score(h, beta)
            step = min(step * 1.02, 0.15)
        else:
            v = 0.5 * v
            step *= 0.8
            if step < 1e-7:
                break
    return best_h, best_c5

def _lp_optimize(h, deadline, radius_init=0.15, max_iters=5000, radius_min=0.001):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        # LP: find direction p that decreases all scores simultaneously
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_lags = len(scores)
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-h[j], -radius)
            hi = min(1.0 - h[j], radius)
            bounds.append((lo, hi))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success:
                p = res.x[:n]
                if float(np.max(np.abs(p))) > 1e-14:
                    # Line search
                    alpha_max = 1.0
                    for j in range(n):
                        if p[j] > 1e-14:
                            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                        elif p[j] < -1e-14:
                            alpha_max = min(alpha_max, h[j] / (-p[j]))
                    alpha_max = max(alpha_max, 1e-12) * 0.95
                    # Golden section
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    lo_a, hi_a = 0.0, alpha_max
                    c1 = hi_a - gr * (hi_a - lo_a)
                    c2 = lo_a + gr * (hi_a - lo_a)
                    f1 = _exact_c5(_project(h + c1 * p, target))
                    f2 = _exact_c5(_project(h + c2 * p, target))
                    for _ in range(25):
                        if hi_a - lo_a < 1e-12:
                            break
                        if f1 < f2:
                            hi_a = c2
                            f2 = f1
                            c2 = c1
                            c1 = hi_a - gr * (hi_a - lo_a)
                            f1 = _exact_c5(_project(h + c1 * p, target))
                        else:
                            lo_a = c1
                            f1 = f2
                            c1 = c2
                            c2 = lo_a + gr * (hi_a - lo_a)
                            f2 = _exact_c5(_project(h + c2 * p, target))
                    opt_a = (lo_a + hi_a) / 2.0
                    opt_h = _project(h + opt_a * p, target)
                    opt_c5 = _exact_c5(opt_h)
                    if opt_c5 < f1:
                        best_a, best_c5_try = opt_a, opt_c5
                    elif f1 < f2:
                        best_a, best_c5_try = c1, f1
                    else:
                        best_a, best_c5_try = c2, f2
                    
                    if best_c5_try < cur_c5 - 1e-14:
                        h = _project(h + best_a * p, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.05, 0.35)
                    else:
                        stall += 1
                        radius *= 0.65
                        if radius < radius_min:
                            break
                    continue
        except Exception:
            pass
        stall += 1
        radius *= 0.65
        if radius < radius_min:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h = _project(h, n / 2.0)
    if rng is not None:
        h += rng.normal(0, 0.05, n)
        h = _project(h, n / 2.0)
    return h

def _random_init(n, rng):
    h = rng.random(n)
    return _project(h, n / 2.0)

def _fourier_init(n, rng, K=5):
    x = np.arange(n) / n
    coeffs = rng.normal(0, 0.3, 2*K)
    h_raw = np.zeros(n) + 0.5
    for k in range(1, K+1):
        h_raw += coeffs[2*(k-1)] * np.cos(2*np.pi*k*x)
        h_raw += coeffs[2*(k-1)+1] * np.sin(2*np.pi*k*x)
    return _project(h_raw, n / 2.0)

def _coordinate_polish(h, deadline, n_cycles=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            lo_v, hi_v = 0.0, 1.0
            for _ in range(35):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0
                h1 = h.copy()
                h1[j] = m1
                h1 = _project(h1, target)
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = m2
                h2 = _project(h2, target)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi_v = m2
                else:
                    lo_v = m1
            v_opt = (lo_v + hi_v) / 2.0
            h_try = h.copy()
            h_try[j] = v_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_polish(h, deadline, n_passes=4):
    """Try transferring mass between pairs of coordinates."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_size = max(1, n // 16)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + step_size, min(j + 2*step_size, n), max(1, step_size//4)):
                if k >= n:
                    break
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(12):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(n, deadline, rng, n_iters=30000):
    """Simulated annealing on binary sequences."""
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.02
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cool
    return best_h, best_c5
