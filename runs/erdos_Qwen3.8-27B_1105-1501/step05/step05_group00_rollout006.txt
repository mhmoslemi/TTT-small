    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _pair_line_search(h, i, j, dx):
    n = len(h)
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-14:
        return h[i], h[j], h
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    alpha_lo = lo - h[i]
    alpha_hi = hi - h[i]
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    n_grid = 40
    grid = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_a = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = alpha_lo, alpha_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_a = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_a * T - opt_a * opt_a * C)) * dx
    if opt_val < best_val:
        best_a, best_val = opt_a, opt_val
    x_new = h[i] + best_a
    y_new = h[j] - best_a
    x_new = np.clip(x_new, 0.0, 1.0)
    y_new = np.clip(y_new, 0.0, 1.0)
    return x_new, y_new, best_val

def _gradient_contrib(h, active_indices, dx):
    n = len(h)
    grad = np.zeros(n)
    for k_idx in active_indices:
        d = k_idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += (1.0 - h[d:])
                grad[d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += -h[g:]
                grad[g:] += (1.0 - h[:m])
    grad *= dx
    return grad

def _lp_direction(h, scores, active_tol, radius):
    n = len(h)
    M = float(np.max(scores))
    slacks = M - scores
    active = np.where(slacks <= active_tol)[0]
    if len(active) == 0:
        active = np.where(slacks <= active_tol * 5.0)[0]
    if len(active) == 0:
        return None
    if len(active) > 25:
        order = np.argsort(slacks[active])[:25]
        active = active[order]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    G = np.zeros((len(active), n + 1))
    for r, k_idx in enumerate(active):
        d = k_idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += -(1.0 - h[:m]) * 0  # placeholder
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += -h[g:]
                G[r, g:] += (1.0 - h[:m])
        G[r, n] = -1.0
    b_ub = -(scores[active] - M)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=G, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(scores))
        slacks = M - scores
        active_tol = max(1e-6, 0.03 * float(np.median(slacks[slacks > 0])) if np.any(slacks > 0) else 1e-6)
        p = _lp_direction(h, scores * dx, active_tol, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search_exact(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            predicted = float(np.max(scores)) + float(np.max(np.correlate(p, 1.0-h, mode='full') - np.correlate(h, p, mode='full'))) * 0 + float(np.max(scores))
            actual_ratio = cur_c5 / max(M, 1e-15)
            if actual_ratio < 0.85:
                radius = min(radius * 1.1, 0.35)
            elif actual_ratio > 1.3:
                radius *= 0.8
            else:
                radius = min(radius * 1.02, 0.30)
        else:
            stall += 1
            radius *= 0.7
            if stall > 40:
                if radius < 0.02:
                    break
                radius = 0.08
                stall = 0
    return best_h, best_c5

def _pair_descent(h, deadline, max_sweeps=200, top_k=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall_global = 0
    rng = np.random.default_rng(4242)
    while time.time() < deadline and stall_global < 15:
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(scores))
        active_tol = max(1e-5, 0.05 * M)
        active_mask = scores >= M - active_tol
        active_indices = np.where(active_mask)[0]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(scores))])
        grad = _gradient_contrib(h, active_indices, dx)
        grad_sum = float(np.sum(grad))
        scores2 = grad - (grad_sum / n)
        idx_sorted = np.argsort(-np.abs(scores2))
        tried = 0
        improved_loc = False
        candidates = idx_sorted[:min(len(idx_sorted), max_sweeps * 2)]
        for ii in range(min(len(candidates), max_sweeps)):
            if time.time() > deadline:
                return best_h, best_c5
            i = int(candidates[ii])
            if tried % 7 == 0:
                j = int(candidates[(ii + len(candidates) // 2) % len(candidates)])
                if j == i:
                    j = (i + 1) % n
            else:
                j = (i + 1 + int(tried % 5)) % n
            if i == j:
                continue
            tried += 1
            old_c5 = _exact_c5(h)
            x_new, y_new, _ = _pair_line_search(h, i, j, dx)
            h[i] = x_new
            h[j] = y_new
            new_c5 = _exact_c5(h)
            if new_c5 < old_c5 - 1e-15:
                improved_loc = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = old_h_i = h[i]
                # revert
                pass
        if not improved_loc:
            stall_global += 1
            # Perturb
            if stall_global < 10:
                h = h + rng.normal(0, 0.005, n)
                h = _project(h, target)
        else:
            stall_global = 0
    return best_h, best_c5

def _pair_descent_v2(h, deadline, max_sweeps=100):
    """More targeted pair descent: only pairs where one is high and other is low."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    rng = np.random.default_rng(31337)
    while time.time() < deadline and stall < 20:
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(scores))
        active_mask = scores >= M - max(1e-5, 0.05 * M)
        active_indices = np.where(active_mask)[0]
        grad = _gradient_contrib(h, active_indices, dx)
        grad_adj = grad - float(np.mean(grad))
        # Top increase and top decrease coordinates
        inc = np.argsort(-grad_adj)[:max_sweeps]
        dec = np.argsort(grad_adj)[:max_sweeps]
        improved = False
        attempts = 0
        for i in inc[:max_sweeps // 2]:
            if time.time() > deadline:
                return best_h, best_c5
            for j in dec[:max_sweeps // 2]:
                if i == j:
                    continue
                if attempts > max_sweeps:
                    break
                attempts += 1
                old_i, old_j = h[i], h[j]
                x_new, y_new, pred_c5 = _pair_line_search(h, int(i), int(j), dx)
                h[int(i)] = x_new
                h[int(j)] = y_new
                new_c5 = _exact_c5(h)
                if new_c5 < best_c5 - 1e-15:
                    improved = True
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    h[int(i)] = old_i
                    h[int(j)] = old_j
        if not improved:
            stall += 1
            h = h + rng.normal(0, 0.003, n)
            h = _project(h, target)
        else:
            stall = 0
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng, n_iter_max=500000):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.006
    T_min = 0.000003
    cooling = 0.99985
    patience = 0
    max_patience = 15000
    total = 0
    while time.time() < deadline:
        if patience > max_patience:
            T = max(T_min, T * 0.97)
            patience = 0
        if T < T_min:
            break
        batch = 200
        for _ in range(batch):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if h[i] == h[j]:
                continue
            # swap
            h[i], h[j] = h[j], h[i]
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            total += 1
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                h[i], h[j] = h[j], h[i]
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start_lp(h_starts, deadline, per_start_time=5.0):
    best_h = None
    best_c5 = float('inf')
    for s in h_starts:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + per_start_time)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _random_init(n, rng):
    h = rng.random(n)
    return _project(h, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 8))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97  # leave 3% for validation

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _block_init(n0, rng)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: LP polish on initial (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair descent (20%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _pair_descent_v2(best_h, d2, max_sweeps=80)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP again (10%)
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.08)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA (20%)
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = _binary_sa(best_h, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Polish binary with LP
    if time.time() < d4 + budget_s * 0.03:
        d4b = min(t0 + budget_s * 0.63, deadline)
        h4b, c5_4b = _lp_loop(best_h, d4b, radius_init=0.06)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Pair descent again (10%)
    d5 = t0 + budget_s * 0.73
    h5, c5_5 = _pair_descent_v2(best_h, d5, max_sweeps=100)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (12%)
    d6 = t0 + budget_s * 0.85
    best_n = len(best_h)
    offsets = [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 250 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10)
        if c5_r < best_c5 and time.time() < d6 - 10:
            d6b = min(d6, time.time() + budget_s * 0.012)
            h_r2, c5_r2 = _pair_descent_v2(h_r, d6b, max_sweeps=50)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
            if time.time() < d6:
                h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + 5), radius_init=0.05)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Aggressive perturbation + re-opt (7%)
    d7 = t0 + budget_s * 0.92
    scale = 0.03
    for attempt in range(300):
        if time.time() > d7:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.06)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.2)

    # Phase 8: Final pair descent polish (5%)
    d8 = deadline
    h8, c5_8 = _pair_descent_v2(best_h, d8, max_sweeps=120)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final LP polish
    if time.time() < deadline - 2:
        hf, cf = _lp_loop(best_h, deadline, radius_init=0.03)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
