    x = np.arange(n) / n
    idx = 0
    # Low-freq sinusoidal modes
    for k in range(1, min(8, q//2 + 1)):

        cos_c = np.cos(2 * np.pi * k * x) - np.mean(np.cos(2 * np.pi * k * x))
        sin_c = np.sin(2 * np.pi * k * x) - np.mean(np.sin(2 * np.pi * k * x))
        nc = np.linalg.norm(cos_c)
        ns = np.linalg.norm(sin_c)

        idx += 2

    while idx < q:

def _sa_block_moves(h, deadline, rng, n_iters=50000, T_init=0.02, T_min=0.0003):
    """Simulated annealing with block-level moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    T = T_init
    cool = (T_min / T_init) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Choose move type
        move_type = rng.integers(3)
        if move_type == 0:
            # Single element change
            i = rng.integers(n)
            delta = rng.uniform(-0.3, 0.3)
            h_new = h.copy()
            h_new[i] = np.clip(h[i] + delta, 0, 1)
            h_new = _project(h_new, target)
        elif move_type == 1:
            # Block shift
            L = int(rng.integers(2, max(3, n // 4)))
            start = int(rng.integers(0, n - L))
            shift = int(rng.integers(1, max(2, n // 4)))
            h_new = h.copy()
            block = h[start:start+L].copy()
            new_start = start + shift
            if new_start + L > n:
                new_start = start - shift
            if new_start < 0 or new_start + L > n:
                continue
            h_new[new_start:new_start+L] = block
            # Need to handle overlap; for simplicity just do element swap
            j = rng.integers(n)
            if j != i:  # i not defined here, swap two elements
                h_new = h.copy()
                h_new[start] = h[start] + rng.uniform(-0.2, 0.2)
                h_new = _project(h_new, target)
        else:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = rng.uniform(-0.1, 0.1)
            max_d = min(h[i], 1.0 - h[j]) if delta > 0 else min(1.0 - h[i], h[j])
            if max_d < 1e-10:
                continue
            d = delta * max_d
            h_new = h.copy()
            h_new[i] += d
            h_new[j] -= d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
        
        f_new = _exact_c5(h_new)
        if f_new < cur_f or rng.random() < np.exp((cur_f - f_new) / T):
            h = h_new
            cur_f = f_new
            if cur_f < best_f:
                best_f = cur_f
                best_h = h.copy()
        T *= cool
    return best_h, best_f

def _resample(h, n_new):

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _make_basis(n, rng, q):
    B = np.zeros((n, q))
    x = np.arange(n) / n
    idx = 0
    # Low-freq sinusoidal
    for k in range(1, min(6, (q+1)//2)):

        s = np.sin(2*np.pi*k*x) - np.sin(2*np.pi*k*x).mean()
        nc, ns = np.linalg.norm(c), np.linalg.norm(s)
        if nc > 1e-10: B[:, idx] = c/nc
        if ns > 1e-10 and idx+1 < q: B[:, idx+1] = s/ns
        idx += 2
    # Block differences
    while idx < q:
        r = int(rng.integers(1, max(2, n//4)+1))
        if 2*r <= n:

            col = np.zeros(n)

            col = rng.standard_normal(n)
            col -= col.mean()
        col /= np.linalg.norm(col) + 1e-15

        idx += 1
