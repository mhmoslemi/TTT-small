def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_solve(h, scores, G, radius, active_lags=None):
    n = len(h)
    n_lags = 2 * n - 1
    if active_lags is None:
        idx = range(n_lags)
    else:
        idx = active_lags
    k = len(idx)
    if k == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    for row, lag in enumerate(idx):
        A_ub[row, :n] = G[lag, :]
        A_ub[row, n] = -1.0
    b_ub = -np.array([scores[lag] for lag in idx])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    # Clip to avoid overflow
    v = np.clip(v, -500, 0)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad
