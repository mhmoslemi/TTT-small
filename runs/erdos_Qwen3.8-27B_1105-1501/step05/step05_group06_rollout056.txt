def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    """Build Jacobian of scores (without dx) w.r.t. h. Shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        j_lo = max(0, -d)
        j_hi = min(n - 1, n - 1 - d)
        if j_hi < j_lo:
            continue
        G[r, j_lo:j_hi+1] += (1.0 - h[j_lo+d:j_hi+1+d])
        G[r, j_lo+d:j_hi+d+1] += -h[j_lo:j_hi+1]
    return G

    """Compute p-norm surrogate value and gradient."""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    L = len(scores)
    s_max = float(np.max(scores))
    if s_max < 1e-30:
        return 0.0, np.zeros(n)
    
    ratio = np.maximum(scores / s_max, 0.0)
    # log of ratio for stable computation
    log_ratio = np.where(ratio > 0, np.log(ratio), -100.0)
    
    # sum_w = sum(ratio^q)
    log_wq = q * log_ratio
    max_lwq = float(np.max(log_wq))
    if max_lwq > 500:
        exp_wq = np.exp(log_wq - max_lwq)
        sum_w = float(np.sum(exp_wq)) * np.exp(max_lwq)
    else:
        exp_wq = np.exp(log_wq)
        sum_w = float(np.sum(exp_wq))
    
    value = s_max * sum_w ** (1.0 / q)
    
    # Gradient weights: ratio^(q-1)
    if q == 1:
        w_grad = np.ones(L)
    else:
        log_wg = (q - 1) * log_ratio
        max_lwg = float(np.max(log_wg))
        if max_lwg > 500:
            w_grad = np.exp(log_wg - max_lwg)
        else:
            w_grad = np.exp(log_wg)
    
    G = _build_G(h)

    grad = G.T @ w_grad

    return value, grad
