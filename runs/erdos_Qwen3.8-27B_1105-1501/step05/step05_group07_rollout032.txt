```python
def _grad_in_h_space(h, w):
    """Compute G.T @ w where G[k,j] = d(c[k])/d(h[j])."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n-1):
        d = k - (n-1)
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        # Direct: j from max(0,-d) to min(n-1, n-1-d)
        js = max(0, -d)
        je = min(n, n - d)
        if je > js:
            grad[js:je] += wk * (1 - h[js+d:je+d])
        # Indirect: j from max(0,d) to min(n-1, n-1+d)
        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad[js2:je2] += wk * (-h[js2-d:je2-d])
    return grad
```

```python
from scipy.optimize import minimize
import time

def _sigmoid(x):
    x = np.clip(x, -500, 500)
    out = np.empty_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _sigmoid_deriv(x):
    s = _sigmoid(x)
    return s * (1.0 - s)

def _find_lambda(q, target):
    n = len(q)
    lo = float(q.min()) - 30.0
    hi = float(q.max()) + 30.0
    # Verify we bracket
    s_lo = float(np.sum(_sigmoid(q - lo)))  # should be ~n
    s_hi = float(np.sum(_sigmoid(q - hi)))  # should be ~0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sigmoid(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

    n = len(q)

    lam = _find_lambda(q, target)
    h = _sigmoid(q - lam)
    # Small correction
    diff = target - float(np.sum(h))

        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _to_q(h):
    h_clip = np.clip(h, 1e-10, 1.0 - 1e-10)
    return np.log(h_clip / (1.0 - h_clip))

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    return float(np.max(_scores(h)))

def _obj_grad_q(q, beta):
    """Objective and gradient in q-space using log-sum-exp with given beta."""
    h = _q_to_h(q)
    S = _scores(h)
    max_S = float(np.max(S))
    shifted = beta * (S - max_S)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(S))
        w = np.zeros(len(S))
        w[idx] = 1.0
        wsum = 1.0
    else:
        w /= wsum
    
    obj = max_S + float(np.log(wsum)) / beta
    
    # Gradient in h-space: dx * (G.T @ w)
    n = len(h)
    dx = 2.0 / n
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        js = max(0, -d)
        je = min(n, n - d)
        if je > js:
            grad_h[js:je] += wk * (1.0 - h[js + d:je + d])
        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad_h[js2:je2] += wk * (-h[js2 - d:je2 - d])
    grad_h *= dx
    
    # Chain rule to q-space
    lam = _find_lambda(q, n / 2.0)
    sig_d = _sigmoid_deriv(q - lam)
    S_sig = float(np.sum(sig_d))
    a = grad_h * sig_d
    grad_q = a - (sig_d / S_sig) * float(np.sum(a))
    
    return obj, grad_q

def _logistic_optimize(q, deadline, beta_start=5.0, beta_end=500.0, n_phases=5, maxiter=500):
    """Run L-BFGS-B in q-space with increasing beta."""
    n = len(q)
    q = q.copy()
    best_h = _q_to_h(q)
    best_c5 = _exact_c5(best_h)
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        def obj_func(qf):
            val, _ = _obj_grad_q(qf, beta)
            return val
        
        def grad_func(qf):
            _, g = _obj_grad_q(qf, beta)
            return g
        
        try:
            res = minimize(obj_func, q, jac=grad_func, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                                  'maxls': 50})
            if np.all(np.isfinite(res.x)):

                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                q = res.x.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    if abs(res) > 1e-12:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

Let me also think about coordinate descent. For each pair (i, j), we can transfer "mass" from one to the other (keeping the sum constant). The effect on the objective is:
- For each score S_k, the change is ΔS_k = dx * [(1-h[i+d_k])*δ_i - h[i-d_k]*δ_i - (1-h[j+d_k])*δ_j + h[j-d_k]*δ_j]

where δ_i = +α, δ_j = -α (or vice versa). So ΔS_k = dx * α * [(1-h[i+d_k]) - h[i-d_k] + (1-h[j+d_k]) - h[j-d_k]]

= dx * α * [g_i - g_j] where g_i = (1-h[i+d_k]) - h[i-d_k] (the partial derivative of c[k] w.r.t. h[i] before the dx factor).

To minimize max(S), we want to reduce the maximum score. The best transfer is one that reduces the most active scores the most.

This is essentially the same as the LP approach but restricted to 2D transfers. The LP approach considers all directions simultaneously, so it's better. But coordinate descent is simpler and can be combined with LP for a "polish" step.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _sigmoid(x):
    x = np.clip(x, -500, 500)
    out = np.empty_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _sigmoid_deriv(x):
    s = _sigmoid(x)
    return s * (1.0 - s)

def _find_lambda(q, target):
    lo = float(q.min()) - 40.0
    hi = float(q.max()) + 40.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sigmoid(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q):
    n = len(q)
    target = n / 2.0
    lam = _find_lambda(q, target)
    h = _sigmoid(q - lam)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _to_q(h):
    hc = np.clip(h, 1e-12, 1.0 - 1e-12)
    return np.log(hc / (1.0 - hc))

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _exact_c5(h):
    return float(np.max(_scores(h)))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _obj_grad_q(q, beta):
    h = _q_to_h(q)
    S = _scores(h)
    max_S = float(np.max(S))
    shifted = beta * (S - max_S)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    wsum = float(np.sum(exp_shifted))
    if wsum < 1e-300:
        obj = max_S
        idx = int(np.argmax(S))
        w = np.zeros(len(S))
        w[idx] = 1.0
    else:
        obj = max_S + float(np.log(wsum)) / beta
        w = exp_shifted / wsum
    n = len(h)
    dx = 2.0 / n
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        js = max(0, -d)
        je = min(n, n - d)
        if je > js:
            grad_h[js:je] += wk * (1.0 - h[js + d:je + d])
        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad_h[js2:je2] += wk * (-h[js2 - d:je2 - d])
    grad_h *= dx
    lam = _find_lambda(q, n / 2.0)
    sig_d = _sigmoid_deriv(q - lam)
    S_sig = float(np.sum(sig_d))
    if S_sig < 1e-30:
        return obj, np.zeros(n)
    a = grad_h * sig_d
    grad_q = a - (sig_d / S_sig) * float(np.sum(a))
    return obj, grad_q

def _logistic_optimize(q, deadline, beta_start=5.0, beta_end=500.0, n_phases=5, maxiter=400):

    q = q.copy()
    best_h = _q_to_h(q)
    best_c5 = _exact_c5(best_h)
    bounds = [(-60.0, 60.0)] * n
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        cur_beta = beta
        def obj_f(qf):
            v, _ = _obj_grad_q(qf, cur_beta)
            return v
        def grad_f(qf):
            _, g = _obj_grad_q(qf, cur_beta)
            return g
        try:
            res = minimize(obj_f, q, jac=grad_f, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                  'maxls': 40, 'maxfun': maxiter * 10})
            if np.all(np.isfinite(res.x)):

                c5_new = _exact_c5(h_new)

                    best_c5 = c5_new
                    best_h = h_new.copy()
                q = np.clip(res.x, -60.0, 60.0)
        except Exception:
            pass
```
