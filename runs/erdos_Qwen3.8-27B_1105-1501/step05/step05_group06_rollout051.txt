    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        j_lo = max(0, -d)
        j_hi = min(n - 1, n - 1 - d)
        if j_lo > j_hi:
            continue
        G[r, j_lo:j_hi + 1] += dx * (1.0 - h[j_lo + d:j_hi + 1 + d])
        k_lo = max(0, j_lo + d)
        k_hi = min(n - 1, j_hi + d)
        if k_lo <= k_hi:
            G[r, k_lo:k_hi + 1] += -dx * h[k_lo - d:k_hi + 1 - d]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_imp = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    np.clip(h, 0.0, 1.0, out=h)
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        scores = _scores(h)
                        G[:, j] = 0
                        G[:, k] = 0
                        _rebuild_G_row(G, h, j)
                        _rebuild_G_row(G, h, k)
                        pass_imp = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        np.clip(h, 0.0, 1.0, out=h)
        if not pass_imp:
            break
    return best_h, best_c5

def _rebuild_G_row(G, h, col):
    n = len(h)
    dx = 2.0 / n
    j = col
    for r in range(2 * n - 1):
        d = r - (n - 1)
        val = 0.0
        j_lo = max(0, -d)
        j_hi = min(n - 1, n - 1 - d)
        if j_lo <= j <= j_hi:
            val += dx * (1.0 - h[j + d])
        if j - d >= 0 and j - d <= n - 1:
            k = j - d
            if k >= j_lo and k <= j_hi:
                val += -dx * h[k]
        G[r, col] = val

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)
