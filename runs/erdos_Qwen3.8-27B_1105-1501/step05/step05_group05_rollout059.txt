def _project(v, target):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}"""

def _sigmoid(x):
    """Numerically stable sigmoid."""
    result = np.empty_like(x, dtype=float)

    result[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    exp_x = np.exp(x[neg])
    result[neg] = exp_x / (1.0 + exp_x)
    return result

def _grad_matrix(h):
    """Build gradient matrix for all 2n-1 lags."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):

            m = n - d

                G[r, d:] += (1.0 - h[:m])

        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    if penalty_lags is not None and len(penalty_lags) > 0:
        for k in penalty_lags:
            b_ub[k] -= penalty_w
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})

            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    """Resample h to new resolution."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            radius = 0.22
            lr = 0.03
            mom = np.zeros(n)
        if mode == 0:
            scores = np.correlate(h, 1.0 - h, mode='full')
            G = _grad_matrix(h)
            p = _lp_solve(h, scores, G, radius)
            if p is not None and np.max(np.abs(p)) > 1e-13:
                alpha, new_c5 = _line_search(h, p, dx, radius)
                if new_c5 < cur_c5 - 1e-12:
                    h = h + alpha * p
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.50)
                    continue
            stall += 1
            radius *= 0.8
        else:
            improved = False
            for beta in [50, 130, 350]:
                d = _logsumexp_grad(h, dx, beta)
                dn = np.linalg.norm(d)
                if dn < 1e-13:
                    continue
                d = d / dn
                mom = 0.82 * mom + 0.18 * d
                mn = np.linalg.norm(mom)
                if mn < 1e-13:
                    continue
                mom = mom / mn
                alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 3.0)
                if new_c5 < cur_c5 - 1e-12:
                    h = h + alpha * lr * mom
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    lr = min(lr * 1.06, 0.45)
                    improved = True
                    break
            if not improved:
                stall += 1
                lr *= 0.8
    return best_h, best_c5

def _population_search(h_best, c5_best, deadline, rng):
    n = len(h_best)
    target = n / 2.0
    pop_size = 10
    population = [(c5_best, h_best.copy())]
    for i in range(pop_size - 1):
        scale = rng.uniform(0.02, 0.12) * (1.0 + 0.4 * i)
        d = rng.normal(0, 1, n)
        d -= d.mean()
        d /= np.linalg.norm(d) + 1e-15
        h_try = h_best + scale * d
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        population.append((c5_try, h_try))
    population.sort(key=lambda x: x[0])
    best_h = population[0][1].copy()
    best_c5 = population[0][0]
    sigma = 0.06
    gen = 0
    no_improve = 0
    while time.time() < deadline:
        if no_improve > 80:
            break
        offspring = []
        for i in range(3):
            offspring.append(population[i])
        while len(offspring) < pop_size and time.time() < deadline:
            pi = int(rng.integers(0, len(population)))
            parent = population[pi][1].copy()
            op = int(rng.integers(0, 5))
            if op == 0:
                d = rng.normal(0, sigma, n)
                d -= d.mean()
                h_try = parent + d
            elif op == 1:
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 == i2:
                    continue
                amt = rng.uniform(-0.4, 0.4)
                h_try = parent.copy()
                h_try[i1] -= amt
                h_try[i2] += amt
            elif op == 2:
                i1 = int(rng.integers(0, max(1, n - 5)))
                width = int(rng.integers(3, max(4, n // 4)))
                width = min(width, n - i1)
                d = np.zeros(n)
                d[i1:i1 + width] = rng.uniform(-sigma, sigma, width)
                d -= d.mean()
                h_try = parent + d
            elif op == 3:
                pj = int(rng.integers(0, len(population)))
                parent2 = population[pj][1]
                cp = int(rng.integers(1, n))
                h_try = np.concatenate([parent[:cp], parent2[cp:]])
            else:
                # Flip a block
                i1 = int(rng.integers(0, n - 3))
                width = int(rng.integers(2, max(3, n // 5)))
                width = min(width, n - i1)
                h_try = parent.copy()
                h_try[i1:i1 + width] = 1.0 - h_try[i1:i1 + width]
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            offspring.append((c5_try, h_try))
        offspring.sort(key=lambda x: x[0])
        population = offspring[:pop_size]
        if population[0][0] < best_c5 - 1e-12:
            best_c5 = population[0][0]
            best_h = population[0][1].copy()
            no_improve = 0
            sigma = min(sigma * 1.1, 0.25)
        else:
            no_improve += 1
            if no_improve % 8 == 0:
                sigma = max(sigma * 0.92, 0.004)
        gen += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Combined local (22%)
    d1 = t0 + budget_s * 0.22
    h1, c5_1 = _combined_local(h, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Logit-space optimization (15%)
    d2 = t0 + budget_s * 0.37
    h2, c5_2 = _logit_optimize(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP refinement (13%)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.20, use_penalty=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Population search (20%)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _population_search(best_h, best_c5, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (14%)
    d5 = t0 + budget_s * 0.84
    best_n = len(best_h)
    tried = set()
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 80 or n_try > 220:
            continue
        if n_try in tried and n_try != best_n:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12)
        if c5_r < best_c5 and time.time() < d5 - 4:
            sub_dl2 = min(d5, time.time() + 5.0)
            h_r, c5_r = _pair_refine(h_r, sub_dl2, max_passes=25, use_random=True, rng=rng)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Kick and reoptimize (7%)
    d6 = t0 + budget_s * 0.91
    h6, c5_6 = _kick_reopt(best_h, d6, rng, n_kicks=25)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final pair refinement (5%)
    d7 = t0 + budget_s * 0.96
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=80, use_random=True, rng=rng)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final LP polish
    d8 = t0 + budget_s - 1.5
    if time.time() < d8:
        h8, c5_8 = _lp_loop(best_h, d8, radius_init=0.06, use_penalty=True)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    if not np.isfinite(final_c5) or final_c5 > 0.5:
        best_h = _project(h0, n0 / 2.0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
