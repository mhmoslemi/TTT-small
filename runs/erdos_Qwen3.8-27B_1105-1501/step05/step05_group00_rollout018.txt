            else:
                g[gg:] += weights[r] * (1.0 - h[:m])
                g[:m] += -weights[r] * h[gg:gg+m]
    
    return g

def _compute_grads(h, active_rows):
    """Compute gradient of each active row with respect to h. Returns (n_active, n) array."""
    n = len(h)
    grads = np.zeros((len(active_rows), n))
    for k, r in enumerate(active_rows):
        d = r - (n - 1)
        m = n - abs(d)
        if m <= 0:

                grads[k, :m] = 1.0 - 2.0 * h[:m]

        if best_signs is not None and best_alpha > 1e-10:
            s1, s2, s3, s4 = best_signs
            h[i] += best_alpha * s1
            h[j] += best_alpha * s2
            h[k] += best_alpha * s3
            h[l] += best_alpha * s4
            h = project(h, target)
            cur_c5 = c5_exact(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _binary_round_optimize(h, deadline, rng):
    """Try rounding to binary and then optimize via swaps."""
    n = len(h)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    cur_c5 = c5_exact(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin <= 0.5)[0]
    patience = 0
    
    while time.time() < deadline and patience < 8000:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        # Try swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = c5_exact(h_bin)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            S0 = np.correlate(h_bin, 1.0 - h_bin, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
                patience = 0
            else:
                patience += 1
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin <= 0.5)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            patience += 1
    
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97  # 3% reserve
    
    # Initialize from parent if available
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)
    
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0
    
    # Phase 1: Directed pair search (35% time)
    d1 = t0 + budget_s * 0.35
    h_p1, c5_p1 = _directed_pair_search(h, 20, d1)
    if c5_p1 < best_c5:
        best_c5 = c5_p1
        best_h = h_p1.copy()
        best_n = n0
    
    # Phase 2: Random pair search (15% time)
    d2 = t0 + budget_s * 0.50
    h_p2, c5_p2 = _random_pair_search(best_h, d2, rng)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()
        best_n = len(best_h)
    
    # Phase 3: Multi-resolution (20% time)
    d3 = t0 + budget_s * 0.70
    offsets = [-4, 4, -6, 6, -8, 8, -2, 2]
    tried_n = {best_n}
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _directed_pair_search(h_r, 15, sub_dl)
        if c5_r < best_c5:
            # Polish further
            sub_dl2 = min(d3, time.time() + budget_s * 0.015)
            h_r, c5_r = _gradient_polish(h_r, sub_dl2)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    
    # Phase 4: Gradient polish (10% time)
    d4 = t0 + budget_s * 0.80
    h_p4, c5_p4 = _gradient_polish(best_h, d4)
    if c5_p4 < best_c5:
        best_c5 = c5_p4
        best_h = h_p4.copy()
    
    # Phase 5: 4-coord search (8% time)
    d5 = t0 + budget_s * 0.88
    h_p5, c5_p5 = _four_coord_search(best_h, d5, rng)
    if c5_p5 < best_c5:
        best_c5 = c5_p5
        best_h = h_p5.copy()
    
    # Phase 6: Binary round + optimize (5% time)
    d6 = t0 + budget_s * 0.93
    h_p6, c5_p6 = _binary_round_optimize(best_h, d6, rng)
    if c5_p6 < best_c5:
        best_c5 = c5_p6
        best_h = h_p6.copy()
        best_n = len(best_h)
    
    # Phase 7: Final directed polish (5% time)
    d7 = t0 + budget_s * 0.97
    h_p7, c5_p7 = _directed_pair_search(best_h, 25, d7)
    if c5_p7 < best_c5:
        best_c5 = c5_p7
        best_h = h_p7.copy()
    
    # Final gradient polish with remaining time
    h_f, c5_f = _gradient_polish(best_h, deadline)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    # Ensure validity
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    
    return best_h, final_c5, len(best_h)
