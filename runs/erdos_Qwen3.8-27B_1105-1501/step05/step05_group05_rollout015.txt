    """Compute gradient of score_k with respect to h."""
    g = np.zeros(n)
    d = k - (n - 1)
    if d >= 0:
        m = n - d
        # From h[j+d] term: contributes (1-h[i-d]) for i=d..n-1
        g[d:] += (1.0 - h[:m])
        # From (1-h[j]) term: contributes -h[i+d] for i=0..m-1

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _score_grad_vec(h, k, n, dx):

    d = k - (n - 1)
    if d >= 0:
        m = n - d
        if m > 0:

            g[:m] += (-h[d:d + m])
    else:
        g2 = -d
        m = n - g2
        if m > 0:
            g[:m] += (1.0 - h[g2:g2 + m])
            g[g2:] += (-h[:m])

def _lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-18:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w[r] * (1.0 - h[:m])
                grad[:m] += w[r] * (-h[d:d + m])
        else:
            g2 = -d
            m = n - g2
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[g2:g2 + m])
                grad[g2:] += w[r] * (-h[:m])
    grad *= dx
