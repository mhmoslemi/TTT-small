import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

For d ≥ 0: score_r = Σ_{m=0}^{n-1-d} h[m+d] * (1-h[m])

∂score_r/∂h[k]:
- h[k] appears as h[m+d] when m = k-d, valid if 0 ≤ k-d ≤ n-1-d, i.e., d ≤ k ≤ n-1
  Contribution: (1 - h[k-d])
- h[k] appears in (1-h[m]) when m = k, valid if 0 ≤ k ≤ n-1-d
  Contribution: -h[k+d]

So: ∂score_r/∂h[k] = (1-h[k-d]) * I(d ≤ k ≤ n-1) + (-h[k+d]) * I(0 ≤ k ≤ n-1-d)

For d < 0, let g = -d > 0:
score_r = Σ_{t=0}^{n-1-g} h[t] * (1-h[t+g])

∂score_r/∂h[k]:
