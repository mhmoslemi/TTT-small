def _project(v, target):
    # Bisection projection onto {x in [0,1]^n : sum(x) = target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0: continue
        if d >= 0:
            G[r, :m] += (1.0 - h[d:d+m])
            G[r, d:] += -h[:m]
        else:
            g = -d
            G[r, :m] += (1.0 - h[g:g+m])
            G[r, g:] += -h[:m]
    return G

def _lp_direction(h, radius):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = _grad_matrix(h)
    # LP: min t s.t. G@x - t <= -scores, sum(x)=0, bounds
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[i]), min(radius, 1.0-h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_graph(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))

    # Gradient of log-sum-exp: sum_r w_r * grad_r
    G = _grad_matrix(h)
    grad = G.T @ w  # shape (n,)
    grad -= grad.mean()
    return grad

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, trust_radius):

    d = d - d.mean()
    if np.max(np.abs(d)) < 1e-15:
        return 0.0, _exact_c5(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40

    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t, best_val = grid[best_i], obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:

    else:
        a, b = t_lo, t_hi

    c2 = a + gr * (b - a)

    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx

    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[r, :m] += (1.0 - h[d:d + m])
            G[r, d:] += -h[:m]
        else:
            g = -d
            G[r, :m] += (1.0 - h[g:g + m])
            G[r, g:] += -h[:m]
    return G

def _lp_direction(h, radius):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.2, max_stall=100, use_penalty=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()

        if use_penalty and stall > 10:

            active = scores >= s_max * 0.99

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []

            hi = min(radius, 1.0 - h[i])
            if lo > hi:
                lo = hi = 0.0
            bounds.append((lo, hi))
        bounds.append((None, None))
        p = None

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= grad.mean()
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, n_random_pairs=3000, rng=None):

    target = n / 2.0

    n_iter = 0

        n_iter += 1
        if rng is not None:
            idx_i = rng.integers(0, n, size=n_random_pairs)
            idx_j = rng.integers(0, n, size=n_random_pairs)
        else:
            idx_i = np.random.randint(0, n, size=n_random_pairs)
            idx_j = np.random.randint(0, n, size=n_random_pairs)
        improved = False
        for k in range(n_random_pairs):
            if time.time() > deadline:
                break

                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True

    return best_h, best_c5

def _kick_reoptimize(h, deadline, rng, n_kicks=30, kick_scale_lo=0.03, kick_scale_hi=0.10):

    target = n / 2.0
    best_h, best_c5 = h.copy(), _exact_c5(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        scale = rng.uniform(kick_scale_lo, kick_scale_hi)
        direction = rng.normal(0, 1, n)
        direction -= direction.mean()
        direction /= (np.linalg.norm(direction) + 1e-15)
        h_k = best_h + scale * direction
        h_k = _project(h_k, target)
        sub_dl = min(deadline, time.time() + 4.0)

        if c5_t < best_c5 - 1e-12:

    return best_h, best_c5

    d2 = t0 + budget_s * 0.70

    res_list = [r for r in res_list if 100 <= r <= 350]

    for n_try in res_list:
        remaining = deadline_abs - time.time()

            break

        if alloc < 8:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + alloc * 0.6)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.18)
