def _slsqp_epigraph(h_init, active_shifts, t_init, reg=0.01, maxiter=100):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(z):
        return z[n] + 0.5 * reg * np.sum((z[:n] - h_init)**2)
    
    def obj_jac(z):
        g = np.zeros(n + 1)
        g[:n] = reg * (z[:n] - h_init)
        g[n] = 1.0
        return g
    
    cons = []
    for s in active_shifts:
        def con_fun(z, s=s):
            hv = z[:n]
            tv = z[n]
            if s >= 0:
                m = n - s
                fv = np.sum(hv[s:s+m] * (1.0 - hv[:m])) * dx
            else:
                g = -s
                m = n - g
                fv = np.sum(hv[:m] * (1.0 - hv[g:g+m])) * dx
            return tv - fv
        def con_jac(z, s=s):
            hv = z[:n]
            gz = np.zeros(n + 1)
            gz[:n] = -_grad_row(hv, s)
            gz[n] = 1.0
            return gz
        cons.append({'type': 'ineq', 'fun': con_fun, 'jac': con_jac})
    
    def eq_fun(z):
        return np.sum(z[:n]) - target
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    z0 = np.concatenate([h_init, [t_init]])
    bounds = [(0, 1)] * n + [(0.0, 1.0)]
    
    res = minimize(obj, z0, jac=obj_jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    
    if res.success and res.x[n] < t_init:
        return res.x[:n], res.x[n]
    return None, None

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _proj_box_sum(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_row(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0: return g
        g[s:] = 1.0 - h[:m]
        g[:m] -= h[s:]
    else:
        g_abs = -s
        m = n - g_abs
        if m <= 0: return g
        g[:m] = 1.0 - h[g_abs:]
        g[g_abs:] -= h[:m]
    return g * dx

def _single_f(h, s):
    """Compute single correlation value for shift s, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    if s >= 0:
        m = n - s
        return float(np.sum(h[s:s+m] * (1.0 - h[:m]))) * dx if m > 0 else 0.0
    else:
        g = -s
        m = n - g
        return float(np.sum(h[:m] * (1.0 - h[g:g+m]))) * dx if m > 0 else 0.0

def _lse_descent(h0, deadline, tau_init=30.0, tau_max=150.0, top_k=40):
    """Log-sum-exp smoothed gradient descent."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    alpha = 2.0 / tau
    stall = 0
    
    while time.time() < deadline:
        if stall > 500:
            break
        f_vals = _corr_full(h)
        f_max = np.max(f_vals)
        # Softmax weights
        shifted = tau * (f_vals - f_max)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        
        # Only use top_k weights
        if top_k < len(f_vals):
            top_idx = np.argsort(-w)[:top_k]
            w_masked = np.zeros_like(w)
            w_masked[top_idx] = w[top_idx]
            w_masked /= np.sum(w_masked)
            w = w_masked
        
        # Compute gradient
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] > 1e-6:
                s = r - (n - 1)
                grad += w[r] * _grad_row(h, s)
        grad -= np.mean(grad)  # project to sum-zero
        
        # Line search
        improved = False
        for _ls in range(40):
            h_trial = _proj_box_sum(h - alpha * grad, target)
            c5_trial = _exact_c5(h_trial)
            if c5_trial < cur_c5 - 1e-14:
                h = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                break
            alpha *= 0.7
        if not improved:
            stall += 1
            alpha *= 0.5
            if alpha < 1e-10:
                alpha = 2.0 / tau
                tau *= 1.1
        
        tau = min(tau * 1.002, tau_max)
    
    return best_h, best_c5

def _slsqp_solve(h_current, active_shifts, t_target, reg=0.005, maxiter=80):
    """SLSQP epigraph solve."""
    n = len(h_current)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(z):
        return z[n] + 0.5 * reg * np.sum((z[:n] - h_current)**2)
    
    def obj_jac(z):
        g = np.zeros(n + 1)
        g[:n] = reg * (z[:n] - h_current)
        g[n] = 1.0
        return g
    
    cons = []
    for s in active_shifts:
        def cf(z, s=s):
            hv = z[:n]
            if s >= 0:
                m = n - s
                return z[n] - float(np.sum(hv[s:s+m] * (1.0 - hv[:m])) * dx) if m > 0 else z[n]
            else:
                g = -s
                m = n - g
                return z[n] - float(np.sum(hv[:m] * (1.0 - hv[g:g+m])) * dx) if m > 0 else z[n]
        def cj(z, s=s):
            hv = z[:n]
            gz = np.zeros(n + 1)
            gz[:n] = -_grad_row(hv, s)
            gz[n] = 1.0
            return gz
        cons.append({'type': 'ineq', 'fun': cf, 'jac': cj})
    
    def eq_f(z):
        return np.sum(z[:n]) - target
    def eq_j(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eq_f, 'jac': eq_j})
    
    z0 = np.concatenate([h_current, [t_target]])
    bounds = [(0, 1)] * n + [(0.0, 1.0)]
    
    try:
        res = minimize(obj, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success:
            h_new = _proj_box_sum(res.x[:n], target)
            return h_new, float(res.x[n])
    except Exception:
        pass
    return None, None

= sum_j h[j]*(1-h[j+k]) - a*sum_j h[j]*d[j+k] + a*sum_j d[j]*(1-h[j+k]) - a^2*sum_j d[j]*d[j+k]
= S0[k] - a * corr(h,d)[k] + a * corr(d, 1-h)[k] - a^2 * corr(d,d)[k]

where I'm using the convention corr(a,b)[k] = sum_j a[j] * b[j+k].

So: new_corr = S0 + a*(corr(d,1-h) - corr(h,d)) - a^2 * corr(d,d)
