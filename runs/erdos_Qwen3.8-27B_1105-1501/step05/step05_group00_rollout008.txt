```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val
```

```python
def _binary_swap_delta(h, p, q, corr, n):
    """Compute the change in correlation when h[p]: 1->0 and h[q]: 0->1."""
    delta = np.zeros(2*n-1)
    k_offset = n - 1
    
    # p goes from 1 to 0
    # p as "i": for lag k, valid if 0 <= p+k < n
    #   old: h[p+k]*0 = 0, new: h_new[p+k]*1
    for k in range(-(n-1), n):
        # p as i: need 0 <= p+k < n
        if 0 <= p + k < n:
            idx = p + k
            if idx == q:
                delta[k + k_offset] += 1.0  # h_new[q] = 1
            else:
                delta[k + k_offset] += h[idx]  # h_new[idx] = h[idx]
        # p as i+k: need i = p-k, 0 <= p-k < n
        i_pk = p - k
        if 0 <= i_pk < n:

                delta[k + k_offset] += -1.0
```

```python
    """Compute delta[m] for all lags m=0..2n-2 when h[p]:1->0, h[q]:0->1."""
    delta = np.zeros(2*n-1)
    ko = n - 1  # k offset: m = k + ko
    
    # p as "i": k range [-p, n-1-p]
    k_lo = -p
    k_hi = n - 1 - p
    if k_lo <= k_hi:
        indices = k_lo + np.arange(k_hi - k_lo + 1)
        m_vals = indices + ko
        # h_new[p+k] = h[p+k] unless p+k == q
        pk_vals = p + indices
        contributions = h[pk_vals].copy()
        # Where p+k == q, i.e., k == q-p
        if k_lo <= q-p <= k_hi:
            pass_idx = q - p - k_lo
            contributions[pass_idx] = 1.0
        delta[m_vals] += contributions
    
    # p as "i+k": k range [p-n+1, p]
    k_lo2 = p - n + 1
    k_hi2 = p
    if k_lo2 <= k_hi2:
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val
```

```python
def _fast_binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    ko = n - 1
    
    # Start from rounded binary
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.005
    T_min = 1e-6
    cooling = 0.99985
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    patience = 0
    max_patience = 50000
    
    while time.time() < deadline and T > T_min:
        if patience > max_patience:
            T = max(T_min, T * 0.995)
            patience = 0
            if T < T_min:
                break
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Batch of swaps
        batch = 200
        for _ in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            
            # Compute delta for swap h[i]:1->0, h[j]:0->1
            delta = np.zeros(2*n-1)
            
            # i as "i" (position i in h[i+k]): k range [-i, n-1-i]
            klo = -i
            khi = n - 1 - i
            if klo <= khi:
                idx = klo + np.arange(khi - klo + 1)
                m = idx + ko
                vals = h[i + idx].copy()
                if klo <= j - i <= khi:
                    vals[j - i - klo] = 1.0
                delta[m] += vals
            
            # i as "i+k" (position i in h[i-k]): k range [i-n+1, i]
            klo2 = i - n + 1
            khi2 = i
            if klo2 <= khi2:
                idx2 = klo2 + np.arange(khi2 - klo2 + 1)
                m2 = idx2 + ko
                vals2 = h[i - idx2] - 1.0
                if klo2 <= i - j <= khi2:
                    vals2[i - j - klo2] = 0.0
                delta[m2] += vals2
            
            # j as "i" (position j in h[j+k]): k range [-j, n-1-j]
            klo3 = -j
            khi3 = n - 1 - j
            if klo3 <= khi3:
                idx3 = klo3 + np.arange(khi3 - klo3 + 1)
                m3 = idx3 + ko
                vals3 = -h[j + idx3].copy()
                if klo3 <= i - j <= khi3:
                    vals3[i - j - klo3] = 0.0
                delta[m3] += vals3
            
            # j as "i+k" (position j in h[j-k]): k range [j-n+1, j]
            klo4 = j - n + 1
            khi4 = j
            if klo4 <= khi4:
                idx4 = klo4 + np.arange(khi4 - klo4 + 1)
                m4 = idx4 + ko
                vals4 = 1.0 - h[j - idx4]
                if klo4 <= j - i <= khi4:
                    vals4[j - i - klo4] = 1.0
                delta[m4] += vals4
            
            new_corr = corr + delta
            new_max = float(np.max(new_corr)) * dx
            d_c5 = new_max - cur_c5
            
            if d_c5 < 0 or (T > T_min and rng.random() < np.exp(-d_c5 / T)):
                # Accept
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                cur_c5 = new_max
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                # Update ones/zeros
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
            # else: reject, h and corr unchanged
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c5
```
