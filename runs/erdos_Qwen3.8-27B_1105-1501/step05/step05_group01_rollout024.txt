import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_grad(h, tau):
    """Compute gradient of LSE objective J_tau = tau * log(sum exp(dx*S_k/tau))"""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    scaled = dx * S / tau
    M = float(np.max(scaled))
    z = np.exp(scaled - M)
    w = z / float(np.sum(z))
    # g = dx * sum_k w_k * dS_k/dh
    g = np.zeros(n)
    for k in range(len(S)):
        wv = w[k] * dx
        if wv < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            m = n - offset
            if m > 0:
                g[offset:] += wv * (1.0 - h[:m])
                g[:m] += -wv * h[offset:]
        else:
            gg = -offset
            m = n - gg
            if m > 0:
                g[:m] += wv * (1.0 - h[gg:])
                g[gg:] += -wv * h[:m]
    return g, float(M + tau * np.log(float(np.sum(z))))

def _line_search_quad(h, d, dx, radius):
    """Exact line search using quadratic model along direction d.
    Returns (alpha, objective_value)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible interval
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    
    for _ in range(50):
        if t_hi - t_lo < 5e-13:
            break
        if f1 < f2:
            t_hi = c2
            c2 = c1
            f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            t_lo = c1
            c1 = c2
            f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * Q)) * dx
    return opt_t, opt_val

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_search(n, deadline, seed=0):

    target = n / 2.0
    if target != int(target):
        return None, None
    n_ones = int(target)

    best_c5 = float('inf')

    h = np.zeros(n)
    h[::2] = 1.0
    # Fix count
    if int(np.sum(h)) != n_ones:
        h = h.copy()
        ones = np.where(h > 0)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > n_ones:
            to_remove = rng.choice(ones, len(ones) - n_ones, replace=False)
            h[to_remove] = 0
        elif len(ones) < n_ones:
            to_add = rng.choice(zeros, n_ones - len(ones), replace=False)
            h[to_add] = 1
    
    cur_c5 = _exact_c5(h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    
    # Hill climbing on binary: try flipping pairs
    stall = 0
    while time.time() < deadline and stall < 500:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        if h[i] == h[j]:
            stall += 1
            continue
        # Flip: h[i] and h[j] swap
        d = np.zeros(n)
        d[i] = 1.0 - 2.0 * h[i]  # if h[i]=0, d[i]=1; if h[i]=1, d[i]=-1

            h[i] = h[j]
            h[j] = tmp
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lse(h, tau):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    scaled = dx * S / tau
    M = float(np.max(scaled))
    z = np.exp(scaled - M)
    sz = float(np.sum(z))
    w = z / sz
    
    g = np.zeros(n)
    for k in range(len(S)):
        wv = w[k] * dx
        if wv < 1e-15:
            continue
        offset = k - (n - 1)
        p = abs(offset)
        if p == 0:
            g += wv * (1.0 - 2.0 * h)
        else:
            m = n - p
            if m > 0:
                g[p:] += wv * (1.0 - h[:m])
                g[:m] += wv * (-h[p:])
    # Note: for symmetric correlation, offset +p and -p give same gradient contribution
    # But we've already counted both in the loop, so this is correct
    
    obj = float(M + tau * np.log(sz))
    return g, obj

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    
    for _ in range(45):
        if t_hi - t_lo < 5e-13:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * Q)) * dx
    return opt_t, opt_val

def _lse_opt(h, deadline, tau_list, max_iter, momentum=0.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    stall = 0
    vel = np.zeros(n)  # momentum
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        improved = False
        for tau in tau_list:
            if time.time() > deadline:
                break
            g, _ = _grad_lse(h, tau)
            g -= np.mean(g)  # project to sum-zero
            # Add momentum
            vel = momentum * vel + (1.0 - momentum) * g
            d = -vel
            mx = float(np.max(np.abs(d)))
            if mx < 1e-14:
                continue
            d /= mx
            
            alpha, new_c5 = _line_search(h, d, dx, 0.5)
            if alpha != 0 and new_c5 < cur - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                actual = _exact_c5(h_new)
                if actual < best_c5 - 1e-13:
                    h = h_new
                    cur = actual
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                    improved = True
                    break
        if not improved:
            vel *= 0.5  # decay momentum on failure
            stall += 1
    
    return best_h, best_c5

def _random_opt(h, deadline, n_iters, seed, radius_init=0.3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    radius = radius_init
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        
        d = rng.standard_normal(n)
        d -= np.mean(d)
        mx = float(np.max(np.abs(d)))
        if mx < 1e-14:
            continue
        d /= mx
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if alpha != 0 and new_c5 < cur - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)

            if actual < best_c5 - 1e-13:
                h = h_new
                cur = actual
                best_c5 = cur
                best_h = h.copy()
                stall = 0
                radius = min(radius * 1.01, 0.6)
            else:
                stall += 1
        else:
            stall += 1
            if stall % 150 == 0:
                radius *= 0.9
    
    return best_h, best_c5
