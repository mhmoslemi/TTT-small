import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target exactly."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    # Micro adjustment
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        p *= 2
    # Add some intermediate widths

    widths = [w for w in widths if 2 * w <= n]

        widths = [max(1, n // 4)]

        n_starts = n - 2 * w + 1

        if n_starts <= 0:
            continue
        for a in range(0, n_starts, stride):
            if len(columns) >= q_target:
                break
            col = np.zeros(n)
            c = 1.0 / np.sqrt(2.0 * w)
            col[a:a+w] = c
            col[a+w:a+2*w] = -c
            # Ensure sum is exactly zero (it is by construction)
            columns.append(col)

    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-10] = 1.0
    B /= norms
    return B

def _build_basis_fourier(n, q_target, rng):
    """Build a basis with Fourier-like sum-zero columns."""
    x = np.arange(n) / n
    columns = []
    for k in range(1, q_target + 1):
        col_c = np.cos(2 * np.pi * k * x)
        col_s = np.sin(2 * np.pi * k * x)
        # Remove mean to ensure sum ~ 0
        col_c -= col_c.mean()
        col_s -= col_s.mean()
        nc = np.linalg.norm(col_c)
        ns = np.linalg.norm(col_s)
        if nc > 1e-10:
            columns.append(col_c / nc)
        if ns > 1e-10:
            columns.append(col_s / ns)
        if len(columns) >= q_target:
            break
    B = np.column_stack(columns[:q_target])
    return B

def _build_basis_mixed(n, q_target, rng, frac_rect=0.5):
    """Build a mixed basis: some rectangular, some random."""
    q_rect = int(q_target * frac_rect)
    q_rand = q_target - q_rect
    
    # Rectangular part
    B_rect = _build_basis(n, 0, q_rect, rng)
    
    # Random part: random sum-zero vectors
    cols_rand = []
    for _ in range(q_rand):

        v -= v.mean()  # Ensure zero sum

            cols_rand.append(v / nv)
    
    if cols_rand:
        B_rand = np.column_stack(cols_rand)
        B = np.hstack([B_rect, B_rand])
    else:
        B = B_rect
    
    return B[:, :q_target]

def _de_search(B, deadline, pop_size=40, F=0.7, CR=0.8, max_gen=2000, y_bounds=2.0, init_y=None):
    """Differential evolution in basis space."""
    n, q = B.shape
    
    # Initialize population
    if init_y is not None:
        pop = np.zeros((pop_size, q))
        pop[0] = init_y
        for i in range(1, pop_size):
            pop[i] = init_y + rng.standard_normal(q) * 0.5
    else:
        pop = rng.uniform(-y_bounds, y_bounds, (pop_size, q))
    
    # Evaluate initial population
    fitness = np.zeros(pop_size)
    for i in range(pop_size):
        raw = 0.5 + B @ pop[i]
        h = _project(raw, n / 2.0)
        fitness[i] = _exact_c5(h)
    
    best_i = int(np.argmin(fitness))
    best_y = pop[best_i].copy()
    best_c5 = float(fitness[best_i])
    best_h = _project(0.5 + B @ best_y, n / 2.0)
    
    gen = 0
    while gen < max_gen and time.time() < deadline:
        gen += 1
        for i in range(pop_size):
            if time.time() > deadline:
                return best_h, best_c5, best_y
            
            # Select three distinct random individuals
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(len(idxs), 3, replace=False)
            a, b, c = idxs[a], idxs[b], idxs[c]
            
            # Mutation
            v = pop[a] + F * (pop[b] - pop[c])
            v = np.clip(v, -y_bounds, y_bounds)
            
            # Crossover
            trial = pop[i].copy()
            for j in range(q):
                if rng.random() < CR or j == rng.integers(q):
                    trial[j] = v[j]
            
            # Evaluate
            raw = 0.5 + B @ trial
            h_trial = _project(raw, n / 2.0)
            f_trial = _exact_c5(h_trial)
            
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c5:

                    best_y = trial.copy()
                    best_h = h_trial.copy()

            if gen % 100 == 0:

    h = _project(raw, n / 2.0)
    return h, _exact_c5(h)

def _de_optimize(B, y_init, deadline, rng, pop_size=40, F=0.65, CR=0.75,
                 max_gen=5000, y_lo=-3.0, y_hi=3.0):
    n, q = B.shape
    pop = np.empty((pop_size, q))
    pop[0] = y_init.copy()
    for i in range(1, pop_size / 2):
        pop[i] = y_init + rng.standard_normal(q) * 0.4
    for i in range(int(pop_size / 2), pop_size):
        pop[i] = rng.uniform(y_lo, y_hi, q)
    fitness = np.empty(pop_size)
    for i in range(pop_size):
        _, fitness[i] = _eval_basis(B, pop[i], n)
    best_i = int(np.argmin(fitness))
    best_c5 = float(fitness[best_i])
    best_y = pop[best_i].copy()
    best_h = _project(0.5 + B @ best_y, n / 2.0)
    gen = 0
    no_improve = 0
    while gen < max_gen and time.time() < deadline:
        gen += 1
        F_cur = F * max(0.3, 1.0 - 0.2 * gen / 2000.0)
        CR_cur = min(0.95, CR + 0.03 * gen / 2000.0)
        improved = False
        for i in range(pop_size):
            if time.time() > deadline:
                return best_h, best_c5, best_y
            r1 = int(rng.integers(pop_size))
            r2 = int(rng.integers(pop_size))
            r3 = int(rng.integers(pop_size))
            while r2 == r1:
                r2 = int(rng.integers(pop_size))
            while r3 == r1 or r3 == r2:
                r3 = int(rng.integers(pop_size))
            if gen % 4 == 0:
                v = pop[best_i] + F_cur * (pop[r1] - pop[r2])
            else:
                v = pop[r1] + F_cur * (pop[r2] - pop[r3])
            v = np.clip(v, y_lo, y_hi)
            trial = pop[i].copy()
            j_rand = int(rng.integers(q))
            mask = rng.random(q) < CR_cur
            mask[j_rand] = True
            trial[mask] = v[mask]
            trial = np.clip(trial, y_lo, y_hi)
            h_trial, f_trial = _eval_basis(B, trial, n)
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c5 - 1e-15:
                    best_c5 = f_trial
                    best_y = trial.copy()
                    best_h = h_trial
                    best_i = i
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve > 40:
                worst = np.argsort(fitness)[pop_size // 2:]
                for wi in worst:
                    pop[wi] = best_y + rng.standard_normal(q) * 0.3
                    _, f_w = _eval_basis(B, pop[wi], n)
                    fitness[wi] = f_w
                    if f_w < best_c5:
                        best_c5 = f_w
                        best_y = pop[wi].copy()
                        best_i = wi
                no_improve = 0
    return best_h, best_c5, best_y

def _coord_search(B, y0, deadline, n, step_init=0.2, max_iter=5000, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    q = B.shape[1]
    y = y0.copy()
    h_cur, c5_cur = _eval_basis(B, y, n)
    best_y = y.copy()
    best_c5 = c5_cur
    best_h = h_cur.copy()
    step = step_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        order = np.arange(q)
        rng.shuffle(order)
        improved = False
        for j in order:
            if time.time() > deadline:
                break
            for sign in [1.0, -1.0]:
                y_try = y.copy()
                y_try[j] += sign * step
                h_try, c5_try = _eval_basis(B, y_try, n)
                if c5_try < c5_cur - 1e-14:
                    y = y_try
                    h_cur = h_try
                    c5_cur = c5_try
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_y = y.copy()
                        best_h = h_cur.copy()
                    break
        if improved:
            stall = 0
            step = min(step * 1.04, 0.8)
        else:
            stall += 1
            step *= 0.6
            if step < 0.003 or stall > 40:
                break
    return best_h, best_c5, best_y

def _pair_sa(h, deadline, rng, max_iters=50000, T_init=0.004, T_min=0.00005):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = T_init
    cool = (T_min / T_init) ** (1.0 / float(max(max_iters, 1)))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cool
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if cur_h[i] >= cur_h[j]:
            amt = (cur_h[i] - cur_h[j]) * min(1.0, 0.02 + T * 30)
            if amt < 1e-9:
                continue
            h_try = cur_h.copy()
            h_try[i] -= amt
            h_try[j] += amt
        else:
            amt = (cur_h[j] - cur_h[i]) * min(1.0, 0.02 + T * 30)
            if amt < 1e-9:
                continue
            h_try = cur_h.copy()
            h_try[j] -= amt
            h_try[i] += amt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = _exact_c5(h_try)
        delta = c5_try - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_h = h_try
            cur_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    t_explore = t0 + budget_s * 0.50
    t_refine = t0 + budget_s * 0.72
    t_polish = t0 + budget_s * 0.96

    # Phase 1: Multi-resolution DE exploration
    res_list = sorted(set([48, 64, 80, 96, 112, 128, 144, 160, 192, 224, 256]))
    for n_try in res_list:
        if time.time() > t_explore:
            break
        q = max(12, min(40, n_try // 3))
        remaining = t_explore - time.time()
        time_alloc = min(budget_s * 0.055, max(8.0, remaining * 0.25))
        if remaining < 8:
            continue
        sub_dl = min(t_explore, time.time() + time_alloc)
        if n_try == best_n:
            h_init = best_h.copy()
        else:
            h_init = _resample(best_h, n_try)
        B = _build_block_basis(n_try, q, rng)
        v = h_init - 0.5
        y0, _, _, _ = np.linalg.lstsq(B, v, rcond=None)
        y0 = np.clip(y0, -3.0, 3.0)
        h_de, c5_de, y_de = _de_optimize(B, y0, sub_dl, rng, pop_size=35,
                                          F=0.65, CR=0.75, max_gen=5000,
                                          y_lo=-3.0, y_hi=3.0)
        if time.time() < t_explore and c5_de < best_c5 + 0.01:
            sub_dl2 = min(t_explore, time.time() + time_alloc * 0.35)
            h_cs, c5_cs, _ = _coord_search(B, y_de, sub_dl2, n_try,
                                            step_init=0.15, max_iter=2000, rng=rng)
            if c5_cs < c5_de:
                h_de, c5_de = h_cs, c5_cs
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            best_n = n_try

    # Phase 2: Refined DE with different bases on best resolution
    if time.time() < t_refine:
        n_best = best_n
        q2 = max(16, min(50, n_best // 2))
        basis_types = [
            ('block', _build_block_basis(n_best, q2, rng)),
            ('fourier', _build_fourier_basis(n_best, q2)),
            ('random', _build_random_basis(n_best, q2, rng)),
        ]
        for bname, B2 in basis_types:
            if time.time() > t_refine:
                break
            h_cur = _resample(best_h, n_best) if len(best_h) != n_best else best_h.copy()
            v = h_cur - 0.5
            y0, _, _, _ = np.linalg.lstsq(B2, v, rcond=None)
            y0 = np.clip(y0, -2.5, 2.5)
            sub_dl = min(t_refine, time.time() + budget_s * 0.05)
            h_de, c5_de, y_de = _de_optimize(B2, y0, sub_dl, rng, pop_size=40,
                                              F=0.6, CR=0.8, max_gen=4000,
                                              y_lo=-2.5, y_hi=2.5)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
                best_n = n_best
            if time.time() < t_refine and c5_de < best_c5 + 0.005:
                sub_dl2 = min(t_refine, time.time() + budget_s * 0.025)
                h_cs, c5_cs, _ = _coord_search(B2, y_de, sub_dl2, n_best,
                                                step_init=0.1, max_iter=3000, rng=rng)
                if c5_cs < best_c5:
                    best_c5 = c5_cs
                    best_h = h_cs.copy()
                    best_n = n_best

    # Phase 3: Random restarts at best resolution
    if time.time() < t_refine:
        n_best = best_n
        q3 = max(12, min(35, n_best // 3))
        B3 = _build_block_basis(n_best, q3, rng)
        n_restarts = 5
        for rs in range(n_restarts):
            if time.time() > t_refine:
                break
            h_rand = np.clip(0.5 + rng.normal(0, 0.15, n_best), 0, 1)
            h_rand = _project(h_rand, n_best / 2.0)
            v = h_rand - 0.5
            y0, _, _, _ = np.linalg.lstsq(B3, v, rcond=None)
            y0 = np.clip(y0, -3.0, 3.0)
            sub_dl = min(t_refine, time.time() + budget_s * 0.015)
            h_de, c5_de, _ = _de_optimize(B3, y0, sub_dl, rng, pop_size=25,
                                           F=0.7, CR=0.8, max_gen=2000,
                                           y_lo=-3.0, y_hi=3.0)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
                best_n = n_best

    # Phase 4: Simulated annealing pair polish
    if time.time() < t_polish:
        h_sa, c5_sa = _pair_sa(best_h, t_polish, rng, max_iters=80000,
                               T_init=0.003, T_min=0.00003)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Final: ensure validity
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    if final_c5 > _exact_c5(h_start):
        best_h = h_start.copy()
        final_c5 = _exact_c5(h_start)
        best_n = n0
    return best_h, final_c5, best_n
