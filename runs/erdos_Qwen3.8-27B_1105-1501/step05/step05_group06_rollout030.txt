def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:

        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _binary_sa(h_start, n, deadline, seed, max_iters=500000):
    rng = np.random.default_rng(seed)
    target = n // 2
    # Binarize starting point
    h = (h_start >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    if int(h.sum()) > target:
        ones = np.where(h == 1)[0]
        to_zero = rng.choice(ones, int(h.sum()) - target, replace=False)
        h[to_zero] = 0
    elif int(h.sum()) < target:
        zeros = np.where(h == 0)[0]
        to_one = rng.choice(zeros, target - int(h.sum()), replace=False)
        h[to_one] = 1
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    dx = 2.0 / n
    T = 0.01
    T_min = 1e-6
    cool = 0.9995
    
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Swap a random 1 with a random 0
        i = rng.integers(len(ones_pos))
        j = rng.integers(len(zeros_pos))
        a = ones_pos[i]
        b = zeros_pos[j]
        
        h[a] = 0
        h[b] = 1
        
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # Update position arrays
            ones_pos[i] = b
            zeros_pos[j] = a
        else:
            h[a] = 1
            h[b] = 0
        
        T *= cool
        if T < T_min:
            T = T_min
    
    return best_h, best_c5

def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

def _lp_refine(h, deadline, top_k_list=None, radius_init=0.12, max_iters=3000):
    n = len(h)
    if top_k_list is None:
        top_k_list = [None, 12, 8, 5, 3]
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    scores = _scores(h)
    G = _build_G(h)
    n_lags = 2 * n - 1
    ti = 0
    n_tk = len(top_k_list)
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        tk = top_k_list[ti % n_tk]
        if tk is not None and tk < n_lags:
            idx = np.argsort(scores)[-tk:]
        else:
            idx = np.arange(n_lags)
        s_k = scores[idx]
        G_k = G[idx]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = G_k
        A_ub[:, n] = -1.0
        b_ub = -s_k
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            p = res.x[:n] if res.success else None
        except Exception:
            p = None
        
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.72
            ti += 1
            if radius < 1e-8:
                break
            continue
        
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(22):
            if hi_a - lo_a < 1e-14:
                break
            if f1 < f2:
                hi_a, f2, c2 = c2, f1, c1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo_a, f1, c1 = c1, f2, c2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(h + c2 * p)
        
        opt_a = (lo_a + hi_a) / 2.0
        h_cand = _project(h + opt_a * p, n / 2.0)
        new_c5 = _exact_c5(h_cand)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h_cand
            cur_c5 = new_c5
            scores = _scores(h)
            G = _build_G(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
    
    return best_h, best_c5

def _binary_sa(h_start, n, deadline, seed, t0_temp=0.008, cool_rate=0.9998, reheat_every=20000):
    rng = np.random.default_rng(seed)
    target = n // 2
    
    h = (h_start >= 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > target:
        one_pos = np.where(h == 1)[0]
        to_zero = rng.choice(one_pos, n_ones - target, replace=False)
        h[to_zero] = 0
    elif n_ones < target:
        zero_pos = np.where(h == 0)[0]
        to_one = rng.choice(zero_pos, target - n_ones, replace=False)
        h[to_one] = 1
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = t0_temp
    it = 0
    
    while time.time() < deadline:
        it += 1
        if it % reheat_every == reheat_every:
            T = max(T, t0_temp * 0.5)
        
        if T < 1e-8:
            break
        
        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        i = int(rng.integers(len(ones_arr)))
        j = int(rng.integers(len(zeros_arr)))
        a = int(ones_arr[i])
        b = int(zeros_arr[j])
        
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        
        T *= cool_rate
    
    return best_h, best_c5

def _sa_with_refine(h_start, n, deadline, seed):
    """Run binary SA in phases with continuous refinement between phases."""
    if time.time() > deadline:
        h_s = _project(h_start, n / 2.0)
        return h_s, _exact_c5(h_s)
    
    frac = 0.6
    sa_dl = t0_sa = time.time() + (deadline - time.time()) * frac
    h_bin, c5_bin = _binary_sa(h_start, n, sa_dl, seed, t0_temp=0.008, cool_rate=0.9998)
    
    h_cont = _project(h_bin, n / 2.0)
    c5_cont = _exact_c5(h_cont)
    
    if time.time() < deadline - 10:
        h_r, c5_r = _lp_refine(h_cont, deadline, top_k_list=[None, 8, 5, 3],
                                radius_init=0.12, max_iters=2000)
        if c5_r < c5_cont:
            h_cont, c5_cont = h_r, c5_r
    
    return h_cont, c5_cont

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _pair_descent_quick(h, deadline, max_passes=100):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-10 or h[j] > 1 - 1e-10:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-10 or h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j] + 1e-10, h[k] - 1.0 + 1e-10)
                a_hi = min(1.0 - h[j] - 1e-10, h[k] - 1e-10)
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(16):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if cur_c5 - float(np.max(scores + alpha * C)) > 1e-7 and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        h = np.clip(h, 0.0, 1.0)
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.98
    
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    # Phase 1: Quick LP refinement of initial
    dl1 = t0 + budget_s * 0.06
    h1, c5_1 = _lp_refine(h_start, dl1, top_k_list=[None, 12, 8, 5],
                           radius_init=0.12, max_iters=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Binary SA + refinement at original resolution
    dl2 = t0 + budget_s * 0.22
    h2, c5_2 = _sa_with_refine(best_h, n0, dl2, seed + 1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = n0
    
    # Phase 3: Binary SA at higher resolutions
    dl3 = t0 + budget_s * 0.60
    for n_try in [n0 + 64, n0 + 128, n0 + 192, 256, 320]:
        if time.time() > dl3:
            break
        if best_c5 < 0.38055:
            break
        h_resample = _resample(best_h, n_try)
        sub_dl = min(dl3, time.time() + budget_s * 0.09)
        h_s, c5_s = _sa_with_refine(h_resample, n_try, sub_dl, seed + 10 + n_try)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n_try
    
    # Phase 4: Diverse quasi-periodic starts with SA
    dl4 = t0 + budget_s * 0.78
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
    for n_try in [n0, 192, 256]:
        if time.time() > dl4:
            break
        for ai in range(min(3, len(alphas))):
            if time.time() > dl4:
                break
            alpha = alphas[ai]
            h_init = _quasi_init(n_try, alpha, rng)
            sub_dl = min(dl4, time.time() + budget_s * 0.06)
            h_s, c5_s = _sa_with_refine(h_init, n_try, sub_dl, seed + 100 + ai * 10 + n_try)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n_try
    
    # Phase 5: Intensive LP + pair descent on best
    if time.time() < t0 + budget_s * 0.88:
        dl5 = t0 + budget_s * 0.90
        h5, c5_5 = _lp_refine(best_h, dl5, top_k_list=[6, 4, 3, None],
                               radius_init=0.06, max_iters=3000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < dl5:
            h_pd, c5_pd = _pair_descent_quick(best_h, dl5, max_passes=200)
            if c5_pd < best_c5:
                best_c5 = c5_pd
                best_h = h_pd.copy()
    
    # Phase 6: Perturbation restarts
    if time.time() < hard_deadline:
        dl6 = hard_deadline
        scale = 0.02
        for attempt in range(600):
            if time.time() > dl6 - 3:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6 - 2, time.time() + budget_s * 0.005)
            h_try, c5_try = _lp_refine(h_pert, sub_dl, top_k_list=[4, 3],
                                        radius_init=0.04, max_iters=200)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.005, scale * 0.82)
            else:
                scale = min(0.05, scale * 1.12)
    
    # Phase 7: Final polish
    if time.time() < hard_deadline - 2:
        h_fin, c5_fin = _pair_descent_quick(best_h, hard_deadline, max_passes=100)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
