def _binary_init_thue_morse(n):
    """Thue-Morse based initialization."""
    # Compute Thue-Morse sequence
    h = np.zeros(n)
    for i in range(n):
        # Count digits of i
        c = 0
        x = i
        while x > 0:
            c += x & 1
            x >>= 1
        h[i] = 1.0 if c % 2 == 0 else 0.0
    h = _project(h, n / 2.0)
    return h

def _binary_init_quadratic(n, p):
    """Quadratic residue based initialization."""
    h = np.zeros(n)
    # Use quadratic residues mod p
    residues = set()
    for a in range(p):
        residues.add((a * a) % p)
    for i in range(n):
        h[i] = 1.0 if i % p in residues else 0.0
    h = _project(h, n / 2.0)
    return h
