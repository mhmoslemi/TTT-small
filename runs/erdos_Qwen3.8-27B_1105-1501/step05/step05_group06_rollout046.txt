import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    # Project v to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(0, n):
        r = n - 1 + d
        m = n - d
        # i from 0 to m-1, term: h[i]*(1-h[i+d])
        G[r, :m] += dx * (1.0 - h[d:d+m])
        G[r, d:d+m] += -dx * h[0:m]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        # d = -s, i from s to n-1, term: h[i]*(1-h[i-s])
        G[r, s:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[s:]
    return G

def _pnorm_obj_grad(h, q):
    F = _scores(h)
    m = F.max()
    if m < 1e-30:
        return m, np.zeros(len(h))
    t = F / m  # all in [0,1]
    tq = t ** q
    T = float(tq.sum())
    obj = m * T ** (1.0 / q)
    if T < 1e-300:
        # All F are essentially zero, gradient is zero
        return obj, np.zeros(len(h))
    w = t ** (q - 1)  # weights, all in [0,1]
    scale = T ** (1.0 / q - 1.0)
    G = _build_G(h)
    grad = G.T @ (w * scale)
    return obj, grad

def _pnorm_optimize(h0, q, deadline, maxiter=200):
    n = len(h0)
    target = n / 2.0
    t0 = time.time()
    
    def objfun(h):
        v, _ = _pnorm_obj_grad(h, q)
        return v
    
    def objgrad(h):
        _, g = _pnorm_obj_grad(h, q)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objfun, h0, jac=objgrad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        pass
    else:
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            c5_old = _exact_c5(h0)
            if c5_new < c5_old:
                return h_new, c5_new
    return h0.copy(), _exact_c5(h0)

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=100):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = _scores(h)
        G = _build_G(h)
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(20):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if abs(alpha) < 1e-10:
                    continue
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > 1e-7:
                    h[j] += alpha
                    h[k] -= alpha
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:
                        cur_c5 = actual
                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5

    hard_deadline = t0 + budget_s * 0.97

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, :m] += dx * (1.0 - h[d:d + m])
        G[r, d:d + m] += -dx * h[:m]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, s:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[s:]
    return G

def _lse_obj_grad(h, beta):
    F = _scores(h)
    Fm = F.max()
    F_shift = F - Fm
    BF = beta * F_shift
    BF_max = 0.0
    BF_shifted = BF - BF_max
    log_sum = BF_max + np.log(np.sum(np.exp(BF_shifted)))
    obj = (Fm + log_sum) / beta
    w = np.exp(BF_shifted)
    w /= w.sum()
    G = _build_G(h)
    grad = G.T @ w
    return obj, grad

def _pnorm_obj_grad(h, q):
    F = _scores(h)
    m = F.max()
    if m < 1e-30:
        return m, np.zeros(len(h))
    t = np.clip(F / m, 0.0, 1.0)
    tq = t ** (q - 1)
    T = float(np.sum(t ** q))
    if T < 1e-300:
        return m, np.zeros(len(h))
    obj = m * T ** (1.0 / q)
    scale = T ** (1.0 / q - 1.0)
    G = _build_G(h)
    grad = G.T @ (tq * scale)
    return obj, grad

    target = n / 2.0

    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = _scores(h)
        G = _build_G(h)
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue

                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if abs(alpha) < 1e-11:
                    continue
                pred = float(np.max(scores + alpha * C))
                if cur_c5 - pred > 1e-8:
                    h[j] += alpha
                    h[k] -= alpha
                    actual = _exact_c5(h)
                    if actual < cur_c5 - 1e-14:

                        scores = _scores(h)
                        G = _build_G(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not improved:
            break
    return best_h, best_c5
