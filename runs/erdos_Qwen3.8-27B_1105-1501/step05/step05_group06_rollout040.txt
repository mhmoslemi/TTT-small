```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgradient(h, scores, tol):
    n = len(h)
    dx = 2.0 / n
    max_s = scores.max()
    active = np.where(scores >= max_s - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += dx * (1.0 - h[d:])
            grad[d:] += -dx * h[:m]
        else:
            d_neg = -d
            m = n - d_neg
            grad[:m] += -dx * h[d_neg:]
            grad[d_neg:] += dx * (1.0 - h[:m])
    grad /= len(active)
    return grad

def _subgrad_descend(h, deadline, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    scores = _scores(h)
    step_size = 0.08
    stall = 0
    tol_base = 1e-4
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        tol = tol_base * (1.0 + 0.5 * stall)
        grad = _subgradient(h, scores, tol=tol)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        
        alpha = step_size
        improved = False
        for _ in range(25):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                scores = _scores(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                step_size = min(alpha * 1.15, 0.25)
                improved = True
                break
            alpha *= 0.65
            if alpha < 1e-10:
                break
        if not improved:
            step_size *= 0.65
            stall += 1
    
    return best_h, best_c5

def _de_search(h_init, n, deadline, pop_size, n_gen_max, F):

    best_local_h = pop[int(np.argmin(pop_c5))].copy()
    
    for gen in range(n_gen_max):
        if time.time() > deadline:
            break

                break

                bi = int(np.argmin(pop_c5))
                mutant = pop[bi] + F * (pop[a] - pop[b])
            else:

                pop_c5[i] = trial_c5

            if pop_c5[gi] < best_local_c5 - 1e-15:
                best_local_c5 = float(pop_c5[gi])
                best_local_h = pop[gi].copy()

def _pair_descent(h, deadline, max_passes=100):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    
    dx = 2.0 / n
    # Build G
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]
    for s2 in range(1, n):
        r = n - 1 - s2
        m = n - s2
        G[r, :m] += dx * (1.0 - h[s2:])
        G[r, s2:] += -dx * h[:m]
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                hv_j = h[j]
                hv_k = h[k]
                if hv_j < 1e-10 and hv_k < 1e-10:
                    continue
                if hv_j > 1.0 - 1e-10 and hv_k > 1.0 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-hv_j, hv_k - 1.0)
                a_hi = min(1.0 - hv_j, hv_k)
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = float(np.max(scores + m1 * C))
                    f2 = float(np.max(scores + m2 * C))
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                if abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    np.clip(h, 0.0, 1.0, out=h)
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        # Update G incrementally (approximate: rebuild)
                        # Only rebuild when improvement found
                        G = np.zeros((2 * n - 1, n))
                        for d in range(n):
                            r = n - 1 + d
                            m = n - d
                            G[r, d:] += dx * (1.0 - h[:m])
                            G[r, :m] += -dx * h[d:]
                        for s2 in range(1, n):
                            r = n - 1 - s2
                            m = n - s2
                            G[r, :m] += dx * (1.0 - h[s2:])
                            G[r, s2:] += -dx * h[:m]
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
                        np.clip(h, 0.0, 1.0, out=h)
        if not pass_improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

    from scipy.ndimage import gaussian_filter1d
    h_s = gaussian_filter1d(h, sigma=sigma, mode='wrap')
    return _project(h_s, n / 2.0)
```

```python
    dx = 2.0 / n
    # Compute softmax weights
    max_s = scores.max()
    exp_scores = np.exp(beta * (scores - max_s))
    weights = exp_scores / exp_scores.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    for r in range(len(scores)):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += w * dx * (1.0 - h[d:])
            grad[d:] += -w * dx * h[:m]
        else:
            d_neg = -d
            m = n - d_neg
            grad[:m] += -w * dx * h[d_neg:]
            grad[d_neg:] += w * dx * (1.0 - h[:m])
    return grad
```

```python
    G = np.zeros((2*n-1, n))
    for d in range(n):

        m = n - d
        G[r, d:] += dx * (1.0 - h[:m])
        G[r, :m] += -dx * h[d:]

        m = n - s2
        G[r, :m] += dx * (1.0 - h[s2:])
        G[r, s2:] += -dx * h[:m]
    return G
```

```python
                alpha = alphas_qp[ri % len(alphas_qp)]
                if s == 2:
                    h_init = _binary_init(n_try, rng)
                else:
                    h_init = _quasi_init(n_try, alpha, rng)

            sub_dl_de = min(dl3, time.time() + budget_s * 0.05)
            h_de, c5_de = _de_search(h_init, n_try, sub_dl_de,
                                      pop_size=30, n_gen_max=350, F=0.75,
                                      seed_val=1000 + ri * 10 + s)

            sub_dl_sg = min(dl3, time.time() + budget_s * 0.03)
            h_sg, c5_sg = _softmax_descend(h_de, sub_dl_sg,
                                            beta_start=30, beta_end=250, max_iters=3000)

            c5_try = c5_sg
            h_try = h_sg
            if n_try <= 160 and time.time() < dl3:
                sub_dl_pd = min(dl3, time.time() + budget_s * 0.015)
                h_pd, c5_pd = _pair_descent(h_try, sub_dl_pd, max_passes=80)
                if c5_pd < c5_try:
                    c5_try = c5_pd
                    h_try = h_pd

            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 4: Smoothing restarts (10%)
    if time.time() < t0 + budget_s * 0.63:
        dl4 = t0 + budget_s * 0.63
        for attempt in range(12):
            if time.time() > dl4:
                break
            sigma_val = 1.0 + 0.4 * attempt
            h_sm = _smooth_reproject(best_h, sigma_val)
            sub_dl = min(dl4, time.time() + budget_s * 0.01)
            h_try, c5_try = _softmax_descend(h_sm, sub_dl,
                                              beta_start=50, beta_end=300, max_iters=1500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                if best_n <= 160 and time.time() < dl4:
                    sub_dl2 = min(dl4, time.time() + budget_s * 0.008)
                    h_pd, c5_pd = _pair_descent(best_h, sub_dl2, max_passes=50)
                    if c5_pd < best_c5:
                        best_c5 = c5_pd
                        best_h = h_pd.copy()

    # Phase 5: Intensive local at best (12%)
    if time.time() < t0 + budget_s * 0.75:
        dl5 = t0 + budget_s * 0.75
        sub_dl_sg = min(dl5, time.time() + budget_s * 0.05)
        h_sg, c5_sg = _softmax_descend(best_h, sub_dl_sg,
                                        beta_start=60, beta_end=400, max_iters=5000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

        if best_n <= 192 and time.time() < dl5:
            sub_dl_pd = min(dl5, time.time() + budget_s * 0.04)
            h_pd, c5_pd = _pair_descent(best_h, sub_dl_pd, max_passes=200)
            if c5_pd < best_c5:
                best_c5 = c5_pd
                best_h = h_pd.copy()

        if time.time() < dl5:
            sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
            h_s2, c5_s2 = _softmax_descend(best_h, sub_dl2,
                                            beta_start=80, beta_end=500, max_iters=2000)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2.copy()

    # Phase 6: Perturbation + short search (12%)
    dl6 = t0 + budget_s * 0.87
    scale = 0.02
    for attempt in range(500):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.003)
        h_try, c5_try = _softmax_descend(h_pert, sub_dl,
                                          beta_start=80, beta_end=300, max_iters=400)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.02)

    # Phase 7: DE at best resolution with tight params (7%)
    if time.time() < t0 + budget_s * 0.94:
        dl7 = t0 + budget_s * 0.94
        h_de, c5_de = _de_search(best_h, best_n, dl7,
                                  pop_size=20, n_gen_max=400, F=0.4,
                                  seed_val=9999)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
        sub_dl = min(hard_deadline, time.time() + budget_s * 0.01)
        h_sg, c5_sg = _softmax_descend(best_h, sub_dl,
                                        beta_start=100, beta_end=500, max_iters=1500)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 8: Absolute final polish
    if time.time() < hard_deadline:
        h_fin, c5_fin = _softmax_descend(best_h, hard_deadline,
                                          beta_start=100, beta_end=600, max_iters=2000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if best_n <= 160 and time.time() < hard_deadline:
            h_pd, c5_pd = _pair_descent(best_h, hard_deadline, max_passes=80)
            if c5_pd < best_c5:
                best_c5 = c5_pd
                best_h = h_pd.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
```
