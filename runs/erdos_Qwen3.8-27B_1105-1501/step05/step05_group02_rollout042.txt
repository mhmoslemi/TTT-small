def _dinic_maxflow(capacity, n_nodes, source, sink):
    """
    capacity: 2D array, capacity[u][v] = capacity of edge u->v
    Returns max flow value and min cut (reachable set from source)
    """
    n = n_nodes
    flow = 0
    while True:
        # BFS to find level graph
        level = [-1] * n
        level[source] = 0
        queue = [source]
        for u in queue:
            for v in range(n):
                if level[v] < 0 and capacity[u][v] > 1e-12:
                    level[v] = level[u] + 1
                    queue.append(v)
        if level[sink] < 0:
            break
        # DFS to find blocking flow
        iter_arr = [0] * n
        def dfs(u, f):
            if u == sink:
                return f
            for i in range(iter_arr[u], n):
                iter_arr[u] = i
                v = i
                if capacity[u][v] > 1e-12 and level[v] == level[u] + 1:
                    d = dfs(v, min(f, capacity[u][v]))
                    if d > 0:
                        capacity[u][v] -= d
                        capacity[v][u] += d
                        return d
            return 0
        while True:
            f = dfs(source, float('inf'))
            if f == 0:
                break
            flow += f
    # Find min cut: nodes reachable from source in residual graph
    min_cut = [False] * n
    queue = [source]
    min_cut[source] = True
    for u in queue:
        for v in range(n):
            if not min_cut[v] and capacity[u][v] > 1e-12:
                min_cut[v] = True
                queue.append(v)
    return flow, min_cut

For the min-cut, I want to minimize:
sum_d w_d^+ * S(+d) + sum_d w_d^- * S(-d) + λ * sum(h)

where S(+d) = sum_{j=0}^{n-1-d} h[j+d]*(1-h[j]) and S(-d) = sum_{j=0}^{n-1-d} h[j]*(1-h[j+d]).

The graph:
- n variable nodes (0 to n-1)
- Source (node n) and sink (node n+1)
- For each d from 1 to max_d:
  - For each j from 0 to n-1-d:
    - Edge (j+d) → j with capacity w_d^+ * dx [for S(+d) term]
    - Edge j → (j+d) with capacity w_d^- * dx [for S(-d) term]
- Unary edges for λ:
  - If λ ≥ 0: edge i → sink with capacity λ for each i [penalizes h[i]=1]
  - If λ < 0: edge source → i with capacity -λ for each i [penalizes h[i]=0]

After solving, the min-cut gives:
- Node i in source side → h[i] = 1
- Node i in sink side → h[i] = 0

The count of 1s is the number of variable nodes on the source side.

To get exactly n/2 ones, I either:
1. Find the right λ (binary search)
2. Or repair: if count > n/2, flip some 1s to 0s; if count < n/2, flip some 0s to 1s.

For repair, I'll compute the marginal objective change for each possible flip and pick the best ones.

def _solve_cut(n, weights_pos, weights_neg, lam, max_d):
    """Solve min-cut for the Erdős problem.
    weights_pos[d] = weight for S(+d) terms
    weights_neg[d] = weight for S(-d) terms
    lam = Lagrange multiplier for balance
    Returns binary vector (1 = source side).
    """
    source = n
    sink = n + 1
    edges = []
    dx = 2.0 / n
    for d in range(1, max_d + 1):
        wp = weights_pos[d] * dx
        wn = weights_neg[d] * dx
        if wp < 1e-15 and wn < 1e-15:
            continue
        for j in range(n - d):
            if wp > 1e-15:
                edges.append((j + d, j, wp))  # S(+d): h[j+d]=1, h[j]=0
            if wn > 1e-15:
                edges.append((j, j + d, wn))  # S(-d): h[j]=1, h[j+d]=0
    # Unary edges for balance
    if lam >= 0:
        for i in range(n):
            edges.append((i, sink, lam))
    else:
        for i in range(n):
            edges.append((source, i, -lam))

def _interpolate_cuts(h_lo, h_hi, target):
    cnt_lo = float(h_lo.sum())
    cnt_hi = float(h_hi.sum())
    if abs(cnt_lo - cnt_hi) < 1e-12:
        return h_lo.copy()
    alpha = (target - cnt_hi) / (cnt_lo - cnt_hi)
    alpha = float(np.clip(alpha, 0.0, 1.0))
    return alpha * h_lo + (1.0 - alpha) * h_hi

def _majorization_step(h, beta, max_d):
    n = len(h)
    target = n / 2.0
    sc = _scores(h)
    sc_max = float(np.max(sc))
    w_raw = np.exp(beta * (sc - sc_max))
    w_pos = np.zeros(max_d + 1)
    w_neg = np.zeros(max_d + 1)
    for d in range(1, min(max_d, n - 1) + 1):
        w_pos[d] = w_raw[n - 1 + d]
        w_neg[d] = w_raw[n - 1 - d]
    lam_vals = [-8.0, -4.0, -2.0, -1.0, -0.5, -0.2, 0.0, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    solutions = []
    for lam in lam_vals:
        h_c = _cut_solve(n, w_pos, w_neg, lam, max_d)
        solutions.append(h_c)
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for i in range(len(solutions)):
        for j in range(i + 1, len(solutions)):
            h_interp = _interpolate_cuts(solutions[i], solutions[j], target)
            h_interp = _project(h_interp, target)
            c5_c = _exact_c5(h_interp)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_interp.copy()
        h_proj = _project(solutions[i], target)
        c5_p = _exact_c5(h_proj)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_proj.copy()
    return best_h, best_c5

def _majorization_search(h, deadline, max_d=40, max_betas=5):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    betas = [5.0, 10.0, 20.0, 35.0, 60.0]
    for beta in betas:
        if time.time() > deadline:
            break
        for _ in range(3):
            if time.time() > deadline:
                break
            h_new, c5_new = _majorization_step(h, beta, max_d)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
    return best_h, best_c5

def _slsqp_fun(x, n, aidx):
    return x[n]

def _slsqp_fun_grad(x, n, aidx):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _slsqp_eq(x, n, aidx):
    return float(np.sum(x[:n]) - n / 2.0)

def _slsqp_eq_grad(x, n, aidx):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _slsqp_inq(x, n, aidx):
    h = x[:n]
    t = x[n]
    sc = _scores(h)
    return t - sc[aidx]

def _slsqp_inq_grad(x, n, aidx):
    h = x[:n]
    dx = 2.0 / n
    g = np.zeros((len(aidx), n + 1))
    for i, j in enumerate(aidx):
        d = j - (n - 1)
        k1lo = max(0, d)
        k1hi = min(n, n + d)
        k2lo = max(0, -d)
        k2hi = min(n, n - d)
        if k1lo < k1hi:
            g[i, k1lo:k1hi] = -(1.0 - h[k1lo - d:k1hi - d]) * dx
        if k2lo < k2hi:
            g[i, k2lo:k2hi] += h[k2lo + d:k2hi + d] * dx
        g[i, n] = 1.0
    return g

def _slsqp_polish(h, topk=30):
    n = len(h)
    sc = _scores(h)
    order = np.argsort(sc)[::-1]
    aidx = order[:topk]
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = float(np.max(sc))
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_grad, 'args': (n, aidx)}]
    cons.append({'type': 'ineq', 'fun': _slsqp_inq, 'jac': _slsqp_inq_grad, 'args': (n, aidx)})
    try:
        res = minimize(_slsqp_fun, x0, args=(n, aidx), method='SLSQP',
                       jac=_slsqp_fun_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        if res.success or (hasattr(res, 'x') and np.all(np.isfinite(res.x))):
            h_new = res.x[:n]
            h_new = _project(h_new, n / 2.0)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h, _exact_c5(h)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _line_search(h, p, n_eval=20):
    n = len(h)
    target = n / 2.0
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(n_eval):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, f2 = c2, f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo, f1 = c1, f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = 0.5 * (lo + hi)
    return opt_a, _exact_c5(_project(h + opt_a * p, target))

def _lp_step(h, radius, topk):
    n = len(h)
    target = n / 2.0
    sc = _scores(h)
    G = _grad_matrix(h)
    nlags = 2 * n - 1
    if topk < nlags:
        idx = np.argsort(sc)[-topk:]
        sc_k = sc[idx]
        G_k = G[idx]
    else:
        sc_k = sc
        G_k = G
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(sc_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -sc_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x
